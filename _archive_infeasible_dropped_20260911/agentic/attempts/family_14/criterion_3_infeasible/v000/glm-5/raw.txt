```python
import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    required = set()
    for _ in range(t):
        required.add(int(next(it)))
    
    # Track which required sites have at least one edge touching them
    has_edge = set()
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        _ = next(it)  # weight, not needed
        if u in required:
            has_edge.add(u)
        if v in required:
            has_edge.add(v)
    
    # Find the smallest required site with no edge
    answer = None
    for site in required:
        if site not in has_edge:
            if answer is None or site < answer:
                answer = site
    
    print(answer)

if __name__ == "__main__":
    main()
```
