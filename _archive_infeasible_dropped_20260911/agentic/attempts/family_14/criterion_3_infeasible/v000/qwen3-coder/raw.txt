```python
n, m, t = map(int, input().split())
required = set(map(int, input().split()))

# Track which sites appear in any cabling option
connected_sites = set()

for _ in range(m):
    u, v, w = map(int, input().split())
    connected_sites.add(u)
    connected_sites.add(v)

# Find required sites that are not connected to any cabling option
unconnected_required = []
for site in required:
    if site not in connected_sites:
        unconnected_required.append(site)

# Output the smallest ID among unconnected required sites
print(min(unconnected_required))
```
