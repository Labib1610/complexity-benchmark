```python
import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))  # k is always 0, not needed
    
    has_link = [False] * (n + 1)
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        has_link[u] = True
        has_link[v] = True
    
    for i in range(1, n + 1):
        if not has_link[i]:
            print(i)
            return

if __name__ == "__main__":
    main()
```
