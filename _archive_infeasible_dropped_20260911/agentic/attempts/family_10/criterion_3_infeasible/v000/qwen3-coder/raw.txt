```python
n, m, k = map(int, input().split())

# Create a set to track all locations that have links
linked_locations = set()

# Process each link
for _ in range(m):
    u, v = map(int, input().split())
    linked_locations.add(u)
    linked_locations.add(v)

# Find the smallest location index that has no links
for i in range(1, n + 1):
    if i not in linked_locations:
        print(i)
        break
```
