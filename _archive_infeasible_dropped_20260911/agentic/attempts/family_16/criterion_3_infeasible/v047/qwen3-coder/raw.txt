```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    
    idx = 2
    scenes = []
    total_gpu = 0
    total_cpu = 0
    
    for i in range(j):
        gpu_time = int(data[idx])
        cpu_time = int(data[idx + 1])
        idx += 2
        scenes.append((gpu_time, cpu_time))
        total_gpu += gpu_time
        total_cpu += cpu_time
        
        # Check JOB condition
        if gpu_time + cpu_time > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD conditions
    if total_gpu > T:
        print("LOAD 1")
        return
    if total_cpu > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    main()
```
