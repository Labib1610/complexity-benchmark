We can solve this by checking each dossier’s total time and the total load on each desk.

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    
    j, T = int(data[0]), int(data[1])
    idx = 2
    desk1_total = 0
    desk2_total = 0
    
    for i in range(1, j + 1):
        p1 = int(data[idx])
        p2 = int(data[idx + 1])
        idx += 2
        
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
        
        desk1_total += p1
        desk2_total += p2
    
    if desk1_total > T:
        print("LOAD 1")
    elif desk2_total > T:
        print("LOAD 2")
    else:
        # If no violation, but problem guarantees one exists,
        # we still need to output something. In practice, this won't happen.
        print("LOAD 1")

if __name__ == "__main__":
    main()
```
