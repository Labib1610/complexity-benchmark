We can read the input, keep track of totals for both resources, and check for the first item that exceeds the given T when adding its p1 and p2. If such an item is found, print `JOB i`. If not, then check the totals for each resource and print `LOAD c` accordingly.

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return

    j = int(data[0])
    T = int(data[1])
    
    total1 = 0
    total2 = 0
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(data[idx])
        p2 = int(data[idx + 1])
        idx += 2
        
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
        
        total1 += p1
        total2 += p2
    
    if total1 > T:
        print("LOAD 1")
    elif total2 > T:
        print("LOAD 2")

if __name__ == "__main__":
    main()
```
