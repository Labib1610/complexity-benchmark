```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    total1 = 0
    total2 = 0
    # first pass to detect isolated task
    tasks = []
    for idx in range(1, j + 1):
        p1 = int(next(it))
        p2 = int(next(it))
        if p1 + p2 > T:
            print(f"JOB {idx}")
            return
        total1 += p1
        total2 += p2
        tasks.append((p1, p2))  # not really needed later
    # no isolated task, check subsystem loads
    if total1 > T:
        print("LOAD 1")
    else:
        # problem guarantees exactly one violation, so total2 must exceed T
        print("LOAD 2")

if __name__ == "__main__":
    solve()
```
