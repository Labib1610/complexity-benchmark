```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    T = int(next(it))
    total1 = 0
    total2 = 0
    job_idx = -1
    for i in range(1, n + 1):
        p1 = int(next(it))
        p2 = int(next(it))
        total1 += p1
        total2 += p2
        if job_idx == -1 and p1 + p2 > T:
            job_idx = i
    out = []
    if job_idx != -1:
        out.append(f"JOB {job_idx}")
    elif total1 > T:
        out.append("LOAD 1")
    else:
        out.append("LOAD 2")
    sys.stdout.write(out[0])

if __name__ == "__main__":
    solve()
```
