```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    
    p1_total = 0
    p2_total = 0
    
    idx = 2
    for i in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx+1])
        idx += 2
        
        # Check if single job exceeds deadline
        if p1 + p2 > T:
            print(f"JOB {i+1}")
            return
            
        p1_total += p1
        p2_total += p2
    
    # Check if either machine is overloaded
    if p1_total > T:
        print("LOAD 1")
        return
    if p2_total > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    main()
```
