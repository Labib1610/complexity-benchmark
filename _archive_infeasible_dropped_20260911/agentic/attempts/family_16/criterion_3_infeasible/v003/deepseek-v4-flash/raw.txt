Here’s the Python solution for the problem:  

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    T = int(data[1])
    idx = 2
    total1 = 0
    total2 = 0
    for i in range(1, n + 1):
        p1 = int(data[idx])
        p2 = int(data[idx + 1])
        idx += 2
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
        total1 += p1
        total2 += p2
    if total1 > T:
        print("LOAD 1")
    elif total2 > T:
        print("LOAD 2")
    else:
        # Should not happen per problem statement, but just in case
        print("LOAD 1")

if __name__ == "__main__":
    main()
```
