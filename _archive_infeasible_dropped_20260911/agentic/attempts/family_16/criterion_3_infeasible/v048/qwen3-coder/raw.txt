```python
j, T = map(int, input().split())

total_p1 = 0
total_p2 = 0

for i in range(1, j + 1):
    p1, p2 = map(int, input().split())
    total_p1 += p1
    total_p2 += p2
    
    # Check for Isolated Request Violation
    if p1 + p2 > T:
        print(f"JOB {i}")
        exit()

# Check for Module Saturation
if total_p1 > T:
    print("LOAD 1")
elif total_p2 > T:
    print("LOAD 2")
```
