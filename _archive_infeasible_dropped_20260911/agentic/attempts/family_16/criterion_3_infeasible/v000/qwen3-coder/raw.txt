```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    
    index = 2
    jobs = []
    p1_total = 0
    p2_total = 0
    
    for i in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        index += 2
        jobs.append((p1, p2))
        p1_total += p1
        p2_total += p2
        # Check if any single job exceeds the deadline
        if p1 + p2 > T:
            print(f"JOB {i+1}")
            return
    
    # If no single job exceeds, check machine loads
    if p1_total > T:
        print("LOAD 1")
        return
    if p2_total > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    main()
```
