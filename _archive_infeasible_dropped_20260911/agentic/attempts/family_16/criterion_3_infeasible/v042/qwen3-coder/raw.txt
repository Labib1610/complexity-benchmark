```python
j, T = map(int, input().split())

jobs = []
total_p1 = 0
total_p2 = 0

for i in range(j):
    p1, p2 = map(int, input().split())
    jobs.append((p1, p2))
    total_p1 += p1
    total_p2 += p2
    
    # Check if this job alone exceeds the deadline
    if p1 + p2 > T:
        print(f"JOB {i+1}")
        exit()

# Check if either facility is overloaded
if total_p1 > T:
    print("LOAD 1")
    exit()
    
if total_p2 > T:
    print("LOAD 2")
    exit()
```
