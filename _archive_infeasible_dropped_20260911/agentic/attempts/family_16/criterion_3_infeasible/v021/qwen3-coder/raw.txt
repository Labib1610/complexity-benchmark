```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    
    headers = []
    payloads = []
    
    index = 2
    for _ in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        headers.append(p1)
        payloads.append(p2)
        index += 2
    
    # Check for individual job exceeding T
    for i in range(j):
        if headers[i] + payloads[i] > T:
            print(f"JOB {i+1}")
            return
    
    # Check for channel saturation
    total_header_time = sum(headers)
    total_payload_time = sum(payloads)
    
    if total_header_time > T:
        print("LOAD 1")
        return
    if total_payload_time > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    main()
```
