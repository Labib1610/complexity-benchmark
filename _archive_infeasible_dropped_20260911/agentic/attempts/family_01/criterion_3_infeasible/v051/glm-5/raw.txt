import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for lexicographic order
    adj_lists = [sorted(s) for s in adj]
    
    # Iterate over all possible triangles in lexicographic order
    # a < b < c, so we iterate a from 1 to n-2, b from a+1 to n-1, c from b+1 to n
    for a in range(1, n - 1):
        for b in adj_lists[a]:
            if b <= a:
                continue
            # We have edge (a, b). We need c > b such that (a, c) and (b, c) exist.
            # Iterate through neighbors of a that are > b
            for c in adj_lists[a]:
                if c <= b:
                    continue
                if c in adj[b]:
                    print(a, b, c)
                    return
    
    # Should not reach here given problem guarantees
    print("-1")

if __name__ == "__main__":
    solve()
