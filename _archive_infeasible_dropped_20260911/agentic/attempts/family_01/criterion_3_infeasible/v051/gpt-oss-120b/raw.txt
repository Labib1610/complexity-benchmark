```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    neigh = [0] * n  # bitmask of neighbors
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        neigh[u] |= 1 << v
        neigh[v] |= 1 << u

    for a in range(n):
        mask_a = neigh[a]
        for b in range(a + 1, n):
            if (mask_a >> b) & 1:          # edge a-b exists
                common = mask_a & neigh[b]  # neighbors of both a and b
                # keep only vertices > b
                common &= ~((1 << (b + 1)) - 1)
                if common:
                    # lowest set bit gives smallest c
                    c_bit = common & -common
                    c = c_bit.bit_length() - 1
                    print(a + 1, b + 1, c + 1)
                    return

if __name__ == "__main__":
    solve()
```
