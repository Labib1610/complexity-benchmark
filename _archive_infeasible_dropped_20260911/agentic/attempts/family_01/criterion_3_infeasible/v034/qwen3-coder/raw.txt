```python
n, m = map(int, input().split())
edges = set()
for _ in range(m):
    u, v = map(int, input().split())
    if u > v:
        u, v = v, u
    edges.add((u, v))

# For each vertex, maintain a set of its neighbors
adj = [set() for _ in range(n + 1)]
for u, v in edges:
    adj[u].add(v)
    adj[v].add(u)

# Find the lexicographically smallest triangle
found = False
for a in range(1, n + 1):
    if found:
        break
    for b in range(a + 1, n + 1):
        if found:
            break
        if b in adj[a]:  # a and b are connected
            for c in range(b + 1, n + 1):
                if c in adj[a] and c in adj[b]:  # a-c and b-c are connected
                    print(a, b, c)
                    found = True
                    break
```
