```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list and edge set for O(1) lookup
    adj = defaultdict(set)
    edges = set()
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        adj[u].add(v)
        adj[v].add(u)
        edges.add((min(u,v), max(u,v)))
    
    # Convert adjacency lists to sorted lists for deterministic iteration
    for u in range(1, n+1):
        adj[u] = sorted(adj[u])
    
    # Iterate through all possible triplets (i,j,k) with i<j<k
    # For efficiency, iterate through each node, then its neighbors, then their common neighbors
    best = None
    
    for i in range(1, n-1):
        for j in adj[i]:
            if j <= i:
                continue
            for k in adj[i]:
                if k <= j:
                    continue
                # Check if j and k are connected
                if k in adj[j]:
                    candidate = (i, j, k)
                    if best is None or candidate < best:
                        best = candidate
    
    print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
```
