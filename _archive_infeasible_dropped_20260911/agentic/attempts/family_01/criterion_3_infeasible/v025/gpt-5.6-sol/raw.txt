import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]

    adjacency = [0] * n
    index = 2
    for _ in range(m):
        u = data[index] - 1
        v = data[index + 1] - 1
        index += 2
        adjacency[u] |= 1 << v
        adjacency[v] |= 1 << u

    all_vertices = (1 << n) - 1
    higher = [all_vertices ^ ((1 << (i + 1)) - 1) for i in range(n)]

    for a in range(n):
        candidates_b = adjacency[a] & higher[a]
        while candidates_b:
            b_bit = candidates_b & -candidates_b
            b = b_bit.bit_length() - 1

            candidates_c = adjacency[a] & adjacency[b] & higher[b]
            if candidates_c:
                c_bit = candidates_c & -candidates_c
                c = c_bit.bit_length() - 1
                print(a + 1, b + 1, c + 1)
                return

            candidates_b ^= b_bit

if __name__ == "__main__":
    main()
