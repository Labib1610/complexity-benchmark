```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    # adjacency matrix using bytearray for O(1) checks
    adj = [bytearray(n + 1) for _ in range(n + 1)]
    neigh = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u][v] = adj[v][u] = 1
        neigh[u].append(v)
        neigh[v].append(u)

    for i in range(1, n + 1):
        neigh[i].sort()

    out = []
    for a in range(1, n + 1):
        na = neigh[a]
        for b in na:
            if b <= a:
                continue
            # c must be > b and neighbor of a
            for c in na:
                if c <= b:
                    continue
                if adj[b][c]:
                    out = [a, b, c]
                    sys.stdout.write(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
```
