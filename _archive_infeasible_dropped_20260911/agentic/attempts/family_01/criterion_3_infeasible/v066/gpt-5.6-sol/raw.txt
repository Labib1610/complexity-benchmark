```python
import sys

data = list(map(int, sys.stdin.buffer.read().split()))
n, m = data[0], data[1]

adj = [0] * n
pos = 2
for _ in range(m):
    u = data[pos] - 1
    v = data[pos + 1] - 1
    pos += 2
    adj[u] |= 1 << v
    adj[v] |= 1 << u

for a in range(n):
    candidates_b = adj[a] >> (a + 1)
    while candidates_b:
        lowest = candidates_b & -candidates_b
        b = a + 1 + lowest.bit_length() - 1

        candidates_c = (adj[a] & adj[b]) >> (b + 1)
        if candidates_c:
            c = b + 1 + (candidates_c & -candidates_c).bit_length() - 1
            print(a + 1, b + 1, c + 1)
            sys.exit()

        candidates_b ^= lowest
```
