import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_mask = [0] * n
    neigh = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            continue
        adj_mask[u] |= 1 << v
        adj_mask[v] |= 1 << u
        neigh[u].append(v)
        neigh[v].append(u)

    for a in range(n):
        # ensure neighbors are sorted for lexicographic order
        neigh[a].sort()
    for a in range(n):
        for b in neigh[a]:
            if b <= a:
                continue
            common = adj_mask[a] & adj_mask[b]
            # remove vertices <= b
            common >>= (b + 1)
            if common:
                # lowest set bit
                low = (common & -common).bit_length() - 1
                c = b + 1 + low
                # output 1‑based indices
                print(a + 1, b + 1, c + 1)
                return

if __name__ == "__main__":
    solve()
