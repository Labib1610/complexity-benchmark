```python
import sys

data = list(map(int, sys.stdin.buffer.read().split()))
n, m = data[0], data[1]

adj = [0] * n
p = 2
for _ in range(m):
    u = data[p] - 1
    v = data[p + 1] - 1
    p += 2
    adj[u] |= 1 << v
    adj[v] |= 1 << u

all_mask = (1 << n) - 1
higher = [all_mask ^ ((1 << (i + 1)) - 1) for i in range(n)]

for a in range(n):
    candidates = adj[a] & higher[a]
    while candidates:
        low = candidates & -candidates
        b = low.bit_length() - 1

        common = adj[a] & adj[b] & higher[b]
        if common:
            c = (common & -common).bit_length() - 1
            print(a + 1, b + 1, c + 1)
            sys.exit()

        candidates ^= low
```
