```python
import sys

data = map(int, sys.stdin.buffer.read().split())
n = next(data)
m = next(data)

adj = [0] * n

for _ in range(m):
    u = next(data) - 1
    v = next(data) - 1
    adj[u] |= 1 << v
    adj[v] |= 1 << u

all_mask = (1 << n) - 1
higher = [all_mask ^ ((1 << (i + 1)) - 1) for i in range(n)]

for a in range(n):
    candidates_b = adj[a] & higher[a]
    while candidates_b:
        bit_b = candidates_b & -candidates_b
        b = bit_b.bit_length() - 1

        common = adj[a] & adj[b] & higher[b]
        if common:
            bit_c = common & -common
            c = bit_c.bit_length() - 1
            print(a + 1, b + 1, c + 1)
            sys.exit()

        candidates_b ^= bit_b
```
