```python
import sys

def main():
    input = sys.stdin.buffer.readline
    n, m = map(int, input().split())

    adjacency = [0] * n
    forward = [[] for _ in range(n)]
    vertex_bit = [1 << i for i in range(n)]

    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        if u > v:
            u, v = v, u
        adjacency[u] |= vertex_bit[v]
        adjacency[v] |= vertex_bit[u]
        forward[u].append(v)

    for neighbors in forward:
        neighbors.sort()

    for a in range(n):
        adj_a = adjacency[a]
        for b in forward[a]:
            common = (adj_a & adjacency[b]) >> (b + 1)
            if common:
                lowest = common & -common
                c = b + 1 + lowest.bit_length() - 1
                print(a + 1, b + 1, c + 1)
                return

main()
```
