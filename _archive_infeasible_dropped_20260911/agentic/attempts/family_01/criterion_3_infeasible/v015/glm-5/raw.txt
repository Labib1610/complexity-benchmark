import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    
    # To find lexicographically smallest triangle (a, b, c) with a < b < c.
    # Iterate a from 1 to n, b from a+1 to n, c from b+1 to n.
    # Check if edges (a,b), (a,c), (b,c) all exist.
    # Use adjacency sets for O(1) lookups.
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # b must be > a and in neighbors_a
        # We want smallest b, then smallest c
        # Get sorted neighbors of a that are > a
        for b in sorted(neighbors_a):
            if b <= a:
                continue
            if b > n:
                continue
            # c must be > b, in neighbors_a and in adj[b]
            # To find smallest c, iterate sorted neighbors of b that are > b and check if in neighbors_a
            for c in sorted(adj[b]):
                if c <= b:
                    continue
                if c in neighbors_a:
                    print(a, b, c)
                    return
                # Since sorted, once c > b and we check in order, first valid is smallest c for this (a,b)
                # But we need to continue checking only until we find one.
                # Actually the loop will find the first valid c.
            # Note: we break out of b loop only after finding a triangle? No, we need smallest b too.
            # The structure: for each a, for each b in sorted order, for each c in sorted order.
            # The first triangle found is lexicographically smallest.
    # Should never reach here per problem statement
    print(-1)

solve()
