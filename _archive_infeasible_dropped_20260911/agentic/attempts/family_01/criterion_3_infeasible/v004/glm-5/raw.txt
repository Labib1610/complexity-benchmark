```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)

    best = None
    for a in range(1, n + 1):
        la = adj[a]
        if len(la) < 2:
            continue
        # candidates b > a
        bs = sorted(x for x in la if x > a)
        if len(bs) < 2:
            continue
        # try to find b < c with b,c in bs and edge b-c
        # lexicographically smallest: smallest b, then smallest c
        for i in range(len(bs)):
            b = bs[i]
            lb = adj[b]
            for j in range(i + 1, len(bs)):
                c = bs[j]
                if c in lb:
                    best = (a, b, c)
                    print(a, b, c)
                    return
    # guaranteed to exist
    if best:
        print(best[0], best[1], best[2])

solve()
```
