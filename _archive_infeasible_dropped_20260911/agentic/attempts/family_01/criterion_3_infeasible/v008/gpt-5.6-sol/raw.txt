```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]

    adj = [0] * n
    pos = 2
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for a in range(n - 2):
        neighbors = adj[a] & (-1 << (a + 1))
        while neighbors:
            bit = neighbors & -neighbors
            b = bit.bit_length() - 1
            neighbors ^= bit

            common = (adj[a] & adj[b]) >> (b + 1)
            if common:
                c = b + 1 + ((common & -common).bit_length() - 1)
                print(a + 1, b + 1, c + 1)
                return

if __name__ == "__main__":
    main()
```
