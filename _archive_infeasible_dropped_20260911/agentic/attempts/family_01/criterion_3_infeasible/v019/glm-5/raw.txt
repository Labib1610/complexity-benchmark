import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1

    adj = [set() for _ in range(n + 1)]
    deg = [0] * (n + 1)

    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        adj[u].add(v)
        adj[v].add(u)
        deg[u] += 1
        deg[v] += 1

    # Order vertices by degree ascending, then by id ascending
    order = sorted(range(1, n + 1), key=lambda x: (deg[x], x))
    rank = [0] * (n + 1)
    for r, v in enumerate(order):
        rank[v] = r

    # For each vertex, keep neighbors with rank >= its own rank
    high = [[] for _ in range(n + 1)]
    for v in range(1, n + 1):
        for u in adj[v]:
            if rank[u] >= rank[v]:
                high[v].append(u)

    # Sort high[v] by rank ascending (which also sorts by id when degrees tie)
    for v in range(1, n + 1):
        high[v].sort(key=lambda x: rank[x])

    # Mark sets for fast intersection
    # We'll iterate vertices in id order to find lex smallest triangle
    # For each v, for each u in high[v] with u > v, check common neighbors w > u
    # that are in both high[v] and high[u]

    # To make it efficient, convert high lists to sets for membership
    high_set = [set() for _ in range(n + 1)]
    for v in range(1, n + 1):
        high_set[v] = set(high[v])

    for v in range(1, n + 1):
        hv = high[v]
        hv_set = high_set[v]
        for u in hv:
            if u <= v:
                continue
            # Find common w in high[v] and high[u] with w > u
            # Iterate over smaller set
            if len(hv) <= len(high[u]):
                for w in hv:
                    if w > u and w in high_set[u]:
                        print(v, u, w)
                        return
            else:
                hu = high[u]
                hu_set = high_set[u]
                for w in hu:
                    if w > u and w in hv_set:
                        print(v, u, w)
                        return

solve()
