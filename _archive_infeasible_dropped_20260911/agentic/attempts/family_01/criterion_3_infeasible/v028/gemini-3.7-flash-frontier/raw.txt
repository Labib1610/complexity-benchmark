```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    
    adj = [0] * (n + 1)
    neighbors_greater = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
        if u < v:
            neighbors_greater[u].append(v)
        else:
            neighbors_greater[v].append(u)
            
    for a in range(1, n + 1):
        neighbors_greater[a].sort()
        for b in neighbors_greater[a]:
            common = adj[a] & adj[b]
            valid = common >> (b + 1)
            if valid:
                c = b + 1 + (valid & -valid).bit_length() - 1
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    main()
```
