```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [set() for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # We'll iterate through all edges and look for a third node that connects to both
    # That would form a triangle (resonance cascade)
    
    # To ensure we get the lexicographically smallest one, we sort the edges appropriately
    # But since we want the lex smallest triple (a,b,c), we should generate candidates 
    # and sort them properly
    
    # However, a better approach is to iterate over potential triples in lexicographical order
    # and check if all three edges exist
    
    # Let's build a set of edges for O(1) lookup
    edge_set = set()
    for u, v in edges:
        if u > v:
            u, v = v, u
        edge_set.add((u, v))
    
    # Now iterate through all possible triples (a,b,c) with a<b<c in lexicographical order
    # which means we iterate a from 1 to n, b from a+1 to n, c from b+1 to n
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if (a, b) not in edge_set:
                continue
            for c in range(b + 1, n + 1):
                if (a, c) in edge_set and (b, c) in edge_set:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
