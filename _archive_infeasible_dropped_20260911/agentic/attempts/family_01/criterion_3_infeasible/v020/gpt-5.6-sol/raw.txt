```python
import sys

data = list(map(int, sys.stdin.buffer.read().split()))
n, m = data[0], data[1]

adj_bits = [0] * n
adj_list = [[] for _ in range(n)]

idx = 2
for _ in range(m):
    u = data[idx] - 1
    v = data[idx + 1] - 1
    idx += 2
    adj_bits[u] |= 1 << v
    adj_bits[v] |= 1 << u
    adj_list[u].append(v)
    adj_list[v].append(u)

for a in range(n):
    adj_list[a].sort()
    for b in adj_list[a]:
        if b <= a:
            continue
        common = (adj_bits[a] & adj_bits[b]) >> (b + 1)
        if common:
            offset = (common & -common).bit_length() - 1
            c = b + 1 + offset
            print(a + 1, b + 1, c + 1)
            sys.exit()
```
