```python
import sys
from collections import defaultdict

def find_triangle(n, edges):
    # Build adjacency list
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # For each node, check its neighbors
    # For every pair of neighbors, check if they are connected
    # That would form a triangle
    
    # We want the lexicographically smallest triangle
    # So we iterate through nodes in order, and for each node,
    # we check pairs of its neighbors in order
    
    for u in range(1, n + 1):
        neighbors = sorted(list(adj[u]))
        for i in range(len(neighbors)):
            for j in range(i + 1, len(neighbors)):
                v = neighbors[i]
                w = neighbors[j]
                # Check if v and w are connected (forming triangle u-v-w)
                if w in adj[v]:
                    return [u, v, w]
    
    return None

# Read input
n, m = map(int, sys.stdin.readline().split())
edges = []
for _ in range(m):
    u, v = map(int, sys.stdin.readline().split())
    edges.append((u, v))

# Find and print the triangle
triangle = find_triangle(n, edges)
print(' '.join(map(str, sorted(triangle))))
```
