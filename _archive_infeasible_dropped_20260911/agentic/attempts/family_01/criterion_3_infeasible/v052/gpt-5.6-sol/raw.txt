```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]

    adj = [0] * n
    pos = 2
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        if u != v:
            adj[u] |= 1 << v
            adj[v] |= 1 << u

    for a in range(n - 2):
        bs = adj[a] & ~((1 << (a + 1)) - 1)
        while bs:
            b_bit = bs & -bs
            b = b_bit.bit_length() - 1
            bs ^= b_bit

            common = adj[a] & adj[b] & ~((1 << (b + 1)) - 1)
            if common:
                c = (common & -common).bit_length() - 1
                print(a + 1, b + 1, c + 1)
                return

main()
```
