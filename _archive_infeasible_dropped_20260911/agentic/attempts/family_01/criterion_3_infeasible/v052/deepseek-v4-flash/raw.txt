We can solve this by checking all triples in increasing order and stopping at the first valid one.  
Since \( n \leq 2000 \), a naive triple loop would be \( O(n^3) \), which is too slow.  
Instead, we can use adjacency sets and iterate over pairs \((a, b)\) and check for a common neighbor \( c > b \).

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b not in adj[a]:
                continue
            # find smallest c > b such that c in adj[a] and c in adj[b]
            for c in range(b + 1, n + 1):
                if c in adj[a] and c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
```
