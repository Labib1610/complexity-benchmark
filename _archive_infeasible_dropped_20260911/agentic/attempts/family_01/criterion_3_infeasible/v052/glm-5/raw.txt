```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)

    best = None
    for u, v in edges:
        au = adj[u]
        av = adj[v]
        if len(au) <= len(av):
            common = au & av
        else:
            common = av & au
        if common:
            w = min(common)
            cand = (u, v, w)
            if best is None or cand < best:
                best = cand

    print(best[0], best[1], best[2])

solve()
```
