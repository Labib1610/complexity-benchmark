```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)          # 1‑based, bits 0..n-1 correspond to vertices 1..n
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        bit_u = 1 << (v - 1)
        bit_v = 1 << (u - 1)
        adj[u] |= bit_u
        adj[v] |= bit_v

    for a in range(1, n - 1):
        bits_a = adj[a]
        # iterate b > a where edge a‑b exists
        # we can scan bits of bits_a starting from a
        mask = bits_a >> a          # bits for vertices > a
        while mask:
            # isolate lowest set bit -> gives b
            low = mask & -mask
            b = (low.bit_length() - 1) + a + 1  # convert to vertex number
            # common neighbours of a and b
            common = bits_a & adj[b]
            # consider only c > b
            common >>= b
            if common:
                # lowest set bit gives smallest c > b
                lowc = common & -common
                c = (lowc.bit_length() - 1) + b + 1
                sys.stdout.write(f"{a} {b} {c}")
                return
            # clear this bit and continue
            mask ^= low

if __name__ == "__main__":
    solve()
```
