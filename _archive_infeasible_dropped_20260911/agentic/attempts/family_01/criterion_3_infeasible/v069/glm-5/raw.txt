import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # For each pair (a, b) with a < b, check if they are incompatible
    # If so, find the smallest c > b such that c is incompatible with both a and b
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Iterate through neighbors b of a where b > a
        for b in neighbors_a:
            if b <= a:
                continue
            neighbors_b = adj[b]
            # Find smallest c > b that is in both neighbors_a and neighbors_b
            # Iterate through neighbors of a that are > b
            for c in neighbors_a:
                if c > b and c in neighbors_b:
                    print(a, b, c)
                    return
    
    # Should not reach here as problem guarantees at least one triad exists
    # But if it does, try alternative approach
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            neighbors_b = adj[b]
            for c in neighbors_b:
                if c > b and c in neighbors_a:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
