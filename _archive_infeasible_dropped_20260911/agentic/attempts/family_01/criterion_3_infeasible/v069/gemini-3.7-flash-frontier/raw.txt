import sys

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    for a in range(1, n + 1):
        # We only consider neighbors b > a
        neighbors_a = adj[a] >> (a + 1)
        if not neighbors_a:
            continue
        
        # Iterate over neighbors b > a
        curr_b = a + 1
        temp = neighbors_a
        while temp:
            lsb = temp & -temp
            b = a + 1 + (lsb.bit_length() - 1)
            temp &= temp - 1
            
            # Common neighbors > b
            common = (adj[a] & adj[b]) >> (b + 1)
            if common:
                c_lsb = common & -common
                c = b + 1 + (c_lsb.bit_length() - 1)
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    main()
