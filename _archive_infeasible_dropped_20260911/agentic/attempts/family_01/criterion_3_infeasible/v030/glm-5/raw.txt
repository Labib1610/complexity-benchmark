```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    
    for i in range(m):
        u = int(input_data[2 + 2 * i])
        v = int(input_data[2 + 2 * i + 1])
        adj[u].add(v)
        adj[v].add(u)
        
    best_triplet = None
    
    for a in range(1, n + 1):
        if best_triplet is not None:
            break
        neighbors_a = adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            if best_triplet is not None and b >= best_triplet[1]:
                break
            neighbors_b = adj[b]
            for c in neighbors_b:
                if c <= b:
                    continue
                if c in neighbors_a:
                    best_triplet = (a, b, c)
                    break
                    
    if best_triplet:
        print(best_triplet[0], best_triplet[1], best_triplet[2])

if __name__ == '__main__':
    solve()
```
