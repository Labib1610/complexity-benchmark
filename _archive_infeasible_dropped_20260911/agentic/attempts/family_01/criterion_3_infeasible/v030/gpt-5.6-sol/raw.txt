```python
import sys

data = list(map(int, sys.stdin.buffer.read().split()))
n, m = data[0], data[1]

bits = [1 << i for i in range(n)]
adj = [0] * n
forward = [[] for _ in range(n)]

p = 2
for _ in range(m):
    u = data[p] - 1
    v = data[p + 1] - 1
    p += 2

    adj[u] |= bits[v]
    adj[v] |= bits[u]

    if u > v:
        u, v = v, u
    forward[u].append(v)

for neighbors in forward:
    neighbors.sort()

for a in range(n):
    row = adj[a]
    for b in forward[a]:
        common = (row & adj[b]) >> (b + 1)
        if common:
            c = b + (common & -common).bit_length() + 1
            print(a + 1, b + 1, c)
            sys.exit()
```
