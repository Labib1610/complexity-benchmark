```python
import sys

def solve():
    input = sys.stdin.buffer
    data = input.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    best = None
    
    for u, v in edges:
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        for w in adj[u]:
            if w in adj[v]:
                a, b, c = u, v, w
                if a > b:
                    a, b = b, a
                if b > c:
                    b, c = c, b
                if a > b:
                    a, b = b, a
                if best is None or (a, b, c) < best:
                    best = (a, b, c)
        if best is not None and best[0] == 1:
            break
    
    print(*best)

solve()
```
