```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    best = None
    
    for a in range(1, n + 1):
        if best is not None and a >= best[0]:
            break
        adj_a = adj[a]
        if len(adj_a) < 2:
            continue
        
        candidates = [x for x in adj_a if x > a]
        if len(candidates) < 2:
            continue
        
        found = False
        for i in range(len(candidates)):
            b = candidates[i]
            if best is not None and b >= best[1]:
                break
            adj_b = adj[b]
            for j in range(i + 1, len(candidates)):
                c = candidates[j]
                if c in adj_b:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
                    found = True
                    break
            if found:
                break
        
        if best is not None and best[0] == a:
            break
    
    print(best[0], best[1], best[2])

solve()
```
