import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Convert sets to lists for faster access
    adj_list = {u: list(neighbors) for u, neighbors in adj.items()}
    
    # Sort adjacency lists to enable early termination optimizations
    for u in adj_list:
        adj_list[u].sort()
    
    # Iterate through all nodes as potential first element
    for a in range(1, n+1):
        if a not in adj_list:
            continue
        neighbors_a = adj_list[a]
        # Check pairs among neighbors of 'a'
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            # Look for common neighbor with a that is greater than b
            neighbors_b = adj_list.get(b, [])
            # Use two pointers to find common neighbors efficiently
            j = 0
            while j < len(neighbors_b) and neighbors_b[j] <= b:
                j += 1
            
            k_idx = i + 1
            while k_idx < len(neighbors_a) and j < len(neighbors_b):
                c_a = neighbors_a[k_idx]
                c_b = neighbors_b[j]
                
                if c_a == c_b:
                    # Found triangle (a, b, c_a)
                    print(a, b, c_a)
                    return
                elif c_a < c_b:
                    k_idx += 1
                else:
                    j += 1
    
    # Fallback - should never reach here due to problem guarantee
    # But let's do a brute force just in case
    edge_set = set(edges)
    for a in range(1, n+1):
        if a not in adj_list:
            continue
        for b in adj_list[a]:
            if b <= a:
                continue
            for c in adj_list[b]:
                if c <= b:
                    continue
                if c in adj_list[a]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
