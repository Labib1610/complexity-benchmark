import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    graph = [set() for _ in range(n + 1)]
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        graph[u].add(v)
        graph[v].add(u)
    
    # Sort edges to process in order
    edges.sort()
    
    # For each node, sort its neighbors
    for i in range(1, n + 1):
        graph[i] = sorted(graph[i])
    
    # Now we look for a triangle: three nodes a<b<c such that (a,b), (a,c), (b,c) exist
    # We'll iterate through all edges (a,b) with a<b, and for each, 
    # we look for a node c > b such that (a,c) and (b,c) also exist.
    
    for a in range(1, n + 1):
        for b in graph[a]:
            if b <= a:
                continue
            # Now we want c such that c > b and (a,c) and (b,c) exist
            # To do this efficiently, we intersect neighbors of a and b,
            # and look for elements > b
            neighbors_a = set(graph[a])
            neighbors_b = set(graph[b])
            common_neighbors = neighbors_a & neighbors_b
            for c in common_neighbors:
                if c > b:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
