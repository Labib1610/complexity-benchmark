```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    edges.sort()
    for u, v in edges:
        au = adj[u]
        av = adj[v]
        if len(au) <= len(av):
            for w in au:
                if w > v and w in av:
                    sys.stdout.write(f"{u} {v} {w}")
                    return
        else:
            for w in av:
                if w > v and w in au:
                    sys.stdout.write(f"{u} {v} {w}")
                    return

solve()
```
