import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    best = None

    # For each edge (u, v) with u < v, find smallest w > v adjacent to both.
    for u, v in edges:
        au = adj[u]
        av = adj[v]
        if len(au) > len(av):
            small, large = av, au
        else:
            small, large = au, av
        # Iterate over neighbors of the smaller-degree vertex that are > v.
        # We want the smallest such w, so scan candidates and take min.
        # To be efficient, iterate the smaller set.
        min_w = None
        if len(small) <= len(large) // 20:
            # small is much smaller: iterate small
            for w in small:
                if w > v and w in large:
                    if min_w is None or w < min_w:
                        min_w = w
        else:
            # iterate larger set but only those > v; still could be big.
            # Fallback: iterate small anyway (it's the smaller one).
            for w in small:
                if w > v and w in large:
                    if min_w is None or w < min_w:
                        min_w = w
        if min_w is not None:
            cand = (u, v, min_w)
            if best is None or cand < best:
                best = cand
                # Since we process edges in input order, we cannot break early
                # because a later edge might have smaller u. But we can prune:
                # if best[0] == 1 and best[1] == 2 and best[2] == 3, it's optimal.
                if best == (1, 2, 3):
                    print(1, 2, 3)
                    return

    if best is not None:
        print(best[0], best[1], best[2])
        return

    # Fallback: brute force over all triples (should not be needed given guarantee).
    for a in range(1, n + 1):
        sa = adj[a]
        for b in sa:
            if b <= a:
                continue
            sb = adj[b]
            for c in sb:
                if c > b and c in sa:
                    print(a, b, c)
                    return

solve()
