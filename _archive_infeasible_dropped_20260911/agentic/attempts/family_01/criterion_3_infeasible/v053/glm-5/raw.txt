import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to process in order (u, v) lexicographically
    edges.sort()
    
    best = None
    
    # Strategy 1: Check triangles that include an edge
    # For each edge (u, v), check if there's a common neighbor w
    # We want the lexicographically smallest triangle (a, b, c)
    # Process edges in sorted order, and for each edge, check neighbors in sorted order
    
    for u, v in edges:
        # We want to find w such that w is a common neighbor of u and v
        # and w > v (to ensure u < v < w)
        # Iterate through neighbors of u that are > v
        for w in sorted(adj[u]):
            if w > v and w in adj[v]:
                candidate = (u, v, w)
                if best is None or candidate < best:
                    best = candidate
                    # Since we process edges in order and neighbors in order,
                    # the first valid triangle found is the best
                    print(u, v, w)
                    return
    
    # Strategy 2: If no triangle found with edge-based approach,
    # check all triples (this handles cases where the triangle might not
    # be found efficiently by the edge approach)
    # But since we need the smallest lexicographic triple, we check in order
    
    for a in range(1, n - 1):
        for b in range(a + 1, n):
            if b not in adj[a]:
                continue
            for c in range(b + 1, n + 1):
                if c in adj[a] and c in adj[b]:
                    print(a, b, c)
                    return
    
    # This should never be reached given problem constraints
    if best:
        print(best[0], best[1], best[2])

if __name__ == "__main__":
    solve()
