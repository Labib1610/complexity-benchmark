```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    
    best = None
    for a in range(1, n + 1):
        if best is not None and a >= best[0]:
            break
        adj_a = adj[a]
        for b in adj_a:
            if b <= a:
                continue
            if best is not None and (a > best[0] or (a == best[0] and b >= best[1])):
                break
            adj_b = adj[b]
            for c in adj_a:
                if c <= b:
                    continue
                if best is not None and (a > best[0] or (a == best[0] and b > best[1]) or (a == best[0] and b == best[1] and c >= best[2])):
                    break
                if c in adj_b:
                    best = (a, b, c)
                    break
            if best is not None and best[0] == a and best[1] == b:
                break
        if best is not None and best[0] == a:
            break
    
    sys.stdout.write(f"{best[0]} {best[1]} {best[2]}\n")

solve()
```
