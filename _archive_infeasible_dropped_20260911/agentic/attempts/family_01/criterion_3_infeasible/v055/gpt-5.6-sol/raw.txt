```python
import sys

data = list(map(int, sys.stdin.buffer.read().split()))
n, m = data[0], data[1]

bits = [1 << i for i in range(n)]
adj = [0] * n

p = 2
for _ in range(m):
    u = data[p] - 1
    v = data[p + 1] - 1
    p += 2
    adj[u] |= bits[v]
    adj[v] |= bits[u]

for a in range(n - 2):
    row = adj[a]
    for b in range(a + 1, n - 1):
        if row & bits[b]:
            common = (row & adj[b]) >> (b + 1)
            if common:
                c = b + (common & -common).bit_length()
                print(a + 1, b + 1, c + 1)
                sys.exit()
```
