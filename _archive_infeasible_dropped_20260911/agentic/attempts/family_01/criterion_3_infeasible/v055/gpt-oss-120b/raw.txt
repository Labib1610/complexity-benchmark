```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj_bits = [0] * (n + 1)          # bitset adjacency
    neigh = [[] for _ in range(n + 1)]  # neighbor lists

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        # store undirected edge
        adj_bits[u] |= 1 << (v - 1)
        adj_bits[v] |= 1 << (u - 1)
        neigh[u].append(v)
        neigh[v].append(u)

    # sort neighbor lists for deterministic order
    for i in range(1, n + 1):
        neigh[i].sort()

    for a in range(1, n + 1):
        for b in neigh[a]:
            if b <= a:
                continue
            common = adj_bits[a] & adj_bits[b]
            # keep only vertices with index > b (i.e., node number > b)
            mask = common >> b
            if mask:
                low = mask & -mask
                k = low.bit_length() - 1   # position after shift
                c = b + 1 + k
                # ensure a<b<c
                if c > b:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
```
