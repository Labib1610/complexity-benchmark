import sys
from collections import defaultdict

def find_deadlock_triplet():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency set for O(1) lookup
    adj = defaultdict(set)
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to process in order
    edges.sort()
    
    # For each edge (u,v), look for a third node w such that 
    # (u,w) and (v,w) are also edges
    for u, v in edges:
        if u >= v:
            continue
        # Check common neighbors of u and v
        # Iterate through neighbors of the smaller degree node
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
            
        for w in adj[u]:
            if w != v and w in adj[v]:
                # Found triangle: u-v, u-w, v-w
                # Ensure output is sorted
                result = sorted([u, v, w])
                print(result[0], result[1], result[2])
                return

find_deadlock_triplet()
