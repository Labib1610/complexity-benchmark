```python
import sys


def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return

    n = int(data[0])
    m = int(data[1])

    adj_mask = [0] * (n + 1)

    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        adj_mask[u] |= 1 << v
        adj_mask[v] |= 1 << u

    for a in range(1, n + 1):
        mask_a = adj_mask[a]
        for b in range(a + 1, n + 1):
            if (mask_a >> b) & 1:
                common = (mask_a & adj_mask[b]) >> (b + 1)
                if common:
                    c = b + 1 + (common & -common).bit_length() - 1
                    print(f"{a} {b} {c}")
                    return


if __name__ == "__main__":
    solve()
```
