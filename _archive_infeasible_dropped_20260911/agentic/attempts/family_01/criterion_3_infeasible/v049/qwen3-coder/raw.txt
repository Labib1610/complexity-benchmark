```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for deterministic iteration
    for key in adj:
        adj[key] = sorted(list(adj[key]))
    
    # We want to find a triangle (a,b,c) with a<b<c such that all three edges exist
    # and it's lexicographically smallest
    
    # Iterate through all nodes as potential 'a'
    for a in range(1, n+1):
        if a not in adj:
            continue
        # For each neighbor b of a where b > a
        for i, b in enumerate(adj[a]):
            if b <= a:
                continue
            # Now look for c > b such that both (b,c) and (a,c) exist
            # To optimize, we iterate through the intersection of adj[a] and adj[b]
            # But since lists are sorted, we can do a merge-like process
            neighbors_a = adj[a]
            neighbors_b = adj[b]
            
            i_a = 0
            i_b = 0
            
            while i_a < len(neighbors_a) and i_b < len(neighbors_b):
                cand_a = neighbors_a[i_a]
                cand_b = neighbors_b[i_b]
                
                if cand_a == cand_b:
                    c = cand_a
                    if c > b:
                        print(a, b, c)
                        return
                    i_a += 1
                    i_b += 1
                elif cand_a < cand_b:
                    i_a += 1
                else:
                    i_b += 1

if __name__ == "__main__":
    main()
```
