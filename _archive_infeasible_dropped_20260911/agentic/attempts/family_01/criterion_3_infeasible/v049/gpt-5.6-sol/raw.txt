import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]

    adjacency = [0] * n
    pos = 2
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        if u != v:
            adjacency[u] |= 1 << v
            adjacency[v] |= 1 << u

    suffix = [0] * (n + 1)
    mask = 0
    for i in range(n - 1, -1, -1):
        mask |= 1 << i
        suffix[i] = mask

    for a in range(n):
        remaining_b = adjacency[a] & suffix[a + 1]
        while remaining_b:
            b_bit = remaining_b & -remaining_b
            b = b_bit.bit_length() - 1
            common = adjacency[a] & adjacency[b] & suffix[b + 1]
            if common:
                c = (common & -common).bit_length() - 1
                print(a + 1, b + 1, c + 1)
                return
            remaining_b ^= b_bit

if __name__ == "__main__":
    main()
