import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]

    adj = [0] * (n + 1)
    pos = 2
    for _ in range(m):
        u, v = data[pos], data[pos + 1]
        pos += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for a in range(1, n + 1):
        aa = adj[a]
        candidates = aa & ~((1 << (a + 1)) - 1)

        while candidates:
            bit = candidates & -candidates
            b = bit.bit_length() - 1
            candidates ^= bit

            common = (aa & adj[b]) >> (b + 1)
            if common:
                c = b + 1 + (common & -common).bit_length() - 1
                print(a, b, c)
                return

if __name__ == "__main__":
    main()
