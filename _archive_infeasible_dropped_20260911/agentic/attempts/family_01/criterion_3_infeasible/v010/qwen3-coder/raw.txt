import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    edge_set = set()
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        edge_set.add((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort vertices by degree descending for optimization
    vertices = list(range(1, n+1))
    
    # For each vertex u, check pairs of its neighbors
    for u in range(1, n+1):
        neighbors = sorted(list(adj[u]))
        for i in range(len(neighbors)):
            v = neighbors[i]
            for j in range(i+1, len(neighbors)):
                w = neighbors[j]
                # Check if v and w are connected
                if (v, w) in edge_set or (w, v) in edge_set:
                    # Found a triangle: u, v, w
                    triple = sorted([u, v, w])
                    print(triple[0], triple[1], triple[2])
                    return

if __name__ == "__main__":
    main()
