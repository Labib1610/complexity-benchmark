```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    edges.sort()
    best = None
    for u, v in edges:
        au = adj[u]
        av = adj[v]
        if len(au) > len(av):
            au, av = av, au
        common = au & av
        if common:
            w = min(common)
            if best is None or (u, v, w) < best:
                best = (u, v, w)
    if best is not None:
        sys.stdout.write(f"{best[0]} {best[1]} {best[2]}\n")
        return
    # Fallback (should not be reached due to problem guarantees)
    for u in range(1, n + 1):
        au = adj[u]
        for v in au:
            if v > u:
                common = au & adj[v]
                if common:
                    w = min(common)
                    sys.stdout.write(f"{u} {v} {w}\n")
                    return

solve()
```
