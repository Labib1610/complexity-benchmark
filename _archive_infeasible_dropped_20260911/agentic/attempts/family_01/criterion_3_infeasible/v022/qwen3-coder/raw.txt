```python
import sys
from collections import defaultdict

def find_triple_alliance():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list and edge set for O(1) lookup
    adj = defaultdict(set)
    edges = set()
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.add((u, v))
    
    # Convert sets to lists and sort adjacency lists for consistent ordering
    for node in adj:
        adj[node] = sorted(list(adj[node]))
    
    # Look for triangles by checking each edge
    for u in range(1, n + 1):
        if u not in adj:
            continue
        for v in adj[u]:
            if v <= u:
                continue
            # Check neighbors of u and v to find common neighbor
            for w in adj[u]:
                if w <= v:
                    continue
                if w in adj[v]:
                    # Found triangle u-v-w
                    result = [u, v, w]
                    result.sort()
                    print(result[0], result[1], result[2])
                    return

find_triple_alliance()
```
