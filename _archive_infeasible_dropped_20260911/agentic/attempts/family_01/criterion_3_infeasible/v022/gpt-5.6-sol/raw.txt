```python
import sys

data = list(map(int, sys.stdin.buffer.read().split()))
n, m = data[0], data[1]

adj = [0] * n
idx = 2
for _ in range(m):
    u = data[idx] - 1
    v = data[idx + 1] - 1
    idx += 2
    adj[u] |= 1 << v
    adj[v] |= 1 << u

for a in range(n - 2):
    candidates = adj[a] >> (a + 1)

    while candidates:
        lowest = candidates & -candidates
        offset = lowest.bit_length() - 1
        b = a + 1 + offset

        common = (adj[a] & adj[b]) >> (b + 1)
        if common:
            c_offset = (common & -common).bit_length() - 1
            c = b + 1 + c_offset
            print(a + 1, b + 1, c + 1)
            sys.exit()

        candidates ^= lowest
```
