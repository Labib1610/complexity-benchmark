```python
import sys

def main():
    input = sys.stdin.buffer.readline
    n, m = map(int, input().split())
    adj = [0] * n

    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for a in range(n - 2):
        candidates = adj[a] >> (a + 1)
        while candidates:
            low = candidates & -candidates
            b = a + low.bit_length()
            candidates ^= low

            common = (adj[a] & adj[b]) >> (b + 1)
            if common:
                c = b + (common & -common).bit_length()
                print(a + 1, b + 1, c + 1)
                return

if __name__ == "__main__":
    main()
```
