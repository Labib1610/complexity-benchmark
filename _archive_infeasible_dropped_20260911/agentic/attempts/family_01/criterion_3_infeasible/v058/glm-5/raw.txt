import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Iterate to find the lexicographically smallest triangle
    for a in range(1, n - 1):
        neighbors_a = adj[a]
        # We need b > a and c > b.
        # Iterate through neighbors of a that are greater than a.
        possible_b = [v for v in neighbors_a if v > a]
        possible_b.sort()
        
        for b in possible_b:
            # Check if b connects to some c > b that also connects to a
            neighbors_b = adj[b]
            # We need c in neighbors_a AND c in neighbors_b AND c > b
            # Iterate over the smaller set for efficiency
            if len(neighbors_a) < len(neighbors_b):
                search_set = neighbors_a
            else:
                search_set = neighbors_b
            
            possible_c = [v for v in search_set if v > b and v in neighbors_a and v in neighbors_b]
            
            if possible_c:
                possible_c.sort()
                c = possible_c[0]
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
