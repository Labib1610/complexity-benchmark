We can solve this by checking all triples in lexicographic order and stopping at the first fully connected one.  
Since \( n \le 2000 \), a naive triple loop would be \( O(n^3) \), which is too slow.  
Instead, we can iterate over the first two nodes and check their common neighbors.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b not in adj[a]:
                continue
            common = adj[a] & adj[b]
            for c in sorted(common):
                if c > b:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
```
