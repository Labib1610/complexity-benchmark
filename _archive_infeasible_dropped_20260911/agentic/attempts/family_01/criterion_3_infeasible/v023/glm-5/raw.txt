```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for lexicographic order
    sorted_adj = [sorted(s) for s in adj]
    
    for a in range(1, n - 1):
        # Get neighbors of a that are > a
        neighbors_a = [b for b in sorted_adj[a] if b > a]
        for b in neighbors_a:
            # Get neighbors of b that are > b
            neighbors_b = [c for c in sorted_adj[b] if c > b]
            for c in neighbors_b:
                if c in adj[a]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
```
