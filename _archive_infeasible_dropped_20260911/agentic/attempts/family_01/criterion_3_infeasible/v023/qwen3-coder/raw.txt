```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to help with finding triangles
    edges.sort()
    
    # For each node, sort its neighbors
    for node in adj:
        adj[node] = sorted(adj[node])
    
    # We'll iterate through all edges and look for a common neighbor
    min_triple = None
    
    # To optimize, we can iterate through all edges and for each edge (u,v),
    # look for a node w such that both (u,w) and (v,w) exist
    for u, v in edges:
        # We want to find a w such that u-w and v-w exist
        # To optimize, iterate through the smaller adjacency list
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        
        for w in adj[u]:
            if w > v:  # To ensure a < b < c in the end
                if w in adj[v]:
                    # Found a triangle: u, v, w
                    # u < v is guaranteed by edge sorting, and w > v
                    # So we have u < v < w
                    triple = (u, v, w)
                    if min_triple is None or triple < min_triple:
                        min_triple = triple
                # Since adj[u] is sorted, and w > v, we can break if w becomes too large
                # but since we want the lexicographically smallest, we continue
    
    if min_triple:
        print(min_triple[0], min_triple[1], min_triple[2])

if __name__ == "__main__":
    main()
```
