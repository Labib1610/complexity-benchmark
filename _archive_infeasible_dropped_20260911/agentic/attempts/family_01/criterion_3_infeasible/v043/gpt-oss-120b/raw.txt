import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_bits = [0] * (n + 1)          # bit i corresponds to vertex i+1
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj_bits[u] |= 1 << (v - 1)
        adj_bits[v] |= 1 << (u - 1)

    for a in range(1, n - 1):
        bits_a = adj_bits[a]
        for b in range(a + 1, n):
            if (bits_a >> (b - 1)) & 1 == 0:
                continue          # no edge a-b
            common = bits_a & adj_bits[b]
            # mask out vertices <= b
            mask = (1 << b) - 1    # bits 0..b-1 set
            common &= ~mask
            if common:
                # lowest set bit gives smallest c
                low = common & -common
                c = low.bit_length()          # 1‑based index
                print(a, b, c)
                return

if __name__ == "__main__":
    solve()
