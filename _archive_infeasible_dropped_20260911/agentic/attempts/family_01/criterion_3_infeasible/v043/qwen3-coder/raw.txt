import sys
from collections import defaultdict

def find_deadlock():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(set)
    edges = set()
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].add(v)
        adj[v].add(u)
        edges.add((min(u, v), max(u, v)))
    
    # Convert to sorted lists for consistent iteration
    for u in range(1, n + 1):
        adj[u] = sorted(list(adj[u]))
    
    # Look for triangle (deadlock of 3 employees)
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c <= b:
                    continue
                # Check if b and c are also connected (forming a triangle)
                if (min(b, c), max(b, c)) in edges:
                    print(a, b, c)
                    return

find_deadlock()
