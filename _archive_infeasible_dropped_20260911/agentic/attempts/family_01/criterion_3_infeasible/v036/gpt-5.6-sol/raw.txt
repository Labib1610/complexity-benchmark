```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]

    adjacency = [0] * n
    forward = [[] for _ in range(n)]

    pos = 2
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2

        if u > v:
            u, v = v, u

        adjacency[u] |= 1 << v
        adjacency[v] |= 1 << u
        forward[u].append(v)

    for a in range(n):
        forward[a].sort()
        for b in forward[a]:
            common = (adjacency[a] & adjacency[b]) >> (b + 1)
            if common:
                c = b + 1 + (common & -common).bit_length() - 1
                print(a + 1, b + 1, c + 1)
                return

if __name__ == "__main__":
    main()
```
