import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    graph = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        edges.append((u, v))
        graph[u].add(v)
        graph[v].add(u)
    
    # We need to find a triangle (three nodes all connected to each other)
    # That means we need three nodes a, b, c such that:
    # (a,b), (a,c), and (b,c) are all edges
    
    # To find the lexicographically smallest such triple:
    # Iterate through all edges (u,v) with u < v
    # For each such edge, look for a node w such that:
    # w > v (to maintain a < b < c)
    # and (u,w) and (v,w) are also edges
    
    # Let's sort the adjacency list for each node to help with efficiency
    for i in range(1, n+1):
        graph[i] = sorted(graph[i])
    
    # Create a set of edges for O(1) lookup
    edge_set = set()
    for u, v in edges:
        if u > v:
            u, v = v, u
        edge_set.add((u, v))
    
    # Now iterate through all possible triples
    # But do it smartly: for each edge (a,b) with a<b, 
    # check for each c > b if (a,c) and (b,c) exist
    
    for a in range(1, n - 1):
        for b in range(a + 1, n):
            if b not in graph[a]:
                continue
            # Now we have edge (a,b). Look for c > b such that (a,c) and (b,c) exist
            # To optimize, intersect neighbors of a and b that are > b
            neighbors_a = set(x for x in graph[a] if x > b)
            neighbors_b = set(x for x in graph[b] if x > b)
            common = sorted(list(neighbors_a & neighbors_b))
            if common:
                c = common[0]
                print(a, b, c)
                return

if __name__ == "__main__":
    main()
