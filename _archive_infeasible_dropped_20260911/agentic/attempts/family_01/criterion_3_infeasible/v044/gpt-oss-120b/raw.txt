```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_int = [0] * (n + 1)
    neigh = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        # store undirected
        adj_int[u] |= 1 << (v - 1)
        adj_int[v] |= 1 << (u - 1)
        neigh[u].append(v)
        neigh[v].append(u)

    # sort neighbor lists for deterministic order
    for i in range(1, n + 1):
        neigh[i].sort()

    full_mask = (1 << n) - 1
    for a in range(1, n + 1):
        mask_a = adj_int[a]
        for b in neigh[a]:
            if b <= a:
                continue
            # common neighbors of a and b
            common = mask_a & adj_int[b]
            # keep only vertices > b
            mask_gt_b = full_mask ^ ((1 << b) - 1)   # bits for indices >= b
            common &= mask_gt_b
            if common:
                # get smallest vertex w
                lsb = common & -common
                w = lsb.bit_length()  # because bits are 0-indexed for vertex-1
                print(a, b, w)
                return

if __name__ == "__main__":
    solve()
```
