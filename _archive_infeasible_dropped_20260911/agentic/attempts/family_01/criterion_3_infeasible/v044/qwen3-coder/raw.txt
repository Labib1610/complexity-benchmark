import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        edges.append((u, v))
        index += 2
    
    # Build adjacency list using sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # Collect all nodes that have at least two connections
    # Because any triangle must have each node connected to two others
    candidates = []
    for i in range(1, n + 1):
        if len(adj[i]) >= 2:
            candidates.append(i)
    
    # Sort candidates to ensure we check in lexicographical order
    candidates.sort()
    
    # For each candidate node 'a', look for a triangle containing 'a'
    for a in candidates:
        # Get neighbors of 'a' and sort them
        neighbors_of_a = sorted(list(adj[a]))
        # Check all pairs of neighbors of 'a' to see if they are connected
        for i in range(len(neighbors_of_a)):
            b = neighbors_of_a[i]
            if b <= a:
                continue  # Ensure a < b
            for j in range(i+1, len(neighbors_of_a)):
                c = neighbors_of_a[j]
                if c <= b:
                    continue  # Ensure b < c
                # Check if b and c are connected
                if c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
