```python
import sys

data = list(map(int, sys.stdin.buffer.read().split()))
n, m = data[0], data[1]

adj = [0] * (n + 1)
upper = [[] for _ in range(n + 1)]

idx = 2
for _ in range(m):
    u, v = data[idx], data[idx + 1]
    idx += 2
    if u > v:
        u, v = v, u
    adj[u] |= 1 << v
    adj[v] |= 1 << u
    upper[u].append(v)

for a in range(1, n + 1):
    upper[a].sort()
    for b in upper[a]:
        common = adj[a] & adj[b] & ~((1 << (b + 1)) - 1)
        if common:
            c = (common & -common).bit_length() - 1
            print(a, b, c)
            sys.exit()
```
