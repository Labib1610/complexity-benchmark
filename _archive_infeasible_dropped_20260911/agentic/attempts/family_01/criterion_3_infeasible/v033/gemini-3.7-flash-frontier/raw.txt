```python
import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])

    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for u in range(1, n + 1):
        mask = adj[u] >> (u + 1)
        while mask > 0:
            lsb = mask & -mask
            v = u + 1 + lsb.bit_length() - 1
            mask &= mask - 1

            common = (adj[u] & adj[v]) >> (v + 1)
            if common:
                lsb_c = common & -common
                w = v + 1 + lsb_c.bit_length() - 1
                print(f"{u} {v} {w}")
                return


if __name__ == "__main__":
    solve()
```
