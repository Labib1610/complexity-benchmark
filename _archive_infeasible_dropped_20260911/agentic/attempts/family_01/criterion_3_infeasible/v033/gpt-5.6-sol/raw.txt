```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]

    adj = [[] for _ in range(n)]
    bits = [0] * n

    pos = 2
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2

        if u == v:
            continue

        adj[u].append(v)
        adj[v].append(u)
        bits[u] |= 1 << v
        bits[v] |= 1 << u

    for neighbors in adj:
        neighbors.sort()

    for a in range(n):
        for b in adj[a]:
            if b <= a:
                continue

            common = (bits[a] & bits[b]) >> (b + 1)
            if common:
                lowest = common & -common
                c = b + 1 + lowest.bit_length() - 1
                print(a + 1, b + 1, c + 1)
                return

if __name__ == "__main__":
    main()
```
