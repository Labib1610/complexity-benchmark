import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    best = None
    
    # Case 1: a vertex with degree >= 2*sqrt(m)
    threshold = 2 * int(m ** 0.5) + 2
    for a in range(1, n + 1):
        if len(adj[a]) >= threshold:
            neighbors = sorted(adj[a])
            L = len(neighbors)
            for i in range(L):
                b = neighbors[i]
                if best is not None and b >= best[1]:
                    break
                for j in range(i + 1, L):
                    c = neighbors[j]
                    if best is not None and c >= best[2]:
                        break
                    if b in adj[c]:
                        if best is None or (a, b, c) < best:
                            best = (a, b, c)
                        break  # smallest c for this (a, b) found
    
    if best is not None:
        print(best[0], best[1], best[2])
        return
    
    # Case 2: low-degree graph, iterate edges
    for u, v in edges:
        if best is not None and u >= best[0]:
            break
        if best is not None and v >= best[1]:
            continue
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        for w in adj[u]:
            if w > v and w in adj[v]:
                if best is None or (u, v, w) < best:
                    best = (u, v, w)
                break  # smallest w for this (u, v) found
    
    print(best[0], best[1], best[2])

solve()
