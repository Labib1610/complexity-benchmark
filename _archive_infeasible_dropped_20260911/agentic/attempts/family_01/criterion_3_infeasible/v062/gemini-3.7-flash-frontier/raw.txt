```python
import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])

    adj = [0] * (n + 1)
    adj_list = [[] for _ in range(n + 1)]

    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        if u < v:
            adj_list[u].append(v)
        else:
            adj_list[v].append(u)

    for a in range(1, n + 1):
        adj_list[a].sort()

    for a in range(1, n + 1):
        for b in adj_list[a]:
            common = (adj[a] & adj[b]) >> (b + 1)
            if common:
                lowbit = common & -common
                c = lowbit.bit_length() + b
                print(f"{a} {b} {c}")
                return


if __name__ == "__main__":
    solve()
```
