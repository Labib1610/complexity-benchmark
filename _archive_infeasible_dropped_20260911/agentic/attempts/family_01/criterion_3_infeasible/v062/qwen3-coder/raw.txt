```python
import sys
from collections import defaultdict

def find_triangle(edges, n):
    # Build adjacency list
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # For each node, check its neighbors
    for a in range(1, n + 1):
        if a not in adj:
            continue
        neighbors = sorted(adj[a])
        # Check every pair of neighbors of 'a'
        for i in range(len(neighbors)):
            b = neighbors[i]
            if b <= a:
                continue
            for j in range(i + 1, len(neighbors)):
                c = neighbors[j]
                if c <= a:
                    continue
                # Check if b and c are connected
                if c in adj[b]:
                    return (a, b, c)
    return None

# Read input
input_lines = sys.stdin.read().splitlines()
n, m = map(int, input_lines[0].split())
edges = [tuple(map(int, line.split())) for line in input_lines[1:]]

# Find triangle
result = find_triangle(edges, n)
print(*result)
```
