import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]

    adjacency = [0] * n
    pos = 2
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        adjacency[u] |= 1 << v
        adjacency[v] |= 1 << u

    full_mask = (1 << n) - 1
    higher = [full_mask ^ ((1 << (i + 1)) - 1) for i in range(n)]

    for a in range(n):
        b_candidates = adjacency[a] & higher[a]
        while b_candidates:
            b_bit = b_candidates & -b_candidates
            b = b_bit.bit_length() - 1

            common = adjacency[a] & adjacency[b] & higher[b]
            if common:
                c = (common & -common).bit_length() - 1
                print(a + 1, b + 1, c + 1)
                return

            b_candidates ^= b_bit

if __name__ == "__main__":
    main()
