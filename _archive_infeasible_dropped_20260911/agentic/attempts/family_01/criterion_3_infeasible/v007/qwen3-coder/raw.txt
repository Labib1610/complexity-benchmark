import sys
from collections import defaultdict

def find_triangle(edges, adj):
    # Iterate through all edges to find a triangle
    for u in sorted(adj.keys()):
        for v in sorted(adj[u]):
            if u >= v:
                continue
            # Check for common neighbors of u and v
            for w in sorted(adj[u]):
                if w <= v:
                    continue
                if w in adj[v]:
                    return [u, v, w]
    return None

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    index = 0
    n = int(data[index])

    index += 2
    
    adj = defaultdict(set)
    edge_list = []
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edge_list.append((u, v))
    
    # Sort adjacency lists for consistent ordering
    for key in adj:
        adj[key] = sorted(list(adj[key]))
        
    # Find triangle
    result = find_triangle(edge_list, adj)
    if result:
        print(result[0], result[1], result[2])
