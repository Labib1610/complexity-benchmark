```python
import sys

data = list(map(int, sys.stdin.buffer.read().split()))
n, m = data[0], data[1]

adj = [0] * (n + 1)
pos = 2
for _ in range(m):
    u, v = data[pos], data[pos + 1]
    pos += 2
    adj[u] |= 1 << v
    adj[v] |= 1 << u

higher = [-(1 << (i + 1)) for i in range(n + 1)]

for a in range(1, n - 1):
    adj_a = adj[a]
    candidates = adj_a & higher[a]

    while candidates:
        bit_b = candidates & -candidates
        b = bit_b.bit_length() - 1

        common = adj_a & adj[b] & higher[b]
        if common:
            bit_c = common & -common
            c = bit_c.bit_length() - 1
            print(a, b, c)
            sys.exit()

        candidates ^= bit_b
```
