import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    deg = [0] * (n + 1)
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u][v] = True
        adj[v][u] = True
        deg[u] += 1
        deg[v] += 1
    
    # Iterate over all possible triplets in lexicographical order
    for a in range(1, n - 1):
        for b in range(a + 1, n):
            if not adj[a][b]:
                continue
            for c in range(b + 1, n + 1):
                if adj[a][c] and adj[b][c]:
                    print(a, b, c)
                    return
    
    # Fallback: if not found by naive iteration (should not happen per problem guarantee),
    # but theoretically possible if naive iteration is too slow.
    # This part uses the degree-based optimization.
    # We check nodes with degree >= 2.
    
    # Re-iterate with optimization
    for a in range(1, n - 1):
        if deg[a] < 2:
            continue
        for b in range(a + 1, n):
            if not adj[a][b]:
                continue
            for c in range(b + 1, n + 1):
                if adj[a][c] and adj[b][c]:
                    print(a, b, c)
                    return

if __name__ == '__main__':
    solve()
