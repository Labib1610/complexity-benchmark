```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    neigh = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        neigh[u].append(v)
        neigh[v].append(u)

    # sort neighbor lists for deterministic order
    for lst in neigh:
        lst.sort()

    full_mask = (1 << (n + 1)) - 1  # bits 0..n

    for a in range(1, n + 1):
        for b in neigh[a]:
            if b <= a:
                continue
            # common neighbors of a and b
            common = adj[a] & adj[b]
            # keep only vertices > b
            mask = full_mask ^ ((1 << (b + 1)) - 1)
            common &= mask
            if common:
                # lowest set bit gives smallest c
                low_bit = common & -common
                c = low_bit.bit_length() - 1
                print(a, b, c)
                return

if __name__ == "__main__":
    solve()
```
