```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    
    best = None
    for a in range(1, n + 1):
        if best is not None and a >= best[0]:
            break
        adj_a = adj[a]
        for b in adj_a:
            if b <= a:
                continue
            if best is not None and (a > best[0] or (a == best[0] and b >= best[1])):
                break
            adj_b = adj[b]
            common = adj_a & adj_b
            for c in common:
                if c > b:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
                    break
    
    if best is not None:
        sys.stdout.write(f"{best[0]} {best[1]} {best[2]}\n")

solve()
```
