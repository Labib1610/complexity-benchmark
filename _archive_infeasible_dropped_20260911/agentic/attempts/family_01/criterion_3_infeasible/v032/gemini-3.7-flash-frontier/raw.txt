```python
import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    adj = [0] * (n + 1)
    adj_list = [[] for _ in range(n + 1)]

    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        if u < v:
            adj_list[u].append(v)
        else:
            adj_list[v].append(u)

    for u in range(1, n + 1):
        adj_list[u].sort()

    for a in range(1, n + 1):
        mask_a = adj[a]
        for b in adj_list[a]:
            common = mask_a & adj[b] & ~((1 << (b + 1)) - 1)
            if common:
                c = (common & -common).bit_length() - 1
                print(f"{a} {b} {c}")
                return


if __name__ == "__main__":
    solve()
```
