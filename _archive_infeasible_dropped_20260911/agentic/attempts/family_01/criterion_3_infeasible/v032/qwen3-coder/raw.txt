```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = set()
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.add((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Convert to sorted lists for deterministic iteration
    for u in range(1, n+1):
        adj[u] = sorted(list(adj[u]))
    
    # Look for a triangle (three nodes all connected to each other)
    for a in range(1, n-1):
        for b in adj[a]:
            if b <= a:
                continue
            for c in adj[a]:
                if c <= b:
                    continue
                # Check if b and c are also connected
                if c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
