We can solve this by storing the incompatible pairs in an adjacency set for fast lookup, then trying all triples in lexicographic order until we find a triangle.  

```python
import sys

def main():
    data = sys.stdin.read().split()
    n = int(data[0])
    m = int(data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
    
    for a in range(1, n - 1):
        for b in range(a + 1, n):
            if b not in adj[a]:
                continue
            common = adj[a] & adj[b]
            if common:
                c = min(common)
                if c > b:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
