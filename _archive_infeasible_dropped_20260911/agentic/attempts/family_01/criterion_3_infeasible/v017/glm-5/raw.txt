import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    idx = 2
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx + 1]); idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    
    best = None
    for a in range(1, n + 1):
        neighbors = sorted(adj[a])
        # Only consider neighbors greater than a to avoid duplicates and ensure a < b < c
        candidates = [x for x in neighbors if x > a]
        # Early termination: if best exists and a >= best[0], we can't improve
        if best is not None and a >= best[0]:
            break
        L = len(candidates)
        for i in range(L):
            b = candidates[i]
            if best is not None and b >= best[1]:
                continue
            for j in range(i + 1, L):
                c = candidates[j]
                if best is not None and c >= best[2]:
                    continue
                if c in adj[b]:
                    best = (a, b, c)
                    break  # found best for this a, b
            if best is not None and best[0] == a and best[1] == b:
                break  # found best for this a
        if best is not None and best[0] == a:
            break  # found best overall (since a is minimized)
    
    sys.stdout.write(f"{best[0]} {best[1]} {best[2]}\n")

solve()
