We can iterate through the vertices and for each vertex, look for two of its neighbors that are directly connected, forming a triangle. To keep the lexicographically smallest trio, we check potential triples sorted by ID.  

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
    
    # Check triangles
    for a in range(1, n + 1):
        for b in adj[a]:
            if b < a:
                continue
            for c in adj[b]:
                if c < b:
                    continue
                if c in adj[a]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
```
