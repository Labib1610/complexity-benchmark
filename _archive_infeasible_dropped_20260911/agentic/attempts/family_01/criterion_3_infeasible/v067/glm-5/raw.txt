```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    best = None
    
    for u, v in edges:
        # Find common neighbors to form triangles
        # Iterate through the smaller adjacency set for efficiency
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        
        for w in adj[u]:
            if w != v and w in adj[v]:
                trio = sorted([u, v, w])
                if best is None or trio < best:
                    best = trio
    
    print(best[0], best[1], best[2])

if __name__ == "__main__":
    solve()
```
