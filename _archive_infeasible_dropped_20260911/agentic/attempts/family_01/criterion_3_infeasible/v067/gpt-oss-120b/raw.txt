```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)          # bitset, node i -> bit i-1
    neigh = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        # set bits
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)
        neigh[u].append(v)
        neigh[v].append(u)

    # sort neighbor lists for deterministic order
    for i in range(1, n + 1):
        neigh[i].sort()

    full_mask = (1 << n) - 1

    for a in range(1, n + 1):
        for b in neigh[a]:
            if b <= a:
                continue
            common = adj[a] & adj[b]
            # keep only vertices with index > b
            mask = (~((1 << b) - 1)) & full_mask
            t = common & mask
            if t:
                low = t & -t
                c = low.bit_length()          # 1‑based vertex number
                sys.stdout.write(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
```
