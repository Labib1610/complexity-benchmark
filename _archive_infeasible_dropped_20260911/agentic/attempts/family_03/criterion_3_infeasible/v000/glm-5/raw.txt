```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    m = int(data[1])
    has_road = [False] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        has_road[u] = True
        has_road[v] = True
    for i in range(1, n + 1):
        if not has_road[i]:
            print(i)
            return

main()
```
