```python
u, s = map(int, input().split())

# Set to store all skills that are covered by at least one consultant
covered_skills = set()

# Process each consultant
for _ in range(s):
    line = list(map(int, input().split()))
    c = line[0]
    skills = line[1:]
    for skill in skills:
        covered_skills.add(skill)

# Find the smallest uncovered skill
for skill_id in range(1, u + 1):
    if skill_id not in covered_skills:
        print(skill_id)
        break
```
