```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    need = k + 1
    used = [False] * (n + 1)
    ans = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if len(ans) >= need:
            continue
        if not used[u] and not used[v]:
            ans.append((u, v))
            used[u] = used[v] = True
    # According to the statement, we are guaranteed to have enough edges.
    out_lines = ['{} {}'.format(u, v) for u, v in ans[:need]]
    sys.stdout.write('\n'.join(out_lines))

if __name__ == "__main__":
    solve()
```
