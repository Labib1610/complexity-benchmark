We can simply read all costs, find the minimum, and print it in the required format.  

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    n = int(data[0])
    B = int(data[1])
    min_cost = float('inf')
    idx = 2
    for _ in range(n):
        cost = int(data[idx])
        # value = int(data[idx+1])  # not needed
        idx += 2
        if cost < min_cost:
            min_cost = cost
    print(f"MIN {min_cost}")

if __name__ == "__main__":
    main()
```
