```python
import sys

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, B = data[0], data[1]
    min_cost = 10**10  # larger than any possible cost
    idx = 2
    for _ in range(n):
        cost = data[idx]
        # value = data[idx + 1]  # not needed
        if cost < min_cost:
            min_cost = cost
        idx += 2
    # According to the problem statement min_cost > B is guaranteed
    sys.stdout.write(f"MIN {min_cost}")

if __name__ == "__main__":
    solve()
```
