We can check the three conditions in order:  

- If the total sum is odd, print `ODD`.  
- Otherwise, check if any single task is greater than half the total, print `MAX`.  
- Otherwise, find a common divisor of all tasks that does not divide half the total, print `MOD d`.  

Here’s the implementation:  

```python
import sys
import math

def main():
    data = sys.stdin.read().strip().split()
    n = int(data[0])
    d = list(map(int, data[1:]))
    
    total = sum(d)
    
    if total % 2 == 1:
        print("ODD")
        return
    
    half = total // 2
    
    if max(d) > half:
        print("MAX")
        return
    
    g = 0
    for x in d:
        g = math.gcd(g, x)
    
    if g >= 2 and half % g != 0:
        print(f"MOD {g}")
        return
    
    # If none of the above, find any divisor that works
    # Since problem guarantees one applies, we can fallback to checking divisors of g
    for div in range(2, int(g**0.5) + 1):
        if g % div == 0:
            if half % div != 0:
                print(f"MOD {div}")
                return
            other = g // div
            if half % other != 0:
                print(f"MOD {other}")
                return
    
    # g itself should work if half % g != 0
    if half % g != 0:
        print(f"MOD {g}")
        return

if __name__ == "__main__":
    main()
```
