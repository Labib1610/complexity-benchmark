```python
import sys
import math

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    costs = [int(next(it)) for _ in range(n)]
    T = int(next(it))

    total = sum(costs)
    if T > total:
        print("SUM")
        return

    g = 0
    for c in costs:
        g = math.gcd(g, c)
    if g >= 2 and T % g != 0:
        print(f"MOD {g}")
        return

    # According to problem guarantees, we should never reach here.
    # Fallback (should not be needed):
    print("SUM")

if __name__ == "__main__":
    solve()
```
