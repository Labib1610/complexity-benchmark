```python
import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    seen = [False] * (k + 1)
    idx = 2
    for _ in range(t):
        x = int(data[idx])
        idx += 3
        seen[x] = True
    for i in range(1, k + 1):
        if not seen[i]:
            print(i)
            return

main()
```
