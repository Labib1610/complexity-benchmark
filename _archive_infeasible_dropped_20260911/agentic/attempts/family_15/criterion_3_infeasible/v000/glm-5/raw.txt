```python
import sys

def main():
    data = sys.stdin.read().split('\n')
    k = int(data[0])
    seqs = [data[i+1] for i in range(k)]
    s1 = seqs[0]
    distinct = sorted(set(s1))
    sets = [set(seq) for seq in seqs]
    print("BLOCK")
    for c in distinct:
        for j in range(2, k+1):
            if c not in sets[j-1]:
                print(f"{c} {j}")
                break

main()
```
