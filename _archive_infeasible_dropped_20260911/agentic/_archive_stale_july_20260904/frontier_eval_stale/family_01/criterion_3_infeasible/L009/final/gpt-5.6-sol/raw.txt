import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]

    adjacency_bits = [0] * n
    higher_neighbors = [[] for _ in range(n)]

    pos = 2
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2

        adjacency_bits[u] |= 1 << v
        adjacency_bits[v] |= 1 << u

        if u > v:
            u, v = v, u
        higher_neighbors[u].append(v)

    for a in range(n):
        higher_neighbors[a].sort()
        for b in higher_neighbors[a]:
            common = (adjacency_bits[a] & adjacency_bits[b]) >> (b + 1)
            if common:
                offset = (common & -common).bit_length() - 1
                c = b + 1 + offset
                print(a + 1, b + 1, c + 1)
                return

if __name__ == "__main__":
    main()
