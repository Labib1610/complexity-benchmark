import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]

    adj = [0] * (n + 1)
    pos = 2
    for _ in range(m):
        u = data[pos]
        v = data[pos + 1]
        pos += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for a in range(1, n + 1):
        candidates = adj[a] & (-1 << (a + 1))
        while candidates:
            bit_b = candidates & -candidates
            b = bit_b.bit_length() - 1
            common = adj[a] & adj[b] & (-1 << (b + 1))
            if common:
                bit_c = common & -common
                c = bit_c.bit_length() - 1
                print(a, b, c)
                return
            candidates ^= bit_b

if __name__ == "__main__":
    main()
