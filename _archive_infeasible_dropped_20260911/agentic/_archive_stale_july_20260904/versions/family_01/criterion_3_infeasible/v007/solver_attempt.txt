import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        idx += 2
        
    # We need the lexicographically smallest triplet (a, b, c)
    # such that a < b < c and (a,b), (a,c), (b,c) are edges.
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        neighbors_a = sorted(list(adj[a]))
        # Iterate b in neighbors of a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that c is in neighbors_a AND c is in adj[b]
            # To ensure lexicographical order, we check neighbors of b
            # that are also in neighbors_a
            neighbors_b = adj[b]
            
            # Find common neighbors c > b
            common = []
            for c in neighbors_a:
                if c > b and c in neighbors_b:
                    common.append(c)
            
            if common:
                print(f"{a} {b} {min(common)}")
                return

if __name__ == '__main__':
    solve()
