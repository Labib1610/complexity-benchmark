import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triplet
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # all edges (a, b) and check if there exists a node c such that
    # (a, c) and (b, c) are also edges.
    # Since we want the smallest a, then b, then c, we iterate a from 1 to n,
    # then b from a+1 to n, and check if c exists.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through all possible pairs (a, b) that are connected
    for a in range(1, n + 1):
        for b in adj[a]:
            # We need a node c > b such that c is in adj[a] and adj[b]
            # Since adj lists are sorted, we can find common elements efficiently
            set_a = set(adj[a])
            set_b = set(adj[b])
            common = sorted(list(set_a.intersection(set_b)))
            
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
