import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (x, y, z) with x < y < z.
    # To minimize x, then y, then z, we iterate x from 1 to n,
    # then y from x+1 to n, then check if z exists.
    
    for x in range(1, n + 1):
        # Get neighbors of x that are > x
        neighbors_x = [v for v in adj[x] if v > x]
        neighbors_x.sort()
        
        for i in range(len(neighbors_x)):
            y = neighbors_x[i]
            # We need a z such that y < z, z is in adj[x], and z is in adj[y]
            # Intersection of adj[x] and adj[y]
            # To optimize, check the smaller adjacency list
            if len(adj[x]) < len(adj[y]):
                candidates = adj[x]
            else:
                candidates = adj[y]
            
            # Find z > y in the intersection
            found_z = -1
            for z in candidates:
                if z > y and z in adj[x] and z in adj[y]:
                    if found_z == -1 or z < found_z:
                        found_z = z
            
            if found_z != -1:
                print(f"{x} {y} {found_z}")
                return

if __name__ == '__main__':
    solve()
