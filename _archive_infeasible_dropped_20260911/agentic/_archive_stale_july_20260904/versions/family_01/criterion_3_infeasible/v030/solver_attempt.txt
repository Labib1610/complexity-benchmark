import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically first (a, b, c) with a < b < c:
    # Iterate through all possible a from 1 to n-2
    # For each a, iterate through all neighbors b of a such that b > a
    # For each such b, check if there exists a common neighbor c such that c > b
    
    # Sort neighbors to ensure we check in lexicographical order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that c > b and c is in both adj[a] and adj[b]
            # Since neighbors_a is sorted, we can iterate through it
            # But we need to check if c is in adj[b] efficiently
            for c in neighbors_a[i+1:]:
                if c > b and c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
