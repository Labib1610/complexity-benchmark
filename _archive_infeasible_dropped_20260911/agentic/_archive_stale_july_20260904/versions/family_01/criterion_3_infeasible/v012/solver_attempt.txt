import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # Sort edges to help find the lexicographically smallest triad
    # We want to minimize a, then b, then c.
    # A triad (a, b, c) with a < b < c exists if (a, b), (a, c), and (b, c) are edges.
    
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through possible a, b, c.
    # Since n is up to 2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    # We can iterate over each edge (u, v) as (a, b) and check for a common neighbor c > v.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # To minimize (a, b, c), we iterate a from 1 to n
    for a in range(1, n + 1):
        # Get neighbors of a that are > a
        neighbors_a = [v for v in adj[a] if v > a]
        if len(neighbors_a) < 2:
            continue
        
        # For each pair of neighbors b, c of a, check if (b, c) is an edge
        # To keep it efficient, iterate b in neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Check neighbors of b that are > b
            neighbors_b = adj[b]
            
            # We need a c such that c is in neighbors_a AND c is in neighbors_b
            # Since both are sorted, we can use two pointers or set intersection
            ptr_a = i + 1
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    print(f"{a} {b} {val_a}")
                    return
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
