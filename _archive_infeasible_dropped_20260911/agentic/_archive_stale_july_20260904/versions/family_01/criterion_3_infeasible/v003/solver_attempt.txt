```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triplet
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # all edges (a, b) and check if there exists a c > b such that
    # (a, c) and (b, c) are also edges.
    # Since we want the smallest a, then b, then c, we iterate through
    # sorted edges (a, b) and for each, check common neighbors.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a < b < c. Since we iterate a from 1 to n,
            # and b is a neighbor of a, we just need to find c > b
            # such that c is in adj[a] and c is in adj[b].
            
            # Optimization: intersect the two sets of neighbors
            # Only check neighbors of b that are greater than b
            for c in adj[b]:
                if c > b and c in adj[a]:
                    print(f"{a} {b} {c}")
                    return

solve()
```
