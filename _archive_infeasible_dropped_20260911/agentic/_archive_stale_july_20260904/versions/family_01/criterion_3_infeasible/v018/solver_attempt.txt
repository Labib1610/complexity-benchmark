import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Sort adjacency lists to help with finding c efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Candidates for b must be neighbors of a and b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # Candidates for c must be neighbors of a and b, and c > b
            # We can intersect sorted_adj[a] and sorted_adj[b]
            # Since we need the smallest c, we just need the first one > b
            
            # Optimization: use set intersection for faster lookup
            # But since we need the smallest c, we can iterate through sorted_adj[a]
            # and check if it's in adj[b]
            for c in sorted_adj[a]:
                if c > b:
                    if c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
