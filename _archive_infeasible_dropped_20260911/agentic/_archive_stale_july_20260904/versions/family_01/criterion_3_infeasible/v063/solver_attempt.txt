```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find the lexicographically smallest (a, b, c) with a < b < c
    # such that a-b, b-c, c-a are edges.
    # Iterate a from 1 to n-2
    for a in range(1, n + 1):
        # Get neighbors of a, sort them to check b < c
        neighbors_a = sorted([x for x in adj[a] if x > a])
        
        # We want to minimize b, then c.
        # Iterate through sorted neighbors as potential b
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # Now we need a neighbor c of a such that c > b and c is also a neighbor of b
            # To minimize c, we check neighbors of b that are also in neighbors_a
            # and are greater than b.
            # Intersection of adj[a] and adj[b]
            common = sorted([x for x in adj[b] if x > b and x in adj[a]])
            
            if common:
                c = common[0]
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
```
