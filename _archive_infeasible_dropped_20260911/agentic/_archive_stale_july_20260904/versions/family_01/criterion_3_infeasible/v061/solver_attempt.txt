import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest (a, b, c)
    # We want to minimize a, then b, then c.
    # Iterate through all possible a, then b, then c.
    # Since we need the lexicographically smallest, we iterate a from 1 to n,
    # b from a+1 to n, and check if c exists.
    
    # To optimize, iterate through edges (a, b) and check if there exists c
    # such that (a, c) and (b, c) are edges.
    
    # Sort edges by (u, v)
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we can iterate through
    # all possible a, then b, then c.
    # Since n <= 2000, we can iterate through all edges (a, b) and check
    # common neighbors.
    
    best = (n + 1, n + 1, n + 1)
    
    # Optimization: iterate a from 1 to n
    for a in range(1, n + 1):
        # If a is already >= best[0], we can't find a smaller one
        if a >= best[0]:
            break
        
        # Get neighbors of a
        neighbors_a = adj[a]
        # Sort neighbors to check b in order
        sorted_neighbors_a = sorted(list(neighbors_a))
        
        for b in sorted_neighbors_a:
            if b <= a:
                continue
            if a >= best[0] and b >= best[1]:
                break
            
            # Find common neighbors c > b
            # Intersection of neighbors_a and neighbors_b
            neighbors_b = adj[b]
            
            # We want the smallest c > b
            # Iterate through sorted neighbors of a that are > b
            for c in sorted_neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    # Found a triangle (a, b, c)
                    if (a, b, c) < best:
                        best = (a, b, c)
                    # Since we want smallest c for fixed a, b, we can break
                    break
    
    print(f"{best[0]} {best[1]} {best[2]}")

solve()
