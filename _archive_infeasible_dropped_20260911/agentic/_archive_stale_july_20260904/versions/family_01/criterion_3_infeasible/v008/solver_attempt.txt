import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triple
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # all edges (a, b) and check if there exists a c such that
    # (a, c) and (b, c) are also edges.
    # Since we want the smallest a, then b, then c, we iterate in order.
    
    # Pre-sort adjacency lists to make finding c easier
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a < b < c. Since a is fixed and b is from neighbors_a,
            # we only need to check c > b.
            # Intersection of neighbors_a and neighbors_b
            neighbors_b = adj[b]
            
            # Use two pointers to find common elements in sorted lists
            ptr_a = i + 1
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    # Found a triple (a, b, val_a)
                    print(f"{a} {b} {val_a}")
                    return
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
