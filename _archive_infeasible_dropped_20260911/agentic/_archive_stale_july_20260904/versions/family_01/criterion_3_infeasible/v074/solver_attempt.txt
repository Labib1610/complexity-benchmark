import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find the lexicographically smallest (a, b, c) with a < b < c
    # such that a, b, c are all mutually incompatible.
    # This means b in adj[a] and c in adj[a] and c in adj[b].
    
    # Iterate through possible a
    for a in range(1, n + 1):
        # Get all neighbors of a, sorted to ensure lexicographical order
        neighbors_a = sorted(list(adj[a]))
        
        # Iterate through possible b
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that c > b and c is in adj[a] and c is in adj[b]
            # This is equivalent to c in (adj[a] intersection adj[b])
            # To find the smallest c, we can iterate through the smaller set
            
            # Optimization: check intersection of adj[a] and adj[b]
            # Since we want the smallest c > b, we can iterate through sorted neighbors
            # of the one with fewer neighbors.
            
            set_a = adj[a]
            set_b = adj[b]
            
            # Check intersection
            if len(set_a) < len(set_b):
                for c in sorted(list(set_a)):
                    if c > b and c in set_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in sorted(list(set_b)):
                    if c > b and c in set_a:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
