import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, b from a+1 to n, and c from b+1 to n.
    # Since we need the smallest a, then b, then c, this order is correct.
    # Optimization: only check if b is in adj[a] and c is in adj[a] and c is in adj[b].
    
    for a in range(1, n + 1):
        # Sort neighbors to ensure we check b and c in increasing order
        neighbors_a = sorted(list(adj[a]))
        if len(neighbors_a) < 2:
            continue
            
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need to check if there exists c > b such that c is in adj[a] and c is in adj[b]
            # This is equivalent to finding the intersection of adj[a] and adj[b]
            # and picking the smallest element > b.
            
            # Optimization: intersect sets
            common = adj[a].intersection(adj[b])
            
            # Filter for c > b
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

solve()
