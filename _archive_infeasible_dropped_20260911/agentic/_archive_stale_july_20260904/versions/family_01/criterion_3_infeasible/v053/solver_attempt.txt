import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triplet
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all edges (a, b) and check if there exists
    # a node c such that (a, c) and (b, c) are edges.
    # Since we want the smallest a, then b, then c, we iterate
    # through sorted edges (a, b) and check common neighbors.
    
    # Optimization: For each edge (a, b), check common neighbors.
    # To ensure we find the smallest (a, b, c), we can iterate
    # through all possible 'a' from 1 to n.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c > b such that c is in adj[a] and adj[b]
            # To minimize c, we look at the intersection of adj[a] and adj[b]
            # that are greater than b.
            
            # Intersection of neighbors of a and b
            # Since we want the smallest c, we can iterate through
            # the smaller adjacency list.
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
            
            common = []
            for c in candidates:
                if c > b and c in adj[a] and c in adj[b]:
                    common.append(c)
            
            if common:
                print(f"{a} {b} {min(common)}")
                return

solve()
