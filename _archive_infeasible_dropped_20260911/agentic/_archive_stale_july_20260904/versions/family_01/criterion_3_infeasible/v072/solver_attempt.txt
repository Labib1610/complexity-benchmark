import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we look at all neighbors b > a.
    # Then we check if there exists c > b such that c is in adj[a] and adj[b].
    
    # Sort neighbors for each node to help find smallest c efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        neighbors_a = sorted_adj[a]
        # Filter neighbors b > a
        b_candidates = [b for b in neighbors_a if b > a]
        
        for i in range(len(b_candidates)):
            b = b_candidates[i]
            # We need c > b such that c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can intersect efficiently
            # or just iterate through neighbors_a after b
            
            # Optimization: only check c > b
            # Find intersection of neighbors_a (after b) and neighbors_b (after b)
            # Since both are sorted, we can use two pointers or set intersection
            
            # Using set for O(1) lookup in adj[b]
            neighbors_b = adj[b]
            for c in neighbors_a:
                if c > b and c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
