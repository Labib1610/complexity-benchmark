```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triangle
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists c > b
    # such that (a, c) and (b, c) are edges.
    # To ensure we find the smallest, we can iterate through sorted edges.
    
    best = None
    
    for a, b in edges:
        # We want to find c > b such that c is in adj[a] and adj[b]
        # Intersection of adj[a] and adj[b]
        common = adj[a].intersection(adj[b])
        if common:
            c = min(common)
            # The triangle is (a, b, c). Since a < b and c > b, 
            # the sorted order is a, b, c.
            current = (a, b, c)
            if best is None or current < best:
                best = current
                
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
