import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all possible 'a' from 1 to n
    # For each 'a', iterate through all neighbors 'b' of 'a' where b > a
    # Check if there exists a 'c' such that c > b and c is a neighbor of both a and b
    
    # Sort neighbors to ensure we find the smallest b and c
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # Since adj[a] and adj[b] are sorted, we can use two pointers or set intersection
            # Given n=2000, set intersection is efficient enough
            
            # Optimization: only check c > b
            # Use set for O(1) lookup
            set_b = set(adj[b])
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

solve()
