```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, iterate c from b+1 to n
    # Check if c is in adj[a] and adj[b]
    
    # Optimization: Sort adjacency lists to iterate efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider neighbors b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # Check if there exists c > b such that c is in adj[a] and adj[b]
            # We can intersect the sorted lists of neighbors of a and b
            # that are greater than b
            
            # To make this fast, use the set property
            # We need c > b such that c in adj[a] and c in adj[b]
            # Since we want the smallest c, we can iterate through the smaller set
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
            
            found_c = -1
            for c in candidates:
                if c > b and c in adj[a] and c in adj[b]:
                    if found_c == -1 or c < found_c:
                        found_c = c
            
            if found_c != -1:
                print(f"{a} {b} {found_c}")
                return

solve()
```
