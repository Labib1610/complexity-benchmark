import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest trio (a, b, c) with a < b < c:
    # Iterate a from 1 to n
    # Iterate b in sorted(adj[a]) such that b > a
    # Check if there exists c in adj[a] intersect adj[b] such that c > b
    
    # Sort adjacency lists to ensure we find the lexicographically smallest trio
    sorted_adj = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Optimization: convert to set for O(1) lookup
        set_a = adj[a]
        
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and adj[b]
            set_b = adj[b]
            
            # Iterate through neighbors of b that are > b
            for c in sorted_adj[b]:
                if c > b:
                    if c in set_a:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
