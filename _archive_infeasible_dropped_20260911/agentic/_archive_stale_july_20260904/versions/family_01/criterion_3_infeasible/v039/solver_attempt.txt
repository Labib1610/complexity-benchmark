```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    edges.sort()
    
    # To find the lexicographically smallest trio (a, b, c)
    # We iterate through all edges (a, b) and check if there exists c > b
    # such that (a, c) and (b, c) are also edges.
    
    best_trio = None
    
    # Sort adjacency lists to help with finding c efficiently
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a, b in edges:
        # We need to find c > b such that c is in adj[a] and c is in adj[b]
        # Since adj lists are sorted, we can use intersection or two-pointer
        list_a = adj[a]
        list_b = adj[b]
        
        # Optimization: only check if a potential c exists
        # We only care about c > b
        ptr_a = 0
        ptr_b = 0
        
        # Find first element > b in both lists
        while ptr_a < len(list_a) and list_a[ptr_a] <= b:
            ptr_a += 1
        while ptr_b < len(list_b) and list_b[ptr_b] <= b:
            ptr_b += 1
            
        while ptr_a < len(list_a) and ptr_b < len(list_b):
            val_a = list_a[ptr_a]
            val_b = list_b[ptr_b]
            
            if val_a == val_b:
                c = val_a
                if best_trio is None or (a, b, c) < best_trio:
                    best_trio = (a, b, c)
                # Since we want the smallest lexicographical trio, 
                # and we iterate edges in order, we can potentially break early
                # but given the constraints, checking all is fine.
                break
            elif val_a < val_b:
                ptr_a += 1
            else:
                ptr_b += 1
                
    if best_trio:
        print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

solve()
```
