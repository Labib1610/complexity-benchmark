```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest triplet (a, b, c)
    # We iterate through all possible a, then b, then check if c exists.
    # Since we need the smallest a, then b, then c, we sort the edges
    # or iterate through sorted nodes.
    
    # Sort edges to help find smallest triplets
    edges.sort()
    
    # To optimize, iterate through all edges (u, v) as potential (a, b)
    # and check if there exists a node w > v such that w is connected to both u and v.
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then iterate through neighbors b of a where b > a,
    # then check if there is a common neighbor c > b.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b < a:
                continue
            
            # We need a c such that c > b and c is in adj[a] and adj[b]
            # Intersection of adj[a] and adj[b]
            # Since we want the smallest c, we can iterate through sorted neighbors
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
