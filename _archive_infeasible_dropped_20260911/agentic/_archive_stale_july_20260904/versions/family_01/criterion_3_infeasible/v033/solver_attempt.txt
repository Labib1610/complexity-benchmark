import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) such that a < b < c.
    # To find the lexicographically smallest (a, b, c), we iterate
    # through possible values of a, then b, then c.
    
    # Sort adjacency lists to help with lexicographical order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    best = None
    
    for a in range(1, n + 1):
        # Optimization: if we already found a solution (a, b, c),
        # and the current 'a' is already greater than the best 'a', we can stop.
        if best and a >= best[0]:
            break
            
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            if best and (a > best[0] or (a == best[0] and b >= best[1])):
                continue
                
            # Check for common neighbors c such that c > b
            # To be efficient, intersect neighbors_a and neighbors_b
            neighbors_b = adj[b]
            
            # We only care about c > b
            for c in neighbors_a:
                if c > b and c in neighbors_b:
                    if best is None or (a < best[0]) or (a == best[0] and b < best[1]) or (a == best[0] and b == best[1] and c < best[2]):
                        best = (a, b, c)
                        # Since we iterate a, b, c in order, the first one we find
                        # might not be the absolute smallest if we don't break carefully.
                        # But with the current loops, we can just keep updating.
    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
