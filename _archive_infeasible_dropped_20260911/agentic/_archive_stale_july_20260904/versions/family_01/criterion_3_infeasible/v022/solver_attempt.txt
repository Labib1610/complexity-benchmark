import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to help find lexicographically smallest triad
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) with a < b.
    # Then check for a common neighbor c such that a < b < c.
    # To ensure lexicographical order, we can iterate a from 1 to n,
    # then b in sorted order of neighbors of a, then check intersection.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a common neighbor c such that c > b
            # Intersection of adj[a] and adj[b]
            # Since we want the smallest c, we can iterate through sorted neighbors
            # of a that are > b and check if they are in adj[b].
            
            # Optimization: iterate through the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                for c in neighbors_a:
                    if c > b and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                neighbors_b = sorted(list(adj[b]))
                for c in neighbors_b:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

solve()
