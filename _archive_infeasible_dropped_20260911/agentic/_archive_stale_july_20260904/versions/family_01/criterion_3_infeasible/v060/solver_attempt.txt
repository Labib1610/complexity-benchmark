import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, check if there exists c > b such that c in adj[a] and c in adj[b]
    
    # Sort adjacency lists to potentially speed up intersection
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Only consider neighbors > a
        neighbors_a = [x for x in neighbors_a if x > a]
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c in adj[a] and c in adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            neighbors_b = adj[b]
            
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
