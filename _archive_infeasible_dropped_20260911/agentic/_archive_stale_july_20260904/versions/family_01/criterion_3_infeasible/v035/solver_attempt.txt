import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to help find the lexicographically smallest (a, b, c)
    # We want smallest a, then smallest b, then smallest c.
    # Iterate through all edges (u, v) as potential (a, b)
    # Then check if there exists a w such that w is in adj[u] and adj[v]
    
    edges.sort()
    
    best = None
    
    for u, v in edges:
        # u is a, v is b. We need c > v such that c is in adj[u] and adj[v]
        # To ensure lexicographical order, we check candidates for c
        # Since we want smallest a, then b, then c, we can iterate through
        # the smaller adjacency list to find common neighbors.
        
        if len(adj[u]) < len(adj[v]):
            candidates = adj[u]
        else:
            candidates = adj[v]
            
        for w in candidates:
            if w > v and w in adj[u] and w in adj[v]:
                # Found a triplet (u, v, w)
                # Since we iterate edges in sorted order, the first one we find
                # might not be the absolute smallest if we don't check carefully.
                # Actually, if we iterate u from 1 to n, and v from u+1 to n,
                # we can find the smallest.
                
                # Let's refine:
                # Iterate a from 1 to n
                #   Iterate b in adj[a] where b > a
                #     Iterate c in adj[a] where c > b
                #       If c in adj[b], we found the smallest (a, b, c)
                pass

    # Correct approach for lexicographical smallest:
    for a in range(1, n + 1):
        sorted_neighbors = sorted(list(adj[a]))
        for i in range(len(sorted_neighbors)):
            b = sorted_neighbors[i]
            if b <= a: continue
            for j in range(i + 1, len(sorted_neighbors)):
                c = sorted_neighbors[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
