import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c,
    # we iterate through all possible a, then b, then check if c exists.
    # Since we need a < b < c, we sort the adjacency lists to check efficiently.
    for i in range(1, n + 1):
        adj[i].sort()
        
    # To check if a, b, c form a triangle, we need to check if b is in adj[a],
    # c is in adj[b], and a is in adj[c].
    # Using a set for O(1) lookup is faster.
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    # Iterate through all possible a
    for a in range(1, n + 1):
        # Iterate through all possible b > a
        for b in adj[a]:
            if b > a:
                # Check for common neighbors c > b
                # Intersection of adj[a] and adj[b] gives potential c's
                # We need c > b
                for c in adj[a]:
                    if c > b and c in adj_sets[b]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
