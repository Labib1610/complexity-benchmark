```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest trio
    edges.sort()
    
    # A trio (a, b, c) with a < b < c is a deadlock if 
    # (a,b), (a,c), and (b,c) are all edges.
    # To find the lexicographically smallest, we iterate through possible a, b, c.
    # Since n <= 2000, we can iterate through all edges (a, b) and check for common neighbors c.
    
    best_trio = None
    
    # To minimize (a, b, c) lexicographically:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # If (a, b) is an edge, check for common neighbors c > b
    
    for a in range(1, n + 1):
        # Optimization: if we already found a trio starting with a smaller 'a',
        # we don't need to check larger 'a's.
        if best_trio and a > best_trio[0]:
            break
            
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # Check for common neighbors c > b
                # To keep it efficient, intersect the smaller adjacency list
                if len(adj[a]) < len(adj[b]):
                    candidates = adj[a]
                else:
                    candidates = adj[b]
                
                for c in candidates:
                    if c > b and c in adj[a] and c in adj[b]:
                        trio = (a, b, c)
                        if best_trio is None or trio < best_trio:
                            best_trio = trio
        
        if best_trio and best_trio[0] < a:
            break
            
    if best_trio:
        print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

solve()
```
