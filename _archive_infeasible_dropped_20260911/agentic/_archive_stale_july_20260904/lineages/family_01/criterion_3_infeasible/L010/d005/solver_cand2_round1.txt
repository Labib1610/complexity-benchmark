```python
import sys

def solve():
    # A deadlock is a set of 3 edges (u1, v1), (u2, v2), (u3, v3)
    # such that each vertex appears an even number of times.
    # With 3 edges, this is only possible if the edges form a triangle (u,v), (v,w), (w,u).
    # The vertices involved are {u, v, w}.
    # We need to find all triangles in the graph and output the lexicographically
    # smallest set of vertices {a, b, c} that form a triangle.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find all triangles efficiently:
    # For each edge (u, v), check common neighbors.
    # Since we need the lexicographically smallest {a, b, c},
    # we can iterate through all possible triangles.
    
    results = []
    
    # Sort vertices by degree to optimize triangle finding (optional but good)
    # For this problem, iterating over edges and checking common neighbors is O(m * sqrt(m))
    for u, v in edges:
        # Find common neighbors of u and v
        # To ensure we don't count the same triangle multiple times,
        # we only look for w such that u < v < w or other orderings.
        # Actually, just find all w such that (u, w) and (v, w) exist.
        
        # Intersection of adj[u] and adj[v]
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
            
        for w in adj[u]:
            if w in adj[v]:
                # Found a triangle {u, v, w}
                triangle = sorted([u, v, w])
                results.append(tuple(triangle))
                
    if results:
        # The problem asks for the lexicographically smallest sequence
        # of the set of workers W.
        ans = min(results)
        print(f"{ans[0]} {ans[1]} {ans[2]}")

solve()
```
