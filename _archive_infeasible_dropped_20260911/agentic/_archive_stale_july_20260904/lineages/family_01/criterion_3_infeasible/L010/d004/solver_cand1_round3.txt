import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # A deadlock of three workers {a, b, c} means the three reports are
    # edges between these three nodes.
    # Condition 1: Exactly 3 distinct workers.
    # Condition 2: No worker appears >= 3 times.
    # With 3 edges and 3 nodes, the only way to have 3 distinct nodes is
    # if the edges are (a, b), (b, c), and (a, c).
    # Each node appears exactly twice in this set of edges.
    # This satisfies condition 2 (no node appears 3 times).
    # Thus, a deadlock is simply a triangle (a, b, c) in the graph.
    
    # We need the lexicographically smallest {a, b, c} with a < b < c.
    # Iterate through all possible a, then b, then c.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        if len_a < 2:
            continue
            
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Check for c such that (a, c) and (b, c) exist
            # c must be in adj[a] and adj[b]
            # To be lexicographically smallest, we want smallest a, then b, then c.
            # Since we iterate a and b in increasing order, we just need to find
            # the smallest c > b that is in both adj[a] and adj[b].
            
            # Intersection of adj[a] and adj[b]
            # Since adj[a] and adj[b] are sets, intersection is fast.
            common = adj[a].intersection(adj[b])
            
            # We need c > b
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

solve()
