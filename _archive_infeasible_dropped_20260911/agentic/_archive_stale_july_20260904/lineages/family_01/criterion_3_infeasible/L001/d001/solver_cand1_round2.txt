import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and adj[b]
    
    # Sort adjacency lists to potentially speed up intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Candidates for b must be connected to a
        neighbors_a = adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Now we need c > b such that c is in adj[a] and adj[b]
            # Intersection of adj[a] and adj[b]
            # Since we want the smallest c, we can iterate through sorted neighbors
            # or use set intersection
            common = adj_sets[a].intersection(adj_sets[b])
            
            best_c = float('inf')
            found = False
            for c in common:
                if c > b:
                    if c < best_c:
                        best_c = c
                        found = True
            
            if found:
                print(f"{a} {b} {best_c}")
                return

if __name__ == '__main__':
    solve()
