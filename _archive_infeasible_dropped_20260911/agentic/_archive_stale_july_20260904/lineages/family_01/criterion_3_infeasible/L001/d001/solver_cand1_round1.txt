import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through all possible 'a' from 1 to n-2.
    # For a fixed 'a', we look for 'b' > 'a' such that 'b' is connected to 'a'.
    # Then we look for 'c' > 'b' such that 'c' is connected to both 'a' and 'b'.
    
    # Sort adjacency lists to help find smallest 'b' and 'c'
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Filter neighbors to only those > a
        b_candidates = [b for b in neighbors_a if b > a]
        
        for b in b_candidates:
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # Since adj[a] and adj[b] are sorted, we can use two pointers or set intersection
            # Given n=2000, set intersection or simple iteration is fine.
            # To be efficient, use the smaller adjacency list to check existence.
            
            # Optimization: check if intersection exists
            set_a = set(adj[a])
            set_b = set(adj[b])
            
            # Find common neighbors > b
            common = sorted(list(set_a.intersection(set_b)))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return
                    
    print("0 0 0")

if __name__ == '__main__':
    solve()
