```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # For each edge (u,v), we look for a third node w such that:
    # w is connected to both u and v
    # This forms a triangle: u-v-w-u
    
    # To find the lexicographically smallest triangle:
    # We iterate through all edges (u,v) with u < v
    # For each such edge, we look for the smallest w > max(u,v) 
    # such that w is connected to both u and v
    
    # First, sort adjacency lists for consistent iteration
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    best = None
    
    # Sort edges to process in order that helps find lex smallest
    edges.sort()
    
    for u, v in edges:
        if u > v:
            u, v = v, u
            
        # Find common neighbors of u and v that are > max(u, v)
        min_w = max(u, v) + 1
        
        # Iterate through neighbors of u to find candidates
        for w in adj[u]:
            if w < min_w:
                continue
            if w in adj[v]:
                # Found a triangle u, v, w
                candidate = tuple(sorted([u, v, w]))
                if best is None or candidate < best:
                    best = candidate
                    break  # Since adj[u] is sorted, first valid is smallest for this u,v
        
        if best:
            # Try to see if there's a better one by checking other edges
            pass
    
    # More systematic approach:
    # For each node i, check its neighbors j,k where i<j<k and j-k is also an edge
    
    found = False
    for i in range(1, n+1):
        if found:
            break
        neighbors = list(adj[i])
        neighbors.sort()
        # For each pair of neighbors of i
        for jj in range(len(neighbors)):
            if found:
                break
            for kk in range(jj+1, len(neighbors)):
                j = neighbors[jj]
                k = neighbors[kk]
                # Check if j and k are connected
                if k in adj[j]:
                    # Triangle found: i - j - k - i
                    res = sorted([i, j, k])
                    print(res[0], res[1], res[2])
                    found = True
                    break

if __name__ == "__main__":
    main()
```
