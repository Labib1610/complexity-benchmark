```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_bits = [0] * (n + 1)          # 1-indexed bitsets
    neigh = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        # store undirected edge
        adj_bits[u] |= 1 << v
        adj_bits[v] |= 1 << u
        neigh[u].append(v)
        neigh[v].append(u)

    # sort neighbor lists for deterministic order
    for i in range(1, n + 1):
        neigh[i].sort()

    for a in range(1, n + 1):
        for b in neigh[a]:
            if b <= a:
                continue
            common = adj_bits[a] & adj_bits[b]
            # remove vertices <= b
            mask = common >> b
            if mask:
                low = mask & -mask
                c = (low.bit_length() - 1) + b
                # a < b < c holds by construction
                sys.stdout.write(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
```
