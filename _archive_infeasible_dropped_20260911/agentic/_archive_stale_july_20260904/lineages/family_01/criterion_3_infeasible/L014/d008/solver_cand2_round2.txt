import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, iterate c from b+1 to n
    # Check if c is in adj[a] and c is in adj[b]
    
    # Optimization: Only iterate over neighbors of a for b
    # Sort neighbors to ensure we check in lexicographical order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # We need b > a and c > b
        # Filter neighbors_a to only include those > a
        valid_b_list = [b for b in neighbors_a if b > a]
        
        for i in range(len(valid_b_list)):
            b = valid_b_list[i]
            # Now we need c > b such that c is in adj[a] and adj[b]
            # This is the intersection of neighbors_a and neighbors_b
            # Since both are sorted, we can find intersection efficiently
            # But we only need the smallest c > b
            
            # Optimization: check intersection of neighbors_a and neighbors_b
            # Only need to check c > b
            neighbors_b = adj[b]
            for c in valid_b_list[i+1:]:
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
