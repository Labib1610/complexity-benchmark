import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Find lexicographically smallest deadlock configuration
    # A deadlock is a triangle (3 nodes where each pair is an edge)
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Convert to sorted list for deterministic iteration
        sorted_neighbors = sorted(neighbors_a)
        for i, b in enumerate(sorted_neighbors):
            if b <= a:
                continue
            for c in sorted_neighbors[i+1:]:
                if c <= b:
                    continue
                # Check if b and c are connected
                if c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
