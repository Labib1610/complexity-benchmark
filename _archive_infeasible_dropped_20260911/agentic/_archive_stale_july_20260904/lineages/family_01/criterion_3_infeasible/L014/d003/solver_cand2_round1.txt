import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A group (a, b, c) with a < b < c is locked if:
    # (a, b) is reactive, (a, c) is reactive, and (b, c) is reactive.
    # This is equivalent to finding a triangle in the graph.
    # We need to find the lexicographically first (a, b, c).
    
    # Iterate in the specified order: a from 1 to n, b from a+1 to n, c from b+1 to n
    for a in range(1, n + 1):
        # Pre-filter neighbors of a that are > a
        neighbors_a = [v for v in adj[a] if v > a]
        # Sort to ensure we check b in increasing order
        neighbors_a.sort()
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c such that c > b, c is in adj[a], and c is in adj[b]
            # Intersection of neighbors of a and neighbors of b
            # Only consider nodes > b
            
            # Optimization: iterate through the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
            
            # Find all c > b that are in both adj[a] and adj[b]
            found_c = []
            for c in candidates:
                if c > b and c in adj[a] and c in adj[b]:
                    found_c.append(c)
            
            if found_c:
                print(f"{a} {b} {min(found_c)}")
                return

if __name__ == '__main__':
    solve()
