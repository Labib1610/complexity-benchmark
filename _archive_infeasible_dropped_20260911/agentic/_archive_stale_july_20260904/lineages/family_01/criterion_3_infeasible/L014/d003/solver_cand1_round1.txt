import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need the lexicographically smallest triple (a, b, c) with a < b < c
    # such that (a, b), (a, c), and (b, c) are all in the adjacency set.
    # Since n <= 2000, an O(n^2) or O(n * m) approach is acceptable.
    
    # Iterate through a from 1 to n
    for a in range(1, n + 1):
        # Get neighbors of a that are greater than a
        neighbors_a = [v for v in adj[a] if v > a]
        neighbors_a.sort()
        
        # Iterate through b in neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # We need a c such that b < c, c is in adj[a], and c is in adj[b]
            # This is equivalent to finding a common neighbor of a and b that is > b
            # Intersection of adj[a] and adj[b]
            # To optimize, iterate through the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
            
            for c in candidates:
                if c > b:
                    if c in adj[a] and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
