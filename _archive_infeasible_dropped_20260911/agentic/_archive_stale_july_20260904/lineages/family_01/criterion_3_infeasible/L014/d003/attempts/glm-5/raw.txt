import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    
    # We want lexicographically smallest triangle (a, b, c) with a < b < c.
    # For each a from 1 to n, for each b in adj[a] with b > a,
    # check if there exists c in adj[a] intersect adj[b] with c > b.
    # The smallest such c gives the best triangle for this (a, b).
    # Since we iterate a ascending, then b ascending, the first found is the answer.
    
    for a in range(1, n + 1):
        adj_a = adj[a]
        # candidates b: neighbors of a with b > a
        # We want smallest b first
        for b in sorted(adj_a):
            if b <= a:
                continue
            adj_b = adj[b]
            # find smallest c > b in both adj_a and adj_b
            # iterate over the smaller set
            if len(adj_a) <= len(adj_b):
                for c in adj_a:
                    if c > b and c in adj_b:
                        print(a, b, c)
                        return
            else:
                for c in adj_b:
                    if c > b and c in adj_a:
                        print(a, b, c)
                        return

solve()
