```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that:
    # b in adj[a], c in adj[a], c in adj[b]
    # This is equivalent to finding a triangle (a, b, c) in the graph.
    # Lexicographical order: smallest a, then smallest b, then smallest c.
    # Since a < b < c, we iterate a from 1 to n-2, b from a+1 to n-1, c from b+1 to n.
    
    # To optimize, iterate a, then iterate b in adj[a] such that b > a,
    # then check if there exists c in adj[a] intersect adj[b] such that c > b.
    
    for a in range(1, n + 1):
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for b in neighbors_a:
            # Find common neighbors c of a and b such that c > b
            # We want the smallest such c.
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
```
