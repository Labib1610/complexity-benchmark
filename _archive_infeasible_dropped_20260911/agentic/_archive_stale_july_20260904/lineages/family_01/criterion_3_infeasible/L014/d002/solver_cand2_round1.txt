import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triangle (i, j, k) with i < j < k:
    # We iterate through i from 1 to n.
    # For each i, we look at all neighbors j of i where j > i.
    # Then we check if there exists a k > j such that k is a neighbor of both i and j.
    
    # Sort neighbors to ensure we find the lexicographically smallest triangle
    for i in range(1, n + 1):
        neighbors_i = sorted(list(adj[i]))
        for idx_j in range(len(neighbors_i)):
            j = neighbors_i[idx_j]
            if j <= i:
                continue
            
            # We need k > j such that k is in adj[i] and adj[j]
            # To be efficient, intersect the two sets
            common = sorted(list(adj[i].intersection(adj[j])))
            for k in common:
                if k > j:
                    print(f"{i} {j} {k}")
                    return

if __name__ == '__main__':
    solve()
