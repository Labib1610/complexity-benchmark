```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
    
    # A locked group of three chemicals {a, b, c} means
    # (a, b), (b, c), and (a, c) are all reactive pairs.
    # This is equivalent to finding a triangle in the graph.
    # We need the lexicographically smallest (a, b, c).
    
    # Sort adjacency lists to iterate in order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    # Iterate through all possible a < b < c
    # To optimize, iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # We need b in neighbors_a such that b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c in neighbors_a AND neighbors_b such that c > b
            # Intersection of neighbors_a and neighbors_b
            # Since we need the smallest lexicographical, we check c in order
            # Optimization: intersect the two sorted lists
            set_b = adj[b]
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
