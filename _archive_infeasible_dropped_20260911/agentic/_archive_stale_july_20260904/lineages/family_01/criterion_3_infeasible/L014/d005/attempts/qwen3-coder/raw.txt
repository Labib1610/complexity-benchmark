import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list using sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        adj[u].add(v)
        adj[v].add(u)
        index += 2

    # Function to check if three nodes form a deadlock
    def is_deadlock(a, b, c):
        # Condition 1: At least one pair is prohibited
        has_prohibited = (b in adj[a]) or (c in adj[a]) or (c in adj[b])
        if not has_prohibited:
            return False
        
        # Condition 2: For any two, their relationship with the third must be the same
        # Check relationships of a and b with c
        a_with_c = (c in adj[a])
        b_with_c = (c in adj[b])
        
        # Check relationships of a and c with b
        a_with_b = (b in adj[a])
        c_with_b = (b in adj[c])
        
        # Check relationships of b and c with a
        b_with_a = (a in adj[b])
        c_with_a = (a in adj[c])
        
        same_bc = (a_with_c == b_with_c)
        same_ac = (a_with_b == c_with_b)
        same_ab = (b_with_a == c_with_a)
        
        return same_bc and same_ac and same_ab

    # Iterate through all combinations in lexicographical order
    for a in range(1, n - 1):
        for b in range(a + 1, n):
            for c in range(b + 1, n + 1):
                if is_deadlock(a, b, c):
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
