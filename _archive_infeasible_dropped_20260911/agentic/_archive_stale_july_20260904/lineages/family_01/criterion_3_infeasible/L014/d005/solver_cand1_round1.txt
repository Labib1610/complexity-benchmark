```python
import sys

def solve():
    # A trio {a, b, c} is a deadlock if every one of the 8 ways to place them
    # into two vessels fails.
    # A placement fails if any prohibited pair is in the same vessel.
    # Let the prohibited pairs among {a, b, c} be E'.
    # A placement is valid if for every {u, v} in E', u and v are in different vessels.
    # This is equivalent to saying the subgraph ( {a, b, c}, E' ) is bipartite.
    # A placement fails if the subgraph is NOT bipartite.
    # A graph with 3 vertices is not bipartite if and only if it contains an odd cycle.
    # The only odd cycle in a 3-vertex graph is a triangle (a cycle of length 3).
    # Thus, a trio {a, b, c} is a deadlock if and only if all three pairs
    # {a, b}, {b, c}, and {a, c} are prohibited.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find the lexicographically smallest {a, b, c} such that
    # {a, b}, {b, c}, and {a, c} are all in the prohibited set.
    # This is equivalent to finding a triangle in the graph.
    
    # To find the lexicographically smallest triangle:
    # Iterate a from 1 to n-2
    #   Iterate b from a+1 to n-1
    #     If b is connected to a:
    #       Find common neighbors c > b
    #       If any exist, the smallest such c gives the smallest {a, b, c}
    
    for a in range(1, n - 1):
        adj_a = adj[a]
        for b in range(a + 1, n):
            if b in adj_a:
                # Check for common neighbors c > b
                adj_b = adj[b]
                # Intersection of adj_a and adj_b
                # We want the smallest c > b
                # Optimization: iterate over the smaller adjacency list
                if len(adj_a) < len(adj_b):
                    for c in sorted(adj_a):
                        if c > b and c in adj_b:
                            print(f"{a} {b} {c}")
                            return
                else:
                    for c in sorted(adj_b):
                        if c > b and c in adj_a:
                            print(f"{a} {b} {c}")
                            return

solve()
```
