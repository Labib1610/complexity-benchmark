import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Build adjacency list using sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A deadlock is a triangle (a, b, c) where each node has degree 2 within the trio.
    # In a graph, a triangle (a, b, c) has edges (a,b), (b,c), (c,a).
    # Each node in the triangle is connected to the other two, so each has degree 2.
    # Thus, any triangle in the graph is a deadlock configuration.
    # We need the lexicographically smallest (a, b, c) with a < b < c.
    
    # Iterate through all possible a, then b, then c
    # To optimize, iterate a from 1 to n-2
    for a in range(1, n + 1):
        # Sort neighbors to ensure we check b and c in order
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        if len_a < 2:
            continue
            
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Check if there exists c such that (a,c) and (b,c) are edges
            # c must be > b
            # Intersection of adj[a] and adj[b] gives common neighbors
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    # Found the lexicographically smallest triangle
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
