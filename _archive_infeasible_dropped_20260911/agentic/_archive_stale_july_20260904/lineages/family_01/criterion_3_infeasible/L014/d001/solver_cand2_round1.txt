import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A volatile triad is a triangle (a, b, c) in the graph.
    # We need the lexicographically smallest (a, b, c) with a < b < c.
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        neighbors_a = sorted(list(adj[a]))
        # Iterate b in neighbors of a, b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c such that c is a neighbor of a and b, and c > b
            # Intersection of adj[a] and adj[b]
            # To optimize, iterate over the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                common = adj[a]
                target = b
            else:
                common = adj[b]
                target = a
            
            found_c = -1
            for c in common:
                if c > b and c in adj[target]:
                    if found_c == -1 or c < found_c:
                        found_c = c
            
            if found_c != -1:
                print(f"{a} {b} {found_c}")
                return

if __name__ == '__main__':
    solve()
