import sys

def solve():
    # A set of three materials is a deadlock if it is impossible to assign them 
    # to two vessels without violating prohibitions. This means the three 
    # materials must form a triangle (a clique of size 3) in the graph of 
    # prohibitions. If three materials form a triangle, any two must be in 
    # different vessels, but with three materials, at least two must share 
    # a vessel, which is prohibited.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c), we iterate through 
    # possible a, then b, then c.
    # A triangle (a, b, c) exists if a < b < c and (a,b), (a,c), (b,c) are edges.
    
    # Sort edges to help find triangles efficiently
    edges.sort()
    
    # Iterate through all possible a
    for a in range(1, n + 1):
        # Get neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        # For each pair of neighbors b, c of a, check if (b, c) is an edge
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if c is in adj[b]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
