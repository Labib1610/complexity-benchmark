```python
import sys

def solve():
    # A configuration of three materials {a, b, c} cannot be safely transported 
    # using fewer than 3 vessels if and only if they form a triangle (a-b, b-c, c-a)
    # in the graph of prohibited pairs.
    # We need to find the lexicographically smallest triangle (a, b, c) with a < b < c.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n,
    # b from a+1 to n, and c from b+1 to n.
    # Optimization: a triangle exists if b is connected to a, c is connected to a,
    # and c is connected to b.
    
    # Sort adjacency lists to potentially speed up intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Filter neighbors to only those > a
        valid_b = [b for b in neighbors_a if b > a]
        for i in range(len(valid_b)):
            b = valid_b[i]
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of neighbors_a[i+1:] and neighbors_b
            neighbors_b = adj_sets[b]
            for j in range(i + 1, len(valid_b)):
                c = valid_b[j]
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
