```python
import sys

def solve():
    # A configuration of three materials {a, b, c} is a deadlock if and only if
    # it is impossible to assign them to two holds such that no hold contains a prohibited pair.
    # This happens if and only if the subgraph induced by {a, b, c} contains an odd cycle.
    # Since the only odd cycle of length 3 is a triangle (K3), a set of 3 materials
    # is a deadlock if and only if they form a triangle in the graph of prohibited pairs.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # We need the lexicographically smallest {a, b, c} such that a < b < c
    # and (a, b), (b, c), (a, c) are all edges in the graph.
    
    # Sort edges to help find lexicographically smallest triangle
    edges.sort()
    
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        # For each pair of neighbors (b, c) of a, check if (b, c) is an edge
        # To ensure lexicographical order, we iterate b then c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if (b, c) is an edge
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
