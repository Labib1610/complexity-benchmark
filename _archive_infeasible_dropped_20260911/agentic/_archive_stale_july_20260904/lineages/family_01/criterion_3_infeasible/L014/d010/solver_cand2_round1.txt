```python
import sys

def solve():
    # A set of three distinct materials {a, b, c} is a deadlock configuration
    # if every permutation (x, y, z) is hazardous.
    # A sequence (x, y, z) is hazardous if (x, y) is prohibited AND (y, z) is prohibited.
    # Let G be the graph where edges represent prohibited pairs.
    # For {a, b, c} to be a deadlock, every permutation must satisfy the condition.
    # Permutations:
    # (a, b, c) -> (a, b) in E and (b, c) in E
    # (a, c, b) -> (a, c) in E and (c, b) in E
    # (b, a, c) -> (b, a) in E and (a, c) in E
    # (b, c, a) -> (b, c) in E and (c, a) in E
    # (c, a, b) -> (c, a) in E and (a, b) in E
    # (c, b, a) -> (c, b) in E and (b, a) in E
    #
    # All these conditions are satisfied if and only if all pairs (a, b), (b, c), (a, c)
    # are in E. That is, {a, b, c} must form a triangle (a clique of size 3) in G.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need the lexicographically smallest {a, b, c} with a < b < c
    # such that a, b, c form a triangle.
    for a in range(1, n + 1):
        # Sort neighbors to ensure we check b and c in order
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Check for c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
