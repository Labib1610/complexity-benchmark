import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    best = None
    
    for u in range(1, n + 1):
        if best is not None and u >= best[0]:
            break
        neighbors = sorted(adj[u])
        for v in neighbors:
            if v <= u:
                continue
            if best is not None and v >= best[1]:
                continue
            common = adj[u] & adj[v]
            for w in common:
                if w <= v:
                    continue
                if best is None or (u, v, w) < best:
                    best = (u, v, w)
    
    print(best[0], best[1], best[2])

solve()
