import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    degree = [0] * (n + 1)
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        degree[u] += 1
        degree[v] += 1
        adj[u].add(v)
        adj[v].add(u)
        
    # A trio {a, b, c} is a deadlock if the number of edges involving 
    # at least one worker NOT in {a, b, c} is m - 3.
    # This is equivalent to saying that the number of edges involving 
    # ONLY workers in {a, b, c} is exactly 3.
    # The edges involving only {a, b, c} are (a, b), (a, c), and (b, c).
    # So we need to check if (a, b), (a, c), and (b, c) are all in the edge set.
    
    # To minimize a, then b, then c, we iterate in lexicographical order.
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b not in adj[a]:
                continue
            for c in range(b + 1, n + 1):
                if c in adj[a] and c in adj[b]:
                    # Found a triangle {a, b, c}.
                    # Check if it's a deadlock:
                    # Edges involving only {a, b, c} are (a, b), (a, c), (b, c).
                    # Count = 3.
                    # Condition: m - count_edges_outside = m - 3
                    # => count_edges_outside = 3
                    # This is true if the only edges in the graph are (a, b), (a, c), (b, c).
                    # Wait, the condition is: "Count how many incident files involve 
                    # at least one worker who is NOT part of the chosen trio."
                    # Let E be the set of all edges.
                    # Let S = {a, b, c}.
                    # We want |{e in E : e intersect (V \ S) is not empty}| = m - 3.
                    # This is equivalent to |{e in E : e is a subset of S}| = 3.
                    # Since we found a triangle, the edges (a, b), (a, c), (b, c) are in E.
                    # So |{e in E : e is a subset of S}| is at least 3.
                    # If it is exactly 3, then the condition is satisfied.
                    
                    count_in_trio = 0
                    if (a, b) in edges or (b, a) in edges: count_in_trio += 1
                    if (a, c) in edges or (c, a) in edges: count_in_trio += 1
                    if (b, c) in edges or (c, b) in edges: count_in_trio += 1
                    
                    # Actually, the condition is simply that the triangle {a, b, c}
                    # must be the only edges involving these vertices, and no other 
                    # edges can exist in the graph? No, that's not right.
                    # The condition is: |{e in E : e \cap (V \ S) \neq \emptyset}| = m - 3.
                    # This is equivalent to |E| - |{e in E : e \subseteq S}| = m - 3.
                    # m - |{e in E : e \subseteq S}| = m - 3.
                    # |{e in E : e \subseteq S}| = 3.
                    # Since {a, b, c} is a triangle, |{e in E : e \subseteq S}| is at least 3.
                    # So we just need to check if there are any other edges 
                    # between a, b, c. But there are only 3 possible edges between 3 nodes.
                    # So any triangle satisfies the condition!
                    print(f"{a} {b} {c}")
                    return

solve()
