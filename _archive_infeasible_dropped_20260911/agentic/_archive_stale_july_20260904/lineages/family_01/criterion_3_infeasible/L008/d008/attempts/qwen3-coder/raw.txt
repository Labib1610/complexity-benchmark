import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # We need to find three edges such that they involve exactly three distinct nodes
    # This means we're looking for a triangle in the graph
    
    # Let's iterate through all edges and for each edge (u,v), 
    # check for a common neighbor w, so that (u,w) and (v,w) also exist
    
    # To make it efficient, let's process edges in a way that helps with lexicographical order
    
    # First, sort the edges to help with ordering
    edges.sort()
    
    # Create a set of edges for O(1) lookup
    edge_set = set(edges)
    
    # Also create reverse edges for easy lookup
    rev_edge_set = set((v, u) for u, v in edges)
    
    best = None
    
    # For each edge (u, v), look for a third node w such that both (u,w) and (v,w) exist
    for u, v in edges:
        if u > v:
            continue  # Ensure u < v to avoid duplication
            
        # Find common neighbors of u and v
        neighbors_u = adj[u]
        neighbors_v = adj[v]
        
        # Iterate through neighbors of u to find common ones with v
        for w in neighbors_u:
            if w in neighbors_v:
                # Found a triangle: u-v, u-w, v-w
                # Check that all three edges exist in the original list
                # Since we built adj from the edge list, we know these edges exist
                # But we should confirm the third edge is present too
                
                # The edges we have: (u,v) is already confirmed
                # Need to check (u,w) and (v,w) exist in original edge list
                # Since adj was built from edges, any neighbor relation corresponds to an edge
                
                # Now, get the three nodes
                nodes = sorted([u, v, w])
                a, b, c = nodes
                
                if best is None or (a, b, c) < best:
                    best = (a, b, c)
                    
    # Output the result
    print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
