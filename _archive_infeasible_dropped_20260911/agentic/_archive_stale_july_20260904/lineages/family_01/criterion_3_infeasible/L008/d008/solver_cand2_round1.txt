import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # We need to find three workers {a, b, c} such that the edges
    # (a, b), (b, c), and (a, c) all exist in the input.
    # This is equivalent to finding a triangle in the graph.
    # We need to minimize a, then b, then c.
    
    found_triplets = []
    
    # Iterate through all edges (u, v) and check for a common neighbor w
    # To minimize a, b, c, we iterate u from 1 to n
    for u in range(1, n + 1):
        neighbors = sorted(list(adj[u]))
        for i in range(len(neighbors)):
            v = neighbors[i]
            # Check common neighbors w such that w > v
            # The triangle is (u, v, w)
            # Since u < v < w is required for the output, 
            # we ensure u < v and v < w.
            for w in neighbors[i+1:]:
                if w in adj[v]:
                    # Found a triangle (u, v, w)
                    # Sort them to satisfy a < b < c
                    triplet = sorted([u, v, w])
                    found_triplets.append(tuple(triplet))
    
    if found_triplets:
        found_triplets.sort()
        print(f"{found_triplets[0][0]} {found_triplets[0][1]} {found_triplets[0][2]}")

solve()
