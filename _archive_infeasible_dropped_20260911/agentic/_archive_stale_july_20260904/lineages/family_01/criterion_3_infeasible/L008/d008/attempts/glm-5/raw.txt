import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    best = None
    
    for u, v in edges:
        if u > v:
            u, v = v, u
        au = adj[u]
        av = adj[v]
        if len(au) <= len(av):
            small, large = au, av
        else:
            small, large = av, au
        for w in small:
            if w in large:
                a, b, c = u, v, w
                if a > b:
                    a, b = b, a
                if b > c:
                    b, c = c, b
                    if a > b:
                        a, b = b, a
                if best is None or (a, b, c) < best:
                    best = (a, b, c)
                    if best == (1, 2, 3):
                        print(1, 2, 3)
                        return
    
    print(best[0], best[1], best[2])

solve()
