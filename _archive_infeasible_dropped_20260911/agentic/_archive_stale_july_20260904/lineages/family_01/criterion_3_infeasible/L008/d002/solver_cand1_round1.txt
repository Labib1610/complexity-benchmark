import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To minimize (i, j, k) lexicographically:
    # Iterate through all edges (u, v) as potential first two members.
    # Since we need to minimize i, then j, then k, we sort edges.
    edges.sort()
    
    # To find a triangle (i, j, k) with i < j < k:
    # For each edge (u, v) with u < v, we look for a common neighbor w.
    # To ensure we find the lexicographically smallest (i, j, k),
    # we can iterate through all possible i, then j, then k.
    
    # Optimization: iterate i from 1 to n, j from i+1 to n.
    # If (i, j) is an edge, check all neighbors of i that are also neighbors of j.
    
    for i in range(1, n + 1):
        # Sort neighbors to ensure we find smallest k first
        neighbors_i = sorted(list(adj[i]))
        for j in neighbors_i:
            if j <= i:
                continue
            
            # Find common neighbors k > j
            # Intersection of adj[i] and adj[j]
            common = sorted(list(adj[i].intersection(adj[j])))
            for k in common:
                if k > j:
                    print(f"{i} {j} {k}")
                    return

solve()
