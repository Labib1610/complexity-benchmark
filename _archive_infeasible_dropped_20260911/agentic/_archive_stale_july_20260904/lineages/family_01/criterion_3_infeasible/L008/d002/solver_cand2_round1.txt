import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) with a < b < c
    # minimizing a, then b, then c.
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # To minimize b, we iterate through sorted neighbors of a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need to find c such that c is a neighbor of a and b,
            # and c > b.
            # Intersection of adj[a] and adj[b]
            # Since we want to minimize c, we check neighbors of b that are also in adj[a]
            # and are > b.
            neighbors_b = adj[b]
            
            # To optimize, iterate through the smaller neighbor set
            if len(neighbors_a) < len(neighbors_b):
                for c in neighbors_a:
                    if c > b and c in neighbors_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                # Sort neighbors of b to find smallest c > b
                sorted_neighbors_b = sorted(list(neighbors_b))
                for c in sorted_neighbors_b:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
