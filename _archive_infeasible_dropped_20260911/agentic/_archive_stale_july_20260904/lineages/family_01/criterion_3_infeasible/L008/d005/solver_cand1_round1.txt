```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # A deadlock is a triangle (a, b, c).
    # Primary: (a, b), Secondary: (b, c), Closing: (a, c)
    # The workers are a, b, c.
    # The problem asks for the triplet (a, b, c) such that a < b < c.
    # We need to find all triangles and pick the lexicographically smallest one.
    
    triangles = []
    
    # To find triangles efficiently:
    # Iterate over each edge (u, v) and check common neighbors.
    # To avoid duplicates and ensure efficiency, iterate over the vertex with smaller degree.
    
    # Sort edges to help with lexicographical order if needed, 
    # but we can just collect all and sort at the end.
    
    for u in range(1, n + 1):
        neighbors = sorted(list(adj[u]))
        for i in range(len(neighbors)):
            v = neighbors[i]
            if v <= u:
                continue
            # Find common neighbors of u and v
            # Intersection of adj[u] and adj[v]
            # Since we want triangles (a, b, c) with a < b < c,
            # for each edge (u, v), any common neighbor w forms a triangle.
            # The triangle is sorted(u, v, w).
            
            # Optimization: only check neighbors w > v to avoid duplicates
            for w in neighbors[i+1:]:
                if w in adj[v]:
                    tri = sorted([u, v, w])
                    triangles.append(tuple(tri))
                    
    triangles.sort()
    print(f"{triangles[0][0]} {triangles[0][1]} {triangles[0][2]}")

if __name__ == '__main__':
    solve()
```
