import sys

def main():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1

    # adjacency as sorted lists for binary search
    adj = [[] for _ in range(n + 1)]
    # also store as sets for fast lookup
    adj_set = [set() for _ in range(n + 1)]

    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        adj_set[u].add(v)
        adj_set[v].add(u)

    # Build sorted adjacency lists
    for i in range(1, n + 1):
        adj[i] = sorted(adj_set[i])

    # For each vertex a, we want to find the smallest b > a such that
    # there exists c > b with edges a-b, a-c, b-c.
    # We iterate a from 1 to n, and for each a, iterate b in sorted order
    # among neighbors of a that are > a. For each such b, we need to find
    # the smallest c > b that is a common neighbor of a and b.

    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # neighbors of a that are > a
        # We iterate b in sorted order
        for b in neighbors_a:
            if b <= a:
                continue
            # find smallest c > b that is in both adj[a] and adj[b]
            # iterate over the smaller adjacency list
            if len(adj_set[a]) <= len(adj_set[b]):
                smaller = adj[a]
                larger_set = adj_set[b]
            else:
                smaller = adj[b]
                larger_set = adj_set[a]
            for c in smaller:
                if c > b and c in larger_set:
                    print(a, b, c)
                    return
    # Should not reach here given problem guarantees
    print(-1)

main()
