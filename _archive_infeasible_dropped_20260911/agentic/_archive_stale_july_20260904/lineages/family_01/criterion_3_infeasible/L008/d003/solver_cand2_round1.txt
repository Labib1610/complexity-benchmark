import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A triad (a, b, c) exists if there is a focal worker 'a' 
    # and two colleagues 'b' and 'c' such that:
    # 1. (a, b) is an incident
    # 2. (a, c) is an incident
    # 3. (b, c) is an incident
    # This is equivalent to finding a triangle in the graph.
    
    # To minimize (a, b, c) lexicographically:
    # Iterate a from 1 to n
    # Iterate b in adj[a] such that b > a
    # Iterate c in adj[a] such that c > b
    # Check if (b, c) is an edge
    
    # To optimize, sort adjacency lists
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors = sorted_adj[a]
        len_n = len(neighbors)
        for i in range(len_n):
            b = neighbors[i]
            if b <= a:
                continue
            for j in range(i + 1, len_n):
                c = neighbors[j]
                # Check if (b, c) is an edge
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
