import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find three workers {a, b, c} such that all three pairs
    # (a, b), (b, c), (a, c) exist in the incident files.
    # This is equivalent to finding a triangle in the graph.
    # To minimize a, then b, then c, we iterate in order.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        if len_a < 2:
            continue
            
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that c is in adj[a] and c is in adj[b]
            # and c > b
            # Intersection of neighbors_a and neighbors_b
            # Since we want to minimize b and then c, we iterate b then c
            
            # Optimization: intersect neighbors_a and neighbors_b
            # Only consider neighbors > b
            for j in range(i + 1, len_a):
                c = neighbors_a[j]
                if c in adj[b]:
                    # Found the triplet (a, b, c)
                    # Since we iterate a from 1, b from neighbors, c from neighbors,
                    # and we check b > a and c > b, this is the lexicographically smallest.
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
