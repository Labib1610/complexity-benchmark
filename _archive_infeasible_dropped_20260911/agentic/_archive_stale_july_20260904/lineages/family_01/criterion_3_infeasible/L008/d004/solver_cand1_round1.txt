import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
        idx += 2
        
    # A deadlock is a triangle (a, b, c) where each edge is one of the incident files.
    # The condition "every worker appears in exactly two of the three files"
    # is satisfied if and only if the three files form a triangle (a, b), (b, c), (a, c).
    # In a triangle (a, b, c), worker 'a' is in (a, b) and (a, c),
    # worker 'b' is in (a, b) and (b, c),
    # worker 'c' is in (a, c) and (b, c).
    # Each worker appears in exactly 2 files.
    
    # We need to find all triangles and pick the lexicographically smallest (a, b, c).
    # Since n <= 2000, we can iterate over all edges (u, v) and check common neighbors.
    
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    best = None
    
    # To minimize (a, b, c), we iterate a from 1 to n, b from a+1 to n, c from b+1 to n.
    # But we only care about triangles.
    for a in range(1, n + 1):
        neighbors = sorted(adj[a])
        for i in range(len(neighbors)):
            b = neighbors[i]
            if b <= a:
                continue
            # Check if there exists c > b such that (a, c) and (b, c) are edges
            # Intersection of adj[a] and adj[b]
            common = adj_sets[a].intersection(adj_sets[b])
            for c in common:
                if c > b:
                    # Found a triangle (a, b, c)
                    if best is None or (a < best[0]) or (a == best[0] and b < best[1]) or (a == best[0] and b == best[1] and c < best[2]):
                        best = (a, b, c)
        
        # Optimization: if we found a triangle with 'a', and we are past 'a', 
        # we can't find a smaller 'a'.
        if best and best[0] < a:
            break
            
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
