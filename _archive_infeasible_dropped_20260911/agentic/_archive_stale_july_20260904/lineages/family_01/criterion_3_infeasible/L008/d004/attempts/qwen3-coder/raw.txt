import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to process in order
    edges.sort()
    
    # For each edge, look for a common neighbor to form a triangle
    for i in range(len(edges)):
        u, v = edges[i]
        # We want to find a node w such that:
        # (u,w) and (v,w) are also edges
        # And we need the triplet (u,v,w) such that each appears in exactly two edges
        # Which means u,v,w form a triangle
        
        # Common neighbors of u and v
        common_neighbors = adj[u] & adj[v]
        
        candidates = sorted(common_neighbors)
        for w in candidates:
            # Ensure u < v < w for consistency
            if w > v and w > u:
                # The three nodes u, v, w form a triangle
                # Check if this triplet gives us the lexicographically smallest deadlock
                triplet = sorted([u, v, w])
                a, b, c = triplet[0], triplet[1], triplet[2]
                
                # Since we're looking for the minimal triplet in lex order,
                # and we're iterating through edges in sorted order,
                # the first valid triplet we find should be the answer.
                print(a, b, c)
                return

if __name__ == "__main__":
    main()
