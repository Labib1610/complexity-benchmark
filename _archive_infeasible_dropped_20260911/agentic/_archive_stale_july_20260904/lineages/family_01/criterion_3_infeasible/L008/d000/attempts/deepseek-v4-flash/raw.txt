We can solve this by checking each possible triple in increasing order and stopping at the first valid one.  
Since n can be up to 2000, checking all triples naively would be too slow (O(n³)).  
Instead, we can use adjacency sets and iterate over pairs, then check for a common neighbor.

```python
import sys

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges for lexicographic order
    edges.sort()
    
    for u, v in edges:
        common = adj[u] & adj[v]
        if common:
            w = min(common)
            a, b, c = sorted([u, v, w])
            print(a, b, c)
            return

if __name__ == "__main__":
    main()
```
