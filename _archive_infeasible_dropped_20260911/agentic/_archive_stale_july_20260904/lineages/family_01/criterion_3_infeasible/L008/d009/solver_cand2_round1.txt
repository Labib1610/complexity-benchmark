import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Each file is a pair (u, v).
    # We need three files (u1, v1), (u2, v2), (u3, v3) such that their
    # symmetric difference is empty.
    # The symmetric difference is the set of workers appearing an odd number of times.
    # For the symmetric difference to be empty, every worker involved must appear
    # an even number of times (0 or 2).
    # With 3 files, each file has 2 workers, total 6 slots.
    # If each worker appears an even number of times, and there are 3 files,
    # the only way is for each worker to appear exactly twice.
    # This means the 3 files must form a triangle: (a, b), (b, c), (c, a).
    
    edges = []
    adj = [{} for _ in range(n + 1)]
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u][v] = i
        adj[v][u] = i
        idx += 2
        
    best = None
    
    # Find all triangles (a, b, c)
    # To minimize a, then b, then c, we iterate carefully.
    for a in range(1, n + 1):
        # Optimization: if we already found a triangle with a smaller 'a', we can stop.
        if best and a > best[0]:
            break
            
        nodes_a = sorted(adj[a].keys())
        for i in range(len(nodes_a)):
            b = nodes_a[i]
            if best and a == best[0] and b > best[1]:
                continue
            
            # Look for c such that (b, c) and (c, a) exist
            nodes_b = adj[b]
            for c in nodes_b:
                if c > b and c in adj[a]:
                    # Found triangle (a, b, c)
                    res = sorted([a, b, c])
                    if best is None or res < best:
                        best = res
                        
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
