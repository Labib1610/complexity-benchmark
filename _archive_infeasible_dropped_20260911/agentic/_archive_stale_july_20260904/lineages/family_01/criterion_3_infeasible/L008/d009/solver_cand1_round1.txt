import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    degree = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        degree[u] += 1
        degree[v] += 1
        idx += 2
        
    # T = deg(a) + deg(b) + deg(c)
    # E = number of edges (u, v) such that exactly one of u, v is in {a, b, c}
    # Let S = {a, b, c}.
    # Let E_in be the number of edges with both endpoints in S.
    # Let E_out be the number of edges with exactly one endpoint in S.
    # deg(a) + deg(b) + deg(c) = 2 * E_in + E_out
    # We want T = E_out + 6
    # 2 * E_in + E_out = E_out + 6
    # 2 * E_in = 6 => E_in = 3
    # This means the three workers must form a triangle (a, b, c).
    
    triangles = []
    for a in range(1, n + 1):
        adj_a = sorted(list(adj[a]))
        for i in range(len(adj_a)):
            b = adj_a[i]
            if b <= a:
                continue
            for j in range(i + 1, len(adj_a)):
                c = adj_a[j]
                if c in adj[b]:
                    triangles.append((a, b, c))
    
    triangles.sort()
    
    for a, b, c in triangles:
        # T = deg(a) + deg(b) + deg(c)
        # E = (deg(a) - 2) + (deg(b) - 2) + (deg(c) - 2)
        # T = E + 6 is always satisfied for any triangle.
        # The problem guarantees at least one valid group exists.
        # Since any triangle satisfies T = E + 6, we just need the lexicographically smallest.
        print(f"{a} {b} {c}")
        return

if __name__ == '__main__':
    solve()
