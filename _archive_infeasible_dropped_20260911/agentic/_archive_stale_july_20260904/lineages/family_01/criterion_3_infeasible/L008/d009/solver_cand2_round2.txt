import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # We need to find a triangle (a, b, c) such that all three edges (a,b), (b,c), (a,c) exist.
    # The problem states that a set of 3 workers {a, b, c} has exactly 3 incident files
    # contained within it. This is exactly the definition of a triangle in the graph.
    
    # To satisfy the tie-breaking rules (min a, then min b, then min c),
    # we iterate through possible a, b, c in increasing order.
    
    for a in range(1, n + 1):
        # Sort neighbors to help with ordering
        neighbors = sorted(list(adj[a]))
        for i in range(len(neighbors)):
            b = neighbors[i]
            if b <= a:
                continue
            
            # Check for common neighbors c such that b < c
            # c must be in adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    # Found a triangle (a, b, c)
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
