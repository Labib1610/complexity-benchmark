```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use a set for O(1) lookup of mandates
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A deadlock of 3 workers {a, b, c} exists if all three pairs 
    # (a,b), (b,c), (a,c) are in the mandate list.
    # This forms a triangle in the graph of mandates.
    # We need to find the lexicographically smallest (a, b, c) 
    # such that a < b < c and all three edges exist.
    
    # Iterate through all possible a, b, c to find the smallest triplet.
    # Since n <= 2000, O(n^3) is too slow.
    # However, we only need to check if a triangle exists.
    # We can iterate over a, then b, then check if c exists.
    
    for a in range(1, n + 1):
        # Get neighbors of a
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        if len_a < 2:
            continue
            
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that c > b and c is connected to both a and b
            # Intersection of neighbors of a and neighbors of b
            # Since we want the smallest a, then b, then c, we iterate
            # through sorted neighbors of b that are > b.
            
            neighbors_b = adj[b]
            for c in neighbors_a[i+1:]:
                if c in neighbors_b:
                    # Found the smallest a, then b, then c
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
