import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A group of 3 workers {a, b, c} is impossible to split into 2 shifts
    # if and only if they form a triangle (all three pairs are in the incident list).
    # We need to find the lexicographically smallest (a, b, c) such that
    # a < b < c and all pairs (a, b), (b, c), (a, c) are in the incident list.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        if len_a < 2:
            continue
            
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that b < c and c is connected to both a and b
            # To minimize (a, b, c), we iterate b and c in increasing order
            for j in range(i + 1, len_a):
                c = neighbors_a[j]
                # Check if b and c are connected
                if c in adj[b]:
                    # Found the smallest triplet (a, b, c)
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
