import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c,
    # we iterate through a from 1 to n-2, b from a+1 to n-1, and c from b+1 to n.
    # To optimize, we only check c if c is in both adj[a] and adj[b].
    # Since we want the lexicographically smallest, we iterate in order.
    
    # Sort adjacency lists to potentially speed up intersection
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider 'a' if it has at least 2 neighbors
        if len(adj[a]) < 2:
            continue
        
        # Neighbors of a that are greater than a
        neighbors_a = adj_list[a]
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a < b < c. Since neighbors_a is sorted, b > a is guaranteed.
            
            # Find common neighbors of a and b that are > b
            # Intersection of adj[a] and adj[b]
            # Since adj[a] and adj[b] are sorted, we can intersect efficiently
            set_b = adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
