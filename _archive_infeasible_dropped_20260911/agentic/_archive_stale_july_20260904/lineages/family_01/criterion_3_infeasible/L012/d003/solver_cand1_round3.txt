```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list using sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) such that a < b < c
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n
    # then b from a+1 to n, then c from b+1 to n.
    # A triangle exists if b is in adj[a] and c is in adj[a] and c is in adj[b].
    
    for a in range(1, n + 1):
        # Sort neighbors to help find smallest b and c
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c such that c is in adj[a] and c is in adj[b]
            # c must be > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            
            # Filter for c > b
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
```
