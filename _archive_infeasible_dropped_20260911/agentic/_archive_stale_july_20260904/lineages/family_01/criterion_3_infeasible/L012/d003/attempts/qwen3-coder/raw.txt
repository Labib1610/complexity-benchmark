import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        edges.append((u, v))
        index += 2
    
    # Build adjacency list
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # We need to find a triangle in the complement graph
    # That is, three nodes such that none of them are connected in the original graph
    # But since we're told that such a triangle exists, we can just search for it
    
    # Let's reframe: we want three nodes a,b,c such that none of the pairs (a,b), (a,c), (b,c) are in edges
    # So we want three nodes that form an independent set of size 3 in the original graph
    
    # Let's create a set of all edges for fast lookup
    edge_set = set()
    for u, v in edges:
        if u > v:
            u, v = v, u
        edge_set.add((u, v))
    
    # Now let's iterate through all combinations of 3 nodes and find one that has no edges between them
    # Since we want the lexicographically smallest, we iterate in order
    for a in range(1, n+1):
        for b in range(a+1, n+1):
            # Check if a and b are NOT connected
            if (a, b) not in edge_set:
                for c in range(b+1, n+1):
                    # Check if a and c are NOT connected and b and c are NOT connected
                    if (a, c) not in edge_set and (b, c) not in edge_set:
                        print(a, b, c)
                        return

if __name__ == "__main__":
    main()
