import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Build adjacency list
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all possible a from 1 to n-2
    for a in range(1, n - 1):
        # Get all neighbors of a
        neighbors_a = sorted(list(adj[a]))
        # For each pair of neighbors b, c of a, check if b and c are connected
        # To ensure a < b < c, we only consider neighbors > a
        valid_neighbors = [x for x in neighbors_a if x > a]
        
        # Sort to ensure we check b and c in lexicographical order
        valid_neighbors.sort()
        
        num_neighbors = len(valid_neighbors)
        for i in range(num_neighbors):
            b = valid_neighbors[i]
            for j in range(i + 1, num_neighbors):
                c = valid_neighbors[j]
                # Check if b and c are connected
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
