```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
    
    # To find the lexicographically smallest (a, b, c) with a < b < c
    # We iterate through a from 1 to n-2
    # Then b from a+1 to n-1
    # Then check if c exists such that (a,b), (b,c), (a,c) are all edges
    # To optimize, we only check c in the intersection of adj[a] and adj[b]
    
    # Sort adjacency lists to help with lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Optimization: only consider b that are connected to a
        for b in adj[a]:
            if b <= a:
                continue
            
            # Find common neighbors c > b
            # Intersection of adj[a] and adj[b]
            # Since adj lists are sorted, we can find common elements efficiently
            # But since we need the smallest c, we can just iterate through the smaller list
            
            # To ensure lexicographical order, we check candidates for c
            # c must be in adj[a] and adj[b] and c > b
            
            # Use the smaller adjacency list to iterate
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
                
            for c in candidates:
                if c > b and c in adj_sets[a] and c in adj_sets[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
