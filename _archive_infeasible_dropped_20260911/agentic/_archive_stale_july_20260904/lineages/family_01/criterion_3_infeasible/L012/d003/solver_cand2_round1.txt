import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # The problem asks for a trio (a, b, c) such that no two are connected by an interference record.
    # This is equivalent to finding an independent set of size 3 in the interference graph.
    # Let G be the interference graph. We want to find {a, b, c} such that
    # (a, b) not in E, (b, c) not in E, (a, c) not in E.
    # This is equivalent to finding a triangle in the complement graph G'.
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n-2,
    # b from a+1 to n-1, and c from b+1 to n.
    # Check if (a, b), (b, c), (a, c) are NOT in adj.
    
    # Optimization: For a fixed a, we need b > a such that (a, b) not in E.
    # Then we need c > b such that (a, c) not in E AND (b, c) not in E.
    
    for a in range(1, n + 1):
        # Candidates for b are nodes > a not connected to a
        non_neighbors_a = [b for b in range(a + 1, n + 1) if b not in adj[a]]
        
        for i in range(len(non_neighbors_a)):
            b = non_neighbors_a[i]
            # Candidates for c are nodes > b not connected to a AND not connected to b
            for j in range(i + 1, len(non_neighbors_a)):
                c = non_neighbors_a[j]
                if c not in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
