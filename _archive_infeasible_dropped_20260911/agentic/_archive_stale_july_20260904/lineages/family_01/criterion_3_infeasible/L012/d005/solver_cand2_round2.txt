import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through all possible a, then b, then check if c exists.
    # Since we need the lexicographically smallest, we iterate a from 1 to n,
    # b from a+1 to n, and check if there exists c > b such that
    # c is in adj[a] and c is in adj[b].
    
    # Sort adjacency lists to make intersection faster and ordered
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # Optimization: iterate through edges (u, v) as potential (a, b)
    # or iterate through all pairs (a, b) that have an edge.
    # Given n=2000, O(n^2) is acceptable if the inner check is fast.
    
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b from a+1 to n.
    for a in range(1, n + 1):
        # Only consider b that are connected to a
        for b in adj[a]:
            if b <= a:
                continue
            
            # Now find the smallest c > b such that c is in adj[a] and adj[b]
            # Since adj lists are sorted, we can use set intersection or two pointers
            # But since we want the smallest c, we can just iterate through adj[a]
            # and check if it's in adj[b].
            
            # To make this efficient, use a boolean array or set for adj[b]
            # But we only need to check for a specific b.
            set_b = set(adj[b])
            for c in adj[a]:
                if c > b and c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
