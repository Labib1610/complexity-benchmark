```python
import sys

def solve():
    # A subset of three elements {a, b, c} is locked if its coherence product is 0
    # for every assignment A: {a, b, c} -> {0, 1}.
    # The coherence product is the product of (A(u) - A(v))^2 for all links {u, v} in T.
    # (A(u) - A(v))^2 is 0 if A(u) == A(v), and 1 if A(u) != A(v).
    # So the product is 0 if there exists at least one link {u, v} such that A(u) == A(v).
    # The subset is locked if for every assignment A, there is a link {u, v} with A(u) == A(v).
    # This is equivalent to saying that the graph induced by the three elements
    # is NOT 2-colorable (i.e., it contains an odd cycle).
    # A triangle (3-cycle) is the only way to have an odd cycle in a 3-element graph.
    # Thus, a subset of 3 elements is locked if and only if it forms a triangle.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
    
    # We need the lexicographically smallest triangle {a, b, c} with a < b < c.
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Iterate b from a+1 to n-1
        # To optimize, only consider b that are connected to a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for b in neighbors_a:
            # Iterate c from b+1 to n
            # c must be connected to both a and b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            # Filter c > b
            valid_c = [x for x in common if x > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
```
