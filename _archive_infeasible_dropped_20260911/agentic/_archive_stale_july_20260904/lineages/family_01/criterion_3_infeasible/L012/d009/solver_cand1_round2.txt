```python
import sys

def solve():
    # A triplet {a, b, c} is unresolvable if and only if it contains an odd cycle.
    # In a triplet, the only possible odd cycle is a triangle (a cycle of length 3).
    # If a triplet contains a triangle, say {a, b, c} are all connected to each other,
    # then any assignment of 0s and 1s to {a, b, c} will have at least two nodes
    # with the same mode (by Pigeonhole Principle), and since they are all connected,
    # that edge will be conflicting.
    # Thus, a triplet is unresolvable if and only if it forms a triangle (a clique of size 3).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triangle {a, b, c} with a < b < c:
    # Iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    
    # Sort adjacency lists to help with lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Iterate b in adj[a] such that b > a
        for b in adj[a]:
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Since adj[a] and adj[b] are sorted, we can find common elements efficiently
            # or just iterate through the smaller list.
            
            # Optimization: intersect adj[a] and adj[b]
            # Only need to check c > b
            set_a = adj_sets[a]
            set_b = adj_sets[b]
            
            # To find the smallest c > b, iterate through the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                for c in adj[a]:
                    if c > b and c in set_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in adj[b]:
                    if c > b and c in set_a:
                        print(f"{a} {b} {c}")
                        return

solve()
```
