```python
import sys

def solve():
    # A group of three processes cannot be scheduled on two processors if and only if
    # they form a triangle (a clique of size 3) in the constraint graph.
    # We need to find the lexicographically smallest triangle (a, b, c) with a < b < c.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # c is in adj[a] and c is in adj[b].
    
    # Optimization: sort adjacency lists to make intersection faster
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider neighbors b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # Since sorted_adj is sorted, we can find common elements efficiently
            # or just use set intersection.
            
            # Using set intersection for speed
            common = adj[a].intersection(adj[b])
            # Filter for c > b
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
```
