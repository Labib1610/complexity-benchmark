```python
import sys

def solve():
    # The problem asks for a group of three processes such that every possible 
    # allocation to two processors violates at least one constraint.
    # With 3 processes, the only way to violate the 2-processor constraint 
    # is if all three processes are mutually exclusive (a triangle in the graph).
    # If they form a triangle, any assignment of 3 processes to 2 processors 
    # will force at least two processes to be on the same processor by the 
    # Pigeonhole Principle, violating the mutual exclusion constraint.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    
    # Sort edges to help find triangles efficiently
    # For each edge (u, v) with u < v, check all w > v such that (u, w) and (v, w) exist.
    
    best = None
    
    # Iterate through all u < v < w
    # To optimize, iterate through u, then iterate through neighbors v of u (v > u),
    # then check if there is a common neighbor w > v.
    
    for u in range(1, n + 1):
        neighbors_u = sorted([v for v in adj[u] if v > u])
        for i in range(len(neighbors_u)):
            v = neighbors_u[i]
            # Intersection of adj[u] and adj[v]
            # We only care about w > v
            common = adj[u].intersection(adj[v])
            for w in common:
                if w > v:
                    if best is None or (u, v, w) < best:
                        best = (u, v, w)
        
        # Since we want the lexicographically smallest, if we found a triangle
        # starting with u, we don't necessarily need to check larger u.
        # However, we need to ensure we found the smallest (a, b, c).
        # The current loop structure finds them in lexicographical order if we 
        # break early or just keep track of the minimum.
        if best and best[0] < u:
            break
            
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
