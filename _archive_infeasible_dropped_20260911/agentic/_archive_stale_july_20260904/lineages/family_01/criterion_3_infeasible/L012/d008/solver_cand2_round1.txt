import sys

def solve():
    # The problem asks for a group of three processes {u, v, w} such that
    # if we consider only the constraints between them, we get a contradiction.
    # A constraint (u, v) means u and v must be on different processors.
    # A set of three processes {u, v, w} forms a contradiction if and only if
    # all three pairs (u, v), (v, w), and (u, w) are present in the constraints.
    # This is because if we have edges (u, v), (v, w), and (u, w):
    # - u on P1 => v on P2
    # - v on P2 => w on P1
    # - w on P1 => u on P2 (Contradiction: u is on P1 and P2)
    # Thus, we are looking for a triangle in the graph.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all possible 'a' from 1 to n-2
    # Then iterate through all 'b' in adj[a] such that b > a
    # Then check if there exists 'c' in adj[a] AND adj[b] such that c > b
    
    # Sort adjacency lists to ensure we find the smallest c
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a < b < c
            # Check intersection of neighbors_a and neighbors_b
            # Only need to check c > b
            neighbors_b = adj_sets[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
