import sys

def solve():
    # The problem asks for a subset of 3 sensors that cannot be assigned to 2 phases.
    # A set of 3 sensors cannot be assigned to 2 phases if and only if they form a triangle (K3).
    # If they form a triangle, any assignment of 3 sensors into 2 phases will result in 
    # at least two sensors being in the same phase, and since they are all connected, 
    # those two will violate the rule.
    # We need to find the lexicographically smallest triangle (a, b, c) with a < b < c.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest triangle (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we want the smallest, we sort the adjacency lists.
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    # Iterate through all possible a
    for a in range(1, n + 1):
        # Iterate through all possible b > a
        for b in sorted_adj[a]:
            if b <= a:
                continue
            
            # Check if there exists c > b such that (a, c) and (b, c) are edges
            # To keep it lexicographically smallest, we check c in increasing order
            # Intersection of sorted_adj[a] and sorted_adj[b] gives potential c's
            # Since sorted_adj is sorted, we can find common elements efficiently
            
            ptr_a = 0
            ptr_b = 0
            list_a = sorted_adj[a]
            list_b = sorted_adj[b]
            
            while ptr_a < len(list_a) and ptr_b < len(list_b):
                val_a = list_a[ptr_a]
                val_b = list_b[ptr_b]
                
                if val_a == val_b:
                    c = val_a
                    if c > b:
                        print(f"{a} {b} {c}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
