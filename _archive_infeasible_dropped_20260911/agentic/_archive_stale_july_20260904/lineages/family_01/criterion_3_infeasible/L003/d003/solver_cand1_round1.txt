```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all edges (u, v) and check for a common neighbor w.
    # To ensure a < b < c, we check:
    # 1. w < u < v: (w, u, v)
    # 2. u < w < v: (u, w, v)
    # 3. u < v < w: (u, v, w)
    
    best = None
    
    # Sort edges to process in a way that helps find the lexicographical minimum
    edges.sort()
    
    for u, v in edges:
        # Find common neighbors
        # We only care about w such that (u, v, w) sorted is the smallest
        # Intersection of adj[u] and adj[v]
        common = adj[u].intersection(adj[v])
        
        for w in common:
            # Form the triplet and sort it
            triplet = sorted([u, v, w])
            
            if best is None or triplet < best:
                best = triplet
                
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
