We can solve this by checking all triples, but that would be too slow for \( n = 2000 \).  
Instead, we can use an adjacency matrix (or set) and iterate over edges, checking for a common neighbor.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n)]
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    for u, v in edges:
        common = adj[u] & adj[v]
        if common:
            w = min(common)
            a, b, c = sorted([u+1, v+1, w+1])
            print(a, b, c)
            return

if __name__ == "__main__":
    solve()
```
