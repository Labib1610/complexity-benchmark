import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    d = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        d[u] += 1
        d[v] += 1
        idx += 2
        
    # The condition:
    # Let S = {a, b, c}.
    # Let G be the set of all grievances.
    # Let G_S be the set of grievances where at least one endpoint is in S.
    # The number of grievances where neither endpoint is in S is:
    # m - |G_S| = m - (d(a) + d(b) + d(c) - (|{a,b} in G| + |{b,c} in G| + |{a,c} in G|))
    # The condition given is:
    # m - |G_S| = m - d(a) - d(b) - d(c) + 3
    # This simplifies to:
    # |G_S| = d(a) + d(b) + d(c) - 3
    # Substituting |G_S|:
    # d(a) + d(b) + d(c) - (|{a,b} in G| + |{b,c} in G| + |{a,c} in G|) = d(a) + d(b) + d(c) - 3
    # Which simplifies to:
    # |{a,b} in G| + |{b,c} in G| + |{a,c} in G| = 3
    # This means the three employees must form a triangle in the grievance graph.
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # Check common neighbors c > b
                # To be lexicographically smallest, we iterate a, then b, then c
                # The first triangle we find will be the lexicographically smallest
                for c in range(b + 1, n + 1):
                    if c in adj[a] and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
