```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)          # bit i-1 corresponds to vertex i
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    for a in range(1, n + 1):
        a_bits = adj[a]
        for b in range(a + 1, n + 1):
            if (a_bits >> (b - 1)) & 1:          # edge a-b exists
                common = a_bits & adj[b]        # vertices adjacent to both a and b
                mask = common >> b               # keep only vertices > b
                if mask:
                    low = mask & -mask            # lowest set bit
                    idx = low.bit_length() - 1   # position relative to shifted mask
                    c = b + 1 + idx
                    sys.stdout.write(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
```
