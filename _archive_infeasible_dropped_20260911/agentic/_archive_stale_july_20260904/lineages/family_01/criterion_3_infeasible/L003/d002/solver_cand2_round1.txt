```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through possible a, then b, then c.
    # A trio (a, b, c) with a < b < c exists if:
    # 1. (a, b) is an edge
    # 2. (a, c) is an edge
    # 3. (b, c) is an edge
    
    # Sort edges to help find smallest a, b, c
    edges.sort()
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Iterate through all pairs (b, c) in neighbors_a
        # such that b < c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if (b, c) is an edge
                # Since b < c, check if c is in adj[b]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

    print("0 0 0")

if __name__ == '__main__':
    solve()
```
