import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we iterate b in adj[a] such that b > a.
    # Then we check if there exists c in adj[a] such that c > b and c in adj[b].
    
    # Sort adjacency lists to help find smallest c
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        # Iterate b in adj[a] where b > a
        sorted_neighbors_a = [x for x in adj[a] if x > a]
        for b in sorted_neighbors_a:
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Since adj[a] is sorted, we can iterate through it
            for c in adj[a]:
                if c <= b:
                    continue
                if c in adj_sets[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
