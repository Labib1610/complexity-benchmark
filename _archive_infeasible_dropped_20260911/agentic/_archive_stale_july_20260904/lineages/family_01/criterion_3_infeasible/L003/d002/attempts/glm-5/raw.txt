import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].add(v)
        adj[v].add(u)

    # Iterate through all possible triplets in lexicographical order
    for a in range(1, n + 1):
        # We need adj[a] to contain at least two other nodes b and c
        # such that b and c are also connected.
        # Since we iterate a in increasing order, the first valid triplet found
        # will be the lexicographically smallest.
        
        neighbors = adj[a]
        # Convert to sorted list to iterate b in increasing order
        # This ensures we find the smallest b first for a given a
        sorted_neighbors = sorted(list(neighbors))
        
        for b in sorted_neighbors:
            if b <= a:
                continue
            
            # Check if b connects to any c that also connects to a
            # We iterate c in increasing order to find the smallest valid c
            for c in sorted_neighbors:
                if c <= b:
                    continue
                
                # Check if edge (b, c) exists
                if c in adj[b]:
                    # Found the lexicographically smallest triplet
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
