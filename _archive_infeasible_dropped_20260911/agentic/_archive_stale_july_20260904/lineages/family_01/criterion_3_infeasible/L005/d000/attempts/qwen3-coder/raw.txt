```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list and edge set for O(1) lookup
    adj = defaultdict(set)
    edges = set()
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        adj[u].add(v)
        adj[v].add(u)
        edges.add((min(u, v), max(u, v)))
    
    # Convert adjacency sets to lists for ordered iteration
    adj_list = {u: sorted(list(adj[u])) for u in range(1, n+1)}
    
    best = None
    
    # Iterate through all pairs of nodes
    for a in range(1, n+1):
        neighbors_a = adj_list[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            # Now look for a common neighbor c of a and b such that c > b
            neighbors_b = adj_list[b]
            j = 0
            # Use two pointers to find common neighbors efficiently
            while j < len(neighbors_b) and neighbors_b[j] <= b:
                j += 1
            
            k = 0
            while k < len(neighbors_a) and j < len(neighbors_b):
                c_a = neighbors_a[k]
                c_b = neighbors_b[j]
                
                if c_a < c_b:
                    k += 1
                elif c_a > c_b:
                    j += 1
                else:
                    # Found a common neighbor
                    c = c_a
                    if c > b:
                        candidate = (a, b, c)
                        if best is None or candidate < best:
                            best = candidate
                    k += 1
                    j += 1
    
    print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
```
