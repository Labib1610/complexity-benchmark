import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    iterator = iter(data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].add(v)
        adj[v].add(u)
    
    # To find the lexicographically smallest triangle (a, b, c),
    # we iterate a from 1 to n, then b from a+1 to n, then c from b+1 to n.
    # We check if (a,b), (a,c), (b,c) are all edges.
    # To optimize, we iterate a, then b > a where (a,b) is an edge,
    # then c > b where (a,c) and (b,c) are edges.
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # We need to iterate through neighbors b of a such that b > a
        for b in neighbors_a:
            if b > a:
                neighbors_b = adj[b]
                # We need to find c such that c > b, c is neighbor of a, and c is neighbor of b
                # Iterate through neighbors of a that are > b
                for c in neighbors_a:
                    if c > b and c in neighbors_b:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
