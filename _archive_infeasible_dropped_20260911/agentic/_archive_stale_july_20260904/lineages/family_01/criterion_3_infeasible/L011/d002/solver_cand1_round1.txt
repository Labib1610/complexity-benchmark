```python
import sys

def solve():
    # The problem asks for a group of three steps such that no two steps 
    # can be on the same line. This is equivalent to finding a triangle 
    # (a clique of size 3) in the graph where edges represent 
    # "cannot be on the same line".
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    #   Iterate b from a+1 to n-1
    #     If b is connected to a:
    #       Check if there exists c > b such that c is connected to both a and b
    
    # Convert sets to sorted lists for faster iteration if needed, 
    # but set lookup is O(1).
    
    for a in range(1, n + 1):
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # To keep it efficient, iterate through the smaller neighbor list
            if len(neighbors_a) - i - 1 < len(adj[b]):
                for j in range(i + 1, len(neighbors_a)):
                    c = neighbors_a[j]
                    if c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                neighbors_b = sorted([x for x in adj[b] if x > b])
                for c in neighbors_b:
                    if c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
```
