import sys

def solve():
    # The problem asks for a subset of 3 steps {a, b, c} such that all 8 
    # assignments to 2 lines result in a collision.
    # A collision occurs if two steps in the subset are connected by an edge.
    # For a subset of 3 nodes to have a collision in every possible 2-coloring,
    # the subset must contain a triangle (a 3-cycle).
    # If a subset of 3 nodes contains a triangle, any 2-coloring will force
    # at least two nodes of the triangle to be the same color, causing a collision.
    # Thus, the problem is equivalent to finding the lexicographically smallest triangle.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triangle
    edges.sort()
    
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check for a common neighbor c > b.
    # Since we want the smallest (a, b, c), we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # (a, c) and (b, c) are edges.
    
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that b < c and c is in adj[a] and adj[b]
            # To keep it efficient, intersect the sorted neighbor lists
            neighbors_b = adj[b]
            for c in neighbors_a[i+1:]:
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
