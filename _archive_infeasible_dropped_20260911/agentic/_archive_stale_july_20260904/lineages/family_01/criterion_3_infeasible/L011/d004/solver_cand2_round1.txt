```python
import sys

def solve():
    # The problem asks for a group of three steps that cannot be assigned to two lines.
    # This is equivalent to finding a triangle (a clique of size 3) in the conflict graph.
    # If three steps form a triangle, each pair has a conflict, so no two can be on the same line.
    # We need the lexicographically smallest (a, b, c) with a < b < c.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c), we iterate through possible a, b, c.
    # Since n <= 2000, we can iterate through all edges (u, v) and check for a common neighbor w.
    # To ensure we find the smallest (a, b, c), we sort the edges and check neighbors.
    
    # Sort edges to help find the smallest (a, b, c)
    edges.sort()
    
    best = None
    
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # For a fixed 'a', we want the smallest 'b' and 'c'
        # 'b' must be a neighbor of 'a', 'c' must be a neighbor of 'a' and 'b'
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Optimization: if we already found a solution with a smaller 'a' or 'b',
            # we can prune. But since we iterate 'a' and 'b' in order, the first
            # triangle we find might not be the lexicographically smallest.
            # Actually, if we iterate a from 1..n and b from neighbors of a,
            # we just need to find the smallest c for each (a, b).
            
            # Find common neighbors of a and b
            common = sorted(list(adj[a].intersection(adj[b])))
            if common:
                c = common[0]
                # The triangle is (a, b, c) sorted
                res = sorted([a, b, c])
                if best is None or res < best:
                    best = res
        
        # If we have a best solution, and the current 'a' is already larger than best[0],
        # we can stop.
        if best and a >= best[0]:
            break
            
    print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
