```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # The vectors are v_i = (x_{i,1}, x_{i,2}, x_{i,3}) where x_{i,j} = 1 if i~j else 0.
    # v_1 = (0, 1 if 1~2 else 0, 1 if 1~3 else 0)
    # v_2 = (1 if 2~1 else 0, 0, 1 if 2~3 else 0)
    # v_3 = (1 if 3~1 else 0, 1 if 3~2 else 0, 0)
    # Let a = 1 if 1~2 else 0, b = 1 if 1~3 else 0, c = 1 if 2~3 else 0.
    # Matrix:
    # [[0, a, b],
    #  [a, 0, c],
    #  [b, c, 0]]
    # Determinant = 0*(0-c^2) - a*(0-bc) + b*(ac-0) = abc + abc = 2abc.
    # The vectors are linearly independent if and only if the determinant is non-zero.
    # 2abc != 0 iff a=1, b=1, c=1.
    # This means 1~2, 1~3, and 2~3 must all be true.
    # In other words, the three steps must form a triangle (a clique of size 3).
    
    # We need the lexicographically smallest (a, b, c) such that a < b < c
    # and {a, b, c} forms a triangle.
    
    for a in range(1, n + 1):
        # Sort neighbors to ensure we check combinations in lexicographical order
        neighbors = sorted(list(adj[a]))
        len_n = len(neighbors)
        for i in range(len_n):
            b = neighbors[i]
            if b <= a:
                continue
            # Check for common neighbors c > b
            for j in range(i + 1, len_n):
                c = neighbors[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
```
