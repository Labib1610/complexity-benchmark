import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # We iterate through all edges (u, v) with u < v.
    # For each edge, we look for a common neighbor w.
    # To ensure we find the lexicographically smallest, we can iterate
    # through possible values of a, then b, then c.
    
    # Sort edges to help find the smallest triangle
    edges.sort()
    
    # Optimization: iterate through all possible 'a' from 1 to n
    # Then iterate through neighbors 'b' of 'a' such that b > a
    # Then check if there exists 'c' such that c > b and c is in adj[a] and adj[b]
    
    # To make this efficient, we use the fact that we need the smallest (a, b, c)
    for a in range(1, n + 1):
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need to find if there exists c > b such that c in adj[a] and c in adj[b]
            # Intersection of adj[a] and adj[b]
            # Since we want the smallest c, we can iterate through sorted neighbors
            # But checking intersection is faster if we use the smaller degree set
            if len(adj[a]) < len(adj[b]):
                common = [x for x in adj[a] if x > b and x in adj[b]]
            else:
                common = [x for x in adj[b] if x > b and x in adj[a]]
            
            if common:
                print(f"{a} {b} {min(common)}")
                return

solve()
