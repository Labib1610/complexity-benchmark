import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A group of 3 steps {u, v, w} forms a bottleneck if exactly 24/64 states are relaxed.
    # Tension T(x_u, x_v, x_w) = |(x_u+u)%4 - (x_v+v)%4| * I(u,v) + 
    #                           |(x_v+v)%4 - (x_w+w)%4| * I(v,w) + 
    #                           |(x_w+w)%4 - (x_u+u)%4| * I(u,w)
    # where I(a,b) is 1 if a,b share a tool, else 0.
    # Let M_u = (x_u + u) % 4, M_v = (x_v + v) % 4, M_w = (x_w + w) % 4.
    # Since x_u, x_v, x_w are independent and uniform in {0,1,2,3},
    # M_u, M_v, M_w are also independent and uniform in {0,1,2,3}.
    # Tension = |M_u - M_v| * I_uv + |M_v - M_w| * I_vw + |M_w - M_u| * I_uw.
    # We need to count (M_u, M_v, M_w) such that Tension == 4.
    
    def count_relaxed(u, v, w):
        i_uv = 1 if v in adj[u] else 0
        i_vw = 1 if w in adj[v] else 0
        i_uw = 1 if w in adj[u] else 0
        
        count = 0
        for mu in range(4):
            for mv in range(4):
                for mw in range(4):
                    tension = abs(mu - mv) * i_uv + abs(mv - mw) * i_vw + abs(mw - mu) * i_uw
                    if tension == 4:
                        count += 1
        return count

    # Iterate through all triplets (a, b, c) with a < b < c
    # To optimize, only check triplets that have at least one edge
    # Actually, the problem implies the bottleneck is a triangle or connected structure.
    # Given the constraints and the nature of the problem, we can iterate efficiently.
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            # Optimization: if a and b are not connected, they might still be part of a bottleneck
            # but usually these problems involve triangles. Let's check all.
            for c in range(b + 1, n + 1):
                if count_relaxed(a, b, c) == 24:
                    print(f"{a} {b} {c}")
                    return

solve()
