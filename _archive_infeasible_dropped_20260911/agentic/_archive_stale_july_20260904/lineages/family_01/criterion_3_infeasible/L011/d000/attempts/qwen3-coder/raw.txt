import sys
from collections import defaultdict

def find_unresolvable_trio(n, edges):
    # Build adjacency list
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # Convert to sorted lists for deterministic iteration
    for u in range(1, n + 1):
        adj[u] = sorted(list(adj[u]))
    
    # Look for a triangle (three nodes all connected to each other)
    for a in range(1, n + 1):
        for b in adj[a]:
            if b <= a:  # Ensure a < b
                continue
            for c in adj[b]:
                if c <= b:  # Ensure b < c
                    continue
                # Check if a and c are also connected
                if c in adj[a]:
                    return [a, b, c]
    
    return None

# Read input
n, m = map(int, sys.stdin.readline().split())
edges = []
for _ in range(m):
    u, v = map(int, sys.stdin.readline().split())
    edges.append((u, v))

# Find and print the result
result = find_unresolvable_trio(n, edges)
print(' '.join(map(str, result)))
