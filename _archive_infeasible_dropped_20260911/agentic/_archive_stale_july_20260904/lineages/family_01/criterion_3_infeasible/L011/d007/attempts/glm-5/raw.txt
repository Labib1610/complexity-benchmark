import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    deg = [0] * (n + 1)
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u][v] = True
        adj[v][u] = True
        deg[u] += 1
        deg[v] += 1
    
    # Precompute sorted adjacency lists for each node
    adj_lists = []
    for u in range(n + 1):
        lst = []
        for v in range(1, n + 1):
            if adj[u][v]:
                lst.append(v)
        adj_lists.append(lst)
    
    # Threshold: if deg[u] * deg[v] > threshold, use hash set intersection
    threshold = 2 * m
    
    result = None
    
    for u in range(1, n + 1):
        for v in adj_lists[u]:
            if v <= u:
                continue
            
            # Find common neighbors w where w > v (to ensure u < v < w)
            common = []
            
            # Choose strategy based on degrees
            if deg[u] * deg[v] <= threshold:
                # Linear scan through smaller adjacency list
                if len(adj_lists[u]) <= len(adj_lists[v]):
                    small_list = adj_lists[u]
                    check_adj = adj[v]
                else:
                    small_list = adj_lists[v]
                    check_adj = adj[u]
                
                for w in small_list:
                    if w > v and check_adj[w]:
                        common.append(w)
            else:
                # Use hash set intersection
                set_u = set(adj_lists[u])
                for w in adj_lists[v]:
                    if w > v and w in set_u:
                        common.append(w)
            
            # Check each triangle (u, v, w)
            for w in common:
                # Count edges in triangle
                edges = 0
                if adj[u][v]:
                    edges += 1
                if adj[u][w]:
                    edges += 1
                if adj[v][w]:
                    edges += 1
                
                # Check if unresolvable bottleneck (exactly 12 relaxed states)
                # Based on analysis: 12 relaxed states occurs when edges = 2
                if edges == 2:
                    result = (u, v, w)
                    print(u, v, w)
                    return
    
    # Fallback (should not reach here as problem guarantees a solution)
    if result is None:
        # Brute force search as last resort
        for a in range(1, n - 1):
            for b in range(a + 1, n):
                for c in range(b + 1, n + 1):
                    edges = 0
                    if adj[a][b]:
                        edges += 1
                    if adj[a][c]:
                        edges += 1
                    if adj[b][c]:
                        edges += 1
                    if edges == 2:
                        print(a, b, c)
                        return

if __name__ == "__main__":
    solve()
