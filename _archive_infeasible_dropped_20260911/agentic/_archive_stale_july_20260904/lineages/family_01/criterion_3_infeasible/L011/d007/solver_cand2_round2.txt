import sys

def solve():
    # A group of three particles {a, b, c} is a frustrated trio if every 
    # assignment of spin states results in at least one alignment penalty.
    # This means the graph formed by these three particles must contain 
    # an odd cycle. Since the only cycle of length 3 is the triangle itself,
    # a group of three particles is a frustrated trio if and only if 
    # all three pairs (a,b), (b,c), and (a,c) are connected by coupling links.
    # In other words, the trio must form a triangle (a clique of size 3).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check for a common neighbor c.
    # To ensure a < b < c, we check c > b.
    
    # Sort edges to help find the lexicographically smallest triangle
    edges.sort()
    
    # Pre-sort adjacency lists for consistent iteration
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need to find c such that b < c and c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can intersect efficiently
            neighbors_b = adj[b]
            
            # Intersection of two sorted lists
            ptr_a = i + 1
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                if val_a == val_b:
                    print(f"{a} {b} {val_a}")
                    return
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
