```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c)
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and adj[b]
    
    # Sort neighbors to potentially speed up intersection
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Only consider neighbors > a
        valid_b = [b for b in neighbors_a if b > a]
        
        for i in range(len(valid_b)):
            b = valid_b[i]
            # Find common neighbors of a and b that are > b
            # Since neighbors are sorted, we can intersect efficiently
            # But for n=2000, simple set intersection is fine
            common = adj[a].intersection(adj[b])
            
            c_list = [c for c in common if c > b]
            if c_list:
                print(f"{a} {b} {min(c_list)}")
                return

if __name__ == '__main__':
    solve()
```
