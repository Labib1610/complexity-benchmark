```python
import sys

def solve():
    # The problem asks for a minimal certificate of three materials {a, b, c}
    # such that no matter how they are divided into two units, at least two
    # materials in the same unit are volatile.
    # This means the subgraph induced by {a, b, c} must contain at least 
    # 3 edges (a triangle). If it has 3 edges, any partition of 3 nodes into 
    # 2 sets will put at least two nodes in the same set, and since it's a 
    # triangle, those two nodes will be connected by an edge.
    # We need the lexicographically smallest {a, b, c} that forms a triangle.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check for a common neighbor c.
    # Since we want the smallest (a, b, c), we sort the edges or iterate carefully.
    
    edges.sort()
    
    best = None
    
    for a, b in edges:
        # We look for c such that (a, c) and (b, c) exist.
        # To ensure a < b < c, we check c > b.
        # To ensure lexicographical order, we check candidates for c.
        
        # Intersection of adj[a] and adj[b]
        common = adj[a].intersection(adj[b])
        for c in common:
            # Ensure a < b < c
            nodes = sorted([a, b, c])
            if best is None or nodes < best:
                best = nodes
                
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
