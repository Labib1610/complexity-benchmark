import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    
    deg = [0] * (n + 1)
    edge_list = []
    for _ in range(m):
        u = int(input_data[idx]); idx += 1
        v = int(input_data[idx]); idx += 1
        edge_list.append((u, v))
        deg[u] += 1
        deg[v] += 1
    
    adj = [set() for _ in range(n + 1)]
    for u, v in edge_list:
        adj[u].add(v)
        adj[v].add(u)
    
    # Orient edges: lower (degree, vertex) -> higher (degree, vertex)
    out_adj = [set() for _ in range(n + 1)]
    for u, v in edge_list:
        if (deg[u], u) < (deg[v], v):
            out_adj[u].add(v)
        else:
            out_adj[v].add(u)
    
    best = None
    for v in range(1, n + 1):
        # Mark out-neighbors of v
        marked = set()
        for u in out_adj[v]:
            marked.add(u)
        # For each out-neighbor u, check out-neighbors of u that are also marked
        for u in out_adj[v]:
            for w in out_adj[u]:
                if w in marked:
                    tri = tuple(sorted([v, u, w]))
                    if best is None or tri < best:
                        best = tri
    
    print(f"{best[0]} {best[1]} {best[2]}")
