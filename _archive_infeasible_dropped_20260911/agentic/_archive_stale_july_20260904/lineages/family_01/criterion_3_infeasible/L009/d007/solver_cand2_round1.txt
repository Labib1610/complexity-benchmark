```python
import sys

def solve():
    # A Resilient Triad is a set of 3 nodes {a, b, c} such that
    # the subgraph induced by {a, b, c} is 2-edge-connected.
    # A graph with 3 nodes is 2-edge-connected if and only if it is a cycle of length 3 (a triangle).
    # Proof:
    # - If the induced subgraph has 3 edges (a triangle), removing any edge leaves a path of length 2,
    #   which is still connected.
    # - If the induced subgraph has < 3 edges, it is not 2-edge-connected.
    # So we just need to find the lexicographically smallest triangle in the graph.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Optimization: sort adjacency lists to iterate efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that c > b and c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since both are sorted, we can find common elements efficiently
            nb_a = neighbors_a
            nb_b = sorted_adj[b]
            
            ptr_a = i + 1
            ptr_b = 0
            
            while ptr_a < len(nb_a) and ptr_b < len(nb_b):
                val_a = nb_a[ptr_a]
                val_b = nb_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
```
