import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A trio {a, b, c} has internal reactivity scores:
    # s_a = (1 if b in adj[a] else 0) + (1 if c in adj[a] else 0)
    # s_b = (1 if a in adj[b] else 0) + (1 if c in adj[b] else 0)
    # s_c = (1 if a in adj[c] else 0) + (1 if b in adj[c] else 0)
    # The product s_a * s_b * s_c = 8.
    # Since each score is at most 2, the only way to get 8 is 2 * 2 * 2 = 8.
    # This means each material must be connected to the other two in the trio.
    # This is equivalent to saying the trio forms a triangle (a clique of size 3).
    
    # Wait, let's re-read: "product of internal reactivity scores is 8".
    # If a trio is {a, b, c}, s_a is the number of edges in the trio connected to a.
    # If the trio is a triangle, s_a=2, s_b=2, s_c=2, product=8.
    # Are there other possibilities?
    # If it's not a triangle, at least one edge is missing.
    # If one edge is missing (e.g., (a,b) exists, (b,c) exists, (a,c) missing):
    # s_a = 1, s_b = 2, s_c = 1. Product = 2.
    # If two edges are missing (e.g., (a,b) exists, (b,c) missing, (a,c) missing):
    # s_a = 1, s_b = 1, s_c = 0. Product = 0.
    # If three edges are missing:
    # s_a = 0, s_b = 0, s_c = 0. Product = 0.
    # So the only way to get 8 is a triangle.
    
    # Iterate through all triangles and find the lexicographically smallest.
    # To optimize, iterate a < b < c.
    # For a fixed a, iterate b in adj[a] where b > a.
    # Then check if c in adj[a] and c in adj[b] where c > b.
    
    adj_list = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        neighbors_a = adj_list[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Find common neighbors of a and b that are > b
            neighbors_b = adj[b]
            for c in neighbors_a[i+1:]:
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
