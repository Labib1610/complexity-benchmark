import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    graph = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        edges.append((u, v))
        graph[u].add(v)
        graph[v].add(u)
    
    # Build adjacency list as sets for fast lookup
    # We need to find a triangle (three nodes all connected to each other)
    # Because if we have three nodes a,b,c all connected, 
    # then we cannot split them into two groups without having volatility in at least one group
    
    # To find the lexicographically smallest triangle:
    # Iterate through all edges, and for each edge (u,v), look for a node w such that 
    # (u,w) and (v,w) are also edges. That forms a triangle u-v-w.
    # Keep track of the minimum such triangle found.
    
    min_cert = None
    
    # Sort edges to help with finding lex smallest
    # But better approach: iterate through nodes in order
    for a in range(1, n + 1):
        neighbors = list(graph[a])
        neighbors.sort()
        # For each pair of neighbors of a, check if they are connected
        for i in range(len(neighbors)):
            b = neighbors[i]
            if b <= a:
                continue
            for j in range(i + 1, len(neighbors)):
                c = neighbors[j]
                if c <= b:
                    continue
                # Check if b and c are connected
                if c in graph[b]:
                    # Found triangle a-b-c
                    cert = (a, b, c)
                    if min_cert is None or cert < min_cert:
                        min_cert = cert
                        # Since we're iterating in sorted order, first one found is lex smallest
                        print(a, b, c)
                        return

if __name__ == "__main__":
    main()
