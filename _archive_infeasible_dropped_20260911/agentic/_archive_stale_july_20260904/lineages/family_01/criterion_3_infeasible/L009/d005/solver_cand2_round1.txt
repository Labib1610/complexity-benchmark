import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A certificate of three materials {a, b, c} means that for any partition
    # into two non-empty sets, at least one set contains a volatile pair.
    # The possible partitions of {a, b, c} are:
    # 1. {a}, {b, c}
    # 2. {b}, {a, c}
    # 3. {c}, {a, b}
    # For this to be a certificate, {b, c} must be volatile, {a, c} must be volatile,
    # and {a, b} must be volatile.
    # This means {a, b, c} must form a triangle (a clique of size 3).
    
    # We need the lexicographically smallest {a, b, c} such that a < b < c
    # and (a,b), (b,c), (a,c) are all in the registry.
    
    for a in range(1, n + 1):
        # Sort neighbors to ensure we find the lexicographically smallest
        neighbors_a = sorted([v for v in adj[a] if v > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Check for c > b such that c is connected to both a and b
            # Intersection of adj[a] and adj[b]
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
