```python
import sys

def solve():
    # A certificate of three materials {a, b, c} is impossible to split into two
    # containers if and only if the subgraph induced by {a, b, c} contains
    # at least 3 edges (i.e., it is a triangle).
    # If a triangle exists, any partition into two sets will have at least one
    # set containing two vertices connected by an edge.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need the lexicographically smallest {a, b, c} such that they form a triangle.
    # Iterate through all possible a, b, c.
    # Since n <= 2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    # A triangle {a, b, c} exists if there is an edge (a, b) and a vertex c
    # such that (a, c) and (b, c) are edges.
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    for a in range(1, n + 1):
        # To find the smallest triangle, we can iterate through all edges (u, v)
        # and check if there exists a common neighbor w.
        # To ensure lexicographical order, we iterate a, then b, then c.
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # Check for common neighbors c > b
                # Intersection of adj[a] and adj[b]
                common = adj[a].intersection(adj[b])
                for c in common:
                    if c > b:
                        print(f"{a} {b} {c}")
                        return

solve()
```
