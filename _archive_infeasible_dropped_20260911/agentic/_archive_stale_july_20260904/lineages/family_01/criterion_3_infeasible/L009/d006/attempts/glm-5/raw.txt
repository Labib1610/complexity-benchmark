    #         w = neighbors[u][1]
    #         Candidate = sorted((u, v, w)).
    #         Update best.
    #         
    #     Is it sufficient to check neighbors[u][0] and neighbors[u][1]?
    #     We want the smallest triplet for this u.
    #     The triplet is {u, v, w}.
    #     We want to minimize the sorted tuple.
    #     This means we want smallest elements.
    #     neighbors[u] is sorted.
    #     So neighbors[u][0] and neighbors[u][1] are the two smallest neighbors.
    #     This minimizes v and w.
    #     Does this minimize the triplet?
    #     Triplet elements are u, v, w.
    #     If u is fixed, we want smallest v and w.
    #     Yes.
    #     
    #     But wait, we iterate u.
    #     The triplet {u, v, w} might not start with u.
    #     Example u=5, v=1, w=2. Triplet {1, 2, 5}.
    #     If we iterate u=1, we might find {1, 3, 4}.
    #     {1, 2, 5} is smaller than {1, 3, 4}.
    #     So for a fixed u, picking smallest neighbors is correct to find the best triplet involving u.
    #     Then we take min over all u.
    #     
    #     One detail: neighbors[u] might contain elements smaller than u.
    #     neighbors[u] is sorted.
    #     neighbors[u][0] is the smallest neighbor.
    #     neighbors[u][1] is the second smallest.
    #     This is correct.
    
    # Implementation details:
    # adj matrix for O(1) edge check.
    # neighbors list for iteration.
    
    # Type 1 search
    for u in range(1, n + 1):
        # We iterate v, w from neighbors[u]
        # To ensure v < w and avoid duplicates/checking both (v,w) and (w,v)
        # and to find lex smallest, we iterate in order.
        # neighbors[u] is sorted.
        # We only care about v, w such that u < v < w to form a candidate (u, v, w).
        # BUT, a triangle could be {1, 2, 3} where u=2, v=1, w=3.
        # If we iterate u=1, we check neighbors 2, 3. adj[2][3]?
        # If adj[2][3] is true, we output 1 2 3.
        # This is correct.
        # If we iterate u=2, we check neighbors 1, 3. adj[1][3]?
        # If we found it at u=1, we would have outputted already.
        # So we only need to find triangles where u is the smallest node?
        # Yes, because we iterate u from 1 to n.
        # If a triangle {a, b, c} exists with a < b < c, it will be found when u=a.
        # We check neighbors b, c. adj[b][c] will be true.
        # So we only need to check pairs v, w in neighbors[u] such that u < v < w.
        # Wait, if we restrict to u < v < w, we might miss triangles?
        # No. If triangle is {a, b, c} with a < b < c.
        # When u=a, neighbors include b, c. We check pair (b, c). adj[b][c] is true. Found.
        # When u=b, neighbors include a, c. We check pair (a, c)? No, a < b.
        # If we only check v, w > u, then at u=b we check nothing (a < b, c > b).
        # So we only check pairs where both are greater than u.
        # This is sufficient to find the triangle exactly once with u being the smallest vertex.
        # And it guarantees u < v < w.
        
        # Optimization: filter neighbors greater than u
        greater_neighbors = [v for v in neighbors[u] if v > u]
        len_g = len(greater_neighbors)
        for i in range(len_g):
            v = greater_neighbors[i]
            for j in range(i + 1, len_g):
                w = greater_neighbors[j]
                if adj[v][w]:
                    print(u, v, w)
                    return
    
    # Type 2 search
    # Graph is triangle-free.
    best = (n + 1, n + 1, n + 1)
    
    for u in range(1, n + 1):
        if deg[u] >= 2:
            # neighbors[u] is sorted
            v = neighbors[u][0]
            w = neighbors[u][1]
            # Form triplet
            # Since we want lex smallest, we sort the triplet
            # But we know v < w.
            # We need to place u.
            # u could be < v, between v and w, or > w.
            # Actually, we can just use tuple(sorted((u, v, w)))
            # Optimization: manual sort to avoid function call overhead if needed, 
            # but Python's sorted is fast.
            
            # Manual sort for speed
            if u < v:
                current = (u, v, w)
            elif u < w:
                current = (v, u, w)
            else:
                current = (v, w, u)
            
            if current < best:
                best = current
                
    print(best[0], best[1], best[2])
