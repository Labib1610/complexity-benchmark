import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate through a from 1 to n, b from a+1 to n,
    # and check if there exists c > b such that c is in adj[a] and adj[b].
    
    # Sort adjacency lists to make finding c easier
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider b that are connected to a
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since both are sorted, we can use two pointers
            nb_a = sorted_adj[a]
            nb_b = sorted_adj[b]
            
            i, j = 0, 0
            while i < len(nb_a) and j < len(nb_b):
                if nb_a[i] == nb_b[j]:
                    c = nb_a[i]
                    if c > b:
                        print(f"{a} {b} {c}")
                        return
                    i += 1
                    j += 1
                elif nb_a[i] < nb_b[j]:
                    i += 1
                else:
                    j += 1

if __name__ == '__main__':
    solve()
