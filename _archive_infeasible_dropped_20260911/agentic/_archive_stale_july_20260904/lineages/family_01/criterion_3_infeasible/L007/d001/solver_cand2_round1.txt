```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically first triangle (a, b, c) with a < b < c:
    # We iterate through all possible a, then b, then check if c exists.
    # Since we need the lexicographically first, we sort the nodes.
    
    best_triangle = None
    
    # Sort nodes to ensure we check in lexicographical order
    for a in range(1, n + 1):
        # Optimization: only consider b > a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Check if there exists c > b such that c is in adj[a] and adj[b]
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    if best_triangle is None or (a, b, c) < best_triangle:
                        best_triangle = (a, b, c)
            
            # If we found a triangle starting with a, b, we don't necessarily 
            # have the lexicographically smallest one yet, but we can prune 
            # if we find a smaller one later.
            if best_triangle and best_triangle[0] < a:
                print(f"{best_triangle[0]} {best_triangle[1]} {best_triangle[2]}")
                return

    if best_triangle:
        print(f"{best_triangle[0]} {best_triangle[1]} {best_triangle[2]}")

solve()
```
