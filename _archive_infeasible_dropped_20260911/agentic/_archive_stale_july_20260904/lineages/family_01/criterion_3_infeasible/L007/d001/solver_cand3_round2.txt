import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest (a, b, c)
    edges.sort()
    
    # To find the smallest (a, b, c), we iterate through possible a, then b, then c.
    # Since we need a < b < c, we iterate through all edges (a, b)
    # and check if there exists a c > b such that (a, c) and (b, c) exist.
    
    # To optimize, we can iterate through all edges (a, b) and check common neighbors.
    # To ensure we find the smallest (a, b, c), we can iterate a from 1 to n,
    # then b from a+1 to n, then check common neighbors.
    
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Now find c > b such that c is in adj[a] and c is in adj[b]
            # We want the smallest such c
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
