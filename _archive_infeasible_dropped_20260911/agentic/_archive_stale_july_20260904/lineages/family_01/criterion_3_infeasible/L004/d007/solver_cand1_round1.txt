import sys

def solve():
    # The condition for a triplet {u, v, w} to be critically flawed:
    # Let A be the adjacency matrix of the subgraph induced by {u, v, w}.
    # The condition is that there exists a non-zero vector w such that Aw = 2w.
    # This means 2 is an eigenvalue of the adjacency matrix of the induced subgraph.
    # For a 3x3 matrix, the eigenvalues are the roots of det(A - lambda*I) = 0.
    # Let the induced subgraph have edges E_U.
    # If the subgraph is a triangle (K3), A = [[0, 1, 1], [1, 0, 1], [1, 1, 0]].
    # Eigenvalues are 2, -1, -1. So a triangle is always critically flawed.
    # If the subgraph has 2 edges (a path of length 2), A = [[0, 1, 0], [1, 0, 1], [0, 1, 0]].
    # Eigenvalues are sqrt(2), 0, -sqrt(2). Not 2.
    # If the subgraph has 1 edge, A = [[0, 1, 0], [1, 0, 0], [0, 0, 0]].
    # Eigenvalues are 1, -1, 0. Not 2.
    # If the subgraph has 0 edges, A = 0. Eigenvalues are 0, 0, 0. Not 2.
    # Thus, a triplet is critically flawed if and only if it forms a triangle (K3).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need the lexicographically smallest triangle (u, v, w) with u < v < w.
    # Iterate u from 1 to n-2
    for u in range(1, n + 1):
        neighbors = sorted(list(adj[u]))
        len_n = len(neighbors)
        if len_n < 2:
            continue
        
        # Iterate v in neighbors such that v > u
        for i in range(len_n):
            v = neighbors[i]
            if v <= u:
                continue
            
            # Check for w in neighbors such that w > v and w is connected to v
            # To optimize, intersect adj[u] and adj[v]
            common = adj[u].intersection(adj[v])
            valid_w = [w for w in common if w > v]
            if valid_w:
                print(f"{u} {v} {min(valid_w)}")
                return

if __name__ == '__main__':
    solve()
