```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest triple
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n.
    # For a fixed a, we iterate b in adj[a] such that b > a.
    # Then we check if there exists c in adj[a] AND adj[b] such that c > b.
    
    for a in range(1, n + 1):
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for b in neighbors_a:
            # Find common neighbors c such that c > b
            # We want the smallest such c
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
