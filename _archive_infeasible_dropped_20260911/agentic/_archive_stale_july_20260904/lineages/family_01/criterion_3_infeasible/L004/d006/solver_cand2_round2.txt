import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # For a subset {v1, v2, v3}, let S be the adjacency matrix of the induced subgraph.
    # S = [[0, a, b], [a, 0, c], [b, c, 0]] where a, b, c are 1 if edge exists, 0 otherwise.
    # S^2 = [[a^2+b^2, bc, ac], [bc, a^2+c^2, ab], [ac, ab, b^2+c^2]]
    # Sum of entries = 2*(a^2 + b^2 + c^2 + ab + bc + ac)
    # We want sum = 12, so a^2 + b^2 + c^2 + ab + bc + ac = 6.
    # Since a, b, c are in {0, 1}, the only way to get 6 is a=1, b=1, c=1.
    # This means the subset must form a triangle (a complete graph K3).
    
    # Wait, the problem says S is the 3x3 matrix of the *original* graph.
    # Let v1, v2, v3 be the chosen nodes.
    # S_ij = 1 if edge(vi, vj) exists in the original graph.
    # S = [[0, e12, e13], [e12, 0, e23], [e13, e23, 0]]
    # S^2 = [[e12^2+e13^2, e13*e23, e12*e23], 
    #        [e13*e23, e12^2+e23^2, e12*e13], 
    #        [e12*e23, e12*e13, e13^2+e23^2]]
    # Sum = 2*(e12^2 + e13^2 + e23^2 + e12*e13 + e12*e23 + e13*e23) = 12
    # e12^2 + e13^2 + e23^2 + e12*e13 + e12*e23 + e13*e23 = 6
    # Let k = e12 + e13 + e23 (number of edges in the triplet).
    # Sum = 2 * (k^2 - e12*e13 - e12*e23 - e13*e23 + e12*e13 + e12*e23 + e13*e23) = 2 * k^2
    # Wait, (e12+e13+e23)^2 = e12^2 + e13^2 + e23^2 + 2(e12*e13 + e12*e23 + e13*e23)
    # Since e^2 = e, e12^2 + e13^2 + e23^2 = e12 + e13 + e23 = k.
    # Sum = 2 * (k + e12*e13 + e12*e23 + e13*e23) = 12
    # k + e12*e13 + e12*e23 + e13*e23 = 6
    # If k=3 (triangle), 3 + 1 + 1 + 1 = 6. Correct.
    # If k=2 (path of 2 edges), 2 + 0 + 0 + 0 = 2 != 6.
    # If k=1, 1 + 0 = 1.
    # So the condition is simply that the three nodes must form a triangle.
    
    for i in range(1, n + 1):
        neighbors = sorted(list(adj[i]))
        for idx_j in range(len(neighbors)):
            j = neighbors[idx_j]
            if j <= i: continue
            for idx_k in range(idx_j + 1, len(neighbors)):
                k = neighbors[idx_k]
                if k <= j: continue
                if k in adj[j]:
                    print(f"{i} {j} {k}")
                    return

solve()
