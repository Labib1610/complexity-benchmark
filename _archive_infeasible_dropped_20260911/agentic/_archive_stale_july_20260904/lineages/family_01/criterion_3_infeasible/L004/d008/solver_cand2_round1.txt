```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # The determinant of Z is:
    # 1 - Z12^2 - Z23^2 - Z13^2 + 2*Z12*Z23*Z13
    # Z12 = u3^3 if (u1, u2) in E else 0
    # Z23 = u1^3 if (u2, u3) in E else 0
    # Z13 = u2^3 if (u1, u3) in E else 0
    #
    # Target: 1 + 2(u1*u2*u3)^3 - u1^6 - u2^6 - u3^6
    #
    # Let a = (u1, u2) in E, b = (u2, u3) in E, c = (u1, u3) in E
    # Det = 1 - (a*u3^3)^2 - (b*u1^3)^2 - (c*u2^3)^2 + 2*(a*u3^3)*(b*u1^3)*(c*u2^3)
    # Det = 1 - a*u3^6 - b*u1^6 - c*u2^6 + 2*a*b*c*(u1*u2*u3)^3
    #
    # We want Det = 1 + 2(u1*u2*u3)^3 - u1^6 - u2^6 - u3^6
    # Comparing terms:
    # 2*a*b*c = 2 => a=1, b=1, c=1
    # -a*u3^6 - b*u1^6 - c*u2^6 = -u1^6 - u2^6 - u3^6
    # Since a=b=c=1, this is -u3^6 - u1^6 - u2^6 = -u1^6 - u2^6 - u3^6
    # This is always true if a=b=c=1.
    # So the condition is simply that all three pairs (u1, u2), (u2, u3), (u1, u3)
    # must be in E. That is, {u1, u2, u3} must form a triangle.
    
    # We need the lexicographically smallest triangle (u1, u2, u3) with u1 < u2 < u3.
    for u1 in range(1, n + 1):
        adj_u1 = adj[u1]
        sorted_adj_u1 = sorted(list(adj_u1))
        for i in range(len(sorted_adj_u1)):
            u2 = sorted_adj_u1[i]
            if u2 <= u1:
                continue
            # Check for u3 > u2 such that (u1, u3) in E and (u2, u3) in E
            adj_u2 = adj[u2]
            # Intersection of adj[u1] and adj[u2] gives potential u3
            common = adj_u1.intersection(adj_u2)
            valid_u3 = [u3 for u3 in common if u3 > u2]
            if valid_u3:
                print(f"{u1} {u2} {min(valid_u3)}")
                return

solve()
```
