import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # The matrix Z for u1 < u2 < u3:
    # Z = [[u1, z12, z13], [z12, u2, z23], [z13, z23, u3]]
    # where zij = 0 if connected, else ui*uj
    # det(Z) = u1*u2*u3 + 2*z12*z23*z13 - u1*z23^2 - u2*z13^2 - u3*z12^2
    # We want det(Z) = u1*u2*u3
    # So: 2*z12*z23*z13 - u1*z23^2 - u2*z13^2 - u3*z12^2 = 0
    
    # Let e12 = 1 if connected, 0 otherwise
    # z12 = u1*u2 * (1 - e12)
    # The condition simplifies based on the number of edges in the triplet:
    # 0 edges: 2(u1*u2*u1*u3*u2*u3) - u1(u2*u3)^2 - u2(u1*u3)^2 - u3(u1*u2)^2 = 0
    #          2(u1^2*u2^2*u3^2) - u1*u2^2*u3^2 - u2*u1^2*u3^2 - u3*u1^2*u2^2 = 0
    #          u1^2*u2^2*u3^2 * (2 - 1/u1 - 1/u2 - 1/u3) = 0 -> Impossible for u >= 1
    # 1 edge: (e.g., e12=1, e13=0, e23=0)
    #         2(0) - u1(u2*u3)^2 - u2(u1*u3)^2 - u3(0) = -u1*u2^2*u3^2 - u2*u1^2*u3^2 != 0
    # 2 edges: (e.g., e12=1, e23=1, e13=0)
    #         2(0) - u1(0) - u2(u1*u3)^2 - u3(0) = -u2*u1^2*u3^2 != 0
    # 3 edges: (e12=1, e23=1, e13=1)
    #         2(0) - u1(0) - u2(0) - u3(0) = 0. This works!
    
    # Wait, the condition is det(Z) = u1*u2*u3.
    # If 3 edges: Z = [[u1, 0, 0], [0, u2, 0], [0, 0, u3]]
    # det(Z) = u1*u2*u3. Yes!
    # If 2 edges (e12=1, e23=1, e13=0):
    # Z = [[u1, 0, u1*u3], [0, u2, 0], [u1*u3, 0, u3]]
    # det(Z) = u1(u2*u3 - 0) - 0 + u1*u3(0 - u2*u1*u3) = u1*u2*u3 - u1^2*u2*u3^2
    # This equals u1*u2*u3 only if u1^2*u2*u3^2 = 0, impossible.
    # So the condition is equivalent to the triplet forming a triangle (3 edges).
    
    for u1 in range(1, n + 1):
        adj_u1 = adj[u1]
        sorted_adj_u1 = sorted([v for v in adj_u1 if v > u1])
        for i in range(len(sorted_adj_u1)):
            u2 = sorted_adj_u1[i]
            for j in range(i + 1, len(sorted_adj_u1)):
                u3 = sorted_adj_u1[j]
                if u3 in adj[u2]:
                    print(f"{u1} {u2} {u3}")
                    return

solve()
