import sys
from itertools import combinations

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    edges = set()
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        if u > v:
            u, v = v, u
        edges.add((u, v))
        adj[u].add(v)
        adj[v].add(u)
    return n, m, edges, adj

def determinant_3x3(W):
    # W is a 3x3 matrix
    # det(W) = W[0][0]*(W[1][1]*W[2][2] - W[1][2]*W[2][1]) 
    #        - W[0][1]*(W[1][0]*W[2][2] - W[1][2]*W[2][0]) 
    #        + W[0][2]*(W[1][0]*W[2][1] - W[1][1]*W[2][0])
    return (W[0][0] * (W[1][1] * W[2][2] - W[1][2] * W[2][1]) 
            - W[0][1] * (W[1][0] * W[2][2] - W[1][2] * W[2][0]) 
            + W[0][2] * (W[1][0] * W[2][1] - W[1][1] * W[2][0]))

def build_matrix(u1, u2, u3, edges):
    W = [[0]*3 for _ in range(3)]
    for i in range(3):
        W[i][i] = -1
    
    # Check edges
    if (u1, u2) in edges:
        W[0][1] = W[1][0] = 1
    if (u1, u3) in edges:
        W[0][2] = W[2][0] = 1
    if (u2, u3) in edges:
        W[1][2] = W[2][1] = 1
        
    return W

def solve():
    n, m, edges, adj = read_input()
    
    # Iterate through all combinations of 3 nodes in lex order
    for u1 in range(1, n - 1):
        for u2 in range(u1 + 1, n):
            if (u1, u2) not in edges:
                continue
            for u3 in range(u2 + 1, n + 1):
                if (u1, u3) not in edges or (u2, u3) not in edges:
                    continue
                W = build_matrix(u1, u2, u3, edges)
                det = determinant_3x3(W)
                if det == 4:
                    print(u1, u2, u3)
                    return

solve()
