```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need to find a triangle (u, v, w) with u < v < w
    # such that (u, v), (v, w), (u, w) are all edges.
    # The problem asks for the lexicographically smallest (u, v, w).
    
    best = None
    
    # Iterate through all edges (u, v) with u < v
    # For each edge, check common neighbors w > v
    # To ensure lexicographical order, we iterate u from 1 to n,
    # then v from u+1 to n.
    
    for u in range(1, n + 1):
        # Optimization: if we already found a triangle starting with a smaller u,
        # we don't need to check larger u.
        if best and u > best[0]:
            break
            
        for v in adj[u]:
            if v <= u:
                continue
            
            # Check common neighbors w
            # To keep it efficient, iterate over the smaller adjacency list
            if len(adj[u]) < len(adj[v]):
                candidates = adj[u]
            else:
                candidates = adj[v]
                
            for w in candidates:
                if w > v and w in adj[u] and w in adj[v]:
                    # Found a triangle (u, v, w)
                    if best is None or (u, v, w) < best:
                        best = (u, v, w)
    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
