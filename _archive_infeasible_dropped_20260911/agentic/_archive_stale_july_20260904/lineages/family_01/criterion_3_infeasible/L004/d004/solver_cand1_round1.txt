```python
import sys

def solve():
    # A subset of 3 transceivers cannot be validly assigned to 2 bands
    # if and only if they form a triangle (a clique of size 3).
    # If they form a triangle, any assignment of 2 bands to 3 nodes
    # will force at least two nodes to share a band, and since they
    # are all connected, that band will have an interference link.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append(tuple(sorted((u, v))))
        idx += 2
        
    # We need to find all triangles (i, j, k) with i < j < k
    # and pick the lexicographically smallest one.
    # Since n <= 2000, we can iterate over edges (i, j) and check
    # for common neighbors k.
    
    best_triplet = None
    
    # Sort edges to help find the smallest triplet
    edges.sort()
    
    for i in range(1, n + 1):
        # Optimization: if we already found a triplet starting with a smaller i,
        # we don't need to check larger i.
        if best_triplet and i > best_triplet[0]:
            break
            
        neighbors = sorted(list(adj[i]))
        for idx_j in range(len(neighbors)):
            j = neighbors[idx_j]
            if j <= i:
                continue
            if best_triplet and i == best_triplet[0] and j > best_triplet[1]:
                break
            
            # Find common neighbors k > j
            # Intersection of adj[i] and adj[j]
            common = adj[i].intersection(adj[j])
            for k in common:
                if k > j:
                    triplet = (i, j, k)
                    if best_triplet is None or triplet < best_triplet:
                        best_triplet = triplet
                        
    if best_triplet:
        print(f"{best_triplet[0]} {best_triplet[1]} {best_triplet[2]}")

solve()
```
