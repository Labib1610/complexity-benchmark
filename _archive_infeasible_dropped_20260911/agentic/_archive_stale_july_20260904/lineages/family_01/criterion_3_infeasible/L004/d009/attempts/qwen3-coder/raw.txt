import sys
from itertools import combinations

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    edges = set()
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        if u > v:
            u, v = v, u
        edges.add((u, v))
    return n, m, edges

def is_critical(u1, u2, u3, edges):
    # Create the adjacency matrix for this triplet
    adj = [[0]*3 for _ in range(3)]
    
    # Check edges between the three nodes
    pairs = [(u1, u2), (u1, u3), (u2, u3)]
    for i in range(3):
        u, v = pairs[i]
        if (u, v) in edges:
            adj[0][1] = adj[1][0] = 1 if (u, v) == (u1, u2) else adj[0][1]
            adj[0][2] = adj[2][0] = 1 if (u, v) == (u1, u3) else adj[0][2]
            adj[1][2] = adj[2][1] = 1 if (u, v) == (u2, u3) else adj[1][2]
    
    # The matrix A in the quadratic form x^T A x
    # A = [ 2   -I12    -3*I13]
    #     [-I12   8    -6*I23]
    #     [-3*I13 -6*I23 18 ]
    
    A = [[2, -adj[0][1], -3*adj[0][2]],
         [-adj[0][1], 8, -6*adj[1][2]],
         [-3*adj[0][2], -6*adj[1][2], 18]]
    
    # Calculate determinant of A
    det = (A[0][0] * (A[1][1]*A[2][2] - A[1][2]*A[2][1]) -
           A[0][1] * (A[1][0]*A[2][2] - A[1][2]*A[2][0]) +
           A[0][2] * (A[1][0]*A[2][1] - A[1][1]*A[2][0]))
    
    # If determinant is 0, then there exists a non-trivial solution
    return abs(det) < 1e-9

def solve():
    n, m, edges = read_input()
    
    # Try all combinations of 3 transceivers in lexicographic order
    for u1, u2, u3 in combinations(range(1, n+1), 3):
        if is_critical(u1, u2, u3, edges):
            print(u1, u2, u3)
            return

solve()
