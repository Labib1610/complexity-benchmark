import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2

    # The energy formula is E = x1^2 + 4x2^2 + 9x3^2 - 2*I12*x1*x2 - 6*I23*x2*x3 - 3*I13*x1*x3
    # This is a quadratic form x^T A x where A is:
    # [[1, -I12, -1.5*I13], [-I12, 4, -3*I23], [-1.5*I13, -3*I23, 9]]
    # E = 0 for non-zero x iff det(A) <= 0 (since the quadratic form is positive definite if det > 0)
    # det(A) = 1(36 - 9*I23^2) + I12(-9*I12 - 4.5*I13*I23) - 1.5*I13(3*I12*I23 + 6*I13)
    # det(A) = 36 - 9*I23^2 - 9*I12^2 - 4.5*I12*I13*I23 - 4.5*I12*I13*I23 - 9*I13^2
    # det(A) = 36 - 9*(I12^2 + I23^2 + I13^2 + I12*I13*I23)
    # det(A) <= 0 iff I12^2 + I23^2 + I13^2 + I12*I13*I23 >= 4
    # Since I_ij are 0 or 1:
    # If all I_ij = 1, sum = 1+1+1+1 = 4. (Triangle)
    # If two I_ij = 1, sum = 1+1+0+0 = 2.
    # If one I_ij = 1, sum = 1.
    # If zero I_ij = 1, sum = 0.
    # Thus, the condition is equivalent to I12=1, I23=1, and I13=1.
    # The problem asks for a triplet (u1, u2, u3) with u1 < u2 < u3 such that they form a triangle.

    for u1 in range(1, n + 1):
        neighbors = sorted(list(adj[u1]))
        len_n = len(neighbors)
        for i in range(len_n):
            u2 = neighbors[i]
            if u2 <= u1:
                continue
            # Check for common neighbors u3 > u2
            for j in range(i + 1, len_n):
                u3 = neighbors[j]
                if u3 in adj[u2]:
                    print(f"{u1} {u2} {u3}")
                    return

solve()
