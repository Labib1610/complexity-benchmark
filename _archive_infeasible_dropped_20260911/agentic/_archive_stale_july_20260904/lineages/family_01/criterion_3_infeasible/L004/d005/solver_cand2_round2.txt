import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # Let S = {a, b, c}.
    # X = number of edges incident to at least one of {a, b, c}.
    # Y = number of edges incident to exactly one of {a, b, c}.
    # X - Y = number of edges incident to at least one of {a, b, c} 
    #         minus number of edges incident to exactly one of {a, b, c}.
    # This is equal to the number of edges incident to at least two of {a, b, c}.
    # Let E(S) be the set of edges with both endpoints in S.
    # Let E(S, V\S) be the set of edges with one endpoint in S and one in V\S.
    # X = |E(S)| + |E(S, V\S)| + |E(V\S, S)| (Wait, X is edges incident to at least one)
    # X = |E(S)| + |E(S, V\S)|
    # Y = |E(S, V\S)|
    # X - Y = |E(S)|.
    # So we need to find a subset of 3 vertices such that the number of edges 
    # between them is exactly 3. This means the subset must form a triangle.
    
    # We need to find all triangles and pick the lexicographically smallest one.
    # To find triangles efficiently:
    # Iterate over all vertices i, then all neighbors j of i, then all neighbors k of j.
    # Check if k is a neighbor of i.
    
    best_tri = None
    
    # Sort neighbors to ensure we only count each triangle once
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for i in range(1, n + 1):
        neighbors = adj_list[i]
        for idx_j, j in enumerate(neighbors):
            if j <= i: continue
            for k in neighbors[idx_j + 1:]:
                if k in adj[j]:
                    # Found triangle (i, j, k)
                    tri = tuple(sorted((i, j, k)))
                    if best_tri is None or tri < best_tri:
                        best_tri = tri
                        
    print(f"{best_tri[0]} {best_tri[1]} {best_tri[2]}")

solve()
