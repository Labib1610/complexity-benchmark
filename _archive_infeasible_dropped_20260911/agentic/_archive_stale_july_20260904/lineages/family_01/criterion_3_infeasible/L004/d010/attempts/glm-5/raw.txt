```python
import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    
    adj = [set() for _ in range(n + 1)]
    degrees = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
        degrees[u] += 1
        degrees[v] += 1
        
    for u in range(1, n + 1):
        if degrees[u] >= 2:
            for v in adj[u]:
                if v > u and degrees[v] >= 2:
                    for w in adj[u]:
                        if w > v and w in adj[v]:
                            print(u, v, w)
                            return

if __name__ == '__main__':
    solve()
```
