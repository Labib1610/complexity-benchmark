import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    # adjacency bitmask (1-indexed nodes, bit position node-1)
    adj_mask = [0] * (n + 1)
    neigh = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        # set bits
        adj_mask[u] |= 1 << (v - 1)
        adj_mask[v] |= 1 << (u - 1)
        neigh[u].append(v)
        neigh[v].append(u)

    # sort neighbor lists for deterministic order
    for i in range(1, n + 1):
        neigh[i].sort()

    for i in range(1, n + 1):
        for j in neigh[i]:
            if j <= i:
                continue
            common = adj_mask[i] & adj_mask[j]
            # mask out bits for nodes <= j
            mask = common & ~((1 << j) - 1)   # bits 0..j-1 cleared
            if mask:
                lowbit = mask & -mask
                k = lowbit.bit_length()          # position (0‑based) + 1
                # k is node number because bit position = node-1
                print(i, j, k)
                return

if __name__ == "__main__":
    solve()
