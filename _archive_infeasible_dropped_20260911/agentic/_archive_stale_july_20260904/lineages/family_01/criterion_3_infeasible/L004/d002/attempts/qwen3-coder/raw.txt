import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Convert to sorted lists for consistent iteration
    for u in adj:
        adj[u] = sorted(adj[u])
    
    # Iterate through all edges, for each edge (u,v), look for a common neighbor w
    best = None
    
    # Process edges in a way that ensures we check triangles in lexicographical order
    # Sort edges by the first node, then by the second
    edges.sort()
    
    for u, v in edges:
        if u > v:
            u, v = v, u
        # Find common neighbors of u and v
        # Iterate through neighbors of u, check if they are also neighbors of v
        for w in adj[u]:
            if w > v:  # To ensure u < v < w for lexicographical order
                if w in adj[v]:
                    # Found a triangle u, v, w
                    candidate = tuple(sorted([u, v, w]))
                    if best is None or candidate < best:
                        best = candidate
        # Also check the other way: neighbors of v that are also neighbors of u
        # But to avoid duplicates, only check neighbors of v that are > u
        # This is already handled by the above loop since we ensure u < v
    
    if best:
        print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
