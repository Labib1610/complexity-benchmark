import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj_mask = [0] * (n + 1)
    adj_list = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        adj_mask[u] |= 1 << v
        adj_mask[v] |= 1 << u
        adj_list[u].append(v)
        adj_list[v].append(u)

    for i in range(1, n + 1):
        adj_list[i].sort()

    all_bits = (1 << (n + 1)) - 1
    high_mask = [0] * (n + 2)
    for b in range(1, n + 1):
        high_mask[b] = all_bits & ~((1 << (b + 1)) - 1)

    masks = adj_mask
    lists = adj_list
    high = high_mask

    for a in range(1, n - 1):
        ma = masks[a]
        for b in lists[a]:
            if b <= a:
                continue
            common = ma & masks[b]
            if common:
                cand = common & high[b]
                if cand:
                    c = (cand & -cand).bit_length() - 1
                    sys.stdout.write(f"{a} {b} {c}\n")
                    return

if __name__ == "__main__":
    solve()
