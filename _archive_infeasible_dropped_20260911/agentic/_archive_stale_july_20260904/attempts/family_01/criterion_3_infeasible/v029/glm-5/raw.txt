import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Convert sets to sorted lists for deterministic iteration
    adj_lists = [sorted(adj[i]) for i in range(n + 1)]
    
    best_a = n + 1
    best_b = n + 1
    best_c = n + 1
    
    for a in range(1, n - 1):
        # Pruning: if current 'a' is already >= best_a, we can't improve
        if a >= best_a:
            break
        
        adj_a = adj[a]
        adj_a_list = adj_lists[a]
        
        # Iterate through neighbors b > a
        for b in adj_a_list:
            if b <= a:
                continue
            # Pruning: if b >= best_b (given current a < best_a), can't improve
            if b >= best_b:
                break
            
            adj_b = adj[b]
            
            # Iterate through neighbors c > b that are also neighbors of a
            for c in adj_a_list:
                if c <= b:
                    continue
                # Pruning: if c >= best_c (given a < best_a and b < best_b), can't improve
                if c >= best_c:
                    break
                
                if c in adj_b:
                    # Found a triad (a, b, c) with a < b < c
                    best_a = a
                    best_b = b
                    best_c = c
                    # Since we iterate c in increasing order, first valid c is optimal for this (a, b)
                    break
            
            # After finding a valid triad for this (a, b), no need to check larger b
            if best_a == a and best_b == b:
                break
    
    print(best_a, best_b, best_c)

if __name__ == "__main__":
    solve()
