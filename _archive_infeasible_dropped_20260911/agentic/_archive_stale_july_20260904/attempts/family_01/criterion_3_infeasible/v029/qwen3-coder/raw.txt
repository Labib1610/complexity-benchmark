```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n, m = int(data[0]), int(data[1])
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Build adjacency list as sets for fast lookup
    # We'll iterate through all edges and look for a third node forming a triangle
    
    best_triad = None
    
    # For tie-breaking, we want the triad with smallest (a, b, c) in lexicographical order
    # So we can iterate through potential triads in some sorted manner and keep the best
    
    # Let's try all edges, and for each edge (u,v), check its neighbors to find triangles
    # We want to ensure u < v to avoid duplicates
    # And we'll sort the edges to help with tie-breaking
    
    sorted_edges = sorted(edges)
    
    for u, v in sorted_edges:
        if u > v:
            u, v = v, u
        # Find common neighbors of u and v
        neighbors_u = adj[u]
        neighbors_v = adj[v]
        common = neighbors_u & neighbors_v
        for w in sorted(common):
            if w != u and w != v:
                # Found a triangle (u, v, w)
                # Sort them to get a < b < c
                a, b, c = sorted([u, v, w])
                candidate = (a, b, c)
                if best_triad is None or candidate < best_triad:
                    best_triad = candidate
    
    print(best_triad[0], best_triad[1], best_triad[2])

if __name__ == "__main__":
    main()
```
