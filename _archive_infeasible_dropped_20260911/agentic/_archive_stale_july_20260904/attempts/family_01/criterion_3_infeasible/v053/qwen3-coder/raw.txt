from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for consistent processing
    for key in adj:
        adj[key] = sorted(adj[key])
    
    # We need to find a triangle (three nodes all connected to each other)
    # Enumerate all possible triplets in lexicographical order
    # To do this efficiently, we iterate through each edge and check for a common neighbor
    
    # First, sort the edges to help with ordering
    edges_sorted = sorted(edges)
    
    # Create a list of neighbors for each node
    neighbors = {}
    for u in adj:
        neighbors[u] = set(adj[u])
    
    # Now, go through each edge and look for triangles
    found = False
    result = None
    
    # Try all combinations of three nodes in lex order
    nodes = sorted(adj.keys())
    for i in range(len(nodes)):
        a = nodes[i]
        for j in range(i+1, len(nodes)):
            b = nodes[j]
            if b not in neighbors[a]:
                continue
            for k in range(j+1, len(nodes)):
                c = nodes[k]
                if c in neighbors[a] and c in neighbors[b]:
                    print(a, b, c)
