 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj_bits = [0] * (n + 1)
    adj_list = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        adj_bits[u] |= 1 << (v - 1)
        adj_bits[v] |= 1 << (u - 1)
        adj_list[u].append(v)
        adj_list[v].append(u)

    for i in range(1, n + 1):
        adj_list[i] = sorted(set(adj_list[i]))

    for a in range(1, n + 1):
        mask_a = adj_bits[a]
        for b in adj_list[a]:
            if b <= a:
                continue
            common = mask_a & adj_bits[b]
            common &= ~((1 << b) - 1)
            if common:
                c = (common & -common).bit_length()
                sys.stdout.write(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
```
