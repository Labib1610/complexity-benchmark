 ```python
import sys

def solve():
    data = sys.stdin.buffer.readline().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])

    adj = [0] * (n + 1)
    neigh = [[] for _ in range(n + 1)]
    readline = sys.stdin.buffer.readline

    for _ in range(m):
        u_str, v_str = readline().split()
        u = int(u_str)
        v = int(v_str)
        if u == v:
            continue
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        neigh[u].append(v)
        neigh[v].append(u)

    for i in range(1, n + 1):
        lst = neigh[i]
        if not lst:
            continue
        lst.sort()
        j = 0
        last = -1
        for x in lst:
            if x != last:
                lst[j] = x
                j += 1
                last = x
        neigh[i] = lst[:j]

    full = (1 << (n + 1)) - 1
    mask_gt = [0] * (n + 1)
    for i in range(1, n + 1):
        mask_gt[i] = full ^ ((1 << (i + 1)) - 1)

    out = sys.stdout.write
    for a in range(1, n + 1):
        for b in neigh[a]:
            if b <= a:
                continue
            common = adj[a] & adj[b]
            x = common & mask_gt[b]
            if x:
                c = (x & -x).bit_length() - 1
                out(f"{a} {b} {c}\n")
                return

    out("-1 -1 -1\n")

if __name__ == "__main__":
    solve()
```
