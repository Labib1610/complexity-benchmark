import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for deterministic iteration
    # We convert to sorted lists to iterate in order
    sorted_adj = [sorted(s) for s in adj]
    
    # To find the lexicographically first triangle (a, b, c),
    # we iterate a from 1 to n, then b from a+1 to n, then c from b+1 to n.
    # We check if (a,b), (b,c), (a,c) exist.
    # To check (a,c) efficiently, we can iterate neighbors of a that are > b.
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # We need b > a, so we iterate neighbors of a that are > a
        # But we also iterate b in ascending order.
        # So we can iterate the sorted neighbors of a.
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Now we need c > b such that c is neighbor of a and neighbor of b.
            # We iterate neighbors of a that are > b.
            # Since neighbors_a is sorted, we can start searching from index after b.
            # But simpler: iterate neighbors_a and check condition.
            
            # To optimize, we can iterate neighbors of b.
            # c must be in neighbors_a AND neighbors_b AND c > b.
            # Since we want the smallest c, we should iterate candidates in order.
            # neighbors_a is sorted. We can iterate it.
            
            for c in neighbors_a:
                if c <= b:
                    continue
                
                # Check if c is connected to b
                if c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == '__main__':
    solve()
