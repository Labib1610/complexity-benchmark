```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        edges.append((u, v))
        index += 2
    
    # Build adjacency list and set for O(1) lookup
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # Convert to sorted lists for consistent iteration
    adj_list = {node: sorted(list(neighbors)) for node, neighbors in adj.items()}
    
    # Iterate through all possible triplets (a,b,c) with a < b < c
    # We want a triangle: a-b, a-c, b-c all exist
    # To find the lexicographically smallest, we iterate a from 1 to n,
    # then b from (a+1) to n, then c from (b+1) to n
    
    for a in range(1, n+1):
        if a not in adj_list:
            continue
        neighbors_a = adj_list[a]
        # Only consider neighbors of a that are greater than a
        candidates_b = [b for b in neighbors_a if b > a]
        candidates_b.sort()
        
        for b in candidates_b:
            if b not in adj_list:
                continue
            # Check if a and b are connected
            if b not in adj[a]:
                continue
            # Now find c such that c > b and both a-c and b-c exist
            neighbors_b = adj_list[b]
            # Get common neighbors of a and b that are > b
            candidates_c = [c for c in neighbors_a if c > b and c in adj[b]]
            if candidates_c:
                c = min(candidates_c)
                print(a, b, c)
                return
                
    # If no triplet found (should not happen per problem constraints)
    # But just in case:
    print(-1, -1, -1)

if __name__ == "__main__":
    main()
```
