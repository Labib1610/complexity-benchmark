```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to process in order
    edges.sort()
    
    # For each edge (u,v), look for a third node w such that (u,w) and (v,w) are also edges
    for u, v in edges:
        if u >= v:
            continue
        # Get common neighbors of u and v
        common_neighbors = adj[u] & adj[v]
        if common_neighbors:
            # Find the smallest w in common neighbors
            w = min(common_neighbors)
            if u < v < w:
                print(u, v, w)
                return
            elif u < w < v:
                print(u, w, v)
                return
            elif w < u < v:
                print(w, u, v)
                return
    
    # This should never happen due to problem guarantee
    pass

if __name__ == "__main__":
    main()
```
