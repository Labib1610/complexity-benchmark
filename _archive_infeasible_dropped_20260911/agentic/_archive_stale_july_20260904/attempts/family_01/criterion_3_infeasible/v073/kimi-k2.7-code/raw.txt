```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    out = [set() for _ in range(n)]
    bits = [1 << i for i in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            continue
        if u > v:
            u, v = v, u
        adj[u] |= bits[v]
        adj[v] |= bits[u]
        out[u].add(v)
    for a in range(n):
        abits = adj[a]
        for b in sorted(out[a]):
            common = abits & adj[b]
            mask = common >> (b + 1)
            if mask:
                lsb = mask & -mask
                c = b + lsb.bit_length()
                sys.stdout.write(f"{a + 1} {b + 1} {c + 1}\n")
                return

if __name__ == "__main__":
    solve()
```
