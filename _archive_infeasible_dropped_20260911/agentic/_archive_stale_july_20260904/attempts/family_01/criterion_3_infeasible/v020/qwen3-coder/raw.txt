import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    edge_set = set()
    adj = defaultdict(set)
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        edge_set.add((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to process in order
    edges.sort()
    
    # We need to find the lexicographically smallest triangle.
    # To do that, we iterate through all edges and for each edge (u,v),
    # we look for a common neighbor w such that u < v < w.
    # This ensures we get the lexicographically smallest triangle.
    
    min_triangle = None
    
    # For each node u, sort its neighbors
    for u in adj:
        adj[u] = sorted(adj[u])
    
    # Iterate through each edge (u,v) with u < v
    for u, v in edges:
        if u >= v:
            continue
        # Find common neighbors of u and v that are greater than v
        # to ensure u < v < w
        for w in adj[u]:
            if w <= v:
                continue
            if w in adj[v]:
                # Found a triangle u, v, w with u < v < w
                print(u, v, w)
                return

if __name__ == "__main__":
    main()
