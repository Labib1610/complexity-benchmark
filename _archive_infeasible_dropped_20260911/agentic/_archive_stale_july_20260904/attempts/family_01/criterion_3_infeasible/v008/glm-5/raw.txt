```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        adj[u].add(v)
        adj[v].add(u)
    
    for a in range(1, n + 1):
        adj_a = adj[a]
        if len(adj_a) < 2:
            continue
        # Check pairs among neighbors of a
        neighbors = sorted(adj_a)
        len_n = len(neighbors)
        for i in range(len_n):
            b = neighbors[i]
            if b < a:
                continue
            adj_b = adj[b]
            for j in range(i + 1, len_n):
                c = neighbors[j]
                if c < a:
                    continue
                if c in adj_b:
                    print(a, b, c)
                    return

solve()
```
