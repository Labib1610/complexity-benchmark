```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * n
    p = 2
    for _ in range(m):
        u = int(data[p]) - 1
        v = int(data[p + 1]) - 1
        p += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    mask_gt = [0] * n
    for i in range(n):
        mask_gt[i] = ~((1 << (i + 1)) - 1)

    write = sys.stdout.write
    for i in range(n):
        B = adj[i] & mask_gt[i]
        while B:
            lsb = B & -B
            j = lsb.bit_length() - 1
            C = adj[i] & adj[j] & mask_gt[j]
            if C:
                k = (C & -C).bit_length() - 1
                write(f"{i + 1} {j + 1} {k + 1}\n")
                return
            B ^= lsb

if __name__ == "__main__":
    solve()
```
