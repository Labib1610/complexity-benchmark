We can solve this by iterating over each possible smallest vertex `a` in sorted orderebababababbabababababababababababababbab, and checking pairs of its neighbors to find a triangle. Since the graph is guaranteed to have a triangle and we want lexicographically smallest, we proceed incrementally.

```python
import sys

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)

    for a in range(1, n + 1):
        neighbors = sorted(adj[a])
        for i in range(len(neighbors)):
            b = neighbors[i]
            if b <= a:
                continue
            for j in range(i + 1, len(neighbors)):
                c = neighbors[j]
                if c <= b:
                    continue
                if c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
