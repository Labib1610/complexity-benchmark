```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list using sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        adj[u].add(v)
        adj[v].add(u)
        index += 2
    
    # Iterate through all triplets (i, j, k) in lexicographical order
    for i in range(1, n - 1):
        for j in range(i + 1, n):
            if j not in adj[i]:
                continue
            for k in range(j + 1, n + 1):
                if k not in adj[i]:
                    continue
                if k not in adj[j]:
                    continue
                # Found a triangle
                print(i, j, k)
                return

if __name__ == "__main__":
    main()
```
