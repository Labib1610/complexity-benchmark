 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    full = (1 << n) - 1
    gt = [0] * (n + 1)
    for v in range(1, n + 1):
        gt[v] = full ^ ((1 << v) - 1)

    for a in range(1, n + 1):
        s = adj[a] & gt[a]
        while s:
            lsb = s & -s
            b = lsb.bit_length()
            cand = adj[b] & s & gt[b]
            if cand:
                c = (cand & -cand).bit_length()
                sys.stdout.write(f"{a} {b} {c}\n")
                return
            s ^= lsb

if __name__ == "__main__":
    solve()
```
