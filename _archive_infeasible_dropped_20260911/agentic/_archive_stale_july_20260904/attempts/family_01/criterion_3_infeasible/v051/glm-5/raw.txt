import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges lexicographically to find the smallest triangle first
    edges.sort()
    
    # Check each edge (u, v) to see if there's a common neighbor w > v
    # This ensures we find the lexicographically smallest triangle
    for u, v in edges:
        # We need w > v to maintain a < b < c ordering
        # Since u < v, we need w > v
        common = adj[u] & adj[v]
        for w in common:
            if w > v:
                print(u, v, w)
                return
    
    # If not found by edge iteration, use the O(n^2) method
    # This handles cases where the smallest triangle might not be found
    # by iterating edges in sorted order (though theoretically it should)
    
    # Convert sets to sorted lists for iteration
    adj_list = [sorted(s) for s in adj]
    
    best = None
    
    for a in range(1, n - 1):
        adj_a = adj_list[a]
        # b must be > a
        for b in adj_a:
            if b <= a:
                continue
            adj_b = adj_list[b]
            # c must be > b
            # Find intersection of adj_a and adj_b where c > b
            i, j = 0, 0
            len_a, len_b = len(adj_a), len(adj_b)
            while i < len_a and j < len_b:
                if adj_a[i] < adj_b[j]:
                    i += 1
                elif adj_a[i] > adj_b[j]:
                    j += 1
                else:
                    c = adj_a[i]
                    if c > b:
                        if best is None or (a, b, c) < best:
                            best = (a, b, c)
                            # Since we iterate in order, first found is smallest
                            print(a, b, c)
                            return
                    i += 1
                    j += 1
    
    if best:
        print(best[0], best[1], best[2])

if __name__ == "__main__":
    solve()
