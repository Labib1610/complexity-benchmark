```python
import sys
from bisect import bisect_right

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    mask = [0] * (n + 1)
    neigh = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        bu = 1 << (u - 1)
        bv = 1 << (v - 1)
        if mask[u] & bv:
            continue
        mask[u] |= bv
        mask[v] |= bu
        neigh[u].append(v)
        neigh[v].append(u)

    for lst in neigh:
        lst.sort()

    out_write = sys.stdout.write
    for a in range(1, n + 1):
        ma = mask[a]
        lst = neigh[a]
        idx = bisect_right(lst, a)
        for i in range(idx, len(lst)):
            b = lst[i]
            common = mask[b] & ma
            shifted = common >> b
            if shifted:
                q = (shifted & -shifted).bit_length() - 1
                c = q + b + 1
                out_write(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
```
