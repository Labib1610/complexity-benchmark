import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list using sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We'll iterate through all combinations (a,b,c) with a < b < c
    # To do this efficiently, we'll look for triangles in the graph.
    # Since we want the lexicographically smallest, we'll iterate a from 1 to n,
    # then b from a+1 to n, then check if there's a c > b such that (a,b,c) forms a triangle.
    
    # For efficiency, let's create a sorted list of neighbors for each node
    neighbors = [sorted(adj[i]) for i in range(n + 1)]
    
    # Iterate over all possible a < b < c
    for a in range(1, n - 1):
        for b in neighbors[a]:
            if b <= a:
                continue
            # Now we need to find c > b such that both (a,c) and (b,c) exist
            # To do this efficiently, we intersect neighbors of a and b that are > b
            # Since neighbors are sorted, we can use two pointers or just iterate
            
            # Get neighbors of a and b that are greater than b
            set_a = set(x for x in adj[a] if x > b)
            set_b = set(x for x in adj[b] if x > b)
            
            common = set_a & set_b
            if common:
                c = min(common)
                print(a, b, c)
                return

if __name__ == "__main__":
    main()
