import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    masks = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        masks[u] |= 1 << (v - 1)
        masks[v] |= 1 << (u - 1)

    for a in range(1, n + 1):
        ma = masks[a]
        sub = ma >> a
        while sub:
            lsb = sub & -sub
            b = a + lsb.bit_length()
            common = ma & masks[b]
            shifted = common >> b
            if shifted:
                c = b + (shifted & -shifted).bit_length()
                print(a, b, c)
                return
            sub ^= lsb

if __name__ == "__main__":
    solve()
