```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list using sets for fast lookup
    adj = defaultdict(set)
    edges = []
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort adjacency lists for consistent iteration
    for key in adj:
        adj[key] = sorted(adj[key])
    
    # We'll iterate through all edges and for each edge (u,v), 
    # we'll look for a common neighbor w such that u-v-w forms a triangle
    # Since we want the lexicographically smallest triplet,
    # we'll process nodes in increasing order
    
    # Convert edges to sorted list to process in order
    edges.sort()
    
    for u, v in edges:
        if u >= v:
            continue  # Ensure u < v
        # Check for common neighbors of u and v
        # Iterate through neighbors of u
        for w in adj[u]:
            if w <= v:
                continue  # To ensure a < b < c
            if w in adj[v]:
                # Found triangle u, v, w
                print(u, v, w)
                return

if __name__ == "__main__":
    main()
```
