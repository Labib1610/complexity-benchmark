import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    for a in range(1, n + 1):
        # neighbors of a with label > a
        nb_mask = adj[a] & ~((1 << a) - 1)
        while nb_mask:
            b_bit = nb_mask & -nb_mask
            b = b_bit.bit_length()
            # common neighbors of a and b with label > b
            inter = adj[a] & adj[b] & ~((1 << b) - 1)
            if inter:
                c = (inter & -inter).bit_length()
                sys.stdout.write(f"{a} {b} {c}")
                return

solve()
