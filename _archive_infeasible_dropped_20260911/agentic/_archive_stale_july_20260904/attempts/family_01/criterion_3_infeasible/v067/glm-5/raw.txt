import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges lexicographically
    edges.sort()
    
    # Sort adjacency lists for ordered iteration
    adj_lists = [sorted(s) for s in adj]
    
    # Try to find triangle containing smallest vertex a
    # Iterate edges in sorted order, so first found triangle is lexicographically smallest
    for u, v in edges:
        # Find common neighbors w > v
        # Since edges are sorted, u < v
        # We need w > v to ensure u < v < w
        # Iterate through neighbors of u that are > v
        neighbors_u = adj_lists[u]
        # Binary search for start index (first neighbor > v)
        lo, hi = 0, len(neighbors_u)
        while lo < hi:
            mid = (lo + hi) // 2
            if neighbors_u[mid] <= v:
                lo = mid + 1
            else:
                hi = mid
        for i in range(lo, len(neighbors_u)):
            w = neighbors_u[i]
            if w in adj[v]:
                print(u, v, w)
                return
    
    # Fallback: shouldn't reach here per problem guarantee
    # But if we do, search all triangles
    for a in range(1, n + 1):
        neighbors_a = adj_lists[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b < a:
                continue
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c < b:
                    continue
                if c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
