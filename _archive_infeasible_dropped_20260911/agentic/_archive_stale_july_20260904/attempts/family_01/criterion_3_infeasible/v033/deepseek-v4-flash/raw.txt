```python
import sys
import sys
input = sys.stdin.readline

def solve():
    n, m = map(int, input().split())
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].add(v)
        adj[v].add(u)
    
    # Check all triples in lexicographic order
    for a in range(n):
        for b in range(a + 1, n):
            if b not in adj[a]:
                continue
            for c in range(b + 1, n):
                if c in adj[a] and c in adj[b]:
                    print(a + 1, b + 1, c + 1)
                    return

if __name__ == "__main__":
    solve()
```
