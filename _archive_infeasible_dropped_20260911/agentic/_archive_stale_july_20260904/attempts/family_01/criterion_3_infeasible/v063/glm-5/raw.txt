import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)

    best = None

    for x in range(1, n + 1):
        if best is not None and x >= best[1]:
            break
        neighbors = adj[x]
        if len(neighbors) < 2:
            continue
        sorted_neighbors = sorted(neighbors)
        limit_y = best[1] if best is not None else n + 1
        for y in sorted_neighbors:
            if y <= x:
                continue
            if best is not None and y >= best[1]:
                break
            adj_y = adj[y]
            for z in sorted_neighbors:
                if z <= y:
                    continue
                if z in adj_y:
                    candidate = (x, y, z)
                    if best is None or candidate < best:
                        best = candidate
                    break

    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == "__main__":
    solve()
