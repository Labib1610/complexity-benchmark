 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    p = 2
    for _ in range(m):
        u = int(data[p])
        v = int(data[p + 1])
        p += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    gt = [0] * (n + 1)
    for i in range(n + 1):
        gt[i] = ~((1 << (i + 1)) - 1)

    for a in range(1, n + 1):
        cand = adj[a] & gt[a]
        while cand:
            lsb = cand & -cand
            b = lsb.bit_length() - 1
            cand ^= lsb
            common = (adj[a] & adj[b]) & gt[b]
            if common:
                c = (common & -common).bit_length() - 1
                sys.stdout.write(f"{a} {b} {c}\n")
                return

if __name__ == "__main__":
    solve()
```
