 ```python
import sys
import bisect

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [0] * (n + 1)
    neigh = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)
        neigh[u].append(v)
        neigh[v].append(u)

    for i in range(1, n + 1):
        neigh[i].sort()

    full_mask = (1 << n) - 1
    higher = [0] * n
    for k in range(n):
        higher[k] = full_mask ^ ((1 << (k + 1)) - 1)

    for a in range(1, n + 1):
        mask_a = adj[a]
        lst = neigh[a]
        pos = bisect.bisect_right(lst, a)
        for i in range(pos, len(lst)):
            b = lst[i]
            cand = mask_a & adj[b] & higher[b - 1]
            if cand:
                c = (cand & -cand).bit_length()
                sys.stdout.write(f"{a} {b} {c}\n")
                return

solve()
```
