 ```python
import sys
import bisect

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    masks = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
        masks[u] |= 1 << v
        masks[v] |= 1 << u

    for i in range(1, n + 1):
        adj[i].sort()

    for a in range(1, n + 1):
        ma = masks[a]
        lst = adj[a]
        idx = bisect.bisect_left(lst, a + 1)
        for j in range(idx, len(lst)):
            b = lst[j]
            common = ma & masks[b]
            if common:
                shifted = common >> (b + 1)
                if shifted:
                    low = shifted & -shifted
                    c = low.bit_length() - 1 + b + 1
                    sys.stdout.write(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
```
