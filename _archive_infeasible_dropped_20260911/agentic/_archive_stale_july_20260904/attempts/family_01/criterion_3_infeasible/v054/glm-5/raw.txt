```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    for u in range(1, n + 1):
        adj[u].discard(u)
    
    for u, v in edges:
        if u > v:
            continue
        if len(adj[u]) > len(adj[v]):
            small, large = v, u
        else:
            small, large = u, v
        for w in adj[small]:
            if w > large and w in adj[large]:
                print(u, v, w)
                return
    
    for u in range(1, n + 1):
        lst = sorted(adj[u])
        L = len(lst)
        for i in range(L):
            vi = lst[i]
            for j in range(i + 1, L):
                vj = lst[j]
                if vj in adj[vi]:
                    print(u, vi, vj)
                    return

if __name__ == "__main__":
    solve()
```
