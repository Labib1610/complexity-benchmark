import sys
from collections import defaultdict

def find_feedback_cascade():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list representation of the graph
    adj = defaultdict(set)
    edges = []
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        index += 2
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v) if u < v else (v, u))
    
    # Sort edges to process in order
    edges.sort()
    
    # For each edge, check if it's part of a triangle
    for u, v in edges:
        # Find common neighbors of u and v
        neighbors_u = adj[u]
        neighbors_v = adj[v]
        
        # Check each neighbor of u to see if it's also a neighbor of v
        # and forms a triangle with u and v
        common = []
        for w in neighbors_u:
            if w in neighbors_v and w != u and w != v:
                # Found a triangle: u, v, w
                triple = sorted([u, v, w])
                return f"{triple[0]} {triple[1]} {triple[2]}"
    
    return ""

print(find_feedback_cascade())
