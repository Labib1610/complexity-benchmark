```python
import sys
from collections import defaultdict

def find_triangle(n, edges):
    # Build adjacency set for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # Also build adjacency list sorted for consistent iteration
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    # For each node, check all pairs of its neighbors
    for node in range(1, n + 1):
        neighbors = adj_list[node]
        for i in range(len(neighbors)):
            for j in range(i + 1, len(neighbors)):
                u = neighbors[i]
                v = neighbors[j]
                if v in adj[u]:  # Triangle found
                    # Return the triple in sorted order
                    triangle = sorted([node, u, v])
                    return triangle[0], triangle[1], triangle[2]
    return None

# Read input
input_lines = sys.stdin.read().splitlines()
n, m = map(int, input_lines[0].split())
edges = []
for i in range(1, m + 1):
    u, v = map(int, input_lines[i].split())
    edges.append((u, v))

# Find and print the triangle
a, b, c = find_triangle(n, edges)
print(a, b, c)
```
