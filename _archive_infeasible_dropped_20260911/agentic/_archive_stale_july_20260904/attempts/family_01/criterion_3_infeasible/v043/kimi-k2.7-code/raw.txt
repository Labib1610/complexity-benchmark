```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [0] * (n + 1)
    edge = [bytearray(n + 1) for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        if not edge[u][v]:
            edge[u][v] = 1
            edge[v][u] = 1
            adj[u] |= 1 << v
            adj[v] |= 1 << u

    all_bits = (1 << (n + 1)) - 1
    suffix = [0] * (n + 1)
    for b in range(1, n + 1):
        suffix[b] = all_bits ^ ((1 << (b + 1)) - 1)

    out = sys.stdout
    for a in range(1, n - 1):
        adja = adj[a]
        rowa = edge[a]
        for b in range(a + 1, n):
            if rowa[b]:
                common = adja & adj[b] & suffix[b]
                if common:
                    c = (common & -common).bit_length() - 1
                    out.write(f"{a} {b} {c}\n")
                    return

if __name__ == "__main__":
    solve()
```
