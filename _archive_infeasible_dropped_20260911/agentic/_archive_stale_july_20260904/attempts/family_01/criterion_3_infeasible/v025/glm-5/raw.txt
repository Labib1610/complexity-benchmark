import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    deg = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]); v = int(input_data[idx+1])
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
        deg[u] += 1
        deg[v] += 1
    
    # Sort vertices by degree
    order = sorted(range(1, n + 1), key=lambda x: deg[x])
    
    # Position of each vertex in the sorted order
    pos = [0] * (n + 1)
    for i, v in enumerate(order):
        pos[v] = i
    
    # For each vertex, store neighbors that appear later in the sorted order
    later_neighbors = [[] for _ in range(n + 1)]
    for v in range(1, n + 1):
        for u in adj[v]:
            if pos[u] > pos[v]:
                later_neighbors[v].append(u)
    
    # Mark vertices that have been processed
    processed = [False] * (n + 1)
    
    for v in order:
        processed[v] = True
        # Collect later neighbors that are not yet processed
        candidates = [u for u in later_neighbors[v] if not processed[u]]
        k = len(candidates)
        if k < 2:
            continue
        
        # Sort candidates by vertex id for lexicographically smallest triangle
        candidates.sort()
        
        # Check all pairs of candidates for an edge
        for i in range(k):
            a = candidates[i]
            adj_a = adj[a]
            for j in range(i + 1, k):
                b = candidates[j]
                if b in adj_a:
                    # Found triangle (v, a, b)
                    res = sorted([v, a, b])
                    sys.stdout.write(f"{res[0]} {res[1]} {res[2]}\n")
                    return
    
    # Should not reach here if input guarantees a triangle
    # But just in case, do a brute force fallback
    for a in range(1, n + 1):
        adj_a = adj[a]
        for b in adj_a:
            if b <= a:
                continue
            for c in adj_a:
                if c > b and c in adj[b]:
                    sys.stdout.write(f"{a} {b} {c}\n")
                    return

solve()
