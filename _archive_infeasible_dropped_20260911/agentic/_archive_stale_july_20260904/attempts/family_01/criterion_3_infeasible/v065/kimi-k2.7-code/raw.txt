 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * n
    neigh = [[] for _ in range(n)]
    p = 2
    for _ in range(m):
        u = int(data[p]) - 1
        v = int(data[p + 1]) - 1
        p += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        neigh[u].append(v)
        neigh[v].append(u)
    for i in range(n):
        neigh[i].sort()
    for a in range(n):
        for b in neigh[a]:
            if b <= a:
                continue
            common = adj[a] & adj[b]
            high = common >> (b + 1)
            if high:
                c = (high & -high).bit_length() - 1 + b + 1
                sys.stdout.write(f"{a + 1} {b + 1} {c + 1}\n")
                return

if __name__ == "__main__":
    solve()
```
