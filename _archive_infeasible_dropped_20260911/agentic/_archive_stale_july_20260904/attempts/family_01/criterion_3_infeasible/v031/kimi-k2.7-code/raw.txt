import sys

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    idx = 0
    n = data[idx]; idx += 1
    m = data[idx]; idx += 1

    adj = [0] * (n + 1)
    for _ in range(m):
        u = data[idx]; idx += 1
        v = data[idx]; idx += 1
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    full = (1 << n) - 1
    answer = None

    for a in range(1, n - 1):
        mask_gt_a = full ^ ((1 << a) - 1)
        nb = adj[a] & mask_gt_a
        while nb:
            lsb = nb & -nb
            b = lsb.bit_length()
            mask_gt_b = full ^ ((1 << b) - 1)
            common = adj[a] & adj[b] & mask_gt_b
            if common:
                c_lsb = common & -common
                c = c_lsb.bit_length()
                answer = (a, b, c)
                break
            nb ^= lsb
        if answer:
            break

    sys.stdout.write(f"{answer[0]} {answer[1]} {answer[2]}")

if __name__ == "__main__":
    solve()
