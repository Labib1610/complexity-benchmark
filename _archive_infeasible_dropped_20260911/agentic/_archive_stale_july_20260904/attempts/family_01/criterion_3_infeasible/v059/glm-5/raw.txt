import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Iterate through all pairs (u, v) in lexicographical order
    for u in range(1, n + 1):
        # To find the lexicographically smallest triangle (u, v, w),
        # we must iterate v in increasing order.
        # We only need to check neighbors v where v > u.
        neighbors_u = adj[u]
        # Create a sorted list of neighbors > u to iterate in order
        valid_vs = [v for v in neighbors_u if v > u]
        valid_vs.sort()
        
        for v in valid_vs:
            # For a fixed u and v, we need the smallest w such that
            # w > v, w is a neighbor of u, and w is a neighbor of v.
            neighbors_v = adj[v]
            
            # Iterate through potential w candidates from u's neighbors
            # We iterate in sorted order to find the smallest valid w immediately.
            for w in valid_vs:
                if w > v and w in neighbors_v:
                    # Found the lexicographically smallest triangle
                    print(u, v, w)
                    return

if __name__ == '__main__':
    solve()
