```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    transports = []
    idx = 2
    for i in range(j):
        f1 = int(input_data[idx])
        f2 = int(input_data[idx+1])
        transports.append((f1, f2))
        idx += 2
        
    # Check for Individual Transport Violation
    for i in range(j):
        if transports[i][0] + transports[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for Facility Bottleneck
    total_f1 = 0
    total_f2 = 0
    for f1, f2 in transports:
        total_f1 += f1
        total_f2 += f2
        
    if total_f1 > t:
        print("LOAD 1")
    elif total_f2 > t:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
