```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    target = int(input_data[1])
    
    alpha_total = 0
    beta_total = 0
    
    idx = 2
    for i in range(1, n + 1):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        idx += 2
        
        # Check if a single block exceeds the target
        if a + b > target:
            print(f"JOB {i}")
            return
        
        alpha_total += a
        beta_total += b
        
    # Check if total load exceeds the target
    if alpha_total > target:
        print("LOAD 1")
    elif beta_total > target:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
