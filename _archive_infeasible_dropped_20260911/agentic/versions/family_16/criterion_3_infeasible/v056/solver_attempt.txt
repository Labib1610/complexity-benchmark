```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    deadline = int(input_data[1])
    
    total_node1 = 0
    total_node2 = 0
    
    idx = 2
    for i in range(1, n + 1):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx + 1])
        idx += 2
        
        # Check Individual Job Bound
        if t1 + t2 > deadline:
            print(f"JOB {i}")
            return
        
        total_node1 += t1
        total_node2 += t2
        
    # Check Node Saturation
    if total_node1 > deadline:
        print("LOAD 1")
    elif total_node2 > deadline:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
