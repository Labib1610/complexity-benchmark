```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    passengers = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        passengers.append((p1, p2))
        idx += 2
        
    # Check for Individual Delay
    for i in range(j):
        if passengers[i][0] + passengers[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for Checkpoint Bottleneck
    total_p1 = sum(p[0] for p in passengers)
    total_p2 = sum(p[1] for p in passengers)
    
    if total_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
