```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    parts = []
    idx = 2
    
    total_p1 = 0
    total_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i failure mode
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        total_p1 += p1
        total_p2 += p2
        
    # Check LOAD c failure mode
    if total_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
