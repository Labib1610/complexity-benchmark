import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all possible a from 1 to n-2
    # Then iterate through all b > a such that (a, b) is an edge
    # Then check if there exists c > b such that (a, c) and (b, c) are edges
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    # Pre-sort adjacency lists to make intersection faster
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # We want the smallest a, then smallest b, then smallest c.
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Iterate b in neighbors of a, where b > a
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need a c such that c > b and c is in both adj[a] and adj[b]
            # Since adj lists are sorted, we can find intersection efficiently
            # or just iterate through the smaller list
            neighbors_b = adj[b]
            
            # Use two pointers or set intersection
            # Since we want the smallest c, we iterate through sorted neighbors
            ptr_a = 0
            ptr_b = 0
            len_a = len(neighbors_a)
            len_b = len(neighbors_b)
            
            while ptr_a < len_a and ptr_b < len_b:
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
