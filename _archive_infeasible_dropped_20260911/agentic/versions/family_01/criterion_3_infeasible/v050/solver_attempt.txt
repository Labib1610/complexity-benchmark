import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all possible 'a' from 1 to n
    # For each 'a', iterate through all neighbors 'b' of 'a' (b > a)
    # For each 'b', check if there exists a 'c' such that c > b and c is in adj[a] and adj[b]
    
    # Sort neighbors to ensure we find the smallest b and c
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c > b such that c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can intersect efficiently
            # or just iterate through neighbors of b
            neighbors_b = adj[b]
            
            # Use two pointers to find common elements > b
            ptr_a = i + 1
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
