import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # We iterate through all edges (u, v) and check for common neighbors.
    # To ensure lexicographical order, we sort the edges or iterate carefully.
    
    edges.sort()
    
    best_triangle = None
    
    for u, v in edges:
        # We want to find w such that (u, w) and (v, w) exist.
        # To keep it lexicographically smallest, we check potential w values.
        # Since we need a < b < c, for a fixed edge (u, v), 
        # the triangle is (u, v, w) if w > v, or (u, w, v) if u < w < v, or (w, u, v) if w < u.
        
        # Optimization: iterate over the smaller adjacency list
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
            
        for w in adj[u]:
            if w in adj[v]:
                # Found a triangle (u, v, w)
                # Sort the three vertices
                t = sorted([u, v, w])
                if best_triangle is None or t < best_triangle:
                    best_triangle = t
                    
    print(f"{best_triangle[0]} {best_triangle[1]} {best_triangle[2]}")

if __name__ == '__main__':
    solve()
