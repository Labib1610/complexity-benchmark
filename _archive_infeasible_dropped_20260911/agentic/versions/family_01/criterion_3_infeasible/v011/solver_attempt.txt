import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest (a, b, c)
    # We want to minimize a, then b, then c.
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # Get all neighbors of a
        neighbors_a = sorted(list(adj[a]))
        if len(neighbors_a) < 2:
            continue
        
        # For each pair of neighbors (b, c) of a, check if b and c are connected
        # To ensure a < b < c, we only consider neighbors > a
        valid_neighbors = [x for x in neighbors_a if x > a]
        
        # Sort valid neighbors to check pairs (b, c) in lexicographical order
        valid_neighbors.sort()
        
        for i in range(len(valid_neighbors)):
            b = valid_neighbors[i]
            for j in range(i + 1, len(valid_neighbors)):
                c = valid_neighbors[j]
                # Check if b and c are connected
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
