import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # We iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we iterate through all neighbors 'b' of 'a' such that b > a.
    # Then we check if there exists a 'c' such that c > b and c is a neighbor of both a and b.
    # To ensure lexicographical order, we sort neighbors of 'a' and 'b'.
    
    # Sort adjacency lists to make searching for c easier
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need to find the smallest c > b such that c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can intersect or just check
            # To optimize, use the set for O(1) lookup
            for c in neighbors_a:
                if c > b:
                    if c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
