```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
    
    # To find the lexicographically first triad (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is connected to a
    # If so, iterate c from b+1 to n
    # Check if c is connected to a and c is connected to b
    
    # Optimization: Sort adjacency lists to iterate efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Only consider b > a that are connected to a
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since neighbors are sorted, we can find common elements efficiently
            # But for n=2000, a simple set intersection or check is fine
            
            # To ensure lexicographical order, we iterate through potential c's
            # The potential c's are neighbors of a that are also neighbors of b
            # and c > b
            
            # Use set for O(1) lookup
            set_b = adj[b]
            for c in neighbors_a:
                if c > b and c in set_b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
