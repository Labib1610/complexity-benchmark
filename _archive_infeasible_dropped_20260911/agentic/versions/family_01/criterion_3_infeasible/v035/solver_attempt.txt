import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest triplet
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check for a common neighbor c
    # such that c > b.
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b from a+1 to n, then check if they are connected.
    # If connected, check for common neighbors c > b.
    
    # Optimization: iterate through sorted edges (a, b)
    # For each edge (a, b), find common neighbors c > b.
    # Since we want the smallest (a, b, c), we can collect all valid
    # triplets and pick the minimum.
    
    best = None
    
    for a in range(1, n + 1):
        # Optimization: if we already found a triplet (a', b', c')
        # and a > a', we can stop.
        if best and a > best[0]:
            break
            
        neighbors_a = adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Check if (a, b, c) is a valid triplet
            # We need c > b such that c is in adj[a] and adj[b]
            # To be lexicographically smallest, we want the smallest c
            # for the current (a, b).
            
            # Intersection of neighbors
            common = neighbors_a.intersection(adj[b])
            for c in common:
                if c > b:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
        
        # If we found a triplet starting with 'a', we don't need to check
        # any 'a' larger than this.
        if best and best[0] == a:
            # We need to ensure we found the smallest b and c for this a.
            # The current loop structure finds all c for a fixed (a, b).
            # We can just continue to finish checking all b for this a.
            pass
            
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
