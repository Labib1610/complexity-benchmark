import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest trio (a, b, c) with a < b < c:
    # Iterate through all possible 'a' from 1 to n-2
    # Then iterate through all neighbors 'b' of 'a' such that b > a
    # Then check if there exists a 'c' such that c > b and c is a neighbor of both a and b
    
    # Sort neighbors to ensure we check in lexicographical order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need to find c > b such that c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can intersect efficiently
            # or just iterate through neighbors of b that are > b
            neighbors_b = sorted_adj[b]
            
            # Use set intersection for speed
            # Only check neighbors of b that are > b
            for c in neighbors_b:
                if c > b and c in adj[a]:
                    print(f"{a} {b} {c}")
                    return

solve()
