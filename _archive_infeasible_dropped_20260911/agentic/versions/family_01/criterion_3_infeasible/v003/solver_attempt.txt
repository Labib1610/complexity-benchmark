```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need to find a, b, c such that (a,b), (a,c), (b,c) are all edges.
    # To find the lexicographically smallest (a, b, c), we iterate through
    # all edges (u, v) and check if there exists a node w such that
    # (u, w) and (v, w) are also edges.
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    best = None
    
    for u, v in edges:
        # We want to find w such that (u, w) and (v, w) exist.
        # To keep (a, b, c) sorted, we check w < u, u < w < v, and v < w.
        
        # Case 1: w < u < v
        # Case 2: u < w < v
        # Case 3: u < v < w
        
        # Intersection of neighbors of u and v
        common = adj[u].intersection(adj[v])
        if common:
            w = min(common)
            # Form the triplet and sort it
            triplet = sorted([u, v, w])
            if best is None or triplet < best:
                best = triplet
                
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
