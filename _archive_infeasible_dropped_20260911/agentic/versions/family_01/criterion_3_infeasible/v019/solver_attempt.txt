import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically first trio (a, b, c) with a < b < c,
    # we iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we look for 'b' > 'a' such that (a, b) is an edge.
    # Then we look for 'c' > 'b' such that (a, c) and (b, c) are edges.
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    # Pre-sort adjacency lists for faster lookup
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # Iterate through all 'b' connected to 'a' where b > a
        for b in adj[a]:
            if b <= a:
                continue
            
            # We need a 'c' such that c > b and c is in both adj[a] and adj[b]
            # Since adj lists are sorted, we can find intersection efficiently
            # or just iterate through the smaller list.
            
            # Optimization: check intersection of adj[a] and adj[b]
            # Only need to check elements > b
            ptr_a = 0
            ptr_b = 0
            list_a = adj[a]
            list_b = adj[b]
            
            while ptr_a < len(list_a) and ptr_b < len(list_b):
                val_a = list_a[ptr_a]
                val_b = list_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

if __name__ == '__main__':
    solve()
