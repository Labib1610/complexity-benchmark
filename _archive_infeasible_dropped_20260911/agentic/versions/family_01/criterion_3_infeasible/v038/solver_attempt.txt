import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triangle
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists c > b
    # such that (a, c) and (b, c) are also edges.
    # To ensure lexicographical order, we check 'a' from 1 to n.
    
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that b < c and c is in both adj[a] and adj[b]
            # To keep it efficient, intersect the sets
            common = adj[a].intersection(adj[b])
            
            # Filter for c > b
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
