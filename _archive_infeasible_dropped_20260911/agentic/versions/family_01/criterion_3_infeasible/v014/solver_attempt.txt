import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, iterate c from b+1 to n
    # Check if c is in adj[a] and adj[b]
    
    # Optimization: Sort adjacency lists to iterate efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # We need c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we need the smallest c, we can iterate through sorted neighbors
            # But we only need to check if there exists any c > b
            # Actually, we need the smallest c for the lexicographical order
            
            # To optimize, use set intersection for c
            # But since we need the smallest c, we can just iterate
            # through the smaller adjacency list
            if len(sorted_adj[a]) < len(sorted_adj[b]):
                candidates = sorted_adj[a]
            else:
                candidates = sorted_adj[b]
            
            for c in candidates:
                if c > b:
                    if c in adj[a] and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

solve()
