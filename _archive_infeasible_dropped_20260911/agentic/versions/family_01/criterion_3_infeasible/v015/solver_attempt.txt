import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # Sort edges to help find lexicographically smallest (a, b, c)
    # We want to minimize a, then b, then c.
    # Iterate through all possible a, then b, then c.
    # Since we need a < b < c, we can iterate through all edges (a, b)
    # and check if there exists a c such that (a, c) and (b, c) exist.
    
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n
    # For a fixed a, we iterate b from a+1 to n
    # If (a, b) is an edge, we check for common neighbors c > b
    
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        for b in neighbors_a:
            # We need a c such that b < c and c is in adj[a] and adj[b]
            # To minimize c, we look at the intersection of adj[a] and adj[b]
            # that are greater than b.
            
            # Optimization: intersect the two sets
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
