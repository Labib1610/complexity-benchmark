```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triple
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # all edges (a, b) and check if there exists a c > b such that
    # (a, c) and (b, c) are both edges.
    # Since we want the smallest a, then b, then c, we iterate through
    # sorted edges (a, b) and for each, check common neighbors.
    
    # Optimization: iterate through all edges (a, b)
    # For a fixed (a, b), we need a c > b such that c is in adj[a] and adj[b].
    # The intersection of adj[a] and adj[b] gives all such c.
    
    best_triple = None
    
    for a in range(1, n + 1):
        # Optimization: if we already found a triple with a smaller 'a', we can stop
        if best_triple and a > best_triple[0]:
            break
            
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if best_triple and (a > best_triple[0] or (a == best_triple[0] and b >= best_triple[1])):
                continue
            
            # Find common neighbors c > b
            # We can intersect the sets adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    if best_triple is None or (a < best_triple[0]) or \
                       (a == best_triple[0] and b < best_triple[1]) or \
                       (a == best_triple[0] and b == best_triple[1] and c < best_triple[2]):
                        best_triple = (a, b, c)
    
    if best_triple:
        print(f"{best_triple[0]} {best_triple[1]} {best_triple[2]}")

solve()
```
