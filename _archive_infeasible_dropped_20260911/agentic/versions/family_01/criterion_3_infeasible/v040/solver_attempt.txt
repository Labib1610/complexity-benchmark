import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triple
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) as the first two nodes of the triple.
    # Then check if there exists a node c > b such that (a, c) and (b, c) are edges.
    # To ensure lexicographical order, we iterate a from 1 to n, 
    # then b from a+1 to n, then check for c.
    
    # Optimization: Only check nodes with higher degree or use adjacency sets.
    # Since n <= 2000, O(n * m) or O(n^2) is acceptable.
    
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that b < c and c is in adj[a] and adj[b]
            # To keep it lexicographically smallest, we check c in increasing order
            # Intersection of adj[a] and adj[b]
            # Since we want the smallest c, we can iterate through the smaller set
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
            
            for c in sorted(list(candidates)):
                if c > b:
                    if c in adj[a] and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

solve()
