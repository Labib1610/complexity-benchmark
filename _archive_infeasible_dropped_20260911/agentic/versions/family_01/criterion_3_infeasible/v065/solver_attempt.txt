```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # Sort edges to help find the lexicographically smallest triad
    # A triad (a, b, c) with a < b < c exists if (a, b), (b, c), (a, c) are edges.
    # We iterate through all edges (a, b) and check for a common neighbor c > b.
    
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # possible a, then b, then c.
    # Since n <= 2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that b < c and c is in both adj[a] and adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
