import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        idx += 2
        
    # To find the lexicographically smallest trio (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Sort adjacency lists to help with finding c efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        if not adj[a]:
            continue
        
        # Neighbors of a that are > a
        neighbors_a = sorted_adj[a]
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need to find c > b such that c is in adj[a] and c is in adj[b]
            # Since neighbors_a is sorted, we can look at elements after b
            
            # Optimization: intersect adj[a] and adj[b]
            # Only need to check elements > b
            set_b = adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
