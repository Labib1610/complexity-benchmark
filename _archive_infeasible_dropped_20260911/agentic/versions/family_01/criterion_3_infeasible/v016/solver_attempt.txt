import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    edges.sort()
    
    # To find the lexicographically first trio (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists a c > b
    # such that (a, c) and (b, c) are also edges.
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b from a+1 to n, then check for c.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # We need the smallest a, then smallest b, then smallest c.
    # Iterate through all possible a
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = [v for v in adj[a] if v > a]
        for b in neighbors_a:
            # Now we have (a, b). We need a c such that b < c,
            # c is in adj[a] and c is in adj[b].
            # Since adj lists are sorted, we can find common elements efficiently.
            
            # Optimization: only check c > b
            # Intersection of adj[a] and adj[b]
            set_a = set(adj[a])
            common = [c for c in adj[b] if c > b and c in set_a]
            
            if common:
                print(f"{a} {b} {min(common)}")
                return

solve()
