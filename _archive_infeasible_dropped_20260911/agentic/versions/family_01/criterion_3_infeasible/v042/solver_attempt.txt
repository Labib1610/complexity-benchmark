import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest trio (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is connected to a
    # If so, check if there exists c > b such that c is connected to both a and b
    
    # Sort neighbors to potentially speed up intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Only consider neighbors > a
        valid_neighbors_a = [x for x in neighbors_a if x > a]
        
        for i in range(len(valid_neighbors_a)):
            b = valid_neighbors_a[i]
            
            # We need to find c > b such that c is in adj[a] and c is in adj[b]
            # Since adj[a] and adj[b] are sorted, we can use two pointers or set intersection
            # But since we want the smallest c, we just iterate through the intersection
            
            # Optimization: use set for O(1) lookup for the second neighbor
            set_b = set(adj[b])
            
            for j in range(i + 1, len(valid_neighbors_a)):
                c = valid_neighbors_a[j]
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

solve()
