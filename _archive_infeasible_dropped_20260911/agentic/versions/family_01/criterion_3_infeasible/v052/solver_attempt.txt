import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2,
    # b from a+1 to n-1, and check if c exists in both adj[a] and adj[b].
    
    # Sort adjacency lists to make searching easier
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # To optimize, we only check b that are connected to a.
    # Then we check c that are connected to both a and b.
    
    # Sort edges to iterate in lexicographical order
    edges.sort()
    
    # We need to find the smallest (a, b, c) such that (a,b), (b,c), (a,c) are edges.
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = [v for v in adj[a] if v > a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Now find c > b such that c is in adj[a] and adj[b]
            # Since adj lists are sorted, we can use intersection
            # But we only need the smallest c
            neighbors_b = adj[b]
            
            # Use two pointers to find the smallest common element > b
            ptr_a = 0
            ptr_b = 0
            len_a = len(neighbors_a)
            len_b = len(neighbors_b)
            
            while ptr_a < len_a and ptr_b < len_b:
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
