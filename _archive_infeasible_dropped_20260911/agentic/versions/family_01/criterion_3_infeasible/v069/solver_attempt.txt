import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, check if there exists c > b such that c is in adj[a] and adj[b]
    
    # Sort adjacency lists to potentially speed up intersection
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider neighbors of a that are > a
        neighbors_a = [x for x in adj_list[a] if x > a]
        if len(neighbors_a) < 2:
            continue
            
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c > b such that c is in adj[a] and adj[b]
            # Since we want the lexicographically smallest, we check neighbors of b
            # that are also in neighbors_a and > b
            
            # Optimization: intersect neighbors_a[i+1:] with adj[b]
            # Using set for adj[b] lookup
            adj_b = adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
