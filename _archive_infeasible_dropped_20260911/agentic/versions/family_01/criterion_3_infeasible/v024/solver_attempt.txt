```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically first triplet (a, b, c) with a < b < c,
    # we iterate through possible values of a, then b, then check if c exists.
    # Since we want the lexicographically smallest, we iterate a from 1 to n,
    # then b from a+1 to n, and check if there exists a c > b such that
    # (a, c) and (b, c) are edges.
    
    # Sort edges to help with iteration
    edges.sort()
    
    # Pre-sort adjacency lists for faster lookup
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # Iterate through all neighbors 'b' of 'a' where b > a
        for b in adj[a]:
            if b > a:
                # Now we need to find a 'c' such that b < c and c is in adj[a] and adj[b]
                # Since adj lists are sorted, we can find common elements efficiently
                # or just iterate through the smaller adjacency list.
                # To optimize, intersect adj[a] and adj[b]
                
                # Using set intersection for speed
                set_a = set(adj[a])
                set_b = set(adj[b])
                common = sorted(list(set_a.intersection(set_b)))
                
                for c in common:
                    if c > b:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
```
