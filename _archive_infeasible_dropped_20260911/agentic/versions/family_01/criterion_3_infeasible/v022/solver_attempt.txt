import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2,
    # b from a+1 to n-1, and check if c exists in both adj[a] and adj[b].
    
    # Sort adjacency lists to make checking faster and potentially find c earlier
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Get neighbors of a that are > a
        neighbors_a = [x for x in adj[a] if x > a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that b < c and c is in both adj[a] and adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # To keep it lexicographically smallest, we check c in increasing order
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj_sets[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
