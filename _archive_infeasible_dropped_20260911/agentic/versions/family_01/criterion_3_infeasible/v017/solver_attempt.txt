```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through possible a, then b, then check if c exists.
    # Since n <= 2000, we can iterate through all edges (u, v) and check
    # for a common neighbor w.
    
    best = (float('inf'), float('inf'), float('inf'))
    
    # Sort edges to help find smallest IDs first
    edges.sort()
    
    # Optimization: iterate through all nodes as 'a'
    for a in range(1, n + 1):
        # If a is already >= best[0], we can't find a better triplet
        if a >= best[0]:
            break
            
        neighbors = sorted(list(adj[a]))
        len_n = len(neighbors)
        
        for i in range(len_n):
            b = neighbors[i]
            if b <= a: continue
            
            # If (a, b) is already worse than best, skip
            if a > best[0] or (a == best[0] and b >= best[1]):
                continue
            
            # Find common neighbors of a and b
            # Intersection of adj[a] and adj[b]
            # Since we want smallest c, we check neighbors of the smaller degree node
            if len(adj[a]) < len(adj[b]):
                common = adj[a].intersection(adj[b])
            else:
                common = adj[b].intersection(adj[a])
            
            for c in common:
                if c > b:
                    if (a, b, c) < best:
                        best = (a, b, c)
    
    print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
