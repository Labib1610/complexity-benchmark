```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) with a < b < c
    # To satisfy the tie-breaking rule, we iterate through possible a, then b, then c
    # Since n <= 2000, an O(n^2) or O(m * sqrt(m)) approach is needed.
    # We can iterate over each edge (u, v) and check for a common neighbor w.
    
    best_triple = None
    
    # To optimize, iterate through nodes by ID
    for i in range(1, n + 1):
        # For a fixed i, we want to find j, k such that i < j < k
        # and (i, j), (j, k), (k, i) are edges.
        # This means j and k are neighbors of i, and j and k are connected.
        neighbors_i = sorted([x for x in adj[i] if x > i])
        
        for idx_j in range(len(neighbors_i)):
            j = neighbors_i[idx_j]
            
            # If we already have a best_triple and i > best_triple[0], we can stop
            if best_triple and i > best_triple[0]:
                print(f"{best_triple[0]} {best_triple[1]} {best_triple[2]}")
                return
            
            # Check neighbors of j that are also neighbors of i
            # To keep it efficient, intersect neighbors of i and j
            # Only consider k > j
            for k in adj[j]:
                if k > j and k in adj[i]:
                    # Found a triangle (i, j, k)
                    if best_triple is None or (i, j, k) < best_triple:
                        best_triple = (i, j, k)
                        
    if best_triple:
        print(f"{best_triple[0]} {best_triple[1]} {best_triple[2]}")

solve()
```
