import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) with a < b < c
    # To minimize a, then b, then c, we iterate a from 1 to n
    # For each a, iterate b in adj[a] such that b > a
    # Then check if there exists c in adj[a] intersect adj[b] such that c > b
    
    # Sort adjacency lists to help with ordering
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Find common neighbors c > b
            # Since neighbors_a and neighbors_b are sorted, we can intersect efficiently
            neighbors_b = adj[b]
            
            # To minimize c, we iterate through sorted neighbors of a
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
