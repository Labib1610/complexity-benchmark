```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # Iterate through all possible 'a' from 1 to n-2
    # Then iterate through all 'b' in adj[a] such that b > a
    # Then check if there exists 'c' in adj[a] intersect adj[b] such that c > b
    
    # Sort adjacency lists to help find smallest c
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need a c such that c > b and c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can iterate through it
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in adj_sets[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
