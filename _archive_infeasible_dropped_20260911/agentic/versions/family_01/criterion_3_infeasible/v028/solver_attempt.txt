import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Sort adjacency lists to potentially speed up intersection
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        sorted_neighbors_a = sorted([x for x in neighbors_a if x > a])
        
        for b in sorted_neighbors_a:
            # We need to find the smallest c > b such that c is in neighbors_a and neighbors_b
            neighbors_b = adj[b]
            
            # Intersection of neighbors_a and neighbors_b
            # Since we want the smallest c > b, we can iterate through sorted neighbors of b
            # that are > b and check if they are in neighbors_a
            for c in sorted([x for x in neighbors_b if x > b]):
                if c in neighbors_a:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
