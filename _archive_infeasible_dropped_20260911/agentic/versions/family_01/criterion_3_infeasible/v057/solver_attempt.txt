import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find the lexicographically smallest triplet (i, j, k)
    # such that i < j < k and (i,j), (j,k), (i,k) are edges.
    # Iterate i from 1 to n-2
    for i in range(1, n + 1):
        # Get neighbors of i that are > i
        neighbors_i = [v for v in adj[i] if v > i]
        neighbors_i.sort()
        
        # Iterate j from neighbors_i
        for idx_j in range(len(neighbors_i)):
            j = neighbors_i[idx_j]
            
            # We need a k > j such that k is in adj[i] and adj[j]
            # To keep it lexicographically smallest, we check k in sorted order
            # Intersection of neighbors of i and j that are > j
            common = sorted(list(adj[i].intersection(adj[j])))
            for k in common:
                if k > j:
                    print(f"{i} {j} {k}")
                    return

if __name__ == '__main__':
    solve()
