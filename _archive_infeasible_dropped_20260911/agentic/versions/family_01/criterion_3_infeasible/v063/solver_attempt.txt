import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest trio (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then c.
    # Since we need a triangle, for a fixed a and b, c must be in adj[a] AND adj[b].
    
    # Optimization: iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # For each pair of neighbors b, c of a, check if b and c are connected
        # To ensure a < b < c, we only look at neighbors > a
        valid_neighbors = [x for x in neighbors_a if x > a]
        
        for i in range(len(valid_neighbors)):
            b = valid_neighbors[i]
            # Check if there exists c > b such that c is in adj[a] and adj[b]
            # We can intersect the sets adj[a] and adj[b]
            # To keep it efficient, iterate through the smaller set
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
            
            for c in candidates:
                if c > b and c in adj[a] and c in adj[b]:
                    # Found a triangle (a, b, c)
                    # Since we iterate a, b, c in increasing order, 
                    # the first one we find is the lexicographically smallest.
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
