import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # We need to find a, b, c such that (a,b), (a,c), (b,c) are all edges.
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n.
    # For a fixed a, we look for b, c in adj[a] such that b < c and b is in adj[c].
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    # To optimize, iterate through all edges (a, b) and check if there exists c
    # such that (a, c) and (b, c) are edges.
    # Since we want the smallest (a, b, c), we can iterate a, then b, then check c.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through all possible a
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Iterate through all pairs (b, c) in neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
