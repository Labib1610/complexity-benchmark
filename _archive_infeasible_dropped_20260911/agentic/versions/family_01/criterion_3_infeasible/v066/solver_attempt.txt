```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to ensure we check smaller 'a' and 'b' first
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # We iterate through all edges (a, b) and check if there exists a common neighbor c.
    # To ensure lexicographical order, we iterate through all possible 'a' from 1 to n,
    # then all neighbors 'b' of 'a' (where b > a), then check if there is a 'c'
    # such that c is in adj[a] and adj[b] (where c > b).
    
    best = None
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # If we already found a solution with a smaller 'a', we can stop
            if best and a > best[0]:
                break
            
            # Find common neighbors c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
        
        if best and a > best[0]:
            break
            
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
