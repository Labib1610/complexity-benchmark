import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest (a, b, c)
    # We want to minimize a, then b, then c.
    # Iterate through all possible a, then b, then check if c exists.
    
    # To optimize, iterate through all edges (u, v) as potential (a, b)
    # and check if there exists a common neighbor w > v.
    
    edges.sort()
    
    best = None
    
    # We need the lexicographically smallest (a, b, c).
    # Iterate a from 1 to n-2
    for a in range(1, n + 1):
        # For a fixed a, we want the smallest b > a such that there exists c > b
        # where (a, b), (a, c), and (b, c) are all edges.
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # If we already found a solution with a smaller 'a' or same 'a' and smaller 'b',
            # we can potentially prune, but since we iterate a and b in order,
            # the first valid c we find for the smallest (a, b) is the answer.
            
            # Find common neighbors of a and b that are > b
            # Intersection of adj[a] and adj[b]
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
