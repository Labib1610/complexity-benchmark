import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c:
    # We iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we look for 'b' > 'a' such that 'b' is connected to 'a'.
    # Then we look for 'c' > 'b' such that 'c' is connected to both 'a' and 'b'.
    
    # Sort adjacency lists to ensure we check smaller indices first
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # To optimize, we only need to check edges (u, v) and see if they share a neighbor w
    # But to guarantee lexicographical order, we iterate a, then b, then c.
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Filter neighbors that are greater than a
        b_candidates = [b for b in neighbors_a if b > a]
        for i in range(len(b_candidates)):
            b = b_candidates[i]
            # We need a c > b such that c is in adj[a] and adj[b]
            # Since adj lists are sorted, we can intersect them efficiently
            # or just iterate through the smaller list
            
            # Optimization: check if there exists c > b in both adj[a] and adj[b]
            # Using sets for O(1) lookup
            set_a = set(adj[a])
            set_b = set(adj[b])
            
            # Iterate through the smaller set to find common neighbors
            if len(adj[a]) < len(adj[b]):
                for c in adj[a]:
                    if c > b and c in set_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in adj[b]:
                    if c > b and c in set_a:
                        print(f"{a} {b} {c}")
                        return

solve()
