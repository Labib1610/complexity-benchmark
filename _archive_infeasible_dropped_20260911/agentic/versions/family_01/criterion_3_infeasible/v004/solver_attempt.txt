```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triple
    edges.sort()
    
    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists a c > b
    # such that (a, c) and (b, c) are also edges.
    # Since we want the smallest a, then smallest b, then smallest c,
    # we iterate through sorted edges (a, b).
    
    for a, b in edges:
        # We need to find c > b such that c is in adj[a] and c is in adj[b]
        # Intersection of adj[a] and adj[b] gives all common neighbors
        common = adj[a].intersection(adj[b])
        
        # Filter for c > b
        candidates = [c for c in common if c > b]
        
        if candidates:
            c = min(candidates)
            print(f"{a} {b} {c}")
            return

if __name__ == '__main__':
    solve()
```
