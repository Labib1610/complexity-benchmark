import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest trio
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists a c > b
    # such that (a, c) and (b, c) are edges.
    # To optimize, we only need to check c in the intersection of adj[a] and adj[b].
    
    best_trio = None
    
    for a in range(1, n + 1):
        # Optimization: if we already found a trio with a smaller 'a', we can stop
        if best_trio and a >= best_trio[0]:
            break
            
        sorted_neighbors_a = sorted(list(adj[a]))
        for i in range(len(sorted_neighbors_a)):
            b = sorted_neighbors_a[i]
            
            # Optimization: if we have a best_trio, and a == best_trio[0] and b >= best_trio[1],
            # we don't need to check this b.
            if best_trio and a == best_trio[0] and b >= best_trio[1]:
                continue
            
            # Find common neighbors c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            
            for c in common:
                if c > b:
                    if best_trio is None or (a < best_trio[0]) or (a == best_trio[0] and b < best_trio[1]) or (a == best_trio[0] and b == best_trio[1] and c < best_trio[2]):
                        best_trio = (a, b, c)
    
    if best_trio:
        print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

if __name__ == '__main__':
    solve()
