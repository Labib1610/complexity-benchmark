```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
    
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through a from 1 to n-2, b from a+1 to n-1,
    # and check if there exists c > b such that c is connected to both a and b.
    
    # Sort adjacency lists to make finding c easier
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Candidates for b must be connected to a
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Now we need a c > b such that c is in adj[a] and adj[b]
            # We can intersect the sorted lists of neighbors of a and b
            # or just iterate through neighbors of b that are > b
            neighbors_b = adj[b]
            
            # To ensure lexicographical order, we check candidates for c
            # in increasing order.
            for c in neighbors_b:
                if c > b and c in adj[a]:
                    # Found the lexicographically smallest (a, b, c)
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

if __name__ == '__main__':
    solve()
```
