```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # Sort edges to help find the lexicographically smallest triangle
    # We want smallest a, then smallest b, then smallest c.
    # A triangle is (a, b, c) with a < b < c.
    # We iterate through all edges (a, b) and check if there exists c > b
    # such that (a, c) and (b, c) are edges.
    
    edges.sort()
    
    best = None
    
    for a, b in edges:
        # We have edge (a, b). Look for c > b such that (a, c) and (b, c) exist.
        # To keep it efficient, iterate over the smaller adjacency list
        if len(adj[a]) < len(adj[b]):
            candidates = adj[a]
            for c in candidates:
                if c > b and c in adj[b]:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
        else:
            candidates = adj[b]
            for c in candidates:
                if c > b and c in adj[a]:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
        
        # Optimization: if we found a triangle (a, b, c), any future triangle
        # starting with a' > a will be lexicographically larger.
        # If we have a fixed 'a', we want the smallest 'b', then smallest 'c'.
        # Since we iterate edges in sorted order, the first time we find a triangle
        # for a specific 'a', it might not be the smallest 'b'.
        # However, we can just collect all and pick the min.
        
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
