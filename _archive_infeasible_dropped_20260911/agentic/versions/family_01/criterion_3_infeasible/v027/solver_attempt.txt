import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to ensure we check lexicographically smallest (u, v) first
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # We iterate through all edges (u, v) as potential (a, b).
    # Then we look for a common neighbor w such that w > v.
    # To ensure we find the lexicographically smallest (a, b, c),
    # we iterate u from 1 to n, then v in adj[u] (sorted), then w in intersection.
    
    for u in range(1, n + 1):
        neighbors_u = sorted(list(adj[u]))
        for v in neighbors_u:
            if v <= u:
                continue
            
            # Now we have a potential (u, v). We need a w > v such that w is in adj[u] and adj[v].
            # To keep it efficient, intersect the smaller adjacency list with the larger.
            if len(adj[u]) < len(adj[v]):
                candidates = adj[u]
                other = adj[v]
            else:
                candidates = adj[v]
                other = adj[u]
            
            for w in sorted(list(candidates)):
                if w > v and w in other:
                    print(f"{u} {v} {w}")
                    return

if __name__ == '__main__':
    solve()
