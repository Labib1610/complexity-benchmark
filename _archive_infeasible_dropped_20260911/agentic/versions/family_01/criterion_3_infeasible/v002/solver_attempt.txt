import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c,
    # we iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we look at all neighbors 'b' of 'a' where b > a.
    # Then we check if there exists a 'c' such that b < c and c is a neighbor of both a and b.
    
    # Sort neighbors for each node to help find the smallest c efficiently
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Iterate b in neighbors of a where b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need to find the smallest c > b such that c is in adj[a] and adj[b]
            # Since adj[a] and adj[b] are sorted, we can use two pointers or set intersection
            # Given the constraints and the requirement for the lexicographically smallest,
            # we check neighbors of b that are also in neighbors of a.
            
            set_a = set(neighbors_a[i+1:])
            for c in adj[b]:
                if c > b and c in set_a:
                    print(f"{a} {b} {c}")
                    return

solve()
