import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that a < b < c and (a,b), (b,c), (a,c) exist.
    # To find the lexicographically first, we iterate a from 1 to n,
    # then b from a+1 to n, then c from b+1 to n.
    # Since we need (a,b), (b,c), (a,c), we check if b is in adj[a],
    # then check if c is in adj[a] AND c is in adj[b].
    
    for a in range(1, n + 1):
        # Get neighbors of a that are greater than a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # To keep it efficient, iterate through the smaller set
            if len(neighbors_a) - i < len(adj[b]):
                for j in range(i + 1, len(neighbors_a)):
                    c = neighbors_a[j]
                    if c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                # Iterate through neighbors of b that are > b
                neighbors_b = sorted([x for x in adj[b] if x > b])
                for c in neighbors_b:
                    if c in adj[a]:
                        # Ensure a < b < c
                        res = sorted([a, b, c])
                        print(f"{res[0]} {res[1]} {res[2]}")
                        return

solve()
