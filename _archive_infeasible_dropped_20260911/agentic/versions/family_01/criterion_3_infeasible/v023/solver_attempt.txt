```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all edges (a, b) with a < b,
    # and check if there exists a c > b such that (a, c) and (b, c) exist.
    # To ensure lexicographical order, we sort the edges.
    edges.sort()
    
    for a, b in edges:
        # We need a c > b such that c is in both adj[a] and adj[b]
        # To find the smallest c, we can intersect the sets
        common = adj[a].intersection(adj[b])
        if common:
            c = min(common)
            # The triple is (a, b, c). Since a < b and b < c,
            # this is already sorted.
            print(f"{a} {b} {c}")
            return

if __name__ == '__main__':
    solve()
```
