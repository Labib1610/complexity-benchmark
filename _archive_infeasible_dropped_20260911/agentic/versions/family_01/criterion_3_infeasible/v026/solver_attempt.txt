import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest trio
    # We want smallest a, then smallest b, then smallest c.
    # A trio (a, b, c) with a < b < c exists if b in adj[a], c in adj[a], and c in adj[b].
    
    # Sort nodes by degree to optimize search? 
    # Actually, just iterate through all edges (u, v) and check common neighbors.
    # To ensure lexicographical order, we iterate a from 1 to n.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # For each pair (a, b) where b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that c > b and c is in adj[a] and adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            neighbors_b = adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
