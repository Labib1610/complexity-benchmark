```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all edges (a, b) and check if there exists
    # a node c such that (a, c) and (b, c) are edges.
    # To ensure lexicographical order, we sort the edges.
    edges.sort()
    
    # We want the smallest a, then smallest b, then smallest c.
    # Since we iterate through sorted edges (a, b), we just need to
    # find the smallest c for each (a, b) and keep track of the best.
    
    best = None
    
    for a, b in edges:
        # We need a < b < c.
        # Check common neighbors of a and b that are > b.
        # To optimize, intersect the smaller set.
        if len(adj[a]) < len(adj[b]):
            candidates = adj[a]
            other = b
        else:
            candidates = adj[b]
            other = a
            
        for c in candidates:
            if c > b and c in adj[other]:
                if best is None or (a, b, c) < best:
                    best = (a, b, c)
        
        # If we have a best, and the current a is already larger than best[0],
        # we can stop because we won't find a lexicographically smaller one.
        if best and a > best[0]:
            break
            
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
