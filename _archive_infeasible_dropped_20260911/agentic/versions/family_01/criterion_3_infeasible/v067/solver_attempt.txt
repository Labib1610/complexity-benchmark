import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # We iterate through all edges (u, v) with u < v.
    # For each edge, we look for a common neighbor w.
    # To minimize (a, b, c), we want to check potential candidates efficiently.
    
    # Sort edges to process in lexicographical order
    edges.sort()
    
    best = (float('inf'), float('inf'), float('inf'))
    
    # Optimization: iterate through all edges (u, v)
    # For each edge, find common neighbors w.
    # Since we want the smallest (a, b, c), we can iterate u from 1 to n.
    for u in range(1, n + 1):
        # If u is already >= best[0], we can't find a better trio starting with u
        if u >= best[0]:
            break
            
        neighbors_u = sorted(list(adj[u]))
        for i in range(len(neighbors_u)):
            v = neighbors_u[i]
            if v <= u: continue
            
            # Current pair is (u, v). We need w > v such that w is in adj[u] and adj[v]
            # To minimize (u, v, w), we want the smallest w > v.
            # Since neighbors_u is sorted, we can intersect efficiently.
            
            # Optimization: only check if u, v could potentially beat best
            if (u, v, 0) >= best:
                continue
                
            # Find common neighbors w > v
            # Using set intersection for speed
            common = adj[u].intersection(adj[v])
            for w in common:
                if w > v:
                    trio = (u, v, w)
                    if trio < best:
                        best = trio
                        
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
