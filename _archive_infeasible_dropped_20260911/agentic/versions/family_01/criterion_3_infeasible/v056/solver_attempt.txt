```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all edges (u, v) as potential (a, b).
    # Since we want the smallest a, then b, then c, we sort edges.
    edges.sort()
    
    # We need to find a third node w such that (a, w) and (b, w) exist.
    # To ensure lexicographical order, we check all possible w for each edge (u, v).
    # Since n <= 2000, O(m * n) is acceptable (200,000 * 2000 is too much, 
    # but we only check nodes w that are neighbors of both u and v).
    
    best = None
    
    for u, v in edges:
        # Potential coalition (u, v, w) where u < v
        # w can be < u, between u and v, or > v.
        # We need to check all w.
        
        # Intersection of neighbors of u and v
        common = adj[u].intersection(adj[v])
        
        for w in common:
            # Form the triplet and sort it
            triplet = sorted([u, v, w])
            if best is None or triplet < best:
                best = triplet
                
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
