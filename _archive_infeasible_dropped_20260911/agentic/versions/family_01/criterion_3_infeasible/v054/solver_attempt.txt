```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically first (a, b, c) with a < b < c:
    # We iterate through all edges (a, b) and check if there exists a c > b
    # such that (a, c) and (b, c) are also edges.
    # To ensure lexicographical order, we sort the edges or iterate carefully.
    
    # Sort edges to help find the lexicographically smallest (a, b, c)
    edges.sort()
    
    # To optimize, iterate through all edges (a, b)
    # For each edge, find common neighbors c > b
    for a, b in edges:
        # We want to find the smallest c > b such that c is in adj[a] and adj[b]
        # Since we want the lexicographically smallest (a, b, c),
        # we should check potential c's in increasing order.
        
        # Intersection of neighbors of a and b
        common = adj[a].intersection(adj[b])
        
        # Filter for c > b
        valid_c = [c for c in common if c > b]
        
        if valid_c:
            c = min(valid_c)
            print(f"{a} {b} {c}")
            return

if __name__ == '__main__':
    solve()
```
