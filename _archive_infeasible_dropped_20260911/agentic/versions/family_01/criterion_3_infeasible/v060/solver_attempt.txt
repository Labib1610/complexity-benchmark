```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through all edges (u, v) and check for common neighbors.
    # Sorting edges by (u, v) ensures we check potential 'a' and 'b' in order.
    edges.sort()
    
    best = None
    
    for u, v in edges:
        # u is the smaller index, v is the larger.
        # We look for a third node w such that (u, w) and (v, w) exist.
        # To ensure lexicographical order, we check w in increasing order.
        
        # Optimization: intersect the smaller adjacency list
        if len(adj[u]) < len(adj[v]):
            common = adj[u].intersection(adj[v])
        else:
            common = adj[v].intersection(adj[u])
            
        for w in common:
            # The triplet is (min(u, v, w), mid(u, v, w), max(u, v, w))
            nodes = sorted([u, v, w])
            if best is None or nodes < best:
                best = nodes
                
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
