```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we iterate b in sorted order of neighbors of a.
    # Then check if there exists c such that c > b and c is in adj[a] and adj[b].
    
    for a in range(1, n + 1):
        neighbors_a = sorted([v for v in adj[a] if v > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Find common neighbors c > b
            # We want the smallest c, so we check neighbors of b that are also in neighbors_a
            # and are greater than b.
            common = sorted(list(adj[b].intersection(adj[a])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
