import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # Iterate a from 1 to n
    # Iterate b in sorted(adj[a]) such that b > a
    # Check if there exists c in adj[a] intersect adj[b] such that c > b
    
    # Sort adjacency lists to ensure we find the smallest b and c
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Optimization: only consider b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need to find the smallest c > b such that c is in both adj[a] and adj[b]
            # Since neighbors_a is sorted, we can iterate through it
            # But we need to check if c is in adj[b] efficiently
            # Using a set for adj[b] is O(1)
            set_b = adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c > b and c in set_b:
                    print(f"{a} {b} {c}")
                    return

solve()
