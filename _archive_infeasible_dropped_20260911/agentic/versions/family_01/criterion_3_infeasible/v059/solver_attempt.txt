import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically first (a, b, c) with a < b < c,
    # we iterate through all possible a, then b, then c.
    # Since we need the smallest a, then smallest b, then smallest c,
    # we can iterate a from 1 to n, b from a+1 to n, and check if c exists.
    
    # Sort edges to help iterate efficiently
    edges.sort()
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # To find the lexicographically first (a, b, c), we iterate a, then b.
    # For a fixed a and b, we need to find the smallest c > b such that
    # c is in adj[a] and c is in adj[b].
    
    for a in range(1, n + 1):
        # Only consider a if it's part of at least one edge
        if not adj[a]:
            continue
        
        # Get neighbors of a that are > a
        neighbors_a = [x for x in adj[a] if x > a]
        
        for b in neighbors_a:
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Since adj lists are sorted, we can find common elements efficiently
            # or just iterate through the smaller adjacency list.
            
            # Optimization: iterate through the smaller list of neighbors
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
            
            for c in candidates:
                if c > b:
                    # Check if c is connected to both a and b
                    if c in adj[a] and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

solve()
