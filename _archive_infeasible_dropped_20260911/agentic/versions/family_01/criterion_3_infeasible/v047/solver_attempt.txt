import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest trio (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is connected to a
    # If so, check if there exists c > b such that c is connected to both a and b
    
    # Sort adjacency lists to potentially speed up intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Neighbors of a that are > a
        neighbors_a = [x for x in adj[a] if x > a]
        for b in neighbors_a:
            # We need a c > b such that c is in adj[a] and adj[b]
            # To find the smallest c, iterate through sorted neighbors of b
            for c in adj[b]:
                if c > b:
                    if c in adj_sets[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
