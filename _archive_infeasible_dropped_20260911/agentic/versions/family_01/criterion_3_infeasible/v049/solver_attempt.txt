```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is connected to a
    # If so, iterate c from b+1 to n
    # Check if c is connected to both a and b
    
    # Optimization: Sort adjacency lists to iterate efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider nodes that have at least 2 neighbors
        if len(adj[a]) < 2:
            continue
            
        # Neighbors of a that are > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # This is the intersection of neighbors_a[i+1:] and adj[b]
            # Since we want the smallest c, we iterate through sorted neighbors of b
            for c in sorted_adj[b]:
                if c > b:
                    if c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
```
