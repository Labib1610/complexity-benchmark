import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and adj[b]
    
    # Convert sets to sorted lists for faster iteration if needed, 
    # but set lookup is O(1).
    
    for a in range(1, n - 1):
        # Neighbors of a that are > a
        neighbors_a = [x for x in adj[a] if x > a]
        neighbors_a.sort()
        
        for b in neighbors_a:
            # We need a c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we need the smallest c, we can iterate through sorted neighbors
            
            # Optimization: only check neighbors of b that are > b
            # and also in neighbors_a
            
            # To make this efficient, we can use set intersection
            # But we need the smallest c.
            
            # Get neighbors of b that are > b
            # Check if any of them are in adj[a]
            
            # To find the smallest c efficiently:
            # Iterate through sorted neighbors of b that are > b
            # Check if they are in adj[a]
            
            neighbors_b = adj[b]
            for c in sorted([x for x in neighbors_b if x > b]):
                if c in adj[a]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
