import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Optimization: sort adjacency lists to make intersection faster
    sorted_adj = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        if len(neighbors_a) < 2:
            continue
            
        # We need to find b, c in neighbors_a such that b < c and b is in adj[c]
        # To ensure lexicographical order, we iterate a, then b, then c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need c > b such that c is in neighbors_a and c is in adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # Since we want the smallest c, we can just iterate through neighbors_a[i+1:]
            # and check if it's in adj[b]
            
            # Using a set for adj[b] for O(1) lookup
            adj_b = adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj_b:
                    print(f"{a} {b} {c}")
                    return

solve()
