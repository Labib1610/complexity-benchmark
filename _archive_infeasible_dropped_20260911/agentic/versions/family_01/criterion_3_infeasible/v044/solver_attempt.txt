import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triplet
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists a c > b
    # such that c is connected to both a and b.
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b from a+1 to n, then check common neighbors.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need to find c > b such that c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can use two pointers or set intersection
            # But since we want the smallest c, we just iterate through neighbors_a
            # starting from i+1
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
