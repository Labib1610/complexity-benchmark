import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that a < b < c and 
    # b in adj[a], c in adj[a], c in adj[b]
    
    # To ensure we find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, b from a+1 to n, and c from b+1 to n.
    # To optimize, we only check b in adj[a] and c in adj[a] & adj[b].
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        if len(neighbors_a) < 2:
            continue
            
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # Since we want the smallest c, we can iterate through sorted neighbors_a
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
