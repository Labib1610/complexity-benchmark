```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    total_p1 = 0
    total_p2 = 0
    
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
        total_p1 += p1
        total_p2 += p2
        
    if total_p1 > T:
        print("LOAD 1")
    elif total_p2 > T:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
