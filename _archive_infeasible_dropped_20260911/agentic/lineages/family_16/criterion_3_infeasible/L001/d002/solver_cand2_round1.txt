```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # S_{i,1} >= 0
    # E_{i,1} = S_{i,1} + p_{i,1}
    # S_{i,2} >= E_{i,1}
    # E_{i,2} = S_{i,2} + p_{i,2}
    # E_{i,2} <= T
    # Implies: p_{i,1} + p_{i,2} <= T
    for i in range(j):
        if p[i][0] + p[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # Sum of p_{i,c} <= T
    sum1 = 0
    sum2 = 0
    for i in range(j):
        sum1 += p[i][0]
        sum2 += p[i][1]
        
    if sum1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
