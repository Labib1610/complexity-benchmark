```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    tasks = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tasks.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Constraint: x_{i,1} >= 0, x_{i,1} + p_{i,1} <= x_{i,2}, x_{i,2} + p_{i,2} <= T
    # This is impossible if p_{i,1} + p_{i,2} > T
    for i in range(j):
        if tasks[i][0] + tasks[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # Constraint: sum of p_{i,c} <= T
    sum1 = 0
    sum2 = 0
    for p1, p2 in tasks:
        sum1 += p1
        sum2 += p2
        
    if sum1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
