```python
j, T = map(int, input().split())

# Check for sequential bottleneck first
for i in range(j):
    p1, p2 = map(int, input().split())
    if p1 + p2 > T:
        print(f"JOB {i + 1}")
        exit()

# If no sequential bottleneck, check resource bottlenecks
# We need to read the input again or store it. Since we exited on sequential bottleneck,
# we can re-read the input for resource bottleneck check.
import sys
input_lines = sys.stdin.readlines()

total_p1 = 0
total_p2 = 0

for line in input_lines[1:]:
    p1, p2 = map(int, line.split())
    total_p1 += p1
    total_p2 += p2

if total_p1 > T:
    print("LOAD 1")
elif total_p2 > T:
    print("LOAD 2")
```
