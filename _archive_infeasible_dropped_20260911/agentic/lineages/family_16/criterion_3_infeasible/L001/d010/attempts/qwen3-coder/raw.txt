```python
import sys
import threading

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for _ in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        idx += 2
    
    # Check for Local Contradiction (JOB i)
    for i in range(j):
        p1 = p1_list[i]
        p2 = p2_list[i]
        if p1 + p2 > T:
            print(f"JOB {i+1}")
            return
    
    # Check for Global Contradiction (LOAD c)
    # For c=1: sum of p1's should not exceed the capacity derived from T and j
    # The maximum sum of integrals for non-overlapping intervals in [1, 2^T] is T*ln(2)
    # So sum of p1's <= T
    # Similarly for c=2
    
    sum_p1 = sum(p1_list)
    sum_p2 = sum(p2_list)
    
    if sum_p1 > T:
        print("LOAD 1")
        return
    if sum_p2 > T:
        print("LOAD 2")
        return

threading.Thread(target=main).start()
```
