```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    params = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        params.append((p1, p2))
        idx += 2
        
    # Check for JOB i (Local Contradiction)
    # Condition: p1 + p2 > T
    for i in range(j):
        if params[i][0] + params[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c (Global Contradiction)
    # Condition: sum(p_i,c) > T
    sum1 = 0
    sum2 = 0
    for i in range(j):
        sum1 += params[i][0]
        sum2 += params[i][1]
        
    if sum1 > T:
        print("LOAD 1")
        return
    if sum2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
