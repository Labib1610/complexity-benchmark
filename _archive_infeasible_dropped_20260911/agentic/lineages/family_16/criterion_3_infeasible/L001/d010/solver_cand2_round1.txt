import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    params = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        params.append((p1, p2))
        idx += 2
        
    # Check for JOB i
    for i in range(j):
        p1, p2 = params[i]
        # M_{i,1} >= p1/2, M_{i,2} >= p2/2
        # (M_{i,1} + p1/2) + (M_{i,2} + p2/2) <= T
        # Min value of LHS is (p1/2 + p1/2) + (p2/2 + p2/2) = p1 + p2
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c
    # For a dimension c, we need M_{i,c} >= p_{i,c}/2 and M_{i,c} + p_{i,c}/2 <= T
    # Let L_{i,c} = M_{i,c} - p_{i,c}/2. Then L_{i,c} >= 0.
    # The condition |M_{i,c} - M_{k,c}| >= (p_{i,c} + p_{k,c})/2
    # becomes |(L_{i,c} + p_{i,c}/2) - (L_{k,c} + p_{k,c}/2)| >= (p_{i,c} + p_{k,c})/2
    # This is equivalent to saying the intervals [L_{i,c}, L_{i,c} + p_{i,c}] are disjoint.
    # Since L_{i,c} >= 0, the total length required is sum(p_{i,c}).
    # The constraint M_{i,c} + p_{i,c}/2 <= T becomes L_{i,c} + p_{i,c} <= T.
    # So we need sum(p_{i,c}) <= T.
    
    sum_p1 = sum(p[0] for p in params)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    sum_p2 = sum(p[1] for p in params)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
