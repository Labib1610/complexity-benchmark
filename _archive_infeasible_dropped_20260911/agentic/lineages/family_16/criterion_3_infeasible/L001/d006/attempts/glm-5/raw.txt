```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    T = int(data[1])
    p1 = [0] * (j + 1)
    p2 = [0] * (j + 1)
    idx = 2
    for i in range(1, j + 1):
        p1[i] = int(data[idx])
        p2[i] = int(data[idx + 1])
        idx += 2

    # Check for local contradiction
    for i in range(1, j + 1):
        if p1[i] + p2[i] > T:
            print(f"JOB {i}")
            return

    # Check for global contradiction in dimension 1
    s1 = sum(p1[1:j + 1])
    if s1 > T:
        print("LOAD 1")
        return

    # Check for global contradiction in dimension 2
    s2 = sum(p2[1:j + 1])
    if s2 > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    solve()
```
