import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    X = []
    Y = []
    idx = 2
    for i in range(j):
        xi = int(input_data[idx])
        yi = int(input_data[idx+1])
        X.append(xi)
        Y.append(yi)
        idx += 2
        
        # Check Local Contradiction: JOB i
        # Conditions: b_i - a_i = X_i, d_i - c_i = Y_i, c_i >= b_i, a_i >= 0, d_i <= T
        # d_i - a_i = (d_i - c_i) + (c_i - b_i) + (b_i - a_i) >= Y_i + 0 + X_i
        # So d_i - a_i >= X_i + Y_i. Since a_i >= 0 and d_i <= T, d_i - a_i <= T.
        # Thus, if X_i + Y_i > T, it's impossible.
        if xi + yi > T:
            print(f"JOB {i + 1}")
            return

    # Check Global Contradiction: LOAD c
    # For group 1: b_i - a_i = X_i, |(a_i + b_i) - (a_k + b_k)| >= X_i + X_k, a_i >= 0, b_i <= T
    # Let m_i = (a_i + b_i) / 2. Then a_i = m_i - X_i/2, b_i = m_i + X_i/2.
    # a_i >= 0 => m_i >= X_i/2. b_i <= T => m_i <= T - X_i/2.
    # Separation: |2m_i - 2m_k| >= X_i + X_k => |m_i - m_k| >= (X_i + X_k) / 2.
    # This is equivalent to packing intervals [X_i/2, T - X_i/2] with length (X_i + X_k)/2 separation.
    # Sum of X_i must fit in T. Specifically, if we sort by m_i, 
    # m_{i+1} - m_i >= (X_{i+1} + X_i) / 2.
    # Summing this: m_j - m_1 >= X_1/2 + X_j/2 + sum_{k=2}^{j-1} X_k.
    # m_j <= T - X_j/2, m_1 >= X_1/2.
    # T - X_j/2 - X_1/2 >= X_1/2 + X_j/2 + sum_{k=2}^{j-1} X_k
    # T >= sum_{i=1}^j X_i.
    
    if sum(X) > T:
        print("LOAD 1")
        return
    
    if sum(Y) > T:
        print("LOAD 2")
        return

solve()
