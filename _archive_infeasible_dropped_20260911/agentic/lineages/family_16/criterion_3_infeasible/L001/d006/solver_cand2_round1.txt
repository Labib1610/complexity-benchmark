import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    K = int(input_data[1])
    
    A = []
    B = []
    idx = 2
    for _ in range(N):
        A.append(int(input_data[idx]))
        B.append(int(input_data[idx+1]))
        idx += 2
        
    # Check Type Alpha: y_{i,1} - x_{i,1} = A_i, y_{i,2} - x_{i,2} = B_i, x_{i,2} >= y_{i,1}
    # Constraints: 0 <= x_{i,1}, y_{i,1} <= K, 0 <= x_{i,2}, y_{i,2} <= K
    # y_{i,1} = x_{i,1} + A_i, y_{i,2} = x_{i,2} + B_i
    # x_{i,1} >= 0, x_{i,1} + A_i <= K => x_{i,1} <= K - A_i
    # x_{i,2} >= 0, x_{i,2} + B_i <= K => x_{i,2} <= K - B_i
    # x_{i,2} >= y_{i,1} = x_{i,1} + A_i
    # So we need x_{i,2} >= x_{i,1} + A_i.
    # Since x_{i,2} <= K - B_i and x_{i,1} >= 0, we need K - B_i >= A_i, or A_i + B_i <= K.
    # If A_i + B_i > K, then JOB i is inconsistent.
    for i in range(N):
        if A[i] + B[i] > K:
            print(f"JOB {i + 1}")
            return

    # Check Type Beta: m=1 or m=2
    # For a fixed m, we have N pairs (x_{i,m}, y_{i,m}) such that y_{i,m} - x_{i,m} = C_i
    # 0 <= x_{i,m}, y_{i,m} <= K => 0 <= x_{i,m} <= K - C_i
    # R3: For all i != k, either x_{i,m} >= y_{k,m} or x_{k,m} >= y_{i,m}
    # This is equivalent to saying the intervals [x_{i,m}, y_{i,m}] are laminar/nested
    # or simply that they can be ordered such that they don't overlap in a way that violates R3.
    # Actually, the condition for N intervals [x_i, x_i + C_i] to satisfy R3 is that
    # they can be ordered such that they are disjoint or nested.
    # The condition for existence is sum(C_i) <= K.
    if sum(A) > K:
        print("LOAD 1")
        return
    if sum(B) > K:
        print("LOAD 2")
        return

solve()
