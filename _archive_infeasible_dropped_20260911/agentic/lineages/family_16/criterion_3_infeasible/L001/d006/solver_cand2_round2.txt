import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    elements = []
    idx = 2
    for i in range(j):
        v1 = int(input_data[idx])
        v2 = int(input_data[idx+1])
        elements.append((v1, v2))
        idx += 2
        
    # Check for Local Element Contradiction (JOB i)
    # Constraints:
    # y1 - x1 = v1, y2 - x2 = v2
    # x2 >= y1
    # x1 >= 0, y2 <= T
    # Substituting:
    # y1 = x1 + v1
    # x2 >= x1 + v1
    # y2 = x2 + v2 >= x1 + v1 + v2
    # Since x1 >= 0, y2 >= v1 + v2
    # So if v1 + v2 > T, it's impossible.
    for i in range(j):
        v1, v2 = elements[i]
        if v1 + v2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for Aggregate Span Contradiction (LOAD c)
    # Constraints:
    # y_i - x_i = v_i
    # (x_i, y_i) disjoint
    # x_i >= 0, y_i <= T
    # This is equivalent to packing intervals of length v_i into [0, T].
    # The sum of lengths must be <= T.
    # Wait, the problem says "open intervals (x_i, y_i) are strictly disjoint".
    # If they are open, they can be adjacent.
    # The total length required is sum(v_i).
    # If we have j intervals of length v_i, they occupy a total length of sum(v_i).
    # Since they are open and disjoint, the total length must be <= T.
    # Actually, for open intervals (x_i, y_i) to be disjoint in [0, T],
    # the sum of lengths must be <= T.
    
    sum1 = sum(e[0] for e in elements)
    if sum1 > t:
        print("LOAD 1")
        return
        
    sum2 = sum(e[1] for e in elements)
    if sum2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
