```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    params = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        params.append((p1, p2))
        idx += 2
        
    # Check for JOB i
    # Condition: y_i - x_i = p1, v_i - u_i = p2, u_i >= y_i, x_i >= 0, v_i <= T
    # v_i = u_i + p2 >= y_i + p2 = x_i + p1 + p2 >= p1 + p2
    # So we need p1 + p2 <= T. If p1 + p2 > T, then JOB i is the answer.
    for i in range(j):
        if params[i][0] + params[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c
    # Condition: For dimension c, we have j intervals [s_i, e_i] with length p_{i,c}
    # such that they are non-overlapping and 0 <= s_i < e_i <= T.
    # This is possible if and only if the sum of lengths <= T.
    
    sum1 = sum(p[0] for p in params)
    if sum1 > T:
        print("LOAD 1")
        return
        
    sum2 = sum(p[1] for p in params)
    if sum2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
