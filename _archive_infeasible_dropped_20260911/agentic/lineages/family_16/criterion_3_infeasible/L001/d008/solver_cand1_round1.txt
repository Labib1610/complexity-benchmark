import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    params = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        params.append((p1, p2))
        idx += 2
        
    # Check for JOB i (Local Contradiction)
    # Condition: R(phi_i) + R(psi_i) <= T
    # Since phi_i(t) <= 1 and phi_i(t) = 0 for t < 0,
    # integral(phi_i) = p1 implies R(phi_i) >= p1.
    # Similarly R(psi_i) >= p2.
    # So if p1 + p2 > T, then R(phi_i) + R(psi_i) > T.
    for i in range(j):
        p1, p2 = params[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c (Global Contradiction)
    # Condition: sum(integral(f_i)) <= T
    # Because f_i(t) <= 1 and f_i(t) = 0 for t < 0,
    # and f_i(t) * f_k(t) = 0 for i != k,
    # the total measure of the union of supports must be at least sum(integral(f_i)).
    # If sum(integral(f_i)) > T, then it's impossible to fit in [0, T].
    
    sum1 = 0
    sum2 = 0
    for p1, p2 in params:
        sum1 += p1
        sum2 += p2
        
    if sum1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
