import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p.append((p1, p2))
        idx += 2
        
    # Check JOB i
    for i in range(j):
        if p[i][0] + p[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # For a fixed c, we have V_i = U_i * 2^p_i, U_i >= 1, V_i <= 2^T
    # The condition (V_i - U_k)(V_k - U_i) <= 0 implies intervals [U_i, V_i] are disjoint.
    # Let the sorted order be i_1, i_2, ..., i_j.
    # Then U_{i_1} >= 1, V_{i_1} <= U_{i_2}, V_{i_2} <= U_{i_3}, ..., V_{i_{j-1}} <= U_{i_j}, V_{i_j} <= 2^T.
    # Since V_i = U_i * 2^{p_i}, we have U_i = V_i * 2^{-p_i}.
    # The chain is: 1 <= U_{i_1}, V_{i_1} <= U_{i_2}, V_{i_2} <= U_{i_3}, ..., V_{i_j} <= 2^T.
    # This implies V_{i_1} >= 2^{p_{i_1}}, V_{i_2} >= V_{i_1} * 2^{p_{i_2}}, ..., V_{i_j} >= V_{i_{j-1}} * 2^{p_{i_j}}.
    # So V_{i_j} >= 2^{p_{i_1} + p_{i_2} + ... + p_{i_j}}.
    # The condition is solvable iff sum(p_i) <= T.
    
    for c in range(1, 3):
        total_p = sum(p[i][c-1] for i in range(j))
        if total_p > T:
            print(f"LOAD {c}")
            return

if __name__ == '__main__':
    solve()
