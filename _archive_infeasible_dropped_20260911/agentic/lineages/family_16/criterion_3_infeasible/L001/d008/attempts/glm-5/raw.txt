```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    for i in range(j):
        p1 = int(input_data[2 + 2*i])
        p2 = int(input_data[2 + 2*i + 1])
        p1_list.append(p1)
        p2_list.append(p2)
        
    # Check for Local Contradiction (JOB i)
    for i in range(j):
        if p1_list[i] + p2_list[i] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for Global Contradiction (LOAD c)
    # For c=1, the condition is sum of the j largest p_{i,1} > T
    # For c=2, the condition is sum of the j largest p_{i,2} > T
    # Since j is the total number of elements, it's just the sum of all p_{i,c}
    if sum(p1_list) > T:
        print("LOAD 1")
    elif sum(p2_list) > T:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
