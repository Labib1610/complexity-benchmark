import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    params = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        params.append((p1, p2))
        idx += 2
        
    # Check for Local Contradiction (JOB i)
    # Condition 3: M_{i,1} + M_{i,2} <= T - (p_{i,1} + p_{i,2})/2
    # Condition 4: M_{i,1} >= p_{i,1}/2, M_{i,2} >= p_{i,2}/2
    # Summing lower bounds: M_{i,1} + M_{i,2} >= (p_{i,1} + p_{i,2})/2
    # Contradiction if: (p_{i,1} + p_{i,2})/2 > T - (p_{i,1} + p_{i,2})/2
    # i.e., p_{i,1} + p_{i,2} > T
    for i in range(j):
        if params[i][0] + params[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for Global Contradiction (LOAD c)
    # For a dimension c, we need to place intervals [M_{i,c} - p_{i,c}/2, M_{i,c} + p_{i,c}/2]
    # such that they are disjoint and M_{i,c} >= p_{i,c}/2.
    # The interval for index i is [M_{i,c} - p_{i,c}/2, M_{i,c} + p_{i,c}/2].
    # Let x_{i,c} = M_{i,c} - p_{i,c}/2. Then x_{i,c} >= 0.
    # Clearance: |M_{i,c} - M_{k,c}| >= (p_{i,c} + p_{k,c})/2
    # This is equivalent to saying the intervals [x_{i,c}, x_{i,c} + p_{i,c}] are disjoint.
    # The upper bound M_{i,c} <= T - p_{i,c}/2 implies x_{i,c} + p_{i,c} <= T.
    # So we need to pack intervals of length p_{i,c} into [0, T].
    # The minimum space required is sum(p_{i,c}).
    # If sum(p_{i,c}) > T, then it's impossible.
    
    for c in range(2):
        total_p = sum(p[c] for p in params)
        if total_p > T:
            print(f"LOAD {c + 1}")
            return

if __name__ == '__main__':
    solve()
