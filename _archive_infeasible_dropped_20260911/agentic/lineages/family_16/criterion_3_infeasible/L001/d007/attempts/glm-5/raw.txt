```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    
    p1_list = []
    p2_list = []
    
    # Check for local contradiction (JOB)
    job_index = -1
    for i in range(1, j + 1):
        p1 = int(next(it))
        p2 = int(next(it))
        p1_list.append(p1)
        p2_list.append(p2)
        # Local contradiction: even if we place A_i at 0 and B_i at 0,
        # the sum of rightmost coordinates is p1 + p2.
        # If p1 + p2 > T, impossible.
        if p1 + p2 > T:
            job_index = i
    
    if job_index != -1:
        print(f"JOB {job_index}")
        return
    
    # Check for global contradiction (LOAD)
    # For category 1: sum of p1 must be <= T
    # For category 2: sum of p2 must be <= T
    sum_p1 = sum(p1_list)
    sum_p2 = sum(p2_list)
    
    if sum_p1 > T:
        print("LOAD 1")
        return
    
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    solve()
```
