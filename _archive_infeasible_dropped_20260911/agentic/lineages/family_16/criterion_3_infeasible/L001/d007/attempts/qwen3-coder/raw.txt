```python
import sys
input = sys.stdin.read

def main():
    data = input().split()
    j = int(data[0])
    T = int(data[1])
    p = []
    for i in range(j):
        p1 = int(data[2 + 2*i])
        p2 = int(data[3 + 2*i])
        p.append((p1, p2))
    
    # Check for Local Contradiction
    for i in range(j):
        p1, p2 = p[i]
        # Minimum rightmost coordinate for A_i is p1 (since it starts at >= 0)
        # Minimum rightmost coordinate for B_i is p2 (since it starts at >= 0)
        # Their sum must be <= T
        if p1 + p2 > T:
            print(f"JOB {i+1}")
            return
    
    # Check for Global Contradiction
    # For category 1 (A intervals)
    sum_p1 = sum(x[0] for x in p)
    # Minimum space needed for A intervals considering separation:
    # If we place them optimally to minimize the span, the total space needed 
    # is at least the sum of lengths (if they are placed end-to-end with only 
    # endpoint intersection allowed). But since they must fit within [0, T], 
    # we need sum_p1 <= T. However, the separation constraint means that 
    # the total "footprint" is at least sum_p1, but we also need to ensure 
    # that we can arrange them without violating the separation and the bound.
    # Actually, the minimum rightmost coordinate when placing j non-overlapping 
    # intervals of lengths p1[0], ..., p1[j-1] starting from >=0 is 
    # sum of p1's if placed consecutively. So if sum_p1 > T, then LOAD 1.
    
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # For category 2 (B intervals)
    sum_p2 = sum(x[1] for x in p)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    main()
```
