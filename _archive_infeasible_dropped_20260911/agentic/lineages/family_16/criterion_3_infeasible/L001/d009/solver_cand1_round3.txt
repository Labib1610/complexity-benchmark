```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    params = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        params.append((p1, p2))
        idx += 2
        
    # Check for JOB i
    # Y_i = X_i * 2^p1, V_i = U_i * 2^p2
    # X_i >= 1, U_i >= 1 => Y_i >= 2^p1, V_i >= 2^p2
    # Y_i * V_i <= 2^T => 2^p1 * 2^p2 <= 2^T => p1 + p2 <= T
    # If p1 + p2 > T, then JOB i is the contradiction.
    for i in range(j):
        if params[i][0] + params[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c
    # Constraint 6: Y_i Y_k + X_i X_k <= Y_i X_i + Y_k X_k
    # (Y_i - X_i)(Y_k - X_k) <= 0
    # Since Y_i = X_i * 2^p1, Y_i - X_i = X_i(2^p1 - 1)
    # Since X_i >= 1 and p1 >= 1, Y_i - X_i > 0.
    # This implies (Y_i - X_i) and (Y_k - X_k) must have opposite signs or be zero.
    # But they are all positive, so this is impossible unless the set is empty or 
    # the constraints force a contradiction.
    # The global constraint Y_i <= 2^T implies X_i * 2^p1 <= 2^T => X_i <= 2^(T - p1).
    # Since X_i >= 1, we need 2^(T - p1) >= 1 => T >= p1.
    # If we have multiple i, the pairwise constraint (Y_i - X_i)(Y_k - X_k) <= 0
    # is always violated if all Y_i - X_i > 0.
    # The only way to satisfy (Y_i - X_i)(Y_k - X_k) <= 0 is if at most one i has Y_i > X_i.
    # But Y_i = X_i * 2^p1 and p1 >= 1, so Y_i > X_i for all i.
    # Thus, if j > 1, it's always a LOAD contradiction.
    
    # Check if LOAD 1 or LOAD 2 is the answer.
    # The problem guarantees one of these is the cause.
    # If j > 1, LOAD 1 and LOAD 2 are both candidates.
    # We check if any p1 > T or p2 > T.
    # Actually, the problem implies that if JOB i is not the answer, 
    # then the global constraints are the cause.
    
    # Check if any p1 > T (for LOAD 1) or p2 > T (for LOAD 2)
    # If p1 > T, then Y_i = X_i * 2^p1 >= 2^p1 > 2^T, violating Y_i <= 2^T.
    # This is a stronger condition for LOAD.
    
    max_p1 = 0
    max_p2 = 0
    for p1, p2 in params:
        if p1 > max_p1: max_p1 = p1
        if p2 > max_p2: max_p2 = p2
        
    if max_p1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

solve()
```
