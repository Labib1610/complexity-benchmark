import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p.append((p1, p2))
        idx += 2
        
    # Check for JOB i
    for i in range(j):
        if p[i][0] + p[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c
    # Constraint 6/7: Vi*Vk + Ui*Uk <= Vi*Ui + Vk*Uk
    # Vi*Vk - Vi*Ui - Vk*Uk + Ui*Uk <= 0
    # (Vi - Uk)(Vk - Ui) <= 0
    # Since Vi = Ui * 2^pi, this is (Ui * 2^pi - Uk)(Uk * 2^pk - Ui) <= 0
    # Let Ui = 2^xi. Then (2^(xi+pi) - 2^xk)(2^(xk+pk) - 2^xi) <= 0
    # This implies the sequence of xi must be monotonic.
    # For a fixed c, let pi be the parameters.
    # The constraints are:
    # 1) Ui >= 1 => xi >= 0
    # 2) Vi <= 2^T => xi + pi <= T => xi <= T - pi
    # 3) (Ui*2^pi - Uk)(Uk*2^pk - Ui) <= 0
    # If we sort indices such that p_i is non-decreasing, 
    # the condition (Vi - Uk)(Vk - Ui) <= 0 forces xi to be non-decreasing.
    # The tightest constraints are xi >= 0 and xi <= T - pi.
    # For a global solution to exist, we need 0 <= xi <= T - pi for all i.
    # If we sort by pi, then xi must be non-decreasing.
    # The most restrictive upper bound is min(T - pi).
    # Actually, the condition (Vi - Uk)(Vk - Ui) <= 0 for all i, k
    # implies that if we sort by pi, then Vi must be non-decreasing.
    # Vi = Ui * 2^pi. If pi is increasing, Ui can be constant (e.g., 1).
    # The global constraint is actually: max(pi) <= T.
    # Wait, the example 2 says LOAD 1 is inconsistent because 4*50 > 150.
    # The condition for LOAD c is: sum(pi) > T? No, max(pi) + (j-1)*min(pi)?
    # The condition is: max(pi) + sum(pi_sorted_i) ... no.
    # The condition for LOAD c is: max(pi) + (j-1) * min(pi) > T.
    
    for c in range(2):
        ps = sorted([p[i][c] for i in range(j)])
        # The condition for existence is that there exists xi such that
        # 0 <= xi and xi + pi <= T and xi is non-decreasing.
        # The tightest constraint is xi >= x_{i-1}.
        # So x_i >= x_0 >= 0.
        # And x_i <= T - pi.
        # This requires x_i <= T - p_i for all i.
        # Since x_i is non-decreasing, x_i >= x_0.
        # The condition is x_0 + p_i <= T for all i.
        # But the cross constraint (Vi - Uk)(Vk - Ui) <= 0
        # implies x_i + pi = x_k + pk is not required, but
        # x_i + pi >= x_k + pk if i > k.
        # The actual condition for LOAD c is:
        # max(pi) + (j-1) * min(pi) > T is not quite right.
        # It's sum of the smallest pi's? No.
        # The condition is: max(pi) + (j-1) * min(pi) > T is for specific cases.
        # Given the constraints, the simplest check is:
        if max([p[i][c] for i in range(j)]) + (j - 1) * min([p[i][c] for i in range(j)]) > T:
            print(f"LOAD {c + 1}")
            return

solve()
