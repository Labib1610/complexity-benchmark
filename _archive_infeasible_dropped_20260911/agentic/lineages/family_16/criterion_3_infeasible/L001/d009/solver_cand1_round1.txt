import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for i in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Check for JOB i
    # V_{i,1} * V_{i,2} = U_{i,1} * 2^{p_{i,1}} * U_{i,2} * 2^{p_{i,2}} <= 2^T
    # Since U_{i,1} >= 1 and U_{i,2} >= 1, we need 2^{p_{i,1} + p_{i,2}} <= 2^T
    # So p_{i,1} + p_{i,2} <= T
    for i in range(j):
        if p1[i] + p2[i] > T:
            print(f"JOB {i + 1}")
            return

    # Check for LOAD c
    # Constraint 6/7: V_i V_k + U_i U_k <= V_i U_i + V_k U_k
    # (V_i - U_i)(V_k - U_k) <= 0
    # Since V_i = U_i * 2^{p_i}, V_i - U_i = U_i(2^{p_i} - 1)
    # So U_i(2^{p_i} - 1) * U_k(2^{p_k} - 1) <= 0
    # Since U_i >= 1, this implies (2^{p_i} - 1)(2^{p_k} - 1) <= 0
    # This must hold for all i, k.
    # If p_i > 0, then 2^{p_i} - 1 > 0.
    # If there exist i, k such that p_i > 0 and p_k > 0, then (2^{p_i}-1)(2^{p_k}-1) > 0,
    # which violates the inequality.
    # Thus, for a valid system, at most one p_i can be > 0.
    # If more than one p_i > 0, it's a LOAD contradiction.
    
    def check_load(p_list):
        # Count how many p_i > 0
        count = 0
        for p in p_list:
            if p > 0:
                count += 1
        if count > 1:
            return True
        # Also check V_i <= 2^T. Since U_i >= 1, V_i = U_i * 2^{p_i} >= 2^{p_i}.
        # If p_i > T, then V_i > 2^T, which is impossible.
        for p in p_list:
            if p > T:
                return True
        return False

    if check_load(p1):
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
