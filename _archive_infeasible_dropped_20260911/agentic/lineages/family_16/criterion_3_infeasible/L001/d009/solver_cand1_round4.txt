import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    params = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        params.append((p1, p2))
        idx += 2
        
    # Check for JOB i
    # Constraint: U1*2^p1 * U2*2^p2 <= 2^T, U1>=1, U2>=1
    # Implies 2^(p1+p2) <= 2^T => p1 + p2 <= T
    for i in range(j):
        if params[i][0] + params[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c
    # Global constraint: For a fixed c, let p_i = p_{i,c}.
    # We need V_i = U_i * 2^p_i, U_i >= 1, V_i <= 2^T.
    # The condition (V_i - U_k)(V_k - U_i) <= 0 implies that the intervals [U_i, V_i]
    # must be disjoint (or touch at endpoints).
    # If we sort indices such that U_1 < V_1 <= U_2 < V_2 <= ... <= U_j < V_j,
    # then V_j >= V_{j-1} * 2^p_j >= ... >= V_1 * 2^(p_1 + ... + p_j)
    # Since V_1 = U_1 * 2^p_1 >= 2^p_1, we have V_j >= 2^(sum(p_i)).
    # The constraint V_i <= 2^T must hold for all i, so V_j <= 2^T.
    # Thus, if sum(p_i) > T, it's a global contradiction.
    
    sum_p1 = sum(p[0] for p in params)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    sum_p2 = sum(p[1] for p in params)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
