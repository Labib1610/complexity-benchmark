```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    channels = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        channels.append((p1, p2))
        idx += 2
        
    # Check for Channel Contradiction (JOB i)
    # Rule 3: V_{i,1} * V_{i,2} <= 2^T
    # Minimum V_{i,1} = 1 * 2^{p_{i,1}} = 2^{p_{i,1}}
    # Minimum V_{i,2} = 1 * 2^{p_{i,2}} = 2^{p_{i,2}}
    # Product = 2^{p_{i,1} + p_{i,2}}
    # Condition: p_{i,1} + p_{i,2} > T
    for i in range(j):
        if channels[i][0] + channels[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for Spectrum Contradiction (LOAD c)
    # Rule 4: No two uplink bands share interior frequencies.
    # To minimize the highest frequency, we pack them starting from 1.
    # Let the bands be [U_i, V_i]. V_i = U_i * 2^{p_i}.
    # To minimize the maximum frequency, we set U_{i+1} = V_i.
    # U_1 = 1, V_1 = 2^{p_1}, U_2 = 2^{p_1}, V_2 = 2^{p_1} * 2^{p_2} = 2^{p_1 + p_2}, ...
    # The highest frequency is 2^{sum(p_i)}.
    # Condition: sum(p_i) > T
    
    sum_p1 = sum(c[0] for c in channels)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    sum_p2 = sum(c[1] for c in channels)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
