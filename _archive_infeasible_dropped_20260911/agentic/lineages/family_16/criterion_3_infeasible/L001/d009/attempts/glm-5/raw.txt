```python
import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    T = int(input_data[1])
    p1 = [0] * j
    p2 = [0] * j
    idx = 2
    for i in range(j):
        p1[i] = int(input_data[idx])
        p2[i] = int(input_data[idx + 1])
        idx += 2

    # Check for local contradiction (JOB i)
    for i in range(j):
        if p1[i] + p2[i] > T:
            print(f"JOB {i + 1}")
            return

    # Check for global contradiction (LOAD c)
    for c, p in [(1, p1), (2, p2)]:
        p_sorted = sorted(p)
        # The minimum possible maximum V is achieved by sorting p in ascending order.
        # The chain of inequalities forces V_max >= 2^(p_sorted[0] + p_sorted[1] + ... + p_sorted[j-1])
        total = sum(p_sorted)
        if total > T:
            print(f"LOAD {c}")
            return

if __name__ == "__main__":
    solve()
```
