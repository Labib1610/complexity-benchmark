import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    prep_times = []
    finish_times = []
    
    idx = 2
    sum_prep = 0
    sum_finish = 0
    
    for i in range(1, j + 1):
        p = int(input_data[idx])
        f = int(input_data[idx + 1])
        idx += 2
        
        if p + f > t:
            print(f"JOB {i}")
            return
        
        sum_prep += p
        sum_finish += f
        
    if sum_prep > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
