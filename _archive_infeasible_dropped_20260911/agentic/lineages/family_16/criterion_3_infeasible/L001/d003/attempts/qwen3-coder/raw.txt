```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for _ in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        idx += 2
        
        # Check Item Span Violation
        if p1 + p2 > T:
            print(f"JOB {_ + 1}")
            return
    
    # Check Stage Overload
    sum_p1 = sum(p1_list)
    sum_p2 = sum(p2_list)
    
    if sum_p1 > T:
        print("LOAD 1")
        return
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    main()
```
