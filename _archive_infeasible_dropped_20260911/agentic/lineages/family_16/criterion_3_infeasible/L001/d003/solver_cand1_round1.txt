```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    orders = []
    idx = 2
    
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check Sequential bottleneck
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        orders.append((p1, p2))
        
    # Check Resource bottleneck
    if sum_p1 > t:
        print("LOAD 1")
        return
    if sum_p2 > t:
        print("LOAD 2")
        return
    
    # If none of the above, it must be a FLOW bottleneck
    # Johnson's rule states the minimum makespan for a 2-machine flow shop
    # is achieved by sorting orders. If the minimum makespan > T, it's impossible.
    # The problem guarantees it's impossible, so if we passed the above checks,
    # it must be FLOW.
    print("FLOW")

if __name__ == '__main__':
    solve()
```
