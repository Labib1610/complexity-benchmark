```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    a = []
    b = []
    idx = 2
    for i in range(j):
        val_a = int(input_data[idx])
        val_b = int(input_data[idx+1])
        a.append(val_a)
        b.append(val_b)
        
        # Check JOB i
        # x_i >= 0, y_i - x_i >= a_i, y_i + b_i <= T
        # x_i + a_i <= y_i <= T - b_i
        # x_i <= T - b_i - a_i
        # Contradiction if T - b_i - a_i < 0 => a_i + b_i > T
        if val_a + val_b > T:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # LOAD 1: x_k >= 0, x_k <= T - a_k, |x_u - x_v| >= min(a_u, a_v)
    # This is equivalent to packing j intervals of length a_k in [0, T]
    # Sort by a_k descending. If we sort x_k such that x_1 <= x_2 <= ... <= x_j
    # Then x_{i+1} - x_i >= a_i (or a_{i+1}).
    # The condition |x_u - x_v| >= min(a_u, a_v) is satisfied if we sort by a_k.
    # Specifically, if we sort such that a_1 >= a_2 >= ... >= a_j,
    # then x_{i+1} - x_i >= a_i is sufficient.
    # Total span needed: sum_{i=1}^{j-1} a_i <= T - max(a_k) is not quite right.
    # The condition is: there exist x_1, ..., x_j in [0, T-a_k] such that |x_u - x_v| >= min(a_u, a_v).
    # Sort a_k ascending: a_1 <= a_2 <= ... <= a_j.
    # The tightest constraint is sum_{i=1}^j a_i > T.
    
    a_sorted = sorted(a)
    if sum(a_sorted) > T:
        print("LOAD 1")
        return
        
    b_sorted = sorted(b)
    if sum(b_sorted) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
