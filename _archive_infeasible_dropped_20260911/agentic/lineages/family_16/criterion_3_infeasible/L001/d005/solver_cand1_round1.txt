```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum_p1 = 0
    sum_p2 = 0
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Average of f_i(x) = p1*x + p2*(1-x) over [0, 1] is (p1 + p2) / 2
        # Condition: (p1 + p2) / 2 > T / 2  => p1 + p2 > T
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # F(1) = sum(p1_i), F(0) = sum(p2_i)
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
