```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    T = int(data[1])
    idx = 2
    total1 = 0
    total2 = 0
    for i in range(1, j + 1):
        p1 = int(data[idx]); p2 = int(data[idx + 1])
        idx += 2
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
        total1 += p1
        total2 += p2
    if total1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

solve()
```
