```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2, i + 1))
        idx += 2
        
    # Check JOB i
    for i in range(j):
        p1, p2, original_idx = jobs[i]
        if p1 + p2 > T:
            print(f"JOB {original_idx}")
            return
            
    # Check LOAD 1 (Channel 1 exclusivity)
    # Sum of p1 must be <= T
    sum_p1 = sum(job[0] for job in jobs)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2 (Channel 2 exclusivity)
    # Sum of p2 must be <= T
    sum_p2 = sum(job[1] for job in jobs)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
