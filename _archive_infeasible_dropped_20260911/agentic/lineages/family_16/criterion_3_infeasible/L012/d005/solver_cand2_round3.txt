import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for _ in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Check JOB i
    # Internal: y_i - x_i >= p1_i
    # Boundaries: x_i >= 0, y_i + p2_i <= T
    # y_i <= T - p2_i
    # y_i - x_i >= p1_i => y_i >= x_i + p1_i
    # So x_i + p1_i <= T - p2_i => x_i <= T - p1_i - p2_i
    # Since x_i >= 0, we need T - p1_i - p2_i >= 0
    # If T - p1_i - p2_i < 0, then JOB i is unsatisfiable.
    for i in range(j):
        if p1[i] + p2[i] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Span X: (x_u + p1_u) - x_v >= sum(p1)
    # Boundaries: x_i >= 0, x_i + p1_i <= T
    # Max value of (x_u + p1_u) is T (since x_u + p1_u <= T)
    # Min value of x_v is 0
    # Max span is T - 0 = T
    # If T < sum(p1), then LOAD 1 is unsatisfiable.
    if sum(p1) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Span Y: (y_u + p2_u) - y_v >= sum(p2)
    # Boundaries: y_i >= 0, y_i + p2_i <= T
    # Max value of (y_u + p2_u) is T
    # Min value of y_v is 0
    # Max span is T - 0 = T
    # If T < sum(p2), then LOAD 2 is unsatisfiable.
    if sum(p2) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
