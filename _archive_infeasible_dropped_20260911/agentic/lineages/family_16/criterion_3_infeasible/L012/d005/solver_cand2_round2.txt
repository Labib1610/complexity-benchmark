```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1 = []
    p2 = []
    
    idx = 2
    for i in range(j):
        p1_val = int(input_data[idx])
        p2_val = int(input_data[idx+1])
        p1.append(p1_val)
        p2.append(p2_val)
        
        # Check JOB i
        if p1_val + p2_val > t:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1
    # Sum of p1_i <= T
    if sum(p1) > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sum of p2_i <= T
    if sum(p2) > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
