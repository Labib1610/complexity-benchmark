```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for _ in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Check JOB i
    # JOB i: y_{i,1} - x_{i,1} = p_{i,1}, y_{i,2} - x_{i,2} = p_{i,2}, y_{i,1} <= x_{i,2}, 0 <= x_{i,1}, y_{i,2} <= T
    # y_{i,2} = x_{i,2} + p_{i,2} >= y_{i,1} + p_{i,2} = x_{i,1} + p_{i,1} + p_{i,2} >= p_{i,1} + p_{i,2}
    # Contradiction if p_{i,1} + p_{i,2} > T
    for i in range(j):
        if p1[i] + p2[i] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # LOAD 1: sum(p_{i,1}) > T
    if sum(p1) > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # LOAD 2: sum(p_{i,2}) > T
    if sum(p2) > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
