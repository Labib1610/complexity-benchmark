```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        p1_list.append(p1)
        p2_list.append(p2)
        
        # Check JOB i:
        # A_i = [x, x + p1], B_i = [y, y + p2]
        # Domain: 0 <= x, x + p1 <= T, 0 <= y, y + p2 <= T
        # Order: max(A_i) <= min(B_i) => x + p1 <= y
        # Minimal y is 0, so x + p1 <= 0 is impossible (p1 > 0).
        # Actually, the condition is: there exist x, y such that
        # 0 <= x, x + p1 <= T, 0 <= y, y + p2 <= T, and x + p1 <= y.
        # This is possible if and only if p1 + p2 <= T.
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
            
    # Check LOAD 1:
    # Sum of lengths of A_i must be <= T
    if sum(p1_list) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2:
    # Sum of lengths of B_i must be <= T
    if sum(p2_list) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
