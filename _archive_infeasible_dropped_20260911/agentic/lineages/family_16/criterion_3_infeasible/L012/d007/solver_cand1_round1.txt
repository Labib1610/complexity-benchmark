```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    subsystems = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        subsystems.append((p1, p2))
        idx += 2
        
    # Scenario 1: Constraint 1 (Precedence)
    # Time for subsystem i is p1 + p2. If p1 + p2 > T, it fails.
    for i in range(j):
        if subsystems[i][0] + subsystems[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Scenario 2: Constraint 2 (Capacitor Grid Capacity)
    # Sum of all p1 must be <= T.
    sum_p1 = sum(s[0] for s in subsystems)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Scenario 3: Constraint 3 (Initialization Grid Capacity)
    # Sum of all p2 must be <= T.
    sum_p2 = sum(s[1] for s in subsystems)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
