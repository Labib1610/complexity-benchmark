```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t_limit = int(input_data[1])
    
    p1 = []
    p2 = []
    
    idx = 2
    for i in range(j):
        c1 = int(input_data[idx])
        c2 = int(input_data[idx+1])
        p1.append(c1)
        p2.append(c2)
        
        # Check JOB i certificate
        if c1 + c2 > t_limit:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1 certificate
    if sum(p1) > t_limit:
        print("LOAD 1")
        return
        
    # Check LOAD 2 certificate
    if sum(p2) > t_limit:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
