```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1 = []
    p2 = []
    
    idx = 2
    for i in range(j):
        val1 = int(input_data[idx])
        val2 = int(input_data[idx+1])
        p1.append(val1)
        p2.append(val2)
        idx += 2
        
    # Check JOB k
    # Min d_k - a_k = p1_k + p2_k
    for k in range(j):
        if p1[k] + p2[k] > t:
            print(f"JOB {k + 1}")
            return
            
    # Check LOAD 1
    # Min measure of union C_k = sum(p1_k)
    if sum(p1) > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Min measure of union I_k = sum(p2_k)
    if sum(p2) > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
