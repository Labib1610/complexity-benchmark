```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    
    p1_list = []
    p2_list = []
    
    for i in range(1, j + 1):
        p1 = int(next(it))
        p2 = int(next(it))
        p1_list.append(p1)
        p2_list.append(p2)
        
        # Check JOB condition: p1 + p2 > T
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
    
    # Check LOAD 1: sum of all p1 > T
    if sum(p1_list) > T:
        print("LOAD 1")
        return
    
    # Check LOAD 2: sum of all p2 > T
    if sum(p2_list) > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    solve()
```
