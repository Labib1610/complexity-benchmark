import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    subsystems = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        subsystems.append((p1, p2, i + 1))
        idx += 2
        
    # Experiment 2: LOAD 1
    # Minimize completion time of all Capacitor Charge bursts.
    # This is simply the sum of all p1.
    sum_p1 = sum(s[0] for s in subsystems)
    if sum_p1 > T:
        print("LOAD 1")
        return
    
    # Experiment 3: LOAD 2
    # Minimize elapsed time of initialization grid.
    # This is the sum of all p2.
    sum_p2 = sum(s[1] for s in subsystems)
    if sum_p2 > T:
        print("LOAD 2")
        return
    
    # Experiment 1: JOB i
    # Minimize completion time of subsystem i: p1_i + p2_i.
    # Wait, the constraint is that we must schedule all j subsystems.
    # To minimize completion time of subsystem i, we should schedule
    # subsystem i first, then all others.
    # Completion time = p1_i + p2_i.
    # Wait, the problem says "Find a valid schedule of all j subsystems... 
    # that minimizes the absolute completion time of a specific subsystem i's 
    # System Initialization burst."
    # If we schedule i first, its completion time is p1_i + p2_i.
    # Is it possible to finish earlier? No, because p1_i must finish before p2_i.
    # So the minimum completion time for subsystem i is p1_i + p2_i.
    
    for i in range(j):
        if subsystems[i][0] + subsystems[i][1] > T:
            print(f"JOB {subsystems[i][2]}")
            return

if __name__ == '__main__':
    solve()
