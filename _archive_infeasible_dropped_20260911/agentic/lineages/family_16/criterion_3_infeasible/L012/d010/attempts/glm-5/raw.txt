```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    T = int(data[1])
    idx = 2
    p1 = [0]*j
    p2 = [0]*j
    for i in range(j):
        p1[i] = int(data[idx]); p2[i] = int(data[idx+1])
        idx += 2

    # Proof Type 1: independent manifolds per subsystem
    # Each subsystem i finishes at p1[i] + p2[i]
    for i in range(j):
        if p1[i] + p2[i] > T:
            sys.stdout.write(f"JOB {i+1}\n")
            return

    # Proof Type 2: only Flux Compression, single manifold
    # Earliest finish = sum of all p1
    if sum(p1) > T:
        sys.stdout.write("LOAD 1\n")
        return

    # Proof Type 3: only Plasma Venting, single manifold
    # Earliest finish = sum of all p2
    if sum(p2) > T:
        sys.stdout.write("LOAD 2\n")
        return

    # Fallback (should not happen per problem guarantee)
    sys.stdout.write("LOAD 1\n")

if __name__ == "__main__":
    solve()
```
