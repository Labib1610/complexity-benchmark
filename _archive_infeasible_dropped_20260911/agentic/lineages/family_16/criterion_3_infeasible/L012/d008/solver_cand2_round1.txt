```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    subsystems = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    max_p1_plus_p2 = 0
    
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        subsystems.append((p1, p2))
        sum_p1 += p1
        sum_p2 += p2
        if p1 + p2 > max_p1_plus_p2:
            max_p1_plus_p2 = p1 + p2
        idx += 2
        
    # Check JOB i
    # Time taken for subsystem i is p1 + p2.
    # If p1 + p2 > T, then JOB i is the bottleneck.
    for i in range(j):
        if subsystems[i][0] + subsystems[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Initialization is instantaneous. Capacitor charges must be processed sequentially.
    # Total time = sum(p1).
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Capacitor charges are instantaneous. Initialization bursts must be processed sequentially.
    # Total time = sum(p2).
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
