```python
j, T = map(int, input().split())

# Check for JOB bottleneck
for i in range(j):
    p1, p2 = map(int, input().split())
    if p1 + p2 > T:
        print(f"JOB {i+1}")
        exit()
    # Store durations for LOAD check
    if i == 0:
        durations = [(p1, p2)]
    else:
        durations.append((p1, p2))

# If we reach here, check for LOAD bottleneck
total_p1 = sum(d[0] for d in durations)
total_p2 = sum(d[1] for d in durations)

if total_p1 > T:
    print("LOAD 1")
elif total_p2 > T:
    print("LOAD 2")
```
