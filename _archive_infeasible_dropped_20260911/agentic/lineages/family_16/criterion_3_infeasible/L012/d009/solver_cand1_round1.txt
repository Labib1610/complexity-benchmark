```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for _ in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Scenario 1: JOB i
    # Constraint 1: g(k) >= f(k) + p1[k]
    # Min completion time = max(p1[k] + p2[k])
    max_val = -1
    max_idx = -1
    for i in range(j):
        if p1[i] + p2[i] > max_val:
            max_val = p1[i] + p2[i]
            max_idx = i + 1
    
    if max_val > t:
        print(f"JOB {max_idx}")
        return
    
    # Scenario 2: LOAD 1
    # p2[k] = 0, Constraint 3 dropped.
    # Constraint 2: [f(k), f(k) + p1[k]) non-intersecting.
    # Min completion time = sum(p1[k])
    if sum(p1) > t:
        print("LOAD 1")
        return
    
    # Scenario 3: LOAD 2
    # p1[k] = 0, Constraint 2 dropped.
    # Constraint 3: [g(k), g(k) + p2[k]) non-intersecting.
    # Min completion time = sum(p2[k])
    if sum(p2) > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
