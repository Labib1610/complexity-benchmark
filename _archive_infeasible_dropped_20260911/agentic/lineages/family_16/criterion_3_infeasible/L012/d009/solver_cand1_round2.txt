```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    subsystems = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        subsystems.append((p1, p2))
        idx += 2
        
    # Experiment 1: Check if any individual subsystem exceeds T
    for i in range(j):
        if subsystems[i][0] + subsystems[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Experiment 2: LOAD 1
    # System Initialization bursts are 0, Capacitor grid is serial.
    # Total time = sum of all Capacitor Charge bursts.
    total_p1 = sum(s[0] for s in subsystems)
    if total_p1 > t:
        print("LOAD 1")
        return
        
    # Experiment 3: LOAD 2
    # Capacitor Charge bursts are 0, Initialization grid is serial.
    # Total time = sum of all System Initialization bursts.
    total_p2 = sum(s[1] for s in subsystems)
    if total_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
