```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    
    idx = 2
    for _ in range(j):
        val1 = int(input_data[idx])
        val2 = int(input_data[idx+1])
        p1.append(val1)
        p2.append(val2)
        idx += 2
        
    # Check JOB i
    # Rule 1: |S_{k,1}| = p_{k,1}, |S_{k,2}| = p_{k,2}
    # Rule 2: max(S_{k,1}) < min(S_{k,2})
    # This is possible if and only if p_{k,1} + p_{k,2} <= T
    # Because we can pick {1, ..., p_{k,1}} and {p_{k,1}+1, ..., p_{k,1}+p_{k,2}}
    # The condition is p_{k,1} + p_{k,2} <= T.
    # If p_{k,1} + p_{k,2} > T, then JOB k is unsatisfiable.
    for i in range(j):
        if p1[i] + p2[i] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # Rule 1: |S_{k,c}| = p_{k,c}
    # Rule 3: S_{k,c} \cap S_{m,c} = \emptyset
    # This is possible if and only if sum(p_{k,c}) <= T
    sum1 = sum(p1)
    if sum1 > T:
        print("LOAD 1")
        return
        
    sum2 = sum(p2)
    if sum2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
