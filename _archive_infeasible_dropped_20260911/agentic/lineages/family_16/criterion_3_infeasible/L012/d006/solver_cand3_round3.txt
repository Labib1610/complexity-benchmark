```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        modules.append((p1, p2))
        idx += 2
        
    # Check JOB i: Phase Progression and Chamber Containment
    # For module i: Primary ends at x1, Secondary starts at x2.
    # x1 >= p1, x2 >= x1 + p2, x2 + p2 <= T
    # Actually, the constraint is:
    # Primary: [s1, e1] where e1 - s1 = p1, 0 <= s1 < e1 <= T
    # Secondary: [s2, e2] where e2 - s2 = p2, 0 <= s2 < e2 <= T
    # Phase Progression: s2 >= e1
    # So: 0 <= s1 < e1 <= s2 < e2 <= T
    # This implies e2 - s1 >= p1 + p2. Since e2 <= T and s1 >= 0,
    # we need p1 + p2 <= T.
    for i in range(j):
        if modules[i][0] + modules[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1: Upper Rail Integrity
    # Sum of p1 must be <= T
    sum_p1 = sum(m[0] for m in modules)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2: Lower Rail Integrity
    # Sum of p2 must be <= T
    sum_p2 = sum(m[1] for m in modules)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
