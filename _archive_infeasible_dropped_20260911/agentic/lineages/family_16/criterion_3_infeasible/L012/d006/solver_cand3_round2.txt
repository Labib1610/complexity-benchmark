```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    pairs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        pairs.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Internal Monotonicity: C_i >= B_i
    # Domain Spans: B_i - A_i = p1, D_i - C_i = p2
    # Bounds: A_i >= 0, B_i <= T, C_i >= 0, D_i <= T
    # From B_i - A_i = p1 and A_i >= 0 => B_i >= p1
    # From D_i - C_i = p2 and C_i >= B_i => D_i >= B_i + p2
    # So D_i >= p1 + p2. If p1 + p2 > T, then JOB i is unsatisfiable.
    for i in range(j):
        p1, p2 = pairs[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD 1: Alpha Separation
    # B_u <= A_v or B_v <= A_u for all u != v
    # This is equivalent to saying that the intervals [A_i, B_i] must be disjoint
    # except possibly at endpoints.
    # Since B_i - A_i = p1_i, we have intervals [A_i, A_i + p1_i].
    # The condition B_u <= A_v or B_v <= A_u means for any two intervals,
    # one must end before the other starts.
    # This is possible if and only if the sum of lengths <= T.
    # Wait, the condition is for all pairs. If we sort by A_i, we need
    # B_1 <= A_2, B_2 <= A_3, ..., B_{j-1} <= A_j.
    # Sum of (B_i - A_i) = sum(p1_i).
    # Total span = A_1 + (B_1 - A_1) + (A_2 - B_1) + (B_2 - A_2) + ... + (B_j - A_j)
    # Total span = A_1 + sum(p1_i) + sum(A_{i+1} - B_i) = B_j
    # Since A_1 >= 0 and A_{i+1} - B_i >= 0, we need sum(p1_i) <= T.
    if sum(p[0] for p in pairs) > T:
        print("LOAD 1")
        return

    # Check LOAD 2: Beta Separation
    # Similar logic for C_i, D_i with p2_i
    if sum(p[1] for p in pairs) > T:
        print("LOAD 2")
        return

solve()
```
