```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    u = []
    v = []
    idx = 2
    for i in range(j):
        u_i = int(input_data[idx])
        v_i = int(input_data[idx+1])
        u.append(u_i)
        v.append(v_i)
        idx += 2
        
    # Check JOB i
    for i in range(j):
        if u[i] + v[i] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    if sum(u) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    if sum(v) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
