```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    features = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        features.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Domain Precedence: min(secondary) >= max(primary)
    # Manifold Boundary: [0, T]
    # Measure Requirement: length(primary) = p1, length(secondary) = p2
    # This means p1 + p2 <= T
    for i in range(j):
        if features[i][0] + features[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Primary Separation: sum(p1) <= T
    sum_p1 = sum(f[0] for f in features)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Secondary Separation: sum(p2) <= T
    sum_p2 = sum(f[1] for f in features)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
