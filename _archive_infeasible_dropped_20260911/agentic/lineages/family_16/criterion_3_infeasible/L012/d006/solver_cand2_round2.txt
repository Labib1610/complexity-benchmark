```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_sum = 0
    p2_sum = 0
    
    # We need to check:
    # 1. JOB i: p1_i + p2_i > T
    # 2. LOAD 1: sum(p1) > T
    # 3. LOAD 2: sum(p2) > T
    
    # The problem guarantees exactly one of these conditions is met.
    # We can iterate through the subsystems to check JOB i,
    # while accumulating sums for LOAD 1 and LOAD 2.
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        p1_sum += p1
        p2_sum += p2
        
    if p1_sum > t:
        print("LOAD 1")
    elif p2_sum > t:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
