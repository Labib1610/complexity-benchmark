```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i: Rule 1 and 2
        # A_i = [s1, s1 + p1], B_i = [s2, s2 + p2]
        # s1 >= 0, s1 + p1 <= s2, s2 + p2 <= T
        # s1 + p1 <= s2 => s2 >= s1 + p1
        # s2 + p2 <= T => s1 + p1 + p2 <= T
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
        
        p1_list.append(p1)
        p2_list.append(p2)
        
    # Check LOAD 1: Rule 1 and 3
    # Sum of p1 <= T
    if sum(p1_list) > T:
        print("LOAD 1")
        return
    
    # Check LOAD 2: Rule 1 and 4
    # Sum of p2 <= T
    if sum(p2_list) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
