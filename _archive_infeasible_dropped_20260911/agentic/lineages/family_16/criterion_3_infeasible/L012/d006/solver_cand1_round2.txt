```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_sum = 0
    p2_sum = 0
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        p1_sum += p1
        p2_sum += p2
        
    # Check LOAD 1
    if p1_sum > t:
        print("LOAD 1")
        return
    
    # Check LOAD 2
    if p2_sum > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
