```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for _ in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Case 1: JOB i
    # x_i >= 0, y_i + p2_i <= T, x_i + p1_i <= y_i
    # x_i + p1_i + p2_i <= y_i + p2_i <= T
    # x_i <= T - p1_i - p2_i
    # Unsatisfiable if T - p1_i - p2_i < 0 => p1_i + p2_i > T
    for i in range(j):
        if p1[i] + p2[i] > T:
            print(f"JOB {i + 1}")
            return
            
    # Case 2: LOAD 1
    # x_i >= 0, x_i + p1_i <= T, and for all i != k, (x_i + p1_i <= x_k) or (x_k + p1_k <= x_i)
    # Let q_i = p1_i. We need x_i in [0, T - q_i].
    # Sort by q_i. If we have many variables, the separation constraints force
    # x_i to be spread out. Specifically, if we sort such that q_1 <= q_2 <= ... <= q_j,
    # the tightest constraints are x_i + q_i <= x_{i+1}.
    # Summing these: x_1 + q_1 + q_2 + ... + q_{j-1} <= x_j <= T - q_j
    # So sum_{i=1}^j q_i <= T.
    # Actually, the condition for existence of such x_i is sum_{i=1}^j q_i <= T.
    # If sum > T, it's unsatisfiable.
    if sum(p1) > T:
        print("LOAD 1")
        return
        
    # Case 3: LOAD 2
    # Similar logic for y_i: y_i >= 0, y_i + p2_i <= T
    # The separation constraint is y_i + p2_i <= y_k or y_k + p2_k <= y_i.
    # This is equivalent to the same condition as LOAD 1.
    if sum(p2) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
