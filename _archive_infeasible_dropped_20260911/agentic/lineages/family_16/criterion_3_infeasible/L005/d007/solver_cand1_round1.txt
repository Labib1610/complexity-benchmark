```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    pairs = []
    idx = 2
    for i in range(j):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        pairs.append((u, v))
        idx += 2
        
    # Check JOB i
    # Constraint 3: x < y for all x in X_i, y in Y_i
    # inf(X_i) >= 1 => sup(X_i) >= 1 + u_i
    # inf(Y_i) >= sup(X_i) >= 1 + u_i
    # sup(Y_i) >= inf(Y_i) + v_i >= 1 + u_i + v_i
    # Constraint 6: sup(Y_i) <= T + 1
    # So 1 + u_i + v_i <= T + 1 => u_i + v_i <= T
    for i in range(j):
        u, v = pairs[i]
        if u + v > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Sum of lengths of X_i must be <= T
    sum_u = sum(p[0] for p in pairs)
    if sum_u > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sum of lengths of Y_i must be <= T
    sum_v = sum(p[1] for p in pairs)
    if sum_v > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
