```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    items = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        items.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # C2,i = S2,i + p2,i >= (S1,i + p1,i) + p2,i >= (1 + p1,i) + p2,i
    # alpha_i = 1 + p1,i + p2,i
    # Condition: alpha_i > T + 1
    for i in range(j):
        p1, p2 = items[i]
        if 1 + p1 + p2 > t + 1:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # beta_1 = sum(p1,k) + 1
    # Condition: beta_1 > T + 1
    sum_p1 = sum(item[0] for item in items)
    if sum_p1 + 1 > t + 1:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # beta_2 = sum(p2,k) + 1
    # Condition: beta_2 > T + 1
    sum_p2 = sum(item[1] for item in items)
    if sum_p2 + 1 > t + 1:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
