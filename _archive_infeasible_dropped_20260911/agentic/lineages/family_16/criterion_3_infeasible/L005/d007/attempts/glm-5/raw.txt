```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    T = int(data[1])
    idx = 2
    deadline = T + 1

    job_idx = -1
    load1 = 0
    load2 = 0

    for i in range(1, j + 1):
        p1 = int(data[idx]); p2 = int(data[idx + 1]); idx += 2
        alpha = p1 + p2 + 1
        if alpha > deadline:
            job_idx = i
            break
        load1 += p1
        load2 += p2

    if job_idx != -1:
        sys.stdout.write(f"JOB {job_idx}\n")
        return

    if load1 + 1 > deadline:
        sys.stdout.write("LOAD 1\n")
    else:
        sys.stdout.write("LOAD 2\n")

if __name__ == "__main__":
    solve()
```
