```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    T = int(data[1])
    
    total1 = 0
    total2 = 0
    job_idx = -1
    
    pos = 2
    for i in range(j):
        p1 = int(data[pos])
        p2 = int(data[pos + 1])
        pos += 2
        total1 += p1
        total2 += p2
        if p1 + p2 > T and job_idx == -1:
            job_idx = i + 1
    
    if job_idx != -1:
        sys.stdout.write(f"JOB {job_idx}\n")
    elif total1 > T:
        sys.stdout.write("LOAD 1\n")
    else:
        sys.stdout.write("LOAD 2\n")

solve()
```
