```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        jobs.append((p1, p2, i + 1))
        idx += 2
        
    # Check for Routing Delay (JOB i)
    # A job i is a routing delay if p1_i + p2_i > T
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > t:
            print(f"JOB {jobs[i][2]}")
            return
            
    # Check for Throughput Saturation (LOAD c)
    # Station 1 total load: sum(p1)
    # Station 2 total load: sum(p2)
    sum1 = 0
    sum2 = 0
    for i in range(j):
        sum1 += jobs[i][0]
        sum2 += jobs[i][1]
        
    if sum1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
