```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    # The deadline is T + 1
    deadline = t + 1
    
    # Check for JOB i certificate
    # Constraint: S1 >= 1, S2 >= S1 + p1, C2 = S2 + p2 <= deadline
    # Minimum C2 = 1 + p1 + p2
    # If 1 + p1 + p2 > deadline, then JOB i is impossible
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        if 1 + p1 + p2 > deadline:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c certificate
    # Facility c is overloaded if sum(p_c) > deadline - 1 (since start time >= 1)
    # i.e., sum(p_c) > T
    sum_p1 = 0
    sum_p2 = 0
    for i in range(j):
        sum_p1 += int(input_data[2 + 2 * i])
        sum_p2 += int(input_data[3 + 2 * i])
        
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
