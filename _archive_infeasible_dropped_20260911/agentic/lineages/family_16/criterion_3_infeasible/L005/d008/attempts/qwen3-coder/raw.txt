```python
import sys
from typing import List, Tuple

def read_input() -> Tuple[int, int, List[Tuple[int, int]]]:
    lines = sys.stdin.read().splitlines()
    j, T = map(int, lines[0].split())
    jobs = []
    for i in range(1, j + 1):
        p1, p2 = map(int, lines[i].split())
        jobs.append((p1, p2))
    return j, T, jobs

def find_job_certificate(jobs: List[Tuple[int, int]], T: int) -> int:
    # Check if any job is impossible even in isolation
    for i, (p1, p2) in enumerate(jobs):
        # Minimum completion time for this job is 1 + p1 + p2
        min_completion = 1 + p1 + p2
        if min_completion > T + 1:
            return i + 1  # 1-based indexing
    return -1

def find_load_certificate(jobs: List[Tuple[int, int]], T: int) -> int:
    # Check facility 1 overload
    durations_1 = [p1 for p1, _ in jobs]
    durations_1.sort()
    
    # Calculate minimum makespan on facility 1
    current_time = 1
    for duration in durations_1:
        current_time += duration
    
    if current_time - 1 > T + 1:
        return 1
    
    # Check facility 2 overload
    durations_2 = [p2 for _, p2 in jobs]
    durations_2.sort()
    
    # Calculate minimum makespan on facility 2
    current_time = 1
    for duration in durations_2:
        current_time += duration
    
    if current_time - 1 > T + 1:
        return 2
    
    return -1

def solve():
    j, T, jobs = read_input()
    
    # First try to find a JOB certificate
    job_index = find_job_certificate(jobs, T)
    if job_index != -1:
        print(f"JOB {job_index}")
        return
    
    # If no JOB certificate, find LOAD certificate
    load_facility = find_load_certificate(jobs, T)
    if load_facility != -1:
        print(f"LOAD {load_facility}")
        return

if __name__ == "__main__":
    solve()
```
