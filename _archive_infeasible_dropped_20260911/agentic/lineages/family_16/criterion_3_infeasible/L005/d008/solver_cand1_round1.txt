```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    items = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        items.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Rule 1: P(U_k,p1) >= P(U_k,0) + p1 >= 1 + p1
    # Rule 3: P(V_k,0) >= P(U_k,p1) >= 1 + p1
    # Rule 2: P(V_k,p2) >= P(V_k,0) + p2 >= 1 + p1 + p2
    # Condition: 1 + p1 + p2 > T + 1 => p1 + p2 > T
    for i in range(j):
        p1, p2 = items[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Rule 1: P(U_k,p1) >= P(U_k,0) + p1
    # Rule 4: For any pair, one sequence follows another.
    # This implies the total span is at least 1 + sum(p1_k for all k)
    # Condition: 1 + sum(p1) > T + 1 => sum(p1) > T
    sum_p1 = sum(item[0] for item in items)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Rule 2: P(V_k,p2) >= P(V_k,0) + p2
    # Rule 5: For any pair, one sequence follows another.
    # This implies the total span is at least 1 + sum(p2_k for all k)
    # Condition: 1 + sum(p2) > T + 1 => sum(p2) > T
    sum_p2 = sum(item[1] for item in items)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
