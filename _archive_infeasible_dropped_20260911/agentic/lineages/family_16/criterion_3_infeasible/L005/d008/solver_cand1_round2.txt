```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    # The deadline is T + 1
    deadline = t + 1
    
    # Check JOB i condition:
    # For each item k, S_{1,k} >= 1.
    # C_{1,k} = S_{1,k} + p_{1,k} >= 1 + p_{1,k}
    # S_{2,k} >= C_{1,k} >= 1 + p_{1,k}
    # C_{2,k} = S_{2,k} + p_{2,k} >= 1 + p_{1,k} + p_{2,k}
    # If 1 + p_{1,k} + p_{2,k} > deadline, then JOB k is a certificate.
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        if 1 + p1 + p2 > deadline:
            print(f"JOB {i}")
            return
            
    # Check LOAD c condition:
    # For facility c, the total length is sum(p_{c,k}).
    # Since they start at >= 1 and are disjoint, the supremum is >= 1 + sum(p_{c,k}).
    # If 1 + sum(p_{c,k}) > deadline, then LOAD c is a certificate.
    
    # We need to re-read or store the p1, p2 values. 
    # Since we already iterated, let's reset the pointer.
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    for _ in range(j):
        sum_p1 += int(input_data[idx])
        sum_p2 += int(input_data[idx + 1])
        idx += 2
        
    if 1 + sum_p1 > deadline:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
