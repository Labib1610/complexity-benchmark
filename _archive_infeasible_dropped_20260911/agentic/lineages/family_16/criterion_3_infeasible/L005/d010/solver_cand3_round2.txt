```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    items = []
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        items.append((p1, p2, i))
        idx += 2
        
    # Check JOB i: p1 + p2 > T
    for p1, p2, i in items:
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
            
    # Check LOAD 1: sum(p1) > T
    sum_p1 = sum(item[0] for item in items)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2: sum(p2) > T
    sum_p2 = sum(item[1] for item in items)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
