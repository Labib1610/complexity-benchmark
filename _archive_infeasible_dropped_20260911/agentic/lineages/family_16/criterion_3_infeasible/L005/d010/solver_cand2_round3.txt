import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check JOB i constraint
    # M2_i_max = T - p2_i/2
    # M1_i_min = p1_i/2
    # M2_i_max - M1_i_min < (p1_i + p2_i)/2
    # T - p2_i/2 - p1_i/2 < (p1_i + p2_i)/2
    # T < p1_i + p2_i
    for i in range(j):
        p1, p2 = jobs[i]
        if t < p1 + p2:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c constraint
    # For machine c, the span required is:
    # (p_{c,1}/2) + sum_{k=1 to j-1} ((p_{c,k} + p_{c,k+1})/2) + (p_{c,j}/2)
    # = p_{c,1}/2 + (p_{c,1}/2 + p_{c,2}) + (p_{c,2}/2 + p_{c,3}/2) ... + p_{c,j}/2
    # = sum(p_{c,k})
    
    sum_p1 = sum(job[0] for job in jobs)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    sum_p2 = sum(job[1] for job in jobs)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
