```python
import sys

def solve():
    # Read j and T
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        j, t = map(int, line1)
    except ValueError:
        return

    # Read jobs
    jobs = []
    for i in range(j):
        try:
            p1, p2 = map(int, sys.stdin.readline().split())
            jobs.append((p1, p2, i + 1))
        except ValueError:
            break

    # Check JOB i constraint:
    # For each job i, we need p1_i + p2_i <= T.
    # If p1_i + p2_i > T, then it's impossible to fit the job.
    for p1, p2, idx in jobs:
        if p1 + p2 > t:
            print(f"JOB {idx}")
            return

    # Check LOAD c constraint:
    # For machine 1, sum of p1_i <= T.
    # For machine 2, sum of p2_i <= T.
    sum_p1 = 0
    sum_p2 = 0
    for p1, p2, idx in jobs:
        sum_p1 += p1
        sum_p2 += p2
    
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
