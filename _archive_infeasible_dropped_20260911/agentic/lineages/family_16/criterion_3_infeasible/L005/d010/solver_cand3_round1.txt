```python
import sys

def solve():
    # Read j and T
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        j, t = map(int, line1)
    except ValueError:
        return

    jobs = []
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        try:
            p1, p2 = map(int, sys.stdin.readline().split())
            jobs.append((p1, p2, i))
            sum_p1 += p1
            sum_p2 += p2
            
            # Check JOB i constraint:
            # A_k has p1 elements, B_k has p2 elements.
            # A_k = {s_a, ..., s_a + p1 - 1}
            # B_k = {s_b, ..., s_b + p2 - 1}
            # Rule 3: max(A_k) < min(B_k) => s_a + p1 - 1 < s_b
            # Smallest s_a is 1, so s_b >= p1 + 1.
            # Largest element in B_k is s_b + p2 - 1 >= p1 + 1 + p2 - 1 = p1 + p2.
            # If p1 + p2 > T, then JOB i is impossible.
            if p1 + p2 > t:
                print(f"JOB {i}")
                return
        except ValueError:
            break
            
    # Check LOAD c constraints:
    # Rule 4: sum of p1 <= T
    # Rule 5: sum of p2 <= T
    if sum_p1 > t:
        print("LOAD 1")
    elif sum_p2 > t:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
