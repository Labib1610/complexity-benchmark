```python
import sys

def solve():
    # Read n and W
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    W = int(input_data[1])
    
    widgets = []
    idx = 2
    for i in range(n):
        w1 = int(input_data[idx])
        w2 = int(input_data[idx+1])
        widgets.append((w1, w2, i + 1))
        idx += 2
        
    # Check JOB i: w1 + w2 > W
    # The interval is [1, W+1], length W.
    # Bottom interval [x, x+w1), Top interval [y, y+w2)
    # Constraint: y >= x+w1.
    # Smallest possible x is 1, so smallest y is 1+w1.
    # Top interval ends at y+w2 = 1+w1+w2.
    # Must be <= W+1, so w1+w2 <= W.
    for i in range(n):
        w1, w2, original_idx = widgets[i]
        if w1 + w2 > W:
            print(f"JOB {original_idx}")
            return
            
    # Check LOAD 1: sum(w1) > W
    sum_w1 = sum(w[0] for w in widgets)
    if sum_w1 > W:
        print("LOAD 1")
        return
        
    # Check LOAD 2: sum(w2) > W
    sum_w2 = sum(w[1] for w in widgets)
    if sum_w2 > W:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
