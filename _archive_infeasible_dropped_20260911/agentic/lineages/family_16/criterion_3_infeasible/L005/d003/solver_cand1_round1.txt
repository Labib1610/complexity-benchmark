```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        jobs.append((p1, p2, i))
        sum_p1 += p1
        sum_p2 += p2
        
        # Check for Routing Delay: Job i breaches deadline in isolation
        # Completion time for job i in isolation is p1 + p2
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
            
    # Check for Throughput Saturation: Station 1 or 2 breaches deadline
    # If station 2 takes 0 time, station 1 must process all jobs (sum_p1)
    if sum_p1 > t:
        print("LOAD 1")
        return
    
    # If station 1 takes 0 time, station 2 must process all jobs (sum_p2)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
