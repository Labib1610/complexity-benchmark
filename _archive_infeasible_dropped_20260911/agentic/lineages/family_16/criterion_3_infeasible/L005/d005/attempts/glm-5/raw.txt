```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    deadline = T + 1
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        if p1 + p2 > deadline:
            print(f"JOB {i}")
            return
    
    sum_p1 = 0
    sum_p2 = 0
    for k in range(j):
        sum_p1 += int(input_data[2 + 2 * k])
        sum_p2 += int(input_data[2 + 2 * k + 1])
        
    if sum_p1 > deadline:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
