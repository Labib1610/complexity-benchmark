```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    first_bad = None
    p1_sum = 0
    p2_sum = 0
    for idx in range(1, j + 1):
        p1 = int(next(it))
        p2 = int(next(it))
        p1_sum += p1
        p2_sum += p2
        if first_bad is None and p1 + p2 > T:
            first_bad = idx
    if first_bad is not None:
        sys.stdout.write(f"JOB {first_bad}")
        return
    if p1_sum > T:
        sys.stdout.write("LOAD 1")
    else:
        # problem guarantees at least one certificate, so sum p2 must exceed T
        sys.stdout.write("LOAD 2")

if __name__ == "__main__":
    solve()
```
