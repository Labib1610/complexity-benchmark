```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t_limit = int(input_data[1])
    deadline = t_limit + 1
    
    items = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        items.append((p1, p2))
        idx += 2
        
    # Check Routing Delay
    for i in range(j):
        p1, p2 = items[i]
        # Start at t=1, finish stage 1 at 1 + p1 - 1 = p1
        # Start stage 2 at p1 + 1, finish at p1 + 1 + p2 - 1 = p1 + p2
        if p1 + p2 > deadline:
            print(f"JOB {i + 1}")
            return
            
    # Check Throughput Saturation
    sum_p1 = 0
    sum_p2 = 0
    for p1, p2 in items:
        sum_p1 += p1
        sum_p2 += p2
        
    # Facility 1 starts at t=1, finishes at 1 + sum_p1 - 1 = sum_p1
    if sum_p1 > deadline:
        print("LOAD 1")
        return
    
    # Facility 2 starts at t=1, finishes at 1 + sum_p2 - 1 = sum_p2
    if sum_p2 > deadline:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
