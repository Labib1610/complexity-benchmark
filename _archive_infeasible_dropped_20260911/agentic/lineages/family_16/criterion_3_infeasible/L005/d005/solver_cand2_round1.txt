```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    items = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        items.append((p1, p2, i))
        sum_p1 += p1
        sum_p2 += p2
        idx += 2
        
    # Check Throughput Saturation
    if sum_p1 >= t:
        print("LOAD 1")
        return
    if sum_p2 >= t:
        print("LOAD 2")
        return
        
    # Check Routing Delay
    # Item i finishes at p1_i + p2_i - 1.
    # If p1_i + p2_i - 1 >= t, then p1_i + p2_i > t.
    # The problem states "final cycle of its processing is cycle T or greater".
    # If it starts at 0, it occupies [0, p1+p2-1].
    # The final cycle is p1+p2-1.
    # So we check if p1_i + p2_i - 1 >= t.
    for p1, p2, i in items:
        if p1 + p2 - 1 >= t:
            print(f"JOB {i}")
            return

if __name__ == '__main__':
    solve()
```
