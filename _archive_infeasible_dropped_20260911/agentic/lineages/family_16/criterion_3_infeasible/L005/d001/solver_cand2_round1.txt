```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    orders = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        orders.append((p1, p2))
        idx += 2
        
    # Check for JOB i
    # Residence time = E_{i,2} - S_{i,1}
    # Delay time = S_{i,2} - E_{i,1}
    # Residence - Delay = (E_{i,2} - S_{i,1}) - (S_{i,2} - E_{i,1})
    # Since E_{i,2} - S_{i,2} = p2 and E_{i,1} - S_{i,1} = p1
    # Residence - Delay = p1 + p2
    for i in range(j):
        if orders[i][0] + orders[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c
    # Completion time - Idle time = Total processing time on stage c
    # Stage 1 total processing time = sum(p1)
    # Stage 2 total processing time = sum(p2)
    
    sum_p1 = 0
    sum_p2 = 0
    for p1, p2 in orders:
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
