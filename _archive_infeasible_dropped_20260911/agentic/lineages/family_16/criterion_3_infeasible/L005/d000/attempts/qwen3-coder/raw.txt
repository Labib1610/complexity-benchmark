```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    
    index = 2
    jobs = []
    for i in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        jobs.append((p1, p2))
        index += 2
    
    # Check for single job too long
    for i, (p1, p2) in enumerate(jobs):
        if p1 + p2 > T:
            print(f"JOB {i+1}")
            return
    
    # Check for machine overload
    total_p1 = sum(p1 for p1, p2 in jobs)
    total_p2 = sum(p2 for p1, p2 in jobs)
    
    if total_p1 > T:
        print("LOAD 1")
        return
    if total_p2 > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    main()
```
