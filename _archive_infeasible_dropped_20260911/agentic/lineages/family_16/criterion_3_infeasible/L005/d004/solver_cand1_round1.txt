```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    items = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        items.append((p1, p2, i + 1))
        idx += 2
        
    # Check for Internal Overlap
    # An item i has internal overlap if p1_i + p2_i > T
    for p1, p2, i in items:
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
            
    # Check for Resource Overlap
    # Facility 1: sum of p1_i > T
    # Facility 2: sum of p2_i > T
    sum1 = sum(item[0] for item in items)
    sum2 = sum(item[1] for item in items)
    
    if sum1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
