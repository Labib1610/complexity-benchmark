```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    items = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        items.append((p1, p2, i + 1))
        idx += 2
        
    # Check for LOAD 1: Sum of p1 > T
    sum_p1 = sum(item[0] for item in items)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check for LOAD 2: Sum of p2 > T
    sum_p2 = sum(item[1] for item in items)
    if sum_p2 > T:
        print("LOAD 2")
        return
        
    # Check for JOB i: p1 + p2 > T
    # The transit span of item i is at least p1_i + p2_i.
    # If p1_i + p2_i > T, then for any sequence, the transit span 
    # (time from start at F1 to end at F2) is at least p1_i + p2_i > T.
    for i in range(j):
        if items[i][0] + items[i][1] > T:
            print(f"JOB {items[i][2]}")
            return

if __name__ == '__main__':
    solve()
```
