import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    deadline = t + 1
    
    # Constraints:
    # JOB i: S1,i >= 1, S2,i >= S1,i + p1,i, C2,i = S2,i + p2,i <= T+1
    # This simplifies to: 1 + p1,i + p2,i <= T+1 => p1,i + p2,i <= T
    # If p1,i + p2,i > T, then JOB i is the certificate.
    
    # LOAD c:
    # For facility c, sum of processing times must be <= T.
    # If sum(p_c,k) > T, then LOAD c is the certificate.
    
    sum_p1 = 0
    sum_p2 = 0
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Check LOAD 1
    if sum_p1 > t:
        print("LOAD 1")
        return
    
    # Check LOAD 2
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
