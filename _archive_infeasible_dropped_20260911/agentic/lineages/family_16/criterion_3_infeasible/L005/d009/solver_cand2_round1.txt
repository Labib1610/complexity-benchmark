```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    # The domain is [1, T+1], so the total length available is T.
    # Constraint 1: Domain [1, T+1] means length is T.
    # Constraint 2: p1, p2.
    # Constraint 3: S2 >= C1.
    # Since C1 = S1 + p1 and C2 = S2 + p2, and 1 <= S1, C2 <= T+1:
    # S2 >= S1 + p1
    # C2 = S2 + p2 >= S1 + p1 + p2
    # Since S1 >= 1, C2 >= 1 + p1 + p2.
    # Also C2 <= T+1, so 1 + p1 + p2 <= T+1 => p1 + p2 <= T.
    
    # Check JOB i:
    # For each item i, if p1_i + p2_i > T, then it's impossible.
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c:
    # Facility 1: sum(p1_i) <= T
    # Facility 2: sum(p2_i) <= T
    sum_p1 = 0
    sum_p2 = 0
    for i in range(j):
        sum_p1 += int(input_data[2 + 2 * i])
        sum_p2 += int(input_data[3 + 2 * i])
        
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
