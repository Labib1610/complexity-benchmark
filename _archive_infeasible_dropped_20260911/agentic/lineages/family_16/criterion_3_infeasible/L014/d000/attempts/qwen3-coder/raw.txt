import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    
    p1_total = 0
    p2_total = 0
    
    for i in range(j):
        p1 = int(data[2 + 2*i])
        p2 = int(data[3 + 2*i])
        
        # Check Individual Span Exceeded
        if p1 + p2 > T:
            print(f"JOB {i+1}")
            return
            
        p1_total += p1
        p2_total += p2
    
    # Check Axis Overcapacity
    if p1_total > T:
        print("LOAD 1")
    elif p2_total > T:
        print("LOAD 2")
    else:
        # This should not happen according to problem statement,
        # but if it does, we can default to one of them.
        # But per problem note, one of the above conditions must hold.
        pass

if __name__ == "__main__":
    main()
