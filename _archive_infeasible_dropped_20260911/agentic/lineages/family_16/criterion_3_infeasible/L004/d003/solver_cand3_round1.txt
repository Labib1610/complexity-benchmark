```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    orders = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        orders.append((p1, p2))
        idx += 2
        
    # Check for JOB i bottleneck
    for i in range(j):
        if orders[i][0] + orders[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c bottleneck
    sum_p1 = sum(order[0] for order in orders)
    sum_p2 = sum(order[1] for order in orders)
    
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
