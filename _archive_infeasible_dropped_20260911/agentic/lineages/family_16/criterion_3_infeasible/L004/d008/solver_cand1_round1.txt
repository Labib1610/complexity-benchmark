```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    r1_list = []
    r2_list = []
    
    idx = 2
    for i in range(j):
        r1 = int(input_data[idx])
        r2 = int(input_data[idx+1])
        r1_list.append(r1)
        r2_list.append(r2)
        idx += 2
        
        # Check for JOB i contradiction
        # Rule 3: M1,k and M2,k concatenated must have linearly independent rows.
        # This means r1 + r2 <= T.
        if r1 + r2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c contradiction
    # Rule 2: Concatenation of all M_c,k must have linearly independent rows.
    # This means sum(r_c,k) <= T.
    sum_r1 = sum(r1_list)
    sum_r2 = sum(r2_list)
    
    if sum_r1 > T:
        print("LOAD 1")
    else:
        # If sum_r1 <= T, then sum_r2 must be > T based on problem guarantee
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
