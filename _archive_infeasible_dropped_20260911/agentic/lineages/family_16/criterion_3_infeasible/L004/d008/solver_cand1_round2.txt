```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        
        # Check for Job-specific contradiction
        # Rule 1: S1,k U S2,k subset {1..T}
        # Rule 3: |S1,k| + |S2,k| = p1 + p2 must be <= T
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check for Collection-specific contradiction
    # Rule 1: S_c,k subset {1..T}
    # Rule 2: S_c,1, ..., S_c,j must be mutually disjoint
    # Total elements needed for collection c: sum(p_c,k)
    # Must be <= T
    
    sum_p1 = sum(p1_list)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    sum_p2 = sum(p2_list)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
