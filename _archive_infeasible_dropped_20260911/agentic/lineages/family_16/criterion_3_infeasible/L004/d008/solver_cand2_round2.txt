import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check for Job-specific contradiction
    # Rule 3 implies the union of S1,k and S2,k must be a contiguous range of size p1 + p2.
    # The smallest range of size p1 + p2 is 1 to p1 + p2.
    # If p1 + p2 > T, then Rule 1 and Rule 3 cannot be satisfied.
    for i in range(j):
        p1, p2 = jobs[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for Collection-specific contradiction
    # Collection 1: Sum of p1 for all jobs must be <= T
    # Collection 2: Sum of p2 for all jobs must be <= T
    sum_p1 = 0
    sum_p2 = 0
    for p1, p2 in jobs:
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > t:
        print("LOAD 1")
    else:
        # The problem guarantees one of the two types of certificates exists.
        # If not LOAD 1, it must be LOAD 2.
        print("LOAD 2")

if __name__ == '__main__':
    solve()
