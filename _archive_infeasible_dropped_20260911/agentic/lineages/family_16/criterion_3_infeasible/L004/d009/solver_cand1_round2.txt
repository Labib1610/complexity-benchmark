import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check for JOB i contradiction
    # Axiom III: max(S2,k) - min(S1,k) = p1 + p2 - 1
    # Axiom I: 1 <= min(S1,k) < max(S1,k) <= max(S2,k) <= T
    # The smallest possible max(S2,k) - min(S1,k) is achieved when
    # S1,k and S2,k are packed as tightly as possible.
    # The condition "every element in S1,k < every element in S2,k"
    # implies that the range [min(S1,k), max(S2,k)] must contain at least
    # p1 + p2 elements.
    # The span is max(S2,k) - min(S1,k).
    # Since there are p1 + p2 distinct integers in the union S1,k U S2,k,
    # and all elements are in [1, T], the span must be at least p1 + p2 - 1.
    # The problem states the span must be exactly p1 + p2 - 1.
    # This is possible if and only if the set S1,k U S2,k is a contiguous block of integers.
    # The constraint is: min(S1,k) >= 1 and max(S2,k) <= T.
    # If we pick the smallest possible range [x, x + p1 + p2 - 1],
    # we need x >= 1 and x + p1 + p2 - 1 <= T.
    # So p1 + p2 <= T.
    # If p1 + p2 > T, then no such sets exist.
    
    for i in range(j):
        p1, p2 = jobs[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c contradiction
    # Axiom II: |Union S_c,k| = sum |S_c,k|
    # This means all sets in collection c must be disjoint.
    # Since all elements are in [1, T], the sum of cardinalities
    # must be <= T.
    
    sum_p1 = 0
    sum_p2 = 0
    for p1, p2 in jobs:
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
