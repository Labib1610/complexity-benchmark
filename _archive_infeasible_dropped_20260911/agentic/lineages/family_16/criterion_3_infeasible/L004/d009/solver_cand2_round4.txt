```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    queries = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        queries.append((p1, p2))
        idx += 2
        
    # Check for JOB i (Query-specific contradiction)
    # Axiom III: 
    # 1. Cartesian product A1 x A2 has no (x, y) with x >= y.
    #    This implies max(A1) < min(A2).
    # 2. No z such that min(A1) < z < max(A2) and z not in (A1 U A2).
    #    This implies the union A1 U A2 must be a contiguous block of integers.
    # Since max(A1) < min(A2), the union is [min(A1), max(A2)].
    # The size of the union is |A1| + |A2| = p1 + p2.
    # Axiom I: All elements in [1, T].
    # So, p1 + p2 <= T.
    # If p1 + p2 > T, then JOB i is the contradiction.
    
    for i in range(j):
        p1, p2 = queries[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c (Dimension-specific contradiction)
    # Axiom II: Sum of sizes = size of union.
    # This implies all sets in dimension c must be mutually disjoint.
    # Total size required = sum(p_c for all k).
    # Axiom I: Total size <= T.
    # If sum(p_c) > T, then LOAD c is the contradiction.
    
    sum_p1 = 0
    sum_p2 = 0
    for p1, p2 in queries:
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
