```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check for JOB i contradiction
    # Rule 3 says: C_k is a contiguous block of 1s.
    # No index u where B_k[u]=1 and v >= u where A_k[v]=1.
    # This implies all 1s in B_k must appear before all 1s in A_k.
    # So C_k has p1 + p2 ones.
    # Rule 1: p1 + p2 <= T
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c contradiction
    # Rule 2: Sum of column sums <= 1.
    # Total ones in load 1 = sum(p1_k)
    # Total ones in load 2 = sum(p2_k)
    # Rule 1: Total ones <= T
    
    sum_p1 = 0
    sum_p2 = 0
    for p1, p2 in jobs:
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
