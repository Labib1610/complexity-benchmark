```python
import sys

def solve():
    # Read j and T
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        j, t = map(int, line1)
    except ValueError:
        return

    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        try:
            p1, p2 = map(int, sys.stdin.readline().split())
        except ValueError:
            break
        
        # Check for JOB i contradiction
        # Rule 3 implies S1,k and S2,k form a contiguous block of size p1 + p2
        # Rule 1 requires this block to fit in [1, T]
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Check for LOAD c contradiction
    # Rule 2 requires all S1,k to be disjoint, total size sum(p1) <= T
    if sum_p1 > t:
        print("LOAD 1")
        return
    
    # Rule 2 requires all S2,k to be disjoint, total size sum(p2) <= T
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
