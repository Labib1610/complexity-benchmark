import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check for Job-specific contradiction
    # Rule 1: 1 <= x_k, z_k <= T + 1
    # Rule 3: x_k < y_k < z_k
    # p1 = y_k - x_k, p2 = z_k - y_k
    # z_k - x_k = p1 + p2
    # Since 1 <= x_k and z_k <= T + 1, the max range is (T + 1) - 1 = T
    # So if p1 + p2 > T, it's impossible.
    for i in range(j):
        p1, p2 = jobs[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for Collection-specific contradiction
    # Collection 1: sum of p1_k <= T
    # Collection 2: sum of p2_k <= T
    sum_p1 = sum(job[0] for job in jobs)
    sum_p2 = sum(job[1] for job in jobs)
    
    if sum_p1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
