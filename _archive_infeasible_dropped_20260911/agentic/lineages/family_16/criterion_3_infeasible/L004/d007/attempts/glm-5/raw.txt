import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    
    p1_list = []
    p2_list = []
    
    job_contradiction_idx = -1
    
    for i in range(1, j + 1):
        p1 = int(next(it))
        p2 = int(next(it))
        p1_list.append(p1)
        p2_list.append(p2)
        
        # Check Rule 1 and Rule 3 for this job
        # x_k >= 1, z_k <= T + 1
        # x_k < y_k < z_k
        # y_k - x_k = p1, z_k - y_k = p2
        # z_k - x_k = p1 + p2
        # We need x_k >= 1 and z_k <= T + 1
        # So x_k >= 1 and x_k + p1 + p2 <= T + 1
        # Thus 1 + p1 + p2 <= T + 1 => p1 + p2 <= T
        if p1 + p2 > T:
            job_contradiction_idx = i
    
    if job_contradiction_idx != -1:
        print(f"JOB {job_contradiction_idx}")
        return
    
    # Check collection-specific contradiction
    # Collection 1: sum of p1 must be <= T
    total_p1 = sum(p1_list)
    if total_p1 > T:
        print("LOAD 1")
        return
    
    # Collection 2: sum of p2 must be <= T
    total_p2 = sum(p2_list)
    if total_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
