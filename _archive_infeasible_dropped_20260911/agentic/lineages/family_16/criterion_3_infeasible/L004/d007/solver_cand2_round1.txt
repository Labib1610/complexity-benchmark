import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check for JOB i contradiction
    # Rule 1: deg(A_k) = a_k + p1_k - 1 < T
    # Rule 1: deg(B_k) = b_k + p2_k - 1 < T
    # Rule 3: deg(A_k) < b_k
    # So: a_k + p1_k - 1 < b_k
    # Combined: a_k + p1_k - 1 < b_k <= T - p2_k
    # Impossibility if: p1_k + p2_k - 1 >= T
    for i in range(j):
        p1, p2 = jobs[i]
        if p1 + p2 - 1 >= t:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c contradiction
    # Rule 2: Sum of p_c,k must be <= T
    sum_p1 = sum(job[0] for job in jobs)
    if sum_p1 >= t:
        print("LOAD 1")
        return
        
    sum_p2 = sum(job[1] for job in jobs)
    if sum_p2 >= t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
