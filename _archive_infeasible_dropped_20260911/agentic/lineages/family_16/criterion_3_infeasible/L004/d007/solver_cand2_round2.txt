import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    # W = 2 * T, so the universe size is T.
    # Rule 1: S_{1,k} and S_{2,k} are subsets of {2, 4, ..., 2T}.
    # There are T such numbers.
    
    # Job-specific contradiction:
    # For a job k, S_{1,k} has size p1, S_{2,k} has size p2.
    # Rule 3: max(S_{1,k}) < min(S_{2,k}).
    # To minimize the space, S_{1,k} = {2, 4, ..., 2*p1},
    # S_{2,k} = {2*(p1+1), ..., 2*(p1+p2)}.
    # Total space needed = p1 + p2.
    # If p1 + p2 > T, then JOB k is impossible.
    
    # Collection-specific contradiction:
    # For collection c, sum of sizes of all S_{c,k} must be <= T.
    # If sum(p_{c,k}) > T, then LOAD c is impossible.
    
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        # Check Job-specific
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Check Collection-specific
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
