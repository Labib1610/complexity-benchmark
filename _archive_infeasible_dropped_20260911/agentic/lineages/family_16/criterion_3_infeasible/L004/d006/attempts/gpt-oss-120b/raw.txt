```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))

    p1_sum = 0
    p2_sum = 0

    # First, look for a job-specific contradiction
    for idx in range(1, j + 1):
        p1 = int(next(it))
        p2 = int(next(it))
        if p1 + p2 > T:
            sys.stdout.write(f"JOB {idx}")
            return
        p1_sum += p1
        p2_sum += p2

    # No job contradiction, check collection loads
    if p1_sum > T:
        sys.stdout.write("LOAD 1")
    else:
        # By problem guarantee, this must be true if previous not
        sys.stdout.write("LOAD 2")

if __name__ == "__main__":
    solve()
```
