import sys

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    T = int(data[1])
    
    jobs = []
    idx = 2
    for i in range(1, j + 1):
        p1 = int(data[idx]); p2 = int(data[idx + 1])
        idx +=  ​2
        jobs.append((p1, p2, i))
    
    # Job-specific contradiction:
    # Need p1 + p2 > T, because we need p1 consecutive ints all < p2 consecutive ints.
    # Minimal span is p1 + p2.

    for p1, p2, i in jobs:
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
    
    # Collection-specific contradiction:
    # For collection c, need sum of p_{c,k} > T.

    sum1 = 0
    sum2 = 0
    for p1, p2, _ in jobs:
        sum1 += p1
        sum2 += p2
    
    if sum1 > T:
        print("LOAD 1")
        return
    if sum2 > T:
        print("LOAD 2")
        return
    
    # The problem guarantees at least one certificate exists.
 
if __name__ == "__main__":
    main()
