```python
import sys

def solve():
    # Read N and M
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    
    A = []
    B = []
    idx = 2
    for i in range(N):
        A.append(int(input_data[idx]))
        B.append(int(input_data[idx+1]))
        idx += 2
        
    # Check for JOB i contradiction
    # Families 1, 2, 3 for index k:
    # y1 = x1 + Ak, y2 = x2 + Bk
    # 0 <= x1, y1 <= M, 0 <= x2, y2 <= M
    # x2 >= y1
    # Substituting: x2 >= x1 + Ak
    # Since x2 >= 0 and y2 = x2 + Bk <= M, we have x2 <= M - Bk
    # So we need M - Bk >= x2 >= x1 + Ak >= Ak
    # Thus, M - Bk >= Ak  =>  Ak + Bk <= M
    # If Ak + Bk > M, then JOB i is a contradiction.
    for i in range(N):
        if A[i] + B[i] > M:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c contradiction
    # Families 1, 2, 4 for dimension c:
    # For c=1: sum(Ak) > M
    # For c=2: sum(Bk) > M
    # Note: The problem states open intervals (x, y) are disjoint.
    # The length of each interval is Ak (or Bk).
    # If we have N intervals of lengths L1, L2, ..., LN,
    # the total length required to fit them in [0, M] is sum(Li) < M.
    # If sum(Li) >= M, it's impossible to fit them.
    
    sum_A = sum(A)
    if sum_A > M:
        print("LOAD 1")
        return
        
    sum_B = sum(B)
    if sum_B > M:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
