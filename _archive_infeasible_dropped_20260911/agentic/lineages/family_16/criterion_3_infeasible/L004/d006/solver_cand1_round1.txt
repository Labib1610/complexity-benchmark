```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    
    idx = 2
    for i in range(j):
        val1 = int(input_data[idx])
        val2 = int(input_data[idx+1])
        p1.append(val1)
        p2.append(val2)
        idx += 2
        
        # Check JOB i
        # C1: Y_k - X_k >= p1_k
        # C2: X_k >= 0, X_k + p1_k <= T, Y_k >= 0, Y_k + p2_k <= T
        # Y_k <= T - p2_k
        # X_k >= 0
        # Y_k - X_k >= p1_k => Y_k >= X_k + p1_k >= p1_k
        # So we need T - p2_k >= p1_k => p1_k + p2_k > T
        if val1 + val2 > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD 1 (X-Separation)
    # C3: |2Xa - 2Xb + p1a - p1b| >= p1a + p1b
    # This is equivalent to:
    # 2Xa - 2Xb + p1a - p1b >= p1a + p1b  => 2Xa - 2Xb >= 2p1b => Xa - Xb >= p1b
    # OR 2Xa - 2Xb + p1a - p1b <= -(p1a + p1b) => 2Xa - 2Xb <= -2p1a => Xb - Xa >= p1a
    # This means for any two, either Xa >= Xb + p1b or Xb >= Xa + p1a.
    # This is the classic interval scheduling problem: can we pack intervals [X_k, X_k + p1_k]
    # into [0, T] such that they are disjoint?
    # Yes, if sum(p1_k) <= T.
    if sum(p1) > T:
        print("LOAD 1")
        return
    
    # Check LOAD 2 (Y-Separation)
    if sum(p2) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
