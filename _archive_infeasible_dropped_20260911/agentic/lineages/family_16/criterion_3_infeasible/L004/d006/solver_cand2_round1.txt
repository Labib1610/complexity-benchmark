```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_sum = 0
    p2_sum = 0
    
    # We need to check for JOB i first.
    # A JOB i exists if p1_i + p2_i > T.
    # If multiple exist, the problem implies we should find one.
    # The problem states "Exactly one of the following will be true",
    # but if multiple JOBs exist, any one of them is valid.
    
    idx = 2
    job_found = -1
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        p1_sum += p1
        p2_sum += p2
        
        if job_found == -1 and (p1 + p2 > t):
            job_found = i
            
    if job_found != -1:
        print(f"JOB {job_found}")
    else:
        # If no JOB i, then it must be a LOAD bottleneck.
        if p1_sum > t:
            print("LOAD 1")
        else:
            print("LOAD 2")

if __name__ == '__main__':
    solve()
```
