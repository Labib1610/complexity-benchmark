```python
import sys

def solve():
    # Read j and T
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        j, t = map(int, line1)
    except ValueError:
        return

    # Rule 3 for a single job i:
    # Load 1 has p1 elements, Load 2 has p2 elements.
    # All elements in Load 1 < all elements in Load 2.
    # Union size = p1 + p2.
    # Span M - m + 1 = p1 + p2.
    # Since elements are in [1, T], the minimum possible span is p1 + p2.
    # If p1 + p2 > T, then Rule 1 and Rule 3 cannot be satisfied.
    
    # Rule 2 for a load type c:
    # Total elements required across all jobs = sum(p_c for all jobs).
    # If sum(p_c) > T, then Rule 1 and Rule 2 cannot be satisfied.

    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        try:
            p1, p2 = map(int, sys.stdin.readline().split())
        except ValueError:
            break
            
        # Check Job-specific contradiction
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Check Load-specific contradiction
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
