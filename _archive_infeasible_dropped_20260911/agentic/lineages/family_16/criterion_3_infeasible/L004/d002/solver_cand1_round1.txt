```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i certificate
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        jobs.append((p1, p2))
        
    # Check LOAD 1 certificate
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2 certificate
    if sum_p2 > t:
        print("LOAD 2")
        return
        
    # Johnson's Rule for two-machine scheduling
    # Split jobs into two sets: p1 < p2 and p1 >= p2
    set1 = []
    set2 = []
    for i in range(j):
        if jobs[i][0] < jobs[i][1]:
            set1.append(jobs[i])
        else:
            set2.append(jobs[i])
            
    # Sort set1 by p1 ascending, set2 by p2 descending
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    schedule = set1 + set2
    
    # Calculate completion time
    m1_time = 0
    m2_time = 0
    for p1, p2 in schedule:
        m1_time += p1
        m2_time = max(m2_time, m1_time) + p2
        
    if m2_time > t:
        # If Johnson's rule fails, it must be due to LOAD 1 or 2
        # because Johnson's rule is optimal for makespan.
        # The problem implies one of the certificates must exist.
        if sum_p1 > t:
            print("LOAD 1")
        else:
            print("LOAD 2")

solve()
```
