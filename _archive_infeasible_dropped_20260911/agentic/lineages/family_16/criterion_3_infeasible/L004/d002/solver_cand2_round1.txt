import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        sum_p1 += p1
        sum_p2 += p2
        idx += 2
        
    # Johnson's Rule for 2-machine flow shop scheduling
    # Sort jobs: those with p1 < p2 first (ascending p1), 
    # then those with p1 >= p2 (descending p2)
    group1 = []
    group2 = []
    for i in range(j):
        if jobs[i][0] < jobs[i][1]:
            group1.append((jobs[i][0], jobs[i][1], i + 1))
        else:
            group2.append((jobs[i][0], jobs[i][1], i + 1))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate completion time
    m1_time = 0
    m2_time = 0
    possible = True
    for p1, p2, _ in order:
        m1_time += p1
        m2_time = max(m1_time, m2_time) + p2
        if m2_time > t:
            possible = False
            break
            
    if possible:
        print("POSSIBLE")
    else:
        # Calculate relaxations
        max_val = -1
        best_cert = ""
        
        # JOB i: p1 + p2
        for i in range(j):
            val = jobs[i][0] + jobs[i][1]
            if val > max_val:
                max_val = val
                best_cert = f"JOB {i + 1}"
        
        # LOAD 1: sum(p1) + min(p2)
        min_p2 = min(job[1] for job in jobs)
        val_load1 = sum_p1 + min_p2
        if val_load1 > max_val:
            max_val = val_load1
            best_cert = "LOAD 1"
            
        # LOAD 2: sum(p2) + min(p1)
        min_p1 = min(job[0] for job in jobs)
        val_load2 = sum_p2 + min_p1
        if val_load2 > max_val:
            max_val = val_load2
            best_cert = "LOAD 2"
            
        print(best_cert)

if __name__ == '__main__':
    solve()
