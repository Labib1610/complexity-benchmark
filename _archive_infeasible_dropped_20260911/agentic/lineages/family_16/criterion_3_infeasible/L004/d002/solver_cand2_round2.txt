```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum1 = 0
    sum2 = 0
    
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        # Check if the job itself exceeds the deadline
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
        
        sum1 += p1
        sum2 += p2
        
    # Check if total load on machine 1 exceeds deadline
    if sum1 > t:
        print("LOAD 1")
        return
    
    # Check if total load on machine 2 exceeds deadline
    if sum2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
