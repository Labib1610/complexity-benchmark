import sys

def solve():
    # Read j and M
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    M = int(input_data[1])
    
    U = []
    V = []
    idx = 2
    for i in range(j):
        u_val = int(input_data[idx])
        v_val = int(input_data[idx+1])
        U.append(u_val)
        V.append(v_val)
        idx += 2
        
    # Check for Local Contradiction (JOB i)
    # Constraint: alpha_i >= 0, alpha_i + U_i <= beta_i, beta_i + V_i <= M
    # Implies: 0 + U_i + V_i <= M
    for i in range(j):
        if U[i] + V[i] > M:
            print(f"JOB {i + 1}")
            return
            
    # Check for Global Contradiction on alpha (LOAD 1)
    # Constraints: alpha_k >= 0, alpha_k <= M - U_k, |alpha_x - alpha_y| >= U_y (or U_x)
    # This is equivalent to packing j intervals of length U_k into [0, M].
    # Actually, the condition |alpha_x - alpha_y| >= max(U_x, U_y) is equivalent to
    # saying that if we sort alpha_k such that alpha_1 < alpha_2 < ... < alpha_j,
    # then alpha_{k+1} >= alpha_k + U_k.
    # The tightest packing is alpha_1 = 0, alpha_2 = U_1, alpha_3 = U_1 + U_2, ...
    # alpha_j = sum_{k=1}^{j-1} U_k.
    # Also alpha_k <= M - U_k.
    # The condition is: sum_{k=1}^{j-1} U_k + U_j > M? No, that's not quite right.
    # The condition for existence is: sort U such that U_1 <= U_2 <= ... <= U_j.
    # The minimal range required is sum_{k=1}^{j-1} U_k.
    # The available range for each alpha_k is [0, M - U_k].
    # The condition for impossibility is: sum_{k=1}^{j-1} sorted_U[k] > M - sorted_U[0]
    # Wait, the condition is: there exist alpha_1 < ... < alpha_j in [0, M-U_k]
    # such that alpha_{k+1} - alpha_k >= U_k.
    # This is possible iff sum_{k=1}^{j-1} U_{(k)} <= M - U_{(j)} where U_{(k)} are sorted.
    
    def check_global(arr):
        s = sorted(arr)
        # Minimal span: sum of j-1 smallest U's.
        # The largest U must be at the end to minimize the constraint M - U_j.
        # Actually, the condition is: sum_{k=0}^{j-2} s[k] > M - s[j-1]
        return sum(s[:-1]) > M - s[-1]

    if check_global(U):
        print("LOAD 1")
        return
        
    # Check for Global Contradiction on beta (LOAD 2)
    # Constraints: beta_k >= V_k, beta_k <= M, |beta_x - beta_y| >= V_y
    # Similar logic: sum_{k=0}^{j-2} s[k] > M - s[j-1]
    if check_global(V):
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
