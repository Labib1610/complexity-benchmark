```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p.append((p1, p2))
        idx += 2
        
    # Check Type A: JOB i
    # Constraints: S1 >= 0, S2 >= 0, S2 - S1 >= p1, S1 + p1 <= T, S2 + p2 <= T
    # S1 >= 0, S1 <= T - p1
    # S2 >= 0, S2 <= T - p2
    # S2 >= S1 + p1
    # Contradiction if: (T - p2) < (0 + p1) => T < p1 + p2
    for i in range(j):
        if p[i][0] + p[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check Type B: LOAD c
    # Resource c: S_{c,k} >= 0, S_{c,k} + p_{c,k} <= T
    # Exclusion: |S_{c,x} - S_{c,y}| >= p_{c,x} or p_{c,y}
    # This is equivalent to saying the sum of processing times must be <= T
    # if we sort them. Actually, for a single resource, the condition for 
    # feasibility is sum(p_{c,k}) <= T.
    
    sum_p1 = sum(item[0] for item in p)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    sum_p2 = sum(item[1] for item in p)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
