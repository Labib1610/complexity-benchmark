```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    orders = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        orders.append((p1, p2, i + 1))
        idx += 2
        
    # Check Relaxation A: JOB i
    for p1, p2, i in orders:
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
            
    # Check Relaxation B: LOAD 1
    sum_p1 = sum(order[0] for order in orders)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check Relaxation C: LOAD 2
    sum_p2 = sum(order[1] for order in orders)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
