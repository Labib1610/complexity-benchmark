```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for i in range(j):
        p1_val = int(input_data[idx])
        p2_val = int(input_data[idx+1])
        p1.append(p1_val)
        p2.append(p2_val)
        idx += 2
        
    # Check Type 1: JOB i
    for i in range(j):
        if p1[i] + p2[i] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check Type 2: LOAD 1
    # Sum of p1_i must be <= T
    if sum(p1) > T:
        print("LOAD 1")
        return
        
    # Check Type 3: LOAD 2
    # Sum of p2_i must be <= T
    if sum(p2) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
