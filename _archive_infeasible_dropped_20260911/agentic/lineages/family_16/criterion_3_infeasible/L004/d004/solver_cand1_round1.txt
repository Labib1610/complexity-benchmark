```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    orders = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        orders.append((p1, p2, i + 1))
        idx += 2
        
    # Check for JOB i (Isolation witness)
    # Condition: p1 + p2 > T
    for p1, p2, i in orders:
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
            
    # Check for LOAD c (Relaxation witness)
    # Condition: sum(pi) > T
    sum1 = 0
    sum2 = 0
    for p1, p2, i in orders:
        sum1 += p1
        sum2 += p2
        
    if sum1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
