```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    orders = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        orders.append((p1, p2, i + 1))
        idx += 2
        
    # Check Case 1: JOB i
    for i in range(j):
        p1, p2, original_idx = orders[i]
        # C1: S_i,1 >= 0
        # C2: S_i,2 >= S_i,1 + p1 => S_i,2 >= p1
        # C3: S_i,2 + p2 <= T => S_i,2 <= T - p2
        # Contradiction if p1 > T - p2 => p1 + p2 > T
        if p1 + p2 > T:
            print(f"JOB {original_idx}")
            return
            
    # Check Case 2: LOAD 1
    # Sum of p1 must be <= T
    sum_p1 = sum(order[0] for order in orders)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check Case 3: LOAD 2
    # Sum of p2 must be <= T
    sum_p2 = sum(order[1] for order in orders)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
