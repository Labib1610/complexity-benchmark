```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check if any single job exceeds the deadline
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        jobs.append((p1, p2))
    
    # Check if machine 1 is overloaded
    sum1 = sum(job[0] for job in jobs)
    if sum1 > t:
        print("LOAD 1")
        return
    
    # Check if machine 2 is overloaded
    sum2 = sum(job[1] for job in jobs)
    if sum2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
