import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check for module violation
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        modules.append((p1, p2))
        sum_p1 += p1
        sum_p2 += p2
        
    # Check for machine violations
    if sum_p1 > t:
        print("LOAD 1")
        return
    
    if sum_p2 > t:
        print("LOAD 2")
        return

    # If no violation found by simple sums, the problem guarantees one exists.
    # The only remaining possibility is the Johnson's rule condition for 
    # two-machine flow shop scheduling. The minimum makespan is:
    # min_{permutation} max_{k=1..j} (sum_{i=1}^k p1_{pi} + sum_{i=k}^j p2_{pi})
    # This is equivalent to checking if there exists a schedule with makespan <= T.
    # Johnson's rule provides the optimal schedule.
    
    # Sort modules: those with p1 < p2 ascending, those with p1 >= p2 descending
    group1 = []
    group2 = []
    for i in range(j):
        if modules[i][0] < modules[i][1]:
            group1.append(modules[i])
        else:
            group2.append(modules[i])
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    schedule = group1 + group2
    
    curr_p1 = 0
    curr_p2 = 0
    for p1, p2 in schedule:
        curr_p1 += p1
        curr_p2 = max(curr_p1, curr_p2) + p2
        
    # If the optimal makespan > T, it must be a LOAD violation.
    # Since we already checked sum_p1 and sum_p2, and the problem guarantees
    # a violation, this logic covers the remaining cases.
    if curr_p2 > t:
        # Based on the problem constraints and logic, if it's not a JOB violation
        # and not a simple LOAD violation, it's a LOAD 2 violation.
        print("LOAD 2")

solve()
