```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    components = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        components.append((p1, p2))
        idx += 2
        
    # Check JOB i certificates
    # A component i is impossible if p1 + p2 > T
    # We need the largest index i such that p1 + p2 > T
    job_certificate = -1
    for i in range(j - 1, -1, -1):
        if components[i][0] + components[i][1] > t:
            job_certificate = i + 1
            break
            
    if job_certificate != -1:
        print(f"JOB {job_certificate}")
        return
        
    # Check LOAD c certificates
    # Machine 1 (build) is saturated if sum(p1) > T
    # Machine 2 (test) is saturated if sum(p2) > T
    sum_p1 = sum(c[0] for c in components)
    sum_p2 = sum(c[1] for c in components)
    
    if sum_p2 > t:
        print("LOAD 2")
    else:
        print("LOAD 1")

if __name__ == '__main__':
    solve()
```
