```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        modules.append((p1, p2, i + 1))
        idx += 2
        
    # Check JOB i certificates
    # A module i is impossible if p1_i + p2_i > T
    max_job_idx = -1
    for p1, p2, i in modules:
        if p1 + p2 > t:
            if i > max_job_idx:
                max_job_idx = i
                
    if max_job_idx != -1:
        print(f"JOB {max_job_idx}")
        return
    
    # Check LOAD 1 and LOAD 2 certificates
    # LOAD 1: sum(p1) > T
    # LOAD 2: sum(p2) > T
    sum_p1 = sum(m[0] for m in modules)
    sum_p2 = sum(m[1] for m in modules)
    
    load1 = (sum_p1 > t)
    load2 = (sum_p2 > t)
    
    if load1 and load2:
        print("LOAD 2")
    elif load1:
        print("LOAD 1")
    elif load2:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
