```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    j = int(data[idx]); idx += 1
    T = int(data[idx]); idx += 1
    
    job_idx = -1
    sum1 = 0
    sum2 = 0
    load1_fail = False
    load2_fail = False
    
    for i in range(1, j + 1):
        p1 = int(data[idx]); idx += 1
        p2 = int(data[idx]); idx += 1
        
        if job_idx == -1 and p1 + p2 > T:
            job_idx = i
        
        sum1 += p1
        sum2 += p2
        
        if not load1_fail and sum1 > T:
            load1_fail = True
        if not load2_fail and sum2 > T:
            load2_fail = True
    
    if job_idx != -1:
        print(f"JOB {job_idx}")
    elif load1_fail:
        print("LOAD 1")
    else:
        print("LOAD 2")

solve()
```
