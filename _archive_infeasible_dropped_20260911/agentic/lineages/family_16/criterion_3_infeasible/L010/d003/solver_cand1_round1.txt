```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        modules.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # A module i is a bottleneck if p1_i + p2_i > T
    # We need the highest index i such that p1_i + p2_i > T
    job_bottleneck = -1
    for i in range(j - 1, -1, -1):
        if modules[i][0] + modules[i][1] > t:
            job_bottleneck = i + 1
            break
            
    if job_bottleneck != -1:
        print(f"JOB {job_bottleneck}")
        return
        
    # Check LOAD 1
    # Sum of p1_i > T
    sum_p1 = sum(m[0] for m in modules)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sum of p2_i > T
    sum_p2 = sum(m[1] for m in modules)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
