```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    components = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        components.append((p1, p2))
        idx += 2
        
    # Check for JOB i bottlenecks
    # We need the largest index i such that p1 + p2 > T
    max_job_idx = -1
    for i in range(j):
        if components[i][0] + components[i][1] > t:
            max_job_idx = i + 1
            
    if max_job_idx != -1:
        print(f"JOB {max_job_idx}")
        return
    
    # Check for LOAD bottlenecks
    sum1 = 0
    sum2 = 0
    for p1, p2 in components:
        sum1 += p1
        sum2 += p2
        
    if sum2 > t:
        print("LOAD 2")
    elif sum1 > t:
        print("LOAD 1")

if __name__ == '__main__':
    solve()
```
