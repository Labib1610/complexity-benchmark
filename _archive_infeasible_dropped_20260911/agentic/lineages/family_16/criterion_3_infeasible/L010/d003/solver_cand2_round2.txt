```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        modules.append((p1, p2))
        idx += 2
        
    # Check for JOB i violation
    for i in range(j):
        if modules[i][0] + modules[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD 1 violation
    total_p1 = sum(m[0] for m in modules)
    if total_p1 > t:
        print("LOAD 1")
        return
        
    # Check for LOAD 2 violation
    total_p2 = sum(m[1] for m in modules)
    if total_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
