```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    # Check for JOB i paradox
    # For a row i, we need p1_i ones in M and p2_i ones in N.
    # Let the columns for M be {m_1, ..., m_{p1_i}} and for N be {n_1, ..., n_{p2_i}}.
    # Constraint: max(m) < min(n).
    # To minimize the range, we pick M columns as 1, 2, ..., p1_i
    # and N columns as T-p2_i+1, ..., T.
    # The condition is p1_i < T - p2_i + 1, or p1_i + p2_i <= T.
    # If p1_i + p2_i > T, then JOB i is impossible.
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        p1_list.append(p1)
        p2_list.append(p2)
        
        sum_p1 += p1
        sum_p2 += p2
        
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
            
    # If no JOB i paradox, check LOAD 1 or LOAD 2.
    # M-Column Sparsity: sum(p1_i) <= T
    # N-Column Sparsity: sum(p2_i) <= T
    
    if sum_p1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
