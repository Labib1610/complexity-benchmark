import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    sum_p1 = 0
    sum_p2 = 0
    
    idx = 2
    max_sep = -1
    max_sep_idx = -1
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Separation Size for entity i is p1_i + p2_i
        sep_size = p1 + p2
        if sep_size > max_sep:
            max_sep = sep_size
            max_sep_idx = i
        
        sum_p1 += p1
        sum_p2 += p2
        
    # X-Domain Size is sum of all p1_i
    # Y-Domain Size is sum of all p2_i
    x_domain_size = sum_p1
    y_domain_size = sum_p2
    
    # Find the strictly largest among all
    # Candidates: max_sep (at max_sep_idx), x_domain_size, y_domain_size
    
    # We are guaranteed a unique strict maximum.
    # Check if max_sep is the unique maximum
    is_max_sep = True
    if max_sep <= x_domain_size or max_sep <= y_domain_size:
        is_max_sep = False
        
    if is_max_sep:
        print(f"JOB {max_sep_idx}")
    else:
        # Check if x_domain_size is the unique maximum
        if x_domain_size > max_sep and x_domain_size > y_domain_size:
            print("LOAD 1")
        else:
            # Must be y_domain_size
            print("LOAD 2")

if __name__ == '__main__':
    solve()
