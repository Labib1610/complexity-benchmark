import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    rows = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        rows.append((p1, p2))
        idx += 2
        
    # 1. Check Local Projection Failure (JOB i)
    # The condition A_{i,*} M B_{i,*}^T = p1_i * p2_i means that all ones in A
    # must be at indices strictly smaller than all ones in B.
    # Let the indices of ones in A be S_A and in B be S_B.
    # We need max(S_A) < min(S_B).
    # This requires that the total number of positions used is at least p1_i + p2_i.
    # Specifically, if we place p1_i ones at 1, ..., p1_i and p2_i ones at 
    # p1_i + 1, ..., p1_i + p2_i, we satisfy the condition.
    # This is possible if and only if p1_i + p2_i <= T.
    for i in range(j):
        if rows[i][0] + rows[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # 2. Check Gramian Exhaustion (LOAD 1 or 2)
    # For A A^T to be diagonal, rows of A must be orthogonal.
    # This means for any two rows i and k, they cannot share a 1 in the same column.
    # Total number of ones in A is sum(p1_i).
    # Since each column can hold at most one '1' across all rows to maintain orthogonality,
    # we must have sum(p1_i) <= T.
    # Similarly for B, sum(p2_i) <= T.
    
    sum_p1 = sum(r[0] for r in rows)
    sum_p2 = sum(r[1] for r in rows)
    
    if sum_p1 > T:
        print("LOAD 1")
    else:
        # If sum_p1 <= T, then by process of elimination, it must be LOAD 2
        # (The problem guarantees exactly one obstruction).
        print("LOAD 2")

if __name__ == '__main__':
    solve()
