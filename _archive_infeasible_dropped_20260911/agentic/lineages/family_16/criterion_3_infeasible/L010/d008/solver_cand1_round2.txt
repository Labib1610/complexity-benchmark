```python
import sys

def solve():
    # Read N and M
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    
    processes = []
    idx = 2
    for i in range(N):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        processes.append((a, b))
        idx += 2
        
    # Check Ordering Failure (JOB i)
    # For process i, we need R_i of size a_i and W_i of size b_i
    # such that max(R_i) < min(W_i).
    # This is possible if and only if a_i + b_i <= M.
    # If a_i + b_i > M, then no such sets exist.
    for i in range(N):
        a, b = processes[i]
        if a + b > M:
            print(f"JOB {i + 1}")
            return
            
    # Check Read-Set Exhaustion (LOAD 1)
    # Total size of all read-sets must be <= M.
    # The problem states Read-Set Independence requires the union size to be sum(a_i).
    # If sum(a_i) > M, it's impossible to pick distinct addresses.
    sum_a = sum(p[0] for p in processes)
    if sum_a > M:
        print("LOAD 1")
        return
        
    # Check Write-Set Exhaustion (LOAD 2)
    # Similarly, if sum(b_i) > M, it's impossible to pick distinct addresses.
    sum_b = sum(p[1] for p in processes)
    if sum_b > M:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
