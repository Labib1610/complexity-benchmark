import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    flows = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    # Check for JOB i (Causal Violation)
    # A flow i is impossible if p1_i + p2_i - 1 > T
    # This is because to avoid the firewall, all egress must happen after all ingress.
    # If we have p1 ingress and p2 egress, we need at least p1 + p2 slots.
    # Wait, the condition is: if we schedule an ingress after an egress, it's blocked.
    # So all egress must happen at or after the last ingress? No, the code says:
    # if egress_seen and ingress_happens: BLOCKED.
    # This means all ingress must happen BEFORE any egress.
    # So we need p1 slots for ingress, then p2 slots for egress.
    # Total slots needed = p1 + p2.
    # If p1 + p2 > T, it's impossible for that flow.
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Check for LOAD 1 (Ingress Saturation)
    if sum_p1 > T:
        print("LOAD 1")
        return
    
    # Check for LOAD 2 (Egress Saturation)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
