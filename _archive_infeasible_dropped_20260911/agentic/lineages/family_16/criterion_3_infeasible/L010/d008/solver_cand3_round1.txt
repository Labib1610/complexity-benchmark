```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    traders = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        traders.append((p1, p2))
        idx += 2
        
    # Level 1: Individual Insolvency
    # A trader is insolvent if p1 + p2 > T.
    # Note: The problem states "even if they were the only trader".
    # If they are the only trader, they need p1 ticks for Audit and p2 for Recon.
    # Since Audit must be strictly earlier than Recon, they need at least p1 + p2 ticks.
    for i in range(j):
        p1, p2 = traders[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Level 2: Audit Network Exhaustion
    # Global Audit requirements > T
    total_p1 = sum(trader[0] for trader in traders)
    if total_p1 > t:
        print("LOAD 1")
        return
        
    # Level 3: Reconciliation Network Exhaustion
    # Global Reconciliation requirements > T
    total_p2 = sum(trader[1] for trader in traders)
    if total_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
