```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    max_val = -1
    certificate = ""
    
    sum_p1 = 0
    sum_p2 = 0
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        sum_p1 += p1
        sum_p2 += p2
        
        job_time = p1 + p2
        if job_time > t:
            if job_time > max_val:
                max_val = job_time
                certificate = f"JOB {i}"
                
    if sum_p1 > t:
        if sum_p1 > max_val:
            max_val = sum_p1
            certificate = "LOAD 1"
            
    if sum_p2 > t:
        if sum_p2 > max_val:
            max_val = sum_p2
            certificate = "LOAD 2"
            
    print(certificate)

if __name__ == '__main__':
    solve()
```
