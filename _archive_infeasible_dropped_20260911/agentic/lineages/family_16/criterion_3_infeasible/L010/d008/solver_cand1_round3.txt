import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    sum_p1 = 0
    sum_p2 = 0
    
    max_job_deficit = -float('inf')
    max_job_idx = -1
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        sum_p1 += p1
        sum_p2 += p2
        
        job_deficit = (p1 + p2) - T
        if job_deficit > max_job_deficit:
            max_job_deficit = job_deficit
            max_job_idx = i
            
    load1_deficit = sum_p1 - T
    load2_deficit = sum_p2 - T
    
    # Priorities: LOAD 1, LOAD 2, JOB i
    # We need the strictly maximum deficit.
    # If there is a tie, use the priority order.
    
    best_deficit = -float('inf')
    best_cert = ""
    
    # Check LOAD 1
    if load1_deficit > best_deficit:
        best_deficit = load1_deficit
        best_cert = "LOAD 1"
        
    # Check LOAD 2
    if load2_deficit > best_deficit:
        best_deficit = load2_deficit
        best_cert = "LOAD 2"
    elif load2_deficit == best_deficit:
        # If tie, LOAD 1 > LOAD 2, so keep LOAD 1
        pass
        
    # Check JOB i
    if max_job_deficit > best_deficit:
        best_deficit = max_job_deficit
        best_cert = f"JOB {max_job_idx}"
    elif max_job_deficit == best_deficit:
        # If tie, LOAD 1 > LOAD 2 > JOB i
        # So only update if best_cert is not LOAD 1 or LOAD 2
        if best_cert not in ["LOAD 1", "LOAD 2"]:
            # If multiple jobs tie, pick smallest index (already handled by loop)
            pass
            
    print(best_cert)

if __name__ == '__main__':
    solve()
