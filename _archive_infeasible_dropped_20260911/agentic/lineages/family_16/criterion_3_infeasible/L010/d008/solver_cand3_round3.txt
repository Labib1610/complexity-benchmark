```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    processes = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        processes.append((p1, p2))
        idx += 2
        
    # Priority 1: Input-Space Exhaustion
    # Sum of all p1 must be <= T
    total_p1 = sum(p[0] for p in processes)
    if total_p1 > t:
        print("LOAD 1")
        return
        
    # Priority 2: Output-Space Exhaustion
    # Sum of all p2 must be <= T
    total_p2 = sum(p[1] for p in processes)
    if total_p2 > t:
        print("LOAD 2")
        return
        
    # Priority 3: Causal Violation
    # For each process i, p1_i + p2_i must be <= T
    # If multiple, smallest index i
    for i in range(j):
        if processes[i][0] + processes[i][1] > t:
            print(f"JOB {i + 1}")
            return

if __name__ == '__main__':
    solve()
```
