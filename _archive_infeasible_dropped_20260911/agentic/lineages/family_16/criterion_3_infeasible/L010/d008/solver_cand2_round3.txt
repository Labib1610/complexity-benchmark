```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    molecules = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        molecules.append((p1, p2))
        idx += 2
        
    # Check Polarity Impossibility
    # For a molecule to satisfy polarity, there must exist a split point k
    # such that alpha bonds are in [1, k] and beta bonds are in [k+1, T].
    # This requires p1 <= k and p2 <= T - k.
    # Summing these: p1 + p2 <= T.
    for i in range(j):
        p1, p2 = molecules[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check Alpha Saturation
    # Total alpha bonds must be <= T
    total_p1 = sum(m[0] for m in molecules)
    if total_p1 > t:
        print("LOAD 1")
        return
        
    # Check Beta Saturation
    # Total beta bonds must be <= T
    total_p2 = sum(m[1] for m in molecules)
    if total_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
