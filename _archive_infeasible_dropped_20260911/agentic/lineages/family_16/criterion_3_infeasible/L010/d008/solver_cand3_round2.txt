import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    orders = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        orders.append((p1, p2))
        idx += 2
        
    # Check for JOB i (Flow Impossibility)
    # For a single order i, we need p1_i base stations and p2_i finish stations.
    # Let the base stations be at indices b_1, ..., b_{p1_i} and finish at f_1, ..., f_{p2_i}.
    # The condition is that all b_k < all f_m.
    # This means the largest base station index must be smaller than the smallest finish station index.
    # To maximize the number of stations, we pick the first p1_i stations for base
    # and the last p2_i stations for finish.
    # The condition is: p1_i + p2_i <= T.
    for i in range(j):
        p1, p2 = orders[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD 1 (Base Coat Bottleneck)
    # Total base stations required: sum(p1_i)
    # Must be <= T
    total_p1 = sum(order[0] for order in orders)
    if total_p1 > t:
        print("LOAD 1")
        return
        
    # Check for LOAD 2 (Finish Coat Bottleneck)
    # Total finish stations required: sum(p2_i)
    # Must be <= T
    total_p2 = sum(order[1] for order in orders)
    if total_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
