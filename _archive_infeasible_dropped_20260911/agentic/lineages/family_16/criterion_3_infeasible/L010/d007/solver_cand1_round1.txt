```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    pairs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        pairs.append((p1, p2))
        idx += 2
        
    # Check for JOB i
    # For a single pair (A_i, B_i), we need to find injective f: A_i -> {1..T}
    # and g: B_i -> {1..T} such that f(x) < g(y) for all x in A_i, y in B_i.
    # This is possible if and only if there exists a split point k in {1..T}
    # such that |A_i| <= k-1 and |B_i| <= T-k.
    # This is equivalent to |A_i| + |B_i| <= T.
    for i in range(j):
        p1, p2 = pairs[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD 1
    # Total size of union of A_i must be <= T
    sum_p1 = sum(p[0] for p in pairs)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check for LOAD 2
    # Total size of union of B_i must be <= T
    sum_p2 = sum(p[1] for p in pairs)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
