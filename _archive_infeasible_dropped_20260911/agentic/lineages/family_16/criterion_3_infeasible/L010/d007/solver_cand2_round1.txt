import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    alphas = []
    betas = []
    
    idx = 2
    for i in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        alphas.append(a)
        betas.append(b)
        
        # Check Precedence Failure:
        # In a totally ordered set of size T, we need to find A_i and B_i
        # such that for all x in A_i, y in B_i, x < y.
        # This is possible if and only if there exist at least alpha_i + beta_i
        # distinct elements in X.
        # Specifically, we can pick the smallest alpha_i elements for A_i
        # and the largest beta_i elements for B_i.
        # They are disjoint if alpha_i + beta_i <= T.
        # Wait, the condition is x < y. If we pick the smallest alpha_i elements
        # as A_i, the largest element of A_i is at position alpha_i.
        # The smallest element of B_i must be at position > alpha_i.
        # So we need alpha_i + beta_i <= T.
        if a + b > T:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check Packing Failure:
    # Sum of sizes must be <= T.
    if sum(alphas) > T:
        print("LOAD 1")
        return
    
    if sum(betas) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
