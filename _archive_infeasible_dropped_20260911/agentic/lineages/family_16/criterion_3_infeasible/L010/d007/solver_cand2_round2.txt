```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    shapes = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        shapes.append((p1, p2))
        sum_p1 += p1
        sum_p2 += p2
        idx += 2
        
    # Check Row 1 Overflow
    if sum_p1 > t:
        print("LOAD 1")
        return
    
    # Check Row 2 Overflow
    if sum_p2 > t:
        print("LOAD 2")
        return
    
    # Check Shape Bottleneck
    # For a shape to be valid, there must exist k in [0, T] such that:
    # k >= p1 and (T - k) >= p2
    # This is equivalent to: k >= p1 and k <= T - p2
    # Such a k exists if and only if p1 <= T - p2, or p1 + p2 <= T
    for i in range(j):
        p1, p2 = shapes[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return

if __name__ == '__main__':
    solve()
```
