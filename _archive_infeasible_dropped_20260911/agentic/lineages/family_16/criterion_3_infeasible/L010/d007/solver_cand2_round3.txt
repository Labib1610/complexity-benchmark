```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    # Check for Separation Failure (JOB i)
    # For each entity i, we need to pick X_i and Y_i such that
    # max(X_i) < min(Y_i).
    # This is possible if and only if p1_i + p2_i <= T.
    # If p1_i + p2_i > T, then JOB i is the certificate.
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
        
        p1_list.append(p1)
        p2_list.append(p2)
        sum_p1 += p1
        sum_p2 += p2
        
    # Check for X-Domain Exhaustion (LOAD 1)
    # X-Orthogonality requires sum(p1_i) <= T.
    if sum_p1 > T:
        print("LOAD 1")
        return
    
    # Check for Y-Domain Exhaustion (LOAD 2)
    # Y-Orthogonality requires sum(p2_i) <= T.
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
