```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    e1 = []
    e2 = []
    
    idx = 2
    sum_e1 = 0
    sum_e2 = 0
    
    for i in range(j):
        val1 = int(input_data[idx])
        val2 = int(input_data[idx+1])
        e1.append(val1)
        e2.append(val2)
        sum_e1 += val1
        sum_e2 += val2
        idx += 2
        
        # Check Component Delay
        if val1 + val2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check Machine Saturation
    if sum_e1 > t:
        print("LOAD 1")
        return
    
    if sum_e2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
