```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_sum = 0
    p2_sum = 0
    
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        # Check for Component Delay
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
        
        p1_sum += p1
        p2_sum += p2
        
    # Check for Machine Saturation
    if p1_sum > t:
        print("LOAD 1")
    else:
        # If LOAD 1 is not the bottleneck, and no JOB i is,
        # then LOAD 2 must be the bottleneck.
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
