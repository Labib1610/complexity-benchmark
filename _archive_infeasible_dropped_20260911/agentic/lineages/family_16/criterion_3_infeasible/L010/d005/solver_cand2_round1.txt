```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    
    idx = 2
    for i in range(j):
        val1 = int(input_data[idx])
        val2 = int(input_data[idx+1])
        p1.append(val1)
        p2.append(val2)
        idx += 2
        
        # Check JOB i: x_{i,2} - x_{i,1} = p_{i,1}, y_{i,2} - y_{i,1} = p_{i,2}, y_{i,1} >= x_{i,2}
        # Combined: y_{i,2} = y_{i,1} + p_{i,2} >= x_{i,2} + p_{i,2} = x_{i,1} + p_{i,1} + p_{i,2}
        # Constraints: x_{i,1} >= 0, y_{i,2} <= T
        # So x_{i,1} + p_{i,1} + p_{i,2} <= y_{i,2} <= T
        # Since x_{i,1} >= 0, we need p_{i,1} + p_{i,2} <= T
        if val1 + val2 > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD 1: sum of p_{i,1} > T
    if sum(p1) > T:
        print("LOAD 1")
        return
    
    # Check LOAD 2: sum of p_{i,2} > T
    if sum(p2) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
