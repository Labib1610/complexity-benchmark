```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    exchanges = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        exchanges.append((p1, p2))
        idx += 2
        
    # 1. Check Causality Failure (JOB i)
    # For a single exchange i, we need to find p1_i elements for Q_i and p2_i elements for R_i
    # such that max(Im(Q_i)) < min(Im(R_i)).
    # This is possible if and only if p1_i + p2_i <= T.
    # If p1_i + p2_i > T, then Causality is impossible.
    for i in range(j):
        if exchanges[i][0] + exchanges[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # 2. Check Query Independence (LOAD 1)
    # We need to fit j query functions with sizes p1_1, ..., p1_j into T.
    # This is possible if and only if sum(p1_i) <= T.
    sum_p1 = sum(ex[0] for ex in exchanges)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # 3. Check Reply Independence (LOAD 2)
    # We need to fit j reply functions with sizes p2_1, ..., p2_j into T.
    # This is possible if and only if sum(p2_i) <= T.
    sum_p2 = sum(ex[1] for ex in exchanges)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
