```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    
    exchanges = []
    index = 2
    for _ in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        exchanges.append((p1, p2))
        index += 2
    
    # Check for causality failure
    for i in range(j):
        p1, p2 = exchanges[i]
        # For causality: max(Q_i) < min(R_i)
        # The best case is Q_i uses the smallest p1 numbers, R_i uses the largest p2 numbers
        # So max(Q_i) = p1, min(R_i) = T - p2 + 1
        # We need p1 < T - p2 + 1 => p1 + p2 <= T
        if p1 + p2 > T:
            print(f"JOB {i+1}")
            return
    
    # Check for query collision
    total_query = sum(p1 for p1, p2 in exchanges)
    if total_query > T:
        print("LOAD 1")
        return
    
    # Check for reply collision
    total_reply = sum(p2 for p1, p2 in exchanges)
    if total_reply > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    main()
```
