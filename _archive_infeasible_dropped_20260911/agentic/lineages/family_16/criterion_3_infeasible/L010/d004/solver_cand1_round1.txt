import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    designs = []
    idx = 2
    for i in range(j):
        w = int(input_data[idx])
        v = int(input_data[idx+1])
        designs.append((w, v))
        idx += 2
        
    # Check JOB i
    # Condition: w_i/2 <= c_i <= T - w_i/2, v_i/2 <= d_i <= T - v_i/2, c_i + w_i/2 <= d_i - v_i/2
    # Min c_i = w_i/2, Max d_i = T - v_i/2
    # Inequality: w_i/2 + w_i/2 <= T - v_i/2 - v_i/2 => w_i + v_i <= T
    # Wait, the condition is c_i + w_i/2 <= d_i - v_i/2.
    # Min c_i = w_i/2, Min d_i = v_i/2.
    # The constraint is c_i <= d_i - (w_i + v_i)/2.
    # Since c_i >= w_i/2 and d_i <= T - v_i/2,
    # we need w_i/2 <= d_i - (w_i + v_i)/2 => d_i >= w_i + v_i/2.
    # Also d_i <= T - v_i/2.
    # So w_i + v_i/2 <= T - v_i/2 => w_i + v_i <= T.
    # If w_i + v_i > T, then JOB i is impossible.
    for i in range(j):
        w, v = designs[i]
        if w + v > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD 1 (Primary Crowding)
    # Sum of w_k <= T
    if sum(d[0] for d in designs) > T:
        print("LOAD 1")
        return
    
    # Check LOAD 2 (Secondary Crowding)
    # Sum of v_k <= T
    if sum(d[1] for d in designs) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
