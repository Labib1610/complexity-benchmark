import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    motifs = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check Motif Span Invariant:
        # Alpha phase [a, a+p1], Beta phase [b, b+p2]
        # Constraint: b >= a + p1
        # Also: 0 <= a, a+p1 <= T, 0 <= b, b+p2 <= T
        # The smallest possible b is a + p1.
        # So we need (a + p1) + p2 <= T, which means a + p1 + p2 <= T.
        # Since a >= 0, the condition is p1 + p2 <= T.
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Check Alpha Channel Invariant:
    # Sum of p1 <= T
    if sum_p1 > t:
        print("LOAD 1")
        return
    
    # Check Beta Channel Invariant:
    # Sum of p2 <= T
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
