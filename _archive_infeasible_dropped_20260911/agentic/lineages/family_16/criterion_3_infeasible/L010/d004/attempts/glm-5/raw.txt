```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    
    p1_list = []
    p2_list = []
    
    # Check Motif Span Invariant for each motif
    for i in range(1, j + 1):
        p1 = int(next(it))
        p2 = int(next(it))
        p1_list.append(p1)
        p2_list.append(p2)
        # Motif i is impossible if p1 + p2 > T
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
    
    # Check Alpha Channel Invariant
    sum_p1 = sum(p1_list)
    if sum_p1 > T:
        print("LOAD 1")
        return
    
    # Check Beta Channel Invariant
    sum_p2 = sum(p2_list)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    solve()
```
