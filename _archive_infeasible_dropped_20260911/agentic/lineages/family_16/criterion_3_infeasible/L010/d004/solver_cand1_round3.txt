```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx + 1])
        jobs.append((d1, d2))
        idx += 2
        
    # Check for Direct Path Contradiction
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for Volume Contradiction
    sum1 = 0
    sum2 = 0
    for d1, d2 in jobs:
        sum1 += d1
        sum2 += d2
        
    if sum1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
