import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    apps = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Lifecycle Failure: For any application i, p1 + p2 must be <= T.
        # The problem states: "all of its setup routines must be assigned to 
        # strictly earlier instruction indices than any of its teardown routines."
        # This means if we have p1 setups and p2 teardowns, we need at least 
        # p1 + p2 distinct slots.
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Setup Congestion: Total p1 must be <= T.
    if sum_p1 > t:
        print("LOAD 1")
        return
    
    # Teardown Congestion: Total p2 must be <= T.
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
