import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    queries = []
    replies = []
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Causality check:
        # M_Q,i >> k = 0 means all set bits of M_Q,i must be in positions [0, k-1].
        # So M_Q,i must fit in k bits. Thus p1 <= k.
        # (M_R,i << (T-k)) % 2^T = 0 means all set bits of M_R,i must be in positions [0, T-k-1].
        # So M_R,i must fit in T-k bits. Thus p2 <= T-k.
        # Combining: p1 <= k and p2 <= T-k => p1 + p2 <= T.
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
        
        queries.append(p1)
        replies.append(p2)
        
    # Query Independence:
    # Sum of Hamming weights must be <= T.
    if sum(queries) > T:
        print("LOAD 1")
        return
        
    # Reply Independence:
    # Sum of Hamming weights must be <= T.
    if sum(replies) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
