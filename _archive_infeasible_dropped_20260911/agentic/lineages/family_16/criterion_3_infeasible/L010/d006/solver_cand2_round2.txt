```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    exchanges = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        exchanges.append((p1, p2))
        idx += 2
        
    # Check Intra-Exchange Isolation (JOB i)
    # Property: Union of Q and R must have p1 + p2 distinct states.
    # Since Q and R are subsets of T, and they must be disjoint to satisfy
    # the sorting property (all Q < all R), the total states needed is p1 + p2.
    # If p1 + p2 > T, it's impossible.
    for i in range(j):
        p1, p2 = exchanges[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check Global Q-Isolation (LOAD 1)
    # Total states needed for all Q-sets is sum(p1_i).
    # If sum(p1_i) > T, it's impossible.
    sum_p1 = 0
    for p1, p2 in exchanges:
        sum_p1 += p1
    
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check Global R-Isolation (LOAD 2)
    # Total states needed for all R-sets is sum(p2_i).
    # If sum(p2_i) > T, it's impossible.
    sum_p2 = 0
    for p1, p2 in exchanges:
        sum_p2 += p2
        
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
