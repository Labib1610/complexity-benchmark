```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    pairs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        pairs.append((p1, p2))
        idx += 2
        
    # 1. Check Transversality Failure (JOB i)
    # Two subspaces V and W of dimensions p1 and p2 in space T
    # intersect trivially if and only if p1 + p2 <= T.
    # If p1 + p2 > T, they must intersect non-trivially.
    for i in range(j):
        if pairs[i][0] + pairs[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # 2. Check Primary Collision (LOAD 1)
    # Sum of dimensions of primary subspaces must be <= T.
    sum_p1 = sum(p[0] for p in pairs)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # 3. Check Secondary Collision (LOAD 2)
    # Sum of dimensions of secondary subspaces must be <= T.
    sum_p2 = sum(p[1] for p in pairs)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
