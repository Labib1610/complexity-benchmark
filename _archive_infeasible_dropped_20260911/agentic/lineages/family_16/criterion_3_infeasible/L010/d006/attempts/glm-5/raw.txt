```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    
    total_setup = 0
    total_teardown = 0
    lifecycle_failed_job = -1
    
    for i in range(1, j + 1):
        p1 = int(next(it))
        p2 = int(next(it))
        total_setup += p1
        total_teardown += p2
        if p1 + p2 > T and lifecycle_failed_job == -1:
            lifecycle_failed_job = i
    
    if lifecycle_failed_job != -1:
        print(f"JOB {lifecycle_failed_job}")
    elif total_setup > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == "__main__":
    solve()
```
