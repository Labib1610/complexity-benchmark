```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    total_setup = 0
    total_teardown = 0
    for idx in range(1, j + 1):
        p1 = int(next(it))
        p2 = int(next(it))
        # Lifecycle check
        if p1 + p2 > T:
            sys.stdout.write(f"JOB {idx}")
            return
        total_setup += p1
        total_teardown += p2
    # Congestion checks
    if total_setup > T:
        sys.stdout.write("LOAD 1")
    else:
        sys.stdout.write("LOAD 2")

if __name__ == "__main__":
    solve()
```
