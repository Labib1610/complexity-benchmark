```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check for Job Violation
    # A job i is impossible if p1_i + p2_i > T
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for Preparation Saturation
    # Sum of p1_i > T
    sum_p1 = sum(job[0] for job in jobs)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check for Execution Saturation
    # Sum of p2_i > T
    sum_p2 = sum(job[1] for job in jobs)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
