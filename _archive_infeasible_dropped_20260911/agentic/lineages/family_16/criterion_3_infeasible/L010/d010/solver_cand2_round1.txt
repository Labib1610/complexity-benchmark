import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    sum_p1 = 0
    sum_p2 = 0
    
    # The state update matrices:
    # Delta: M_D = [[-1, 2], [-1, 2]]
    # Snapshot: M_S = [[-1, 4], [-1, 3]]
    # Initial state: [1, 1]^T
    # After p1 Delta and p2 Snapshot chunks, the state is M_S^p2 * M_D^p1 * [1, 1]^T
    # M_D * [1, 1]^T = [1, 1]^T. So M_D^p1 * [1, 1]^T = [1, 1]^T.
    # Thus, the final state is M_S^p2 * [1, 1]^T.
    # M_S = [[-1, 4], [-1, 3]]
    # Eigenvalues of M_S: det(M_S - lambda*I) = (-1-lambda)(3-lambda) + 4 = lambda^2 - 2lambda + 1 = (lambda-1)^2 = 0.
    # Eigenvalue is 1 with multiplicity 2.
    # M_S = I + N, where N = [[-2, 4], [-1, 2]]. Note N^2 = 0.
    # M_S^p2 = (I + N)^p2 = I + p2 * N = [[1 - 2*p2, 4*p2], [-p2, 1 + 2*p2]]
    # Final state = [[1 - 2*p2, 4*p2], [-p2, 1 + 2*p2]] * [1, 1]^T
    # X = 1 - 2*p2 + 4*p2 = 2*p2 + 1
    # Y = -p2 + 1 + 2*p2 = p2 + 1
    # This matches the target state (2*p2 + 1, p2 + 1) for ANY p1, p2.
    # The constraint is that p1 + p2 <= T for each database individually.
    
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
