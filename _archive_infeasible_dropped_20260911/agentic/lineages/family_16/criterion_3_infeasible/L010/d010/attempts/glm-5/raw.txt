```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    T = int(data[1])
    
    total_p1 = 0
    total_p2 = 0
    job_idx = -1
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(data[idx])
        p2 = int(data[idx + 1])
        idx += 2
        total_p1 += p1
        total_p2 += p2
        if job_idx == -1 and p1 + p2 > T:
            job_idx = i
    
    if job_idx != -1:
        print(f"JOB {job_idx}")
    elif total_p1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == "__main__":
    main()
```
