import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    sum_p1 = 0
    sum_p2 = 0
    
    # Process each database
    # A database i is valid if it can fit its p1_i Delta chunks and p2_i Snapshot chunks
    # such that the final tracker U_i = 0.
    # The tracker U_i starts at p2_i.
    # Each Snapshot chunk decreases U_i by 1.
    # Each Delta chunk resets U_i to p2_i.
    # To end at 0, the last operation for database i must be a Snapshot chunk,
    # and after the last Delta chunk, there must be at least p2_i Snapshot chunks.
    # If there are p1_i Delta chunks, there are p1_i resets.
    # Between each reset (and before the first), there must be enough Snapshot chunks
    # to reach 0. But wait, the tracker resets to p2_i.
    # To reach 0, we need p2_i consecutive Snapshot chunks after the last Delta chunk.
    # This means we need at least p2_i slots for Snapshots after the last Delta.
    # Total slots needed for database i:
    # If p1_i > 0: we need at least p1_i Delta chunks and p2_i Snapshot chunks.
    # The last Delta chunk must be followed by p2_i Snapshot chunks.
    # So we need at least p1_i + p2_i seconds.
    # If p1_i = 0: we need p2_i Snapshot chunks, so p2_i seconds.
    # Actually, the problem says "exactly p1_i Delta and p2_i Snapshot".
    # If p1_i > 0, we need p1_i + p2_i seconds.
    # If p1_i = 0, we need p2_i seconds.
    
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        sum_p1 += p1
        sum_p2 += p2
        
        # Check Phase Violation
        needed = (p1 + p2) if p1 > 0 else p2
        if needed > T:
            print(f"JOB {i + 1}")
            return
            
    # Check Saturation
    if sum_p1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
