```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    sum_p1 = 0
    sum_p2 = 0
    
    # The tracker M_i starts at 1.
    # Operations:
    # Snapshot (S): M = M + 1
    # Delta (D): M = M * 2
    # Both (B): M = (M + 1) * 2 = 2M + 2
    #
    # We need final M = 2^p1 + p2.
    # Let x be the number of times we perform 'Both' (B).
    # Let y be the number of times we perform 'Snapshot' (S).
    # Let z be the number of times we perform 'Delta' (D).
    # Total Delta chunks: z + x = p1
    # Total Snapshot chunks: y + x = p2
    #
    # The tracker value after p1 Delta and p2 Snapshot chunks:
    # If we perform all S before all D, M = (1 + p2) * 2^p1 = 2^p1 + p2 * 2^p1.
    # To get 2^p1 + p2, we must perform all additions (S) such that they are 
    # never multiplied by 2. This means no 'Both' operations are allowed (x=0),
    # and all 'Snapshot' operations must occur after all 'Delta' operations.
    #
    # If x > 0, the value will be > 2^p1 + p2.
    # If x = 0, we need p1 Delta operations and p2 Snapshot operations.
    # The minimum time required for one database is p1 + p2.
    # If p1 + p2 > T, it's a Phase Violation.
    
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
