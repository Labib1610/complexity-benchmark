```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    tasks = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tasks.append((p1, p2))
        idx += 2
        
    # Check for JOB i violation
    # The register W_i starts at p1_i.
    # If an Execution cycle occurs when W_i > 0, W_i increases.
    # To end at 0, all p1_i Preparation cycles must occur before any Execution cycle.
    # This requires p1_i + p2_i <= T.
    for i in range(j):
        p1, p2 = tasks[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD 1 (Preparation Saturation)
    sum_p1 = sum(task[0] for task in tasks)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check for LOAD 2 (Execution Saturation)
    sum_p2 = sum(task[1] for task in tasks)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
