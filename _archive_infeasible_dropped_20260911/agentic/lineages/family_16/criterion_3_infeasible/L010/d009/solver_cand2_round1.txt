```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    fields = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        fields.append((p1, p2))
        idx += 2
        
    # Check for JOB i (Curl Violation)
    # Orientation Rule: For a field i, we need p1_i vectors from Pool X (1, u)
    # and p2_i vectors from Pool Y (1, v).
    # Cross product: (1, u) x (1, v) = 1*v - u*1 = v - u.
    # We need v - u > 0 for all selected u, v.
    # This means min(v) > max(u).
    # To maximize the number of vectors we can pick, we pick the smallest possible u's
    # and the largest possible v's.
    # Pool X: u in {1, ..., T}, Pool Y: v in {1, ..., T}.
    # To satisfy v > u for all pairs, we pick u in {1, ..., p1_i} and v in {T-p2_i+1, ..., T}.
    # The condition is min(v) > max(u) => (T - p2_i + 1) > p1_i.
    # So p1_i + p2_i <= T.
    
    for i in range(j):
        p1, p2 = fields[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD 1 (Pool X Depletion)
    sum_p1 = sum(f[0] for f in fields)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check for LOAD 2 (Pool Y Depletion)
    sum_p2 = sum(f[1] for f in fields)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
