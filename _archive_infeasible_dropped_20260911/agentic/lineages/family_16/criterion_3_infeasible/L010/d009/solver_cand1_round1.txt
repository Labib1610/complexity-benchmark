import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    orders = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        orders.append((p1, p2))
        idx += 2
        
    # Check for Spatial Violation:
    # For product i, we need p1_i cells in Row 1 and p2_i cells in Row 2.
    # Condition: max(C1_i) < min(C2_i).
    # This means we need at least p1_i columns for Row 1 and p2_i columns for Row 2,
    # and there must be at least one column gap or the last column of Row 1 must be
    # strictly before the first column of Row 2.
    # The minimum columns required for product i is p1_i + p2_i.
    # If p1_i + p2_i > T, it's impossible.
    for i in range(j):
        if orders[i][0] + orders[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for Row 1 Saturation:
    # Total cells in Row 1 must be <= T.
    sum_p1 = sum(o[0] for o in orders)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check for Row 2 Saturation:
    # Total cells in Row 2 must be <= T.
    sum_p2 = sum(o[1] for o in orders)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
