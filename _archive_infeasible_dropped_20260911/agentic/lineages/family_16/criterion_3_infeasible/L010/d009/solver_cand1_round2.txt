```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    sum_p1 = 0
    sum_p2 = 0
    
    # Process each database
    # A database i is valid if p1_i + p2_i <= T.
    # If p1_i + p2_i > T, it's a JOB i violation.
    # Otherwise, check global saturation.
    
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        # Check Phase Violation
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Check Delta Saturation
    if sum_p1 > T:
        print("LOAD 1")
        return
    
    # Check Snapshot Saturation
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
