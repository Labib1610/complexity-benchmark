```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    T = int(input_data[1])
    
    weights = []
    idx = 2
    for i in range(j):
        wa = int(input_data[idx])
        wb = int(input_data[idx+1])
        weights.append((wa, wb))
        idx += 2
        
    # Check JOB i
    # x_i >= 0, x_i + wA_i <= y_i, y_i + wB_i <= T
    # => 0 + wA_i + wB_i <= T => wA_i + wB_i > T
    for i in range(j):
        if weights[i][0] + weights[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Condition 2: sum of integrals of phi_i(t)^2 * 2^-|t| = sum of integrals of phi_i(t) * 2^-|t|
    # This implies phi_i(t) * phi_k(t) = 0 for i != k almost everywhere.
    # Since phi_i(t) = 1 on [x_i, x_i + wA_i], they must be disjoint.
    # The total measure of the union of supports is sum(wA_i).
    # The domain is [0, T].
    # However, the weight 2^-|t| is non-zero everywhere.
    # The condition is equivalent to saying the supports must be disjoint.
    # If sum(wA_i) > T, then they cannot be disjoint in [0, T].
    if sum(w[0] for w in weights) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Similarly, for psi_i(t) = 1 on [y_i, y_i + wB_i], they must be disjoint.
    # The domain is [0, T].
    if sum(w[1] for w in weights) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
