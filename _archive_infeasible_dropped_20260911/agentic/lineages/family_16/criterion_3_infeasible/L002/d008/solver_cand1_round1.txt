```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for _ in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # The system of equations:
    # T*u + T*v = 0 => u = -v
    # 2T*u + 3T*v + T*w = 0 => 2T(-v) + 3T*v + T*w = 0 => T*v + T*w = 0 => v = -w
    # So u = w, v = -w
    # Third equation: 2T*u + 4T*v + (T + X)*w = 1
    # 2T(w) + 4T(-w) + (T + X)*w = 1
    # 2T*w - 4T*w + T*w + X*w = 1
    # (X - T) * w = 1
    # w = 1 / (X - T)
    # For w > 0, we need X - T > 0, i.e., X > T.
    
    # For JOB i: X = p1[i] + p2[i]
    # For LOAD 1: X = sum(p1)
    # For LOAD 2: X = sum(p2)
    
    # Check LOAD 1
    s1 = sum(p1)
    if s1 > T:
        print("LOAD 1")
        return
    
    # Check LOAD 2
    s2 = sum(p2)
    if s2 > T:
        print("LOAD 2")
        return
    
    # Check JOB i
    for i in range(j):
        if p1[i] + p2[i] > T:
            print(f"JOB {i + 1}")
            return

if __name__ == '__main__':
    solve()
```
