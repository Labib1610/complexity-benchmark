```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    components = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        components.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # x_i >= 0, x_i + p1_i <= y_i, y_i + p2_i <= T
    # x_i + p1_i + p2_i <= T => x_i <= T - p1_i - p2_i
    # If T - p1_i - p2_i < 0, then x_i < 0, which contradicts x_i >= 0
    for i in range(j):
        p1, p2 = components[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Sum of p1_i <= T
    sum_p1 = sum(c[0] for c in components)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sum of p2_i <= T
    sum_p2 = sum(c[1] for c in components)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
