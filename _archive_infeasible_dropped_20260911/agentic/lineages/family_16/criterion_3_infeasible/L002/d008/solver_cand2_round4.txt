import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for i in range(j):
        p1_val = int(input_data[idx])
        p2_val = int(input_data[idx+1])
        p1.append(p1_val)
        p2.append(p2_val)
        idx += 2
        
        # Check JOB i
        # u_i >= p1_i, v_i - u_i >= p1_i + p2_i => v_i >= 2*p1_i + p2_i
        # v_i <= 2*T - p2_i
        # Unsatisfiable if 2*p1_i + p2_i > 2*T - p2_i
        # 2*p1_i + 2*p2_i > 2*T => p1_i + p2_i > T
        if p1_val + p2_val > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD 1
    # u_i >= p1_i, u_i <= 2*T - p1_i
    # Range length: (2*T - p1_i) - p1_i = 2*(T - p1_i)
    # Sort p1 to find the tightest constraint
    sorted_p1 = sorted(p1)
    # For LOAD 1, we need to place j points such that |u_i - u_k| >= p1_i + p1_k
    # This is equivalent to placing points x_i such that x_i >= 0 and x_j - x_1 >= sum(2*p1_i) - (p1_1 + p1_j)
    # Actually, the condition for existence is:
    # sum_{i=1}^j (2*p1_i) - (p1_min + p1_max) <= 2*T - (p1_min + p1_max)
    # Simplified: sum(2*p1_i) <= 2*T
    if sum(p1) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # v_i >= p2_i, v_i <= 2*T - p2_i
    # Similar logic: sum(2*p2_i) <= 2*T
    if sum(p2) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
