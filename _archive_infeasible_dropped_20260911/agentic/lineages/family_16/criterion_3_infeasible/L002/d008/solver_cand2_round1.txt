import sys

def solve():
    # The integral J(k, m) = integral of (sin(pi*p_k*f)*sin(pi*p_m*f) / (pi^2 * f^2)) * cos(pi*f*(2*(z_k - z_m) + p_k - p_m)) df
    # This is a known integral: integral of (sin(a*f)*sin(b*f)/f^2) * cos(c*f) df = (pi/2) * min(a, b, |a+b+c|/2, |a+b-c|/2, ...)
    # Specifically, for the given form, the integral evaluates to min(p_k, p_m, z_k + p_k - z_m, z_m + p_m - z_k)
    # The condition sum_{k,m} J(k,m) = sum p_i simplifies to:
    # sum_{k,m} min(p_k, p_m, z_k + p_k - z_m, z_m + p_m - z_k) = sum p_i
    # Given z_i >= 0 and z_i + p_i <= T, the maximum possible value of the sum is achieved
    # when z_i are spread out. The constraint is violated if the minimum possible value
    # of the sum exceeds sum(p_i).
    # For LOAD c, the condition is sum_{k,m} min(p_k, p_m, z_k + p_k - z_m, z_m + p_m - z_k) = sum(p_i)
    # where 0 <= z_i <= T - p_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for i in range(j):
        val1 = int(input_data[idx])
        val2 = int(input_data[idx+1])
        p1.append(val1)
        p2.append(val2)
        # Check JOB i
        if val1 + val2 > T:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1
    # sum_{k,m} min(p1_k, p1_m, z1_k + p1_k - z1_m, z1_m + p1_m - z1_k)
    # If we set z1_i = 0, the sum is sum_{k,m} min(p1_k, p1_m, p1_k, p1_m) = sum_{k,m} min(p1_k, p1_m)
    # This is always >= sum(p1_i). The constraint is violated if the minimum possible
    # value of the sum over all valid z1_i is > sum(p1_i).
    # The minimum is achieved by packing z1_i as tightly as possible.
    
    def check_load(p_list):
        # The sum is sum_{k,m} min(p_k, p_m, z_k + p_k - z_m, z_m + p_m - z_k)
        # This is equivalent to the sum of lengths of intersection of intervals [z_i, z_i + p_i]
        # The condition sum_{k,m} J(k,m) = sum p_i is equivalent to the intervals being disjoint.
        # If they cannot be disjoint within [0, T], then LOAD is impossible.
        # Total length of intervals is sum(p_i). If sum(p_i) > T, it's impossible.
        if sum(p_list) > T:
            return True
        return False

    if check_load(p1):
        print("LOAD 1")
    else:
        print("LOAD 2")

solve()
