```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    # Stream i is unroutable if for all alpha, beta > 0:
    # alpha * p1 + beta * p2 > T * min(alpha, beta)
    # This is equivalent to:
    # If alpha <= beta: alpha * p1 + beta * p2 > T * alpha => p1 + (beta/alpha) * p2 > T
    # Since beta/alpha >= 1, min value is p1 + p2.
    # If beta <= alpha: alpha * p1 + beta * p2 > T * beta => (alpha/beta) * p1 + p2 > T
    # Since alpha/beta >= 1, min value is p1 + p2.
    # So stream i is unroutable if p1 + p2 > T.
    
    # Phase c is overloaded if there exists q_k >= 0 (not all 0) such that:
    # sum(q_k * p_{c,k}) > T * max(q_k)
    # Let q_k = 1 for all k. Then sum(p_{c,k}) > T.
    # This is equivalent to sum(p_{c,k}) > T.
    
    # Check JOB i
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    sum1 = 0
    sum2 = 0
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        sum1 += p1
        sum2 += p2
        
    if sum1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
