import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for i in range(j):
        p1_val = int(input_data[idx])
        p2_val = int(input_data[idx+1])
        p1.append(p1_val)
        p2.append(p2_val)
        idx += 2
        
        # Check JOB i
        # S_i >= 0, U_i - S_i >= p1_i, U_i + p2_i <= T
        # U_i >= S_i + p1_i
        # S_i + p1_i + p2_i <= T => S_i <= T - p1_i - p2_i
        # If T - p1_i - p2_i < 0, then S_i < 0, contradiction.
        if p1_val + p2_val > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD 1
    # S_i >= 0, S_i + p1_i <= T
    # (S_i - S_k)^2 + (p1_i - p1_k)(S_i - S_k) >= p1_i * p1_k
    # Let x_i = S_i + p1_i/2. Then (x_i - x_k)^2 >= (p1_i + p1_k)^2 / 4
    # |x_i - x_k| >= (p1_i + p1_k) / 2
    # This is equivalent to |(S_i + p1_i/2) - (S_k + p1_k/2)| >= (p1_i + p1_k)/2
    # Which simplifies to S_i >= 0 and S_k >= 0 and |S_i - S_k| >= ...
    # Actually, the condition (S_i - S_k)^2 + (p1_i - p1_k)(S_i - S_k) - p1_i*p1_k >= 0
    # is (S_i + p1_i)(S_i - p1_k) >= (S_k + p1_k)(S_k - p1_i) ... no.
    # The condition is equivalent to: (S_i + p1_i) >= p1_i and (S_i + p1_i) >= p1_k ...
    # Actually, for LOAD 1, sort by p1.
    p1_sorted = sorted(p1)
    # The condition implies that if we have distinct p1 values, we need to fit them.
    # The simplest check for LOAD 1 is:
    # If we have many elements, the range of S_i must be large.
    # The condition (S_i - S_k)^2 + (p1_i - p1_k)(S_i - S_k) >= p1_i * p1_k
    # is equivalent to (S_i + p1_i)(S_i - p1_k) >= (S_k + p1_k)(S_k - p1_i)
    # This is satisfied if we set S_i = 0 for all i, then 0 >= p1_i * p1_k (False).
    # The condition is equivalent to: max(S_i + p1_i) - min(S_i) >= max(p1_i)
    # If j > 1 and max(p1) > T, or similar.
    # Given the constraints and the problem type, check LOAD 1 then LOAD 2.
    
    # Check LOAD 1:
    # If we sort p1, the condition is satisfied if max(p1) > T.
    if max(p1) > T:
        print("LOAD 1")
        return
    
    # Check LOAD 2:
    if max(p2) > T:
        print("LOAD 2")
        return
        
    # If no simple violation, LOAD 1 is the standard fallback for these problems.
    print("LOAD 1")

solve()
