```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for i in range(j):
        p1_val = int(input_data[idx])
        p2_val = int(input_data[idx+1])
        p1.append(p1_val)
        p2.append(p2_val)
        idx += 2
        
        # Check JOB i
        # y_i - x_i >= p1_i, y_i + p2_i <= T, x_i >= 0
        # y_i >= x_i + p1_i
        # x_i + p1_i + p2_i <= T => x_i <= T - p1_i - p2_i
        # Since x_i >= 0, we need T - p1_i - p2_i >= 0
        if p1_val + p2_val > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD 1
    # x_i >= 0, x_i + p1_i <= T
    # (x_i - x_k)^2 + (x_i - x_k)(p1_i - p1_k) >= p1_i * p1_k
    # This is equivalent to (x_i + p1_i)(x_i) >= (x_k + p1_k)(x_k) is not quite right.
    # The condition (x_i - x_k)^2 + (x_i - x_k)(p1_i - p1_k) - p1_i * p1_k >= 0
    # factors to (x_i + p1_i)(x_i - p1_k) >= (x_k + p1_k)(x_k - p1_i)
    # Actually, the condition is equivalent to:
    # Let u_i = x_i + p1_i and v_i = x_i. Then u_i * v_i >= u_k * v_k is not it.
    # The condition is equivalent to: max(x_i) - min(x_i) >= sum of gaps.
    # For LOAD 1, sort by p1_i. If we have many elements, the span of x_i must be large.
    # The condition (x_i - x_k)^2 + (x_i - x_k)(p1_i - p1_k) >= p1_i * p1_k
    # is equivalent to (x_i + p1_i)(x_i) >= (x_k + p1_k)(x_k) is wrong.
    # It is equivalent to: (x_i + p1_i/2)^2 - (p1_i/2)^2 >= (x_k + p1_k/2)^2 - (p1_k/2)^2
    # Let z_i = x_i + p1_i/2. Then z_i^2 - z_k^2 >= (p1_i/2)^2 - (p1_k/2)^2
    # This is equivalent to |z_i - z_k| >= |p1_i/2 - p1_k/2|? No.
    # The condition is simply |x_i - x_k| >= |p1_i - p1_k| is not correct.
    # The condition is equivalent to: x_i + p1_i >= x_k and x_k + p1_k >= x_i.
    # Actually, for LOAD 1, if we sort by p1, the condition implies x_{i+1} - x_i >= p1_i.
    # So sum(p1_i) <= T.
    
    p1.sort()
    if sum(p1) > T:
        print("LOAD 1")
        return
        
    p2.sort()
    if sum(p2) > T:
        print("LOAD 2")
        return

solve()
```
