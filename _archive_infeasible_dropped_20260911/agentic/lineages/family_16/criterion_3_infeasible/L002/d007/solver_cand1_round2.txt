import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    elements = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        elements.append((p1, p2, i + 1))
        idx += 2
        
    # Check JOB i
    # M1,i - p1/2 >= 0 => M1,i >= p1/2
    # M1,i + p1/2 <= M2,i - p2/2 => M2,i >= M1,i + (p1+p2)/2
    # M2,i + p2/2 <= T => M2,i <= T - p2/2
    # Combined: p1/2 + (p1+p2)/2 <= M2,i <= T - p2/2
    # p1 + p2/2 <= T - p2/2 => p1 + p2 <= T
    # If p1 + p2 > T, then JOB i is the certificate.
    for p1, p2, i in elements:
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
            
    # Check LOAD 1
    # M1,i - p1/2 >= 0 => M1,i >= p1/2
    # M1,i + p1/2 <= T => M1,i <= T - p1/2
    # |M1,i - M1,k| >= (p1,i + p1,k)/2
    # Sort by p1,i. Let sorted p1 be q1, q2, ..., qj.
    # The minimum span required is sum(p1,i) for all i.
    # The available span is T.
    # Actually, the condition for packing points x_i in [L, R] with distance |x_i - x_k| >= (p_i + p_k)/2
    # is equivalent to sum(p_i) <= R - L.
    # Here L = min(p1,i/2), R = max(T - p1,i/2).
    # The condition is sum(p1,i) <= T.
    if sum(e[0] for e in elements) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Similar logic for LOAD 2: sum(p2,i) <= T
    if sum(e[1] for e in elements) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
