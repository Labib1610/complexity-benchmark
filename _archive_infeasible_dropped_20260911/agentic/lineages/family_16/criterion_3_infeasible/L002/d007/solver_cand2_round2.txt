```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    elements = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        elements.append((p1, p2, i + 1))
        idx += 2
        
    # Check JOB i
    # S_i >= 0, E_i - S_i = p1, V_i - U_i = p2, E_i <= U_i, V_i <= T
    # S_i + p1 <= U_i, V_i = U_i + p2 <= T => U_i <= T - p2
    # S_i + p1 <= T - p2 => S_i <= T - p1 - p2
    # Since S_i >= 0, if T - p1 - p2 < 0, then JOB i is impossible.
    for p1, p2, i in elements:
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
            
    # Check LOAD 1
    # Sum of p1_i <= T
    sum_p1 = sum(e[0] for e in elements)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sum of p2_i <= T
    sum_p2 = sum(e[1] for e in elements)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
