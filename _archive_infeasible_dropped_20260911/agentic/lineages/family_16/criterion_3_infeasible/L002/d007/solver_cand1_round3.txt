```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    
    idx = 2
    for i in range(j):
        p1_val = int(input_data[idx])
        p2_val = int(input_data[idx+1])
        p1.append(p1_val)
        p2.append(p2_val)
        
        # Check JOB i
        # x_i >= 0, x_i + p1_i <= y_i, y_i + p2_i <= T
        # x_i + p1_i + p2_i <= T => x_i <= T - p1_i - p2_i
        # If T - p1_i - p2_i < 0, then x_i < 0, contradiction.
        if p1_val + p2_val > T:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1
    # sum(p1_i) > T * (something)?
    # The condition for LOAD 1 is:
    # sum(p1_i) must be <= T * (number of intervals that can fit)
    # Actually, the condition for LOAD 1 is:
    # sum(p1_i) <= T * (max number of overlapping intervals)
    # The orthogonality condition sum(phi_i^2) = sum(p1_i) implies that
    # the intervals [x_i, x_i + p1_i) must be disjoint.
    # If they are disjoint and contained in [0, T], then sum(p1_i) <= T.
    if sum(p1) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    if sum(p2) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
