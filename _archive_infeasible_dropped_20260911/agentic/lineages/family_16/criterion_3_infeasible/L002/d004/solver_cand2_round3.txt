```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    oven_total = 0
    broiler_total = 0
    
    idx = 2
    for i in range(1, j + 1):
        o = int(input_data[idx])
        b = int(input_data[idx + 1])
        idx += 2
        
        if o + b > t:
            print(f"JOB {i}")
            return
        
        oven_total += o
        broiler_total += b
        
    if oven_total > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
