```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    v1_sum = 0
    v2_sum = 0
    
    idx = 2
    for i in range(1, j + 1):
        v1 = int(input_data[idx])
        v2 = int(input_data[idx + 1])
        idx += 2
        
        if v1 + v2 > t:
            print(f"JOB {i}")
            return
        
        v1_sum += v1
        v2_sum += v2
        
    if v1_sum > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
