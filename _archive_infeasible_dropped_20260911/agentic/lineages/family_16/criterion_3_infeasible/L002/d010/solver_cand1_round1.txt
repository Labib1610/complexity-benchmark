import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    elements = []
    idx = 2
    for i in range(j):
        q1 = int(input_data[idx])
        q2 = int(input_data[idx+1])
        elements.append((q1, q2))
        idx += 2
        
    # Check JOB i
    # P(x; u, v) = x^2 + 2ux + v^2
    # Admissible: discriminant >= 0 => (2u)^2 - 4v^2 >= 0 => u^2 >= v^2 => |u| >= |v|
    # Strongly Admissible: |u| >= |v| AND sum of roots < 0 => -2u < 0 => u > 0
    # So, Strongly Admissible <=> u >= |v|
    # Since q > 0, |v| = sqrt(q^2) = q. So u >= q.
    
    # JOB i:
    # 1. w1 >= q1
    # 2. (w2 - w1) >= q1 + q2
    # 3. (2T - w2) >= q2
    # Summing: w1 + w2 - w1 + 2T - w2 >= q1 + q1 + q2 + q2
    # 2T >= 2(q1 + q2) => T >= q1 + q2
    
    for i in range(j):
        q1, q2 = elements[i]
        if q1 + q2 > T:
            print(f"JOB {i + 1}")
            return
            
    # LOAD c:
    # For stage 1:
    # w1,i >= q1,i
    # 2T - w1,i >= q1,i => w1,i <= 2T - q1,i
    # Dispersion: |w1,i - w1,k| >= q1,i + q1,k
    # Let x_i = w1,i. Interval [q1,i, 2T - q1,i]. Length L_i = 2T - 2*q1,i.
    # We need to place j points such that |x_i - x_k| >= q1,i + q1,k.
    # This is equivalent to placing points y_i = x_i - q1,i such that |y_i - y_k| >= 2*min(q1,i, q1,k) ? No.
    # Actually, the condition |x_i - x_k| >= q1,i + q1,k is equivalent to:
    # (x_i + q1,i) <= (x_k - q1,k) OR (x_k + q1,k) <= (x_i - q1,i)
    # Let L_i = q1,i and R_i = 2T - q1,i. We need x_i in [L_i, R_i].
    # The condition |x_i - x_k| >= q1,i + q1,k is equivalent to:
    # x_i + q1,i <= x_k - q1,k  OR  x_k + q1,k <= x_i - q1,i
    # Let a_i = x_i - q1,i and b_i = x_i + q1,i.
    # We need a_i >= 0, b_i <= 2T, and |x_i - x_k| >= q1,i + q1,k.
    # This is a known problem: can we place j intervals [q1,i, 2T - q1,i] such that
    # they are separated by the sum of their q values?
    # The condition is equivalent to: max(q1,i) + max(q1,k) <= 2T is not enough.
    # The condition is: sum of (2 * q1,i) for all i <= 2T.
    
    sum_q1 = sum(e[0] for e in elements)
    if sum_q1 > T:
        print("LOAD 1")
        return
        
    sum_q2 = sum(e[1] for e in elements)
    if sum_q2 > T:
        print("LOAD 2")
        return

solve()
