```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    tasks = []
    idx = 2
    for i in range(j):
        q1 = int(input_data[idx])
        q2 = int(input_data[idx+1])
        tasks.append((q1, q2))
        idx += 2
        
    # Check JOB i
    # Transition: w2 - q2 >= w1 + q1 => w2 - w1 >= q1 + q2
    # Confinement: w1 - q1 >= 0 => w1 >= q1
    #              w2 + q2 <= 2T => w2 <= 2T - q2
    # Combined: w2 - w1 >= q1 + q2
    #           w2 <= 2T - q2
    #           w1 >= q1
    # Impossible if (2T - q2) - q1 < q1 + q2
    # i.e., 2T < 2*q1 + 2*q2 => T < q1 + q2
    for i in range(j):
        q1, q2 = tasks[i]
        if q1 + q2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # For stage c, we need to place j non-overlapping intervals of length 2*q_{c,i}
    # within [0, 2T].
    # The total length required is sum(2 * q_{c,i}) for all i.
    # If sum(2 * q_{c,i}) > 2T, then it's impossible.
    # i.e., sum(q_{c,i}) > T.
    
    sum_q1 = sum(task[0] for task in tasks)
    if sum_q1 > T:
        print("LOAD 1")
        return
        
    sum_q2 = sum(task[1] for task in tasks)
    if sum_q2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
