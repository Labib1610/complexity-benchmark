```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    q1 = []
    q2 = []
    idx = 2
    for _ in range(j):
        q1.append(int(input_data[idx]))
        q2.append(int(input_data[idx+1]))
        idx += 2
        
    # Check JOB i
    # P_i = [x_i - q1_i, x_i + q1_i] x [y_i - q2_i, y_i + q2_i]
    # Bounding Square: 0 <= x_i - q1_i, x_i + q1_i <= 2T, 0 <= y_i - q2_i, y_i + q2_i <= 2T
    # Diagonal Separation: u <= v for all (u, v) in P_i
    # Max u = x_i + q1_i, Min v = y_i - q2_i
    # Condition: x_i + q1_i <= y_i - q2_i => y_i - x_i >= q1_i + q2_i
    # Constraints on x_i: q1_i <= x_i <= 2T - q1_i
    # Constraints on y_i: q2_i <= y_i <= 2T - q2_i
    # Min y_i - Max x_i >= q1_i + q2_i
    # q2_i - (2T - q1_i) >= q1_i + q2_i => -2T >= 0 (Impossible)
    # Actually:
    # x_i >= q1_i, x_i <= 2T - q1_i
    # y_i >= q2_i, y_i <= 2T - q2_i
    # y_i - x_i >= q1_i + q2_i
    # Max(y_i - x_i) = (2T - q2_i) - q1_i = 2T - (q1_i + q2_i)
    # We need 2T - (q1_i + q2_i) >= q1_i + q2_i => 2T >= 2(q1_i + q2_i) => T >= q1_i + q2_i
    # If T < q1_i + q2_i, then JOB i is impossible.
    
    for i in range(j):
        if q1[i] + q2[i] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # Sum of lengths of projections must be <= 2T
    # Length of projection of P_i on X-axis is 2 * q1_i
    # Length of projection of P_i on Y-axis is 2 * q2_i
    # Sum of lengths = sum(2 * q_i)
    # If sum(2 * q_i) > 2T, then LOAD c is impossible.
    
    sum_q1 = sum(q1)
    if sum_q1 > T:
        print("LOAD 1")
        return
        
    sum_q2 = sum(q2)
    if sum_q2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
