import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    T = int(input_data[1])
    
    elements = []
    idx = 2
    for i in range(j):
        q1 = int(input_data[idx])
        q2 = int(input_data[idx+1])
        elements.append((q1, q2))
        idx += 2
        
    # Check JOB i
    # R1,i = (w1 - q1, w1 + q1) subset (0, inf) => w1 - q1 >= 0 => w1 >= q1
    # R2,i = (w2 - q2, w2 + q2) subset (-inf, 2T) => w2 + q2 <= 2T => w2 <= 2T - q2
    # Precedence: for all x in R1, y in R2, y >= x
    # This is equivalent to inf(R2) >= sup(R1)
    # w2 - q2 >= w1 + q1 => w2 - w1 >= q1 + q2
    # Combining: w2 <= 2T - q2 and w1 >= q1
    # So w2 - w1 <= 2T - q2 - q1
    # We need w2 - w1 >= q1 + q2
    # Thus, we need q1 + q2 <= 2T - q2 - q1 => 2(q1 + q2) <= 2T => q1 + q2 <= T
    # If q1 + q2 > T, then JOB i is impossible.
    
    for i in range(j):
        q1, q2 = elements[i]
        if q1 + q2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # LOAD 1: R1,i = (w1,i - q1,i, w1,i + q1,i) subset (0, 2T)
    # Exclusivity: disjoint intervals of length 2*q1,i
    # Total length of intervals = sum(2 * q1,i)
    # Must be <= 2T
    # LOAD 2: R2,i = (w2,i - q2,i, w2,i + q2,i) subset (0, 2T)
    # Total length of intervals = sum(2 * q2,i)
    # Must be <= 2T
    
    sum_q1 = sum(e[0] for e in elements)
    if 2 * sum_q1 > 2 * T:
        print("LOAD 1")
        return
    
    print("LOAD 2")

if __name__ == '__main__':
    solve()
