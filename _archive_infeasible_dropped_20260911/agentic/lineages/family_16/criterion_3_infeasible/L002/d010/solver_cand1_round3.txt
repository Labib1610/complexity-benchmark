```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    items = []
    idx = 2
    for i in range(j):
        q1 = int(input_data[idx])
        q2 = int(input_data[idx+1])
        items.append((q1, q2))
        idx += 2
        
    # Check JOB i
    # Precedence: S2 >= E1 = S1 + q1
    # Boundaries: S1 >= 0, E2 = S2 + q2 <= T
    # So S2 >= S1 + q1 >= q1
    # E2 = S2 + q2 >= q1 + q2
    # If q1 + q2 > T, then JOB i is impossible
    for i in range(j):
        if items[i][0] + items[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # LOAD c is impossible if sum(q_c,i) > T
    sum_q1 = sum(item[0] for item in items)
    if sum_q1 > T:
        print("LOAD 1")
        return
        
    sum_q2 = sum(item[1] for item in items)
    if sum_q2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
