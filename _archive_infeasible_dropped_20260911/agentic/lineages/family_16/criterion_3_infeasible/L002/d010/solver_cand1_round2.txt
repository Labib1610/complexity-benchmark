```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        q1 = int(input_data[idx])
        q2 = int(input_data[idx+1])
        jobs.append((q1, q2, i + 1))
        idx += 2
        
    # Check JOB i condition: q1_i + q2_i > T
    for i in range(j):
        q1, q2, original_idx = jobs[i]
        if q1 + q2 > t:
            print(f"JOB {original_idx}")
            return
            
    # Check LOAD c condition: sum(q_c) > T
    sum1 = 0
    sum2 = 0
    for i in range(j):
        sum1 += jobs[i][0]
        sum2 += jobs[i][1]
        
    if sum1 > t:
        print("LOAD 1")
        return
    if sum2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
