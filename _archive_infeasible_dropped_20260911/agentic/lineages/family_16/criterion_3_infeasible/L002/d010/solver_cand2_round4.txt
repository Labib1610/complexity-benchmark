```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        q1 = int(input_data[idx])
        q2 = int(input_data[idx+1])
        jobs.append((q1, q2))
        idx += 2
        
    # Check JOB i certificate
    for i in range(j):
        q1, q2 = jobs[i]
        if q1 + q2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1 certificate
    sum_q1 = sum(job[0] for job in jobs)
    if sum_q1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2 certificate
    sum_q2 = sum(job[1] for job in jobs)
    if sum_q2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
