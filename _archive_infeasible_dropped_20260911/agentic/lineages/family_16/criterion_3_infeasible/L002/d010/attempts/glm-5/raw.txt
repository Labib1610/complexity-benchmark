```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    T = int(data[1])
    
    # Check JOB certificate: q1 + q2 > T
    idx = 2
    for i in range(1, j + 1):
        q1 = int(data[idx])
        q2 = int(data[idx + 1])
        idx += 2
        if q1 + q2 > T:
            sys.stdout.write(f"JOB {i}\n")
            return
    
    # Check LOAD certificate: sum of q_c > T
    sum1 = 0
    sum2 = 0
    idx = 2
    for _ in range(j):
        q1 = int(data[idx])
        q2 = int(data[idx + 1])
        idx += 2
        sum1 += q1
        sum2 += q2
    
    if sum1 > T:
        sys.stdout.write("LOAD 1\n")
    else:
        sys.stdout.write("LOAD 2\n")

solve()
```
