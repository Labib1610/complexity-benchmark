```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    q1 = []
    q2 = []
    
    idx = 2
    for i in range(j):
        val1 = int(input_data[idx])
        val2 = int(input_data[idx+1])
        q1.append(val1)
        q2.append(val2)
        idx += 2
        
        # Check JOB i
        # N1,i = (w1 - q1, w1 + q1) subset (0, inf) => w1 - q1 > 0 => w1 > q1
        # N2,i = (w2 - q2, w2 + q2) subset (-inf, 2T) => w2 + q2 < 2T => w2 < 2T - q2
        # Separation: max(N1,i) < min(N2,i) => w1 + q1 <= w2 - q2
        # So: q1 + q1 < w1 + q1 <= w2 - q2 < 2T - q2 - q2
        # 2*q1 < w1 + q1, w2 - q2 < 2T - 2*q2
        # Condition: 2*q1 + 2*q2 >= 2*T => q1 + q2 >= T
        if q1[-1] + q2[-1] >= T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1: sum(2 * q1_i) > 2T
    if sum(q1) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2: sum(2 * q2_i) > 2T
    if sum(q2) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
