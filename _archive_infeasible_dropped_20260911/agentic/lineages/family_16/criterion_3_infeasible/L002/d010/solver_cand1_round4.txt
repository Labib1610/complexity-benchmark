```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    q1 = []
    q2 = []
    
    idx = 2
    for i in range(j):
        val1 = int(input_data[idx])
        val2 = int(input_data[idx+1])
        q1.append(val1)
        q2.append(val2)
        
        # Check JOB i
        # X + Y >= 2*q1, X - Y >= 2*q2 => 2X >= 2*q1 + 2*q2 => X >= q1 + q2
        # Sum Bound: X <= 2T - q1 - q2
        # Unsatisfiable if q1 + q2 > 2T - q1 - q2 => 2(q1 + q2) > 2T => q1 + q2 > T
        if val1 + val2 > T:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1
    # Let U_i = X_i + Y_i. 2*q1_i <= U_i <= 4T - 2*q1_i
    # Separation: |U_i - U_k| >= 2(q1_i + q1_k)
    # Let V_i = U_i / 2. q1_i <= V_i <= 2T - q1_i
    # |V_i - V_k| >= q1_i + q1_k
    # Let W_i = V_i - q1_i. 0 <= W_i <= 2T - 2*q1_i
    # |(W_i + q1_i) - (W_k + q1_k)| >= q1_i + q1_k
    # This is equivalent to checking if we can pack intervals [q1_i, 2T - q1_i]
    # with separation q1_i + q1_k.
    # Actually, the condition is: sort by q1_i.
    # If we have sorted q1_i, the condition is equivalent to:
    # sum of (2 * q1_i) for all i <= 2T.
    if sum(2 * x for x in q1) > 2 * T:
        print("LOAD 1")
        return
    
    # Check LOAD 2
    # Similar logic for q2
    if sum(2 * x for x in q2) > 2 * T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
