```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    queries = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        queries.append((p1, p2, i + 1))
        idx += 2
        
    # Scenario A: Check if any single query i fails: p1 + p2 > T
    for p1, p2, i in queries:
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
            
    # Scenario B: Check if sum of p1 > T
    sum_p1 = sum(q[0] for q in queries)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Scenario C: Check if sum of p2 > T
    sum_p2 = sum(q[1] for q in queries)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
