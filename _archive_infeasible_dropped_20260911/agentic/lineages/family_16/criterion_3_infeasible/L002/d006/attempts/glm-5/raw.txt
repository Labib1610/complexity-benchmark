```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    
    p1_list = []
    p2_list = []
    
    for _ in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        p1_list.append(p1)
        p2_list.append(p2)
    
    # Check JOB certificate: S_i >= 0, S_i + p1_i <= U_i, U_i + p2_i <= T
    # This implies p1_i + p2_i <= T
    for i in range(j):
        if p1_list[i] + p2_list[i] > T:
            print(f"JOB {i + 1}")
            return
    
    # Check LOAD 1 certificate: sum of p1_i > T
    total_p1 = sum(p1_list)
    if total_p1 > T:
        print("LOAD 1")
        return
    
    # Check LOAD 2 certificate: sum of p2_i > T
    total_p2 = sum(p2_list)
    if total_p2 > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    solve()
```
