```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    queries = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        queries.append((p1, p2, i + 1))
        idx += 2
        
    # Scenario A: Each query i finishes at p1_i + p2_i.
    # If any p1_i + p2_i > T, it fails.
    for i in range(j):
        if queries[i][0] + queries[i][1] > t:
            print(f"JOB {queries[i][2]}")
            return
            
    # Scenario B: Component 1 is the bottleneck.
    # Total time = sum(p1_i) + max(p2_i) is not correct.
    # Actually, in Scenario B, Component 1 processes all p1 sequentially.
    # Component 2 can process all p2 in parallel starting at time 0.
    # So the last query finishes Component 1 at sum(p1_i).
    # The last query finishes Component 2 at p2_i.
    # The total time is max(sum(p1_i), max(p2_i)).
    sum_p1 = sum(q[0] for q in queries)
    max_p2 = max(q[1] for q in queries)
    if max(sum_p1, max_p2) > t:
        print("LOAD 1")
        return
        
    # Scenario C: Component 2 is the bottleneck.
    # Similar to B, total time = max(sum(p2_i), max(p1_i)).
    sum_p2 = sum(q[1] for q in queries)
    max_p1 = max(q[0] for q in queries)
    if max(sum_p2, max_p1) > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
