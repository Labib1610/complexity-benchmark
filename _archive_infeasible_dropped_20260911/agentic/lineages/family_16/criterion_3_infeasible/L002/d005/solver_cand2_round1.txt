```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for k in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        p1_list.append(p1)
        p2_list.append(p2)
        
        # Type A check: f_k(x) = min(x, p1) + min(T-x, p2)
        # The maximum value of f_k(x) is min(p1 + p2, T)
        # We need f_k(x) = p1 + p2.
        # This is possible if and only if p1 + p2 <= T.
        if p1 + p2 > T:
            print(f"JOB {k}")
            return
            
    # Type B check: sum of p1_k <= T
    if sum(p1_list) > T:
        print("LOAD 1")
        return
        
    # Type C check: sum of p2_k <= T
    if sum(p2_list) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
