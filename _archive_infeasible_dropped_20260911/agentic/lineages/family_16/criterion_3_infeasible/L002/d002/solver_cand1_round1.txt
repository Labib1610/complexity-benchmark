```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        jobs.append((p1, p2, i))
        idx += 2
        
    # Check JOB i condition
    for p1, p2, i in jobs:
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
            
    # Check LOAD 1 condition
    sum_p1 = 0
    for p1, p2, i in jobs:
        sum_p1 += p1
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2 condition
    sum_p2 = 0
    for p1, p2, i in jobs:
        sum_p2 += p2
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
