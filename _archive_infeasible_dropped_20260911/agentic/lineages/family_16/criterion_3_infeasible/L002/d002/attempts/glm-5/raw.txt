```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    T = int(data[1])
    idx = 2
    for i in range(1, j + 1):
        p1 = int(data[idx])
        p2 = int(data[idx + 1])
        idx += 2
        if T - p1 < p2:
            print(f"JOB {i}")
            return
    # Check LOAD
    budget1 = T
    budget2 = T
    idx = 2
    for i in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx + 1])
        idx += 2
        budget1 -= p1
        budget2 -= p2
        if budget1 < 0:
            print("LOAD 1")
            return
        if budget2 < 0:
            print("LOAD 2")
            return
    # Should not reach here per problem guarantees
    print("LOAD 1")

if __name__ == "__main__":
    main()
```
