```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    
    jobs = []
    oven_total = 0
    broiler_total = 0
    
    index = 2
    for i in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        index += 2
        jobs.append((p1, p2))
        oven_total += p1
        broiler_total += p2
        
        # Check if this job alone cannot fit
        if p1 + p2 > T:
            print(f"JOB {i+1}")
            return
    
    # Check if oven is overloaded
    if oven_total > T:
        print("LOAD 1")
        return
        
    # Check if broiler is overloaded
    if broiler_total > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    main()
```
