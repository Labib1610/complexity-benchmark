```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for i in range(j):
        val1 = int(input_data[idx])
        val2 = int(input_data[idx+1])
        p1.append(val1)
        p2.append(val2)
        
        # Check JOB i
        # V_S,i >= p1,i
        # V_T,i - V_S,i >= p1,i + p2,i => V_T,i >= V_S,i + p1,i + p2,i >= 2*p1,i + p2,i
        # V_T,i <= 2*T - p2,i
        # Unsatisfiable if 2*p1,i + p2,i > 2*T - p2,i
        # 2*p1,i + 2*p2,i > 2*T => p1,i + p2,i > T
        if val1 + val2 > T:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1
    # V_S,i in [p1,i, 2*T - p1,i]
    # |V_S,i - V_S,k| >= p1,i + p1,k
    # Let x_i = V_S,i. Sort by p1,i.
    # This is equivalent to packing intervals of length p1,i + p1,k.
    # A known result for this specific constraint:
    # If we sort such that p1_1 <= p1_2 <= ... <= p1_j,
    # the condition is sum_{i=1}^j (2*p1_i) <= 2*T - 2*p1_1 (or similar).
    # Actually, the condition for existence is:
    # sum_{i=1}^j 2*p1_i <= 2*T - 2*max(p1_i) is not quite right.
    # The condition is: sum_{i=1}^j 2*p1_i - 2*max(p1_i) <= 2*T - 2*max(p1_i)
    # which simplifies to sum_{i=1}^j p1_i <= T.
    if sum(p1) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    if sum(p2) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
