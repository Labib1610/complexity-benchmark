```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    clients = []
    idx = 2
    for i in range(j):
        e1 = int(input_data[idx])
        e2 = int(input_data[idx+1])
        clients.append((e1, e2, i + 1))
        idx += 2
        
    # Check JOB i
    # u_i >= E1_i, v_i - u_i >= E1_i + E2_i, v_i <= 2T - E2_i
    # v_i >= u_i + E1_i + E2_i >= E1_i + E1_i + E2_i = 2*E1_i + E2_i
    # So we need 2*E1_i + E2_i <= 2T - E2_i => 2*E1_i + 2*E2_i <= 2T => E1_i + E2_i <= T
    # If E1_i + E2_i > T, then JOB i is a valid certificate.
    for i in range(j):
        e1, e2, original_idx = clients[i]
        if e1 + e2 > T:
            print(f"JOB {original_idx}")
            return

    # Check LOAD 1
    # u_i >= E1_i, u_i <= 2T - E1_i, u_y - u_x >= E1_x + E1_y
    # Let u'_i = u_i - E1_i. Then u'_i >= 0.
    # u_y - u_x >= E1_x + E1_y => (u'_y + E1_y) - (u'_x + E1_x) >= E1_x + E1_y
    # u'_y - u'_x >= 2*E1_x.
    # Summing over permutation: u'_last - u'_first = sum_{m=1}^{j-1} (u'_{pi[m+1]} - u'_{pi[m]})
    # >= sum_{m=1}^{j-1} 2*E1_{pi[m]}
    # Max possible value of u'_last - u'_first is (2T - E1_last - E1_last) - 0 = 2T - 2*E1_last
    # Actually, the condition is: there exists a permutation such that sum of 2*E1_x for all but last is <= 2T - 2*E1_last
    # sum_{all} 2*E1_i <= 2T.
    # If sum(2*E1_i) > 2T, then LOAD 1 is a valid certificate.
    if sum(2 * c[0] for c in clients) > 2 * T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # v_i >= E2_i, v_i <= 2T - E2_i, v_y - v_x >= E2_x + E2_y
    # Similar logic: sum(2*E2_i) > 2T
    if sum(2 * c[1] for c in clients) > 2 * T:
        print("LOAD 2")
        return

solve()
```
