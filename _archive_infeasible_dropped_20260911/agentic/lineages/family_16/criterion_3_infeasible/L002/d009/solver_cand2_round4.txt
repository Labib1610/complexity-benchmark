```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    W = int(input_data[1])
    
    A = []
    B = []
    idx = 2
    for i in range(N):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        A.append(a)
        B.append(b)
        idx += 2
        
        # Check JOB i
        # R2 - L2 = B, R1 - L1 = A, L2 >= R1, L1 >= 0, R2 <= W
        # R2 = L2 + B >= R1 + B >= L1 + A + B >= A + B
        # So if A + B > W, JOB i is a certificate
        if a + b > W:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Sum of A_i > W
    if sum(A) > W:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sum of B_i > W
    if sum(B) > W:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
