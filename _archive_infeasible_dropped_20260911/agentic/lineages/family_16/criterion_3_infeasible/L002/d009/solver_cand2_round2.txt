```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    
    idx = 2
    for i in range(j):
        val1 = int(input_data[idx])
        val2 = int(input_data[idx+1])
        p1.append(val1)
        p2.append(val2)
        
        # Check JOB i
        # a_i + b_i <= 2T - 2p1 - 2p2
        # Since a_i >= 0, b_i >= 0, a_i + b_i >= 0
        # If 2T - 2p1 - 2p2 < 0, then JOB i is a certificate
        if 2 * T - 2 * val1 - 2 * val2 < 0:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1
    # Constraints: a_i >= 0, a_i <= 2T - 2p1_i, and A-Dispersion
    # Sort by p1_i ascending. If we have sorted p1_i as q1, q2, ..., qj
    # The dispersion condition implies a_(k) >= a_(k-1) + 2*p1_(k-1)
    # This leads to a_(j) >= a_(1) + sum_{k=1}^{j-1} 2*p1_(k)
    # Since a_(1) >= 0, we need a_(j) >= sum_{k=1}^{j-1} 2*p1_(k)
    # Also a_(j) <= 2T - 2*p1_(j)
    # So if sum_{k=1}^{j-1} 2*p1_(k) > 2T - 2*p1_(j), it's impossible.
    
    def check_load(p_list, c):
        sorted_p = sorted(p_list)
        # sum of 2 * p_k for k=0 to j-2
        # must be <= 2T - 2 * p_last
        # sum_{k=0}^{j-2} 2 * sorted_p[k] > 2T - 2 * sorted_p[j-1]
        current_sum = 0
        for k in range(len(sorted_p) - 1):
            current_sum += 2 * sorted_p[k]
        
        if current_sum > 2 * T - 2 * sorted_p[-1]:
            print(f"LOAD {c}")
            return True
        return False

    if check_load(p1, 1):
        return
    if check_load(p2, 2):
        return

solve()
```
