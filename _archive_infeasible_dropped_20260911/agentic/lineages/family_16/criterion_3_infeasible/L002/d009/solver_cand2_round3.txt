```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    pairs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        pairs.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Transmitter center u_i must be >= p1_i
    # Receiver center v_i must be <= 2T - p2_i
    # Intra-pair separation: v_i - u_i >= p1_i + p2_i
    # So: (2T - p2_i) - p1_i >= p1_i + p2_i
    # 2T >= 2*p1_i + 2*p2_i => T >= p1_i + p2_i
    for i in range(j):
        p1, p2 = pairs[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1: Transmitters
    # Each transmitter i occupies [u_i - p1_i, u_i + p1_i]
    # Contained in [0, 2T] => p1_i <= u_i <= 2T - p1_i
    # Isolation: |u_i - u_k| >= p1_i + p1_k
    # Let x_i = u_i - p1_i. Then 0 <= x_i and x_i + 2*p1_i <= 2T
    # Isolation: |(x_i + p1_i) - (x_k + p1_k)| >= p1_i + p1_k
    # This is equivalent to packing segments of length 2*p1_i into [0, 2T]
    # Sum of lengths must be <= 2T
    sum_p1 = sum(2 * p[0] for p in pairs)
    if sum_p1 > 2 * T:
        print("LOAD 1")
        return
        
    # Check LOAD 2: Receivers
    sum_p2 = sum(2 * p[1] for p in pairs)
    if sum_p2 > 2 * T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
