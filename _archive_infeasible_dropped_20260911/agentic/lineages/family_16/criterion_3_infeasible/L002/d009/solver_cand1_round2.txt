```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    q = []
    idx = 2
    for i in range(j):
        q1 = int(input_data[idx])
        q2 = int(input_data[idx+1])
        q.append((q1, q2))
        idx += 2
        
    # Check JOB i
    # w1 >= q1, w2 <= 2T - q2, w2 - w1 >= q1 + q2
    # w2 >= w1 + q1 + q2 >= q1 + q1 + q2 = 2*q1 + q2
    # So 2*q1 + q2 <= 2T - q2  => 2*q1 + 2*q2 <= 2T => q1 + q2 <= T
    # If q1 + q2 > T, then JOB i is impossible.
    for i in range(j):
        if q[i][0] + q[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # S(w, q) >= 0 means q <= w <= 2T - q
    # The interval length for element i is (2T - q) - q = 2T - 2q
    # Dispersion: |w_i - w_k| >= q_i + q_k
    # This is equivalent to packing intervals [q_i, 2T - q_i]
    # The total length required for n elements is sum(2*q_i) + sum_{i=1}^{n-1} (q_i + q_{i+1})
    # Actually, for a set of elements, they can be placed if:
    # sum(2*q_i) <= 2T - (q_min_first + q_min_last)
    # A simpler condition for LOAD c:
    # Sort elements by q_c,i. Let them be q_{(1)} <= q_{(2)} <= ... <= q_{(j)}
    # The condition for existence is:
    # 2 * sum(q_{(i)}) + sum_{i=1}^{j-1} (q_{(i)} + q_{(i+1)}) <= 2T
    # Which simplifies to: 2*q_{(1)} + 4*sum_{i=2}^{j-1} q_{(i)} + 2*q_{(j)} <= 2T
    
    for c in range(2):
        qc = sorted([x[c] for x in q])
        # The minimum length required to place j elements is:
        # q_{(1)} + (q_{(1)} + q_{(2)}) + (q_{(2)} + q_{(3)}) + ... + (q_{(j-1)} + q_{(j)}) + q_{(j)}
        # = 2 * sum(qc)
        if 2 * sum(qc) > 2 * T:
            print(f"LOAD {c + 1}")
            return

if __name__ == '__main__':
    solve()
```
