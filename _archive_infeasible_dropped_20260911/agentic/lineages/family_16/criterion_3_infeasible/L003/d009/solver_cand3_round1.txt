```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        idx += 2
        
    # Check JOB i
    # The minimal length for team i is p1_i + p2_i.
    # The problem states the chain is: start gap (0) + p1_i + transit (0) + p2_i + end gap (0)
    # So the sum is p1_i + p2_i.
    for i in range(j):
        if p1_list[i] + p2_list[i] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Sum of all p1_i must be <= T
    if sum(p1_list) > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sum of all p2_i must be <= T
    if sum(p2_list) > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
