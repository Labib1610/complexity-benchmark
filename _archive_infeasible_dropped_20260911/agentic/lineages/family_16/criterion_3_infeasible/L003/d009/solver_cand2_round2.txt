```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        idx += 2
        
    # Check JOB i:
    # For each product i, check if p_{i,1} + p_{i,1} + p_{i,2} > T
    # Wait, the condition is x_{i,2} - x_{i,1} >= p_{i,1}
    # x_{i,1} >= 0, x_{i,2} + p_{i,2} <= T
    # So x_{i,2} >= x_{i,1} + p_{i,1}
    # x_{i,2} + p_{i,2} >= x_{i,1} + p_{i,1} + p_{i,2}
    # T >= x_{i,2} + p_{i,2} >= p_{i,1} + p_{i,2}
    # Actually, the condition is x_{i,2} - x_{i,1} >= p_{i,1}
    # x_{i,1} >= 0, x_{i,2} + p_{i,2} <= T
    # x_{i,2} >= x_{i,1} + p_{i,1} >= p_{i,1}
    # x_{i,2} + p_{i,2} >= p_{i,1} + p_{i,2}
    # So if p_{i,1} + p_{i,2} > T, it's impossible.
    for i in range(j):
        if p1_list[i] + p2_list[i] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c:
    # Sum of lengths on line c must be <= T
    sum1 = sum(p1_list)
    if sum1 > T:
        print("LOAD 1")
        return
        
    sum2 = sum(p2_list)
    if sum2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
