```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1 = []
    p2 = []
    
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(j):
        v1 = int(input_data[2 + 2 * i])
        v2 = int(input_data[3 + 2 * i])
        p1.append(v1)
        p2.append(v2)
        sum_p1 += v1
        sum_p2 += v2
        
        # Check JOB i: p_{i,1} + p_{i,2} > T
        if v1 + v2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1: sum(p_{i,1}) > T
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2: sum(p_{i,2}) > T
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
