```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    T = int(data[1])
    
    # Check JOB condition: p1 + p2 > T
    for i in range(j):
        p1 = int(data[2 + 2*i])
        p2 = int(data[3 + 2*i])
        if p1 + p2 > T:
            print(f"JOB {i+1}")
            return
    
    # Check LOAD conditions
    sum1 = 0
    sum2 = 0
    for i in range(j):
        p1 = int(data[2 + 2*i])
        p2 = int(data[3 + 2*i])
        sum1 += p1
        sum2 += p2
    
    if sum1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

solve()
```
