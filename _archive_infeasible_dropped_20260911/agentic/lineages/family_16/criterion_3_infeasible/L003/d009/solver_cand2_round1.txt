```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    pairs = []
    idx = 2
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        pairs.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # JOB i is impossible if p_{i,1} + p_{i,2} > T
    for i in range(j):
        if pairs[i][0] + pairs[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # LOAD 1 is impossible if sum(p_{i,1}) > T
    sum_p1 = sum(p[0] for p in pairs)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # LOAD 2 is impossible if sum(p_{i,2}) > T
    sum_p2 = sum(p[1] for p in pairs)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
