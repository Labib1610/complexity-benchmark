import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t_meters = int(input_data[1])
    t_cm = t_meters * 100
    
    # The problem states T is in meters, but the constraints on p are in cm.
    # Wait, the problem says T is in meters and p is in meters? 
    # Let's re-read: "Each lane has a usable parking length of exactly T meters."
    # "Each team i consists of a primary vehicle of length p_{i,1} meters..."
    # So T is in meters, p is in meters. No conversion needed.
    
    t = t_meters
    
    # Check JOB i
    # Team i isolation: 2*C1 >= p1, 2*C2 <= 2T - p2, 2*(C2 - C1) >= p1 + p2
    # Summing: 2*C2 - 2*C1 >= p1 + p2
    # 2*C2 >= 2*C1 + p1 + p2 >= p1 + p1 + p2 = 2*p1 + p2
    # Also 2*C2 <= 2T - p2
    # So 2*p1 + p2 <= 2T - p2 => 2*p1 + 2*p2 <= 2T => p1 + p2 > T
    
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Lane 1 isolation: sum of p_{i,1} > T
    sum_p1 = 0
    for i in range(j):
        sum_p1 += int(input_data[2 + 2 * i])
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Lane 2 isolation: sum of p_{i,2} > T
    sum_p2 = 0
    for i in range(j):
        sum_p2 += int(input_data[3 + 2 * i])
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
