```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    
    idx = 2
    for _ in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Check JOB i:
    # A system JOB i is impossible if p_{i,1} + p_{i,2} > T
    # because x_{i,1} >= 0, x_{i,1} + p_{i,1} <= x_{i,2}, x_{i,2} + p_{i,2} <= T
    # implies p_{i,1} + p_{i,2} <= T.
    for i in range(j):
        if p1[i] + p2[i] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c:
    # A system LOAD c is impossible if sum(p_{i,c}) > T
    # because all filters of phase c must run sequentially within [0, T].
    sum_p1 = sum(p1)
    sum_p2 = sum(p2)
    
    if sum_p1 > T:
        print("LOAD 1")
        return
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
