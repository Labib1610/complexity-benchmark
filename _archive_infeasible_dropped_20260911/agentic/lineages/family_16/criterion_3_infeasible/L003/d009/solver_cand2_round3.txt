```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    alphas = []
    betas = []
    
    idx = 2
    for i in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        alphas.append(a)
        betas.append(b)
        
        # Check JOB i
        if a + b > T:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1
    if sum(alphas) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    if sum(betas) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
