```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1 = []
    p2 = []
    
    idx = 2
    for i in range(j):
        val1 = int(input_data[idx])
        val2 = int(input_data[idx+1])
        p1.append(val1)
        p2.append(val2)
        
        # Check JOB i
        # E_{i,2} >= S_{i,2} + p_{i,2} >= E_{i,1} + p_{i,2} >= S_{i,1} + p_{i,1} + p_{i,2}
        # Since S_{i,1} >= 0, E_{i,2} >= p_{i,1} + p_{i,2}
        if val1 + val2 > t:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1
    # Sum of spans in domain 1 must be <= T
    if sum(p1) > t:
        print("LOAD 1")
        return
    
    # Check LOAD 2
    # Sum of spans in domain 2 must be <= T
    if sum(p2) > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
