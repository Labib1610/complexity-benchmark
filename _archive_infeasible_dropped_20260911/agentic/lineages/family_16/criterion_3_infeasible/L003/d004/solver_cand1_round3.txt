```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    row1_sum = 0
    row2_sum = 0
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB condition
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        row1_sum += p1
        row2_sum += p2
        
    # Check LOAD condition
    if row1_sum > t:
        print("LOAD 1")
    elif row2_sum > t:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
