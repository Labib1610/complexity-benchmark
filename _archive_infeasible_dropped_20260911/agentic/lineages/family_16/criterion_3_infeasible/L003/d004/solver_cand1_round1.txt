```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    pairs = []
    idx = 2
    sum_u = 0
    sum_v = 0
    
    for i in range(j):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        pairs.append((u, v))
        
        # Check JOB i condition
        if u + v > t:
            print(f"JOB {i + 1}")
            return
        
        sum_u += u
        sum_v += v
        idx += 2
        
    # The integral of sum F_k(x - S_k) is sum(u_k)
    # The integral of sum G_k(x - R_k) is sum(v_k)
    if sum_u > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
