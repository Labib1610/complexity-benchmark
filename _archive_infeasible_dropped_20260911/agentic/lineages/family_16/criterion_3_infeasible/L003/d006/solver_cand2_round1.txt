```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    D = []
    idx = 2
    for i in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        D.append((d1, d2))
        idx += 2
        
    # Check JOB i
    # JOB i is infeasible if D[i][0] + D[i][1] > T
    for i in range(j):
        if D[i][0] + D[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # LOAD c is infeasible if sum of D[i][c] for all i > T
    # Constraint (E) for LOAD c: max(x_u,c - y_v,c, x_v,c - y_u,c) >= 0
    # This is equivalent to saying that for any two jobs u, v, 
    # either x_u,c >= y_v,c or x_v,c >= y_u,c.
    # Combined with y_i,c - x_i,c >= D_i,c, this implies that the intervals
    # [x_i,c, y_i,c] of length D_i,c must be packed into [0, T] without overlap.
    # This is possible if and only if sum(D_i,c) <= T.
    
    sum1 = sum(d[0] for d in D)
    if sum1 > T:
        print("LOAD 1")
        return
        
    sum2 = sum(d[1] for d in D)
    if sum2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
