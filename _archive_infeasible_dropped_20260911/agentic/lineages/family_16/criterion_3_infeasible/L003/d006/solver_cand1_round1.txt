```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    d1 = []
    d2 = []
    
    idx = 2
    for i in range(j):
        val1 = int(input_data[idx])
        val2 = int(input_data[idx+1])
        d1.append(val1)
        d2.append(val2)
        idx += 2
        
        # Check JOB i: Measure Axiom, Precedence Axiom, Bounding Axiom
        # mu(I_{i,1}) >= D_{i,1}, mu(I_{i,2}) >= D_{i,2}
        # sup(I_{i,1}) <= inf(I_{i,2})
        # I_{i,1}, I_{i,2} subset [0, T]
        # This implies D_{i,1} + D_{i,2} <= T
        if val1 + val2 > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD c: Measure Axiom, Disjointness Axiom, Bounding Axiom
    # For a fixed c, sum_{i=1}^j mu(I_{i,c}) >= sum_{i=1}^j D_{i,c}
    # Since I_{i,c} subset [0, T] and disjoint, sum mu(I_{i,c}) <= T
    # So sum D_{i,c} <= T
    
    sum1 = sum(d1)
    if sum1 > T:
        print("LOAD 1")
        return
        
    sum2 = sum(d2)
    if sum2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
