```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    durations = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        durations.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # A signal i is impossible if D_{i,1} + D_{i,2} > T
    for i in range(j):
        if durations[i][0] + durations[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Stage 1 is impossible if sum(D_{i,1}) > T
    sum1 = sum(d[0] for d in durations)
    if sum1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Stage 2 is impossible if sum(D_{i,2}) > T
    sum2 = sum(d[1] for d in durations)
    if sum2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
