```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    durations = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        durations.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Constraint: D_{i,1} + D_{i,2} <= T
    for i in range(j):
        if durations[i][0] + durations[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Constraint: sum(D_{i,1}) <= T
    sum1 = 0
    for i in range(j):
        sum1 += durations[i][0]
        if sum1 > t:
            print("LOAD 1")
            return
            
    # Check LOAD 2
    # Constraint: sum(D_{i,2}) <= T
    sum2 = 0
    for i in range(j):
        sum2 += durations[i][1]
        if sum2 > t:
            print("LOAD 2")
            return

if __name__ == '__main__':
    solve()
```
