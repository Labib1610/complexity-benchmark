```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    T = int(input_data[1])
    
    p = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p.append((p1, p2))
        idx += 2
        
    # Check JOB i
    for i in range(j):
        p1, p2 = p[i]
        # Condition 1: v >= p1, Condition 2: z >= p2
        # Condition 3: u >= v, Condition 4: w + z <= 2T
        # Condition 5: w - z >= u + v
        # From 3 and 5: w - z >= 2v => w >= 2v + z
        # From 4: w + z <= 2T => (2v + z) + z <= 2T => 2v + 2z <= 2T => v + z <= T
        # Contradiction if p1 + p2 > T
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Condition 1: v_i >= p1_i, Condition 3: u_i >= v_i
    # Condition 6: u_i + v_i <= u_k - v_k or u_k + v_k <= u_i - v_i
    # Boundary: u_i + v_i <= 2T
    # Let x_i = u_i - v_i, y_i = u_i + v_i. 
    # Condition 3: x_i >= 0. Condition 6: intervals [x_i, y_i] are disjoint.
    # Total length of intervals is sum(y_i - x_i) = sum(2*v_i) >= sum(2*p1_i).
    # The intervals are in [0, 2T].
    # If sum(2*p1_i) > 2T, then LOAD 1 is impossible.
    sum_p1 = sum(item[0] for item in p)
    if 2 * sum_p1 > 2 * T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Condition 2: z_i >= p2_i, Condition 4: w_i + z_i <= 2T
    # Condition 7: w_i + z_i <= w_k - z_k or w_k + z_k <= w_i - z_i
    # Boundary: w_i - z_i >= 0
    # Let a_i = w_i - z_i, b_i = w_i + z_i.
    # Condition 4: b_i <= 2T. Boundary: a_i >= 0.
    # Condition 7: intervals [a_i, b_i] are disjoint.
    # Length of interval [a_i, b_i] is b_i - a_i = 2*z_i >= 2*p2_i.
    # Total length sum(2*p2_i) must be <= 2T.
    sum_p2 = sum(item[1] for item in p)
    if 2 * sum_p2 > 2 * T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
