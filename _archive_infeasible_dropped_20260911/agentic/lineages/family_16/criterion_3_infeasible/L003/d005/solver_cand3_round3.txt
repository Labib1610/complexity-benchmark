import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    containers = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        containers.append((p1, p2))
        idx += 2
        
    # Precedence 1: Isolated Container Failure
    for i in range(j):
        if containers[i][0] + containers[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Precedence 2: Isolated Stage Bottleneck
    sum1 = 0
    sum2 = 0
    for p1, p2 in containers:
        sum1 += p1
        sum2 += p2
        
    if sum1 > t:
        print("LOAD 1")
        return
    if sum2 > t:
        print("LOAD 2")
        return
        
    # Precedence 3: Coupled Scheduling Conflict
    # Johnson's Rule / Flow Shop Scheduling
    # The minimum makespan for a 2-machine flow shop is found by sorting
    # containers into two groups: those with p1 < p2 (sorted by p1 ascending)
    # and those with p1 >= p2 (sorted by p2 descending).
    
    group1 = []
    group2 = []
    for p1, p2 in containers:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    schedule = group1 + group2
    
    # Calculate completion time
    time1 = 0
    time2 = 0
    for p1, p2 in schedule:
        time1 += p1
        if time2 < time1:
            time2 = time1 + p2
        else:
            time2 += p2
            
    if time2 > t:
        print("INTERLOCK")
    else:
        # This case should not be reached based on problem guarantee
        pass

if __name__ == '__main__':
    solve()
