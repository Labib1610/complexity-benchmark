```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check Single-Job Bottleneck
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        jobs.append((p1, p2))
        sum_p1 += p1
        sum_p2 += p2
        
    # Check Machine Overload
    if sum_p1 > t:
        print("LOAD 1")
        return
    if sum_p2 > t:
        print("LOAD 2")
        return
    
    # If not JOB or LOAD, it must be COMPLEX
    # Johnson's rule provides the minimum makespan for 2-machine flow shop.
    # The minimum makespan is the minimum time to complete all jobs.
    # If the minimum makespan > T, it's impossible.
    # Since the problem guarantees impossibility, if it's not JOB or LOAD,
    # it must be due to the sequential interaction (COMPLEX).
    print("COMPLEX")

if __name__ == '__main__':
    solve()
```
