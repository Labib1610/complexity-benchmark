import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    P = []
    idx = 2
    for _ in range(j):
        P.append((int(input_data[idx]), int(input_data[idx+1])))
        idx += 2
        
    # Check JOB i
    # JOB i: Y_{i,1} - X_{i,1} >= P_{i,1}, Y_{i,2} - X_{i,2} >= P_{i,2}, X_{i,2} >= Y_{i,1}, Y_{i,c} <= T
    # Y_{i,2} >= X_{i,2} + P_{i,2} >= Y_{i,1} + P_{i,2} >= X_{i,1} + P_{i,1} + P_{i,2}
    # Since X_{i,1} >= 0, Y_{i,2} >= P_{i,1} + P_{i,2}
    # Contradiction if P_{i,1} + P_{i,2} > T
    for i in range(j):
        if P[i][0] + P[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # LOAD c: Y_{i,c} - X_{i,c} >= P_{i,c}, Y_{i,c} <= T, X_{i,c} >= 0
    # Family 5: max(X_{i,c} - Y_{k,c}, X_{k,c} - Y_{i,c}) >= 0
    # This implies that for any i, k, either X_{i,c} >= Y_{k,c} or X_{k,c} >= Y_{i,c}.
    # This means the intervals [X_{i,c}, Y_{i,c}] must be disjoint (or touch).
    # Since X_{i,c} >= 0 and Y_{i,c} - X_{i,c} >= P_{i,c}, the total length is sum(P_{i,c}).
    # The maximum Y_{i,c} must be at least sum(P_{i,c}).
    # Contradiction if sum(P_{i,c}) > T
    
    sum1 = sum(p[0] for p in P)
    if sum1 > T:
        print("LOAD 1")
        return
        
    sum2 = sum(p[1] for p in P)
    if sum2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
