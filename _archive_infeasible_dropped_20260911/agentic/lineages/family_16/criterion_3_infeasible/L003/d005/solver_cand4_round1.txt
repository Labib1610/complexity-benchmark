```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    particles = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        particles.append((p1, p2))
        idx += 2
        
    # Check JOB i:
    # For particle i, G_i = [a, b], X_i = [c, d]
    # Widths: b - a >= p1, d - c >= p2
    # Ordering: max(G_i) <= min(X_i) => b <= c
    # Bounds: 0 <= a, d <= T
    # Combining: d >= c + p2 >= b + p2 >= a + p1 + p2 >= p1 + p2
    # If p1 + p2 > T, then JOB i is impossible.
    for i in range(j):
        if particles[i][0] + particles[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c:
    # For state 1: sum of p1_i <= T
    # For state 2: sum of p2_i <= T
    sum1 = 0
    sum2 = 0
    for p1, p2 in particles:
        sum1 += p1
        sum2 += p2
        
    if sum1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
