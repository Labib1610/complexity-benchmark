import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    d = []
    idx = 2
    for i in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        d.append((d1, d2))
        idx += 2
        
    # Check JOB i
    # Constraints: y_{i,2} >= x_{i,2} + d_{i,2}, x_{i,2} >= y_{i,1}, y_{i,1} >= x_{i,1} + d_{i,1}
    # So y_{i,2} >= x_{i,1} + d_{i,1} + d_{i,2}. Since x_{i,1} >= 0, y_{i,2} >= d_{i,1} + d_{i,2}.
    # If d_{i,1} + d_{i,2} > T, then JOB i is inconsistent.
    for i in range(j):
        if d[i][0] + d[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # Constraints: y_{i,c} >= x_{i,c} + d_{i,c}, y_{i,c} <= T, x_{i,c} >= 0
    # Exclusion: intervals [x_{i,c}, y_{i,c}) are disjoint.
    # Total length required is sum(d_{i,c}) <= T.
    
    sum_d1 = 0
    sum_d2 = 0
    for i in range(j):
        sum_d1 += d[i][0]
        sum_d2 += d[i][1]
        
    if sum_d1 > T:
        print("LOAD 1")
        return
    if sum_d2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
