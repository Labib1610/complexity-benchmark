```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    deltas = []
    idx = 2
    for i in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        deltas.append((d1, d2))
        idx += 2
        
    # Check JOB i
    # Constraints: x >= 0, y1 - x1 >= d1, y2 - x2 >= d2, x2 - y1 >= 0, y1 <= T, y2 <= T
    # y2 >= x2 + d2 >= y1 + d2 >= x1 + d1 + d2 >= d1 + d2
    # So if d1 + d2 > T, then y2 > T, contradiction.
    for i in range(j):
        if deltas[i][0] + deltas[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # Constraints: x_i >= 0, y_i - x_i >= d_i, y_i <= T, |y_i - y_k| >= 0 (separation)
    # For a fixed dimension c, let the sorted deltas be d_(1) <= d_(2) <= ... <= d_(j)
    # The constraints imply y_(k) >= y_(k-1) + d_(k)
    # Thus y_(j) >= sum(d_i for all i)
    # If sum(d_i) > T, then y_(j) > T, contradiction.
    
    sum1 = sum(d[0] for d in deltas)
    if sum1 > T:
        print("LOAD 1")
        return
        
    sum2 = sum(d[1] for d in deltas)
    if sum2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
