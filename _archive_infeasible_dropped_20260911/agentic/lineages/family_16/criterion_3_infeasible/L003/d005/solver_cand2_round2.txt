```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    durations = []
    idx = 2
    sum_col1 = 0
    sum_col2 = 0
    
    for i in range(1, j + 1):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i condition
        if d1 + d2 > t:
            print(f"JOB {i}")
            return
        
        sum_col1 += d1
        sum_col2 += d2
        
    # Check LOAD c conditions
    if sum_col1 > t:
        print("LOAD 1")
    elif sum_col2 > t:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
