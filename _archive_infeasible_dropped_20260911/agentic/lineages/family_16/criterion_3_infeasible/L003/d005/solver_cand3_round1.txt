import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    t = int(input_data[1])
    
    p = []
    idx = 2
    for i in range(j):
        p.append((int(input_data[idx]), int(input_data[idx+1])))
        idx += 2
        
    # Check JOB i
    for i in range(j):
        # x_{i,1} >= 0, y_{i,1} = x_{i,1} + p_{i,1} <= T
        # x_{i,2} >= y_{i,1} = x_{i,1} + p_{i,1}
        # y_{i,2} = x_{i,2} + p_{i,2} <= T
        # So x_{i,1} + p_{i,1} + p_{i,2} <= T
        # Since x_{i,1} >= 0, we need p_{i,1} + p_{i,2} <= T
        if p[i][0] + p[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # For a fixed c, we have y_{i,c} - x_{i,c} >= p_{i,c}
    # Disjunction: for any i != k, x_{i,c} >= y_{k,c} or x_{k,c} >= y_{i,c}
    # This implies that the intervals [x_{i,c}, y_{i,c}] must be disjoint
    # except possibly at endpoints.
    # Total length of intervals is sum(p_{i,c}).
    # The union of intervals must fit in [0, T].
    # So sum(p_{i,c}) <= T.
    for c in range(2):
        total_p = sum(p[i][c] for i in range(j))
        if total_p > t:
            print(f"LOAD {c + 1}")
            return
            
    # Check SYSTEM
    # We need to satisfy:
    # 1. y_{i,1} - x_{i,1} >= p_{i,1}
    # 2. y_{i,2} - x_{i,2} >= p_{i,2}
    # 3. x_{i,2} >= y_{i,1}
    # 4. Intervals [x_{i,1}, y_{i,1}] disjoint, [x_{i,2}, y_{i,2}] disjoint
    # 5. 0 <= x_{i,c} < y_{i,c} <= T
    # Let S1 = sum(p_{i,1}), S2 = sum(p_{i,2})
    # The chain condition x_{i,2} >= y_{i,1} implies that the union of [x_{i,2}, y_{i,2}]
    # must be "after" the union of [x_{i,1}, y_{i,1}].
    # Specifically, if we order indices such that [x_{i,1}, y_{i,1}] are sorted,
    # then the total span required is at least S1 + S2.
    # Actually, the condition is: there exist disjoint intervals I_{i,1} and I_{i,2}
    # such that |I_{i,c}| = p_{i,c} and for all i, min(I_{i,2}) >= max(I_{i,1}).
    # This is possible if and only if sum(p_{i,1}) + sum(p_{i,2}) <= T.
    s1 = sum(p[i][0] for i in range(j))
    s2 = sum(p[i][1] for i in range(j))
    if s1 + s2 > t:
        print("SYSTEM")
        return

solve()
