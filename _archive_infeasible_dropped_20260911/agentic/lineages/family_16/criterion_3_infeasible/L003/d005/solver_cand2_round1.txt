import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    A = []
    B = []
    idx = 2
    for i in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        A.append(a)
        B.append(b)
        idx += 2
        
    # Check JOB i
    # C1: X_i >= A_i, X_i <= 2T - A_i
    # C2: Y_i >= B_i, Y_i <= 2T - B_i
    # C3: Y_i - X_i >= A_i + B_i => Y_i >= X_i + A_i + B_i
    # Combined: X_i + A_i + B_i <= Y_i <= 2T - B_i
    # So X_i + A_i + B_i <= 2T - B_i => X_i <= 2T - A_i - 2B_i
    # Also X_i >= A_i. So A_i <= 2T - A_i - 2B_i => 2A_i + 2B_i <= 2T => A_i + B_i <= T
    # If A_i + B_i > T, then JOB i is impossible.
    for i in range(j):
        if A[i] + B[i] > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD 1: X_i >= A_i, X_i <= 2T - A_i, |X_i - X_k| >= A_i + A_k
    # Let X'_i = X_i - A_i. Then X'_i >= 0.
    # X_i <= 2T - A_i => X'_i <= 2T - 2A_i.
    # |(X'_i + A_i) - (X'_k + A_k)| >= A_i + A_k
    # If we sort such that X'_1 <= X'_2 <= ... <= X'_j,
    # then X'_{i+1} - X'_i >= A_{i+1} + A_i.
    # Summing: X'_j - X'_1 = sum_{i=1}^{j-1} (X'_{i+1} - X'_i) >= sum_{i=1}^{j-1} (A_{i+1} + A_i)
    # X'_j >= X'_1 + A_1 + 2*A_2 + 2*A_3 + ... + 2*A_{j-1} + A_j
    # Since X'_1 >= 0, X'_j >= A_1 + 2*sum(A[1:j-1]) + A_j = 2*sum(A) - A_1 - A_j
    # Also X'_j <= 2T - 2*A_j.
    # So 2*sum(A) - A_1 - A_j <= 2T - 2*A_j => 2*sum(A) - A_1 + A_j <= 2T
    # Actually, the condition for existence is:
    # Sort A such that A_1 <= A_2 <= ... <= A_j.
    # The tightest constraint is 2*sum(A) - A_1 - A_j <= 2T - 2*A_j (if we order by A_i)
    # Wait, the condition is: sum_{i=1}^j 2*A_i - A_1 - A_j <= 2T - 2*A_j? No.
    # The condition for existence is: 2 * sum(A) - A_1 - A_j <= 2T - 2*A_j is wrong.
    # The condition is: 2 * sum(A) - A_1 - A_j <= 2T - 2*A_j is not quite right.
    # The condition is: 2 * sum(A) <= 2T.
    
    def check_load(arr, T):
        arr.sort()
        # The minimum span required is sum_{i=1}^{j-1} (A_i + A_{i+1}) = 2*sum(A) - A_1 - A_j
        # The available span is (2T - A_j) - A_1 = 2T - A_1 - A_j
        # So 2*sum(A) - A_1 - A_j <= 2T - A_1 - A_j => 2*sum(A) <= 2T => sum(A) <= T
        return sum(arr) > T

    if check_load(A, T):
        print("LOAD 1")
    else:
        print("LOAD 2")

solve()
