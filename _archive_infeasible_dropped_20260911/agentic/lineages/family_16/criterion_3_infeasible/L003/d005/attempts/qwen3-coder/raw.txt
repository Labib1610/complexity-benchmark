```python
j, T = map(int, input().split())
D = [list(map(int, input().split())) for _ in range(j)]

# Check for JOB certificate
for i in range(j):
    if D[i][0] + D[i][1] > T:
        print(f"JOB {i+1}")
        exit()

# Check for LOAD certificate
sum_stage1 = sum(D[i][0] for i in range(j))
sum_stage2 = sum(D[i][1] for i in range(j))

if sum_stage1 > T:
    print("LOAD 1")
    exit()
elif sum_stage2 > T:
    print("LOAD 2")
    exit()
```
