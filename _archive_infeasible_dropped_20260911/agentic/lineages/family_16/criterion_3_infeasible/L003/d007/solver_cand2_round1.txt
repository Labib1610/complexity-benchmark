import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Delta(i) = 0 implies:
    # Y_{i,1} - X_{i,1} = P_{i,1}
    # Y_{i,2} - X_{i,2} = P_{i,2}
    # X_{i,2} = Y_{i,1}
    # Substituting: Y_{i,2} - X_{i,1} = P_{i,1} + P_{i,2}
    # Since 0 <= X_{i,1} <= Y_{i,1} <= T and 0 <= X_{i,2} <= Y_{i,2} <= T
    # We have Y_{i,2} = X_{i,1} + P_{i,1} + P_{i,2}
    # Since X_{i,1} >= 0, we need Y_{i,2} >= P_{i,1} + P_{i,2}
    # Also Y_{i,2} <= T. So if P_{i,1} + P_{i,2} > T, then JOB i is impossible.
    for i in range(j):
        if p[i][0] + p[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # Gamma_c(i, k) = 0 means intervals [X_{i,c}, Y_{i,c}] are disjoint.
    # For a fixed c, we have j intervals of length P_{i,c}.
    # The total length required is sum(P_{i,c}).
    # If sum(P_{i,c}) > T, then LOAD c is impossible.
    sum1 = sum(x[0] for x in p)
    sum2 = sum(x[1] for x in p)
    
    if sum1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
