```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    d1 = []
    d2 = []
    
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        d1.append(p1)
        d2.append(p2)
        
        # Check JOB i
        # Sum: (E1-S1) + (E2-S2) + (S2-E1) + (T-E2) + S1 >= D1 + D2 + 0 + 0 + 0
        # T >= D1 + D2
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1
    # Sum: sum(D1) <= T
    if sum(d1) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sum: sum(D2) <= T
    if sum(d2) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
