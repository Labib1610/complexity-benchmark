```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    tasks = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tasks.append((p1, p2, i + 1))
        idx += 2
        
    # Check JOB i: p1_i + p2_i > T
    for p1, p2, i in tasks:
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
            
    # Check LOAD 1: sum(p1_i) > T
    # Rule 1, 2, 5, and E1_i <= T implies sum(p1_i) <= T
    sum_p1 = sum(task[0] for task in tasks)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2: sum(p2_i) > T
    # Rule 1, 3, 6, and 7 implies sum(p2_i) <= T
    sum_p2 = sum(task[1] for task in tasks)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
