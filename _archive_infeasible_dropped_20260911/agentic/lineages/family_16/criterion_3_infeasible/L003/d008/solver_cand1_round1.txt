```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1 = []
    p2 = []
    
    idx = 2
    for i in range(j):
        v1 = int(input_data[idx])
        v2 = int(input_data[idx+1])
        p1.append(v1)
        p2.append(v2)
        idx += 2
        
    # Check JOB i
    # Team i isolation: p1[i] + p2[i] > T
    for i in range(j):
        if p1[i] + p2[i] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Lane 1 isolation: sum(p1) > T
    if sum(p1) > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Lane 2 isolation: sum(p2) > T
    if sum(p2) > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
