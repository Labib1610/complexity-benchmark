```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        idx += 2
        
        # Check JOB i: v_i - u_i = p1, z_i - w_i = p2, 0 <= u_i < v_i <= w_i < z_i <= T
        # This implies z_i - u_i = (z_i - w_i) + (w_i - v_i) + (v_i - u_i) = p2 + (w_i - v_i) + p1
        # Since w_i >= v_i, z_i - u_i >= p1 + p2.
        # If p1 + p2 > T, then JOB i is impossible.
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD 1: sum of p1_i > T
    if sum(p1_list) > T:
        print("LOAD 1")
        return
    
    # Check LOAD 2: sum of p2_i > T
    if sum(p2_list) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
