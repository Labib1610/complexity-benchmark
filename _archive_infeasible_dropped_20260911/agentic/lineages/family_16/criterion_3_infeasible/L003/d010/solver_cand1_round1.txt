import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    params = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        params.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Rule 1: end1 - start1 = p1, end2 - start2 = p2
    # Rule 3: 0 <= start1, end1 <= T, 0 <= start2, end2 <= T
    # Rule 5: end1 <= start2
    # Combined: 0 <= start1, start1 + p1 <= start2, start2 + p2 <= T
    # This is possible iff p1 + p2 <= T
    for i in range(j):
        if params[i][0] + params[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # Rule 1: end - start = p
    # Rule 3: 0 <= start, end <= T => 0 <= start, start + p <= T => 0 <= start <= T - p
    # Rule 4: |(startA + endA) - (startB + endB)| >= pA + pB
    # Let sum_coord = start + end = 2*start + p
    # |(2*startA + pA) - (2*startB + pB)| >= pA + pB
    # Let x = 2*start. 0 <= x <= 2*(T - p)
    # |(x_A + pA) - (x_B + pB)| >= pA + pB
    # This is equivalent to:
    # (x_A + pA) - (x_B + pB) >= pA + pB  => x_A - x_B >= 2*pB
    # OR (x_B + pB) - (x_A + pA) >= pA + pB => x_B - x_A >= 2*pA
    # This means the distance between x_A and x_B must be at least 2*max(pA, pB)
    # Actually, the condition |(2*startA + pA) - (2*startB + pB)| >= pA + pB
    # simplifies to |2*startA - 2*startB| >= pA + pB - (pA - pB) = 2*pB (if pA > pB)
    # Wait, the condition is |(2*startA + pA) - (2*startB + pB)| >= pA + pB
    # Let S_A = 2*startA + pA. S_A is in [pA, 2*T - pA].
    # |S_A - S_B| >= pA + pB.
    # This is a classic packing problem. If we sort elements by p, we can check feasibility.
    # For a set of elements with parameters p_i, we need to place S_i in [p_i, 2T - p_i]
    # such that |S_i - S_j| >= p_i + p_j.
    # This is possible iff sum(p_i) + sum(p_i for i < n) <= 2T? No.
    # The condition is equivalent to: if we sort such that p_1 <= p_2 <= ... <= p_n,
    # then the minimum span required is sum(p_i) + sum(p_i) - (p_1 + p_n)?
    # Actually, the condition for existence is: 2 * sum(p_i) - max(p_i) <= 2 * T.
    # Wait, the condition is: sum(p_i) + sum(p_i) - (p_max) <= 2*T?
    # Let's re-evaluate: S_i are centers of intervals of length p_i.
    # The condition |S_i - S_j| >= p_i + p_j is equivalent to saying the intervals
    # [S_i - p_i, S_i + p_i] are disjoint.
    # These intervals are [2*start_i, 2*start_i + 2*p_i].
    # The total length is sum(2*p_i). The range is [0, 2*T].
    # So sum(2*p_i) <= 2*T => sum(p_i) <= T.
    
    sum1 = sum(p[0] for p in params)
    if sum1 > T:
        print("LOAD 1")
        return
        
    sum2 = sum(p[1] for p in params)
    if sum2 > T:
        print("LOAD 2")
        return

solve()
