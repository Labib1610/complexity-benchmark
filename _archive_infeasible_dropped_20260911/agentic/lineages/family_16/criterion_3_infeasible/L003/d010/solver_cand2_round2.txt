```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Constraints: 0 <= x_i, x_i + p1 <= T, 0 <= y_i, y_i + p2 <= T, x_i + p1 <= y_i
    # This implies x_i + p1 + p2 <= y_i + p2 <= T
    # So x_i + p1 + p2 <= T. Since x_i >= 0, we need p1 + p2 <= T.
    # If p1 + p2 > T, then JOB i is a contradiction.
    for i in range(j):
        if p[i][0] + p[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Constraints: 0 <= x_i, x_i + p1 <= T, |x_i - x_k| >= max(p1_i, p1_k)
    # This is equivalent to packing intervals [x_i, x_i + p1_i] in [0, T]
    # such that they don't overlap.
    # The sum of p1_i must be <= T.
    sum_p1 = sum(item[0] for item in p)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Same logic for y_i
    sum_p2 = sum(item[1] for item in p)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
