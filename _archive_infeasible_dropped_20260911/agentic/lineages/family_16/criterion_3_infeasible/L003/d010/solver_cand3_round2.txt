```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i:
        # y1 >= p1/2, y2 <= T - p2/2, y2 - y1 >= (p1 + p2)/2
        # y2 >= y1 + (p1 + p2)/2 >= p1/2 + (p1 + p2)/2 = p1 + p2/2
        # So we need p1 + p2/2 <= T - p2/2 => p1 + p2 <= T
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        p1_list.append(p1)
        p2_list.append(p2)
        
    # Check LOAD 1:
    # Sum of lengths must be <= T
    # The constraint for non-overlapping is |yi - yk| >= (pi + pk)/2
    # This is equivalent to saying the total length occupied is sum(pi)
    # The bounds are [p_i/2, T - p_i/2]
    # The total length required is sum(pi) <= T
    if sum(p1_list) > t:
        print("LOAD 1")
        return
    
    # Check LOAD 2:
    if sum(p2_list) > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
