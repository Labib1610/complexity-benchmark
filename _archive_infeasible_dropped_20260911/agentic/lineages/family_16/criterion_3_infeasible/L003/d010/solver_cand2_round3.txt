```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    pairs = []
    idx = 2
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        pairs.append((p1, p2))
        idx += 2
        
    # Check JOB i: Rule 1, 2, 3, 5 for pair i
    # Sup(I_{i,1}) <= Inf(I_{i,2}) means the intervals are [a, a+p1] and [b, b+p2]
    # with a+p1 <= b. Since both are in [0, T], we need a >= 0, b+p2 <= T.
    # Thus, p1 + p2 <= T. If p1 + p2 > T, JOB i is impossible.
    for i in range(j):
        if pairs[i][0] + pairs[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1: Rule 1, 2, 3, 4 for Domain 1
    # Sum of integrals of all I_{i,1} must be <= T.
    sum1 = sum(p[0] for p in pairs)
    if sum1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2: Rule 1, 2, 3, 4 for Domain 2
    # Sum of integrals of all I_{i,2} must be <= T.
    sum2 = sum(p[1] for p in pairs)
    if sum2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
