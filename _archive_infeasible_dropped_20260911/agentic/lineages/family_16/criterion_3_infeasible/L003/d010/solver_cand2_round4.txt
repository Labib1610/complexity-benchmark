```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i:
        # The instruction buffer in Bank 1 occupies [s1, s1 + p1 - 1].
        # The data buffer in Bank 2 occupies [s2, s2 + p2 - 1].
        # Constraint: s2 >= s1 + p1.
        # Since s2 <= T - p2, we have s1 + p1 <= T - p2.
        # Since s1 >= 0, we need p1 + p2 <= T.
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
        
        p1_list.append(p1)
        p2_list.append(p2)
        
    # Check LOAD 1:
    if sum(p1_list) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2:
    if sum(p2_list) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
