import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    pairs = []
    for i in range(j):
        p1 = int(data[2 + 2*i])
        p2 = int(data[3 + 2*i])
        pairs.append((p1, p2))
    
    # Check each JOB i
    for i in range(j):
        p1, p2 = pairs[i]
        # For JOB i, we only consider the primary and secondary elements of index i
        # Primary element: start1, end1 = start1, start1 + p1
        # Secondary element: start2, end2 = start2, start2 + p2
        # Precedence: end1 <= start2 => start1 + p1 <= start2
        # Bounds: 0 <= start1, start1 + p1 <= T, 0 <= start2, start2 + p2 <= T
        # So: start1 >= 0, start1 + p1 <= T => start1 <= T - p1
        #     start2 >= 0, start2 + p2 <= T => start2 <= T - p2
        # Precedence: start1 + p1 <= start2
        # So we need: start1 + p1 <= start2
        #             0 <= start1 <= T - p1
        #             0 <= start2 <= T - p2
        #             start1 + p1 <= start2
        # This implies: start1 + p1 <= T - p2 => start1 <= T - p1 - p2
        # Also start1 >= 0
        # So we need 0 <= T - p1 - p2 => p1 + p2 <= T
        # And also start1 + p1 <= start2, start2 >= 0 => start1 + p1 >= 0 => start1 >= -p1
        # But start1 >= 0, so that's fine.
        # We also need start1 <= T - p1, start1 <= T - p1 - p2
        # So start1 <= T - max(p1, p1 + p2) = T - p1 - p2
        # We need 0 <= start1 <= T - p1 - p2
        # This is possible iff 0 <= T - p1 - p2 => p1 + p2 <= T
        if p1 + p2 > T:
            print(f"JOB {i+1}")
            return
    
    # If all JOBs are fine, check LOAD 1 and LOAD 2
    # For LOAD c, we only consider elements of class c with their parameters
    # Each element i has parameter p_{i,c}, and must satisfy:
    # 0 <= start_i, start_i + p_{i,c} <= T => 0 <= start_i <= T - p_{i,c}
    # For any two elements i and k in the same class:
    # |(start_i + end_i) - (start_k + end_k)| >= p_{i,c} + p_{k,c}
    # That is, |(2*start_i + p_{i,c}) - (2*start_k + p_{k,c})| >= p_{i,c} + p_{k,c}
    # Let s_i = 2*start_i + p_{i,c}, then |s_i - s_k| >= p_{i,c} + p_{k,c}
    # Also, 0 <= start_i <= T - p_{i,c} => p_{i,c} <= s_i <= 2*T - p_{i,c}
    # So for all elements in class c, we have intervals [p_{i,c}, 2*T - p_{i,c}] for s_i
    # and we need |s_i - s_k| >= p_{i,c} + p_{k,c} for all i != k
    
    # This is equivalent to placing points s_i on the line such that
    # each point s_i is in [p_{i,c}, 2*T - p_{i,c}] and 
    # |s_i - s_k| >= p_{i,c} + p_{k,c}
    
    # This is a classic scheduling problem. We can sort the jobs by p_{i,c} 
    # and try to place them greedily.
    
    # For LOAD 1: primary elements with parameters p1
    primary_params = [p1 for p1, p2 in pairs]
    # For LOAD 2: secondary elements with parameters p2
    secondary_params = [p2 for p1, p2 in pairs]
    
    def is_impossible_class(params):
        # Check if placing elements with given params in one class is impossible
        # We sort by parameter value
        params_sorted = sorted(params)
        # Try to place them greedily
        # s_i = 2*start_i + p_i, so start_i = (s_i - p_i)/2
        # We need s_i in [p_i, 2*T - p_i]
        # and |s_i - s_k| >= p_i + p_k for i < k in sorted order
        # So s_{k} >= s_i + p_i + p_k
        # We place s_k at max(p_k, s_i + p_i + p_k)
        s = params_sorted[0]  # Place first one at its minimum
        for i in range(1, len(params_sorted)):
            p_prev = params_sorted[i-1]
            p_curr = params_sorted[i]
            s = max(p_curr, s + p_prev + p_curr)
            # Check if s is within bounds
            if s > 2*T - p_curr:
                return True
        return False
    
    if is_impossible_class(primary_params):
        print("LOAD 1")
        return
    if is_impossible_class(secondary_params):
        print("LOAD 2")
        return

if __name__ == "__main__":
    main()
