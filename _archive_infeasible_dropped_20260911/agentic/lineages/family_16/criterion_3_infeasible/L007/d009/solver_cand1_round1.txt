```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        
        # Check JOB i:
        # V_i has p1 ones, W_i has p2 ones.
        # Support Precedence: max index of V_i < min index of W_i.
        # This implies we need at least p1 + p2 distinct indices.
        # So p1 + p2 <= T.
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1:
    # Sum of p1_i <= T
    if sum(p1_list) > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2:
    # Sum of p2_i <= T
    if sum(p2_list) > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
