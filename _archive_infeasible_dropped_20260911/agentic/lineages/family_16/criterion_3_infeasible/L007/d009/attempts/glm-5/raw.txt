import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        j = int(next(iterator))
        T = int(next(iterator))
    except StopIteration:
        return
    
    # Check JOB condition: for each graph i, p1 + p2 <= T
    # We need to find the first 1-indexed graph that violates this.
    # We iterate through all graphs to find the minimum index i.
    job_violation_index = -1
    
    # Check LOAD condition: sum of p1 <= T and sum of p2 <= T
    # We need to check which phase is overloaded.
    # If both are overloaded, the problem statement guarantees exactly one condition applies.
    # However, if we find a JOB violation first, we prioritize that?
    # The problem states "exactly one of the following isolated conditions is the root cause".
    # This implies we won't have ambiguous cases where both JOB and LOAD are violated in a way that creates conflict.
    # If a JOB violation exists, it is the root cause.
    # If no JOB violation exists, then a LOAD violation must exist.
    
    sum_p1 = 0
    sum_p2 = 0
    
    # We need to process all inputs anyway to consume them, and to calculate sums.
    # We can do a single pass.
    
    p1_list = []
    p2_list = []
    
    for i in range(1, j + 1):
        try:
            p1 = int(next(iterator))
            p2 = int(next(iterator))
        except StopIteration:
            break
        
        if job_violation_index == -1:
            if p1 + p2 > T:
                job_violation_index = i
        
        sum_p1 += p1
        sum_p2 += p2
    
    if job_violation_index != -1:
        print(f"JOB {job_violation_index}")
    else:
        # Must be a LOAD violation
        if sum_p1 > T:
            print("LOAD 1")
        else:
            print("LOAD 2")

if __name__ == '__main__':
    solve()
