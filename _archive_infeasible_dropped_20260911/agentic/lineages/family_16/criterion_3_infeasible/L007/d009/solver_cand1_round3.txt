```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    # The constraints are:
    # 1. JOB i: p_{i,1} + p_{i,2} > T
    #    Strict Phase Ordering: Machine 1 must finish before Machine 2 starts.
    #    If task i runs on M1 for p_{i,1} slots and M2 for p_{i,2} slots,
    #    the earliest M1 can finish is at time p_{i,1}.
    #    The earliest M2 can start is at time p_{i,1} + 1.
    #    The latest M2 can finish is at time p_{i,1} + p_{i,2}.
    #    Thus, p_{i,1} + p_{i,2} <= T.
    
    # 2. LOAD c: sum_{i=1}^j p_{i,c} > T
    #    Machine Capacity: A machine can process at most one task at any time.
    #    Total slots required on machine c is sum(p_{i,c}).
    #    This must be <= T.
    
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        # Check JOB i constraint
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Check LOAD c constraint
    if sum_p1 > T:
        print("LOAD 1")
    elif sum_p2 > T:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
