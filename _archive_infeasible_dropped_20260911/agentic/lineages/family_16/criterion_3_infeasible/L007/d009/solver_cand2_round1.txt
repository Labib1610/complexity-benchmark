import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        
        # Check JOB i condition:
        # For operation i, we need S_{i,1} subset of [1, T] with |S_{i,1}| = p_{i,1}
        # and S_{i,2} subset of [1, T] with |S_{i,2}| = p_{i,2}.
        # Interference Void: {x | min(S_{i,2}) <= x <= max(S_{i,1})} must be empty.
        # This means max(S_{i,1}) < min(S_{i,2}).
        # To maximize the gap, we pick S_{i,1} = {1, ..., p_{i,1}} and S_{i,2} = {T-p_{i,2}+1, ..., T}.
        # The condition max(S_{i,1}) < min(S_{i,2}) becomes p_{i,1} < T - p_{i,2} + 1,
        # or p_{i,1} + p_{i,2} <= T.
        # If p_{i,1} + p_{i,2} > T, it is impossible.
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD c condition:
    # For channel c, we need to pick j disjoint sets of size p_{i,c} from [1, T].
    # This is possible if and only if the sum of cardinalities <= T.
    sum_p1 = sum(p1_list)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    sum_p2 = sum(p2_list)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
