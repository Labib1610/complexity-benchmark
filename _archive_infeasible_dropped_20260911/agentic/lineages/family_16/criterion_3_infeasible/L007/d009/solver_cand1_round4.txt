```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        
        # Check JOB i:
        # For entity i, we need to pick p1_i distinct values in [1, T] for phi_{i,1}
        # and p2_i distinct values in [1, T] for phi_{i,2}.
        # Let max(image(phi_{i,1})) = M1 and min(image(phi_{i,2})) = M2.
        # We need M1 < M2.
        # To minimize M1, we pick {1, 2, ..., p1_i}, so M1 = p1_i.
        # To maximize M2, we pick {T-p2_i+1, ..., T}, so M2 = T-p2_i+1.
        # Condition: p1_i < T - p2_i + 1  => p1_i + p2_i <= T.
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD c:
    # For index c, we need to fit all p_{i,c} into [1, T] without overlap.
    # Total required space is sum(p_{i,c}).
    # Condition: sum(p_{i,c}) <= T.
    sum1 = sum(p1_list)
    if sum1 > t:
        print("LOAD 1")
        return
        
    sum2 = sum(p2_list)
    if sum2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
