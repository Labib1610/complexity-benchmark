```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    clients = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        clients.append((p1, p2, i))
        sum_p1 += p1
        sum_p2 += p2
        
        # Check JOB i condition:
        # For client i, we need p1 channels <= k and p2 channels > k.
        # This is possible if and only if there exists k in [0, T] such that
        # p1 <= k and p2 <= T - k.
        # This is equivalent to p1 + p2 <= T.
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
            
    # Check LOAD c condition:
    # For fiber c, we need sum(p_i,c) <= T.
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
