```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check JOB i condition:
    # For a single graph i, we need p1 + p2 distinct stages from 1 to T.
    # This is possible if and only if p1 + p2 <= T.
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c condition:
    # For phase 1, we need sum(p_{i,1}) <= T.
    # For phase 2, we need sum(p_{i,2}) <= T.
    sum_p1 = 0
    sum_p2 = 0
    for p1, p2 in jobs:
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
