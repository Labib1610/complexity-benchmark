```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    timeframe = int(input_data[1])
    
    commissions = []
    idx = 2
    for i in range(n):
        p = int(input_data[idx])
        s = int(input_data[idx + 1])
        commissions.append((p, s))
        idx += 2
        
    # Check for JOB i
    for i in range(n):
        if commissions[i][0] + commissions[i][1] > timeframe:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c
    sum1 = 0
    sum2 = 0
    for p, s in commissions:
        sum1 += p
        sum2 += s
        
    if sum1 > timeframe:
        print("LOAD 1")
    elif sum2 > timeframe:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
