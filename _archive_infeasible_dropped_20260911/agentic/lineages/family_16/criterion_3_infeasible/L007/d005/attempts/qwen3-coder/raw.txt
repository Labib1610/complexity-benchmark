        p2 = int(data[idx+1])

            return

        # So to satisfy exclusivity, we must have max(a,c) >= min(b,d), i.e., b <= c or d <= a.

        # But that's not right either, because we can have overlap at a point.

        for c in [1, 2]:

                return
