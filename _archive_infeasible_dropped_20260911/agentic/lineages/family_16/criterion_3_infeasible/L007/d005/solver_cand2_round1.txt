```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i:
        # D_{i,1} = [s1, s1 + p1], D_{i,2} = [s2, s2 + p2]
        # Causality: s1 + p1 <= s2
        # Constraints: 0 <= s1, s1 + p1 <= T, 0 <= s2, s2 + p2 <= T
        # So s1 >= 0, s1 <= T - p1, s2 >= s1 + p1, s2 <= T - p2
        # Possible if there exists s1 such that s1 + p1 <= T - p2
        # i.e., p1 + p2 <= T
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        p1_list.append(p1)
        p2_list.append(p2)
        
    # Check LOAD c:
    # Exclusivity: D_{i,c} \cap D_{k,c} has at most one point.
    # This means intervals [s_i, s_i + p_i] must be ordered such that
    # s_{i+1} >= s_i + p_i.
    # Total length required: sum(p_i) + (j - 1) * 0 (since they can touch at one point)
    # Wait, the problem says "at most one real number".
    # If they touch at one point, the total length is sum(p_i).
    # The range is [0, T], so the total length must be <= T.
    
    sum_p1 = sum(p1_list)
    sum_p2 = sum(p2_list)
    
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
