```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    
    idx = 2
    for i in range(j):
        v1 = int(input_data[idx])
        v2 = int(input_data[idx+1])
        p1.append(v1)
        p2.append(v2)
        
        # Check JOB i
        # Rule 1: S_{i,1} + p_{i,1} <= T, S_{i,2} + p_{i,2} <= T
        # Rule 2: S_{i,2} >= S_{i,1} + p_{i,1}
        # Combined: S_{i,1} + p_{i,1} + p_{i,2} <= T
        if v1 + v2 > T:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1
    # Rule 1 & 3: Sum of p_{i,1} <= T
    if sum(p1) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Rule 1 & 3: Sum of p_{i,2} <= T
    if sum(p2) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
