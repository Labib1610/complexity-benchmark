```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    items = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    job_violation = -1
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        items.append((p1, p2))
        sum_p1 += p1
        sum_p2 += p2
        
        if p1 + p2 > t:
            if job_violation == -1:
                job_violation = i
                
    if sum_p2 > t:
        print("LOAD 2")
    elif sum_p1 > t:
        print("LOAD 1")
    else:
        print(f"JOB {job_violation}")

if __name__ == '__main__':
    solve()
```
