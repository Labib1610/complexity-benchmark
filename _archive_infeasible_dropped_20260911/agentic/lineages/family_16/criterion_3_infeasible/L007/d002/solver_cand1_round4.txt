import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    items = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        items.append((p1, p2))
        
    if sum_p1 > t:
        print("LOAD 1")
        return
    
    if sum_p2 > t:
        print("LOAD 2")
        return
    
    # If we are here, it means sum_p1 <= T, sum_p2 <= T, and p1+p2 <= T for all i.
    # Johnson's rule guarantees that if a schedule exists, it can be found by
    # sorting items with p1 < p2 by p1 ascending, and p1 >= p2 by p2 descending.
    # If this optimal schedule fails, then no schedule exists.
    
    group1 = []
    group2 = []
    for i in range(j):
        if items[i][0] < items[i][1]:
            group1.append((items[i][0], items[i][1]))
        else:
            group2.append((items[i][0], items[i][1]))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    schedule = group1 + group2
    
    curr_p1 = 0
    curr_p2 = 0
    for p1, p2 in schedule:
        curr_p1 += p1
        curr_p2 = max(curr_p2, curr_p1) + p2
        if curr_p2 > t:
            # This case should theoretically not be reachable given the problem 
            # constraints and the nature of the "impossible" prompt, but 
            # if it were, it would be a scheduling violation.
            # Based on the problem description, we only need to check JOB and LOAD.
            break

solve()
