import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    pairs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        pairs.append((p1, p2))
        idx += 2
        
    # Check JOB i:
    # Intra-Pair Orthogonality and Precedence:
    # u_i has p1 ones, v_i has p2 ones.
    # Let u_i have ones at indices x_1 < x_2 < ... < x_{p1}.
    # Let v_i have ones at indices y_1 < y_2 < ... < y_{p2}.
    # Condition: no x such that u_{i,x}=1 and y <= x such that v_{i,y}=1.
    # This means for all x where u_{i,x}=1, all y where v_{i,y}=1 must satisfy y > x.
    # The smallest possible index for a 1 in v_i is y_1.
    # The largest possible index for a 1 in u_i is x_{p1}.
    # We need x_{p1} < y_1.
    # Since x_1 < x_2 < ... < x_{p1} are distinct indices in {1, ..., T},
    # the smallest possible x_{p1} is p1.
    # Since y_1 < y_2 < ... < y_{p2} are distinct indices in {1, ..., T},
    # the largest possible y_1 is T - p2 + 1.
    # So we need p1 < T - p2 + 1, or p1 + p2 <= T.
    
    for i in range(j):
        p1, p2 = pairs[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c:
    # LOAD 1: sum(p1) > T
    # LOAD 2: sum(p2) > T
    
    sum1 = 0
    sum2 = 0
    for p1, p2 in pairs:
        sum1 += p1
        sum2 += p2
        
    if sum1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
