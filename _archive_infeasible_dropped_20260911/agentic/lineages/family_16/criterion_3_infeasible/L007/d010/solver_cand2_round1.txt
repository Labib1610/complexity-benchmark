```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check JOB i condition:
    # For a single job i, we need to pick p1 indices for P_{i,1} and p2 indices for P_{i,2}
    # such that if P_{i,1}e_x = e_x and P_{i,2}e_y = e_y, then x < y.
    # This means all indices in P_{i,1} must be strictly less than all indices in P_{i,2}.
    # We need p1 + p2 distinct indices in {1, ..., T}.
    # Thus, p1 + p2 <= T.
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c condition:
    # Phase 1: sum of p_{i,1} <= T
    # Phase 2: sum of p_{i,2} <= T
    sum_p1 = sum(job[0] for job in jobs)
    sum_p2 = sum(job[1] for job in jobs)
    
    if sum_p1 > t:
        print("LOAD 1")
    elif sum_p2 > t:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
