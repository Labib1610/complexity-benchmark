import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    rows = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        rows.append((p1, p2))
        idx += 2
        
    # Check JOB i: Row i cannot satisfy p1, p2 with T columns
    # Strict Abscissa Precedence: 
    # S1 has p1 points, S2 has p2 points.
    # Let S1 columns be x1 < x2 < ... < xp1
    # Let S2 columns be y1 < y2 < ... < yp2
    # Condition: for all (i, x) in S1 and (i, y) in S2, x < y.
    # This means max(S1_cols) < min(S2_cols).
    # To minimize the range, we pick S1 as {1, ..., p1} and S2 as {p1+1, ..., p1+p2}.
    # So we need p1 + p2 <= T.
    for i in range(j):
        if rows[i][0] + rows[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c: Set c is over-constrained
    # Column Sparsity: S1 can have at most one coordinate in column k.
    # Total coordinates in S1 across all rows is sum(p1_i).
    # Since there are T columns, we need sum(p1_i) <= T.
    sum_p1 = 0
    sum_p2 = 0
    for p1, p2 in rows:
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
