import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    vms = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        vms.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Rule 2: Bank 1 block [s1, s1 + p1 - 1] and Bank 2 block [s2, s2 + p2 - 1]
    # Sequential Addressing: max(Bank 1) < min(Bank 2)
    # s1 + p1 - 1 < s2
    # Since s2 >= 1, we need s1 + p1 - 1 < s2.
    # The smallest possible s2 is 1.
    # So we need s1 + p1 - 1 < 1 => s1 + p1 < 2.
    # Since s1 >= 1, s1 + p1 >= 1 + p1.
    # So 1 + p1 < 2 is impossible for p1 >= 1.
    # Wait, the rule says "every byte address in Bank 1 must be strictly less than every byte address in Bank 2".
    # Let Bank 1 block be [s1, e1] and Bank 2 block be [s2, e2].
    # e1 < s2.
    # Since s1 >= 1, e1 = s1 + p1 - 1 >= p1.
    # Since e2 <= T, s2 = e2 - p2 + 1 <= T - p2 + 1.
    # So we need p1 < s2.
    # The smallest s2 can be is 1. But we need s2 > e1 >= p1.
    # So s2 must be at least p1 + 1.
    # Also s2 + p2 - 1 <= T => s2 <= T - p2 + 1.
    # So we need p1 + 1 <= T - p2 + 1 => p1 + p2 <= T.
    
    for i in range(j):
        p1, p2 = vms[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    sum_p1 = sum(vm[0] for vm in vms)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    sum_p2 = sum(vm[1] for vm in vms)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
