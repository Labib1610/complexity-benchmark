```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    adjustments = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        adjustments.append((p1, p2))
        idx += 2
        
    # Check for JOB i
    # For adjustment i, we need p1 timestamps for primary and p2 for secondary.
    # Rule: All primary timestamps < all secondary timestamps.
    # This means we need at least p1 + p2 distinct timestamps in [1, T].
    # The smallest set of timestamps is {1, 2, ..., p1+p2}.
    # The primary engine uses {1, ..., p1} and secondary uses {p1+1, ..., p1+p2}.
    # This is valid if p1 + p2 <= T.
    # If p1 + p2 > T, then adjustment i is internally unsatisfiable.
    for i in range(j):
        p1, p2 = adjustments[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c
    # Total primary engine time required: sum(p1)
    # Total secondary engine time required: sum(p2)
    # If sum(p1) > T, then LOAD 1 is overloaded.
    # If sum(p2) > T, then LOAD 2 is overloaded.
    sum_p1 = 0
    sum_p2 = 0
    for p1, p2 in adjustments:
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
