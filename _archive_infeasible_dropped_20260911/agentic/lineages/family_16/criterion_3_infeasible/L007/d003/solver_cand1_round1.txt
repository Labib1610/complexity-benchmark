import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    items = []
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i condition
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        items.append((p1, p2))
        
    # Check LOAD 1 condition
    sum_p1 = sum(item[0] for item in items)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2 condition
    sum_p2 = sum(item[1] for item in items)
    if sum_p2 > t:
        print("LOAD 2")
        return

    # If none of the above, it must be impossible due to Johnson's rule logic
    # The minimum time to complete all jobs is given by the optimal schedule.
    # If it's impossible, there must be a bottleneck.
    # The problem guarantees it's impossible, so one of the above must hold.
    # If not, we check the total time constraint via Johnson's rule.
    # However, the problem implies one of the three certificates is always valid.
    
    # Re-check: The problem states "In every provided record, it is physically 
    # impossible... You must output a concise certificate".
    # If the above don't trigger, we can use the fact that if it's impossible,
    # the minimum makespan > T.
    
    # Sort items for Johnson's rule
    # Group 1: p1 < p2, sorted by p1 ascending
    # Group 2: p1 >= p2, sorted by p2 descending
    g1 = sorted([x for x in items if x[0] < x[1]], key=lambda x: x[0])
    g2 = sorted([x for x in items if x[0] >= x[1]], key=lambda x: x[1], reverse=True)
    order = g1 + g2
    
    time1 = 0
    time2 = 0
    for p1, p2 in order:
        time1 += p1
        time2 = max(time1, time2) + p2
        
    # If time2 > t, it's impossible. The certificate must be one of the three.
    # Given the constraints and problem type, one of the three must be true.
    pass

if __name__ == '__main__':
    solve()
