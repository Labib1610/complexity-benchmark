import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    items = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i condition: p1 + p2 > T
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        items.append((p1, p2))
        
    # Check LOAD c condition: sum(p1) > T or sum(p2) > T
    if sum_p1 > t:
        print("LOAD 1")
        return
    if sum_p2 > t:
        print("LOAD 2")
        return

    # If neither condition is met, the problem guarantees it's impossible,
    # but the problem statement says one of these MUST be present.
    # Johnson's rule logic: if it's impossible, it must be one of the above.
    # The problem guarantees one of these violations exists.
    pass

if __name__ == '__main__':
    solve()
