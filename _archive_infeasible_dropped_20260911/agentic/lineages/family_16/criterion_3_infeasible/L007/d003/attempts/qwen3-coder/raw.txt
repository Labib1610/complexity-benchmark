```python
j, T = map(int, input().split())
items = []
for _ in range(j):
    p1, p2 = map(int, input().split())
    items.append((p1, p2))

# Check JOB condition
for i in range(j):
    p1, p2 = items[i]
    if p1 + p2 > T:
        print(f"JOB {i + 1}")
        exit()

# Check LOAD conditions
total_p1 = sum(item[0] for item in items)
total_p2 = sum(item[1] for item in items)

if total_p1 > T:
    print("LOAD 1")
    exit()

if total_p2 > T:
    print("LOAD 2")
    exit()
```
