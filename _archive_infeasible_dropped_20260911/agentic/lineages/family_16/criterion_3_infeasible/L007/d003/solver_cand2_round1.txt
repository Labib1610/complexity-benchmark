```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    items = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        items.append((p1, p2, i + 1))
        idx += 2
        
    # Check JOB i
    for i in range(j):
        p1, p2, original_idx = items[i]
        if p1 + p2 > t:
            print(f"JOB {original_idx}")
            return
            
    # Check LOAD 1
    sum_p1 = sum(item[0] for item in items)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    sum_p2 = sum(item[1] for item in items)
    if sum_p2 > t:
        print("LOAD 2")
        return
        
    # Johnson's Rule for Flow Shop Scheduling
    # Items with p1 < p2 sorted by p1 ascending
    # Items with p1 >= p2 sorted by p2 descending
    group1 = []
    group2 = []
    for item in items:
        if item[0] < item[1]:
            group1.append(item)
        else:
            group2.append(item)
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    schedule = group1 + group2
    
    # Calculate completion time
    curr_p1 = 0
    curr_p2 = 0
    for p1, p2, _ in schedule:
        curr_p1 += p1
        curr_p2 = max(curr_p1, curr_p2) + p2
        
    if curr_p2 > t:
        # If Johnson's rule fails, it must be a LOAD 2 issue 
        # (though the problem implies one of these must be the cause)
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
