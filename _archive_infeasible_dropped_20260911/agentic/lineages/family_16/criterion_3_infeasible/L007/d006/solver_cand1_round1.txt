```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    items = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        items.append((p1, p2))
        idx += 2
        
    # Check JOB i: p_{i,1} + p_{i,2} > T
    for i in range(j):
        if items[i][0] + items[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1: sum(p_{i,1}) > T + (j - 1)
    # Exclusivity: intersection at most one point means sum of lengths + (j-1) <= T
    # sum(p_{i,c}) + (j-1) <= T  => sum(p_{i,c}) <= T - j + 1
    
    sum1 = sum(item[0] for item in items)
    if sum1 > T - j + 1:
        print("LOAD 1")
        return
        
    sum2 = sum(item[1] for item in items)
    if sum2 > T - j + 1:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
