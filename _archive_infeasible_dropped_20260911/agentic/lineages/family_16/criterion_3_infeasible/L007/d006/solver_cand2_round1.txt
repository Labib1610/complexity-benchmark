import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    # Precedence constraint: (chi_{i,1} * chi_{i,2})(t) = 0 for t < 0
    # This implies that the support of chi_{i,1} must end before or at the start of chi_{i,2}.
    # Let I1 = [a1, a1 + p1] and I2 = [a2, a2 + p2].
    # Precedence means a1 + p1 <= a2.
    # The convex hull of the support of (chi_{i,1} + chi_{i,2}) is [a1, a2 + p2].
    # The length of this interval is (a2 + p2) - a1.
    # Since a2 >= a1 + p1, the minimum length is (a1 + p1 + p2) - a1 = p1 + p2.
    # If p1 + p2 > T, then JOB i is violated.
    
    # LOAD c constraint: sum of measures in dimension c must be <= T.
    # sum(p_{i,c}) <= T.
    
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > T:
        print("LOAD 1")
    elif sum_p2 > T:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
