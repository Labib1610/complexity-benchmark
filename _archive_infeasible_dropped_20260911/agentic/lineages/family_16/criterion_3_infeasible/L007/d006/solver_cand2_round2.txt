```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    tasks = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tasks.append((p1, p2))
        idx += 2
        
    # Check JOB i condition:
    # max(A_i) <= min(B_i)
    # A_i = [s1, s1 + p1], B_i = [s2, s2 + p2]
    # s1 + p1 <= s2
    # Since s1 >= 0 and s2 + p2 <= T, we need s2 >= s1 + p1
    # So s1 + p1 + p2 <= T
    for i in range(j):
        if tasks[i][0] + tasks[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c condition:
    # Sum of lengths <= T
    # For timeline 1: sum(p_{i,1}) <= T
    # For timeline 2: sum(p_{i,2}) <= T
    sum1 = 0
    sum2 = 0
    for p1, p2 in tasks:
        sum1 += p1
        sum2 += p2
        
    if sum1 > t:
        print("LOAD 1")
        return
    if sum2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
