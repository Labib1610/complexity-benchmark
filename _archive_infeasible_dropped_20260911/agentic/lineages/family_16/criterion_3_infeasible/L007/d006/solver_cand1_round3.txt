```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        
        # Check JOB i condition: min span = p1 + p2
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD c condition: min span = sum of lengths
    sum1 = sum(p1_list)
    if sum1 > T:
        print("LOAD 1")
        return
        
    sum2 = sum(p2_list)
    if sum2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
