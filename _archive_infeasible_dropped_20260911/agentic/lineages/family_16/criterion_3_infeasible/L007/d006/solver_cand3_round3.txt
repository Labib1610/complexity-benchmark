```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    processes = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        processes.append((p1, p2))
        idx += 2
        
    # Check LOAD 1: sum of all p1 > T
    sum_p1 = sum(p[0] for p in processes)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2: sum of all p2 > T
    sum_p2 = sum(p[1] for p in processes)
    if sum_p2 > t:
        print("LOAD 2")
        return
        
    # Check JOB i: p1_i + p2_i > T
    # The problem states: "minimum possible distance from the lowest coordinate 
    # of process i's primary block to the highest coordinate of its secondary block"
    # Since primary must end before secondary starts, the minimum distance is 
    # (start_p1 + p1_i) + p2_i. The smallest start_p1 is 0.
    # So min distance is p1_i + p2_i.
    for i in range(j):
        if processes[i][0] + processes[i][1] > t:
            print(f"JOB {i + 1}")
            return

if __name__ == '__main__':
    solve()
```
