```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    T = int(input_data[1])
    
    # Parse pulse widths
    p1 = []
    p2 = []
    idx = 2
    for i in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Condition 1: JOB i
    # Bounded Support: 0 <= k <= T - p
    # Non-negative Lag: R_i(tau) = 0 for tau < 0
    # R_i(tau) = integral s_{i,1}(t) * s_{i,2}(t + tau) dt
    # This is the overlap of [k1, k1+p1] and [k2-tau, k2-tau+p2]
    # For R_i(tau) = 0 for all tau < 0, the intervals must not overlap for any tau < 0.
    # This means the end of the first interval must be <= the start of the second interval.
    # k1 + p1 <= k2.
    # Combined with 0 <= k1, k1+p1 <= T, 0 <= k2, k2+p2 <= T:
    # We need 0 <= k1, k1+p1 <= k2, k2+p2 <= T.
    # This is possible if and only if p1 + p2 <= T.
    for i in range(j):
        if p1[i] + p2[i] > T:
            print(f"JOB {i + 1}")
            return
            
    # Condition 2: LOAD c
    # Bounded Support: 0 <= k_i <= T - p_i
    # Orthogonal Multiplexing: integral s_{i,c}(t) * s_{m,c}(t) dt = 0 for i != m
    # This means the intervals [k_i, k_i + p_i] must be disjoint.
    # The sum of lengths of disjoint intervals must be <= T.
    # sum(p_i) <= T.
    sum_p1 = sum(p1)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    sum_p2 = sum(p2)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
