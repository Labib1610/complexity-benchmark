```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        idx += 2
        
    # Check JOB i: p_{i,1} + p_{i,2} > T
    for i in range(j):
        if p1_list[i] + p2_list[i] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1: sum(p_{i,1}) > T + (j - 1)
    # The disjointness rule says intersection is at most one point.
    # For j items, we need j-1 points of contact.
    # Total length required: sum(p_{i,1}) - (j-1) <= T
    # sum(p_{i,1}) <= T + j - 1
    if sum(p1_list) > T + j - 1:
        print("LOAD 1")
        return
        
    # Check LOAD 2: sum(p_{i,2}) > T + (j - 1)
    if sum(p2_list) > T + j - 1:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
