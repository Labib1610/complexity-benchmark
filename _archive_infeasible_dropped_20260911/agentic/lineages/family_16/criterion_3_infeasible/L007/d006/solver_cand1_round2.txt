import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i bottleneck
        # Routing-feasible: start2 >= end1. 
        # end1 = start1 + p1, start2 >= end1 => start2 >= start1 + p1
        # end2 = start2 + p2 <= T => start2 <= T - p2
        # So we need start1 + p1 <= T - p2 => p1 + p2 <= T
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Check LOAD c bottleneck
    # Capacity-feasible: sum of processing times on machine c <= T
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
