```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # For each entity i, the interval must cover D_{i,1} and D_{i,2}.
    # Since D_{i,1} has length p_{i,1} and D_{i,2} has length p_{i,2},
    # and causality requires all x in D_{i,1} <= all y in D_{i,2},
    # the minimum length of the union is p_{i,1} + p_{i,2}.
    for i in range(j):
        if p[i][0] + p[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # For a layer c, the exclusivity axiom says D_{e,c} and D_{k,c} 
    # intersect at most at one point. This means the total length 
    # required to fit all j entities is the sum of their lengths.
    # Sum_{e=1}^j p_{e,c} <= T.
    sum1 = 0
    sum2 = 0
    for i in range(j):
        sum1 += p[i][0]
        sum2 += p[i][1]
        
    if sum1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
