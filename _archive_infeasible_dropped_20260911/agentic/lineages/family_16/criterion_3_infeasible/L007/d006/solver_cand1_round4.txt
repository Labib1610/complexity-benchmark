```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    pairs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        pairs.append((p1, p2))
        idx += 2
        
    # Check JOB i: p1 + p2 > T
    for i in range(j):
        if pairs[i][0] + pairs[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1: sum of red blocks > T
    sum_red = sum(p[0] for p in pairs)
    if sum_red > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2: sum of blue blocks > T
    sum_blue = sum(p[1] for p in pairs)
    if sum_blue > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
