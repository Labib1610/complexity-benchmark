import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    tasks = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tasks.append((p1, p2))
        idx += 2
        
    # Check for JOB bottleneck
    # Precedence: D_{i,1} = [s1, s1+p1], D_{i,2} = [s2, s2+p2]
    # s1 + p1 <= s2, s1 >= 0, s2 + p2 <= T
    # So p1 + p2 <= T
    for i in range(j):
        if tasks[i][0] + tasks[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD bottleneck
    # Capacity: For track c, sum of p_{i,c} <= T
    # Since intersection is at most one point, the total length is sum(p_{i,c})
    # Actually, the condition is that the tasks must fit in [0, T].
    # If they are placed contiguously, the total length is sum(p_{i,c}).
    # However, they can overlap at endpoints.
    # The total length required for n tasks is sum(p_{i,c}) - (n-1) * 0 = sum(p_{i,c})
    # Wait, the capacity rule says intersection is at most one point.
    # This means they can be placed such that the end of one is the start of the next.
    # So the total length is sum(p_{i,c}) - (j-1) * 0 = sum(p_{i,c})?
    # No, if they share an endpoint, the total length is sum(p_{i,c}) - (j-1) * 0.
    # Actually, if they share an endpoint, the total length is sum(p_{i,c}) - (j-1) * 0.
    # Let's re-read: "intersection ... at most one real number".
    # This means they can be packed tightly: [0, p1], [p1, p1+p2], [p1+p2, p1+p2+p3]...
    # The total length is sum(p_{i,c}).
    
    sum1 = 0
    sum2 = 0
    for p1, p2 in tasks:
        sum1 += p1
        sum2 += p2
        
    if sum1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

solve()
