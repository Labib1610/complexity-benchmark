```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    tracks = []
    idx = 2
    sum1 = 0
    sum2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        tracks.append((p1, p2))
        sum1 += p1
        sum2 += p2
        
    # Check LOAD 1
    if sum1 > t:
        print("LOAD 1")
        return
    
    # Check LOAD 2
    if sum2 > t:
        print("LOAD 2")
        return
    
    # Check JOB i
    # Sequentiality: vocal ends at v_end, instrumental starts at i_start.
    # v_end <= i_start.
    # For track i: vocal occupies [v_start_i, v_end_i], length p1_i.
    # instrumental occupies [i_start_i, i_end_i], length p2_i.
    # v_end_i <= i_start_i.
    # Total span for track i: v_start_i to i_end_i.
    # Since v_start_i >= 0 and i_end_i <= T, we need p1_i + p2_i <= T.
    # Wait, the condition is: v_end_i <= i_start_i.
    # The total length required for track i is p1_i + p2_i.
    # If p1_i + p2_i > T, then it's impossible.
    for i in range(j):
        if tracks[i][0] + tracks[i][1] > t:
            print(f"JOB {i + 1}")
            return

if __name__ == '__main__':
    solve()
```
