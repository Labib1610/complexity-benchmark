```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i obstruction
        # Valid if there exists S1=[a, a+p1] and S2=[b, b+p2]
        # such that 0 <= a, a+p1 <= T, 0 <= b, b+p2 <= T, and a+p1 <= b
        # This is possible if p1 + p2 <= T
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
        
        p1_list.append(p1)
        p2_list.append(p2)
        
    # Check LOAD c obstruction
    # Valid if sum of lengths <= T
    # For stage 1: sum(p1_i) <= T
    if sum(p1_list) > T:
        print("LOAD 1")
        return
    
    # For stage 2: sum(p2_i) <= T
    if sum(p2_list) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
