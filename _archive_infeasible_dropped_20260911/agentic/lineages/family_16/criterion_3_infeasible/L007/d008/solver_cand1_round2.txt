```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    items = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        items.append((p1, p2))
        idx += 2
        
    # Check JOB i:
    # For each item i, we need h_{i,1} and h_{i,2} such that:
    # 1. Contiguity: h_{i,c} is 1 on [a_c, a_c + p_{i,c}]
    # 2. Intra-Item Ordering: max(h_{i,1}) <= min(h_{i,2})
    #    If h_{i,1} is [s1, e1] and h_{i,2} is [s2, e2], then e1 <= s2.
    #    Since e1 = s1 + p_{i,1} and e2 = s2 + p_{i,2},
    #    we need s1 + p_{i,1} <= s2.
    #    Also, both must be within [0, T], so s1 >= 0, e2 <= T.
    #    Thus, s1 + p_{i,1} + p_{i,2} <= T.
    #    This is possible if and only if p_{i,1} + p_{i,2} <= T.
    for i in range(j):
        if items[i][0] + items[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c:
    # For a fixed component c, we need to place j intervals of length p_{i,c}
    # in [0, T] such that they are disjoint.
    # This is possible if and only if sum(p_{i,c}) <= T.
    sum1 = sum(item[0] for item in items)
    if sum1 > T:
        print("LOAD 1")
        return
        
    sum2 = sum(item[1] for item in items)
    if sum2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
