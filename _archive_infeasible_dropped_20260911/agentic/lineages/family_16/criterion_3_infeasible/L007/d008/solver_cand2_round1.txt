```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # For each item i, we need to find disjoint sets of indices S1, S2 in {1..T}
    # such that |S1| >= p1 and |S2| >= p2, and for all x in S1, y in S2, x < y.
    # This is possible if and only if there exists a split point k such that
    # we can pick p1 indices <= k and p2 indices > k.
    # The smallest possible max index for S1 is p1, and the smallest possible
    # min index for S2 is p1 + 1.
    # So we need p1 + p2 <= T.
    for i in range(j):
        if p[i][0] + p[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # For component c, we need sum of dimensions <= T.
    # Sum of p_{i,c} for all i must be <= T.
    sum1 = 0
    sum2 = 0
    for i in range(j):
        sum1 += p[i][0]
        sum2 += p[i][1]
        
    if sum1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
