```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        
        # Condition 1: Intra-item Nullification
        # K_{i,1} (p1 x T) and K_{i,2} (p2 x T)
        # K_{i,1} K_{i,2}^T = 0 implies row spaces are orthogonal.
        # Dimension of row space of K_{i,1} is p1, K_{i,2} is p2.
        # Total dimension T. So p1 + p2 <= T.
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Condition 2: Inter-item Nullification
    # For component c, sum of rows p_{i,c} must be <= T.
    sum_p1 = sum(p1_list)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    sum_p2 = sum(p2_list)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
