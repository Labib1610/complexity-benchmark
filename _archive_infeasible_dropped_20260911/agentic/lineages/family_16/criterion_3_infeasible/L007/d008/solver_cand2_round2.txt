```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i condition:
        # For task i, we need p1 slots in [1, T] and p2 slots in [1, T]
        # such that max(S_{i,1}) < min(S_{i,2}).
        # This implies we need at least p1 + p2 distinct slots in [1, T].
        # So p1 + p2 > T is the condition for impossibility.
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        p1_list.append(p1)
        p2_list.append(p2)
        
    # Check LOAD c condition:
    # For core c, we need to allocate sum(p_{i,c}) distinct slots in [1, T].
    # This is impossible if sum(p_{i,c}) > T.
    if sum(p1_list) > t:
        print("LOAD 1")
        return
    
    if sum(p2_list) > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
