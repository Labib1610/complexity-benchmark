```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i condition:
        # For item i, component 1 needs p1 time, component 2 needs p2 time.
        # Causal precedence: component 2 starts after component 1 finishes.
        # Total time needed: p1 + p2.
        # Must be <= T.
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
        
        p1_list.append(p1)
        p2_list.append(p2)
        
    # Check LOAD c condition:
    # For component c, total energy sum(p_{i,c}) must be <= T.
    sum_p1 = sum(p1_list)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    sum_p2 = sum(p2_list)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
