import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # For an isometry f: [0, p] -> [0, T], the range is an interval of length p.
    # Specifically, f(x) = x + a or f(x) = -x + a.
    # Range is [a, a+p] or [a-p, a].
    # For f_{i,1} and f_{i,2} to satisfy f_{i,1}(x) <= f_{i,2}(y) for all x, y:
    # max(range(f_{i,1})) <= min(range(f_{i,2}))
    # Let range(f_{i,1}) = [a, a+p_{i,1}] and range(f_{i,2}) = [b, b+p_{i,2}]
    # We need a+p_{i,1} <= b.
    # Also, ranges must be in [0, T]:
    # 0 <= a and a+p_{i,1} <= T => a <= T - p_{i,1}
    # 0 <= b and b+p_{i,2} <= T => b <= T - p_{i,2}
    # So we need p_{i,1} <= b <= T - p_{i,2}.
    # This is possible if and only if p_{i,1} <= T - p_{i,2}, i.e., p_{i,1} + p_{i,2} <= T.
    
    for i in range(j):
        if p[i][0] + p[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # For component c, we need to place j intervals of length p_{i,c} in [0, T]
    # such that they overlap at most at one point.
    # This is possible if and only if the sum of lengths <= T + (j-1) * 0?
    # Wait, "overlap at most at one point" means they can be placed end-to-end.
    # The total length required is sum(p_{i,c}) - (j-1) * 0 = sum(p_{i,c}).
    # Actually, if they are [a_i, a_i + p_{i,c}], they overlap at most at one point
    # if a_{i+1} >= a_i + p_{i,c}.
    # Total span = sum(p_{i,c}) <= T.
    
    sum1 = sum(item[0] for item in p)
    if sum1 > T:
        print("LOAD 1")
        return
        
    sum2 = sum(item[1] for item in p)
    if sum2 > T:
        print("LOAD 2")
        return

solve()
