import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    # Constraints:
    # 1. Bounded Support: 0 <= k_{i,c} and k_{i,c} + p_{i,c} <= T
    #    => 0 <= k_{i,c} <= T - p_{i,c}
    #    This is possible if and only if p_{i,c} <= T.
    
    # 2. Non-negative Lag: R_i(tau) = 0 for tau < 0.
    #    R_i(tau) = integral s_{i,1}(t) s_{i,2}(t + tau) dt
    #    This is the overlap of [k_{i,1}, k_{i,1} + p_{i,1}] and [k_{i,2} - tau, k_{i,2} - tau + p_{i,2}]
    #    For tau < 0, the overlap is non-zero if there exists t such that:
    #    k_{i,1} <= t <= k_{i,1} + p_{i,1} AND k_{i,2} - tau <= t <= k_{i,2} - tau + p_{i,2}
    #    The condition R_i(tau) = 0 for all tau < 0 means the intervals must not overlap for any tau < 0.
    #    This is equivalent to saying the start of the second interval must be >= end of the first.
    #    k_{i,2} >= k_{i,1} + p_{i,1}
    #    Combined with 0 <= k_{i,1} and k_{i,2} + p_{i,2} <= T:
    #    k_{i,1} >= 0, k_{i,2} >= k_{i,1} + p_{i,1}, k_{i,2} <= T - p_{i,2}
    #    This is possible iff 0 + p_{i,1} <= T - p_{i,2} => p_{i,1} + p_{i,2} <= T.
    
    # 3. Orthogonal Multiplexing: sum of p_{i,c} <= T for each c.
    
    # Check JOB i:
    # For each i, check if p_{i,1} + p_{i,2} > T.
    # If multiple exist, the problem implies exactly one root cause.
    # The problem states "exactly one of the following... is the root cause".
    # We check JOB i first.
    
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD c:
    # Sum of p_{i,c} <= T for all i.
    sum1 = 0
    sum2 = 0
    for i in range(j):
        sum1 += int(input_data[2 + 2 * i])
        sum2 += int(input_data[3 + 2 * i])
        
    if sum1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
