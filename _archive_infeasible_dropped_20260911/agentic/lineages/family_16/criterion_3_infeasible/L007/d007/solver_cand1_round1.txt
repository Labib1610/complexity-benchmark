import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    # Constraints:
    # 1. JOB i: 
    #    f_{i,1} and f_{i,2} are in [0, 1] and supported on [0, T].
    #    Integral f_{i,1} = p_{i,1}, Integral f_{i,2} = p_{i,2}.
    #    Integral f_{i,1}(x) * (Integral_{0}^x f_{i,2}(t) dt) dx = 0.
    #    Since f_{i,1}, f_{i,2} >= 0, the integral is 0 iff f_{i,1}(x) * F_{i,2}(x) = 0 a.e.
    #    where F_{i,2}(x) is the cumulative integral of f_{i,2}.
    #    If f_{i,1} is non-zero on some set of positive measure, F_{i,2} must be 0 on that set.
    #    Since F_{i,2} is non-decreasing, if F_{i,2}(x) = 0 on a set, it must be 0 on [0, x].
    #    Thus, f_{i,1} must be supported on a set where f_{i,2} is 0.
    #    This implies the support of f_{i,1} must be to the left of the support of f_{i,2}.
    #    The total length of the support of f_{i,1} is at least p_{i,1} (since f <= 1).
    #    The total length of the support of f_{i,2} is at least p_{i,2}.
    #    So p_{i,1} + p_{i,2} <= T. If p_{i,1} + p_{i,2} > T, JOB i is impossible.
    
    # 2. LOAD c:
    #    For a fixed c, sum of integrals of f_{i,c} <= T (since they are orthogonal and f <= 1).
    #    Sum_{i=1}^j p_{i,c} <= T. If sum > T, LOAD c is impossible.

    # Check JOB i
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD c
    sum1 = 0
    sum2 = 0
    for i in range(j):
        sum1 += int(input_data[2 + 2 * i])
        sum2 += int(input_data[3 + 2 * i])
    
    if sum1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
