import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p.append((p1, p2))
        idx += 2
        
    # Check JOB i:
    # K_{i,1} is [a, a+p1], K_{i,2} is [b, b+p2]
    # Subspace Confinement: 0 <= a, a+p1 <= T, 0 <= b, b+p2 <= T
    # Asymmetric Ordering: K_{i,1} x K_{i,2} subset H => v >= u for all u in K_{i,1}, v in K_{i,2}
    # This is equivalent to min(K_{i,2}) >= max(K_{i,1}) => b >= a + p1
    # So we need a + p1 <= b and b + p2 <= T
    # This implies a + p1 + p2 <= T.
    # If p1 + p2 > T, it's impossible.
    for i in range(j):
        if p[i][0] + p[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c:
    # Subspace Confinement: K_{i,c} subset [0, T]
    # Interior Disjointness: interiors of K_{i,c} are disjoint.
    # Since K_{i,c} are connected and compact, they are intervals [a_i, a_i + p_{i,c}].
    # The sum of lengths of disjoint intervals in [0, T] must be <= T.
    # sum(p_{i,c}) <= T
    
    sum1 = sum(x[0] for x in p)
    if sum1 > T:
        print("LOAD 1")
        return
        
    sum2 = sum(x[1] for x in p)
    if sum2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
