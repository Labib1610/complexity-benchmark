```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    requests = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        requests.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Separation rule: S_{i,1} = [x, x+p1-1], S_{i,2} = [y, y+p2-1]
    # x >= 1, x+p1-1 < y, y+p2-1 <= T
    # Minimum y is x+p1. So x+p1+p2-1 <= T => p1+p2 <= T
    for i in range(j):
        if requests[i][0] + requests[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c
    # Exclusivity rule: sum of sizes for index c must be <= T
    sum1 = 0
    sum2 = 0
    for p1, p2 in requests:
        sum1 += p1
        sum2 += p2
        
    if sum1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
