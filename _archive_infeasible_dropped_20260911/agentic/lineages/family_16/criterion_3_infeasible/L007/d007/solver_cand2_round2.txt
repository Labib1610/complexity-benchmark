```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    items = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        items.append((p1, p2))
        idx += 2
        
    # Check JOB i: y_i >= x_i + p1, x_i >= 0, x_i + p1 <= T, y_i >= 0, y_i + p2 <= T
    # This implies y_i >= x_i + p1 >= p1.
    # So we need y_i + p2 >= p1 + p2.
    # Since y_i + p2 <= T, we need p1 + p2 <= T.
    # If p1 + p2 > T, it's impossible.
    for i in range(j):
        p1, p2 = items[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD c: sum of lengths <= T
    sum_p1 = 0
    sum_p2 = 0
    for p1, p2 in items:
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
