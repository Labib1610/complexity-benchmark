```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    items = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        items.append((p1, p2, i + 1))
        idx += 2
        
    # Check JOB i
    # Item i cannot be satisfied if p1 + p2 > T
    for p1, p2, i in items:
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
            
    # Check LOAD c
    # Stage 1: sum of p1 > T
    # Stage 2: sum of p2 > T
    sum_p1 = sum(item[0] for item in items)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    sum_p2 = sum(item[1] for item in items)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
