```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    items = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        items.append((p1, p2, i))
        sum_p1 += p1
        sum_p2 += p2
        idx += 2
        
    # Check JOB i
    # Requirement 1 & 2: S_{i,1} + p_{i,1} + p_{i,2} <= T
    # Since S_{i,1} >= 0, this is impossible if p_{i,1} + p_{i,2} > T
    for p1, p2, i in items:
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
            
    # Check LOAD c
    # Requirement 1 & 3: Sum of p_{i,c} <= T
    if sum_p1 > t:
        print("LOAD 1")
        return
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
