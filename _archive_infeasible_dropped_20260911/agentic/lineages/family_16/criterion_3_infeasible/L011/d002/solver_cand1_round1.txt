```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    transmissions = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        transmissions.append((p1, p2))
        idx += 2
        
    # Check for JOB i
    for i in range(j):
        p1, p2 = transmissions[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c
    sum1 = 0
    sum2 = 0
    for p1, p2 in transmissions:
        sum1 += p1
        sum2 += p2
        
    if sum1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
