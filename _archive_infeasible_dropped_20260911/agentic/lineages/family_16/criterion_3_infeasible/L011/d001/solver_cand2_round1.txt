```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2, i + 1))
        idx += 2
        
    # Check for JOB i bottleneck
    for i in range(j):
        p1, p2, original_idx = jobs[i]
        if p1 + p2 > t:
            print(f"JOB {original_idx}")
            return
            
    # Check for LOAD 1 bottleneck
    sum1 = 0
    for i in range(j):
        sum1 += jobs[i][0]
    if sum1 > t:
        print("LOAD 1")
        return
        
    # Check for LOAD 2 bottleneck
    sum2 = 0
    for i in range(j):
        sum2 += jobs[i][1]
    if sum2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
