import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    jobs = []
    idx = 2
    load1 = 0
    load2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check if individual job exceeds T
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
        
        load1 += p1
        load2 += p2
        
    # Check if machine loads exceed T
    if load1 > T:
        print("LOAD 1")
    elif load2 > T:
        print("LOAD 2")
    else:
        # If neither individual jobs nor machine loads exceed T, 
        # Johnson's rule guarantees the schedule meets the deadline.
        # The problem implies it is impossible, so this branch shouldn't be reached.
        pass

if __name__ == '__main__':
    solve()
