```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check Job Sequential Bottleneck
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check Machine Aggregate Bottleneck
    sum1 = sum(job[0] for job in jobs)
    if sum1 > t:
        print("LOAD 1")
        return
        
    sum2 = sum(job[1] for job in jobs)
    if sum2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
