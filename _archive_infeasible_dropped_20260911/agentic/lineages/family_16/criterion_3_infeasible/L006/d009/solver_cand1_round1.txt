```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum_p1 = 0
    sum_p2 = 0
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Work done by F = i + j from (0,0) to (p1, p2) is p1 + p2
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Work done by F = i + j from (0,0) to (sum_p1, 0) is sum_p1
    if sum_p1 > t:
        print("LOAD 1")
        return
    
    # Work done by F = i + j from (0,0) to (0, sum_p2) is sum_p2
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
