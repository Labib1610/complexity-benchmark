```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    jobs = []
    idx = 2
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check LOAD 1: sum of p1 > T
    sum_p1 = sum(job[0] for job in jobs)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2: sum of p2 > T
    sum_p2 = sum(job[1] for job in jobs)
    if sum_p2 > T:
        print("LOAD 2")
        return
        
    # Check JOB i: p1_i + p2_i > T
    # The hypothetical sequence for job i:
    # E_i = p1_i
    # T_i = max(E_i, T_{i-1}) + p2_i
    # Since all other jobs are 0, T_{i-1} = 0.
    # So T_i = p1_i + p2_i.
    # The final T_j will be T_i because all subsequent jobs have 0 time.
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > T:
            print(f"JOB {i + 1}")
            return

if __name__ == '__main__':
    solve()
```
