```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    
    jobs = []
    for _ in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        jobs.append((p1, p2))
    
    # Check LOAD family first
    # LOAD 1: p2 all zero, T_j = sum of all p1
    total_p1 = sum(p1 for p1, p2 in jobs)
    if total_p1 > T:
        print("LOAD 1")
        return
    
    # LOAD 2: p1 all zero, T_j = sum of all p2
    total_p2 = sum(p2 for p1, p2 in jobs)
    if total_p2 > T:
        print("LOAD 2")
        return
    
    # Check JOB family
    # For job i alone: T_j = p1[i] + p2[i]
    for i, (p1, p2) in enumerate(jobs, 1):
        if p1 + p2 > T:
            print(f"JOB {i}")
            return

if __name__ == "__main__":
    solve()
```
