```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    requests = []
    idx = 2
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        requests.append((p1, p2))
        idx += 2
        
    # Check Isolated Job Obstruction
    # For a single request i:
    # secured = p1
    # exposed = max(0, 0 - p1) + p2 = p2
    # metric = p1 + p2
    for i in range(j):
        p1, p2 = requests[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check Continuous Utilization Exhaustion
    # c = 1: Skip step 3 (p2 = 0)
    # secured = sum(p1)
    # exposed = 0
    # metric = sum(p1)
    sum_p1 = sum(r[0] for r in requests)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # c = 2: Skip steps 1 and 2 (p1 = 0)
    # secured = 0
    # exposed = sum(p2)
    # metric = sum(p2)
    sum_p2 = sum(r[1] for r in requests)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
