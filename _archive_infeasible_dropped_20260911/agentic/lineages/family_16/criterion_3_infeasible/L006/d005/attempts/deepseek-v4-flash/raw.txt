We can solve this by first checking the sequential bottleneck condition for each job, and if none match, then checking the load condition for each machine.  

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    j = int(data[0])
    T = int(data[1])
    
    sum1 = 0
    sum2 = 0
    idx = 2
    for i in range(1, j + 1):
        p1 = int(data[idx])
        p2 = int(data[idx + 1])
        idx += 2
        sum1 += p1
        sum2 += p2
        if T - p1 < p2:
            print(f"JOB {i}")
            return
    
    if T < sum1:
        print("LOAD 1")
    elif T < sum2:
        print("LOAD 2")

if __name__ == "__main__":
    main()
```
