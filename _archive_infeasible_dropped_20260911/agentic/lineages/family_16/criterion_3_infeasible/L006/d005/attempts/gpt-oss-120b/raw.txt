```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    sum1 = 0
    sum2 = 0
    job_idx = None
    for i in range(1, j + 1):
        p1 = int(next(it))
        p2 = int(next(it))
        sum1 += p1
        sum2 += p2
        if job_idx is None and T - p1 < p2:
            job_idx = i
    out = []
    if job_idx is not None:
        out.append(f"JOB {job_idx}")
    else:
        if T < sum1:
            out.append("LOAD 1")
        else:
            out.append("LOAD 2")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```
