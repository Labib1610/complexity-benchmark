```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum_p1 = 0
    sum_p2 = 0
    
    # Iterate through jobs to check for Sequential Bottleneck and calculate totals
    # for Machine Capacity Limit
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        # Sequential Bottleneck check:
        # Job i requires p1 + p2 time. If p1 + p2 > T, it's a violation.
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Machine Capacity Limit check:
    # Machine 1 must process all jobs, total time sum_p1.
    # Machine 2 must process all jobs, total time sum_p2.
    if sum_p1 > t:
        print("LOAD 1")
    else:
        # Since the problem guarantees it's impossible to meet the deadline,
        # if sum_p1 <= t and no JOB i violates, then sum_p2 must be > t.
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
