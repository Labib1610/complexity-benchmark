```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum_p1 = 0
    sum_p2 = 0
    
    # Process jobs and check for Sequential Bottleneck
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        sum_p1 += p1
        sum_p2 += p2
        
        # Sequential Bottleneck: T - p1 < p2  =>  T < p1 + p2
        if t < p1 + p2:
            print(f"JOB {i + 1}")
            return
            
    # Machine Capacity Limit: T - sum_p < 0 => T < sum_p
    if t < sum_p1:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
