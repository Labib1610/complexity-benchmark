```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_sum = 0
    p2_sum = 0
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check Sequential Bottleneck
        if t - p1 < p2:
            print(f"JOB {i}")
            return
        
        p1_sum += p1
        p2_sum += p2
        
    # Check Machine Capacity Limit
    if p1_sum > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
