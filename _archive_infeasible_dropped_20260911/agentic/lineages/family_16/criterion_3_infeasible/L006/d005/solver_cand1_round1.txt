```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    sum1 = 0
    sum2 = 0
    
    # Iterate through jobs to check for Sequential Bottleneck and calculate total loads
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        # Sequential Bottleneck:
        # A job i is invalid if p1 + p2 > T.
        # If p1 + p2 > T, then for any t1 + t2 = T, 
        # either t1 < p1 or t2 < p2 must hold.
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
        
        sum1 += p1
        sum2 += p2
        
    # Machine Capacity Limit:
    # If the sum of all processing times on a machine exceeds T,
    # then the set of all jobs is a subset that violates the capacity.
    if sum1 > T:
        print("LOAD 1")
    elif sum2 > T:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
