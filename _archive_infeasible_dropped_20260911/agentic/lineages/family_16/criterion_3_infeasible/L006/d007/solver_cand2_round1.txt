```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for _ in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Check LOAD 1: sum(p1) > T
    if sum(p1) > T:
        print("LOAD 1")
        return
    
    # Check LOAD 2: sum(p2) > T
    if sum(p2) > T:
        print("LOAD 2")
        return
    
    # Check JOB i: p1[i] + p2[i] > T
    # The hypothetical sequence for request i:
    # A_i = p1[i], B_i = p1[i] + p2[i]
    # For k > i, A_k = p1[i], B_k = max(p1[i], B_{k-1}) + 0
    # Since B_i = p1[i] + p2[i], B_j will be at least p1[i] + p2[i]
    # Actually, the problem defines the hypothetical sequence as:
    # p1_k = 0, p2_k = 0 for k != i
    # A_i = p1[i], B_i = p1[i] + p2[i]
    # For k > i, A_k = A_i = p1[i], B_k = max(A_i, B_{k-1}) + 0
    # This implies B_j = B_i = p1[i] + p2[i]
    for i in range(j):
        if p1[i] + p2[i] > T:
            print(f"JOB {i + 1}")
            return

if __name__ == '__main__':
    solve()
```
