```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    entities = []
    idx = 2
    for i in range(j):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        entities.append((x, y, i + 1))
        idx += 2
        
    # Check JOB certificate
    for x, y, i in entities:
        if x + y > T:
            print(f"JOB {i}")
            return
            
    # Check LOAD 1: sum of X_k > T
    # In LOAD 1, we have f(k) + X_k <= g(k).
    # Since intervals [f(k), f(k)+X_k) are disjoint and f(k) >= 0,
    # the minimum span is sum(X_k).
    sum_x = sum(e[0] for e in entities)
    if sum_x > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2: sum of Y_k > T
    # In LOAD 2, X_k = 0, so g(k) >= f(k).
    # Axiom A3 says [g(k), g(k)+Y_k) are disjoint.
    # Since g(k) >= 0, the minimum span is sum(Y_k).
    sum_y = sum(e[1] for e in entities)
    if sum_y > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
