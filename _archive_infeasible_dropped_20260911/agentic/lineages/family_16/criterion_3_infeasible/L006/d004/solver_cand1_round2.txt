```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    elements = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        elements.append((p1, p2))
        idx += 2
        
    # Check for JOB i certificate
    # A(i) length p1, B(i) length p2, A(i) <= B(i)
    # Min length required is p1 + p2
    for i in range(j):
        p1, p2 = elements[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD c certificate
    # Sum of lengths of all elements in phase c must be <= T
    sum_p1 = 0
    sum_p2 = 0
    for p1, p2 in elements:
        sum_p1 += p1
        sum_p2 += p2
        
    if sum_p1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
