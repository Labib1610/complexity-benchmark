```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum1 = 0
    sum2 = 0
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check Sequential Job Limit
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum1 += p1
        sum2 += p2
        
    # Check Total Machine Workload
    if sum1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
