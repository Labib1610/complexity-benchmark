```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum1 = 0
    sum2 = 0
    
    job_witness = -1
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        if p1 + p2 > t:
            if job_witness == -1:
                job_witness = i
        
        sum1 += p1
        sum2 += p2
        
    load_witness = -1
    if sum1 > t:
        load_witness = 1
    elif sum2 > t:
        load_witness = 2
        
    if job_witness != -1 and load_witness != -1:
        print(f"LOAD {load_witness}")
    elif job_witness != -1:
        print(f"JOB {job_witness}")
    else:
        print(f"LOAD {load_witness}")

if __name__ == '__main__':
    solve()
```
