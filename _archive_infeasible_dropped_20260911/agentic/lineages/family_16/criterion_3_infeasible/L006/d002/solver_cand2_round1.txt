```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check for Individual Job Bottleneck
    job_idx = -1
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > t:
            job_idx = i + 1
            break
            
    # Check for Machine Capacity Bottleneck
    load1 = 0
    load2 = 0
    for p1, p2 in jobs:
        load1 += p1
        load2 += p2
        
    load_c = -1
    if load1 > t:
        load_c = 1
    elif load2 > t:
        load_c = 2
        
    # Priority rule: If both exist, output LOAD c.
    # Wait, the prompt says: "If at least one individual-job witness exists 
    # and at least one machine-load witness exists, output LOAD c".
    # This means LOAD has priority over JOB.
    
    if load_c != -1:
        print(f"LOAD {load_c}")
    elif job_idx != -1:
        print(f"JOB {job_idx}")

if __name__ == '__main__':
    solve()
```
