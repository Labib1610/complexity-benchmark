```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    s1 = 0
    s2 = 0
    
    # Process jobs and check for Isolated Job Bottleneck
    # We store the sum while checking for the job bottleneck
    for i in range(j):
        p1 = int(input_data[2 + 2 * i])
        p2 = int(input_data[3 + 2 * i])
        
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
        
        s1 += p1
        s2 += p2
        
    # If no job bottleneck, check for Machine Saturation
    if s1 > t:
        print("LOAD 1")
    elif s2 > t:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
