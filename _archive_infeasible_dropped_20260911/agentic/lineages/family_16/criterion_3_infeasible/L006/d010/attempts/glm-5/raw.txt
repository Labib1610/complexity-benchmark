import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    
    p1_list = []
    p2_list = []
    for _ in range(j):
        p1_list.append(int(next(it)))
        p2_list.append(int(next(it)))
    
    # Check Type A: find i such that f_i alone exceeds T
    # f_i(0, 0) = (p1_i, max(p1_i + p2_i, p2_i))
    # V = max(p1_i + p2_i, p2_i)
    for i in range(j):
        p1 = p1_list[i]
        p2 = p2_list[i]
        V = max(p1 + p2, p2)
        if V > T:
            print(f"JOB {i + 1}")
            return
    
    # Check Type B
    # For c=1: g_k(u, v) = (u + p1_k, max(u + p1_k, v))
    # V = sum of all p1_k
    total_p1 = sum(p1_list)
    if total_p1 > T:
        print("LOAD 1")
        return
    
    # For c=2: h_k(u, v) = (u, max(u + p2_k, v + p2_k))
    # V = max over all prefixes of (prefix_sum_p1 + p2_k)
    # where prefix_sum_p1 is sum of p1_1 to p1_{k-1}
    prefix_p1 = 0
    max_V_c2 = 0
    for k in range(j):
        p2 = p2_list[k]
        candidate = prefix_p1 + p2
        if candidate > max_V_c2:
            max_V_c2 = candidate
        prefix_p1 += p1_list[k]
    
    if max_V_c2 > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    solve()
