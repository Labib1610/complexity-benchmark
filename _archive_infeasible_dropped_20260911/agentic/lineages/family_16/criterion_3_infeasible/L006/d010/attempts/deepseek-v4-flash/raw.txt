We can solve this by simulating both possible failure modes efficiently.

For **Type A**, we need to check each job individually: if we apply only that job and skip all others, does the final `v` exceed `T`?  
We can precompute prefix and suffix effects to answer each job in O(1).

For **Type B**, we simulate the two modified pipelines directly in O(j) each.

Here’s the implementation:

```python
import sys

def solve():
    input = sys.stdin.readline
    j, T = map(int, input().split())
    p1 = [0] * (j + 1)
    p2 = [0] * (j + 1)
    for i in range(1, j + 1):
        a, b = map(int, input().split())
        p1[i] = a
        p2[i] = b

    # Type A check
    # prefix_u[i] = u after applying jobs 1..i
    # prefix_v[i] = v after applying jobs 1..i
    pref_u = [0] * (j + 1)
    pref_v = [0] * (j + 1)
    for i in range(1, j + 1):
        u = pref_u[i-1]
        v = pref_v[i-1]
        nu = u + p1[i]
        nv = max(nu + p2[i], v + p2[i])
        pref_u[i] = nu
        pref_v[i] = nv

    # suffix_u[i] = u after applying jobs i..j starting from (0,0)
    # suffix_v[i] = v after applying jobs i..j starting from (0,0)
    suff_u = [0] * (j + 2)
    suff_v = [0] * (j + 2)
    for i in range(j, 0, -1):
        u = suff_u[i+1]
        v = suff_v[i+1]
        nu = u + p1[i]
        nv = max(nu + p2[i], v + p2[i])
        suff_u[i] = nu
        suff_v[i] = nv

    # For each i, simulate only job i applied to (0,0)
    for i in range(1, j + 1):
        u = p1[i]
        v = max(u + p2[i], p2[i])
        if v > T:
            print(f"JOB {i}")
            return

    # Type B check
    # c=1: g_k(u,v) = (u+p1, max(u+p1, v))
    u = 0
    v = 0
    for i in range(1, j + 1):
        u = u + p1[i]
        v = max(u, v)
    if v > T:
        print("LOAD 1")
        return

    # c=2: h_k(u,v) = (u, max(u+p2, v+p2))
    u = 0
    v = 0
    for i in range(1, j + 1):
        u = u  # unchanged
        v = max(u + p2[i], v + p2[i])
    if v > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    solve()
```
