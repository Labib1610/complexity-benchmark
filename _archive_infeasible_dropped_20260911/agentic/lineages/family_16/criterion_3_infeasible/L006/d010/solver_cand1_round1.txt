```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    params = []
    idx = 2
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        params.append((p1, p2))
        idx += 2
        
    # Check Type A: JOB i
    # f_i(0, 0) = (p1_i, max(p1_i + p2_i, p2_i)) = (p1_i, p1_i + p2_i)
    for i in range(j):
        p1, p2 = params[i]
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
            
    # Check Type B: LOAD c
    # c=1: g_k(u, v) = (u + p1_k, max(u + p1_k, v))
    # Start (0,0). After k steps: (sum(p1_1..k), max(sum(p1_1..k), sum(p1_1..k-1), ..., p1_1))
    # Actually, g_k(u, v) = (u + p1_k, max(u + p1_k, v))
    # (0,0) -> (p1_1, p1_1) -> (p1_1+p1_2, max(p1_1+p1_2, p1_1)) = (p1_1+p1_2, p1_1+p1_2)
    # So final V = sum(p1_1..j)
    s1 = sum(p[0] for p in params)
    if s1 > t:
        print("LOAD 1")
        return
        
    # c=2: h_k(u, v) = (u, max(u + p2_k, v + p2_k))
    # (0,0) -> (0, p2_1) -> (0, max(p2_1, p2_2)) -> (0, max(p2_1, p2_2, ..., p2_j))
    # Wait, let's re-check h_k:
    # h_1(0,0) = (0, max(0+p2_1, 0+p2_1)) = (0, p2_1)
    # h_2(0, p2_1) = (0, max(0+p2_2, p2_1+p2_2)) = (0, p2_1+p2_2)
    # So final V = sum(p2_1..j)
    s2 = sum(p[1] for p in params)
    if s2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
