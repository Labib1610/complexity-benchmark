```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for _ in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Check LOAD 1: sum(p1) > T
    if sum(p1) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2: sum(p2) + min(p1) > T
    # The duration for machine 2 is sum(p2) + first_start_time
    # The first job starts on M2 at p1[0], subsequent jobs start at max(finish_m1, finish_m2)
    # Actually, for LOAD 2, the duration is p1[0] + sum(p2)
    if p1[0] + sum(p2) > T:
        print("LOAD 2")
        return
        
    # Check JOB i: p1[i] + p2[i] > T
    for i in range(j):
        if p1[i] + p2[i] > T:
            print(f"JOB {i+1}")
            return

if __name__ == '__main__':
    solve()
```
