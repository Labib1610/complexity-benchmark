```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for _ in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Check LOAD 1: sum of all p1 > T
    if sum(p1) > t:
        print("LOAD 1")
        return
    
    # Check LOAD 2: sum of all p2 > T
    if sum(p2) > t:
        print("LOAD 2")
        return
    
    # If not LOAD, it must be JOB i.
    # The condition for JOB i is: p1[i] + p2[i] > T
    # Because in the modified network for job i, the max path is 
    # (sum of p1 from 1 to i) + (sum of p2 from i to j)
    # But since all other jobs are 0, this simplifies to p1[i] + p2[i].
    # Wait, the path is:
    # Stage 1 of job 1 -> ... -> Stage 1 of job i -> Stage 2 of job i -> ... -> Stage 2 of job j
    # If all other jobs are 0, the path weight is p1[i] + p2[i].
    # So we just need to find i such that p1[i] + p2[i] > T.
    
    for i in range(j):
        if p1[i] + p2[i] > t:
            print(f"JOB {i + 1}")
            return

if __name__ == '__main__':
    solve()
```
