```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    jobs = []
    idx = 2
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Hypothetical: E_i = p1_i, T_i = p1_i + p2_i. 
    # For all k != i, E_k = E_{k-1}, T_k = T_{k-1}.
    # So T_j = T_i = p1_i + p2_i.
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Suppress machine 2: E_k = sum(p1_1...p1_k), T_k = max(E_k, T_{k-1})
    # Since E_k is non-decreasing, T_k = E_k.
    # T_j = sum(p1_1...p1_j)
    sum_p1 = sum(job[0] for job in jobs)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Suppress machine 1: E_k = 0, T_k = max(0, T_{k-1}) + p2_k
    # T_k = sum(p2_1...p2_k)
    sum_p2 = sum(job[1] for job in jobs)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
