```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    
    p1_times = []
    p2_times = []
    
    index = 2
    for _ in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        p1_times.append(p1)
        p2_times.append(p2)
        index += 2
    
    # Check for Isolated Job Obstruction
    for i in range(j):
        if p1_times[i] + p2_times[i] > T:
            print(f"JOB {i+1}")
            return
    
    # Check for Continuous Utilization Exhaustion
    total_p1 = sum(p1_times)
    total_p2 = sum(p2_times)
    
    if total_p1 > T:
        print("LOAD 1")
        return
    elif total_p2 > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    main()
```
