```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum1 = 0
    sum2 = 0
    
    # We need to check for JOB i first or simultaneously.
    # The problem guarantees exactly one certificate exists.
    # We can iterate through the jobs to check for JOB i and accumulate sums.
    
    idx = 2
    for i in range(1, j + 1):
        s1 = int(input_data[idx])
        s2 = int(input_data[idx + 1])
        idx += 2
        
        # Check Job Bottleneck
        if s1 + s2 > t:
            print(f"JOB {i}")
            return
        
        sum1 += s1
        sum2 += s2
        
    # Check Machine Bottleneck
    if sum1 > t:
        print("LOAD 1")
    elif sum2 > t:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
