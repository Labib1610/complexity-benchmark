```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum_a = 0
    sum_e = 0
    
    for i in range(j):
        a = int(input_data[2 + 2 * i])
        e = int(input_data[3 + 2 * i])
        
        # Check JOB condition
        if a + e > t:
            print(f"JOB {i + 1}")
            return
        
        sum_a += a
        sum_e += e
        
    # Check LOAD condition
    if sum_a > t:
        print("LOAD 1")
    elif sum_e > t:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
