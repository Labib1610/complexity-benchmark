```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum1 = 0
    sum2 = 0
    
    for i in range(j):
        x = int(input_data[2 + 2 * i])
        y = int(input_data[3 + 2 * i])
        
        if x + y > t:
            print(f"JOB {i + 1}")
            return
        
        sum1 += x
        sum2 += y
        
    if sum1 > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
