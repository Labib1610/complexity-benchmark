```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum1 = 0
    sum2 = 0
    
    # We need to check conditions in order.
    # Since we need to output the first one that fails, 
    # we can process the input and check as we go.
    
    # First, check LOAD 1 and LOAD 2
    # We need to read all p1, p2 to calculate sums
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sum1 += p1
        sum2 += p2
        jobs.append((p1, p2))
        idx += 2
        
    if sum1 > t:
        print("LOAD 1")
        return
    
    if sum2 > t:
        print("LOAD 2")
        return
    
    # If neither LOAD 1 nor LOAD 2, check JOB i
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > t:
            print(f"JOB {i + 1}")
            return

if __name__ == '__main__':
    solve()
```
