```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum1 = 0
    sum2 = 0
    first_job_idx = -1
    
    idx = 2
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        sum1 += p1
        sum2 += p2
        
        if first_job_idx == -1 and (p1 + p2 > t):
            first_job_idx = i
            
    if sum1 > t:
        print("LOAD 1")
    elif sum2 > t:
        print("LOAD 2")
    else:
        print(f"JOB {first_job_idx}")

if __name__ == '__main__':
    solve()
```
