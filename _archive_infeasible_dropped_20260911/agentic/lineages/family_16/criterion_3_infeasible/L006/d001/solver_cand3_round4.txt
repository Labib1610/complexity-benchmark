```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum_a = 0
    sum_b = 0
    
    idx = 2
    for i in range(1, j + 1):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        idx += 2
        
        # Check Sequential Bottleneck
        if a + b > t:
            print(f"JOB {i}")
            return
        
        sum_a += a
        sum_b += b
        
    # Check Throughput Bottleneck
    if sum_a > t:
        print("LOAD 1")
    else:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
