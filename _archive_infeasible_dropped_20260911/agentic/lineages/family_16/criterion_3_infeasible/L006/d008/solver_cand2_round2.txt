import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for _ in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Check JOB i
    for i in range(j):
        # E({i}, {i})
        # X_k = p1[i] if k == i else 0
        # Y_k = max(X_k, Y_{k-1}) + (p2[i] if k == i else 0)
        # For k < i: X_k = 0, Y_k = 0
        # For k = i: X_i = p1[i], Y_i = p1[i] + p2[i]
        # For k > i: X_k = p1[i], Y_k = max(p1[i], Y_{k-1}) = p1[i] + p2[i]
        if p1[i] + p2[i] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1: E({1..j}, {})
    # X_k = sum(p1[0..k])
    # Y_k = max(X_k, Y_{k-1})
    # Y_j = X_j = sum(p1)
    s1_sum = sum(p1)
    if s1_sum > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2: E({}, {1..j})
    # X_k = 0
    # Y_k = max(0, Y_{k-1}) + p2[k] = Y_{k-1} + p2[k]
    # Y_j = sum(p2)
    s2_sum = sum(p2)
    if s2_sum > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
