import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for _ in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Check Isolated Mask Obstruction: JOB i
    # S1_i = p1[i], S2_i = p1[i] + p2[i]
    # For k > i, S1_k = p1[i], S2_k = S2_{k-1} + max(p1[i] - S2_{k-1}, 0)
    # This simplifies to S2_k = max(S2_{k-1}, p1[i])
    # So S2_j = max(p1[i] + p2[i], p1[i]) = p1[i] + p2[i]
    for i in range(j):
        if p1[i] + p2[i] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check Asymmetric Mask Exhaustion: LOAD c
    # c = 1: M_k = 1, N_k = 0
    # S1_k = sum(p1[0...k]), S2_k = S2_{k-1} + max(S1_k - S2_{k-1}, 0)
    # This is equivalent to S2_k = max(S1_k, S2_{k-1})
    # Since S1_k is non-decreasing, S2_k = S1_k
    # So S2_j = sum(p1)
    s1_sum = sum(p1)
    if s1_sum > t:
        print("LOAD 1")
        return
        
    # c = 2: M_k = 0, N_k = 1
    # S1_k = 0, S2_k = S2_{k-1} + max(0 - S2_{k-1}, 0) + p2_k
    # S2_k = S2_{k-1} + p2_k
    # So S2_j = sum(p2)
    s2_sum = sum(p2)
    if s2_sum > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
