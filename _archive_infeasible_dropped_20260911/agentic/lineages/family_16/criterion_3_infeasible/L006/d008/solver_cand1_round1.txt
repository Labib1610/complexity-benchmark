```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    requests = []
    idx = 2
    for _ in range(j):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        requests.append((u, v))
        idx += 2
        
    # Check Isolated Job Obstruction
    # For a single job i: S = u_i, D = v_i. Metric = u_i + v_i.
    for i in range(j):
        u, v = requests[i]
        if u + v > T:
            print(f"JOB {i + 1}")
            return
            
    # Check Continuous Utilization Exhaustion
    # LOAD 1: u = p1_k, v = 0.
    # S = sum(p1_k), D = 0. Metric = sum(p1_k).
    sum_u = sum(r[0] for r in requests)
    if sum_u > T:
        print("LOAD 1")
        return
        
    # LOAD 2: u = 0, v = p2_k.
    # S = 0, D = sum(p2_k). Metric = sum(p2_k).
    sum_v = sum(r[1] for r in requests)
    if sum_v > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
