import sys

def compute_final_metric(requests):
    S = 0
    D = 0
    for u, v in requests:
        S += u
        D = max(0, D - u) + v
    return S + D

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    T = int(data[1])
    
    requests = []
    index = 2
    for _ in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        requests.append((p1, p2))
        index += 2
    
    # Check Isolated Job Obstruction
    for i in range(j):
        # Create a hypothetical sequence where only request i is active
        hyp_requests = [(0, 0)] * j
        hyp_requests[i] = requests[i]
        final_metric = compute_final_metric(hyp_requests)
        if final_metric > T:
            print(f"JOB {i+1}")
            return
    
    # Check Continuous Utilization Exhaustion
    # Case c=1: use u=p1,k and v=0
    hyp_requests_c1 = [(u, 0) for u, v in requests]
    final_metric_c1 = compute_final_metric(hyp_requests_c1)
    if final_metric_c1 > T:
        print("LOAD 1")
        return
        
    # Case c=2: use u=0 and v=p2,k
    hyp_requests_c2 = [(0, v) for u, v in requests]
    final_metric_c2 = compute_final_metric(hyp_requests_c2)
    if final_metric_c2 > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    main()
