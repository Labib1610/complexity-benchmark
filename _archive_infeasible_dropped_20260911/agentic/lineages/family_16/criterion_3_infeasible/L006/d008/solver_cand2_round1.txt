```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    requests = []
    idx = 2
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        requests.append((p1, p2))
        idx += 2
        
    # Check for JOB i
    # For a single job i:
    # X_i = p1_i, D_i = p1_i, Y_i = p1_i + p2_i
    # For k > i, X_k = p1_i, Y_k = p1_i + p2_i
    # So Y_j = p1_i + p2_i
    for i in range(j):
        if requests[i][0] + requests[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check for LOAD 1
    # All requests enabled for machine 1, none for machine 2
    # u_k = p1_k, v_k = 0
    # X_k = sum_{m=1}^k p1_m
    # D_k = max(0, X_k - Y_{k-1})
    # Y_k = Y_{k-1} + D_k
    # Since X_k is non-decreasing, X_k - Y_{k-1} >= 0 is likely
    # Y_k = X_k = sum_{m=1}^k p1_m
    # So Y_j = sum_{m=1}^j p1_m
    sum_p1 = sum(r[0] for r in requests)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check for LOAD 2
    # All requests enabled for machine 2, none for machine 1
    # u_k = 0, v_k = p2_k
    # X_k = 0
    # D_k = max(0, 0 - Y_{k-1}) = 0
    # Y_k = Y_{k-1} + 0 + p2_k = sum_{m=1}^k p2_m
    # So Y_j = sum_{m=1}^j p2_m
    sum_p2 = sum(r[1] for r in requests)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
