```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t_total = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        sum_p1 += p1
        sum_p2 += p2
        idx += 2
        
    # Check LOAD 1
    if sum_p1 > t_total:
        print("LOAD 1")
        return
    
    # Check LOAD 2
    if sum_p2 > t_total:
        print("LOAD 2")
        return
    
    # Check JOB i
    # Condition 2: Y(t) * (P1 - X(t)) = 0
    # For each index i, Y_i(t) can only be > 0 if X_i(t) = P1_i.
    # X_i(t) reaches P1_i at time t_i >= P1_i (due to Condition 3).
    # So Y_i(t) can only grow in the interval [P1_i, T].
    # The max growth of Y_i(t) is T - P1_i.
    # We need T - P1_i >= P2_i, or P1_i + P2_i > T.
    
    for i in range(j):
        if p1_list[i] + p2_list[i] > t_total:
            print(f"JOB {i + 1}")
            return

if __name__ == '__main__':
    solve()
```
