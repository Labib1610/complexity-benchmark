```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Constraint 1: 0 <= x_i, x_i + p1_i <= T
    # Constraint 2: 0 <= y_i, y_i + p2_i <= T
    # Constraint 3: x_i + p1_i <= y_i
    # Combined: p1_i + p2_i <= T
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Constraint 1: 0 <= x_k, x_k + p1_k <= T
    # Constraint 4: |x_u - x_v| >= max(p1_u, p1_v)
    # This is equivalent to scheduling jobs on a single machine of capacity T.
    # The minimum time required is the sum of all p1_k.
    if sum(job[0] for job in jobs) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Constraint 2: 0 <= y_k, y_k + p2_k <= T
    # Constraint 5: |y_u - y_v| >= max(p2_u, p2_v)
    # Similar to LOAD 1, the minimum time required is the sum of all p2_k.
    if sum(job[1] for job in jobs) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
