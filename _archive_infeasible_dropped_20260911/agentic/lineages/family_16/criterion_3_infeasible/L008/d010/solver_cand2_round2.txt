import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    tasks = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i: Property 2 requires all CPU cycles < all GPU cycles.
        # This implies the sets are disjoint and there exists a split point.
        # The minimum cycles required for task i is p1 + p2.
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Check LOAD 1: Sum of all CPU cycles must be <= T
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2: Sum of all GPU cycles must be <= T
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
