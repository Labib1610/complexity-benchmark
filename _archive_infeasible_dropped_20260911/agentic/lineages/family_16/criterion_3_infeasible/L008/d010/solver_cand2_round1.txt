```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for i in range(j):
        p1.append(int(input_data[idx]))
        p2.append(int(input_data[idx+1]))
        idx += 2
        
    # Check JOB i
    # Constraints 1-5: 0 <= A_i, B_i = A_i + p1_i, B_i <= C_i, C_i + p2_i = D_i, D_i <= T
    # A_i + p1_i <= C_i => A_i + p1_i + p2_i <= D_i
    # Since A_i >= 0 and D_i <= T, we need p1_i + p2_i <= T
    for i in range(j):
        if p1[i] + p2[i] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Constraints 1, 3, 6: 0 <= A_k, B_k = A_k + p1_k, B_k <= T
    # |A_u - A_v| + |B_u - B_v| >= p1_u + p1_v
    # Since B_k - A_k = p1_k, |A_u - A_v| + |(A_u + p1_u) - (A_v + p1_v)| >= p1_u + p1_v
    # Let f(A) = 2*A + p1. The condition is |f(A_u) - f(A_v)| >= p1_u + p1_v
    # This is equivalent to saying the intervals [A_k, A_k + p1_k] must be disjoint?
    # Actually, |A_u - A_v| + |B_u - B_v| >= p1_u + p1_v is equivalent to:
    # Either A_u + p1_u <= A_v or A_v + p1_v <= A_u.
    # This means the intervals [A_k, A_k + p1_k] must be disjoint.
    # Total length required = sum(p1_k) <= T.
    if sum(p1) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Constraints 2, 4, 7: 0 <= C_k, D_k = C_k + p2_k, D_k <= T
    # |C_u - C_v| + |D_u - D_v| >= p2_u + p2_v
    # Similarly, this implies intervals [C_k, C_k + p2_k] must be disjoint.
    # Total length required = sum(p2_k) <= T.
    if sum(p2) > T:
        print("LOAD 2")
        return

solve()
```
