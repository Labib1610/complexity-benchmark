```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    pairs = []
    idx = 2
    for i in range(j):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        pairs.append((u, v))
        idx += 2
        
    # Check JOB i
    # C1: x_i >= 0
    # C2: y_i <= T - v_i
    # C3: y_i >= x_i + u_i
    # Combined: x_i + u_i <= y_i <= T - v_i
    # So x_i + u_i <= T - v_i => x_i <= T - v_i - u_i
    # Since x_i >= 0, we need 0 <= T - v_i - u_i => u_i + v_i <= T
    for i in range(j):
        u, v = pairs[i]
        if u + v > T:
            print(f"JOB {i + 1}")
            return

    # Check LOAD 1
    # x_k >= 0, x_k + u_k <= T
    # C4: max(x_m, x_n) >= min(x_m + u_m, x_n + u_n)
    # This is equivalent to saying the intervals [x_k, x_k + u_k] have disjoint interiors.
    # If we sort them, the total length must be <= T.
    # Actually, the condition is: for any m, n, the intervals [x_m, x_m + u_m] and [x_n, x_n + u_n]
    # cannot overlap. This means the sum of u_k must be <= T.
    if sum(p[0] for p in pairs) > T:
        print("LOAD 1")
        return

    # Check LOAD 2
    # Similar to LOAD 1, but for y_k and v_k.
    # y_k >= 0, y_k + v_k <= T
    # C5: intervals [y_k, y_k + v_k] must be disjoint.
    if sum(p[1] for p in pairs) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
