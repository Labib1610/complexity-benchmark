```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    linkages = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        linkages.append((p1, p2))
        idx += 2
        
    # Check JOB i: Constraint 3 implies that for a single linkage,
    # the interval for V1 (length p1) and V2 (length p2) must be disjoint
    # or sequential. Specifically, V1(x) < p1 implies V2(x) = 0.
    # This means the interval where V2' = 1 must start after the interval
    # where V1' = 1 ends. Total length required = p1 + p2.
    for i in range(j):
        p1, p2 = linkages[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1: Sum of p1 must be <= T
    sum_p1 = sum(l[0] for l in linkages)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2: Sum of p2 must be <= T
    sum_p2 = sum(l[1] for l in linkages)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
