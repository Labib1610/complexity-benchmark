```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    sum_ma = 0
    sum_mb = 0
    
    # We need to check for JOB i, LOAD 1, or LOAD 2.
    # JOB i: m_A_i + m_B_i > T
    # LOAD 1: sum(m_A) > T
    # LOAD 2: sum(m_B) > T
    
    # The problem states exactly one of these is triggered.
    # We check them in order.
    
    for i in range(j):
        ma = int(input_data[2 + 2 * i])
        mb = int(input_data[3 + 2 * i])
        
        if ma + mb > t:
            print(f"JOB {i + 1}")
            return
        
        sum_ma += ma
        sum_mb += mb
        
    if sum_ma > t:
        print("LOAD 1")
    elif sum_mb > t:
        print("LOAD 2")

if __name__ == '__main__':
    solve()
```
