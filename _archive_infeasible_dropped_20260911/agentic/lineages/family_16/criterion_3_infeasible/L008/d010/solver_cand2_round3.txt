```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    processes = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        processes.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Constraint 1: x_i >= p1_i/2, x_i <= T - p1_i/2
    # Constraint 2: y_i >= p2_i/2, y_i <= T - p2_i/2
    # Constraint 3: x_i + p1_i/2 <= y_i - p2_i/2 => y_i - x_i >= (p1_i + p2_i)/2
    # Max y_i - x_i = (T - p2_i/2) - (p1_i/2) = T - (p1_i + p2_i)/2
    # So we need T - (p1_i + p2_i)/2 >= (p1_i + p2_i)/2
    # T >= p1_i + p2_i
    for i in range(j):
        p1, p2 = processes[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Sum of p1_k <= T
    # Constraint 4: |x_u - x_v| >= (p1_u + p1_v)/2
    # This is equivalent to packing intervals of length p1_k into [0, T]
    # The total length required is sum(p1_k) <= T
    if sum(p[0] for p in processes) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Similarly, sum of p2_k <= T
    if sum(p[1] for p in processes) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
