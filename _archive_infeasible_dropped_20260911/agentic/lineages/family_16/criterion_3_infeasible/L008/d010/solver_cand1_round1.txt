```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    rows = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        rows.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Constraint 3: Alpha 1s at [a_start, a_end], Beta 1s at [b_start, b_end]
    # a_end = a_start + p1 - 1
    # b_start = b_end - p2 + 1
    # Constraint 3: a_end < b_start => a_start + p1 - 1 < b_end - p2 + 1
    # Minimum columns needed: p1 + p2. If p1 + p2 > T, then impossible.
    for i in range(j):
        if rows[i][0] + rows[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Sum of p1_k <= T
    sum_p1 = sum(r[0] for r in rows)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sum of p2_k <= T
    sum_p2 = sum(r[1] for r in rows)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
