```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t_limit = int(input_data[1])
    
    jobs = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        jobs.append((p1, p2))
        sum_p1 += p1
        sum_p2 += p2
        
        # Check JOB i condition
        if p1 + p2 > t_limit:
            print(f"JOB {i}")
            return
            
    # Check LOAD 1 condition
    if sum_p1 > t_limit:
        print("LOAD 1")
        return
        
    # Check LOAD 2 condition
    if sum_p2 > t_limit:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
