def solve2():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    
    jobs = []
    for i in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        jobs.append((p1, p2))
    
    # Check JOB condition
    for i, (p1, p2) in enumerate(jobs, 1):
        if p1 + p2 > T:
            print(f"JOB {i}")
            return
    
    # Check LOAD 1: sum of all p1 > T
    total_p1 = sum(p1 for p1, p2 in jobs)
    if total_p1 > T:
        print("LOAD 1")
        return
    
    # Check LOAD 2: sum of all p2 > T
    total_p2 = sum(p2 for p1, p2 in jobs)
    if total_p2 > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    solve2()
