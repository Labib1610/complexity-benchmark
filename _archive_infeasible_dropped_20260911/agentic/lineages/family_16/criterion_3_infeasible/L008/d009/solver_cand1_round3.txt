```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    p1_list = []
    p2_list = []
    
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p1_list.append(p1)
        p2_list.append(p2)
        
        # Check JOB i
        # P_i length p1, Q_i length p2, P_i <= Q_i, P_i, Q_i subset [0, T]
        # Let P_i = [a, a+p1], Q_i = [b, b+p2]
        # a >= 0, a+p1 <= T, b >= 0, b+p2 <= T
        # a+p1 <= b
        # So p1 + p2 <= b + p2 <= T
        if p1 + p2 > t:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1
    # Sum of lengths of P_i must be <= T
    if sum(p1_list) > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sum of lengths of Q_i must be <= T
    if sum(p2_list) > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
