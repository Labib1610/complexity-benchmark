```python
import sys

def solve():
    # Read j and T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    linkages = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        linkages.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Constraint 3: For linkage k, if rho1 is [s1, e1] and rho2 is [s2, e2],
    # then for all x in [s1, e1] and y in [s2, e2], y >= x.
    # This is equivalent to s2 >= e1.
    # Since e1 = s1 + p1 and e2 = s2 + p2, and 0 <= s1, e1 <= T, 0 <= s2, e2 <= T:
    # We need s1 >= 0, s1 + p1 <= T, s2 >= s1 + p1, s2 + p2 <= T.
    # This implies s1 + p1 + p2 <= T.
    # So, p1 + p2 <= T must hold for each linkage.
    for i in range(j):
        if linkages[i][0] + linkages[i][1] > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Constraint 4: integral (sum rho1_k)^2 dx <= sum p1_k
    # Let f(x) = sum rho1_k(x). f(x) is the number of intervals covering x.
    # The condition is integral f(x)^2 dx <= sum p1_k.
    # Since f(x) is integer-valued, f(x)^2 >= f(x).
    # integral f(x)^2 dx >= integral f(x) dx = sum p1_k.
    # The only way to satisfy the condition is if f(x)^2 = f(x) almost everywhere,
    # which means f(x) must be 0 or 1 almost everywhere.
    # This means the intervals must be disjoint (except at endpoints).
    # The total length of the union of intervals is sum p1_k.
    # If they are disjoint, the measure of the union is sum p1_k.
    # Since they are all within [0, T], we must have sum p1_k <= T.
    sum_p1 = sum(l[0] for l in linkages)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Same logic as LOAD 1
    sum_p2 = sum(l[1] for l in linkages)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
