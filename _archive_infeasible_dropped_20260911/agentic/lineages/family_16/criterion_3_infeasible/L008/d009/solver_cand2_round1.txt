import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    T = int(input_data[1])
    
    particles = []
    idx = 2
    for i in range(j):
        alpha = int(input_data[idx])
        beta = int(input_data[idx+1])
        particles.append((alpha, beta))
        idx += 2
        
    # Check JOB i
    # Cond 1: y >= (a+b)/2
    # Cond 2: y + a <= x <= y + 2T - a
    # Cond 3: b - y <= x <= 2T - b - y
    # Combined:
    # x >= y + a, x <= y + 2T - a
    # x >= b - y, x <= 2T - b - y
    # For a fixed y, we need:
    # max(y+a, b-y) <= min(y+2T-a, 2T-b-y)
    # This implies:
    # y+a <= y+2T-a => 2a <= 2T => a <= T
    # y+a <= 2T-b-y => 2y <= 2T-a-b => y <= T - (a+b)/2
    # b-y <= y+2T-a => 2y >= b-2T+a => y >= (a+b)/2 - T
    # b-y <= 2T-b-y => 2b <= 2T => b <= T
    # Combined with y >= (a+b)/2:
    # (a+b)/2 <= y <= T - (a+b)/2
    # Contradiction if (a+b)/2 > T - (a+b)/2 => a+b > T
    # Also need a <= T and b <= T.
    for i in range(j):
        a, b = particles[i]
        if a + b > T or a > T or b > T:
            print(f"JOB {i+1}")
            return

    # LOAD 1: Cond 2 & 4
    # Cond 2: y + a <= x <= y + 2T - a  =>  x - y >= a, x - y <= 2T - a
    # Let u = x - y. Then a <= u <= 2T - a.
    # Cond 4: (x_u - x_v)^2 + (y_u - y_v)^2 - 2(x_u - x_v)(y_u - y_v) >= (a_u + a_v)^2
    # ( (x_u - y_u) - (x_v - y_v) )^2 >= (a_u + a_v)^2
    # |u_u - u_v| >= a_u + a_v
    # This is a 1D packing problem: u_i in [a_i, 2T - a_i] such that |u_u - u_v| >= a_u + a_v.
    # Sort by a_i? No, just check if total length is enough.
    # The range for u_i is [a_i, 2T - a_i], length is 2T - 2a_i.
    # This is equivalent to placing points u_i in [0, 2T - 2a_i] with distance a_u + a_v.
    # Actually, the condition |u_u - u_v| >= a_u + a_v is equivalent to:
    # Sort particles such that u_1 < u_2 < ... < u_j.
    # u_{i+1} - u_i >= a_{i+1} + a_i.
    # Total span: u_j - u_1 = sum_{i=1}^{j-1} (u_{i+1} - u_i) >= sum_{i=1}^{j-1} (a_{i+1} + a_i).
    # Also u_j <= 2T - a_j and u_1 >= a_1.
    # So 2T - a_j - a_1 >= sum_{i=1}^{j-1} (a_{i+1} + a_i) = a_1 + 2*sum_{i=2}^{j-1} a_i + a_j.
    # 2T >= 2*sum_{i=1}^j a_i.
    if sum(p[0] for p in particles) > T:
        print("LOAD 1")
        return

    # LOAD 2: Cond 3 & 5
    # Cond 3: b - y <= x <= 2T - b - y => x + y >= b, x + y <= 2T - b
    # Let v = x + y. Then b <= v <= 2T - b.
    # Cond 5: (x_u - x_v)^2 + (y_u - y_v)^2 + 2(x_u - x_v)(y_u - y_v) >= (b_u + b_v)^2
    # ( (x_u + y_u) - (x_v + y_v) )^2 >= (b_u + b_v)^2
    # |v_u - v_v| >= b_u + b_v
    # Same logic as LOAD 1.
    if sum(p[1] for p in particles) > T:
        print("LOAD 2")
        return

solve()
