```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    signals = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        signals.append((p1, p2))
        idx += 2
        
    # Check JOB i:
    # Constraint 3: (a_k + p1_k/2) + p1_k/2 <= (c_k + p2_k/2) - p2_k/2
    # a_k + p1_k <= c_k
    # Since a_k >= 0, we need c_k >= p1_k.
    # Since d_k = c_k + p2_k <= T, we need c_k <= T - p2_k.
    # So we need p1_k <= T - p2_k, or p1_k + p2_k <= T.
    for i in range(j):
        p1, p2 = signals[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1:
    # Sum of p1_k <= T
    sum_p1 = sum(s[0] for s in signals)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2:
    # Sum of p2_k <= T
    sum_p2 = sum(s[1] for s in signals)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
