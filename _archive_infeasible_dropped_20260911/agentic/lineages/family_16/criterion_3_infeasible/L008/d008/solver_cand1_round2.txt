```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    p1 = []
    p2 = []
    idx = 2
    for i in range(j):
        p1_val = int(input_data[idx])
        p2_val = int(input_data[idx+1])
        p1.append(p1_val)
        p2.append(p2_val)
        
        # Check JOB i
        # Constraint 1: p1/2 <= c1 <= T - p1/2
        # Constraint 2: p2/2 <= c2 <= T - p2/2
        # Constraint 3: c2 - c1 >= (p1 + p2) / 2 => c2 >= c1 + (p1 + p2) / 2
        # Min c2 = p1/2 + (p1 + p2) / 2 = p1 + p2/2
        # Max c2 = T - p2/2
        # Contradiction if p1 + p2/2 > T - p2/2 => p1 + p2 > T
        if p1_val + p2_val > T:
            print(f"JOB {i + 1}")
            return
        idx += 2
        
    # Check LOAD 1
    # Constraint 1: p1_k/2 <= c1_k <= T - p1_k/2
    # Constraint 4: |c1_u - c1_v| >= (p1_u + p1_v) / 2
    # Let x_k = c1_k - p1_k/2. Then 0 <= x_k <= T - p1_k.
    # |(x_u + p1_u/2) - (x_v + p1_v/2)| >= (p1_u + p1_v) / 2
    # |x_u - x_v + (p1_u - p1_v)/2| >= (p1_u + p1_v) / 2
    # This is equivalent to packing items of length p1_k in a track of length T.
    # The total length required is sum(p1_k) <= T.
    if sum(p1) > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    if sum(p2) > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
