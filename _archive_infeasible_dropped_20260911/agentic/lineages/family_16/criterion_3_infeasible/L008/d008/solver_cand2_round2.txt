```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    entities = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        entities.append((p1, p2))
        idx += 2
        
    # Check JOB i
    for i in range(j):
        p1, p2 = entities[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    sum_p1 = sum(e[0] for e in entities)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    sum_p2 = sum(e[1] for e in entities)
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
