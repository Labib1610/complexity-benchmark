def read_input():
    lines = []

def parse_input(lines):

        linkages.append((p1, p2))

def check_job_impossible(T, p1, p2):

    # Constraint 2: c2 >= p2/2 and c2 <= T - p2/2
    # Constraint 3: c2 - c1 >= (p1 + p2) / 2
    
    min_c1 = p1 / 2
    max_c1 = T - p1 / 2
    min_c2 = p2 / 2
    max_c2 = T - p2 / 2
    min_diff = (p1 + p2) / 2
    
    # If the ranges are invalid themselves
    if min_c1 > max_c1 or min_c2 > max_c2:
        return True
        
    # From constraint 3: c2 >= c1 + min_diff
    # So we need c1 + min_diff <= max_c2 => c1 <= max_c2 - min_diff
    # And c2 >= min_c1 + min_diff => c2 >= min_c1 + min_diff
    # Also c2 <= max_c2
    
    # Check if there's a feasible range for c1
    upper_bound_c1_from_c2 = max_c2 - min_diff
    lower_bound_c1_from_c2 = min_c2 - min_diff
    
    # Effective bounds for c1
    eff_min_c1 = max(min_c1, lower_bound_c1_from_c2)
    eff_max_c1 = min(max_c1, upper_bound_c1_from_c2)
    
    if eff_min_c1 > eff_max_c1:
        return True
        
    return False
