import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2, i + 1))
        idx += 2
        
    # Check JOB i
    # Constraint 1: alpha1 - alpha0 >= p1
    # Constraint 2: beta1 - beta0 >= p2
    # Constraint 3: alpha0 + p1 <= beta0
    # Markers: 0 <= alpha0 <= alpha1 <= T, 0 <= beta0 <= beta1 <= T
    # alpha0 >= 0, alpha1 <= T => alpha1 - alpha0 <= T
    # beta0 >= 0, beta1 <= T => beta1 - beta0 <= T
    # From 1: alpha1 - alpha0 >= p1
    # From 2: beta1 - beta0 >= p2
    # From 3: beta0 >= alpha0 + p1
    # Combined: beta1 = (beta1 - beta0) + beta0 >= p2 + alpha0 + p1
    # Since alpha0 >= 0, beta1 >= p1 + p2
    # Also beta1 <= T, so p1 + p2 <= T
    # If p1 + p2 > T, then JOB i is impossible.
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > t:
            print(f"JOB {jobs[i][2]}")
            return
            
    # Check LOAD 1
    # Sum of p1_k <= T
    sum_p1 = sum(job[0] for job in jobs)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sum of p2_k <= T
    sum_p2 = sum(job[1] for job in jobs)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
