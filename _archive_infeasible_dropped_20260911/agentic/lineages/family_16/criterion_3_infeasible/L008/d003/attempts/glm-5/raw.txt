```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    T = int(data[1])
    idx = 2
    total_p1 = 0
    total_p2 = 0
    job_violation = -1
    for i in range(j):
        p1 = int(data[idx]); p2 = int(data[idx+1])
        idx += 2
        total_p1 += p1
        total_p2 += p2
        if job_violation == -1 and p1 + p2 > T:
            job_violation = i + 1
    if job_violation != -1:
        print(f"JOB {job_violation}")
    elif total_p1 > T:
        print("LOAD 1")
    else:
        print("LOAD 2")

solve()
```
