```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t_limit = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check JOB i violation
    for i in range(j):
        p1, p2 = jobs[i]
        if p1 + p2 > t_limit:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1 violation
    sum_p1 = 0
    for p1, p2 in jobs:
        sum_p1 += p1
    if sum_p1 > t_limit:
        print("LOAD 1")
        return
        
    # Check LOAD 2 violation
    sum_p2 = 0
    for p1, p2 in jobs:
        sum_p2 += p2
    if sum_p2 > t_limit:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
