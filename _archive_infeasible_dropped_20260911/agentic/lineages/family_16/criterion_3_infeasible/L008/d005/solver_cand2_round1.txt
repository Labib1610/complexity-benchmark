```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Precedence: X_k = [a, a+p1], Y_k = [b, b+p2] with a+p1 <= b
    # The union X_k U Y_k is contained in [0, T].
    # The minimum length required for X_k and Y_k to exist is p1 + p2.
    # If p1 + p2 > T, then no such intervals exist.
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > t:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Gimbal Capacity: sum of p1_k <= T
    sum_p1 = sum(job[0] for job in jobs)
    if sum_p1 > t:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sensor Capacity: sum of p2_k <= T
    sum_p2 = sum(job[1] for job in jobs)
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
