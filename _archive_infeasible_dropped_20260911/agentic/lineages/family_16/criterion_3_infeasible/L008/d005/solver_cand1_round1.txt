```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t_total = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check JOB i
    # Condition: g_i(t1)s_i(t2) = 0 for all t1 >= t2
    # This means the gimbal interval [start_g, end_g] and sensor interval [start_s, end_s]
    # must satisfy end_g <= start_s.
    # Since 0 <= start_g and end_s <= T, we need p1 + p2 <= T.
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > t_total:
            print(f"JOB {i + 1}")
            return
            
    # Check LOAD 1
    # Sum of p1_k <= T
    sum_p1 = sum(job[0] for job in jobs)
    if sum_p1 > t_total:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    # Sum of p2_k <= T
    sum_p2 = sum(job[1] for job in jobs)
    if sum_p2 > t_total:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
