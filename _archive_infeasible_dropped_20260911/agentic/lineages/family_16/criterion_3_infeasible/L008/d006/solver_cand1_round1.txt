```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    jobs = []
    idx = 2
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Check LOAD 1
    sum_p1 = sum(job[0] for job in jobs)
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    sum_p2 = sum(job[1] for job in jobs)
    if sum_p2 > T:
        print("LOAD 2")
        return
        
    # Check JOB i
    # Condition 1: G_i(T) = p1, S_i(T) = p2
    # Condition 2: S_i(t) > 0 implies G_i(t) = p1
    # Condition 3: G_i(t) increases at most at rate 1
    # Condition 4: S_i(t) increases at most at rate 1
    # To satisfy Condition 2, G_i(t) must reach p1 before S_i(t) can start.
    # G_i(t) takes at least p1 time to reach p1.
    # So S_i(t) can only start at t >= p1.
    # The remaining time is T - p1.
    # In this time, S_i(t) can increase by at most T - p1.
    # So we need p2 <= T - p1, or p1 + p2 <= T.
    
    for i in range(j):
        p1, p2 = jobs[i]
        if p1 + p2 > T:
            print(f"JOB {i + 1}")
            return

if __name__ == '__main__':
    solve()
```
