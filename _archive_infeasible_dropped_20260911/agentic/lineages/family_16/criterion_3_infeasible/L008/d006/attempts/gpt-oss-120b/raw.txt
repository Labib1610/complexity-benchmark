```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    T = int(next(it))
    p1 = []
    p2 = []
    for _ in range(j):
        p1.append(int(next(it)))
        p2.append(int(next(it)))
    # Check per-job contradictions
    for idx, (a, b) in enumerate(zip(p1, p2), start=1):
        if a > T or b > T - a:
            print(f"JOB {idx}")
            return
    # Check LOAD 1
    if sum(p1) > T:
        print("LOAD 1")
        return
    # Check LOAD 2
    if sum(p2) > T:
        print("LOAD 2")
        return

if __name__ == "__main__":
    solve()
```
