```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    T = int(input_data[1])
    
    pairs = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    job_idx = -1
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        pairs.append((p1, p2))
        sum_p1 += p1
        sum_p2 += p2
        
        # System 1: exists A_i, B_i s.t. sup(A_i) <= inf(B_i)
        # This is possible if and only if p1_i + p2_i <= T
        if job_idx == -1 and (p1 + p2 > T):
            job_idx = i
            
    if job_idx != -1:
        print(f"JOB {job_idx}")
        return
    
    # System 2: exists {A_1, ..., A_j} s.t. mu(A_a \cap A_b) = 0
    # This is possible if and only if sum(p1_k) <= T
    if sum_p1 > T:
        print("LOAD 1")
        return
    
    # System 3: exists {B_1, ..., B_j} s.t. mu(B_a \cap B_b) = 0
    # This is possible if and only if sum(p2_k) <= T
    if sum_p2 > T:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
