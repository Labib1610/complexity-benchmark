```python
import sys
import threading

def main():
    import sys
    j, T = map(int, sys.stdin.readline().split())
    jobs = []
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(j):
        p1, p2 = map(int, sys.stdin.readline().split())
        jobs.append((p1, p2))
        sum_p1 += p1
        sum_p2 += p2
        # Check JOB i
        if p1 + p2 > T:
            print(f"JOB {i+1}")
            return
    
    # Check LOAD 1
    if sum_p1 > T:
        print("LOAD 1")
        return
        
    # Check LOAD 2
    if sum_p2 > T:
        print("LOAD 2")
        return

threading.Thread(target=main).start()
```
