import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    job_i = -1
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check for Isolated Job Failure
        if p1 + p2 > t:
            if job_i == -1:
                job_i = i
        
        sum_p1 += p1
        sum_p2 += p2
        
    if job_i != -1:
        print(f"JOB {job_i}")
    else:
        # Check for Subsystem Saturation
        if sum_p1 > t:
            print("LOAD 1")
        else:
            print("LOAD 2")

if __name__ == '__main__':
    solve()
