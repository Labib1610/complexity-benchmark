```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        sum_p1 += p1
        sum_p2 += p2
        idx += 2
        
    # Check LOAD 1
    if sum_p1 > t:
        print("LOAD 1")
        return
    
    # Check LOAD 2
    if sum_p2 > t:
        print("LOAD 2")
        return
    
    # Check JOB i
    # Concurrent idle time = T - (p1 + p2)
    # If T - (p1 + p2) < 0, then p1 + p2 > T
    for i in range(j):
        if jobs[i][0] + jobs[i][1] > t:
            print(f"JOB {i + 1}")
            return

if __name__ == '__main__':
    solve()
```
