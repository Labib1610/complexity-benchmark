```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    t = int(input_data[1])
    
    jobs = []
    idx = 2
    sum_p1 = 0
    sum_p2 = 0
    
    for i in range(1, j + 1):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx + 1])
        idx += 2
        
        # Check JOB i condition:
        # If job i is processed alone, gimbal phase finishes at p1.
        # Remaining time is T - p1.
        # Violation if T - p1 < p2, i.e., p1 + p2 > T.
        if p1 + p2 > t:
            print(f"JOB {i}")
            return
        
        sum_p1 += p1
        sum_p2 += p2
        
    # Check LOAD c condition:
    # Total uptime required > T
    if sum_p1 > t:
        print("LOAD 1")
        return
    if sum_p2 > t:
        print("LOAD 2")
        return

if __name__ == '__main__':
    solve()
```
