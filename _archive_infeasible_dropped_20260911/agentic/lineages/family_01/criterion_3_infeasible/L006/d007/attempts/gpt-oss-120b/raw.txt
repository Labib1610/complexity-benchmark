```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    neigh = [0] * (n + 1)          # bitset of neighbors, bit i-1 corresponds to node i
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        bit_u = 1 << (v - 1)
        bit_v = 1 << (u - 1)
        neigh[u] |= bit_u
        neigh[v] |= bit_v

    for a in range(1, n - 1):
        na = neigh[a]
        for b in range(a + 1, n):
            if (na >> (b - 1)) & 1:          # edge a-b exists
                common = na & neigh[b]      # nodes connected to both a and b
                # keep only nodes with index > b
                common >>= b                 # shift away bits <= b
                if common:
                    # lowest set bit gives smallest c
                    low = common & -common
                    idx = low.bit_length() - 1   # zero‑based index after shift
                    c = b + 1 + idx
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
```
