import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        
        # Forward audit trail: smaller -> larger
        if u < v:
            adj[u].add(v)
            edges.append((u, v))
        else:
            adj[v].add(u)
            edges.append((v, u))
            
    # We need to find (i, j, k) such that i < j < k
    # and (i, j) exists, (j, k) exists, and (i, k) exists.
    # Lexicographical order: minimize i, then j, then k.
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    # To find the lexicographically smallest (i, j, k):
    # Iterate through all possible i, then j, then k.
    # Since we need i < j < k, we can iterate i from 1 to n-2.
    
    for i in range(1, n + 1):
        # Get all j such that (i, j) exists
        js = sorted([j for j in adj[i] if j > i])
        for j in js:
            # Now we need k such that (j, k) exists and (i, k) exists
            # k must be > j
            # Intersection of adj[j] and adj[i]
            common = sorted(list(adj[i].intersection(adj[j])))
            for k in common:
                if k > j:
                    print(f"{i} {j} {k}")
                    return

if __name__ == '__main__':
    solve()
