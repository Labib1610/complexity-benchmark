import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # A deadlock kernel is a triangle (3-clique).
    # We need the lexicographically smallest (a, b, c) with a < b < c.
    # Iterate through all edges (u, v) and check for a common neighbor w.
    # To ensure lexicographical order, we can iterate through possible 'a',
    # then 'b', then 'c'.
    
    # Sort edges to help find triangles systematically
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n-2
    for a in range(1, n + 1):
        # Neighbors of a
        neighbors_a = sorted(list(adj[a]))
        if len(neighbors_a) < 2:
            continue
            
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b. Since we want smallest (a, b, c), 
            # we check neighbors of b that are also neighbors of a.
            # To ensure c > b, we only look at neighbors of b that are > b.
            
            # Optimization: intersect neighbors of a and b
            # Only consider c > b
            for c in neighbors_a[i+1:]:
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
