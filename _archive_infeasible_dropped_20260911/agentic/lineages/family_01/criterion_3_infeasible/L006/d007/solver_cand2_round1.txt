```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
    
    # We need to find a, b, c such that a < b < c
    # Condition:
    # 1. b in adj[a] and c in adj[a]
    # 2. c in adj[b]
    # This is equivalent to finding a triangle (a, b, c) where a < b < c
    # and the edges are (a, b), (a, c), (b, c).
    # Since the problem guarantees at least one such configuration,
    # we iterate through all possible a, b, c in lexicographical order.
    
    # To optimize, iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Get neighbors of a that are > a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        
        # For each pair b, c in neighbors_a with b < c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if c is in adj[b]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
