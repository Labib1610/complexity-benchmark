import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # A perfect triad {a, b, c} with a < b < c has exactly 3 edges.
    # This means the subgraph induced by {a, b, c} is a triangle (a-b, b-c, a-c).
    # We need the lexicographically smallest (a, b, c).
    
    # Sort edges to iterate in lexicographical order of (a, b)
    edges.sort()
    
    best_triad = None
    
    for u, v in edges:
        # u < v. We look for a third node w such that {u, v, w} is a triangle.
        # w must be connected to both u and v.
        # To ensure lexicographical order, we check w in increasing order.
        
        # Candidates for w are nodes connected to both u and v.
        common = adj[u].intersection(adj[v])
        
        for w in common:
            # Form the triad {a, b, c}
            triad = sorted([u, v, w])
            
            if best_triad is None or triad < best_triad:
                best_triad = triad
                
    print(f"{best_triad[0]} {best_triad[1]} {best_triad[2]}")

if __name__ == '__main__':
    solve()
