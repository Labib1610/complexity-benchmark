```python
import sys
from itertools import combinations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    edge_set = set()
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        edge_set.add((u, v))
    
    # We need to find a triplet {x1, x2, x3} with exactly 3 edges among them
    # That means the subgraph induced by these 3 nodes has exactly 3 edges
    # i.e., it's a triangle (3 edges connecting all 3 nodes)
    
    # Since we want the lexicographically smallest triplet, we iterate through
    # all combinations of 3 nodes in increasing order
    for combo in combinations(range(1, n+1), 3):
        x1, x2, x3 = combo
        # Check how many edges exist between these three nodes
        count = 0
        if (x1, x2) in edge_set:
            count += 1
        if (x1, x3) in edge_set:
            count += 1
        if (x2, x3) in edge_set:
            count += 1
        
        if count == 3:
            print(x1, x2, x3)
            return

if __name__ == "__main__":
    main()
```
