import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A resonance anomaly is a set {x1, x2, x3} such that there are exactly 3 pulses
    # among them. This means the subgraph induced by {x1, x2, x3} has 3 edges.
    # This is equivalent to saying {x1, x2, x3} forms a triangle in the graph.
    
    # To find the lexicographically smallest {x1, x2, x3}, we iterate x1 from 1 to n,
    # x2 from x1+1 to n, and x3 from x2+1 to n.
    # For a fixed x1 and x2, they form an edge if x2 in adj[x1].
    # If they form an edge, we need to find x3 > x2 such that x3 is connected to both x1 and x2.
    
    # Optimization: iterate x1, then x2 such that (x1, x2) is an edge.
    # Then find common neighbors of x1 and x2 that are > x2.
    
    # Convert sets to sorted lists for faster intersection or just iterate
    # Since n <= 2000, we can use bitsets or simple intersection.
    
    adj_list = [sorted(list(s)) for s in adj]
    
    for x1 in range(1, n + 1):
        for x2 in range(x1 + 1, n + 1):
            if x2 in adj[x1]:
                # Find common neighbors x3 > x2
                # Intersection of adj[x1] and adj[x2]
                # Since we need the smallest x3, we can just iterate
                s1 = adj[x1]
                s2 = adj[x2]
                
                # To be efficient, iterate over the smaller set
                if len(s1) < len(s2):
                    for x3 in s1:
                        if x3 > x2 and x3 in s2:
                            print(f"{x1} {x2} {x3}")
                            return
                else:
                    for x3 in s2:
                        if x3 > x2 and x3 in s1:
                            print(f"{x1} {x2} {x3}")
                            return

solve()
