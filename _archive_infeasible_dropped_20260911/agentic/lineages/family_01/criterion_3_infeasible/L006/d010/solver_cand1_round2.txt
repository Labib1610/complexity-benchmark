import sys

def solve():
    # The resonance condition:
    # Sum_{i,j,k} (V_A[i]*V_A[j]) * (V_B[i]*V_B[k]) * (V_C[j]*V_C[k]) > 0
    # This is equivalent to:
    # Sum_{i,j,k} (V_A[i]*V_B[i]) * (V_A[j]*V_C[j]) * (V_B[k]*V_C[k]) > 0
    # Let E_uv be the set of pulse indices where registries u and v were pulsed together.
    # V_u[p] = 1 iff pulse p involves u.
    # V_A[i]*V_B[i] = 1 iff pulse i involves both A and B.
    # So, the sum is > 0 iff there exist pulses i, j, k such that:
    # pulse i involves {A, B}, pulse j involves {A, C}, pulse k involves {B, C}.
    # This means A and B are connected, A and C are connected, and B and C are connected.
    # This is exactly the condition that A, B, and C form a triangle in the graph
    # where edges are the pulses.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v: u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # We need to find a triangle (x1, x2, x3) with x1 < x2 < x3.
    # To get the lexicographically smallest, we iterate x1 from 1 to n,
    # then x2 from x1+1 to n, then x3 from x2+1 to n.
    
    for x1 in range(1, n + 1):
        # Neighbors of x1
        neighbors_x1 = sorted(list(adj[x1]))
        for i in range(len(neighbors_x1)):
            x2 = neighbors_x1[i]
            if x2 < x1: continue
            
            # We need x3 such that x3 > x2, x3 in adj[x1], and x3 in adj[x2]
            # To ensure lexicographical order, we check x2 candidates in order
            # and for each x2, check x3 candidates in order.
            
            # Intersection of adj[x1] and adj[x2]
            common = sorted(list(adj[x1].intersection(adj[x2])))
            for x3 in common:
                if x3 > x2:
                    # Found the smallest triangle
                    res = sorted([x1, x2, x3])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
