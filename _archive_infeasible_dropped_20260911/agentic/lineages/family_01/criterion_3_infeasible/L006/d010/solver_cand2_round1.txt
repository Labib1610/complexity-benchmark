import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # V_a . V_b = sum_{k=1}^m V_a[k] * V_b[k]
    # V_a[k] is -k if registry a was involved in pulse k, else 0.
    # So V_a . V_b = sum of k^2 for all pulses k where both a and b were involved.
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for k in range(1, m + 1):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append((v, k * k))
        adj[v].append((u, k * k))
        
    # Precompute dot products
    # dot[a][b] = sum of k^2 for pulses involving both a and b
    dot = [{} for _ in range(n + 1)]
    for u in range(1, n + 1):
        for v, val in adj[u]:
            dot[u][v] = dot[u].get(v, 0) + val
            
    def get_dot(a, b):
        return dot[a].get(b, 0)

    # We need x1 < x2 < x3 such that (d12 * d13 * d23) > 0
    # d12 = V_x1 . V_x2, etc.
    # Since k^2 > 0, d_ab >= 0.
    # The product is > 0 if and only if all three dot products are > 0.
    # This means x1, x2, x3 must form a triangle in the graph where edges exist
    # if the dot product is > 0.
    
    # Build adjacency list for the "resonance graph"
    res_adj = [[] for _ in range(n + 1)]
    for u in range(1, n + 1):
        for v in dot[u]:
            if u < v:
                res_adj[u].append(v)
    
    for x1 in range(1, n + 1):
        res_adj[x1].sort()
        for i in range(len(res_adj[x1])):
            x2 = res_adj[x1][i]
            # Check for x3 > x2 such that (x1, x3) and (x2, x3) are edges
            # Intersection of sorted lists
            ptr1 = 0
            ptr2 = 0
            list1 = res_adj[x1]
            list2 = res_adj[x2]
            while ptr1 < len(list1) and ptr2 < len(list2):
                val1 = list1[ptr1]
                val2 = list2[ptr2]
                if val1 == val2:
                    if val1 > x2:
                        print(f"{x1} {x2} {val1}")
                        return
                    ptr1 += 1
                    ptr2 += 1
                elif val1 < val2:
                    ptr1 += 1
                else:
                    ptr2 += 1

solve()
