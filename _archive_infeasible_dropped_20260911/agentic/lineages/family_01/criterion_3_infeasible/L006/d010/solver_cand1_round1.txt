import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # V_a . V_b = sum_{k=1}^m (V_a[k] * V_b[k])
    # V_a[k] is W_k if pulse k involves registry a, else 0.
    # So V_a . V_b = sum_{k: pulse k involves both a and b} (W_k^2)
    # W_k = 1 for k=1, 2, 3. W_k = -k for k > 3.
    # W_k^2 = 1 for k=1, 2, 3. W_k^2 = k^2 for k > 3.
    
    adj = [{} for _ in range(n + 1)]
    
    idx = 2
    for k in range(1, m + 1):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        
        if u > v:
            u, v = v, u
        
        w_sq = 1 if k <= 3 else k * k
        
        adj[u][v] = adj[u].get(v, 0) + w_sq
        
    def get_dot(u, v):
        if u > v:
            u, v = v, u
        return adj[u].get(v, 0)

    # We need x1 < x2 < x3 such that (d12 * d13 * d23) > 0
    # d12, d13, d23 are dot products. Since W_k^2 > 0, dot products are >= 0.
    # If any dot product is 0, the product is 0.
    # So we need d12 > 0, d13 > 0, d23 > 0.
    # This means x1-x2, x1-x3, and x2-x3 must all have been involved in at least one common pulse.
    
    for x1 in range(1, n + 1):
        # Find all neighbors of x1
        neighbors_x1 = sorted(adj[x1].keys())
        # Also check nodes where x1 is the larger index
        for i in range(1, x1):
            if x1 in adj[i]:
                neighbors_x1.append(i)
        neighbors_x1.sort()
        
        for i in range(len(neighbors_x1)):
            x2 = neighbors_x1[i]
            if x2 <= x1: continue
            
            # We need x3 > x2 such that x3 is a neighbor of x1 and x2
            # Check neighbors of x2
            neighbors_x2 = set(adj[x2].keys())
            for j in range(1, x2):
                if x2 in adj[j]:
                    neighbors_x2.add(j)
            
            for j in range(i + 1, len(neighbors_x1)):
                x3 = neighbors_x1[j]
                if x3 in neighbors_x2:
                    # Found triplet (x1, x2, x3) with all dot products > 0
                    print(f"{x1} {x2} {x3}")
                    return

solve()
