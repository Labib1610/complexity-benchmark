import sys
from collections import defaultdict

def find_triangle():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(set)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort vertices by ID for lexicographic ordering
    vertices = sorted(adj.keys())
    
    # For each edge, check if it's part of a triangle
    for u, v in edges:
        # Find common neighbors
        if u > v:
            u, v = v, u
        common = adj[u] & adj[v]
        if common:
            # Get the smallest common neighbor
            z = min(common)
            # Output in sorted order
            result = sorted([u, v, z])
            print(result[0], result[1], result[2])
            return

find_triangle()
