import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to ensure we check candidates in lexicographical order
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) as the first two nodes
    # Then check if there exists a node c such that (a, c) and (b, c) are edges
    # To get the smallest (a, b, c), we iterate a from 1 to n, b from a+1 to n
    # and check common neighbors c > b.
    
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that c > b and c is in both adj[a] and adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # To optimize, iterate through the smaller set
            if len(neighbors_a) - i - 1 < len(adj[b]):
                for j in range(i + 1, len(neighbors_a)):
                    c = neighbors_a[j]
                    if c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                # Iterate through neighbors of b that are > b
                neighbors_b = sorted([v for v in adj[b] if v > b])
                for c in neighbors_b:
                    if c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

solve()
