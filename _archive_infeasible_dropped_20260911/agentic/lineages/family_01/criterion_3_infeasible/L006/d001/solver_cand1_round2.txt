```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triangle
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check for a common neighbor c.
    # To ensure we find the smallest, we can iterate through all possible a, b, c
    # but that's O(n^3). Instead, iterate through edges (a, b) and check
    # intersection of adj[a] and adj[b].
    
    best = None
    
    for a in range(1, n + 1):
        # Optimization: if we already found a triangle starting with a smaller 'a',
        # we don't need to check larger 'a's.
        if best and a > best[0]:
            break
            
        for b in sorted(list(adj[a])):
            if a >= b:
                continue
            
            # Check if there exists c > b such that c is in adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
