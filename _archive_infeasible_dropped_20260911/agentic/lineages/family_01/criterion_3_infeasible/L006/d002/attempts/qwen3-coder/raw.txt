```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Create a list of neighbors for each node
    neighbors = {i: adj[i] for i in range(1, n+1)}
    
    # We want to find a triangle (3-clique) in the graph
    # Iterate through all edges (u,v)
    # For each edge, look for a common neighbor w
    # That would make a triangle u-v-w
    
    # To get the lexicographically smallest, we'll process nodes in order
    best = None
    
    # Convert to sorted lists for consistent ordering
    for u in range(1, n+1):
        if u not in neighbors:
            continue
        for v in neighbors[u]:
            if v <= u:
                continue  # Ensure u < v
            # Look for common neighbors w where w > v
            for w in neighbors[u]:
                if w <= v:
                    continue
                if w in neighbors[v]:
                    # Found triangle u-v-w
                    candidate = (u, v, w)
                    if best is None or candidate < best:
                        best = candidate
    
    if best:
        print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
```
