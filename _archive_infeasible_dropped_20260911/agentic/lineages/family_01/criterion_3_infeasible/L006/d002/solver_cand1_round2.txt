import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triangle
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check for a common neighbor c.
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b from a+1 to n, then check if c exists.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We have edge (a, b). Now look for c > b such that c is in adj[a] and adj[b]
            # Since adj lists are sorted, we can use two-pointer intersection
            nb_a = neighbors_a
            nb_b = adj[b]
            
            ptr_a = i + 1
            ptr_b = 0
            
            while ptr_a < len(nb_a) and ptr_b < len(nb_b):
                val_a = nb_a[ptr_a]
                val_b = nb_b[ptr_b]
                
                if val_a == val_b:
                    # Found a triangle (a, b, val_a)
                    # Since we iterate a, then b, then c, the first one found is the smallest
                    print(f"{a} {b} {val_a}")
                    return
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
