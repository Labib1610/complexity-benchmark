import sys
from itertools import combinations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list for the graph
    adj = [set() for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        adj[u].add(v)
        adj[v].add(u)
        index += 2
    
    # Iterate through all combinations of 3 nodes in increasing order
    for x1, x2, x3 in combinations(range(1, n + 1), 3):
        # Check if the required edges exist for resonance anomaly
        if x2 in adj[x1] and x3 in adj[x1] and x3 in adj[x2]:
            print(x1, x2, x3)
            return

if __name__ == "__main__":
    main()
