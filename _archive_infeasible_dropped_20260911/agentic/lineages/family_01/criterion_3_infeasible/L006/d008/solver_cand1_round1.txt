import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # The condition R_{x1}(x2) * R_{x1}(x3) * R_{x2}(x3) = 1
    # means that there are edges (x1, x2), (x1, x3), and (x2, x3)
    # in the graph formed by the pulses.
    # This is equivalent to finding a triangle (x1, x2, x3) in the graph.
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        
    # We need the lexicographically smallest (x1, x2, x3) with x1 < x2 < x3.
    # Iterate x1 from 1 to n-2
    for x1 in range(1, n - 1):
        # Get neighbors of x1 that are > x1
        neighbors_x1 = sorted([v for v in adj[x1] if v > x1])
        
        # Iterate x2 in neighbors_x1
        for i in range(len(neighbors_x1)):
            x2 = neighbors_x1[i]
            
            # We need x3 such that x3 > x2 and x3 is in adj[x1] and adj[x2]
            # Check intersection of adj[x1] and adj[x2]
            # Since we need the smallest x3, we can iterate through sorted neighbors
            # or use set intersection.
            
            # Optimization: check smaller set
            if len(adj[x1]) < len(adj[x2]):
                candidates = adj[x1]
                other = adj[x2]
            else:
                candidates = adj[x2]
                other = adj[x1]
            
            for x3 in sorted(candidates):
                if x3 > x2 and x3 in other:
                    print(f"{x1} {x2} {x3}")
                    return

solve()
