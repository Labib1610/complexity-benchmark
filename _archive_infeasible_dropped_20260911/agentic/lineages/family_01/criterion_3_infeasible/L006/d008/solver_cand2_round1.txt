```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use bitsets (integers in Python) to represent adjacency
    # adj[i] is a bitmask where the j-th bit is 1 if there is an edge between i and j
    adj = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find a, b, c such that a < b < c
    # 1. a is connected to b and c
    # 2. b is connected to c
    # This is equivalent to finding a triangle (a, b, c)
    # The condition is:
    # (adj[a] & (1 << b)) and (adj[a] & (1 << c)) and (adj[b] & (1 << c))
    
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n-2,
    # b from a+1 to n-1, and c from b+1 to n.
    
    for a in range(1, n - 1):
        adj_a = adj[a]
        for b in range(a + 1, n):
            if (adj_a >> b) & 1:
                # b is connected to a, now check for c
                # c must be connected to both a and b
                common = adj_a & adj[b]
                # We only care about c > b
                # Mask out bits <= b
                common >>= (b + 1)
                if common > 0:
                    # Find the smallest c
                    # (common & -common) gives the lowest set bit
                    c = (common & -common).bit_length() + b
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
