```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for k in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Iterate to find the lexicographically smallest triplet
    for x1 in range(1, n - 1):
        # Neighbors of x1 strictly greater than x1
        # We need to find x2, x3 in this set such that they are connected
        neighbors = []
        for v in adj[x1]:
            if v > x1:
                neighbors.append(v)
        neighbors.sort()
        
        # Check pairs in neighbors
        # Since neighbors is sorted, iterating gives increasing x2
        # For a fixed x2, iterating subsequent neighbors gives increasing x3
        k_limit = len(neighbors)
        for i in range(k_limit):
            x2 = neighbors[i]
            for j in range(i + 1, k_limit):
                x3 = neighbors[j]
                if x3 in adj[x2]:
                    print(x1, x2, x3)
                    return

if __name__ == '__main__':
    solve()
```
