import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n  # bitset of neighbors, 0‑based
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for i in range(n):
        ai = adj[i]
        for j in range(i + 1, n):
            if (ai >> j) & 1:          # edge i‑j exists
                common = ai & adj[j]   # neighbors common to i and j
                c = common >> (j + 1)  # keep only k > j
                if c:
                    low = (c & -c).bit_length() - 1
                    k = j + 1 + low
                    print(i + 1, j + 1, k + 1)
                    return

if __name__ == "__main__":
    solve()
