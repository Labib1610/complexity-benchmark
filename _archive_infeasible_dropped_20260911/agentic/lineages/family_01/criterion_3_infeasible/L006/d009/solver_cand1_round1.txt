import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[i] will store the set of registries connected to i
    # The dot product V_a . V_b is the number of pulses where both a and b were involved.
    # This is equivalent to the number of times the pair (a, b) appeared in the input.
    # Since each pulse is a pair (u, v), V_a . V_b is the count of edges between a and b.
    # The condition (V_x1 . V_x2) * (V_x1 . V_x3) * (V_x2 . V_x3) > 0
    # means that all three pairs (x1, x2), (x1, x3), and (x2, x3) must have at least one pulse.
    # This is equivalent to saying that x1, x2, x3 form a triangle in the graph
    # where edges are defined by the pulses.
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u != v:
            adj[u].add(v)
            adj[v].add(u)
            
    # We need the lexicographically smallest (x1, x2, x3) with x1 < x2 < x3
    # such that x1, x2, x3 form a triangle.
    for x1 in range(1, n + 1):
        # Sort neighbors to ensure we check x2, x3 in order
        neighbors_x1 = sorted(list(adj[x1]))
        for i in range(len(neighbors_x1)):
            x2 = neighbors_x1[i]
            if x2 <= x1:
                continue
            
            # Check for x3 > x2 such that x3 is in adj[x1] and adj[x2]
            neighbors_x2 = adj[x2]
            for j in range(i + 1, len(neighbors_x1)):
                x3 = neighbors_x1[j]
                if x3 in neighbors_x2:
                    print(f"{x1} {x2} {x3}")
                    return

if __name__ == '__main__':
    solve()
