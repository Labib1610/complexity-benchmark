import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # The matrix M is:
    # 0 Z12 Z13
    # Z12 0 Z23
    # Z13 Z23 0
    # Determinant = 0*... - Z12*(0 - Z13*Z23) + Z13*(Z12*Z23 - 0)
    # Det = Z12*Z13*Z23 + Z13*Z12*Z23 = 2 * Z12 * Z13 * Z23
    # For Det to be 2, Z12, Z13, and Z23 must all be 1.
    # This means x1, x2, x3 must form a triangle in the graph.
    
    # We need the lexicographically smallest (x1, x2, x3) with x1 < x2 < x3.
    # Iterate x1 from 1 to n-2
    for x1 in range(1, n - 1):
        # Iterate x2 from x1+1 to n-1
        # To optimize, only check x2 that are connected to x1
        neighbors_x1 = sorted([v for v in adj[x1] if v > x1])
        for x2 in neighbors_x1:
            # We need x3 > x2 such that x3 is connected to both x1 and x2
            # Intersection of neighbors of x1 and x2
            common = adj[x1].intersection(adj[x2])
            # Filter for x3 > x2
            valid_x3 = [x3 for x3 in common if x3 > x2]
            if valid_x3:
                print(f"{x1} {x2} {min(valid_x3)}")
                return

solve()
