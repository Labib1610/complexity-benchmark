import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # P_i(j) is 1 if the number of pulses between i and j is even, -1 if odd.
    # We only care about the parity of the count of pulses between i and j.
    # Using a dictionary to store edges that have been toggled an odd number of times.
    # Since n <= 2000, we can use a 2D array or a list of sets.
    # Given the constraints, a 2D bitset or adjacency list is efficient.
    
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        
        if v in adj[u]:
            adj[u].remove(v)
            adj[v].remove(u)
        else:
            adj[u].add(v)
            adj[v].add(u)
            
    # P_i(j) = -1 if j in adj[i], else 1.
    # Condition: P_x1(x2) + P_x1(x3) + P_x2(x3) = -3
    # This means P_x1(x2) = -1, P_x1(x3) = -1, P_x2(x3) = -1.
    # So we need to find x1 < x2 < x3 such that:
    # x2 is in adj[x1], x3 is in adj[x1], and x3 is in adj[x2].
    
    # Iterate x1 from 1 to n-2
    for x1 in range(1, n - 1):
        neighbors_x1 = sorted(list(adj[x1]))
        # We need x2, x3 in neighbors_x1 such that x2 < x3 and x3 in adj[x2]
        for i in range(len(neighbors_x1)):
            x2 = neighbors_x1[i]
            if x2 <= x1:
                continue
            
            # Now check for x3 > x2 such that x3 is in neighbors_x1 AND x3 is in adj[x2]
            # Intersection of neighbors_x1[i+1:] and adj[x2]
            for j in range(i + 1, len(neighbors_x1)):
                x3 = neighbors_x1[j]
                if x3 in adj[x2]:
                    print(f"{x1} {x2} {x3}")
                    return

solve()
