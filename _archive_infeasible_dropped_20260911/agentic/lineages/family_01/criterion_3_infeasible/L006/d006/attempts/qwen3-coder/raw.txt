```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = set()
    graph = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.add((u, v))
        graph[u].add(v)
    
    # We need to find a, b, c such that a < b < c
    # and (a,b), (b,c), (a,c) all exist
    
    # Convert to sorted list for consistent ordering
    nodes = sorted(graph.keys())
    
    for a in nodes:
        # Get all neighbors of a that are > a
        neighbors_a = sorted(list(graph[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            # Now check if there's a neighbor of b that is > b and also connected to a
            neighbors_b = sorted(list(graph[b]))
            for c in neighbors_b:
                if c <= b:
                    continue
                # Check if (a, c) exists
                if (a, c) in edges:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
