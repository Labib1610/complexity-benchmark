import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
    
    # For each node, sort its neighbors for ordered iteration
    adj_lists = [sorted(list(s)) for s in adj]
    
    # visited[v] stores the smallest u that has v as a neighbor
    visited = [0] * (n + 1)
    
    for u in range(1, n + 1):
        neighbors = adj_lists[u]
        # Check for intersection between neighbors of u
        # Since neighbors are sorted, we can iterate and check visited marks
        for v in neighbors:
            if visited[v] != 0:
                # Found a triangle: visited[v], u, v
                # visited[v] < u < v is guaranteed by loop structure
                print(visited[v], u, v)
                return
        # Mark all neighbors of u with u
        for v in neighbors:
            visited[v] = u
    
    # Should not reach here given the problem guarantees

if __name__ == '__main__':
    solve()
