```python
import sys

def solve():
    # The problem asks for a set of three suppliers {a, b, c} such that
    # it is impossible to assign them to two tiers without a conflict.
    # This is equivalent to finding a triangle (a clique of size 3) in the graph.
    # We need the lexicographically smallest (a, b, c) where a < b < c.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c), we iterate through
    # possible values of a, then b, then c.
    # Since n <= 2000, we can iterate through all edges (a, b) and check
    # if there exists a c such that (a, c) and (b, c) are edges.
    
    # Sort edges to help find the smallest (a, b, c)
    edges.sort()
    
    best = None
    
    for a in range(1, n + 1):
        # Optimization: if we already found a solution with a smaller 'a',
        # we don't need to check larger 'a's.
        if best and a >= best[0]:
            break
            
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if best and (a > best[0] or (a == best[0] and b >= best[1])):
                continue
            
            # Find common neighbors of a and b that are > b
            # Intersection of adj[a] and adj[b]
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
                    # Since we want the smallest c for this (a, b),
                    # and we sorted common, the first one is the best for this (a, b)
                    break
                    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
