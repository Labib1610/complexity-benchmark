import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        
    # We need to find a, b, c such that a < b < c
    # with edges (a, b), (b, c), and (a, c).
    # To find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, b from a+1 to n, and c from b+1 to n.
    
    for a in range(1, n + 1):
        # Sort neighbors to check b in increasing order
        neighbors_a = sorted(list(adj[a]))
        for b in neighbors_a:
            # We need c such that (b, c) exists and (a, c) exists.
            # So c must be in adj[b] AND in adj[a].
            # To find the smallest c, we intersect adj[a] and adj[b].
            
            # Optimization: only check c > b
            # Intersection of sorted lists or sets
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
