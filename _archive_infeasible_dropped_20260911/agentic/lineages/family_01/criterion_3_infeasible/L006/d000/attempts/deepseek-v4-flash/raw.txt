Here's a solution using an adjacency matrix in a bit-level approach:  

```python
import sys

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [0] * n
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        edges.append((u, v))

    for a in range(n):
        for b in range(a + 1, n):
            if adj[a] >> b & 1:
                common = adj[a] & adj[b]
                # find smallest c > b that is connected to both a and b
                mask = common & ~((1 << (b + 1)) - 1)
                if mask:
                    c = (mask & -mask).bit_length() - 1
                    print(a + 1, b + 1, c + 1)
                    return

if __name__ == "__main__":
    main()
```
