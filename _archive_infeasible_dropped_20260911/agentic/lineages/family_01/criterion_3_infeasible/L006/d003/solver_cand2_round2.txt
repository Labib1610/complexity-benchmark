```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to ensure we check combinations in lexicographical order
    # We want smallest a, then smallest b, then smallest c.
    # Iterate through all edges (u, v) as potential (a, b)
    # Then check if there exists a node c such that (a, c) and (b, c) exist.
    
    edges.sort()
    
    best = None
    
    for u, v in edges:
        # u < v is guaranteed by the input processing
        # We look for c such that (u, c) and (v, c) exist.
        # To find the lexicographically smallest (u, v, c),
        # we iterate through all possible c.
        
        # Intersection of neighbors of u and v
        common = adj[u].intersection(adj[v])
        if common:
            c = min(common)
            # The triplet is (u, v, c). Sort them to get (a, b, c)
            triplet = sorted([u, v, c])
            if best is None or triplet < best:
                best = triplet
                
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
