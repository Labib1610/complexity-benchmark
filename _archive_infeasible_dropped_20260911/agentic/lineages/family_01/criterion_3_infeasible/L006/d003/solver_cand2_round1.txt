import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need to find a triangle (u, v, w) such that u < v < w.
    # To find the lexicographically smallest, we iterate through possible u, then v, then w.
    # Sort edges to help with iteration
    edges.sort()
    
    # To optimize, iterate through each edge (u, v) and check for a common neighbor w.
    # To ensure lexicographical order, we check u from 1 to n.
    for u in range(1, n + 1):
        neighbors = sorted(list(adj[u]))
        # For each pair of neighbors v, w of u, check if (v, w) is an edge.
        # Since we want the lexicographically smallest (u, v, w),
        # we iterate u, then v, then w.
        for i in range(len(neighbors)):
            v = neighbors[i]
            if v <= u:
                continue
            for j in range(i + 1, len(neighbors)):
                w = neighbors[j]
                # Check if (v, w) is an edge
                if w in adj[v]:
                    # Found triangle (u, v, w)
                    res = sorted([u, v, w])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
