import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
    
    # We need to find a, b, c such that (a,b), (b,c), (a,c) are all edges.
    # This is equivalent to finding a triangle in the graph.
    # To find the lexicographically smallest (a, b, c), we iterate through
    # possible a, then b, then c.
    
    # Sort edges to help with finding triangles efficiently
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n-2
    for a in range(1, n + 1):
        # Get neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        if len(neighbors_a) < 2:
            continue
            
        # For each pair of neighbors b, c of a, check if (b, c) is an edge
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    # Found a triangle (a, b, c)
                    # Since we iterate a, then b, then c in increasing order,
                    # the first one we find is the lexicographically smallest.
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
