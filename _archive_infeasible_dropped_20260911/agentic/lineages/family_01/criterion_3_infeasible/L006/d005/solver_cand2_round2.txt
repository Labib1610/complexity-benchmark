import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
    
    # We need to find a triangle (a, b, c) such that a < b < c.
    # To find the lexicographically smallest (a, b, c), we iterate
    # through all possible a, then b, then c.
    
    # Sort edges to help find triangles efficiently
    # A triangle exists if for an edge (u, v), there exists w such that
    # w is in adj[u] and w is in adj[v].
    
    best_trio = None
    
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n
    for a in range(1, n + 1):
        # Get neighbors of a
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need c such that c is a neighbor of both a and b, and c > b
            # To ensure lexicographical order, we check neighbors of b
            # that are also neighbors of a.
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    # Found a triangle (a, b, c) with a < b < c
                    trio = tuple(sorted((a, b, c)))
                    if best_trio is None or trio < best_trio:
                        best_trio = trio
        
        # Optimization: if we found a trio starting with 'a', 
        # we don't necessarily need to check higher 'a's if we are sure
        # we found the smallest. Since we iterate a from 1, the first
        # 'a' that yields a result will contain the smallest 'a'.
        if best_trio is not None:
            break
            
    if best_trio:
        print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

solve()
