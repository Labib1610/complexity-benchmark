```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need the lexicographically smallest (a, b, c) with a < b < c
    # Iterate through all possible a, then b, then c
    # To optimize, only iterate b in adj[a] and c in adj[a]
    
    # Sort adjacency lists to ensure we find the lexicographically smallest
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need c such that b < c and c is in adj[a] and c is in adj[b]
            # Since neighbors_a is sorted, we can iterate through it
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if c is connected to b
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
```
