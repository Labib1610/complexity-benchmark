```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need to find a triangle (a, b, c) with a < b < c.
    # To find the lexicographically smallest, we iterate through possible a, then b, then c.
    # Since n <= 2000, we can iterate through all edges (u, v) and check for a common neighbor.
    
    best = None
    
    # Sort edges to help find the lexicographically smallest triangle
    edges.sort()
    
    for u, v in edges:
        # Triangle is (u, v, w) where w is a common neighbor
        # We need to order them as a < b < c
        # Possible cases for w:
        # 1. w < u < v: (w, u, v)
        # 2. u < w < v: (u, w, v)
        # 3. u < v < w: (u, v, w)
        
        # Intersection of adj[u] and adj[v]
        common = adj[u].intersection(adj[v])
        for w in common:
            t = sorted([u, v, w])
            if best is None or t < best:
                best = t
                
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
