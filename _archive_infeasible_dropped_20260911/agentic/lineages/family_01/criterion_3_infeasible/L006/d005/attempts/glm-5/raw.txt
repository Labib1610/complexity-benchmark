```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    
    for a in range(1, n + 1):
        neighbors = sorted(adj[a])
        for i in range(len(neighbors)):
            b = neighbors[i]
            if b < a:
                continue
            for j in range(i + 1, len(neighbors)):
                c = neighbors[j]
                if c < b:
                    continue
                if c in adj[b]:
                    print(a, b, c)
                    return

solve()
```
