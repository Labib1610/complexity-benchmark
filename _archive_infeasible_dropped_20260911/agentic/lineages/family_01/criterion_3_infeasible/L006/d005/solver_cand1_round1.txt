import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c
    # We iterate a from 1 to n-2
    # Then b from a+1 to n-1
    # Then check if there exists c > b such that c is in adj[a] and adj[b]
    
    # Sort adjacency lists to potentially speed up intersection
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # We need c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we need the smallest c, we can iterate through sorted neighbors
            # But we only care about c > b
            
            # Optimization: use set intersection for speed
            # But we need the smallest c > b
            set_a = adj[a]
            set_b = adj[b]
            
            # Find common neighbors
            common = set_a.intersection(set_b)
            for c in common:
                if c > b:
                    # Found a triplet (a, b, c)
                    # Since we iterate a then b, and we want the smallest c,
                    # we should collect all valid c for this (a, b) and pick min
                    pass
            
            # Actually, to be lexicographically smallest, we just need the first
            # valid (a, b, c) found in the nested loops.
            # To ensure we get the smallest c for a fixed (a, b), we sort common
            common_sorted = sorted([c for c in common if c > b])
            if common_sorted:
                print(f"{a} {b} {common_sorted[0]}")
                return

solve()
