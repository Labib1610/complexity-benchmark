```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # We need to find a, b, c such that:
    # (a, b) is an edge, (a, c) is an edge, (b, c) is an edge.
    # This is a triangle (a, b, c).
    # The problem states: "a is restricted from sharing a tier with b, 
    # and there is a third supplier c who cannot be placed in a tier with a 
    # nor in a tier with b."
    # This means (a, b), (a, c), and (b, c) must all be edges.
    # We need the lexicographically smallest (a, b, c) with a < b < c.
    
    # Sort edges to help find the smallest lexicographical triplet
    edges.sort()
    
    best = None
    
    # Iterate through all edges (a, b)
    for i in range(m):
        u, v = edges[i]
        a, b = (u, v) if u < v else (v, u)
        
        # Optimization: if we already have a 'best' and a >= best[0], 
        # we might be able to skip, but since we need the smallest, 
        # we check all triangles.
        
        # Find common neighbors c of a and b
        # To ensure a < b < c, we only look for c > b
        common = adj[a].intersection(adj[b])
        for c in common:
            if c > b:
                # Found a triangle (a, b, c)
                triplet = (a, b, c)
                if best is None or triplet < best:
                    best = triplet
                    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
