import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    bits = [0] * (n + 1)
    edges = []
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        if u > v:
            u, v = v, u
        bits[u] |= 1 << v
        bits[v] |= 1 << u
        edges.append((u, v))

    # sort edges to ensure lexicographic order when scanning a,b
    edges.sort()
    full_mask = (1 << (n + 1)) - 1
    for a, b in edges:
        # common neighbors of a and b
        common = bits[a] & bits[b]
        # keep only vertices > b
        mask = full_mask ^ ((1 << (b + 1)) - 1)   # bits b+1 .. n
        common &= mask
        if common:
            # get smallest c
            lowbit = common & -common
            c = lowbit.bit_length() - 1
            print(a, b, c)
            return

if __name__ == "__main__":
    solve()
