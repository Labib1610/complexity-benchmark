```python
import sys

def solve():
    # Read N and M
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    
    # We need to find a triangle (a, b, c) such that edges (a,b), (b,c), (a,c) exist.
    # Adjacency list representation
    adj = [set() for _ in range(N + 1)]
    idx = 2
    for _ in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to N-2
    # Iterate b from a+1 to N-1
    # Check if b is in adj[a]
    # If so, check if there exists c > b such that c is in adj[a] and adj[b]
    
    # Sort adjacency lists to make finding c easier
    sorted_adj = [sorted(list(adj[i])) for i in range(N + 1)]
    
    for a in range(1, N + 1):
        # Only consider neighbors b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # We need a c > b such that c is in adj[a] and adj[b]
            # Since sorted_adj[a] and sorted_adj[b] are sorted, 
            # we can find the intersection efficiently.
            
            # Optimization: iterate through the smaller adjacency list
            if len(sorted_adj[a]) < len(sorted_adj[b]):
                for c in sorted_adj[a]:
                    if c > b and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in sorted_adj[b]:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
```
