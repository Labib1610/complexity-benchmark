```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need to find a triangle (a, b, c) such that a < b < c.
    # To find the lexicographically smallest (a, b, c), we iterate
    # through possible values of a, then b, then c.
    
    # Sort edges to help find triangles efficiently
    # For each edge (u, v) with u < v, check all w such that (u, w) and (v, w) exist.
    # If w exists, we have a triangle (min(u,v,w), mid(u,v,w), max(u,v,w)).
    
    found_triangles = []
    
    # To optimize, iterate through nodes as the smallest vertex 'a'
    for a in range(1, n + 1):
        neighbors = sorted(list(adj[a]))
        len_n = len(neighbors)
        for i in range(len_n):
            b = neighbors[i]
            # We only care about triangles where a < b < c
            # So b must be > a
            if b < a:
                continue
            
            # Check common neighbors c > b
            for j in range(i + 1, len_n):
                c = neighbors[j]
                if c in adj[b]:
                    # Found triangle (a, b, c)
                    # Since we iterate a from 1..n and b, c from neighbors,
                    # the first one we find might not be the lexicographically smallest.
                    # However, if we collect all and sort, it's fine.
                    # Actually, we can just store the smallest one found.
                    res = sorted([a, b, c])
                    found_triangles.append(tuple(res))
    
    if found_triangles:
        ans = min(found_triangles)
        print(f"{ans[0]} {ans[1]} {ans[2]}")

solve()
```
