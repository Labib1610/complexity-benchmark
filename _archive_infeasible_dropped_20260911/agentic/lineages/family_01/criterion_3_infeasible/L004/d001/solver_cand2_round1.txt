import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[u] stores a set of workers v such that u -> v (where u < v)
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
    
    # We need to find a, b, c such that a < b < c
    # and (a->b), (a->c), (b->c) exist.
    # To find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, then b from a+1 to n, then c from b+1 to n.
    
    for a in range(1, n + 1):
        # Optimization: only consider a if it has at least 2 outgoing edges
        if len(adj[a]) < 2:
            continue
        
        # Sort neighbors of a to check b in increasing order
        sorted_neighbors_a = sorted(list(adj[a]))
        
        for i in range(len(sorted_neighbors_a)):
            b = sorted_neighbors_a[i]
            
            # Check if there exists c in adj[a] such that c > b and c is in adj[b]
            # We need to find the smallest c > b such that c in adj[a] and c in adj[b]
            for j in range(i + 1, len(sorted_neighbors_a)):
                c = sorted_neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
