import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        idx += 2
        
    # We need to find a, b, c such that a < b < c
    # and a->b, a->c, b->c exist.
    # To find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, then b from a+1 to n, then c from b+1 to n.
    
    # Pre-process adjacency lists into sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    # Sort adjacency lists to potentially speed up or help logic
    for i in range(1, n + 1):
        adj[i].sort()
        
    for a in range(1, n + 1):
        # Candidates for b are nodes v such that a -> v
        b_candidates = adj[a]
        for b in b_candidates:
            if b <= a:
                continue
            
            # Now we need a c such that b < c, a -> c, and b -> c
            # c must be in adj[a] AND in adj[b]
            # To find the smallest c, we can intersect the sorted lists
            # or just iterate through the smaller list.
            
            # Since we want the smallest c, we iterate through sorted adj[a]
            # and check if it's in adj[b] and > b.
            for c in adj[a]:
                if c > b:
                    if c in adj_sets[b]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
