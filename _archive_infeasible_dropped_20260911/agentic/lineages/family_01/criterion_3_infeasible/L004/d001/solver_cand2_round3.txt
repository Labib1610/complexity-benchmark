```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[a] will store a sorted list of b such that a -> b
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        
        if u < v:
            adj[u].append(v)
        else:
            adj[v].append(u)
            
    for i in range(1, n + 1):
        adj[i].sort()
        
    # We need a < b < c such that a->b, a->c, b->c
    # Iterate through a
    for a in range(1, n + 1):
        if not adj[a]:
            continue
        
        # For a fixed a, we need to find b, c in adj[a] such that b < c and b -> c
        # To get lexicographically smallest, we iterate b then c
        targets = adj[a]
        # Use a set for O(1) lookup of b -> c
        # Since we only need to check if b -> c exists, we can use a set of sets
        # or just a set of tuples.
        
        # To optimize, let's build a set of edges for quick lookup
        # Actually, we can just check if c is in adj[b]
        
        # Pre-convert adj lists to sets for O(1) lookup
        adj_sets = [set(adj[i]) for i in range(n + 1)]
        
        for i in range(len(targets)):
            b = targets[i]
            for j in range(i + 1, len(targets)):
                c = targets[j]
                # Check if b -> c exists
                if c in adj_sets[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
```
