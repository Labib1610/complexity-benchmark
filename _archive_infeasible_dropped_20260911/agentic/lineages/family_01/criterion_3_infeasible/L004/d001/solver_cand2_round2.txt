import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].append(v)
    
    for i in range(1, n + 1):
        adj[i].sort()
        
    # We need a < b < c such that a->b, a->c, b->c
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        # Neighbors of a are b's and c's
        neighbors_a = adj[a]
        if len(neighbors_a) < 2:
            continue
            
        # To find lexicographically smallest (a, b, c),
        # we iterate b in increasing order from neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Now we need a c such that b->c and c is in neighbors_a
            # c must be > b
            neighbors_b = set(adj[b])
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
