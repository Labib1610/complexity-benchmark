```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest triad
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists c > b
    # such that (a, c) and (b, c) are edges.
    # To ensure lexicographical order, we iterate through possible a, then b.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through all edges (a, b)
    for a, b in edges:
        # We need to find c > b such that c is in adj[a] and c is in adj[b]
        # Since adj lists are sorted, we can use two pointers or set intersection
        # Given the constraints and the requirement for lexicographical order,
        # we check candidates c in order.
        
        # Optimization: intersect adj[a] and adj[b]
        # Since we need c > b, we only look at elements > b
        set_a = set(adj[a])
        set_b = set(adj[b])
        common = sorted(list(set_a.intersection(set_b)))
        
        for c in common:
            if c > b:
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
```
