```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[u] will store all v such that there is a dependency u -> v
    # Since dependency is always from smaller ID to larger ID,
    # u < v is guaranteed for all edges (u, v) in adj.
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].append(v)
    
    # Sort adjacency lists to ensure we find the lexicographically smallest triple
    for i in range(1, n + 1):
        adj[i].sort()
        
    # We need a < b < c such that a->b, a->c, and b->c.
    # Iterate through possible 'a'
    for a in range(1, n + 1):
        # For a fixed 'a', we need to find b, c in adj[a] such that b < c and b->c.
        # adj[a] is sorted.
        neighbors_a = adj[a]
        
        # To find the lexicographically smallest (a, b, c), we iterate b and c
        # in increasing order.
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Check if there exists c in neighbors_a such that c > b and b->c
            # We can use a set for O(1) lookup of edges from b
            # But since we need to check b->c, let's pre-process edges from b
            # into a set or boolean array for fast lookup.
            
            # Optimization: only check b if it has outgoing edges
            if not adj[b]:
                continue
            
            # Convert adj[b] to a set for O(1) lookup
            b_targets = set(adj[b])
            
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in b_targets:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
