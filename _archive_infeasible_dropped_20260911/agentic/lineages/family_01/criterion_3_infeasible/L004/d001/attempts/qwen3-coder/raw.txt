```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to help with lexicographical ordering
    edges.sort()
    
    # For each edge (u,v), look for a third node w such that (u,w) and (v,w) exist
    # We want the lexicographically smallest triplet
    
    # To optimize, we can iterate through all edges and for each edge (u,v)
    # check all nodes w that are neighbors of both u and v
    
    # Convert adjacency list to sorted lists for consistent iteration
    adj_list = {}
    for node in adj:
        adj_list[node] = sorted(list(adj[node]))
    
    best = None
    
    # Iterate through edges in sorted order to ensure we check smaller u first
    for u, v in edges:
        if u >= v:
            continue
        # Get common neighbors of u and v
        if u not in adj_list or v not in adj_list:
            continue
        neighbors_u = set(adj_list[u])
        neighbors_v = set(adj_list[v])
        common = neighbors_u & neighbors_v
        
        # For each common neighbor w, form triplet (u, v, w) with u < v < w
        for w in sorted(common):
            if v < w:
                candidate = (u, v, w)
                if best is None or candidate < best:
                    best = candidate
    
    if best:
        print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
```
