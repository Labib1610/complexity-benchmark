import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            continue
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        if u < v:
            edges.append((u, v))
        else:
            edges.append((v, u))

    # For each a<b, check common neighbors > b
    for a in range(n):
        # neighbors of a greater than a
        neigh_a = adj[a] >> (a + 1)
        # iterate b > a where edge exists
        mask = neigh_a
        while mask:
            lsb = mask & -mask
            b = (lsb.bit_length() - 1) + a + 1
            # common neighbors of a and b
            common = adj[a] & adj[b]
            # keep only c > b
            cand = common >> (b + 1)
            if cand:
                lsb_c = cand & -cand
                c = (lsb_c.bit_length() - 1) + b + 1
                print(a + 1, b + 1, c + 1)
                return
            mask ^= lsb

if __name__ == "__main__":
    solve()
