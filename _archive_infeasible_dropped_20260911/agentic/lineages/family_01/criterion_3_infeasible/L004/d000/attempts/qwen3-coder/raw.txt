```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for consistent processing
    for key in adj:
        adj[key] = sorted(adj[key])
    
    # We'll look for the lexicographically smallest triangle
    # Iterate through all nodes in increasing order
    min_triple = None
    
    for a in range(1, n+1):
        if a not in adj:
            continue
        neighbors_a = adj[a]
        # For each pair of neighbors of a, check if they are connected
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            for j in range(i+1, len(neighbors_a)):
                c = neighbors_a[j]
                if c <= b:
                    continue
                # Check if b and c are connected
                if c in adj[b]:
                    triple = (a, b, c)
                    if min_triple is None or triple < min_triple:
                        min_triple = triple
        
        # Alternative approach: for each neighbor b of a, look at neighbors of b
        for b in neighbors_a:
            if b <= a:
                continue
            # Now look for c that is neighbor of both a and b
            neighbors_b = adj[b]
            # Use two pointers or set intersection
            common = set(neighbors_a).intersection(neighbors_b)
            for c in common:
                if c > b:
                    triple = (a, b, c)
                    if min_triple is None or triple < min_triple:
                        min_triple = triple
                        
    if min_triple:
        print(min_triple[0], min_triple[1], min_triple[2])

if __name__ == "__main__":
    main()
```
