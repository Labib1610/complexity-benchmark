import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) with a < b < c
    # minimizing a, then b, then c.
    # Iterate through all possible 'a' from 1 to n
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # For a fixed 'a', we need to find b, c such that b, c are neighbors of a
        # and b is a neighbor of c.
        # To minimize b, then c, we iterate through sorted neighbors of a.
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need a c such that c is a neighbor of a and c is a neighbor of b
            # c must be > b
            # Intersection of adj[a] and adj[b]
            # Since we want to minimize b, we check b's in increasing order.
            # For a fixed a and b, we want the smallest c > b.
            
            # Optimization: check intersection
            # To make this efficient, use set intersection
            common = adj[a].intersection(adj[b])
            
            # Filter common neighbors that are > b
            valid_c = [x for x in common if x > b]
            if valid_c:
                c = min(valid_c)
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
