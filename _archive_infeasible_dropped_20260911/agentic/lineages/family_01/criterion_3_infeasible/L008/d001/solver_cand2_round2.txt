import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c,
    # we iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we look for 'b' in adj[a] such that b > a.
    # Then we look for 'c' in adj[a] intersect adj[b] such that c > b.
    
    # Sort adjacency lists to help find smallest b and c
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # To optimize, we iterate a, then b, then check if any c exists.
    # Since we need the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, b from sorted adj[a], and check intersection.
    
    # Convert adj to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Neighbors of a that are > a
        neighbors_a = [x for x in adj[a] if x > a]
        for b in neighbors_a:
            # We need a c such that c > b and c is in adj[a] and adj[b]
            # To get the smallest c, we iterate through the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                for c in adj[a]:
                    if c > b and c in adj_sets[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in adj[b]:
                    if c > b and c in adj_sets[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
