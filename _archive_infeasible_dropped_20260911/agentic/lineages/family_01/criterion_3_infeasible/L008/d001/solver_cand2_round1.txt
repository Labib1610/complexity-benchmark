import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Get neighbors of a that are > a
        neighbors_a = [v for v in adj[a] if v > a]
        neighbors_a.sort()
        
        # Iterate b from neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Check if there exists c > b such that c is in adj[a] and adj[b]
            # We want the smallest c, so iterate through sorted neighbors of b
            neighbors_b = adj[b]
            
            # Optimization: intersect neighbors_a and neighbors_b
            # Since we want the smallest c, we can iterate through sorted neighbors_a
            # and check if they are in neighbors_b
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
