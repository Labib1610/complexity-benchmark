```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    vetoes = []
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        vetoes.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # We need to find a triangle in the graph where all three edges exist
    # That means we need to find three nodes x, y, z such that
    # (x,y), (y,z), (x,z) are all in the veto list
    
    # Convert vetoes to a set for O(1) lookup
    veto_set = set()
    for u, v in vetoes:
        if u > v:
            u, v = v, u
        veto_set.add((u, v))
    
    # Iterate through all possible trios in lexicographical order
    for a in range(1, n+1):
        for b in range(a+1, n+1):
            if (a, b) not in veto_set:
                continue
            for c in range(b+1, n+1):
                if (a, c) in veto_set and (b, c) in veto_set:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
