```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency set for O(1) lookup of edges
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A closed culinary triad is a triangle in the graph.
    # In a triangle (x, y, z), the edges are (x, y), (y, z), and (z, x).
    # Each ingredient is part of exactly two recipes (edges) within the triad.
    # This is exactly the definition of a triangle in an undirected graph.
    
    # We need to find all triangles (x, y, z) with x < y < z
    # and pick the lexicographically smallest one.
    
    # To iterate efficiently:
    # Sort neighbors to help with the x < y < z condition
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    best = None
    
    # Iterate over all possible x
    for x in range(1, n + 1):
        # Optimization: if we already found a triad with a smaller x, we can stop
        if best and x > best[0]:
            break
            
        # Iterate over neighbors y of x such that y > x
        for y in adj_list[x]:
            if y <= x:
                continue
            if best and x == best[0] and y > best[1]:
                continue
                
            # Check for z such that (x, z) and (y, z) are edges
            # z must be > y
            for z in adj_list[y]:
                if z <= y:
                    continue
                
                # Check if z is connected to x
                if z in adj[x]:
                    # Found a triangle (x, y, z)
                    if best is None or (x < best[0]) or (x == best[0] and y < best[1]) or (x == best[0] and y == best[1] and z < best[2]):
                        best = (x, y, z)
    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
