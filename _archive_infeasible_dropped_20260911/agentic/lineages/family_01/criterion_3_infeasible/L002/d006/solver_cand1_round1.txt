import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # We need to find a triad (a, b, c) such that (a, b), (b, c), and (a, c)
    # are all in the set of incompatibilities.
    # Using an adjacency list of sets for O(1) lookup.
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, iterate c from b+1 to n
    # Check if c is in adj[a] and c is in adj[b]
    
    for a in range(1, n + 1):
        # Only consider 'a' if it has at least 2 neighbors
        if len(adj[a]) < 2:
            continue
        
        # Sort neighbors to ensure we check in order
        neighbors_a = sorted([x for x in adj[a] if x > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Check if there exists c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors of a and b
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
