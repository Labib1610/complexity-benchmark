import sys

def solve():
    # A complete synergy trio is a set of 3 components {a, b, c} 
    # such that there are exactly 3 stable pairings forming them.
    # This is equivalent to saying {a, b, c} forms a triangle (a-b, b-c, c-a).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest triangle {a, b, c} with a < b < c:
    # We iterate through all possible 'a' from 1 to n-2.
    # Then iterate through all neighbors 'b' of 'a' such that b > a.
    # Then check if there exists a 'c' such that c > b and c is a neighbor of both a and b.
    
    # Sort neighbors to ensure we find the lexicographically smallest trio
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # Since adj[a] and adj[b] are sorted, we can use two pointers or set intersection
            # Given the constraints and the requirement for the smallest trio,
            # we can just iterate through the sorted neighbors of b.
            
            neighbors_b = adj[b]
            # Find intersection of neighbors_a and neighbors_b
            # Since we need the smallest c > b, we look for the first common element > b
            ptr_a = i + 1
            ptr_b = 0
            
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
