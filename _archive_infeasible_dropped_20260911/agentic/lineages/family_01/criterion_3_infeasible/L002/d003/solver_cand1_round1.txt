```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if v not in adj[u]:
            adj[u].add(v)
        if u not in adj[v]:
            adj[v].add(u)
            
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # Iterate through a from 1 to n-2
    # Iterate through b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Optimization: sort adjacency lists to iterate efficiently
    sorted_adj = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need a c such that c > b, c in adj[a], c in adj[b]
            # Intersect neighbors_a and neighbors_b
            # Since both are sorted, we can find common elements efficiently
            # But we only need the smallest c > b
            
            neighbors_b = sorted_adj[b]
            
            # Use two pointers to find common elements > b
            ptr_a = 0
            ptr_b = 0
            len_a = len(neighbors_a)
            len_b = len(neighbors_b)
            
            while ptr_a < len_a and ptr_b < len_b:
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
```
