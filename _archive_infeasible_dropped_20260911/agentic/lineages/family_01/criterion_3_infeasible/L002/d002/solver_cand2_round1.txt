```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find the lexicographically smallest (a, b, c) with a < b < c
    # such that (a,b), (a,c), (b,c) are edges.
    # Iterate through a from 1 to n-2
    for a in range(1, n - 1):
        neighbors_a = sorted([x for x in adj[a] if x > a])
        if len(neighbors_a) < 2:
            continue
            
        # Iterate through b in neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Iterate through c in neighbors_a after b
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
