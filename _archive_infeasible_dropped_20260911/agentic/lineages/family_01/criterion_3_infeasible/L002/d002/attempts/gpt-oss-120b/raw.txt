import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    # adjacency matrix for O(1) edge test
    adj_mat = [bytearray(n + 1) for _ in range(n + 1)]
    # adjacency bitsets (bits 0..n-1 correspond to vertices 1..n)
    adj_bits = [0] * (n + 1)

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj_mat[u][v] = adj_mat[v][u] = 1
        adj_bits[u] |= 1 << (v - 1)
        adj_bits[v] |= 1 << (u - 1)

    full_mask = (1 << n) - 1

    for a in range(1, n - 1):
        bits_a = adj_bits[a]
        for b in range(a + 1, n):
            if not adj_mat[a][b]:
                continue
            # common neighbours of a and b
            common = bits_a & adj_bits[b]
            # keep only vertices > b
            mask = full_mask ^ ((1 << b) - 1)   # bits b..n-1 (0‑based), we need >b so b index is b
            common &= mask
            if common:
                # lowest set bit gives smallest c
                low = common & -common
                c_idx = low.bit_length() - 1   # 0‑based
                c = c_idx + 1
                sys.stdout.write(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
