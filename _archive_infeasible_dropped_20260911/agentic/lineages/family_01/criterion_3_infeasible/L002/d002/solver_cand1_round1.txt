```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # We need to find a triangle (a, b, c) such that edges (a,b), (b,c), (a,c) exist.
    # The problem asks for the lexicographically smallest triplet (a, b, c) with a < b < c.
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n-2,
    # b from a+1 to n-1, and c from b+1 to n.
    # Since we need to check if (a,b), (b,c), (a,c) exist, we check if c is in adj[a] and adj[b].
    
    # Optimization: iterate through sorted adjacency lists to find common neighbors
    # For a fixed a and b, we need to find c > b such that c in adj[a] and c in adj[b].
    
    best = None
    
    # Sort adjacency lists to make intersection faster and ordered
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        for b in sorted_adj[a]:
            if b <= a:
                continue
            
            # Find common neighbors of a and b that are > b
            # Since sorted_adj[a] and sorted_adj[b] are sorted, we can intersect them
            # efficiently.
            ptr_a = 0
            ptr_b = 0
            list_a = sorted_adj[a]
            list_b = sorted_adj[b]
            
            while ptr_a < len(list_a) and ptr_b < len(list_b):
                val_a = list_a[ptr_a]
                val_b = list_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        # Found a triangle (a, b, val_a)
                        # Since we iterate a and b in increasing order, 
                        # the first one we find might not be the global minimum,
                        # but we can keep track of the minimum.
                        res = tuple(sorted((a, b, val_a)))
                        if best is None or res < best:
                            best = res
                        # Since we want the lexicographically smallest, 
                        # we can break early if we find a candidate for a fixed a, b
                        break
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1
                    
    print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
