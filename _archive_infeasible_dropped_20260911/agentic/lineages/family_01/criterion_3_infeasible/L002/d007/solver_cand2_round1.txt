import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A Pipelined Trio {a, b, c} requires that for all 6 permutations,
    # the first can be pipelined with the second.
    # This means (a,b), (b,a), (a,c), (c,a), (b,c), (c,b) must all be in the set.
    # Since the relationship is symmetric, this is equivalent to saying
    # that a, b, and c must form a clique of size 3 (a triangle).
    
    # To find the lexicographically smallest trio (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Iterate c from b+1 to n
    # Check if b in adj[a], c in adj[a], and c in adj[b]
    
    for a in range(1, n - 1):
        # Optimization: only consider b if it's connected to a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for b in neighbors_a:
            # Optimization: only consider c if it's connected to both a and b
            # Intersection of neighbors of a and b
            neighbors_b = adj[b]
            # We need c > b and c in neighbors_a and c in neighbors_b
            # To be efficient, iterate over the smaller neighbor set
            if len(adj[a]) < len(adj[b]):
                candidates = [c for c in adj[a] if c > b and c in neighbors_b]
            else:
                candidates = [c for c in adj[b] if c > b and c in adj[a]]
            
            if candidates:
                print(f"{a} {b} {min(candidates)}")
                return

if __name__ == '__main__':
    solve()
