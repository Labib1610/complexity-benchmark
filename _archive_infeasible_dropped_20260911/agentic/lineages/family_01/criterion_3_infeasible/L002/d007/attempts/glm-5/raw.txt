```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    for i in range(1, n + 1):
        adj[i] |= 1 << i
    for a in range(1, n + 1):
        da = adj[a]
        for b in range(a + 1, n + 1):
            if (da >> b) & 1:
                db = adj[b]
                common = da & db
                if common:
                    c = (common & -common).bit_length() - 1
                    if c > b:
                        print(a, b, c)
                        return
solve()
```
