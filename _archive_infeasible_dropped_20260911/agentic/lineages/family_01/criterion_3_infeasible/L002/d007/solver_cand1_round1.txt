import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[i] will store the set of candidates that candidate i has a veto with.
    # The problem defines the incompatibility domain of i as {i} U {v | i vetoes v}.
    # A trio {a, b, c} is deadlocked if {a, b, c} is a subset of the incompatibility domain of a, b, and c.
    # This means:
    # {a, b, c} subset of {a} U adj[a] => b in adj[a] and c in adj[a]
    # {a, b, c} subset of {b} U adj[b] => a in adj[b] and c in adj[b]
    # {a, b, c} subset of {c} U adj[c] => a in adj[c] and b in adj[c]
    # This is equivalent to saying {a, b, c} forms a clique of size 3 in the veto graph.
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest trio {a, b, c} with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Optimization: convert adj sets to sorted lists for faster intersection
    adj_list = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        # Candidates b > a that are connected to a
        neighbors_a = [x for x in adj_list[a] if x > a]
        for b in neighbors_a:
            # Candidates c > b that are connected to both a and b
            # We need to find the smallest c > b such that c in adj[a] and c in adj[b]
            # Since adj_list is sorted, we can use intersection
            # But we only need the smallest one.
            
            # Using set intersection for efficiency
            common = adj[a].intersection(adj[b])
            for c in sorted(list(common)):
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
