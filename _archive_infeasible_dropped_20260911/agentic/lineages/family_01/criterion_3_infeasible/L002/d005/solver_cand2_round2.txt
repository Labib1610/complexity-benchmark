```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A completely fractured triad is a triangle in the conflict graph.
    # We need the lexicographically smallest (a, b, c) with a < b < c.
    # Iterate through all possible a, then b, then c.
    # To optimize, iterate a from 1 to n, b from a+1 to n, and check if b is in adj[a].
    # If so, check if there exists c > b such that c is in adj[a] and adj[b].
    
    # Sort adjacency lists to help find c efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Neighbors of a that are greater than a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a[i+1:] and sorted_adj[b]
            # Since both are sorted, we can use a two-pointer approach
            ptr_a = i + 1
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(sorted_adj[b]):
                val_a = neighbors_a[ptr_a]
                val_b = sorted_adj[b][ptr_b]
                
                if val_a == val_b:
                    print(f"{a} {b} {val_a}")
                    return
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

if __name__ == '__main__':
    solve()
```
