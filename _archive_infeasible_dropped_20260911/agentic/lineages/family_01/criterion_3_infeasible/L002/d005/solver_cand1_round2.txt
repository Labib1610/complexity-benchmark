import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through possible a, then b, then c.
    # A triad (a, b, c) is fractured if (a, b), (a, c), and (b, c) are edges.
    
    # Sort edges to help iterate efficiently
    edges.sort()
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Iterate through all possible 'b' such that a < b and (a, b) is an edge
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a 'c' such that b < c, (a, c) is an edge, and (b, c) is an edge.
            # This means c must be in the intersection of adj[a] and adj[b].
            # Since adj lists are sorted, we can find intersection efficiently.
            
            # Optimization: only check c > b
            # Use set for O(1) lookup if needed, but intersection of sorted lists is fast
            # Given constraints, intersection of two sorted lists is efficient.
            
            # Find intersection of neighbors_a[i+1:] and adj[b]
            # Since we want the smallest c, we just need the first match.
            
            ptr_a = i + 1
            ptr_b = 0
            
            # adj[b] is sorted, neighbors_a is sorted.
            # We need c > b.
            while ptr_a < len(neighbors_a) and ptr_b < len(adj[b]):
                val_a = neighbors_a[ptr_a]
                val_b = adj[b][ptr_b]
                
                if val_a == val_b:
                    # Found the smallest c > b
                    print(f"{a} {b} {val_a}")
                    return
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
