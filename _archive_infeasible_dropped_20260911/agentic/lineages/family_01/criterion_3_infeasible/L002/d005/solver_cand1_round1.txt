```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # dossiers[i] will store the set of log indices where technician i+1 appears
    dossiers = [set() for _ in range(n)]
    
    idx = 2
    for i in range(1, m + 1):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        dossiers[u].add(i)
        dossiers[v].add(i)
        
    # Precompute pairwise intersections to speed up checks
    # Since we need the lexicographically smallest, we iterate i, j, k
    # A triad (i, j, k) is fractured if:
    # dossiers[i] & dossiers[j] is not empty
    # dossiers[i] & dossiers[k] is not empty
    # dossiers[j] & dossiers[k] is not empty
    
    # To optimize, we can pre-calculate adjacency: adj[i][j] = True if intersection is non-empty
    adj = [[False for _ in range(n)] for _ in range(n)]
    for i in range(n):
        for j in range(i + 1, n):
            if dossiers[i].isdisjoint(dossiers[j]):
                continue
            else:
                adj[i][j] = adj[j][i] = True
                
    # Iterate to find the lexicographically smallest triad
    for i in range(n):
        for j in range(i + 1, n):
            if not adj[i][j]:
                continue
            for k in range(j + 1, n):
                if adj[i][k] and adj[j][k]:
                    print(f"{i + 1} {j + 1} {k + 1}")
                    return

if __name__ == '__main__':
    solve()
```
