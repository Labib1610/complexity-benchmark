```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list using sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        index += 2
    
    # For each edge (u,v), check for a common neighbor w
    # such that (u,v,w) forms a triangle
    triangles = []
    
    for u, v in edges:
        if u > v:
            continue
        # Find common neighbors of u and v
        for w in adj[u]:
            if w > v and w in adj[v]:
                # Found a triangle (u, v, w)
                triad = tuple(sorted([u, v, w]))
                triangles.append(triad)
    
    # Sort triangles lexicographically and return the first one
    triangles.sort()
    result = triangles[0]
    print(result[0], result[1], result[2])

if __name__ == "__main__":
    main()
```
