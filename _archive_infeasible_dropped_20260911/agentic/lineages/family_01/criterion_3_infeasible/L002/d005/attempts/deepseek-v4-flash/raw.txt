We can solve this by iterating over each edge and checking for a common neighbor that forms a triangle. To keep it efficient, we’ll use adjacency sets for O(1) neighbor checks.

```python
import sys

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    # Sort edges by smallest vertex first for lexicographic search
    edges.sort(key=lambda x: (min(x), max(x)))

    for u, v in edges:
        # Find common neighbors
        common = adj[u] & adj[v]
        if common:
            w = min(common)
            a, b, c = sorted((u, v, w))
            print(a, b, c)
            return

if __name__ == "__main__":
    main()
```
