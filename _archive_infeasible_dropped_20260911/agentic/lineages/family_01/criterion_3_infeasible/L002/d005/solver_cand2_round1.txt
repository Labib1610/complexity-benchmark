import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency set to store undirected edges
    # A triad {a, b, c} is completely fractured if all 3 pairs
    # {a, b}, {b, c}, {a, c} exist in the grievance log.
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
    
    # We need to find the lexicographically smallest (a, b, c) with a < b < c
    # such that {a, b}, {b, c}, and {a, c} are all edges.
    # Iterate through a from 1 to n-2
    for a in range(1, n - 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted(list(adj[a]))
        
        # For each pair of neighbors b, c of a, check if {b, c} is an edge
        # Since we want lexicographical order, we iterate b and c in order
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if {b, c} exists (b < c is guaranteed by sorted neighbors_a)
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
