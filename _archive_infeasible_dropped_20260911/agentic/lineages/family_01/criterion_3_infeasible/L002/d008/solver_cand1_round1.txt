```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[u] stores a set of neighbors of u
    # edge_id maps (min(u, v), max(u, v)) to an event ID (0 to m-1)
    adj = [set() for _ in range(n + 1)]
    edge_to_id = {}
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edge_to_id[(u, v)] = i
        
    # A trio (x, y, z) is deadlocked if |(Px ∩ Py) ∪ (Py ∩ Pz) ∪ (Pz ∩ Px)| == 3
    # Px ∩ Py is the edge (x, y) if it exists, else empty.
    # So the shared portfolio is the set of edges between the three pairs:
    # {(x, y), (y, z), (z, x)} intersected with the set of existing edges.
    # The condition is that exactly 3 edges exist among the trio (x, y, z).
    # This means the trio must form a triangle in the graph.
    
    # Iterate through all triangles (x, y, z) with x < y < z
    # To find the lexicographically smallest, we iterate x from 1 to n,
    # y from x+1 to n, and z from y+1 to n.
    
    for x in range(1, n + 1):
        adj_x = adj[x]
        # Sort neighbors to help find y and z in order
        sorted_adj_x = sorted(list(adj_x))
        for i in range(len(sorted_adj_x)):
            y = sorted_adj_x[i]
            if y <= x:
                continue
            
            # z must be a neighbor of both x and y
            adj_y = adj[y]
            # Intersection of neighbors of x and y
            common = sorted(list(adj_x.intersection(adj_y)))
            
            for z in common:
                if z <= y:
                    continue
                
                # Trio (x, y, z) is a triangle.
                # The shared portfolio is the set of edges between them.
                # Since it's a triangle, all 3 edges (x,y), (y,z), (z,x) exist.
                # The shared portfolio size is 3.
                print(f"{x} {y} {z}")
                return

solve()
```
