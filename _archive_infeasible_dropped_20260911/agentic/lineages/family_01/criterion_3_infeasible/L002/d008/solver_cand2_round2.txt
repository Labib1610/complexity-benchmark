import sys

def solve():
    # A trio {a, b, c} is deadlocked if it cannot be partitioned into two sets
    # without any internal vetoes. This is equivalent to saying the trio
    # contains an odd cycle. In a set of 3, the only odd cycle is a triangle (K3).
    # Thus, a deadlocked trio is simply a triangle in the graph of vetoes.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through a from 1 to n-2
    # Iterate through b from a+1 to n-1
    # Check if there exists c > b such that (a, c) and (b, c) are edges.
    
    # Sort adjacency lists to optimize
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that b < c and c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can intersect efficiently
            neighbors_b = sorted_adj[b]
            
            # Use two pointers to find common neighbor c > b
            ptr_a = i + 1
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
