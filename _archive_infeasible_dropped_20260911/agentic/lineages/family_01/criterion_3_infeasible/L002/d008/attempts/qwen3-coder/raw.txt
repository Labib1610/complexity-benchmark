```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Iterate through all possible trios in lexicographical order
    for a in range(1, n - 1):
        for b in range(a + 1, n):
            for c in range(b + 1, n + 1):
                # Check if this trio is deadlocked
                # A trio {a,b,c} is deadlocked if the subgraph induced by these three nodes
                # is a triangle (i.e., all three edges exist), because then it's impossible
                # to bipartition them without having a vetoed pair in the same set.
                
                has_ab = b in adj[a]
                has_ac = c in adj[a]
                has_bc = c in adj[b]
                
                if has_ab and has_ac and has_bc:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
