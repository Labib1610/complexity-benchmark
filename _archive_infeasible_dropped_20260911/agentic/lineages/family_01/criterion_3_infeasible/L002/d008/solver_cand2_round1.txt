```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # A trio {a, b, c} has total penalty P(a,b) + P(b,c) + P(c,a) = 3.
    # P(x, y) is 1 if {x, y} is a veto, 2 otherwise.
    # Let k be the number of vetoes in the trio.
    # Total penalty = k * 1 + (3 - k) * 2 = 3
    # k + 6 - 2k = 3 => k = 3.
    # So a deadlocked trio is a triangle in the veto graph.
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need the lexicographically smallest {a, b, c} with a < b < c.
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Iterate b from a+1 to n-1
        for b in range(a + 1, n):
            if b in adj[a]:
                # Check for c > b such that {a, c} and {b, c} are vetoes
                # Intersection of adj[a] and adj[b]
                common = adj[a].intersection(adj[b])
                # Filter for c > b
                valid_c = [c for c in common if c > b]
                if valid_c:
                    print(f"{a} {b} {min(valid_c)}")
                    return

solve()
```
