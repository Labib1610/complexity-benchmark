import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # Sort edges to help find lexicographically smallest triad
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check for a common neighbor c
    # such that c > b.
    
    best_triad = None
    
    # Optimization: sort adjacency lists to iterate efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a < b < c. Since we iterate a then b, 
            # we only care about c > b.
            # Check intersection of neighbors_a and neighbors_b
            # Only check c > b
            neighbors_b = adj[b]
            
            for c in neighbors_a[i+1:]:
                if c in neighbors_b:
                    # Found a triad (a, b, c)
                    # Since we iterate a from 1..n, b from neighbors of a,
                    # and c from neighbors of a, the first one we find 
                    # might not be the absolute lexicographical smallest.
                    # However, if we find one, we can keep track of the best.
                    if best_triad is None or (a, b, c) < best_triad:
                        best_triad = (a, b, c)
            
            # If we have a best_triad, and the current 'a' is already 
            # greater than best_triad[0], we can stop.
            if best_triad and a > best_triad[0]:
                break
        if best_triad and a > best_triad[0]:
            break
            
    print(f"{best_triad[0]} {best_triad[1]} {best_triad[2]}")

if __name__ == '__main__':
    solve()
