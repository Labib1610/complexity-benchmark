```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest (a, b, c)
    # We want smallest a, then smallest b, then smallest c.
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # all edges (a, b) and check if there exists a c > b such that
    # (a, c) and (b, c) are both edges.
    
    best = None
    
    # Optimization: iterate through edges (a, b)
    for a, b in edges:
        # We need to find c > b such that c is in adj[a] and adj[b]
        # To keep it lexicographically smallest, we check candidates in increasing order
        # Intersection of adj[a] and adj[b]
        common = adj[a].intersection(adj[b])
        
        for c in common:
            if c > b:
                # Found a triplet (a, b, c)
                # Since we want the lexicographically smallest, we compare
                if best is None or (a, b, c) < best:
                    best = (a, b, c)
        
        # If we have a best, and the current 'a' is already larger than best[0],
        # we can't find a smaller one.
        if best and a > best[0]:
            break
            
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
