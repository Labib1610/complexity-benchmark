import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    best = None

    # Strategy 1: iterate over edges (u, v) with u < v, then iterate over
    # common neighbors w > v. This naturally finds triangles in sorted order.
    for u, v in edges:
        if u > v:
            u, v = v, u
        # common neighbors w with w > v
        if len(adj[u]) <= len(adj[v]):
            small, large = adj[u], adj[v]
        else:
            small, large = adj[v], adj[u]
        for w in small:
            if w > v and w in large:
                cand = (u, v, w)
                if best is None or cand < best:
                    best = cand
                    return  # lexicographically smallest possible, can stop early

    if best is not None:
        print(best[0], best[1], best[2])
        return

    # Strategy 2: fallback for triangles where the smallest two vertices are
    # not connected by an edge (e.g., 1-3, 1-4, 3-4 with no 1-3 edge... wait
    # that's impossible). Actually every triangle has all three edges, so
    # strategy 1 covers all triangles. But to be safe, do a full scan.
    for a in range(1, n + 1):
        neighbors_a = sorted(adj[a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b < a:
                continue
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c < b:
                    continue
                if c in adj[b]:
                    cand = (a, b, c)
                    if best is None or cand < best:
                        best = cand
                        print(best[0], best[1], best[2])
                        return

    if best is not None:
        print(best[0], best[1], best[2])

solve()
