import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    
    # Iterate in lexicographic order of (a, b, c)
    for a in range(1, n - 1):
        adj_a = adj[a]
        # b must be > a and in adj_a
        # We need b < c, both in adj_a, and c in adj[b]
        # To get lexicographically smallest, iterate b in increasing order
        for b in adj_a:
            if b <= a:
                continue
            adj_b = adj[b]
            # Find smallest c > b such that c in adj_a and c in adj_b
            # Since sets are unordered, we iterate over the smaller set
            if len(adj_a) <= len(adj_b):
                for c in adj_a:
                    if c > b and c in adj_b:
                        print(a, b, c)
                        return
            else:
                for c in adj_b:
                    if c > b and c in adj_a:
                        print(a, b, c)
                        return

if __name__ == "__main__":
    solve()
