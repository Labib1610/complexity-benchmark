We can solve this by checking each possible triple efficiently.  
Since \( n \) can be up to 2000, checking all triples naively would be too slow.  
Instead, we can iterate over the first two vertices and check if they have a common neighbor that is larger than both, and pick the smallest such neighbor.

```python
import sys

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # find smallest c > b such that c is connected to both a and b
                for c in range(b + 1, n + 1):
                    if c in adj[a] and c in adj[b]:
                        print(a, b, c)
                        return

if __name__ == "__main__":
    main()
```
