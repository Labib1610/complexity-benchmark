import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use a bitset-like approach with sets or boolean arrays for adjacency
    # Since n <= 2000, an adjacency matrix of bitsets is efficient.
    # adj[i] will store a bitset (integer) where the j-th bit is 1 if 
    # there is a grievance between i and j.
    adj = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        # The problem states "appears in at least one log entry... with each of the other two"
        # This means the relationship is undirected for the purpose of the triad.
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    # We need to find the lexicographically smallest (a, b, c) with a < b < c
    # such that b is in adj[a], c is in adj[a], and c is in adj[b].
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        adj_a = adj[a]
        # Find all neighbors of a that are > a
        neighbors_a = []
        temp_adj_a = adj_a >> (a + 1)
        for b in range(a + 1, n):
            if (adj_a >> b) & 1:
                neighbors_a.append(b)
        
        # For each pair (b, c) in neighbors_a, check if c is in adj[b]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            adj_b = adj[b]
            # We need c > b such that c is in neighbors_a and c is in adj[b]
            # This is equivalent to c in (neighbors_a[i+1:] AND adj[b])
            common = 0
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if (adj_b >> c) & 1:
                    print(f"{a} {b} {c}")
                    return

solve()
