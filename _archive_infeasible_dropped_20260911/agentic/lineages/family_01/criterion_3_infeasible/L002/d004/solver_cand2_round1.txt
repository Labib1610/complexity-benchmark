import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # V_x[i] is 1 if x filed grievance i, -1 if x was filed against in i, 0 otherwise.
    # Dot product V_a . V_b = sum_{i=1}^m V_a[i] * V_b[i]
    # V_a[i] * V_b[i] is:
    #  1 if (a filed i and b filed i) or (a was filed against i and b was filed against i)
    # -1 if (a filed i and b was filed against i) or (a was filed against i and b filed i)
    #  0 otherwise
    
    # Let F_x be the set of grievances filed by x, A_x be the set of grievances against x.
    # V_a . V_b = |F_a \cap F_b| + |A_a \cap A_b| - |F_a \cap A_b| - |A_a \cap F_b|
    
    f_sets = [set() for _ in range(n + 1)]
    a_sets = [set() for _ in range(n + 1)]
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        f_sets[u].add(i)
        a_sets[v].add(i)
        idx += 2
        
    # Precompute dot products for all pairs (i, j)
    # Since n <= 2000, n^2 is 4*10^6, which is manageable.
    dots = [[0] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        # To compute dots[i][j] efficiently:
        # |F_i \cap F_j| = count of grievances where both i and j filed
        # |A_i \cap A_j| = count of grievances where both i and j were filed against
        # |F_i \cap A_j| = count of grievances where i filed and j was filed against
        # |A_i \cap F_j| = count of grievances where i was filed against and j filed
        
        # We can iterate over all grievances to fill this
        pass
    
    # Actually, iterating over all grievances is O(m).
    # For each grievance i: u filed against v
    # For all x, if x filed i, increment dots[x][u] and decrement dots[x][v]
    # If x was filed against i, decrement dots[x][u] and increment dots[x][v]
    
    # Let's use the sets to compute dot products
    for i in range(1, n + 1):
        for j in range(i + 1, n + 1):
            val = len(f_sets[i] & f_sets[j]) + len(a_sets[i] & a_sets[j]) - \
                  len(f_sets[i] & a_sets[j]) - len(a_sets[i] & f_sets[j])
            dots[i][j] = dots[j][i] = val
            
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if dots[a][b] < 0:
                for c in range(b + 1, n + 1):
                    if dots[b][c] < 0 and dots[a][c] < 0:
                        print(f"{a} {b} {c}")
                        return

solve()
