import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # S[x] will store the set of log IDs for technician x
    # Using bitsets (integers in Python) to represent sets of log IDs
    # Since m can be up to 2*10^5, bitsets are efficient for intersection checks
    s = [0] * (n + 1)
    
    idx = 2
    for log_id in range(1, m + 1):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        
        # log_id is 1-indexed, shift to 0-indexed for bit manipulation
        bit = 1 << (log_id - 1)
        s[u] |= bit
        s[v] |= bit
        
    # We need to find the lexicographically smallest (a, b, c) with a < b < c
    # such that (S[a] & S[b]) != 0, (S[b] & S[c]) != 0, (S[a] & S[c]) != 0
    
    # Precompute adjacency: adj[i] is a bitmask of technicians connected to i
    # Two technicians are connected if their intersection is non-empty
    adj = [0] * (n + 1)
    for i in range(1, n + 1):
        for j in range(i + 1, n + 1):
            if s[i] & s[j]:
                adj[i] |= (1 << j)
                adj[j] |= (1 << i)
                
    # Iterate to find the lexicographically smallest triad
    for a in range(1, n + 1):
        # Get all neighbors of a
        neighbors_a = []
        mask_a = adj[a]
        for b in range(a + 1, n + 1):
            if (mask_a >> b) & 1:
                neighbors_a.append(b)
        
        # For each pair (b, c) in neighbors_a, check if they are connected
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                if (adj[b] >> c) & 1:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
