import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A trio {a, b, c} with a < b < c
    # Internal conflict sets:
    # S_a = {x in {b, c} | x in adj[a]}
    # S_b = {x in {a, c} | x in adj[b]}
    # S_c = {x in {a, b} | x in adj[c]}
    # Condition 1: S_a, S_b, S_c are non-empty
    # Condition 2: S_a != S_b, S_a != S_c, S_b != S_c
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            # Optimization: check if a and b can be part of a trio
            # S_a must contain at least one of {b, c}
            # S_b must contain at least one of {a, c}
            for c in range(b + 1, n + 1):
                s_a = set()
                if b in adj[a]: s_a.add(b)
                if c in adj[a]: s_a.add(c)
                
                if not s_a: continue
                
                s_b = set()
                if a in adj[b]: s_b.add(a)
                if c in adj[b]: s_b.add(c)
                
                if not s_b: continue
                
                s_c = set()
                if a in adj[c]: s_c.add(a)
                if b in adj[c]: s_c.add(b)
                
                if not s_c: continue
                
                if s_a != s_b and s_a != s_c and s_b != s_c:
                    print(f"{a} {b} {c}")
                    return

solve()
