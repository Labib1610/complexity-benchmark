import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1])
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    for a in range(1, n + 1):
        bit_a = 1 << a
        adj_a = adj[a]
        for b in range(a + 1, n + 1):
            if not (adj_a & (1 << b)):
                continue
            bit_b = 1 << b
            adj_b = adj[b]
            common = adj_a & adj_b
            common_high = common & ~((1 << b) - 1)
            c = b + 1
            while c <= n:
                if common_high & (1 << c):
                    bit_c = 1 << c
                    adj_c = adj[c]
                    
                    sa = 0
                    if adj_a & bit_b: sa |= bit_b
                    if adj_a & bit_c: sa |= bit_c
                    
                    sb = 0
                    if adj_b & bit_a: sb |= bit_a
                    if adj_b & bit_c: sb |= bit_c
                    
                    sc = 0
                    if adj_c & bit_a: sc |= bit_a
                    if adj_c & bit_b: sc |= bit_b
                    
                    if sa and sb and sc and sa != sb and sb != sc and sa != sc:
                        print(a, b, c)
                        return
                c += 1

solve()
