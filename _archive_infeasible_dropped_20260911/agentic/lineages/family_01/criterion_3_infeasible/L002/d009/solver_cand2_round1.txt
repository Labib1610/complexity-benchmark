```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency matrix for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A system of equations z_u + z_v = 0 for u, v in {a, b, c}
    # has only the solution z_a = z_b = z_c = 0 if and only if
    # the graph formed by these 3 vertices and the edges (u, v)
    # is not bipartite.
    # A graph with 3 vertices is not bipartite if and only if it contains
    # an odd cycle. The only odd cycle possible in a 3-vertex graph is
    # a triangle (a cycle of length 3).
    # Thus, the condition is that the subgraph induced by {a, b, c}
    # must contain all three edges (a, b), (b, c), and (a, c).
    
    # We need the lexicographically smallest (a, b, c) such that
    # a < b < c and edges (a, b), (b, c), (a, c) all exist.
    
    for a in range(1, n + 1):
        # Optimization: only iterate over neighbors of a
        neighbors_a = sorted(list(adj[a]))
        len_na = len(neighbors_a)
        for i in range(len_na):
            b = neighbors_a[i]
            if b <= a:
                continue
            # We need c such that c > b and c is connected to both a and b
            # Check neighbors of b that are also neighbors of a
            for c in neighbors_a[i+1:]:
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
```
