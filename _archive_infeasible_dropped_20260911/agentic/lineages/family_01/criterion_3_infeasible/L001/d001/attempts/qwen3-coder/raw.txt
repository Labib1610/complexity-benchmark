```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    edge_set = set()
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        edge_set.add((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to help with finding lexicographically smallest
    edges.sort()
    
    # For each edge, check its neighbors to find triangles
    found = False
    result = None
    
    # We'll iterate through all edges and try to find a triangle
    for u, v in edges:
        # Check for common neighbors of u and v
        for w in adj[u]:
            if w != v and (min(v,w), max(v,w)) in edge_set:
                # Found a triangle: u, v, w
                triple = tuple(sorted([u, v, w]))
                if not found or triple < result:
                    result = triple
                    found = True
        if found:
            break
    
    if not found:
        # Fallback: check all combinations (less efficient but guaranteed to work)
        nodes = list(range(1, n+1))
        for i in range(n):
            for j in range(i+1, n):
                for k in range(j+1, n):
                    u, v, w = nodes[i], nodes[j], nodes[k]
                    if (min(u,v), max(u,v)) in edge_set and \
                       (min(u,w), max(u,w)) in edge_set and \
                       (min(v,w), max(v,w)) in edge_set:
                        print(u, v, w)
                        return
    
    if result:
        print(result[0], result[1], result[2])

if __name__ == "__main__":
    main()
```
