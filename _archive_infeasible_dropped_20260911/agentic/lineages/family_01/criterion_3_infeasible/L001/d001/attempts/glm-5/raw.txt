import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    deg = [0] * (n + 1)
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u][v] = True
        adj[v][u] = True
        deg[u] += 1
        deg[v] += 1
    
    # We need to find a triangle (3-clique).
    # For each edge (u, v), check if there exists w such that (u, w) and (v, w) are edges.
    # To find the lexicographically smallest triangle {a, b, c} (a < b < c),
    # we iterate a from 1 to n, b from a+1 to n, and check for c.
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if adj[a][b]:
                # We have edge (a, b). We need a common neighbor c > b.
                # Optimization: iterate c from b+1 to n.
                # Since n <= 2000, this is O(n^3) in worst case, but acceptable.
                # With m constraints, it's faster.
                for c in range(b + 1, n + 1):
                    if adj[a][c] and adj[b][c]:
                        print(a, b, c)
                        return

if __name__ == '__main__':
    solve()
