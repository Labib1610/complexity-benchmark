import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Sort adjacency lists to potentially speed up intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Use sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider nodes with at least 2 neighbors
        if len(adj[a]) < 2:
            continue
        
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # Since neighbors_a is sorted, we can iterate through it
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c > b and c in adj_sets[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
