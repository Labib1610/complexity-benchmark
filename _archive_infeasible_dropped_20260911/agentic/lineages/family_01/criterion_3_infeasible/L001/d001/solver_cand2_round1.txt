import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # Sort edges to help find the lexicographically smallest (a, b, c)
    # We want to minimize a, then b, then c.
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # Get all neighbors of a
        neighbors_a = sorted(list(adj[a]))
        if len(neighbors_a) < 2:
            continue
            
        # Iterate through all pairs (b, c) in neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Since we want a < b < c, we only care about b > a
            if b <= a:
                continue
            
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
