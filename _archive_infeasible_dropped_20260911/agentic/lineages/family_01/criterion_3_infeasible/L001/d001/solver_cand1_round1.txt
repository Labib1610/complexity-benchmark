```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # We need to find a triangle (a, b, c) with a < b < c.
    # To find the lexicographically smallest, we iterate through possible a, then b, then c.
    # A triangle exists if for an edge (u, v) with u < v, there exists a node w
    # such that w is connected to both u and v.
    # The triangle is (min(u, v, w), mid(u, v, w), max(u, v, w)).
    
    best = None
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    for u, v in edges:
        # Triangle is (u, v, w) where w is in adj[u] and adj[v]
        # We want to minimize the sorted tuple (a, b, c)
        # Intersection of adj[u] and adj[v]
        common = adj[u].intersection(adj[v])
        for w in common:
            # Form the sorted triplet
            triplet = sorted([u, v, w])
            if best is None or triplet < best:
                best = triplet
                
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
