import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triad
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # We iterate through all edges (a, b) and check if there exists a c
    # such that (a, c) and (b, c) are edges.
    # To ensure lexicographical order, we check 'a' from 1 to n.
    
    best_triad = None
    
    # Optimization: iterate through all edges (u, v) as potential (a, b)
    # Since we need the smallest a, then b, then c, we can iterate
    # through all possible 'a' and 'b' and check for 'c'.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Now we need to find if there is a c > b such that c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can use two pointers or set intersection
            neighbors_b = adj[b]
            
            # Intersection of sorted lists
            ptr_a = 0
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    c = val_a
                    if c > b:
                        # Found a triad (a, b, c)
                        # Since we iterate a from 1, b from smallest, c from smallest,
                        # the first one we find is the lexicographically smallest.
                        print(f"{a} {b} {c}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
