import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) such that a < b < c.
    # To minimize a, then b, then c, we iterate in order.
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Check for common neighbors c such that c > b
            # Since we want to minimize b, we check all b for a fixed a
            # and then find the smallest c.
            # To ensure we find the lexicographically smallest (a, b, c),
            # we should iterate b in increasing order.
            
            # Find intersection of adj[a] and adj[b]
            # Only consider c > b
            common = []
            for c in neighbors_a[i+1:]:
                if c in adj[b]:
                    common.append(c)
            
            if common:
                print(f"{a} {b} {min(common)}")
                return

if __name__ == '__main__':
    solve()
