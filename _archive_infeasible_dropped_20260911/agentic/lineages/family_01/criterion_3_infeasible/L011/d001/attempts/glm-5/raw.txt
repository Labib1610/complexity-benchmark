import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].add(v)
        adj[v].add(u)
    
    # Iterate to find the lexicographically smallest triangle
    for a in range(1, n + 1):
        neighbors = adj[a]
        # We need b and c such that b < c, a < b, and edge (b, c) exists.
        # Since we iterate a in increasing order, the first found is minimal a.
        # To find minimal (b, c) for current a, we iterate b in increasing order.
        # We only need to check neighbors of a that are greater than a.
        # Convert to sorted list to iterate in order
        sorted_neighbors = sorted(neighbors)
        for b in sorted_neighbors:
            if b <= a:
                continue
            
            # Check if b has neighbor c > b that is also neighbor of a
            # We iterate neighbors of b that are > b
            # To get smallest c, we iterate sorted neighbors of b
            # But we can iterate over the smaller set for efficiency
            # Let's iterate sorted neighbors of b
            # Optimization: iterate intersection of adj[b] and adj[a]?
            # adj[b] and adj[a] are sets.
            # We need c in adj[b] intersect adj[a] such that c > b.
            
            # Iterate neighbors of b
            for c in sorted(adj[b]):
                if c <= b:
                    continue
                if c in adj[a]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
