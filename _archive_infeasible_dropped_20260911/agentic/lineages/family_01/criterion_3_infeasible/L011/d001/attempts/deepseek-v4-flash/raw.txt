Here’s the Python solution for the problem:  

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    # Try to find a triangle with smallest lexicographic order
    # We'll iterate over edges and check common neighbors
    best = None
    for u, v in edges:
        if u > v:
            u, v = v, u
        common = adj[u] & adj[v]
        for w in common:
            if w > v:
                cand = (u, v, w)
                if best is None or cand < best:
                    best = cand
        if best is not None and best[0] < u:
            # Since we iterate edges in input order, we can't break early
            # But we can still keep the smallest found
            pass

    # The above may not guarantee smallest lexicographic if we don't iterate smartly.
    # Better approach: iterate over a from 1 to n, then b > a, then c > b
    # But that's O(n^3) worst case. Instead, we can use adjacency sets and check
    # for each a, iterate over b in adj[a] with b > a, then check c in adj[a] & adj[b] with c > b.
    # This is efficient enough because m <= 200000.

    best = None
    for a in range(1, n + 1):
        for b in adj[a]:
            if b <= a:
                continue
            for c in adj[a] & adj[b]:
                if c > b:
                    cand = (a, b, c)
                    if best is None or cand < best:
                        best = cand
        if best is not None and best[0] < a:
            # Since we iterate a increasing, if we already found a triangle with smaller a,
            # we can stop because any later a will be larger.
            break

    print(*best)

if __name__ == "__main__":
    solve()
```
