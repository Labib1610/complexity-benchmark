import sys

def solve():
    # The condition weight(Sa ^ Sb ^ Sc) = weight(Sa) + weight(Sb) + weight(Sc) - 6
    # is equivalent to saying that the number of positions k where 
    # (Sa[k], Sb[k], Sc[k]) is (1, 1, 0), (1, 0, 1), (0, 1, 1), or (1, 1, 1)
    # must be exactly 3.
    # Specifically, let I(u, v, k) be 1 if sequence u has a 1 at position k, else 0.
    # The condition simplifies to: the number of positions k where 
    # at least two of {Sa, Sb, Sc} have a 1 is exactly 3.
    # Since each operation k sets exactly two sequences to 1, 
    # this means there are exactly 3 operations that involve pairs from {a, b, c}.
    # The pairs are (a, b), (a, c), and (b, c).
    # If an operation involves (a, b), it contributes to the count.
    # If an operation involves (a, c), it contributes.
    # If an operation involves (b, c), it contributes.
    # If an operation involves (a, b) and (a, c), that's not possible as each op is one pair.
    # So we need exactly 3 operations among the pairs (a, b), (a, c), (b, c).
    # This means the triangle (a, b, c) must have exactly 3 edges in the graph 
    # where edges are the operations.
    # Wait, the condition is weight(Sa ^ Sb ^ Sc) = weight(Sa) + weight(Sb) + weight(Sc) - 6.
    # Let x_k = Sa[k], y_k = Sb[k], z_k = Sc[k].
    # weight(Sa ^ Sb ^ Sc) = sum_k (x_k ^ y_k ^ z_k)
    # weight(Sa) + weight(Sb) + weight(Sc) = sum_k (x_k + y_k + z_k)
    # The difference is sum_k (x_k + y_k + z_k - (x_k ^ y_k ^ z_k)) = 6.
    # x + y + z - (x ^ y ^ z) is 2 if exactly two are 1, and 4 if all three are 1.
    # Since each operation sets exactly two bits, x_k + y_k + z_k is at most 2.
    # So the difference is 2 * (number of k where exactly two of {Sa, Sb, Sc} are 1).
    # We need this to be 6, so we need exactly 3 positions k where exactly two of {Sa, Sb, Sc} are 1.
    # This happens if and only if the edges (a, b), (b, c), (a, c) all exist in the graph.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a: continue
            # Check for common neighbors c > b
            for j in range(i + 1, len_a):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
