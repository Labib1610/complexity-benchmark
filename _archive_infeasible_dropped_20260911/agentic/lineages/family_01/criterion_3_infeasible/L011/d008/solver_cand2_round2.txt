import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    d = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        d[u] += 1
        d[v] += 1
        
    # Precompute adjacency matrix for O(1) lookup
    adj_mat = [set() for _ in range(n + 1)]
    for i in range(1, n + 1):
        for neighbor in adj[i]:
            adj_mat[i].add(neighbor)
            
    # Let T = {a, b, c}.
    # D(a) + D(b) + D(c) = B(a, b, c) + 6
    # Let E(T) be the number of edges with both endpoints in T.
    # B(a, b, c) = (D(a) - 2*E_in(a)) + (D(b) - 2*E_in(b)) + (D(c) - 2*E_in(c))
    # where E_in(x) is the number of edges connecting x to another node in T.
    # Note: sum(E_in) = 2 * E(T).
    # So B(a, b, c) = D(a) + D(b) + D(c) - 2 * (2 * E(T))
    # B(a, b, c) = D(a) + D(b) + D(c) - 4 * E(T)
    # The condition D(a) + D(b) + D(c) = B(a, b, c) + 6 becomes:
    # D(a) + D(b) + D(c) = D(a) + D(b) + D(c) - 4 * E(T) + 6
    # 4 * E(T) = 6
    # This implies E(T) must be 1.5, which is impossible.
    # Re-reading: "B(a, b, c) is the number of tests that activate exactly one component from the trio"
    # Let E_in(a, b, c) be the number of edges with exactly one endpoint in {a, b, c}.
    # Let E_T be the number of edges with both endpoints in {a, b, c}.
    # D(a) = (edges with a and one of {b, c}) + (edges with a and one outside)
    # D(a) + D(b) + D(c) = (edges with both in T) * 2 + (edges with exactly one in T)
    # D(a) + D(b) + D(c) = 2 * E_T + B(a, b, c)
    # Given D(a) + D(b) + D(c) = B(a, b, c) + 6
    # 2 * E_T + B(a, b, c) = B(a, b, c) + 6 => 2 * E_T = 6 => E_T = 3.
    # So we need to find a trio {a, b, c} that forms a triangle (3 edges).
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj_mat[a]:
                for c in range(b + 1, n + 1):
                    if c in adj_mat[a] and c in adj_mat[b]:
                        print(f"{a} {b} {c}")
                        return

solve()
