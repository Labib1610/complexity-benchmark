```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[u] stores the set of sequences v such that there is an operation (u, v)
    # We use a list of sets to store the neighbors
    adj = [set() for _ in range(n + 1)]
    # weights[i] stores W(S_i)
    weights = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
        weights[u] += 1
        weights[v] += 1
        
    # The condition d(Sa, Sb) + d(Sb, Sc) + d(Sc, Sa) = 2(W(Sa) + W(Sb) + W(Sc)) - 6
    # simplifies based on the fact that each operation adds 1 to exactly two sequences.
    # Let N(u, v) be 1 if there is an operation between u and v, else 0.
    # d(Sa, Sb) = W(Sa) + W(Sb) - 2 * N(a, b)
    # Substituting this:
    # (W(Sa) + W(Sb) - 2*N(a, b)) + (W(Sb) + W(Sc) - 2*N(b, c)) + (W(Sc) + W(Sa) - 2*N(c, a))
    # = 2(W(Sa) + W(Sb) + W(Sc)) - 2(N(a, b) + N(b, c) + N(c, a))
    # The condition becomes:
    # 2(W(Sa) + W(Sb) + W(Sc)) - 2(N(a, b) + N(b, c) + N(c, a)) = 2(W(Sa) + W(Sb) + W(Sc)) - 6
    # Which simplifies to:
    # N(a, b) + N(b, c) + N(c, a) = 3
    # This means a, b, and c must form a triangle in the graph where edges are operations.
    
    # Iterate to find the lexicographically smallest triangle (a, b, c)
    for a in range(1, n + 1):
        adj_a = adj[a]
        sorted_adj_a = sorted(list(adj_a))
        for i in range(len(sorted_adj_a)):
            b = sorted_adj_a[i]
            if b <= a:
                continue
            # Check for c > b such that (a, c) and (b, c) are edges
            adj_b = adj[b]
            # Intersection of adj_a and adj_b gives potential c's
            common = sorted(list(adj_a.intersection(adj_b)))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
