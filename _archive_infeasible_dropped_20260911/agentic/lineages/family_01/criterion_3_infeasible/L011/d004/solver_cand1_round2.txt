import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # x^T W x = 6 means sum_{i,j} W_{i,j} x_i x_j = 6.
    # Since x has three 1s at indices a, b, c, the sum is:
    # W_{a,b} + W_{b,a} + W_{a,c} + W_{c,a} + W_{b,c} + W_{c,b} = 6
    # Since W_{i,j} is 1 if connected and 0 otherwise, this sum is 2 * (W_{a,b} + W_{a,c} + W_{b,c})
    # So we need W_{a,b} + W_{a,c} + W_{b,c} = 3.
    # This means a, b, and c must form a triangle in the graph.
    
    # We need the lexicographically smallest (a, b, c) such that a < b < c
    # and a, b, c form a triangle.
    
    for a in range(1, n + 1):
        adj_a = adj[a]
        sorted_adj_a = sorted(list(adj_a))
        len_adj_a = len(sorted_adj_a)
        
        for i in range(len_adj_a):
            b = sorted_adj_a[i]
            if b <= a:
                continue
            
            # Check for c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of adj[a] and adj[b]
            adj_b = adj[b]
            
            # To optimize, iterate over the smaller adjacency list
            if len(adj_a) < len(adj_b):
                for c in sorted_adj_a:
                    if c > b and c in adj_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                sorted_adj_b = sorted(list(adj_b))
                for c in sorted_adj_b:
                    if c > b and c in adj_a:
                        print(f"{a} {b} {c}")
                        return

solve()
