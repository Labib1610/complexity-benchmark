import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # A complete tripartite resonance is a triangle (a, b, c)
    # where the number of edges between them is exactly 3.
    # This is exactly the definition of a triangle in the graph.
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all edges (u, v) and check for a common neighbor w.
    # Since we need a < b < c, we ensure this order.
    
    results = []
    
    # Sort edges to help find the lexicographically smallest triangle
    edges.sort()
    
    # To optimize, iterate over edges (u, v) and check common neighbors
    # Use the smaller degree node to iterate over neighbors
    for u, v in edges:
        # Triangle (u, v, w)
        # We need u < v < w or other permutations, but we want the sorted triplet
        # Let's find all w such that (u, w) and (v, w) exist
        
        # Optimization: iterate over the smaller adjacency list
        if len(adj[u]) < len(adj[v]):
            common = adj[u].intersection(adj[v])
        else:
            common = adj[v].intersection(adj[u])
            
        for w in common:
            triplet = sorted([u, v, w])
            results.append(tuple(triplet))
            
    # The problem guarantees at least one exists.
    # We need the lexicographically smallest (a, b, c).
    # Since we want the smallest a, then b, then c, we can just sort the results.
    
    # To avoid storing all triangles if there are many, we can use a min-heap
    # or just sort the unique ones.
    unique_results = sorted(list(set(results)))
    
    if unique_results:
        a, b, c = unique_results[0]
        print(f"{a} {b} {c}")

solve()
