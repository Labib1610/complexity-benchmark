```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To minimize S = x * 25000000 + y * 5000 + z,
    # we iterate x from 1 to n, y from x+1 to n, and z from y+1 to n.
    # A configuration (x, y, z) has score 3 if x-y, y-z, and z-x are all edges.
    
    # Optimization: iterate x, then iterate y in adj[x] such that y > x.
    # Then check if there exists z in adj[x] intersect adj[y] such that z > y.
    
    best_s = float('inf')
    best_triple = None
    
    # Sort adjacency lists to help with ordering
    sorted_adj = [sorted(list(s)) for s in adj]
    
    for x in range(1, n + 1):
        # Optimization: if x is already too large, we can't beat best_s
        if x * 25000000 > best_s:
            break
            
        for y in sorted_adj[x]:
            if y <= x:
                continue
            
            # Current signature part
            s_xy = x * 25000000 + y * 5000
            if s_xy > best_s:
                continue
                
            # Find common neighbors z > y
            # Using set intersection for efficiency
            common = adj[x].intersection(adj[y])
            for z in common:
                if z > y:
                    s = s_xy + z
                    if s < best_s:
                        best_s = s
                        best_triple = (x, y, z)
                        
    if best_triple:
        print(f"{best_triple[0]} {best_triple[1]} {best_triple[2]}")

solve()
```
