import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        idx += 2
        
    # We need to find a, b, c such that a < b < c
    # and (a,b), (a,c), (b,c) are in the log.
    # To satisfy the smallest a, then b, then c requirement:
    
    found = None
    
    # Iterate through possible a
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted(list(adj[a]))
        
        # For each pair of neighbors b, c of a, check if (b, c) is an edge
        # To optimize, iterate b in neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Check if there exists c in neighbors_a such that c > b and c is in adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    # Found a triangle (a, b, c)
                    # Since we iterate a from 1..n, and b, c are sorted,
                    # the first one we find might not be the lexicographically smallest.
                    # Actually, if we iterate a, then b, then c in increasing order,
                    # the first one we find is guaranteed to be the smallest.
                    print(f"{a} {b} {c}")
                    return

solve()
