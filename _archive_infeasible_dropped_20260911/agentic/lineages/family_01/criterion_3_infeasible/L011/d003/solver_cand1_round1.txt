```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) such that a < b < c.
    # To satisfy the smallest lexicographical requirement, we iterate
    # through a from 1 to n, then b from a+1 to n, then check if c exists.
    
    # To optimize, we only check b that are neighbors of a,
    # and then check if there is a common neighbor c > b.
    
    best = None
    
    for a in range(1, n + 1):
        # Optimization: if we already found a solution with a smaller 'a',
        # we can stop.
        if best and a > best[0]:
            break
            
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            if best and (a > best[0] or (a == best[0] and b >= best[1])):
                continue
                
            # Find common neighbors c such that c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    if best is None or (a < best[0]) or (a == best[0] and b < best[1]) or (a == best[0] and b == best[1] and c < best[2]):
                        best = (a, b, c)
    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
