import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that a < b < c
    # and (a,b), (a,c), (b,c) are all in the adjacency list.
    # To ensure we find the lexicographically smallest (a, b, c),
    # we iterate through a, then b, then c.
    
    # Sort neighbors to help with lexicographical order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Check if there exists c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we need smallest a, b, c, we look for the smallest c > b
            # that is in both adj[a] and adj[b].
            
            # Optimization: use set intersection for faster lookup
            # But we need the smallest c > b.
            # Since neighbors_a is sorted, we can iterate through it.
            
            set_b = adj[b]
            for c in neighbors_a:
                if c > b and c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
