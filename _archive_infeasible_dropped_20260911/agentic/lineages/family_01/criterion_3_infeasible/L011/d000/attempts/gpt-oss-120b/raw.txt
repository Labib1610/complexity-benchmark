import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)          # 1‑based, bit i‑1 corresponds to vertex i
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        # set bits for both directions
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    full_mask = (1 << n) - 1

    for a in range(1, n - 1):
        a_mask = adj[a]
        for b in range(a + 1, n):
            # edge a‑b must exist
            if not (a_mask >> (b - 1)) & 1:
                continue
            # common neighbours of a and b
            common = a_mask & adj[b]
            # keep only vertices > b
            mask = full_mask ^ ((1 << b) - 1)   # bits with index >= b (0‑based) i.e. vertices > b
            common &= mask
            if common:
                # lowest set bit gives smallest c
                low = common & -common
                c = low.bit_length()            # because bits are 0‑based, vertex = index+1 = bit_length
                sys.stdout.write(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
