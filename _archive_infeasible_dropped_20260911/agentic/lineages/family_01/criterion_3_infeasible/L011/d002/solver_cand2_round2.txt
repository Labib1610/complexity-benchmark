```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all edges (u, v) and check for a common neighbor w.
    # Since we need the smallest a, then b, then c, we sort the edges
    # and check neighbors.
    
    # Sort edges to help find the smallest (a, b, c)
    edges.sort()
    
    best = None
    
    # For each edge (u, v), find common neighbors w
    # To ensure we find the lexicographically smallest (a, b, c),
    # we can iterate through all possible triples or optimize.
    # Given n=2000, O(m * sqrt(m)) or O(n * m) is acceptable.
    
    for u in range(1, n + 1):
        neighbors = sorted(list(adj[u]))
        len_n = len(neighbors)
        for i in range(len_n):
            for j in range(i + 1, len_n):
                v = neighbors[i]
                w = neighbors[j]
                # Check if (v, w) is an edge
                if w in adj[v]:
                    # Found a triangle (u, v, w)
                    # Sort them to get (a, b, c)
                    triple = sorted([u, v, w])
                    if best is None or triple < best:
                        best = triple
        # Optimization: if we have found a triple starting with a smaller 'a',
        # we can potentially stop, but since we need the absolute smallest,
        # we continue or break carefully.
        if best is not None and u > best[0]:
            break
            
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
