```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # A complete deadlock is a triangle (a, b, c) such that 
    # (a, b), (b, c), and (a, c) are all in the log.
    # We need to find all such triangles and pick the lexicographically smallest.
    
    found_triangles = []
    
    # To find triangles efficiently:
    # Iterate over each edge (u, v) and check common neighbors
    # To avoid O(N^3), use the property that we only need to check neighbors
    # of the vertex with the smaller degree.
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    for u, v in edges:
        # Find common neighbors of u and v
        # To optimize, iterate over the smaller adjacency list
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        
        for neighbor in adj[u]:
            if neighbor in adj[v]:
                # Found a triangle (u, v, neighbor)
                a, b, c = sorted([u, v, neighbor])
                found_triangles.append((a, b, c))
                
    # The problem guarantees at least one exists.
    # Sort the found triangles to get the lexicographically smallest.
    found_triangles.sort()
    
    res = found_triangles[0]
    print(f"{res[0]} {res[1]} {res[2]}")

if __name__ == '__main__':
    solve()
```
