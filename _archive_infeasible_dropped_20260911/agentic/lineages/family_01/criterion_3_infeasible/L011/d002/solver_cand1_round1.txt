```python
import sys

def solve():
    # A complete deadlock of three employees is an odd cycle of length 3.
    # This is simply a triangle in the graph of incompatibilities.
    # We need to find a triangle (a, b, c) with a < b < c.
    # We want the lexicographically smallest (a, b, c).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c), we iterate through
    # possible values of a, then b, then c.
    # A triangle (a, b, c) exists if a < b < c, (a, b) in E, (a, c) in E, (b, c) in E.
    
    # Sort adjacency lists to help with finding c efficiently
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    best = None
    
    # Iterate through all edges (a, b) as the first two nodes of the triangle
    # Since we want smallest (a, b, c), we iterate a from 1 to n, b from a+1 to n
    for a in range(1, n + 1):
        # Optimization: if we already found a triangle with a smaller 'a', we can stop
        if best and a > best[0]:
            break
            
        for b in adj[a]:
            if b <= a:
                continue
            
            # Now find c such that (a, c) in E and (b, c) in E
            # c must be > b
            # We can intersect the sorted adjacency lists of a and b
            # to find common neighbors c > b
            
            # Using two pointers on sorted lists
            ptr_a = 0
            ptr_b = 0
            list_a = adj[a]
            list_b = adj[b]
            
            while ptr_a < len(list_a) and ptr_b < len(list_b):
                val_a = list_a[ptr_a]
                val_b = list_b[ptr_b]
                
                if val_a == val_b:
                    c = val_a
                    if c > b:
                        if best is None or (a, b, c) < best:
                            best = (a, b, c)
                        # Since we want the smallest c for this (a, b),
                        # and we are iterating b in increasing order,
                        # we can potentially break here if we find the smallest c
                        break
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1
                    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
