import sys
from collections import defaultdict

def find_complete_deadlock():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list representation
    adj = defaultdict(set)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # For each edge (u,v), check if there exists a node w such that 
    # (u,w) and (v,w) are also edges
    found = False
    result = None
    
    # Sort edges to ensure we process in lexicographical order
    edges.sort()
    
    for u, v in edges:
        if found:
            break
        # Check common neighbors of u and v
        common_neighbors = adj[u] & adj[v]
        for w in sorted(common_neighbors):
            # Found triangle u, v, w
            triangle = tuple(sorted([u, v, w]))
            if not result or triangle < result:
                result = triangle
                found = True
                break
    
    print(result[0], result[1], result[2])

find_complete_deadlock()
