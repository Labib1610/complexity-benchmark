import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    D = [len(adj[i]) for i in range(n + 1)]
    
    # Resonance Level = sum_{k=1}^m 3^{S_k}
    # S_k is the number of devices from {a, b, c} in transmission k.
    # S_k can be 0, 1, or 2 (since a transmission involves 2 devices).
    # 3^0 = 1, 3^1 = 3, 3^2 = 9.
    # Resonance = sum_{k=1}^m (1 + 2 * S_k + 6 * (S_k == 2))
    # Resonance = m + 2 * sum(S_k) + 6 * (number of edges between {a, b, c})
    # sum(S_k) = D_a + D_b + D_c
    # Resonance = m + 2(D_a + D_b + D_c) + 6 * (number of edges between {a, b, c})
    # We want Resonance = m + 2(D_a + D_b + D_c) + 12
    # This implies 6 * (number of edges between {a, b, c}) = 12
    # So, number of edges between {a, b, c} must be 2.
    
    # A trio {a, b, c} has 2 edges if and only if they form a path of length 2.
    # This means two of them are connected to the same middle node.
    
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for i in range(1, n + 1):
        neighbors = sorted(adj[i])
        len_n = len(neighbors)
        if len_n < 2:
            continue
        for j_idx in range(len_n):
            for k_idx in range(j_idx + 1, len_n):
                u = neighbors[j_idx]
                v = neighbors[k_idx]
                # Trio is {u, i, v}. Sorted:
                trio = sorted([u, i, v])
                a, b, c = trio
                
                # Check if number of edges is 2.
                # Edges are (u, i), (v, i), and possibly (u, v).
                # If (u, v) is an edge, edges = 3. If not, edges = 2.
                if v not in adj_sets[u]:
                    print(f"{a} {b} {c}")
                    return

solve()
