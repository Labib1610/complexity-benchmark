```python
import sys

def solve():
    # The problem asks for a triplet (a, b, c) such that the 3x3 system
    # formed by equations a, b, c restricted to variables x_a, x_b, x_c
    # has a unique solution.
    # Let A_i be the set of variables appearing in equation i.
    # The equation i restricted to {a, b, c} is:
    # sum_{v in A_i intersect {a, b, c}} x_v = 0
    # The system is:
    # Eq a: (a in A_a ? x_a : 0) + (b in A_a ? x_b : 0) + (c in A_a ? x_c : 0) = 0
    # Eq b: (a in A_b ? x_a : 0) + (b in A_b ? x_b : 0) + (c in A_b ? x_c : 0) = 0
    # Eq c: (a in A_c ? x_a : 0) + (b in A_c ? x_b : 0) + (c in A_c ? x_c : 0) = 0
    # This is a system M * [x_a, x_b, x_c]^T = 0.
    # Unique solution iff det(M) != 0.
    # M_ij = 1 if j in A_i, else 0.
    # Note: The update (u, v) adds x_v to Eq u and x_u to Eq v.
    # So u in A_v and v in A_u.
    # M_uv = 1 iff there is an update between u and v.
    # M_uu = 0 because no update (u, u) exists.
    # So M is the adjacency matrix of the graph formed by the updates.
    # M_ij = 1 if i != j and (i, j) is an edge, 0 otherwise.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # det(M) for a 3x3 matrix where M_ii = 0:
    # M = [[0, M12, M13], [M21, 0, M23], [M31, M32, 0]]
    # det = 0*(...) - M12*(M21*0 - M23*M31) + M13*(M21*M32 - 0*M31)
    # det = M12*M23*M31 + M13*M21*M32
    # Since M_ij = M_ji (undirected graph), det = 2 * M12 * M23 * M31
    # det != 0 iff M12=1, M23=1, M31=1.
    # This means a, b, c must form a triangle in the graph.
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # Find common neighbors c > b
                common = adj[a].intersection(adj[b])
                for c in sorted(common):
                    if c > b:
                        print(f"{a} {b} {c}")
                        return

solve()
```
