```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n, m = int(data[0]), int(data[1])
    edges = []
    adj = [set() for _ in range(n+1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Convert sets to lists for faster iteration
    adj_list = [list(adj[i]) for i in range(n+1)]
    deg = [len(adj[i]) for i in range(n+1)]
    
    # Build neighbor sets for fast intersection
    neigh_sets = [set(adj[i]) for i in range(n+1)]
    
    # We want to find a,b,c such that |T_a ∪ T_b ∪ T_c| = |T_a| + |T_b| + |T_c| - 3
    # This is equivalent to saying that the three sets T_a, T_b, T_c are pairwise disjoint except for the tasks they share within the trio.
    # Actually, let's think differently:
    # |T_a ∪ T_b ∪ T_c| = |T_a| + |T_b| + |T_c| - |T_a ∩ T_b| - |T_a ∩ T_c| - |T_b ∩ T_c| + |T_a ∩ T_b ∩ T_c|
    # So the condition becomes:
    # |T_a| + |T_b| + |T_c| - |T_a ∩ T_b| - |T_a ∩ T_c| - |T_b ∩ T_c| + |T_a ∩ T_b ∩ T_c| = |T_a| + |T_b| + |T_c| - 3
    # Simplifying:
    # - |T_a ∩ T_b| - |T_a ∩ T_c| - |T_b ∩ T_c| + |T_a ∩ T_b ∩ T_c| = -3
    # Rearranging:
    # |T_a ∩ T_b| + |T_a ∩ T_c| + |T_b ∩ T_c| - |T_a ∩ T_b ∩ T_c| = 3
    
    # But since each task connects exactly two employees, |T_a ∩ T_b ∩ T_c| = 0.
    # So the condition simplifies to:
    # |T_a ∩ T_b| + |T_a ∩ T_c| + |T_b ∩ T_c| = 3
    
    # That means the total number of tasks shared between pairs in the trio is 3.
    # Since each task connects exactly two employees, this means there are exactly 3 tasks among the trio,
    # and these 3 tasks cover all three pairs (a,b), (a,c), (b,c).
    # In other words, the subgraph induced by {a,b,c} has exactly 3 edges forming a triangle.
    
    # So we need to find a triangle in the graph.
    
    # Create edge existence map for O(1) check
    edge_exists = [[False] * (n+1) for _ in range(n+1)]
    for u,v in edges:
        edge_exists[u][v] = True
        edge_exists[v][u] = True
    
    # Iterate through all triangles
    best = None
    
    # To find a triangle efficiently, we can iterate over all edges and for each edge (u,v),
    # look for a common neighbor w of u and v.
    found = False
    for u in range(1, n+1):
        if found:
            break
        for v in adj_list[u]:
            if v <= u:
                continue
            if found:
                break
            # Look for common neighbors of u and v
            if deg[u] > deg[v]:
                u, v = v, u
            for w in adj_list[u]:
                if w == v or w <= u:
                    continue
                if edge_exists[v][w]:
                    # Found triangle u-v-w
                    trio = tuple(sorted([u, v, w]))
                    if best is None or trio < best:
                        best = trio
                    found = True
                    break
    
    print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
```
