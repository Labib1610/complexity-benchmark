import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    deg = [0] * (n + 1)
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        deg[u] += 1
        deg[v] += 1
    
    # Sort employees by degree for the low-degree optimization
    order = sorted(range(1, n + 1), key=lambda x: deg[x])
    
    # Iterate over all pairs (a, b) with a < b
    # For each pair, check if they are connected (common task)
    # Then look for c > b such that (a, c) and (b, c) are connected
    # and deg[a] + deg[b] + deg[c] - 3 == |union of tasks|
    # The union condition means exactly 3 tasks are shared among the three pairs.
    # Since each pair has exactly one task, the three tasks are:
    # task(a,b), task(a,c), task(b,c).
    # The union size is deg[a] + deg[b] + deg[c] - (shared tasks counted multiple times).
    # For the union to be deg[a] + deg[b] + deg[c] - 3, the three tasks must be distinct
    # and there are no other shared tasks between any two of the three employees.
    # Wait, let's re-evaluate:
    # |A U B U C| = |A| + |B| + |C| - |A∩B| - |A∩C| - |B∩C| + |A∩B∩C|
    # We want this to be |A| + |B| + |C| - 3.
    # So |A∩B| + |A∩C| + |B∩C| - |A∩B∩C| = 3.
    # Since each pair of employees shares at least 1 task (if they are connected),
    # and tasks are between exactly 2 employees, |A∩B∩C| = 0.
    # So we need |A∩B| + |A∩C| + |B∩C| = 3.
    # Since each pair in the trio must be connected (share at least 1 task),
    # we must have |A∩B| = 1, |A∩C| = 1, |B∩C| = 1.
    # This means the three employees form a triangle in the task graph,
    # and there are no multiple edges between any two of them (which is guaranteed).
    # So we just need to find a triangle in the graph!
    
    # To find a triangle efficiently:
    # For each edge (u, v), we check for common neighbors w.
    # To avoid checking all pairs, we orient edges from lower degree to higher degree
    # (or lower index if degrees are equal). This is the "forward" direction.
    # Then for each node u, we look at its forward neighbors.
    # For each pair of forward neighbors (v, w), we check if (v, w) is an edge.
    # The complexity is O(m * sqrt(m)) for triangle finding.
    
    # But we need the lexicographically smallest triangle.
    # So we can iterate a from 1 to n, then b from a+1 to n, then c from b+1 to n.
    # But this is O(n^3) which is too slow for n=2000.
    
    # Alternative: iterate a from 1 to n.
    # For a fixed a, we want to find b, c > a such that (a,b), (a,c), (b,c) are edges.
    # We can iterate b from a+1 to n. If (a,b) is an edge, we look for c > b such that

    # To do this efficiently, we can precompute for each a, the set of neighbors of a
    # that are > a. Then for each b > a with (a,b) an edge, we intersect the neighbors of b
    # that are > b with the neighbors of a that are > b.
    # But intersection of sets can be slow.
    
    # Let's use the degree orientation.
    # For a fixed a, we want to find b, c > a.
    # We can just iterate b from a+1 to n. If (a,b) is an edge,
    # we iterate c from b+1 to n. If (a,c) and (b,c) are edges, we found a triangle.
    # To make the inner loop fast, we can represent the adjacency list as a bitset
    # or use the fact that we can check (b,c) in O(1) with a hash set.
    # But iterating c from b+1 to n is O(n) per b, so O(n^2) per a, O(n^3) total.

    # Since we want the lexicographically smallest, we process a from 1 to n.

    # For a from 1 to n:
    #   For b from a+1 to n:
    #     if b in adj[a]:
    #       # We need c > b such that c in adj[a] and c in adj[b]
    #       # We want the smallest such c.
    #       # So we can iterate over the smaller of adj[a] and adj[b] (for elements > b)
    #       # and find the minimum.
    #       # But we need to do this for all b, and we want the first b that has any such c.
    #       # Actually, we want the lexicographically smallest (a, b, c).
    #       # So for a fixed a, we want the smallest b, and for that b, the smallest c.
    #       # So we can just iterate b from a+1 to n, and for the first b that is in adj[a]
    #       # and has a common neighbor c > b with a, we take the smallest such c and return.
    #       # But wait, what if a smaller b has no c, but a larger b has a smaller c?
    #       # No, because we want lexicographically smallest, so (a, b1, c1) < (a, b2, c2) if b1 < b2,

    #       # and for that b, the smallest c.
    #       # So we can break early once we find the first valid b and its smallest c.

    #   For b from a+1 to n:
    #     if b in adj[a]:
    #       # find smallest c > b in adj[a] ∩ adj[b]

    # The complexity: for each a, we iterate b. For each b in adj[a], we do an intersection.
    # The total work is sum over a of sum over b in adj[a] of min(deg(a), deg(b)).
    # This is O(m * sqrt(m)) in the worst case if we use the degree orientation,
    # but here we are iterating b in order, not by degree.
    # However, since we break early, it might be fast enough.

    # Let's think about the worst case.
    # If the graph is a complete bipartite graph plus one triangle at the end,

    #   # To do this, we can iterate b in sorted(neighbors_a):
    #   #   for c in sorted(neighbors_a):
    #   #     if c > b and c in adj[b]:
    #   #       return a, b, c
    #   # But this is O(deg(a)^2) per a, which could be O(n^2) per a, O(n^3) total.
    #   # We need to be faster.

    #   #   we want the smallest c in neighbors_a with c > b and c in adj[b].
    #   #   We can iterate over the smaller of neighbors_a and adj[b] (restricted to > b).
    #   #   But neighbors_a is sorted, so we can binary search for the first element > b,
    #   #   then iterate from there and check if in adj[b].
    #   #   But checking in adj[b] is O(1), so the cost is O(number of elements in neighbors_a > b).
    #   #   In the worst case, this is O(deg(a)) per b, so O(deg(a)^2) per a.
    #   #   If deg(a) is large, this is bad.

    #   # We want the minimum such c.
    #   # So for each b in neighbors_a:
    #   #   for c in adj[b]:
    #   #     if c > b and c in adj[a]:
    #   #       keep track of min c.

    #   # The cost is sum over b in neighbors_a of deg(b).
    #   # This could be large if deg(b) is large.
    
    #   # We can choose the smaller of the two: for each b, either iterate over neighbors_a > b or adj[b].
    #   # But we need to find the minimum c, so we can't just check existence.
    
    #   # Let's think differently: we want the lexicographically smallest triangle.
    #   # So we process a from 1 to n. For each a, we want the smallest b > a such that

    #   # So we can iterate b from a+1 to n. If b in adj[a], we compute the intersection.
    #   # To compute the smallest c > b in adj[a] ∩ adj[b] efficiently:
    #   # We can iterate over the smaller of adj[a] and adj[b], and filter for > b.

    #   # but for membership testing, set is faster.
    #   # We can keep both: a set and a sorted list for each adjacency list.

    #   # In a dense graph, deg(a) can be up to 1999.
    #   # For a=1, we might iterate b from 2 to 2000. For each b in adj[1], we do an intersection.

    #   # The first b that yields a valid c gives us the answer (with the smallest c for that b).
    #   # So we can break early.
    
    #   # Let's implement this.
    
    for a in range(1, n + 1):
        # We want to iterate b in increasing order
        # So we get the sorted list of neighbors of a that are > a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for b in neighbors_a:
            # find smallest c > b in adj[a] ∩ adj[b]
            set_a = adj[a]
            set_b = adj[b]
            if len(set_a) > len(set_b):
                smaller = set_b
                larger = set_a
            else:
                smaller = set_a
                larger = set_b
            min_c = n + 1
            for c in smaller:
                if c > b and c in larger:
                    if c < min_c:
                        min_c = c
            if min_c <= n:
                print(a, b, min_c)
