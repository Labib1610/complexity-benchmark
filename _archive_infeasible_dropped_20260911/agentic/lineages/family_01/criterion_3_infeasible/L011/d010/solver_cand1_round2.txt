import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[u] stores the set of neighbors of u
    # Since we need to check if a task exists between two employees,
    # an adjacency matrix or a set of sets is useful.
    # Given n <= 2000, an adjacency matrix of bits or a list of sets is fine.
    adj = [set() for _ in range(n + 1)]
    degree = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        degree[u] += 1
        degree[v] += 1
        idx += 2
        
    # The condition |Ta U Tb U Tc| = |Ta| + |Tb| + |Tc| - 3
    # Let E(a, b) be 1 if there is a task between a and b, else 0.
    # |Ta U Tb U Tc| = |Ta| + |Tb| + |Tc| - (|Ta n Tb| + |Ta n Tc| + |Tb n Tc|) + |Ta n Tb n Tc|
    # Since each task is between exactly two employees, |Ta n Tb| is 1 if there is a task, 0 otherwise.
    # |Ta n Tb n Tc| is always 0 because a task involves only 2 people.
    # So the condition simplifies to:
    # E(a, b) + E(a, c) + E(b, c) = 3
    # This means a, b, and c must form a triangle in the graph.
    
    # We need the lexicographically smallest (a, b, c) with a < b < c.
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Iterate b from a+1 to n-1
        for b in range(a + 1, n):
            if b in adj[a]:
                # Check for c > b such that c is connected to both a and b
                # Intersection of adj[a] and adj[b]
                common = adj[a].intersection(adj[b])
                # Filter for c > b
                valid_c = [c for c in common if c > b]
                if valid_c:
                    print(f"{a} {b} {min(valid_c)}")
                    return

if __name__ == '__main__':
    solve()
