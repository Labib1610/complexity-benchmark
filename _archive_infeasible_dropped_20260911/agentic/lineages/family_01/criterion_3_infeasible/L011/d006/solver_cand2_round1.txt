import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Dot product S_i . S_j = sum_{k=1}^m S_i[k] * S_j[k]
    # For operation k with indices u, v:
    # S_u[k] = k*(u-v), S_v[k] = k*(v-u)
    # If operation k involves {u, v}, then S_u[k]*S_v[k] = -k^2 * (u-v)^2
    # If operation k involves {u, w} where w != v, S_u[k] = k*(u-w), S_v[k] = 0
    # If operation k involves {v, w} where w != u, S_v[k] = k*(v-w), S_u[k] = 0
    # If operation k involves {x, y} where x, y not in {u, v}, S_u[k] = 0, S_v[k] = 0
    
    # Dot product S_i . S_j = sum_{k: {i,j} is op k} (-k^2 * (i-j)^2)
    # Let W_{ij} = sum_{k: {i,j} is op k} k^2. Then S_i . S_j = -W_{ij} * (i-j)^2.
    # Since W_{ij} > 0 and (i-j)^2 > 0, S_i . S_j is always negative if they share an operation.
    # If they don't share an operation, S_i . S_j = 0.
    
    # We need S_a . S_b < 0, S_a . S_c < 0, S_b . S_c < 0.
    # This means {a, b}, {a, c}, and {b, c} must all be operations in the input.
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v: u, v = v, u
        adj[u].add(v)
        
    # Find lexicographically smallest a, b, c such that b in adj[a], c in adj[a], c in adj[b]
    # (assuming u < v for all edges in adj)
    for a in range(1, n + 1):
        sorted_neighbors_a = sorted(list(adj[a]))
        for i in range(len(sorted_neighbors_a)):
            b = sorted_neighbors_a[i]
            # We need c > b such that c in adj[a] and c in adj[b]
            # Intersection of adj[a] and adj[b]
            # Since we need c > b, we look at neighbors of b that are > b
            # and also in adj[a]
            for c in sorted_neighbors_a[i+1:]:
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
