import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # We need to find a, b, c such that a < b < c and
    # the number of operations involving exactly two of {a, b, c} is exactly 3.
    # This is equivalent to finding a triangle in the graph of operations.
    
    # We use a 'forward' adjacency list representation to facilitate finding
    # the lexicographically smallest triangle efficiently.
    # adj[u] contains a list of vertices v such that u < v and there is an edge (u, v).
    
    adj = [[] for _ in range(n + 1)]
    
    # We also need a fast lookup to check if an edge exists between two nodes.
    # Since n is up to 2000, an adjacency matrix is feasible.
    exists = [[False] * (n + 1) for _ in range(n + 1)]
    
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
        except StopIteration:
            break
        
        if u == v:
            continue
            
        # Populate adjacency matrix
        exists[u][v] = True
        exists[v][u] = True
        
        # Populate forward adjacency list for ordered traversal
        mn = min(u, v)
        mx = max(u, v)
        adj[mn].append(mx)
    
    # To ensure we find the lexicographically smallest triangle (a, b, c),
    # we iterate 'a' from 1 to n, then 'b' from a+1 to n, then 'c' from b+1 to n.
    # We iterate neighbors of 'a' to find 'b', and neighbors of 'b' to find 'c'.
    # To guarantee we check candidates in increasing order, we must sort the adjacency lists.
    for i in range(1, n + 1):
        adj[i].sort()
        
    # Search for the triangle
    for a in range(1, n + 1):
        for b in adj[a]:
            # We have edge (a, b). Now we search for c > b such that (a, c) and (b, c) exist.
            # Since adj[a] and adj[b] are sorted, we can iterate through potential 'c' values.
            # The potential 'c' must be a common neighbor of 'a' and 'b' that is greater than 'b'.
            
            # Iterate through neighbors of 'b' to find 'c'
            for c in adj[b]:
                if c <= b:
                    continue
                
                # Check if 'a' is connected to 'c'
                # Using the adjacency matrix for O(1) check
                if exists[a][c]:
                    # Found the lexicographically smallest valid triplet
                    print(a, b, c)
                    return

if __name__ == '__main__':
    solve()
