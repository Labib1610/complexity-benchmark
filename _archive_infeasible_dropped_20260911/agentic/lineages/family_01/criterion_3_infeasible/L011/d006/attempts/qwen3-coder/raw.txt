import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    operations = []
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        if u > v:
            u, v = v, u
        operations.append((u, v))
        index += 2
    
    # For each sequence, maintain a set of operations (by index) where it is involved
    seq_ops = [set() for _ in range(n)]
    for i, (u, v) in enumerate(operations):
        seq_ops[u].add(i)
        seq_ops[v].add(i)
    
    # For each pair of sequences, store the operations where both are involved
    shared_ops = [[set() for _ in range(n)] for _ in range(n)]
    for i, (u, v) in enumerate(operations):
        shared_ops[u][v].add(i)
        shared_ops[v][u].add(i)
    
    # Function to compute the score for a triplet (a,b,c)
    def compute_score(a, b, c):
        if a >= b or b >= c or a >= c:
            return -1
        Sa_ops = seq_ops[a]
        Sb_ops = seq_ops[b]
        Sc_ops = seq_ops[c]
        
        # Operations where exactly one of a,b,c is involved
        only_a = Sa_ops - Sb_ops - Sc_ops
        only_b = Sb_ops - Sa_ops - Sc_ops
        only_c = Sc_ops - Sa_ops - Sb_ops
        
        # Operations where exactly two of a,b,c are involved
        only_ab = shared_ops[a][b] - Sc_ops
        only_ac = shared_ops[a][c] - Sb_ops
        only_bc = shared_ops[b][c] - Sa_ops
        
        # Operations where all three a,b,c are involved - impossible since each op involves exactly 2
        # So we don't need to consider this case
        
        # Count positions where |Sa[k] + Sb[k] + Sc[k]| != |Sa[k]| + |Sb[k]| + |Sc[k]|
        # This happens when not all non-zero terms have the same sign
        # Since each sequence has entries in {-1, 0, 1}, let's analyze:
        # In positions involving only one sequence: contribution is 1, so |sum| = sum of abs
        # In positions involving exactly two sequences: 
        #     if signs are same: |sum|=sum of abs
        #     if signs differ: |sum| < sum of abs
        # In positions involving all three sequences: impossible
        
        # Let's count violations:
        violations = 0
        
        # Check operations involving exactly a and b (but not c)
        for k in only_ab:
            u_k, v_k = operations[k]
            # Determine which is min and max to know who gets 1 and -1
            if u_k == a:
                val_a = 1
                val_b = -1
            else:
                val_a = -1
                val_b = 1
            val_c = 0
            total_abs = abs(val_a) + abs(val_b) + abs(val_c)  # Always 2
            abs_sum = abs(val_a + val_b + val_c)
            if abs_sum != total_abs:
                violations += 1
                
        # Check operations involving exactly a and c (but not b)
        for k in only_ac:
            u_k, v_k = operations[k]
            if u_k == a:
                val_a = 1
                val_c = -1
            else:
                val_a = -1
                val_c = 1
            val_b = 0
            total_abs = abs(val_a) + abs(val_b) + abs(val_c)  # Always 2
            abs_sum = abs(val_a + val_b + val_c)
            if abs_sum != total_abs:
                violations += 1
                
        # Check operations involving exactly b and c (but not a)
        for k in only_bc:
            u_k, v_k = operations[k]
            if u_k == b:
                val_b = 1
                val_c = -1
            else:
                val_b = -1
                val_c = 1
            val_a = 0
            total_abs = abs(val_a) + abs(val_b) + abs(val_c)  # Always 2
            abs_sum = abs(val_a + val_b + val_c)
            if abs_sum != total_abs:
                violations += 1
                
        # Operations involving only one of them contribute nothing to violation
        # Because |1| = |-1| = 1, and others are 0, so |sum| = sum of abs
        
        # We want:
        # sum_k |Sa[k] + Sb[k] + Sc[k]| = sum_k (|Sa[k]| + |Sb[k]| + |Sc[k]|) - 6
        # sum_k |sum| = sum_k abs_sum - 6
        # This means total_abs_sum - sum_of_abs = 6
        # But total_abs_sum = sum_k |sum| and sum_of_abs = sum_k (|Sa[k]| + |Sb[k]| + |Sc[k]|)  
        # Difference = sum_k (|Sa[k]| + |Sb[k]| + |Sc[k]| - |Sa[k] + Sb[k] + Sc[k]|) = violations * 2?
        # Wait, let's re-examine:
        # Each operation contributes some |Sa[k]| + |Sb[k]| + |Sc[k]| and |Sa[k] + Sb[k] + Sc[k]|
        # Difference per operation is (|Sa[k]| + |Sb[k]| + |Sc[k]|) - |Sa[k] + Sb[k] + Sc[k]|
        # We want the total difference to be 6.
        
        # Let's compute correctly:
        total_diff = 0
        # Operations with only a
        for k in only_a:
            val_a = 1 if operations[k][0] == a else -1
            val_b = 0
            val_c = 0
            abs_sum = abs(val_a + val_b + val_c)
            sum_abs = abs(val_a) + abs(val_b) + abs(val_c)
            total_diff += sum_abs - abs_sum
            
        # Operations with only b
        for k in only_b:
            val_b = 1 if operations[k][0] == b else -1
            val_a = 0
            val_c = 0
            abs_sum = abs(val_a + val_b + val_c)
            sum_abs = abs(val_a) + abs(val_b) + abs(val_c)
            total_diff += sum_abs - abs_sum
            
        # Operations with only c
        for k in only_c:
            val_c = 1 if operations[k][0] == c else -1
            val_a = 0
            val_b = 0
            abs_sum = abs(val_a + val_b + val_c)
            sum_abs = abs(val_a) + abs(val_b) + abs(val_c)
            total_diff += sum_abs - abs_sum
            
        # Operations with a,b but not c
        for k in only_ab:
            u_k, v_k = operations[k]
            if u_k == a:
                val_a, val_b, val_c = 1, -1, 0
            else:
                val_a, val_b, val_c = -1, 1, 0
            abs_sum = abs(val_a + val_b + val_c)
            sum_abs = abs(val_a) + abs(val_b) + abs(val_c)
            total_diff += sum_abs - abs_sum
            
        # Operations with a,c but not b
        for k in only_ac:
            u_k, v_k = operations[k]
            if u_k == a:
                val_a, val_b, val_c = 1, 0, -1
            else:
                val_a, val_b, val_c = -1, 0, 1
            abs_sum = abs(val_a + val_b + val_c)
            sum_abs = abs(val_a) + abs(val_b) + abs(val_c)
            total_diff += sum_abs - abs_sum
            
        # Operations with b,c but not a
        for k in only_bc:
            u_k, v_k = operations[k]
            if u_k == b:
                val_a, val_b, val_c = 0, 1, -1
            else:
                val_a, val_b, val_c = 0, -1, 1
            abs_sum = abs(val_a + val_b + val_c)
            sum_abs = abs(val_a) + abs(val_b) + abs(val_c)
            total_diff += sum_abs - abs_sum
            
        return total_diff
    
    # Iterate through all triplets (a,b,c) with a<b<c in lexicographical order
    for a in range(n):
        for b in range(a+1, n):
            for c in range(b+1, n):
                diff = compute_score(a, b, c)
                if diff == 6:
                    print(a+1, b+1, c+1)
                    return

if __name__ == "__main__":
    main()
