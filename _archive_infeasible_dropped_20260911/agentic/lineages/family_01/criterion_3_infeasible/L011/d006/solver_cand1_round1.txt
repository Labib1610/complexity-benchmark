```python
import sys

def solve():
    # The condition sum |Sa + Sb + Sc| = sum (|Sa| + |Sb| + |Sc|) - 6
    # is equivalent to saying that there are 3 indices k1, k2, k3 such that
    # the sum of the three sequences at these positions results in a cancellation.
    # Specifically, for a trio (a, b, c), the sum of absolute values is 
    # sum(|Sa| + |Sb| + |Sc|) = 2 * (number of operations involving a, b, or c).
    # The condition simplifies to finding a triangle in the graph of operations
    # where the edges are (u, v).
    # If we have a triangle (a, b, c), then at the positions where these edges
    # were defined, the values cancel out.
    # Specifically, if edges (a, b), (b, c), and (a, c) exist,
    # at position k1 (a, b): Sa=1, Sb=-1, Sc=0 -> |1-1+0| = 0, |1|+|-1|+0 = 2
    # at position k2 (b, c): Sa=0, Sb=1, Sc=-1 -> |0+1-1| = 0, 0+|1|+|-1| = 2
    # at position k3 (a, c): Sa=1, Sc=-1, Sb=0 -> |1+0-1| = 0, |1|+0+|-1| = 2
    # Total reduction is 2+2+2 = 6.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        
    # To find the lexicographically smallest (a, b, c), iterate a, b, c
    for a in range(1, n + 1):
        # Sort neighbors to ensure we check b and c in order
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a < b < c. Since we iterate a, we just need b > a.
            # Check if there exists c such that (a, c) and (b, c) exist.
            # (a, c) is in adj[a], (b, c) is in adj[b] or adj[c]
            # Since we only stored u < v in adj[u], we need to be careful.
            # Edge (u, v) with u < v is in adj[u].
            
            # Possible c:
            # 1. a < b < c: (a, b), (a, c), (b, c)
            # 2. a < c < b: (a, c), (c, b), (a, b)
            # 3. c < a < b: (c, a), (c, b), (a, b)
            
            # Check case 1: c > b
            # Need c in adj[a] and c in adj[b]
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return
            
            # Check case 2: a < c < b
            # Need c in adj[a] and b in adj[c]
            for c in neighbors_a:
                if c > a and c < b:
                    if b in adj[c]:
                        print(f"{a} {c} {b}")
                        return
            
            # Check case 3: c < a < b
            # Need a in adj[c] and b in adj[c]
            for c in range(1, a):
                if a in adj[c] and b in adj[c]:
                    print(f"{c} {a} {b}")
                    return

solve()
```
