import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[k] stores the pair (u, v) for operation k
    # We need to calculate D(x) and W(x, y)
    # D(x) = sum of S_x[i]
    # W(x, y) = sum |S_x[i] - S_y[i]|
    # Note: S_x[i] is 1 if x is one of the two sequences in operation i, else 0.
    # D(x) is the number of operations involving x.
    # W(x, y) = D(x) + D(y) - 2 * (number of operations involving both x and y)
    # The condition D(a) + D(b) > W(a, b) simplifies to:
    # D(a) + D(b) > D(a) + D(b) - 2 * count(a, b)
    # 0 > -2 * count(a, b) => count(a, b) > 0
    # So the condition is simply that for every pair, they must share at least one operation.
    
    # count[u][v] stores how many operations involve both u and v
    # Since n <= 2000, we can use a list of dictionaries or a flat array
    # Given the constraints and the nature of the problem, we need to find a, b, c
    # such that count(a, b) > 0, count(a, c) > 0, and count(b, c) > 0.
    
    adj = [[] for _ in range(n + 1)]
    count = {}
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v: u, v = v, u
        adj[u].append(v)
        count[(u, v)] = count.get((u, v), 0) + 1
        
    def get_count(u, v):
        if u > v: u, v = v, u
        return count.get((u, v), 0)
    
    # Iterate to find the lexicographically smallest (a, b, c)
    for a in range(1, n + 1):
        # Candidates for b must be connected to a
        neighbors_a = sorted(list(set(adj[a])))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a: continue
            
            # Candidates for c must be connected to both a and b
            # To be efficient, intersect neighbors of a and b
            neighbors_b = set(adj[b])
            for c in neighbors_a[i+1:]:
                if c in neighbors_b:
                    # Check if b and c are connected
                    if get_count(b, c) > 0:
                        print(f"{a} {b} {c}")
                        return

solve()
