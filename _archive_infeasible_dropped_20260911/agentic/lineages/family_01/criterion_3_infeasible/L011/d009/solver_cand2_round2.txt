```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # The equation E(a, b) + E(b, c) + E(c, a) = 2(a^2 + b^2 + c^2)
    # Let S = {a, b, c}.
    # If all three pairs are tested: (a^2+b^2) + (b^2+c^2) + (c^2+a^2) = 2(a^2+b^2+c^2).
    # This is always true if all three pairs are tested.
    # If only two pairs are tested (e.g., {a,b} and {b,c}): (a^2+b^2) + (b^2+c^2) + 0 = 2b^2 + a^2 + c^2.
    # This is also always true.
    # If only one pair is tested (e.g., {a,b}): (a^2+b^2) + 0 + 0 = a^2 + b^2.
    # This is only true if a^2 + b^2 = 2(a^2 + b^2 + c^2), which is impossible for positive a,b,c.
    # If zero pairs are tested: 0 = 2(a^2 + b^2 + c^2), impossible.
    
    # Thus, we need to find a, b, c (a < b < c) such that at least two pairs are tested.
    # To find the lexicographically smallest, we iterate a from 1 to n, b from a+1 to n, c from b+1 to n.
    # Since we need at least two pairs, we check if (a,b) and (b,c) are tested, 
    # or (a,b) and (a,c), or (b,c) and (a,c).
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            ab_exists = b in adj[a]
            for c in range(b + 1, n + 1):
                bc_exists = c in adj[b]
                ac_exists = c in adj[a]
                
                count = (1 if ab_exists else 0) + (1 if bc_exists else 0) + (1 if ac_exists else 0)
                if count >= 2:
                    print(f"{a} {b} {c}")
                    return

solve()
```
