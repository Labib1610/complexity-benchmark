```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[u] stores the set of sequences v such that {u, v} is an operation
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # W(Sa ^ Sb ^ Sc) = W(Sa) + W(Sb) + W(Sc) - 2 * (intersection(Sa, Sb) + intersection(Sb, Sc) + intersection(Sa, Sc)) + 6 * intersection(Sa, Sb, Sc)
    # The condition W(Sa ^ Sb ^ Sc) = W(Sa) + W(Sb) + W(Sc) - 6
    # implies 2 * (intersection(Sa, Sb) + intersection(Sb, Sc) + intersection(Sa, Sc)) - 6 * intersection(Sa, Sb, Sc) = 6
    # intersection(Sa, Sb) + intersection(Sb, Sc) + intersection(Sa, Sc) - 3 * intersection(Sa, Sb, Sc) = 3
    # Since each operation is a pair {u, v}, intersection(Sa, Sb) is 1 if {a, b} is an operation, else 0.
    # intersection(Sa, Sb, Sc) is 1 if {a, b}, {b, c}, and {a, c} are all operations (a triangle).
    # Let x_ab = 1 if {a, b} is an edge, 0 otherwise.
    # Condition: x_ab + x_bc + x_ac - 3 * (x_ab * x_bc * x_ac) = 3
    # If x_ab=1, x_bc=1, x_ac=1, then 1+1+1 - 3(1) = 0 != 3.
    # If two are 1 and one is 0, then 1+1+0 - 3(0) = 2 != 3.
    # If one is 1 and two are 0, then 1+0+0 - 3(0) = 1 != 3.
    # Wait, the condition is W(Sa ^ Sb ^ Sc) = W(Sa) + W(Sb) + W(Sc) - 6.
    # This happens if and only if there are exactly 3 shared operations between the pairs.
    # Specifically, if {a, b}, {b, c}, {a, c} are NOT all edges, but there are 3 operations shared.
    # Actually, the condition simplifies to: the number of operations shared by pairs (a,b), (b,c), (a,c) is 3.
    
    # Let N(u, v) be the number of common neighbors of u and v.
    # The condition is N(a, b) + N(b, c) + N(a, c) = 3 (if no triangle) or something similar.
    # Actually, the condition is simply: N(a, b) + N(b, c) + N(a, c) = 3.
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            nab = len(adj[a].intersection(adj[b]))
            if nab > 3: continue
            
            # We need to find c > b such that nab + nbc + nac = 3
            # Iterate over all c > b
            for c in range(b + 1, n + 1):
                nbc = len(adj[b].intersection(adj[c]))
                nac = len(adj[a].intersection(adj[c]))
                if nab + nbc + nac == 3:
                    print(f"{a} {b} {c}")
                    return

solve()
```
