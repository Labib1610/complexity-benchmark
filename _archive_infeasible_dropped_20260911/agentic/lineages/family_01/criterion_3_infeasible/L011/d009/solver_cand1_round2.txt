```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[u] will store a list of neighbors v such that {u, v} is an operation
    adj = [[] for _ in range(n + 1)]
    # To quickly check if {u, v} is an operation
    # Since m <= 200,000, we can use a set of frozensets or a list of sets
    # But given the constraints and the nature of the problem, 
    # we need to find a triangle (a, b, c) such that the edges (a,b), (b,c), (c,a) exist.
    # The score condition:
    # F(Si, Sj) = sum_{k=1}^m (3 if Si[k]=1 and Sj[k]=1 else 1)
    # F(Si, Sj) = m + 2 * (number of positions k where Si[k]=1 and Sj[k]=1)
    # Let c(i, j) be the number of positions where both Si and Sj have 1.
    # Since each operation k sets exactly two sequences to 1, 
    # c(i, j) is 1 if {i, j} is an operation, and 0 otherwise.
    # So F(Si, Sj) = m + 2 if {i, j} is an operation, else m.
    # The condition: F(Sa, Sb) + F(Sb, Sc) + F(Sc, Sa) = 3m + 6
    # (m + 2*c(a, b)) + (m + 2*c(b, c)) + (m + 2*c(c, a)) = 3m + 6
    # 3m + 2*(c(a, b) + c(b, c) + c(c, a)) = 3m + 6
    # c(a, b) + c(b, c) + c(c, a) = 3
    # Since c(i, j) is 1 if {i, j} is an operation, this means {a, b}, {b, c}, {c, a}
    # must all be operations.
    # Thus, we are looking for a triangle in the graph where edges are operations.
    
    adj_sets = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v: u, v = v, u
        adj[u].append(v)
        adj[v].append(u)
        adj_sets[u].add(v)
        adj_sets[v].add(u)
        edges.append((u, v))
    
    # Sort adjacency lists to help find lexicographically smallest
    for i in range(1, n + 1):
        adj[i].sort()
        
    # Find the lexicographically smallest triangle (a, b, c)
    for a in range(1, n + 1):
        for b in adj[a]:
            if b <= a: continue
            # Check for c in adj[a] and adj[b]
            # To ensure a < b < c, we only look for c > b
            for c in adj[b]:
                if c <= b: continue
                if c in adj_sets[a]:
                    print(f"{a} {b} {c}")
                    return

solve()
```
