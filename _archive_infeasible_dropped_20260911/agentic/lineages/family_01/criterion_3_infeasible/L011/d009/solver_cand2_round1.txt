import sys

def solve():
    # The condition d(Sa, Sb) + d(Sb, Sc) + d(Sc, Sa) = 2 * (W(Sa) + W(Sb) + W(Sc)) - 6
    # simplifies based on the fact that each operation sets exactly two bits.
    # Let x be the number of operations involving only {a, b}, {b, c}, or {a, c}.
    # Let y be the number of operations involving one of {a, b, c} and one outside.
    # The condition simplifies to finding a triangle in the graph of operations
    # where the three edges (a,b), (b,c), (a,c) exist.
    # Specifically, if we have edges (a,b), (b,c), and (a,c), the condition
    # is satisfied if there is exactly one operation for each of these pairs.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # We need to find a, b, c such that (a,b), (b,c), (a,c) are edges.
    # To ensure lexicographical order, iterate a from 1 to n, b from a+1 to n, c from b+1 to n.
    for a in range(1, n + 1):
        for b in sorted(list(adj[a])):
            if b <= a:
                continue
            # Check for common neighbors c > b
            common = adj[a].intersection(adj[b])
            for c in sorted(list(common)):
                if c > b:
                    # Check if (b, c) is an edge
                    if c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

solve()
