```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[u] stores the set of sequences v such that {u, v} is an operation
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # Let x = S_a[k], y = S_b[k], z = S_c[k].
    # The term (x+y+z)^4 - (x+y+z) is non-zero only if at least two of x, y, z are 1.
    # If exactly two are 1, (1+1+0)^4 - (1+1+0) = 16 - 2 = 14.
    # If exactly three are 1, (1+1+1)^4 - (1+1+1) = 81 - 3 = 78.
    # Let N2 be the number of k where exactly two of {a, b, c} are 1.
    # Let N3 be the number of k where all three are 1.
    # 14 * N2 + 78 * N3 = 42.
    # Since N2, N3 are non-negative integers, we check:
    # If N3 = 0, 14 * N2 = 42 => N2 = 3.
    # If N3 = 1, 14 * N2 = 42 - 78 = -36 (impossible).
    # So we need exactly 3 positions where exactly two of {a, b, c} are 1,
    # and 0 positions where all three are 1.
    
    # N2 is the number of k where (S_a[k], S_b[k], S_c[k]) is (1,1,0), (1,0,1), or (0,1,1).
    # This is equivalent to the number of operations {a,b}, {a,c}, or {b,c}.
    # Let count(u, v) = 1 if {u, v} is an operation, else 0.
    # N2 = count(a, b) + count(a, c) + count(b, c).
    # N3 = number of k where S_a[k]=S_b[k]=S_c[k]=1.
    # This happens if there exists k such that {a,b}, {a,c}, {b,c} are all operations.
    # But the problem says each operation is a pair {u, v}.
    # S_a[k]=1 and S_b[k]=1 means {a, b} is the k-th operation.
    # So N3 = 1 if {a,b}, {a,c}, {b,c} are all operations, which is impossible
    # because each operation is a unique pair. Thus N3 is always 0.
    # We need count(a, b) + count(a, c) + count(b, c) = 3.
    # This means {a, b}, {a, c}, and {b, c} must all be operations.
    
    for a in range(1, n + 1):
        adj_a = adj[a]
        sorted_adj_a = sorted(list(adj_a))
        len_a = len(sorted_adj_a)
        for i in range(len_a):
            b = sorted_adj_a[i]
            if b <= a: continue
            adj_b = adj[b]
            # Intersection of adj[a] and adj[b] gives all c such that {a,c} and {b,c} are ops
            common = adj_a.intersection(adj_b)
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
