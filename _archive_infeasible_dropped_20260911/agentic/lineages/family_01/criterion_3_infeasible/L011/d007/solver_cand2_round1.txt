import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # For a trio (a, b, c) with a < b < c, the matrix M_k is:
    # [[1, 1, 1], [Sa, Sb, Sc], [Sa^2, Sb^2, Sc^2]]
    # det(M_k) = (Sb - Sa)(Sc - Sa)(Sc - Sb)
    # The values of (Sa, Sb, Sc) at position k can only be:
    # (1, -1, 0) -> det = (-1-1)(0-1)(0-(-1)) = (-2)(-1)(1) = 2, det^2 = 4
    # (1, 0, -1) -> det = (0-1)(-1-1)(-1-0) = (-1)(-2)(-1) = -2, det^2 = 4
    # (0, 1, -1) -> det = (1-0)(-1-0)(-1-1) = (1)(-1)(-2) = 2, det^2 = 4
    # Any other combination involving 0s results in det = 0.
    # We need sum(det^2) = 12. Since each non-zero contribution is 4,
    # we need exactly 3 operations where the trio (a, b, c) are involved.
    
    # An operation (u, v) involves a, b, c if {u, v} is a subset of {a, b, c}.
    # Since u != v, the possible pairs are {a, b}, {a, c}, {b, c}.
    # If all three pairs {a, b}, {a, c}, {b, c} appear as operations,
    # then for each pair, one of the three det^2 cases will be triggered.
    # Specifically, if {a, b}, {a, c}, {b, c} are operations,
    # at k1=(a,b), Sa=1, Sb=-1, Sc=0 -> det^2=4
    # at k2=(a,c), Sa=1, Sb=0, Sc=-1 -> det^2=4
    # at k3=(b,c), Sa=0, Sb=1, Sc=-1 -> det^2=4
    # Total sum = 12.
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(input_data[2 + 2 * i])
        v = int(input_data[3 + 2 * i])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        
    # We need to find a, b, c such that (a,b), (a,c), (b,c) are all edges.
    # This is equivalent to finding a triangle in the graph.
    # Iterate a from 1 to n, b from a+1 to n, check if (a,b) is an edge.
    # If so, check common neighbors c > b.
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # Find common neighbors c > b
                # Intersection of adj[a] and adj[b]
                common = adj[a].intersection(adj[b])
                for c in common:
                    if c > b:
                        print(f"{a} {b} {c}")
                        return

solve()
