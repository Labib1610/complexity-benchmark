import sys

        # and sum over k of (S_a[k] + S_b[k] + S_c[k]) + 42

        set_b = seq_positions_set[b]
        set_c = seq_positions_set[c]

        union_pos = set_a | set_b | set_c
        
        sum_lhs = 0
        sum_rhs = 0

            val_a = 1 if k in set_a else 0
            val_b = 1 if k in set_b else 0
            val_c = 1 if k in set_c else 0
            s = val_a + val_b + val_c  # This is 0, 1, 2, or 3
            sum_lhs += s ** 4
            sum_rhs += s
        
        sum_rhs += 42
        return sum_lhs == sum_rhs
    
    # We need to find a, b, c such that the equation holds
    # Let's analyze what the equation means:
    # Let x_k = S_a[k] + S_b[k] + S_c[k]. Then x_k can be 0, 1, 2, or 3.
    # We want sum(x_k^4) = sum(x_k) + 42

            if b == u or b == v:
                members += 1
            if c == u or c == v:
                members += 1
            if members == 2:
                count += 1
        return count

                    print(a+1, b+1, c+1)
                    return
