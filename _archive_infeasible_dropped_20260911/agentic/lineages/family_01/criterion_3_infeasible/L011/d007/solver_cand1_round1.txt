import sys

def solve():
    # The expression (S_a[k] - S_b[k]) * (S_b[k] - S_c[k]) * (S_c[k] - S_a[k])
    # is non-zero only if S_a[k], S_b[k], S_c[k] are distinct.
    # Since each operation involves only two sequences (one 1, one -1),
    # at any position k, the values are {1, -1, 0}.
    # For the product to be non-zero, the set {S_a[k], S_b[k], S_c[k]} must be {1, -1, 0}.
    # If {S_a[k], S_b[k], S_c[k]} = {1, -1, 0}, the product is (1 - (-1)) * (-1 - 0) * (0 - 1) = 2 * -1 * -1 = 2.
    # The square is 4.
    # The sum is 12, so there must be exactly 3 positions k where the set is {1, -1, 0}.
    # This happens if and only if the three sequences form a triangle in the graph
    # where edges are operations. Specifically, if we have edges (a,b), (b,c), (a,c),
    # let the operations be:
    # op1: {a,b}, op2: {b,c}, op3: {a,c}
    # At op1: S_a=1, S_b=-1, S_c=0
    # At op2: S_b=1, S_c=-1, S_a=0
    # At op3: S_c=1, S_a=-1, S_b=0
    # Sum of squares = 4 + 4 + 4 = 12.
    # So we need to find a triangle (a, b, c) in the graph of operations.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a: continue
            
            # Check common neighbors c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in sorted(list(common)):
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
