```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[u] will store the set of sequences v such that S_u . S_v > 0
    # S_u . S_v > 0 if there exists at least one k such that S_u[k]=1 and S_v[k]=1.
    # The input gives us pairs (u, v) that share a 1 at some position k.
    # If (u, v) is an operation, then S_u[k]=1 and S_v[k]=1.
    # This means S_u . S_v >= 1.
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that b in adj[a], c in adj[a], and c in adj[b].
    # Since we need the lexicographically smallest (a, b, c), we iterate a from 1 to n,
    # then b from a+1 to n, then c from b+1 to n.
    
    for a in range(1, n + 1):
        # Neighbors of a
        neighbors_a = sorted(list(adj[a]))
        if len(neighbors_a) < 2:
            continue
            
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c such that c is in adj[a] and c is in adj[b]
            # To keep it lexicographically smallest, we check neighbors of b
            # that are also in neighbors_a.
            
            # Optimization: intersect sets
            common = adj[a].intersection(adj[b])
            
            # Filter for c > b
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

solve()
```
