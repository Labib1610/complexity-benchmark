import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Represent the graph as an adjacency list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # The coefficient of x_a * x_b * x_c in Q = P * (x_1 + ... + x_n)
    # is the number of edges in the triangle (a, b, c) in the graph P.
    # We want the coefficient to be 3, which means (a, b, c) must form a triangle.
    # We need to find the lexicographically smallest (a, b, c) such that
    # (a, b), (b, c), and (a, c) are all edges in P.
    
    # Iterate through all possible a, b, c with a < b < c
    # To optimize, iterate through edges (a, b) and check if there exists c > b
    # such that (a, c) and (b, c) are edges.
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that b < c and c is in both adj[a] and adj[b]
            # To keep it lexicographically smallest, we check c in increasing order
            # Intersection of neighbors_a and neighbors_b
            neighbors_b = adj[b]
            
            # Find common neighbors c > b
            common = []
            for c in neighbors_a:
                if c > b and c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
