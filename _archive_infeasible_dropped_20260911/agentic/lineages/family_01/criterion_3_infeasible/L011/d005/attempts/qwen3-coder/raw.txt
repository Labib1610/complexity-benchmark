import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Read operations
    operations = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        operations.append((u, v))
        index += 2
    
    # For each sequence, maintain a set of sequences it shares positions with
    # We'll store for each sequence i, a set of sequences it has a dot product with
    # Actually, let's think differently.
    
    # For each pair of sequences, count how many times they both are 1
    # But m can be up to 200000, so we can't store all pairwise dot products
    
    # Let's store for each sequence, which positions it has 1 in
    # But that's also too much memory
    
    # Key insight: we only care about dot products > 0
    # So we can store for each sequence, a set of sequences it intersects with
    
    # Let's build adjacency list: for each sequence, which sequences does it share a 1 with
    # But we need counts, not just existence
    
    # Let's store for each sequence, a dictionary of neighbors and their dot product counts
    neighbors = [defaultdict(int) for _ in range(n)]
    
    # Process each operation
    for u, v in operations:
        neighbors[u][v] += 1
        neighbors[v][u] += 1
    
    # Now we want to find a triangle a-b-c where all edges have weight > 0
    # And we want the lexicographically smallest such triple
    
    # Convert neighbors to sorted lists for consistent iteration
    adj = [[] for _ in range(n)]
    for i in range(n):
        for j, count in neighbors[i].items():
            if count > 0:
                adj[i].append(j)
        adj[i].sort()
    
    # For efficiency, let's only consider neighbors
    # We'll look for triangles by checking each pair of nodes that are connected
    # and seeing if they share a common neighbor
    
    # But we want the lexicographically smallest, so let's iterate in order
    for a in range(n):
        # Get neighbors of a
        neigh_a = adj[a]
        # For each pair of neighbors of a, check if they are connected
        for i in range(len(neigh_a)):
            b = neigh_a[i]
            if b <= a:
                continue
            for j in range(i+1, len(neigh_a)):
                c = neigh_a[j]
                if c <= b:
                    continue
                # Check if b and c are connected
                if c in neighbors[b]:
                    # Found a triangle
                    print(a+1, b+1, c+1)  # Convert back to 1-based indexing
                    return

if __name__ == "__main__":
    main()
