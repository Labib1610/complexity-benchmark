import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need a, b, c such that (a,b), (a,c), (b,c) are all edges.
    # This is a triangle in the graph.
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n.
    # For a fixed a, we look for b, c in adj[a] such that b < c and (b, c) is an edge.
    
    edges.sort()
    
    # To optimize, iterate through all edges (u, v) and check common neighbors
    # Since n <= 2000, we can use bitsets or just intersection.
    # Given the constraints and the requirement for lexicographical order:
    
    for a in range(1, n + 1):
        neighbors = sorted(list(adj[a]))
        if len(neighbors) < 2:
            continue
        
        for i in range(len(neighbors)):
            b = neighbors[i]
            for j in range(i + 1, len(neighbors)):
                c = neighbors[j]
                # Check if (b, c) is an edge
                if c in adj[b]:
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
