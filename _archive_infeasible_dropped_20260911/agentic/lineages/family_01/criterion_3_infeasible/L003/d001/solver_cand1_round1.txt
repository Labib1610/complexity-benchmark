```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need a trio (a, b, c) such that (a, b), (a, c), and (b, c) are all edges.
    # This is equivalent to finding a triangle in the graph.
    # The problem asks for the lexicographically smallest (a, b, c) with a < b < c.
    
    edges.sort()
    
    # Iterate through all edges (a, b) and check for a common neighbor c
    # To ensure lexicographical order, we iterate a from 1 to n, then b from a+1 to n
    for a in range(1, n + 1):
        # Sort neighbors to check c in increasing order
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we have edge (a, b). Find common neighbors c > b.
            # Intersection of adj[a] and adj[b]
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
