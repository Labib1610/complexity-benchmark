import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        edges.append((u, v))
        index += 2
    
    # Build adjacency list
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # Collect all edges and sort them to process in order
    edge_list = []
    for u, v in edges:
        if u > v:
            u, v = v, u
        edge_list.append((u, v))
    edge_list.sort()
    
    # For each edge (a,b), look for a neighbor of a or b that connects to the other
    for a, b in edge_list:
        # Check neighbors of a
        for c in adj[a]:
            if c == b:
                continue
            if c in adj[b]:
                # Found triangle a-b, b-c, a-c
                res = sorted([a, b, c])
                print(res[0], res[1], res[2])
                return
        
        # Check neighbors of b
        for c in adj[b]:
            if c == a:
                continue
            if c in adj[a]:
                # Found triangle a-b, b-c, a-c
                res = sorted([a, b, c])
                print(res[0], res[1], res[2])
                return

if __name__ == "__main__":
    main()
