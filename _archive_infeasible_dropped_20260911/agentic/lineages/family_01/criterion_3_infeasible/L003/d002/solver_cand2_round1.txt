import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # A routing deadlock is a triangle (a, b, c).
    # We need the lexicographically smallest (a, b, c) with a < b < c.
    # Iterate through all possible 'a' from 1 to n-2.
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # For a fixed 'a', we need to find b, c such that:
        # 1. a < b < c
        # 2. b is a neighbor of a
        # 3. c is a neighbor of a
        # 4. b is a neighbor of c
        
        # To ensure lexicographical order, iterate b then c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in sorted(list(common)):
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
