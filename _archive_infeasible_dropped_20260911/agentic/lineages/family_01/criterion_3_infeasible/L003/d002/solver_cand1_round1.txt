```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (i, j, k) with i < j < k
    # To prioritize (i, j, k) lexicographically:
    # Iterate i from 1 to n-2
    #   Iterate j from i+1 to n-1
    #     If j is connected to i:
    #       Find common neighbors k > j
    
    for i in range(1, n + 1):
        # Sort neighbors to help with lexicographical order
        neighbors_i = sorted(list(adj[i]))
        
        for j in neighbors_i:
            if j <= i:
                continue
            
            # We need a k such that k > j and k is connected to both i and j
            # Intersection of adj[i] and adj[j]
            # Since we want the smallest k, we can iterate through sorted neighbors
            neighbors_j = adj[j]
            
            # To optimize, iterate through the smaller neighbor list
            if len(adj[i]) < len(neighbors_j):
                candidates = [k for k in neighbors_i if k > j and k in neighbors_j]
            else:
                candidates = [k for k in sorted(list(neighbors_j)) if k > j and k in adj[i]]
            
            if candidates:
                print(f"{i} {j} {candidates[0]}")
                return

if __name__ == '__main__':
    solve()
```
