import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges lexicographically: (u, v) ascending
    edges.sort()
    
    # For each edge (a, b) with a < b, find smallest c > b such that
    # both (a, c) and (b, c) are edges. We want lexicographically smallest (a, b, c).
    for a, b in edges:
        # Candidates: common neighbors of a and b that are > b
        # Iterate over the smaller adjacency set for efficiency
        if len(adj[a]) > len(adj[b]):
            a, b = b, a
        # Now adj[a] is smaller (but a might be > b now, doesn't matter for membership)
        # We need c > max(original_a, original_b) = original_b
        # original_a < original_b, so c > original_b
        # But we swapped a and b, so we need to be careful
        # Actually we just need common neighbors > max(a_orig, b_orig)
        # Let's restore original order
        # Wait, we swapped, so let's track original max
        orig_max = max(a, b)
        orig_min = min(a, b)
        
        # Iterate over neighbors of the node with smaller degree
        small_node = a  # has smaller degree
        large_node = b
        small_adj = adj[small_node]
        large_adj = adj[large_node]
        
        # Find common neighbors > orig_max
        # Iterate through small_adj
        for c in small_adj:
            if c > orig_max and c in large_adj:
                # Found triangle (orig_min, orig_max, c)
                print(f"{orig_min} {orig_max} {c}")
                return
    
    # Should not reach here per problem statement
    print("-1 -1 -1")

solve()
