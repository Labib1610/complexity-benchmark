```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # We'll iterate through all edges and for each edge (u,v), 
    # we'll look for a common neighbor w, which would form a triangle.
    # To get the lexicographically smallest, we'll process nodes in order.
    
    # First, let's sort the adjacency lists for consistent processing
    for i in range(1, n + 1):
        adj[i] = sorted(adj[i])
    
    best = None
    
    # Iterate through all edges
    for u, v in edges:
        if u >= v:
            continue
        # Find common neighbors of u and v
        # To optimize, iterate through the smaller adjacency list
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        set_v = set(adj[v])
        for w in adj[u]:
            if w in set_v:
                # Found a triangle: u, v, w
                # Ensure ordering a < b < c
                a, b, c = sorted([u, v, w])
                if best is None or (a, b, c) < best:
                    best = (a, b, c)
    
    print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
```
