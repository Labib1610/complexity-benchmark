import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then check if c exists.
    # Since we need the smallest a, then b, then c, we sort the edges
    # or iterate systematically.
    
    # Sort edges to help find smallest a, b, c
    edges.sort()
    
    # For each edge (u, v), check if there exists a w such that
    # (u, w) and (v, w) are also edges.
    # To ensure lexicographical order, we check candidates for a, b, c.
    
    # Optimization: iterate through all u, then all neighbors v of u,
    # then check if there is a common neighbor w.
    
    best = None
    
    # Sort adjacency lists to ensure we find the smallest c for a given (a, b)
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a < b < c. Since a is fixed, b is fixed.
            # We need to find the smallest c > b such that c is in adj[a] and adj[b].
            # Intersection of sorted lists
            ptr_a = i + 1
            ptr_b = 0
            neighbors_b = adj[b]
            
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    c = val_a
                    # Found a triangle (a, b, c)
                    # Since we iterate a from 1, b from neighbors of a,
                    # and c is the smallest common neighbor > b,
                    # the first one we find is the lexicographically smallest.
                    print(f"{a} {b} {c}")
                    return
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1
                    
solve()
