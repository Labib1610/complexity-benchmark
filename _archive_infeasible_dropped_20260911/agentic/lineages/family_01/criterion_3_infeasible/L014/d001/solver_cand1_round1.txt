import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest trio
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # all edges (a, b) and check if there exists a c > b such that
    # (a, c) and (b, c) are also edges.
    
    # To optimize, iterate through all edges (a, b)
    # For each edge (a, b), find common neighbors c > b
    
    # Pre-sort adjacency lists to make intersection faster
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    best_trio = None
    
    for a, b in edges:
        # We want to find c > b such that c is in adj[a] and c is in adj[b]
        # Since adj lists are sorted, we can use two pointers or set intersection
        # Given constraints, set intersection is fine if we use the smaller list
        
        # Optimization: only check if a potential c could improve the current best
        # If we already have a best_trio (a_best, b_best, c_best),
        # we only care about (a, b, c) < (a_best, b_best, c_best)
        
        list_a = adj[a]
        list_b = adj[b]
        
        # Only look for c > b
        # Use two pointers to find common elements in list_a and list_b
        i = 0
        j = 0
        while i < len(list_a) and j < len(list_b):
            val_a = list_a[i]
            val_b = list_b[j]
            
            if val_a == val_b:
                if val_a > b:
                    c = val_a
                    if best_trio is None or (a, b, c) < best_trio:
                        best_trio = (a, b, c)
                    # Since we want the smallest c for this (a, b), we can break
                    break
                i += 1
                j += 1
            elif val_a < val_b:
                i += 1
            else:
                j += 1
                
    if best_trio:
        print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

solve()
