import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we look at all neighbors b of a.
    # Then we check if there exists a c such that c is a neighbor of a and b.
    
    # Sort neighbors to ensure we check in lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # To optimize, iterate a, then b in adj[a], then check intersection
    # of adj[a] and adj[b].
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and adj[b]
            # Since adj lists are sorted, we can use two pointers or set intersection
            # Given constraints, set intersection or simple scan is fine.
            # To be lexicographically smallest, we want the smallest c.
            
            # Optimization: use a set for O(1) lookup for the second neighbor
            set_a = set(neighbors_a)
            for c in adj[b]:
                if c > b and c in set_a:
                    print(f"{a} {b} {c}")
                    return

solve()
