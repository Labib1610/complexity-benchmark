import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = {}
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        if u not in adj:
            adj[u] = set()
        adj[u].add(v)
        if v not in adj:
            adj[v] = set()
        adj[v].add(u)
    
    # Sort edges to help find the lexicographically smallest triad
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (u, v) as potential (a, b)
    # Then check if there exists a common neighbor w > v
    
    # Sort keys to iterate in order
    nodes = sorted(adj.keys())
    
    best_triad = None
    
    for u in nodes:
        neighbors_u = sorted(list(adj[u]))
        for i in range(len(neighbors_u)):
            v = neighbors_u[i]
            if u >= v:
                continue
            
            # Now we have a potential (u, v) as (a, b)
            # We need to find the smallest w > v such that w is in adj[u] and adj[v]
            common = adj[u].intersection(adj[v])
            
            # Filter for w > v
            valid_w = [w for w in common if w > v]
            if valid_w:
                w = min(valid_w)
                triad = tuple(sorted((u, v, w)))
                if best_triad is None or triad < best_triad:
                    best_triad = triad
        
        # Optimization: if we found a triad starting with u, 
        # and we have checked all possible v for this u, 
        # we can potentially stop if we are sure no smaller a exists.
        # Since we iterate u in increasing order, the first u that yields 
        # a triad is the smallest 'a'.
        if best_triad is not None:
            # We need to ensure we checked all v for this u
            # Actually, just continue to finish u's loop
            pass

    if best_triad:
        print(f"{best_triad[0]} {best_triad[1]} {best_triad[2]}")

solve()
