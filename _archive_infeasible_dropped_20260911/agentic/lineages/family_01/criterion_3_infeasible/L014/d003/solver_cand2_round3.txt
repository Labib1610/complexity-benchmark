import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that a < b < c and they form a triangle.
    # To satisfy the lexicographical requirement, we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # c is in adj[a] and c is in adj[b].
    
    # To optimize, we can iterate a, then iterate b in adj[a] where b > a.
    # Then find the intersection of adj[a] and adj[b] and pick the smallest c > b.
    
    for a in range(1, n + 1):
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for b in neighbors_a:
            # Find common neighbors c > b
            # Intersection of adj[a] and adj[b]
            # Since we want the smallest c > b, we can iterate through the smaller set
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
            
            best_c = float('inf')
            found = False
            for c in candidates:
                if c > b:
                    if c in adj[a] and c in adj[b]:
                        if c < best_c:
                            best_c = c
                            found = True
            
            if found:
                print(f"{a} {b} {best_c}")
                return

if __name__ == '__main__':
    solve()
