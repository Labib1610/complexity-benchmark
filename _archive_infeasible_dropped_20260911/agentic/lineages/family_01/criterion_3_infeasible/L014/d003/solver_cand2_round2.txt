```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = {}
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        if u not in adj:
            adj[u] = set()
        if v not in adj:
            adj[v] = set()
        adj[u].add(v)
        adj[v].add(u)
    
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c:
    # We iterate through all edges (u, v) and check for common neighbors w.
    # To ensure we find the minimum, we can sort the edges or just check all.
    # Given n=2000, m=2e5, O(m * sqrt(m)) is acceptable.
    
    best_triplet = None
    
    # Sort edges to help find the lexicographically smallest triplet
    edges.sort()
    
    # Pre-sort adjacency lists to iterate in order
    for u in adj:
        adj[u] = sorted(list(adj[u]))
        
    for u, v in edges:
        # We want to find w such that (u, v, w) is a triangle.
        # Let a = min(u, v, w), b = mid(u, v, w), c = max(u, v, w)
        # We want to minimize a, then b, then c.
        
        # Check common neighbors of u and v
        set_u = set(adj[u])
        for w in adj[v]:
            if w in set_u:
                # Found a triangle (u, v, w)
                triplet = sorted([u, v, w])
                if best_triplet is None or triplet < best_triplet:
                    best_triplet = triplet
                    
    if best_triplet:
        print(f"{best_triplet[0]} {best_triplet[1]} {best_triplet[2]}")

solve()
```
