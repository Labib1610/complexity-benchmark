import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # The problem implies we are dealing with satellite identifiers 1..n.
    # We need to track which satellites are currently in the cache.
    # Since the problem asks for the minimum number of cache misses,
    # this is the classic Belady's Optimal Page Replacement Algorithm.
    
    requests = [int(x) for x in input_data[2:]]
    
    # Precompute next occurrence of each satellite
    next_occurrence = [float('inf')] * (n + 1)
    future_indices = [[] for _ in range(n + 1)]
    for i in range(len(requests) - 1, -1, -1):
        future_indices[requests[i]].append(i)
    
    # Current cache state
    cache = set()
    misses = 0
    
    for i in range(len(requests)):
        sat = requests[i]
        # Remove current request from its future list
        future_indices[sat].pop()
        
        if sat in cache:
            continue
        
        misses += 1
        if len(cache) < m:
            cache.add(sat)
        else:
            # Find the element in cache that is used furthest in the future
            furthest_idx = -1
            to_remove = -1
            
            for item in cache:
                if not future_indices[item]:
                    to_remove = item
                    break
                else:
                    idx = future_indices[item][-1]
                    if idx > furthest_idx:
                        furthest_idx = idx
                        to_remove = item
            
            cache.remove(to_remove)
            cache.add(sat)
            
    sys.stdout.write(str(misses) + '\n')

if __name__ == '__main__':
    solve()
