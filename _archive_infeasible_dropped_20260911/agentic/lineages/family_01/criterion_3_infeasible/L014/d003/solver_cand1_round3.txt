import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest (a, b, c)
    # We want smallest a, then smallest b, then smallest c.
    # Iterate through all edges (u, v) as (a, b) where a < b
    edges.sort()
    
    for a, b in edges:
        # Find common neighbors c such that a < b < c or a < c < b or c < a < b
        # Since we need a < b < c, we look for c > b such that c is in adj[a] and adj[b]
        # To ensure we find the smallest a, b, c, we iterate through sorted edges
        # and for each edge (a, b), check common neighbors c > b.
        
        # Optimization: intersect the smaller adjacency list
        if len(adj[a]) < len(adj[b]):
            common = adj[a].intersection(adj[b])
        else:
            common = adj[b].intersection(adj[a])
            
        for c in common:
            if c > b:
                # Found a triangle (a, b, c) with a < b < c
                # Since we iterate edges (a, b) in sorted order, 
                # the first one we find might not be the global minimum.
                # We need to keep track of the best (a, b, c).
                pass
    
    # Actually, the constraints n=2000 allow O(n*m/word_size) or O(n^2)
    # Let's iterate a from 1 to n, then b in adj[a], then c in adj[a]
    best = None
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a: continue
            
            # Check if there exists c > b such that c is in adj[a] and adj[b]
            # To be efficient, intersect adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
            if best and best[0] < a:
                break
        if best and best[0] < a:
            break
            
    print(f"{best[0]} {best[1]} {best[2]}")

solve()
