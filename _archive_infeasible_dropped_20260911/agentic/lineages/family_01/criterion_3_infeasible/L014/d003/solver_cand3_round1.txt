import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = {}
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        if u not in adj: adj[u] = set()
        if v not in adj: adj[v] = set()
        adj[u].add(v)
        adj[v].add(u)
        
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c:
    # We iterate through all possible 'a' in ascending order.
    # Then iterate through all 'b' in adj[a] such that b > a.
    # Then check if there exists 'c' in adj[a] intersect adj[b] such that c > b.
    
    sorted_nodes = sorted(adj.keys())
    
    best_triplet = None
    
    for a in sorted_nodes:
        neighbors_a = sorted(list(adj[a]))
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Optimization: check if we already found a better 'a' or 'b'
            if best_triplet and (a > best_triplet[0] or (a == best_triplet[0] and b >= best_triplet[1])):
                continue
                
            # Find common neighbors c > b
            # Using set intersection for efficiency
            common = adj[a].intersection(adj[b])
            for c in sorted(list(common)):
                if c > b:
                    triplet = (a, b, c)
                    if best_triplet is None or triplet < best_triplet:
                        best_triplet = triplet
                    # Since we want the lexicographically smallest, 
                    # and we iterate a, b, c in order, the first one we find 
                    # for a specific (a, b) is the smallest c for that (a, b).
                    break
                    
    if best_triplet:
        print(f"{best_triplet[0]} {best_triplet[1]} {best_triplet[2]}")

if __name__ == '__main__':
    solve()
