import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c
    # Iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, O(n * m) or O(n^2) is acceptable.
    
    # Sort adjacency lists to help with lexicographical order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that c > b and c is in both adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we want the smallest c, we can iterate through neighbors_b
            # and check if they are in neighbors_a
            
            # Optimization: use set for O(1) lookup
            set_a = adj[a]
            for c in sorted_adj[b]:
                if c > b and c in set_a:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
