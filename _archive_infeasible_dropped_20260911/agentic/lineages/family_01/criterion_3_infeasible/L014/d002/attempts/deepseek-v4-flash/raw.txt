We can solve this by checking each possible triple in increasing order and stopping at the first valid triangle.  
Since \( n \leq 2000 \), a naive \( O(n^3) \) check would be too slow, but we can use an adjacency matrix for \( O(1) \) edge lookups and break early.

```python
import sys

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u][v] = adj[v][u] = True

    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if not adj[a][b]:
                continue
            for c in range(b + 1, n + 1):
                if adj[a][c] and adj[b][c]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
