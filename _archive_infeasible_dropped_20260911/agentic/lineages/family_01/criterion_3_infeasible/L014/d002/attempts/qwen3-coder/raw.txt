```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # We need to find a triangle (three nodes all connected to each other)
    # We'll iterate through all edges and for each edge (u,v), we look for a node w
    # such that w is connected to both u and v
    
    # To optimize, we can sort the adjacency list and process in order
    # But since we want the lexicographically smallest triple,
    # we should iterate through nodes in sorted order
    
    # Let's restructure: for each node, sort its neighbors
    for i in range(1, n + 1):
        adj[i] = sorted(adj[i])
    
    best = None
    
    # Iterate through all edges
    for u, v in edges:
        if u > v:
            u, v = v, u  # Ensure u < v for consistent ordering
        
        # For each neighbor of u, check if it's also a neighbor of v
        # and if it forms a valid triangle with u and v
        set_v = adj[v]
        set_u = adj[u]
        
        # To optimize intersection, we'll iterate through the smaller list
        if len(set_u) > len(set_v):
            set_u, set_v = set_v, set_u
            u, v = v, u
            
        set_v = set(set_v)  # Make lookup O(1)
        
        for w in set_u:
            if w in set_v:
                # Found a triangle: u, v, w
                # Order them: a < b < c
                a, b, c = sorted([u, v, w])
                if best is None or (a, b, c) < best:
                    best = (a, b, c)
                    
    print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
```
