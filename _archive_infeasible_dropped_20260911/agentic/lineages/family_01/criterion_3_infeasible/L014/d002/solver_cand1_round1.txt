import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest triangle
    # We want to minimize a, then b, then c.
    # A triangle is (a, b, c) with a < b < c.
    # Iterate through all edges (a, b) and check if there exists c > b
    # such that (a, c) and (b, c) are edges.
    
    edges.sort()
    
    best = None
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # possible values of a, then b, then c.
    # Since n <= 2000, we can iterate through all edges (a, b)
    # and check common neighbors.
    
    for a in range(1, n + 1):
        # Optimization: if we already found a triangle with a smaller 'a', stop
        if best and a > best[0]:
            break
            
        neighbors_a = adj[a]
        # Sort neighbors to check b in increasing order
        sorted_neighbors_a = sorted(list(neighbors_a))
        
        for b in sorted_neighbors_a:
            if b <= a:
                continue
            if best and a == best[0] and b > best[1]:
                break
            
            # Find common neighbors c > b
            neighbors_b = adj[b]
            common = neighbors_a.intersection(neighbors_b)
            
            for c in common:
                if c > b:
                    if best is None or (a < best[0]) or (a == best[0] and b < best[1]) or (a == best[0] and b == best[1] and c < best[2]):
                        best = (a, b, c)
    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
