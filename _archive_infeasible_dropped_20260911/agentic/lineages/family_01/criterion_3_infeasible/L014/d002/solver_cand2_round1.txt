import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # Sort edges to help find the lexicographically smallest triplet
    edges.sort()
    
    # To find the smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check for a common neighbor c
    # To ensure we find the smallest, we can iterate through sorted edges
    # and for each edge (a, b), check neighbors.
    
    # Optimization: iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # For each pair of neighbors b, c of a, check if (b, c) is an edge
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                if c in adj[b]:
                    # Found a triangle (a, b, c)
                    # Since we iterate a, then b, then c in increasing order,
                    # the first one we find is the lexicographically smallest.
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
