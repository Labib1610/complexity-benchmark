```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to help find lexicographically smallest triple
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we can iterate through all edges (a, b)
    # and check if there exists a c > b such that (a, c) and (b, c) exist.
    
    # To optimize, iterate through all pairs (a, b) that are connected.
    # Since we want the smallest a, then b, then c:
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Now check for c > b such that c is in both adj[a] and adj[b]
            # To get the smallest c, we look at the intersection of adj[a] and adj[b]
            # and pick the smallest element > b.
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
