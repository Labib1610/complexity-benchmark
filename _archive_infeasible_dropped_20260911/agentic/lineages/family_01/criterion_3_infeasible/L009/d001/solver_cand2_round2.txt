import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest trio (a, b, c) with a < b < c:
    # We iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we iterate through all neighbors 'b' of 'a' such that b > a.
    # Then we check if there exists a 'c' such that c > b and c is a neighbor of both a and b.
    # To ensure lexicographical order, we sort the neighbors of 'a' and 'b'.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c > b such that c is in adj[a] and adj[b]
            # Intersection of sorted lists is efficient
            neighbors_b = adj[b]
            
            # Check candidates for c
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c > b and c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
