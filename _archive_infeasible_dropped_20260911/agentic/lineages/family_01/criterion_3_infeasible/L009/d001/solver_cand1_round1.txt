import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triple
    # We want to minimize x, then y, then z.
    # Iterate through all possible x from 1 to n-2
    for x in range(1, n + 1):
        # Get all neighbors of x that are greater than x
        neighbors_x = sorted([v for v in adj[x] if v > x])
        
        # For each pair of neighbors (y, z) of x, check if they are connected
        # To minimize x, y, z, we iterate y and z in increasing order
        for i in range(len(neighbors_x)):
            y = neighbors_x[i]
            for j in range(i + 1, len(neighbors_x)):
                z = neighbors_x[j]
                if z in adj[y]:
                    print(f"{x} {y} {z}")
                    return

if __name__ == '__main__':
    solve()
