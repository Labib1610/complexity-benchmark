```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # For each node, sort its neighbors for efficient intersection
    for i in range(1, n + 1):
        adj[i] = sorted(adj[i])
    
    # We want the lexicographically smallest triangle
    # So we iterate through all edges and for each edge (u,v) with u<v,
    # we look for a third node w > v such that (u,w) and (v,w) are also edges
    
    # To make it faster, for each edge (u,v), we find common neighbors of u and v
    # that are greater than both u and v
    
    best = None
    
    # Let's go through all edges
    for u, v in edges:
        if u >= v:
            continue
        # Find common neighbors of u and v that are > max(u, v)
        nu = adj[u]
        nv = adj[v]
        
        i, j = 0, 0
        while i < len(nu) and j < len(nv):
            if nu[i] == nv[j]:
                w = nu[i]
                if w > v:  # since u < v, w > v implies w > u and w > v
                    candidate = tuple(sorted([u, v, w]))
                    if best is None or candidate < best:
                        best = candidate
                i += 1
                j += 1
            elif nu[i] < nv[j]:
                i += 1
            else:
                j += 1
    
    if best is None:
        # Should not happen according to problem statement
        return
    
    print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
```
