import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # To minimize x, then y, then z, we iterate through possible x, y, z.
    # Since we need a triangle (x, y, z) with x < y < z,
    # we can iterate through all edges (u, v) and check for a common neighbor w.
    # To ensure we find the lexicographically smallest (x, y, z),
    # we can iterate x from 1 to n, then y from x+1 to n, then check if they share a neighbor z > y.
    
    # Optimization: sort adjacency lists to make finding z easier
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for x in range(1, n + 1):
        neighbors_x = adj[x]
        for i in range(len(neighbors_x)):
            y = neighbors_x[i]
            if y <= x:
                continue
            
            # We need a z such that z > y and z is in both adj[x] and adj[y]
            # Since adj lists are sorted, we can use set intersection or two pointers
            # Given the constraints, set intersection is efficient enough.
            
            # Find common neighbors of x and y
            # Only consider z > y
            set_x = set(adj[x])
            set_y = set(adj[y])
            common = sorted(list(set_x.intersection(set_y)))
            
            for z in common:
                if z > y:
                    print(f"{x} {y} {z}")
                    return
                    
    print("-1 -1 -1")

if __name__ == '__main__':
    solve()
