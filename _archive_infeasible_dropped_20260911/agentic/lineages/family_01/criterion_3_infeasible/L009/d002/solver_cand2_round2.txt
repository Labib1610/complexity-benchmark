import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest (x, y, z),
    # we iterate through all possible x, then y, then z.
    # Since we need x < y < z, we iterate x from 1 to n.
    # For a fixed x, we look at all neighbors y of x.
    # If y > x, we check if there exists a z such that z > y
    # and z is a neighbor of both x and y.
    
    for x in range(1, n + 1):
        neighbors_x = sorted(list(adj[x]))
        for i in range(len(neighbors_x)):
            y = neighbors_x[i]
            if y <= x:
                continue
            
            # Find common neighbors z such that z > y
            # We can intersect adj[x] and adj[y]
            common = adj[x].intersection(adj[y])
            valid_z = [z for z in common if z > y]
            
            if valid_z:
                z = min(valid_z)
                print(f"{x} {y} {z}")
                return

if __name__ == '__main__':
    solve()
