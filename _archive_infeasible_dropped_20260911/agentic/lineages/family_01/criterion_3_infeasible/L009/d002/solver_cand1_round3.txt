import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # To minimize x, then y, then z, we iterate through x from 1 to n.
    # For a fixed x, we iterate through all neighbors y of x.
    # If x < y, we check if there exists a z such that y < z and 
    # z is a neighbor of both x and y.
    
    # Sort neighbors to ensure we check in order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for x in range(1, n + 1):
        neighbors_x = adj[x]
        for i in range(len(neighbors_x)):
            y = neighbors_x[i]
            if y <= x:
                continue
            
            # Now we need to find z > y such that z is in adj[x] and adj[y]
            # Since adj lists are sorted, we can intersect them efficiently
            # or just iterate through the smaller set.
            
            # Optimization: iterate through the smaller adjacency list
            if len(adj[x]) < len(adj[y]):
                for z in adj[x]:
                    if z > y and z in adj[y]:
                        print(f"{x} {y} {z}")
                        return
            else:
                for z in adj[y]:
                    if z > y and z in adj[x]:
                        print(f"{x} {y} {z}")
                        return

solve()
