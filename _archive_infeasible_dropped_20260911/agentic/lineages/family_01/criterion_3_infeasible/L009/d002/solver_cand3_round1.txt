import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To minimize x, then y, then z, we iterate x from 1 to n
    # For a fixed x, we iterate y from x+1 to n
    # If x and y are connected, we look for a z > y such that z is connected to both x and y
    
    # Sort neighbors to help with the lexicographical requirement
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for x in range(1, n + 1):
        neighbors_x = sorted_adj[x]
        # Only consider y > x
        for i in range(len(neighbors_x)):
            y = neighbors_x[i]
            if y <= x:
                continue
            
            # We need z > y such that z is in neighbors_x and neighbors_y
            # To minimize y and z, we check neighbors of x that are > y
            # and check if they are also in neighbors of y
            
            # Optimization: use set intersection for faster lookup
            # But since we need the smallest z, we iterate through sorted neighbors of x
            for j in range(i + 1, len(neighbors_x)):
                z = neighbors_x[j]
                if z > y:
                    if z in adj[y]:
                        print(f"{x} {y} {z}")
                        return

if __name__ == '__main__':
    solve()
