import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (x, y, z), we iterate x from 1 to n.
    # For a fixed x, we look for y > x such that (x, y) is an edge.
    # Then we look for z > y such that (x, z) and (y, z) are edges.
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    # Pre-sort adjacency lists to iterate in order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for x in range(1, n + 1):
        neighbors_x = adj[x]
        for i in range(len(neighbors_x)):
            y = neighbors_x[i]
            if y <= x:
                continue
            
            # We need z > y such that z is in both adj[x] and adj[y]
            # Since adj lists are sorted, we can use two pointers or set intersection
            # Given n=2000, O(n^2) is acceptable.
            
            # Optimization: iterate through the smaller adjacency list
            if len(adj[x]) < len(adj[y]):
                for z in adj[x]:
                    if z > y and z in adj[y]:
                        print(f"{x} {y} {z}")
                        return
            else:
                for z in adj[y]:
                    if z > y and z in adj[x]:
                        print(f"{x} {y} {z}")
                        return
                        
    print("-1 -1 -1")

if __name__ == '__main__':
    solve()
