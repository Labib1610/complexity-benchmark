import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # Sort edges to help find the lexicographically smallest triple
    # We want to minimize i, then j, then k.
    # A triple (i, j, k) with i < j < k exists if j in adj[i] and k in adj[i] and k in adj[j]
    
    # Sort vertices by degree to optimize intersection, but we need lexicographical order.
    # Iterate i from 1 to n
    for i in range(1, n + 1):
        neighbors_i = sorted(list(adj[i]))
        len_ni = len(neighbors_i)
        if len_ni < 2:
            continue
            
        # Iterate j in neighbors_i
        for idx_j in range(len_ni):
            j = neighbors_i[idx_j]
            # We need k > j such that k is in adj[i] and k is in adj[j]
            # Intersection of adj[i] and adj[j]
            # Since neighbors_i is sorted, we can look for k > j
            
            # Optimization: intersect smaller set
            if len(adj[i]) < len(adj[j]):
                common = adj[i].intersection(adj[j])
            else:
                common = adj[j].intersection(adj[i])
            
            for k in sorted(list(common)):
                if k > j:
                    print(f"{i} {j} {k}")
                    return

if __name__ == '__main__':
    solve()
