import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To minimize x, then y, then z, we iterate x from 1 to n-2
    for x in range(1, n - 1):
        # Get neighbors of x that are > x
        neighbors_x = [v for v in adj[x] if v > x]
        neighbors_x.sort()
        
        # Iterate through pairs of neighbors (y, z)
        # y must be > x, z must be > y
        for i in range(len(neighbors_x)):
            y = neighbors_x[i]
            # Check if y is connected to any z > y that is also connected to x
            # To optimize, intersect adj[y] and adj[x]
            # We only care about z > y
            for j in range(i + 1, len(neighbors_x)):
                z = neighbors_x[j]
                if z in adj[y]:
                    print(f"{x} {y} {z}")
                    return
                    
    print("-1 -1 -1")

if __name__ == '__main__':
    solve()
