import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    degree = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        degree[u] += 1
        degree[v] += 1
        idx += 2
        
    # The condition: Total reports involving at least one of {a, b, c}
    # Let S = {a, b, c}. The number of reports involving at least one of S is:
    # (D_a + D_b + D_c) - (edges between pairs in S)
    # The condition given is: (D_a + D_b + D_c) - (edges in S) = (D_a + D_b + D_c) - 3
    # This simplifies to: (edges in S) = 3.
    # This means {a, b, c} must form a triangle in the graph.
    
    # We need the lexicographically smallest triangle (a, b, c) with a < b < c.
    # Iterate through all possible a, b, c.
    # Since n <= 2000, O(n^3) is too slow.
    # However, we only need to find the first triangle.
    # We can iterate a from 1 to n, b from a+1 to n, and check if c exists.
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # Find common neighbors of a and b
                common = adj[a].intersection(adj[b])
                # We need c > b
                valid_c = [c for c in common if c > b]
                if valid_c:
                    print(f"{a} {b} {min(valid_c)}")
                    return

if __name__ == '__main__':
    solve()
