import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A group G = {a, b, c} justifies the failure if for every x in G,
    # the number of individuals in G \ {x} that are NOT in F(x) is exactly 1.
    # Let G = {a, b, c}.
    # For a: count of {b, c} \ F(a) is 1.
    # This means exactly one of (a,b) or (a,c) is NOT in the friction set.
    # In other words, exactly one of (a,b) or (a,c) is an edge in the complement graph.
    # Let E' be the set of edges in the complement graph.
    # The condition is:
    # For a: |{b, c} \ F(a)| = 1  => (a,b) not in F(a) XOR (a,c) not in F(a)
    # This is equivalent to saying:
    # (a,b) in E' XOR (a,c) in E'
    # (b,a) in E' XOR (b,c) in E'
    # (c,a) in E' XOR (c,b) in E'
    # Summing these: each edge in E' within G is counted twice.
    # The condition is satisfied if the subgraph of the complement graph induced by G
    # has degree 1 for every vertex in G.
    # A graph with 3 vertices where every vertex has degree 1 is impossible.
    # Wait, let's re-read: "number of individuals who are present in G but not present in F(x) is exactly one."
    # Let G = {a, b, c}.
    # For x=a: {b, c} \ F(a). Size is 1.
    # This means exactly one of b or c is NOT connected to a in the friction graph.
    # This means exactly one of (a,b) or (a,c) is an edge in the complement graph.
    # Let E' be the complement graph.
    # For a: deg_{G}(a) in E' is 1.
    # For b: deg_{G}(b) in E' is 1.
    # For c: deg_{G}(c) in E' is 1.
    # This is only possible if the complement graph contains a triangle (a,b,c)
    # where each vertex has degree 2, OR if it's a matching.
    # But a triangle has degree 2. A matching of 3 vertices is impossible.
    # Let's re-check:
    # If G = {a, b, c}, the edges in the complement graph within G are E'_G.
    # deg_{G}(a) = 1, deg_{G}(b) = 1, deg_{G}(c) = 1.
    # Sum of degrees = 3. But sum of degrees must be 2 * |E'_G|.
    # 3 is odd, so this is impossible.
    # Re-reading again: "number of individuals who are present in G but not present in F(x) is exactly one."
    # G = {a, b, c}. x is in G.
    # Individuals in G but not in F(x) are:
    # If x=a, individuals are {b, c}.
    # If (a,b) is not in F(a) and (a,c) is in F(a), then count is 1.
    # This means (a,b) is in E' and (a,c) is not in E'.
    # So in the complement graph E', a is connected to exactly one of {b, c}.
    # This means deg_{G}(a) = 1.
    # This is indeed impossible for 3 vertices.
    # Let's re-read carefully: "number of individuals who are present in G but not present in F(x) is exactly one."
    # Does "present in G" include x itself?
    # "For every individual x in G, the number of individuals who are present in G but not present in F(x) is exactly one."
    # If x is in G, x is never in F(x). So x is always "present in G but not in F(x)".
    # So the count is 1 + (number of y in G\{x} such that y is not in F(x)).
    # For the count to be 1, the number of y in G\{x} such that y is not in F(x) must be 0.
    # This means for all y in G\{x}, y must be in F(x).
    # This means (x,y) must be an edge in the friction graph for all x, y in G, x != y.
    # This means G must be a clique in the friction graph.
    # Let's check Example 1: 1-2, 1-3, 2-3, 2-4, 3-4.
    # G={1,2,3}. F(1)={2,3}, F(2)={1,3,4}, F(3)={1,2,4}.
    # x=1: G\{1}={2,3}. Both 2 and 3 are in F(1). Count = 1 (only 1 itself). Correct.
    # x=2: G\{2}={1,3}. Both 1 and 3 are in F(2). Count = 1 (only 2 itself). Correct.
    # x=3: G\{3}={1,2}. Both 1 and 2 are in F(3). Count = 1 (only 3 itself). Correct.
    # So G must be a clique of size 3.
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                for c in range(b + 1, n + 1):
                    if c in adj[a] and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

solve()
