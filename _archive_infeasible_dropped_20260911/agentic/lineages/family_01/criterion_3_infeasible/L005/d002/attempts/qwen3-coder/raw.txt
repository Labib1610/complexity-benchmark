import sys
from collections import defaultdict

def find_triangle(edges, n):
    # Create adjacency list
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # Create a list of neighbors for each node, sorted
    neighbors = {}
    for node in range(1, n+1):
        neighbors[node] = sorted(list(adj[node]))
    
    # For each edge, check if it forms a triangle with any common neighbor
    # We want the lexicographically smallest triangle
    best = None
    
    # Try all edges as the "first" edge of triangle
    for u in range(1, n+1):
        for v in neighbors[u]:
            if v <= u:
                continue  # Ensure u < v to avoid duplicates
            # Check common neighbors of u and v
            common_neighbors = adj[u] & adj[v]
            for w in common_neighbors:
                if w <= u or w <= v:
                    continue  # Ensure u < v < w
                candidate = (u, v, w)
                if best is None or candidate < best:
                    best = candidate
    
    return best

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        edges.append((u, v))
        idx += 2
    
    result = find_triangle(edges, n)
    print(result[0], result[1], result[2])

if __name__ == "__main__":
    main()
