import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # A roster failure is a triangle (a, b, c).
    # We need to find all triangles and pick the lexicographically smallest.
    # To optimize, iterate over edges (u, v) and check common neighbors.
    # Since we need the lexicographically smallest (a, b, c),
    # we can iterate through all possible a, then b, then c.
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    best = None
    
    # Iterate through all possible a
    for a in range(1, n + 1):
        # Iterate through all neighbors b of a such that a < b
        neighbors_a = sorted(list(adj[a]))
        for b in neighbors_a:
            if b < a:
                continue
            
            # Now find c such that (a, c) and (b, c) exist and b < c
            # To minimize lexicographical order, we want the smallest c
            # for the smallest (a, b).
            
            # Intersection of adj[a] and adj[b]
            # We only care about c > b
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    res = tuple(sorted((a, b, c)))
                    if best is None or res < best:
                        best = res
        
        # If we found a triangle starting with 'a', and we have checked all
        # triangles starting with 'a', we can potentially stop if we are sure
        # no smaller 'a' exists. Since we iterate 'a' from 1 to n,
        # the first 'a' that yields a triangle will be the smallest 'a'.
        if best is not None:
            # We need to ensure we found the lexicographically smallest (a, b, c)
            # for this specific 'a'.
            print(f"{best[0]} {best[1]} {best[2]}")
            return

solve()
