import sys

def solve():
    # The problem asks for a group of three individuals {a, b, c} such that
    # for every way to split them into two periods (2 vs 1), the two in the same
    # period have a conflict.
    # This means:
    # (a, b) must have a conflict
    # (b, c) must have a conflict
    # (a, c) must have a conflict
    # In other words, the three individuals must form a triangle (a clique of size 3).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) with a < b.
    # Then check for a common neighbor c such that a < b < c or a < c < b or c < a < b.
    # Actually, the condition is just that {a, b, c} forms a triangle.
    # We want the smallest (a, b, c) where a < b < c.
    
    # Sort edges to help find lexicographical order
    edges.sort()
    
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    #   Iterate b from a+1 to n-1
    #     If (a, b) is an edge:
    #       Check for c > b such that (a, c) and (b, c) are edges.
    
    for a in range(1, n + 1):
        # Sort neighbors to check c in order
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we have edge (a, b). Look for c > b such that (a, c) and (b, c) exist.
            # We want the smallest c.
            for c in neighbors_a[i+1:]:
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
