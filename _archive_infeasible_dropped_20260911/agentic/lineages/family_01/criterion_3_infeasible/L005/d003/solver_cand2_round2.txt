```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A group of three {a, b, c} justifies the failure if:
    # 1. (a, b) is an edge
    # 2. (a, c) is an edge
    # 3. (b, c) is an edge
    # This is simply a triangle in the graph.
    # The problem description says:
    # "selecting any one of the three individuals for the first period and any second 
    # individual from the three for the second period reveals a preliminary report 
    # between them... the remaining third individual is blocked from both periods 
    # because they share a preliminary report with the individual in the first period 
    # and a preliminary report with the individual in the second period."
    # This is exactly the definition of a triangle (a clique of size 3).
    
    # To find the lexicographically first triangle (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, iterate c from b+1 to n
    # Check if c is in adj[a] and c is in adj[b]
    
    for a in range(1, n + 1):
        # Sort neighbors to potentially speed up or just iterate
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # To be efficient, check the smaller set
            if len(adj[b]) < len(neighbors_a) - i:
                for c in sorted(adj[b]):
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for j in range(i + 1, len(neighbors_a)):
                    c = neighbors_a[j]
                    if c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

solve()
```
