```python
import sys

def solve():
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr += 1
    m = int(input[ptr])
    ptr += 1
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    deg = [0] * (n + 1)
    
    for _ in range(m):
        u = int(input[ptr])
        ptr += 1
        v = int(input[ptr])
        ptr += 1
        adj[u][v] = True
        adj[v][u] = True
        deg[u] += 1
        deg[v] += 1
    
    # Find lexicographically smallest triangle (all three edges exist)
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if adj[a][b]:
                for c in range(b + 1, n + 1):
                    if adj[a][c] and adj[b][c]:
                        print(a, b, c)
                        return
    
    # Find lexicographically smallest complement triangle (no edges among them)
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if not adj[a][b]:
                for c in range(b + 1, n + 1):
                    if not adj[a][c] and not adj[b][c]:
                        print(a, b, c)
                        return

if __name__ == "__main__":
    solve()
```
