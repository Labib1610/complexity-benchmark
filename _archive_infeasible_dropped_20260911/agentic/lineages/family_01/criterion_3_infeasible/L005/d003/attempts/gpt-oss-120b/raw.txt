import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    # adjacency as bitsets (0‑based vertices)
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    full_mask = (1 << n) - 1
    for a in range(n):
        row_a = adj[a]
        for b in range(a + 1, n):
            if not (row_a >> b) & 1:   # edge a‑b absent
                continue
            # vertices that are adjacent to both a and b
            common = row_a & adj[b]
            # keep only vertices with index > b
            mask = full_mask ^ ((1 << (b + 1)) - 1)
            cand = common & mask
            if cand:
                # lowest set bit gives smallest c > b
                c = (cand & -cand).bit_length() - 1
                # output 1‑based, sorted (a<b<c)
                print(a + 1, b + 1, c + 1)
                return

if __name__ == "__main__":
    solve()
