import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Create adjacency matrix for quick lookup
    adj_matrix = [[False] * (n+1) for _ in range(n+1)]
    for u, v in edges:
        adj_matrix[u][v] = True
        adj_matrix[v][u] = True
    
    # Iterate through all triplets (i, j, k) in lexicographical order
    for i in range(1, n+1):
        for j in range(i+1, n+1):
            if not adj_matrix[i][j]:
                continue
            for k in range(j+1, n+1):
                if not adj_matrix[i][k] or not adj_matrix[j][k]:
                    continue
                
                # Check if this triangle is a minimal failing subset
                # There are 3 ways to assign periods:
                # 1. i in period 1, j,k in period 2
                # 2. j in period 1, i,k in period 2
                # 3. k in period 1, i,j in period 2
                
                # For each assignment, check if conflicting pairs share period
                fail = True
                
                # Assignment 1: i alone in period 1, j,k together in period 2
                # Conflicts within period 2: (j,k)
                if not adj_matrix[j][k]:
                    fail = False
                
                if fail:
                    # Assignment 2: j alone in period 1, i,k together in period 2
                    # Conflicts within period 2: (i,k)
                    if not adj_matrix[i][k]:
                        fail = False
                
                if fail:
                    # Assignment 3: k alone in period 1, i,j together in period 2
                    # Conflicts within period 2: (i,j)
                    if not adj_matrix[i][j]:
                        fail = False
                
                if fail:
                    print(i, j, k)
                    return

if __name__ == "__main__":
    main()
