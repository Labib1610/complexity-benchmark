import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    c = [0] * (n + 1)
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
        c[u] += 1
        c[v] += 1
        edges.append((min(u, v), max(u, v)))
    
    # The interference checksum for {a, b, c} is:
    # sum_{e in E} (k_e ^ 1)
    # k_e is the number of endpoints of e in {a, b, c}.
    # k_e = 0: (0 ^ 1) = 1
    # k_e = 1: (1 ^ 1) = 0
    # k_e = 2: (2 ^ 1) = 3
    # Sum = (count of edges with 0 endpoints) * 1 + (count of edges with 1 endpoint) * 0 + (count of edges with 2 endpoints) * 3
    # Let E_0, E_1, E_2 be the counts. E_0 + E_1 + E_2 = m.
    # Sum = E_0 + 3 * E_2.
    # E_2 is the number of edges within {a, b, c}.
    # E_1 is the number of edges with exactly one endpoint in {a, b, c}.
    # E_1 = (C_a + C_b + C_c) - 2 * E_2.
    # E_0 = m - E_1 - E_2 = m - (C_a + C_b + C_c) + E_2.
    # Sum = (m - (C_a + C_b + C_c) + E_2) + 3 * E_2 = m - (C_a + C_b + C_c) + 4 * E_2.
    # We want Sum = m - (C_a + C_b + C_c) + 12.
    # This simplifies to 4 * E_2 = 12, so E_2 = 3.
    # The condition is simply that the three employees form a triangle.
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # Find common neighbors c > b
                common = adj[a].intersection(adj[b])
                for c_val in sorted(list(common)):
                    if c_val > b:
                        print(f"{a} {b} {c_val}")
                        return

solve()
