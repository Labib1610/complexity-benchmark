import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all edges (u, v) as potential (a, b).
    # Since we want the smallest a, then b, then c, we sort edges.
    edges.sort()
    
    best = None
    
    for u, v in edges:
        # Potential triplet (u, v, w) where u < v < w
        # Check all neighbors of u that are also neighbors of v
        # To keep it efficient, iterate over the smaller adjacency list
        if len(adj[u]) < len(adj[v]):
            candidates = adj[u]
            other = v
        else:
            candidates = adj[v]
            other = u
            
        for w in candidates:
            if w == other:
                continue
            
            # Ensure w is connected to the other node
            if w in adj[other]:
                # Found a triangle (u, v, w)
                # Sort to get (a, b, c) such that a < b < c
                a, b, c = sorted([u, v, w])
                if best is None or (a, b, c) < best:
                    best = (a, b, c)
                    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
