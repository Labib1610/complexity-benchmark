import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
    
    # A group of 3 {a, b, c} has 0 conflict-free assignments if and only if
    # it forms a triangle (K3) in the incompatibility graph.
    # Proof:
    # If it's a triangle, any assignment of 3 people to 2 shifts must have
    # at least two people in the same shift (Pigeonhole Principle).
    # Since all three are connected, those two will have a conflict.
    # Thus, 0 assignments are conflict-free.
    # If it's not a triangle, there is at least one pair not connected,
    # allowing at least one conflict-free assignment.
    
    # We need the lexicographically first triangle {a, b, c} with a < b < c.
    for a in range(1, n + 1):
        adj_a = adj[a]
        # Sort neighbors to ensure we find the lexicographically smallest b and c
        sorted_adj_a = sorted([x for x in adj_a if x > a])
        for i in range(len(sorted_adj_a)):
            b = sorted_adj_a[i]
            adj_b = adj[b]
            # Check for c > b such that c is connected to both a and b
            # Intersection of adj_a and adj_b
            common = sorted([x for x in adj_a if x > b and x in adj_b])
            if common:
                print(f"{a} {b} {common[0]}")
                return

solve()
