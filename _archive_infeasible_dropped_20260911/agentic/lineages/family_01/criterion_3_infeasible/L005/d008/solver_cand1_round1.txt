```python
import sys

def solve():
    # A parity failure is a set of three logs (u1, v1), (u2, v2), (u3, v3)
    # such that their XOR sum is 0.
    # Let the logs be represented as vectors in GF(2)^n.
    # The XOR sum is 0 if and only if each index appears an even number of times.
    # With 3 logs, each having 2 ones, there are 6 ones total.
    # For the XOR sum to be 0, each index must appear exactly twice.
    # This means the three logs must form a triangle: (u, v), (v, w), (w, u).
    # The bitwise OR sum of these three logs will have 1s at positions u, v, w.
    # We need to find a triangle (u, v, w) such that the sorted tuple (min(u,v,w), mid(u,v,w), max(u,v,w))
    # is lexicographically smallest.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    best = None
    
    # To find the lexicographically smallest (a, b, c), we iterate through possible a, b, c.
    # Since n <= 2000, we can iterate through all edges (u, v) and check for common neighbors.
    # To optimize, iterate through all nodes as the smallest vertex 'a'.
    for a in range(1, n + 1):
        neighbors = sorted(list(adj[a]))
        for i in range(len(neighbors)):
            b = neighbors[i]
            # We need a third node c such that (a, c) and (b, c) exist.
            # c must be > b to ensure a < b < c.
            # Check intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    res = (a, b, c)
                    if best is None or res < best:
                        best = res
        
        # If we found a triangle starting with 'a', and we have checked all triangles
        # starting with 'a', we can potentially stop if we are sure no smaller 'a' exists.
        # But we iterate 'a' from 1 to n, so the first 'a' that yields a result
        # might not be the global minimum if there are multiple triangles with the same 'a'.
        # Actually, the current approach finds the best for each 'a'.
        
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
