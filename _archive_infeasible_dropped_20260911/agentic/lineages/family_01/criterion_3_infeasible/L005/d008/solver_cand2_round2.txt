import sys

def solve():
    # The problem asks for three individuals (a, b, c) such that there are 
    # three logs involving them, and these three logs involve exactly these 
    # three individuals. This is equivalent to finding a triangle in the 
    # incompatibility graph (a cycle of length 3).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically first triangle (a, b, c) with a < b < c:
    # Iterate through all possible 'a' from 1 to n.
    # For each 'a', iterate through all neighbors 'b' of 'a' (b > a).
    # For each 'b', check if there exists a 'c' such that 'c' is a neighbor 
    # of both 'a' and 'b'. Since we want the lexicographically smallest (a, b, c),
    # we sort the neighbors and check.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c > b such that c is in adj[a] and adj[b]
            # To optimize, intersect the sorted neighbor lists
            neighbors_b = adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
