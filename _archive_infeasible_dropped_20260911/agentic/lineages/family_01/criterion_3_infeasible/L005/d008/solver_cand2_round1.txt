```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # A group of 3 {a, b, c} is a certifying group if there are 3 logs
    # involving only these 3. This means the subgraph induced by {a, b, c}
    # must have exactly 3 edges. This is equivalent to saying {a, b, c}
    # forms a triangle (a-b, b-c, c-a).
    
    # We need to find the lexicographically smallest triangle (a, b, c)
    # where a < b < c.
    
    # Sort nodes to iterate in lexicographical order
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        for i in range(len_a):
            b = neighbors_a[i]
            # We need a < b < c. Since a is fixed, we need b > a.
            if b <= a:
                continue
            
            # Check for c such that (a, c) and (b, c) are edges
            # c must be > b
            # Intersection of adj[a] and adj[b]
            # To optimize, iterate over the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
                for c in candidates:
                    if c > b and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                candidates = adj[b]
                for c in candidates:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

solve()
```
