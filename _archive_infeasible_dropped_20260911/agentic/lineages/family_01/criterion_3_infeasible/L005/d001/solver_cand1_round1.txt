import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # A report (u, v) is a confirmed incompatibility if there exists w
    # such that (u, w) and (v, w) are also reports.
    # This means u, v, w form a triangle in the original graph.
    # A group of three (a, b, c) is a deadlock if all three edges exist.
    # The problem asks for a triangle (a, b, c) in the graph.
    
    # Sort edges to process lexicographically
    edges.sort()
    
    # To find the lexicographically first triangle (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists c > b
    # such that (a, c) and (b, c) are edges.
    
    # Optimization: use bitsets or intersection of adjacency lists
    # Since n <= 2000, we can use bitsets (integers in Python)
    adj_bits = [0] * (n + 1)
    for u, v in edges:
        adj_bits[u] |= (1 << v)
        adj_bits[v] |= (1 << u)
        
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if (adj_bits[a] >> b) & 1:
                # Check for common neighbors c > b
                common = adj_bits[a] & adj_bits[b]
                # We only care about c > b
                # Mask out bits <= b
                common &= ~((1 << (b + 1)) - 1)
                if common:
                    # Find the smallest c
                    c = (common & -common).bit_length() - 1
                    print(f"{a} {b} {c}")
                    return

solve()
