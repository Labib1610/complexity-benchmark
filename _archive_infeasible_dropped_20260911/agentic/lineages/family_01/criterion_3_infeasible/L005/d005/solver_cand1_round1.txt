import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # Let the group be {a, b, c}.
    # Step 1: Retrieve all documents containing at least one of {a, b, c}.
    # Let S be the set of these documents.
    # Step 2: Each individual signs documents they are in.
    # Total signatures = (count of docs in S containing a) + (count of docs in S containing b) + (count of docs in S containing c).
    # Since every doc in S contains at least one of {a, b, c}, the total signatures is:
    # (number of docs in S containing exactly one of {a, b, c}) * 1 +
    # (number of docs in S containing exactly two of {a, b, c}) * 2 +
    # (number of docs in S containing exactly three of {a, b, c}) * 3.
    # We want total signatures = |S| + 3.
    # Let x be the number of docs in S containing exactly one of {a, b, c}.
    # Let y be the number of docs in S containing exactly two of {a, b, c}.
    # Let z be the number of docs in S containing exactly three of {a, b, c}.
    # |S| = x + y + z.
    # Total signatures = x + 2y + 3z.
    # We want x + 2y + 3z = x + y + z + 3 => y + 2z = 3.
    # Since z can only be 0 or 1 (only one doc can contain {a, b, c} if it exists),
    # Case 1: z = 1, y = 1.
    # Case 2: z = 0, y = 3.
    
    # Precompute degrees for efficiency
    deg = [len(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            # Find common neighbors of a and b
            common = adj[a].intersection(adj[b])
            
            # Case 1: z=1, y=1. {a, b, c} forms a triangle (z=1).
            # y=1 means there is exactly one other edge connected to {a, b, c}
            # that is not part of the triangle.
            for c in common:
                if c > b:
                    # Triangle {a, b, c} exists.
                    # y is the number of edges connecting exactly two of {a, b, c}.
                    # The edges are (a,b), (b,c), (a,c).
                    # y = (edges connected to a excluding b,c) + (edges connected to b excluding a,c) + (edges connected to c excluding a,b)
                    # y = (deg[a]-2) + (deg[b]-2) + (deg[c]-2)
                    if (deg[a] - 2) + (deg[b] - 2) + (deg[c] - 2) == 1:
                        print(f"{a} {b} {c}")
                        return
            
            # Case 2: z=0, y=3. No triangle.
            # y is the number of edges connecting exactly two of {a, b, c}.
            # y = (edges between a,b) + (edges between b,c) + (edges between a,c)
            # y = 1 + (1 if c in adj[b]) + (1 if c in adj[a])
            # Since z=0, c is not in adj[a] and c is not in adj[b].
            # So y = 1 + 0 + 0 = 1. This doesn't satisfy y=3.
            # Wait, y is the number of edges connecting exactly two of {a, b, c}.
            # If z=0, y = (1 if b in adj[a]) + (1 if c in adj[b]) + (1 if c in adj[a])
            # y = 1 + (1 if c in adj[b]) + (1 if c in adj[a])
            # For y=3, we need c in adj[b] and c in adj[a], which means z=1.
            # So only Case 1 is possible.
            pass

solve()
