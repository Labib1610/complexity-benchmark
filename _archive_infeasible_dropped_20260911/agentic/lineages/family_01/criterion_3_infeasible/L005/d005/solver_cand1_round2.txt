```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    L = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        L[u] += 1
        L[v] += 1
        idx += 2
        
    # The condition: L_a + L_b + L_c + (m - (logs involving a, b, or c)) = m + 3
    # Let S be the set {a, b, c}.
    # Logs involving none of {a, b, c} = m - (logs involving at least one of a, b, c)
    # Let E(S) be the number of logs involving at least one of a, b, c.
    # The condition is: L_a + L_b + L_c + m - E(S) = m + 3
    # L_a + L_b + L_c - E(S) = 3
    # E(S) = (L_a + L_b + L_c) - 3
    # Let k be the number of edges within the triangle {a, b, c}.
    # E(S) = L_a + L_b + L_c - (edges between a,b, c, a) - (edges between a,b, c, others)
    # Actually, E(S) = L_a + L_b + L_c - (edges between a,b, c) - (edges between a,b, c) + k
    # Wait, inclusion-exclusion:
    # E(S) = L_a + L_b + L_c - (edges between {a,b,c} and {a,b,c}) - (edges between {a,b,c} and others)
    # The number of edges involving at least one of {a, b, c} is:
    # (L_a + L_b + L_c) - (edges between a,b, c) - (edges between a,b, c) + k
    # Simplified: E(S) = L_a + L_b + L_c - (edges between {a,b,c} and {a,b,c})
    # Let k be the number of edges among {a, b, c}.
    # E(S) = L_a + L_b + L_c - k
    # The condition L_a + L_b + L_c - E(S) = 3 becomes:
    # L_a + L_b + L_c - (L_a + L_b + L_c - k) = 3
    # k = 3
    # So the condition is simply that {a, b, c} must form a triangle.
    
    # We need the lexicographically first triangle.
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # Check for c > b such that c is connected to both a and b
                # Intersection of adj[a] and adj[b]
                common = adj[a].intersection(adj[b])
                for c in sorted(common):
                    if c > b:
                        print(f"{a} {b} {c}")
                        return

solve()
```
