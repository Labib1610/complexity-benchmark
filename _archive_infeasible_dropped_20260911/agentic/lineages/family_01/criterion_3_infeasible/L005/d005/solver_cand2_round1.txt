import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2

    # Let the group be {a, b, c}.
    # Let E_in be the number of edges with both endpoints in {a, b, c}.
    # Let E_out be the number of edges with exactly one endpoint in {a, b, c}.
    # Total weight = 3 * E_in + 1 * E_out.
    # Number of reports with weight > 0 = E_in + E_out.
    # Condition: 3 * E_in + E_out = (E_in + E_out) + 6
    # 2 * E_in = 6 => E_in = 3.
    # This means the group {a, b, c} must form a triangle (a, b, c).
    
    # Iterate through all triangles (a, b, c) with a < b < c
    # To find the lexicographically first, we iterate a from 1 to n-2,
    # b from a+1 to n-1, c from b+1 to n.
    
    for a in range(1, n - 1):
        for b in range(a + 1, n):
            if b in adj[a]:
                # Check for c such that (a, c) and (b, c) exist
                # Intersection of adj[a] and adj[b]
                common = adj[a].intersection(adj[b])
                if common:
                    sorted_common = sorted(list(common))
                    for c in sorted_common:
                        if c > b:
                            # Found a triangle (a, b, c)
                            # Check if it satisfies the condition
                            # E_in = 3 (the triangle edges)
                            # E_out = (degree(a) - 2) + (degree(b) - 2) + (degree(c) - 2)
                            # Total weight = 3*3 + E_out = 9 + E_out
                            # Reports with weight > 0 = 3 + E_out
                            # Condition: 9 + E_out = (3 + E_out) + 6 => 9 + E_out = 9 + E_out
                            # This is always true for any triangle!
                            # The problem asks for the lexicographically first triangle.
                            print(f"{a} {b} {c}")
                            return

solve()
