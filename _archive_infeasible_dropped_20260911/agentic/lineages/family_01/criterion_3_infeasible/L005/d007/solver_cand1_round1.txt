import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # A group of 3 {a, b, c} has 3 edges if and only if it forms a triangle.
    # We need the lexicographically smallest {a, b, c} such that a < b < c.
    # Iterate through all edges (u, v) and check for a common neighbor w.
    
    best = None
    
    # To find the lexicographically smallest, we iterate a from 1 to n-2
    for a in range(1, n + 1):
        # Optimization: if we already found a solution with a smaller 'a', we can stop
        if best and a > best[0]:
            break
            
        # Sort neighbors to help find b and c in order
        neighbors_a = sorted(list(adj[a]))
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if best and (a > best[0] or (a == best[0] and b >= best[1])):
                continue
            
            # Find common neighbors of a and b
            # Since we need a < b < c, we only care about w > b
            for c in adj[b]:
                if c > b and c in adj[a]:
                    # Found a triangle {a, b, c}
                    candidate = (a, b, c)
                    if best is None or candidate < best:
                        best = candidate
    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
