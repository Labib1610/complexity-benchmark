```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    # Create a set of edges for O(1) lookup
    edge_set = set(edges)
    
    # For each edge, check if it's part of a triangle
    for u, v in edges:
        # Check common neighbors of u and v
        for w in adj[u]:
            if w != v and w in adj[v]:
                # Found a triangle: u, v, w
                # Ensure correct order
                a, b, c = sorted([u, v, w])
                print(a, b, c)
                return

if __name__ == "__main__":
    main()
```
