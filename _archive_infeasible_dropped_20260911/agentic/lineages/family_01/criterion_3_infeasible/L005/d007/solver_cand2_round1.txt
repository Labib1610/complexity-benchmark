import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A group of 3 {a, b, c} fails the 2-period assignment if and only if
    # it contains at least 3 edges (i.e., it forms a triangle).
    # If a triangle exists, any assignment of 3 people to 2 periods
    # will force at least two people to be in the same period.
    # If those two are connected, the assignment is invalid.
    # In a triangle (a, b, c), any assignment of 3 people to 2 periods
    # must put at least two people in the same period. Since all pairs
    # in a triangle are connected, any assignment is invalid.
    
    # We need the lexicographically smallest triangle (a, b, c) with a < b < c.
    # Iterate through all possible a, then b, then c.
    for a in range(1, n + 1):
        # Sort neighbors to ensure we check b and c in increasing order
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Check for common neighbors c such that c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in sorted(list(common)):
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
