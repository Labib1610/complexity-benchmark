import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # A valid proof of 3 logs (u1, v1), (u2, v2), (u3, v3)
    # such that every involved individual is in at least 2 logs.
    # This structure is exactly a triangle (3-cycle).
    # If we have a triangle (a, b, c), the logs are (a, b), (b, c), (a, c).
    # Individual 'a' is in (a, b) and (a, c) -> 2 logs.
    # Individual 'b' is in (a, b) and (b, c) -> 2 logs.
    # Individual 'c' is in (b, c) and (a, c) -> 2 logs.
    # This satisfies the condition.
    
    # We need to find the lexicographically smallest (a, b, c) such that
    # a < b < c and (a, b), (b, c), (a, c) are all edges.
    
    # To find the lexicographically smallest, we iterate a from 1 to n,
    # b from a+1 to n, and check if c exists such that (a, c) and (b, c) exist.
    
    for a in range(1, n + 1):
        # Sort neighbors to ensure we find the smallest b and c
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Find common neighbors of a and b
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
