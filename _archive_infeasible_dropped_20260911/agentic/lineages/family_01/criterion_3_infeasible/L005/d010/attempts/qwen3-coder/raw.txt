import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    edge_set = set()
    adj = defaultdict(set)
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        edge_set.add((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges for consistent processing
    edges.sort()
    
    # We need to find three edges such that every node involved is in at least two edges
    # That means we're looking for a triangle in the graph
    
    # Let's iterate through all edges and for each edge (u,v), look for a third node w
    # such that (u,w) and (v,w) also exist
    
    found = False
    result = None
    
    for i in range(m):
        u, v = edges[i]
        if found:
            break
        # Get common neighbors of u and v
        common_neighbors = adj[u] & adj[v]
        for w in common_neighbors:
            if found:
                break
            # We have edges (u,v), (u,w), (v,w)
            # Check if all three edges exist (they should since w is in both adj[u] and adj[v])
            if (u < v and v < w):  # Ensure a < b < c
                a, b, c = u, v, w
            elif (u < w and w < v):
                a, b, c = u, w, v
            elif (w < u and u < v):
                a, b, c = w, u, v
            elif (v < u and u < w):
                a, b, c = v, u, w
            elif (v < w and w < u):
                a, b, c = v, w, u
            else:
                a, b, c = w, v, u
            
            if (a, b) in edge_set and (a, c) in edge_set and (b, c) in edge_set:
                if not found or (a, b, c) < result:
                    result = (a, b, c)
                    found = True
    
    # If not found yet, we need to check other combinations properly
    if not found:
        # Build adjacency list with sorted neighbors for deterministic iteration
        adj_list = {}
        for node in adj:
            adj_list[node] = sorted(list(adj[node]))
        
        best = None
        # For each node, look at pairs of its neighbors
        for u in sorted(adj_list.keys()):
            neighbors = adj_list[u]
            for i in range(len(neighbors)):
                v = neighbors[i]
                for j in range(i+1, len(neighbors)):
                    w = neighbors[j]
                    # Now we have u connected to both v and w
                    # Check if v and w are connected
                    if w in adj[v]:
                        # We have a triangle: u-v, u-w, v-w
                        nodes = sorted([u, v, w])
                        candidate = tuple(nodes)
                        if best is None or candidate < best:
                            best = candidate
        
        result = best
    
    print(result[0], result[1], result[2])

if __name__ == "__main__":
    main()
