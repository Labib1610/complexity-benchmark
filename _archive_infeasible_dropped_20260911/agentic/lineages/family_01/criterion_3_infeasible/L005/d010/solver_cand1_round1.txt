```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[u] will store the set of neighbors of u
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # The dot product of two sequences is the number of positions where both have 1.
    # A position k has 1 for u and v if the k-th log is (u, v).
    # Let S_u be the set of logs involving u.
    # The dot product of u and v is |S_u \cap S_v|.
    # Since each log is a pair (u, v), |S_u \cap S_v| is 1 if (u, v) is a log, 0 otherwise.
    # The condition is: dot(a, b) + dot(b, c) + dot(a, c) == 3.
    # This means all three pairs (a, b), (b, c), and (a, c) must be in the logs.
    # This is equivalent to finding a triangle in the graph.
    
    # We need to find the lexicographically smallest (a, b, c) such that a < b < c
    # and (a, b), (b, c), (a, c) are all edges in the graph.
    
    for a in range(1, n + 1):
        adj_a = adj[a]
        # Sort neighbors to ensure we check b and c in increasing order
        sorted_adj_a = sorted(list(adj_a))
        len_a = len(sorted_adj_a)
        for i in range(len_a):
            b = sorted_adj_a[i]
            if b <= a:
                continue
            
            # We need c such that c > b and c is in both adj[a] and adj[b]
            adj_b = adj[b]
            # Intersection of adj_a and adj_b
            common = sorted(list(adj_a.intersection(adj_b)))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
