```python
import sys

def solve():
    # The problem asks for three edges (u1, v1), (u2, v2), (u3, v3) such that
    # every vertex involved appears an even number of times.
    # Let the edges be (a, b), (b, c), (c, a).
    # The vertices involved are a, b, b, c, c, a.
    # Each vertex appears exactly twice.
    # This corresponds to a triangle in the graph.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # We need to find a triangle (a, b, c) with a < b < c.
    # To find the lexicographically smallest (a, b, c), we iterate
    # through possible values of a, then b, then c.
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    # To find the smallest triangle efficiently:
    # Iterate through all edges (u, v) and check for a common neighbor w.
    # Since we need the lexicographically smallest (a, b, c),
    # we can iterate a from 1 to n, then b from a+1 to n,
    # then check if there exists c > b such that (a, c) and (b, c) are edges.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        for b in adj[a]:
            if b <= a:
                continue
            # Now we have edge (a, b). Look for c > b such that (a, c) and (b, c) exist.
            # Intersection of adj[a] and adj[b]
            set_a = set(adj[a])
            for c in adj[b]:
                if c > b and c in set_a:
                    print(f"{a} {b} {c}")
                    return

solve()
```
