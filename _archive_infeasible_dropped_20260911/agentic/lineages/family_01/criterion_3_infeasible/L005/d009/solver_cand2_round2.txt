import sys

def solve():
    # The problem asks for three logs (u1, v1), (u2, v2), (u3, v3) such that
    # every individual involved appears an even number of times.
    # Let the set of individuals be {a, b, c}.
    # Each log involves 2 people. Total appearances = 3 * 2 = 6.
    # If we have 3 individuals, each must appear an even number of times.
    # The only way to distribute 6 appearances among 3 people such that each
    # is even is for each person to appear exactly 2 times.
    # This means the three logs must be (a, b), (b, c), and (c, a).
    # This is equivalent to finding a triangle in the graph.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) where a < b < c
    # such that (a, b), (b, c), (a, c) are edges.
    # We iterate through all possible a, then b, then c.
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    # To optimize, iterate over a, then b, then check if c exists.
    # Since we need the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, b from a+1 to n, and check if there exists c > b
    # such that (a, c) and (b, c) are edges.
    
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        for b in neighbors_a:
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of neighbors_a and neighbors_b
            neighbors_b = adj[b]
            # Find common neighbors c > b
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
