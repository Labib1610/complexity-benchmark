import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Try to find a triangle (three edges forming a cycle of length 3)
    # This is the most common case for valid proof
    for u, v in edges:
        # Find common neighbors of u and v
        # Iterate through the smaller adjacency list
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        for w in adj[u]:
            if w in adj[v]:
                # Found triangle u-v-w
                res = sorted([u, v, w])
                print(f"{res[0]} {res[1]} {res[2]}")
                return
    
    # If no triangle found, we need to find a path of length 2
    # where the endpoints are connected (forming a cycle of length 3)
    # But since we already checked all edges for triangles above,
    # we need to check for paths where endpoints are connected
    
    # Actually, the triangle check above covers all cases where
    # three edges form a cycle. Let me reconsider...
    
    # The parity condition means: collect all IDs from 3 logs,
    # each ID appears even times. With 3 edges (6 IDs total),
    # each ID appears 0, 2, 4, or 6 times.
    # Since each edge has 2 distinct IDs, and we have 3 edges,
    # the only way is: 3 distinct IDs, each appearing exactly 2 times.
    # This forms a triangle!
    
    # So the triangle check should find all valid proofs.
    # The problem guarantees at least one exists.
    
    # If we reach here, something went wrong (shouldn't happen per problem)
    # But let's add a fallback check
    
    # Re-check with different approach: for each vertex, check pairs of neighbors
    for u in range(1, n + 1):
        neighbors = list(adj[u])
        for i in range(len(neighbors)):
            for j in range(i + 1, len(neighbors)):
                v = neighbors[i]
                w = neighbors[j]
                if w in adj[v]:
                    res = sorted([u, v, w])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return
    
    # Should not reach here given problem guarantees
    pass

if __name__ == "__main__":
    solve()
