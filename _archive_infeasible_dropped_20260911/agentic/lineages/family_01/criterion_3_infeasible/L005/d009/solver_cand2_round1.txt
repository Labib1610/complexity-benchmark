```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A subset of three {a, b, c} is unassignable if and only if
    # it contains an odd cycle. Since the subset size is 3,
    # the only odd cycle possible is a triangle (a-b, b-c, c-a).
    # If {a, b, c} forms a triangle, any assignment of 2 periods
    # will force at least two people into the same period,
    # and since they are all connected, that pair will conflict.
    
    # We need the lexicographically smallest {a, b, c} such that
    # a < b < c and they form a triangle.
    
    # Iterate through all possible a, b, c
    # To optimize, iterate a from 1 to n-2
    for a in range(1, n + 1):
        # Get neighbors of a
        neighbors_a = sorted(list(adj[a]))
        if len(neighbors_a) < 2:
            continue
            
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Check for c such that a-c and b-c exist
            # c must be > b
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    # Found the lexicographically smallest triangle
                    # because we iterate a, then b, then c in increasing order
                    print(f"{a} {b} {c}")
                    return

solve()
```
