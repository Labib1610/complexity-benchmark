import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # For each protein x, store the list of catalysts z that appear with it
    protein_to_catalysts = [[] for _ in range(k + 1)]
    # For each protein x, store its associated receptor y
    protein_to_receptor = [0] * (k + 1)
    
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        protein_to_catalysts[x].append(z)
        protein_to_receptor[x] = y
    
    # Build bipartite graph between proteins (left) and catalysts (right)
    # We need to find a perfect matching
    
    # Adjacency list for each protein
    adj = protein_to_catalysts
    
    # Hopcroft-Karp algorithm
    INF = float('inf')
    
    # pairU[u] = v matched with u (protein -> catalyst), 0 if unmatched
    pairU = [0] * (k + 1)
    # pairV[v] = u matched with v (catalyst -> protein), 0 if unmatched
    pairV = [0] * (k + 1)
    # dist[u] = distance in BFS for protein u
    dist = [0] * (k + 1)
    
    from collections import deque
    
    def bfs():
        queue = deque()
        for u in range(1, k + 1):
            if pairU[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if pairV[v] == 0:
                        # Unmatched catalyst, can reach dummy node 0
                        if dist[0] == INF:
                            dist[0] = dist[u] + 1
                    else:
                        # Matched catalyst, follow to its pair
                        if dist[pairV[v]] == INF:
                            dist[pairV[v]] = dist[u] + 1
                            queue.append(pairV[v])
        return dist[0] != INF
    
    def dfs(u):
        for v in adj[u]:
            if pairV[v] == 0 or (dist[pairV[v]] == dist[u] + 1 and dfs(pairV[v])):
                pairU[u] = v
                pairV[v] = u
                return True
        dist[u] = INF
        return False
    
    matching_size = 0
    while bfs():
        for u in range(1, k + 1):
            if pairU[u] == 0:
                if dfs(u):
                    matching_size += 1
    
    if matching_size != k:
        print("NO")
        return
    
    print("YES")
    for x in range(1, k + 1):
        y = protein_to_receptor[x]
        z = pairU[x]
        print(x, y, z)

if __name__ == "__main__":
    solve()
