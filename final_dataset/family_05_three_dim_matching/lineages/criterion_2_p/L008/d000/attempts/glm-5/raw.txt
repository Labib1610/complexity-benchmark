```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    idx = 2
    # For each specialist x, collect the set of allowed time slots z.
    # Machine y is fixed per specialist and unique across specialists.
    spec_slots = [set() for _ in range(k + 1)]
    spec_machine = [0] * (k + 1)
    for _ in range(t):
        x = int(data[idx]); y = int(data[idx+1]); z = int(data[idx+2])
        idx += 3
        spec_slots[x].add(z)
        spec_machine[x] = y

    # Specialists with no allowed time slots cannot be assigned.
    for x in range(1, k + 1):
        if not spec_slots[x]:
            print("NO")
            return

    # Hopcroft-Karp bipartite matching: left = specialists, right = time slots.
    pair_u = [0] * (k + 1)
    pair_v = [0] * (k + 1)
    dist = [0] * (k + 1)
    INF = 10**9
    from collections import deque

    def bfs():
        queue = deque()
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in spec_slots[u]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for v in spec_slots[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True

    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1

    if matching == k:
        print("YES")
        out = []
        for x in range(1, k + 1):
            z = pair_u[x]
            y = spec_machine[x]
            out.append(f"{x} {y} {z}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        print("NO")

solve()
```
