from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    plans = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        plans.append((x, y, z))
        index += 3
    
    # Since each driver is assigned to exactly one truck and no two drivers share a truck,
    # we can think of this as a bipartite matching problem between drivers and routes.
    # For each plan (driver x, truck y, route z), since truck y is fixed for driver x,
    # we can represent it as an edge from driver x to route z.
    
    # Build a graph: for each driver, which routes they can do
    graph = defaultdict(list)
    plan_map = {}
    
    for i in range(t):
        x, y, z = plans[i]
        graph[x].append(z)
        plan_map[(x, z)] = (x, y, z)
    
    # Now we want to match each driver to exactly one route such that all routes are covered
    # This is a maximum bipartite matching problem where we need a perfect matching
    
    # Since k drivers and k routes, we need a perfect matching
    
    # Let's implement Hopcroft-Karp for maximum bipartite matching
    # But since k <= 1200, a simpler approach with DFS should work
    
    def dfs(u, match_route, match_driver, visited):
        for v in graph[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            if match_route[v] == -1 or dfs(match_route[v], match_route, match_driver, visited):
                match_route[v] = u
                match_driver[u] = v
                return True
        return False
    
    # Initialize matching arrays
    match_route = [-1] * (k + 1)  # route z matched to which driver
    match_driver = [-1] * (k + 1) # driver x matched to which route
    
    matched = 0
    for u in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(u, match_route, match_driver, visited):
            matched += 1
    
    if matched != k:
        print("NO")
        return
    
    print("YES")
    for u in range(1, k + 1):
        v = match_driver[u]
        print(*plan_map[(u, v)])

solve()
