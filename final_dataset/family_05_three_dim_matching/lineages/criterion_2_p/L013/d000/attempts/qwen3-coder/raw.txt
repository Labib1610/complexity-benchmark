from collections import defaultdict, deque

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
        
    # Since each agent has exactly one alias and all aliases are distinct,
    # we can think of this as a bipartite matching problem between
    # agents (and their fixed aliases) and frequencies.
    
    # For each agent x, we have a fixed alias y, and a set of allowed frequencies z
    # We need to match each agent to a distinct frequency
    
    # Build adjacency list for agents to frequencies
    adj = defaultdict(list)
    edges = []
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        adj[x].append(z)
        edges.append((x, y, z))
    
    # Check if each agent has at least one frequency option
    for i in range(1, k + 1):
        if i not in adj:
            print("NO")
            return
    
    # Use BFS-based approach to find maximum matching
    # This is essentially finding a maximum bipartite matching
    # between agents (1..k) and frequencies (1..k)
    
    def bfs_match():
        # Build residual graph
        # For each agent, keep track of which frequencies they can use
        # Perform Ford-Fulkerson or use Hungarian algorithm concepts
        
        # But since it's simpler, let's model it as bipartite matching
        # and use a standard algorithm
        
        # match[freq] = agent assigned to this frequency, or -1 if none
        match_freq_to_agent = [-1] * (k + 1)
        match_agent_to_freq = [-1] * (k + 1)
        
        def dfs(agent, visited):
            for freq in adj[agent]:
                if visited[freq]:
                    continue
                visited[freq] = True
                
                # If frequency is free or we can reassign its current agent
                if match_freq_to_agent[freq] == -1 or dfs(match_freq_to_agent[freq], visited):
                    match_freq_to_agent[freq] = agent
                    match_agent_to_freq[agent] = freq
                    return True
            return False
        
        matched = 0
        for agent in range(1, k + 1):
            visited = [False] * (k + 1)
            if dfs(agent, visited):
                matched += 1
        
        if matched == k:
            # Reconstruct solution
            print("YES")
            # We need to know the alias for each agent
            agent_alias = {}
            for x, y, z in edges:
                agent_alias[x] = y
            
            for agent in range(1, k + 1):
                freq = match_agent_to_freq[agent]
                alias = agent_alias[agent]
                print(agent, alias, freq)
        else:
            print("NO")
    
    bfs_match()

solve()
