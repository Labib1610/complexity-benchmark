import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    idx = 2
    channels = []
    for _ in range(t):
        s = int(data[idx]); p = int(data[idx+1]); a = int(data[idx+2])
        idx += 3
        channels.append((s, p, a))
    
    # Group channels by sensor (and processor, since they are locked together)
    # Each sensor must pick exactly one antenna, and all antennas must be distinct.
    # This is a bipartite matching: sensors on left, antennas on right.
    
    # Build adjacency: for each sensor, list of (channel_index, antenna)
    sensor_to_channels = [[] for _ in range(k+1)]
    for i, (s, p, a) in enumerate(channels):
        sensor_to_channels[s].append((i, a))
    
    # Check that every sensor has at least one channel
    for s in range(1, k+1):
        if not sensor_to_channels[s]:
            print("NO")
            return
    
    # Bipartite matching: sensors 1..k to antennas 1..k
    # match_antenna[a] = sensor assigned to antenna a
    match_antenna = [0] * (k+1)
    match_sensor = [0] * (k+1)  # sensor -> antenna
    # For reconstruction, store which channel index was used
    used_channel = [-1] * (k+1)  # sensor -> channel index
    
    def try_kuhn(v, visited):
        for ch_idx, a in sensor_to_channels[v]:
            if not visited[a]:
                visited[a] = True
                if match_antenna[a] == 0 or try_kuhn(match_antenna[a], visited):
                    match_antenna[a] = v
                    match_sensor[v] = a
                    used_channel[v] = ch_idx
                    return True
        return False
    
    for s in range(1, k+1):
        visited = [False] * (k+1)
        if not try_kuhn(s, visited):
            print("NO")
            return
    
    print("YES")
    for s in range(1, k+1):
        ch_idx = used_channel[s]
        s_val, p_val, a_val = channels[ch_idx]
        print(f"{s_val} {p_val} {a_val}")

solve()
