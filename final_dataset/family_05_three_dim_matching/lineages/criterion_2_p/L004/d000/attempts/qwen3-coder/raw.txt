from collections import defaultdict, deque

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
    
    channels = []
    sensor_to_processor = {}
    processor_to_sensor = {}
    
    for _ in range(t):
        s, p, a = map(int, input().split())
        channels.append((s, p, a))
        
        # Verify the hardware lock constraint
        if s in sensor_to_processor:
            if sensor_to_processor[s] != p:
                print("NO")
                return
        else:
            sensor_to_processor[s] = p
            
        if p in processor_to_sensor:
            if processor_to_sensor[p] != s:
                print("NO")
                return
        else:
            processor_to_sensor[p] = s
    
    # Check if we have exactly k sensors and k processors
    if len(sensor_to_processor) != k or len(processor_to_sensor) != k:
        print("NO")
        return
    
    # Build a bipartite graph between sensors and antennas
    # An edge exists if there's a channel with that sensor-antenna pair
    antenna_adj = defaultdict(list)  # antenna -> list of sensors
    sensor_adj = defaultdict(list)   # sensor -> list of antennas
    
    for s, p, a in channels:
        antenna_adj[a].append(s)
        sensor_adj[s].append(a)
    
    # Try to find a perfect matching using Hopcroft-Karp or similar
    # We need to match each sensor to a unique antenna
    
    match_sensor = {}  # sensor -> antenna
    match_antenna = {} # antenna -> sensor
    
    def bfs():
        queue = deque()
        dist = {}
        
        for s in range(1, k+1):
            if s not in match_sensor:
                dist[s] = 0
                queue.append(s)
            else:
                dist[s] = float('inf')
        dist[None] = float('inf')
        
        while queue:
            s = queue.popleft()
            if dist[s] < dist[None]:
                for a in sensor_adj[s]:
                    if a in match_antenna:
                        nxt_s = match_antenna[a]
                        if dist[nxt_s] == float('inf'):
                            dist[nxt_s] = dist[s] + 1
                            queue.append(nxt_s)
                    else:
                        dist[None] = dist[s] + 1
        return dist[None] != float('inf')
    
    def dfs(s):
        if s is None:
            return True
        for a in sensor_adj[s]:
            if a in dist and dist[a] == dist[s] + 1:
                if dfs(match_antenna.get(a)):
                    match_sensor[s] = a
                    match_antenna[a] = s
                    return True
        dist[s] = float('inf')
        return False
    
    matching = 0
    while bfs():
        for s in range(1, k+1):
            if s not in match_sensor:
                if dfs(s):
                    matching += 1
    
    if matching != k:
        print("NO")
        return
    
    # Reconstruct the solution
    result = []
    used_channels = set()
    
    for s in range(1, k+1):
        if s in match_sensor:
            a = match_sensor[s]
            p = sensor_to_processor[s]
            result.append((s, p, a))
        else:
            print("NO")
            return
    
    print("YES")
    for s, p, a in result:
        print(s, p, a)

solve()
