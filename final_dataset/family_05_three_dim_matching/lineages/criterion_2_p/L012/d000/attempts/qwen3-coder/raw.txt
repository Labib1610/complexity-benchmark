from collections import defaultdict

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
    
    # Read configurations
    configs = []
    catalyst_to_reagent = {}
    catalyst_reagent_to_solvents = defaultdict(list)
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        configs.append((x, y, z))
        
        # Verify the constraint: each catalyst always uses the same reagent
        if x in catalyst_to_reagent:
            assert catalyst_to_reagent[x] == y
        else:
            catalyst_to_reagent[x] = y
            
        # Group solvents by (catalyst, reagent) pair
        catalyst_reagent_to_solvents[(x, y)].append(z)
    
    # Check if we have all catalysts 1 to k
    if len(catalyst_to_reagent) != k:
        print("NO")
        return
        
    # Check if reagents are exactly 1 to k (bijective mapping)
    reagents = set(catalyst_to_reagent.values())
    if len(reagents) != k or reagents != set(range(1, k + 1)):
        print("NO")
        return
    
    # Build bipartite graph for matching catalysts to solvents
    # Each catalyst needs to be matched to a unique solvent
    # But we must respect that catalyst x must use reagent catalyst_to_reagent[x]
    
    # For each catalyst, we have a list of possible solvents
    catalyst_solvents = {}
    for x in range(1, k + 1):
        y = catalyst_to_reagent[x]
        catalyst_solvents[x] = catalyst_reagent_to_solvents[(x, y)]
    
    # Try to find a perfect matching using backtracking/Hungarian-like approach
    # This is essentially a bipartite matching: catalysts <-> solvents
    
    # Build adjacency list for catalyst -> available solvents
    adj = defaultdict(list)
    for x in range(1, k + 1):
        adj[x] = catalyst_solvents[x]
    
    # Try to match each catalyst to a unique solvent
    match_solvent_to_catalyst = {}
    match_catalyst_to_solvent = {}
    
    def dfs(catalyst, visited_solvents):
        for solvent in adj[catalyst]:
            if solvent in visited_solvents:
                continue
            visited_solvents.add(solvent)
            
            # If solvent is unmatched or we can find augmenting path
            if solvent not in match_solvent_to_catalyst:
                match_solvent_to_catalyst[solvent] = catalyst
                match_catalyst_to_solvent[catalyst] = solvent
                return True
            else:
                # Try to rematch the catalyst currently using this solvent
                old_catalyst = match_solvent_to_catalyst[solvent]
                if dfs(old_catalyst, visited_solvents):
                    match_solvent_to_catalyst[solvent] = catalyst
                    match_catalyst_to_solvent[catalyst] = solvent
                    return True
        return False
    
    # Try to match each catalyst
    matched = 0
    for catalyst in range(1, k + 1):
        visited_solvents = set()
        if dfs(catalyst, visited_solvents):
            matched += 1
    
    if matched != k:
        print("NO")
        return
    
    # Output the solution
    print("YES")
    for catalyst in range(1, k + 1):
        reagent = catalyst_to_reagent[catalyst]
        solvent = match_catalyst_to_solvent[catalyst]
        print(catalyst, reagent, solvent)

solve()
