from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    configs = []
    index = 2
    for _ in range(t):
        d = int(data[index])
        f = int(data[index+1])
        s = int(data[index+2])
        configs.append((d, f, s))
        index += 3
    
    # Build bipartite graph: directors -> screening rooms
    # For each director, we know their film already
    # So we just need to match directors to screening rooms
    # This is a maximum bipartite matching problem
    
    # Create adjacency list from directors to screening rooms
    adj = defaultdict(list)
    for d, f, s in configs:
        adj[d].append(s)
    
    # Try to find a perfect matching
    # Since each director must be matched to exactly one screening room
    # And each screening room must be matched to exactly one director
    
    match_screening_room = [0] * (k + 1)  # match_screening_room[s] = d
    match_director = [0] * (k + 1)       # match_director[d] = s
    
    def dfs(d, visited):
        for s in adj[d]:
            if visited[s]:
                continue
            visited[s] = True
            
            # If s is unmatched or we can rematch its current director
            if match_screening_room[s] == 0 or dfs(match_screening_room[s], visited):
                match_screening_room[s] = d
                match_director[d] = s
                return True
        return False
    
    # Try to match each director
    matched = 0
    for d in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(d, visited):
            matched += 1
    
    if matched < k:
        print("NO")
        return
    
    print("YES")
    # Reconstruct the selected configurations
    result_configs = []
    for d in range(1, k + 1):
        s = match_director[d]
        # Find the film for this director
        # We need to find the actual config that has this d, s
        for config_d, config_f, config_s in configs:
            if config_d == d and config_s == s:
                result_configs.append((config_d, config_f, config_s))
                break
    
    for d, f, s in result_configs:
        print(d, f, s)

solve()
