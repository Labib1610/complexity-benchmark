from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    if t < k:
        print("NO")
        return
    
    dispatches = []
    route_to_dispatch = {}
    vehicle_to_route = {}
    
    idx = 2
    for _ in range(t):
        x = int(data[idx])
        y = int(data[idx+1])
        z = int(data[idx+2])
        idx += 3
        dispatches.append((x, y, z))
        
        # Since each route is paired with exactly one vehicle
        if x not in route_to_dispatch:
            route_to_dispatch[x] = y
        elif route_to_dispatch[x] != y:
            # This should not happen according to problem statement
            pass
            
        # Different routes use different vehicles
        if y in vehicle_to_route:
            if vehicle_to_route[y] != x:
                # This should not happen according to problem statement
                pass
        else:
            vehicle_to_route[y] = x
    
    # Build a bipartite graph between routes and drivers
    # For each route, we have a specific vehicle. We need to match drivers.
    route_drivers = defaultdict(list)
    route_vehicle = {}
    
    for x, y, z in dispatches:
        if x not in route_vehicle:
            route_vehicle[x] = y
        route_drivers[x].append((z, y))  # driver, vehicle
    
    # Check if we have exactly k routes
    if len(route_vehicle) != k:
        print("NO")
        return
        
    # For each route, we know its vehicle. Now build graph for matching drivers to routes
    # This is a matching problem: routes <-> drivers
    route_driver_edges = defaultdict(list)
    all_drivers = set()
    
    for x, y, z in dispatches:
        route_driver_edges[x].append(z)
        all_drivers.add(z)
    
    # Create adjacency list for Hungarian algorithm or use network flow
    # But since k <= 1200, we can use a simpler approach with DFS matching
    
    # Build graph: routes -> drivers
    graph = defaultdict(list)
    for x, y, z in dispatches:
        graph[x].append(z)
        
    # Try to find maximum matching using DFS
    match_driver_to_route = {}
    match_route_to_driver = {}
    
    def dfs(route, visited):
        for driver in graph[route]:
            if driver in visited:
                continue
            visited.add(driver)
            
            if driver not in match_driver_to_route:
                match_driver_to_route[driver] = route
                match_route_to_driver[route] = driver
                return True
                
            # Find augmenting path
            old_route = match_driver_to_route[driver]
            if dfs(old_route, visited):
                match_driver_to_route[driver] = route
                match_route_to_driver[route] = driver
                return True
                
        return False
    
    matched = 0
    routes = list(route_vehicle.keys())
    
    for route in routes:
        visited = set()
        if dfs(route, visited):
            matched += 1
    
    if matched != k:
        print("NO")
        return
        
    print("YES")
    for route in routes:
        driver = match_route_to_driver[route]
        vehicle = route_vehicle[route]
        print(route, vehicle, driver)

solve()
