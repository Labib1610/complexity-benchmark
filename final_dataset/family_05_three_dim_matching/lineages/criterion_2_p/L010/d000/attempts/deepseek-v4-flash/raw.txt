We can model this as a bipartite matching problem.  
Each catalyst is paired with exactly one solvent (given by the lab procedure), so we can group reactions by catalyst–solvent pair.  
For each such pair, we have a set of possible temperatures.  
We need to assign each catalyst–solvent pair a distinct temperature, using only temperatures that appear in the log for that pair.  
This is exactly a bipartite matching between the catalyst–solvent pairs and the temperature indices.  
Since each catalyst has a fixed solvent, the number of distinct pairs is at most k.  
We can run a standard DFS-based bipartite matching (Kuhn’s algorithm) on up to 1200 nodes per side, which is fine.  
If the matching size equals k, we output YES and the chosen reactions.  

```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # For each catalyst, store its fixed solvent (if any)
    cat_solvent = {}
    # For each catalyst, store set of temperatures
    cat_temps = {}
    
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        if x not in cat_solvent:
            cat_solvent[x] = y
        # According to problem, x always has same y, so we can trust it
        if x not in cat_temps:
            cat_temps[x] = set()
        cat_temps[x].add(z)
    
    # Build list of catalyst indices that have at least one reaction
    catalysts = list(cat_temps.keys())
    n = len(catalysts)
    if n < k:
        print("NO")
        return
    
    # Map catalyst to index in our list
    cat_to_idx = {c:i for i,c in enumerate(catalysts)}
    
    # Build adjacency: for each catalyst index, list of temperatures
    adj = [[] for _ in range(n)]
    for c in catalysts:
        idx = cat_to_idx[c]
        adj[idx] = list(cat_temps[c])
    
    # Kuhn's algorithm for bipartite matching
    match_temp = [-1] * (k+1)  # temperature -> catalyst index
    match_cat = [-1] * n
    
    def dfs(cat_idx, visited):
        for temp in adj[cat_idx]:
            if not visited[temp]:
                visited[temp] = True
                if match_temp[temp] == -1 or dfs(match_temp[temp], visited):
                    match_temp[temp] = cat_idx
                    match_cat[cat_idx] = temp
                    return True
        return False
    
    for i in range(n):
        visited = [False] * (k+1)
        dfs(i, visited)
    
    # Check if all catalysts matched
    if sum(1 for x in match_cat if x != -1) < k:
        print("NO")
        return
    
    # Output
    print("YES")
    for c in catalysts:
        idx = cat_to_idx[c]
        temp = match_cat[idx]
        sol = cat_solvent[c]
        print(f"{c} {sol} {temp}")

if __name__ == "__main__":
    solve()
```
