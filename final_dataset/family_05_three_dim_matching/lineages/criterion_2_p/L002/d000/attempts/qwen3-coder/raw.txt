from collections import defaultdict, deque

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
        
    # Since each specialist is paired with exactly one machine,
    # and no two specialists share the same machine,
    # we can think of this as a bipartite matching problem
    # between specialists (or machines) and time slots.
    
    # For each specialist x, we know their machine y_x
    # So each allowed combination (x, y, z) actually means
    # specialist x can work at time z (on their dedicated machine y)
    
    # Build a graph: for each specialist, which time slots they can use
    adj = defaultdict(list)  # specialist -> list of time slots
    spec_to_machine = {}     # specialist -> their machine
    machine_to_spec = {}     # machine -> their specialist
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        adj[x].append(z)
        spec_to_machine[x] = y
        machine_to_spec[y] = x
    
    # Check if we have exactly k specialists
    if len(adj) != k:
        print("NO")
        return
    
    # Now we need to match each specialist to a unique time slot
    # This is maximum bipartite matching
    
    # Use Hungarian algorithm / Ford-Fulkerson approach
    match_time_to_spec = {}  # time slot -> specialist matched to it
    match_spec_to_time = {}  # specialist -> time slot matched to it
    
    def dfs(spec, visited):
        for time_slot in adj[spec]:
            if time_slot in visited:
                continue
            visited.add(time_slot)
            
            # If time slot is unmatched or we can find augmenting path
            if time_slot not in match_time_to_spec or dfs(match_time_to_spec[time_slot], visited):
                match_time_to_spec[time_slot] = spec
                match_spec_to_time[spec] = time_slot
                return True
        return False
    
    # Try to match each specialist
    matched = 0
    for spec in range(1, k+1):
        if spec not in adj:
            print("NO")
            return
        visited = set()
        if dfs(spec, visited):
            matched += 1
    
    if matched != k:
        print("NO")
        return
    
    print("YES")
    for spec in range(1, k+1):
        time_slot = match_spec_to_time[spec]
        machine = spec_to_machine[spec]
        print(spec, machine, time_slot)

solve()
