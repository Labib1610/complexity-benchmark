import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    brackets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        brackets.append((x - 1, y - 1, z - 1))
        idx += 3
        
    # We need to select k brackets such that all x, y, z are distinct.
    # This is equivalent to finding a perfect matching in a 3D bipartite-like structure,
    # or more simply, a 3D matching problem. Since k is small (up to 22),
    # we can use backtracking with pruning.
    
    # Group brackets by x-coordinate
    by_x = [[] for _ in range(k)]
    for b in brackets:
        by_x[b[0]].append(b)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x):
        if x == k:
            return True
        
        for b in by_x[x]:
            y, z = b[1], b[2]
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
