import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3
        
    # Exact Cover Problem (3D Matching)
    # Since k <= 22, we can use backtracking with pruning.
    # State: (used_x_mask, used_y_mask, used_z_mask)
    
    # Pre-filter triples: only keep those that could potentially be part of a solution
    # Actually, with k=22, simple backtracking with bitmasks is efficient enough.
    
    solution = []
    
    def backtrack(idx, mask_x, mask_y, mask_z):
        if mask_x == (1 << k) - 1:
            return True
        
        if idx == t:
            return False
        
        # Pruning: if remaining triples are not enough to cover remaining slots
        # (Not strictly necessary for k=22 if the search space is sparse)
        
        # Try including triples[idx]
        x, y, z = triples[idx]
        if not (mask_x & (1 << x)) and not (mask_y & (1 << y)) and not (mask_z & (1 << z)):
            solution.append(triples[idx])
            if backtrack(idx + 1, mask_x | (1 << x), mask_y | (1 << y), mask_z | (1 << z)):
                return True
            solution.pop()
            
        # Try skipping triples[idx]
        if backtrack(idx + 1, mask_x, mask_y, mask_z):
            return True
            
        return False

    # Optimization: Sort triples or use Dancing Links if needed.
    # Given the constraints and the nature of 3D matching, 
    # simple backtracking might be slow. Let's use a more direct approach.
    
    # Using a simple recursive search with bitmasks
    if backtrack(0, 0, 0, 0):
        print("YES")
        for x, y, z in solution:
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")

solve()
