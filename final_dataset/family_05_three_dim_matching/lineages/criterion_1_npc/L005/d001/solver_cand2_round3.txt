import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(m):
        c = int(input_data[idx]) - 1
        i = int(input_data[idx+1]) - 1
        d = int(input_data[idx+2]) - 1
        triples.append((c, i, d))
        idx += 3
        
    # Group triples by compute node
    by_compute = [[] for _ in range(n)]
    for c, i, d in triples:
        by_compute[c].append((i, d))
        
    # State: (compute_node_index, used_images_mask, used_databases_mask)
    # Since n <= 22, we use bitmasks.
    # This is an exact cover problem (3D matching).
    # Backtracking with pruning.
    
    memo = {}

    def backtrack(c_idx, img_mask, db_mask):
        if c_idx == n:
            return []
        
        state = (c_idx, img_mask, db_mask)
        if state in memo:
            return None
        
        for i, d in by_compute[c_idx]:
            if not (img_mask & (1 << i)) and not (db_mask & (1 << d)):
                res = backtrack(c_idx + 1, img_mask | (1 << i), db_mask | (1 << d))
                if res is not None:
                    return [(c_idx + 1, i + 1, d + 1)] + res
        
        memo[state] = False
        return None

    result = backtrack(0, 0, 0)
    
    if result is not None:
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
