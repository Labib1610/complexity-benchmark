We need to select exactly k triples such that all x, y, and z values from 1 to k appear exactly once each.  

This is equivalent to finding a perfect matching in a 3-partite 3-uniform hypergraph, which is NP-hard in general.  
But k ≤ 22, so we can use a bitmask DP over two dimensions while iterating over the third.

We can fix the order of x (frequency bands) and try to assign y and z using DP over subsets.

Here’s the implementation:

```python
import sys

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    triples = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        triples.append((x-1, y-1, z-1))
    
    # Group by x
    by_x = [[] for _ in range(k)]
    for x, y, z in triples:
        by_x[x].append((y, z))
    
    # DP over subsets of y and z
    # dp[mask_y][mask_z] = True if possible
    # We'll iterate x from 0 to k-1
    dp = [[False]*(1<<k) for _ in range(1<<k)]
    dp[0][0] = True
    
    # To reconstruct, store parent
    parent = [[None]*(1<<k) for _ in range(1<<k)]
    
    for x in range(k):
        new_dp = [[False]*(1<<k) for _ in range(1<<k)]
        for my in range(1<<k):
            for mz in range(1<<k):
                if not dp[my][mz]:
                    continue
                for y, z in by_x[x]:
                    if (my>>y)&1 or (mz>>z)&1:
                        continue
                    nmy = my | (1<<y)
                    nmz = mz | (1<<z)
                    if not new_dp[nmy][nmz]:
                        new_dp[nmy][nmz] = True
                        parent[nmy][nmz] = (my, mz, x, y, z)
        dp = new_dp
    
    full_mask = (1<<k)-1
    if not dp[full_mask][full_mask]:
        print("NO")
        return
    
    print("YES")
    # Reconstruct
    my, mz = full_mask, full_mask
    result = []
    for x in range(k-1, -1, -1):
        pmy, pmz, px, py, pz = parent[my][mz]
        result.append((px+1, py+1, pz+1))
        my, mz = pmy, pmz
    result.reverse()
    for line in result:
        print(*line)

if __name__ == "__main__":
    solve()
```
