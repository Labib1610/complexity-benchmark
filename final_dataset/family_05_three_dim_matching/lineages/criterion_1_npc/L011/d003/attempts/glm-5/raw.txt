import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    configs = []
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        configs.append((x-1, y-1, z-1))  # 0-indexed
    
    # Build adjacency: for each mold x, list of (dye y, packaging z) pairs
    adj = [[] for _ in range(k)]
    for idx, (x, y, z) in enumerate(configs):
        adj[x].append((y, z, idx))
    
    # match_y[y] = x that y is matched to
    # match_z[z] = x that z is matched to
    match_y = [-1] * k
    match_z = [-1] * k
    
    def dfs(x, visited_y, visited_z):
        for y, z, idx in adj[x]:
            if visited_y[y] or visited_z[z]:
                continue
            visited_y[y] = True
            visited_z[z] = True
            old_x_y = match_y[y]
            old_x_z = match_z[z]
            match_y[y] = x
            match_z[z] = x
            if old_x_y == -1 and old_x_z == -1:
                return True
            # Need to rematch old_x_y and old_x_z
            # They might be the same mold or different
            molds_to_rematch = set()
            if old_x_y != -1:
                molds_to_rematch.add(old_x_y)
            if old_x_z != -1:
                molds_to_rematch.add(old_x_z)
            
            success = True
            for old_x in molds_to_rematch:
                if not dfs(old_x, visited_y, visited_z):
                    success = False
                    break
            if success:
                return True
            # Backtrack
            match_y[y] = old_x_y
            match_z[z] = old_x_z
        return False
    
    for x in range(k):
        visited_y = [False] * k
        visited_z = [False] * k
        if not dfs(x, visited_y, visited_z):
            print("NO")
            return
    
    # Build result: for each mold x, find its matched (y, z)
    result = []
    for x in range(k):
        for y, z, idx in adj[x]:
            if match_y[y] == x and match_z[z] == x:
                result.append((x+1, y+1, z+1))
                break
    
    print("YES")
    for r in result:
        print(r[0], r[1], r[2])

solve()
