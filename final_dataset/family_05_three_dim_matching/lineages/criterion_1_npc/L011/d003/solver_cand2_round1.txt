import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    designs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        designs.append((x, y, z))
        idx += 3

    # We need to select k designs such that all x are distinct, all y are distinct,
    # and all z are distinct. This is equivalent to finding a perfect matching
    # in a 3-partite hypergraph, or more simply, a set of k triples (x_i, y_i, z_i)
    # such that {x_i} = {0..k-1}, {y_i} = {0..k-1}, {z_i} = {0..k-1}.
    
    # Since k <= 22, we can use backtracking with pruning.
    # Sort designs to potentially find a solution faster.
    
    used_x = [False] * k
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(idx_in_designs, count):
        if count == k:
            return True
        if idx_in_designs == t:
            return False
        
        # Pruning: if remaining designs are not enough
        if t - idx_in_designs < k - count:
            return False
            
        # Option 1: Include designs[idx_in_designs]
        x, y, z = designs[idx_in_designs]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append((x + 1, y + 1, z + 1))
            if backtrack(idx_in_designs + 1, count + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Option 2: Skip designs[idx_in_designs]
        if backtrack(idx_in_designs + 1, count):
            return True
            
        return False

    if backtrack(0, 0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

solve()
