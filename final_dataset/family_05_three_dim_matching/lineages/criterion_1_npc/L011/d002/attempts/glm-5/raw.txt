import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    triples = []
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        triples.append((x-1, y-1, z-1))  # 0-indexed
    
    # Build adjacency: for each x, list of (y, z) pairs
    adj = [[] for _ in range(k)]
    for idx, (x, y, z) in enumerate(triples):
        adj[x].append((y, z, idx))
    
    # match_y[y] = x that is matched to y
    match_y = [-1] * k
    # match_z[z] = x that is matched to z
    match_z = [-1] * k
    
    def dfs(x, seen_y, seen_z):
        for y, z, idx in adj[x]:
            if seen_y[y] or seen_z[z]:
                continue
            seen_y[y] = True
            seen_z[z] = True
            
            my = match_y[y]
            mz = match_z[z]
            
            if my == -1 and mz == -1:
                match_y[y] = x
                match_z[z] = x
                return True
            elif my == -1:
                # mz != -1, need to reassign match_z[z]
                match_z[z] = x
                match_y[y] = x
                if dfs(mz, seen_y, seen_z):
                    return True
                match_z[z] = mz
                match_y[y] = -1
            elif mz == -1:
                # my != -1, need to reassign match_y[y]
                match_y[y] = x
                match_z[z] = x
                if dfs(my, seen_y, seen_z):
                    return True
                match_y[y] = my
                match_z[z] = -1
            else:
                # Both my and mz are matched, need to reassign both
                match_y[y] = x
                match_z[z] = x
                if dfs(my, seen_y, seen_z) and dfs(mz, seen_y, seen_z):
                    return True
                match_y[y] = my
                match_z[z] = mz
        return False
    
    matched_x = [False] * k
    for x in range(k):
        seen_y = [False] * k
        seen_z = [False] * k
        if dfs(x, seen_y, seen_z):
            matched_x[x] = True
    
    if sum(matched_x) == k:
        print("YES")
        # Find the selected triples
        # match_y[y] = x, match_z[z] = x
        # For each x, find the triple that matches
        x_to_yz = {}
        for y in range(k):
            x = match_y[y]
            if x != -1:
                x_to_yz[x] = (y, -1)  # placeholder for z
        for z in range(k):
            x = match_z[z]
            if x != -1:
                if x in x_to_yz:
                    x_to_yz[x] = (x_to_yz[x][0], z)
                else:
                    x_to_yz[x] = (-1, z)
        
        # Now for each x, find the triple with matching y and z
        for x in range(k):
            y, z = x_to_yz.get(x, (-1, -1))
            # Find the triple index
            for ty, tz, tidx in adj[x]:
                if ty == y and tz == z:
                    print(f"{x+1} {y+1} {z+1}")
                    break
    else:
        print("NO")

solve()
