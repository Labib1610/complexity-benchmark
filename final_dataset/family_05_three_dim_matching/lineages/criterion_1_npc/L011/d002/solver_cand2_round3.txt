```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x].append((y, z))
        idx += 3
        
    # We need to select k triples (x, y, z) such that x, y, z are all distinct
    # and cover {1..k}. This is equivalent to finding a perfect matching in a 
    # 3D hypergraph, or more simply, finding a set of k edges in a bipartite 
    # graph where each edge is (y, z) and we assign each x to one such edge.
    
    # Since k <= 22, we can use backtracking with pruning.
    # State: current x (1 to k), used_y (bitmask), used_z (bitmask)
    
    used_y = 0
    used_z = 0
    result = []
    
    def backtrack(x):
        if x > k:
            return True
        
        for y, z in adj[x]:
            y_bit = 1 << (y - 1)
            z_bit = 1 << (z - 1)
            if not (used_y & y_bit) and not (used_z & z_bit):
                nonlocal used_y, used_z
                used_y |= y_bit
                used_z |= z_bit
                result.append((x, y, z))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y &= ~y_bit
                used_z &= ~z_bit
        return False

    # Optimization: sort adj[x] by length to fail fast
    for i in range(1, k + 1):
        adj[i].sort(key=lambda item: len([1 for j in range(1, k+1) if (item[0], item[1]) in adj[j]]))
        
    if backtrack(1):
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

solve()
```
