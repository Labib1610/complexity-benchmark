import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    triples = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        triples.append((x, y, z))

    # Exact cover with DLX
    n_cols = 3 * k  # columns: X0..Xk-1, Y0..Yk-1, Z0..Zk-1

    # Build column headers
    class Node:
        __slots__ = ('left', 'right', 'up', 'down', 'col', 'row')
        def __init__(self):
            self.left = self.right = self.up = self.down = self
            self.col = None
            self.row = -1

    header = Node()
    cols = [Node() for _ in range(n_cols)]
    for i, col in enumerate(cols):
        col.col = col
        col.up = col.down = col
        col.left = header.left
        col.right = header
        header.left.right = col
        header.left = col
        col.size = 0  # number of nodes in column

    # Add rows
    row_nodes = []  # list of first node of each row
    for r, (x, y, z) in enumerate(triples):
        nodes = []
        for c in (x, k + y, 2 * k + z):
            col = cols[c]
            node = Node()
            node.col = col
            node.row = r
            # vertical links
            node.up = col.up
            node.down = col
            col.up.down = node
            col.up = node
            col.size += 1
            nodes.append(node)
        # horizontal links
        for i in range(3):
            nodes[i].right = nodes[(i + 1) % 3]
            nodes[i].left = nodes[(i - 1) % 3]
        row_nodes.append(nodes[0])

    solution_rows = []

    sys.setrecursionlimit(10000)

    def cover(col):
        col.right.left = col.left
        col.left.right = col.right
        i = col.down
        while i != col:
            j = i.right
            while j != i:
                j.down.up = j.up
                j.up.down = j.down
                j.col.size -= 1
                j = j.right
            i = i.down

    def uncover(col):
        i = col.up
        while i != col:
            j = i.left
            while j != i:
                j.col.size += 1
                j.down.up = j
                j.up.down = j
                j = j.left
            i = i.up
        col.right.left = col
        col.left.right = col

    def search(k_depth=0):
        if header.right == header:
            return True
        # choose column with minimal size
        c = header.right
        min_sz = c.size
        col = c
        while c != header:
            if c.size < min_sz:
                min_sz = c.size
                col = c
                if min_sz == 0:
                    break
            c = c.right
        if min_sz == 0:
            return False
        cover(col)
        r = col.down
        while r != col:
            solution_rows.append(r.row)
            j = r.right
            while j != r:
                cover(j.col)
                j = j.right
            if search(k_depth + 1):
                return True
            # backtrack
            j = r.left
            while j != r:
                uncover(j.col)
                j = j.left
            solution_rows.pop()
            r = r.down
        uncover(col)
        return False

    possible = search()
    out = []
    if possible:
        out.append("YES")
        for rid in solution_rows:
            x, y, z = triples[rid]
            out.append(f"{x+1} {y+1} {z+1}")
    else:
        out.append("NO")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
