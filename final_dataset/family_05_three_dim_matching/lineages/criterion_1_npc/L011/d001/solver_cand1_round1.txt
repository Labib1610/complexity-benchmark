import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    protocols = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        protocols.append((x - 1, y - 1, z - 1))
        idx += 3

    # We need to select k protocols (x_i, y_i, z_i) such that
    # all x_i are distinct, all y_i are distinct, all z_i are distinct.
    # This is equivalent to finding a set of k triples (x, y, z)
    # such that x, y, z are permutations of 0..k-1.
    # This is the exact cover problem or finding a transversal in a Latin Square.
    # Since k <= 22, we use backtracking with pruning.
    
    adj = [[] for _ in range(k)]
    for x, y, z in protocols:
        adj[x].append((y, z))
        
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        # Optimization: sort by number of available choices
        candidates = []
        for y, z in adj[x]:
            if not used_y[y] and not used_z[z]:
                candidates.append((y, z))
        
        # Heuristic: try candidates that are most constrained
        for y, z in candidates:
            used_y[y] = True
            used_z[z] = True
            result.append((x + 1, y + 1, z + 1))
            
            if backtrack(x + 1):
                return True
            
            result.pop()
            used_y[y] = False
            used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
