import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    vectors = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        vectors.append((x, y, z))
        idx += 3

    # Group vectors by x to facilitate backtracking
    adj = [[] for _ in range(k)]
    for v in vectors:
        adj[v[0]].append(v)

    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        for v in adj[x]:
            _, y, z = v
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(v)
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for v in result:
            print(f"{v[0] + 1} {v[1] + 1} {v[2] + 1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
