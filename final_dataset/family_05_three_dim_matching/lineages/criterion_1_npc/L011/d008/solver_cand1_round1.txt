import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        configs.append((x, y, z))
        idx += 3
        
    target = sum(1 << i for i in range(1, k + 1))
    
    # We need to select k configurations such that:
    # sum(2^x_i) = target, sum(2^y_i) = target, sum(2^z_i) = target
    # This is equivalent to saying that for each dimension, 
    # the set of values {x_1, ..., x_k} must be exactly {1, 2, ..., k}.
    
    # Since k is small (up to 22), but t can be up to k^3, 
    # we use backtracking with pruning.
    
    selected = []
    
    # Bitmasks for tracking used values in each dimension
    # mask_x, mask_y, mask_z
    
    def backtrack(idx, mask_x, mask_y, mask_z, count):
        if count == k:
            if mask_x == target and mask_y == target and mask_z == target:
                return True
            return False
        
        if idx == t:
            return False
        
        # Pruning: if remaining configs are not enough to fill k
        if count + (t - idx) < k:
            return False
            
        # Try including configs[idx]
        x, y, z = configs[idx]
        bit_x, bit_y, bit_z = 1 << x, 1 << y, 1 << z
        
        if not (mask_x & bit_x) and not (mask_y & bit_y) and not (mask_z & bit_z):
            selected.append(configs[idx])
            if backtrack(idx + 1, mask_x | bit_x, mask_y | bit_y, mask_z | bit_z, count + 1):
                return True
            selected.pop()
            
        # Try excluding configs[idx]
        if backtrack(idx + 1, mask_x, mask_y, mask_z, count):
            return True
            
        return False

    if backtrack(0, 0, 0, 0, 0):
        print("YES")
        for item in selected:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
