from itertools import combinations
import sys

def solve():
    k, t = map(int, sys.stdin.readline().split())
    
    if t < k:
        print("NO")
        return
    
    configs = []
    for _ in range(t):
        x, y, z = map(int, sys.stdin.readline().split())
        configs.append((x, y, z))
    
    # Calculate baseline threshold
    B_k = sum(2**j for j in range(1, k+1))
    
    # Try all combinations of k configurations
    for combo in combinations(configs, k):
        # Calculate resonances
        freq_res = sum(2**x for x, _, _ in combo)
        mod_res = sum(2**y for _, y, _ in combo)
        enc_res = sum(2**z for _, _, z in combo)
        
        if freq_res == B_k and mod_res == B_k and enc_res == B_k:
            print("YES")
            for x, y, z in combo:
                print(x, y, z)
            return
    
    print("NO")

solve()
