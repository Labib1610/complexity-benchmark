import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        configs.append((x, y, z))
        idx += 3

    target = (1 << (k + 1)) - 2
    
    # Meet-in-the-middle: split t configs into two halves
    mid = t // 2
    left = configs[:mid]
    right = configs[mid:]
    
    # Map (sum_x, sum_y, sum_z) -> list of indices
    # Since k <= 22, sum_x, sum_y, sum_z are at most 2^23-1
    # We need to pick exactly k items total.
    
    memo = {}
    
    def get_subsets(arr):
        n = len(arr)
        # subsets[count] = { (sx, sy, sz): [indices] }
        subsets = [{} for _ in range(n + 1)]
        subsets[0][(0, 0, 0)] = []
        
        for i, (x, y, z) in enumerate(arr):
            val_x, val_y, val_z = 1 << x, 1 << y, 1 << z
            for count in range(i, -1, -1):
                for (sx, sy, sz), path in subsets[count].items():
                    new_s = (sx + val_x, sy + val_y, sz + val_z)
                    if new_s[0] <= target and new_s[1] <= target and new_s[2] <= target:
                        subsets[count + 1][new_s] = path + [i]
        return subsets

    left_subsets = get_subsets(left)
    right_subsets = get_subsets(right)
    
    for c1 in range(len(left_subsets)):
        c2 = k - c1
        if c2 < 0 or c2 >= len(right_subsets):
            continue
        
        for (sx1, sy1, sz1), path1 in left_subsets[c1].items():
            target_s = (target - sx1, target - sy1, target - sz1)
            if target_s in right_subsets[c2]:
                path2 = right_subsets[c2][target_s]
                print("YES")
                for idx in path1:
                    print(f"{left[idx][0]} {left[idx][1]} {left[idx][2]}")
                for idx in path2:
                    print(f"{right[idx][0]} {right[idx][1]} {right[idx][2]}")
                return

    print("NO")

solve()
