import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    configs = []
    idx = 2
    for _ in range(t):
        configs.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    # The condition d1 * d2 * d3 = k^3 with d1, d2, d3 <= k implies d1 = d2 = d3 = k.
    # This means we need to pick k configurations such that all k frequency bands,
    # all k modulation schemes, and all k encryption keys are used exactly once.
    # This is equivalent to finding a transversal of size k in a 3D grid,
    # or a Latin Square of order k if we view it as a set of triples.
    
    # Since k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    selected = []
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for i in range(start_idx, t):
            x, y, z = configs[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = True
                used_y[y] = True
                used_z[z] = True
                selected.append(configs[i])
                
                if backtrack(i + 1, count + 1):
                    return True
                
                selected.pop()
                used_x[x] = False
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for item in selected:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
