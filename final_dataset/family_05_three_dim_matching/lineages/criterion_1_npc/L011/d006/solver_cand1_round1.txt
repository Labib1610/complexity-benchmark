```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        configs.append((x, y, z))
        idx += 3

    # We need to select k configurations (x_i, y_i, z_i) such that
    # all x_i are distinct, all y_i are distinct, and all z_i are distinct.
    # This is equivalent to finding a set of k edges in a 3-partite hypergraph
    # that form a perfect matching.
    # Since k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * k
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(start_idx, count):
        if count == k:
            return True
        
        # Optimization: if remaining configs are not enough
        if t - start_idx < k - count:
            return False
            
        for i in range(start_idx, t):
            x, y, z = configs[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = True
                used_y[y] = True
                used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(i + 1, count + 1):
                    return True
                
                result.pop()
                used_x[x] = False
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

solve()
```
