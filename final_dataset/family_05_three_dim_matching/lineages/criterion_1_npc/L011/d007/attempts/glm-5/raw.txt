import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    configs = []
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        configs.append((x, y, z))
    
    # We need to select exactly k configurations such that:
    # - Each x value from 1 to k appears exactly once
    # - Each y value from 1 to k appears exactly once
    # - Each z value from 1 to k appears exactly once
    # This is equivalent to finding a perfect matching in a 3D assignment problem
    
    # Build adjacency: for each x, which configs have this x
    configs_by_x = [[] for _ in range(k + 1)]
    for idx, (x, y, z) in enumerate(configs):
        configs_by_x[x].append((idx, y, z))
    
    # Check if any x has no configs
    for x in range(1, k + 1):
        if not configs_by_x[x]:
            print("NO")
            return
    
    # DP with bitmask for y and z
    # dp[mask_y][mask_z] = list of config indices achieving this state
    # mask_y and mask_z are bitmasks of size k (bits 0 to k-1 represent values 1 to k)
    
    full_mask = (1 << k) - 1
    
    # Use dictionary for sparse DP
    # dp[(mask_y, mask_z)] = list of config indices
    dp = {((0, 0)): []}
    
    for x in range(1, k + 1):
        new_dp = {}
        for (mask_y, mask_z), selected in dp.items():
            for idx, y, z in configs_by_x[x]:
                bit_y = 1 << (y - 1)
                bit_z = 1 << (z - 1)
                if (mask_y & bit_y) == 0 and (mask_z & bit_z) == 0:
                    new_mask_y = mask_y | bit_y
                    new_mask_z = mask_z | bit_z
                    new_selected = selected + [idx]
                    if (new_mask_y, new_mask_z) not in new_dp:
                        new_dp[(new_mask_y, new_mask_z)] = new_selected
        dp = new_dp
        if not dp:
            print("NO")
            return
    
    # Check if we reached full mask
    if (full_mask, full_mask) in dp:
        print("YES")
        selected_indices = dp[(full_mask, full_mask)]
        for idx in selected_indices:
            x, y, z = configs[idx]
            print(x, y, z)
    else:
        print("NO")

solve()
