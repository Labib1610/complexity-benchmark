import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    blocks = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        blocks.append((x, y, z))
        idx += 3

    target = (1 << k) - 1
    
    # Meet-in-the-middle: split blocks into two halves
    mid = t // 2
    left_blocks = blocks[:mid]
    right_blocks = blocks[mid:]
    
    # Map (sum_x, sum_y, sum_z, count) -> list of blocks
    # Since we need exactly k blocks, we store count
    left_map = {}
    
    def get_subsets(arr):
        n = len(arr)
        # Using recursion to generate subsets
        res = {}
        def backtrack(i, cur_x, cur_y, cur_z, count, path):
            if count > k:
                return
            if i == n:
                key = (cur_x, cur_y, cur_z, count)
                if key not in res:
                    res[key] = path
                return
            # Include
            backtrack(i + 1, cur_x + (1 << (arr[i][0]-1)), cur_y + (1 << (arr[i][1]-1)), cur_z + (1 << (arr[i][2]-1)), count + 1, path + [arr[i]])
            # Exclude
            backtrack(i + 1, cur_x, cur_y, cur_z, count, path)
        
        backtrack(0, 0, 0, 0, 0, [])
        return res

    left_map = get_subsets(left_blocks)
    
    # Search for right side
    n_right = len(right_blocks)
    def find_right(i, cur_x, cur_y, cur_z, count, path):
        if count > k:
            return None
        if i == n_right:
            needed_x = target - cur_x
            needed_y = target - cur_y
            needed_z = target - cur_z
            needed_count = k - count
            key = (needed_x, needed_y, needed_z, needed_count)
            if key in left_map:
                return left_map[key] + path
            return None
        
        # Include
        res = find_right(i + 1, cur_x + (1 << (right_blocks[i][0]-1)), cur_y + (1 << (right_blocks[i][1]-1)), cur_z + (1 << (right_blocks[i][2]-1)), count + 1, path + [right_blocks[i]])
        if res: return res
        # Exclude
        return find_right(i + 1, cur_x, cur_y, cur_z, count, path)

    result = find_right(0, 0, 0, 0, 0, [])
    
    if result:
        print("YES")
        for b in result:
            print(f"{b[0]} {b[1]} {b[2]}")
    else:
        print("NO")

solve()
