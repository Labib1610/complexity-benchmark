import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        configs.append((x, y, z))
        idx += 3
        
    # This is a 3D matching problem, which is NP-complete.
    # However, k <= 22, so we can use backtracking with pruning.
    
    selected = []
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        # Heuristic: if remaining configs are not enough, prune
        if t - start_idx < k - count:
            return False
            
        for i in range(start_idx, t):
            x, y, z = configs[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = True
                used_y[y] = True
                used_z[z] = True
                selected.append(configs[i])
                
                if backtrack(i + 1, count + 1):
                    return True
                
                selected.pop()
                used_x[x] = False
                used_y[y] = False
                used_z[z] = False
        return False

    # Sort configs to potentially improve pruning
    configs.sort()
    
    if backtrack(0, 0):
        print("YES")
        for item in selected:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
