import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    rows = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        rows.append((x, y, z))

    ncol = 3 * k
    # DLX structures
    max_nodes = t * 3 + ncol + 1
    U = [0] * max_nodes
    D = [0] * max_nodes
    L = [0] * max_nodes
    R = [0] * max_nodes
    C = [0] * max_nodes  # column header for each node
    S = [0] * (ncol + 1)  # size of column
    H = [-1] * t          # row header

    # header node 0
    for i in range(ncol + 1):
        U[i] = D[i] = i
        L[i] = i - 1
        R[i] = i + 1
    L[0] = ncol
    R[ncol] = 0

    node_cnt = ncol + 1

    def add_node(row_id, col):
        nonlocal node_cnt
        nd = node_cnt
        node_cnt += 1
        C[nd] = col
        S[col] += 1
        # vertical links
        U[nd] = U[col]
        D[nd] = col
        D[U[col]] = nd
        U[col] = nd
        # horizontal links
        if H[row_id] == -1:
            H[row_id] = L[nd] = R[nd] = nd
        else:
            first = H[row_id]
            L[nd] = L[first]
            R[nd] = first
            R[L[first]] = nd
            L[first] = nd

    for rid, (x, y, z) in enumerate(rows):
        add_node(rid, 1 + x)          # columns 1..k for x
        add_node(rid, 1 + k + y)      # columns k+1..2k for y
        add_node(rid, 1 + 2 * k + z)  # columns 2k+1..3k for z

    ans_rows = []

    def cover(col):
        # remove column header
        L[R[col]] = L[col]
        R[L[col]] = R[col]
        i = D[col]
        while i != col:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(col):
        i = U[col]
        while i != col:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[col]] = col
        R[L[col]] = col

    sys.setrecursionlimit(10000)
    found = False

    def search(kd):
        nonlocal found
        if R[0] == 0:
            # all columns covered
            found = True
            return
        # choose column with minimal size
        c = R[0]
        min_s = S[c]
        j = R[c]
        while j != 0:
            if S[j] < min_s:
                min_s = S[j]
                c = j
                if min_s == 0:
                    break
            j = R[j]
        if min_s == 0:
            return
        cover(c)
        r = D[c]
        while r != c:
            ans_rows.append(r)
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            search(kd + 1)
            if found:
                return
            # backtrack
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            ans_rows.pop()
            r = D[r]
        uncover(c)

    search(0)

    if not found:
        print("NO")
        return

    # retrieve original triples from selected rows
    selected = []
    for node in ans_rows:
        # any node in the row, traverse to get row id
        # we stored row id via H mapping: need to map node to row id
        # we can find by walking left until we hit the header node of the row (which is the first node of that row)
        # but easier: we stored rows list, and each row's nodes are linked; we can map by checking which row header contains this node
        # Build a map from node to row id during construction
        pass

    # Build node->row map
    node_to_row = {}
    for rid in range(t):
        if H[rid] == -1:
            continue
        nd = H[rid]
        while True:
            node_to_row[nd] = rid
            nd = R[nd]
            if nd == H[rid]:
                break

    for node in ans_rows:
        selected.append(rows[node_to_row[node]])

    print("YES")
    out = sys.stdout
    for x, y, z in selected:
        out.write(f"{x+1} {y+1} {z+1}\n")

if __name__ == "__main__":
    solve()
