import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    configs = []
    idx = 2
    for _ in range(t):
        configs.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    if t < k:
        print("NO")
        return

    # The matrix B is 3k x 3k.
    # Rows 1..k: x
    # Rows k+1..2k: k+y
    # Rows 2k+1..3k: 2k+z
    # For each (x, y, z):
    # B[x][k+y] += 1
    # B[k+y][2k+z] += 1
    # B[2k+z][x] += 1
    # This is a directed graph with 3k nodes.
    # Each configuration is a set of 3 edges: (x -> k+y), (k+y -> 2k+z), (2k+z -> x).
    # This forms a cycle of length 3.
    # The matrix B is the adjacency matrix of a multigraph where each config adds 3 edges.
    # The determinant of an adjacency matrix of a graph is non-zero iff the graph
    # has a perfect matching (in the bipartite sense) or is a union of disjoint cycles.
    # Here, each config is a 3-cycle. We need to pick k configs such that the 
    # resulting 3k edges form a permutation (a set of disjoint cycles covering all 3k nodes).
    # Since each config is a 3-cycle, we need to pick k configs such that every node
    # has in-degree 1 and out-degree 1.
    
    # Let's use a randomized approach (backtracking with pruning or local search).
    # Given k <= 22, we can try to find a subset of size k.
    
    import time
    start_time = time.time()
    
    while time.time() - start_time < 4.5:
        subset = random.sample(configs, k)
        
        in_degree = [0] * (3 * k + 1)
        out_degree = [0] * (3 * k + 1)
        
        possible = True
        for x, y, z in subset:
            # Edges: x -> k+y, k+y -> 2k+z, 2k+z -> x
            nodes = [x, k + y, 2 * k + z]
            for i in range(3):
                u, v = nodes[i], nodes[(i + 1) % 3]
                out_degree[u] += 1
                in_degree[v] += 1
                if out_degree[u] > 1 or in_degree[v] > 1:
                    possible = False
                    break
            if not possible: break
        
        if possible:
            # Check if all degrees are exactly 1
            if all(d == 1 for d in in_degree[1:]) and all(d == 1 for d in out_degree[1:]):
                print("YES")
                for c in subset:
                    print(f"{c[0]} {c[1]} {c[2]}")
                return

    print("NO")

solve()
