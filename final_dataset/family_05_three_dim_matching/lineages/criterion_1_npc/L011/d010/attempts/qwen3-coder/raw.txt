def solve():
    import sys
    from itertools import permutations
    
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(data[idx])
        y = int(data[idx+1])
        z = int(data[idx+2])
        configs.append((x, y, z))
        idx += 3
    
    # Special case: k=0
    if k == 0:
        print("YES")
        return
    
    # For small k, we can try all permutations of k elements
    # But since k can be up to 22, we need a better approach
    # This is essentially a 3D matching problem or finding a perfect matching in a tripartite hypergraph
    # We'll model this as finding a maximum matching in a bipartite graph representation
    
    # However, given the constraints, we can use a backtracking approach with pruning
    # or try all possible subsets of k configurations from t configurations (if t is not too large)
    
    # Since t <= k^3 and k <= 22, t can be up to 10648, which is too large to check all subsets
    # But we can think of this as finding a perfect matching where each element from 1 to k 
    # appears exactly once in each of the three positions
    
    # Let's model this as a matching problem:
    # We want to select k triplets such that:
    # - All x values are distinct (and cover 1 to k)
    # - All y values are distinct (and cover 1 to k)
    # - All z values are distinct (and cover 1 to k)
    
    # This is equivalent to finding a Latin square or a 3D matching
    
    # For implementation, we'll use backtracking with bitmasks for efficiency
    
    # Create mappings for faster lookup
    freq_used = [0] * (k + 1)  # bitmask for frequencies
    mod_used = [0] * (k + 1)   # bitmask for modulations
    key_used = [0] * (k + 1)   # bitmask for keys
    
    # Precompute which configs use each freq, mod, key for pruning
    freq_configs = [[] for _ in range(k + 1)]
    mod_configs = [[] for _ in range(k + 1)]
    key_configs = [[] for _ in range(k + 1)]
    
    for i, (x, y, z) in enumerate(configs):
        freq_configs[x].append(i)
        mod_configs[y].append(i)
        key_configs[z].append(i)
    
    selected = []
    result = []
    
    def backtrack(pos):
        nonlocal result
        if pos == k:
            result = selected[:]
            return True
            
        # Try to extend the matching
        # We'll try all unused frequencies and see if we can match them
        for freq in range(1, k + 1):
            if freq_used[freq] == 0:  # freq not used yet
                # Find configs with this freq that also have unused mod and key
                for config_idx in freq_configs[freq]:
                    x, y, z = configs[config_idx]
                    if mod_used[y] == 0 and key_used[z] == 0:
                        # Use this config
                        freq_used[x] = 1
                        mod_used[y] = 1
                        key_used[z] = 1
                        selected.append((x, y, z))
                        
                        if backtrack(pos + 1):
                            return True
                            
                        # Backtrack
                        freq_used[x] = 0
                        mod_used[y] = 0
                        key_used[z] = 0
                        selected.pop()
                # If we couldn't find a match for this frequency, no solution
                return False
        return False
    
    # Optimization: start with the least frequent element
    if backtrack(0):
        print("YES")
        for x, y, z in result:
            print(x, y, z)
    else:
        print("NO")

solve()
