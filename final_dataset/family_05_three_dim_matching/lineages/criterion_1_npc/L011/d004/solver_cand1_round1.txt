import sys

def solve():
    # The atomic number of Titanium is 22.
    # The problem asks to select k triplets (x, y, z) such that
    # the set of x's is {1..k}, the set of y's is {1..k}, and the set of z's is {1..k}.
    # This is equivalent to finding a perfect matching in a 3D bipartite-like structure,
    # specifically a Latin Square completion problem or a 3D matching problem.
    # Given k <= 22, we can model this as an exact cover problem or use backtracking.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triplets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triplets.append((x, y, z))
        idx += 3
        
    # Use backtracking with pruning
    # State: used_x, used_y, used_z (bitmasks)
    # Since k is small (up to 22), we can use bitmasks.
    
    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(triplets):
        adj[x].append((y, z, i))
        
    used_y = 0
    used_z = 0
    result = []
    
    def backtrack(x):
        nonlocal used_y, used_z
        if x == k:
            return True
        
        for y, z, i in adj[x]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                used_y |= (1 << y)
                used_z |= (1 << z)
                result.append(triplets[i])
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    if backtrack(0):
        print("YES")
        for x, y, z in result:
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")

solve()
