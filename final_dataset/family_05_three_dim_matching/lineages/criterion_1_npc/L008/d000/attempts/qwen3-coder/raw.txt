from itertools import permutations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    routes = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        routes.append((x, y, z))
        index += 3
    
    # Try all permutations of k routes from the available routes
    from itertools import combinations
    
    # For small k, we can try all combinations of k routes and check if they form a perfect matching
    for combo in combinations(range(t), k):
        selected_routes = [routes[i] for i in combo]
        
        # Check if this combination uses each source, transformer, and destination exactly once
        sources = [r[0] for r in selected_routes]
        transformers = [r[1] for r in selected_routes]
        destinations = [r[2] for r in selected_routes]
        
        if len(set(sources)) == k and len(set(transformers)) == k and len(set(destinations)) == k:
            # Verify that sources, transformers, destinations are exactly 1..k
            if set(sources) == set(range(1, k+1)) and set(transformers) == set(range(1, k+1)) and set(destinations) == set(range(1, k+1)):
                print("YES")
                for r in selected_routes:
                    print(r[0], r[1], r[2])
                return
    
    print("NO")

solve()
