import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    recipes = []
    idx = 2
    for _ in range(m):
        recipes.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    # This is a 3D matching problem, which is NP-hard.
    # However, n is at most 22. We can use backtracking with pruning.
    
    used_b = [False] * (n + 1)
    used_c = [False] * (n + 1)
    used_e = [False] * (n + 1)
    
    chosen = []
    
    def backtrack(start_idx, count):
        if count == n:
            return True
        
        for i in range(start_idx, m):
            b, c, e = recipes[i]
            if not used_b[b] and not used_c[c] and not used_e[e]:
                used_b[b] = True
                used_c[c] = True
                used_e[e] = True
                chosen.append(recipes[i])
                
                if backtrack(i + 1, count + 1):
                    return True
                
                chosen.pop()
                used_b[b] = False
                used_c[c] = False
                used_e[e] = False
        return False

    # Optimization: sort recipes to try to find a solution faster
    # Or just run backtracking. Given n <= 22, it might be tight.
    # But the constraints on b, c, e being <= n make it a sparse 3D matching.
    
    if backtrack(0, 0):
        print("YES")
        for r in chosen:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
