import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # We need to select k triplets (x_i, y_i, z_i) such that
    # all x_i are distinct, all y_i are distinct, and all z_i are distinct.
    # This is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph,
    # or more simply, finding a set of k edges in a 3D grid that form a Latin Square.
    # Since k <= 22, we can use backtracking with pruning.
    
    edges = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        edges.append((x, y, z))
        idx += 3
        
    # Group edges by x to simplify the search
    adj = [[] for _ in range(k)]
    for e in edges:
        adj[e[0]].append(e)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x):
        if x == k:
            return True
        
        for edge in adj[x]:
            _, y, z = edge
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(edge)
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for e in result:
            print(f"{e[0]+1} {e[1]+1} {e[2]+1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
