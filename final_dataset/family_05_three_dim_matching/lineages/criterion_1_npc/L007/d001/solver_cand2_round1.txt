import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    points = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx+1]) - 1
        c = int(input_data[idx+2]) - 1
        points.append((a, b, c))
        idx += 3
        
    # Group points by Alpha level
    adj = [[] for _ in range(k)]
    for p in points:
        adj[p[0]].append((p[1], p[2]))
        
    # We need to pick one point for each Alpha level such that
    # all Beta levels and all Gamma levels are distinct.
    # This is equivalent to finding a system of distinct representatives
    # or a matching in a 3D structure.
    # Since k <= 22, we can use backtracking with pruning.
    
    used_beta = [False] * k
    used_gamma = [False] * k
    result = []
    
    def backtrack(alpha):
        if alpha == k:
            return True
        
        for b, c in adj[alpha]:
            if not used_beta[b] and not used_gamma[c]:
                used_beta[b] = True
                used_gamma[c] = True
                result.append((alpha + 1, b + 1, c + 1))
                
                if backtrack(alpha + 1):
                    return True
                
                result.pop()
                used_beta[b] = False
                used_gamma[c] = False
        return False

    # Sort adj to try more constrained branches first (heuristic)
    for i in range(k):
        adj[i].sort(key=lambda x: (used_beta[x[0]], used_gamma[x[1]]))
        
    if backtrack(0):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
