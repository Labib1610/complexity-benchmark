import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        c = int(input_data[idx]) - 1
        n = int(input_data[idx+1]) - 1
        i = int(input_data[idx+2]) - 1
        triples.append((c, n, i))
        idx += 3

    # We need to select k triples such that each c, n, i appears exactly once.
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking
    # with bitmasks for n and i, iterating through chassis c = 0 to k-1.
    
    adj = [[] for _ in range(k)]
    for c, n, i in triples:
        adj[c].append((n, i))
        
    ans = []
    
    def backtrack(c, used_n, used_i):
        if c == k:
            return True
        
        # Optimization: try to pick triples for chassis c
        for n, i in adj[c]:
            if not (used_n & (1 << n)) and not (used_i & (1 << i)):
                ans.append((c + 1, n + 1, i + 1))
                if backtrack(c + 1, used_n | (1 << n), used_i | (1 << i)):
                    return True
                ans.pop()
        return False

    # Heuristic: sort chassis by number of available options to prune faster
    # But for k=22, simple backtracking with bitmasks is usually sufficient.
    if backtrack(0, 0, 0):
        print("YES")
        for triple in ans:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

solve()
