import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # We need to select k triples (c, n, i) such that each c, n, i appears exactly once.
    # This is equivalent to finding a perfect matching in a 3D matching problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    triples = []
    idx = 2
    for _ in range(t):
        c = int(input_data[idx]) - 1
        n = int(input_data[idx+1]) - 1
        i = int(input_data[idx+2]) - 1
        triples.append((c, n, i))
        idx += 3
        
    # Group triples by c to process one by one
    by_c = [[] for _ in range(k)]
    for c, n, i in triples:
        by_c[c].append((n, i))
        
    used_n = [False] * k
    used_i = [False] * k
    result = []
    
    def backtrack(c_idx):
        if c_idx == k:
            return True
        
        # Heuristic: sort by number of available options to prune early
        options = []
        for n, i in by_c[c_idx]:
            if not used_n[n] and not used_i[i]:
                options.append((n, i))
        
        # Sort options by how many future possibilities they block (MRV heuristic)
        # For k=22, simple backtracking with bitmasks or just careful ordering is usually enough
        for n, i in options:
            used_n[n] = True
            used_i[i] = True
            result.append((c_idx + 1, n + 1, i + 1))
            
            if backtrack(c_idx + 1):
                return True
            
            result.pop()
            used_n[n] = False
            used_i[i] = False
            
        return False

    # Optimization: Sort c_idx by number of available choices
    order = sorted(range(k), key=lambda x: len(by_c[x]))
    
    # Re-map to handle the sorted order
    new_by_c = [by_c[i] for i in order]
    
    def backtrack_optimized(idx_in_order):
        if idx_in_order == k:
            return True
        
        c = order[idx_in_order]
        options = []
        for n, i in by_c[c]:
            if not used_n[n] and not used_i[i]:
                options.append((n, i))
        
        for n, i in options:
            used_n[n] = True
            used_i[i] = True
            result.append((c + 1, n + 1, i + 1))
            if backtrack_optimized(idx_in_order + 1):
                return True
            result.pop()
            used_n[n] = False
            used_i[i] = False
        return False

    if backtrack_optimized(0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

solve()
