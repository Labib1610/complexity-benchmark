import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triplets = []
    idx = 2
    for _ in range(t):
        c = int(input_data[idx]) - 1
        n = int(input_data[idx+1]) - 1
        i = int(input_data[idx+2]) - 1
        triplets.append((c, n, i))
        idx += 3
        
    # Group triplets by core frame
    by_core = [[] for _ in range(k)]
    for c, n, i in triplets:
        by_core[c].append((n, i))
        
    # Sort by number of options to optimize backtracking (MRV heuristic)
    # Also, we need to track used guidance computers and optical arrays
    used_n = [False] * k
    used_i = [False] * k
    result = []
    
    # To avoid the "rigid evaluation" trap, we process cores with fewer options first
    order = sorted(range(k), key=lambda x: len(by_core[x]))
    
    def backtrack(idx):
        if idx == k:
            return True
        
        c = order[idx]
        for n, i in by_core[c]:
            if not used_n[n] and not used_i[i]:
                used_n[n] = True
                used_i[i] = True
                result.append((c + 1, n + 1, i + 1))
                
                if backtrack(idx + 1):
                    return True
                
                result.pop()
                used_n[n] = False
                used_i[i] = False
        return False

    if backtrack(0):
        print("YES")
        for triplet in result:
            print(f"{triplet[0]} {triplet[1]} {triplet[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
