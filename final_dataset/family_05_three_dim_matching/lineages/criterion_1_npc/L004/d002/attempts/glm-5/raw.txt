import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    configs = []
    idx = 2
    for _ in range(t):
        p1 = int(data[idx]); p2 = int(data[idx+1]); p3 = int(data[idx+2])
        idx += 3
        configs.append((p1, p2, p3))
    
    # For each value v in 1..k, collect configurations where p1 == v
    by_p1 = [[] for _ in range(k+1)]
    for i, (a, b, c) in enumerate(configs):
        by_p1[a].append(i)
    
    # Check feasibility: every primary value must have at least one config
    for v in range(1, k+1):
        if not by_p1[v]:
            print("NO")
            return
    
    # We need to select exactly one config for each primary value (since we need k configs
    # and all k primary values must appear, each exactly once).
    # Then check if secondary and tertiary values each cover 1..k.
    
    # Try all combinations: for each primary value, try each config with that primary.
    # k <= 22, but number of configs per primary could be large.
    # However total configs t <= k^3, and we need a perfect matching in a 3D sense.
    
    # This is equivalent to finding a system of distinct representatives:
    # choose one config per primary value such that secondary values are all distinct
    # and tertiary values are all distinct, each covering 1..k.
    
    # We can model as: for each primary value v, we have a set of (b,c) pairs.
    # We need to pick one (b,c) per v such that b's form a permutation of 1..k and c's form a permutation of 1..k.
    
    # This is a 3-dimensional matching problem. With k <= 22, we can use DP with bitmask.
    # State: mask of used secondary values, mask of used tertiary values, and we process primary values 1..k.
    # But 2^22 * 2^22 is too large.
    
    # However, we can use the fact that we process primary values one by one, and the state is (mask_b, mask_c).
    # That's 2^22 * 2^22 = 2^44, too large.
    
    # Alternative: since we need both b and c to be permutations, we can think of it as
    # finding a perfect matching in a bipartite graph where left side is primary values,
    # right side is (b,c) pairs, but with the constraint that b's and c's are each permutations.
    
    # With k <= 22, maybe we can use inclusion-exclusion or a smarter approach.
    # Actually, we can use the fact that the number of configs per primary is limited.
    
    # Let's try a backtracking approach with pruning, since k <= 22 but the structure might be sparse.
    
    # For each primary value, list of (b, c) pairs.
    options = []
    for v in range(1, k+1):
        opts = []
        for i in by_p1[v]:
            a, b, c = configs[i]
            opts.append((b, c, i))
        options.append(opts)
    
    # Sort options by number of choices (heuristic for backtracking)
    order = sorted(range(k), key=lambda x: len(options[x]))
    
    used_b = 0
    used_c = 0
    chosen = [None] * k  # chosen[i] = config index for primary value order[i]
    
    # Precompute for each position the list of (b, c, config_idx)
    # We'll do backtracking in the order given by 'order'
    
    def backtrack(pos):
        nonlocal used_b, used_c
        if pos == k:
            return True
        p = order[pos]
        for (b, c, ci) in options[p]:
            mb = 1 << (b-1)
            mc = 1 << (c-1)
            if (used_b & mb) == 0 and (used_c & mc) == 0:
                used_b |= mb
                used_c |= mc
                chosen[pos] = ci
                if backtrack(pos+1):
                    return True
                used_b &= ~mb
                used_c &= ~mc
                chosen[pos] = None
        return False
    
    if backtrack(0):
        print("YES")
        # Map chosen back to primary values
        result = [None] * k
        for pos in range(k):
            p = order[pos]
            result[p] = chosen[pos]
        for v in range(k):
            a, b, c = configs[result[v]]
            print(f"{a} {b} {c}")
    else:
        print("NO")

solve()
