import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    configs = []
    idx = 2
    for _ in range(m):
        configs.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    # We need to select n configurations such that:
    # 1. All first parameters are distinct (1..n)
    # 2. All second parameters are distinct (1..n)
    # 3. All third parameters are distinct (1..n)
    # This is equivalent to finding a transversal in a 3D Latin Square or 
    # an exact cover problem. Since n <= 22, we use backtracking with pruning.
    
    # Group configs by first parameter
    by_first = [[] for _ in range(n + 1)]
    for c in configs:
        by_first[c[0]].append(c)
        
    used_second = [False] * (n + 1)
    used_third = [False] * (n + 1)
    result = []

    def backtrack(first_idx):
        if first_idx > n:
            return True
        
        for c in by_first[first_idx]:
            s, t = c[1], c[2]
            if not used_second[s] and not used_third[t]:
                used_second[s] = True
                used_third[t] = True
                result.append(c)
                
                if backtrack(first_idx + 1):
                    return True
                
                result.pop()
                used_second[s] = False
                used_third[t] = False
        return False

    if backtrack(1):
        print("YES")
        for c in result:
            print(f"{c[0]} {c[1]} {c[2]}")
    else:
        print("NO")

solve()
