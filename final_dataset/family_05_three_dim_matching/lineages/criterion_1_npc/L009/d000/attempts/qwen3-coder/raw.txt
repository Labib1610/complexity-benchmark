from itertools import permutations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    triples = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        triples.append((x, y, z))
        index += 3
    
    # We need to find a subset of k triples such that:
    # - All x's are distinct
    # - All y's are distinct
    # - All z's are distinct
    
    # Try all combinations of k triples from the list
    from itertools import combinations
    
    for combo in combinations(range(t), k):
        # Check if this combination satisfies the constraints
        used_x = [False] * (k + 1)
        used_y = [False] * (k + 1)
        used_z = [False] * (k + 1)
        selected = []
        
        valid = True
        for i in combo:
            x, y, z = triples[i]
            if used_x[x] or used_y[y] or used_z[z]:
                valid = False
                break
            used_x[x] = True
            used_y[y] = True
            used_z[z] = True
            selected.append((x, y, z))
        
        if valid:
            print("YES")
            for x, y, z in selected:
                print(x, y, z)
            return
    
    print("NO")

solve()
