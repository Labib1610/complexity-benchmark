import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    triples = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        triples.append((x, y, z))

    ncol = 3 * k  # columns: 0..k-1 specialists, k..2k-1 machines, 2k..3k-1 times

    # DLX structures
    max_nodes = t * 3 + ncol + 1
    left = [0] * max_nodes
    right = [0] * max_nodes
    up = [0] * max_nodes
    down = [0] * max_nodes
    col = [0] * max_nodes
    row_id = [0] * max_nodes
    size = [0] * (ncol + 1)  # column sizes, 1-indexed for headers

    header = 0
    # init header links
    for i in range(ncol + 1):
        left[i] = i - 1
        right[i] = i + 1
        up[i] = down[i] = i
    left[0] = ncol
    right[ncol] = 0

    node_cnt = ncol + 1  # next free node index

    # add rows
    for rid, (x, y, z) in enumerate(triples, 1):
        cols = [x, k + y, 2 * k + z]  # zero‑based column numbers
        first = None
        for c in cols:
            c += 1  # shift to 1‑based for DLX headers
            node = node_cnt
            node_cnt += 1
            col[node] = c
            row_id[node] = rid
            # vertical links
            up[node] = up[c]
            down[node] = c
            down[up[c]] = node
            up[c] = node
            size[c] += 1
            # horizontal links
            if first is None:
                left[node] = right[node] = node
                first = node
            else:
                left[node] = left[first]
                right[node] = first
                right[left[first]] = node
                left[first] = node

    solution_rows = []

    sys.setrecursionlimit(10000)

    def cover(c):
        # remove column header
        left[right[c]] = left[c]
        right[left[c]] = right[c]
        # remove rows
        i = down[c]
        while i != c:
            j = right[i]
            while j != i:
                down[up[j]] = down[j]
                up[down[j]] = up[j]
                size[col[j]] -= 1
                j = right[j]
            i = down[i]

    def uncover(c):
        i = up[c]
        while i != c:
            j = left[i]
            while j != i:
                size[col[j]] += 1
                down[up[j]] = j
                up[down[j]] = j
                j = left[j]
            i = up[i]
        left[right[c]] = c
        right[left[c]] = c

    def search():
        if right[header] == header:
            return True
        # choose column with minimal size
        c = right[header]
        min_sz = size[c]
        j = right[c]
        while j != header:
            if size[j] < min_sz:
                c = j
                min_sz = size[j]
                if min_sz == 0:
                    break
            j = right[j]
        if min_sz == 0:
            return False
        cover(c)
        r = down[c]
        while r != c:
            solution_rows.append(row_id[r])
            j = right[r]
            while j != r:
                cover(col[j])
                j = right[j]
            if search():
                return True
            # backtrack
            j = left[r]
            while j != r:
                uncover(col[j])
                j = left[j]
            solution_rows.pop()
            r = down[r]
        uncover(c)
        return False

    if search():
        out = ["YES"]
        for rid in solution_rows:
            x, y, z = triples[rid - 1]
            out.append(f"{x+1} {y+1} {z+1}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
