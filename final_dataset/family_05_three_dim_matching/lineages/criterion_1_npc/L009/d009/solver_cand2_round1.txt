import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    paths = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        paths.append((x, y, z))
        idx += 3
        
    # The condition is that the sum of 1/(s-x) + 1/(s-(k+y)) + 1/(s-(2k+z))
    # for k paths must equal sum_{j=1}^{3k} 1/(s-j).
    # This means the multiset of all 3k denominators {x_i, k+y_i, 2k+z_i}
    # must be exactly {1, 2, ..., 3k}.
    # This implies:
    # 1. The set of all x_i must be {1, 2, ..., k}
    # 2. The set of all y_i must be {1, 2, ..., k}
    # 3. The set of all z_i must be {1, 2, ..., k}
    
    # This is a variation of the Exact Cover problem or 3D Matching.
    # Since k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    result = []
    
    def backtrack(idx, count):
        if count == k:
            return True
        if idx == t:
            return False
        
        # Pruning: if remaining paths are not enough
        if count + (t - idx) < k:
            return False
            
        x, y, z = paths[idx]
        
        # Try including paths[idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append(paths[idx])
            if backtrack(idx + 1, count + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try excluding paths[idx]
        if backtrack(idx + 1, count):
            return True
            
        return False

    if backtrack(0, 0):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
