import sys
from itertools import combinations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    paths = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        paths.append((x, y, z))
        index += 3
    
    # Precompute capacities
    C = [0] * (3 * k + 1)
    C[1] = 1
    for i in range(2, 3 * k + 1):
        C[i] = (k + 1) * C[i-1] + 1
    
    # Compute target sum
    target = sum(C[1:3*k + 1])
    
    # For each path, compute its energy signature
    path_energies = []
    for x, y, z in paths:
        level1 = x
        level2 = k + y
        level3 = 2 * k + z
        energy = C[level1] + C[level2] + C[level3]
        path_energies.append(energy)
    
    # Try all combinations of k paths
    from itertools import combinations
    for indices in combinations(range(t), k):
        total_energy = sum(path_energies[i] for i in indices)
        if total_energy == target:
            print("YES")
            for i in indices:
                x, y, z = paths[i]
                print(x, y, z)
            return
    
    print("NO")

if __name__ == "__main__":
    main()
