import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    links = []
    idx = 2
    for _ in range(t):
        links.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    # The condition sum(50^x) = sum(50^y) = sum(50^z) = sum_{i=1}^k 50^i
    # is equivalent to saying that the multiset of x's, y's, and z's
    # must each be {1, 2, ..., k}.
    # This is because 50 is larger than k (k <= 22), so the base-50 
    # representation is unique.
    
    target = [0] * (k + 1)
    for i in range(1, k + 1):
        target[i] = 1

    # We need to select k links (x_i, y_i, z_i) such that:
    # count(x) = {1:1, 2:1, ..., k:1}
    # count(y) = {1:1, 2:1, ..., k:1}
    # count(z) = {1:1, 2:1, ..., k:1}
    
    # This is an Exact Cover problem. Since k is small (up to 22),
    # we can use backtracking with pruning or randomized search.
    
    def backtrack(idx, current_links, count_x, count_y, count_z):
        if len(current_links) == k:
            return current_links
        
        if idx == t:
            return None
        
        # Pruning: if remaining links are not enough to fill the counts
        rem = t - idx
        if len(current_links) + rem < k:
            return None
            
        # Try including links[idx]
        x, y, z = links[idx]
        if count_x[x] == 0 and count_y[y] == 0 and count_z[z] == 0:
            count_x[x] = 1
            count_y[y] = 1
            count_z[z] = 1
            current_links.append(links[idx])
            res = backtrack(idx + 1, current_links, count_x, count_y, count_z)
            if res is not None:
                return res
            current_links.pop()
            count_x[x] = 0
            count_y[y] = 0
            count_z[z] = 0
            
        # Try excluding links[idx]
        return backtrack(idx + 1, current_links, count_x, count_y, count_z)

    # Shuffle to randomize search
    random.shuffle(links)
    
    res = backtrack(0, [], [0]*(k+1), [0]*(k+1), [0]*(k+1))
    
    if res:
        print("YES")
        for link in res:
            print(f"{link[0]} {link[1]} {link[2]}")
    else:
        print("NO")

solve()
