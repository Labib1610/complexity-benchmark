import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    paths = []
    idx = 2
    for _ in range(t):
        paths.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    # C_1 = 2
    # C_i = 2 + sum_{j=1}^{i-1} C_j
    # C_2 = 2 + C_1 = 4
    # C_3 = 2 + C_1 + C_2 = 8
    # C_i = 2^(i)
    # Let's verify:
    # C_1 = 2^1 = 2
    # C_2 = 2^2 = 4
    # C_3 = 2^3 = 8
    # C_i = 2^i
    # Target = sum_{i=1}^{3k} 2^i = 2^{3k+1} - 2
    
    capacities = [2**(i) for i in range(1, 3*k + 1)]
    target = sum(capacities)
    
    # Path (x, y, z) uses levels x, k+y, 2k+z
    # Energy = C_x + C_{k+y} + C_{2k+z}
    # We need to pick k paths such that sum of energies = target
    # Let v_i be a binary variable for path i
    # sum_{i=1}^t v_i * (C_{x_i} + C_{k+y_i} + C_{2k+z_i}) = target
    # sum_{i=1}^t v_i = k
    
    # Since k is small (up to 22), we can use meet-in-the-middle or recursion with pruning.
    # Given the constraints and the nature of powers of 2, this is a variation of the subset sum problem.
    
    path_values = []
    for x, y, z in paths:
        val = capacities[x-1] + capacities[k+y-1] + capacities[2*k+z-1]
        path_values.append(val)
        
    memo = {}

    def find_subset(idx, count, current_sum):
        if count == k:
            return [] if current_sum == target else None
        if idx == t or count > k or (k - count) > (t - idx):
            return None
        
        state = (idx, count, current_sum)
        if state in memo:
            return memo[state]
        
        # Include path idx
        res = find_subset(idx + 1, count + 1, current_sum + path_values[idx])
        if res is not None:
            memo[state] = [paths[idx]] + res
            return memo[state]
        
        # Exclude path idx
        res = find_subset(idx + 1, count, current_sum)
        if res is not None:
            memo[state] = res
            return memo[state]
        
        memo[state] = None
        return None

    # The target is 2^(3k+1) - 2.
    # Since k <= 22, 3k+1 <= 67. This fits in a 64-bit integer.
    # Meet-in-the-middle is better for k=22.
    
    mid = t // 2
    left_paths = path_values[:mid]
    right_paths = path_values[mid:]
    
    left_map = {} # (count, sum) -> list of indices
    def get_left(idx, count, current_sum, current_indices):
        if count <= k:
            if (count, current_sum) not in left_map:
                left_map[(count, current_sum)] = current_indices
        if idx == len(left_paths) or count == k:
            return
        get_left(idx + 1, count + 1, current_sum + left_paths[idx], current_indices + [idx])
        get_left(idx + 1, count, current_sum, current_indices)

    get_left(0, 0, 0, [])
    
    def check_right(idx, count, current_sum, current_indices):
        if count <= k:
            needed_count = k - count
            needed_sum = target - current_sum
            if (needed_count, needed_sum) in left_map:
                return left_map[(needed_count, needed_sum)] + [i + mid for i in current_indices]
        if idx == len(right_paths) or count == k:
            return None
        res = check_right(idx + 1, count + 1, current_sum + right_paths[idx], current_indices + [idx])
        if res: return res
        return check_right(idx + 1, count, current_sum, current_indices)

    result_indices = check_right(0, 0, 0, [])
    
    if result_indices:
        print("YES")
        for i in result_indices:
            print(f"{paths[i][0]} {paths[i][1]} {paths[i][2]}")
    else:
        print("NO")

solve()
