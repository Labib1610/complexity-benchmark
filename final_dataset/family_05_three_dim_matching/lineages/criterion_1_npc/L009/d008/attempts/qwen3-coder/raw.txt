from itertools import combinations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    paths = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        paths.append((x, y, z))
        index += 3
    
    # Precompute capacities
    C = [0] * (3 * k + 1)
    C[1] = 2
    for i in range(2, 3*k + 1):
        C[i] = 2 + sum(C[j] for j in range(1, i))
    
    total_capacity = sum(C[i] for i in range(1, 3*k + 1))
    
    # Try all combinations of k paths
    from itertools import combinations
    
    for combo in combinations(range(t), k):
        used = set()
        total_energy = 0
        
        # Check if this combination uses each level exactly once
        levels_used = [0] * (3 * k + 1)
        
        valid = True
        for idx in combo:
            x, y, z = paths[idx]
            level1, level2, level3 = x, k + y, 2*k + z
            
            if levels_used[level1] == 1 or levels_used[level2] == 1 or levels_used[level3] == 1:
                valid = False
                break
                
            levels_used[level1] = 1
            levels_used[level2] = 1
            levels_used[level3] = 1
            total_energy += C[level1] + C[level2] + C[level3]
            
        if valid and total_energy == total_capacity:
            print("YES")
            for idx in combo:
                x, y, z = paths[idx]
                print(x, y, z)
            return
    
    print("NO")

solve()
