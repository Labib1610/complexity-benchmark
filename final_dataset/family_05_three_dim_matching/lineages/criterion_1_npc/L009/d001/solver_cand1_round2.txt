import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    combinations = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        combinations.append((x, y, z))
        idx += 3

    # We need to select k combinations such that:
    # Union of x's is {0..k-1}, Union of y's is {0..k-1}, Union of z's is {0..k-1}
    # Since we pick exactly k combinations, this implies each x, y, z must appear exactly once.
    # This is equivalent to finding a transversal in a 3D matching problem.
    # Given k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * k
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(idx_comb, count):
        if count == k:
            return True
        if idx_comb == t:
            return False
        
        # Pruning: if remaining combinations are not enough
        if t - idx_comb < k - count:
            return False
            
        x, y, z = combinations[idx_comb]
        
        # Try including this combination
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append((x + 1, y + 1, z + 1))
            if backtrack(idx_comb + 1, count + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try excluding this combination
        if backtrack(idx_comb + 1, count):
            return True
            
        return False

    # Sort combinations to potentially hit a solution faster
    if backtrack(0, 0):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
