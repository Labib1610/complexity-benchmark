import sys
sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    triples = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        triples.append((x, y, z))
    # adjacency for quick lookup
    by_x = [[] for _ in range(k)]
    by_y = [[] for _ in range(k)]
    by_z = [[] for _ in range(k)]
    for idx, (x, y, z) in enumerate(triples):
        by_x[x].append(idx)
        by_y[y].append(idx)
        by_z[z].append(idx)

    # state
    used = [False] * t
    chosen = []
    uncovered_x = (1 << k) - 1
    uncovered_y = (1 << k) - 1
    uncovered_z = (1 << k) - 1

    # ordering of triples: maybe sort by degree (fewest other triples sharing its elements)
    deg = [len(by_x[x]) + len(by_y[y]) + len(by_z[z]) for (x, y, z) in triples]
    order = sorted(range(t), key=lambda i: deg[i])

    # map from position in order to original index
    pos_to_idx = order

    # precompute for pruning: for each remaining position, union of elements that can still be covered
    suffix_union_x = [0] * (t + 1)
    suffix_union_y = [0] * (t + 1)
    suffix_union_z = [0] * (t + 1)
    for i in range(t - 1, -1, -1):
        idx = pos_to_idx[i]
        x, y, z = triples[idx]
        suffix_union_x[i] = suffix_union_x[i + 1] | (1 << x)
        suffix_union_y[i] = suffix_union_y[i + 1] | (1 << y)
        suffix_union_z[i] = suffix_union_z[i + 1] | (1 << z)

    found = False
    ans = []

    def dfs(pos: int, cnt: int, ux: int, uy: int, uz: int):
        nonlocal found, ans
        if found:
            return
        # prune: not enough slots left
        if cnt > k:
            return
        if cnt + (t - pos) < k:
            return
        # prune: uncovered elements must be possible to cover later
        need_x = ux
        need_y = uy
        need_z = uz
        if (need_x & ~suffix_union_x[pos]) != 0:
            return
        if (need_y & ~suffix_union_y[pos]) != 0:
            return
        if (need_z & ~suffix_union_z[pos]) != 0:
            return
        if cnt == k:
            if ux == 0 and uy == 0 and uz == 0:
                found = True
                ans = chosen.copy()
            return
        if pos == t:
            return
        # option 1: take this triple
        idx = pos_to_idx[pos]
        if not used[idx]:
            used[idx] = True
            chosen.append(idx)
            x, y, z = triples[idx]
            dfs(pos + 1, cnt + 1,
                ux & ~(1 << x),
                uy & ~(1 << y),
                uz & ~(1 << z))
            chosen.pop()
            used[idx] = False
            if found:
                return
        # option 2: skip
        dfs(pos + 1, cnt, ux, uy, uz)

    dfs(0, 0, uncovered_x, uncovered_y, uncovered_z)

    out = []
    if found:
        out.append("YES")
        for idx in ans:
            x, y, z = triples[idx]
            out.append(f"{x+1} {y+1} {z+1}")
    else:
        out.append("NO")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
