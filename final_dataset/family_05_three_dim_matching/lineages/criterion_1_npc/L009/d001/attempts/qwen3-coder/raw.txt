from itertools import combinations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    idx = 2
    allowed = []
    for i in range(t):
        x = int(data[idx]) - 1  # Convert to 0-based indexing
        y = int(data[idx+1]) - 1
        z = int(data[idx+2]) - 1
        allowed.append((x, y, z))
        idx += 3
    
    # We need to find a subset of k combinations such that each operator, apparatus, and shift appears at least once
    # This is essentially finding a matching in a 3-dimensional structure
    
    # Try all combinations of k allowed combinations
    for combo_indices in combinations(range(t), k):
        operators_used = [False] * k
        apparatuses_used = [False] * k
        shifts_used = [False] * k
        
        valid = True
        selected_combos = []
        
        for i in combo_indices:
            x, y, z = allowed[i]
            operators_used[x] = True
            apparatuses_used[y] = True
            shifts_used[z] = True
            selected_combos.append((x+1, y+1, z+1))  # Convert back to 1-based indexing
            
        # Check if all operators, apparatuses, and shifts are covered
        if all(operators_used) and all(apparatuses_used) and all(shifts_used):
            print("YES")
            for combo in selected_combos:
                print(combo[0], combo[1], combo[2])
            return
            
    print("NO")

solve()
