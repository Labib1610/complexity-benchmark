```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # specialists are 1-indexed, convert to 0-indexed
    # We need to pick k triplets (x, y, z) such that:
    # 1. Each specialist x in {0..k-1} is used exactly once.
    # 2. Each machine y in {0..k-1} is used exactly once.
    # 3. Each time slot z in {0..k-1} is used exactly once.
    
    # This is equivalent to finding a perfect matching in a 3D bipartite-like structure
    # or more simply, an exact cover problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    adj = [[] for _ in range(k)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        adj[x].append((y, z))
        idx += 3
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(spec_idx):
        if spec_idx == k:
            return True
        
        for y, z in adj[spec_idx]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((spec_idx + 1, y + 1, z + 1))
                
                if backtrack(spec_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    # Sort specialists by number of options to optimize backtracking
    # But since we must pick each specialist exactly once, the order of 
    # specialists is fixed (0 to k-1). We can sort the options for each.
    if backtrack(0):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
```
