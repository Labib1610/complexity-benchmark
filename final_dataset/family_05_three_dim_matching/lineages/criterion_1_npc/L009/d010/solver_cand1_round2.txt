import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    paths = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        paths.append((x, y, z))
        idx += 3

    target = (1 << (k + 1)) - 2
    
    # We need to select k paths (x_i, y_i, z_i) such that:
    # sum(2^x_i) = target, sum(2^y_i) = target, sum(2^z_i) = target
    # Since 1 <= x, y, z <= k, the sum of 2^x_i is a binary representation
    # where the i-th bit is 1 if frequency i is used.
    # target = 2^1 + 2^2 + ... + 2^k.
    # This means we must use each frequency exactly once, each polarization exactly once,
    # and each phase shift exactly once.
    # This is equivalent to finding a permutation P and Q of {1..k} such that
    # the set of paths is {(i, P[i], Q[i]) for i in 1..k}.
    
    # This is a 3D matching problem, but with a specific structure.
    # We can model this as finding a perfect matching in a 3-partite hypergraph
    # or using backtracking with pruning. Given k <= 22, we use backtracking.
    
    adj = [[] for _ in range(k + 1)]
    for x, y, z in paths:
        adj[x].append((y, z))
        
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    result = []

    def backtrack(x):
        if x > k:
            return True
        for y, z in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x, y, z))
                if backtrack(x + 1):
                    return True
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
