def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    paths = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        index += 3
        paths.append((x, y, z))
    
    # Target sum
    target = (1 << (3 * k + 1)) - 2
    
    # Function to compute energy signature
    def energy(x, y, z):
        return (1 << x) + (1 << (k + y)) + (1 << (2 * k + z))
    
    # Precompute energies for all paths
    energies = [energy(x, y, z) for x, y, z in paths]
    
    # Bitmask DP approach
    # We'll use meet-in-the-middle since k can be up to 22 and t up to k^3
    
    # Split paths into two halves
    mid = t // 2
    first_half = paths[:mid]
    second_half = paths[mid:]
    first_energies = energies[:mid]
    second_energies = energies[mid:]
    
    # Generate all subsets of first half with their counts and sums
    first_dict = {}
    for mask in range(1 << mid):
        count = bin(mask).count('1')
        total = 0
        subset = []
        for i in range(mid):
            if mask & (1 << i):
                total += first_energies[i]
                subset.append(first_half[i])
        if count not in first_dict:
            first_dict[count] = []
        first_dict[count].append((total, subset))
    
    # Generate all subsets of second half with their counts and sums
    second_dict = {}
    for mask in range(1 << (t - mid)):
        count = bin(mask).count('1')
        total = 0
        subset = []
        for i in range(t - mid):
            if mask & (1 << i):
                total += second_energies[i]
                subset.append(second_half[i])
        if count not in second_dict:
            second_dict[count] = []
        second_dict[count].append((total, subset))
    
    # Check all combinations
    for i in range(k + 1):
        j = k - i
        if i in first_dict and j in second_dict:
            # Create a dictionary for second half: sum -> subset
            second_map = {}
            for total, subset in second_dict[j]:
                if total not in second_map:
                    second_map[total] = subset
                else:
                    # If multiple subsets have same sum, we just need one
                    pass
            
            for total1, subset1 in first_dict[i]:
                needed = target - total1
                if needed in second_map:
                    subset2 = second_map[needed]
                    result = subset1 + subset2
                    print("YES")
                    for x, y, z in result:
                        print(x, y, z)
                    return
    
    print("NO")

solve()
