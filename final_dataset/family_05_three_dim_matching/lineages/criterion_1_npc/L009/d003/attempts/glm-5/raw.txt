import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    paths = []
    idx = 2
    for _ in range(t):
        x = int(data[idx]); y = int(data[idx+1]); z = int(data[idx+2])
        idx += 3
        paths.append((x, y, z))
    
    # Build adjacency: input x -> list of (y, z)
    adj = [[] for _ in range(k+1)]
    for i, (x, y, z) in enumerate(paths):
        adj[x].append((y, z, i))
    
    # Check feasibility: each input must have at least one path
    for x in range(1, k+1):
        if not adj[x]:
            print("NO")
            return
    
    # State: (mask_y, mask_z) -> (prev_x, prev_state, path_index)
    # We process inputs in order 1..k
    # Use dictionary for DP
    full_mask = (1 << k) - 1
    dp = {}
    dp[(0, 0)] = None  # initial state
    
    for x in range(1, k+1):
        new_dp = {}
        for (my, mz), _ in dp.items():
            for (y, z, pi) in adj[x]:
                yb = 1 << (y-1)
                zb = 1 << (z-1)
                if (my & yb) or (mz & zb):
                    continue
                nmy = my | yb
                nmz = mz | zb
                key = (nmy, nmz)
                if key not in new_dp:
                    new_dp[key] = (x, (my, mz), pi)
        dp = new_dp
        if not dp:
            print("NO")
            return
    
    # Find final state with full masks
    final_key = (full_mask, full_mask)
    if final_key not in dp:
        print("NO")
        return
    
    # Reconstruct
    result = []
    cur_x, cur_state, cur_pi = dp[final_key]
    result.append(paths[cur_pi])
    cur_state = (cur_state[0], cur_state[1])
    
    # We need to backtrack through previous inputs
    # But we only stored one transition per state in the last layer
    # We need to store all layers. Let's redo with layer storage.
    
    # Redo DP with layer storage
    dp_layers = [None] * (k+1)
    dp_layers[0] = {(0, 0): None}
    
    for x in range(1, k+1):
        new_dp = {}
        prev_dp = dp_layers[x-1]
        for (my, mz), _ in prev_dp.items():
            for (y, z, pi) in adj[x]:
                yb = 1 << (y-1)
                zb = 1 << (z-1)
                if (my & yb) or (mz & zb):
                    continue
                nmy = my | yb
                nmz = mz | zb
                key = (nmy, nmz)
                if key not in new_dp:
                    new_dp[key] = (my, mz, pi)
        dp_layers[x] = new_dp
        if not new_dp:
            print("NO")
            return
    
    if final_key not in dp_layers[k]:
        print("NO")
        return
    
    result = []
    cur_state = final_key
    for x in range(k, 0, -1):
        my, mz, pi = dp_layers[x][cur_state]
        result.append(paths[pi])
        cur_state = (my, mz)
    
    print("YES")
    for p in result:
        print(f"{p[0]} {p[1]} {p[2]}")

solve()
