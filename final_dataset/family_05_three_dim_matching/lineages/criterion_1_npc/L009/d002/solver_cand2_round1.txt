import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    links = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        links.append((x, y, z))
        idx += 3
        
    # This is a 3D matching problem, which is NP-complete.
    # Given k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    # Group links by x to optimize search
    adj = [[] for _ in range(k + 1)]
    for link in links:
        adj[link[0]].append(link)
        
    result = []
    
    def backtrack(x_idx):
        if x_idx > k:
            return True
        
        for link in adj[x_idx]:
            _, y, z = link
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(link)
                
                if backtrack(x_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for link in result:
            print(f"{link[0]} {link[1]} {link[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
