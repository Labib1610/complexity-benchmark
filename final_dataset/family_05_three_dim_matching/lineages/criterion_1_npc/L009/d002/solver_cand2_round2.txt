import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3

    # Group triples by operator (x)
    by_x = [[] for _ in range(k)]
    for x, y, z in triples:
        by_x[x].append((y, z))

    # We need to pick one (y, z) for each x such that all y's are distinct
    # and all z's are distinct. This is an exact cover problem.
    # Since k <= 22, we use backtracking with bitmasks.
    
    used_y = 0
    used_z = 0
    result = []

    def backtrack(x):
        nonlocal used_y, used_z
        if x == k:
            return True
        
        for y, z in by_x[x]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                used_y |= (1 << y)
                used_z |= (1 << z)
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    # Optimization: sort by number of choices to prune search tree
    # But for k=22, simple backtracking might be slow.
    # Let's reorder x by the number of available choices.
    order = sorted(range(k), key=lambda i: len(by_x[i]))
    
    # Re-map the backtracking to use the sorted order
    def backtrack_ordered(idx_in_order):
        nonlocal used_y, used_z
        if idx_in_order == k:
            return True
        
        x = order[idx_in_order]
        for y, z in by_x[x]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                used_y |= (1 << y)
                used_z |= (1 << z)
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack_ordered(idx_in_order + 1):
                    return True
                
                result.pop()
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    if backtrack_ordered(0):
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

solve()
