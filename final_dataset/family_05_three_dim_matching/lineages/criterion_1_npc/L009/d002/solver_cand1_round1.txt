import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    combinations = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        combinations.append((x, y, z))
        idx += 3

    # We need to pick k combinations (x_i, y_i, z_i) such that
    # all x_i are distinct, all y_i are distinct, and all z_i are distinct.
    # This is equivalent to finding a Latin Square of size k using the given triples.
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    # Group combinations by x
    by_x = [[] for _ in range(k)]
    for c in combinations:
        by_x[c[0]].append(c)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        for c in by_x[x]:
            _, y, z = c
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(c)
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for c in result:
            print(f"{c[0]+1} {c[1]+1} {c[2]+1}")
    else:
        print("NO")

solve()
