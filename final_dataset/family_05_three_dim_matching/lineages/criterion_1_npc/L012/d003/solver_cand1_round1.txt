import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    if t < k:
        print("NO")
        return
    
    pockets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        pockets.append((x, y, z))
        idx += 3
        
    # We need to select k pockets such that all x, y, z coordinates 0..k-1 are covered.
    # This is an exact cover problem variant. Since k is small (up to 22),
    # but we need exactly k items, we can use backtracking with pruning.
    
    selected = []
    
    def backtrack(start_idx, count, x_mask, y_mask, z_mask):
        if count == k:
            if x_mask == (1 << k) - 1 and y_mask == (1 << k) - 1 and z_mask == (1 << k) - 1:
                return True
            return False
        
        # Pruning: if remaining pockets are not enough to cover missing coordinates
        rem = k - count
        if (bin(x_mask).count('0') - (22 - k)) > rem: return False
        if (bin(y_mask).count('0') - (22 - k)) > rem: return False
        if (bin(z_mask).count('0') - (22 - k)) > rem: return False
        
        for i in range(start_idx, t):
            x, y, z = pockets[i]
            new_x = x_mask | (1 << x)
            new_y = y_mask | (1 << y)
            new_z = z_mask | (1 << z)
            
            selected.append(pockets[i])
            if backtrack(i + 1, count + 1, new_x, new_y, new_z):
                return True
            selected.pop()
            
        return False

    # Optimization: If k is large, simple backtracking might be slow.
    # However, the constraints and problem type suggest a search.
    # Given 5s time limit, let's use a more efficient search order.
    
    if backtrack(0, 0, 0, 0, 0):
        print("YES")
        for p in selected:
            print(f"{p[0]+1} {p[1]+1} {p[2]+1}")
    else:
        print("NO")

solve()
