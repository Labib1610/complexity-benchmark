```python
import sys

sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    triples = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        triples.append((x, y, z))

    # columns: 0..k-1 for x, k..2k-1 for y, 2k..3k-1 for z
    ncol = 3 * k
    # build column lists
    col_nodes = [[] for _ in range(ncol)]
    for idx, (x, y, z) in enumerate(triples):
        col_nodes[x].append(idx)
        col_nodes[k + y].append(idx)
        col_nodes[2 * k + z].append(idx)

    # quick impossibility check
    for c in range(ncol):
        if not col_nodes[c]:
            print("NO")
            return

    # DLX structures
    # each row (triple) will have three nodes linking to its columns
    # We'll store for each column the list of rows (indices) that are still available
    # Use simple recursive exact cover with bitsets for columns and row usage.

    # Represent column availability with bitmask (up to 66 bits -> use int)
    full_mask = (1 << ncol) - 1

    # For each row, precompute its column mask
    row_mask = []
    for (x, y, z) in triples:
        mask = (1 << x) | (1 << (k + y)) | (1 << (2 * k + z))
        row_mask.append(mask)

    # For each column, list rows covering it
    col_rows = [set() for _ in range(ncol)]
    for r, mask in enumerate(row_mask):
        for c in range(ncol):
            if mask >> c & 1:
                col_rows[c].add(r)

    solution = []

    # heuristic: choose column with fewest rows
    def search(col_mask, used_rows):
        if col_mask == 0:
            return True
        # pick column with minimal options
        # iterate over bits set in col_mask
        c = -1
        min_len = 10**9
        m = col_mask
        while m:
            lsb = m & -m
            idx = (lsb.bit_length() - 1)
            sz = len(col_rows[idx] - used_rows)
            if sz < min_len:
                min_len = sz
                c = idx
                if sz == 0:
                    break
            m ^= lsb
        if min_len == 0:
            return False

        rows = list(col_rows[c] - used_rows)
        for r in rows:
            solution.append(r)
            new_used = used_rows | {r}
            new_mask = col_mask & ~row_mask[r]
            # temporarily remove rows that conflict on any column (they share a column)
            # but our used_rows set already prevents reusing same row; we also need to
            # ensure that future selections don't reuse columns already covered, which
            # is handled by col_mask.
            if search(new_mask, new_used):
                return True
            solution.pop()
        return False

    if search(full_mask, set()):
        print("YES")
        for r in solution:
            x, y, z = triples[r]
            print(x + 1, y + 1, z + 1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
