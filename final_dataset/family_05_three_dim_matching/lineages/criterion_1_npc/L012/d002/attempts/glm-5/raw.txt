import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    pts = []
    idx = 2
    for _ in range(t):
        x = int(data[idx]); y = int(data[idx+1]); z = int(data[idx+2])
        pts.append((x, y, z))
        idx += 3

    if k == 0:
        print("YES")
        return
    if t < k:
        print("NO")
        return

    # Build adjacency: for each x, list of (y,z)
    adj = [[] for _ in range(k+1)]
    for i, (x, y, z) in enumerate(pts):
        adj[x].append((y, z, i))

    # Check feasibility: each x must have at least one point
    for x in range(1, k+1):
        if not adj[x]:
            print("NO")
            return

    # We need a perfect matching in a 3-partite 3-uniform hypergraph
    # x is fixed 1..k, need to choose for each x a (y,z) so that y's and z's are permutations
    # Use backtracking with bitsets
    FULL = (1 << k) - 1
    ymask = [0] * (k+1)
    zmask = [0] * (k+1)
    for x in range(1, k+1):
        ym = 0
        zm = 0
        for (y, z, i) in adj[x]:
            ym |= (1 << (y-1))
            zm |= (1 << (z-1))
        ymask[x] = ym
        zmask[x] = zm

    # Quick pruning: union of all y masks must be FULL, same for z
    yu = 0
    zu = 0
    for x in range(1, k+1):
        yu |= ymask[x]
        zu |= zmask[x]
    if yu != FULL or zu != FULL:
        print("NO")
        return

    # Hall's condition check for y and z separately (necessary)
    # For every subset S of x's, |N_y(S)| >= |S| and |N_z(S)| >= |S|
    # k <= 22, 2^22 = 4M, feasible
    # Precompute for each subset the union of ymask and zmask
    # But we can do DP over subsets
    # Actually we can iterate all subsets
    # Let's map x in 1..k to bit (x-1)
    # Precompute union_y[mask], union_z[mask]
    size = 1 << k
    # Use DP: union[mask] = union[mask without lowest bit] | mask_of_lowest_bit
    union_y = [0] * size
    union_z = [0] * size
    # We'll fill iteratively
    # To save memory, maybe compute on the fly? 4M ints is okay (~32MB each)
    # Let's do it
    for mask in range(1, size):
        low = mask & (-mask)
        rest = mask ^ low
        xi = low.bit_length()  # 1-based x
        union_y[mask] = union_y[rest] | ymask[xi]
        union_z[mask] = union_z[rest] | zmask[xi]
        pc = rest.bit_count() + 1
        if union_y[mask].bit_count() < pc or union_z[mask].bit_count() < pc:
            print("NO")
            return

    # Backtracking search
    # Order x's by number of available options (heuristic)
    order = sorted(range(1, k+1), key=lambda x: len(adj[x]))
    # For each x in order, try each (y,z)
    chosen = [-1] * (k+1)  # chosen[x] = index in pts
    used_y = 0
    used_z = 0

    # Precompute list of (y,z,i) for each x in order
    adj_order = [adj[x] for x in order]

    def backtrack(pos, used_y, used_z):
        if pos == k:
            return True
        x = order[pos]
        for (y, z, i) in adj_order[pos]:
            yb = 1 << (y-1)
            zb = 1 << (z-1)
            if (used_y & yb) or (used_z & zb):
                continue
            # Pruning: remaining x's must be able to cover remaining y's and z's
            # remaining mask of x positions
            rem_mask = 0
            for j in range(pos+1, k):
                rem_mask |= (1 << (order[j]-1))
            # available y for remaining = union_y[rem_mask] & ~used_y' but used_y' changes
            # Actually we need: for remaining positions, the union of their y masks minus used_y must cover all remaining y bits
            # remaining y bits needed = FULL & ~(used_y | yb)
            # union of remaining y masks = union_y[rem_mask]
            # Need: (union_y[rem_mask] & ~(used_y | yb)).bit_count() >= (k - pos - 1)
            # Similarly for z
            new_used_y = used_y | yb
            new_used_z = used_z | zb
            rem_y = union_y[rem_mask] & ~new_used_y
            rem_z = union_z[rem_mask] & ~new_used_z
            if rem_y.bit_count() < (k - pos - 1) or rem_z.bit_count() < (k - pos - 1):
                continue
            chosen[x] = i
            if backtrack(pos+1, new_used_y, new_used_z):
                return True
            chosen[x] = -1
        return False

    if backtrack(0, 0, 0):
        print("YES")
        out = []
        for x in range(1, k+1):
            i = chosen[x]
            out.append(f"{pts[i][0]} {pts[i][1]} {pts[i][2]}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        print("NO")

solve()
