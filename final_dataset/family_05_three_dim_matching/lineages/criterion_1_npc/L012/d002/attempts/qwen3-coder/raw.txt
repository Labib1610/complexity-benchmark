from itertools import combinations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    pockets = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        pockets.append((x, y, z))
        index += 3
    
    # We need to find a subset of exactly k pockets such that:
    # - All x coordinates cover {1, 2, ..., k}
    # - All y coordinates cover {1, 2, ..., k}
    # - All z coordinates cover {1, 2, ..., k}
    
    # Try all combinations of k pockets from the t pockets
    from itertools import combinations
    
    for combo in combinations(pockets, k):
        xs = set(p[0] for p in combo)
        ys = set(p[1] for p in combo)
        zs = set(p[2] for p in combo)
        
        if len(xs) == k and len(ys) == k and len(zs) == k:
            # Check if the sets are exactly {1, 2, ..., k}
            if xs == set(range(1, k+1)) and ys == set(range(1, k+1)) and zs == set(range(1, k+1)):
                print("YES")
                for pocket in combo:
                    print(pocket[0], pocket[1], pocket[2])
                return
    
    print("NO")

solve()
