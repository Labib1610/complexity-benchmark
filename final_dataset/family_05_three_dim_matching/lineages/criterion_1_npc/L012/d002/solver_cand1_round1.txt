import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    pockets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        pockets.append((x, y, z))
        idx += 3

    # We need to select k pockets such that:
    # {x_i} = {0..k-1}, {y_i} = {0..k-1}, {z_i} = {0..k-1}
    # This is an exact cover problem. Since k <= 22, we can use backtracking.
    # To optimize, we use bitmasks for the sets of x, y, z covered.
    
    target = (1 << k) - 1
    
    # Pre-filter: only keep pockets that could potentially be part of a solution
    # (though with k=22, simple backtracking with pruning is needed)
    
    selected = []
    
    def backtrack(idx, count, mask_x, mask_y, mask_z):
        if count == k:
            if mask_x == target and mask_y == target and mask_z == target:
                return True
            return False
        
        if idx == t or (k - count) > (t - idx):
            return False
        
        # Pruning: if remaining pockets cannot fill the missing bits
        rem = k - count
        if (bin(mask_x).count('0') - (22 - k)) > rem or \
           (bin(mask_y).count('0') - (22 - k)) > rem or \
           (bin(mask_z).count('0') - (22 - k)) > rem:
            pass # This is a loose heuristic
            
        # Try including pockets[idx]
        x, y, z = pockets[idx]
        if not ((mask_x >> x) & 1) or not ((mask_y >> y) & 1) or not ((mask_z >> z) & 1):
            selected.append(pockets[idx])
            if backtrack(idx + 1, count + 1, mask_x | (1 << x), mask_y | (1 << y), mask_z | (1 << z)):
                return True
            selected.pop()
            
        # Try excluding pockets[idx]
        if backtrack(idx + 1, count, mask_x, mask_y, mask_z):
            return True
            
        return False

    # The search space is large, but the constraints are tight.
    # Using a more efficient approach: Dancing Links or simple recursive search
    # Given the 5s time limit, let's refine the search.
    
    if backtrack(0, 0, 0, 0, 0):
        print("YES")
        for p in selected:
            print(f"{p[0]+1} {p[1]+1} {p[2]+1}")
    else:
        print("NO")

solve()
