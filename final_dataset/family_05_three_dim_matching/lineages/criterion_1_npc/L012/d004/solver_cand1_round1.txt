import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    pockets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        pockets.append((x, y, z))
        idx += 3

    target = (1 << (k + 1)) - 2
    
    # We need to select k pockets such that:
    # sum(2^x_i) = target, sum(2^y_i) = target, sum(2^z_i) = target
    # Since 1 <= x, y, z <= k, and sum(2^i for i in 1..k) = 2^(k+1) - 2,
    # this is equivalent to saying that for each dimension, the set of 
    # coordinates {c_1, c_2, ..., c_k} must be exactly {1, 2, ..., k}.
    
    # This is a variation of the Exact Cover problem.
    # Given the constraints (k <= 22), we can use backtracking with pruning.
    
    selected = []
    
    # Bitmasks for used coordinates in each dimension
    # target_mask = (1 << (k + 1)) - 2
    target_mask = (1 << (k + 1)) - 2
    
    def backtrack(idx, count, mask_x, mask_y, mask_z):
        if count == k:
            if mask_x == target_mask and mask_y == target_mask and mask_z == target_mask:
                return True
            return False
        
        if idx == t:
            return False
        
        # Pruning: if remaining pockets are not enough to fill k
        if count + (t - idx) < k:
            return False
            
        # Try including pockets[idx]
        x, y, z = pockets[idx]
        bit_x, bit_y, bit_z = 1 << x, 1 << y, 1 << z
        
        if not (mask_x & bit_x) and not (mask_y & bit_y) and not (mask_z & bit_z):
            selected.append(pockets[idx])
            if backtrack(idx + 1, count + 1, mask_x | bit_x, mask_y | bit_y, mask_z | bit_z):
                return True
            selected.pop()
            
        # Try excluding pockets[idx]
        if backtrack(idx + 1, count, mask_x, mask_y, mask_z):
            return True
            
        return False

    if backtrack(0, 0, 0, 0, 0):
        print("YES")
        for p in selected:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
