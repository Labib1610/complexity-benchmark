import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    pockets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        pockets.append((x, y, z))
        idx += 3

    target = (4**(k + 1) - 4) // 3
    
    # The energy values are 4^1, 4^2, ..., 4^k.
    # In base 4, these are 10, 100, 1000, ...
    # The sum target is 11...1 (k times) in base 4.
    # This means for each coordinate (x, y, z), we need exactly one pocket 
    # for each value 1, 2, ..., k.
    
    # Represent each pocket as a tuple of powers: (4^(x-1), 4^(y-1), 4^(z-1))
    # Actually, the problem says energy is 4^x.
    # Target sum is sum_{i=1}^k 4^i = 4 + 16 + ... + 4^k.
    # This is equivalent to saying that for each dimension, the set of 
    # coordinates {x_1, ..., x_k} must be exactly {1, 2, ..., k}.
    
    # This is a variation of the Exact Cover problem.
    # Since k is small (up to 22), but we need to pick k items, 
    # we can use backtracking with pruning.
    
    pockets.sort()
    
    selected = []
    
    # State: current index in pockets, current set of x, y, z used
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    def backtrack(idx, count):
        if count == k:
            return True
        if idx == t or (k - count) > (t - idx):
            return False
        
        # Try including pockets[idx]
        x, y, z = pockets[idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            selected.append(pockets[idx])
            if backtrack(idx + 1, count + 1):
                return True
            selected.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try excluding pockets[idx]
        if backtrack(idx + 1, count):
            return True
            
        return False

    if backtrack(0, 0):
        print("YES")
        for p in selected:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
