from itertools import combinations

def solve():
    k, t = map(int, input().split())
    pockets = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        pockets.append((x, y, z))
    
    if k == 0:
        print("YES")
        return
    
    # Try all combinations of k pockets
    for combo in combinations(pockets, k):
        # Check if all coordinates from 1 to k are present in each dimension
        x_coords = set(p[0] for p in combo)
        y_coords = set(p[1] for p in combo)
        z_coords = set(p[2] for p in combo)
        
        if (len(x_coords) == k and len(y_coords) == k and len(z_coords) == k and
            x_coords == set(range(1, k+1)) and y_coords == set(range(1, k+1)) and z_coords == set(range(1, k+1))):
            print("YES")
            for pocket in combo:
                print(pocket[0], pocket[1], pocket[2])
            return
    
    print("NO")

solve()
