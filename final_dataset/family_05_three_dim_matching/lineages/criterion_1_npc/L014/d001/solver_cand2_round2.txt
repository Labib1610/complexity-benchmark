import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    reactions = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        reactions.append((x, y, z))
        idx += 3

    # This is an exact cover problem.
    # We have k base fluids, k catalysts, k thermal profiles.
    # Each reaction (x, y, z) covers (base_x, cat_y, therm_z).
    # Total 3k constraints.
    
    # Group reactions by base fluid
    by_base = [[] for _ in range(k)]
    for r in reactions:
        by_base[r[0]].append(r)
        
    # Use backtracking with pruning
    # State: (base_idx, used_catalysts_mask, used_thermals_mask)
    # Since k <= 22, masks are integers.
    
    memo = {}

    def backtrack(b_idx, cat_mask, therm_mask):
        if b_idx == k:
            return []
        
        state = (b_idx, cat_mask, therm_mask)
        if state in memo:
            return None
        
        for r in by_base[b_idx]:
            _, c, t_val = r
            if not (cat_mask & (1 << c)) and not (therm_mask & (1 << t_val)):
                res = backtrack(b_idx + 1, cat_mask | (1 << c), therm_mask | (1 << t_val))
                if res is not None:
                    return [r] + res
        
        memo[state] = False
        return None

    result = backtrack(0, 0, 0)
    
    if result is not None:
        print("YES")
        for r in result:
            print(f"{r[0]+1} {r[1]+1} {r[2]+1}")
    else:
        print("NO")

solve()
