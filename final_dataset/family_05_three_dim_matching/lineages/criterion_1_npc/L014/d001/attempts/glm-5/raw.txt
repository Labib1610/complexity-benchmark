import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    reactions = []
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        reactions.append((x - 1, y - 1, z - 1))
    
    # Build adjacency list for bipartite matching: base -> list of (catalyst, reaction_index)
    adj = [[] for _ in range(k)]
    for idx, (x, y, z) in enumerate(reactions):
        adj[x].append((y, idx))
    
    # Find maximum matching between bases and catalysts
    match_catalyst = [-1] * k  # catalyst -> base
    match_base = [-1] * k      # base -> catalyst
    
    def dfs(u, visited, match_catalyst_local):
        for v, idx in adj[u]:
            if visited[v]:
                continue
            visited[v] = True
            if match_catalyst_local[v] == -1 or dfs(match_catalyst_local[v], visited, match_catalyst_local):
                match_catalyst_local[v] = u
                return True
        return False
    
    for u in range(k):
        visited = [False] * k
        dfs(u, visited, match_catalyst)
    
    # Reconstruct match_base from match_catalyst
    for c, b in enumerate(match_catalyst):
        if b != -1:
            match_base[b] = c
    
    # Check if we have a perfect matching
    if any(m == -1 for m in match_base):
        print("NO")
        return
    
    # Now we need to find reactions that match (base, catalyst) pairs and also cover all thermal profiles
    # Build mapping: (base, catalyst) -> list of (thermal, reaction_index)
    bc_to_thermal = {}
    for idx, (x, y, z) in enumerate(reactions):
        key = (x, y)
        if key not in bc_to_thermal:
            bc_to_thermal[key] = []
        bc_to_thermal[key].append((z, idx))
    
    # For each base, get the matched catalyst and available thermals
    base_thermals = []
    valid = True
    for b in range(k):
        c = match_base[b]
        key = (b, c)
        if key not in bc_to_thermal:
            valid = False
            break
        thermals = bc_to_thermal[key]
        base_thermals.append(thermals)
    
    if not valid:
        print("NO")
        return
    
    # Now we need to select one thermal from each base's list such that all thermals are distinct
    # This is another bipartite matching: bases -> thermals
    n = k
    match_thermal = [-1] * n  # thermal -> base
    
    def dfs2(u, visited2):
        for z, idx in base_thermals[u]:
            if visited2[z]:
                continue
            visited2[z] = True
            if match_thermal[z] == -1 or dfs2(match_thermal[z], visited2):
                match_thermal[z] = u
                return True
        return False
    
    for u in range(n):
        visited2 = [False] * n
        if not dfs2(u, visited2):
            print("NO")
            return
    
    # Reconstruct the solution
    result = [None] * n
    for z, b in enumerate(match_thermal):
        # Find the reaction index for base b with thermal z
        c = match_base[b]
        key = (b, c)
        for tz, idx in bc_to_thermal[key]:
            if tz == z:
                result[b] = idx
                break
    
    print("YES")
    for idx in result:
        x, y, z = reactions[idx]
        print(x + 1, y + 1, z + 1)

solve()
