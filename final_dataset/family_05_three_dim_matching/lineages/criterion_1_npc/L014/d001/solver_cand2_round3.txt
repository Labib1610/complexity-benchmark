import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    reactions = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        reactions.append((x, y, z))
        idx += 3

    # We need to select k reactions such that each x, y, z appears exactly once.
    # This is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph.
    # Since k <= 22, we can use backtracking with pruning.
    
    # Group reactions by base fluid x
    by_x = [[] for _ in range(k)]
    for r in reactions:
        by_x[r[0]].append(r)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x_idx):
        if x_idx == k:
            return True
        
        for r in by_x[x_idx]:
            _, y, z = r
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(r)
                
                if backtrack(x_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for r in result:
            print(f"{r[0]+1} {r[1]+1} {r[2]+1}")
    else:
        print("NO")

solve()
