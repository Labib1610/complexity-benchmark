import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    reactions = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        reactions.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem.
    # We have k base fluids, k catalysts, k thermal profiles.
    # Each reaction (x, y, z) covers base fluid x, catalyst y, thermal profile z.
    # We need to pick k reactions such that each fluid, catalyst, and profile is used exactly once.
    # This is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph.
    # Since k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    result = []
    
    def backtrack(idx_r, count):
        if count == k:
            return True
        if idx_r == t:
            return False
        
        # Pruning: if remaining reactions are not enough to fill the rest
        if t - idx_r < k - count:
            return False
            
        # Try including reactions[idx_r]
        x, y, z = reactions[idx_r]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append(reactions[idx_r])
            if backtrack(idx_r + 1, count + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try excluding reactions[idx_r]
        if backtrack(idx_r + 1, count):
            return True
            
        return False

    # Sort reactions to potentially hit a solution faster
    reactions.sort()
    
    if backtrack(0, 0):
        print("YES")
        for r in result:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
