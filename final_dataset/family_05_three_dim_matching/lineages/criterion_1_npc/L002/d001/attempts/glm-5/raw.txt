import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    paths = []
    idx = 2
    for _ in range(t):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        w = int(data[idx+2]) - 1
        paths.append((u, v, w))
        idx += 3

    # Group paths by input port
    by_input = [[] for _ in range(k)]
    for p in paths:
        by_input[p[0]].append(p)

    # If any input port has no paths, impossible
    for i in range(k):
        if not by_input[i]:
            print("NO")
            return

    # DP over input ports: state = (output_mask, freq_mask)
    # Store parent pointer for reconstruction
    # Use dict for sparse states
    # Initial state: input 0, masks = 0
    # We process inputs 0..k-1

    # current: dict mapping (out_mask, freq_mask) -> (path_index, prev_state)
    current = {(0, 0): None}
    parents = [dict() for _ in range(k)]  # parents[i] stores dict for step i

    for i in range(k):
        next_states = {}
        pi = parents[i]
        for state, _ in current.items():
            out_mask, freq_mask = state
            for pidx, (u, v, w) in enumerate(by_input[i]):
                # u should be i
                if (out_mask >> v) & 1:
                    continue
                if (freq_mask >> w) & 1:
                    continue
                new_out = out_mask | (1 << v)
                new_freq = freq_mask | (1 << w)
                new_state = (new_out, new_freq)
                if new_state not in next_states:
                    next_states[new_state] = (pidx, state)
                    pi[new_state] = (pidx, state)
        current = next_states
        if not current:
            print("NO")
            return

    if not current:
        print("NO")
        return

    # Reconstruct
    final_state = next(iter(current.keys()))
    result = []
    state = final_state
    for i in range(k-1, -1, -1):
        pidx, prev_state = parents[i][state]
        u, v, w = by_input[i][pidx]
        result.append((u+1, v+1, w+1))
        state = prev_state

    result.reverse()
    print("YES")
    for u, v, w in result:
        print(u, v, w)

solve()
