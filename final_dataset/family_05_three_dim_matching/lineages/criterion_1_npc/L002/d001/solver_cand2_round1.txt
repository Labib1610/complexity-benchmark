import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    assemblies = []
    idx = 2
    for _ in range(t):
        m = int(input_data[idx]) - 1
        e = int(input_data[idx+1]) - 1
        s = int(input_data[idx+2]) - 1
        assemblies.append((m, e, s))
        idx += 3
        
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    # We need to pick k assemblies such that each m, e, s is used exactly once.
    # Represent each assembly as a bitmask or just use sets.
    
    used_m = [False] * k
    used_e = [False] * k
    used_s = [False] * k
    chosen = []
    
    def backtrack(idx):
        if len(chosen) == k:
            return True
        if idx == t:
            return False
        
        # Option 1: Include assemblies[idx]
        m, e, s = assemblies[idx]
        if not used_m[m] and not used_e[e] and not used_s[s]:
            used_m[m] = used_e[e] = used_s[s] = True
            chosen.append(assemblies[idx])
            if backtrack(idx + 1):
                return True
            chosen.pop()
            used_m[m] = used_e[e] = used_s[s] = False
            
        # Option 2: Skip assemblies[idx]
        # Optimization: if remaining assemblies are not enough to fill k
        if (t - 1 - idx) >= (k - len(chosen)):
            if backtrack(idx + 1):
                return True
        
        return False

    # The search space is small enough for k=22 if we prune well.
    # However, standard backtracking might be slow. Let's use Dancing Links logic
    # or simply optimize the search.
    
    if backtrack(0):
        print("YES")
        for m, e, s in chosen:
            print(f"{m+1} {e+1} {s+1}")
    else:
        print("NO")

solve()
