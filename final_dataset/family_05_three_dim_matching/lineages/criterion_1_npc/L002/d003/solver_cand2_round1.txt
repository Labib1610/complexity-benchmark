```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    pathways = []
    idx = 2
    for _ in range(t):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        pathways.append((u, v, w))
        idx += 3
        
    # We need to select k pathways such that:
    # {u_i} = {1, ..., k}, {v_i} = {1, ..., k}, {w_i} = {1, ..., k}
    # This is equivalent to finding a transversal in a 3D Latin Square structure.
    # Since k <= 22, we can use backtracking with pruning.
    
    used_u = [False] * (k + 1)
    used_v = [False] * (k + 1)
    used_w = [False] * (k + 1)
    
    # Group pathways by u to simplify selection
    by_u = [[] for _ in range(k + 1)]
    for p in pathways:
        by_u[p[0]].append(p)
        
    result = []
    
    def backtrack(u_idx):
        if u_idx > k:
            return True
        
        for p in by_u[u_idx]:
            u, v, w = p
            if not used_v[v] and not used_w[w]:
                used_v[v] = True
                used_w[w] = True
                result.append(p)
                
                if backtrack(u_idx + 1):
                    return True
                
                result.pop()
                used_v[v] = False
                used_w[w] = False
        return False

    if backtrack(1):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
```
