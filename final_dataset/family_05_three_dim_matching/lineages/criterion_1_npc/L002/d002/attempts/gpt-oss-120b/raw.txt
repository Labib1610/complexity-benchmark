import sys

sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    edges_by_u = [[] for _ in range(k)]
    for _ in range(t):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it)) - 1
        edges_by_u[u].append((v, w, u, v, w))  # store original 1‑based later
    # order inputs by fewest options to improve pruning
    order = sorted(range(k), key=lambda x: len(edges_by_u[x]))
    used_out = 0
    used_freq = 0
    result = [None] * k

    def dfs(pos: int) -> bool:
        nonlocal used_out, used_freq
        if pos == k:
            return True
        u = order[pos]
        for v, w, ou, ov, ow in edges_by_u[u]:
            if (used_out >> v) & 1:
                continue
            if (used_freq >> w) & 1:
                continue
            # choose
            used_out ^= 1 << v
            used_freq ^= 1 << w
            result[pos] = (ou + 1, ov + 1, ow + 1)
            if dfs(pos + 1):
                return True
            # backtrack
            used_out ^= 1 << v
            used_freq ^= 1 << w
        return False

    if dfs(0):
        out_lines = ["YES"]
        for triple in result:
            out_lines.append(f"{triple[0]} {triple[1]} {triple[2]}")
        sys.stdout.write("\n".join(out_lines))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
