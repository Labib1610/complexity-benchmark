```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # We need to select k triplets (u, v, w) such that:
    # {u_1, ..., u_k} = {1, ..., k}
    # {v_1, ..., v_k} = {1, ..., k}
    # {w_1, ..., w_k} = {1, ..., k}
    # This is equivalent to finding a perfect matching in a 3D matching problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    routes = []
    idx = 2
    for _ in range(t):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2]) - 1
        routes.append((u, v, w))
        idx += 3
        
    # Group routes by u to iterate through inputs 0 to k-1
    by_u = [[] for _ in range(k)]
    for r in routes:
        by_u[r[0]].append(r)
        
    used_v = [False] * k
    used_w = [False] * k
    result = []
    
    def backtrack(u_idx):
        if u_idx == k:
            return True
        
        for r in by_u[u_idx]:
            _, v, w = r
            if not used_v[v] and not used_w[w]:
                used_v[v] = True
                used_w[w] = True
                result.append(r)
                
                if backtrack(u_idx + 1):
                    return True
                
                result.pop()
                used_v[v] = False
                used_w[w] = False
        return False

    if backtrack(0):
        print("YES")
        for r in result:
            print(f"{r[0]+1} {r[1]+1} {r[2]+1}")
    else:
        print("NO")

solve()
```
