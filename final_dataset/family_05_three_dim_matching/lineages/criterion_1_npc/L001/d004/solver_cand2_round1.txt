import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    vectors = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        vectors.append((x, y, z))
        idx += 3
        
    if k == 1:
        if t >= 1:
            print("YES")
            print(f"{vectors[0][0]} {vectors[0][1]} {vectors[0][2]}")
        else:
            print("NO")
        return

    # A set is harmonious if for any two vectors (x1, y1, z1) and (x2, y2, z2),
    # x1 != x2, y1 != y2, and z1 != z2.
    # This is equivalent to finding a clique of size k in a graph where
    # an edge exists if Hamming distance is 3.
    
    adj = [[] for _ in range(t)]
    for i in range(t):
        for j in range(i + 1, t):
            v1 = vectors[i]
            v2 = vectors[j]
            if v1[0] != v2[0] and v1[1] != v2[1] and v1[2] != v2[2]:
                adj[i].append(j)
                adj[j].append(i)
                
    # Since k is small (up to 22), we can use backtracking with pruning
    def find_clique(current_clique, candidates):
        if len(current_clique) == k:
            return current_clique
        
        if not candidates:
            return None
        
        # Heuristic: pick candidate with most connections to other candidates
        for i in range(len(candidates)):
            v = candidates[i]
            new_candidates = [c for c in candidates[i+1:] if c in adj[v]]
            
            # Pruning: if remaining candidates + current_clique < k, stop
            if len(current_clique) + 1 + len(new_candidates) < k:
                continue
                
            res = find_clique(current_clique + [v], new_candidates)
            if res:
                return res
        return None

    result = find_clique([], list(range(t)))
    
    if result:
        print("YES")
        for idx in result:
            print(f"{vectors[idx][0]} {vectors[idx][1]} {vectors[idx][2]}")
    else:
        print("NO")

solve()
