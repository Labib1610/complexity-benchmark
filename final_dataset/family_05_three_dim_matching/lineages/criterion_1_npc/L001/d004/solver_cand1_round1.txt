import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    points = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        points.append((x, y, z))
        idx += 3
        
    # The condition min(|Px-Qx|, |Py-Qy|, |Pz-Qz|) >= 1 means
    # |Px-Qx| >= 1 AND |Py-Qy| >= 1 AND |Pz-Qz| >= 1.
    # This is equivalent to saying that no two points share the same X, Y, or Z coordinate.
    # Since we need to pick k points from a k x k x k grid such that all X, Y, Z are distinct,
    # this is equivalent to finding a matching in a 3D bipartite-like structure,
    # or more simply, a set of k points (x_i, y_i, z_i) such that {x_i} = {1..k}, {y_i} = {1..k}, {z_i} = {1..k}.
    
    # This is a variation of the exact cover problem or finding a transversal.
    # Given k <= 22, we can use backtracking with pruning.
    
    points.sort()
    
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    # Group points by x
    by_x = [[] for _ in range(k + 1)]
    for p in points:
        by_x[p[0]].append(p)
        
    result = []
    
    def backtrack(x):
        if x > k:
            return True
        
        for p in by_x[x]:
            y, z = p[1], p[2]
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(p)
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
