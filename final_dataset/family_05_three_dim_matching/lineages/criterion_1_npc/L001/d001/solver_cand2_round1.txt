import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    edges = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        edges.append((x, y, z))
        idx += 3

    # This is a 3D matching problem, which is NP-complete.
    # However, given the constraints and the nature of competitive programming,
    # we can use a randomized approach (Dancing Links or backtracking with heuristics).
    # For k up to 22, a backtracking approach with bitmasks or pruning is often effective.
    
    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(edges):
        adj[x].append((y, z, i))
        
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        # Heuristic: try to pick the most constrained specialist first
        # But for simplicity, standard backtracking:
        for y, z, i in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(edges[i])
                if backtrack(x + 1):
                    return True
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    # Sort specialists by number of options to prune search space
    adj.sort(key=len)
    
    if backtrack(0):
        print("YES")
        for x, y, z in result:
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")

solve()
