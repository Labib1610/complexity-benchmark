```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem (3D matching).
    # Since k <= 22, we use backtracking with pruning.
    
    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(triples):
        adj[x].append(i)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(person_idx):
        if person_idx == k:
            return True
        
        for triple_idx in adj[person_idx]:
            _, y, z = triples[triple_idx]
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(triples[triple_idx])
                
                if backtrack(person_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for x, y, z in result:
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
