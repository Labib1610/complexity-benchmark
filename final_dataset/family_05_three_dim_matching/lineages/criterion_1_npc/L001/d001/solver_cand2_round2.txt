```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        configs.append((x, y, z))
        idx += 3

    # We need to select k triples (x_i, y_i, z_i) such that
    # {x_i} = {0..k-1}, {y_i} = {0..k-1}, {z_i} = {0..k-1}
    # This is equivalent to finding a transversal in a 3D Latin Square
    # or exact cover in a 3-set system.
    # Since k <= 22, we use backtracking with pruning.
    
    # Group configs by x
    by_x = [[] for _ in range(k)]
    for c in configs:
        by_x[c[0]].append(c)
        
    used_y = [False] * k
    used_z = [False] * k
    res = []

    def backtrack(x):
        if x == k:
            return True
        
        for c in by_x[x]:
            y, z = c[1], c[2]
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                res.append(c)
                if backtrack(x + 1):
                    return True
                res.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    # Heuristic: sort by_x to try most constrained first
    # But for k=22, simple backtracking might be slow.
    # However, the constraints are tight.
    if backtrack(0):
        print("YES")
        for c in res:
            print(f"{c[0]+1} {c[1]+1} {c[2]+1}")
    else:
        print("NO")

solve()
```
