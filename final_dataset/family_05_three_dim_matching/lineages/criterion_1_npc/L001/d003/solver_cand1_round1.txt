import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    protocols = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        protocols.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    # We need to pick k protocols such that each x, y, z is used exactly once.
    # Represent as a matrix where rows are protocols and columns are:
    # (x, i), (y, i), (z, i) for i in 1..k. Total 3k columns.
    
    adj = [[] for _ in range(k + 1)]
    for i, (x, y, z) in enumerate(protocols):
        adj[x].append(i)
        
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    result = []
    
    def backtrack(x):
        if x > k:
            return True
        
        for idx in adj[x]:
            _, y, z = protocols[idx]
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(protocols[idx])
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
