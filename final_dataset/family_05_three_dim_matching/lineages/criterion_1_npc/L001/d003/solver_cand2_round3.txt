import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        triples.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    # To optimize, we use bitmasks for the constraints.
    # Each triple (x, y, z) covers:
    # Technician x (1..k), Bay y (1..k), Core z (1..k)
    
    # Pre-filter triples: only keep those that could potentially be part of a solution
    # (Though with k=22, simple backtracking with pruning is usually sufficient)
    
    used_x = 0
    used_y = 0
    used_z = 0
    
    solution = []
    
    def backtrack(idx):
        if len(solution) == k:
            return True
        
        if idx == len(triples):
            return False
        
        # Try including triples[idx]
        x, y, z = triples[idx]
        bit_x = 1 << (x - 1)
        bit_y = 1 << (y - 1)
        bit_z = 1 << (z - 1)
        
        if not (used_x & bit_x or used_y & bit_y or used_z & bit_z):
            # Include
            nonlocal used_x, used_y, used_z
            used_x |= bit_x
            used_y |= bit_y
            used_z |= bit_z
            solution.append((x, y, z))
            
            if backtrack(idx + 1):
                return True
            
            # Backtrack
            solution.pop()
            used_x &= ~bit_x
            used_y &= ~bit_y
            used_z &= ~bit_z
            
        # Skip
        if backtrack(idx + 1):
            return True
            
        return False

    # Heuristic: sort triples to try to fill constraints faster
    if backtrack(0):
        print("YES")
        for s in solution:
            print(f"{s[0]} {s[1]} {s[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
