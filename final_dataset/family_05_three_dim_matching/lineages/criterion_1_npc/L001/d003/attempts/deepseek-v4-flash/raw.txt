We need to find a perfect matching in a 3-partite hypergraph, which is NP-hard in general. However, with k ≤ 22, we can use a backtracking approach with bitmasking to keep track of used technicians, bays, and cores.

We’ll represent each test as a tuple (x, y, z). We’ll sort tests to help pruning. We’ll use recursion with memoization on the used masks for technicians and bays (cores can be derived from the count of selected tests, but we still need to check availability).

Here’s the implementation:

```python
import sys

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    tests = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        tests.append((x-1, y-1, z-1))
    
    # Group tests by technician
    by_tech = [[] for _ in range(k)]
    for idx, (x, y, z) in enumerate(tests):
        by_tech[x].append((y, z, idx))
    
    # Sort each list for deterministic order
    for lst in by_tech:
        lst.sort()
    
    used_bay = 0
    used_core = 0
    chosen = []
    
    # Memoization: (tech_index, used_bay_mask, used_core_mask) -> bool
    memo = {}
    
    def dfs(tech, used_bay, used_core):
        if tech == k:
            return True
        key = (tech, used_bay, used_core)
        if key in memo:
            return memo[key]
        for y, z, idx in by_tech[tech]:
            if (used_bay >> y) & 1:
                continue
            if (used_core >> z) & 1:
                continue
            chosen.append((tech+1, y+1, z+1))
            if dfs(tech+1, used_bay | (1 << y), used_core | (1 << z)):
                memo[key] = True
                return True
            chosen.pop()
        memo[key] = False
        return False
    
    if dfs(0, 0, 0):
        print("YES")
        for x, y, z in chosen:
            print(x, y, z)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
