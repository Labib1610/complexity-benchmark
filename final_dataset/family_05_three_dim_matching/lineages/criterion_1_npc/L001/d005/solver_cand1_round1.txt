import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    models = []
    idx = 2
    for _ in range(t):
        f = int(input_data[idx])
        p = int(input_data[idx+1])
        a = int(input_data[idx+2])
        models.append((f, p, a))
        idx += 3
        
    # The condition |Uf| + |Up| + |Ua| = 3k with k items means:
    # |Uf| = k, |Up| = k, |Ua| = k.
    # This implies that for the k chosen resonators (f_i, p_i, a_i),
    # the set of f_i must be {1, ..., k}, the set of p_i must be {1, ..., k},
    # and the set of a_i must be {1, ..., k}.
    # This is equivalent to finding a Latin Square of order k, 
    # where each row is a resonator (f, p, a).
    # Specifically, we need to pick k triples such that all f are distinct,
    # all p are distinct, and all a are distinct.
    
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    used_f = [False] * (k + 1)
    used_p = [False] * (k + 1)
    used_a = [False] * (k + 1)
    
    chosen = []
    
    def backtrack(idx):
        if len(chosen) == k:
            return True
        
        for i in range(idx, t):
            f, p, a = models[i]
            if not used_f[f] and not used_p[p] and not used_a[a]:
                used_f[f] = True
                used_p[p] = True
                used_a[a] = True
                chosen.append(models[i])
                
                if backtrack(i + 1):
                    return True
                
                chosen.pop()
                used_f[f] = False
                used_p[p] = False
                used_a[a] = False
        return False

    if backtrack(0):
        print("YES")
        for m in chosen:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

solve()
