```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    triples = []
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        c = int(next(it)) - 1
        triples.append((a, b, c))

    # columns: 0..k-1 freq, k..2k-1 phase, 2k..3k-1 amp
    ncol = 3 * k
    col_rows = [[] for _ in range(ncol)]
    for idx, (a, b, c) in enumerate(triples):
        col_rows[a].append(idx)
        col_rows[k + b].append(idx)
        col_rows[2 * k + c].append(idx)

    # quick impossibility check: each column must have at least one row
    for lst in col_rows:
        if not lst:
            print("NO")
            return

    used_row = [False] * t
    covered = [False] * ncol
    solution = []

    # heuristic: precompute column order by size
    col_order = sorted(range(ncol), key=lambda x: len(col_rows[x]))

    sys.setrecursionlimit(10000)

    def dfs(depth: int) -> bool:
        if depth == k:
            return True
        # choose uncovered column with fewest candidates
        best_col = -1
        best_len = 10**9
        for c in col_order:
            if not covered[c]:
                cnt = 0
                for r in col_rows[c]:
                    if not used_row[r]:
                        # check other two columns not yet covered
                        a, b, c3 = triples[r]
                        cols = (a, k + b, 2 * k + c3)
                        if not covered[cols[0]] and not covered[cols[1]] and not covered[cols[2]]:
                            cnt += 1
                if cnt == 0:
                    return False
                if cnt < best_len:
                    best_len = cnt
                    best_col = c
        # iterate rows for best_col
        for r in col_rows[best_col]:
            if used_row[r]:
                continue
            a, b, c3 = triples[r]
            cols = (a, k + b, 2 * k + c3)
            if any(covered[x] for x in cols):
                continue
            # select
            used_row[r] = True
            for x in cols:
                covered[x] = True
            solution.append(r)
            if dfs(depth + 1):
                return True
            # backtrack
            solution.pop()
            for x in cols:
                covered[x] = False
            used_row[r] = False
        return False

    if dfs(0):
        print("YES")
        out = sys.stdout
        for idx in solution:
            a, b, c = triples[idx]
            out.write(f"{a+1} {b+1} {c+1}\n")
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
