import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    models = []
    idx = 2
    for _ in range(t):
        models.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    # The total load is sum(A_x^2) + sum(B_x^2) + sum(G_x^2).
    # Since sum(A_x) = k, sum(A_x^2) >= k, with equality iff all A_x are 1.
    # Thus, total load >= k + k + k = 3k.
    # Equality 3k holds if and only if all A_x, B_x, G_x are 1.
    # This means the chosen k models must have distinct alpha, distinct beta, and distinct gamma values.
    # This is equivalent to finding a transversal of size k in a 3D grid,
    # or more simply, a set of k models (a_i, b_i, g_i) such that
    # {a_1, ..., a_k} = {1, ..., k}, {b_1, ..., b_k} = {1, ..., k}, {g_1, ..., g_k} = {1, ..., k}.
    
    # Given k <= 22, but the search space for combinations is C(t, k).
    # If t is large, we need a more efficient approach than brute force.
    # However, the problem constraints and the nature of the condition suggest
    # we are looking for a permutation-like structure.
    
    import itertools
    
    # Backtracking search to find the set
    def find_set(idx, current_models, used_a, used_b, used_g):
        if len(current_models) == k:
            return current_models
        
        for i in range(idx, t):
            a, b, g = models[i]
            if not (used_a & (1 << a)) and not (used_b & (1 << b)) and not (used_g & (1 << g)):
                res = find_set(i + 1, current_models + [models[i]], 
                               used_a | (1 << a), used_b | (1 << b), used_g | (1 << g))
                if res:
                    return res
        return None

    # Optimization: If t is large, the problem might be solvable via backtracking
    # because the constraints are quite restrictive.
    result = find_set(0, [], 0, 0, 0)
    
    if result:
        print("YES")
        for m in result:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

solve()
