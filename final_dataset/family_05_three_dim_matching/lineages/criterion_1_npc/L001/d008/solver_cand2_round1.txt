import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    towers = []
    idx = 2
    for _ in range(t):
        towers.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    # The condition "number of towers with band >= x is k - x + 1"
    # implies that for each band, the set of values must be exactly {1, 2, ..., k}.
    # Specifically, if we sort the alpha values of the k chosen towers, they must be 1, 2, ..., k.
    # Same for beta and gamma.
    
    # This is equivalent to saying that for each band, the chosen towers must have
    # distinct values from 1 to k.
    
    # We need to select k towers (a_i, b_i, g_i) such that:
    # {a_1, ..., a_k} = {1, ..., k}
    # {b_1, ..., b_k} = {1, ..., k}
    # {g_1, ..., g_k} = {1, ..., k}
    
    # This is a variation of the exact cover problem or 3D matching.
    # Since k <= 22, we can use backtracking with pruning.
    
    towers.sort(key=lambda x: (x[0], x[1], x[2]))
    
    used_a = [False] * (k + 1)
    used_b = [False] * (k + 1)
    used_g = [False] * (k + 1)
    
    res = []
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for i in range(start_idx, t):
            a, b, g = towers[i]
            if not used_a[a] and not used_b[b] and not used_g[g]:
                used_a[a] = True
                used_b[b] = True
                used_g[g] = True
                res.append(towers[i])
                
                if backtrack(i + 1, count + 1):
                    return True
                
                res.pop()
                used_a[a] = False
                used_b[b] = False
                used_g[g] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for t_val in res:
            print(f"{t_val[0]} {t_val[1]} {t_val[2]}")
    else:
        print("NO")

solve()
