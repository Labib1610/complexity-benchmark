import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        configs.append((x, y, z))
        idx += 3
        
    # Group configs by technician
    tech_groups = [[] for _ in range(k)]
    for x, y, z in configs:
        tech_groups[x].append((y, z))
        
    # dp[tech_idx][mask_console][mask_sector]
    # Since k <= 22, we use bitmasks. 
    # However, 2^22 * 2^22 is too large.
    # This is a 3D matching problem. With k=22, we use backtracking with pruning.
    
    used_console = 0
    used_sector = 0
    result = []
    
    def backtrack(tech_idx):
        if tech_idx == k:
            return True
        
        for console, sector in tech_groups[tech_idx]:
            if not (used_console & (1 << console)) and not (used_sector & (1 << sector)):
                used_console |= (1 << console)
                used_sector |= (1 << sector)
                result.append((tech_idx + 1, console + 1, sector + 1))
                
                if backtrack(tech_idx + 1):
                    return True
                
                result.pop()
                used_console &= ~(1 << console)
                used_sector &= ~(1 << sector)
        return False

    # Sort tech_groups by number of options to prune search space
    # But we need to keep track of original tech index
    order = sorted(range(k), key=lambda i: len(tech_groups[i]))
    
    # Re-map tech_groups to the sorted order
    sorted_tech_groups = [tech_groups[i] for i in order]
    
    def backtrack_sorted(idx_in_order):
        if idx_in_order == k:
            return True
        
        tech_idx = order[idx_in_order]
        for console, sector in sorted_tech_groups[idx_in_order]:
            if not (used_console & (1 << console)) and not (used_sector & (1 << sector)):
                used_console |= (1 << console)
                used_sector |= (1 << sector)
                result.append((tech_idx + 1, console + 1, sector + 1))
                
                if backtrack_sorted(idx_in_order + 1):
                    return True
                
                result.pop()
                used_console &= ~(1 << console)
                used_sector &= ~(1 << sector)
        return False

    if backtrack_sorted(0):
        print("YES")
        for r in result:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
