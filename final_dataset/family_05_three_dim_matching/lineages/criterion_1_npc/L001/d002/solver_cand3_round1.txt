import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    recipes = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        recipes.append((x, y, z))
        idx += 3
        
    # Group recipes by flavor extract (x)
    by_x = [[] for _ in range(k)]
    for r in recipes:
        by_x[r[0]].append(r)
        
    # Backtracking with bitmask for y and z
    # used_y and used_z are bitmasks
    selected = []
    
    def backtrack(x_idx, used_y, used_z):
        if x_idx == k:
            return True
        
        for r in by_x[x_idx]:
            y, z = r[1], r[2]
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                selected.append(r)
                if backtrack(x_idx + 1, used_y | (1 << y), used_z | (1 << z)):
                    return True
                selected.pop()
        return False

    if backtrack(0, 0, 0):
        print("YES")
        for r in selected:
            print(f"{r[0]+1} {r[1]+1} {r[2]+1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
