import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    triplets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triplets.append((x, y, z))
        idx += 3

    # We need to select k triplets such that each red, green, blue node is used exactly once.
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking with pruning.
    # Represent state as bitmasks for green and blue nodes used.
    # Red nodes are covered by the index of the triplet we are considering.
    
    # Group triplets by red node
    by_red = [[] for _ in range(k)]
    for x, y, z in triplets:
        by_red[x].append((y, z))
    
    # Optimization: if any red node has no triplets, impossible
    for r in range(k):
        if not by_red[r]:
            print("NO")
            return

    used_green = 0
    used_blue = 0
    result = []

    def backtrack(r):
        if r == k:
            return True
        
        for y, z in by_red[r]:
            if not (used_green & (1 << y)) and not (used_blue & (1 << z)):
                nonlocal used_green, used_blue
                used_green |= (1 << y)
                used_blue |= (1 << z)
                result.append((r + 1, y + 1, z + 1))
                
                if backtrack(r + 1):
                    return True
                
                result.pop()
                used_green &= ~(1 << y)
                used_blue &= ~(1 << z)
        return False

    if backtrack(0):
        print("YES")
        for triplet in result:
            print(f"{triplet[0]} {triplet[1]} {triplet[2]}")
    else:
        print("NO")

solve()
