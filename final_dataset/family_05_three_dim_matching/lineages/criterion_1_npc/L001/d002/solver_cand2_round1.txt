import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph.
    # Since k <= 22, we can use backtracking with pruning or bitmask DP.
    # Given the constraints and the nature of the problem (exact cover), 
    # backtracking with bitmasks is efficient.
    
    edges = []
    idx = 2
    for _ in range(t):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2]) - 1
        edges.append((u, v, w))
        idx += 3
        
    # Group edges by specialist to simplify recursion
    adj = [[] for _ in range(k)]
    for u, v, w in edges:
        adj[u].append((v, w))
        
    used_v = 0
    used_w = 0
    result = []
    
    def backtrack(u):
        nonlocal used_v, used_w
        if u == k:
            return True
        
        for v, w in adj[u]:
            if not (used_v & (1 << v)) and not (used_w & (1 << w)):
                used_v |= (1 << v)
                used_w |= (1 << w)
                result.append((u + 1, v + 1, w + 1))
                
                if backtrack(u + 1):
                    return True
                
                result.pop()
                used_v &= ~(1 << v)
                used_w &= ~(1 << w)
        return False

    if backtrack(0):
        print("YES")
        for row in result:
            print(f"{row[0]} {row[1]} {row[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
