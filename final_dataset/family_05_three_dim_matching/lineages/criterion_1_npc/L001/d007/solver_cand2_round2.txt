import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    models = []
    idx = 2
    for _ in range(t):
        models.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    # Newton's sums: sum_{i=1}^k x_i^j = sum_{i=1}^k i^j for j=1..k
    # This is equivalent to the set {x_1, ..., x_k} being equal to {1, ..., k}
    # as multisets.
    
    target_set = sorted(range(1, k + 1))
    
    # We need to pick k models (a_i, b_i, g_i) such that:
    # {a_1, ..., a_k} = {1, ..., k}
    # {b_1, ..., b_k} = {1, ..., k}
    # {g_1, ..., g_k} = {1, ..., k}
    
    # Since k is small (up to 22), but t can be large, we use backtracking
    # with pruning.
    
    def backtrack(idx, current_selection, used_a, used_b, used_g):
        if len(current_selection) == k:
            if all(used_a) and all(used_b) and all(used_g):
                return current_selection
            return None
        
        if idx == t:
            return None
        
        # Pruning: if remaining models are not enough to fill the sets
        remaining = t - idx
        needed = k - len(current_selection)
        if remaining < needed:
            return None
            
        # Option 1: Include models[idx]
        a, b, g = models[idx]
        if not used_a[a-1] and not used_b[b-1] and not used_g[g-1]:
            used_a[a-1] = used_b[b-1] = used_g[g-1] = True
            current_selection.append(models[idx])
            res = backtrack(idx + 1, current_selection, used_a, used_b, used_g)
            if res is not None:
                return res
            current_selection.pop()
            used_a[a-1] = used_b[b-1] = used_g[g-1] = False
            
        # Option 2: Skip models[idx]
        return backtrack(idx + 1, current_selection, used_a, used_b, used_g)

    result = backtrack(0, [], [False]*k, [False]*k, [False]*k)
    
    if result:
        print("YES")
        for m in result:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

solve()
