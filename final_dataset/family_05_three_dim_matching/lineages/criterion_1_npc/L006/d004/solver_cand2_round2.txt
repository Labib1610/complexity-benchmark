import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    W = int(input_data[0])
    N = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(N):
        p = int(input_data[idx]) - 1
        s = int(input_data[idx+1]) - 1
        e = int(input_data[idx+2]) - 1
        configs.append((p, s, e))
        idx += 3
        
    # We need to select W triples (p_i, s_i, e_i) such that all p_i are distinct,
    # all s_i are distinct, and all e_i are distinct.
    # This is equivalent to finding a matching of size W in a 3D bipartite-like structure.
    # Since W is small (up to 22), we can use backtracking with pruning.
    
    # Group by p
    by_p = [[] for _ in range(W)]
    for p, s, e in configs:
        by_p[p].append((s, e))
        
    used_s = [False] * W
    used_e = [False] * W
    result = []
    
    def backtrack(p_idx):
        if p_idx == W:
            return True
        
        # Optimization: if no configurations for this p, impossible
        if not by_p[p_idx]:
            return False
            
        for s, e in by_p[p_idx]:
            if not used_s[s] and not used_e[e]:
                used_s[s] = True
                used_e[e] = True
                result.append((p_idx + 1, s + 1, e + 1))
                
                if backtrack(p_idx + 1):
                    return True
                
                result.pop()
                used_s[s] = False
                used_e[e] = False
        return False

    # Sort by_p to try to fail faster or succeed faster
    # Heuristic: try p with fewer options first
    order = sorted(range(W), key=lambda i: len(by_p[i]))
    
    # Re-map to handle the order
    new_by_p = [by_p[i] for i in order]
    
    def backtrack_ordered(idx_in_order):
        if idx_in_order == W:
            return True
        
        p_orig = order[idx_in_order]
        for s, e in new_by_p[idx_in_order]:
            if not used_s[s] and not used_e[e]:
                used_s[s] = True
                used_e[e] = True
                result.append((p_orig + 1, s + 1, e + 1))
                
                if backtrack_ordered(idx_in_order + 1):
                    return True
                
                result.pop()
                used_s[s] = False
                used_e[e] = False
        return False

    if backtrack_ordered(0):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
