```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    struts = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        struts.append((x, y, z))
        idx += 3

    # This is an Exact Cover problem (3D matching).
    # Since k <= 22, we use backtracking with pruning.
    # Represent struts as a list of lists for faster access.
    # adj[x][y] = list of z values
    adj = [[[] for _ in range(k)] for _ in range(k)]
    for x, y, z in struts:
        adj[x][y].append(z)
    
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        # Try all y for current x
        for y in range(k):
            if not used_y[y]:
                for z in adj[x][y]:
                    if not used_z[z]:
                        used_y[y] = True
                        used_z[z] = True
                        result.append((x + 1, y + 1, z + 1))
                        
                        if backtrack(x + 1):
                            return True
                        
                        result.pop()
                        used_z[z] = False
                        used_y[y] = False
        return False

    # Optimization: check if any y or z is impossible to cover
    # Count occurrences of each y and z
    count_y = [0] * k
    count_z = [0] * k
    for x, y, z in struts:
        count_y[y] += 1
        count_z[z] += 1
    
    for i in range(k):
        if count_y[i] == 0 or count_z[i] == 0:
            print("NO")
            return

    if backtrack(0):
        print("YES")
        for s in result:
            print(f"{s[0]} {s[1]} {s[2]}")
    else:
        print("NO")

solve()
```
