import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    struts = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        struts.append((x, y, z))
        idx += 3
        
    # We need to select k struts such that all x, y, z are distinct.
    # This is equivalent to finding a matching of size k in a 3D bipartite-like structure.
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    # Group struts by x
    by_x = [[] for _ in range(k + 1)]
    for s in struts:
        by_x[s[0]].append(s)
        
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    result = []
    
    def backtrack(x):
        if x > k:
            return True
        
        for s in by_x[x]:
            _, y, z = s
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(s)
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for s in result:
            print(f"{s[0]} {s[1]} {s[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
