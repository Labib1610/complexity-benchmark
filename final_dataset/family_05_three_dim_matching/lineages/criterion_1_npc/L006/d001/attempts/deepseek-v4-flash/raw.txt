We need to select exactly k struts such that all x, y, and z coordinates are distinct.  
This is equivalent to finding a perfect matching in a 3-partite 3D matching problem, which is NP-hard in general, but here k ≤ 22, so we can use backtracking with pruning.

We’ll represent available struts as a set of triples.  
We’ll try to build a selection of k struts, ensuring no coordinate repeats.

Here’s the Python solution:

```python
import sys

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    struts = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        struts.append((x-1, y-1, z-1))  # 0-index for easier bitmasking

    # Sort struts to help pruning
    struts.sort()

    # We'll use bitmasks for used x, y, z
    used_x = 0
    used_y = 0
    used_z = 0
    chosen = []

    # Backtracking
    def dfs(idx, count):
        nonlocal used_x, used_y, used_z
        if count == k:
            return True
        if idx >= t:
            return False
        # Prune: if remaining struts not enough
        if t - idx < k - count:
            return False

        x, y, z = struts[idx]
        # Try taking this strut
        if not (used_x & (1 << x)) and not (used_y & (1 << y)) and not (used_z & (1 << z)):
            used_x |= (1 << x)
            used_y |= (1 << y)
            used_z |= (1 << z)
            chosen.append((x+1, y+1, z+1))
            if dfs(idx + 1, count + 1):
                return True
            chosen.pop()
            used_x ^= (1 << x)
            used_y ^= (1 << y)
            used_z ^= (1 << z)

        # Try skipping this strut
        if dfs(idx + 1, count):
            return True
        return False

    if dfs(0, 0):
        print("YES")
        for x, y, z in chosen:
            print(x, y, z)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
