import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    protocols = []
    idx = 2
    for _ in range(t):
        m = int(input_data[idx]) - 1
        c = int(input_data[idx+1]) - 1
        f = int(input_data[idx+2]) - 1
        protocols.append((m, c, f))
        idx += 3

    # We need to select k protocols such that each m, c, f appears exactly once.
    # This is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph.
    # Since k <= 22, we can use backtracking with pruning.
    
    # Group protocols by m
    by_m = [[] for _ in range(k)]
    for p in protocols:
        by_m[p[0]].append(p)
        
    used_c = [False] * k
    used_f = [False] * k
    result = []

    def backtrack(m_idx):
        if m_idx == k:
            return True
        
        for p in by_m[m_idx]:
            c, f = p[1], p[2]
            if not used_c[c] and not used_f[f]:
                used_c[c] = True
                used_f[f] = True
                result.append(p)
                
                if backtrack(m_idx + 1):
                    return True
                
                result.pop()
                used_c[c] = False
                used_f[f] = False
        return False

    if backtrack(0):
        print("YES")
        for p in result:
            print(f"{p[0]+1} {p[1]+1} {p[2]+1}")
    else:
        print("NO")

solve()
