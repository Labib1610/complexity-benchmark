import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    blocks = []
    idx = 2
    for _ in range(t):
        x = int(data[idx]); y = int(data[idx+1]); z = int(data[idx+2])
        idx += 3
        blocks.append((x, y, z))
    
    # Build adjacency: for each shape x, list of (y, z)
    adj = [[] for _ in range(k+1)]
    for (x, y, z) in blocks:
        adj[x].append((y, z))
    
    # We need perfect matching in 3-partite 3-uniform hypergraph.
    # Use backtracking with bitmasks for colors and materials.
    # k <= 22, so bitmask fits in int.
    
    # Sort adjacency lists by degree (heuristic) - but shapes are fixed 1..k, we assign in order.
    # We'll do recursive assignment of shapes 1..k.
    
    color_used = 0
    mat_used = 0
    assignment = [None] * (k+1)  # assignment[x] = (y, z)
    
    # Precompute for each shape the list of (y,z)
    # Use recursion with early pruning.
    
    # To speed up, represent available as bitmask.
    # For each shape, iterate over its blocks.
    
    # Order shapes by number of options (ascending) for better pruning.
    shape_order = list(range(1, k+1))
    shape_order.sort(key=lambda x: len(adj[x]))
    
    # Map shape_order to positions
    # We'll recurse over index in shape_order.
    
    found = [False]
    result = [None]
    
    def backtrack(i, color_used, mat_used):
        if found[0]:
            return
        if i == k:
            found[0] = True
            result[0] = assignment[:]
            return
        x = shape_order[i]
        for (y, z) in adj[x]:
            cmask = 1 << y
            zmask = 1 << z
            if not (color_used & cmask) and not (mat_used & zmask):
                assignment[x] = (y, z)
                backtrack(i+1, color_used | cmask, mat_used | zmask)
                if found[0]:
                    return
                assignment[x] = None
    
    backtrack(0, 0, 0)
    
    if found[0]:
        out = ["YES"]
        for x in range(1, k+1):
            y, z = result[0][x]
            out.append(f"{x} {y} {z}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

solve()
