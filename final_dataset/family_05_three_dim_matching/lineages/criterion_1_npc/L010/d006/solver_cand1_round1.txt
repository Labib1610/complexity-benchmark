import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    relays = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        relays.append((a, b, c))
        idx += 3
        
    # The condition |S1| * |S2| * |S3| = k^3 where |Si| <= k
    # implies |S1| = |S2| = |S3| = k.
    # This means we need to pick k relays such that the set of first elements
    # has size k, the set of second elements has size k, and the set of third
    # elements has size k. Since each parameter is in [1, k], this means
    # each set must be exactly {1, 2, ..., k}.
    
    # Given k <= 22, t can be up to k^3 = 10648.
    # We need to pick k relays from t.
    # This is a variation of the Exact Cover problem or hitting set.
    # Since k is small, we can use backtracking with pruning.
    
    def backtrack(start_idx, current_relays, s1, s2, s3):
        if len(current_relays) == k:
            if len(s1) == k and len(s2) == k and len(s3) == k:
                return current_relays
            return None
        
        if start_idx >= t:
            return None
        
        # Pruning: if remaining relays are not enough to reach k
        if len(current_relays) + (t - start_idx) < k:
            return None
            
        for i in range(start_idx, t):
            r = relays[i]
            new_s1 = s1 | {r[0]}
            new_s2 = s2 | {r[1]}
            new_s3 = s3 | {r[2]}
            
            # Optimization: if any set size exceeds k, skip
            if len(new_s1) > k or len(new_s2) > k or len(new_s3) > k:
                continue
                
            res = backtrack(i + 1, current_relays + [r], new_s1, new_s2, new_s3)
            if res:
                return res
        return None

    # The search space is large, but the constraint is very tight.
    # Let's try a randomized approach or simple backtracking.
    # Given the time limit of 5s, backtracking might work if we order well.
    
    result = backtrack(0, [], set(), set(), set())
    
    if result:
        print("YES")
        for r in result:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
