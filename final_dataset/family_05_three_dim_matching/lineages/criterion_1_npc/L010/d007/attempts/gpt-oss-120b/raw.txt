import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    triples = []
    by_a = [[] for _ in range(k)]
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        c = int(next(it)) - 1
        triples.append((a, b, c))
        by_a[a].append((b, c, len(triples)-1))

    full_mask = (1 << k) - 1
    memo = {}

    def dfs(mask_a, mask_b, mask_c):
        if mask_a == full_mask:
            return []
        key = (mask_a, mask_b, mask_c)
        if key in memo:
            return None
        # choose an uncovered a with fewest candidates
        # compute candidates count for each uncovered a
        best_a = -1
        best_list = None
        min_len = 10**9
        for a in range(k):
            if not (mask_a >> a) & 1:
                cand = []
                for b, c, idx in by_a[a]:
                    if ((mask_b >> b) & 1) or ((mask_c >> c) & 1):
                        continue
                    cand.append((b, c, idx))
                if not cand:
                    memo[key] = None
                    return None
                if len(cand) < min_len:
                    min_len = len(cand)
                    best_a = a
                    best_list = cand
                if min_len == 1:
                    break
        # try each candidate
        for b, c, idx in best_list:
            res = dfs(mask_a | (1 << best_a), mask_b | (1 << b), mask_c | (1 << c))
            if res is not None:
                return [idx] + res
        memo[key] = None
        return None

    result = dfs(0, 0, 0)
    if result is None:
        print("NO")
    else:
        print("YES")
        for idx in result:
            a, b, c = triples[idx]
            print(a+1, b+1, c+1)

if __name__ == "__main__":
    solve()
