import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    events = []
    idx = 2
    for _ in range(t):
        events.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    # The condition (activation score of F1) * (activation score of F2) * (activation score of F3) = k^3
    # Since the max activation score for any profile is k, the only way the product is k^3
    # is if each activation score is exactly k.
    # This means each profile must have exactly k distinct indices with count > 0.
    # Since we select exactly k events, this implies each profile must have exactly one event
    # for each index from 1 to k.
    # In other words, the set of k events must be a permutation of (1..k) for each dimension.
    
    # This is equivalent to finding a set of k events such that:
    # 1. All 'a' values are distinct (1..k)
    # 2. All 'b' values are distinct (1..k)
    # 3. All 'c' values are distinct (1..k)
    # This is exactly the definition of a Latin Square of order k, 
    # but we are selecting k cells from a k x k x k cube.
    
    # Since k is small (up to 22), but t can be up to k^3, we use backtracking.
    # To optimize, we group events by their 'a' coordinate.
    
    events_by_a = [[] for _ in range(k + 1)]
    for e in events:
        events_by_a[e[0]].append(e)
        
    used_b = [False] * (k + 1)
    used_c = [False] * (k + 1)
    
    result = []
    
    def backtrack(a_idx):
        if a_idx > k:
            return True
        
        for event in events_by_a[a_idx]:
            _, b, c = event
            if not used_b[b] and not used_c[c]:
                used_b[b] = True
                used_c[c] = True
                result.append(event)
                
                if backtrack(a_idx + 1):
                    return True
                
                result.pop()
                used_b[b] = False
                used_c[c] = False
        return False

    if backtrack(1):
        print("YES")
        for e in result:
            print(f"{e[0]} {e[1]} {e[2]}")
    else:
        print("NO")

solve()
