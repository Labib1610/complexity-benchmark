import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    relays = []
    idx = 2
    for _ in range(t):
        relays.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    h = [0] * (k + 1)
    h[1] = 1
    for i in range(2, k + 1):
        h[i] = 23 * h[i-1] + 1
        
    tk = sum(h[j] for j in range(1, k + 1))
    
    # Since k is small (up to 22), and we need to pick exactly k relays,
    # if t is small, we can use backtracking or meet-in-the-middle.
    # Given the constraints and the nature of the problem, 
    # a simple recursive search with pruning is efficient enough.
    
    def find_subset(idx, count, current_a, current_b, current_c, chosen):
        if count == k:
            if current_a == tk and current_b == tk and current_c == tk:
                return chosen
            return None
        
        if idx == t or (k - count) > (t - idx):
            return None
        
        # Option 1: Include relays[idx]
        a, b, c = relays[idx]
        if current_a + h[a] <= tk and current_b + h[b] <= tk and current_c + h[c] <= tk:
            res = find_subset(idx + 1, count + 1, current_a + h[a], current_b + h[b], current_c + h[c], chosen + [relays[idx]])
            if res is not None:
                return res
        
        # Option 2: Skip relays[idx]
        res = find_subset(idx + 1, count, current_a, current_b, current_c, chosen)
        if res is not None:
            return res
            
        return None

    result = find_subset(0, 0, 0, 0, 0, [])
    
    if result:
        print("YES")
        for r in result:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
