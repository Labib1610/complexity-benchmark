Looking at this problem, I need to understand when a set of k relays forms a valid selection.

**Key insight**: The matrix M where M[i,j] = min(a_i, j) has non-zero determinant if and only if the multiset of a-values is exactly {1, 2, ..., k} (each appearing exactly once).

**Proof sketch**: 
- If any value v in {1,...,k} is missing, then column v equals column v+1 (both have all entries ≤ v), making the determinant zero.
- If the values are exactly {1,...,k}, sorting gives a lower triangular matrix with diagonal entries 1,2,...,k, so determinant = k! ≠ 0.

So I need to find k relays such that for each channel type, the selected relays have exactly the values {1, 2, ..., k}.

This is a 3-dimensional bipartite matching problem. I'll use maximum bipartite matching with augmenting paths, trying all possible assignments for the first channel, then checking if the remaining relays can form valid matchings for channels 2 and 3.

```python
import sys
from itertools import permutations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    idx = 0
    k = int(input_data[idx]); idx += 1
    t = int(input_data[idx]); idx += 1
    relays = []
    for _ in range(t):
        a = int(input_data[idx]); idx += 1
        b = int(input_data[idx]); idx += 1
        c = int(input_data[idx]); idx += 1
        relays.append((a, b, c))
    
    if k == 0:
        print("YES")
        return
    
    # Group relays by their 'a' value
    by_a = [[] for _ in range(k + 1)]
    for i, (a, b, c) in enumerate(relays):
        by_a[a].append(i)
    
    # Check if any a-value has no relays
    for v in range(1, k + 1):
        if not by_a[v]:
            print("NO")
            return
    
    # Bipartite matching with augmenting paths
    def can_match(adj, n_left, n_right):
        match_r = [-1] * (n_right + 1)
        def try_kuhn(v, visited):
            for u in adj[v]:
                if not visited[u]:
                    visited[u] = True
                    if match_r[u] == -1 or try_kuhn(match_r[u], visited):
                        match_r[u] = v
                        return True
            return False
        for v in range(1, n_left + 1):
            visited = [False] * (n_right + 1)
            if not try_kuhn(v, visited):
                return None
        return match_r
    
    # Try all permutations of which relay is assigned to each a-value
    # For each, check if b and c channels can also form perfect matchings
    for perm in permutations(range(k)):
        # perm[i] = index into by_a[i+1] of the relay assigned a-value (i+1)
        selected = []
        valid = True
        for i in range(k):
            lst = by_a[i + 1]
            if perm[i] >= len(lst):
                valid = False
                break
            selected.append(relays[lst[perm[i]]])
        if not valid:
            continue
        
        # Check b-channel matching
        adj_b = [[] for _ in range(k + 1)]
        for i in range(k):
            adj_b[i + 1].append(selected[i][1])
        if can_match(adj_b, k, k) is None:
            continue
        
        # Check c-channel matching
        adj_c = [[] for _ in range(k + 1)]
        for i in range(k):
            adj_c[i + 1].append(selected[i][2])
        if can_match(adj_c, k, k) is None:
            continue
        
        print("YES")
        for (a, b, c) in selected:
            print(a, b, c)
        return
    
    print("NO")

solve()
```
