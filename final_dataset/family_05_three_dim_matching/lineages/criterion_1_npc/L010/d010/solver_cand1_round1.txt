import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    relays = []
    idx = 2
    for _ in range(t):
        relays.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    def check_det(vals, k):
        # The matrix M_{i,j} = min(a_i, j)
        # Sort vals: a_1 <= a_2 <= ... <= a_k
        # The determinant of this matrix is non-zero if and only if
        # the values {a_1, ..., a_k} are such that when sorted,
        # a_i >= i for all i.
        # Actually, for M_{i,j} = min(a_i, j), the determinant is non-zero
        # if and only if the values {a_1, ..., a_k} are a permutation of {1, ..., k}.
        # Wait, let's re-verify:
        # If a_i = i, M is the identity matrix (det=1).
        # If we have a set of values, the matrix is non-singular iff
        # the values are a permutation of {1, 2, ..., k}.
        # Let's check Example 1: k=2, relays (1,1,1) and (2,2,2).
        # Primary: {1, 2}. Sorted: 1, 2. M = [[min(1,1), min(1,2)], [min(2,1), min(2,2)]] = [[1, 1], [1, 2]]. Det = 1.
        # Yes, the condition is that the set of values must be exactly {1, 2, ..., k}.
        s = sorted(vals)
        for i in range(k):
            if s[i] != i + 1:
                return False
        return True

    # Since k is small (up to 22), but t can be up to k^3, we need an efficient way.
    # However, the condition "must be a permutation of 1..k" is very strong.
    # We need to pick k relays such that their primary values are a permutation of 1..k,
    # their secondary values are a permutation of 1..k,
    # and their tertiary values are a permutation of 1..k.
    
    # This is equivalent to finding a system of distinct representatives or a matching.
    # We can model this as a 3D matching problem, but k=22 is too large for general 3D matching.
    # Wait, the condition is simply that we need to pick k relays (a_i, b_i, c_i)
    # such that {a_1...a_k} = {1..k}, {b_1...b_k} = {1..k}, {c_1...c_k} = {1..k}.
    
    # This is exactly the problem of finding a transversal in a 3-partite hypergraph.
    # Given the constraints and the nature of the problem, backtracking with pruning should work.
    
    chosen = []
    used_a = [False] * (k + 1)
    used_b = [False] * (k + 1)
    used_c = [False] * (k + 1)
    
    def backtrack(idx, count):
        if count == k:
            return True
        if idx == t:
            return False
        
        # Try including relays[idx]
        a, b, c = relays[idx]
        if not used_a[a] and not used_b[b] and not used_c[c]:
            used_a[a] = used_b[b] = used_c[c] = True
            chosen.append(relays[idx])
            if backtrack(idx + 1, count + 1):
                return True
            chosen.pop()
            used_a[a] = used_b[b] = used_c[c] = False
            
        # Try skipping
        if backtrack(idx + 1, count):
            return True
        return False

    if backtrack(0, 0):
        print("YES")
        for r in chosen:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
