import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    transactions = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        transactions.append((x, y, z))
        idx += 3

    # We need to select k transactions such that:
    # - Each x in {0..k-1} appears exactly once.
    # - Each y in {0..k-1} appears exactly once.
    # - Each z in {0..k-1} appears exactly once.
    # This is equivalent to finding a perfect matching in a 3D hypergraph,
    # or more specifically, a set of k triples (x_i, y_i, z_i) such that
    # {x_i} = {0..k-1}, {y_i} = {0..k-1}, {z_i} = {0..k-1}.
    
    # Since k <= 22, we can use meet-in-the-middle or backtracking with pruning.
    # Given the constraints and the nature of the problem, backtracking with
    # bitmasks is efficient.
    
    used_x = 0
    used_y = 0
    used_z = 0
    
    # Group transactions by x to simplify the search
    by_x = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(transactions):
        by_x[x].append((y, z, i))
        
    ans = []
    
    def backtrack(x_idx, current_used_y, current_used_z):
        if x_idx == k:
            return True
        
        for y, z, original_idx in by_x[x_idx]:
            if not (current_used_y & (1 << y)) and not (current_used_z & (1 << z)):
                ans.append(transactions[original_idx])
                if backtrack(x_idx + 1, current_used_y | (1 << y), current_used_z | (1 << z)):
                    return True
                ans.pop()
        return False

    if backtrack(0, 0, 0):
        print("YES")
        for t in ans:
            print(f"{t[0]+1} {t[1]+1} {t[2]+1}")
    else:
        print("NO")

solve()
