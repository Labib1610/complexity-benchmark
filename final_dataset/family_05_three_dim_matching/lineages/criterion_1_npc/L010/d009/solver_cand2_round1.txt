import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    candidates = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        candidates.append((a, b, c))
        idx += 3
        
    # The condition "coverage length is exactly k" means that for each spectrum,
    # the set of focal points must be exactly {1, 2, ..., k}.
    # This is because the union of [x-1, x] for x in {1, ..., k} is [0, k],
    # which has length k. If any value in {1, ..., k} is missing, the length
    # will be less than k.
    
    # We need to pick k stabilizers (a_i, b_i, c_i) such that:
    # {a_1, ..., a_k} = {1, ..., k}
    # {b_1, ..., b_k} = {1, ..., k}
    # {c_1, ..., c_k} = {1, ..., k}
    
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    chosen = []
    
    def backtrack(idx, mask_a, mask_b, mask_c, count):
        if count == k:
            if mask_a == (1 << k) - 1 and mask_b == (1 << k) - 1 and mask_c == (1 << k) - 1:
                return True
            return False
        
        if idx == t:
            return False
        
        # Pruning: if remaining candidates are not enough to fill the masks
        if (t - idx) < (k - count):
            return False
            
        # Try including candidates[idx]
        a, b, c = candidates[idx]
        new_mask_a = mask_a | (1 << (a - 1))
        new_mask_b = mask_b | (1 << (b - 1))
        new_mask_c = mask_c | (1 << (c - 1))
        
        chosen.append(candidates[idx])
        if backtrack(idx + 1, new_mask_a, new_mask_b, new_mask_c, count + 1):
            return True
        chosen.pop()
        
        # Try excluding candidates[idx]
        if backtrack(idx + 1, mask_a, mask_b, mask_c, count):
            return True
            
        return False

    if backtrack(0, 0, 0, 0, 0):
        print("YES")
        for item in chosen:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
