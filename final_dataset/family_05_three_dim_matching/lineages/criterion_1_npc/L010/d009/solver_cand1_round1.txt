import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    relays = []
    idx = 2
    for _ in range(t):
        relays.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    if k > t:
        print("NO")
        return

    # The condition that the determinant is non-zero means the chosen relays
    # must have distinct primary, secondary, and tertiary channels.
    # This is equivalent to finding a subset of size k such that:
    # {a_i} = {1..k}, {b_i} = {1..k}, {c_i} = {1..k}.
    # This is a 3D matching problem, which is NP-hard.
    # However, k is small (up to 22). We can use a randomized approach.
    
    # Represent each relay as a vector in a space where we can use hashing
    # to check if a set of k relays forms a permutation in each dimension.
    # Assign a random 64-bit integer to each channel value for each dimension.
    
    random.seed(42)
    dim1_vals = [random.getrandbits(64) for _ in range(k + 1)]
    dim2_vals = [random.getrandbits(64) for _ in range(k + 1)]
    dim3_vals = [random.getrandbits(64) for _ in range(k + 1)]
    
    target = sum(dim1_vals[1:]) + sum(dim2_vals[1:]) + sum(dim3_vals[1:])
    
    # Try randomized search
    for _ in range(200000):
        subset = random.sample(relays, k)
        s1 = sum(dim1_vals[r[0]] for r in subset)
        s2 = sum(dim2_vals[r[1]] for r in subset)
        s3 = sum(dim3_vals[r[2]] for r in subset)
        
        if s1 == sum(dim1_vals[1:]) and s2 == sum(dim2_vals[1:]) and s3 == sum(dim3_vals[1:]):
            # Verify
            p = [r[0] for r in subset]
            s = [r[1] for r in subset]
            t_ = [r[2] for r in subset]
            if len(set(p)) == k and len(set(s)) == k and len(set(t_)) == k:
                print("YES")
                for r in subset:
                    print(f"{r[0]} {r[1]} {r[2]}")
                return

    print("NO")

solve()
