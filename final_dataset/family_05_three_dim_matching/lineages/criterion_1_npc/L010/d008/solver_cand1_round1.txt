import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    relays = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        relays.append((a, b, c))
        idx += 3
        
    # The condition sum(2^{a_i-1}) = 2^k - 1 means that for each position j in {1..k},
    # exactly one relay must have a_i = j.
    # This must hold for all three dimensions (a, b, c).
    # This is equivalent to finding a subset of size k such that:
    # The set of 'a' values is {1, 2, ..., k}
    # The set of 'b' values is {1, 2, ..., k}
    # The set of 'c' values is {1, 2, ..., k}
    
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    # However, the constraints on t are small (t <= k^3).
    
    # We need to pick k relays such that each value 1..k appears exactly once in each column.
    
    def backtrack(idx, count, used_a, used_b, used_c, current_selection):
        if count == k:
            return current_selection
        
        if idx == t:
            return None
        
        # Pruning: if remaining relays are not enough
        if count + (t - idx) < k:
            return None
            
        a, b, c = relays[idx]
        
        # Try including relays[idx]
        if not (used_a & (1 << (a - 1))) and not (used_b & (1 << (b - 1))) and not (used_c & (1 << (c - 1))):
            res = backtrack(idx + 1, count + 1, used_a | (1 << (a - 1)), used_b | (1 << (b - 1)), used_c | (1 << (c - 1)), current_selection + [relays[idx]])
            if res is not None:
                return res
                
        # Try excluding relays[idx]
        res = backtrack(idx + 1, count, used_a, used_b, used_c, current_selection)
        if res is not None:
            return res
            
        return None

    result = backtrack(0, 0, 0, 0, 0, [])
    
    if result:
        print("YES")
        for r in result:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
