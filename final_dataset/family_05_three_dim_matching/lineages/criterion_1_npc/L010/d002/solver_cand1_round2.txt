import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triplets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triplets.append((x, y, z))
        idx += 3
        
    # Group triplets by technician x
    by_x = [[] for _ in range(k)]
    for x, y, z in triplets:
        by_x[x].append((y, z))
        
    # dp[mask_y][mask_z] stores the list of triplets used
    # Since k <= 22, we use a recursive search with pruning/memoization
    # or backtracking. Given the constraints and the nature of 3D matching,
    # we can use backtracking with bitmasks.
    
    memo = {}

    def backtrack(x, mask_y, mask_z, current_squads):
        if x == k:
            return current_squads
        
        state = (x, mask_y, mask_z)
        if state in memo:
            return None
        
        for y, z in by_x[x]:
            if not (mask_y & (1 << y)) and not (mask_z & (1 << z)):
                res = backtrack(x + 1, mask_y | (1 << y), mask_z | (1 << z), current_squads + [(x + 1, y + 1, z + 1)])
                if res is not None:
                    return res
        
        memo[state] = False
        return None

    result = backtrack(0, 0, 0, [])
    
    if result:
        print("YES")
        for s in result:
            print(f"{s[0]} {s[1]} {s[2]}")
    else:
        print("NO")

solve()
