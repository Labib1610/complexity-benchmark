import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem is equivalent to finding a system of distinct representatives
    # or a perfect matching in a 3-partite hypergraph.
    # Since k is small (max 11), we can use backtracking with pruning.
    
    edges = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        edges.append((x, y, z))
        idx += 3
        
    # Group edges by supervisor to simplify selection
    adj = [[] for _ in range(k)]
    for x, y, z in edges:
        adj[x].append((y, z))
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x):
        if x == k:
            return True
        
        for y, z in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    # Heuristic: sort supervisors by number of available options
    # This is a standard optimization for exact cover problems
    if backtrack(0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
