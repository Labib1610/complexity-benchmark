        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))

        by_x[x].append((y, z))

    # This is equivalent to finding a perfect matching in a 3D hypergraph,
    # or more simply, a set of k triples (x_i, y_i, z_i) such that {x_i} = {0..k-1}, {y_i} = {0..k-1}, {z_i} = {0..k-1}.
    
    # Use backtracking with bitmasks for y and z.
    # To optimize, pick the technician with the fewest options first (MRV heuristic).
    
    used_y = 0
    used_z = 0
    result = []

    def backtrack(x_idx, current_used_y, current_used_z):
        if x_idx == k:
            return True
        
        # MRV: find x with fewest available options
        best_x = -1
        min_options = k + 1

                for y, z in by_x[i]:
                    if not (current_used_y & (1 << y)) and not (current_used_z & (1 << z)):
                        count += 1
                if count < min_options:
                    min_options = count
                    best_x = i
        
        if best_x == -1 or min_options == 0:
            return False

        assigned_x |= (1 << best_x)
        
        for y, z in by_x[best_x]:
            if not (current_used_y & (1 << y)) and not (current_used_z & (1 << z)):
