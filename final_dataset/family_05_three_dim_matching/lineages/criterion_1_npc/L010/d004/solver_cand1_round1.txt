import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    relays = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        relays.append((x, y, z))
        idx += 3
        
    # We need to pick k relays (x_i, y_i, z_i) such that all x_i are distinct,
    # all y_i are distinct, and all z_i are distinct.
    # This is equivalent to finding a perfect matching in a 3-partite hypergraph,
    # or more simply, a set of k triples such that each coordinate 1..k appears exactly once.
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    # Group relays by x to simplify backtracking
    by_x = [[] for _ in range(k + 1)]
    for r in relays:
        by_x[r[0]].append(r)
        
    result = []
    
    def backtrack(x_idx):
        if x_idx > k:
            return True
        
        for r in by_x[x_idx]:
            _, y, z = r
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(r)
                
                if backtrack(x_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for r in result:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
