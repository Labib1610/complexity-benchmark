import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    relays = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        relays.append((x, y, z))
        idx += 3

    # The condition sum(6^x_i) = sum(6^j) for j=1..k means that
    # for each value v in {1..k}, the count of relays with parameter x=v must be exactly 1.
    # This is equivalent to saying that the set of x-coordinates must be a permutation of 1..k,
    # the set of y-coordinates must be a permutation of 1..k,
    # and the set of z-coordinates must be a permutation of 1..k.
    # This is exactly the definition of a Latin Square of order k,
    # but we are selecting k cells from a k x k x k cube such that each line
    # parallel to the axes contains exactly one selected cell.
    
    # This is equivalent to finding a perfect matching in a 3-partite hypergraph,
    # or more simply, finding a set of k triplets (x_i, y_i, z_i) such that
    # {x_1, ..., x_k} = {1, ..., k}, {y_1, ..., y_k} = {1, ..., k}, {z_1, ..., z_k} = {1, ..., k}.
    
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    chosen = []

    def backtrack(idx):
        if len(chosen) == k:
            return True
        
        for i in range(idx, t):
            x, y, z = relays[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = used_y[y] = used_z[z] = True
                chosen.append(relays[i])
                if backtrack(i + 1):
                    return True
                chosen.pop()
                used_x[x] = used_y[y] = used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for r in chosen:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
