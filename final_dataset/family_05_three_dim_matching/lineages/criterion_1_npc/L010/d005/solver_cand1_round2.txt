import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    relays = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        relays.append((x, y, z))
        idx += 3
        
    if k == 1:
        if t >= 1:
            print("YES")
            print(f"{relays[0][0]} {relays[0][1]} {relays[0][2]}")
        else:
            print("NO")
        return

    # The condition product > 0 means for all i != j:
    # x_i != x_j AND y_i != y_j AND z_i != z_j
    # This is equivalent to saying that the set of x-coordinates is a permutation of 1..k,
    # the set of y-coordinates is a permutation of 1..k,
    # and the set of z-coordinates is a permutation of 1..k.
    # This is exactly the definition of a Latin Square of order k represented as a set of triples.
    
    # Since k is small (up to 22), we can use backtracking with pruning.
    # We need to pick k relays such that all x are distinct, all y are distinct, all z are distinct.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    chosen = []
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for i in range(start_idx, t):
            x, y, z = relays[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = True
                used_y[y] = True
                used_z[z] = True
                chosen.append(relays[i])
                
                if backtrack(i + 1, count + 1):
                    return True
                
                chosen.pop()
                used_x[x] = False
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for r in chosen:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
