```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    relays = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        relays.append((x, y, z))
        idx += 3
        
    target = (5**(k + 1) - 5) // 4
    
    # The condition sum(5^x_i) = sum(5^y_i) = sum(5^z_i) = sum_{j=1}^k 5^j
    # is equivalent to representing the sum in base 5.
    # Since 5^1 + 5^2 + ... + 5^k = (5^1)(5^k - 1)/(5 - 1) = (5^{k+1} - 5)/4,
    # the condition is satisfied if and only if for each band, the count of 
    # relays with parameter value j is exactly 1 for each j in {1, ..., k}.
    
    # This is a variation of the exact cover problem or subset sum.
    # Given k is small (up to 22), we can use backtracking with pruning.
    
    memo = {}

    def backtrack(idx, count, sum_x, sum_y, sum_z):
        if count == k:
            return [] if sum_x == target and sum_y == target and sum_z == target else None
        if idx == t or count + (t - idx) < k:
            return None
        
        state = (idx, count, sum_x, sum_y, sum_z)
        if state in memo:
            return memo[state]
        
        # Try including relays[idx]
        rx, ry, rz = relays[idx]
        nx, ny, nz = sum_x + 5**rx, sum_y + 5**ry, sum_z + 5**rz
        
        if nx <= target and ny <= target and nz <= target:
            res = backtrack(idx + 1, count + 1, nx, ny, nz)
            if res is not None:
                memo[state] = [relays[idx]] + res
                return memo[state]
        
        # Try excluding relays[idx]
        res = backtrack(idx + 1, count, sum_x, sum_y, sum_z)
        memo[state] = res
        return res

    result = backtrack(0, 0, 0, 0, 0)
    
    if result is not None:
        print("YES")
        for r in result:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
```
