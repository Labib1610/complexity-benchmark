import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    deployments = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        deployments.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    # We need to pick k deployments such that each x, y, z is used exactly once.
    
    # Group deployments by x
    by_x = [[] for _ in range(k)]
    for d in deployments:
        by_x[d[0]].append(d)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x_idx):
        if x_idx == k:
            return True
        
        for d in by_x[x_idx]:
            _, y, z = d
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(d)
                
                if backtrack(x_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for d in result:
            print(f"{d[0]+1} {d[1]+1} {d[2]+1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
