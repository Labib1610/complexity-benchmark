```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triplets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triplets.append((x, y, z))
        idx += 3
        
    # dp[mask_y][mask_z] stores the mask_x that can form a valid set of crews
    # using machines in mask_y and time intervals in mask_z.
    # Since k <= 22, we use a meet-in-the-middle or optimized search.
    # Given the constraints and the nature of 3D matching, we use backtracking with pruning.
    
    adj = [[] for _ in range(k)]
    for x, y, z in triplets:
        adj[x].append((y, z))
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x):
        if x == k:
            return True
        
        for y, z in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    # Sort adj to try to find a solution faster
    for i in range(k):
        adj[i].sort()
        
    if backtrack(0):
        print("YES")
        for crew in result:
            print(f"{crew[0]} {crew[1]} {crew[2]}")
    else:
        print("NO")

solve()
```
