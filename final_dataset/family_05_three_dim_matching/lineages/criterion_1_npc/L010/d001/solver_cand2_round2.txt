```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    adj = [[] for _ in range(k)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        adj[x].append((y, z))
        idx += 3
        
    # We need to select k triples (x, y, z) such that each x, y, z is used exactly once.
    # Since x is 0..k-1, we must pick exactly one (y, z) for each x.
    # This is equivalent to finding a perfect matching in a 3-partite hypergraph,
    # or more simply, selecting one edge from each x such that all y's and z's are distinct.
    
    # State: (index of x, mask of used y, mask of used z)
    # Since k <= 22, we use backtracking with pruning.
    
    memo = {}

    def backtrack(x, y_mask, z_mask, current_selection):
        if x == k:
            return current_selection
        
        state = (x, y_mask, z_mask)
        if state in memo:
            return None
        
        for y, z in adj[x]:
            if not (y_mask & (1 << y)) and not (z_mask & (1 << z)):
                res = backtrack(x + 1, y_mask | (1 << y), z_mask | (1 << z), current_selection + [(x + 1, y + 1, z + 1)])
                if res is not None:
                    return res
        
        memo[state] = False
        return None

    result = backtrack(0, 0, 0, [])
    
    if result:
        print("YES")
        for crew in result:
            print(f"{crew[0]} {crew[1]} {crew[2]}")
    else:
        print("NO")

solve()
```
