We can model this as a matching problem:  
We need to assign to each frequency `x` a unique amplitude `y` and a unique phase `z`, using only the given modules.

Since `k ≤ 22`, we can use bitmask DP over amplitudes and phases.

We’ll do a recursive backtracking with memoization:  
State: `(freq_index, amplitude_mask, phase_mask)`  
We try to pick a module for the current frequency that hasn’t used its amplitude or phase yet.

```python
import sys

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    modules = [[] for _ in range(k + 1)]
    for _ in range(t):
        x, y, z = map(int, input().split())
        modules[x].append((y, z))
    
    # Precompute for each frequency the list of (amplitude, phase) pairs
    # We'll use bitmasks for amplitudes and phases (0-indexed)
    freq_options = []
    for x in range(1, k + 1):
        opts = []
        for y, z in modules[x]:
            opts.append((y - 1, z - 1))
        freq_options.append(opts)
    
    from functools import lru_cache
    
    @lru_cache(None)
    def dfs(i, amp_mask, phase_mask):
        if i == k:
            return True
        for y, z in freq_options[i]:
            if not (amp_mask >> y) & 1 and not (phase_mask >> z) & 1:
                if dfs(i + 1, amp_mask | (1 << y), phase_mask | (1 << z)):
                    return True
        return False
    
    if not dfs(0, 0, 0):
        print("NO")
        return
    
    # Reconstruct solution
    amp_mask = 0
    phase_mask = 0
    result = []
    for i in range(k):
        for y, z in freq_options[i]:
            if not (amp_mask >> y) & 1 and not (phase_mask >> z) & 1:
                if dfs(i + 1, amp_mask | (1 << y), phase_mask | (1 << z)):
                    result.append((i + 1, y + 1, z + 1))
                    amp_mask |= (1 << y)
                    phase_mask |= (1 << z)
                    break
    
    print("YES")
    for x, y, z in result:
        print(x, y, z)

if __name__ == "__main__":
    solve()
```
