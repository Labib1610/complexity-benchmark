import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for _ in range(t):
        modules.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    target = (3**k - 1) // 2
    
    # The condition sum(3^(k-v_i)) = (3^k - 1) / 2
    # In base 3, (3^k - 1) / 2 is represented as a string of k ones: 11...1 (k times)
    # This means for each parameter, we need exactly one module with value 1, 
    # one with value 2, ..., one with value k.
    # Wait, let's re-check: 
    # If we have one of each value 1, 2, ..., k:
    # Sum = 3^(k-1) + 3^(k-2) + ... + 3^(k-k) = 3^(k-1) + ... + 3^0
    # This is a geometric series: (3^k - 1) / (3 - 1) = (3^k - 1) / 2.
    # Yes, the condition is equivalent to: 
    # For each parameter, the set of k modules must contain exactly one of each value {1, 2, ..., k}.
    
    # This is an Exact Cover problem.
    # We need to pick k modules such that:
    # Frequency values are a permutation of 1..k
    # Amplitude values are a permutation of 1..k
    # Phase values are a permutation of 1..k
    
    # Since k <= 22, we can use backtracking with pruning.
    # Given the constraints and the nature of the problem, 
    # we can treat this as finding a set of k modules that satisfy the constraints.
    
    selected = []
    used_f = [False] * (k + 1)
    used_a = [False] * (k + 1)
    used_p = [False] * (k + 1)
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        # Heuristic: if remaining modules are not enough, return
        if t - start_idx < k - count:
            return False
            
        for i in range(start_idx, t):
            f, a, p = modules[i]
            if not used_f[f] and not used_a[a] and not used_p[p]:
                used_f[f] = used_a[a] = used_p[p] = True
                selected.append(modules[i])
                if backtrack(i + 1, count + 1):
                    return True
                selected.pop()
                used_f[f] = used_a[a] = used_p[p] = False
        return False

    # Sort modules to potentially hit a solution faster
    if backtrack(0, 0):
        print("YES")
        for m in selected:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

solve()
