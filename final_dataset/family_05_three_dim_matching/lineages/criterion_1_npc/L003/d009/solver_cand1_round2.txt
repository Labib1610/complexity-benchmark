import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        modules.append((x, y, z))
        idx += 3

    # This is an Exact Cover problem.
    # We need to select k modules such that:
    # 1. Each x in 0..k-1 is used exactly once.
    # 2. Each y in 0..k-1 is used exactly once.
    # 3. Each z in 0..k-1 is used exactly once.
    # This is equivalent to finding a transversal in a 3D k x k x k binary matrix.
    
    # Since k <= 22, we use backtracking with pruning (Dancing Links style).
    # Represent constraints:
    # x_i: module with frequency i
    # y_j: module with amplitude j
    # z_l: module with phase l
    
    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(modules):
        adj[x].append((y, z, i))
        
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        # Heuristic: pick x with fewest options
        for y, z, i in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(i)
                if backtrack(x + 1):
                    return True
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for idx in result:
            x, y, z = modules[idx]
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")

solve()
