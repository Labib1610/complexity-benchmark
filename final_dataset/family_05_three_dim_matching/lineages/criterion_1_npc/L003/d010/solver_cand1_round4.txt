import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        modules.append((x, y, z))
        idx += 3
        
    # This is equivalent to finding a transversal in a 3D Latin Square or 
    # a matching in a 3-partite hypergraph. Since k <= 22, we can use 
    # backtracking with pruning.
    
    adj = [[] for _ in range(k + 1)]
    for m in modules:
        adj[m[0]].append((m[1], m[2]))
        
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    result = []
    
    def backtrack(x):
        if x > k:
            return True
        
        for y, z in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x, y, z))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for m in result:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
