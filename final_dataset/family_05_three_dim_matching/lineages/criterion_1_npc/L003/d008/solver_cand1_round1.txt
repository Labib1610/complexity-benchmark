import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    if k == 0:
        print("YES")
        return
    
    modules = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        modules.append((x, y, z))
        idx += 3
        
    # The matrix M is a Vandermonde matrix.
    # The determinant of a k x k Vandermonde matrix is the product of (v_j - v_i) for 1 <= i < j <= k.
    # This determinant is non-zero if and only if all v_i are distinct.
    # Thus, we need to select k modules such that all frequencies are distinct,
    # all amplitudes are distinct, and all phases are distinct.
    
    # Since x, y, z are in range [1, k], and we need k modules,
    # each parameter must be a permutation of 1..k.
    
    # This is equivalent to finding a set of k triples (x_i, y_i, z_i)
    # such that {x_1, ..., x_k} = {1, ..., k}, {y_1, ..., y_k} = {1, ..., k},
    # and {z_1, ..., z_k} = {1, ..., k}.
    # This is the exact cover problem or a 3D matching problem.
    
    # Given k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    result = []
    
    def backtrack(idx):
        if len(result) == k:
            return True
        
        if idx >= t:
            return False
        
        # Try including modules[idx]
        x, y, z = modules[idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append(modules[idx])
            if backtrack(idx + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try skipping modules[idx]
        # Pruning: if remaining modules are not enough to fill k
        if (t - 1 - idx) >= (k - len(result)):
            if backtrack(idx + 1):
                return True
        
        return False

    if backtrack(0):
        print("YES")
        for m in result:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

solve()
