import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for _ in range(t):
        modules.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    if k == 0:
        print("YES")
        return

    # E_v = k * E_{v-1} + 1
    # E_1 = 1
    # E_2 = k + 1
    # E_3 = k(k+1) + 1 = k^2 + k + 1
    # E_v = (k^v - 1) / (k - 1) if k > 1, else E_v = 1
    
    if k == 1:
        e = [0, 1]
    else:
        e = [0] * (k + 1)
        e[1] = 1
        for i in range(2, k + 1):
            e[i] = k * e[i-1] + 1
            
    target = sum(e[1:])
    
    # We need to pick k modules (x_i, y_i, z_i) such that
    # sum(e[x_i]) = target, sum(e[y_i]) = target, sum(e[z_i]) = target
    # Since x, y, z are in [1, k], e[x] is a value.
    # This is a variation of the subset sum problem.
    # Given k is small (up to 22), we can use meet-in-the-middle or backtracking.
    # However, t can be up to k^3 = 10648.
    
    memo = {}

    def backtrack(idx, count, s1, s2, s3, path):
        if count == k:
            return (s1 == target and s2 == target and s3 == target)
        if idx == t or count + (t - idx) < k:
            return False
        
        state = (idx, count, s1, s2, s3)
        if state in memo:
            return memo[state]
        
        # Try including modules[idx]
        m = modules[idx]
        if s1 + e[m[0]] <= target and s2 + e[m[1]] <= target and s3 + e[m[2]] <= target:
            path.append(m)
            if backtrack(idx + 1, count + 1, s1 + e[m[0]], s2 + e[m[1]], s3 + e[m[2]], path):
                return True
            path.pop()
            
        # Try excluding modules[idx]
        if backtrack(idx + 1, count, s1, s2, s3, path):
            return True
            
        memo[state] = False
        return False

    path = []
    if backtrack(0, 0, 0, 0, 0, path):
        print("YES")
        for m in path:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

solve()
