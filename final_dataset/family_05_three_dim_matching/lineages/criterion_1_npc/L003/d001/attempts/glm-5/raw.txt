import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    for i in range(t):
        x = int(input_data[2 + 3*i])
        y = int(input_data[3 + 3*i])
        z = int(input_data[4 + 3*i])
        configs.append((x, y, z))
        
    adj = [[] for _ in range(k + 1)]
    for x, y, z in configs:
        adj[x].append((y, z))
        
    match_y = [0] * (k + 1)
    match_z = [0] * (k + 1)
    
    def dfs(u, visited):
        for v, w in adj[u]:
            if visited[v][w]:
                continue
            visited[v][w] = True
            if match_y[v] == 0 or dfs(match_y[v], visited):
                match_y[v] = u
                match_z[w] = u
                return True
        return False

    for u in range(1, k + 1):
        visited = [[False] * (k + 1) for _ in range(k + 1)]
        if not dfs(u, visited):
            print("NO")
            return
            
    print("YES")
    ans = [None] * (k + 1)
    for v in range(1, k + 1):
        u = match_y[v]
        w = match_z[v]
        ans[u] = (u, v, w)
        
    for i in range(1, k + 1):
        print(f"{ans[i][0]} {ans[i][1]} {ans[i][2]}")

if __name__ == '__main__':
    solve()
