            return memo[(mask_y, mask_z)]

                if dfs(mask_y | (1 << y), mask_z | (1 << z)):
                    choice[(mask_y, mask_z)] = (y, z)
                    memo[(mask_y, mask_z)] = True
                    return True
        memo[(mask_y, mask_z)] = False
        return False

        # Reconstruct
        result = []
        mask_y, mask_z = 0, 0

            mask_z |= (1 << z)
        print("YES")
        for x, y, z in result:
            print(x, y, z)
    else:
        print("NO")

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    configs = [[] for _ in range(k)]
    for _ in range(t):
        x, y, z = map(int, input().split())
        configs[x-1].append((y-1, z-1))
    
    full_mask = (1 << k) - 1
    
    # Precompute for each actuator the list of (y,z) as bitmask pairs? Not needed.
    
    # We'll use a recursive backtracking with dynamic ordering.
    
    # To speed up, we can precompute for each actuator the list of (y,z) as (1<<y, 1<<z)
    options = []
    for i in range(k):
        opts = []
        for y, z in configs[i]:
            opts.append((1 << y, 1 << z))
        options.append(opts)
    
    # We'll store the solution path
    solution = []
    
    def backtrack(mask_y, mask_z, remaining):
        if remaining == 0:
            return mask_y == full_mask and mask_z == full_mask
        
        # Choose the actuator with fewest available options
        best_idx = -1
        best_count = k+1
        best_opts = None
        # Iterate over remaining actuators
        rem = remaining
        while rem:
            # get lowest set bit
            lsb = rem & -rem
            idx = (lsb.bit_length() - 1)
            rem ^= lsb
            # count available options
            cnt = 0
            opts = []
            for y_mask, z_mask in options[idx]:
                if not (mask_y & y_mask) and not (mask_z & z_mask):
                    cnt += 1
                    opts.append((y_mask, z_mask))
            if cnt == 0:
                return False
            if cnt < best_count:
                best_count = cnt
                best_idx = idx
                best_opts = opts
                if cnt == 1:
                    break  # can't do better
        
        # Try each option for best_idx
        for y_mask, z_mask in best_opts:
            # Add to solution
            solution.append((best_idx, y_mask, z_mask))
            if backtrack(mask_y | y_mask, mask_z | z_mask, remaining ^ (1 << best_idx)):
                return True
            solution.pop()
        return False
    
    if backtrack(0, 0, full_mask):
        print("YES")
        for idx, y_mask, z_mask in solution:
            y = (y_mask.bit_length() - 1) + 1
            z = (z_mask.bit_length() - 1) + 1
            print(idx+1, y, z)
    else:
        print("NO")

    solve()
