import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        configs.append((x, y, z))
        idx += 3
        
    # We need to select k triples (x_i, y_i, z_i) such that
    # {x_0, ..., x_{k-1}} = {0, ..., k-1}
    # {y_0, ..., y_{k-1}} = {0, ..., k-1}
    # {z_0, ..., z_{k-1}} = {0, ..., k-1}
    
    # This is an exact cover problem. Since k <= 22, we can use backtracking.
    # To optimize, we group configs by x.
    by_x = [[] for _ in range(k)]
    for c in configs:
        by_x[c[0]].append(c)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x_idx):
        if x_idx == k:
            return True
        
        for c in by_x[x_idx]:
            _, y, z = c
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(c)
                
                if backtrack(x_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for c in result:
            print(f"{c[0]+1} {c[1]+1} {c[2]+1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
