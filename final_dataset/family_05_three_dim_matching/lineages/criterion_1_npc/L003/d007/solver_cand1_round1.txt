import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    N = int(input_data[0])
    M = int(input_data[1])
    nodes = []
    idx = 2
    for _ in range(M):
        nodes.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    target = (1 << (N + 1)) - 2
    
    # We need sum(2^a_i) = target, sum(2^b_i) = target, sum(2^c_i) = target
    # Since 1 <= a_i, b_i, c_i <= N, and sum(2^h) = 2^1 + 2^2 + ... + 2^N = 2^{N+1} - 2
    # This is equivalent to saying that for each harmonic, the count of nodes 
    # with harmonic value h must be exactly 1 for each h in {1, ..., N}.
    
    # This is a variation of the exact cover problem.
    # We need to pick N nodes such that:
    # 1. Each harmonic value 1..N appears exactly once in Alpha
    # 2. Each harmonic value 1..N appears exactly once in Beta
    # 3. Each harmonic value 1..N appears exactly once in Gamma
    
    # Since N is small (up to 22), we can use backtracking with pruning.
    # State: (index_in_nodes, count_nodes, used_alpha_mask, used_beta_mask, used_gamma_mask, chosen_nodes)
    
    target_mask = (1 << (N + 1)) - 2
    
    memo = {}

    def backtrack(idx, count, a_mask, b_mask, g_mask, chosen):
        if count == N:
            if a_mask == target_mask and b_mask == target_mask and g_mask == target_mask:
                return chosen
            return None
        
        if idx == M:
            return None
        
        state = (idx, count, a_mask, b_mask, g_mask)
        if state in memo:
            return None
        
        # Try including nodes[idx]
        a, b, g = nodes[idx]
        if not (a_mask & (1 << a)) and not (b_mask & (1 << b)) and not (g_mask & (1 << g)):
            res = backtrack(idx + 1, count + 1, a_mask | (1 << a), b_mask | (1 << b), g_mask | (1 << g), chosen + [nodes[idx]])
            if res is not None:
                return res
        
        # Try excluding nodes[idx]
        # Pruning: if remaining nodes are not enough to reach N
        if (M - 1 - idx) >= (N - count):
            res = backtrack(idx + 1, count, a_mask, b_mask, g_mask, chosen)
            if res is not None:
                return res
        
        memo[state] = False
        return None

    result = backtrack(0, 0, 0, 0, 0, [])
    
    if result:
        print("YES")
        for node in result:
            print(f"{node[0]} {node[1]} {node[2]}")
    else:
        print("NO")

solve()
