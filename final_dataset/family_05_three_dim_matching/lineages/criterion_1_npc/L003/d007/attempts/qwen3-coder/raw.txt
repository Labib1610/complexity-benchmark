from itertools import combinations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    if t == 0:
        if k == 0:
            print("YES")
        else:
            print("NO")
        return
    
    modules = []
    idx = 2
    for i in range(t):
        x = int(data[idx])
        y = int(data[idx+1])
        z = int(data[idx+2])
        modules.append((x, y, z))
        idx += 3
    
    target = (1 << k) - 1  # 2^k - 1
    
    # Try all combinations of k modules
    for combo in combinations(range(t), k):
        freq_sum = 0
        amp_sum = 0
        phase_sum = 0
        
        selected_modules = []
        for i in combo:
            x, y, z = modules[i]
            freq_sum += 1 << (x - 1)
            amp_sum += 1 << (y - 1)
            phase_sum += 1 << (z - 1)
            selected_modules.append((x, y, z))
        
        if freq_sum == target and amp_sum == target and phase_sum == target:
            print("YES")
            for mod in selected_modules:
                print(mod[0], mod[1], mod[2])
            return
    
    print("NO")

solve()
