import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        modules.append((x, y, z))
        idx += 3
        
    target = (1 << k) - 1
    
    # The condition sum(2^{v-1}) = 2^k - 1 for v in {1, ..., k}
    # means that for each parameter, the set of values {v_1, ..., v_k}
    # must be exactly {1, 2, ..., k}.
    # This is because 2^0 + 2^1 + ... + 2^{k-1} is the unique binary 
    # representation of 2^k - 1.
    
    # We need to select k modules (x_i, y_i, z_i) such that:
    # {x_1, ..., x_k} = {1, ..., k}
    # {y_1, ..., y_k} = {1, ..., k}
    # {z_1, ..., z_k} = {1, ..., k}
    
    # This is equivalent to finding a transversal in a 3D matching problem
    # or a specific case of exact cover. Since k is small (up to 22),
    # but t can be up to k^3, we use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    res = []
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for i in range(start_idx, t):
            x, y, z = modules[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = True
                used_y[y] = True
                used_z[z] = True
                res.append(modules[i])
                
                if backtrack(i + 1, count + 1):
                    return True
                
                res.pop()
                used_x[x] = False
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for m in res:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

solve()
