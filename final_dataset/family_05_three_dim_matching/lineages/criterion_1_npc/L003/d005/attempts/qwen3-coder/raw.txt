```python
from itertools import combinations

def hamming_distance(codeword1, codeword2):
    return sum(1 for a, b in zip(codeword1, codeword2) if a != b)

def solve():
    k, t = map(int, input().split())
    
    if t == 0:
        print("NO")
        return
    
    codewords = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        codewords.append((x, y, z))
    
    # Try all combinations of k codewords
    for combo in combinations(range(t), k):
        selected = [codewords[i] for i in combo]
        
        # Check if all pairs have Hamming distance 3
        valid = True
        for i in range(k):
            for j in range(i + 1, k):
                if hamming_distance(selected[i], selected[j]) != 3:
                    valid = False
                    break
            if not valid:
                break
        
        if valid:
            print("YES")
            for codeword in selected:
                print(codeword[0], codeword[1], codeword[2])
            return
    
    print("NO")

solve()
```
