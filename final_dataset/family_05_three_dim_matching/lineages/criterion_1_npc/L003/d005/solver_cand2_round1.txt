```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    codewords = []
    idx = 2
    for _ in range(t):
        codewords.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    # Hamming distance 3 means all 3 positions must be different.
    # Precompute adjacency list for the graph where nodes are codewords
    # and edges exist if Hamming distance is 3.
    adj = [[] for _ in range(t)]
    for i in range(t):
        for j in range(i + 1, t):
            c1 = codewords[i]
            c2 = codewords[j]
            if c1[0] != c2[0] and c1[1] != c2[1] and c1[2] != c2[2]:
                adj[i].append(j)
                adj[j].append(i)
                
    # We need a clique of size k. Since k is small (up to 22),
    # we can use backtracking with pruning.
    
    def get_clique(current_clique, candidates):
        if len(current_clique) == k:
            return current_clique
        
        if not candidates:
            return None
        
        # Heuristic: pick candidate with most connections to other candidates
        for i in range(len(candidates)):
            v = candidates[i]
            new_candidates = [c for c in candidates[i+1:] if c in adj[v]]
            
            res = get_clique(current_clique + [v], new_candidates)
            if res:
                return res
        return None

    # Optimization: sort candidates by degree
    initial_candidates = sorted(range(t), key=lambda x: len(adj[x]), reverse=True)
    result = get_clique([], initial_candidates)
    
    if result:
        print("YES")
        for idx in result:
            print(f"{codewords[idx][0]} {codewords[idx][1]} {codewords[idx][2]}")
    else:
        print("NO")

solve()
```
