```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        configs.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem.
    # We have 3*k constraints:
    # 1..k: Actuator i used
    # k+1..2k: Scanner j used
    # 2k+1..3k: Processor l used
    
    # Since k <= 22, we use backtracking with pruning.
    # Represent constraints as bitmasks or sets.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    result = []
    
    def backtrack(idx):
        if len(result) == k:
            return True
        
        if idx >= t:
            return False
        
        # Try including configs[idx]
        x, y, z = configs[idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append(configs[idx])
            if backtrack(idx + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try skipping configs[idx]
        # Optimization: if remaining configs are not enough to fill k, return False
        if (t - (idx + 1)) < (k - len(result)):
            return False
            
        if backtrack(idx + 1):
            return True
            
        return False

    # Sort configs to potentially find a solution faster
    if backtrack(0):
        print("YES")
        for row in result:
            print(f"{row[0]} {row[1]} {row[2]}")
    else:
        print("NO")

solve()
```
