We can model this as a 3D matching problem, but since k ≤ 22, we can use a backtracking approach with pruning.

We’ll store the valid triples in a list. Then we’ll try to assign each left pillar exactly one triple, ensuring no right pillar or keystone is reused.

Here’s the Python solution:

```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    triples = []
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        triples.append((x-1, y-1, z-1))  # 0-indexed
    
    # Group triples by left pillar
    by_left = [[] for _ in range(k)]
    for idx, (x, y, z) in enumerate(triples):
        by_left[x].append(idx)
    
    used_right = [False] * k
    used_keystone = [False] * k
    chosen = [None] * k  # chosen[left] = triple index
    
    # Sort each left's triples to try more constrained first (optional)
    # We'll just use as is.
    
    def dfs(left_idx):
        if left_idx == k:
            return True
        for tidx in by_left[left_idx]:
            x, y, z = triples[tidx]
            if not used_right[y] and not used_keystone[z]:
                used_right[y] = True
                used_keystone[z] = True
                chosen[left_idx] = tidx
                if dfs(left_idx + 1):
                    return True
                used_right[y] = False
                used_keystone[z] = False
        return False
    
    if dfs(0):
        out = ["YES"]
        for tidx in chosen:
            x, y, z = triples[tidx]
            out.append(f"{x+1} {y+1} {z+1}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
```
