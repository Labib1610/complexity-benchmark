```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # configurations[job][machine] = list of priorities
    adj = [[[] for _ in range(k + 1)] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x][y].append(z)
        idx += 3
        
    # We need to select k pairs (x, y) such that each x, y, and z is unique.
    # This is equivalent to finding a perfect matching in a 3D bipartite-like structure.
    # Since k <= 22, we can use backtracking with pruning.
    
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    result = []
    
    def backtrack(job_idx):
        if job_idx > k:
            return True
        
        for y in range(1, k + 1):
            if not used_y[y]:
                for z in adj[job_idx][y]:
                    if not used_z[z]:
                        used_y[y] = True
                        used_z[z] = True
                        result.append((job_idx, y, z))
                        
                        if backtrack(job_idx + 1):
                            return True
                        
                        result.pop()
                        used_z[z] = False
                        used_y[y] = False
        return False

    if backtrack(1):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
```
