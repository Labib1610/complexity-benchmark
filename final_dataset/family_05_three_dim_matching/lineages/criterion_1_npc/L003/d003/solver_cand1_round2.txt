import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triplets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triplets.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem.
    # We have 3*k constraints:
    # 1. Each left pillar (0..k-1) used once.
    # 2. Each right pillar (0..k-1) used once.
    # 3. Each keystone (0..k-1) used once.
    # Total constraints: 3*k.
    
    # Since k <= 22, we can use backtracking with bitmasks.
    # State: (index of left pillar to satisfy, used_right_mask, used_keystone_mask)
    
    adj = [[] for _ in range(k)]
    for x, y, z in triplets:
        adj[x].append((y, z))
        
    solution = []
    
    def backtrack(x, used_y, used_z):
        if x == k:
            return True
        
        for y, z in adj[x]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                solution.append((x + 1, y + 1, z + 1))
                if backtrack(x + 1, used_y | (1 << y), used_z | (1 << z)):
                    return True
                solution.pop()
        return False

    if backtrack(0, 0, 0):
        print("YES")
        for s in solution:
            print(f"{s[0]} {s[1]} {s[2]}")
    else:
        print("NO")

solve()
