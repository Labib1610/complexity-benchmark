```python
import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    blueprints = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        blueprints.append((x, y, z))
        idx += 3
        
    # The condition is: count(distinct x) + count(distinct y) + count(distinct z) >= 3k
    # Since each count is at most k, the only way to reach 3k is if:
    # count(distinct x) = k, count(distinct y) = k, count(distinct z) = k.
    # This means the set of k blueprints must contain every x from 1 to k,
    # every y from 1 to k, and every z from 1 to k.
    
    # This is equivalent to finding a transversal in a 3D grid, or more simply,
    # a set of k points such that all x, y, z coordinates are distinct.
    # This is a variation of the exact cover problem or finding a Latin transversal.
    
    # Given k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    result = []
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for i in range(start_idx, t):
            x, y, z = blueprints[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = used_y[y] = used_z[z] = True
                result.append(blueprints[i])
                
                if backtrack(i + 1, count + 1):
                    return True
                
                result.pop()
                used_x[x] = used_y[y] = used_z[z] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for b in result:
            print(f"{b[0]} {b[1]} {b[2]}")
    else:
        print("NO")

solve()
```
