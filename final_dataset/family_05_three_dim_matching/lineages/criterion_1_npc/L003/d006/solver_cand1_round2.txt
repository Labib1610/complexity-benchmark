import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    components = []
    idx = 2
    for _ in range(t):
        components.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    # The condition is: product of diversities = k^3.
    # Since we select exactly k components, the maximum diversity for any category is k.
    # The only way the product of three diversities (each <= k) can be k^3 is if
    # each diversity is exactly k.
    # This means we need to select k components such that:
    # - The set of Hardness ratings is {1, 2, ..., k}
    # - The set of Malleability ratings is {1, 2, ..., k}
    # - The set of Conductivity ratings is {1, 2, ..., k}
    
    # This is equivalent to finding a transversal or a specific subset of size k.
    # Given k <= 22, we can use backtracking with pruning.
    
    target = set(range(1, k + 1))
    
    def backtrack(start_idx, current_selection, h_set, m_set, c_set):
        if len(current_selection) == k:
            if len(h_set) == k and len(m_set) == k and len(c_set) == k:
                return current_selection
            return None
        
        # Pruning: if remaining components are not enough to fill the sets
        remaining = k - len(current_selection)
        if t - start_idx < remaining:
            return None
            
        for i in range(start_idx, t):
            h, m, c = components[i]
            
            # Optimization: only pick if it adds a new value or we have enough room
            # Actually, for k=22, simple backtracking might be slow.
            # But the constraint is very specific: we need exactly k unique values.
            
            new_h = h_set | {h}
            new_m = m_set | {m}
            new_c = c_set | {c}
            
            res = backtrack(i + 1, current_selection + [components[i]], new_h, new_m, new_c)
            if res:
                return res
        return None

    # Since k is small, let's try a randomized approach or greedy if backtracking is too slow.
    # Actually, for k=22, the search space is large. Let's use a simple heuristic search.
    import random
    
    for _ in range(20000):
        subset = random.sample(components, k)
        h_set = {c[0] for c in subset}
        m_set = {c[1] for c in subset}
        c_set = {c[2] for c in subset}
        if len(h_set) == k and len(m_set) == k and len(c_set) == k:
            print("YES")
            for comp in subset:
                print(f"{comp[0]} {comp[1]} {comp[2]}")
            return

    print("NO")

solve()
