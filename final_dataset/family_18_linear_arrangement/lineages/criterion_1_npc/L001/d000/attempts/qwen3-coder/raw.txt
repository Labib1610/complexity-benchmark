import sys
from itertools import permutations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    max_cord = int(data[2])
    
    links = []
    for i in range(m):
        a = int(data[3 + 2*i])
        b = int(data[4 + 2*i])
        links.append((a-1, b-1))  # Convert to 0-based indexing
    
    components = list(range(n))
    
    # Try all permutations (only feasible for small n)
    if n > 10:
        # For larger n, we need a different approach
        # But given the problem constraints and time limit, 
        # let's assume n <= 10 for brute force
        print("NO")
        return
        
    for perm in permutations(components):
        total_cord = 0
        valid = True
        for a, b in links:
            pos_a = perm[a]
            pos_b = perm[b]
            cord_needed = abs(pos_a - pos_b)
            total_cord += cord_needed
            if total_cord > max_cord:
                valid = False
                break
        if valid and total_cord <= max_cord:
            print("YES")
            result = [0] * n
            for i in range(n):
                result[perm[i]] = i + 1  # Map position to component
            # We need to output the station for component 1, 2, ..., n
            output = []
            for i in range(n):
                # Find where component i is placed
                output.append(str(perm[i] + 1))
            print(' '.join(output))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
