from collections import defaultdict, deque

    idx += 1

    parent = [-1] * (n+1)
    children = [[] for _ in range(n+1)]
    
    # BFS to build rooted tree with node 1 as root
    visited = [False] * (n+1)
    queue = deque()

        u = queue.popleft()

            if not visited[v]:
                visited[v] = True
                parent[v] = u
                children[u].append(v)
                queue.append(v)

    # A greedy approach: place nodes close to their parents
    
    position = [-1] * (n+1)
    assigned = [False] * (n+1)
    
    # Try all possible assignments using backtracking with pruning
    def calculate_cost(pos):
        total = 0
        for u, v in edges:
            if pos[u] != -1 and pos[v] != -1:
                total += abs(pos[u] - pos[v])
        return total

    # One good heuristic is to do a BFS traversal and assign positions accordingly

        order.append(node)
        for child in children[node]:
            dfs_order(child)

    # Assign positions 1 to n in the order obtained
    for i in range(n):
        position[order[i]] = i + 1
    
    cost = calculate_cost(position)
    
    if cost <= B:

            result.append(str(position[i]))
        print("YES")

    # But with n up to 10^5, full permutation is impossible
    
    # Let's try a more sophisticated approach
    # For trees, minimum linear arrangement can be approximated
    
    # Another idea: centroid decomposition or heavy-light decomposition
    # But again, might be overkill
    
    # Let's think differently. What if we model this properly?
    # It's essentially finding a linear arrangement of a tree that minimizes sum of edge lengths
    
    # There's a known algorithm for this:
    # Repeatedly remove leaves and place them optimally
    
    # But let's implement a simpler version that works well in practice
    
    # Try a greedy placement: start with root at middle position
    position = [-1] * (n+1)
    assigned = [False] * (n+1)
    
    # Place root at center
    position[1] = (n+1)//2
    assigned[1] = True
    
    # Queue for BFS placement
    queue = deque([1])
    placed = 1

        pos_u = position[u]
        
        # Try to place children near the parent
        for v in children[u]:
            if not assigned[v]:
                # Find next available position closest to pos_u

                        dist = abs(p - pos_u)
                        if dist < min_dist:
                            min_dist = dist
                            best_pos = p
                if best_pos != -1:
                    position[v] = best_pos
                    assigned[v] = True
                    placed += 1
                    queue.append(v)
    
    cost = calculate_cost(position)
    
    if cost <= B:
        result = []
        for i in range(1, n+1):
            result.append(str(position[i]))

    order_list = []

        if len(children[node]) >= 1:
            inorder(children[node][0])
        order_list.append(node)
        for i in range(1, len(children[node])):
            inorder(children[node][i])

    cost = 0

        result = []
