import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # Compute subtree sizes
    size = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    
    stack = [(1, 0, 0)]
    while stack:
        u, p, state = stack.pop()
        if state == 0:
            parent[u] = p
            order.append(u)
            stack.append((u, p, 1))
            for v in adj[u]:
                if v != p:
                    stack.append((v, u, 0))
        else:
            sz = 1
            for v in adj[u]:
                if v != p:
                    sz += size[v]
            size[u] = sz
    
    # DP: dp[u][k] = True if subtree rooted at u can be placed in k slots
    # We use a set of achievable sizes for efficiency
    # dp[u] = set of sizes achievable by subtree u
    # For each child v, we convolve dp[u] with dp[v]
    
    # We'll use a different approach: for each node, compute the set of achievable sizes
    # using bitset-like operations (Python sets)
    
    dp = [set() for _ in range(n+1)]
    
    # Process in reverse order (bottom-up)
    for u in reversed(order):
        # Start with size 1 (just the node itself)
        current = {1}
        for v in adj[u]:
            if v == parent[u]:
                continue
            # Convolve current with dp[v]
            new_current = set()
            for a in current:
                for b in dp[v]:
                    new_current.add(a + b)
            current = new_current
        dp[u] = current
    
    # Check if size n is achievable (which it should be for a tree)
    # Actually, for a tree, dp[1] should contain n
    # But we need to check if there's a valid assignment with cost <= B
    
    # The minimum cost is achieved by a DFS-like ordering
    # We need to find if there's an assignment with cost <= B
    
    # Let's compute the minimum possible cost
    # The minimum cost is achieved by placing the tree in a "compact" manner
    # Specifically, we can use a greedy approach: place root at some position,
    # then recursively place children in contiguous blocks
    
    # The minimum cost for a tree is:
    # For each edge (u, v) where v is child of u, the contribution is size[v]
    # So total minimum cost = sum of sizes of all nodes except root
    # Wait, let's verify: if we place root at position p, and children in contiguous blocks,
    # the edge from root to child v contributes size[v] (the block size)
    # So total cost = sum of subtree sizes of all children of all nodes
    # = sum of (size[child] for each edge)
    # = sum of (size[v] for each edge where v is child)
    # = sum of (size[v] for v != root)
    
    # Actually, let me reconsider. If we place root at position p, and children in contiguous
    # blocks to the left and right, the cost is:
    # For each child v, the edge (root, v) contributes size[v] (distance from root to block)
    # Then within each child's block, recursively...
    
    # The minimum cost is achieved by placing the tree in DFS order
    # Let's compute this minimum cost
    
    min_cost = 0
    for u in range(2, n+1):
        min_cost += size[u]
    
    if min_cost > B:
        print("NO")
        return
    
    # If min_cost <= B, we need to find an assignment
    # We can use a greedy approach: place nodes in DFS order
    # This achieves the minimum cost
    
    # Construct the assignment
    assignment = [0] * (n + 1)
    current_pos = 1
    
    # DFS to assign positions
    stack = [(1, 0)]
    while stack:
        u, state = stack.pop()
        if state == 0:
            assignment[u] = current_pos
            current_pos += 1
            stack.append((u, 1))
            # Push children in reverse order so they're processed in order
            children = []
            for v in adj[u]:
                if v != parent[u]:
                    children.append(v)
            for v in reversed(children):
                stack.append((v, 0))
    
    # Verify the cost
    total_cost = 0
    for u in range(1, n+1):
        for v in adj[u]:
            if v > u:  # Count each edge once
                total_cost += abs(assignment[u] - assignment[v])
    
    if total_cost <= B:
        print("YES")
        print(' '.join(map(str, assignment[1:])))
    else:
        print("NO")

solve()
