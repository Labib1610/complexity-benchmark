We can model this as a **2-SAT** problem.  

Each machine has two possible modes: `0` or `1`.  
Each interlock forbids exactly one combination of modes for its two machines.  

For example:  
- If an interlock has `(a, b)`, it forbids `(a=0, b=0)`.  
- If it has `(-a, b)`, it forbids `(a=1, b=0)`.  

We can convert each forbidden combination into two clauses in 2-SAT form.  

Here’s the implementation:  

```python
import sys
input = sys.stdin.readline

n, m = map(int, input().split())
N = 2 * n  # each variable x has two nodes: x (false) and x+n (true)

g = [[] for _ in range(N)]
gr = [[] for _ in range(N)]

def add_imp(u, v):
    g[u].append(v)
    gr[v].append(u)

# For each interlock, add implications
for _ in range(m):
    c1, c2 = map(int, input().split())
    # Convert to variable indices
    # For positive c: variable = c-1, true means mode 0
    # For negative c: variable = -c-1, true means mode 1
    # We'll use: x_i = True means mode 0 for machine i+1
    # So for positive c: literal is (c-1) true
    # For negative c: literal is (-c-1) false

    def lit(c):
        var = abs(c) - 1
        if c > 0:
            return var  # true
        else:
            return var + n  # false

    a = lit(c1)
    b = lit(c2)

    # Forbid both true: (not a) or (not b)
    # not a => a+n if a < n else a-n
    na = a + n if a < n else a - n
    nb = b + n if b < n else b - n

    add_imp(a, nb)
    add_imp(b, na)

# Kosaraju SCC
visited = [False] * N
order = []

def dfs1(u):
    visited[u] = True
    for v in g[u]:
        if not visited[v]:
            dfs1(v)
    order.append(u)

for i in range(N):
    if not visited[i]:
        dfs1(i)

comp = [-1] * N
def dfs2(u, cid):
    comp[u] = cid
    for v in gr[u]:
        if comp[v] == -1:
            dfs2(v, cid)

cid = 0
for u in reversed(order):
    if comp[u] == -1:
        dfs2(u, cid)
        cid += 1

# Check satisfiability
for i in range(n):
    if comp[i] == comp[i + n]:
        print("NO")
        sys.exit()

# Assign values
assignment = [0] * n
for i in range(n):
    # comp[i] > comp[i+n] means i is true
    assignment[i] = 0 if comp[i] > comp[i + n] else 1

print("YES")
print(" ".join(map(str, assignment)))
```
