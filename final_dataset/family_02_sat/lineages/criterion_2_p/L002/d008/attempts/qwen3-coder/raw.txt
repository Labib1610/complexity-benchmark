import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    # For each clause (a OR b), we'll build implications
    # If both a and b are False, then it's invalid
    # So we need: (x_i = 0) -> (x_j = 1) or (x_j = 0) -> (x_i = 1)
    # This translates to 2-SAT
    
    # Variables: x_1, ..., x_n correspond to machines 1..n being in mode 1 (True) or mode 0 (False)
    # For each constraint like (c1 c2), where c1 = +i means x_i should be True for rigidity,
    #                                  c1 = -i means x_i should be False for rigidity
    # We want NOT both linkages rigid => NOT(rigid1 AND rigid2) => NOT(c1) OR NOT(c2)
    # Which gives us the clause: (NOT c1 OR NOT c2)
    # But SAT solvers usually work with clauses of the form (a OR b)
    # So we negate: NOT(NOT c1 OR NOT c2) = (c1 AND c2)
    # Actually let's think differently:
    # We have that for each interlock, it's unsafe if both linkages are rigid.
    # Let's say var i represents machine i being in mode 1 (True).
    # Then machine i in mode 0 is represented by NOT var i.
    # If c > 0, linkage is rigid when var |c| is False (mode 0)
    # If c < 0, linkage is rigid when var |c| is True (mode 1)
    # So for interlock with linkages c1, c2:
    #   rigid1 = (c1 > 0) ? NOT var(|c1|) : var(|c1|)
    #   rigid2 = (c2 > 0) ? NOT var(|c2|) : var(|c2|)
    # We want NOT(rigid1 AND rigid2) which is equivalent to (NOT rigid1 OR NOT rigid2)
    # So our clause is: (NOT rigid1 OR NOT rigid2)
    # Let's compute what this means:
    # NOT rigid1 = NOT((c1 > 0) ? NOT var(|c1|) : var(|c1|)) = (c1 > 0) ? var(|c1|) : NOT var(|c1|)
    # Similarly for NOT rigid2
    
    # Build implication graph for 2-SAT
    # Nodes: 0..2*n-1 where [0..n-1] represent variables (xi=True), [n..2n-1] represent negations (xi=False)
    adj = [[] for _ in range(2 * n)]
    rev_adj = [[] for _ in range(2 * n)]
    
    def node_var(i):
        return i - 1  # 1-based to 0-based index for variable xi
    
    def node_not_var(i):
        return i - 1 + n  # 1-based to 0-based index for NOT xi
    
    # For each clause (NOT rigid1 OR NOT rigid2):
    for _ in range(m):
        c1, c2 = map(int, sys.stdin.readline().split())
        
        # Compute NOT rigid1 and NOT rigid2 as literals
        # Literal: positive means variable is True, negative means variable is False
        # But in our graph: 0..n-1 are xi=True, n..2n-1 are xi=False
        
        # NOT rigid1:
        if c1 > 0:
            # rigid1 = NOT x_{|c1|}
            # NOT rigid1 = x_{|c1|}
            lit1 = node_var(c1)  # x_{c1} is True
        else:
            # rigid1 = x_{|c1|}
            # NOT rigid1 = NOT x_{|c1|}
            lit1 = node_not_var(-c1)  # x_{|c1|} is False
            
        # NOT rigid2:
        if c2 > 0:
            # rigid2 = NOT x_{|c2|}
            # NOT rigid2 = x_{|c2|}
            lit2 = node_var(c2)
        else:
            # rigid2 = x_{|c2|}
            # NOT rigid2 = NOT x_{|c2|}
            lit2 = node_not_var(-c2)
            
        # Add implications: if NOT rigid1 is False, then NOT rigid2 must be True
        # That is: if (rigid1 is True), then (NOT rigid2 is True)
        # In terms of literals: if NOT( NOT rigid1 ), then (NOT rigid2)
        # Which is: if (rigid1), then (NOT rigid2)
        # But we already have NOT rigid1 and NOT rigid2 as lit1 and lit2
        # The clause (NOT rigid1 OR NOT rigid2) is equivalent to:
        # (rigid1 -> NOT rigid2) AND (rigid2 -> NOT rigid1)
        # Since rigid1 = NOT lit1_in_positive_form, etc., it gets complex.
        # Let's reframe:
        # We have clause (a OR b). In implication graph:
        # (NOT a -> b) and (NOT b -> a)
        
        # Let's define:
        # a = NOT rigid1
        # b = NOT rigid2
        # Clause is (a OR b) which implies edges (NOT a -> b) and (NOT b -> a)
        
        not_lit1 = lit1 ^ n  # Flip between true/false nodes
        not_lit2 = lit2 ^ n
        
        adj[not_lit1].append(lit2)
        adj[not_lit2].append(lit1)
        rev_adj[lit2].append(not_lit1)
        rev_adj[lit1].append(not_lit2)
    
    # Now run SCC (Strongly Connected Components) to solve 2-SAT
    order = []
    visited = [False] * (2 * n)
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Reverse graph traversal
    visited = [False] * (2 * n)
    components = [-1] * (2 * n)
    comp_count = 0
    
    def dfs2(u):
        visited[u] = True
        components[u] = comp_count
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
                
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            comp_count += 1
            
    # Check if any variable and its negation are in the same component
    for i in range(n):
        if components[i] == components[i + n]:
            print("NO")
            return
            
    # Assign values based on component numbers
    assignment = [False] * n
    comp_order = list(range(comp_count))
    comp_assigned = [False] * comp_count
    comp_value = [False] * comp_count
    
    # Process components in reverse topological order
    for i in range(2 * n):
        c = components[i]
        if not comp_assigned[c]:
            comp_assigned[c] = True
            # If this component corresponds to a variable node and hasn't been assigned yet
            pass  # We'll assign based on the order they get processed
            
    # Actually, simpler approach: process components in topological order of DAG of SCCs
    # But since we're using Kosaraju, the components are in reverse topological order in "components"
    # Wait, no. The first DFS gives finishing times. Second DFS processes in reverse order.
    # So components are numbered in the order of completion of second DFS.
    # To get topological order of condensation, we process components from highest number down.
    
    # Assign truth values: if a variable's positive node is in a later component than its negative node, set to True
    for i in range(n):
        comp_true = components[i]
        comp_false = components[i + n]
        if comp_true > comp_false:
            assignment[i] = True
        elif comp_true < comp_false:
            assignment[i] = False
        else:
            # This case shouldn't happen due to check above
            pass
            
    print("YES")
    print(' '.join(str(int(b)) for b in assignment))

threading.Thread(target=main).start()
