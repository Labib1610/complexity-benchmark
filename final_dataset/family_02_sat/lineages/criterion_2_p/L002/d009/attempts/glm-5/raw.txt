def tarjan_scc(N, adj_start, adj_to):
    disc = [-1] * N
    low = [0] * N
    comp = [-1] * N
    on_stack = bytearray(N)
    scc_stack = []
    comp_id = 0
    index = 0
    # iterative
    for root in range(N):
        if disc[root] != -1:
            continue
        work = [(root, adj_start[root])]
        disc[root] = low[root] = index; index += 1
        scc_stack.append(root)
        on_stack[root] = 1
        while work:
            v, ptr = work[-1]
            if ptr < adj_start[v+1]:
                work[-1] = (v, ptr+1)
                w = adj_to[ptr]
                if disc[w] == -1:
                    disc[w] = low[w] = index; index += 1
                    scc_stack.append(w)
                    on_stack[w] = 1
                    work.append((w, adj_start[w]))
                elif on_stack[w]:
                    if disc[w] < low[v]:
                        low[v] = disc[w]
            else:
                work.pop()
                if low[v] == disc[v]:
                    while True:
                        w = scc_stack.pop()
                        on_stack[w] = 0
                        comp[w] = comp_id
                        if w == v:
                            break
                    comp_id += 1
                if work:
                    u = work[-1][0]
                    if low[v] < low[u]:
                        low[u] = low[v]
    return comp, comp_id
