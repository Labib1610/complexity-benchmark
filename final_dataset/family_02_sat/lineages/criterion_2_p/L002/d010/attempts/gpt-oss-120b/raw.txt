```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = n * 2
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def var(i):  # i from 1..n, returns index of xi true
        return (i - 1) * 2

    def neg(x):
        return x ^ 1

    for _ in range(m):
        c1 = int(next(it))
        c2 = int(next(it))

        # literal for channel: v = 1 corresponds to literal being true
        # if c>0: v = s_i => literal is xi
        # if c<0: v = 1 - s_i => literal is ¬xi
        if c1 > 0:
            lit1 = var(c1)          # xi
        else:
            lit1 = neg(var(-c1))    # ¬xi

        if c2 > 0:
            lit2 = var(c2)
        else:
            lit2 = neg(var(-c2))

        # clause (lit1 OR lit2) => implications (¬lit1 -> lit2) and (¬lit2 -> lit1)
        g[neg(lit1)].append(lit2)
        g[neg(lit2)].append(lit1)
        rg[lit2].append(neg(lit1))
        rg[lit1].append(neg(lit2))

    visited = [False] * N
    order = []

    def dfs(v):
        visited[v] = True
        for to in g[v]:
            if not visited[to]:
                dfs(to)
        order.append(v)

    for v in range(N):
        if not visited[v]:
            dfs(v)

    comp = [-1] * N

    def rdfs(v, k):
        comp[v] = k
        for to in rg[v]:
            if comp[to] == -1:
                rdfs(to, k)

    k = 0
    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, k)
            k += 1

    assignment = [0] * n
    for i in range(1, n + 1):
        if comp[var(i)] == comp[neg(var(i))]:
            print("NO")
            return
        assignment[i - 1] = 1 if comp[var(i)] > comp[neg(var(i))] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
