```python
import sys

sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = 2 * n
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def var_node(idx: int, val: int) -> int:
        """return node index for x_idx == val (val 0 or 1)"""
        return 2 * (idx - 1) + val

    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        a = abs(l1)
        b = abs(l2)
        # literal L: x == target
        L1 = var_node(a, 1 if l1 > 0 else 0)
        L2 = var_node(b, 1 if l2 > 0 else 0)
        notL1 = L1 ^ 1
        notL2 = L2 ^ 1
        # add implications
        g[notL1].append(L2)
        g[notL2].append(L1)
        rg[L2].append(notL1)
        rg[L1].append(notL2)

    visited = [False] * N
    order = []

    def dfs(v: int):
        stack = [(v, 0)]
        while stack:
            v, state = stack.pop()
            if state == 0:
                if visited[v]:
                    continue
                visited[v] = True
                stack.append((v, 1))
                for to in g[v]:
                    if not visited[to]:
                        stack.append((to, 0))
            else:
                order.append(v)

    for v in range(N):
        if not visited[v]:
            dfs(v)

    comp = [-1] * N
    cid = 0

    def rdfs(v: int, label: int):
        stack = [v]
        while stack:
            v = stack.pop()
            if comp[v] != -1:
                continue
            comp[v] = label
            for to in rg[v]:
                if comp[to] == -1:
                    stack.append(to)

    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, cid)
            cid += 1

    assignment = [0] * n
    for i in range(1, n + 1):
        if comp[2 * (i - 1)] == comp[2 * (i - 1) + 1]:
            print("NO")
            return
        # variable is true if its SCC comes later (higher topological order)
        assignment[i - 1] = 1 if comp[2 * (i - 1) + 1] > comp[2 * (i - 1)] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
