import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    # Build implication graph
    # Variable x: 2*x = x is True (ON), 2*x+1 = x is False (OFF)
    # We use 0-indexed switches 0..n-1
    
    # For 2-SAT, we need to determine the number of variables
    # Each switch is a variable with two states
    
    # Collect edges
    edges = []
    
    # For each interlock (l1, l2):
    # V(|l1|)*sgn(l1) + V(|l2|)*sgn(l2) >= 0
    # V(x) = 1 if ON, -1 if OFF
    # sgn(l) = 1 if l>0, -1 if l<0
    # 
    # Let's denote: a = sgn(l1), b = sgn(l2)
    # V1*a + V2*b >= 0
    # V1, V2 in {1, -1}
    # 
    # Possible (V1, V2) pairs: (1,1), (1,-1), (-1,1), (-1,-1)
    # a*V1 + b*V2 >= 0
    # 
    # The forbidden case is when a*V1 + b*V2 < 0, i.e., a*V1 + b*V2 = -2
    # This happens when a*V1 = -1 and b*V2 = -1
    # i.e., V1 = -a, V2 = -b
    # 
    # So the forbidden assignment is: V1 = -sgn(l1), V2 = -sgn(l2)
    # V1 = 1 means ON, V1 = -1 means OFF
    # V1 = -sgn(l1): if sgn(l1)=1, V1=-1 (OFF); if sgn(l1)=-1, V1=1 (ON)
    # 
    # In terms of boolean: ON=True, OFF=False
    # V1 = -sgn(l1) means:
    #   if sgn(l1)=1: V1=-1, so OFF, so x1=False
    #   if sgn(l1)=-1: V1=1, so ON, so x1=True
    # So forbidden: x1 = (sgn(l1) == -1), x2 = (sgn(l2) == -1)
    # 
    # We need NOT(x1 = a AND x2 = b) where a = (sgn(l1)==-1), b = (sgn(l2)==-1)
    # This is equivalent to: (not x1=a) OR (not x2=b)
    # i.e., (x1 != a) OR (x2 != b)
    # 
    # In 2-SAT: (x1 != a) OR (x2 != b)
    # If x1 == a, then x2 != b
    # If x2 == b, then x1 != a
    # 
    # x1 == a means: if a is True, x1=True; if a is False, x1=False
    # x1 != a means: if a is True, x1=False; if a is False, x1=True
    # 
    # Let's define literal: for variable x, literal can be x (True) or ~x (False)
    # x1 == a: if a=True, literal is x1; if a=False, literal is ~x1
    # x1 != a: if a=True, literal is ~x1; if a=False, literal is x1
    # 
    # So the clause is: (x1 != a) OR (x2 != b)
    # Implications: (x1 == a) -> (x2 != b), (x2 == b) -> (x1 != a)
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        a = 1 if l1 > 0 else -1
        b = 1 if l2 > 0 else -1
        
        # Forbidden: V1 = -a, V2 = -b
        # V1 = -a: if a=1, V1=-1 (OFF, x1=False); if a=-1, V1=1 (ON, x1=True)
        # So forbidden x1 = (a == -1)
        # Similarly forbidden x2 = (b == -1)
        
        # Clause: NOT(x1 == (a==-1) AND x2 == (b==-1))
        # = (x1 != (a==-1)) OR (x2 != (b==-1))
        
        # Let's use standard 2-SAT encoding
        # For variable i (0-indexed): 2*i = i is True, 2*i+1 = i is False
        # literal "x_i is True" -> node 2*i
        # literal "x_i is False" -> node 2*i+1
        # ~literal: if literal is 2*i, then ~literal is 2*i+1, and vice versa
        
        idx1 = abs(l1) - 1
        idx2 = abs(l2) - 1
        
        # forbidden: x1 = (a == -1), x2 = (b == -1)
        # If a == -1: forbidden x1 = True, so literal is 2*idx1 (x1 is True)
        # If a == 1: forbidden x1 = False, so literal is 2*idx1+1 (x1 is False)
        lit1 = 2 * idx1 if a == -1 else 2 * idx1 + 1
        lit2 = 2 * idx2 if b == -1 else 2 * idx2 + 1
        
        # Clause: (~lit1) OR (~lit2)
        # Implications: lit1 -> ~lit2, lit2 -> ~lit1
        not_lit1 = lit1 ^ 1
        not_lit2 = lit2 ^ 1
        
        edges.append((lit1, not_lit2))
        edges.append((lit2, not_lit1))
    
    # Build adjacency list
    num_nodes = 2 * n
    adj = [[] for _ in range(num_nodes)]
    radj = [[] for _ in range(num_nodes)]
    
    for u, v in edges:
        adj[u].append(v)
        radj[v].append(u)
    
    # Kosaraju's algorithm
    visited = [False] * num_nodes
    order = []
    
    # Use iterative DFS to avoid recursion limit
    for i in range(num_nodes):
        if not visited[i]:
            stack = [(i, 0)]
            while stack:
                node, idx = stack[-1]
                if idx == 0:
                    visited[node] = True
                if idx < len(adj[node]):
                    neighbor = adj[node][idx]
                    stack[-1] = (node, idx + 1)
                    if not visited[neighbor]:
                        stack.append((neighbor, 0))
                else:
                    order.append(node)
                    stack.pop()
    
    visited = [False] * num_nodes
    comp = [-1] * num_nodes
    comp_id = 0
    
    for i in reversed(order):
        if not visited[i]:
            # DFS on reverse graph
            stack = [i]
            visited[i] = True
            while stack:
                node = stack.pop()
                comp[node] = comp_id
                for neighbor in radj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        stack.append(neighbor)
            comp_id += 1
    
    # Check satisfiability
    for i in range(n):
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return
    
    # Assign values
    assignment = [0] * n
    for i in range(n):
        # If comp[2*i] > comp[2*i+1], then x_i = True (ON)
        # If comp[2*i] < comp[2*i+1], then x_i = False (OFF)
        if comp[2 * i] > comp[2 * i + 1]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(" ".join(map(str, assignment)))

solve()
