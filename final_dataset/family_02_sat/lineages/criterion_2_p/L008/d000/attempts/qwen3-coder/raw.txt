import sys
sys.setrecursionlimit(1000000)

def solve():
    n, m = map(int, input().split())
    
    # For each constraint (a OR b), we create implications
    # (!a => b) and (!b => a)
    # Variables: 1..n for mode 1, -(1..n) for mode 0
    # We'll use 1-based indexing, so var i means relay i in mode 1
    # and var -i means relay i in mode 0
    
    adj = [[] for _ in range(2 * n + 1)]  # adjacency list for implications
    adj_rev = [[] for _ in range(2 * n + 1)]  # reverse graph
    
    # Helper function to get index in 1..2n range
    def var_index(x):
        if x > 0:
            return x
        else:
            return n + (-x)
    
    # Helper function to get variable from index
    def from_index(idx):
        if idx <= n:
            return idx
        else:
            return -(idx - n)
    
    # For each clause (a OR b), add edges (!a -> b) and (!b -> a)
    for _ in range(m):
        a, b = map(int, input().split())
        # !a -> b
        not_a_idx = var_index(-a)
        b_idx = var_index(b)
        adj[not_a_idx].append(b_idx)
        adj_rev[b_idx].append(not_a_idx)
        
        # !b -> a
        not_b_idx = var_index(-b)
        a_idx = var_index(a)
        adj[not_b_idx].append(a_idx)
        adj_rev[a_idx].append(not_b_idx)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n + 1)
    order = []
    
    def dfs1(u):
        if visited[u]:
            return
        visited[u] = True
        for v in adj[u]:
            dfs1(v)
        order.append(u)
    
    def dfs2(u, component):
        if visited[u]:
            return
        visited[u] = True
        component.append(u)
        for v in adj_rev[u]:
            dfs2(v, component)
    
    # First DFS pass
    for i in range(1, 2 * n + 1):
        dfs1(i)
    
    # Second DFS pass on reversed graph
    visited = [False] * (2 * n + 1)
    components = []
    while order:
        u = order.pop()
        if not visited[u]:
            component = []
            dfs2(u, component)
            components.append(component)
    
    # Check if satisfiable
    assignment = [0] * (n + 1)  # 1-based indexing
    
    # Map each node to its component
    comp_id = [0] * (2 * n + 1)
    for idx, component in enumerate(components):
        for node in component:
            comp_id[node] = idx
    
    # Check for contradiction: x and !x in same component
    for i in range(1, n + 1):
        pos_idx = var_index(i)      # i in mode 1
        neg_idx = var_index(-i)     # i in mode 0
        if comp_id[pos_idx] == comp_id[neg_idx]:
            print("NO")
            return
    
    # Assign values based on topological order of SCCs
    # Components are in reverse topological order due to Kosaraju
    for component in components:
        for node in component:
            var = from_index(node)
            if var > 0:
                if assignment[var] == 0:  # Not assigned yet
                    assignment[var] = 1
            else:  # var < 0
                relay = -var
                if assignment[relay] == 0:  # Not assigned yet
                    assignment[relay] = 0
    
    print("YES")
    print(*assignment[1:])  # Skip index 0

solve()
