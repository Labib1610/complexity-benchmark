def tarjan_scc(N, adj_start, adj_list):
    # returns comp array, number of components
    index = [-1]*N
    low = [0]*N
    on_stack = bytearray(N)
    comp = [-1]*N
    stack = []  # SCC stack
    call_stack = []  # for iterative DFS: (node, edge pointer)
    counter = 0
    ncomp = 0
    
    for root in range(N):
        if index[root] != -1:
            continue
        # iterative DFS
        call_stack.append((root, adj_start[root]))
        index[root] = low[root] = counter
        counter += 1
        stack.append(root)
        on_stack[root] = 1
        while call_stack:
            u, ptr = call_stack[-1]
            if ptr < adj_start[u+1]:
                # advance pointer
                call_stack[-1] = (u, ptr+1)
                v = adj_list[ptr]
                if index[v] == -1:
                    index[v] = low[v] = counter
                    counter += 1
                    stack.append(v)
                    on_stack[v] = 1
                    call_stack.append((v, adj_start[v]))

                    if on_stack[v] and index[v] < low[u]:
                        low[u] = index[v]

                if call_stack:
                    p = call_stack[-1][0]
                    if low[u] < low[p]:
                        low[p] = low[u]

                    while True:
                        w = stack.pop()
                        on_stack[w] = 0
                        comp[w] = ncomp
                        if w == u:

                    ncomp += 1
