We can model each artwork as a boolean variable (Spotlight = true, Ambient = false).  
Each mandate is a clause of the form `(l1 OR l2)`, which is a 2-SAT problem.  

We’ll build an implication graph and use SCC (Tarjan’s or Kosaraju’s) to check satisfiability and extract a solution.

```python
import sys
sys.setrecursionlimit(10**6)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    
    # 2-SAT: variable i (1..n) -> node 2*i for true, 2*i+1 for false
    # We'll use 0-based indexing: node 2*i = true, 2*i+1 = false
    N = 2 * n
    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]
    
    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)
    
    def node(var):
        # var > 0: true, var < 0: false
        if var > 0:
            return 2 * (var - 1)      # true node
        else:
            return 2 * (-var - 1) + 1 # false node
    
    def neg_node(var):
        if var > 0:
            return 2 * (var - 1) + 1
        else:
            return 2 * (-var - 1)
    
    for _ in range(m):
        a, b = map(int, input().split())
        # clause (a OR b) => (!a -> b) and (!b -> a)
        add_imp(neg_node(a), node(b))
        add_imp(neg_node(b), node(a))
    
    # Kosaraju's SCC
    visited = [False] * N
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(N):
        if not visited[i]:
            dfs1(i)
    
    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)
    
    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1
    
    # Check satisfiability
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return
    
    # Build assignment: choose value based on SCC order
    assignment = [False] * n
    for i in range(n):
        # In Kosaraju, comp id order gives topological order of SCCs in reverse
        # If true node has higher comp id, set true
        assignment[i] = comp[2*i] > comp[2*i+1]
    
    print("YES")
    print(" ".join('1' if x else '0' for x in assignment))

if __name__ == "__main__":
    solve()
```
