import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    head = [-1] * (n + 1)
    to = [0] * (2 * m)
    weight = [0] * (2 * m)
    nxt = [0] * (2 * m)
    edge_cnt = 0
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        u = abs(l1)
        v = abs(l2)
        w = 1 if (l1 < 0) == (l2 < 0) else -1
        to[edge_cnt] = v
        weight[edge_cnt] = w
        nxt[edge_cnt] = head[u]
        head[u] = edge_cnt
        edge_cnt += 1
        to[edge_cnt] = u
        weight[edge_cnt] = w
        nxt[edge_cnt] = head[v]
        head[v] = edge_cnt
        edge_cnt += 1
    
    color = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    for start in range(1, n + 1):
        if not visited[start]:
            visited[start] = True
            color[start] = 1
            stack = [start]
            while stack:
                u = stack.pop()
                e = head[u]
                while e != -1:
                    v = to[e]
                    w = weight[e]
                    if not visited[v]:
                        visited[v] = True
                        color[v] = color[u] * w
                        stack.append(v)
                    e = nxt[e]
    
    for u in range(1, n + 1):
        e = head[u]
        while e != -1:
            v = to[e]
            w = weight[e]
            if color[u] * color[v] != w:
                print("NO")
                return
            e = nxt[e]
    
    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

solve()
