import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    
    # For each qubit i, we track:
    # forced[i] = 0 means must be in S (state 0)
    # forced[i] = 1 means must NOT be in S (state 1)
    # forced[i] = -1 means undetermined
    forced = [-1] * (n + 1)
    
    # We also need to handle implications: if P is satisfied, then N must be excluded.
    # But the key insight: a route faults iff P ⊆ S and S ∩ N = ∅.
    # To avoid fault: for each route, either some element of P is NOT in S, or some element of N IS in S.
    
    # Strategy: process routes and use forced values to propagate constraints.
    # If a route has P with all elements forced to 0 (in S), then at least one element of N must be forced to 1 (not in S).
    # If a route has N with all elements forced to 0 (in S), then at least one element of P must be forced to 1 (not in S).
    
    # Actually, let's think differently. This is a 2-SAT-like problem.
    # For each route, the fault condition is: (all p in P are in S) AND (all n in N are not in S)
    # To avoid: NOT(all p in P in S) OR NOT(all n in N not in S)
    # = (some p in P not in S) OR (some n in N in S)
    
    # Let's use a simpler approach: greedy with propagation.
    # We maintain for each route whether it's already "satisfied" (cannot fault).
    
    routes = []
    # For each qubit, list of routes where it's in P (forcing it to 0 would help satisfy)
    # and routes where it's in N (forcing it to 1 would help satisfy)
    
    # Actually, let's think about it as: we want to assign values to avoid all faults.
    # A route is "satisfied" if either:
    #   - some element of P is assigned 1 (not in S), OR
    #   - some element of N is assigned 0 (in S)
    
    # If a route has P = empty, then we need some element of N to be 0 (in S).
    # If a route has N = empty, then we need some element of P to be 1 (not in S).
    # If both P and N are empty... that can't happen since l1, l2 are non-zero.
    
    # Let's collect routes and check for immediate contradictions.
    
    from collections import defaultdict
    
    # routes_p[i] = list of route indices where qubit i is in P
    # routes_n[i] = list of route indices where qubit i is in N
    routes_p = defaultdict(list)
    routes_n = defaultdict(list)
    
    route_P = []
    route_N = []
    
    for idx in range(m):
        parts = input().split()
        l1 = int(parts[0])
        l2 = int(parts[1])
        
        P = set()
        N = set()
        
        if l1 > 0:
            P.add(l1)
        else:
            N.add(-l1)
        
        if l2 > 0:
            P.add(l2)
        else:
            N.add(-l2)
        
        route_P.append(P)
        route_N.append(N)
        
        for p in P:
            routes_p[p].append(idx)
        for nn in N:
            routes_n[nn].append(idx)
    
    # Check for routes where P and N intersect (same qubit in both)
    # If qubit i is in both P and N of a route, then:
    #   - if i in S (state 0), then N has an element in S, so route is satisfied
    #   - if i not in S (state 1), then P has an element not in S, so route is satisfied
    # So such routes are always satisfied. We can ignore them.
    
    # Check for routes where P is empty: need some element of N to be 0 (in S)
    # Check for routes where N is empty: need some element of P to be 1 (not in S)
    
    # A route with P empty and N = {i} means i must be in S (forced 0).
    # A route with N empty and P = {i} means i must not be in S (forced 1).
    
    # Let's use a queue-based propagation.
    # satisfied[idx] = True if route idx is already satisfied
    satisfied = [False] * m
    
    # unsatisfied_count_p[idx] = number of elements in P not yet forced to 1
    # unsatisfied_count_n[idx] = number of elements in N not yet forced to 0
    # Actually, let's track differently.
    
    # For route idx:
    # - It's satisfied if some p in P is forced 1, or some n in N is forced 0.
    # - If all p in P are forced 0 and all n in N are forced 1, then it's a fault (contradiction).
    
    # Let's track:
    # p_forced_1[idx] = True if some element of P is forced to 1
    # n_forced_0[idx] = True if some element of N is forced to 0
    # p_count[idx] = number of elements in P
    # n_count[idx] = number of elements in N
    # p_forced_0_count[idx] = number of elements of P forced to 0
    # n_forced_1_count[idx] = number of elements of N forced to 1
    
    p_count = [0] * m
    n_count = [0] * m
    p_forced_0_count = [0] * m
    n_forced_1_count = [0] * m
    p_forced_1 = [False] * m
    n_forced_0 = [False] * m
    
    for idx in range(m):
        P = route_P[idx]
        N = route_N[idx]
        p_count[idx] = len(P)
        n_count[idx] = len(N)
        
        # Check if route is already satisfied (P and N intersect)
        if P & N:
            satisfied[idx] = True
    
    # Now, process forced assignments.
    # A route becomes "critical" if:
    #   - p_forced_0_count == p_count (all P forced to 0) and not p_forced_1
    #     => need some n in N to be forced to 0
    #   - n_forced_1_count == n_count (all N forced to 1) and not n_forced_0
    #     => need some p in P to be forced to 1
    
    # If a route is critical and N is empty (n_count == 0), then contradiction.
    # If a route is critical and P is empty (p_count == 0), then contradiction.
    
    # If a route is critical with all P forced to 0, and N has exactly one undetermined element,
    # we must force that element to 0.
    # Similarly for the other case.
    
    # Let's use a different approach: since this is essentially 2-SAT, but with clauses
    # that can have more than 2 literals, it's actually more general.
    # However, each route has at most 2 elements in P and at most 2 in N.
    
    # Actually, since |l1|, |l2| <= n and there are only 2 values, P has at most 2 elements
    # and N has at most 2 elements.
    
    # So each route is: (p1 in S or p2 in S ... ) -> (n1 not in S or n2 not in S or ...)
    # Wait, the fault is: all P in S AND all N not in S.
    # To avoid: NOT(all P in S) OR NOT(all N not in S)
    # = (some P not in S) OR (some N in S)
    
    # This is a clause: (NOT p1_in_S) OR (NOT p2_in_S) OR ... OR (n1_in_S) OR (n2_in_S) OR ...
    # where p_in_S means "p is in S" (state 0), and n_in_S means "n is in S" (state 0).
    
    # So the clause is: (p1=1) OR (p2=1) OR ... OR (n1=0) OR (n2=0) OR ...
    # where 1 means not in S, 0 means in S.
    
    # This is a general SAT problem, but with special structure.
    # Since each clause has at most 4 literals (2 from P, 2 from N), it's not directly 2-SAT.
    
    # However, let's think about it more carefully.
    # If P has 2 elements and N has 2 elements, the clause is:
    # (p1=1) OR (p2=1) OR (n1=0) OR (n2=0)
    # This is a 4-SAT clause, which is NP-hard in general.
    
    # But wait, the problem says l1 and l2 are the two values. So P and N together have at most 2 elements.
    # If l1 > 0 and l2 > 0: P = {l1, l2}, N = {} (clause: l1=1 OR l2=1)
    # If l1 > 0 and l2 < 0: P = {l1}, N = {|l2|} (clause: l1=1 OR |l2|=0)
    # If l1 < 0 and l2 > 0: P = {l2}, N = {|l1|} (clause: l2=1 OR |l1|=0)
    # If l1 < 0 and l2 < 0: P = {}, N = {|l1|, |l2|} (clause: |l1|=0 OR |l2|=0)
    
    # So each clause has at most 2 literals! This is 2-SAT!
    
    # Let's verify:
    # - Route "3 -5": P={3}, N={5}. Clause: (3=1) OR (5=0). 
    #   Meaning: either 3 is not in S, or 5 is in S.
    # - Route "-2 -2": P={}, N={2}. Clause: (2=0).
    #   Meaning: 2 must be in S.
    # - Route "1 2": P={1,2}, N={}. Clause: (1=1) OR (2=1).
    #   Meaning: either 1 is not in S, or 2 is not in S.
    
    # Great, so this is 2-SAT! Let's solve it using 2-SAT.
    
    # Variables: x_i = 1 if qubit i is in S (state 0), x_i = 0 if not in S (state 1).
    # Wait, let me redefine to match 2-SAT convention.
    # Let x_i = True if qubit i is in S (state 0), False if not in S (state 1).
    
    # Clause for route with P and N:
    # (some p in P is not in S) OR (some n in N is in S)
    # = (x_p1 = False) OR (x_p2 = False) OR (x_n1 = True) OR (x_n2 = True)
    
    # But since P and N together have at most 2 elements, let's enumerate:
    
    # Case 1: P = {a, b}, N = {} (both positive, a != b)
    #   Clause: (x_a = False) OR (x_b = False)
    #   = NOT x_a OR NOT x_b
    #   Implications: x_a -> NOT x_b, x_b -> NOT x_a
    
    # Case 2: P = {a}, N = {} (both positive, a == b, i.e., l1 = l2 = a)
    #   Clause: (x_a = False) OR (x_a = False) = NOT x_a
    #   So x_a must be False (qubit a is not in S, state 1)
    
    # Case 3: P = {a}, N = {b} (one positive, one negative, a != b)
    #   Clause: (x_a = False) OR (x_b = True)
    #   = NOT x_a OR x_b
    #   Implications: x_a -> x_b, NOT x_b -> NOT x_a
    
    # Case 4: P = {a}, N = {a} (one positive, one negative, same value, i.e., l1 = a, l2 = -a)
    #   Clause: (x_a = False) OR (x_a = True) = True (always satisfied)
    #   No constraints.
    
    # Case 5: P = {}, N = {a, b} (both negative, a != b)
    #   Clause: (x_a = True) OR (x_b = True)
    #   = x_a OR x_b
    #   Implications: NOT x_a -> x_b, NOT x_b -> x_a
    
    # Case 6: P = {}, N = {a} (both negative, same value, i.e., l1 = l2 = -a)
    #   Clause: (x_a = True) OR (x_a = True) = x_a
    #   So x_a must be True (qubit a is in S, state 0)
    
    # So we have a standard 2-SAT problem. Let's solve it.
    
    # Build implication graph
    # Node 2*i represents x_i = True (qubit i in S, state 0)
    # Node 2*i+1 represents x_i = False (qubit i not in S, state 1)
    
    # For 2-SAT, we use Tarjan's SCC algorithm.
    
    # Number of nodes: 2*n (for n variables)
    # Node indexing: 0 to 2*n-1
    # For variable i (1-indexed): 
    #   - node 2*(i-1) represents x_i = True
    #   - node 2*(i-1)+1 represents x_i = False
    
    # Implication: a -> b means add edge from a to b
    
    adj = [[] for _ in range(2 * n)]
    
    def add_implication(a, b):
        # a -> b
        adj[a].append(b)
    
    def add_clause_or(a_val, a_is_true, b_val, b_is_true):
        # (a_is_true ? x_a : NOT x_a) OR (b_is_true ? x_b : NOT x_b)
        # = NOT(a_is_true ? x_a : NOT x_a) -> (b_is_true ? x_b : NOT x_b)
        # = NOT(b_is_true ? x_b : NOT x_b) -> (a_is_true ? x_a : NOT x_a)
        a_node = 2 * (a_val - 1) + (0 if a_is_true else 1)
        b_node = 2 * (b_val - 1) + (0 if b_is_true else 1)
        a_not = 2 * (a_val - 1) + (1 if a_is_true else 0)
        b_not = 2 * (b_val - 1) + (1 if b_is_true else 0)
        add_implication(a_not, b_node)
        add_implication(b_not, a_node)
    
    def add_unit_clause(val, is_true):
        # x_val = is_true
        # This means NOT x_val -> x_val (if is_true) or x_val -> NOT x_val (if not is_true)
        node = 2 * (val - 1) + (0 if is_true else 1)
        not_node = 2 * (val - 1) + (1 if is_true else 0)
        add_implication(not_node, node)
    
    for idx in range(m):
        P = route_P[idx]
        N = route_N[idx]
        
        if len(P) == 2 and len(N) == 0:
            # Case 1: NOT x_a OR NOT x_b
            a, b = list(P)
            add_clause_or(a, False, b, False)
        elif len(P) == 1 and len(N) == 0:
            # Case 2: NOT x_a (unit clause)
            a = list(P)[0]
            add_unit_clause(a, False)
        elif len(P) == 1 and len(N) == 1:
            a = list(P)[0]
            b = list(N)[0]
            if a == b:
                # Case 4: always satisfied
                pass
            else:
                # Case 3: NOT x_a OR x_b
                add_clause_or(a, False, b, True)
        elif len(P) == 0 and len(N) == 2:
            # Case 5: x_a OR x_b
            a, b = list(N)
            add_clause_or(a, True, b, True)
        elif len(P) == 0 and len(N) == 1:
            # Case 6: x_a (unit clause)
            a = list(N)[0]
            add_unit_clause(a, True)
    
    # Tarjan's SCC
    index_counter = [0]
    stack = []
    on_stack = [False] * (2 * n)
    index = [-1] * (2 * n)
    lowlink = [0] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = [0]
    
    import sys
    sys.setrecursionlimit(300000)
    
    # Use iterative Tarjan to avoid recursion limit issues
    def tarjan_iterative():
        for start in range(2 * n):
            if index[start] != -1:
                continue
            # Iterative DFS
            call_stack = [(start, 0)]
            while call_stack:
                v, ei = call_stack[-1]
                if ei == 0:
                    index[v] = index_counter[0]
                    lowlink[v] = index_counter[0]
                    index_counter[0] += 1
                    stack.append(v)
                    on_stack[v] = True
                
                if ei < len(adj[v]):
                    w = adj[v][ei]
                    call_stack[-1] = (v, ei + 1)
                    if index[w] == -1:
                        call_stack.append((w, 0))
                    elif on_stack[w]:
                        lowlink[v] = min(lowlink[v], index[w])
                else:
                    if lowlink[v] == index[v]:
                        while True:
                            w = stack.pop()
                            on_stack[w] = False
                            scc_id[w] = scc_count[0]
                            if w == v:
                                break
                        scc_count[0] += 1
                    
                    call_stack.pop()
                    if call_stack:
                        parent = call_stack[-1][0]
                        lowlink[parent] = min(lowlink[parent], lowlink[v])
    
    tarjan_iterative()
    
    # Check satisfiability
    for i in range(1, n + 1):
        true_node = 2 * (i - 1)
        false_node = 2 * (i - 1) + 1
        if scc_id[true_node] == scc_id[false_node]:
            print("NO")
            return
    
    # Assign values: x_i = True if scc_id[true_node] > scc_id[false_node]
    # (Tarjan's SCCs are in reverse topological order, so higher scc_id means earlier in topo order)
    result = [0] * (n + 1)
    for i in range(1, n + 1):
        true_node = 2 * (i - 1)
        false_node = 2 * (i - 1) + 1
        if scc_id[true_node] > scc_id[false_node]:
            result[i] = 0  # in S, state 0
        else:
            result[i] = 1  # not in S, state 1
    
    print("YES")
    print(" ".join(map(str, result[1:])))

solve()
