We can model each dependency as a 2-SAT clause:  

- `1 2` means `(x1 OR x2)`  
- `-1 -2` means `(¬x1 OR ¬x2)`  

So the problem reduces to checking satisfiability of a 2-SAT formula with up to \(10^5\) variables and \(2 \times 10^5\) clauses.  

We can solve it with implication graph + SCC (Tarjan or Kosaraju).  

Here’s the implementation:  

```python
import sys
sys.setrecursionlimit(1 << 20)
input = sys.stdin.readline

n, m = map(int, input().split())
N = 2 * n  # each variable x has nodes: 2*x for false, 2*x+1 for true

def node(var):
    # var > 0: x_var true -> node 2*(var-1)+1
    # var < 0: x_var false -> node 2*(-var-1)
    if var > 0:
        return 2 * (var - 1) + 1
    else:
        return 2 * (-var - 1)

g = [[] for _ in range(N)]
gr = [[] for _ in range(N)]

for _ in range(m):
    a, b = map(int, input().split())
    u = node(a)
    v = node(b)
    # (a OR b) => (¬a -> b) and (¬b -> a)
    g[u ^ 1].append(v)
    g[v ^ 1].append(u)
    gr[v].append(u ^ 1)
    gr[u].append(v ^ 1)

# Kosaraju SCC
visited = [False] * N
order = []

def dfs1(u):
    visited[u] = True
    for w in g[u]:
        if not visited[w]:
            dfs1(w)
    order.append(u)

for i in range(N):
    if not visited[i]:
        dfs1(i)

comp = [-1] * N

def dfs2(u, cid):
    comp[u] = cid
    for w in gr[u]:
        if comp[w] == -1:
            dfs2(w, cid)

cid = 0
for u in reversed(order):
    if comp[u] == -1:
        dfs2(u, cid)
        cid += 1

# Check satisfiability
for i in range(n):
    if comp[2 * i] == comp[2 * i + 1]:
        print("NO")
        sys.exit()

print("YES")
res = []
for i in range(n):
    # If false node has higher comp id than true node, assign false (0)
    res.append('1' if comp[2 * i + 1] > comp[2 * i] else '0')
print(' '.join(res))
```
