import sys
sys.setrecursionlimit(1000000)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    clauses = []
    index = 2
    for _ in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        index += 2
        clauses.append((l1, l2))
    
    # Build implication graph for 2-SAT
    # Variables: 0 to 2*n-1
    # For switch i (1-based), var 2*(i-1) is x_i (0), var 2*(i-1)+1 is not x_i (1)
    # But in clauses, positive means x_i = 1 (ON), negative means x_i = 0 (OFF)
    
    # For variable x_i:
    # Node 2*(i-1) represents x_i (switch i is ON)
    # Node 2*(i-1)+1 represents not x_i (switch i is OFF)
    
    adj = [[] for _ in range(2 * n)]
    adj_rev = [[] for _ in range(2 * n)]
    
    # For each clause (l1 or l2), add implications
    # not l1 => l2  and  not l2 => l1
    for l1, l2 in clauses:
        # l1 is switch |l1|, sign indicates polarity
        # if l1 > 0: switch |l1| should be ON (x_|l1| = 1)
        # if l1 < 0: switch |l1| should be OFF (x_|l1| = 0)
        
        # Convert to variable indices
        if l1 > 0:
            var1 = 2 * (l1 - 1)     # x_l1
            not_var1 = var1 + 1     # not x_l1
        else:
            var1 = 2 * (-l1 - 1) + 1  # not x_|l1|
            not_var1 = var1 - 1       # x_|l1|
            
        if l2 > 0:
            var2 = 2 * (l2 - 1)     # x_l2
            not_var2 = var2 + 1     # not x_l2
        else:
            var2 = 2 * (-l2 - 1) + 1  # not x_|l2|
            not_var2 = var2 - 1       # x_|l2|
        
        # Add implications: not l1 => l2, not l2 => l1
        adj[not_var1].append(var2)
        adj[not_var2].append(var1)
        adj_rev[var2].append(not_var1)
        adj_rev[var1].append(not_var2)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Second DFS on transpose graph
    visited = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in adj_rev[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            scc_count += 1
    
    # Check if satisfiable
    for i in range(n):
        # Variable i corresponds to nodes 2*i and 2*i+1
        if scc_id[2*i] == scc_id[2*i+1]:
            print("NO")
            return
    
    # Build assignment
    assignment = [False] * n
    # Process SCCs in reverse topological order
    # We can do this by processing scc_id in order
    scc_order = [[] for _ in range(scc_count)]
    for i in range(2 * n):
        scc_order[scc_id[i]].append(i)
    
    assigned = [False] * n
    
    # Process SCCs in reverse order of finishing time in second DFS
    # But since we used the order from first DFS, we process sccs in order of their id
    for component in scc_order:
        for node in component:
            var_index = node // 2
            if not assigned[var_index]:
                # If node is 2*i, then x_i = False
                # If node is 2*i+1, then x_i = True
                if node % 2 == 0:
                    assignment[var_index] = False
                else:
                    assignment[var_index] = True
                assigned[var_index] = True
    
    print("YES")
    result = []
    for i in range(n):
        if assignment[i]:
            result.append("1")
        else:
            result.append("0")
    print(" ".join(result))

if __name__ == "__main__":
    main()
