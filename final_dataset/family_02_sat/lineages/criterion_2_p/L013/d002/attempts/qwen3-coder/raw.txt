import sys
from collections import defaultdict

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each rule (l1, l2), we need:
    # NOT ( (x[|l1|] != (l1>0)) AND (x[|l2|] != (l2>0)) )
    # Which is equivalent to:
    # (x[|l1|] == (l1>0)) OR (x[|l2|] == (l2>0))
    # This is a 2-SAT clause: (a OR b)
    # where a = (x[|l1|] == (l1>0)) and b = (x[|l2|] == (l2>0))
    
    # In 2-SAT terms:
    # variable i represents x[i] = 1
    # variable -i represents x[i] = 0
    # The clause (x[i] == 1 OR x[j] == 1) translates to (i OR j)
    # The clause (x[i] == 1 OR x[j] == 0) translates to (i OR -j)
    # The clause (x[i] == 0 OR x[j] == 1) translates to (-i OR j)
    # The clause (x[i] == 0 OR x[j] == 0) translates to (-i OR -j)
    
    # Given a literal l, var(l) = abs(l)
    # If l > 0: x[var(l)] == 1
    # If l < 0: x[var(l)] == 0
    
    # For each rule (l1, l2), we want:
    # (x[|l1|] == (l1>0)) OR (x[|l2|] == (l2>0))
    # This is equivalent to:
    # (NOT(x[|l1|] != (l1>0)) OR NOT(x[|l2|] != (l2>0)))
    # Which is:
    # NOT(x[|l1|] != (l1>0)) OR NOT(x[|l2|] != (l2>0))
    # Which means:
    # (x[|l1|] == (l1>0)) OR (x[|l2|] == (l2>0))
    
    # To convert to standard 2-SAT clause (a OR b):
    # If l1 > 0, then a = x[|l1|] = True  => literal is |l1|
    # If l1 < 0, then a = x[|l1|] = False => literal is -|l1|
    # Same for l2
    
    # So the clause is (l1 OR l2)
    
    # In implication graph:
    # (a OR b) is equivalent to (NOT a => b) and (NOT b => a)
    
    # Build implication graph
    # Nodes: 1..n for x[i]=True, -1..-n for x[i]=False
    
    graph = defaultdict(list)
    reverse_graph = defaultdict(list)
    
    for _ in range(m):
        l1, l2 = map(int, sys.stdin.readline().split())
        # Add clause (l1 OR l2)
        # This is equivalent to (NOT l1 => l2) and (NOT l2 => l1)
        # NOT l1 is -l1, NOT l2 is -l2
        graph[-l1].append(l2)
        graph[-l2].append(l1)
        reverse_graph[l2].append(-l1)
        reverse_graph[l1].append(-l2)
    
    # Now we do Kosaraju's algorithm to find strongly connected components
    visited = [False] * (2 * n + 1)
    order = []
    
    # Nodes are from -n to -1 and 1 to n
    node_list = list(range(-n, 0)) + list(range(1, n+1))
    
    def dfs1(u):
        visited[u + n] = True  # offset by n to handle negative indices
        for v in graph[u]:
            if not visited[v + n]:
                dfs1(v)
        order.append(u)
    
    for u in node_list:
        if not visited[u + n]:
            dfs1(u)
    
    # Second DFS on reverse graph
    visited = [False] * (2 * n + 1)
    scc = [0] * (2 * n + 1)
    scc_id = 0
    
    def dfs2(u):
        visited[u + n] = True
        scc[u + n] = scc_id
        for v in reverse_graph[u]:
            if not visited[v + n]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u + n]:
            scc_id += 1
            dfs2(u)
    
    # Check for contradiction: variable and its negation in same SCC
    for i in range(1, n+1):
        if scc[i + n] == scc[-i + n]:
            print("NO")
            return
    
    # Assign values based on topological order of SCCs
    # Variables in later SCCs (higher scc_id) are assigned True
    assignment = [0] * (n + 1)
    
    for i in range(1, n+1):
        if scc[i + n] > scc[-i + n]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(" ".join(map(str, assignment[1:])))

if __name__ == "__main__":
    main()
