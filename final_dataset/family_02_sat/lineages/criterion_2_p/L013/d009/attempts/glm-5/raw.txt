import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        u = abs(l1) - 1
        v = abs(l2) - 1
        sign1 = 1 if l1 > 0 else -1
        sign2 = 1 if l2 > 0 else -1
        
        # For each brace, we need total displacement != -2
        # Displacement from connection = sign * (2*s - 1)
        # Total = sign1*(2*s_u-1) + sign2*(2*s_v-1)
        # If total == -2, fracture.
        # We add constraint: NOT(total == -2)
        
        # Case analysis:
        # If signs are same (both +1 or both -1):
        #   total = sign * (2*s_u + 2*s_v - 2)
        #   total == -2 when sign*(2*s_u + 2*s_v - 2) == -2
        #   If sign == 1: 2*s_u + 2*s_v - 2 == -2 => s_u + s_v == 0 => s_u = s_v = 0
        #   If sign == -1: -(2*s_u + 2*s_v - 2) == -2 => s_u + s_v == 2 => s_u = s_v = 1
        #   So we forbid (s_u = s_v = 0) for sign=1, or (s_u = s_v = 1) for sign=-1
        #   Constraint: NOT(s_u = 0 AND s_v = 0) for sign=1
        #   Constraint: NOT(s_u = 1 AND s_v = 1) for sign=-1
        #   This is a clause: (s_u OR s_v) for sign=1, or (!s_u OR !s_v) for sign=-1
        
        # If signs are different:
        #   total = sign1*(2*s_u-1) + sign2*(2*s_v-1)
        #   Let's say sign1 = 1, sign2 = -1:
        #   total = (2*s_u - 1) - (2*s_v - 1) = 2*s_u - 2*s_v
        #   total == -2 when s_u = 0, s_v = 1
        #   Constraint: NOT(s_u = 0 AND s_v = 1)
        #   Clause: (s_u OR !s_v)
        #   If sign1 = -1, sign2 = 1:
        #   total = -(2*s_u - 1) + (2*s_v - 1) = -2*s_u + 2*s_v
        #   total == -2 when s_u = 1, s_v = 0
        #   Constraint: NOT(s_u = 1 AND s_v = 0)
        #   Clause: (!s_u OR s_v)
        
        # 2-SAT encoding:
        # Variable x_i is True if s_i = 1, False if s_i = 0
        # Clause (a OR b) means: add edges (!a -> b) and (!b -> a)
        
        if sign1 == sign2:
            if sign1 == 1:
                # Clause: (x_u OR x_v)
                adj[2*u + 1].append(2*v)     # !x_u -> x_v
                adj[2*v + 1].append(2*u)     # !x_v -> x_u
            else:
                # Clause: (!x_u OR !x_v)
                adj[2*u].append(2*v + 1)     # x_u -> !x_v
                adj[2*v].append(2*u + 1)     # x_v -> !x_u
        else:
            if sign1 == 1:
                # sign1 = 1, sign2 = -1
                # Clause: (x_u OR x_v) which is same as (!(!x_u) OR x_v)
                # Actually: forbid (s_u=0, s_v=1) means forbid (!x_u, x_v)
                # Clause: (x_u OR !x_v)
                adj[2*u + 1].append(2*v + 1) # !x_u -> !x_v
                adj[2*v].append(2*u)         # x_v -> x_u
            else:
                # sign1 = -1, sign2 = 1
                # Forbid (s_u=1, s_v=0) means forbid (x_u, !x_v)
                # Clause: (!x_u OR x_v)
                adj[2*u].append(2*v)         # x_u -> x_v
                adj[2*v + 1].append(2*u + 1) # !x_v -> !x_u
    
    # Kosaraju's algorithm for SCC
    order = []
    visited = [False] * (2 * n)
    
    def dfs1(v):
        stack = [(v, 0)]
        while stack:
            node, idx = stack[-1]
            if not visited[node]:
                visited[node] = True
            if idx < len(adj[node]):
                nei = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[nei]:
                    stack.append((nei, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    radj = [[] for _ in range(2 * n)]
    for u in range(2 * n):
        for v in adj[u]:
            radj[v].append(u)
    
    comp = [-1] * (2 * n)
    cid = 0
    
    def dfs2(v, cid):
        stack = [v]
        comp[v] = cid
        while stack:
            node = stack.pop()
            for nei in radj[node]:
                if comp[nei] == -1:
                    comp[nei] = cid
                    stack.append(nei)
    
    for v in reversed(order):
        if comp[v] == -1:
            dfs2(v, cid)
            cid += 1
    
    assignment = [0] * n
    possible = True
    
    for i in range(n):
        if comp[2*i] == comp[2*i + 1]:
            possible = False
            break
        assignment[i] = 1 if comp[2*i] > comp[2*i + 1] else 0
    
    if not possible:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, assignment)))

solve()
