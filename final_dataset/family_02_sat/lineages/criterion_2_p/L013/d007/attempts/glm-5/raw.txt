import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n + 2)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        def get_node(l):
            idx = abs(l)
            base = 1 if l < 0 else 0
            return 2 * idx + base
        
        u = get_node(l1)
        v = get_node(l2)
        
        adj[u ^ 1].append(v)
        adj[v ^ 1].append(u)
    
    visited = [False] * (2 * n + 2)
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, state = stack.pop()
            if state == 0:
                if visited[node]:
                    continue
                visited[node] = True
                stack.append((node, 1))
                for v in adj[node]:
                    if not visited[v]:
                        stack.append((v, 0))
            else:
                order.append(node)
    
    for i in range(2, 2 * n + 2):
        if not visited[i]:
            dfs1(i)
    
    visited = [False] * (2 * n + 2)
    comp_id = [-1] * (2 * n + 2)
    current_comp = 0
    
    def dfs2(u, cid):
        stack = [u]
        while stack:
            node = stack.pop()
            if visited[node]:
                continue
            visited[node] = True
            comp_id[node] = cid
            for v in adj[node]:
                if not visited[v]:
                    stack.append(v)
    
    for u in reversed(order):
        if not visited[u]:
            dfs2(u, current_comp)
            current_comp += 1
    
    for i in range(1, n + 1):
        if comp_id[2 * i] == comp_id[2 * i + 1]:
            print("NO")
            return
    
    print("YES")
    res = []
    for i in range(1, n + 1):
        if comp_id[2 * i] > comp_id[2 * i + 1]:
            res.append(1)
        else:
            res.append(0)
    print(' '.join(map(str, res)))

if __name__ == "__main__":
    solve()
