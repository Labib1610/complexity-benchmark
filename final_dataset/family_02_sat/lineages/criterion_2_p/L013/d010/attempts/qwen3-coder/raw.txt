import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each brace, we have the system:
    # x + t1 * y = 0
    # t2 * x + y = 0
    # This is equivalent to:
    # x = -t1 * y
    # y = -t2 * x
    # Substituting: y = -t2 * (-t1 * y) = t1 * t2 * y
    # So: y = t1 * t2 * y
    # If t1 * t2 != 1, then y = 0, and thus x = 0. So the brace is safe.
    # If t1 * t2 = 1, then there are non-trivial solutions, so the brace fractures.
    # So for each brace, we need t1 * t2 != 1.
    
    # Let's define the constraints.
    # For a brace with connections l1 and l2:
    # t1 = 1 - s[|l1|] if l1 > 0, else s[|l1|]
    # t2 = 1 - s[|l2|] if l2 > 0, else s[|l2|]
    # We want t1 * t2 != 1
    
    # Let's think in terms of boolean logic.
    # Let a = s[|l1|], b = s[|l2|]
    # If l1 > 0: t1 = 1 - a
    # If l1 < 0: t1 = a
    # If l2 > 0: t2 = 1 - b
    # If l2 < 0: t2 = b
    # t1 * t2 = (1 - a if l1 > 0 else a) * (1 - b if l2 > 0 else b)
    # We want this product != 1
    
    # Let's enumerate all 4 cases for a and b (00, 01, 10, 11) and see when t1*t2 = 1
    # Case 1: a=0, b=0
    #  t1 = (1-0)=1 if l1>0 else 0
    #  t2 = (1-0)=1 if l2>0 else 0
    #  t1*t2 = 1*1=1 if both l1>0 and l2>0
    #  t1*t2 = 1*0=0 if l1>0 and l2<0
    #  t1*t2 = 0*1=0 if l1<0 and l2>0
    #  t1*t2 = 0*0=0 if l1<0 and l2<0
    # So only forbidden if both l1>0 and l2>0
    
    # Case 2: a=0, b=1
    #  t1 = (1-0)=1 if l1>0 else 0
    #  t2 = (1-1)=0 if l2>0 else 1
    #  t1*t2 = 1*0=0 if l1>0 and l2>0
    #  t1*t2 = 1*1=1 if l1>0 and l2<0
    #  t1*t2 = 0*0=0 if l1<0 and l2>0
    #  t1*t2 = 0*1=0 if l1<0 and l2<0
    # So only forbidden if l1>0 and l2<0
    
    # Case 3: a=1, b=0
    #  t1 = (1-1)=0 if l1>0 else 1
    #  t2 = (1-0)=1 if l2>0 else 0
    #  t1*t2 = 0*1=0 if l1>0 and l2>0
    #  t1*t2 = 0*0=0 if l1>0 and l2<0
    #  t1*t2 = 1*1=1 if l1<0 and l2>0
    #  t1*t2 = 1*0=0 if l1<0 and l2<0
    # So only forbidden if l1<0 and l2>0
    
    # Case 4: a=1, b=1
    #  t1 = (1-1)=0 if l1>0 else 1
    #  t2 = (1-1)=0 if l2>0 else 1
    #  t1*t2 = 0*0=0 if l1>0 and l2>0
    #  t1*t2 = 0*1=0 if l1>0 and l2<0
    #  t1*t2 = 1*0=0 if l1<0 and l2>0
    #  t1*t2 = 1*1=1 if l1<0 and l2<0
    # So only forbidden if l1<0 and l2<0
    
    # Summary of forbidden assignments:
    # l1>0, l2>0: forbid (0,0)
    # l1>0, l2<0: forbid (0,1)
    # l1<0, l2>0: forbid (1,0)
    # l1<0, l2<0: forbid (1,1)
    
    # This is a 2-SAT problem.
    # For each variable s_i, we have two nodes: i (for s_i = 0) and i+n (for s_i = 1)
    # For each forbidden pair (a,b) on struts i,j:
    # If (s_i=0, s_j=0) is forbidden, add clause (s_i=1 OR s_j=1) => add edges (~i -> j+n, ~j -> i+n)
    # If (s_i=0, s_j=1) is forbidden, add clause (s_i=1 OR s_j=0) => add edges (~i -> ~j, j -> i+n)
    # If (s_i=1, s_j=0) is forbidden, add clause (s_i=0 OR s_j=1) => add edges (i -> j, ~j -> ~i)
    # If (s_i=1, s_j=1) is forbidden, add clause (s_i=0 OR s_j=0) => add edges (i -> ~j, j -> ~i)
    
    # Build implication graph
    adj = defaultdict(list)
    
    index = 2
    for i in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        index += 2
        
        i1 = abs(l1)
        i2 = abs(l2)
        
        # Convert to 0-based indexing
        i1 -= 1
        i2 -= 1
        
        s1 = 1 if l1 > 0 else 0  # 1 means direct coupling (1-s), 0 means inverted (s)
        s2 = 1 if l2 > 0 else 0
        
        # Determine forbidden assignment
        if s1 == 1 and s2 == 1:  # l1>0, l2>0: forbid (0,0)
            # ~i1 -> i2+n, ~i2 -> i1+n
            adj[i1].append(i2 + n)
            adj[i2].append(i1 + n)
        elif s1 == 1 and s2 == 0:  # l1>0, l2<0: forbid (0,1)
            # ~i1 -> ~i2, i2 -> i1+n
            adj[i1].append(i2)
            adj[i2 + n].append(i1 + n)
        elif s1 == 0 and s2 == 1:  # l1<0, l2>0: forbid (1,0)
            # i1 -> i2, ~i2 -> ~i1
            adj[i1 + n].append(i2)
            adj[i2].append(i1)
        else:  # l1<0, l2<0: forbid (1,1)
            # i1 -> ~i2, i2 -> ~i1
            adj[i1 + n].append(i2)
            adj[i2 + n].append(i1)
    
    # Now we need to solve 2-SAT
    # Kosaraju's algorithm for SCC
    def kosaraju():
        # First DFS to get finishing times
        visited = [False] * (2 * n)
        order = []
        
        def dfs1(u):
            visited[u] = True
            for v in adj[u]:
                if not visited[v]:
                    dfs1(v)
            order.append(u)
        
        for i in range(2 * n):
            if not visited[i]:
                dfs1(i)
        
        # Build reverse graph
        rev_adj = defaultdict(list)
        for u in adj:
            for v in adj[u]:
                rev_adj[v].append(u)
        
        # Second DFS in reverse order
        visited = [False] * (2 * n)
        scc = [-1] * (2 * n)
        scc_id = 0
        
        def dfs2(u):
            visited[u] = True
            scc[u] = scc_id
            for v in rev_adj[u]:
                if not visited[v]:
                    dfs2(v)
        
        while order:
            u = order.pop()
            if not visited[u]:
                dfs2(u)
                scc_id += 1
        
        return scc
    
    scc = kosaraju()
    
    # Check if satisfiable
    for i in range(n):
        if scc[i] == scc[i + n]:
            print("NO")
            return
    
    # Assign values
    # Variables in later SCCs are assigned first
    assignment = [-1] * n
    # Create a mapping from SCC id to nodes
    scc_nodes = defaultdict(list)
    for i in range(2 * n):
        scc_nodes[scc[i]].append(i)
    
    # Process SCCs in reverse topological order
    scc_order = list(range(scc_id))
    # We need the topological order of SCCs. Since we used finishing times,
    # the reverse of the finishing order of SCCs in the original graph gives us this.
    # But Kosaraju already gives SCC ids in a valid topological order (low to high).
    # So we process from high to low.
    
    for cid in reversed(range(scc_id)):
        for node in scc_nodes[cid]:
            var = node % n
            if assignment[var] == -1:
                if node < n:  # node represents s_var = 0
                    assignment[var] = 0
                else:  # node represents s_var = 1
                    assignment[var] = 1
    
    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    main()
