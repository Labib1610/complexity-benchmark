import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n + 2)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        # For each literal l:
        # l > 0: switch |l| should be 1. Violation: switch is 0. Node: 2*|l|
        # l < 0: switch |l| should be 0. Violation: switch is 1. Node: 2*|l|+1
        def get_node(l):
            idx = abs(l)
            if l > 0:
                return 2 * idx      # violation: s[idx] = 0
            else:
                return 2 * idx + 1  # violation: s[idx] = 1
        
        u = get_node(l1)
        v = get_node(l2)
        
        # Implication: NOT u -> v (if u is not violated, v must be violated)
        adj[u ^ 1].append(v)
        adj[v ^ 1].append(u)
    
    # 2-SAT using iterative Kosaraju's algorithm
    N = 2 * n + 2
    visited = [False] * N
    order = []
    
    # First pass: build finish order
    for start in range(2, N):
        if not visited[start]:
            stack = [(start, 0)]
            while stack:
                node, idx = stack[-1]
                if not visited[node]:
                    visited[node] = True
                if idx < len(adj[node]):
                    nei = adj[node][idx]
                    stack[-1] = (node, idx + 1)
                    if not visited[nei]:
                        stack.append((nei, 0))
                else:
                    stack.pop()
                    order.append(node)
    
    # Second pass: find SCCs
    radj = [[] for _ in range(N)]
    for u in range(N):
        for v in adj[u]:
            radj[v].append(u)
    
    comp = [-1] * N
    cid = 0
    
    for node in reversed(order):
        if comp[node] == -1:
            stack = [node]
            comp[node] = cid
            while stack:
                u = stack.pop()
                for v in radj[u]:
                    if comp[v] == -1:
                        comp[v] = cid
                        stack.append(v)
            cid += 1
    
    # Check for contradictions
    for i in range(1, n + 1):
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return
    
    # Build assignment
    assignment = [0] * (n + 1)
    for i in range(1, n + 1):
        # s[i] = 1 if comp[violation_0] > comp[violation_1]
        # violation_0 = 2*i (s[i]=0), violation_1 = 2*i+1 (s[i]=1)
        if comp[2 * i] > comp[2 * i + 1]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(' '.join(map(str, assignment[1:])))

if __name__ == "__main__":
    solve()
