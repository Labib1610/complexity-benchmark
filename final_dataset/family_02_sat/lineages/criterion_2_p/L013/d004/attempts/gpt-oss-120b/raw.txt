```python
import sys

sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    N = n * 2
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def lit(idx: int, val: int) -> int:
        """return node id for literal (x_idx == val)"""
        # false node = 2*idx, true node = 2*idx+1
        return 2 * idx + val

    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        i1, a = abs(l1) - 1, 1 if l1 > 0 else 0
        i2, b = abs(l2) - 1, 1 if l2 > 0 else 0

        u = lit(i1, a)
        v = lit(i2, b)

        nu = lit(i1, 1 - a)   # negation of u
        nv = lit(i2, 1 - b)   # negation of v

        # (!u -> v)
        g[nu].append(v)
        rg[v].append(nu)
        # (!v -> u)
        g[nv].append(u)
        rg[u].append(nv)

    visited = [False] * N
    order = []

    def dfs(v: int):
        stack = [(v, 0)]
        while stack:
            node, state = stack.pop()
            if state == 0:
                if visited[node]:
                    continue
                visited[node] = True
                stack.append((node, 1))
                for to in g[node]:
                    if not visited[to]:
                        stack.append((to, 0))
            else:
                order.append(node)

    for v in range(N):
        if not visited[v]:
            dfs(v)

    comp = [-1] * N
    cid = 0

    def rdfs(v: int, label: int):
        stack = [v]
        while stack:
            node = stack.pop()
            if comp[node] != -1:
                continue
            comp[node] = label
            for to in rg[node]:
                if comp[to] == -1:
                    stack.append(to)

    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, cid)
            cid += 1

    assignment = [0] * n
    for i in range(n):
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return
        # component with larger topological order gets false
        assignment[i] = 0 if comp[2 * i] > comp[2 * i + 1] else 1

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
