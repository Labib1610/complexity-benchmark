import sys
sys.setrecursionlimit(10**6)

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each constraint (a b), we create implications:
    # If NOT a, then b
    # If NOT b, then a
    # This translates to:
    # If a is false (opposite of what it says), then b must be true
    # If b is false (opposite of what it says), then a must be true
    
    # Variables: for variable x (1-based), we use index 2*x (true) and 2*x+1 (false)
    # But actually, let's think differently.
    # Let's say x means variable x is true (energized)
    # -x means variable x is false (isolated)
    # So variable x has two literals: x (positive) and -x (negative)
    
    # For constraint (a b):
    # If (NOT a) then b  =>  (NOT a) -> b  which is equivalent to a OR b
    # If (NOT b) then a  =>  (NOT b) -> a  which is equivalent to b OR a
    # So both are just a OR b
    
    # This is 2-SAT: each clause is (a OR b)
    # We can model this as implication graph:
    # (NOT a) -> b  and (NOT b) -> a
    
    # In 2-SAT, for literal x, we denote:
    # x*2 for positive literal (variable x is True)
    # x*2+1 for negative literal (variable x is False)
    # But our variables are 1-based, and can be negative.
    
    # Let's reindex:
    # For variable v (1-based), 
    # if v > 0: literal v corresponds to variable v being True
    # if v < 0: literal v corresponds to variable (-v) being False
    
    # So for variable x (1-based):
    # Literal +x: node 2*(x-1)     (x is True)
    # Literal -x: node 2*(x-1)+1   (x is False)
    
    # Given constraint (a b): (NOT a) => b  and (NOT b) => a
    # Nodes:
    # NOT a => b : node_of(NOT a) => node_of(b)
    # NOT b => a : node_of(NOT b) => node_of(a)
    
    # node_of(literal):
    def node(lit):
        # lit is a signed integer with |lit| <= n
        var_index = abs(lit)
        if lit > 0:
            # lit = +var_index, so node is "var_index is True"
            return 2 * (var_index - 1)
        else:
            # lit = -var_index, so node is "var_index is False"
            return 2 * (var_index - 1) + 1
    
    # NOT node:
    def neg_node(u):
        # u = 2*i or 2*i+1
        if u % 2 == 0:
            # u represents "i-th variable is True", so NOT u is "i-th variable is False"
            return u + 1
        else:
            # u represents "i-th variable is False", so NOT u is "i-th variable is True"
            return u - 1
    
    adj = [[] for _ in range(2 * n)]
    rev_adj = [[] for _ in range(2 * n)]
    
    idx = 2
    for i in range(m):
        a = int(data[idx])
        b = int(data[idx + 1])
        idx += 2
        
        # Clause: a OR b
        # Implications: NOT a => b, NOT b => a
        
        # Add edge: NOT a -> b
        adj[node(-a)].append(node(b))
        rev_adj[node(b)].append(node(-a))
        
        # Add edge: NOT b -> a
        adj[node(-b)].append(node(a))
        rev_adj[node(a)].append(node(-b))
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for u in range(2 * n):
        if not visited[u]:
            dfs1(u)
    
    visited = [False] * (2 * n)
    components = [-1] * (2 * n)
    comp_count = 0
    
    def dfs2(u):
        visited[u] = True
        components[u] = comp_count
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
    
    for u in reversed(order):
        if not visited[u]:
            dfs2(u)
            comp_count += 1
    
    # Check if satisfiable
    for i in range(n):
        node_true = 2 * i       # variable (i+1) is True
        node_false = 2 * i + 1  # variable (i+1) is False
        if components[node_true] == components[node_false]:
            print("NO")
            return
    
    # Extract solution
    assignment = [0] * n
    comp_order = list(range(comp_count))
    # Sort components in topological order (we can use the finish time from first DFS)
    # Actually, in Kosaraju, components are labeled in reverse topological order of DAG
    # So component with highest number comes first.
    
    # For each variable, if comp[node_true] > comp[node_false], assign True
    for i in range(n):
        node_true = 2 * i
        node_false = 2 * i + 1
        if components[node_true] > components[node_false]:
            assignment[i] = 1
        else:
            assignment[i] = 0
            
    print("YES")
    print(" ".join(map(str, assignment)))

solve()
