We can model this as a **2-SAT** problem.  
Each rule is a clause of two literals, and we need to find a satisfying assignment.

We’ll:
- Build an implication graph.
- Use Tarjan’s algorithm to find strongly connected components.
- If a variable and its negation are in the same SCC → unsatisfiable.
- Otherwise, assign values based on topological order of SCCs.

```python
import sys
sys.setrecursionlimit(1 << 20)

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # nodes: 0..n-1 for x_i, n..2n-1 for ~x_i

    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def idx(lit):
        # lit > 0: x_lit -> index lit-1
        # lit < 0: ~x_(-lit) -> index n + (-lit) - 1
        if lit > 0:
            return lit - 1
        else:
            return n + (-lit) - 1

    def neg_idx(i):
        return (i + n) % N

    for _ in range(m):
        a, b = map(int, input().split())
        ia = idx(a)
        ib = idx(b)
        # clause (a or b) => (~a -> b) and (~b -> a)
        g[neg_idx(ia)].append(ib)
        g[neg_idx(ib)].append(ia)
        gr[ib].append(neg_idx(ia))
        gr[ia].append(neg_idx(ib))

    # Kosaraju's algorithm
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    for i in range(n):
        if comp[i] == comp[i + n]:
            print("NO")
            return

    print("YES")
    # assignment: if comp[x] > comp[~x] then True else False
    ans = [0] * n
    for i in range(n):
        ans[i] = 1 if comp[i] > comp[i + n] else 0
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    main()
```
