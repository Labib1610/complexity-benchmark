import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n + 2)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        u = abs(l1)
        v = abs(l2)
        
        # l1: if positive, prefers Primary (1); if negative, prefers Secondary (0)
        # l2: if positive, prefers Primary (1); if negative, prefers Secondary (0)
        # Protocol is breached if BOTH preferences fail.
        # So we need: NOT (fail1 AND fail2) = (NOT fail1) OR (NOT fail2)
        # fail1 is true when l1's preference is not met.
        # If l1 > 0, prefers Primary. fail1 = (state[u] == 0). NOT fail1 = (state[u] == 1).
        # If l1 < 0, prefers Secondary. fail1 = (state[u] == 1). NOT fail1 = (state[u] == 0).
        # Same for l2.
        
        # We encode state[u] == 1 as node u, state[u] == 0 as node u+n.
        # Implication A -> B is edges (A, B) and (not B, not A).
        
        if l1 > 0:
            # NOT fail1 is state[u] == 1, represented by u
            a = u
        else:
            # NOT fail1 is state[u] == 0, represented by u+n
            a = u + n
            
        if l2 > 0:
            # NOT fail2 is state[v] == 1, represented by v
            b = v
        else:
            # NOT fail2 is state[v] == 0, represented by v+n
            b = v + n
            
        # We need (NOT fail1) OR (NOT fail2), which is a OR b.
        # This is equivalent to (not a -> b) and (not b -> a).
        # not a is a+n if a <= n, else a-n. Same for b.
        
        na = a + n if a <= n else a - n
        nb = b + n if b <= n else b - n
        
        adj[na].append(b)
        adj[nb].append(a)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n + 2)
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack[-1]
            if not visited[node]:
                visited[node] = True
            if idx < len(adj[node]):
                nei = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[nei]:
                    stack.append((nei, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(1, 2 * n + 2):
        if not visited[i]:
            dfs1(i)
    
    radj = [[] for _ in range(2 * n + 2)]
    for u in range(1, 2 * n + 2):
        for v in adj[u]:
            radj[v].append(u)
    
    comp_id = [0] * (2 * n + 2)
    cid = 0
    
    def dfs2(u, cid):
        stack = [u]
        while stack:
            node = stack.pop()
            if comp_id[node] == 0:
                comp_id[node] = cid
                for nei in radj[node]:
                    if comp_id[nei] == 0:
                        stack.append(nei)
    
    for u in reversed(order):
        if comp_id[u] == 0:
            cid += 1
            dfs2(u, cid)
    
    for i in range(1, n + 1):
        if comp_id[i] == comp_id[i + n]:
            print("NO")
            return
    
    print("YES")
    ans = [0] * (n + 1)
    for i in range(1, n + 1):
        # If comp_id[i] > comp_id[i+n], then i is assigned True (Primary)
        # because topological order of i's component is after i+n's component.
        # In Kosaraju, later in `order` means earlier in topological sort of SCCs.
        # Higher comp_id means later in topological sort.
        # We want the literal that appears later in topological order.
        ans[i] = 1 if comp_id[i] > comp_id[i + n] else 0
    
    print(' '.join(map(str, ans[1:])))

if __name__ == '__main__':
    solve()
