```python
import sys

def solve() -> None:
    sys.setrecursionlimit(1 << 25)
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = 2 * n
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def lit_to_node(l: int) -> int:
        v = abs(l) - 1
        return 2 * v if l > 0 else 2 * v + 1

    def neg(node: int) -> int:
        return node ^ 1

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        u = lit_to_node(a)
        v = lit_to_node(b)
        nu = neg(u)
        nv = neg(v)
        # (a OR b) => (~a -> b) and (~b -> a)
        g[nu].append(v)
        rg[v].append(nu)
        g[nv].append(u)
        rg[u].append(nv)

    # Kosaraju
    visited = [False] * N
    order = []

    sys.setrecursionlimit(10**6)
    def dfs(v: int):
        stack = [(v, 0)]
        while stack:
            node, idx = stack[-1]
            if not visited[node]:
                visited[node] = True
            adj = g[node]
            if idx < len(adj):
                nxt = adj[idx]
                stack[-1] = (node, idx + 1)
                if not visited[nxt]:
                    stack.append((nxt, 0))
            else:
                order.append(node)
                stack.pop()

    for v in range(N):
        if not visited[v]:
            dfs(v)

    comp = [-1] * N
    cid = 0
    def rdfs(v: int, cid: int):
        stack = [v]
        while stack:
            node = stack.pop()
            if comp[node] != -1:
                continue
            comp[node] = cid
            for nxt in rg[node]:
                if comp[nxt] == -1:
                    stack.append(nxt)

    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, cid)
            cid += 1

    assignment = [0] * n
    for i in range(n):
        t = 2 * i
        f = t + 1
        if comp[t] == comp[f]:
            print("NO")
            return
        assignment[i] = 1 if comp[t] > comp[f] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
