```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    m = int(data[1])
    N = 2 * n

    # literal node encoding: value v -> 2v-2 (v>0) or -2v-1 (v<0)
    # node 2i   : variable i+1 is 1
    # node 2i+1 : variable i+1 is 0
    lits = [2 * v - 2 if v > 0 else -2 * v - 1 for v in map(int, data[2:2 + 2 * m])]
    del data

    adj = [[] for _ in range(N)]
    radj = [[] for _ in range(N)]

    it = iter(lits)
    for a, b in zip(it, it):
        na = a ^ 1
        nb = b ^ 1
        adj[na].append(b)
        adj[nb].append(a)
        radj[b].append(na)
        radj[a].append(nb)
    del lits, it

    # Kosaraju pass 1: post-order on adj
    visited = bytearray(N)
    order = []
    order_append = order.append
    stack = []
    stack_append = stack.append
    stack_pop = stack.pop

    for sv in range(N):
        if visited[sv]:
            continue
        visited[sv] = 1
        stack_append(sv)
        while stack:
            u = stack_pop()
            if u >= 0:
                stack_append(~u)
                for w in adj[u]:
                    if not visited[w]:
                        visited[w] = 1
                        stack_append(w)
            else:
                order_append(~u)

    comp = [-1] * N
    c = 0
    for sv in reversed(order):
        if comp[sv] >= 0:
            continue
        comp[sv] = c
        stack_append(sv)
        while stack:
            u = stack_pop()
            for w in radj[u]:
                if comp[w] < 0:
                    comp[w] = c
                    stack_append(w)
        c += 1

    # assignment
    out = []
    out_append = out.append
    for p, q in zip(comp[0::2], comp[1::2]):
        if p == q:
            sys.stdout.write("NO\n")
            return
        out_append('1' if p > q else '0')

    sys.stdout.write("YES\n" + " ".join(out) + "\n")

main()
```

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    m = int(data[1])
    N = 2 * n

    # Literal node encoding:
    #   value v > 0  -> node 2*(v-1)      (variable v is 1)
    #   value v < 0  -> node 2*(-v-1)+1   (variable -v is 0)
    # Clause (a OR b) adds implications: ~a -> b, ~b -> a  (~x = x^1)
    lits = [v + v - 2 if v > 0 else -v - v - 1 for v in map(int, data[2:2 + 2 * m])]
    del data

    adj = [[] for _ in range(N)]
    radj = [[] for _ in range(N)]

    it = iter(lits)
    for a, b in zip(it, it):
        na = a ^ 1
        nb = b ^ 1
        adj[na].append(b)
        adj[nb].append(a)
        radj[b].append(na)
        radj[a].append(nb)
    del it
    del lits

    # ---- Kosaraju pass 1: post-order (finish order) on adj ----
    visited = bytearray(N)
    order = []
    order_append = order.append
    stack = []
    stack_append = stack.append
    stack_pop = stack.pop

    for sv in range(N):
        if visited[sv]:
            continue
        visited[sv] = 1
        stack_append(sv)
        while stack:
            u = stack_pop()
            if u >= 0:
                stack_append(~u)
                for w in adj[u]:
                    if not visited[w]:
                        visited[w] = 1
                        stack_append(w)
            else:
                order_append(~u)

    # ---- Kosaraju pass 2: label SCCs on radj, reverse finish order ----
    # Labels are in topological order of the condensation (source SCC = 0).
    comp = [-1] * N
    c = 0
    for sv in reversed(order):
        if comp[sv] >= 0:
            continue
        comp[sv] = c
        stack_append(sv)
        while stack:
            u = stack_pop()
            for w in radj[u]:
                if comp[w] < 0:
                    comp[w] = c
                    stack_append(w)
        c += 1

    # ---- Build assignment ----
    # x_i = 1 iff comp(true literal) > comp(false literal)
    out = []
    out_append = out.append
    for p, q in zip(comp[0::2], comp[1::2]):
        if p == q:
            sys.stdout.write("NO\n")
            return
        out_append('1' if p > q else '0')

    sys.stdout.write("YES\n" + " ".join(out) + "\n")

main()
```
