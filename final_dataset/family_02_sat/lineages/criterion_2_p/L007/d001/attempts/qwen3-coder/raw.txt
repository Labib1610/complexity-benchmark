```python
import sys
from collections import defaultdict

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    # For each zone, we have two variables: x_i (state 1) and ~x_i (state 0)
    # We'll use 2*i for x_i (state 1) and 2*i+1 for ~x_i (state 0)
    # But since zones are 1-indexed, we'll use 2*(i-1) and 2*(i-1)+1 for zone i
    
    # We build a graph for 2-SAT
    # Each node is a literal. Positive literal for state 1, negative for state 0
    # For a constraint (l1, l2) that forbids (not designated_state1, not designated_state2)
    # It means we must have (designated_state1 OR designated_state2)
    # Which translates to (~x1 OR ~x2) where x1 and x2 are the literals that make the constraint violated
    
    adj = defaultdict(list)
    adj_rev = defaultdict(list)
    
    # Process constraints
    for _ in range(m):
        l1, l2 = map(int, sys.stdin.readline().split())
        # Zone |l1|, designated state 1 if l1>0 else 0
        # Zone |l2|, designated state 1 if l2>0 else 0
        # Forbidden combination: (opposite of designated_state1, opposite of designated_state2)
        
        # Let's define:
        # For zone i: 
        #   var_i_1 (index 2*(i-1)) represents state 1
        #   var_i_0 (index 2*(i-1)+1) represents state 0
        
        z1 = abs(l1)
        s1_des = 1 if l1 > 0 else 0
        z2 = abs(l2)
        s2_des = 1 if l2 > 0 else 0
        
        # Forbidden is (not s1_des, not s2_des)
        # So we must have (s1_des OR s2_des)
        
        # In terms of literals:
        # If s1_des == 1, we use var_z1_1 (2*(z1-1))
        # If s1_des == 0, we use var_z1_0 (2*(z1-1)+1)
        # Similarly for z2
        
        # But in 2-SAT, we add implication clauses
        # (a OR b) is equivalent to (~a => b) and (~b => a)
        
        # Literal for s1_des
        if s1_des == 1:
            lit1 = 2 * (z1 - 1)  # x1
        else:
            lit1 = 2 * (z1 - 1) + 1  # ~x1
            
        # Literal for s2_des
        if s2_des == 1:
            lit2 = 2 * (z2 - 1)  # x2
        else:
            lit2 = 2 * (z2 - 1) + 1  # ~x2
            
        # We need (s1_des OR s2_des) which is (~(~s1_des) OR ~(~s2_des)) = (~(s1_opp) OR ~(s2_opp))
        # Where s1_opp is the opposite of s1_des, etc.
        # So we need (~s1_opp OR ~s2_opp)
        # Which means if s1_opp then ~s2_opp, and if s2_opp then ~s1_opp
        
        # s1_opp literal:
        if s1_des == 0:
            s1_opp = 2 * (z1 - 1)  # x1
        else:
            s1_opp = 2 * (z1 - 1) + 1  # ~x1
            
        # s2_opp literal:
        if s2_des == 0:
            s2_opp = 2 * (z2 - 1)  # x2
        else:
            s2_opp = 2 * (z2 - 1) + 1  # ~x2
            
        # Add clauses: s1_opp => ~s2_opp, which is ~s1_opp OR ~s2_opp
        # This is equivalent to edge from s1_opp to ~s2_opp and from s2_opp to ~s1_opp
        # But ~s2_opp is the opposite of s2_opp
        # s2_opp is at index i, then ~s2_opp is at index i^1
        adj[s1_opp].append(s2_opp ^ 1)
        adj[s2_opp].append(s1_opp ^ 1)
        
        adj_rev[s2_opp ^ 1].append(s1_opp)
        adj_rev[s1_opp ^ 1].append(s2_opp)
    
    # Now we do 2-SAT with Kosaraju's algorithm
    # Nodes are 0 to 2*n-1
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Second DFS on transpose graph to find SCCs
    visited = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in adj_rev[u]:
            if not visited[v]:
                dfs2(v)
    
    for i in reversed(order):
        if not visited[i]:
            dfs2(i)
            scc_count += 1
            
    # Check if for every variable x, x and ~x are in different SCCs
    for i in range(n):
        var_1 = 2 * i  # x_i (state 1)
        var_0 = 2 * i + 1  # ~x_i (state 0)
        if scc_id[var_1] == scc_id[var_0]:
            print("NO")
            return
            
    # Otherwise, there exists a solution
    # Assign values based on topological order of SCCs
    # Variables in later SCCs (higher scc_id) are assigned True first
    assignment = [False] * n
    for i in range(n):
        var_1 = 2 * i  # x_i 
        var_0 = 2 * i + 1  # ~x_i
        # If scc_id[var_1] > scc_id[var_0], then x_i should be True
        if scc_id[var_1] > scc_id[var_0]:
            assignment[i] = True
        else:
            assignment[i] = False
            
    print("YES")
    print(' '.join(str(int(b)) for b in assignment))

if __name__ == "__main__":
    main()
```
