import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n + 1)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        # yield(l1) < 0 and yield(l2) < 0 is forbidden
        # yield(v) < 0 iff state[|v|] = 0 and v > 0, or state[|v|] = 1 and v < 0
        # Let x_i = 1 if state[i] = 1, x_i = 0 if state[i] = 0
        # yield(v) < 0 <=> (v > 0 and x_{|v|} = 0) or (v < 0 and x_{|v|} = 1)
        # <=> (v > 0 and !x_{|v|}) or (v < 0 and x_{|v|})
        # <=> (v > 0 -> !x_{|v|}) and (v < 0 -> x_{|v|})
        # Let a = condition for yield(l1) < 0, b = condition for yield(l2) < 0
        # We forbid (a and b). So we require (!a or !b).
        # This is a 2-SAT clause: (!a or !b).
        
        # Determine literals for a and b
        # a is true <=> yield(l1) < 0
        if l1 > 0:
            a_lit = l1 * 2      # x_{l1} is false -> a is true. a_lit represents x_{l1}=false
            a_neg = l1 * 2 + 1
        else:
            a_lit = (-l1) * 2 + 1  # x_{|l1|} is true -> a is true. a_lit represents x_{|l1|}=true
            a_neg = (-l1) * 2
        
        if l2 > 0:
            b_lit = l2 * 2
            b_neg = l2 * 2 + 1
        else:
            b_lit = (-l2) * 2 + 1
            b_neg = (-l2) * 2
        
        # Clause: (!a or !b)
        # a_lit -> !a_lit is a_neg
        # b_lit -> !b_lit is b_neg
        adj[a_lit].append(b_neg)
        adj[b_lit].append(a_neg)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n + 2)
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack.pop()
            if idx == 0:
                if visited[node]:
                    continue
                visited[node] = True
            if idx < len(adj[node]):
                child = adj[node][idx]
                stack.append((node, idx + 1))
                if not visited[child]:
                    stack.append((child, 0))
            else:
                order.append(node)
    
    for i in range(1, 2 * n + 2):
        if not visited[i]:
            dfs1(i)
    
    radj = [[] for _ in range(2 * n + 2)]
    for u in range(1, 2 * n + 2):
        for v in adj[u]:
            radj[v].append(u)
    
    comp = [-1] * (2 * n + 2)
    cid = 0
    
    def dfs2(u, c):
        stack = [u]
        comp[u] = c
        while stack:
            node = stack.pop()
            for nei in radj[node]:
                if comp[nei] == -1:
                    comp[nei] = c
                    stack.append(nei)
    
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1
    
    result = [0] * (n + 1)
    for i in range(1, n + 1):
        # Node 2*i represents x_i = false (state 0)
        # Node 2*i+1 represents x_i = true (state 1)
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return
        if comp[2 * i + 1] > comp[2 * i]:
            result[i] = 1
        else:
            result[i] = 0
    
    print("YES")
    print(' '.join(map(str, result[1:])))

solve()
