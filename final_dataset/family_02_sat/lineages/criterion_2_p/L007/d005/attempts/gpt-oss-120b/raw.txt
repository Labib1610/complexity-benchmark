```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = 2 * n
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def lit_true(v):   # literal index for x_v = True
        return 2 * v + 1

    def lit_false(v):  # literal index for x_v = False
        return 2 * v

    def add_imp(u, v):
        g[u].append(v)
        rg[v].append(u)

    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))

        # litA = opposite of condition l1
        if l1 > 0:
            # condition: x_l1 = False, opposite is True
            litA = lit_true(l1 - 1)
        else:
            # condition: x_-l1 = True, opposite is False
            litA = lit_false(-l1 - 1)

        if l2 > 0:
            litB = lit_true(l2 - 1)
        else:
            litB = lit_false(-l2 - 1)

        # clause (litA ∨ litB)
        add_imp(litA ^ 1, litB)
        add_imp(litB ^ 1, litA)

    visited = [False] * N
    order = []

    def dfs(v):
        visited[v] = True
        for to in g[v]:
            if not visited[to]:
                dfs(to)
        order.append(v)

    for v in range(N):
        if not visited[v]:
            dfs(v)

    comp = [-1] * N
    cur = 0

    def rdfs(v, k):
        comp[v] = k
        for to in rg[v]:
            if comp[to] == -1:
                rdfs(to, k)

    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, cur)
            cur += 1

    assignment = [0] * n
    for i in range(n):
        if comp[lit_true(i)] == comp[lit_false(i)]:
            print("NO")
            return
        # true literal gets higher component number => earlier in topological order
        assignment[i] = 1 if comp[lit_true(i)] > comp[lit_false(i)] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
