```python
import sys

sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = n * 2
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def var(i):  # i 0‑based
        return i << 1

    def neg(x):
        return x ^ 1

    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        v1 = abs(l1) - 1
        v2 = abs(l2) - 1
        # literal true when (l>0) == (x_i == 1)
        lit1 = var(v1) if l1 > 0 else neg(var(v1))
        lit2 = var(v2) if l2 > 0 else neg(var(v2))
        # clause (lit1 OR lit2) => (~lit1 -> lit2) and (~lit2 -> lit1)
        nl1 = neg(lit1)
        nl2 = neg(lit2)
        g[nl1].append(lit2)
        g[nl2].append(lit1)
        rg[lit2].append(nl1)
        rg[lit1].append(nl2)

    visited = [False] * N
    order = []

    def dfs(v):
        stack = [(v, 0)]
        while stack:
            v, idx = stack.pop()
            if idx == 0:
                if visited[v]:
                    continue
                visited[v] = True
                stack.append((v, 1))
                for to in g[v]:
                    if not visited[to]:
                        stack.append((to, 0))
            else:
                order.append(v)

    for v in range(N):
        if not visited[v]:
            dfs(v)

    comp = [-1] * N
    cid = 0

    def rdfs(v, label):
        stack = [v]
        while stack:
            cur = stack.pop()
            if comp[cur] != -1:
                continue
            comp[cur] = label
            for to in rg[cur]:
                if comp[to] == -1:
                    stack.append(to)

    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, cid)
            cid += 1

    assignment = [0] * n
    for i in range(n):
        if comp[var(i)] == comp[neg(var(i))]:
            print("NO")
            return
        assignment[i] = 1 if comp[var(i)] > comp[neg(var(i))] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
