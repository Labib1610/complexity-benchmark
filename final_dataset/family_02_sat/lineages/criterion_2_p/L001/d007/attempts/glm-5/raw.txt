import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n + 2)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        # Clause: (x_a == 1) or (x_b == 1)
        # Implications:
        # x_a == 0 -> x_b == 1
        # x_b == 0 -> x_a == 1
        
        # Map variable v to node index
        # v > 0: node v (index v)
        # v < 0: node -v (index -v + n)
        def get_node(v):
            return v if v > 0 else -v + n
        
        a = l1
        b = l2
        
        u = get_node(a)
        v = get_node(b)
        
        # u is the node for a, v is the node for b
        # If a > 0, u represents x_a. If a < 0, u represents x_{-a}.
        # We need to add edges based on the implication.
        
        # If a > 0: we need x_a == 1. The implication x_a == 0 -> ... is from negation.
        # If a < 0: we need x_{-a} == 0. The implication x_{-a} == 1 -> ... is from negation.
        
        # Let's define the "true" node and "false" node for each variable i:
        # T_i = i, F_i = i + n
        
        # If a > 0: we need T_a. Negation is F_a.
        # If a < 0: we need F_{-a}. Negation is T_{-a}.
        
        if a > 0:
            u_true = a
            u_false = a + n
        else:
            u_true = -a + n
            u_false = -a
            
        if b > 0:
            v_true = b
            v_false = b + n
        else:
            v_true = -b + n
            v_false = -b
            
        adj[u_false].append(v_true)
        adj[v_false].append(u_true)
        
        # Also add reverse edges for Tarjan's (not strictly necessary for logic but good for graph)
        # No, standard adjacency list is fine.
        
    # Tarjan's SCC
    N = 2 * n + 1
    index = 0
    indices = [0] * (N + 1)
    lowlink = [0] * (N + 1)
    on_stack = [False] * (N + 1)
    stack = []
    scc_id = [0] * (N + 1)
    current_scc = 0
    
    def strongconnect(v):
        nonlocal index, current_scc
        index += 1
        indices[v] = index
        lowlink[v] = index
        stack.append(v)
        on_stack[v] = True
        
        for w in adj[v]:
            if indices[w] == 0:
                strongconnect(w)
                lowlink[v] = min(lowlink[v], lowlink[w])
            elif on_stack[w]:
                lowlink[v] = min(lowlink[v], indices[w])
        
        if lowlink[v] == indices[v]:
            current_scc += 1
            while True:
                w = stack.pop()
                on_stack[w] = False
                scc_id[w] = current_scc
                if w == v:
                    break
    
    for v in range(1, N + 1):
        if indices[v] == 0:
            strongconnect(v)
            
    # Check for contradiction
    for i in range(1, n + 1):
        if scc_id[i] == scc_id[i + n]:
            print("NO")
            return
            
    # Topological sort is implicit in SCC numbering (reverse order)
    # Assign values
    res = [0] * (n + 1)
    for i in range(1, n + 1):
        # If scc_id[i] < scc_id[i+n], then i is "more true" (appears earlier in reverse topological order)
        # We want to set to False if it appears later.
        # Actually, standard algorithm: if scc_id[x] > scc_id[not x], set x to true.
        # Because SCCs are processed in reverse topological order.
        # Let's verify: if scc_id[i] > scc_id[i+n], it means i is in a "later" SCC (topologically),
        # which depends on earlier SCCs. If we process in reverse topological order,
        # we encounter i first. We set it to True.
        # Wait, the SCCs are numbered in order of finishing (top of stack popped).
        # So current_scc increases as we finish SCCs.
        # The root SCC (sink) gets the highest ID? No.
        # In Kosaraju/Tarjan, the order depends.
        # Let's stick to the condition: if scc_id[i] < scc_id[i+n] then res[i] = 1?
        # Let's trace: if x -> !x, then x and !x are in same SCC (contradiction).
        # If !x -> x, then !x is in earlier SCC (topologically). x is in later SCC.
        # Tarjan processes nodes. The root of the DFS tree finishes last.
        # Actually, the standard property is: if scc_id[x] > scc_id[!x], set x to 1.
        # Let's check Example 1.
        # 1 2 -> (1 or 2). Edges: F1->T2, F2->T1.
        # -1 -2 -> (-1 or -2). Edges: T1->F2, T2->F1.
        # Graph:
        # T1 -> F2
        # T2 -> F1
        # F1 -> T2
        # F2 -> T1
        # Cycle: T1 -> F2 -> T1. SCC 1: {T1, F2}.
        # Cycle: T2 -> F1 -> T2. SCC 2: {T2, F1}.
        # Order of processing:
        # Start T1. Recurse F2. Recurse T1 (visited). Backtrack. F2 finishes. T1 finishes. SCC 1 found.
        # Next T2. Recurse F1. Recurse T2 (visited). Backtrack. F1 finishes. T2 finishes. SCC 2 found.
        # scc_id[T1] = 1, scc_id[F2] = 1.
        # scc_id[T2] = 2, scc_id[F1] = 2.
        # Check:
        # i=1: scc_id[T1]=1, scc_id[F1]=2. 1 < 2. res[1] = 1.
        # i=2: scc_id[T2]=2, scc_id[F2]=1. 2 > 1. res[2] = 0.
        # Result: 1 0. Correct.
        
        if scc_id[i] < scc_id[i + n]:
            res[i] = 1
        else:
            res[i] = 0
            
    print("YES")
    print(*(res[1:]))

solve()
