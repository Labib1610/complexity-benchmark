import sys
sys.setrecursionlimit(1_000_000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = n * 2
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def var(i, val):  # i: 0-indexed variable, val: 0 or 1 (literal true means su=1)
        return i * 2 + (1 if val else 0)

    def add_imp(u, v):
        g[u].append(v)
        rg[v].append(u)

    def add_clause(xi, xv, yi, yv):
        # (x == xv) OR (y == yv)
        # implications: (x != xv) -> (y == yv) ; (y != yv) -> (x == xv)
        add_imp(var(xi, not xv), var(yi, yv))
        add_imp(var(yi, not yv), var(xi, xv))

    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        u = abs(l1) - 1
        v = abs(l2) - 1
        s1 = 1 if l1 > 0 else -1
        s2 = 1 if l2 > 0 else -1
        # check all 4 assignments
        for su in (0, 1):
            a = 2 * su - 1
            for sv in (0, 1):
                b = 2 * sv - 1
                if s1 * (u + 1) * a + s2 * (v + 1) * b == -( (u + 1) + (v + 1) ):
                    # this pair is forbidden: add clause (su != su) OR (sv != sv)
                    # which is (su == not su) OR (sv == not sv)
                    add_clause(u, 1 - su, v, 1 - sv)

    order = []
    visited = [False] * N
    def dfs(v):
        visited[v] = True
        for to in g[v]:
            if not visited[to]:
                dfs(to)
        order.append(v)
    for i in range(N):
        if not visited[i]:
            dfs(i)

    comp = [-1] * N
    def rdfs(v, k):
        comp[v] = k
        for to in rg[v]:
            if comp[to] == -1:
                rdfs(to, k)

    k = 0
    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, k)
            k += 1

    assignment = [0] * n
    for i in range(n):
        if comp[var(i, 0)] == comp[var(i, 1)]:
            print("NO")
            return
        assignment[i] = 1 if comp[var(i, 1)] > comp[var(i, 0)] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
