import sys
from collections import defaultdict

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each constraint l1(2su-1) + l2(2sv-1) = -(|l1|+|l2|), simplify
    # Let's rewrite the equation:
    # l1(2su - 1) + l2(2sv - 1) = -(|l1| + |l2|)
    # 2*l1*su - l1 + 2*l2*sv - l2 = -|l1| - |l2|
    # 2*l1*su + 2*l2*sv = l1 + l2 - |l1| - |l2|
    #
    # Let R = l1 + l2 - |l1| - |l2|
    # If l1 > 0 and l2 > 0: R = l1 + l2 - l1 - l2 = 0
    # If l1 > 0 and l2 < 0: R = l1 + l2 - l1 + l2 = 2*l2 < 0
    # If l1 < 0 and l2 > 0: R = l1 + l2 + l1 - l2 = 2*l1 < 0
    # If l1 < 0 and l2 < 0: R = l1 + l2 + l1 + l2 = 2*(l1 + l2) < 0
    #
    # So we get: 2*l1*su + 2*l2*sv = R
    # Divide by 2: l1*su + l2*sv = R/2
    #
    # This gives us a system of linear equations over GF(2)
    # But let's think differently.
    #
    # Let x_u = 2*s_u - 1, so x_u is either +1 or -1
    # The constraint becomes: l1*x_u + l2*x_v = -(|l1| + |l2|)
    #
    # Case analysis:
    # 1. l1>0, l2>0: x_u + x_v = -2 => x_u=-1, x_v=-1 => s_u=0, s_v=0
    # 2. l1>0, l2<0: x_u - x_v = -2+2|l2|/(???)
    # 
    # Actually, let's re-examine the constraint more carefully.
    # We want: l1*(2*s_u - 1) + l2*(2*s_v - 1) = -(|l1| + |l2|)
    # Let's denote x_u = 2*s_u - 1 (so x_u in {-1, +1})
    # Constraint: l1*x_u + l2*x_v = -(|l1| + |l2|)
    #
    # Now case analysis on signs of l1, l2:
    # Case 1: l1 > 0, l2 > 0
    #   x_u + x_v = -2 => x_u = -1, x_v = -1 => s_u = 0, s_v = 0
    #
    # Case 2: l1 > 0, l2 < 0 (so l2 = -|l2|)
    #   x_u - |l2|*x_v = -(|l1| + |l2|) 
    #   Since |l1|=l1, this is: x_u - |l2|*x_v = -(l1 + |l2|)
    #   But x_u = +-1, x_v = +-1, so left side is integer between -1-|l2| and 1+|l2|.
    #   And right side is -(l1 + |l2|).
    #   Hmm, maybe there's a better way.
    #
    # Let's reconsider the original equation with s_u, s_v in {0,1}:
    # l1*(2*s_u - 1) + l2*(2*s_v - 1) = -(|l1| + |l2|)
    # 2*l1*s_u - l1 + 2*l2*s_v - l2 = -|l1| - |l2|
    # 2*l1*s_u + 2*l2*s_v = l1 + l2 - |l1| - |l2|
    # l1*s_u + l2*s_v = (l1 + l2 - |l1| - |l2|)/2
    #
    # Let's call RHS = C_{uv}
    # So we have modular linear equation: l1*s_u + l2*s_v ≡ C_{uv} (mod 2)
    # Because we're dealing with integers and want integer solutions, 
    # we need (l1 + l2 - |l1| - |l2|) to be even for the division by 2 to work.
    #
    # Let's check when l1 + l2 - |l1| - |l2| is even:
    # - If l1>0, l2>0: 0 (even)
    # - If l1>0, l2<0: l1 + l2 - l1 + l2 = 2*l2 (even)
    # - If l1<0, l2>0: l1 + l2 + l1 - l2 = 2*l1 (even)
    # - If l1<0, l2<0: l1 + l2 + l1 + l2 = 2*(l1 + l2) (even)
    # So it's always even. Good.
    #
    # So our constraints are:
    # l1*s_u + l2*s_v ≡ (l1 + l2 - |l1| - |l2|)//2  (mod 2)
    #
    # Let's compute C = (l1 + l2 - |l1| - |l2|)//2 for each constraint:
    # Case 1: l1>0, l2>0: C = (l1+l2-l1-l2)//2 = 0
    # Case 2: l1>0, l2<0: C = (l1+l2-l1+|l2|)//2 = (l2+|l2|)//2 = 0 (since l2<0, |l2|=-l2)
    # Case 3: l1<0, l2>0: C = (l1+l2+|l1|-l2)//2 = (l1+|l1|)//2 = 0 (since l1<0, |l1|=-l1)
    # Case 4: l1<0, l2<0: C = (l1+l2+|l1|+|l2|)//2 = (l1+l2-l1-l2)//2 = 0
    #
    # Wait, that can't be right. Let me redo this calculation.
    # l1 + l2 - |l1| - |l2|
    # Case 1: l1>0, l2>0: l1 + l2 - l1 - l2 = 0
    # Case 2: l1>0, l2<0: l1 + l2 - l1 + l2 = 2*l2 (but l2<0!)
    # Case 3: l1<0, l2>0: l1 + l2 + l1 - l2 = 2*l1 (but l1<0!)
    # Case 4: l1<0, l2<0: l1 + l2 + l1 + l2 = 2*(l1 + l2) 
    #
    # So C = (l1 + l2 - |l1| - |l2|)//2
    # Case 1: 0//2 = 0
    # Case 2: 2*l2//2 = l2 (which is negative, but we take mod 2 anyway)
    # Case 3: 2*l1//2 = l1 (negative)
    # Case 4: 2*(l1+l2)//2 = l1 + l2
    #
    # In mod 2 arithmetic:
    # Case 1: l1*s_u + l2*s_v ≡ 0 (mod 2)
    # Case 2: l1*s_u + l2*s_v ≡ l2 (mod 2). Since l2<0, l2≡0 or 1 (mod 2)
    # ...
    #
    # Maybe it's easier to just evaluate what values of s_u,s_v satisfy the constraint.
    # Given l1, l2, we want l1*(2*s_u-1) + l2*(2*s_v-1) = -(|l1|+|l2|)
    # Let's try all 4 possibilities for (s_u,s_v):
    # (0,0): l1*(-1) + l2*(-1) = -l1 - l2
    # (0,1): l1*(-1) + l2*(+1) = -l1 + l2  
    # (1,0): l1*(+1) + l2*(-1) = l1 - l2
    # (1,1): l1*(+1) + l2*(+1) = l1 + l2
    #
    # We want this to equal -(|l1| + |l2|).
    # So for each constraint, we find which of (0,0),(0,1),(1,0),(1,1) works.
    # If none works, impossible.
    # If one works, add implication.
    # If multiple work, still add implications?
    # Actually, we should build a graph where variables are s_i and their negations.
    # This is a 2-SAT problem.
    #
    # For each constraint, we figure out which assignments to s_u, s_v satisfy it.
    # Then we encode as 2-SAT.
    
    # Build 2-SAT instance
    # Variables: 0..n-1 represent s_1 .. s_n (0-based)
    #            n..2n-1 represent NOT s_1 .. NOT s_n
    
    def var(i): # 1-based index to 0-based variable
        return i - 1
    
    def not_var(i): # negation of variable i (1-based)
        return (i - 1) + n
    
    # For each pair (u,v), we'll collect forbidden pairs of values
    constraints = []
    
    for _ in range(m):
        l1, l2 = map(int, sys.stdin.readline().split())
        u = abs(l1)
        v = abs(l2)
        
        # Try all four combinations
        valid_clauses = []
        target = -(abs(l1) + abs(l2))
        
        # (s_u=0, s_v=0) corresponds to (x_u=-1, x_v=-1)
        val00 = l1 * (-1) + l2 * (-1)
        if val00 == target:
            # This combination is allowed, so we don't forbid it
            pass
        else:
            # This combination is forbidden
            # Add clause: NOT(s_u=0 AND s_v=0) = (s_u=1 OR s_v=1)
            valid_clauses.append((var(u)+n, var(v)+n)) # actually means "not u or not v"
            
        val01 = l1 * (-1) + l2 * (+1)  
        if val01 == target:
            pass
        else:
            # Forbid (s_u=0, s_v=1) => clause (s_u=1 OR s_v=0)
            valid_clauses.append((var(u)+n, var(v)))
            
        val10 = l1 * (+1) + l2 * (-1)
        if val10 == target:
            pass
        else:
            # Forbid (s_u=1, s_v=0) => clause (s_u=0 OR s_v=1)  
            valid_clauses.append((var(u), var(v)+n))
            
        val11 = l1 * (+1) + l2 * (+1)
        if val11 == target:
            pass
        else:
            # Forbid (s_u=1, s_v=1) => clause (s_u=0 OR s_v=0)
            valid_clauses.append((var(u), var(v)))
            
        # All clauses must be satisfied
        constraints.extend(valid_clauses)
    
    # Now we have a set of clauses (a OR b) that must all be true
    # This is 2-SAT. We can solve using SCC.
    
    # Number of variables in SAT: 2*n (variables and their negations)
    # Clauses are of form (a OR b) where a,b are in [0, 2*n-1]
    
    # Build implication graph
    adj = [[] for _ in range(2 * n)]
    rev_adj = [[] for _ in range(2 * n)]
    
    # Clause (a OR b) is equivalent to (NOT a -> b) AND (NOT b -> a)
    for a, b in constraints:
        # NOT a implies b
        adj[a ^ 1].append(b)  # NOT a is a^1 (flip last bit)
        rev_adj[b].append(a ^ 1)
        # NOT b implies a  
        adj[b ^ 1].append(a)
        rev_adj[a].append(b ^ 1)
        
    # Kosaraju's algorithm for SCC
    order = []
    visited = [False] * (2 * n)
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
        
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
            
    # Second DFS on reverse graph
    comp = [-1] * (2 * n)
    num_scc = 0
    
    def dfs2(u):
        comp[u] = num_scc
        for v in rev_adj[u]:
            if comp[v] == -1:
                dfs2(v)
                
    for i in reversed(order):
        if comp[i] == -1:
            dfs2(i)
            num_scc += 1
            
    # Check if satisfiable
    for i in range(n):
        if comp[var(i+1)] == comp[not_var(i+1)]:
            print("NO")
            return
            
    # Extract solution
    assignment = [0] * n
    # Process components in topological order
    # Component with higher index in order comes first in topological sort of condensation
    # So if comp[x] > comp[neg(x)], then x should be TRUE
    
    for i in range(n):
        u = var(i+1)      # represents s_{i+1}  
        not_u = not_var(i+1)  # represents NOT s_{i+1}
        if comp[u] > comp[not_u]:
            assignment[i] = 1
        else:
            assignment[i] = 0
            
    print("YES")
    print(" ".join(map(str, assignment)))

solve()
