We can model this as a **2-SAT** problem.  
Each hub has two possible states: ACTIVE (true) or DORMANT (false).  
Each contingency plan is a clause:  
`(l1 OR l2)` where a positive literal means the hub is ACTIVE, and a negative literal means the hub is DORMANT.

We need to check satisfiability and, if possible, output an assignment.

We’ll implement an iterative Kosaraju or Tarjan SCC algorithm for 2-SAT.

```python
import sys
sys.setrecursionlimit(1 << 20)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # each variable has two nodes: 2*i for false, 2*i+1 for true

    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def node(lit):
        # lit > 0: variable lit-1 true -> node 2*(lit-1)+1
        # lit < 0: variable -lit-1 false -> node 2*(-lit-1)
        if lit > 0:
            return 2 * (lit - 1) + 1
        else:
            return 2 * (-lit - 1)

    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)

    for _ in range(m):
        a, b = map(int, input().split())
        # clause (a or b) => (!a -> b) and (!b -> a)
        na = node(-a)
        nb = node(-b)
        add_imp(na, node(b))
        add_imp(nb, node(a))

    # Kosaraju SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        stack = [u]
        while stack:
            v = stack.pop()
            if visited[v]:
                continue
            visited[v] = True
            for w in g[v]:
                if not visited[w]:
                    stack.append(w)
            order.append(v)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        stack = [u]
        while stack:
            v = stack.pop()
            if comp[v] != -1:
                continue
            comp[v] = cid
            for w in gr[v]:
                if comp[w] == -1:
                    stack.append(w)

    cid = 0
    for v in reversed(order):
        if comp[v] == -1:
            dfs2(v, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    # Build assignment
    assignment = [0] * n
    for i in range(n):
        # comp[2*i] < comp[2*i+1] means false is earlier in topological order => true
        assignment[i] = 1 if comp[2*i] > comp[2*i+1] else 0

    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
