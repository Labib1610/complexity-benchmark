```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    m = int(data[1])
    N = 2 * n

    # Build implication graph: clause (a OR b) -> edges (not a -> b), (not b -> a)
    # Node of literal v: 2v-2 if v>0 else -2v-1 ; negation = node ^ 1
    adj = [[] for _ in range(N)]
    it = map(int, data[2:2 + 2 * m])
    for a, b in zip(it, it):
        na = 2 * a - 2 if a > 0 else -2 * a - 1
        nb = 2 * b - 2 if b > 0 else -2 * b - 1
        adj[na ^ 1].append(nb)
        adj[nb ^ 1].append(na)

    # Iterative Tarjan SCC
    index_arr = [0] * N      # discovery index, 0 = unvisited (1-based)
    low = [0] * N
    on = bytearray(N)
    comp = [0] * N
    tstack = []
    tappend = tstack.append
    tpop = tstack.pop
    vstack = []
    vappend = vstack.append
    vpop = vstack.pop
    istack = []
    iappend = istack.append
    ipop = istack.pop
    counter = 1
    scc = 0

    for root in range(N):
        if index_arr[root]:
            continue
        index_arr[root] = counter
        low[root] = counter
        counter += 1
        on[root] = 1
        tappend(root)
        vappend(root)
        iappend(iter(adj[root]))
        while vstack:
            v = vstack[-1]
            it2 = istack[-1]
            lowv = low[v]
            for w in it2:
                iw = index_arr[w]
                if not iw:
                    low[v] = lowv
                    index_arr[w] = counter
                    low[w] = counter
                    counter += 1
                    on[w] = 1
                    tappend(w)
                    vappend(w)
                    iappend(iter(adj[w]))
                    break
                elif on[w] and iw < lowv:
                    lowv = iw
            else:
                low[v] = lowv
                vpop()
                ipop()
                if vstack:
                    p = vstack[-1]
                    if lowv < low[p]:
                        low[p] = lowv
                if lowv == index_arr[v]:
                    while True:
                        w = tpop()
                        on[w] = 0
                        comp[w] = scc
                        if w == v:
                            break
                    scc += 1

    # Build answer
    out = []
    oappend = out.append
    for i in range(0, N, 2):
        a = comp[i]
        b = comp[i + 1]
        if a == b:
            sys.stdout.write('NO\n')
            return
        oappend('1' if a < b else '0')
    sys.stdout.write('YES\n' + ' '.join(out) + '\n')

main()
```
