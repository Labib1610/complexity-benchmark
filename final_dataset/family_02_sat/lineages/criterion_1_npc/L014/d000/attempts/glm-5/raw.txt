import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # Build 2-SAT: each zone must satisfy at least one of 3 behaviors.
    # Behavior x: x>0 means station x is 1 (true), x<0 means station |x| is 0 (false).
    # For each zone with behaviors a,b,c, add clauses: (a or b), (a or c), (b or c).
    
    # Variables: station i -> 2*i (true: station i is 1), 2*i+1 (false: station i is 0)
    # i from 1..n, so indices 2..2n+1
    
    adj = [[] for _ in range(2*n + 2)]
    radj = [[] for _ in range(2*n + 2)]
    
    def lit_to_node(x):
        # x positive: station x is 1 -> node 2*x
        # x negative: station |x| is 0 -> node 2*|x|+1
        if x > 0:
            return 2*x
        else:
            return 2*(-x) + 1
    
    def neg(node):
        return node ^ 1
    
    def add_clause(u, v):
        # add (u or v)
        adj[neg(u)].append(v)
        adj[neg(v)].append(u)
        radj[v].append(neg(u))
        radj[u].append(neg(v))
    
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        na = lit_to_node(a)
        nb = lit_to_node(b)
        nc = lit_to_node(c)
        add_clause(na, nb)
        add_clause(na, nc)
        add_clause(nb, nc)
    
    # Kosaraju SCC
    visited = [False] * (2*n + 2)
    order = []
    
    sys.setrecursionlimit(1000000)
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack[-1]
            if idx == 0:
                if visited[node]:
                    stack.pop()
                    continue
                visited[node] = True
            if idx < len(adj[node]):
                child = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[child]:
                    stack.append((child, 0))
            else:
                order.append(node)
                stack.pop()
    
    for i in range(2, 2*n + 2):
        if not visited[i]:
            dfs1(i)
    
    comp = [-1] * (2*n + 2)
    
    def dfs2(u, cid):
        stack = [u]
        comp[u] = cid
        while stack:
            node = stack[-1]
            found = False
            for child in radj[node]:
                if comp[child] == -1:
                    comp[child] = cid
                    stack.append(child)
                    found = True
                    break
            if not found:
                stack.pop()
    
    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1
    
    assignment = [0] * (n + 1)
    for i in range(1, n + 1):
        true_node = 2*i
        false_node = 2*i + 1
        if comp[true_node] == comp[false_node]:
            print("NO")
            return
        assignment[i] = 1 if comp[true_node] > comp[false_node] else 0
    
    print("YES")
    print(" ".join(str(assignment[i]) for i in range(1, n + 1)))

solve()
