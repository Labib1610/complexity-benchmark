We can model each protocol as a clause with three literals, where each literal is either a variable or its negation.  
This is exactly the **3-SAT** problem. Since \( n \le 90 \) and \( m \le 400 \), we can solve it using a backtracking algorithm with unit propagation and pure literal elimination (DPLL).

Here’s the Python solution:  

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        c1 = int(next(it))
        c2 = int(next(it))
        c3 = int(next(it))
        clauses.append((c1, c2, c3))
    
    # DPLL solver
    assignment = [None] * (n + 1)  # None = unassigned, True/False = value

    def evaluate_literal(lit, assign):
        var = abs(lit)
        if assign[var] is None:
            return None
        val = assign[var]
        if lit > 0:
            return val
        else:
            return not val

    def propagate(clauses, assign):
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                vals = [evaluate_literal(l, assign) for l in clause]
                if all(v is False for v in vals):
                    return False  # conflict
                # unit clause?
                unassigned = [l for l, v in zip(clause, vals) if v is None]
                if len(unassigned) == 1 and all(v is False for v in vals if v is not None):
                    lit = unassigned[0]
                    var = abs(lit)
                    val = lit > 0
                    if assign[var] is not None and assign[var] != val:
                        return False
                    assign[var] = val
                    changed = True
        return True

    def dpll(clauses, assign):
        if not propagate(clauses, assign):
            return False
        # Check if all clauses satisfied
        all_sat = True
        for clause in clauses:
            vals = [evaluate_literal(l, assign) for l in clause]
            if not any(v is True for v in vals):
                all_sat = False
                break
        if all_sat:
            return True
        # Choose unassigned variable
        for var in range(1, n + 1):
            if assign[var] is None:
                # Try True
                assign[var] = True
                if dpll(clauses, assign):
                    return True
                assign[var] = False
                if dpll(clauses, assign):
                    return True
                assign[var] = None
                return False
        return False

    if dpll(clauses, assignment):
        print("YES")
        print(" ".join("1" if assignment[i] else "0" for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
