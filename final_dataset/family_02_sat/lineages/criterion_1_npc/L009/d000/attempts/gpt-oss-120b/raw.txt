```python
import sys, random, time

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        c = [int(next(it)), int(next(it)), int(next(it))]
        clauses.append(c)

    if m == 0:
        print("YES")
        print(" ".join("0" for _ in range(n)))
        return

    # Preprocess: for each variable, list of clauses containing it (optional)
    var_clauses = [[] for _ in range(n)]
    for idx, cl in enumerate(clauses):
        for lit in cl:
            var = abs(lit) - 1
            var_clauses[var].append(idx)

    def clause_satisfied(cl, assign):
        for lit in cl:
            var = abs(lit) - 1
            val = assign[var]
            if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                return True
        return False

    start_time = time.time()
    TIME_LIMIT = 1.8  # seconds, leave margin

    max_tries = 200
    for _ in range(max_tries):
        # random initial assignment
        assign = [random.randint(0, 1) for _ in range(n)]

        # check satisfied quickly
        unsat = [i for i, cl in enumerate(clauses) if not clause_satisfied(cl, assign)]
        if not unsat:
            print("YES")
            print(" ".join(str(v) for v in assign))
            return

        max_flips = 2 * m
        for _ in range(max_flips):
            if not unsat:
                print("YES")
                print(" ".join(str(v) for v in assign))
                return
            # pick a random unsatisfied clause
            ci = random.choice(unsat)
            cl = clauses[ci]
            # pick a random literal in the clause to flip its variable
            lit = random.choice(cl)
            var = abs(lit) - 1
            assign[var] ^= 1  # flip

            # update unsatisfied list efficiently
            # recompute only clauses that contain var
            affected = var_clauses[var]
            new_unsat = []
            for idx in affected:
                cl2 = clauses[idx]
                sat = clause_satisfied(cl2, assign)
                if sat:
                    if idx in unsat:
                        unsat.remove(idx)
                else:
                    if idx not in unsat:
                        unsat.append(idx)

            # also need to check if flipping made previously satisfied clause unsatisfied
            # but above handles both add/remove.

            # early exit on time
            if time.time() - start_time > TIME_LIMIT:
                break
        if time.time() - start_time > TIME_LIMIT:
            break

    print("NO")

if __name__ == "__main__":
    solve()
```
