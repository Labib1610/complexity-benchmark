We can model this as a **3-SAT** problem, but since each clause has exactly 3 literals and we need to find a satisfying assignment, we can use a backtracking search with unit propagation, or reduce it to an **implication graph** approach if it were 2-SAT — but it’s not.  

Given the constraints (n ≤ 90, m ≤ 400), a simple **DPLL**-style backtracking with early pruning should work fine.  

Here’s the implementation:  

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    clauses = []
    for _ in range(m):
        l1, l2, l3 = map(int, input().split())
        clauses.append((l1, l2, l3))
    
    assignment = [None] * (n + 1)  # 1-indexed, None = unassigned

    def evaluate_literal(lit):
        var = abs(lit)
        val = assignment[var]
        if val is None:
            return None
        if lit > 0:
            return val == 1
        else:
            return val == 0

    def clause_satisfied(clause):
        for lit in clause:
            res = evaluate_literal(lit)
            if res is True:
                return True
        return False

    def clause_unsat(clause):
        # returns True if all literals are assigned and none are true
        all_assigned = True
        for lit in clause:
            res = evaluate_literal(lit)
            if res is True:
                return False
            if res is None:
                all_assigned = False
        return all_assigned

    def backtrack(var_idx):
        if var_idx > n:
            return True
        # Try both values
        for val in (0, 1):
            assignment[var_idx] = val
            # Check all clauses for immediate conflict
            conflict = False
            for clause in clauses:
                if clause_unsat(clause):
                    conflict = True
                    break
            if not conflict:
                if backtrack(var_idx + 1):
                    return True
            assignment[var_idx] = None
        return False

    if backtrack(1):
        print("YES")
        print(" ".join(str(assignment[i]) for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
