from collections import defaultdict
import sys

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each requirement, we have a clause with 3 literals
    # Literal x means variable x is True
    # Literal -x means variable x is False
    clauses = []
    for _ in range(m):
        clause = list(map(int, input().split()))
        clauses.append(clause)
    
    # We'll try all 2^n possible assignments
    # But n can be up to 90, so 2^90 is too large
    # However, since this is a 3-SAT problem with specific structure,
    # and given the constraints, let's try a more efficient approach
    
    # Let's use brute force with optimization
    # Since n <= 90 but that's too much, we might need a smarter approach
    # But let's first check if a simple brute force up to a certain point works
    
    # Actually, let's implement a proper 3-SAT solver using backtracking
    
    def is_satisfied(assignment):
        # assignment is a list where assignment[i] = 0 means variable i+1 is False
        # and assignment[i] = 1 means variable i+1 is True
        for clause in clauses:
            satisfied = False
            for literal in clause:
                var_index = abs(literal) - 1
                if literal > 0:  # variable should be True
                    if assignment[var_index] == 1:
                        satisfied = True
                        break
                else:  # variable should be False
                    if assignment[var_index] == 0:
                        satisfied = True
                        break
            if not satisfied:
                return False
        return True
    
    # Try all possible assignments
    # But since n can be large, let's limit this approach
    # For small n, we can do brute force
    
    if n <= 20:  # Brute force for small n
        for mask in range(1 << n):
            assignment = []
            for i in range(n):
                assignment.append((mask >> i) & 1)
            
            if is_satisfied(assignment):
                print("YES")
                print(" ".join(map(str, assignment)))
                return
        print("NO")
        return
    
    # For larger n, we need a more sophisticated approach
    # Let's implement a simple backtracking solver
    
    def backtrack(pos, assignment):
        if pos == n:
            return is_satisfied(assignment)
        
        # Try setting variable pos+1 to False (0)
        assignment[pos] = 0
        if all_clauses_consistent_so_far(assignment, pos):
            if backtrack(pos + 1, assignment):
                return True
        
        # Try setting variable pos+1 to True (1)
        assignment[pos] = 1
        if all_clauses_consistent_so_far(assignment, pos):
            if backtrack(pos + 1, assignment):
                return True
                
        return False
    
    def all_clauses_consistent_so_far(assignment, upto):
        # Check if all clauses that only involve variables 0..upto are satisfied
        # or can still be satisfied
        for clause in clauses:
            vars_in_clause = [abs(lit) - 1 for lit in clause]
            if all(var <= upto for var in vars_in_clause):
                # This clause only involves variables we've assigned
                satisfied = False
                for literal in clause:
                    var_index = abs(literal) - 1
                    if literal > 0:  # variable should be True
                        if assignment[var_index] == 1:
                            satisfied = True
                            break
                    else:  # variable should be False
                        if assignment[var_index] == 0:
                            satisfied = True
                            break
                if not satisfied:
                    return False
        return True
    
    # Better approach: implement a proper 3-SAT solver
    # Let's use the brute force but with early termination
    
    assignment = [0] * n
    found = False
    result_assignment = None
    
    def dfs(pos):
        nonlocal found, result_assignment
        if found:
            return
            
        if pos == n:
            if is_satisfied(assignment):
                found = True
                result_assignment = assignment[:]
            return
        
        # Pruning: check if current partial assignment violates any fully determined clause
        prune = False
        for clause in clauses:
            vars_in_clause = [abs(lit) - 1 for lit in clause]
            if all(var < pos for var in vars_in_clause):
                # All variables in this clause are already assigned
                satisfied = False

                    else:  # variable should be False

                            break
                if not satisfied:
                    prune = True
                    break
        
        if prune:
            return
            
        # Try 0
        assignment[pos] = 0
        dfs(pos + 1)
        
        # Try 1
        assignment[pos] = 1
        dfs(pos + 1)
    
    dfs(0)
    
    if found:
        print("YES")
        print(" ".join(map(str, result_assignment)))
    else:
        print("NO")
