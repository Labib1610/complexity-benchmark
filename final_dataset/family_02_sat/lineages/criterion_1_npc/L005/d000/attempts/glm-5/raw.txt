import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # Build implication graph for 2-SAT
    # Variable i: node 2*i (False), 2*i+1 (True) for i in 0..n-1
    # For each clause (a or b or c), we add implications:
    # not a -> b, not a -> c
    # not b -> a, not b -> c
    # not c -> a, not c -> b
    
    adj = [[] for _ in range(2 * n)]
    radj = [[] for _ in range(2 * n)]
    
    def get_node(var, is_true):
        # var is 1-indexed, convert to 0-indexed
        idx = abs(var) - 1
        if var > 0:
            return 2 * idx + 1  # True
        else:
            return 2 * idx      # False
    
    def get_neg(node):
        return node ^ 1
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        
        a = get_node(l1, True)
        b = get_node(l2, True)
        c = get_node(l3, True)
        
        # not a -> b, not a -> c
        adj[get_neg(a)].append(b)
        adj[get_neg(a)].append(c)
        radj[b].append(get_neg(a))
        radj[c].append(get_neg(a))
        
        # not b -> a, not b -> c
        adj[get_neg(b)].append(a)
        adj[get_neg(b)].append(c)
        radj[a].append(get_neg(b))
        radj[c].append(get_neg(b))
        
        # not c -> a, not c -> b
        adj[get_neg(c)].append(a)
        adj[get_neg(c)].append(b)
        radj[a].append(get_neg(c))
        radj[b].append(get_neg(c))
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(v):
        stack = [(v, 0)]
        while stack:
            node, idx = stack[-1]
            if idx == 0:
                visited[node] = True
            if idx < len(adj[node]):
                neighbor = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[neighbor]:
                    stack.append((neighbor, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    comp = [-1] * (2 * n)
    def dfs2(v, c):
        stack = [v]
        while stack:
            node = stack.pop()
            if comp[node] != -1:
                continue
            comp[node] = c
            for neighbor in radj[node]:
                if comp[neighbor] == -1:
                    stack.append(neighbor)
    
    cid = 0
    for v in reversed(order):
        if comp[v] == -1:
            dfs2(v, cid)
            cid += 1
    
    # Check if any variable has both True and False in same SCC
    for i in range(n):
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return
    
    # Assign values
    assignment = [0] * n
    for i in range(n):
        # If False node has higher topo order than True node, assign True
        if comp[2 * i] > comp[2 * i + 1]:
            assignment[i] = 1
    
    print("YES")
    print(" ".join(map(str, assignment)))

solve()
