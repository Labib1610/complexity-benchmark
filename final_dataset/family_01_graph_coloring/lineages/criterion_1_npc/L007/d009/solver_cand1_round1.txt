import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # D_{u,v} = T_{u,v} + 5 * (R_{u,v})^2
    # T_{u,v} = P^2 - 11P + 6S + 7
    # R_{u,v} = P - S^2 + 6S - 11
    # We want D_{u,v} = 7 for all edges.
    # Let's check all pairs (s_u, s_v) in {1, 2, 3} x {1, 2, 3}:
    # (1,1): S=2, P=1. T=1-11+12+7=9. R=1-4+12-11=-2. D=9+5(4)=29
    # (1,2): S=3, P=2. T=4-22+18+7=7. R=2-9+18-11=0. D=7+0=7
    # (1,3): S=4, P=3. T=9-33+24+7=7. R=3-16+24-11=0. D=7+0=7
    # (2,1): S=3, P=2. D=7
    # (2,2): S=4, P=4. T=16-44+24+7=3. R=4-16+24-11=1. D=3+5(1)=8
    # (2,3): S=5, P=6. T=36-66+30+7=7. R=6-25+30-11=0. D=7+0=7
    # (3,1): S=4, P=3. D=7
    # (3,2): S=5, P=6. D=7
    # (3,3): S=6, P=9. T=81-99+36+7=25. R=9-36+36-11=-2. D=25+5(4)=45
    
    # Valid pairs (s_u, s_v) are those where {s_u, s_v} is {1, 2}, {1, 3}, or {2, 3}.
    # This means s_u != s_v.
    # So we need a 3-coloring of the graph such that every edge connects different colors.
    # If the graph is bipartite, we can use 2 colors (1 and 2).
    # If not, we might need 3 colors.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [-1] * n
    
    def can_color(u, c):
        colors[u] = c
        for v in adj[u]:
            if colors[v] == -1:
                if not can_color(v, 1 - c if c < 2 else 0):
                    return False
            elif colors[v] == colors[u]:
                return False
        return True

    # Try to color with 3 colors using backtracking
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            valid = True
            for v in adj[u]:
                if colors[v] == c:
                    valid = False
                    break
            if valid:
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = -1
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
