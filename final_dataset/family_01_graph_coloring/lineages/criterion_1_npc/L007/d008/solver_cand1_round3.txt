import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The determinant condition:
    # M_uu = 2*s_u, M_vv = 2*s_v, M_uv = s_u + s_v
    # Det = (2*s_u * 2*s_v) - (s_u + s_v)^2
    # Det = 4*s_u*s_v - (s_u^2 + 2*s_u*s_v + s_v^2)
    # Det = 2*s_u*s_v - s_u^2 - s_v^2 = -(s_u - s_v)^2
    # The condition is -(s_u - s_v)^2 < 0, which is equivalent to s_u != s_v.
    # This means the graph must be bipartite.
    
    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    # If bipartite, we have two sets of vertices (color 0 and color 1).
    # We need to assign s_i in {1, 2, 3} such that for all edges (u, v), s_u != s_v.
    # We can assign all nodes in color 0 to state 1, and all nodes in color 1 to state 2.
    # This satisfies s_u != s_v for all edges.
    
    res = [0] * n
    for i in range(n):
        if color[i] == 0:
            res[i] = 1
        else:
            res[i] = 2
            
    print("YES")
    print(*(res))

solve()
