```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # C(s_u, s_v) = (s_u - s_v)^4 - 5(s_u - s_v)^2 + (s_u + s_v) + 4
    # Let x = s_u - s_v, y = s_u + s_v
    # C = x^4 - 5x^2 + y + 4
    # W = sum(C_uv) = sum((s_u - s_v)^4 - 5(s_u - s_v)^2 + s_u + s_v + 4)
    # V = sum(d_i * s_i)
    # W - V = sum((s_u - s_v)^4 - 5(s_u - s_v)^2 + 4) + sum(s_u + s_v) - sum(d_i * s_i)
    # Note: sum(s_u + s_v) over all edges is sum(d_i * s_i)
    # So W - V = sum((s_u - s_v)^4 - 5(s_u - s_v)^2 + 4)
    # We need sum((s_u - s_v)^4 - 5(s_u - s_v)^2 + 4) = 0
    
    # Possible values for s_u - s_v:
    # If s_u, s_v in {1, 2, 3}:
    # diff = 0: C_diff = 0 - 0 + 4 = 4
    # diff = 1 or -1: C_diff = 1 - 5 + 4 = 0
    # diff = 2 or -2: C_diff = 16 - 20 + 4 = 0
    
    # Let k be the number of edges where s_u == s_v.
    # Then W - V = k * 4 + (m - k) * 0 = 4k.
    # We need 4k = 0, so k = 0.
    # This means we need s_u != s_v for all edges (u, v).
    # This is equivalent to checking if the graph is 3-colorable.
    
    # Since n <= 40, we can use backtracking to find a 3-coloring.
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
