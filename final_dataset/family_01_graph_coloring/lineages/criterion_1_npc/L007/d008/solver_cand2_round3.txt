import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The formula C(s_u, s_v) simplifies to 17 if s_u != s_v.
    # If s_u == s_v, C(s_u, s_u) = 17 + (s_u-1)(s_u-2)(s_u-3) * something.
    # Actually, evaluating C(s_u, s_v):
    # C(1,1) = 1+1-3-3+2+2-5-5+5+5+4+4+17 = 17
    # C(2,2) = 32+32-96-96+64+64-40-40+40+40+8+8+17 = 31
    # C(3,3) = 243+243-729-729+486+486-135-135+135+135+12+12+17 = -17
    # C(1,2) = 17, C(1,3) = 17, C(2,3) = 17
    # So C(s_u, s_v) = 17 if s_u != s_v.
    # If s_u == s_v, C = 17 + (s_u-1)(s_u-2)(s_u-3) * (something).
    # Wait, the formula is symmetric. Let's re-check:
    # C(1,1) = 17
    # C(2,2) = 31
    # C(3,3) = -17
    # C(1,2) = 17
    # C(1,3) = 17
    # C(2,3) = 17
    # We need sum(C) = 17 * m.
    # This is satisfied if for every edge (u, v), s_u != s_v.
    # This is equivalent to checking if the graph is 3-colorable.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * n
    
    def is_bipartite():
        color = [-1] * n
        for i in range(n):
            if color[i] == -1:
                stack = [(i, 0)]
                color[i] = 0
                while stack:
                    u, c = stack.pop()
                    for v in adj[u]:
                        if color[v] == -1:
                            color[v] = 1 - c
                            stack.append((v, 1 - c))
                        elif color[v] == c:
                            return False, []
        return True, [c + 1 for c in color]

    # Try 2-coloring first
    possible, res = is_bipartite()
    if possible:
        print("YES")
        print(*(res))
        return

    # If not bipartite, try 3-coloring using backtracking
    res = [0] * n
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            ok = True
            for v in adj[u]:
                if res[v] == c:
                    ok = False
                    break
            if ok:
                res[u] = c
                if backtrack(u + 1):
                    return True
                res[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
