import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Stability condition: (s_u + s_v) % s_u != 0 or (s_u + s_v) % s_v != 0
    # This is equivalent to: s_v % s_u != 0 or s_u % s_v != 0
    # Pairs (s_u, s_v) that are NOT stable:
    # (1, 1): 1%1 == 0 and 1%1 == 0 (Unstable)
    # (1, 2): 2%1 == 0 (Unstable)
    # (2, 1): 2%1 == 0 (Unstable)
    # (1, 3): 3%1 == 0 (Unstable)
    # (3, 1): 3%1 == 0 (Unstable)
    # (2, 2): 2%2 == 0 (Unstable)
    # (3, 3): 3%3 == 0 (Unstable)
    # Stable pairs: (2, 3) and (3, 2)
    
    # So, for every edge (u, v), we must have {s_u, s_v} = {2, 3}.
    # This means the graph must be bipartite, and we must be able to 
    # color it with 2 and 3.
    
    colors = [0] * n
    for i in range(n):
        if colors[i] == 0:
            stack = [(i, 2)]
            colors[i] = 2
            while stack:
                u, c = stack.pop()
                next_c = 3 if c == 2 else 2
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_c
                        stack.append((v, next_c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors))

solve()
