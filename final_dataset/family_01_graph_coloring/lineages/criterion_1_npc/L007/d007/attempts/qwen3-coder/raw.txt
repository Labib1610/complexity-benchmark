from itertools import product

def compute_coupling(s_u, s_v):
    return (s_u**5 + s_v**5 
            - 3*s_u**4*s_v - 3*s_u*s_v**4 
            + 2*s_u**3*s_v**2 + 2*s_u**2*s_v**3 
            - 5*s_u**3 - 5*s_v**3 
            + 5*s_u**2*s_v + 5*s_u*s_v**2 
            + 4*s_u + 4*s_v)

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
        
    # Precompute all possible coupling values for state pairs
    coupling_values = {}
    for s1 in [1,2,3]:
        for s2 in [1,2,3]:
            coupling_values[(s1,s2)] = compute_coupling(s1, s2)
            
    # Try all 3^n combinations
    for combo in product([1,2,3], repeat=n):
        total = 0
        for u, v in edges:
            total += coupling_values[(combo[u], combo[v])]
        if total == 0:
            print("YES")
            print(" ".join(map(str, combo)))
            return
            
    print("NO")

solve()
