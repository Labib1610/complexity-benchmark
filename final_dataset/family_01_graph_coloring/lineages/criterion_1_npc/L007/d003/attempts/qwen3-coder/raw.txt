def solve():
    import sys
    from collections import defaultdict
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Try all 3-colorings using backtracking
    color = [-1] * n
    
    def is_valid(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
            
        for c in range(3):
            if is_valid(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = -1
        return False
    
    # Since n <= 40, but actually constraints seem tighter, let's try smarter approach
    # For graph 3-coloring, we can use a more systematic approach
    
    # Let's try all possible colorings with pruning
    found = backtrack(0)
    
    if not found:
        print("NO")
    else:
        print("YES")
        result = [str(c + 1) for c in color]
        print(" ".join(result))

solve()
