import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        edges.append((u, v))
        idx += 2

    # B(u, v) = (u^2+1)(v^2+1)s_u + 2(u^2+1)(v^2+1)s_v + 3uvs_u s_v
    # Modulo 3:
    # B(u, v) = (u^2+1)(v^2+1)s_u + 2(u^2+1)(v^2+1)s_v (mod 3)
    # Let K = (u^2+1)(v^2+1) mod 3
    # B(u, v) = K * s_u + 2 * K * s_v = K * (s_u - s_v) (mod 3)
    # Condition: B(u, v) is not a multiple of 3
    # K * (s_u - s_v) != 0 (mod 3)
    # Since K = (u^2+1)(v^2+1), K is 0 mod 3 if u^2+1=0 or v^2+1=0.
    # u^2+1 = 0 (mod 3) has no solution (0^2+1=1, 1^2+1=2, 2^2+1=5=2).
    # So K is never 0 mod 3.
    # Thus, K * (s_u - s_v) != 0 (mod 3) is equivalent to s_u - s_v != 0 (mod 3),
    # which means s_u != s_v (mod 3).
    # Since s_i in {1, 2, 3}, this is equivalent to s_u != s_v.
    
    # The problem reduces to: Can we color the graph with 3 colors such that
    # no two adjacent vertices have the same color?
    # This is the 3-coloring problem, which is NP-complete, but n <= 40.
    # We can use backtracking with pruning.
    
    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
