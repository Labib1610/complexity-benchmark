import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition is s_u^2 + s_u*s_v + s_v^2 != 0 (mod 3).
    # s_i in {1, 2, 3}. Modulo 3, these are {1, 2, 0}.
    # If s_i = 3 (mod 3 = 0):
    #   0^2 + 0*s_v + s_v^2 = s_v^2.
    #   If s_v = 1, s_v^2 = 1 != 0.
    #   If s_v = 2, s_v^2 = 4 = 1 != 0.
    #   If s_v = 0, s_v^2 = 0 == 0.
    # So, if s_u = 3, s_v cannot be 3.
    # If s_u, s_v in {1, 2}:
    #   1, 1: 1 + 1 + 1 = 3 = 0 (mod 3) -> Invalid
    #   2, 2: 4 + 4 + 4 = 12 = 0 (mod 3) -> Invalid
    #   1, 2: 1 + 2 + 4 = 7 = 1 != 0 (mod 3) -> Valid
    # Summary:
    # 1. If s_u = 3, s_v must be in {1, 2}.
    # 2. If s_u, s_v in {1, 2}, they must be different (one 1, one 2).
    
    # This means the graph must be 3-colorable, but with a twist:
    # Color 3 is independent (no two 3s adjacent).
    # Colors 1 and 2 form a bipartite graph.
    # This is equivalent to: Can we partition vertices into sets A, B, C
    # such that C is an independent set, and A, B form a bipartite graph?
    
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning 3
        can_be_3 = True
        for v in adj[u]:
            if assignment[v] == 3:
                can_be_3 = False
                break
        if can_be_3:
            assignment[u] = 3
            if backtrack(u + 1): return True
            assignment[u] = 0
            
        # Try assigning 1 or 2
        for color in [1, 2]:
            possible = True
            for v in adj[u]:
                if assignment[v] == color:
                    possible = False
                    break
            if possible:
                assignment[u] = color
                if backtrack(u + 1): return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
