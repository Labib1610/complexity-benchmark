```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition: min of f(x) = (x-su)^2 + (x-sv)^2 is at x = (su+sv)/2.
    # We need min(su, sv) < (su+sv)/2 < max(su, sv).
    # This is equivalent to saying su != sv.
    # If su == sv, then (su+sv)/2 = su, which is not strictly inside (su, su).
    # If su != sv, then (su+sv)/2 is exactly the midpoint, which is strictly between.
    # So the problem is equivalent to: can we color the graph with 3 colors such that
    # no two adjacent nodes have the same color?
    # This is the 3-coloring problem, which is NP-complete.
    # However, the constraint n <= 40 (since n(n-1)/2 <= 780) allows backtracking.

    colors = [0] * (n + 1)

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
```
