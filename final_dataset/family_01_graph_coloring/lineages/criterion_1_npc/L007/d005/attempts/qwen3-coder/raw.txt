def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # For each edge (u,v), we need s_u^2 + s_u*s_v + s_v^2 not divisible by 3
    # Let's compute forbidden pairs:
    # We have s_u, s_v in {1,2,3}
    # Compute s_u^2 + s_u*s_v + s_v^2 mod 3 for all combinations
    forbidden = set()
    for su in [1,2,3]:
        for sv in [1,2,3]:
            val = (su*su + su*sv + sv*sv) % 3
            if val == 0:
                forbidden.add((su,sv))
                forbidden.add((sv,su))  # Since graph is undirected
    
    # The forbidden pairs are those where the expression is 0 mod 3
    # Let's see which pairs are actually forbidden:
    # (1,1): 1+1+1=3 → 0 mod 3 → forbidden
    # (1,2): 1+2+4=7 → 1 mod 3 → ok
    # (1,3): 1+3+9=13 → 1 mod 3 → ok
    # (2,2): 4+4+4=12 → 0 mod 3 → forbidden
    # (2,3): 4+6+9=19 → 1 mod 3 → ok
    # (3,3): 9+9+9=27 → 0 mod 3 → forbidden
    
    forbidden_pairs = {(1,1), (2,2), (3,3)}
    
    # This is a graph coloring problem with constraints
    # We can try all 3^n possibilities for small n, but since n<=40, that's too much
    # However, let's think differently. 
    # This looks like a constraint satisfaction problem.
    # We can model this as a 3-coloring where certain color pairs are forbidden on edges.
    # In fact, the forbidden pairs are (1,1), (2,2), (3,3)
    # So adjacent nodes cannot have the same color.
    # This is just normal graph 3-coloring!
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Try to 3-color the graph
    # Colors: 1, 2, 3
    color = [0] * n  # 0 means uncolored
    
    def is_safe(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def solve(node):
        if node == n:
            return True
            
        for c in [1, 2, 3]:
            if is_safe(node, c):
                color[node] = c
                if solve(node + 1):
                    return True
                color[node] = 0
                
        return False
    
    if solve(0):
        print("YES")
        print(' '.join(map(str, color)))
    else:
        print("NO")

if __name__ == "__main__":
    main()
