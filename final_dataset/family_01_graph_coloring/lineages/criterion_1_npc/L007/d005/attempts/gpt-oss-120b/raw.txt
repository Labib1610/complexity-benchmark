```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # precompute allowed pairs (x,y) where x,y in {0,1,2} (mod 3)
    allowed = [[True]*3 for _ in range(3)]
    for x in range(3):
        for y in range(3):
            f = (x*x + x*y + y*y) % 3
            allowed[x][y] = f != 0

    # domains: bitmask of possible colors (bits 0..2)
    domains = [(1<<3)-1 for _ in range(n)]

    order = sorted(range(n), key=lambda i: -len(adj[i]))  # static order by degree

    assignment = [-1]*n

    # helper to iterate bits
    def bits(mask):
        while mask:
            b = mask & -mask
            yield (b.bit_length()-1)
            mask ^= b

    sys.setrecursionlimit(10000)

    def dfs(idx):
        if idx == n:
            return True
        # select unassigned variable with smallest domain (MRV)
        var = -1
        min_sz = 4
        for i in range(n):
            if assignment[i] == -1:
                sz = bin(domains[i]).count('1')
                if sz < min_sz:
                    min_sz = sz
                    var = i
                    if sz == 1:
                        break
        if var == -1:
            return True
        cur_mask = domains[var]
        for col in bits(cur_mask):
            # try assign col
            # record changes to revert
            changes = []
            ok = True
            assignment[var] = col
            # propagate to neighbors
            for nb in adj[var]:
                if assignment[nb] != -1:
                    if not allowed[col][assignment[nb]]:
                        ok = False
                        break
                else:
                    # remove colors from neighbor that conflict with col
                    forbid = 0
                    for c2 in range(3):
                        if not allowed[col][c2]:
                            forbid |= 1<<c2
                    new_mask = domains[nb] & ~forbid
                    if new_mask != domains[nb]:
                        changes.append((nb, domains[nb]))
                        domains[nb] = new_mask
                    if new_mask == 0:
                        ok = False
                        break
            if ok and dfs(idx+1):
                return True
            # revert
            for nb, old in reversed(changes):
                domains[nb] = old
            assignment[var] = -1
        return False

    if dfs(0):
        # map 0->3, 1->1, 2->2
        out = []
        for c in assignment:
            out.append(str(3 if c == 0 else c))
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(out))
    else:
        sys.stdout.write("NO\n")

if __name__ == "__main__":
    solve()
```
