```python
import sys

def solve():
    # The interference factor is the number of divisors of 12^s_u * 18^s_v.
    # 12^s_u * 18^s_v = (2^2 * 3)^s_u * (2 * 3^2)^s_v = 2^(2*s_u + s_v) * 3^(s_u + 2*s_v).
    # The number of divisors is (2*s_u + s_v + 1) * (s_u + 2*s_v + 1).
    # We need this to NOT be a perfect square.
    # Let f(s_u, s_v) = (2*s_u + s_v + 1) * (s_u + 2*s_v + 1).
    # Possible values for s_u, s_v in {1, 2, 3}:
    # s_u=1, s_v=1: (2+1+1)*(1+2+1) = 4*4 = 16 (Square)
    # s_u=1, s_v=2: (2+2+1)*(1+4+1) = 5*6 = 30
    # s_u=1, s_v=3: (2+3+1)*(1+6+1) = 6*8 = 48
    # s_u=2, s_v=1: (4+1+1)*(2+2+1) = 6*5 = 30
    # s_u=2, s_v=2: (4+2+1)*(2+4+1) = 7*7 = 49 (Square)
    # s_u=2, s_v=3: (4+3+1)*(2+6+1) = 8*9 = 72
    # s_u=3, s_v=1: (6+1+1)*(3+2+1) = 8*6 = 48
    # s_u=3, s_v=2: (6+2+1)*(3+4+1) = 9*8 = 72
    # s_u=3, s_v=3: (6+3+1)*(3+6+1) = 10*10 = 100 (Square)
    
    # Constraint: f(s_u, s_v) is not a square.
    # This means (s_u, s_v) cannot be (1,1), (2,2), or (3,3).
    # This is equivalent to s_u != s_v.
    # So for every edge (u, v), we must have s_u != s_v.
    # This is exactly the condition for a graph to be 3-colorable.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
