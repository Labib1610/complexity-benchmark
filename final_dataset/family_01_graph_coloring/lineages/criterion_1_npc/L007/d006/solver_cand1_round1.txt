import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The energy formula simplifies significantly.
    # Let f(s) = s^4 - 5s^2.
    # E_i = d_i * f(s_i)
    # C_{u,v} = 2 * s_u * s_v * (3 * s_u * s_v + 5 - 2 * s_u^2 - 2 * s_v^2)
    #
    # For s in {1, 2, 3}:
    # f(1) = 1 - 5 = -4
    # f(2) = 16 - 20 = -4
    # f(3) = 81 - 45 = 36
    #
    # If s_i = 1 or 2, f(s_i) = -4. If s_i = 3, f(s_i) = 36.
    #
    # Let's test if s_i in {1, 2} for all i works.
    # If s_i in {1, 2}, then f(s_i) = -4.
    # Sum E_i = sum(d_i * (-4)) = -4 * sum(d_i) = -4 * (2m) = -8m.
    #
    # C_{u,v} for s_u, s_v in {1, 2}:
    # If s_u=1, s_v=1: C = 2(1)(3+5-2-2) = 8
    # If s_u=1, s_v=2: C = 2(2)(6+5-2-8) = 4
    # If s_u=2, s_v=1: C = 4
    # If s_u=2, s_v=2: C = 2(4)(12+5-8-8) = 8
    #
    # This doesn't look like it will sum to -4m easily.
    # Let's use backtracking since n <= 40 is small enough for some cases,
    # but 3^40 is too large. However, the constraints and the target energy
    # suggest a specific structure.
    
    states = [0] * n
    
    def get_energy(s):
        total = 0
        for i in range(n):
            d_i = len(adj[i])
            si = s[i]
            total += d_i * (si**4 - 5*si**2)
        for u in range(n):
            for v in adj[u]:
                if u < v:
                    su, sv = s[u], s[v]
                    total += 2 * su * sv * (3 * su * sv + 5 - 2 * su**2 - 2 * sv**2)
        return total

    # Try simple assignments or heuristics
    # Given the target -4m, let's try random search or greedy
    import random
    
    target = -4 * m
    for _ in range(100000):
        s = [random.choice([1, 2, 3]) for _ in range(n)]
        if get_energy(s) == target:
            print("YES")
            print(*(s))
            return

    print("NO")

solve()
