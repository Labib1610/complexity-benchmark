import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        w = int(input_data[idx+1])
        edges.append((u, w))
        idx += 2

    def get_det(x, y, z):
        s0 = 3
        s1 = x + y + z
        s2 = x*x + y*y + z*z
        s3 = x**3 + y**3 + z**3
        s4 = x**4 + y**4 + z**4
        # Matrix M = [[s0, s1, s2], [s1, s2, s3], [s2, s3, s4]]
        # det = s0(s2*s4 - s3*s3) - s1(s1*s4 - s2*s3) + s2(s1*s3 - s2*s2)
        return s0 * (s2 * s4 - s3 * s3) - s1 * (s1 * s4 - s2 * s3) + s2 * (s1 * s3 - s2 * s2)

    # The determinant of the Vandermonde-like matrix M is (x-y)^2 * (x-z)^2 * (y-z)^2
    # Since x, y, z are in {1, 2, 3}, the determinant is zero if and only if
    # at least two of {x, y, z} are equal.
    # z = u + w + 3.
    # If u + w + 3 is 1, 2, or 3, then z is 1, 2, or 3.
    # Since u, w >= 1, u + w + 3 >= 5.
    # Thus z is always >= 5.
    # The determinant is (x-y)^2 * (x-z)^2 * (y-z)^2.
    # Since z >= 5 and x, y in {1, 2, 3}, x != z and y != z.
    # The determinant is zero iff x == y.
    # So the condition is simply x != y for every edge (u, w).
    # This is equivalent to 3-coloring the graph.

    adj = [[] for _ in range(n + 1)]
    for u, w in edges:
        adj[u].append(w)
        adj[w].append(u)

    colors = [0] * (n + 1)

    def is_valid(node, c):
        for neighbor in adj[node]:
            if colors[neighbor] == c:
                return False
        return True

    def backtrack(node):
        if node > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
