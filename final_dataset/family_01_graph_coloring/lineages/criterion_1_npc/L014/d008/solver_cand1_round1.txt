import sys

def solve():
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m = map(int, line1)
    except ValueError:
        return

    adj = []
    for _ in range(m):
        adj.append(list(map(int, sys.stdin.readline().split())))

    def check(u, w, x, y):
        # V_u = (w * x^2 + y) % x
        # V_w = (u * y^2 + x) % y
        # If x=1, V_u = (w*1 + y) % 1 = 0
        # If y=1, V_w = (u*1 + x) % 1 = 0
        vu = (w * (x * x) + y) % x
        vw = (u * (y * y) + x) % y
        return vu != vw

    assignment = [0] * (n + 1)

    def backtrack(node):
        if node > n:
            return True
        
        for speed in [1, 2, 3]:
            assignment[node] = speed
            valid = True
            for u, w in adj:
                if u == node and assignment[w] != 0:
                    if not check(u, w, assignment[u], assignment[w]):
                        valid = False
                        break
                elif w == node and assignment[u] != 0:
                    if not check(u, w, assignment[u], assignment[w]):
                        valid = False
                        break
            
            if valid:
                if backtrack(node + 1):
                    return True
            assignment[node] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
