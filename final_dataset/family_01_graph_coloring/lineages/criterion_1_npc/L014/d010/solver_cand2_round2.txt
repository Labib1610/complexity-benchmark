import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The polynomial P(t) = t^3 - (x+y+K)t^2 + (xy + K(x+y))t - Kxy
    # This factors as (t - x)(t - y)(t - K) = 0.
    # Roots are x, y, and K = u + w + 10.
    # Harmonic resonance occurs if any roots are equal:
    # 1. x = y
    # 2. x = K = u + w + 10
    # 3. y = K = u + w + 10
    # Since x, y in {1, 2, 3}, and u, w >= 1, K = u + w + 10 >= 12.
    # Thus, x = K and y = K are impossible.
    # The only condition for resonance is x = y.
    # So, for every edge (u, w), we must have x_u != x_w.
    # This is equivalent to 3-coloring the graph.

    colors = [0] * (n + 1)

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
