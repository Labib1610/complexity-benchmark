import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition F_u(lambda) = 0 and F_w(lambda) = 0 implies a common root.
    # F_u(lambda) = (lambda + 1)^3 * (lambda + 2 - x) + u * (lambda + 1) * (lambda + 1 - x)
    # This simplifies to:
    # (lambda + 1) * [(lambda + 1)^2 * (lambda + 2 - x) + u * (lambda + 1 - x)] = 0
    # The roots are lambda = -1 and the roots of the cubic.
    # By testing values, the condition F_u(lambda) = F_w(lambda) = 0 for some lambda
    # simplifies to x == y. If x != y, the roots are distinct.
    # Thus, the problem is equivalent to: can we color the graph with 3 colors
    # such that no two adjacent nodes have the same color?
    # This is the 3-coloring problem, which is NP-complete, but n <= 40.
    # We can use backtracking with pruning.

    colors = [0] * (n + 1)

    def is_valid(node, c):
        for neighbor in adj[node]:
            if colors[neighbor] == c:
                return False
        return True

    def backtrack(node):
        if node > n:
            return True
        
        # Heuristic: try colors that are most constrained or just 1, 2, 3
        for c in [1, 2, 3]:
            if is_valid(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
