import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        w = int(input_data[idx+1])
        adj.append((u, w))
        idx += 2

    def is_compatible(u, x, w, y):
        wu = 100 * x + u
        ww = 100 * y + w
        
        max_w = max(wu, ww)
        min_w = min(wu, ww)
        
        su = max_w * x + min_w * y
        sw = max_w * y + min_w * x
        
        return su != sw

    assignment = [0] * (n + 1)

    def backtrack(task_idx):
        if task_idx > n:
            return True
        
        for val in [1, 2, 3]:
            assignment[task_idx] = val
            possible = True
            for u, w in adj:
                if u == task_idx and assignment[w] != 0:
                    if not is_compatible(u, val, w, assignment[w]):
                        possible = False
                        break
                elif w == task_idx and assignment[u] != 0:
                    if not is_compatible(u, assignment[u], w, val):
                        possible = False
                        break
            
            if possible:
                if backtrack(task_idx + 1):
                    return True
            assignment[task_idx] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
