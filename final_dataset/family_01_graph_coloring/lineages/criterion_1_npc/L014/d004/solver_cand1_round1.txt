```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning 1, 2, or 3
        for val in [1, 2, 3]:
            possible = True
            for neighbor in adj[u]:
                if assignment[neighbor] != 0:
                    # The condition is that the set of voltages must be {1, 2}, {1, 3}, or {2, 3}
                    # This is equivalent to saying the two voltages cannot be equal.
                    if assignment[neighbor] == val:
                        possible = False
                        break
            
            if possible:
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
