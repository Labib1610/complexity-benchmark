import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem asks for a 3-coloring of the graph.
    # Since n <= 40, we can use backtracking with heuristics.
    # Ordering by degree (MRV) is effective.
    
    assignment = [0] * (n + 1)
    
    def get_mrv_node(assigned):
        best_node = -1
        min_remaining = 4
        for i in range(1, n + 1):
            if not assigned[i]:
                # Count available colors
                used = set()
                for neighbor in adj[i]:
                    if assignment[neighbor] != 0:
                        used.add(assignment[neighbor])
                remaining = 3 - len(used)
                if remaining < min_remaining:
                    min_remaining = remaining
                    best_node = i
        return best_node

    def backtrack(count):
        if count == n:
            return True
        
        u = get_mrv_node([assignment[i] != 0 for i in range(n + 1)])
        if u == -1:
            return True
            
        used = set()
        for neighbor in adj[u]:
            if assignment[neighbor] != 0:
                used.add(assignment[neighbor])
        
        for color in [1, 2, 3]:
            if color not in used:
                assignment[u] = color
                if backtrack(count + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
