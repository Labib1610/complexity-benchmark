import sys

def solve():
    # The condition for f(x) = ax^2 + (a+b)x + b to have two distinct real roots
    # is that the discriminant D > 0.
    # D = (a+b)^2 - 4ab = a^2 + 2ab + b^2 - 4ab = a^2 - 2ab + b^2 = (a-b)^2.
    # For D > 0, we need (a-b)^2 > 0, which means a != b.
    # Thus, for every critical pair (u, v), we must have sensitivity[u] != sensitivity[v].
    # This is equivalent to finding a 3-coloring of the graph where critical pairs are edges.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if colors[v] == c:
                    possible = False
                    break
            if possible:
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
