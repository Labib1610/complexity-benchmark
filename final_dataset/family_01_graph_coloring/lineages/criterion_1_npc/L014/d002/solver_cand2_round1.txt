import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # Heuristic: Degree of saturation (DSATUR algorithm)
    # This is effective for graph coloring
    def get_next_node():
        best_node = -1
        max_sat = -1
        max_deg = -1
        
        for i in range(1, n + 1):
            if assignment[i] == 0:
                neighbor_colors = {assignment[neighbor] for neighbor in adj[i] if assignment[neighbor] != 0}
                sat = len(neighbor_colors)
                deg = len([neighbor for neighbor in adj[i] if assignment[neighbor] == 0])
                
                if sat > max_sat or (sat == max_sat and deg > max_deg):
                    max_sat = sat
                    max_deg = deg
                    best_node = i
        return best_node

    def backtrack():
        u = get_next_node()
        if u == -1:
            return True
        
        neighbor_colors = {assignment[neighbor] for neighbor in adj[u] if assignment[neighbor] != 0}
        for color in [1, 2, 3]:
            if color not in neighbor_colors:
                assignment[u] = color
                if backtrack():
                    return True
                assignment[u] = 0
        return False

    if backtrack():
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
