```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Sort components by degree descending to optimize backtracking
    nodes = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    assignment = [0] * n
    
    def backtrack(idx):
        if idx == n:
            return True
        
        u = nodes[idx]
        
        # Find used colors by neighbors
        used = set()
        for v in adj[u]:
            if assignment[v] != 0:
                used.add(assignment[v])
        
        for color in [1, 2, 3]:
            if color not in used:
                assignment[u] = color
                if backtrack(idx + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
