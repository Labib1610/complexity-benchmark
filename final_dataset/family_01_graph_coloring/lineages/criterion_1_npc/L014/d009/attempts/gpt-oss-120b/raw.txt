import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    edges = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        w = int(next(it)) - 1
        edges[u].append(w)
        edges[w].append(u)

    # precompute forbidden color pairs
    colors = [1, 2, 3]
    forbidden = [[False]*3 for _ in range(3)]
    for i, x in enumerate(colors):
        for j, y in enumerate(colors):
            # compute V_u and V_w (u,w values cancel out in equality test)
            # we can set u=0,w=0 for simplicity because terms with u,w are linear in x,y and symmetric
            # Actually V_u - V_w simplifies to (x-y)*(x*y*(x-y) + (x**2 - y**2))
            # We'll just compute directly with u=0,w=0
            V_u = x*x * (0 * x + 0 * y) + y*y * (x*x + y)
            V_w = y*y * (0 * y + 0 * x) + x*x * (y*y + x)
            if V_u == V_w:
                forbidden[i][j] = True

    allowed = [[not forbidden[i][j] for j in range(3)] for i in range(3)]

    assign = [-1]*n
    # order vertices by degree descending
    order = sorted(range(n), key=lambda v: -len(edges[v]))

    # adjacency check helper
    def can_place(v, col):
        for nb in edges[v]:
            c2 = assign[nb]
            if c2 != -1 and not allowed[col][c2]:
                return False
        return True

    sys.setrecursionlimit(10000)
    def dfs(pos):
        if pos == n:
            return True
        v = order[pos]
        for col in range(3):
            if can_place(v, col):
                assign[v] = col
                if dfs(pos+1):
                    return True
                assign[v] = -1
        return False

    if dfs(0):
        print("YES")
        print(' '.join(str(c+1) for c in assign))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
