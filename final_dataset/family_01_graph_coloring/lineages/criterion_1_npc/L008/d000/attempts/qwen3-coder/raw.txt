import sys
from collections import defaultdict

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(set)
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Try all 3-colorings using backtracking
    color = [0] * (n + 1)  # color[i] represents the shift for operator i
    
    def is_valid(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
            
        for c in range(1, 4):
            if is_valid(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = 0
        return False
    
    # Optimization: start with node that has maximum degree
    max_degree_node = 1
    for i in range(1, n + 1):
        if len(adj[i]) > len(adj[max_degree_node]):
            max_degree_node = i
    
    # But we still need to assign all nodes, so let's just try standard backtracking
    if backtrack(1):
        print("YES")
        print(" ".join(map(str, color[1:])))
    else:
        print("NO")

solve()
