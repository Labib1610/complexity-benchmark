import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Precompute valid pairs (e1, e2)
    # Phi = e1^3 + e2^3, B = e1^2 + e2^2
    # Condition: Phi % B != 0
    valid_pairs = []
    for e1 in [1, 2, 3]:
        for e2 in [1, 2, 3]:
            phi = e1**3 + e2**3
            b = e1**2 + e2**2
            if phi % b != 0:
                valid_pairs.append((e1, e2))

    assignment = [0] * n

    def is_valid(u, val):
        for v_idx, v_val in enumerate(assignment):
            if v_val != 0:
                # Check if there is an edge between u and v_idx
                has_edge = False
                for edge in edges:
                    if (edge[0] == u and edge[1] == v_idx) or (edge[0] == v_idx and edge[1] == u):
                        has_edge = True
                        break
                if has_edge:
                    # Check if (val, v_val) is a valid pair
                    pair_ok = False
                    for p1, p2 in valid_pairs:
                        if p1 == val and p2 == v_val:
                            pair_ok = True
                            break
                    if not pair_ok:
                        return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for val in [1, 2, 3]:
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
