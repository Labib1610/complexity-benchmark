import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition is e_u + 2*e_v != 0 (mod 3).
    # Let x_i = e_i - 1, so x_i in {0, 1, 2}.
    # e_u + 2*e_v = (x_u + 1) + 2*(x_v + 1) = x_u + 2*x_v + 3 = x_u + 2*x_v (mod 3).
    # Condition: x_u + 2*x_v != 0 (mod 3).
    # This is equivalent to x_u != -2*x_v = x_v (mod 3).
    # Wait, let's recheck:
    # If x_u = 0, x_v != 0.
    # If x_u = 1, x_v != 1.
    # If x_u = 2, x_v != 2.
    # So for every edge (u, v), x_u != x_v.
    # This is exactly the definition of a graph coloring problem with 3 colors.
    # Since n <= 40, we can use backtracking with pruning.

    assignment = [-1] * n

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 0, 1, 2
        for color in range(3):
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = -1
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment[i] + 1 for i in range(n)))
    else:
        print("NO")

solve()
