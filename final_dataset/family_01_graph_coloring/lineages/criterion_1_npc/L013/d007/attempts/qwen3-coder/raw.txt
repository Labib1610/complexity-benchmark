def is_valid(e_u, e_v):
    phi = e_u**3 + e_v**3
    B = e_u**2 + e_v**2
    return phi % B != 0

def check_all_conditions(assignment, conditions):
    for u, v in conditions:
        if not is_valid(assignment[u-1], assignment[v-1]):
            return False
    return True

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    conditions = []
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        conditions.append((u, v))
        index += 2
    
    # Try all possible assignments
    # Since each emitter can have state 1, 2, or 3, we have 3^n possibilities
    # For n up to 40, this is too large, but we can try a more intelligent approach
    
    # Let's try backtracking with constraint checking
    def backtrack(pos, assignment):
        if pos == n:
            return assignment[:]
        
        for state in [1, 2, 3]:
            assignment[pos] = state
            # Check all conditions involving current position
            valid = True
            for u, v in conditions:
                if u-1 <= pos and v-1 <= pos:  # Both are assigned
                    if not is_valid(assignment[u-1], assignment[v-1]):
                        valid = False
                        break
            if valid:
                result = backtrack(pos + 1, assignment)
                if result:
                    return result
        return None
    
    # This might still be too slow for large n, let's try a different approach
    # Try random assignments for small number of attempts
    import random
    
    # First, let's see if a simple greedy approach works
    # Or just try many random assignments
    
    if n <= 10:  # For small n, we can try all combinations
        # Generate all possible assignments
        def generate_all(pos, current, all_assignments):
            if pos == n:
                all_assignments.append(current[:])
                return
            for state in [1, 2, 3]:
                current[pos] = state
                generate_all(pos + 1, current, all_assignments)
        
        all_assignments = []
        generate_all(0, [0]*n, all_assignments)
        
        for assignment in all_assignments:
            if check_all_conditions(assignment, conditions):
                print("YES")
                print(" ".join(map(str, assignment)))
                return
    else:
        # For larger n, try random assignments
        for _ in range(10000):  # Try 10000 random assignments
            assignment = [random.choice([1, 2, 3]) for _ in range(n)]
            if check_all_conditions(assignment, conditions):
                print("YES")
                print(" ".join(map(str, assignment)))
                return
    
    print("NO")

solve()
