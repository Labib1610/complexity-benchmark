```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            continue
        adj[u].add(v)
        adj[v].add(u)

    order = sorted(range(n), key=lambda x: -len(adj[x]))
    color = [0] * n  # 0 = unassigned, else 1..3

    # Precompute neighbor list in order indices for fast check
    pos_in_order = {v: i for i, v in enumerate(order)}

    # adjacency restricted to later vertices in order for forward checking
    later_adj = [[] for _ in range(n)]
    for i, v in enumerate(order):
        for nb in adj[v]:
            if pos_in_order[nb] > i:
                later_adj[v].append(nb)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = order[idx]
        # try colors 1..3
        for c in (1, 2, 3):
            ok = True
            for nb in adj[v]:
                if color[nb] == c:
                    ok = False
                    break
            if not ok:
                continue
            color[v] = c
            # simple forward checking: ensure each uncolored neighbor still has at least one possible color
            fail = False
            for nb in later_adj[v]:
                if color[nb] == 0:
                    # check if nb has at least one color different from already colored neighbors
                    possible = False
                    for cc in (1, 2, 3):
                        conflict = False
                        for nn in adj[nb]:
                            if color[nn] == cc:
                                conflict = True
                                break
                        if not conflict:
                            possible = True
                            break
                    if not possible:
                        fail = True
                        break
            if not fail and dfs(idx + 1):
                return True
            color[v] = 0
        return False

    if dfs(0):
        print("YES")
        # output in original order 1..n
        print(' '.join(str(color[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
