import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    conditions = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        conditions.append((u, v))
        idx += 2
        
    # Precompute forbidden pairs for each (u, v)
    # e_u, e_v in {1, 2, 3}
    # Condition: (u * e_u + v * e_v) % (u + v) != 0
    forbidden = [[] for _ in range(n + 1)]
    for u, v in conditions:
        forbidden_states = []
        for eu in [1, 2, 3]:
            for ev in [1, 2, 3]:
                if (u * eu + v * ev) % (u + v) == 0:
                    forbidden_states.append((eu, ev))
        forbidden.append((u, v, forbidden_states))

    assignment = [0] * (n + 1)

    def is_valid(idx, val):
        assignment[idx] = val
        for u, v, f_states in forbidden:
            if u <= idx and v <= idx:
                if (assignment[u], assignment[v]) in f_states:
                    return False
        return True

    def backtrack(idx):
        if idx > n:
            return True
        
        for val in [1, 2, 3]:
            # Check constraints involving current idx and already assigned nodes
            possible = True
            for u, v, f_states in forbidden:
                if u == idx and v < idx:
                    if (val, assignment[v]) in f_states:
                        possible = False; break
                elif v == idx and u < idx:
                    if (assignment[u], val) in f_states:
                        possible = False; break
            
            if possible:
                assignment[idx] = val
                if backtrack(idx + 1):
                    return True
        
        assignment[idx] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
