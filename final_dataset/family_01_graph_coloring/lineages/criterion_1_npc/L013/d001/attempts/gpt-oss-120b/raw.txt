```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # order vertices by degree descending
    order = sorted(range(n), key=lambda x: -adj[x].bit_count())
    color = [-1] * n
    # map from position in order to vertex
    pos2v = order

    sys.setrecursionlimit(10000)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = pos2v[idx]
        # compute forbidden colors from already colored neighbors
        forbid = 0
        nb = adj[v]
        while nb:
            u = (nb & -nb).bit_length() - 1
            nb &= nb - 1
            c = color[u]
            if c != -1:
                forbid |= 1 << c
        for c in range(3):
            if not (forbid >> c) & 1:
                color[v] = c
                if dfs(idx + 1):
                    return True
                color[v] = -1
        return False

    possible = dfs(0)
    out = []
    if possible:
        out.append("YES")
        out.append(' '.join(str(c + 1) for c in color))
    else:
        out.append("NO")
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
```
