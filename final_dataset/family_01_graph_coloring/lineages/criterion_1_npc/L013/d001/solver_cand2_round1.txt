import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_safe(u, shift):
        for v in adj[u]:
            if assignment[v] == shift:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning shift 1, 2, 3
        for shift in range(1, 4):
            if is_safe(u, shift):
                assignment[u] = shift
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Optimization: Sort nodes by degree descending to fail faster
    # However, with n=40, simple backtracking might be slow.
    # Let's use a simple heuristic: order by degree
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to the sorted order
    mapping = {node: i for i, node in enumerate(nodes)}
    new_adj = [[] for _ in range(n)]
    for u in range(1, n + 1):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    res = [0] * n
    
    def backtrack_optimized(idx):
        if idx == n:
            return True
        
        u = idx
        for shift in range(1, 4):
            safe = True
            for v in new_adj[u]:
                if res[v] == shift:
                    safe = False
                    break
            if safe:
                res[u] = shift
                if backtrack_optimized(idx + 1):
                    return True
                res[u] = 0
        return False

    if backtrack_optimized(0):
        print("YES")
        final_assignment = [0] * n
        for i in range(n):
            final_assignment[nodes[i]-1] = res[i]
        print(*(final_assignment))
    else:
        print("NO")

solve()
