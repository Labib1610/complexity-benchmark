import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    conditions = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        conditions.append((u, v))
        idx += 2

    # e_i in {1, 2, 3}
    # e_i^2 in {1, 4, 9}
    # F1 = (u*e_u^2 + v*e_v^2) % 41
    # F2 = (u*e_v^2 + v*e_u^2) % 41
    # F1 != F2 <=> (u-v)*e_u^2 != (u-v)*e_v^2 (mod 41)
    # Since u != v, u-v != 0 (mod 41) for 1 <= u, v <= 40
    # So e_u^2 != e_v^2 (mod 41)
    # e_i^2 mod 41: 1^2=1, 2^2=4, 3^2=9
    # The condition simplifies to e_u^2 != e_v^2
    # Since 1, 4, 9 are distinct, this is equivalent to e_u != e_v
    
    # This is a graph coloring problem with 3 colors.
    # Since n <= 40, we can use backtracking.
    
    adj = [[] for _ in range(n + 1)]
    for u, v in conditions:
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
