def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # If no conditions, any assignment works
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    conditions = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        conditions.append((u, v))
        index += 2
    
    # The constraint (e_u - e_v)^2 >= 1 means e_u != e_v
    # So we need to 3-color a graph where each edge represents u and v must have different colors
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in conditions:
        adj[u].append(v)
        adj[v].append(u)
    
    # Try all 3^n assignments? No, that's too slow for n=40
    # We need to do a backtracking search with pruning
    
    # Since n <= 40, but 3^40 is too large, let's try smart backtracking
    # Actually, let's model this as a graph coloring problem and use backtracking
    
    color = [-1] * n  # -1 means unassigned
    
    def is_safe(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
        
        for c in [1, 2, 3]:
            if is_safe(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = -1
        return False
    
    # To make backtracking faster, we should pick the node with the most constraints first
    # Let's sort nodes by degree
    degrees = [(len(adj[i]), i) for i in range(n)]
    degrees.sort(reverse=True)
    order = [x[1] for x in degrees]
    
    # Reorder the adjacency list based on this heuristic
    # But for simplicity in recursion, we'll just use the original indices
    
    # A better approach: we can try to find a valid coloring using a greedy method first,
    # then fall back to backtracking if needed.
    
    # For small graphs, simple backtracking might work
    # Let's proceed with basic backtracking but with a good variable ordering
    
    # Sort variables by how constrained they are (most connected first)
    def solve():
        # Use the degree-based order
        def backtrack_ordered(idx):
            if idx == n:
                return True
            node = order[idx]
            for c in [1, 2, 3]:
                if is_safe(node, c):
                    color[node] = c
                    if backtrack_ordered(idx + 1):
                        return True
                    color[node] = -1
            return False
            
        return backtrack_ordered(0)
    
    if solve():
        print("YES")
        # Map back to 1-based indexing in output
        print(" ".join(map(str, color)))
    else:
        print("NO")

if __name__ == "__main__":
    main()
