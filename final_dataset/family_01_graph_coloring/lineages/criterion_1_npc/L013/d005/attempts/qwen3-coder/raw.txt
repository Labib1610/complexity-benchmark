def is_resonant(eu, ev):
    # Calculate e_u^(e_v) + e_v^(e_u) + e_u * e_v
    val = pow(eu, ev) + pow(ev, eu) + eu * ev
    return val % 3 == 0

def solve():
    n, m = map(int, input().split())
    
    # If no conditions, any assignment works
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Precompute which pairs are resonant
    resonant = [[False] * 4 for _ in range(4)]
    for i in range(1, 4):
        for j in range(1, 4):
            resonant[i][j] = is_resonant(i, j)
    
    # Try all 3^n assignments
    for mask in range(3**n):
        assignment = []
        temp_mask = mask
        for i in range(n):
            assignment.append((temp_mask % 3) + 1)
            temp_mask //= 3
        
        # Check if this assignment satisfies all constraints
        valid = True
        for u, v in edges:
            eu = assignment[u - 1]
            ev = assignment[v - 1]
            if resonant[eu][ev]:
                valid = False
                break
        
        if valid:
            print("YES")
            print(" ".join(map(str, assignment)))
            return
    
    print("NO")

solve()
