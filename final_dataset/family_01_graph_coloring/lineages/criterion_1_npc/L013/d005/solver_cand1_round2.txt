import sys

def solve():
    # The condition:
    # s_u * X + Y = s_v
    # s_v * X + Y = s_u
    # Subtracting the two: (s_u - s_v) * X = s_v - s_u
    # If s_u != s_v, then X = -1.
    # Substituting X = -1: -s_u + Y = s_v => Y = s_u + s_v.
    # This system always has a unique solution (X=-1, Y=s_u+s_v) if s_u != s_v.
    # If s_u == s_v, the equations become:
    # s_u * X + Y = s_u
    # s_u * X + Y = s_u
    # This is one equation: s_u * X + Y = s_u.
    # This has infinitely many solutions (e.g., X=1, Y=0 or X=0, Y=s_u).
    # Thus, the condition "unique solution" is satisfied if and only if s_u != s_v.
    # The problem reduces to: Can we color the graph with 3 colors such that
    # every edge (u, v) has s_u != s_v?
    # This is equivalent to checking if the graph is 3-colorable.
    # Since n <= 40, we can use backtracking with pruning.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Heuristic: try colors that are most constrained or just 1, 2, 3
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
