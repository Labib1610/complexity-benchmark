```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # W(eu, ev) = (eu - ev)^4 - 5(eu - ev)^2 + 7
    # Let d = |eu - ev|
    # If d = 0: W = 0 - 0 + 7 = 7
    # If d = 1: W = 1 - 5 + 7 = 3
    # If d = 2: W = 16 - 20 + 7 = 3
    # We want sum(W) <= 3m.
    # Since W is 3 if |eu - ev| >= 1 and 7 if |eu - ev| = 0,
    # the condition sum(W) <= 3m is equivalent to saying
    # that for every edge (u, v), |eu - ev| must be >= 1.
    # This means eu != ev for all edges (u, v).
    # This is exactly the 3-coloring problem.

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
