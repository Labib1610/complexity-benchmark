def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency matrix A
    A = [[0] * n for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        A[u][v] = 1
        A[v][u] = 1
        idx += 2
    
    # Try all possible assignments of excitation states (1, 2, 3)
    # Since n <= 40, brute force is infeasible. However, we can try a smart approach.
    # But given the constraints and nature of the problem, let's attempt a mathematical approach.
    
    # Let's analyze what C[i][j] = 0 means for all i,j
    # C = X^2 A + A X^2 + X A X - 6(X A + A X) + 11 A
    # For i != j, A[i][j] = 0 or 1
    # If A[i][j] = 0, then C[i][j] = 0 automatically
    # If A[i][j] = 1, then we need C[i][j] = 0
    # C[i][j] = X^2[i][i] * A[i][j] + A[i][j] * X^2[j][j] + X[i][i] * A[i][j] * X[j][j] - 6*(X[i][i] * A[i][j] + A[i][j] * X[j][j]) + 11 * A[i][j]
    # Since A[i][j] = 1:
    # C[i][j] = e_i^2 * 1 + 1 * e_j^2 + e_i * 1 * e_j - 6*(e_i * 1 + 1 * e_j) + 11 * 1
    # C[i][j] = e_i^2 + e_j^2 + e_i * e_j - 6*e_i - 6*e_j + 11
    # This must be 0 for every edge (i,j)
    # So for every edge (i,j): e_i^2 + e_j^2 + e_i * e_j - 6*e_i - 6*e_j + 11 = 0
    
    # For diagonal entries i=j:
    # C[i][i] = X^2[i][i] * A[i][i] + A[i][i] * X^2[i][i] + X[i][i] * A[i][i] * X[i][i] - 6*(X[i][i] * A[i][i] + A[i][i] * X[i][i]) + 11 * A[i][i]
    # Since A[i][i] = 0, C[i][i] = 0 automatically
    
    # So we only need to satisfy the equations for edges:
    # For each edge (u,v): e_u^2 + e_v^2 + e_u * e_v - 6*e_u - 6*e_v + 11 = 0
    
    # Let's precompute which pairs (a,b) with a,b in {1,2,3} satisfy:
    # a^2 + b^2 + a*b - 6*a - 6*b + 11 = 0
    valid_pairs = set()
    for a in [1, 2, 3]:
        for b in [1, 2, 3]:
            if a**2 + b**2 + a*b - 6*a - 6*b + 11 == 0:
                valid_pairs.add((a, b))
                valid_pairs.add((b, a))  # Since the equation is symmetric
    
    # Now we model this as a graph coloring problem:
    # Each node is an emitter, and for every edge, the states at the endpoints must form a valid pair
    # This is a constraint satisfaction problem. We can use backtracking.
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    edges = []
    for i in range(n):
        for j in range(i+1, n):
            if A[i][j] == 1:
                adj[i].append(j)
                adj[j].append(i)
                edges.append((i, j))
    
    # If there are no edges, any assignment works
    if not edges:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Backtracking function to assign states
    def is_valid_assignment(states):
        for u, v in edges:
            if (states[u], states[v]) not in valid_pairs:
                return False
        return True
    
    # Try all possible assignments using backtracking
    def backtrack(node, states):
        if node == n:
            return is_valid_assignment(states)
        
        for state in [1, 2, 3]:
            states[node] = state
            # Check if this assignment is consistent with all neighbors so far
            ok = True
            for neighbor in adj[node]:
                if states[neighbor] != 0:  # already assigned
                    if (states[node], states[neighbor]) not in valid_pairs:
                        ok = False
                        break
            if ok:
                if backtrack(node + 1, states):
                    return True
        states[node] = 0
        return False
    
    # Initialize states array (0 means unassigned)
    states = [0] * n
    
    # Start backtracking from node 0
    if backtrack(0, states):
        print("YES")
        print(" ".join(map(str, states)))
    else:
        print("NO")

solve()
