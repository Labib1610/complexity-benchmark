import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning cycle 1, 2, or 3
        neighbor_colors = 0
        for v in range(n):
            if (adj[u] & (1 << v)) and assignment[v] != 0:
                neighbor_colors |= (1 << (assignment[v] - 1))
        
        for color in range(1, 4):
            if not (neighbor_colors & (1 << (color - 1))):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Heuristic: sort nodes by degree to prune search space
    degrees = [(bin(adj[i]).count('1'), i) for i in range(n)]
    degrees.sort(key=lambda x: x[0], reverse=True)
    order = [x[1] for x in degrees]
    
    # Re-map to process in order
    new_assignment = [0] * n
    
    def backtrack_ordered(idx_in_order):
        if idx_in_order == n:
            return True
        
        u = order[idx_in_order]
        neighbor_colors = 0
        for v in range(n):
            if (adj[u] & (1 << v)) and new_assignment[v] != 0:
                neighbor_colors |= (1 << (new_assignment[v] - 1))
        
        for color in range(1, 4):
            if not (neighbor_colors & (1 << (color - 1))):
                new_assignment[u] = color
                if backtrack_ordered(idx_in_order + 1):
                    return True
                new_assignment[u] = 0
        return False

    if backtrack_ordered(0):
        print("YES")
        print(*(new_assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
