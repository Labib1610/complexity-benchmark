```python
import sys

# The problem asks for a 3-coloring of a graph with up to 40 nodes.
# Since 3-coloring is NP-complete, but n is small (up to 40),
# we can use a backtracking approach with pruning.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # Sort nodes by degree descending to improve pruning
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(idx_in_nodes):
        if idx_in_nodes == n:
            return True
        
        u = nodes[idx_in_nodes]
        
        # Try colors 1, 2, 3
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(idx_in_nodes + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors[i] for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
