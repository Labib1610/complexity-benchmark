import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # F(s_u, s_v) = s_u^3 + s_v^3 - s_u^2 s_v - s_u s_v^2
    # F(s_u, s_v) = (s_u + s_v)(s_u^2 - s_u s_v + s_v^2) - s_u s_v(s_u + s_v)
    # F(s_u, s_v) = (s_u + s_v)(s_u^2 - 2s_u s_v + s_v^2) = (s_u + s_v)(s_u - s_v)^2
    
    # We need F(s_u, s_v) mod 7 != 0 for all edges.
    # Possible values for s in {1, 2, 3}:
    # s=1: F(1,1)=0, F(1,2)=3(1)=3, F(1,3)=4(4)=16=2
    # s=2: F(2,1)=3, F(2,2)=0, F(2,3)=5(1)=5
    # s=3: F(3,1)=2, F(3,2)=5, F(3,3)=0
    
    # F(s_u, s_v) is 0 mod 7 if s_u = s_v.
    # So we must have s_u != s_v for all edges (u, v).
    # This means the graph must be bipartite.
    
    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v_idx in range(m):
                    u_edge, v_edge = edges[v_idx]
                    if u_edge == u:
                        if color[v_edge] == -1:
                            color[v_edge] = 1 - c
                            stack.append((v_edge, 1 - c))
                        elif color[v_edge] == c:
                            print("NO")
                            return
                    elif v_edge == u:
                        if color[u_edge] == -1:
                            color[u_edge] = 1 - c
                            stack.append((u_edge, 1 - c))
                        elif color[u_edge] == c:
                            print("NO")
                            return
                            
    # If bipartite, color 0 nodes get 1, color 1 nodes get 2.
    # Check if F(1, 2) = 3 mod 7 != 0. Yes.
    # Check if F(1, 3) = 2 mod 7 != 0. Yes.
    # Check if F(2, 3) = 5 mod 7 != 0. Yes.
    # So any assignment of {1, 2} to the two partitions works.
    
    res = []
    for i in range(n):
        if color[i] == 0:
            res.append(1)
        else:
            res.append(2)
            
    print("YES")
    print(*(res))

solve()
