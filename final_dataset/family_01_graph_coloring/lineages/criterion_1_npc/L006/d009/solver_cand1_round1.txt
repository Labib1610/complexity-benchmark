import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # W_j = s_u^3 + s_v^3 - s_u^2 s_v - s_u s_v^2
    # W_j = (s_u - s_v)^2 * (s_u + s_v)
    # Possible values for s in {1, 2, 3}:
    # If s_u = s_v, W_j = 0.
    # If {s_u, s_v} = {1, 2}: W_j = (1-2)^2 * (1+2) = 1 * 3 = 3
    # If {s_u, s_v} = {1, 3}: W_j = (1-3)^2 * (1+3) = 4 * 4 = 16 = 2 (mod 7)
    # If {s_u, s_v} = {2, 3}: W_j = (2-3)^2 * (2+3) = 1 * 5 = 5
    
    # Z_m = product(W_j) mod 7. We need Z_m != 0.
    # Z_m != 0 iff all W_j != 0, which means s_u != s_v for all edges.
    # This is equivalent to saying the graph is 3-colorable using {1, 2, 3}.
    # Since n <= 40, we can use a randomized approach or backtracking.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    # Try to 3-color the graph. If it's not 3-colorable, output NO.
    # For n=40, a simple backtracking with pruning or randomized search works.
    assignment = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if assignment[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3 in random order
        colors = [1, 2, 3]
        random.shuffle(colors)
        for c in colors:
            if is_valid(u, c):
                assignment[u] = c
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # The problem asks for Z_m != 0. If the graph is not 3-colorable,
    # then for any assignment, there exists an edge with s_u = s_v,
    # making W_j = 0, so Z_m = 0.
    # Thus, we just need to check if the graph is 3-colorable.
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
