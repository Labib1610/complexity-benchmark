import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # I(u, v) = (s_u - s_v)^2 * (5 - (s_u - s_v)^2)
    # Possible values for d = |s_u - s_v|:
    # d = 0: I = 0 * 5 = 0
    # d = 1: I = 1 * (5 - 1) = 4
    # d = 2: I = 4 * (5 - 4) = 4
    # So I(u, v) = 4 if s_u != s_v, and 0 if s_u == s_v.
    # We want sum(I(u, v)) = 4 * m.
    # Since each I(u, v) is either 0 or 4, we need I(u, v) = 4 for all m edges.
    # This means s_u != s_v for all edges (u, v).
    # This is equivalent to checking if the graph is 2-colorable (bipartite).
    # However, we have 3 states {1, 2, 3}.
    # If the graph is bipartite, we can color it with 1 and 2.
    # If it's not bipartite, we might use 3 to satisfy the condition.
    
    # Actually, the condition is s_u != s_v for all edges.
    # This is exactly the definition of a graph coloring problem with 3 colors.
    # Since n <= 40, we can use a backtracking approach or meet-in-the-middle.
    # Given the constraints and the nature of the problem, a simple backtracking
    # with pruning or a randomized approach works well.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        for color in [1, 2, 3]:
            valid = True
            for v in adj[u]:
                if assignment[v] == color:
                    valid = False
                    break
            if valid:
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
