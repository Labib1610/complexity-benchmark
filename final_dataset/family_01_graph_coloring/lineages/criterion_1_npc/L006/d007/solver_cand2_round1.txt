import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Classification string: XOOXCOCCX (1-indexed)
    # 1: X, 2: O, 3: O, 4: X, 5: C, 6: O, 7: C, 8: C, 9: X
    # Forbidden products (X): 1, 4, 9
    # Possible states: 1, 2, 3
    # Forbidden pairs (s_u * s_v):
    # 1*1 = 1 (X) -> Forbidden
    # 1*2 = 2 (O) -> OK
    # 1*3 = 3 (O) -> OK
    # 2*1 = 2 (O) -> OK
    # 2*2 = 4 (X) -> Forbidden
    # 2*3 = 6 (O) -> OK
    # 3*1 = 3 (O) -> OK
    # 3*2 = 6 (O) -> OK
    # 3*3 = 9 (X) -> Forbidden
    
    # Forbidden: (1,1), (2,2), (3,3)
    # This is equivalent to a 3-coloring problem where adjacent nodes must have different colors.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
