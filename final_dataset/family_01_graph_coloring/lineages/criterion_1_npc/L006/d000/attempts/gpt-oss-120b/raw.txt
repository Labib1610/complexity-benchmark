```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    # order vertices by degree descending for better pruning
    order = sorted(range(n), key=lambda x: -len(adj[x]))
    color = [0] * n
    sys.setrecursionlimit(10000)

    def dfs(pos: int) -> bool:
        if pos == n:
            return True
        v = order[pos]
        # try colors 1..3
        used = {color[nb] for nb in adj[v] if color[nb] != 0}
        for c in (1, 2, 3):
            if c not in used:
                color[v] = c
                if dfs(pos + 1):
                    return True
                color[v] = 0
        return False

    if dfs(0):
        # output in original order (1-indexed workers)
        out = ' '.join(str(color[i]) for i in range(n))
        sys.stdout.write("YES\n" + out + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == "__main__":
    solve()
```
