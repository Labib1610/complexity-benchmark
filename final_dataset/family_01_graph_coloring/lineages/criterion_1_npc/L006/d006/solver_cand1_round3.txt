import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Destructive harmonic resonance condition:
    # (c_u^2 + c_v^2) % (c_u * c_v) == 0
    # Possible values for c_i are {1, 2, 3}
    # Pairs (c_u, c_v):
    # (1, 1): (1+1) % 1 == 0 -> Destructive
    # (1, 2): (1+4) % 2 != 0 -> Stable
    # (1, 3): (1+9) % 3 != 0 -> Stable
    # (2, 1): (4+1) % 2 != 0 -> Stable
    # (2, 2): (4+4) % 4 == 0 -> Destructive
    # (2, 3): (4+9) % 6 != 0 -> Stable
    # (3, 1): (9+1) % 3 != 0 -> Stable
    # (3, 2): (9+4) % 6 != 0 -> Stable
    # (3, 3): (9+9) % 9 == 0 -> Destructive
    
    # So, c_u == c_v is forbidden.
    # This is equivalent to a graph coloring problem with 3 colors.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
