```python
import sys

def solve():
    # The condition (c_u ^ c_v) + c_u = c_v is equivalent to:
    # If c_u = 1, c_v = 2 or 3.
    # If c_u = 2, c_v = 3.
    # If c_u = 3, no c_v satisfies this.
    #
    # Let's analyze the directed edges:
    # 1 -> 2, 1 -> 3, 2 -> 3.
    # If c_u = 1 and c_v = 2, edge 1 -> 2.
    # If c_u = 1 and c_v = 3, edge 1 -> 3.
    # If c_u = 2 and c_v = 3, edge 2 -> 3.
    #
    # This is a strict ordering: 1 < 2 < 3.
    # If we assign values such that c_u < c_v, we get a directed edge u -> v.
    # If c_u > c_v, we get v -> u.
    # If c_u = c_v, no edge is formed.
    #
    # To avoid cycles, we need to assign values such that the graph is a DAG.
    # Since we only have 3 values, we can partition the vertices into 3 sets:
    # S1 (val 1), S2 (val 2), S3 (val 3).
    # Edges only exist between different sets.
    # If u in Si, v in Sj, and i < j, edge is u -> v.
    # If i > j, edge is v -> u.
    # If i = j, no edge.
    #
    # This is equivalent to finding a 3-coloring of the graph such that
    # if we orient edges from lower color to higher color, there are no cycles.
    # Actually, any 3-coloring works because the edges are always directed
    # from lower color to higher color. A cycle would imply a sequence of
    # colors c1 < c2 < ... < ck < c1, which is impossible.
    #
    # So the problem is simply: can we 3-color the graph?
    # Wait, the condition is:
    # (c_u ^ c_v) + c_u = c_v
    # If c_u=1, c_v=2: (1^2)+1 = 3+1 = 4 != 2. Wait.
    # Let's re-evaluate:
    # c_u=1, c_v=2: (1^2)+1 = 3+1 = 4. 4 != 2.
    # c_u=1, c_v=3: (1^3)+1 = 2+1 = 3. 3 == 3. (Edge 1 -> 3)
    # c_u=2, c_v=3: (2^3)+2 = 1+2 = 3. 3 == 3. (Edge 2 -> 3)
    # c_u=1, c_v=2: (1^2)+1 = 4. No.
    #
    # Let's re-check:
    # 1^2 = 3. 3+1 = 4 != 2.
    # 1^3 = 2. 2+1 = 3 == 3. (1->3)
    # 2^3 = 1. 1+2 = 3 == 3. (2->3)
    # 1^2 = 3. 3+2 = 5 != 1.
    #
    # The only edges are 1->3 and 2->3.
    # This means if we assign 1 or 2, they never have an edge between them.
    # If we assign 3, it can have incoming edges from 1 and 2.
    # This is always a DAG!
    
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We can use 2-coloring to split into {1, 2} and {3}.
    # If we color with 1 and 3, we get no edges.
    # If we color with 2 and 3, we get edges 2->3.
    # If we color with 1 and 2, we get no edges.
    #
    # Let's try to 2-color the graph. If bipartite, use 1 and 3.
    # If not bipartite, we need 3.
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        # Not bipartite, use 3rd color
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
