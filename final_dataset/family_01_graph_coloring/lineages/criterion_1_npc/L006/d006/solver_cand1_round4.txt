import sys

def solve():
    # The condition s_u^3 + s_v <= s_v * s_u^2 + s_u
    # can be rewritten as:
    # s_u^3 - s_u <= s_v * s_u^2 - s_v
    # s_u(s_u^2 - 1) <= s_v(s_u^2 - 1)
    # If s_u = 1: 1(0) <= s_v(0) => 0 <= 0 (Always true)
    # If s_u = 2: 2(3) <= s_v(3) => 6 <= 3*s_v => 2 <= s_v
    # If s_u = 3: 3(8) <= s_v(8) => 24 <= 8*s_v => 3 <= s_v
    
    # Summary of directed channel u -> v:
    # If s_u = 1: Always true
    # If s_u = 2: True if s_v >= 2
    # If s_u = 3: True if s_v = 3
    
    # Let's check the cycle condition.
    # If we assign s_i based on a topological sort of the graph,
    # we can avoid cycles.
    # Actually, if we assign s_i = 1 for all i, we get cycles if the graph has edges.
    # If we assign s_i = 3 for all i, we get cycles if the graph has edges.
    # If we assign s_i based on the rank in a topological sort,
    # we can try to map ranks to {1, 2, 3}.
    
    # A simpler approach:
    # If we assign s_i = 1, 2, or 3 based on the node index,
    # we can try to see if a specific ordering works.
    # Actually, the problem is equivalent to finding an ordering of nodes
    # such that for every edge (u, v), we don't have cycles.
    # If we assign s_i = 1 for all i, every edge (u, v) creates a cycle (u->v and v->u).
    # If we assign s_i = 3 for all i, every edge (u, v) creates a cycle (u->v and v->u).
    # If we assign s_i = 2 for all i, every edge (u, v) creates a cycle (u->v and v->u).
    
    # Wait, the condition is:
    # u -> v if f(s_u, s_v)
    # v -> u if f(s_v, s_u)
    # If we set s_i = 1 for some nodes and s_i = 3 for others:
    # If s_u = 1, s_v = 3:
    # u -> v: f(1, 3) is True (0 <= 0)
    # v -> u: f(3, 1) is False (24 <= 8 is False)
    # So if we have a DAG, we can assign s_i = 1 to nodes "early" in the topo sort
    # and s_i = 3 to nodes "late" in the topo sort.
    
    import sys
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Try to find a topological ordering using Kahn's or DFS
    # Since n is small (40), we can just use a simple greedy approach
    # or check if the graph is bipartite? No, it's not necessarily bipartite.
    # Actually, if we can partition the graph into two sets A and B such that
    # all edges are between A and B, we can set s_A = 1 and s_B = 3.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    res = []
    for i in range(1, n + 1):
        res.append(str(1 if color[i] == 0 else 3))
    print(" ".join(res))

solve()
