```python
import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    def get_energy(s1, s2):
        diff = s1 - s2
        return 5 * (diff**4) - 25 * (diff**2) + 13

    # E(u, v) values:
    # s1=s2: diff=0 -> 13
    # |s1-s2|=1: diff=1 or -1 -> 5(1)-25(1)+13 = -7
    # |s1-s2|=2: diff=2 or -2 -> 5(16)-25(4)+13 = 80-100+13 = -7
    # Wait, let's recheck:
    # If |s1-s2| = 1: 5(1)^4 - 25(1)^2 + 13 = 5 - 25 + 13 = -7
    # If |s1-s2| = 2: 5(2)^4 - 25(2)^2 + 13 = 5(16) - 25(4) + 13 = 80 - 100 + 13 = -7
    # If s1=s2: 5(0)^4 - 25(0)^2 + 13 = 13
    
    # So E(u, v) = -7 if s_u != s_v, and 13 if s_u == s_v.
    # We want sum E(u, v) = -7 * m.
    # Let k be the number of edges where s_u == s_v.
    # Total energy = (m - k) * (-7) + k * 13 = -7m + 7k + 13k = -7m + 20k.
    # We want -7m + 20k = -7m, which implies 20k = 0, so k = 0.
    # We need to assign states such that no two connected nodes have the same state.
    # This is equivalent to checking if the graph is 3-colorable.
    
    # Since n <= 40, we can use a randomized approach or backtracking.
    # 3-coloring is NP-complete, but n=40 is small enough for a heuristic.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        colors = [1, 2, 3]
        random.shuffle(colors)
        for c in colors:
            can_use = True
            for v in adj[u]:
                if assignment[v] == c:
                    can_use = False
                    break
            if can_use:
                assignment[u] = c
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Try a few times with random order
    for _ in range(1000):
        if backtrack(0):
            print("YES")
            print(*(assignment))
            return
            
    print("NO")

solve()
```
