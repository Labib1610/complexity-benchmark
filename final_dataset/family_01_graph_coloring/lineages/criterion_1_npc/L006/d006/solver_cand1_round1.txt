```python
import sys

def solve():
    # The condition for a directed channel from u to v:
    # c_u^2 * c_v + c_v^3 + 12*c_u + 10*c_v = 6*c_u^2 + 6*c_v^2 + 12
    # Let f(u, v) = c_u^2 * c_v + c_v^3 + 12*c_u + 10*c_v - 6*c_u^2 - 6*c_v^2 - 12
    # Let's evaluate f(c_u, c_v) for c_u, c_v in {1, 2, 3}:
    # f(1, 1) = 1 + 1 + 12 + 10 - 6 - 6 - 12 = 0
    # f(1, 2) = 4 + 8 + 12 + 20 - 6 - 24 - 12 = 2
    # f(1, 3) = 9 + 27 + 12 + 30 - 6 - 54 - 12 = 6
    # f(2, 1) = 4 + 1 + 24 + 10 - 24 - 6 - 12 = -3
    # f(2, 2) = 8 + 8 + 24 + 20 - 24 - 24 - 12 = 0
    # f(2, 3) = 18 + 27 + 24 + 30 - 24 - 54 - 12 = 9
    # f(3, 1) = 9 + 1 + 36 + 10 - 54 - 6 - 12 = -16
    # f(3, 2) = 18 + 8 + 36 + 20 - 54 - 24 - 12 = -8
    # f(3, 3) = 27 + 27 + 36 + 30 - 54 - 54 - 12 = 0
    
    # A channel u -> v exists if f(c_u, c_v) == 0.
    # f(c_u, c_v) == 0 only when c_u == c_v.
    # If c_u == c_v, then f(c_u, c_v) is always 0.
    # This means if c_u == c_v, there is a channel u -> v AND v -> u.
    # A cycle of length 2 (u -> v -> u) is formed if c_u == c_v.
    # To avoid cycles, we must have c_u != c_v for all tested pairs (u, v).
    # This is equivalent to the graph coloring problem with 3 colors.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
