```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Try to 3-color the graph
    color = [-1] * n  # -1 means uncolored, 0,1,2 are colors
    
    def dfs(u, c):
        color[u] = c
        for v in adj[u]:
            if color[v] == -1:
                if not dfs(v, 1 - c if c != 1 else 2):  # Assign a different color
                    return False
            elif color[v] == c:
                return False
        return True
    
    # Check each connected component
    valid = True
    for i in range(n):
        if color[i] == -1:
            if not dfs(i, 0):
                valid = False
                break
    
    if not valid:
        print("NO")
    else:
        print("YES")
        result = [str(c + 1) for c in color]
        print(" ".join(result))

if __name__ == "__main__":
    main()
```
