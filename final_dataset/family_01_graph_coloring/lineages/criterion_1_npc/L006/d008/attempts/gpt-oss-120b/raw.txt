```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # precompute forbidden pairs (a,b) where equation holds
    forb = [[False]*4 for _ in range(4)]
    for a in (1,2,3):
        for b in (1,2,3):
            left = (a**3 + b**3) % 7
            right = (a*a*b + a*b*b) % 7
            if left == right:
                forb[a][b] = True

    # domain bitmask: bit 1<<c (c=1..3)
    full_mask = 0b1110  # bits 1..3 used (ignore bit0)
    domains = [full_mask]*n
    assign = [0]*n
    # order of vertices for heuristic: degree descending
    order = sorted(range(n), key=lambda x: -len(adj[x]))

    # helper to get first unassigned vertex with smallest domain size
    def select_var():
        best = -1
        best_cnt = 10
        for v in order:
            if assign[v]==0:
                cnt = bin(domains[v]).count('1')
                if cnt < best_cnt:
                    best_cnt = cnt
                    best = v
                    if cnt == 1:
                        break
        return best

    sys.setrecursionlimit(10000)
    def dfs():
        if all(assign[v]!=0 for v in range(n)):
            return True
        v = select_var()
        mask = domains[v]
        # try each possible color
        for c in (1,2,3):
            if mask & (1<<c):
                # record changes
                changes = []
                assign[v] = c
                ok = True
                for nb in adj[v]:
                    if assign[nb]==0:
                        # remove colors that together with c are forbidden
                        forbid_mask = 0
                        for d in (1,2,3):
                            if forb[c][d]:
                                forbid_mask |= 1<<d
                        new_mask = domains[nb] & ~forbid_mask
                        if new_mask != domains[nb]:
                            changes.append((nb, domains[nb]))
                            domains[nb] = new_mask
                        if domains[nb]==0:
                            ok = False
                            break
                    else:
                        if forb[c][assign[nb]]:
                            ok = False
                            break
                if ok and dfs():
                    return True
                # revert
                for nb, old in reversed(changes):
                    domains[nb] = old
                assign[v] = 0
        return False

    if dfs():
        print("YES")
        print(' '.join(str(assign[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
