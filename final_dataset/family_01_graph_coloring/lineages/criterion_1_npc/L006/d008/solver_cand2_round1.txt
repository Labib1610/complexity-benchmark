```python
import sys

def solve():
    # The matrix M = [[s_u, -s_v], [s_v, s_u]]
    # Let z = s_u + i * s_v. Then M corresponds to multiplication by z in complex numbers.
    # M^4 * [1, 0]^T corresponds to z^4.
    # z^4 = (s_u + i * s_v)^4.
    # The second coordinate is the imaginary part of (s_u + i * s_v)^4.
    # Im((s_u + i * s_v)^4) = 4*s_u^3*s_v - 4*s_u*s_v^3 = 4*s_u*s_v*(s_u^2 - s_v^2).
    # We need 4*s_u*s_v*(s_u^2 - s_v^2) != 0.
    # Since s_u, s_v in {1, 2, 3}, s_u*s_v is never 0.
    # So we need s_u^2 - s_v^2 != 0, which means s_u != s_v.
    # The condition is simply s_u != s_v for all coupled pairs (u, v).
    # This is equivalent to checking if the graph is 3-colorable.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
