import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The energy function E(u, v) = (c_u - c_v)^2 * (c_u - c_v - 1) * (c_u - c_v + 1)
    # Let x = c_u - c_v. E = x^2 * (x-1) * (x+1) = x^2 * (x^2 - 1) = x^4 - x^2.
    # Possible values for c_u, c_v in {1, 2, 3}:
    # If c_u = c_v, x = 0, E = 0.
    # If |c_u - c_v| = 1, x^2 = 1, E = 1^2 * (1 - 1) = 0.
    # If |c_u - c_v| = 2, x^2 = 4, E = 4 * (4 - 1) = 12.
    # So E(u, v) = 0 if and only if |c_u - c_v| <= 1.
    # This is equivalent to saying that for every edge (u, v), |c_u - c_v| <= 1.
    # This is a 3-coloring problem where colors are {1, 2, 3} and adjacent nodes
    # must have colors with difference at most 1.
    
    # This is equivalent to finding a homomorphism to the graph P3: 1-2-3.
    # A graph is 3-colorable with this constraint if and only if it is bipartite?
    # No, the constraint |c_u - c_v| <= 1 is satisfied if we use only 1s and 2s,
    # or only 2s and 3s, or only 1s, 2s, and 3s such that no 1 is adjacent to 3.
    # Actually, if a graph is bipartite, we can color it with 1 and 2.
    # If it's not bipartite, we need to check if it's a subgraph of a graph
    # that allows this. The condition |c_u - c_v| <= 1 means we can't have 1 and 3 adjacent.
    # This is exactly the definition of a graph being 2-colorable (bipartite)
    # if we consider the "forbidden" edge to be (1, 3).
    # Wait, the condition is simply that the graph must be bipartite.
    # If the graph is bipartite, we can color it with 1 and 2.
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
