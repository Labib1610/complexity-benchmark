```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            adj[u].append(v)
            adj[v].append(u)

    # E(a, b) = a^b + b^a + a*b
    # Modulo 3:
    # 1^b = 1, 2^b = (-1)^b
    # E(1, 1) = 1 + 1 + 1 = 3 = 0 (mod 3) -> Invalid
    # E(1, 2) = 1^2 + 2^1 + 1*2 = 1 + 2 + 2 = 5 = 2 (mod 3) -> Valid
    # E(1, 3) = 1^3 + 3^1 + 1*3 = 1 + 0 + 0 = 1 (mod 3) -> Valid
    # E(2, 2) = 2^2 + 2^2 + 2*2 = 4 + 4 + 4 = 12 = 0 (mod 3) -> Invalid
    # E(2, 3) = 2^3 + 3^2 + 2*3 = 8 + 9 + 6 = 23 = 2 (mod 3) -> Valid
    # E(3, 3) = 3^3 + 3^3 + 3*3 = 0 + 0 + 0 = 0 (mod 3) -> Invalid
    
    # Valid pairs: (1, 2), (1, 3), (2, 3)
    # Invalid pairs: (1, 1), (2, 2), (3, 3)
    # This means no two connected nodes can have the same value.
    # This is exactly the definition of a 3-coloring problem.

    colors = [0] * n

    def is_safe(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_safe(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
