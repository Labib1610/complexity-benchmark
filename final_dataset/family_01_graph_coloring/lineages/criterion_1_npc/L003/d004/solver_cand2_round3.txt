import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition q1 != q2 means floor(pu/pv) != floor(pv/pu).
    # Possible codes: 1, 2, 3.
    # Pairs (pu, pv):
    # (1, 1): q1=1, q2=1 (Invalid)
    # (2, 2): q1=1, q2=1 (Invalid)
    # (3, 3): q1=1, q2=1 (Invalid)
    # (1, 2): q1=0, q2=2 (Valid)
    # (1, 3): q1=0, q2=3 (Valid)
    # (2, 3): q1=0, q2=1 (Valid)
    # So, pu != pv is the only condition.
    # This is equivalent to 3-coloring the graph.
    
    colors = [0] * (n + 1)

    def is_safe(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_safe(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
