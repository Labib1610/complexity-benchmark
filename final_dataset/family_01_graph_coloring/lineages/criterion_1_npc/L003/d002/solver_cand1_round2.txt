```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Configuration codes: 1, 2, 3
    # Products:
    # 1*1=1 (sq), 1*2=2, 1*3=3
    # 2*1=2, 2*2=4 (sq), 2*3=6
    # 3*1=3, 3*2=6, 3*3=9 (sq)
    # Constraint: product != square
    # 1*1, 2*2, 3*3 are squares.
    # So, if u and v are connected, they cannot have the same code.
    # This is equivalent to a graph coloring problem with 3 colors.
    
    colors = [0] * n
    
    def is_safe(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            if is_safe(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
