import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The problem asks for a 3-coloring of the graph such that for every edge (u, v),
    # color[u] != color[v]. This is equivalent to checking if the graph is 3-colorable.
    # Since n <= 40, we can use backtracking with pruning.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # Heuristic: sort nodes by degree descending to fail faster
    nodes = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to the sorted order
    mapping = {old: new for new, old in enumerate(nodes)}
    new_adj = [[] for _ in range(n)]
    for u in range(n):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    # Re-run backtracking on the re-mapped graph
    res_colors = [0] * n
    def backtrack_mapped(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            valid = True
            for v in new_adj[u]:
                if res_colors[v] == c:
                    valid = False
                    break
            if valid:
                res_colors[u] = c
                if backtrack_mapped(u + 1):
                    return True
                res_colors[u] = 0
        return False

    if backtrack_mapped(0):
        print("YES")
        final_colors = [0] * n
        for i in range(n):
            final_colors[nodes[i]] = res_colors[i]
        print(*(final_colors))
    else:
        print("NO")

solve()
