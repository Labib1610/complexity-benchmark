import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Resonance condition: product is a perfect square.
    # Pairs (p1, p2) that are forbidden:
    # (1, 1) -> 1*1 = 1 (square)
    # (2, 2) -> 2*2 = 4 (square)
    # (3, 3) -> 3*3 = 9 (square)
    # (1, 2) -> 2 (not square)
    # (1, 3) -> 3 (not square)
    # (2, 3) -> 6 (not square)
    # So, same power settings are forbidden if they are connected.
    # This is equivalent to 3-coloring the graph.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
