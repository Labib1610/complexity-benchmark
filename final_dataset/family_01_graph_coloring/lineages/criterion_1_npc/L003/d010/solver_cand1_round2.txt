import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Interference I(x, y) = 6 if x == y else -3
    # Total interference = sum_{edges (u,v)} I(mode[u], mode[v])
    # Let n1, n2, n3 be the number of antennas with modes 1, 2, 3.
    # Let e1, e2, e3 be the number of edges where both endpoints have mode 1, 2, 3.
    # Total interference = 6*(e1 + e2 + e3) - 3*(m - (e1 + e2 + e3))
    # = 9*(e1 + e2 + e3) - 3*m
    # We want total interference = -3*m, so 9*(e1 + e2 + e3) = 0
    # This implies e1 = e2 = e3 = 0.
    # This means no two antennas connected by an edge can have the same mode.
    # This is equivalent to checking if the graph is 3-colorable.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
