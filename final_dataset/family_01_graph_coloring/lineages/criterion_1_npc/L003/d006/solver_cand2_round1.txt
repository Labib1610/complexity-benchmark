import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # R(p_u, p_v) = p_u^4 + p_v^4 - 4*p_u^3*p_v - 4*p_u*p_v^3 + 6*p_u^2*p_v^2 - 5*p_u^2 - 5*p_v^2 + 10*p_u*p_v + 4
    # Precompute R for p_u, p_v in {1, 2, 3}
    def get_r(p1, p2):
        return (p1**4 + p2**4 - 4*(p1**3)*p2 - 4*p1*(p2**3) + 6*(p1**2)*(p2**2) 
                - 5*(p1**2) - 5*(p2**2) + 10*p1*p2 + 4)

    r_table = {}
    for p1 in [1, 2, 3]:
        for p2 in [1, 2, 3]:
            r_table[(p1, p2)] = get_r(p1, p2)

    # Meet-in-the-middle: n <= 40 is too large for 3^n.
    # However, n=40 is small enough for backtracking with pruning or 
    # if the graph is sparse. Given the constraints and the nature of the problem,
    # we use backtracking with a simple heuristic.
    
    assignment = [0] * n
    
    def backtrack(idx):
        if idx == n:
            total_r = 0
            for u, v in edges:
                total_r += r_table[(assignment[u], assignment[v])]
            return total_r == 0
        
        for p in [1, 2, 3]:
            assignment[idx] = p
            # Pruning: check edges fully contained in [0, idx]
            current_r = 0
            possible = True
            for u, v in edges:
                if u <= idx and v <= idx:
                    current_r += r_table[(assignment[u], assignment[v])]
            
            if backtrack(idx + 1):
                return True
        return False

    # Since n=40 is large, check if we can use a simpler approach.
    # The resonance formula R(p_u, p_v) is symmetric.
    # R(1,1)=0, R(2,2)=0, R(3,3)=0.
    # R(1,2)=R(2,1)=0, R(2,3)=R(3,2)=0, R(1,3)=R(3,1)=16.
    # So R=0 iff no pair (u, v) has {p_u, p_v} == {1, 3}.
    # This is equivalent to: if p_u=1, p_v cannot be 3. If p_u=3, p_v cannot be 1.
    # This is a 3-coloring problem variant: 
    # Assign 1, 2, 3 such that no edge connects 1 and 3.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    res = [0] * n
    def solve_coloring(u):
        if u == n: return True
        for color in [1, 2, 3]:
            ok = True
            for v in adj[u]:
                if res[v] != 0:
                    if (color == 1 and res[v] == 3) or (color == 3 and res[v] == 1):
                        ok = False
                        break
            if ok:
                res[u] = color
                if solve_coloring(u + 1): return True
                res[u] = 0
        return False

    if solve_coloring(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
