```python
import sys
import math

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
            
    # Precompute validity of (x, y)
    # W1 = 2^x + y, W2 = 2^y + x
    # gcd(W1, W2) == 1
    valid = [[False for _ in range(4)] for _ in range(4)]
    for x in range(1, 4):
        for y in range(1, 4):
            w1 = (1 << x) + y
            w2 = (1 << y) + x
            if math.gcd(w1, w2) == 1:
                valid[x][y] = True
                
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        for mode in range(1, 4):
            possible = True
            for v in adj[u]:
                if assignment[v] != 0:
                    if not valid[mode][assignment[v]]:
                        possible = False
                        break
            
            if possible:
                assignment[u] = mode
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
