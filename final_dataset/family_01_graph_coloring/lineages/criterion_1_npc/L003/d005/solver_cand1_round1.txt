import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # P(u, v) = 2 + 2 * cos((p_u - p_v) * 2pi/3)
    # If p_u == p_v, cos(0) = 1, P = 4
    # If p_u != p_v, cos(2pi/3) = -0.5, P = 2 + 2*(-0.5) = 1
    # Total power = sum_{edges} P(u, v) = m
    # Let k be the number of edges where p_u == p_v
    # Total power = 4*k + 1*(m - k) = 3k + m
    # We need 3k + m = m => 3k = 0 => k = 0
    # This means no two connected emitters can have the same phase.
    # This is equivalent to 3-coloring the graph.

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
