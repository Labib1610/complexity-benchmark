import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Precompute validity for pairs (c1, c2) where c1, c2 in {1, 2, 3}
    # Heating: (c1^3 + c2^3) % 97
    # Cooling: (c1^2 * c2 + c1 * c2^2) % 97
    # Condition: Heating > Cooling
    valid_pairs = []
    for c1 in [1, 2, 3]:
        for c2 in [1, 2, 3]:
            heating = (c1**3 + c2**3) % 97
            cooling = (c1**2 * c2 + c1 * c2**2) % 97
            if heating > cooling:
                valid_pairs.append((c1, c2))

    assignment = [0] * n

    def is_safe(u, val):
        for v_idx, v_val in enumerate(assignment):
            if v_val != 0:
                # Check if u and v_idx are connected
                connected = False
                for edge in edges:
                    if (edge[0] == u and edge[1] == v_idx) or (edge[0] == v_idx and edge[1] == u):
                        connected = True
                        break
                if connected:
                    # Check if (val, v_val) is a valid pair
                    found = False
                    for p1, p2 in valid_pairs:
                        if p1 == val and p2 == v_val:
                            found = True
                            break
                    if not found:
                        return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for val in [1, 2, 3]:
            if is_safe(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
