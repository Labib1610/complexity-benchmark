```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    # order vertices by descending degree for better pruning
    order = sorted(range(n), key=lambda x: -len(adj[x]))
    pos_in_order = [0]*n
    for i, v in enumerate(order):
        pos_in_order[v] = i

    colors = [0]*n  # 0 = uncolored, else 1..3

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = order[idx]
        # collect colors used by already colored neighbors
        used = 0
        for nb in adj[v]:
            c = colors[nb]
            if c:
                used |= 1 << c
        for col in (1, 2, 3):
            if not (used & (1 << col)):
                colors[v] = col
                if dfs(idx + 1):
                    return True
                colors[v] = 0
        return False

    possible = dfs(0)
    out = sys.stdout
    if not possible:
        out.write("NO")
    else:
        out.write("YES\n")
        out.write(" ".join(str(colors[i]) for i in range(n)))

if __name__ == "__main__":
    solve()
```
