import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    links = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        links.append((u, v))
        idx += 2

    # The condition: det(R) = x^2 - y^2 != 0 => x != y and x != -y.
    # Since x, y in {1, 2, 3}, x != y is the only constraint.
    # If u == v, then x^2 - x^2 = 0, which is always 0.
    # Wait, the problem says: "If the two involved outputs are assigned offset modes x and y, 
    # their acoustic interference is modeled by the localized resonance matrix R = [[x, y], [y, x]].
    # The matrix R must have full rank (det(R) != 0)."
    # det(R) = x^2 - y^2.
    # If x, y in {1, 2, 3}, x^2 - y^2 = 0 iff x = y.
    # So for any link (u, v), we must have mode[u] != mode[v].
    # If u == v, then mode[u] != mode[u] is impossible.
    
    for u, v in links:
        if u == v:
            print("NO")
            return

    adj = [[] for _ in range(n)]
    for u, v in links:
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
