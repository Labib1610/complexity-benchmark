```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition "dot product must be 0" means if u and v share a channel,
    # they cannot have the same signature code.
    # This is equivalent to a graph coloring problem with 3 colors.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
