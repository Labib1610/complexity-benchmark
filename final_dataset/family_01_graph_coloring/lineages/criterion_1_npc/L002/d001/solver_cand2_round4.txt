import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    def can_color(u):
        for i in range(1, 4):
            possible = True
            for v in adj[u]:
                if colors[v] == i:
                    possible = False
                    break
            if possible:
                colors[u] = i
                return True
        return False

    # Backtracking with simple heuristic: color nodes with higher degree first
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    def backtrack(idx):
        if idx == n:
            return True
        
        u = nodes[idx]
        for c in range(1, 4):
            valid = True
            for v in adj[u]:
                if colors[v] == c:
                    valid = False
                    break
            if valid:
                colors[u] = c
                if backtrack(idx + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
