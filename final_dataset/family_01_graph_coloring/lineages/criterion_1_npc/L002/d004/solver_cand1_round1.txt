import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2
        
    # Discriminant D = B^2 - 4AC
    # A = E_u, B = -(E_u^2 + E_v^2), C = E_u * E_v^2
    # D = (E_u^2 + E_v^2)^2 - 4 * E_u * (E_u * E_v^2)
    # D = E_u^4 + 2*E_u^2*E_v^2 + E_v^4 - 4*E_u^2*E_v^2
    # D = E_u^4 - 2*E_u^2*E_v^2 + E_v^4
    # D = (E_u^2 - E_v^2)^2
    # Condition: D > 0 => (E_u^2 - E_v^2)^2 > 0
    # This is equivalent to E_u^2 != E_v^2, which means E_u != E_v
    # Since E_i in {1, 2, 3}, E_u^2 != E_v^2 is equivalent to E_u != E_v
    # Wait, check: if E_u = 1, E_v = 2: D = (1-4)^2 = 9 > 0. Correct.
    # If E_u = 1, E_v = 3: D = (1-9)^2 = 64 > 0. Correct.
    # If E_u = 2, E_v = 3: D = (4-9)^2 = 25 > 0. Correct.
    # If E_u = E_v: D = 0. Not > 0.
    # So the condition is simply E_u != E_v for all (u, v) in constraints.
    
    # This is a graph coloring problem with 3 colors.
    # Since n is small (up to 40), we can use backtracking with pruning.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for i, j in adj:
            if i == u:
                if colors[j] != 0 and colors[j] == c:
                    return False
            elif j == u:
                if colors[i] != 0 and colors[i] == c:
                    return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
