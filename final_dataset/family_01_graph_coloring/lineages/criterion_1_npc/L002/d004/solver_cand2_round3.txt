import sys

def solve():
    # The equation is x^2 - (P[u] + P[v])x + P[u]P[v] = 0
    # This factors to (x - P[u])(x - P[v]) = 0
    # The solutions are x = P[u] and x = P[v].
    # For the solutions to be strictly distinct, P[u] must not equal P[v].
    # This is equivalent to saying the graph must be 3-colorable.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # Optimization: Sort nodes by degree (heuristic)
    # But for n=40, simple backtracking with pruning is usually fine for 3-coloring
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
