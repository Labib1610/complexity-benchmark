import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The Trace Verification Protocol:
    # G starts as identity matrix I.
    # Swap row 1 and row S[u].
    # Swap column 1 and column S[v].
    # Trace(G) < 3.
    # Let G be the matrix after row swap (R_u) and column swap (C_v).
    # If S[u] == S[v], G is a permutation matrix with 1s at (1,1), (S[u], S[u]), and others.
    # Trace is 3. So S[u] != S[v] is necessary.
    # If S[u] != S[v], the trace is 1.
    # Thus, the condition is simply S[u] != S[v] for all tests (u, v).
    # This is equivalent to finding a 3-coloring of the graph.

    colors = [0] * (n + 1)

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        # Try colors 1, 2, 3
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # Optimization: Sort nodes by degree (heuristic)
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to process in order of degree
    mapping = {node: i for i, node in enumerate(nodes)}
    new_adj = [[] for _ in range(n)]
    for u in range(1, n + 1):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])

    res_colors = [0] * n
    def backtrack_optimized(idx):
        if idx == n:
            return True
        u = idx
        for c in range(1, 4):
            possible = True
            for v in new_adj[u]:
                if res_colors[v] == c:
                    possible = False
                    break
            if possible:
                res_colors[u] = c
                if backtrack_optimized(idx + 1):
                    return True
                res_colors[u] = 0
        return False

    if backtrack_optimized(0):
        print("YES")
        final_ans = [0] * n
        for i in range(n):
            final_ans[nodes[i]-1] = res_colors[i]
        print(*(final_ans))
    else:
        print("NO")

solve()
