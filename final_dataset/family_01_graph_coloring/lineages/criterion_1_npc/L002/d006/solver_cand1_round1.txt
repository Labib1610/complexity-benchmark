import sys
import math

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    constraints = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        constraints.append((u, v))
        idx += 2

    # Global bounds analysis:
    # 1. sum(C[i]^2) < 10*n^2 + 42. Since C[i] in {1, 2, 3}, max sum is 9n. 9n < 10n^2 + 42 is always true for n >= 1.
    # 2. ln(C[i]) < 1.5. ln(1)=0, ln(2)≈0.69, ln(3)≈1.09. All < 1.5. Always true.
    # 3. sum(1/C[i]) >= n/4. 
    # 4. sum(|C[i+1]-C[i]|) < 1000n. Max diff is 2, max sum is 2(n-1). 2(n-1) < 1000n. Always true.
    # Pairwise: cos(pi/2 * (C[u] - C[v])) <= 0.5
    # Possible values for C[u]-C[v] are {-2, -1, 0, 1, 2}
    # cos(-pi) = -1 <= 0.5 (True)
    # cos(-pi/2) = 0 <= 0.5 (True)
    # cos(0) = 1 <= 0.5 (False)
    # cos(pi/2) = 0 <= 0.5 (True)
    # cos(pi) = -1 <= 0.5 (True)
    # So C[u] != C[v] is the only constraint.

    # We need to assign C[i] in {1, 2, 3} such that:
    # 1. For all (u, v) in constraints, C[u] != C[v]
    # 2. sum(1/C[i]) >= n/4

    # This is a 3-coloring problem with an additional constraint.
    # Since n is small (up to 40), we can use backtracking.
    
    adj = [[] for _ in range(n)]
    for u, v in constraints:
        adj[u].append(v)
        adj[v].append(u)

    assignment = [0] * n

    def backtrack(idx):
        if idx == n:
            # Check harmonic sum
            s = sum(1.0 / assignment[i] for i in range(n))
            if s >= n / 4.0 - 1e-9:
                return True
            return False
        
        for val in [1, 2, 3]:
            valid = True
            for neighbor in adj[idx]:
                if assignment[neighbor] == val:
                    valid = False
                    break
            if valid:
                assignment[idx] = val
                if backtrack(idx + 1):
                    return True
                assignment[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
