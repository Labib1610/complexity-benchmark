import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    # Since n <= 40, this is a 3-coloring problem.
    # While 3-coloring is NP-complete, n=40 is small enough for a 
    # backtracking algorithm with pruning or a simple search.
    
    def is_safe(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        # Heuristic: try colors in order 1, 2, 3
        for color in [1, 2, 3]:
            if is_safe(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # To handle potentially dense graphs, we can sort nodes by degree
    # but for n=40, simple backtracking is often sufficient.
    # If it hits a worst-case, we can use a more robust approach.
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
