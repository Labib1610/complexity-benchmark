import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * n
    
    # Sort nodes by degree descending to prune search space
    nodes = sorted(range(n), key=lambda x: bin(adj[x]).count('1'), reverse=True)
    
    def backtrack(idx):
        if idx == n:
            return True
        
        u = nodes[idx]
        
        # Find forbidden colors
        forbidden = 0
        for v in range(n):
            if (adj[u] & (1 << v)) and assignment[v] != 0:
                forbidden |= (1 << (assignment[v] - 1))
        
        # Try colors 1, 2, 3
        for color in range(3):
            if not (forbidden & (1 << color)):
                assignment[u] = color + 1
                if backtrack(idx + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
