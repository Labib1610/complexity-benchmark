```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Energy contribution for diff = |b_u - b_v|:
    # diff = 0: 5(0)^2 - (0)^4 = 0
    # diff = 1: 5(1)^2 - (1)^4 = 4
    # diff = 2: 5(2)^2 - (2)^4 = 20 - 16 = 4
    # The condition is sum(energy) = 4m.
    # Since each edge contributes 4 if |b_u - b_v| is 1 or 2,
    # and 0 if |b_u - b_v| is 0, the condition sum(energy) = 4m
    # is equivalent to saying that for every edge (u, v),
    # |b_u - b_v| must be 1 or 2.
    # This means b_u != b_v for all edges.
    # This is equivalent to saying the graph is 3-colorable.
    
    # Since n <= 40, we can use backtracking with pruning to find a 3-coloring.
    colors = [-1] * n
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    def is_valid(node, c):
        for neighbor in adj[node]:
            if colors[neighbor] == c:
                return False
        return True

    def backtrack(node):
        if node == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            if is_valid(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = -1
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
