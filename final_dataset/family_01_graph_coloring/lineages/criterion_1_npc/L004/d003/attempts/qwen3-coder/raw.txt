from itertools import product

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u-1, v-1))  # Convert to 0-indexed
    
    # Possible protocol versions
    versions = [1, 2, 3]
    
    # Try all possible assignments
    for assignment in product(versions, repeat=n):
        total_energy = 0
        valid = True
        
        for u, v in edges:
            diff = assignment[u] - assignment[v]
            energy = 5 * diff * diff - diff * diff * diff * diff
            total_energy += energy
            
        if total_energy == 4 * m:
            print("YES")
            print(" ".join(map(str, assignment)))
            return
    
    print("NO")

solve()
