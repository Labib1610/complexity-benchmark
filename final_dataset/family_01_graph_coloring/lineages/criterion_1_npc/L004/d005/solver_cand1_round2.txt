```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition (b_u, b_v) in {(1,2), (2,1), (1,3), (3,1), (2,3), (3,2)}
    # is equivalent to saying b_u != b_v.
    # This is exactly the definition of a graph being 3-colorable.
    # Since n <= 40, we can use backtracking with pruning.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        # Heuristic: try colors in order
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # To optimize, we can process nodes in order of degree or connected components
    # But for n=40, simple backtracking is often sufficient for 3-coloring.
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
