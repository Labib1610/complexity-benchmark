import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem asks for an assignment to phases 1, 2, 3 such that
    # if (u, v) is a reactive pair, phase(u) != phase(v).
    # Actually, the constraint is: "one compound must be assigned to a 
    # strictly earlier phase than the other".
    # This means if (u, v) is a reactive pair, phase(u) != phase(v).
    # If we have a chain of dependencies, we need a topological sort.
    # Wait, the constraint is: "one must be strictly earlier".
    # This implies that if we have a reactive pair (u, v), we can assign
    # phase(u) = 1 and phase(v) = 2, or phase(u) = 2 and phase(v) = 3, etc.
    # This is equivalent to finding the longest path in the graph.
    # If the longest path has length L, we need at least L+1 phases.
    # Here we have 3 phases, so the longest path must be at most 2.
    # This means the graph must be 3-colorable? No, it's a directed acyclic graph.
    # The constraint "strictly earlier" means we must be able to orient the edges
    # such that the graph is a DAG and the longest path is at most 2.
    # This is equivalent to saying the graph is a subgraph of a 3-partite graph.
    # Actually, it's simpler: we need to partition the vertices into 3 sets
    # V1, V2, V3 such that all edges go from V1 to V2, V2 to V3, or V1 to V3.
    # This is exactly the definition of a graph being 3-colorable where
    # the colors are ordered.
    
    # Since n is small (max 40), we can use backtracking or a greedy approach.
    assignment = [0] * (n + 1)
    
    def is_valid(u, phase):
        for v in adj[u]:
            if assignment[v] != 0:
                if assignment[v] == phase:
                    return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning phase 1, 2, or 3
        for p in [1, 2, 3]:
            if is_valid(u, p):
                assignment[u] = p
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
