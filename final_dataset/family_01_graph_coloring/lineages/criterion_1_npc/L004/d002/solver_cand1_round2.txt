```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Frequencies: 1 -> 2, 2 -> 3, 3 -> 5
    # Products:
    # 1,1: 2*2=4 (Square) - INVALID
    # 2,2: 3*3=9 (Square) - INVALID
    # 3,3: 5*5=25 (Square) - INVALID
    # 1,2: 2*3=6
    # 1,3: 2*5=10
    # 2,3: 3*5=15
    # None of 6, 10, 15 are squares.
    # So, tasks with conflicts cannot be in the same block.
    # This is equivalent to 3-coloring the graph.
    
    assignment = [-1] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = -1
        return False
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
```
