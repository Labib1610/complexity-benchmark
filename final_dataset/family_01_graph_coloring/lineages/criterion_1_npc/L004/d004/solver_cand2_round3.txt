import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Energy function f(d) where d = |b_u - b_v|
    # d can be 0, 1, 2
    # f(0) = 17(0)^6 - 86(0)^4 + 73(0)^2 = 0
    # f(1) = 17(1)^6 - 86(1)^4 + 73(1)^2 = 17 - 86 + 73 = 4
    # f(2) = 17(2)^6 - 86(2)^4 + 73(2)^2 = 17(64) - 86(16) + 73(4) = 1088 - 1376 + 292 = 4
    # So f(d) = 4 if d != 0, and f(d) = 0 if d == 0.
    # Total energy = sum_{edges} f(|b_u - b_v|) = 4 * (number of edges where b_u != b_v)
    # We want total energy = 4 * m.
    # This means for all edges (u, v), we must have b_u != b_v.
    # This is equivalent to saying the graph must be 3-colorable.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
