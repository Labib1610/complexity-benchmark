import sys

def solve():
    # V(x) = [x^3 + 2x + 1, x^2 + x + 2, x^3 + 2x^2 + x, 2x^3 + x^2 + 1]
    # R1 = [73, 22, 100, 145], R2 = [136, 32, 180, 276]
    
    def get_v(x):
        return [x**3 + 2*x + 1, x**2 + x + 2, x**3 + 2*x**2 + x, 2*x**3 + x**2 + 1]

    v1 = get_v(1) # [4, 4, 4, 4]
    v2 = get_v(2) # [13, 8, 18, 25]
    v3 = get_v(3) # [34, 14, 48, 64]
    r1 = [73, 22, 100, 145]
    r2 = [136, 32, 180, 276]

    # The condition is that for every link (u, v), rank(M_{u,v}) = 4.
    # M_{u,v} = [V(b_u), V(b_v), R1, R2].
    # Rank is 4 if and only if det(M_{u,v}) != 0.
    # Since we have 3 choices for each node, this is a 3-coloring problem.
    # Let's check the determinant for all pairs (b_u, b_v).
    
    import numpy as np
    
    vs = {1: np.array(v1), 2: np.array(v2), 3: np.array(v3)}
    R = np.array([r1, r2]).T
    
    adj = {}
    def check(b1, b2):
        M = np.column_stack((vs[b1], vs[b2], R))
        return np.linalg.matrix_rank(M) == 4

    # Precompute valid pairs
    valid_pairs = []
    for i in [1, 2, 3]:
        for j in [1, 2, 3]:
            if check(i, j):
                valid_pairs.append((i, j))
    
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
        
    # Backtracking to find 3-coloring
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            possible = True
            for v_node, v_color in enumerate(assignment):
                if v_color != 0:
                    # Check if there is an edge between u and v_node
                    is_edge = False
                    for edge in edges:
                        if (edge[0] == u and edge[1] == v_node) or (edge[0] == v_node and edge[1] == u):
                            is_edge = True
                            break
                    if is_edge:
                        if not check(color, v_color):
                            possible = False
                            break
            if possible:
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
