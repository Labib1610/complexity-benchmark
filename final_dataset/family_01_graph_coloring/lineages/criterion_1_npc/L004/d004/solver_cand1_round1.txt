import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    d = [len(adj[i]) for i in range(n)]

    def get_node_energy(b):
        return 7 * (b**4) - 3 * (b**2) + 12

    def get_link_energy(bu, bv):
        return 8 * (bu**2 + bv**2 - bu**4 - bv**4 - 3) + 2 * bu * bv * (2 * bu**2 - 3 * bu * bv + 2 * bv**2 - 5)

    # The total energy formula simplifies significantly.
    # Let E_node = sum_{i=1}^n d_i * f(b_i)
    # Let E_link = sum_{(u,v) in E} g(b_u, b_v)
    # Total Energy = sum_{(u,v) in E} (f(b_u) + f(b_v) + g(b_u, b_v))
    # Let h(bu, bv) = f(bu) + f(bv) + g(bu, bv)
    # f(b) = 7b^4 - 3b^2 + 12
    # g(bu, bv) = 8(bu^2 + bv^2 - bu^4 - bv^4 - 3) + 2*bu*bv*(2*bu^2 - 3*bu*bv + 2*bv^2 - 5)
    # h(bu, bv) = 7bu^4 - 3bu^2 + 12 + 7bv^4 - 3bv^2 + 12 + 8bu^2 + 8bv^2 - 8bu^4 - 8bv^4 - 24 + 4bu^3*bv - 6bu^2*bv^2 + 4bu*bv^3 - 10bu*bv
    # h(bu, bv) = -bu^4 - bv^4 + 5bu^2 + 5bv^2 + 4bu^3*bv - 6bu^2*bv^2 + 4bu*bv^3 - 10bu*bv
    
    # We want sum_{(u,v) in E} h(b_u, b_v) = 4m
    
    def h(bu, bv):
        return -bu**4 - bv**4 + 5*bu**2 + 5*bv**2 + 4*(bu**3)*bv - 6*(bu**2)*(bv**2) + 4*bu*(bv**3) - 10*bu*bv

    # Since n is small (40), we can use backtracking with pruning.
    b = [0] * n
    
    def backtrack(idx):
        if idx == n:
            total_energy = 0
            for u, v in edges:
                total_energy += h(b[u], b[v])
            return total_energy == 4 * m
        
        for val in [1, 2, 3]:
            b[idx] = val
            # Pruning: check if partial sum is already too large or impossible
            # But given the constraints and the nature of the equation, 
            # simple backtracking might be slow. However, n=40 is small enough 
            # for some heuristics or just raw search if the state space is constrained.
            if backtrack(idx + 1):
                return True
        return False

    # The problem is equivalent to finding a 3-coloring variant.
    # Let's try a randomized approach or simple search.
    import random
    for _ in range(10000):
        b = [random.randint(1, 3) for _ in range(n)]
        total_energy = sum(h(b[u], b[v]) for u, v in edges)
        if total_energy == 4 * m:
            print("YES")
            print(*(b))
            return

    print("NO")

solve()
