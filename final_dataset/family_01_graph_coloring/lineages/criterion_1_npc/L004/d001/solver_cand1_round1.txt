import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning 1, 2, or 3
        for color in range(1, 4):
            can_assign = True
            for neighbor in adj[u]:
                if assignment[neighbor] == color:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Sort nodes by degree descending to optimize backtracking
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to process in order of degree
    mapping = [0] * (n + 1)
    for i, node in enumerate(nodes):
        mapping[i + 1] = node
        
    # Re-build adjacency for the re-mapped order
    new_adj = [[] for _ in range(n + 1)]
    rev_mapping = [0] * (n + 1)
    for i in range(1, n + 1):
        rev_mapping[mapping[i]] = i
        
    for u in range(1, n + 1):
        for v in adj[u]:
            new_adj[rev_mapping[u]].append(rev_mapping[v])
            
    final_assignment = [0] * (n + 1)
    
    def solve_backtrack(idx):
        if idx > n:
            return True
        
        u = idx
        for color in range(1, 4):
            can_assign = True
            for neighbor in new_adj[u]:
                if final_assignment[neighbor] == color:
                    can_assign = False
                    break
            if can_assign:
                final_assignment[u] = color
                if solve_backtrack(idx + 1):
                    return True
                final_assignment[u] = 0
        return False

    if solve_backtrack(1):
        print("YES")
        res = [0] * n
        for i in range(1, n + 1):
            res[mapping[i] - 1] = final_assignment[i]
        print(*(res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
