import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning wing 1, 2, or 3
        for wing in [1, 2, 3]:
            can_assign = True
            for neighbor in adj[u]:
                if assignment[neighbor] == wing:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[u] = wing
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Optimization: Sort nodes by degree descending to fail faster
    # However, for n=40, simple backtracking might be slow.
    # Let's use a simple heuristic: order nodes by degree.
    nodes = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to the sorted order
    new_adj = [[] for _ in range(n)]
    mapping = {old: new for new, old in enumerate(nodes)}
    for u in range(n):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    res_assignment = [0] * n
    
    def backtrack_optimized(u):
        if u == n:
            return True
        
        for wing in [1, 2, 3]:
            can_assign = True
            for neighbor in new_adj[u]:
                if res_assignment[neighbor] == wing:
                    can_assign = False
                    break
            if can_assign:
                res_assignment[u] = wing
                if backtrack_optimized(u + 1):
                    return True
                res_assignment[u] = 0
        return False

    if backtrack_optimized(0):
        print("YES")
        final_assignment = [0] * n
        for i in range(n):
            final_assignment[nodes[i]] = res_assignment[i]
        print(*(final_assignment))
    else:
        print("NO")

solve()
