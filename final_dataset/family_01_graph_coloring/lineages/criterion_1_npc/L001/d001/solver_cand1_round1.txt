import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning zone 1, 2, or 3
        for zone in [1, 2, 3]:
            can_assign = True
            for neighbor in adj[u]:
                if assignment[neighbor] == zone:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[u] = zone
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Heuristic: sort nodes by degree descending to fail faster
    # However, for n=40, simple backtracking might be slow.
    # But 3-coloring is NP-complete, and n=40 is small enough for 
    # optimized backtracking or meet-in-the-middle.
    # Given the constraints and problem type, standard backtracking 
    # with pruning is usually expected.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
