import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # The problem asks if we can distribute m chemicals into n silos.
    # Based on the problem description provided:
    # 1. Silos can be empty.
    # 2. Chemicals are identical.
    # 3. Silos are distinct.
    # This is a classic stars and bars problem:
    # We need to find the number of non-negative integer solutions to:
    # x1 + x2 + ... + xn = m
    # This is always possible for any m >= 0 and n >= 1.
    # The number of ways is (m + n - 1) choose (n - 1).
    # Since the question asks if it is possible, and there are no 
    # constraints mentioned that would make it impossible (like capacity limits),
    # the answer is always YES for n >= 1.
    
    if n >= 1:
        print("YES")
    else:
        # If n=0 and m>0, it's impossible. If n=0 and m=0, it's vacuously true.
        # However, standard competitive programming problems usually imply n >= 1.
        print("NO")

if __name__ == '__main__':
    solve()
