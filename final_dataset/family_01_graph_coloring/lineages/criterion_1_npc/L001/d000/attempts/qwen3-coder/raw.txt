def solve():
    import sys
    import itertools
    
    # Read input
    n, m = map(int, sys.stdin.readline().split())
    
    # Initialize adjacency list
    adj = [[] for _ in range(n + 1)]
    
    # Read reactive pairs
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Try all possible assignments
    for assignment in itertools.product([1, 2, 3], repeat=n):
        valid = True
        # Check if any reactive pair is in the same silo
        for i in range(1, n + 1):
            for j in adj[i]:
                if assignment[i-1] == assignment[j-1]:
                    valid = False
                    break
            if not valid:
                break
        
        if valid:
            print("YES")
            print(*assignment)
            return
    
    print("NO")

solve()
