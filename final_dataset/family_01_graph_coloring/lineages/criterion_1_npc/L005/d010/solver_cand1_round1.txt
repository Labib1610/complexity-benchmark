```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Frequencies: 1: 5409, 2: 5616, 3: 5200
    # Sums:
    # 1+1 = 10818 (No)
    # 2+2 = 11232 (No)
    # 3+3 = 10400 (No)
    # 1+2 = 11025 = 105^2 (Yes)
    # 1+3 = 10609 = 103^2 (Yes)
    # 2+3 = 10816 = 104^2 (Yes)
    
    # So, any pair (u, v) must have different packages.
    # This is equivalent to finding a 3-coloring of the graph.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
