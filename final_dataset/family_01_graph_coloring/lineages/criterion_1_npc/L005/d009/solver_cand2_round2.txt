import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Shard 1: (1, 2), Shard 2: (2, 3), Shard 3: (1, 3)
    # Let S1 = (1 2), S2 = (2 3), S3 = (1 3)
    # S1*S1 = I, S2*S2 = I, S3*S3 = I
    # S1*S2 = (1 2 3), S2*S1 = (1 3 2)
    # S1*S3 = (1 3 2), S3*S1 = (1 2 3)
    # S2*S3 = (1 2 3), S3*S2 = (1 3 2)
    # Collision occurs if S_u * S_v = I.
    # This happens if S_u = S_v.
    # So, for every edge (u, v), we must have S_u != S_v.
    # This is equivalent to 3-coloring the graph.

    colors = [0] * n

    def is_valid(node, c):
        for neighbor in adj[node]:
            if colors[neighbor] == c:
                return False
        return True

    def backtrack(node):
        if node == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
