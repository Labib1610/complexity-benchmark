import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * n
    
    # Heuristic: sort nodes by degree descending to prune search space faster
    nodes = sorted(range(n), key=lambda i: bin(adj[i]).count('1'), reverse=True)
    
    def backtrack(idx):
        if idx == n:
            return True
        
        u = nodes[idx]
        
        # Try vaults 1, 2, 3
        for vault in range(1, 4):
            can_place = True
            for v in range(n):
                if (adj[u] & (1 << v)) and assignment[v] == vault:
                    can_place = False
                    break
            
            if can_place:
                assignment[u] = vault
                if backtrack(idx + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
