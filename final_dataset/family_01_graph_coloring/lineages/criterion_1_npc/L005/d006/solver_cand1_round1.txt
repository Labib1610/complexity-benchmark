import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        edges.append((u, v))
        idx += 2

    def check(c):
        for u, v in edges:
            cu, cv = c[u-1], c[v-1]
            vals = [cu * u, cv * v, cu * v, cv * u]
            diff = max(vals) - min(vals)
            if diff <= abs(u - v) * min(cu, cv):
                return False
        return True

    # Since n is small (up to 40), we can use backtracking with pruning.
    # However, the constraints on the power levels are quite restrictive.
    # Let's try a simple backtracking approach.
    assignment = [0] * n

    def backtrack(k):
        if k == n:
            return True
        for val in [1, 2, 3]:
            assignment[k] = val
            # Check constraints involving nodes already assigned
            valid = True
            for u, v in edges:
                if u <= k + 1 and v <= k + 1:
                    cu, cv = assignment[u-1], assignment[v-1]
                    vals = [cu * u, cv * v, cu * v, cv * u]
                    if max(vals) - min(vals) <= abs(u - v) * min(cu, cv):
                        valid = False
                        break
            if valid:
                if backtrack(k + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
