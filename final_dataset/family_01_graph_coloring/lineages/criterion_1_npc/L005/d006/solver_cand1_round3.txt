```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The polynomial is P(x) = x^3 - (c_u + c_v)x^2 + (c_u c_v)x
    # P(x) = x(x^2 - (c_u + c_v)x + c_u c_v) = x(x - c_u)(x - c_v)
    # The roots are 0, c_u, c_v.
    # For these to be three distinct real roots, we need:
    # 1. c_u != 0 (always true as c_i in {1, 2, 3})
    # 2. c_v != 0 (always true)
    # 3. c_u != c_v
    # So the condition is simply c_u != c_v for all interacting pairs.
    # This is equivalent to 3-coloring the graph.

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
