import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Collision condition: (c_u^2 + c_v^2) % (c_u + c_v) == 0
    # Possible pairs (c_u, c_v) where 1 <= c_u, c_v <= 3:
    # (1, 1): (1+1) % 2 == 0 -> Collision
    # (1, 2): (1+4) % 3 == 5 % 3 != 0 -> OK
    # (1, 3): (1+9) % 4 == 10 % 4 != 0 -> OK
    # (2, 1): OK
    # (2, 2): (4+4) % 4 == 0 -> Collision
    # (2, 3): (4+9) % 5 == 13 % 5 != 0 -> OK
    # (3, 1): OK
    # (3, 2): OK
    # (3, 3): (9+9) % 6 == 0 -> Collision
    
    # So, collision occurs if c_u == c_v.
    # This means the problem is equivalent to finding a 3-coloring of the graph.
    
    colors = [0] * (n + 1)
    
    def is_valid(node, c):
        for neighbor in adj[node]:
            if colors[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
