```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    # order vertices by descending degree
    order = sorted(range(n), key=lambda x: -len(adj[x]))
    color = [-1] * n  # 0,1,2

    # for quick conflict check, maintain neighbor colors set per vertex
    neighbor_colors = [set() for _ in range(n)]

    def dfs(pos: int) -> bool:
        if pos == n:
            return True
        v = order[pos]
        # try colors 0..2
        for c in range(3):
            if c in neighbor_colors[v]:
                continue
            # assign
            color[v] = c
            # update neighbor info
            changed = []
            for nb in adj[v]:
                if c not in neighbor_colors[nb]:
                    neighbor_colors[nb].add(c)
                    changed.append(nb)
            if dfs(pos + 1):
                return True
            # backtrack
            for nb in changed:
                neighbor_colors[nb].remove(c)
            color[v] = -1
        return False

    possible = dfs(0)
    out = []
    if not possible:
        out.append("NO")
    else:
        out.append("YES")
        out.append(' '.join(str(c + 1) for c in color))
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
```
