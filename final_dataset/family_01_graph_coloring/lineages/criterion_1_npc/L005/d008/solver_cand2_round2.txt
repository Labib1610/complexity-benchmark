import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition "for every pair (u, v), phase(u) != phase(v)"
    # means that the graph must be 3-colorable.
    # Since n <= 40, we can use backtracking with pruning.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # Heuristic: sort nodes by degree descending to fail faster
    nodes = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to the sorted order
    new_adj = [[] for _ in range(n)]
    mapping = {old: new for new, old in enumerate(nodes)}
    for u in range(n):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    adj = new_adj
    
    if backtrack(0):
        print("YES")
        res = [0] * n
        for i in range(n):
            res[nodes[i]] = colors[i]
        print(*(res))
    else:
        print("NO")

solve()
