```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Matrix commutation check:
    # M1 = [[0, 1], [1, 0]]
    # M2 = [[1, 1], [0, 1]]
    # M3 = [[1, 0], [1, 1]]
    # M1*M2 = [[0, 1], [1, 1]], M2*M1 = [[1, 1], [1, 0]] -> No
    # M1*M3 = [[1, 1], [1, 0]], M3*M1 = [[0, 1], [1, 1]] -> No
    # M2*M3 = [[2, 1], [1, 1]], M3*M2 = [[1, 1], [1, 2]] -> No
    # Mi*Mi = Mi*Mi -> Yes
    # So, collision occurs if and only if c_u == c_v.
    # This is a graph coloring problem with 3 colors.
    
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        # Try to color v with a color different from u
                        used = {colors[u]}
                        for neighbor in adj[v]:
                            if colors[neighbor] != 0:
                                used.add(colors[neighbor])
                        
                        found = False
                        for c in [1, 2, 3]:
                            if c not in used:
                                colors[v] = c
                                found = True
                                break
                        if not found:
                            print("NO")
                            return
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
