import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # S = c_u^3 + c_v^3 - c_u^2 * c_v - c_u * c_v^2
    # S = (c_u - c_v)^2 * (c_u + c_v)
    # For S to be non-zero, c_u must not equal c_v.
    # If c_u != c_v, then (c_u - c_v)^2 is always > 0.
    # Since c_u, c_v are in {1, 2, 3}, c_u + c_v is always > 0.
    # Thus, S != 0 if and only if c_u != c_v.
    # This is equivalent to finding a 3-coloring of the graph.

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n

    def is_valid(node, c):
        for neighbor in adj[node]:
            if colors[neighbor] == c:
                return False
        return True

    def backtrack(node):
        if node == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
