import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # The condition R^2 = 0 means that for any two nodes i and j,
    # there is no node k such that R_{i,k} = 1 and R_{k,j} = 1.
    # R_{i,k} = 1 iff i, k are connected in the query graph AND c_i == c_k.
    # This means if we look at the subgraph induced by nodes of the same shard,
    # there should be no path of length 2.
    # A graph with no path of length 2 is a collection of disjoint stars (or isolated vertices).
    # So, for each shard, the induced subgraph must be a collection of stars.
    # This is equivalent to saying that for each shard, the maximum degree of any node
    # in the induced subgraph is such that no two edges share a vertex? No,
    # a star graph has edges sharing a center. A collection of stars means
    # that for any node, its neighbors in the same shard must not be connected to each other.
    # Wait, R^2 = 0 means there is no k such that (i, k) are in the same shard and connected,
    # AND (k, j) are in the same shard and connected.
    # This means for any node k, all its neighbors in the same shard must be independent.
    # This is exactly the definition of a graph being a collection of stars.
    
    assignment = [0] * (n + 1)
    
    def is_valid(node, shard, current_assignment):
        # Check if adding 'node' to 'shard' creates a path of length 2
        # within the same shard.
        # A path of length 2 exists if there is a neighbor 'v' of 'node'
        # such that 'v' is already in 'shard', and 'v' has another neighbor 'w'
        # that is also in 'shard'.
        
        # Actually, the condition R^2 = 0 is equivalent to:
        # For every shard s, the induced subgraph G_s must not contain a path of length 2.
        # A graph with no path of length 2 is a disjoint union of stars.
        # In a star, the center is connected to leaves, but leaves are not connected to each other.
        # So for any node k, all its neighbors in the same shard must be independent.
        
        neighbors_in_shard = [v for v in adj[node] if current_assignment[v] == shard]
        for v in neighbors_in_shard:
            # If v is in the shard, it cannot have any other neighbors in the same shard
            # except for 'node'.
            for w in adj[v]:
                if w != node and current_assignment[w] == shard:
                    return False
            # Also, 'node' cannot have two neighbors in the same shard that are connected.
            # But the loop above covers this.
        return True

    def backtrack(node):
        if node > n:
            return True
        for s in [1, 2, 3]:
            if is_valid(node, s, assignment):
                assignment[node] = s
                if backtrack(node + 1):
                    return True
                assignment[node] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
