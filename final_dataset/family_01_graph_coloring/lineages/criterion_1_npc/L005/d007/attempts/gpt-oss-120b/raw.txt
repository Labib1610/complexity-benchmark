```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    colors = [-1] * n
    order = sorted(range(n), key=lambda x: -len(adj[x]))  # initial ordering by degree

    # Precompute neighbor list for speed
    neigh = adj

    sys.setrecursionlimit(10000)
    found = False

    def select_vertex():
        # choose uncolored vertex with most colored neighbors (DSATUR heuristic)
        best = -1
        best_score = -1
        for v in range(n):
            if colors[v] != -1:
                continue
            used = set()
            for nb in neigh[v]:
                if colors[nb] != -1:
                    used.add(colors[nb])
            score = len(used)
            if score > best_score:
                best_score = score
                best = v
        return best

    def dfs(colored_cnt: int):
        nonlocal found
        if found:
            return
        if colored_cnt == n:
            found = True
            return
        v = select_vertex()
        # try colors 0,1,2 representing 1,2,3
        for c in range(3):
            ok = True
            for nb in neigh[v]:
                if colors[nb] == c:
                    ok = False
                    break
            if not ok:
                continue
            colors[v] = c
            dfs(colored_cnt + 1)
            if found:
                return
            colors[v] = -1

    dfs(0)

    if not found:
        print("NO")
    else:
        print("YES")
        # convert to 1-based shard IDs
        print(' '.join(str(c+1) for c in colors))

if __name__ == "__main__":
    solve()
```
