import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # Sort nodes by degree descending to handle adversarial cases
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    def backtrack(idx_in_nodes):
        if idx_in_nodes == n:
            return True
        
        u = nodes[idx_in_nodes]
        
        # Find forbidden colors
        forbidden = 0
        for v in adj[u]:
            if colors[v] != 0:
                forbidden |= (1 << (colors[v] - 1))
        
        # Try colors 1, 2, 3
        for c in range(1, 4):
            if not (forbidden & (1 << (c - 1))):
                colors[u] = c
                if backtrack(idx_in_nodes + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
