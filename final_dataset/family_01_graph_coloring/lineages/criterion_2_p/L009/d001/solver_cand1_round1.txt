```python
import sys

# The problem states that for every linked pair (u, v), the bitwise XOR of their 
# power modes must be 3.
# Power modes are in {1, 2}.
# 1 XOR 2 = 3.
# 1 XOR 1 = 0.
# 2 XOR 2 = 0.
# Thus, for every linked pair (u, v), the modes must be different.
# This is equivalent to 2-coloring the graph.
# If the graph is bipartite, we can assign modes 1 and 2.
# To get the lexicographically smallest assignment, we want to assign 1 to 
# nodes in the first partition if possible, but we must check both 
# colorings (1, 2) and (2, 1) for each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            is_bipartite = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible assignments for this component:
            # Option 1: nodes with color 1 get mode 1, color 2 get mode 2
            # Option 2: nodes with color 1 get mode 2, color 2 get mode 1
            # We want the lexicographically smallest result.
            # Compare the two options at the smallest index in the component.
            
            comp_nodes = sorted(component)
            first_node = comp_nodes[0]
            
            # Option 1: color[first_node] -> mode 1
            # If color[first_node] == 1, then nodes with color 1 get 1, color 2 get 2
            # If color[first_node] == 2, then nodes with color 2 get 1, color 1 get 2
            
            # Let's define mapping:
            # If we pick mode 1 for color[first_node], then:
            # mode(v) = 1 if color(v) == color(first_node) else 2
            
            # Check which option is lexicographically smaller at the first node
            # Option A: mode(first_node) = 1
            # Option B: mode(first_node) = 2
            # Clearly Option A is always better for the first node.
            
            for node in component:
                if color[node] == color[first_node]:
                    ans[node] = 1
                else:
                    ans[node] = 2
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
