import sys
from collections import deque

# Set recursion limit just in case, though we use iterative BFS
sys.setrecursionlimit(1 << 25)

def solve():
    input = sys.stdin.read
    data = input().split()
    iterator = iter(data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    degree = [0] * n
    
    # We need to handle 1-based indexing from input
    # We will store edges to identify the component of each edge later if needed, 
    # but actually we just need to check validity and assign modes.
    
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
        
    # Result array: 0 means unvisited, 1 means mode 1, 2 means mode 2
    # We want lexicographically smallest, so we prefer 1.
    # We initialize with 0.
    res = [0] * n
    
    # We iterate through all machines. 
    # Since we prefer 1, if a machine is unvisited, we try to assign it 1.
    # If that leads to a contradiction, we try 2.
    # If both fail, it's impossible.
    
    for i in range(n):
        if res[i] == 0:
            # Start BFS for this component
            # We try to assign mode 1 to the starting node 'i'
            
            # We need a way to reset if assignment 1 fails.
            # Instead of complex rollback, we can perform the BFS logic to check validity first?
            # Or just perform BFS and if we hit a contradiction, we restart the BFS for this component with start node assigned 2.
            
            # Let's implement a function or a loop that tries start_mode = 1, then 2.
            
            # Optimization: 
            # If we assign 1 to 'i', and later find a neighbor 'v' of 'i' must be 1 (contradiction),
            # then 'i' must be 2.
            # If 'i' is 2, and neighbor 'v' must be 2, then 'i' must be 1.
            # Basically, if we find a contradiction, we flip the assignment of the whole connected component.
            # However, we must ensure that the flipped assignment is valid.
            # Since the constraints are local (edge stress), flipping (1->2, 2->1) changes stress:
            # 1-1 (stress 2) -> 2-2 (stress 8). Invalid.
            # 1-2 (stress 5) -> 2-1 (stress 5). Valid.
            # 2-2 (stress 8) -> 1-1 (stress 2). Invalid.
            # So, if a configuration is valid, its flipped version is valid ONLY IF there were no 1-1 or 2-2 links.
            # If there were 1-1 links, the original config was already invalid (stress 2 != 5).
            # So a valid configuration will NEVER have 1-1 or 2-2 links.
            # Therefore, in a valid configuration, all edges connect a 1 and a 2.
            # This means the graph must be bipartite.
            # And if it is bipartite, flipping the colors maintains the bipartite property and the 1-2 edge property.
            
            # So the strategy is:
            # 1. Try to color the component starting with 'i' as 1.
            # 2. If successful, great.
            # 3. If we find a contradiction (non-bipartite), then it's impossible to color this component validly.
            #    Wait. Is it possible that starting with 1 fails, but starting with 2 works?
            #    If the graph is bipartite, starting with 1 works AND starting with 2 works.
            #    If the graph is not bipartite (has odd cycle), neither works.
            #    So we just need to check bipartiteness.
            #    If it is bipartite, we have two valid colorings: (Set A=1, Set B=2) or (Set A=2, Set B=1).
            #    We want lexicographically smallest.
            #    Since we iterate i from 0 to n-1, and we want smaller values first:
            #    If 'i' is in the component, we prefer 'i' to be 1.
            #    So we attempt the coloring starting with 'i'=1.
            #    This will yield one of the two valid bipartitions.
            #    Does this guarantee lexicographically smallest result?
            #    Yes. The first node of the component determines the flip.
            #    If we set the first node to 1, we get the configuration where the first node is 1.
            #    The other valid configuration would have the first node as 2.
            #    1 < 2, so our choice is correct.
            
            queue = deque([i])
            res[i] = 1
            possible = True
            
            while queue:
                u = queue.popleft()
                
                for v in adj[u]:
                    if res[v] == 0:
                        res[v] = 3 - res[u] # Assign opposite mode (1->2, 2->1)
                        queue.append(v)
                    elif res[v] == res[u]:
                        # Contradiction found: adjacent nodes have same mode
                        # This implies stress is not 5 (it's 2 or 8).
                        possible = False
                        # We can break, but need to drain queue or just mark and exit
                        # Actually, once possible is False, we can stop.
                        break
                if not possible:
                    break
            
            if not possible:
                print("NO")
                return

    print("YES")
    print(*(res[i] for i in range(n)))

if __name__ == '__main__':
    solve()
