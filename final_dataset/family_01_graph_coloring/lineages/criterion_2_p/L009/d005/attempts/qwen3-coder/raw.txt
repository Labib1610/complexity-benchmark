```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build graph
    graph = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        graph[u].append(v)
        graph[v].append(u)
        edges.append((u, v))
    
    # For each edge, the link stress must be 5
    # Possible stresses:
    # 1^2 + 1^2 = 2
    # 1^2 + 2^2 = 5
    # 2^2 + 1^2 = 5
    # 2^2 + 2^2 = 8
    # So each edge must have one endpoint with mode 1 and the other with mode 2
    # This is equivalent to checking if the graph is bipartite
    
    color = [-1] * n  # -1: unvisited, 0: color A, 1: color B
    
    # Check if the graph is bipartite using BFS/DFS on each component
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 0
        while queue:
            u = queue.popleft()
            for v in graph[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    queue.append(v)
                elif color[v] == color[u]:
                    return False
        return True
    
    for i in range(n):
        if color[i] == -1:
            if not bfs(i):
                print("NO")
                return
    
    # If bipartite, construct the lexicographically smallest solution
    # Assign 1 to color 0 nodes, 2 to color 1 nodes to get lex smallest
    result = []
    for i in range(n):
        if color[i] == 0:
            result.append('1')
        else:
            result.append('2')
    
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
