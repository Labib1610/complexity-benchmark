```python
import sys

# The raw link stress for modes (x, y) is x^2 + y^2.
# The adjusted diagnostic value is (x^2 + y^2) ^ 7.
# Possible modes are 1 and 2.
# If (x, y) = (1, 1): stress = 1^2 + 1^2 = 2. Value = 2 ^ 7 = 5.
# If (x, y) = (1, 2): stress = 1^2 + 2^2 = 5. Value = 5 ^ 7 = 2.
# If (x, y) = (2, 1): stress = 2^2 + 1^2 = 5. Value = 5 ^ 7 = 2.
# If (x, y) = (2, 2): stress = 2^2 + 2^2 = 8. Value = 8 ^ 7 = 15.
# Let n1 be the number of links between (1, 1), n2 be links between (1, 2) or (2, 1), n3 be links between (2, 2).
# Total links m = n1 + n2 + n3.
# Total score = 5*n1 + 2*n2 + 15*n3 = 2m = 2*(n1 + n2 + n3).
# 5*n1 + 2*n2 + 15*n3 = 2*n1 + 2*n2 + 2*n3
# 3*n1 + 13*n3 = 0.
# Since n1, n3 >= 0, this implies n1 = 0 and n3 = 0.
# Thus, all links must be between (1, 2) or (2, 1).
# This means the graph must be bipartite.
# We need to find the lexicographically smallest assignment (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # To get lexicographically smallest, we want 1s at the start.
    # For each connected component, we have two choices:
    # Either color[i] = 1 or color[i] = 2.
    # We pick the one that makes the first node in the component 1 if possible.
    
    final_assignment = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Check if we can set the first node (smallest index) to 1
            first_node = min(component)
            # If color[first_node] is 1, we keep it. If 2, we flip.
            flip = (color[first_node] == 2)
            for node in component:
                c = color[node]
                if flip:
                    c = 3 - c
                final_assignment[node] = c
                
    print("YES")
    print(*(final_assignment[1:]))

solve()
```
