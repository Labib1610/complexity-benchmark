```python
import sys

# The condition "average exactly 1.5" for tiers {1, 2} means
# (p_u + p_v) / 2 = 1.5 => p_u + p_v = 3.
# Since p_i is 1 or 2, this implies p_u != p_v.
# This is a bipartite coloring problem.
# We need to color the graph with 2 colors (1 and 2).
# If the graph is not bipartite, output NO.
# If it is, each connected component has 2 possible colorings (swap 1 and 2).
# We want to maximize sum_{i in S} 1/i!, where S = {i | p_i = 1}.
# To maximize this, for each component, we choose the coloring that puts
# the smaller indices into set S (since 1/i! decreases rapidly as i increases).
# Specifically, compare the sum of 1/i! for both colorings.
# Since 1/i! > sum_{j=i+1}^n 1/j!, we can greedily decide by looking at the
# smallest index in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two colorings:
            # 1: color 1 -> 1, color 2 -> 2
            # 2: color 1 -> 2, color 2 -> 1
            # We want to maximize sum_{v in S} 1/v!
            # Let S1 be nodes with color 1, S2 be nodes with color 2.
            # Option 1: S = S1. Score = sum_{v in S1} 1/v!
            # Option 2: S = S2. Score = sum_{v in S2} 1/v!
            # Compare by finding the smallest index where they differ.
            
            s1_nodes = [u for u, c in component if c == 1]
            s2_nodes = [u for u, c in component if c == 2]
            
            # To compare sum_{v in S1} 1/v! and sum_{v in S2} 1/v!,
            # find the smallest index i in (S1 union S2).
            # If i in S1, Option 1 is better. If i in S2, Option 2 is better.
            min_idx = min(u for u, c in component)
            
            if min_idx in s1_nodes:
                for u, c in component:
                    res[u] = c
            else:
                for u, c in component:
                    res[u] = 3 - c
                    
    print("YES")
    print(*(res[1:]))

solve()
```
