```python
import sys

# The hash penalty for a query (u, v) is (f_u % f_v) + (f_v % f_u).
# Possible values for f_i are {1, 2}.
# If f_u = 1, f_v = 1: (1%1) + (1%1) = 0 + 0 = 0
# If f_u = 1, f_v = 2: (1%2) + (2%1) = 1 + 0 = 1
# If f_u = 2, f_v = 1: (2%1) + (1%2) = 0 + 1 = 1
# If f_u = 2, f_v = 2: (2%2) + (2%2) = 0 + 0 = 0
#
# The sum of penalties must be exactly m.
# Since each query contributes either 0 or 1, and we want the sum to be m,
# every query must contribute exactly 1.
# This means for every query (u, v), we must have f_u != f_v.
# This is a classic 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest string, we prefer 1 over 2.
# We use BFS/DFS to color the graph components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign 1 to make it lexicographically smallest
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
