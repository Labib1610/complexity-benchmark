```python
import sys

# The stability index I_k for a pair (u, v) with tiers p_u, p_v:
# Vertices: (0,0), (p_u, 1/p_u), (p_v, 1/p_v)
# Area = 0.5 * |x1*y2 - x2*y1|
# If p_u = p_v, Area = 0.
# If p_u = 1, p_v = 2: Area = 0.5 * |1*(1/2) - 2*(1)| = 0.5 * |0.5 - 2| = 0.5 * 1.5 = 0.75 = 3/4.
# If p_u = 2, p_v = 1: Area = 0.5 * |2*(1) - 1*(1/2)| = 0.5 * |2 - 0.5| = 0.75 = 3/4.
# So, I_k = 3/4 if p_u != p_v, and 0 if p_u == p_v.
# We need sum(I_k) = 3m/4.
# This means exactly m dependencies must have p_u != p_v.
# This is equivalent to saying the graph must be bipartite, and we must choose
# a partition (A, B) such that the number of edges between A and B is exactly m.
# Since the graph has m edges, this means ALL edges must connect nodes of different tiers.
# Thus, the graph must be bipartite, and for every connected component, we must
# assign tiers such that it is a valid 2-coloring.
# If the graph is not bipartite, output NO.
# If it is bipartite, each connected component has 2 possible colorings (swap 1 and 2).
# We want to maximize sum((2 - p_i) / i!).
# (2 - p_i) is 1 if p_i = 1, and 0 if p_i = 2.
# So we want to maximize sum_{i: p_i=1} (1/i!).
# For each component, we have two choices:
# Choice 1: nodes in set S1 get tier 1, nodes in S2 get tier 2. Score = sum_{i in S1} 1/i!
# Choice 2: nodes in S1 get tier 2, nodes in S2 get tier 1. Score = sum_{i in S2} 1/i!
# We pick the one that gives a larger sum for each component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    ans = [0] * (n + 1)
    
    import math
    # Precompute factorials is not needed, just compare sums
    # Since 1/i! decreases very rapidly, we can compare sums greedily.
    # Actually, for any two sets S1, S2, the one containing the smaller index
    # will always have a larger sum of 1/i!.
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            s0 = [node for node, c in comp if c == 0]
            s1 = [node for node, c in comp if c == 1]
            
            # Compare sum_{j in s0} 1/j! vs sum_{j in s1} 1/j!
            # The set containing the smallest index will have the larger sum.
            min0 = min(s0)
            min1 = min(s1)
            
            if min0 < min1:
                for node in s0: ans[node] = 1
                for node in s1: ans[node] = 2
            else:
                for node in s0: ans[node] = 2
                for node in s1: ans[node] = 1
                
    print("YES")
    print(*(ans[1:]))

solve()
```
