```python
import sys

# The problem asks for an assignment of 1 or 2 to each node such that for every edge (u, v),
# one node is 1 and the other is 2. This is exactly the definition of a bipartite graph.
# If the graph is bipartite, we can color it with 2 colors.
# The condition "at least one is 1 and at least one is 2" is satisfied if and only if
# the edge connects nodes of different colors.
# Lexicographically smallest means we want as many 1s as possible at the beginning.
# For each connected component, we have two possible colorings (swap 1 and 2).
# We choose the coloring that makes the first node in the component 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if we should flip colors to make the first node 1
            # Actually, the first node i is already 1.
            # If we want lexicographically smallest, we want the smallest index
            # in the component to be 1. Since we iterate 1 to n, the first node
            # encountered in a component is always the smallest index.
            # So color[i] = 1 is already optimal for this component.
            
    print("YES")
    print(*(color[1:]))

solve()
```
