import sys

# The condition is D = sum(d_i^2).
# D = sum((d_i * p_i - N_i)^2)
# Let x_i = p_i - 1. Then p_i = x_i + 1, where x_i in {0, 1}.
# N_i = sum_{j in adj(i)} p_j = sum_{j in adj(i)} (x_j + 1) = sum_{j in adj(i)} x_j + d_i.
# V_i - N_i = d_i(x_i + 1) - (sum_{j in adj(i)} x_j + d_i) = d_i * x_i - sum_{j in adj(i)} x_j.
# D = sum_{i=1}^n (d_i * x_i - sum_{j in adj(i)} x_j)^2.
# We want D = sum d_i^2.
# This is a classic problem: the condition simplifies to x_i = 0 for all i,
# or more generally, it relates to the Laplacian of the graph.
# Specifically, the condition D = sum d_i^2 is satisfied if and only if
# for every connected component, either all x_i = 0 or all x_i = 1.
# If x_i = 0, then p_i = 1. If x_i = 1, then p_i = 2.
# To get the lexicographically smallest sequence, we want p_i = 1 whenever possible.
# For each connected component, we can either set all p_i = 1 or all p_i = 2.
# To be lexicographically smallest, we set all p_i = 1 for every component.
# Wait, let's re-check:
# If x_i = 0 for all i, D = sum(0 - 0)^2 = 0.
# If x_i = 1 for all i, D = sum(d_i - d_i)^2 = 0.
# The condition is D = sum d_i^2.
# If x_i = 0 for all i, D = 0. This equals sum d_i^2 only if all d_i = 0.
# If x_i = 1 for all i, D = 0. This equals sum d_i^2 only if all d_i = 0.
# Actually, the condition D = sum d_i^2 is satisfied if and only if the graph is bipartite
# and we assign tiers based on the bipartition.
# Let the bipartition be sets A and B. If we set p_i = 1 for i in A and p_i = 2 for i in B:
# For i in A: V_i = d_i, N_i = sum_{j in adj(i)} 2 = 2 * d_i. V_i - N_i = -d_i.
# For i in B: V_i = 2 * d_i, N_i = sum_{j in adj(i)} 1 = d_i. V_i - N_i = d_i.
# D = sum_{i in A} (-d_i)^2 + sum_{i in B} (d_i)^2 = sum d_i^2.
# This works if the graph is bipartite. If not bipartite, NO.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # For each component, we can choose color 0 -> p=1, color 1 -> p=2
            # OR color 0 -> p=2, color 1 -> p=1.
            # To be lexicographically smallest, we compare the two options.
            # Option 1: color 0 -> 1, color 1 -> 2
            # Option 2: color 0 -> 2, color 1 -> 1
            # We pick the one that gives a smaller value at the first index of the component.
            
            comp_nodes = sorted([u for u, c in component])
            first_node = comp_nodes[0]
            if color[first_node] == 0:
                # Option 1: first_node gets 1
                for u, c in component:
                    res[u] = 1 if c == 0 else 2
            else:
                # Option 1: first_node gets 2
                # Option 2: first_node gets 1
                for u, c in component:
                    res[u] = 2 if c == 0 else 1
                    
    print("YES")
    print(*(res[1:]))

solve()
