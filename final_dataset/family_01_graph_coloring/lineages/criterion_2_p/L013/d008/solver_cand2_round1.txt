import sys

# The condition: f(x) = x^2 - (g_u + g_v)x + 2 has a minimum of -0.25.
# The minimum of a quadratic ax^2 + bx + c is at x = -b/(2a).
# Here, a=1, b=-(g_u + g_v), c=2.
# Min value = f((g_u + g_v)/2) = ((g_u + g_v)/2)^2 - (g_u + g_v)^2/2 + 2 = -0.25
# -(g_u + g_v)^2 / 4 + 2 = -0.25
# (g_u + g_v)^2 / 4 = 2.25
# (g_u + g_v)^2 = 9
# g_u + g_v = 3
# Since g_i \in {1, 2}, the only way g_u + g_v = 3 is if one is 1 and the other is 2.
# This means the graph must be bipartite, and for every edge (u, v), g_u != g_v.
# Stability metric: sum_{i=1}^n (2 - g_i) / (i-1)!
# To maximize this, we want (2 - g_i) to be 1 (i.e., g_i = 1) for small i.
# If g_i = 1, term is 1/(i-1)!. If g_i = 2, term is 0.
# We want g_i = 1 for as many small i as possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            # BFS to check bipartiteness and assign colors
            stack = [(i, 0)]
            color[i] = 0
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # For each component, we have two choices:
            # Choice 0: nodes with color 0 get g=1, nodes with color 1 get g=2
            # Choice 1: nodes with color 0 get g=2, nodes with color 1 get g=1
            # We pick the choice that makes the smallest index i have g_i = 1.
            
            min_node = min(node for node, c in component)
            # Find color of min_node
            c_min = [c for node, c in component if node == min_node][0]
            
            # If c_min == 0, Choice 0 gives g_{min_node} = 1.
            # If c_min == 1, Choice 1 gives g_{min_node} = 1.
            if c_min == 0:
                for node, c in component:
                    color[node] = c
            else:
                for node, c in component:
                    color[node] = 1 - c

    print("YES")
    res = []
    for i in range(1, n + 1):
        res.append(str(color[i] + 1))
    print(" ".join(res))

solve()
