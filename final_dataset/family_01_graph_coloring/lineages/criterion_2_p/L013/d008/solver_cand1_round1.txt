import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
        
    # Stability index:
    # p=1, p=1: 1/(1+1)^2 = 1/4
    # p=1, p=2: 2/(1+2)^2 = 2/9
    # p=2, p=2: 4/(2+2)^2 = 4/16 = 1/4
    # Let x be number of (1,2) edges, y be number of (1,1) edges, z be number of (2,2) edges.
    # x + y + z = m
    # y/4 + 2x/9 + z/4 = 2m/9
    # (y+z)/4 + 2x/9 = 2m/9
    # Since y+z = m-x:
    # (m-x)/4 + 2x/9 = 2m/9
    # 9(m-x) + 8x = 8m
    # 9m - 9x + 8x = 8m
    # x = m
    # So all m edges must be (1,2) edges.
    # This means the graph must be bipartite, and we must color it with 1 and 2.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # We have two possible assignments:
    # Assignment A: color 0 -> p=1, color 1 -> p=2
    # Assignment B: color 0 -> p=2, color 1 -> p=1
    # Confidence score: sum (2-p_i)/i!
    # For p=1, term is 1/i!. For p=2, term is 0.
    # We want to maximize sum of 1/i! for i where p_i=1.
    
    # Let S0 be set of nodes with color 0, S1 be set of nodes with color 1.
    # Assignment A: p_i=1 if i in S0, p_i=2 if i in S1. Score = sum_{i in S0} 1/i!
    # Assignment B: p_i=2 if i in S0, p_i=1 if i in S1. Score = sum_{i in S1} 1/i!
    
    s0_score = 0
    s1_score = 0
    for i in range(n):
        if color[i] == 0:
            s0_score += 1.0 / (math_factorial(i + 1))
        else:
            s1_score += 1.0 / (math_factorial(i + 1))
            
    print("YES")
    if s0_score >= s1_score:
        res = [1 if color[i] == 0 else 2 for i in range(n)]
    else:
        res = [2 if color[i] == 0 else 1 for i in range(n)]
    print(*(res))

def math_factorial(n):
    res = 1
    for i in range(2, n + 1):
        res *= i
        if res > 1e18: return 1e18 # Precision is enough for comparison
    return res

solve()
