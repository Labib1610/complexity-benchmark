```python
import sys

# The problem states that a conflict at (u, v) means row u and column v cannot form a 2x2 square.
# A Type 1 tile is 1x2 (width 1, height 2).
# A Type 2 tile is 2x1 (width 2, height 1).
# A 2x2 square is formed if:
# - Row u has Type 1 (height 2) and Column v has Type 2 (width 2).
#   Wait, the problem says: "A configuration is valid if, for every conflict cell, 
#   a tile of its row's assigned type and a tile of its column's assigned type 
#   cannot be joined to form a 2x2 square."
# - Type 1: 1x2 (W=1, H=2). Type 2: 2x1 (W=2, H=1).
# - A 2x2 square is formed if we have a Type 1 (H=2) and Type 2 (W=2) adjacent.
# - Specifically, if row u is Type 1 and column v is Type 2, they form a 2x2.
# - So, for each conflict (u, v), we cannot have (p_u == 1 and p_v == 2).
# - This is equivalent to: if p_u == 1, then p_v must be 1.
# - This is an implication: p_u = 1 => p_v = 1.
# - By contrapositive: p_v = 2 => p_u = 2.
# - This is a 2-SAT problem where each variable p_i is 1 or 2.
# - p_u=1 => p_v=1 is equivalent to (p_u=2 OR p_v=1).
# - We want the lexicographically smallest sequence, so we try to set p_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        # p_u=1 => p_v=1
        # p_v=2 => p_u=2
        adj[u].append(v)
        adj[v + n].append(u + n)

    # 2-SAT using Tarjan's or simple implication graph
    # Variables: x_i (p_i=1) and not x_i (p_i=2)
    # p_i=1 is node i, p_i=2 is node i+n
    # p_u=1 => p_v=1 is edge u -> v
    # p_v=2 => p_u=2 is edge v+n -> u+n
    
    dfn = [-1] * (2 * n + 1)
    low = [-1] * (2 * n + 1)
    stack = []
    in_stack = [False] * (2 * n + 1)
    scc = [-1] * (2 * n + 1)
    timer = 0
    scc_cnt = 0
    
    def tarjan(u):
        nonlocal timer, scc_cnt
        dfn[u] = low[u] = timer
        timer += 1
        stack.append(u)
        in_stack[u] = True
        for v in adj[u]:
            if dfn[v] == -1:
                tarjan(v)
                low[u] = min(low[u], low[v])
            elif in_stack[v]:
                low[u] = min(low[u], dfn[v])
        if low[u] == dfn[u]:
            while True:
                node = stack.pop()
                in_stack[node] = False
                scc[node] = scc_cnt
                if node == u: break
            scc_cnt += 1

    for i in range(1, 2 * n + 1):
        if dfn[i] == -1:
            tarjan(i)
            
    res = [0] * (n + 1)
    for i in range(1, n + 1):
        if scc[i] == scc[i + n]:
            print("NO")
            return
        # To get lexicographically smallest, we prefer p_i=1 (node i)
        # If scc[i] < scc[i+n], then i is reachable from i+n, so p_i must be 1
        # In Tarjan's, smaller SCC index means later in topological order
        res[i] = 1 if scc[i] < scc[i + n] else 2
        
    print("YES")
    print(*(res[1:]))

solve()
```
