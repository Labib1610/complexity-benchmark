import sys

# The condition p_u * p_v = 2 where p_i in {1, 2} implies:
# If p_u = 1, then p_v = 2.
# If p_u = 2, then p_v = 1.
# This is a bipartite coloring problem.
# For each connected component, we can pick the color of the first node
# to be 1 or 2, which determines the rest of the component.
# To get the lexicographically smallest sequence, for each component,
# we try to set the smallest index node to 1. If that leads to a conflict,
# we try setting it to 2. If both fail, output NO.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try setting color[i] = 1
            component = []
            stack = [(i, 1)]
            color[i] = 1
            possible = True
            
            comp_nodes = []
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                comp_nodes.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible: break
            
            if not possible:
                # Try setting color[i] = 2
                for node in comp_nodes:
                    color[node] = 0
                
                possible = True
                q = [i]
                color[i] = 2
                head = 0
                while head < len(q):
                    u = q[head]
                    head += 1
                    for v in adj[u]:
                        if color[v] == 0:
                            color[v] = 3 - color[u]
                            q.append(v)
                        elif color[v] == color[u]:
                            possible = False
                            break
                    if not possible: break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(color[1:]))

solve()
