```python
import sys

# The condition f(p_i) * f(p_j) < 0 means cos(pi * p_i) * cos(pi * p_j) < 0.
# Since p_i in {1, 2}, cos(pi * 1) = -1 and cos(pi * 2) = 1.
# So the condition is (-1)^p_i * (-1)^p_j < 0, which means (-1)^(p_i + p_j) < 0.
# This implies p_i + p_j must be odd, so p_i != p_j.
# This is a standard 2-coloring problem on a graph.
# We want the lexicographically smallest sequence p_1, ..., p_n.
# This means we want p_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unvisited, 1 = state 1, 2 = state 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component.
            # To get lexicographically smallest, we prefer color 1 for the first node.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
