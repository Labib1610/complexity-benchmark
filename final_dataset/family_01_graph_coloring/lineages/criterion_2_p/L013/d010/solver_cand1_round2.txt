```python
import sys

# The determinant of the matrix [[x, 1, 0], [1, y, 1], [0, 1, x]] is:
# x(xy - 1) - 1(x - 0) + 0 = x^2y - x - x = x^2y - 2x
# We require x^2y - 2x = 0, which is x(xy - 2) = 0.
# Since x is in {1, 2}:
# If x = 1: 1(1*y - 2) = 0 => y = 2.
# If x = 2: 2(2*y - 2) = 0 => 2y - 2 = 0 => y = 1.
# Thus, for every conflict (u, v), p_u and p_v must be different.
# This means the graph where edges are (u, v) must be bipartite.
# We want the lexicographically smallest sequence p_1, ..., p_n.
# This means we want p_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # If u == v, then p_u != p_u, which is impossible.
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we try to set p_i = 1 for i=1..n
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
