```python
import sys

# The problem states that for each flagged pair (u, v), the sum of their 
# assigned vehicle identifiers must be 3. Since identifiers are 1 or 2,
# this means if asset u is 1, asset v must be 2, and vice versa.
# This is equivalent to a 2-coloring problem on a graph where edges
# represent the hedging mandates.
# We want the lexicographically smallest sequence of assignments (1s and 2s).
# This means for each connected component, we want to assign 1 to the 
# smallest index node if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer assigning 1 to the 
            # smallest index in the component.
            component = []
            stack = [i]
            assignment[i] = 1
            component.append(i)
            
            # Standard 2-coloring check
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # Check if swapping 1s and 2s in this component gives a smaller sequence
            # The current assignment is one valid 2-coloring.
            # The other valid 2-coloring is swapping 1s and 2s.
            # We compare the two colorings for the smallest index in the component.
            # Actually, we just need to ensure the smallest index in the component
            # gets a 1 if possible.
            min_node = min(component)
            if assignment[min_node] == 2:
                for node in component:
                    assignment[node] = 3 - assignment[node]
                    
    print("YES")
    print(*(assignment[1:]))

solve()
```
