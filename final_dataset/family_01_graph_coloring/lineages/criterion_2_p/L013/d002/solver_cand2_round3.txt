import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] = 0 (unvisited), 1 (Portfolio 1), 2 (Portfolio 2)
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # For each connected component, we try assigning the first node to 1.
    # If that fails, we try 2.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To ensure lexicographical smallest, we prioritize assigning 1
            # whenever possible.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Standard bipartite check
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # The current assignment is fixed by the choice of the first node.
            # Since we want the lexicographically smallest, we check if 
            # swapping colors in this component makes it smaller.
            # Actually, for each component, the choice is binary.
            # We want the smallest color for the smallest index in the component.
            # Let min_idx be the smallest index in this component.
            # We want color[min_idx] to be 1.
            min_idx = min(component)
            if color[min_idx] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
