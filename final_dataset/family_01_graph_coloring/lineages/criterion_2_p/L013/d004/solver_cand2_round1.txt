import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        # Each row has two 1s at u and v.
        # The condition is s_u + s_v = 3.
        # Since s_i in {1, 2}, this means if s_u = 1, s_v = 2, and vice versa.
        # This is a bipartite coloring problem.
        adj[u].append((v, i))
        adj[v].append((u, i))
        
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if ans[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we prefer 1 over 2.
            # We try setting ans[i] = 1.
            stack = [(i, 1)]
            ans[i] = 1
            component = []
            while stack:
                u, val = stack.pop()
                component.append((u, val))
                for v, _ in adj[u]:
                    if ans[v] == 0:
                        ans[v] = 3 - val
                        stack.append((v, ans[v]))
                    elif ans[v] == val:
                        print("NO")
                        return
            
    # Verify all rows
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if ans[u] + ans[v] != 3:
            print("NO")
            return
            
    print("YES")
    print(*(ans[1:]))

solve()
