import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    visited = [False] * (n + 1)
    parent = [0] * (n + 1)
    stack = []
    setrecursionlimit(1 << 25)
    
    for start in range(1, n + 1):
        if not visited[start]:
            stack.append((start, 0, iter(adj[start])))
            visited[start] = True
            color[start] = 1
            while stack:
                node, p, children = stack[-1]
                advanced = False
                for child in children:
                    if child == p:
                        continue
                    if not visited[child]:
                        visited[child] = True
                        color[child] = 3 - color[node]
                        parent[child] = node
                        stack.append((child, node, iter(adj[child])))
                        advanced = True
                        break
                    else:
                        if color[child] == color[node]:
                            print("NO")
                            return
                if not advanced:
                    stack.pop()
    
    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

solve()
