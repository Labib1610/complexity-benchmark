```python
import sys

# The condition M_u * M_v = [[0, 1], [1, 0]] means:
# If M_u = I, then M_v must be Swap.
# If M_u = Swap, then M_v must be I.
# This is equivalent to saying M_u != M_v.
# This is a 2-coloring problem on a graph where edges represent "must be different".
# Lexicographically smallest means we prefer 1 over 2 for each node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unassigned, 1 = Mode 1, 2 = Mode 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning 1 first to get lexicographically smallest
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, color[v]))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
