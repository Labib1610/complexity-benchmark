```python
import sys
from collections import defaultdict, deque

def solve():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list and degree count
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # Color nodes with 2-coloring (BFS)
    color = [-1] * n  # -1: unvisited, 0: color A, 1: color B
    visited = [False] * n
    
    for start in range(n):
        if visited[start] or not adj[start]:
            continue
            
        queue = deque()
        queue.append(start)
        color[start] = 0
        visited[start] = True
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            
            for neighbor in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = 1 - current_color
                    queue.append(neighbor)
                elif color[neighbor] == current_color:
                    print("NO")
                    return
    
    # Verify that the coloring satisfies the constraints for each edge
    for u, v in edges:
        # Mode 1 means all edges go OUT from the node
        # Mode 2 means all edges go IN to the node
        # So if u is mode 1, edge must be u -> v
        # If u is mode 2, edge must be v -> u
        # Same for v
        
        # With our coloring:
        # color 0 -> mode 1 (source)
        # color 1 -> mode 2 (destination)
        
        # For the edge (u,v) to be directed correctly:
        # if u is color 0 (mode 1), then u->v is okay 
        # if u is color 1 (mode 2), then v->u is needed, so v must be color 0
        # That means: if color[u] == 1, then color[v] must be 0
        # Equivalently: color[u] + color[v] == 1
        
        if color[u] + color[v] != 1:
            print("NO")
            return
    
    # Convert colors to modes
    # color 0 -> mode 1
    # color 1 -> mode 2
    result = []
    for i in range(n):
        if color[i] == -1:  # Isolated node - assign mode 1 for lex smallest
            result.append(1)
        else:
            result.append(color[i] + 1)
    
    print("YES")
    print(" ".join(map(str, result)))

solve()
```
