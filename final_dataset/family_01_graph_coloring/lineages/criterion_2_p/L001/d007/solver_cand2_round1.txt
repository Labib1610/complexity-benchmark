```python
import sys

# The problem states that for a link between u and v, the sum of powers must be 3.
# Tier 1 provides 1 unit, Tier 2 provides 2 units.
# 1 + 2 = 3, so one endpoint must be Tier 1 and the other must be Tier 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 2 colors (1 and 2).
# To get the lexicographically smallest assignment, we want server 1 to be Tier 1.
# If a component is connected, fixing one node determines all others.
# If a component is not connected, we can choose the tier for the smallest index node
# in that component to be 1 to keep it lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we prefer Tier 1 for the first node
            assignment[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(assignment[1:]))

solve()
```
