import sys

# The wave equation: M_i = (3 + cos(theta_{i,j})) / 2
# If theta_{i,j} = 0, M_i = (3 + 1) / 2 = 2
# If theta_{i,j} = pi, M_i = (3 - 1) / 2 = 1
# For a link (u, v):
# theta_{u,v} = 0 => M_u = 2
# theta_{u,v} = pi => M_u = 1
# theta_{v,u} = (theta_{u,v} + pi) mod 2pi
# If theta_{u,v} = 0, then theta_{v,u} = pi => M_v = 1
# If theta_{u,v} = pi, then theta_{v,u} = 0 => M_v = 2
# Thus, for any link (u, v), M_u != M_v.
# This means the graph must be bipartite.
# If the graph is bipartite, we can assign modes 1 and 2 to the two partitions.
# To get the lexicographically smallest sequence, for each connected component,
# we have two choices: (partition A = 1, partition B = 2) or (partition A = 2, partition B = 1).
# We want M_i = 1 whenever possible for the smallest index i in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    modes = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [(i, 1)]
            visited[i] = True
            modes[i] = 1
            component.append(i)
            
            is_bipartite = True
            idx_in_comp = 0
            while idx_in_comp < len(component):
                u = component[idx_in_comp]
                idx_in_comp += 1
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        modes[v] = 3 - modes[u]
                        component.append(v)
                    elif modes[v] == modes[u]:
                        print("NO")
                        return
            
            # Check if we should flip the modes in this component
            # We want the smallest index in the component to have mode 1
            first_node = min(component)
            if modes[first_node] == 2:
                for node in component:
                    modes[node] = 3 - modes[node]
                    
    print("YES")
    print(*(modes[1:]))

solve()
