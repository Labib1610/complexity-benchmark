import sys

# The equation T_i = 1.5 + 0.5 * cos(theta_{i,j})
# If T_i = 1: 1 = 1.5 + 0.5 * cos(theta_{i,j}) => cos(theta_{i,j}) = -1 => theta_{i,j} = pi
# If T_i = 2: 2 = 1.5 + 0.5 * cos(theta_{i,j}) => cos(theta_{i,j}) = 1 => theta_{i,j} = 0
#
# For a link between u and v:
# theta_{u,v} = pi if T_u = 1, else 0
# theta_{v,u} = pi if T_v = 1, else 0
#
# We know theta_{v,u} = theta_{u,v} + pi (mod 2pi)
# If T_u = 1, theta_{u,v} = pi. Then theta_{v,u} = 2pi = 0.
# This implies T_v must be 2.
# If T_u = 2, theta_{u,v} = 0. Then theta_{v,u} = pi.
# This implies T_v must be 1.
#
# Thus, for every link (u, v), T_u != T_v.
# This is a bipartite graph coloring problem.
# A graph is bipartite if and only if it contains no odd cycles.
# To get the lexicographically smallest sequence, we greedily assign 1 to the
# smallest index in each connected component, then propagate the coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer 1 over 2
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
