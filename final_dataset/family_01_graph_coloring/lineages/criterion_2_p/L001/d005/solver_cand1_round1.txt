```python
import sys

# The condition sin(pi * c / x) = 0 for 1 <= x <= c means pi * c / x = k * pi,
# which simplifies to c / x = k, or x is a divisor of c.
# The number of divisors of c must be equal to c.
# For c=1, divisors are {1}, count=1. (Valid)
# For c=2, divisors are {1, 2}, count=2. (Valid)
# For c=3, divisors are {1, 3}, count=2 != 3. (Invalid)
# For c=4, divisors are {1, 2, 4}, count=3 != 4. (Invalid)
# For c > 2, the number of divisors d(c) is always less than c.
# Proof: d(c) <= 2 * sqrt(c). For c >= 3, 2 * sqrt(c) < c.
# So only c=1 and c=2 are valid configuration values.

# The interference equation:
# c1^2(c1 + c2) + c2^2(c2 - c1) = 9 + c1*c2(c1 - c2)
# c1^3 + c1^2*c2 + c2^3 - c2^2*c1 = 9 + c1^2*c2 - c1*c2^2
# c1^3 + c2^3 = 9
# If c1, c2 in {1, 2}:
# 1^3 + 1^3 = 2 != 9
# 1^3 + 2^3 = 9 (Valid)
# 2^3 + 1^3 = 9 (Valid)
# 2^3 + 2^3 = 16 != 9
# So for any baseline (u, v), one must be 1 and the other must be 2.
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    # To get lexicographically smallest, we prefer 1 over 2.
    # For each connected component, try assigning 1 to the smallest index.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
