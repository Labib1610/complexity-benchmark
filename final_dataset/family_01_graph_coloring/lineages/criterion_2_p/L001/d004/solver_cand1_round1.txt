import sys

# The condition for k is that the number of divisors d(k) = k.
# For k=1: d(1) = 1. (Valid)
# For k=2: d(2) = 2. (Valid)
# For k=3: d(3) = 2 != 3.
# For k=4: d(4) = 3 != 4.
# For k > 2, d(k) < k.
# So the only possible values for k are 1 and 2.
# The baseline condition: u^3 + v^3 = 9.
# If u=1, v^3 = 8 => v=2.
# If u=2, v^3 = 1 => v=1.
# So for any connected pair (u, v), one must be 1 and the other must be 2.
# This means the graph must be bipartite.
# We need to assign 1 or 2 to each node such that for every edge (u, v),
# color[u] != color[v].
# To get the lexicographically smallest sequence, for each connected component,
# we try assigning 1 to the smallest index node and propagate.
# If that fails, we try assigning 2 to the smallest index node and propagate.
# We pick the one that results in a smaller value at the smallest index.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    result = [0] * (n + 1)

    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning 1 to node i
            comp = []
            stack = [(i, 1)]
            color[i] = 1
            possible1 = True
            comp.append((i, 1))
            
            head = 0
            while head < len(comp):
                u, c = comp[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        comp.append((v, 3 - c))
                    elif color[v] == c:
                        possible1 = False
                        break
                if not possible1: break
            
            if possible1:
                # Check if assigning 2 to node i is better
                # Actually, we just need the lexicographically smallest.
                # If we assign 1 to the smallest node in the component, 
                # it is always better than assigning 2.
                for node, c in comp:
                    result[node] = c
            else:
                # Try assigning 2 to node i
                color_reset = [node for node, c in comp]
                for node in color_reset: color[node] = 0
                
                comp = []
                stack = [(i, 2)]
                color[i] = 2
                possible2 = True
                comp.append((i, 2))
                
                head = 0
                while head < len(comp):
                    u, c = comp[head]
                    head += 1
                    for v in adj[u]:
                        if color[v] == 0:
                            color[v] = 3 - c
                            comp.append((v, 3 - c))
                        elif color[v] == c:
                            possible2 = False
                            break
                    if not possible2: break
                
                if possible2:
                    for node, c in comp:
                        result[node] = c
                else:
                    print("NO")
                    return

    print("YES")
    print(*(result[1:]))

solve()
