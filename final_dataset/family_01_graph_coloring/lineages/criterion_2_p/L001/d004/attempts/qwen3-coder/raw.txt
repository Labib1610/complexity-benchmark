import sys
from collections import defaultdict, deque

def find_valid_numbers():
    # We need positive integers where number of divisors equals the number itself
    # Let's check small numbers
    valid = []
    for i in range(1, 100):  # Check first 100 numbers
        count = 0
        for j in range(1, i + 1):
            if i % j == 0:
                count += 1
        if count == i:
            valid.append(i)
    return valid

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # First, find valid configuration values
    # A number d has exactly d divisors iff d is a valid configuration
    valid_configs = []
    for i in range(1, 1000):  # Check enough numbers
        divisor_count = 0
        for j in range(1, i + 1):
            if i % j == 0:
                divisor_count += 1
        if divisor_count == i:
            valid_configs.append(i)
    
    # The only numbers where d(n) = n are 1 and 2
    # d(1) = 1 (divisors: 1)
    # d(2) = 2 (divisors: 1, 2)
    # For n >= 3, d(n) < n (this can be proven)
    
    valid_values = [1, 2]
    
    # For constraint: x^3 + y^3 = 9
    # Possible pairs from {1,2}:
    # 1^3 + 1^3 = 2
    # 1^3 + 2^3 = 1 + 8 = 9  ✓
    # 2^3 + 1^3 = 8 + 1 = 9  ✓
    # 2^3 + 2^3 = 16
    
    # So only valid pairs are (1,2) and (2,1)
    edges = []
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u-1, v-1))  # Convert to 0-indexed
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)
    
    # This is a 2-coloring problem with colors {1,2}
    # Each connected component must be bipartite
    color = [-1] * n  # -1: unvisited, 0: color 1, 1: color 2
    assignment = [0] * n
    
    for start in range(n):
        if color[start] == -1:
            # BFS to check if bipartite and assign colors
            queue = deque([start])
            color[start] = 0
            assignment[start] = 1  # Color 0 -> value 1
            
            while queue:
                node = queue.popleft()
                current_value = assignment[node]
                neighbor_value = 2 if current_value == 1 else 1
                
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - color[node]
                        assignment[neighbor] = neighbor_value
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        # Not bipartite - impossible to satisfy constraints
                        print("NO")
                        return
    
    # Verify all edges satisfy the constraint
    for u, v in edges:
        if assignment[u]**3 + assignment[v]**3 != 9:
            print("NO")
            return
    
    # To get lexicographically smallest, we prefer assigning 1 whenever possible
    # In our BFS approach, we're already doing that since we start with color 0 (value 1)
    
    print("YES")
    print(*assignment)

solve()
