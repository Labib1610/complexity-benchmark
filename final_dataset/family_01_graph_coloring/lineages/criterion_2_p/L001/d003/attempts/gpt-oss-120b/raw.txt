import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    parity = [-1] * (n + 1)          # 0/1 colors
    answer = [0] * (n + 1)           # final multipliers (1 or 2)

    for start in range(1, n + 1):
        if parity[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        parity[start] = 0
        comp = [start]
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if parity[v] == -1:
                    parity[v] = parity[u] ^ 1
                    q.append(v)
                    comp.append(v)
                elif parity[v] == parity[u]:
                    print("NO")
                    return
        # decide orientation: smallest index gets multiplier 1
        min_node = min(comp)
        min_par = parity[min_node]
        # if min_par == 0 -> parity 0 -> 1, else parity 0 -> 2
        if min_par == 0:
            map0, map1 = 1, 2
        else:
            map0, map1 = 2, 1
        for v in comp:
            answer[v] = map0 if parity[v] == 0 else map1

    print("YES")
    print(' '.join(str(answer[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
