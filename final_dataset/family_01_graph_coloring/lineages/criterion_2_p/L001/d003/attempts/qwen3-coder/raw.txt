```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Color the graph with 2 colors (1 and 2)
    # If we can't 2-color it, then it's impossible
    color = [-1] * n
    
    for i in range(n):
        if color[i] == -1:
            # BFS to color component
            queue = deque([i])
            color[i] = 1
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 3 - color[node]  # 1->2, 2->1
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # Check if all edges satisfy the constraint
    for u, v in edges:
        if color[u] * color[v] != 2:
            print("NO")
            return
    
    # If we get here, we have a valid coloring
    print("YES")
    result = [str(c) for c in color]
    print(" ".join(result))

solve()
```
