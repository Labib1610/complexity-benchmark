```python
import sys

# The problem states:
# 1. Each module's tier (x) must be a positive integer such that x^2 <= 4.
#    Since x is a positive integer, x can be 1 or 2.
# 2. For any two modules u and v connected by a link, the sum of the squares
#    of their tiers must be 5.
#    Let x_u and x_v be the tiers. x_u^2 + x_v^2 = 5.
#    Since x_u, x_v in {1, 2}, the only way to get 5 is {1^2 + 2^2} = {1 + 4} = 5.
#    This means if u and v are connected, one must be 1 and the other must be 2.
# 3. This is equivalent to checking if the graph is bipartite.
# 4. We want the lexicographically smallest sequence of tiers.
#    For each connected component, we have two choices for the bipartition (A, B).
#    We want to assign 1 to the smaller index whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1) # 0: unvisited, 1: tier 1, 2: tier 2
    result = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible assignments for this component:
            # Option 1: nodes with color 1 get tier 1, nodes with color 2 get tier 2
            # Option 2: nodes with color 1 get tier 2, nodes with color 2 get tier 1
            # We want the lexicographically smallest sequence.
            # Find the smallest index in the component.
            # If we assign 1 to the smallest index, we must check which color it has.
            
            # Let's compare the two options for the component
            # Option A: color 1 -> 1, color 2 -> 2
            # Option B: color 1 -> 2, color 2 -> 1
            
            # We pick the one that gives a smaller value at the smallest index in the component.
            # Actually, we just need to ensure the smallest index in the component gets 1 if possible.
            # The smallest index in the component is min(component).
            # If color[min_idx] == 1, Option A gives 1, Option B gives 2. Pick A.
            # If color[min_idx] == 2, Option B gives 1, Option A gives 2. Pick B.
            
            min_idx = min(component)
            if color[min_idx] == 1:
                for node in component:
                    result[node] = color[node]
            else:
                for node in component:
                    result[node] = 3 - color[node]
                    
    print("YES")
    print(*(result[1:]))

solve()
```
