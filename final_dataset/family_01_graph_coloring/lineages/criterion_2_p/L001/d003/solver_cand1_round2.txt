```python
import sys

# The condition: for any two modules u and v connected by a critical link,
# the sum of the cubes of their assigned tier values must be 9.
# Tier values are 1 or 2.
# 1^3 = 1, 2^3 = 8.
# 1^3 + 2^3 = 1 + 8 = 9.
# This means for every edge (u, v), one must be 1 and the other must be 2.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we want to assign 1 to
# the smallest possible index whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is not colored, we start a BFS/DFS.
    # To make it lexicographically smallest, we prefer assigning 1 to the
    # current node if it's the first in its connected component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
