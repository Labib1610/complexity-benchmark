```python
import sys

# The problem states that for any two dishes u and v connected by a baseline,
# the product of their phase multipliers must be 2.
# Since the multipliers must be positive integers dividing 2, they can only be 1 or 2.
# If u * v = 2, then one must be 1 and the other must be 2.
# This is a bipartite coloring problem where we want to assign 1 or 2 to each node.
# If a component is bipartite, there are two possible assignments (swapping 1s and 2s).
# To get the lexicographically smallest sequence, for each connected component,
# we should pick the assignment that results in a smaller value for the smallest index in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible assignments for this component:
            # Option 1: color[i] = 1, others follow
            # Option 2: color[i] = 2, others follow (swap 1s and 2s)
            # We want the lexicographically smallest result.
            # Find the smallest index in this component.
            min_node = min(component)
            
            # If we set color[min_node] = 1, is it better than color[min_node] = 2?
            # Yes, 1 is always better than 2.
            # So we force the assignment such that color[min_node] = 1.
            
            # Current assignment: color[min_node] is some value (1 or 2).
            # If color[min_node] == 2, we must flip all colors in this component.
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
```
