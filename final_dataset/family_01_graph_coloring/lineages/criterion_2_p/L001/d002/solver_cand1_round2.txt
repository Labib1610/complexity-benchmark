import sys

# The problem states that each module's tier must be a positive integer that 
# evenly divides 2. The divisors of 2 are 1 and 2.
# The condition is that for any two modules u and v connected by a link,
# tier(u) XOR tier(v) = 3.
# Possible values are {1, 2}.
# 1 XOR 2 = 3.
# 1 XOR 1 = 0.
# 2 XOR 2 = 0.
# Thus, for every link (u, v), tier(u) must be different from tier(v).
# This is a bipartite graph coloring problem where colors are {1, 2}.
# We want the lexicographically smallest sequence.
# For each connected component, we can try assigning 1 to the smallest index
# and see if it's valid. If it's valid, we keep it. If not, we try 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if ans[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer 1 over 2.
            # For the first node in the component, we pick 1.
            stack = [(i, 1)]
            ans[i] = 1
            component = []
            
            # Standard bipartite check
            q = [i]
            component.append(i)
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if ans[v] == 0:
                        ans[v] = 3 - ans[u]
                        q.append(v)
                        component.append(v)
                    elif ans[v] == ans[u]:
                        print("NO")
                        return
            
            # Check if this assignment is the lexicographically smallest for this component.
            # Since we fixed the first node (smallest index in component) to 1,
            # the coloring is forced.
            
    print("YES")
    print(*(ans[1:]))

solve()
