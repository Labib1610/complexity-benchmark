import sys

# The problem states the tier must be a positive integer that divides 2.
# The divisors of 2 are 1 and 2.
# The condition is that for any connected u and v, tier[u] ^ tier[v] = 3.
# Since tier values are in {1, 2}:
# 1 ^ 2 = 3.
# This means if u and v are connected, they must have different values (one is 1, one is 2).
# This is a standard bipartite graph coloring problem.
# We want the lexicographically smallest sequence.
# For each connected component, we can either:
# 1. Assign 1 to the first node visited, 2 to its neighbors, etc.
# 2. Assign 2 to the first node visited, 1 to its neighbors, etc.
# To get the lexicographically smallest sequence, for each component, we look at the
# smallest index node 'i' in that component. We prefer tier[i] = 1.
# If we set tier[i] = 1, all other nodes in the component are fixed.
# If we set tier[i] = 2, all other nodes in the component are fixed.
# We compare the two resulting sequences for the component and pick the one that is smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            idx_stack = 0
            while idx_stack < len(component):
                u, c = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Try setting color[i] = 1
            seq1 = [0] * len(component)
            # Try setting color[i] = 2
            seq2 = [0] * len(component)
            
            # Map node to its position in component list
            node_to_pos = {node: j for j, (node, c) in enumerate(component)}
            
            # Option 1: color[i] = 1
            # If current color[i] is 1, use as is. If 2, flip.
            flip1 = (color[i] != 1)
            for node, c in component:
                val = (3 - c) if flip1 else c
                seq1[node_to_pos[node]] = val
                
            # Option 2: color[i] = 2
            flip2 = (color[i] != 2)
            for node, c in component:
                val = (3 - c) if flip2 else c
                seq2[node_to_pos[node]] = val
            
            # Compare lexicographically
            # Sort by node index to compare
            comp_nodes = sorted(component, key=lambda x: x[0])
            s1 = [seq1[node_to_pos[node]] for node, c in comp_nodes]
            s2 = [seq2[node_to_pos[node]] for node, c in comp_nodes]
            
            best_seq = s1 if s1 < s2 else s2
            for idx_c, (node, c) in enumerate(comp_nodes):
                ans[node] = best_seq[idx_c]
                
    print("YES")
    print(*(ans[1:]))

solve()
