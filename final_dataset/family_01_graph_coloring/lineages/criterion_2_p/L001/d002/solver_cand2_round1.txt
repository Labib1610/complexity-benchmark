```python
import sys

# The resonance condition: (s_u^3 + s_v^3) % 9 == 0
# s_i can be 1 or 2.
# 1^3 = 1
# 2^3 = 8
# Possible sums:
# 1+1 = 2 (not divisible by 9)
# 1+8 = 9 (divisible by 9)
# 8+8 = 16 (not divisible by 9)
# Thus, for every link (u, v), one node must be 1 and the other must be 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we have two sets of nodes (color 0 and color 1).
# For each connected component, we have two choices:
# Choice A: nodes in color 0 get state 1, nodes in color 1 get state 2.
# Choice B: nodes in color 0 get state 2, nodes in color 1 get state 1.
# We want to minimize E = sum(s_i * 2^(n-i)).
# This is equivalent to minimizing the value as a binary-like number where 
# state 1 is smaller than state 2.
# For each component, we look at the smallest index i in that component.
# If we pick Choice A, s_i = 1. If we pick Choice B, s_i = 2.
# Since 1 < 2, we prefer the choice that assigns 1 to the smallest index in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            component.append(i)
            
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        component.append(v)
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find smallest index in component
            min_node = min(component)
            # If color[min_node] == 0, Choice A makes s[min_node] = 1
            # If color[min_node] == 1, Choice B makes s[min_node] = 1
            # We want s[min_node] = 1 to minimize the sum
            if color[min_node] == 0:
                for node in component:
                    assignment[node] = 1 if color[node] == 0 else 2
            else:
                for node in component:
                    assignment[node] = 2 if color[node] == 0 else 1
                    
    print("YES")
    print(*(assignment[1:]))

solve()
```
