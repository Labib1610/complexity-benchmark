```python
import sys

# The condition: P * m_u = Q * m_v and P * Q = 2, where P, Q are positive integers.
# Possible (P, Q) pairs: (1, 2) or (2, 1).
# If (P, Q) = (1, 2): 1 * m_u = 2 * m_v => m_u = 2 * m_v.
# Since m_i is 1 or 2, this implies m_u = 2 and m_v = 1.
# If (P, Q) = (2, 1): 2 * m_u = 1 * m_v => m_v = 2 * m_u.
# This implies m_v = 2 and m_u = 1.
# In both cases, m_u != m_v.
# Thus, the condition is equivalent to m_u != m_v.
# This is a standard 2-coloring problem on a graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we want to assign 1
    # whenever possible. We iterate from 1 to n.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [i]
            color[i] = 1
            component = []
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if this component can be flipped to make the sequence smaller
            # Lexicographical order: we want the first node in the component
            # to be 1 if possible.
            # The current assignment is one valid 2-coloring.
            # The other valid 2-coloring is flipping all 1s to 2s and 2s to 1s.
            # We pick the one that makes the first node in the component 1.
            first_node = min(component)
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
```
