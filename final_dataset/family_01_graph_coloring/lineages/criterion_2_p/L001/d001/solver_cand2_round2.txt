```python
import sys

# The problem states that for each link (u, v), the product of multipliers must be 2.
# Multipliers are 1 or 2.
# If u * v = 2, then one must be 1 and the other must be 2.
# This is a bipartite coloring problem.
# If a component is bipartite, there are two possible colorings (swapping 1s and 2s).
# To get the lexicographically smallest configuration, for each connected component,
# we should choose the coloring that assigns 1 to the smallest index in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        component.append((v, next_c))
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_node = min(node for node, c in component)
            # Find the color assigned to the smallest index
            for node, c in component:
                if node == min_node:
                    min_color = c
                    break
            
            # If min_color is 2, we must flip all colors in this component
            # to make the smallest index have color 1.
            if min_color == 2:
                for node, c in component:
                    color[node] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
```
