```python
import sys

# The problem asks for a 2-coloring of a graph where edges represent the constraint
# that the two connected nodes must have different values (1 and 2).
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we want to assign '1' to the
# smallest possible node index in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is uncolored, it starts a new component.
    # To make the sequence lexicographically smallest, we should assign 1
    # to the first node encountered in each component.
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
