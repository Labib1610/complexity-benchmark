import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Color the graph with 2 colors (1 and 2)
    # If we can't 2-color it, then impossible
    color = [0] * (n + 1)
    
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 1
        while queue:
            u = queue.popleft()
            c = color[u]
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - c  # flips 1->2, 2->1
                    queue.append(v)
                elif color[v] == c:
                    return False
        return True
    
    # Check if the graph is bipartite
    for i in range(1, n + 1):
        if color[i] == 0:
            if not bfs(i):
                print("NO")
                return
    
    # If bipartite, assign colors. We want lexicographically smallest.
    # That means we want earliest nodes to be 1 if possible.
    # In our BFS, we started with color 1 for each component.
    # But to get lex smallest, we need to decide for each component:
    # whether to use (1,2) or (2,1) coloring.
    
    # For each component, we can choose which color is 1 and which is 2.
    # To get lex smallest, for the first node in the component (smallest index),
    # we want it to be 1.
    
    visited = [False] * (n + 1)
    final_color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            # Collect all nodes in this component
            component = []
            queue = deque()
            queue.append(i)
            visited[i] = True
            component.append(i)
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        component.append(v)
                        queue.append(v)
            
            # In original coloring, find the color of the first node in component
            first_node = min(component)
            desired_color_first = 1
            actual_color_first = color[first_node]
            
            # If actual color of first node is 2, we need to flip the colors in this component
            flip = (actual_color_first == 2)
            
            for node in component:
                if flip:
                    final_color[node] = 3 - color[node]
                else:
                    final_color[node] = color[node]
    
    print("YES")
    print(" ".join(str(final_color[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    main()
