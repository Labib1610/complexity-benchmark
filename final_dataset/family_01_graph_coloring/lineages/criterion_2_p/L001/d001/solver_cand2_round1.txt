```python
import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# the sum of phase states is 3. Since phase states are 1 or 2, this means
# phase(u) + phase(v) = 3, which is equivalent to phase(u) != phase(v).
# This is exactly the definition of a bipartite graph.
# We need the lexicographically smallest assignment (1s before 2s).
# For each connected component, we can pick a starting node and assign it 1.
# To make it lexicographically smallest, we should iterate through nodes 1 to n.
# If a node is not visited, assign it 1, then perform BFS/DFS to color its component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS from i, assign color 1 to i
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
