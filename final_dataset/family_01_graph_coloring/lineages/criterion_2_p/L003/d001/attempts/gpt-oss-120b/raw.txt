import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    visited = [False] * (n + 1)
    color = [0] * (n + 1)          # 0/1 provisional
    ans = [0] * (n + 1)

    for start in range(1, n + 1):
        if visited[start]:
            continue
        # BFS/DFS to collect component
        stack = [start]
        visited[start] = True
        color[start] = 0
        comp = [start]
        ok = True
        while stack:
            u = stack.pop()
            cu = color[u]
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    color[v] = cu ^ 1
                    stack.append(v)
                    comp.append(v)
                elif color[v] == cu:
                    # not bipartite
                    print("NO")
                    return
        # decide orientation to make start node have mode 1
        flip = (color[start] == 1)   # if true, we invert colors
        for node in comp:
            final_color = color[node] ^ flip
            ans[node] = 1 if final_color == 0 else 2

    print("YES")
    print(' '.join(str(ans[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
