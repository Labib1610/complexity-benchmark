import sys

def mat_mul(A, B):
    C = [[0]*4 for _ in range(4)]
    for i in range(4):
        for k in range(4):
            if A[i][k] == 0: continue
            for j in range(4):
                C[i][j] += A[i][k] * B[k][j]
    return C

def is_zero(A):
    for row in A:
        for val in row:
            if val != 0: return False
    return True

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        edges.append((int(input_data[idx])-1, int(input_data[idx+1])-1))
        idx += 2

    A = [[-8, -5, 4, 5], [-10, -6, 5, 6], [-16, -10, 8, 10], [-10, -6, 5, 6]]
    B = [[6, 1, -2, -2], [10, -4, -2, 0], [10, 13, -6, -10], [13, -12, -1, 4]]
    A2 = mat_mul(A, A)
    B2 = mat_mul(B, B)

    def get_mat(i, s):
        if i < 4:
            return A if s == 1 else B
        else:
            if s == 1:
                c1, c2 = (i % 3) + 1, (i % 2)
                res = [[0]*4 for _ in range(4)]
                for r in range(4):
                    for c in range(4):
                        res[r][c] = c1 * A[r][c] + c2 * A2[r][c]
                return res
            else:
                c1, c2 = (i % 5) + 1, (i % 3)
                res = [[0]*4 for _ in range(4)]
                for r in range(4):
                    for c in range(4):
                        res[r][c] = c1 * B[r][c] + c2 * B2[r][c]
                return res

    mats = [[get_mat(i, 1), get_mat(i, 2)] for i in range(n)]
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        bad = []
        for s1 in [0, 1]:
            for s2 in [0, 1]:
                if is_zero(mat_mul(mats[u][s1], mats[v][s2])) or is_zero(mat_mul(mats[v][s2], mats[u][s1])):
                    bad.append((s1, s2))
        adj[u].append((v, bad))
        adj[v].append((u, [(s2, s1) for s1, s2 in bad]))

    ans = [-1] * n
    for i in range(n):
        if ans[i] == -1:
            stack = [(i, 0)]
            ans[i] = 0
            comp = [(i, 0)]
            while stack:
                u, s = stack.pop()
                for v, bad in adj[u]:
                    for s_u, s_v in bad:
                        if s_u == s:
                            if ans[v] == -1:
                                ans[v] = 1 - s_v
                                stack.append((v, 1 - s_v))
                                comp.append((v, 1 - s_v))
                            elif ans[v] != 1 - s_v:
                                print("NO")
                                return
    
    print("YES")
    print(*(x + 1 for x in ans))

solve()
