```python
import sys

# The problem asks to assign s_i in {1, 2} such that for every edge (u, v),
# M_u * M_v != 0 and M_v * M_u != 0.
# By calculating the matrices for arbitrary i, we find that:
# M_i(s=1) is a rank-1 matrix. Specifically, M_i(1) = k_i * (8i-9) * A,
# where A is a constant matrix.
# M_i(s=2) is also a rank-1 matrix.
# The condition M_u * M_v = 0 is equivalent to saying the column space of M_v
# is in the null space of M_u.
# For these specific matrices, M_u * M_v = 0 if and only if s_u = s_v = 1.
# If s_u = 2 or s_v = 2, the product is non-zero.
# Thus, the constraint is: for every edge (u, v), we cannot have s_u = 1 AND s_v = 1.
# This is equivalent to: if s_u = 1, then s_v must be 2.
# This is a 2-coloring problem on a graph where we want to avoid (1, 1).
# This is exactly the Independent Set problem on a graph, but we want the
# lexicographically smallest sequence s_1, ..., s_n.
# s_i = 1 is "smaller" than s_i = 2.
# We want to set as many s_i = 1 as possible, starting from i=1.
# This is equivalent to finding the Maximum Independent Set, but we need the
# lexicographically smallest sequence.
# Lexicographically smallest means we greedily try to set s_i = 1.
# If we set s_i = 1, all neighbors of i must be 2.
# This is a standard greedy approach:
# For i = 1 to n:
#   If s_i is not assigned:
#     Try setting s_i = 1. This is valid if no neighbor of i is already 1.
#     If valid, set s_i = 1 and all neighbors to 2.
#     Else, set s_i = 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    s = [0] * (n + 1)
    
    # Greedy approach:
    # To get lexicographically smallest, we want s_i = 1 whenever possible.
    # If we set s_i = 1, all neighbors must be 2.
    # If we set s_i = 2, neighbors can be 1 or 2.
    
    for i in range(1, n + 1):
        if s[i] == 0:
            # Can we set s[i] = 1?
            # Only if no neighbor is already 1.
            can_be_1 = True
            for neighbor in adj[i]:
                if s[neighbor] == 1:
                    can_be_1 = False
                    break
            
            if can_be_1:
                s[i] = 1
                # Neighbors must be 2
                for neighbor in adj[i]:
                    if s[neighbor] == 1: # Should not happen if graph is bipartite-like
                        print("NO")
                        return
                    s[neighbor] = 2
            else:
                s[i] = 2
                
    # Final check
    for u in range(1, n + 1):
        for v in adj[u]:
            if s[u] == 1 and s[v] == 1:
                print("NO")
                return
                
    print("YES")
    print(*(s[1:]))

solve()
```
