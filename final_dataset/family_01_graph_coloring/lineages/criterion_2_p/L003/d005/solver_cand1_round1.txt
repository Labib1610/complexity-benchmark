```python
import sys

# Let A = [[3, -4], [2, -3]].
# A^1 = [[3, -4], [2, -3]]
# A^2 = [[3, -4], [2, -3]] * [[3, -4], [2, -3]] = [[9-8, -12+12], [6-6, -8+9]] = [[1, 0], [0, 1]] = I.
# So M_i = A if s_i = 1, and M_i = I if s_i = 2.
# C_{u,v} = M_u * M_v.
# If s_u = 2 and s_v = 2, C_{u,v} = I * I = I (Feedback loop!)
# If s_u = 1 and s_v = 2, C_{u,v} = A * I = A != I.
# If s_u = 2 and s_v = 1, C_{u,v} = I * A = A != I.
# If s_u = 1 and s_v = 1, C_{u,v} = A * A = I (Feedback loop!)
# Thus, the constraint C_{u,v} != I means s_u != s_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence s_1, ..., s_n.
# This means we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we try color 1 first
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
