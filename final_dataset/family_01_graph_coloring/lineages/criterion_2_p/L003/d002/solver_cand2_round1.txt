```python
import sys

# The condition is: s_u * u + s_v * v is NOT a multiple of (u + v).
# Let s_u, s_v in {1, 2}.
# If s_u = 1, s_v = 1: 1*u + 1*v = u + v. This is a multiple of (u + v).
# So we cannot have s_u = 1 and s_v = 1 simultaneously.
# If s_u = 2, s_v = 2: 2*u + 2*v = 2(u + v). This is a multiple of (u + v).
# So we cannot have s_u = 2 and s_v = 2 simultaneously.
# Thus, for every constraint (u, v), we must have s_u != s_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence s_1, s_2, ..., s_n.
# This means we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we greedily pick 1 for the first node
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
