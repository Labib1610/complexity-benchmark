```python
import sys

# The resonance coefficient R = s_u/s_v + s_v/s_u.
# Possible values for s_i are {1, 2}.
# If s_u = 1, s_v = 1: R = 1/1 + 1/1 = 2 (Integer, invalid)
# If s_u = 2, s_v = 2: R = 2/2 + 2/2 = 2 (Integer, invalid)
# If s_u = 1, s_v = 2: R = 1/2 + 2/1 = 2.5 (Not an integer, valid)
# If s_u = 2, s_v = 1: R = 2/1 + 1/2 = 2.5 (Not an integer, valid)
# Thus, for every pair (u, v), we must have s_u != s_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence s_1, ..., s_n.
# This means we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 to the first node
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
