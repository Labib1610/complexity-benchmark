```python
import sys

# The problem states:
# s_i is in {1, 2}.
# Frequency parameters are (s_i + 1).
# Possible values for (s_i + 1) are {2, 3}.
# Constraint: gcd(s_u + 1, s_v + 1) == 1.
# If s_u = 1, s_u + 1 = 2.
# If s_u = 2, s_u + 1 = 3.
# The constraint gcd(s_u + 1, s_v + 1) == 1 means:
# If s_u = 1 (val 2) and s_v = 1 (val 2), gcd(2, 2) = 2 != 1. (Invalid)
# If s_u = 2 (val 3) and s_v = 2 (val 3), gcd(3, 3) = 3 != 1. (Invalid)
# If s_u = 1 (val 2) and s_v = 2 (val 3), gcd(2, 3) = 1. (Valid)
# This is equivalent to a 2-coloring problem:
# For every edge (u, v), s_u must be different from s_v.
# We want the lexicographically smallest sequence s_1, ..., s_n.
# This means we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 first
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
