```python
import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# color(u) != color(v). This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest assignment (s1, s2, ..., sn),
# we want s1 to be 1 if possible, then s2 to be 1 if possible, and so on.
# Since the graph might have multiple connected components, we process each
# component independently. For each component, we can pick the color of the
# first node (the one with the smallest index) to be 1, which forces the
# colors of all other nodes in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we assign 1 to the first node
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_c
                        stack.append((v, next_c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
```
