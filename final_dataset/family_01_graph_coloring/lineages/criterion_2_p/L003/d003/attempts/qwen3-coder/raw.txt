```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if n == 0:
        print("YES")
        return
    if n == 1:
        print("YES")
        print("1")
        return
    
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Try to 2-color the graph (bipartite check)
    color = [-1] * n
    
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 0
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 1 - color[u]
                    queue.append(v)
                elif color[v] == color[u]:
                    return False
        return True
    
    for i in range(n):
        if color[i] == -1:
            if not bfs(i):
                print("NO")
                return
    
    # Now we have a bipartite graph. Color 0 -> mode 1, Color 1 -> mode 2
    # But we need to check for length 4 cycles where both nodes have same mode
    # Actually, let's think differently.
    # A merged signal is s_u + s_v sync pulses.
    # It's bad if it can be split into two identical halves => if s_u + s_v is even and the two halves are identical.
    # Since all pulses in each signal are identical, this happens iff s_u == s_v and s_u + s_v is even.
    # Since s_u, s_v ∈ {1,2}, s_u + s_v is even iff s_u and s_v have same parity.
    # So s_u + s_v is even and halves are identical iff s_u = s_v.
    # So for every edge (u,v), we must have s_u ≠ s_v.
    # This is exactly a 2-coloring problem.
    
    # We already did 2-coloring. Let's assign modes based on colors.
    # We want lexicographically smallest assignment.
    # That means we prefer s_i = 1 over s_i = 2.
    # In our coloring, let's say color 0 gets mode 1, color 1 gets mode 2.
    # But what if we flip the assignment? color 0 -> 2, color 1 -> 1?
    # Then node 0 would get mode 2 instead of 1, which is worse lex order.
    # So we should assign color 0 -> 1, color 1 -> 2.
    
    result = [0] * n
    for i in range(n):
        if color[i] == 0:
            result[i] = 1
        else:
            result[i] = 2
            
    # Verify the solution
    for u, v in edges:
        if result[u] == result[v]:
            # Try flipping the coloring
            for i in range(n):
                result[i] = 3 - result[i]  # 1->2, 2->1
            # Verify again
            ok = True
            for uu, vv in edges:
                if result[uu] == result[vv]:
                    ok = False
                    break
            if not ok:
                print("NO")
                return
            else:
                print("YES")
                print(' '.join(map(str, result)))
                return
            break
    else:
        print("YES")
        print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
```
