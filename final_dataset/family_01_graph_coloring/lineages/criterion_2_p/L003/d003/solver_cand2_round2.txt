```python
import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# the colors are different (s_u != s_v). This is equivalent to checking if the
# graph is bipartite. If it is, we want the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for smaller indices.
# We can achieve this by iterating from 1 to n and assigning 1 if possible,
# otherwise 2, using BFS/DFS to propagate constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this connected component
            # To get lexicographically smallest, we try to assign 1 to the
            # smallest uncolored node in the component.
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
