```python
import sys

# The condition that W_u and W_v form a basis for R^2 means:
# det([W_u, W_v]) != 0
# W_u = [s_u, 3-s_u]^T, W_v = [s_v, 3-s_v]^T
# det = s_u * (3-s_v) - (3-s_u) * s_v
#     = 3*s_u - s_u*s_v - 3*s_v + s_u*s_v
#     = 3*(s_u - s_v)
# For the determinant to be non-zero, s_u - s_v must be non-zero.
# Thus, s_u != s_v.
# This is a standard 2-coloring problem on a graph.
# We want to minimize E = sum_{i=1}^n s_i * 2^{n-i}.
# This is equivalent to minimizing the integer represented by (s_1-1, s_2-1, ..., s_n-1)
# in binary, where s_i=1 corresponds to 0 and s_i=2 corresponds to 1.
# To minimize this, we want s_i=1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To minimize the energy, we greedily pick s_i = 1 (color 1)
            # whenever we have a choice.
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
