import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n          # 0 / 1 bipartite colors
    comp_nodes = []           # list of lists of nodes per component
    from collections import deque
    for start in range(n):
        if color[start] != -1:
            continue
        q = deque([start])
        color[start] = 0
        cur = []
        while q:
            v = q.popleft()
            cur.append(v)
            for to in adj[v]:
                if color[to] == -1:
                    color[to] = color[v] ^ 1
                    q.append(to)
                elif color[to] == color[v]:
                    print("NO")
                    return
        comp_nodes.append(cur)

    # precompute powers of two weights: weight[i] = 2^{n-1-i}
    # (i from 0)
    weights = [1 << (n - 1 - i) for i in range(n)]

    s = [1] * n   # default 1
    for nodes in comp_nodes:
        sum0 = 0
        sum1 = 0
        for v in nodes:
            if color[v] == 0:
                sum0 += weights[v]
            else:
                sum1 += weights[v]
        # assign 2 to the side with smaller weight sum
        if sum0 <= sum1:
            for v in nodes:
                if color[v] == 0:
                    s[v] = 2
        else:
            for v in nodes:
                if color[v] == 1:
                    s[v] = 2

    print("YES")
    print(' '.join(map(str, s)))

if __name__ == "__main__":
    solve()
