import sys
import sys
sys.setrecursionlimit(1 << 20)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    color = [-1] * n
    comp = [-1] * n
    comp_id = 0
    
    # BFS to check bipartiteness and assign components
    from collections import deque
    for i in range(n):
        if color[i] == -1:
            q = deque([i])
            color[i] = 0
            comp[i] = comp_id
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        comp[v] = comp_id
                        q.append(v)
                    else:
                        if color[v] == color[u]:
                            print("NO")
                            return
            comp_id += 1
    
    # Build component graph
    comp_adj = [set() for _ in range(comp_id)]
    for u, v in edges:
        cu, cv = comp[u], comp[v]
        if cu != cv:
            comp_adj[cu].add(cv)
            comp_adj[cv].add(cu)
    
    # For each component, we have two possible orientations.
    # We need to choose orientation for each component to minimize lexicographically.
    # Since each component is bipartite, we can flip all colors in a component.
    # We'll do a BFS/DFS over component graph to decide flips.
    
    # First, find connected components of the component graph
    visited_comp = [False] * comp_id
    comp_orientation = [0] * comp_id  # 0: keep, 1: flip
    
    # We'll process each connected component of the component graph
    for start in range(comp_id):
        if visited_comp[start]:
            continue
        # BFS to collect all components in this connected part
        q = deque([start])
        visited_comp[start] = True
        comps = []
        while q:
            u = q.popleft()
            comps.append(u)
            for v in comp_adj[u]:
                if not visited_comp[v]:
                    visited_comp[v] = True
                    q.append(v)
        
        # Try both orientations for the first component in this connected part
        best = None
        for first_flip in [0, 1]:
            # Assign orientations
            orient = {}
            orient[start] = first_flip
            possible = True
            # BFS to propagate constraints
            q = deque([start])
            while q:
                u = q.popleft()
                for v in comp_adj[u]:
                    if v not in orient:
                        orient[v] = orient[u] ^ 1  # must be different because edge connects different colors
                        q.append(v)
                    else:
                        if orient[v] == orient[u]:
                            possible = False
                            break
                if not possible:
                    break
            if not possible:
                continue
            
            # Build full assignment for this connected part
            temp = [0] * n
            for node in range(n):
                c = comp[node]
                if c in orient:
                    val = color[node] ^ orient[c]
                    temp[node] = val + 1
                else:
                    temp[node] = color[node] + 1
            
            # Check if this assignment is valid (should be)
            valid = True
            for u, v in edges:
                if temp[u] == temp[v]:
                    valid = False
                    break
            if not valid:
                continue
            
            if best is None:
                best = temp[:]
            else:
                # Compare lexicographically
                if temp < best:
                    best = temp[:]
        
        if best is None:
            print("NO")
            return
        
        # Apply best orientation to global answer
        for node in range(n):
            c = comp[node]
            if c in comp_orientation:
                pass
        # Actually we need to store the best assignment for this connected part
        # Let's store in a global answer array
        if 'ans' not in locals():
            ans = [0] * n
        for node in range(n):
            if comp[node] in set(comps):
                ans[node] = best[node]
    
    # If we reach here, we have an answer
    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
