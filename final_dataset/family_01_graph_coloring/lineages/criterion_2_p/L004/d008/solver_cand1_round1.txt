```python
import sys

# The resonance factor R_{u,v} = s_u^2 + s_u*s_v + s_v^2.
# Possible values for s_i are {1, 2}:
# If s_u=1, s_v=1: R = 1 + 1 + 1 = 3
# If s_u=1, s_v=2: R = 1 + 2 + 4 = 7
# If s_u=2, s_v=1: R = 4 + 2 + 1 = 7
# If s_u=2, s_v=2: R = 4 + 4 + 4 = 12
# We want the product of all R_{u,v} to be 7^m.
# Since each R_{u,v} is 3, 7, or 12, the only way to get 7^m is if every
# R_{u,v} is exactly 7.
# R_{u,v} = 7 if and only if (s_u=1 and s_v=2) or (s_u=2 and s_v=1).
# This means s_u != s_v for every hazard (u, v).
# This is equivalent to checking if the graph is bipartite.
# We want the lexicographically smallest sequence (s_1, ..., s_n).
# We greedily assign s_i = 1 if possible, else s_i = 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    s = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if s[i] == 0:
            # Start BFS/DFS for this component
            stack = [i]
            s[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if s[v] == 0:
                        s[v] = 3 - s[u]
                        stack.append(v)
                    elif s[v] == s[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(s[1:]))

solve()
```
