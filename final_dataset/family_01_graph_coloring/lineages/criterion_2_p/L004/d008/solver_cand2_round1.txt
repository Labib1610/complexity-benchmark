import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # W_{u,v} = 2 - (s_u - s_v)^2
    # If s_u == s_v, W_{u,v} = 2 - 0 = 2
    # If s_u != s_v, W_{u,v} = 2 - 1 = 1
    # Total interference = sum_{edges} W_{u,v}
    # Let k be the number of edges where s_u != s_v.
    # Total = 2 * (m - k) + 1 * k = 2m - k
    # We need 2m - k = m (mod m+1)
    # 2m - k = m + c(m+1)
    # m - k = c(m+1)
    # Since 0 <= k <= m, -m <= m-k <= m.
    # The only multiple of (m+1) in this range is 0.
    # So m - k = 0 => k = m.
    # This means for every edge (u, v), s_u must not equal s_v.
    # This is equivalent to checking if the graph is bipartite.

    color = [0] * (n + 1)
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # To get lexicographically smallest, we want s_i = 1 whenever possible.
            # For each component, we have two choices:
            # 1. color 1 -> 1, color 2 -> 2
            # 2. color 1 -> 2, color 2 -> 1
            # We pick the one that makes the first node in the component 1.
            first_node = min(node for node, c in component)
            # Find the color of the first node
            for node, c in component:
                if node == first_node:
                    first_color = c
                    break
            
            if first_color == 1:
                for node, c in component:
                    assignment[node] = c
            else:
                for node, c in component:
                    assignment[node] = 3 - c
                    
    print("YES")
    print(*(assignment[1:]))

solve()
