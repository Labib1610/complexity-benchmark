import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    d = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        d[u] += 1
        d[v] += 1
        idx += 2
        
    # R = 3 * sum(d_i * s_i) - 2 * sum(s_u * s_v) = 5m
    # Let s_i = 1 + x_i, where x_i in {0, 1}
    # R = 3 * sum(d_i * (1 + x_i)) - 2 * sum((1 + x_u)(1 + x_v))
    # R = 3 * sum(d_i) + 3 * sum(d_i * x_i) - 2 * sum(1 + x_u + x_v + x_u * x_v)
    # Since sum(d_i) = 2m:
    # R = 6m + 3 * sum(d_i * x_i) - 2 * (m + sum(x_u + x_v) + sum(x_u * x_v))
    # R = 4m + 3 * sum(d_i * x_i) - 2 * sum(x_u + x_v) - 2 * sum(x_u * x_v)
    # Note sum(x_u + x_v) = sum(d_i * x_i)
    # R = 4m + sum(d_i * x_i) - 2 * sum(x_u * x_v) = 5m
    # sum(d_i * x_i) - 2 * sum(x_u * x_v) = m
    
    # This is equivalent to: sum_{edges (u,v)} (x_u + x_v - 2*x_u*x_v) = m
    # x_u + x_v - 2*x_u*x_v is 1 if x_u != x_v, and 0 if x_u == x_v.
    # So we need the number of edges with x_u != x_v to be m.
    # This means the graph must be bipartite, and we must pick a partition.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # We have two possible assignments for each connected component:
    # Either (s_i = 1+color[i]) or (s_i = 1+(1-color[i]))
    # To get lexicographically smallest, for each component, we want s_min_idx = 1.
    # Let min_idx be the smallest index in the component.
    # If color[min_idx] == 0, we prefer s_min_idx = 1+0 = 1.
    # If color[min_idx] == 1, we prefer s_min_idx = 1+1 = 2.
    # Wait, we want s_i = 1 or 2.
    # If color[i] = 0, s_i = 1 or 2. If color[i] = 1, s_i = 2 or 1.
    # For each component, pick the assignment that makes the smallest index 1.
    
    res = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            comp = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            min_idx = min(comp)
            # If color[min_idx] == 0, setting s[min_idx] = 1 means s[j] = 1+color[j]
            # If color[min_idx] == 1, setting s[min_idx] = 1 means s[j] = 1+(1-color[j])
            if color[min_idx] == 0:
                for node in comp:
                    res[node] = 1 + color[node]
            else:
                for node in comp:
                    res[node] = 1 + (1 - color[node])
                    
    print("YES")
    print(*(res[1:]))

solve()
