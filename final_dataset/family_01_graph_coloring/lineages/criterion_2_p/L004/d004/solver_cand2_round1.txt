import sys

# The condition |C_i| = d_i means that for each satellite i, 
# either all hazards involving i have i as the emitter (C_i = d_i)
# or all hazards involving i have i as the absorber (C_i = -d_i).
# This is equivalent to a 2-coloring problem on the graph of hazards.
# For each edge (u, v), u and v must have different roles (one emitter, one absorber).
# If a component is bipartite, we can color it with two colors (1 and 2).
# If a component is not bipartite, no solution exists.
# Lexicographically smallest means we want s_i = 1 whenever possible.
# For each connected component, we can choose the coloring (1, 2) or (2, 1).
# We pick the one that makes the first node in the component 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            is_bipartite = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if we should flip colors to make the first node 1
            # The first node i is already color 1.
            # If we wanted to make it 2, we would flip all colors in this component.
            # But we want the lexicographically smallest sequence.
            # Since we process nodes 1 to n, for each component, we want the 
            # smallest node in the component to be 1.
            # Actually, the problem asks for the lexicographically smallest sequence s_1, ..., s_n.
            # This means we want s_1 to be 1, then s_2 to be 1, etc.
            # For each component, we check the smallest index node 'min_node'.
            # If color[min_node] == 2, we flip all colors in that component.
            
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
