```python
import sys

# The problem asks us to assign s_i in {1, 2} to each satellite.
# P_i(x) = x - 1 if s_i = 1, P_i(x) = x + 1 if s_i = 2.
# The condition P_u(x) * P_v(x) = x^2 - 1 means:
# If s_u = 1 (x-1) and s_v = 2 (x+1), product is (x-1)(x+1) = x^2 - 1. (Valid)
# If s_u = 2 (x+1) and s_v = 1 (x-1), product is (x+1)(x-1) = x^2 - 1. (Valid)
# If s_u = 1 and s_v = 1, product is (x-1)^2 != x^2 - 1. (Invalid)
# If s_u = 2 and s_v = 2, product is (x+1)^2 != x^2 - 1. (Invalid)
# Thus, for every hazard (u, v), s_u must be different from s_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (s_1, ..., s_n).
# To get the lexicographically smallest, we greedily try s_i = 1 for each component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    s = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If node i is not colored, we start a BFS/DFS and assign s_i = 1.
    for i in range(1, n + 1):
        if s[i] == 0:
            s[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if s[v] == 0:
                        s[v] = 3 - s[u]
                        stack.append(v)
                    elif s[v] == s[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(s[1:]))

solve()
```
