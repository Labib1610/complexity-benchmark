import sys

# The condition is integral_{-1}^{1} T_u(x) T_v(x) dx = 0.
# T1(x) = x, T2(x) = |x|.
# Case 1: T_u = x, T_v = x. Integral of x^2 from -1 to 1 is 2/3 != 0.
# Case 2: T_u = |x|, T_v = |x|. Integral of |x|^2 = x^2 from -1 to 1 is 2/3 != 0.
# Case 3: T_u = x, T_v = |x|. Integral of x|x| from -1 to 1 is 0 (odd function).
# Thus, for every hazard (u, v), T_u must be different from T_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence of 1s and 2s.
# This means for each connected component, we try to assign 1 to the smallest index.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if this assignment is the lexicographically smallest
            # We have two choices for the component: 
            # 1. color[i] = 1, others follow
            # 2. color[i] = 2, others follow
            # Since we want the smallest sequence, we compare the two options
            # for the first node i in the component.
            # Actually, the greedy choice is always to set the smallest index 
            # in the component to 1.
    
    print("YES")
    print(*(color[1:]))

solve()
