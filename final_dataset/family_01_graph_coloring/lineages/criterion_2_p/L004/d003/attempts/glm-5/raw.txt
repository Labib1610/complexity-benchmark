import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # Precompute valid pairs (s_u, s_v) that satisfy the equation
    # Equation: s_u^3 + s_v^3 + (s_u * s_v)^2 = 13
    # Possible values: 1 and 2
    # Check all 4 combinations:
    # (1,1): 1 + 1 + 1 = 3 != 13
    # (1,2): 1 + 8 + 4 = 13 (valid)
    # (2,1): 8 + 1 + 4 = 13 (valid)
    # (2,2): 8 + 8 + 16 = 32 != 13
    # So valid pairs are (1,2) and (2,1) - satellites must have different values.
    
    color = [0] * (n + 1)  # 0: uncolored, 1: color 1, 2: color 2
    possible = True
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS to color this component
            # Start with color 1 for lexicographically smallest result
            queue = deque([i])
            color[i] = 1
            while queue and possible:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]  # Assign opposite color
                        queue.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break
    
    if not possible:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, color[1:])))

if __name__ == "__main__":
    solve()
