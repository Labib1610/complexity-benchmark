```python
import sys

# The resonance equation: x^3 + y^3 + (x*y)^2 = 13
# x, y in {1, 2}
# If x=1, y=1: 1 + 1 + 1 = 3 != 13
# If x=1, y=2: 1 + 8 + 4 = 13 (Valid)
# If x=2, y=1: 8 + 1 + 4 = 13 (Valid)
# If x=2, y=2: 8 + 8 + 16 = 32 != 13
# Thus, for every hazard (u, v), u and v must have different values.
# This is a 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest sequence, we greedily assign 1 to 
# the smallest unvisited node in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we prefer color 1 for the 
            # smallest node in the component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
