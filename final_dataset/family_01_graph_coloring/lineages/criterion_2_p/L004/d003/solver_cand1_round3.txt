```python
import sys

# The equation is s_u^3 + s_v^3 + (s_u * s_v)^2 = 13
# Possible values for s_u, s_v are {1, 2}.
# If s_u = 1, s_v = 1: 1^3 + 1^3 + (1*1)^2 = 1 + 1 + 1 = 3 != 13
# If s_u = 1, s_v = 2: 1^3 + 2^3 + (1*2)^2 = 1 + 8 + 4 = 13 (Valid)
# If s_u = 2, s_v = 1: 2^3 + 1^3 + (2*1)^2 = 8 + 1 + 4 = 13 (Valid)
# If s_u = 2, s_v = 2: 2^3 + 2^3 + (2*2)^2 = 8 + 8 + 16 = 32 != 13
# Thus, for every hazard (u, v), s_u must be different from s_v.
# This is a standard 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest sequence, we greedily assign 1 to the 
# smallest available node in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign 1 to i to keep it lexicographically smallest
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
