```python
import sys

# The condition is that for every hazard (u, v), s_u^2 + s_v^2 must be 5.
# Since s_i \in {1, 2}, s_i^2 \in {1, 4}.
# The only way s_u^2 + s_v^2 = 5 is if one is 1 and the other is 4.
# This means s_u \neq s_v.
# This is equivalent to a 2-coloring problem on the graph of hazards.
# A graph is 2-colorable if and only if it is bipartite.
# We need the lexicographically smallest assignment, so we prefer 1 over 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If a node is not colored, we assign it 1 (the smaller value).
    # Then we perform BFS/DFS to color its connected component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
