```python
import sys

# The interference profile is I(x) = (x - s_u)(x - s_v) = x^2 - (s_u + s_v)x + s_u*s_v.
# The global minimum of a parabola ax^2 + bx + c (a > 0) is at x = -b / (2a).
# Here, a = 1, b = -(s_u + s_v).
# So the minimum is at x = (s_u + s_v) / 2.
# We require the minimum to be at x = 1.5, so (s_u + s_v) / 2 = 1.5,
# which means s_u + s_v = 3.
# Since s_i \in {1, 2}, this implies s_u != s_v.
# This is a standard 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we prefer 1 over 2.
    # We iterate from 1 to n. If a node is uncolored, we assign it 1.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
