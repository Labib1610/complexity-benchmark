```python
import sys

# The function g(x) = (2x - 3) / (x - 1)
# g(g(x)) = (2((2x-3)/(x-1)) - 3) / (((2x-3)/(x-1)) - 1)
#         = (4x - 6 - 3x + 3) / (2x - 3 - x + 1)
#         = (x - 3) / (x - 2)
# g(g(g(x))) = (2((x-3)/(x-2)) - 3) / (((x-3)/(x-2)) - 1)
#            = (2x - 6 - 3x + 6) / (x - 3 - x + 2)
#            = -x / -1 = x
# So g^(3)(x) = x.
# The condition H_{u,v}(x) = g^(s_u)(g^(s_v)(x)) = x means g^(s_u + s_v)(x) = x.
# Since g^(3)(x) = x, this is equivalent to s_u + s_v ≡ 0 (mod 3).
# s_i ∈ {1, 2}.
# If s_u = 1, then 1 + s_v ≡ 0 (mod 3) => s_v = 2.
# If s_u = 2, then 2 + s_v ≡ 0 (mod 3) => s_v = 1.
# Thus, for every hazard (u, v), s_u must be different from s_v.
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, we prefer 1 over 2.
            # Start BFS/DFS with color 1 for the first node of each component.
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
