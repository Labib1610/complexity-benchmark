import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    if m == 0:
        print("YES")
        print(*( [1] * n ))
        return

    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Energy E = sum_{i=1}^n D_i(s_i^2 - 6s_i) + sum_{(u,v) in links} 2*s_u*s_v + 9m
    # Let x_i = s_i - 1, so x_i in {0, 1}. s_i = x_i + 1.
    # s_i^2 - 6s_i = (x_i+1)^2 - 6(x_i+1) = x_i^2 + 2x_i + 1 - 6x_i - 6 = x_i^2 - 4x_i - 5
    # Since x_i^2 = x_i, this is x_i - 4x_i - 5 = -3x_i - 5.
    # Node energy sum: sum D_i(-3x_i - 5) = -3 sum D_i x_i - 5 sum D_i
    # Since sum D_i = 2m, this is -3 sum D_i x_i - 10m.
    # Link energy: sum 2(x_u+1)(x_v+1) = 2 sum (x_u x_v + x_u + x_v + 1)
    # = 2 sum x_u x_v + 2 sum (x_u + x_v) + 2m
    # Note sum (x_u + x_v) = sum D_i x_i.
    # Total E = -3 sum D_i x_i - 10m + 2 sum x_u x_v + 2 sum D_i x_i + 2m + 9m
    # E = 2 sum x_u x_v - sum D_i x_i + m
    # We want E = 0, so sum D_i x_i - 2 sum x_u x_v = m.
    
    # This is equivalent to: sum_{ (u,v) in links } (x_u + x_v - 2 x_u x_v) = m
    # Note x_u + x_v - 2 x_u x_v is 1 if x_u != x_v, and 0 if x_u == x_v.
    # This is exactly the number of edges connecting nodes with different x values.
    # We want the number of edges between different x values to be m.
    # This means all edges must connect nodes with different x values.
    # This is possible if and only if the graph is bipartite.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, we have two sets of nodes (color 0 and color 1).
    # We can assign x_i = 0 for all i in one set and x_i = 1 for all i in the other.
    # This makes all edges connect different x values.
    # To get lexicographically smallest s_i (s_i = x_i + 1), we want x_i = 0 as much as possible.
    # For each connected component, we can either set (color 0 -> x=0, color 1 -> x=1)
    # or (color 0 -> x=1, color 1 -> x=0).
    # We choose the one that makes the first node in the component have x=0.
    
    res = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            comp = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # In this component, color[i] is the color of the first node.
            # We want x_i = 0, so we map color[i] to x=0.
            c_i = color[i]
            for node in comp:
                if color[node] == c_i:
                    res[node] = 1
                else:
                    res[node] = 2
                    
    print("YES")
    print(*(res[1:]))

solve()
