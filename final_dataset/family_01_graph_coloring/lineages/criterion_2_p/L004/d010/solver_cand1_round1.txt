```python
import sys

# Let s_i = 1 correspond to vector A = (1, 2, 3, 4)
# Let s_i = 2 correspond to vector B = (4, -3, -2, 1)
# Dot product A·A = 1*1 + 2*2 + 3*3 + 4*4 = 1 + 4 + 9 + 16 = 30
# Dot product B·B = 4*4 + (-3)*(-3) + (-2)*(-2) + 1*1 = 16 + 9 + 4 + 1 = 30
# Dot product A·B = 1*4 + 2*(-3) + 3*(-2) + 4*1 = 4 - 6 - 6 + 4 = -4
# Dot product B·A = -4
#
# Total penalty = sum_{hazards (u,v)} (dot product of C_u and C_v)
# We want total penalty = -4 * m.
# Let x_i = 0 if s_i = 1, x_i = 1 if s_i = 2.
# If s_u = s_v, dot product is 30.
# If s_u != s_v, dot product is -4.
# Let k be the number of hazards where s_u != s_v.
# Let m-k be the number of hazards where s_u == s_v.
# Total penalty = (m-k)*30 + k*(-4) = 30m - 30k - 4k = 30m - 34k.
# We want 30m - 34k = -4m => 34m = 34k => k = m.
# This means for every hazard (u, v), we must have s_u != s_v.
# This is equivalent to checking if the graph is bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
