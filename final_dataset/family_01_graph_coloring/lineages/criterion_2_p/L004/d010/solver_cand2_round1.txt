```python
import sys

# The augmented states are A_i = 5*s_i - 1.
# If s_i = 1, A_i = 5(1) - 1 = 4.
# If s_i = 2, A_i = 5(2) - 1 = 9.
# The resonance factor R_{u,v} = gcd(A_u, A_v).
# If s_u = 1 and s_v = 1, R_{u,v} = gcd(4, 4) = 4.
# If s_u = 2 and s_v = 2, R_{u,v} = gcd(9, 9) = 9.
# If s_u = 1 and s_v = 2, R_{u,v} = gcd(4, 9) = 1.
# If s_u = 2 and s_v = 1, R_{u,v} = gcd(9, 4) = 1.
# The global resonance is the product of R_{u,v} for all hazards (u, v).
# For the product to be 1, every R_{u,v} must be 1.
# This means for every hazard (u, v), we must have s_u != s_v.
# This is exactly the condition for a graph to be bipartite.
# We need to check if the graph is bipartite and find the lexicographically smallest 2-coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # To get lexicographically smallest, we prefer color 1 for the first node
            # of each connected component.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
