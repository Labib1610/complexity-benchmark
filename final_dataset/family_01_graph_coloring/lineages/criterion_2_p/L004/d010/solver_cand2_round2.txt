```python
import sys

# The interference multiplier M_{u,v} = gcd(30*s_u + 5*s_v, 5*s_u + 30*s_v)
# Case 1: s_u = 1, s_v = 1
# M = gcd(30+5, 5+30) = gcd(35, 35) = 35
# Case 2: s_u = 2, s_v = 2
# M = gcd(60+10, 10+60) = gcd(70, 70) = 70
# Case 3: s_u = 1, s_v = 2
# M = gcd(30+10, 5+60) = gcd(40, 65) = 5
# Case 4: s_u = 2, s_v = 1
# M = gcd(60+5, 10+30) = gcd(65, 40) = 5
#
# We need the product of M_{u,v} for all m hazards to be 5^m.
# This means every hazard must have M_{u,v} = 5.
# This happens if and only if s_u != s_v.
# This is a standard 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we prefer 1 over 2.
    # We iterate 1 to n, if not colored, start BFS/DFS.
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
