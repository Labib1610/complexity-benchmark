import sys

# The condition "sum of phase states is 3" means if one is 1, the other must be 2.
# This is equivalent to saying the two satellites must have different phase states.
# This is a standard 2-coloring problem on a graph.
# We want the lexicographically smallest sequence, so we iterate from 1 to n.
# If a node is unvisited, we try assigning it 1 (the smaller value).
# If that leads to a contradiction, we try 2. If both fail, output NO.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Try assigning 1 to the first node of the component
            stack = [(i, 1)]
            colors[i] = 1
            component = []
            
            possible = True
            idx_comp = 0
            component.append(i)
            
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        component.append(v)
                    elif colors[v] == colors[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Try assigning 2 instead
                for node in component:
                    colors[node] = 0
                
                colors[i] = 2
                stack = [i]
                possible = True
                idx_comp = 0
                component = [i]
                
                while idx_comp < len(component):
                    u = component[idx_comp]
                    idx_comp += 1
                    for v in adj[u]:
                        if colors[v] == 0:
                            colors[v] = 3 - colors[u]
                            component.append(v)
                        elif colors[v] == colors[u]:
                            possible = False
                            break
                    if not possible:
                        break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(colors[1:]))

solve()
