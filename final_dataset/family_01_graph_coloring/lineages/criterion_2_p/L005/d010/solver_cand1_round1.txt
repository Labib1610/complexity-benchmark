```python
import sys

# The problem states: for each machine i, the sum of power modes of its partners
# plus d_i * (power mode of i) equals 3 * d_i.
# Let c_i be the power mode of machine i (c_i in {1, 2}).
# Let N(i) be the set of neighbors of i.
# The condition is: sum_{j in N(i)} c_j + d_i * c_i = 3 * d_i.
# This can be rewritten as: sum_{j in N(i)} c_j = d_i * (3 - c_i).
# If c_i = 1, then sum_{j in N(i)} c_j = 2 * d_i.
# If c_i = 2, then sum_{j in N(i)} c_j = 1 * d_i.
# Let x_i = c_i - 1, so x_i in {0, 1}.
# If c_i = 1 (x_i = 0), sum_{j in N(i)} (x_j + 1) = 2 * d_i => sum x_j + d_i = 2 * d_i => sum x_j = d_i.
# If c_i = 2 (x_i = 1), sum_{j in N(i)} (x_j + 1) = 1 * d_i => sum x_j + d_i = d_i => sum x_j = 0.
# This means if x_i = 1, all neighbors must have x_j = 0.
# If x_i = 0, the number of neighbors with x_j = 1 must be exactly d_i.
# This implies that if x_i = 0, all neighbors must have x_j = 1.
# Thus, this is a bipartite graph problem where one set is x=0 and the other is x=1.
# If the graph is not bipartite, no solution exists.
# If it is bipartite, we can color it with two colors (0 and 1).
# For each connected component, we have two choices for the bipartition.
# We want the lexicographically smallest c_i, which means we want x_i = 0 as much as possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Try both assignments for the component to find lexicographically smallest
            # Assignment 1: color 0 -> x=0 (c=1), color 1 -> x=1 (c=2)
            # Assignment 2: color 0 -> x=1 (c=2), color 1 -> x=0 (c=1)
            def check(assignment):
                # assignment[0] is value for color 0, assignment[1] for color 1
                vals = [0] * (n + 1)
                for node, col in component:
                    vals[node] = assignment[col]
                for node, col in component:
                    s = sum(vals[neighbor] for neighbor in adj[node])
                    if vals[node] == 1: # c=1
                        if s != 2 * len(adj[node]): return False
                    else: # c=2
                        if s != 1 * len(adj[node]): return False
                return True

            can1 = check([1, 2])
            can2 = check([2, 1])
            
            if not can1 and not can2:
                print("NO")
                return
            
            if can1 and can2:
                # Pick the one that makes the first node in component smaller
                if 1 < 2: # Always prefer 1
                    for node, col in component: res[node] = 1 if col == 0 else 2
                else:
                    for node, col in component: res[node] = 2 if col == 0 else 1
            elif can1:
                for node, col in component: res[node] = 1 if col == 0 else 2
            else:
                for node, col in component: res[node] = 2 if col == 0 else 1
                
    print("YES")
    print(*(res[1:]))

solve()
```
