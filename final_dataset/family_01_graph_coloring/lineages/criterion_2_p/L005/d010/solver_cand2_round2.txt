```python
import sys

# The problem asks for a calibration such that the sum of tokens equals 3m.
# Tokens: (1,1) -> 1, (2,2) -> 2, (1,2) or (2,1) -> 3.
# Let x_i be the mode of machine i (1 or 2).
# Let c1 be the number of pairs (u, v) where x_u = x_v = 1.
# Let c2 be the number of pairs (u, v) where x_u = x_v = 2.
# Let c3 be the number of pairs (u, v) where x_u != x_v.
# We need c1 + 2*c2 + 3*c3 = 3m.
# Since c1 + c2 + c3 = m, we have c1 + 2*c2 + 3*(m - c1 - c2) = 3m
# c1 + 2*c2 + 3m - 3*c1 - 3*c2 = 3m
# -2*c1 - c2 = 0 => 2*c1 + c2 = 0.
# Since c1 and c2 are counts of pairs, they must be >= 0.
# Thus, c1 = 0 and c2 = 0.
# This means there can be no pairs (u, v) such that x_u = x_v.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest, we use BFS/DFS to color components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

solve()
```
