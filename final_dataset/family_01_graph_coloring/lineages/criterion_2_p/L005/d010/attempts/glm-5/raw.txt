```python
import sys
from collections import deque

def solve():
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr]); ptr += 1
    m = int(input[ptr]); ptr += 1
    
    adj = [[] for _ in range(n)]
    deg = [0] * n
    
    for _ in range(m):
        u = int(input[ptr]) - 1; ptr += 1
        v = int(input[ptr]) - 1; ptr += 1
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1
        deg[v] += 1
    
    # c[i] = 0: unassigned, 1: mode 1, 2: mode 2
    c = [0] * n
    
    for i in range(n):
        if c[i] == 0:
            # BFS to assign connected component
            # Try assigning 1 first (lexicographically smallest)
            queue = deque([i])
            c[i] = 1
            ok = True
            
            while queue:
                u = queue.popleft()
                du = deg[u]
                if du == 0:
                    # Isolated node: must satisfy 0 = 0, any mode works. 1 is smallest.
                    continue
                
                # Calculate sum of neighbors' modes
                s = 0
                for v in adj[u]:
                    if c[v] == 0:
                        # Neighbor unassigned: assign opposite mode to satisfy equation
                        # Equation: s + du * c[u] = 3*du
                        # If c[u]=1: s = 2*du, so each neighbor must be 2
                        # If c[u]=2: s = du, so each neighbor must be 1
                        c[v] = 3 - c[u]
                        queue.append(v)
                        s += c[v]
                    else:
                        s += c[v]
                
                # Check if equation is satisfied
                if s + du * c[u] != 3 * du:
                    ok = False
                    break
            
            if not ok:
                print("NO")
                return
    
    print("YES")
    print(' '.join(map(str, c)))

if __name__ == "__main__":
    solve()
```
