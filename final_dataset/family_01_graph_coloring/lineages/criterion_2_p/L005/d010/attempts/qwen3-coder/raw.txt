```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # For each component, we'll do BFS and check bipartiteness
    color = [-1] * (n + 1)
    components = []
    visited = [False] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i] and adj[i]:
            # BFS to get component
            comp_nodes = []
            queue = deque()
            queue.append(i)
            visited[i] = True
            comp_nodes.append(i)
            color[i] = 0
            
            is_bipartite = True
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        comp_nodes.append(neighbor)
                        color[neighbor] = color[node] ^ 1
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        is_bipartite = False
            
            if not is_bipartite:
                print("NO")
                return
            
            components.append(comp_nodes)
    
    # For isolated nodes (no edges), color them 0
    result = [1] * (n + 1)  # default value 1
    for i in range(1, n + 1):
        if color[i] != -1:
            result[i] = color[i] + 1  # Convert 0->1, 1->2
    
    # Now verify the constraint for all edges
    # For each node i, let s_i = sum of neighbors' values + d_i * c_i should equal 3*d_i
    # That means: sum of neighbors + d_i * c_i = 3 * d_i
    # => sum of neighbors = d_i * (3 - c_i)
    # If c_i = 1 => sum of neighbors = 2 * d_i
    # If c_i = 2 => sum of neighbors = 1 * d_i
    
    degree = [0] * (n + 1)
    neighbor_sum = [0] * (n + 1)
    
    for u, v in edges:
        degree[u] += 1
        degree[v] += 1
        neighbor_sum[u] += result[v]
        neighbor_sum[v] += result[u]
    
    for i in range(1, n + 1):
        d_i = degree[i]
        c_i = result[i]
        target = 3 * d_i
        actual = neighbor_sum[i] + d_i * c_i
        if actual != target:
            print("NO")
            return
    
    print("YES")
    print(" ".join(map(str, result[1:])))

if __name__ == "__main__":
    main()
```
