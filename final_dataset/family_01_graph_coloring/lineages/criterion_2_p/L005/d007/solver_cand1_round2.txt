```python
import sys

# The matrix W depends on x and y. The condition that the largest eigenvalue is sqrt(5)
# is equivalent to the condition x != y.
# Proof:
# Let x, y be in {1, 2}.
# If x = y:
# W = (x/9) * [[8, 7, 7], [7, -4, -4], [7, -4, -4]]
# The eigenvalues of this matrix are 0, 0, and (x/9)*(8-4-4) = 0.
# Wait, let's re-evaluate:
# If x=1, y=1: W = (1/9) * [[8, 7, 7], [7, -4, -4], [7, -4, -4]]
# Characteristic polynomial: det(W - lambda*I) = 0
# For x=y=1, eigenvalues are 1, -1, 0. Largest is 1 != sqrt(5).
# If x=1, y=2: W = (1/9) * [[12, 9, 12], [9, -12, -6], [12, -6, 0]]
# The largest eigenvalue is indeed sqrt(5).
# Thus, the condition "largest eigenvalue is sqrt(5)" is equivalent to x != y.
# This is a 2-coloring problem on a graph where edges represent x != y.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS to color component, greedy for lexicographical smallest
            # To get lexicographically smallest, we want color[i] = 1 if possible.
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
