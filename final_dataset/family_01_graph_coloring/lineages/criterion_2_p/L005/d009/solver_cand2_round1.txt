import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Let x_i = c_i - 1, where x_i in {0, 1}.
    # c_i = x_i + 1.
    # v_i = (x_i + 1) * ((x_i + 1) * k_i - sum_{j in N(i)} (x_j + 1))
    # v_i = (x_i + 1) * (x_i * k_i + k_i - sum x_j - k_i)
    # v_i = (x_i + 1) * (x_i * k_i - sum x_j)
    # v_i = x_i^2 * k_i + x_i * k_i - x_i * sum x_j - sum x_j
    # Since x_i^2 = x_i, v_i = x_i * k_i + x_i * k_i - x_i * sum x_j - sum x_j
    # v_i = 2 * x_i * k_i - x_i * sum x_j - sum x_j
    # Sum v_i = sum (2 * x_i * k_i) - sum (x_i * sum x_j) - sum (sum x_j)
    # Note sum (x_i * sum x_j) = 2 * m_x, where m_x is number of edges between nodes with x=1.
    # sum (sum x_j) = sum (x_i * k_i)
    # Sum v_i = 2 * sum (x_i * k_i) - 2 * m_x - sum (x_i * k_i)
    # Sum v_i = sum (x_i * k_i) - 2 * m_x
    # We want Sum v_i = m.
    # Let S be the set of nodes where x_i = 1.
    # Sum_{i in S} k_i - 2 * (edges within S) = m.
    # Let E be the set of all edges. m = |E|.
    # Sum_{i in S} k_i = (edges within S) + (edges between S and S^c)
    # So (edges within S) + (edges between S and S^c) - 2 * (edges within S) = m
    # (edges between S and S^c) - (edges within S) = m
    # This is equivalent to: (edges with at least one endpoint in S) - 2 * (edges within S) = m
    # Let x_i = 0 for all i. Then sum = 0. If m=0, YES 1 1 ... 1.
    # If we set all x_i = 1, then sum = m - 2m = -m.
    # Try setting x_i = 0 for all i: sum = 0. If m=0, YES 1 1 ... 1.
    # Try setting x_i = 1 for all i: sum = -m.
    
    # Actually, the formula simplifies to:
    # If we set all c_i = 1, v_i = 1 * (k_i - k_i) = 0. Sum = 0.
    # If we set all c_i = 2, v_i = 2 * (2*k_i - 2*k_i) = 0. Sum = 0.
    # If m = 0, any combination works. Lexicographically smallest is 1 1 ... 1.
    
    if m == 0:
        print("YES")
        print(*( [1] * n ))
        return

    # Check if any single node flip works?
    # If we set c_i = 2 and others 1:
    # v_i = 2 * (2*k_i - (k_i * 1)) = 2 * k_i
    # For j in N(i), v_j = 1 * (1*k_j - (sum_{l in N(j), l!=i} 1 + 2)) = k_j - (k_j - 1 + 2) = -1
    # For j not in N(i) and j != i, v_j = 0.
    # Sum = 2*k_i + k_i * (-1) = k_i.
    # If k_i = m, then YES.
    
    for i in range(1, n + 1):
        if len(adj[i]) == m:
            res = [1] * n
            res[i-1] = 2
            print("YES")
            print(*res)
            return
            
    print("NO")

solve()
