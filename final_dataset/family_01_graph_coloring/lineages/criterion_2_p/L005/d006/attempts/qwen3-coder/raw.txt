```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # For the matrix S = [[0,x,y],[x,0,0],[y,0,0]]
    # The characteristic polynomial is -t^3 + xy
    # So the eigenvalues are the roots of t^3 = xy
    # The largest eigenvalue is (xy)^(1/3)
    # We need this to be sqrt(5), so xy = (sqrt(5))^3 = 5*sqrt(5)
    # But since x,y are in {1,2}, xy is in {1,2,4}
    # And 5*sqrt(5) is about 11.18, which is not in {1,2,4}
    # Wait, let's recompute properly.
    
    # The characteristic polynomial of S is:
    # det([[ -t,   x,   y ],
    #      [  x,  -t,   0 ],
    #      [  y,   0,  -t ]])
    # = -t*(-t*(-t)) - x*(x*(-t)) - y*(y*(-t)) 
    # = -t^3 + x^2*t + y^2*t
    # = -t^3 + t*(x^2 + y^2)
    # = t*(-t^2 + x^2 + y^2)
    # So eigenvalues are 0 and +/-sqrt(x^2 + y^2)
    # The largest eigenvalue is sqrt(x^2 + y^2)
    # We need sqrt(x^2 + y^2) = sqrt(5)
    # So x^2 + y^2 = 5
    
    # Since x,y in {1,2}:
    # 1^2 + 1^2 = 2
    # 1^2 + 2^2 = 5
    # 2^2 + 1^2 = 5
    # 2^2 + 2^2 = 8
    # So we need exactly one 1 and one 2 in each pair.
    
    # This is a 2-coloring problem: each connected component must be bipartite.
    
    color = [-1] * n
    
    for i in range(n):
        if color[i] == -1:
            # BFS to 2-color the component
            queue = deque()
            queue.append(i)
            color[i] = 0
            
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    # If we reach here, a valid 2-coloring exists.
    # We need to assign modes such that for each edge (u,v), 
    # we have one machine with mode 1 and one with mode 2.
    # This means the coloring directly gives us the assignment.
    # To get lexicographically smallest, we prefer color 0 -> mode 1, color 1 -> mode 2.
    
    result = []
    for i in range(n):
        if color[i] == 0:
            result.append("1")
        else:
            result.append("2")
    
    print("YES")
    print(" ".join(result))

solve()
```
