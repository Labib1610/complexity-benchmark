```python
import sys

# The interference score S = integral from 0 to pi of (cos((x-y)theta) - cos((x+y)theta)) dtheta
# S = [sin((x-y)theta)/(x-y) - sin((x+y)theta)/(x+y)] from 0 to pi
# If x=y, S = integral of (cos(0) - cos(2x*theta)) dtheta = integral of (1 - cos(2x*theta)) dtheta
# S = [theta - sin(2x*theta)/(2x)] from 0 to pi = pi - 0 = pi.
# Since the protocol requires S = 0, we must have x != y.
# If x != y, then x and y are in {1, 2}. The only way x != y is if one is 1 and the other is 2.
# Thus, for every flagged pair (u, v), we must have setting[u] != setting[v].
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # We want the lexicographically smallest configuration.
    # This means we prefer 1 over 2 whenever possible.
    # For each connected component, we can assign the first node 1 or 2.
    # To be lexicographically smallest, we try to assign 1 to the smallest index node.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
