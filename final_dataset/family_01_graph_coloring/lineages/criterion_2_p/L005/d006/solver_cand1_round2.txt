```python
import sys

# The compatibility equation:
# integral_0^pi (cos((x-y)theta) - cos((x+y)theta)) dtheta = 0
# Using the identity cos(A) - cos(B) = -2 sin((A+B)/2) sin((A-B)/2):
# integral_0^pi (2 sin(x*theta) sin(y*theta)) dtheta = 0
# Since x, y are in {1, 2}:
# If x=1, y=1: integral_0^pi (2 sin^2(theta)) dtheta = integral_0^pi (1 - cos(2*theta)) dtheta = pi != 0
# If x=2, y=2: integral_0^pi (2 sin^2(2*theta)) dtheta = integral_0^pi (1 - cos(4*theta)) dtheta = pi != 0
# If x=1, y=2: integral_0^pi (2 sin(theta) sin(2*theta)) dtheta = integral_0^pi (4 sin^2(theta) cos(theta)) dtheta
# Let u = sin(theta), du = cos(theta) dtheta. Limits 0 to 0. Integral = 0.
# Thus, the condition is satisfied if and only if x != y.
# This is a 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
