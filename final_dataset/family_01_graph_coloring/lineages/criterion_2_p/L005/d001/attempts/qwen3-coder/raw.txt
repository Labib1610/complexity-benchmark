```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Color the graph with 2 colors (1 and 2)
    # Each connected component must be bipartite
    color = [0] * (n + 1)  # 0 means unvisited
    visited = [False] * (n + 1)
    
    for start in range(1, n + 1):
        if not visited[start]:
            # BFS to color this component
            queue = deque([start])
            color[start] = 1
            visited[start] = True
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                next_color = 3 - current_color  # 1->2, 2->1
                
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        color[neighbor] = next_color
                        queue.append(neighbor)
                    elif color[neighbor] != next_color:
                        # Conflict found - not bipartite
                        print("NO")
                        return
    
    # Check if the solution satisfies all constraints
    for u, v in edges:
        if color[u] + color[v] != 3:
            print("NO")
            return
    
    # Output the lexicographically smallest solution
    # Since we start with color 1 for each component, this is already minimal
    print("YES")
    print(" ".join(str(color[i]) for i in range(1, n + 1)))

solve()
```
