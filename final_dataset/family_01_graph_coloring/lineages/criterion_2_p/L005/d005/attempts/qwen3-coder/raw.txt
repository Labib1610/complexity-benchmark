```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # For matrix M = [[x, y], [y, x]], eigenvalues are x+y and x-y
    # We need exactly one eigenvalue equal to 3
    # So either x+y=3 or x-y=3, but not both
    # x+y=3 means x=1,y=2 or x=2,y=1
    # x-y=3 means x=3,y=0 or x=4,y=1 etc. But since x,y in {1,2}, x-y in {-1,0,1}
    # So x-y=3 is impossible
    # Therefore, we must have x+y=3 for all edges, meaning (x,y) = (1,2) or (2,1)
    
    # This is equivalent to a 2-coloring problem where adjacent nodes must have different colors
    # Color 1 represents setting 1, color 2 represents setting 2
    # If x=1 and y=2, then x+y=3 ✓
    # If x=2 and y=1, then x+y=3 ✓
    # If x=1 and y=1, then x+y=2 ✗
    # If x=2 and y=2, then x+y=4 ✗
    
    # So we need a bipartite graph - each connected component must be bipartite
    
    color = [0] * (n + 1)  # 0: unvisited, 1: color 1, -1: color -1
    
    def bfs(start):
        queue = deque([start])
        color[start] = 1
        while queue:
            node = queue.popleft()
            for neighbor in adj[node]:
                if color[neighbor] == 0:
                    color[neighbor] = -color[node]
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:
                    return False
        return True
    
    # Check if all components are bipartite
    for i in range(1, n + 1):
        if color[i] == 0 and adj[i]:  # Unvisited node with edges
            if not bfs(i):
                print("NO")
                return
    
    # If bipartite, assign settings
    # We want lexicographically smallest, so we prefer color 1 (setting 1) when possible
    # Since we used 1 and -1 as colors, we map:
    # color 1 -> setting 1
    # color -1 -> setting 2
    
    # But we need to be careful about the lexicographically smallest assignment
    # We should ensure that in each connected component, the node with smaller index
    # gets setting 1 if possible
    
    # Actually, our BFS already ensures a valid 2-coloring. To get lexicographically
    # smallest, we need to make sure nodes with smaller indices get 1 when possible
    
    # Let's re-visit this. In our BFS, we started with color[start] = 1
    # To get lexicographically smallest, if node 1 is in a component, it should be 1
    # If node 1 is not in the component, then the smallest node in the component should be 1
    
    visited = [False] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i] and adj[i]:
            # Find the minimum node in this component
            component = []
            queue = deque([i])
            visited[i] = True
            component.append(i)
            
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        queue.append(neighbor)
                        component.append(neighbor)
            
            # Determine if we should flip colors for lex smallest
            # We want the smallest node in component to have setting 1 (color 1)
            min_node = min(component)
            if color[min_node] == -1:  # Need to flip
                for node in component:
                    color[node] = -color[node]
    
    # Handle isolated nodes (no edges) - assign 1
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
    
    # Verify the solution
    for u, v in edges:
        x, y = (1 if color[u] == 1 else 2), (1 if color[v] == 1 else 2)
        # Check eigenvalues of [[x, y], [y, x]] are x+y and x-y
        # We need exactly one to be 3
        eig1, eig2 = x + y, x - y
        if sorted([eig1, eig2]).count(3) != 1:
            print("NO")
            return
    
    print("YES")
    result = []
    for i in range(1, n + 1):
        result.append("1" if color[i] == 1 else "2")
    print(" ".join(result))

solve()
```
