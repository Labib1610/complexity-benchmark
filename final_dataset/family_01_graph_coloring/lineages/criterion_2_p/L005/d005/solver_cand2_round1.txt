```python
import sys

# The matrices are:
# T1 = [[1, 1], [0, 1]]
# T2 = [[1, 0], [1, 1]]
# T1*T1 = [[1, 2], [0, 1]] (not symmetric)
# T2*T2 = [[1, 0], [2, 1]] (not symmetric)
# T1*T2 = [[2, 1], [1, 1]] (symmetric)
# T2*T1 = [[1, 1], [1, 2]] (symmetric)
#
# The condition "T_A * T_B is symmetric" is satisfied if and only if
# the operators chosen for the pair (u, v) are different (one is T1, one is T2).
# This is a standard 2-coloring problem on a graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unvisited, 1 = T1, 2 = T2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we prefer 1 over 2
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```
