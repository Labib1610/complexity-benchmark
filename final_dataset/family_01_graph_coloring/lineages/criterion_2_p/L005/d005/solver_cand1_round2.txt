```python
import sys

# The interference matrix M = [[x, y], [y, x]] has eigenvalues:
# det(M - lambda*I) = (x - lambda)^2 - y^2 = 0
# (x - lambda) = +/- y
# lambda = x +/- y
# We need exactly one eigenvalue to be 3.
# Possible settings for x, y are {1, 2}.
# Case 1: x=1, y=1 => M = [[1, 1], [1, 1]], eigenvalues: 0, 2. (No 3)
# Case 2: x=1, y=2 => M = [[1, 2], [2, 1]], eigenvalues: 3, -1. (One 3)
# Case 3: x=2, y=1 => M = [[2, 1], [1, 2]], eigenvalues: 3, 1. (One 3)
# Case 4: x=2, y=2 => M = [[2, 2], [2, 2]], eigenvalues: 0, 4. (No 3)
#
# Conclusion: For every pair (u, v), we must have x != y.
# This means the graph must be bipartite.
# If bipartite, we can color with 1 and 2.
# To get the lexicographically smallest, we want as many 1s as possible.
# For each connected component, we have two choices:
# Choice A: nodes in set U get 1, nodes in set V get 2.
# Choice B: nodes in set U get 2, nodes in set V get 1.
# We want to pick the choice that results in a smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two ways to color:
            # Way 1: color 0 -> 1, color 1 -> 2
            # Way 2: color 0 -> 2, color 1 -> 1
            # Compare lexicographically
            way1 = {}
            way2 = {}
            for u, c in component:
                way1[u] = 1 if c == 0 else 2
                way2[u] = 2 if c == 0 else 1
            
            # Find the first node in this component to decide
            comp_nodes = sorted([u for u, c in component])
            first = comp_nodes[0]
            if way1[first] < way2[first]:
                for u in comp_nodes: res[u] = way1[u]
            else:
                for u in comp_nodes: res[u] = way2[u]
                
    print("YES")
    print(*(res[1:]))

solve()
```
