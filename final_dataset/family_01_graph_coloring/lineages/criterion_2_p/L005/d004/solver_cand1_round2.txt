import sys

# The problem states that for every pair (u, v), the product of their power settings must be 2.
# Since power settings are in {1, 2}, the only way for the product to be 2 is if one machine is 1 and the other is 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest configuration, we want as many 1s as possible at the beginning.
# For each connected component, we have two choices of coloring (swapping 1s and 2s).
# We choose the coloring that results in a lexicographically smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            is_bipartite = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings:
            # Option 1: color[v] as is
            # Option 2: 3 - color[v]
            # We want the lexicographically smallest result.
            # Compare the two options for the first node in the component.
            
            # Let's try to make the first node in the component 1 if possible.
            # Actually, we just need to compare the two colorings for the whole component.
            
            c1 = []
            c2 = []
            for node in component:
                c1.append((node, color[node]))
                c2.append((node, 3 - color[node]))
            
            # Sort by node index to compare lexicographically
            c1.sort()
            c2.sort()
            
            # Pick the one that is lexicographically smaller
            best = c1
            for j in range(len(c1)):
                if c1[j][1] < c2[j][1]:
                    best = c1
                    break
                elif c2[j][1] < c1[j][1]:
                    best = c2
                    break
            
            for node, val in best:
                ans[node] = val
                
    print("YES")
    print(*(ans[1:]))

solve()
