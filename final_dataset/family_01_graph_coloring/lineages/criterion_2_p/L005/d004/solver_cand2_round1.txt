```python
import sys

# The problem states:
# Protocol 1: [X, Y, Z] -> [Y, Z, X]
# Protocol 2: [X, Y, Z] -> [Z, X, Y]
# A link between u and v requires that applying the protocol of u then v (or v then u)
# results in the identity transformation [X, Y, Z] -> [X, Y, Z].
# Let P1 be the permutation (1 2 3) and P2 be (3 2 1).
# P1 * P1 = (1 3 2)
# P2 * P2 = (1 3 2)
# P1 * P2 = Identity
# P2 * P1 = Identity
# Thus, for every link (u, v), the protocols must be different (one 1, one 2).
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we prefer 1 over 2.
    # We iterate 1 to n. If not colored, start BFS/DFS.
    # To make it lexicographically smallest, we greedily assign 1 to the 
    # smallest available node in each connected component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
