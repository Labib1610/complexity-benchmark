```python
import sys

# The condition for each machine i with frequency f_i and degree k_i is:
# Sum_{j in neighbors(i)} (f_j^3) = 15 * k_i - 7 * k_i * f_i
# Let f_i = 1: Sum(f_j^3) = 15 * k_i - 7 * k_i = 8 * k_i
# Let f_i = 2: Sum(f_j^3) = 15 * k_i - 14 * k_i = k_i
#
# Let x_i be a variable such that f_i = 1 corresponds to x_i = 0 and f_i = 2 corresponds to x_i = 1.
# Then f_i = x_i + 1.
# f_i^3 = (x_i + 1)^3 = x_i^3 + 3x_i^2 + 3x_i + 1.
# Since x_i is 0 or 1, x_i^3 = x_i^2 = x_i.
# So f_i^3 = x_i + 3x_i + 3x_i + 1 = 7x_i + 1.
#
# The condition for f_i = 1 (x_i = 0): Sum(7x_j + 1) = 8 * k_i
# 7 * Sum(x_j) + k_i = 8 * k_i  => 7 * Sum(x_j) = 7 * k_i => Sum(x_j) = k_i
# The condition for f_i = 2 (x_i = 1): Sum(7x_j + 1) = k_i
# 7 * Sum(x_j) + k_i = k_i => 7 * Sum(x_j) = 0 => Sum(x_j) = 0
#
# Summary:
# If x_i = 0, Sum_{j in neighbors(i)} x_j = k_i (all neighbors must be 1, i.e., x_j = 1)
# If x_i = 1, Sum_{j in neighbors(i)} x_j = 0 (all neighbors must be 0, i.e., x_j = 0)
# This is exactly the definition of a bipartite graph coloring where x_i is the color.
# If x_i = 0, all neighbors must have x_j = 1.
# If x_i = 1, all neighbors must have x_j = 0.
# This is possible if and only if the graph is bipartite.
# To get the lexicographically smallest configuration (f_i), we want x_i = 0 (f_i = 1)
# as much as possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # color[i] is 0 or 1. f_i = color[i] + 1.
    # This is the lexicographically smallest because we prefer 1 over 2.
    res = [str(color[i] + 1) for i in range(1, n + 1)]
    print("YES")
    print(" ".join(res))

solve()
```
