import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # If a machine is paired with itself, the condition is:
            # (mode_i) + 1 * mode_i = 3 * 1 => 2 * mode_i = 3
            # Impossible since mode_i is 1 or 2.
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The condition: sum_{j in neighbors(i)} (mode_j) + k_i * mode_i = 3 * k_i
    # Let x_i = mode_i - 1. Since mode_i in {1, 2}, x_i in {0, 1}.
    # sum_{j in neighbors(i)} (x_j + 1) + k_i * (x_i + 1) = 3 * k_i
    # sum_{j in neighbors(i)} x_j + k_i + k_i * x_i + k_i = 3 * k_i
    # sum_{j in neighbors(i)} x_j + k_i * x_i = k_i
    # This is a system of linear equations over integers.
    # If the graph is bipartite, we can set x_i = 0 for one partition and x_i = 1 for the other.
    # Let's check if the graph is bipartite.
    
    color = [-1] * (n + 1)
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                assignment[u] = c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # Check if the bipartite assignment satisfies the condition
    # sum_{j in neighbors(i)} x_j + k_i * x_i = k_i
    # If x_i = 0, sum_{j in neighbors(i)} x_j = k_i. Since x_j in {0, 1}, this means all neighbors must have x_j = 1.
    # If x_i = 1, sum_{j in neighbors(i)} x_j + k_i = k_i => sum_{j in neighbors(i)} x_j = 0. This means all neighbors must have x_j = 0.
    # This is exactly what bipartite coloring provides.
    
    # Lexicographically smallest: we want as many 1s as possible.
    # The bipartite components can be flipped.
    # For each component, we have two choices: (x_i = 0 or x_i = 1).
    # We want to choose the one that makes the sequence of (x_i + 1) lexicographically smallest.
    
    visited = [False] * (n + 1)
    final_x = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Try both flips
            # Option 1: assignment[i] stays as is
            # Option 2: assignment[i] flips
            
            def get_seq(flip):
                res = []
                for node in component:
                    val = (assignment[node] ^ flip)
                    res.append(val)
                return res
            
            # We want the smallest sequence of (x_i + 1)
            # Since we want lexicographically smallest, we compare the sequences
            # sorted by node index.
            comp_sorted = sorted(component)
            
            seq0 = [assignment[node] for node in comp_sorted]
            seq1 = [1 - assignment[node] for node in comp_sorted]
            
            if seq0 <= seq1:
                for node in component:
                    final_x[node] = assignment[node]
            else:
                for node in component:
                    final_x[node] = 1 - assignment[node]
                    
    print("YES")
    print(*(x + 1 for x in final_x[1:]))

solve()
