import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    d = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        d[u] += 1
        d[v] += 1
        adj[u].append(v)
        adj[v].append(u)
    
    # The condition 3L - 2S = 5m
    # L = sum(d_i * c_i)
    # S = sum(c_u * c_v) for each edge (u, v)
    # 3 * sum(d_i * c_i) - 2 * sum(c_u * c_v) = 5m
    # Let c_i = 1 + x_i, where x_i in {0, 1}
    # 3 * sum(d_i * (1 + x_i)) - 2 * sum((1 + x_u)(1 + x_v)) = 5m
    # 3 * sum(d_i) + 3 * sum(d_i * x_i) - 2 * sum(1 + x_u + x_v + x_u * x_v) = 5m
    # Note sum(d_i) = 2m
    # 6m + 3 * sum(d_i * x_i) - 2m - 2 * sum(x_u + x_v) - 2 * sum(x_u * x_v) = 5m
    # 4m + 3 * sum(d_i * x_i) - 2 * sum(d_i * x_i) - 2 * sum(x_u * x_v) = 5m
    # sum(d_i * x_i) - 2 * sum(x_u * x_v) = m
    
    # This is equivalent to sum(x_i * (d_i - 2 * sum_{j in adj(i)} x_j)) = m
    # If we set x_i = 0 for all i, we get 0 = m. If m=0, YES 1 1 ... 1.
    # If we set x_i = 1 for all i, we get sum(d_i) - 2m = 2m - 2m = 0 = m. If m=0, YES 2 2 ... 2.
    
    # Let's try to find a solution where x_i is 0 or 1.
    # The equation is sum_{i=1}^n x_i * d_i - 2 * sum_{(u,v) \in E} x_u * x_v = m
    
    # Try setting x_i = 1 for all i: sum(d_i) - 2m = 2m - 2m = 0. This equals m only if m=0.
    # Try setting x_i = 0 for all i: 0 = m. Only if m=0.
    
    # What if we set x_i = 1 for some subset?
    # If we set x_i = 1 for all i, we get 0.
    # If we set x_i = 0 for all i, we get 0.
    
    # Actually, the equation simplifies to:
    # sum_{i=1}^n (c_i - 1) * d_i - 2 * sum_{(u,v) \in E} (c_u - 1)(c_v - 1) = m
    # Let y_i = c_i - 1. y_i \in {0, 1}.
    # sum y_i * d_i - 2 * sum y_u * y_v = m
    
    # If we pick a subset of vertices V' and set y_i = 1 for i in V', y_i = 0 otherwise:
    # sum_{i \in V'} d_i - 2 * |edges within V'| = m
    # This is exactly the number of edges with exactly one endpoint in V'.
    # We need to find a subset V' such that the number of edges crossing the cut (V', V\V') is m.
    # The total number of edges is m. So we need all edges to cross the cut.
    # This means the graph must be bipartite.
    
    # If the graph is bipartite, let the two partitions be A and B.
    # If we set V' = A, then all edges cross the cut. The number of crossing edges is m.
    # So y_i = 1 for i in A, y_i = 0 for i in B.
    # c_i = 2 for i in A, c_i = 1 for i in B.
    
    import collections
    color = {}
    for i in range(1, n + 1):
        if i not in color:
            color[i] = 0
            queue = collections.deque([i])
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if v not in color:
                        color[v] = 1 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    # We have two possible assignments:
    # 1. c_i = 2 if color[i] == 0, c_i = 1 if color[i] == 1
    # 2. c_i = 1 if color[i] == 0, c_i = 2 if color[i] == 1
    # We want the lexicographically smallest.
    
    res1 = [0] * n
    res2 = [0] * n
    for i in range(1, n + 1):
        res1[i-1] = 2 if color[i] == 0 else 1
        res2[i-1] = 1 if color[i] == 0 else 2
        
    if res1 < res2:
        print("YES")
        print(*(res1))
    else:
        print("YES")
        print(*(res2))

solve()
