```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)          # final colors: 1 or 2
    temp = [ -1 ] * (n + 1)        # temporary bipartite colors: 0 or 1

    for start in range(1, n + 1):
        if color[start]:
            continue
        # BFS this component
        q = deque([start])
        temp[start] = 0
        comp = [start]
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if temp[v] == -1:
                    temp[v] = temp[u] ^ 1
                    q.append(v)
                    comp.append(v)
                elif temp[v] == temp[u]:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            print("NO")
            return
        # decide mapping to final colors to make start = 1
        if temp[start] == 0:
            map0, map1 = 1, 2
        else:
            map0, map1 = 2, 1
        for v in comp:
            color[v] = map0 if temp[v] == 0 else map1
            temp[v] = -1   # reset for possible future components (not needed but clean)

    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
