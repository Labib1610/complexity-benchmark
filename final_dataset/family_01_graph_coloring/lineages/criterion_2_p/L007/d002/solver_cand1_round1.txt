```python
import sys

# The condition f_u^{f_v} + f_v^{f_u} = 3
# If f_u = 1, f_v = 1: 1^1 + 1^1 = 2 != 3
# If f_u = 1, f_v = 2: 1^2 + 2^1 = 1 + 2 = 3 (Valid)
# If f_u = 2, f_v = 1: 2^1 + 1^2 = 2 + 1 = 3 (Valid)
# If f_u = 2, f_v = 2: 2^2 + 2^2 = 4 + 4 = 8 != 3
# Thus, for every edge (u, v), f_u must be different from f_v.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest assignment, we want f_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we prefer color 1
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
