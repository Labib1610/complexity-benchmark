```python
import sys

# The problem states that for each link (u, v), the settings must satisfy
# max(f_u, f_v) / min(f_u, f_v) = 2.
# Since settings are in {1, 2}, this means one must be 1 and the other must be 2.
# This is equivalent to saying the graph must be bipartite.
# We need the lexicographically smallest assignment, which means we prefer 1 over 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we want to assign 1
    # to as many nodes as possible, prioritizing smaller indices.
    # In each connected component, we can pick a starting node and assign it 1.
    # Then the rest of the component is forced.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign color 1
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
