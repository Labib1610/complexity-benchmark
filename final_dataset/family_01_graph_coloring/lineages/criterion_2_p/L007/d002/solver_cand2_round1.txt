```python
import sys

# The problem states that for any critical link (u, v), the ratio of the 
# maximum configuration value to the minimum must be exactly 2.
# Since the values are {1, 2}, this means one tower must be 1 and the other 2.
# This is equivalent to a 2-coloring problem on a graph.
# A graph is 2-colorable if and only if it is bipartite.
# We need the lexicographically smallest assignment, which means we want 
# tower 1 to be 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a component is not colored, we try assigning 1 to the first node.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
