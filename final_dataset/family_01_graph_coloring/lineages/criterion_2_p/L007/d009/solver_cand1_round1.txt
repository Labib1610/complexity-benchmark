```python
import sys

# The condition is f_v(f_u(x)) = x for all x in {0, 1, 2, 3, 4}.
# f1(x) = 2x + 1 mod 5
# f2(x) = 3x + 2 mod 5
#
# Check combinations:
# f1(f1(x)) = 2(2x+1)+1 = 4x+3 mod 5 (Not x)
# f1(f2(x)) = 2(3x+2)+1 = 6x+5 = x mod 5 (Valid)
# f2(f1(x)) = 3(2x+1)+2 = 6x+5 = x mod 5 (Valid)
# f2(f2(x)) = 3(3x+2)+2 = 9x+8 = 4x+3 mod 5 (Not x)
#
# So, for an edge (u, v), we must have (b_u, b_v) in {(1, 2), (2, 1)}.
# This is a bipartite coloring problem.
# We want to minimize the sequence b_1, b_2, ..., b_n lexicographically.
# This means we prefer b_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To minimize lexicographically, try setting color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

solve()
```
