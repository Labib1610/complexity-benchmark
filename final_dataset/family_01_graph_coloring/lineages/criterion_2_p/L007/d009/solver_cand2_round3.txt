import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Condition: integral from -1 to 1 of P_u(x) * P_v(x) dx < 0
    # If b_i = 1, P_i(x) = x - 1
    # If b_i = 2, P_i(x) = x + 1
    
    # Case 1: b_u = 1, b_v = 1 => (x-1)(x-1) = x^2 - 2x + 1. Integral = [x^3/3 - x^2 + x] from -1 to 1 = (1/3 - 1 + 1) - (-1/3 - 1 - 1) = 1/3 + 7/3 = 8/3 > 0
    # Case 2: b_u = 2, b_v = 2 => (x+1)(x+1) = x^2 + 2x + 1. Integral = [x^3/3 + x^2 + x] from -1 to 1 = (1/3 + 1 + 1) - (-1/3 + 1 - 1) = 7/3 + 1/3 = 8/3 > 0
    # Case 3: b_u = 1, b_v = 2 => (x-1)(x+1) = x^2 - 1. Integral = [x^3/3 - x] from -1 to 1 = (1/3 - 1) - (-1/3 + 1) = -2/3 - 2/3 = -4/3 < 0
    
    # The condition is satisfied if and only if b_u != b_v.
    # This is a bipartite graph coloring problem.
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Greedy approach for lexicographical smallest:
            # Try assigning 1 to the first node of each component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
