import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The equation B^T L B = m where b_i in {1, 2}
    # Let x_i = b_i - 1, so x_i in {0, 1}.
    # B = X + 1 (vector of ones).
    # (X+1)^T L (X+1) = X^T L X + X^T L 1 + 1^T L X + 1^T L 1 = m
    # Since L is the Laplacian, L 1 = 0 and 1^T L = 0.
    # So X^T L X = m.
    # X^T L X = sum_{(u,v) in E} (x_u - x_v)^2 = m.
    # Since x_u, x_v in {0, 1}, (x_u - x_v)^2 is 1 if x_u != x_v and 0 if x_u == x_v.
    # The equation becomes: number of edges with different x_u, x_v equals m.
    # This means every edge must have different x_u, x_v.
    # This is equivalent to saying the graph is bipartite.
    
    color = [-1] * (n + 1)
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                assignment[u] = c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # We want lexicographically smallest B = (x_1+1, x_2+1, ..., x_n+1).
    # This is equivalent to lexicographically smallest X = (x_1, ..., x_n).
    # For each connected component, we have two choices for coloring (0,1) or (1,0).
    # To make X lexicographically smallest, for each component, pick the coloring
    # that results in a smaller value for the first node encountered in the component.
    
    visited = [False] * (n + 1)
    final_x = [0] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Check which coloring is better: color[i]=0 or color[i]=1
            # If we set color[i]=0, x_i=0. If color[i]=1, x_i=1.
            # Since we want smallest X, we prefer x_i=0.
            # The current assignment already has x_i = color[i].
            # If color[i] == 1, we flip all colors in this component.
            if color[i] == 1:
                for node in component:
                    assignment[node] = 1 - assignment[node]
                    
    print("YES")
    print(*(assignment[i] + 1 for i in range(1, n + 1)))

solve()
