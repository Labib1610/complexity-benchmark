```python
import sys
from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # If no edges, any assignment works, prefer all 1s
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Determine what combinations are invalid:
    # C1 = [[1,1],[1,1]], C2 = [[1,-1],[-1,1]]
    # C1 + C1 = [[2,2],[2,2]] -> det = 0 (invalid)
    # C1 + C2 = [[2,0],[0,2]] -> det = 4 (valid)
    # C2 + C1 = [[2,0],[0,2]] -> det = 4 (valid)
    # C2 + C2 = [[2,-2],[-2,2]] -> det = 0 (invalid)
    # So: u and v must have different bases (i.e., not both 1 and not both 2)
    # This is a 2-coloring problem: each edge is a constraint that endpoints have different colors
    
    # We model this as a graph where each edge requires its endpoints to have different values
    # Let's use 0 for basis 1 and 1 for basis 2
    
    color = [-1] * n  # -1 means unvisited
    
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 0  # Assign basis 1 (0) to start node
        
        while queue:
            u = queue.popleft()
            c = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 1 - c
                    queue.append(v)
                elif color[v] == c:
                    return False  # Conflict: adjacent nodes have same color
        return True
    
    # Check if the graph is bipartite
    for i in range(n):
        if color[i] == -1:
            if not bfs(i):
                print("NO")
                return
    
    # If we reach here, a valid coloring exists
    # Convert back to basis labels (0 -> 1, 1 -> 2)
    assignment = [str(c + 1) for c in color]
    print("YES")
    print(" ".join(assignment))

solve()
```
