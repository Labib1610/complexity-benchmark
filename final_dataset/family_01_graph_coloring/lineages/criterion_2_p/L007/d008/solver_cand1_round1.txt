import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    deg = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1
        deg[v] += 1
        idx += 2
        
    # Let x_i = b_i - 1, so x_i in {0, 1}.
    # b_i = x_i + 1.
    # 2 * sum_{(u,v) in E} (x_u+1)(x_v+1) + 5m = 3 * sum_{i=1}^n d_i(x_i+1)
    # 2 * sum (x_u x_v + x_u + x_v + 1) + 5m = 3 * sum d_i x_i + 3 * sum d_i
    # Note: sum_{(u,v) in E} (x_u + x_v) = sum_{i=1}^n d_i x_i
    # 2 * sum x_u x_v + 2 * sum d_i x_i + 2m + 5m = 3 * sum d_i x_i + 3 * 2m
    # 2 * sum x_u x_v + 7m = 3 * sum d_i x_i + 6m
    # 2 * sum x_u x_v + m = 3 * sum d_i x_i
    
    # This is a constraint satisfaction problem.
    # Since we want the lexicographically smallest b_i, we want the lexicographically
    # smallest x_i (0 before 1).
    
    # Try setting x_i greedily.
    x = [0] * (n + 1)
    
    # The equation is 2 * sum_{u<v} A_uv x_u x_v + m = 3 * sum d_i x_i
    # where A_uv is 1 if edge exists, 0 otherwise.
    # Let f(x) = 3 * sum d_i x_i - 2 * sum_{u<v} A_uv x_u x_v
    # We need f(x) = m.
    
    # Since we need lexicographically smallest, we try x_i = 0, then x_i = 1.
    # This is a backtracking problem. However, n is 2*10^5.
    # Let's re-examine: 2 * sum x_u x_v + m = 3 * sum d_i x_i
    # This is equivalent to sum_{i=1}^n x_i (3 d_i - 2 * sum_{j in adj(i), j < i} x_j) = m
    # Wait, the sum is over all edges. 2 * sum_{u<v} x_u x_v = sum_{i=1}^n x_i * (sum_{j in adj(i), j < i} x_j)
    # So sum_{i=1}^n x_i (3 d_i - 2 * sum_{j in adj(i), j < i} x_j) = m
    
    # This allows a greedy approach!
    # For i = 1 to n:
    #   Try x_i = 0: check if it's possible to satisfy the remaining sum.
    #   This is still hard. But notice the equation:
    #   sum_{i=1}^n x_i (3 d_i - 2 * sum_{j in adj(i), j < i} x_j) = m
    
    # Actually, the equation is 2 * sum_{u<v} x_u x_v + m = 3 * sum d_i x_i
    # Let S = sum_{i=1}^n x_i (3 d_i - 2 * sum_{j in adj(i), j < i} x_j)
    # We need S = m.
    
    x = [0] * (n + 1)
    current_sum = 0
    for i in range(1, n + 1):
        # Try x_i = 0
        val0 = 0
        # If we pick x_i = 0, contribution is 0.
        # If we pick x_i = 1, contribution is (3 * d_i - 2 * sum_{j in adj(i), j < i} x_j)
        contrib1 = 3 * deg[i] - 2 * sum(x[j] for j in adj[i] if j < i)
        
        if current_sum == m:
            x[i] = 0
        elif current_sum + contrib1 <= m:
            # This is a heuristic. Is it always possible?
            # The problem might be simpler. Let's check if a solution exists.
            x[i] = 1
            current_sum += contrib1
        else:
            x[i] = 0
            
    if current_sum == m:
        print("YES")
        print(*(x[i] + 1 for i in range(1, n + 1)))
    else:
        print("NO")

solve()
