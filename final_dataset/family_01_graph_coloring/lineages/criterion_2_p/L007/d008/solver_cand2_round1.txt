```python
import sys

# Let V1 = (5, 2, -1, 3) and V2 = (-4, -3, -1, 3)
# V1 . V1 = 25 + 4 + 1 + 9 = 39
# V2 . V2 = 16 + 9 + 1 + 9 = 35
# V1 . V2 = -20 - 6 + 1 + 9 = -16
#
# Condition: V_u . V_v < 0
# If u and v have same basis:
#   If both 1: V1 . V1 = 39 > 0 (Invalid)
#   If both 2: V2 . V2 = 35 > 0 (Invalid)
# If u and v have different basis:
#   V1 . V2 = -16 < 0 (Valid)
#
# Thus, the condition V_u . V_v < 0 is equivalent to saying
# that u and v must have different bases.
# This is a 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color component
            # To get lexicographically smallest, we prefer basis 1 (color 1)
            # for the first node of each component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
