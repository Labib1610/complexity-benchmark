import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Let f_i be the flag for tower i.
    # f_i = 1 means rotation +2pi/3, f_i = 2 means rotation -2pi/3.
    # Let x_i = 1 if f_i = 1, x_i = -1 if f_i = 2.
    # theta_i(t) = t * x_i * (2pi/3).
    # Condition: cos(theta_u(t) + theta_v(t)) = 1 for all t >= 1.
    # cos(t * (x_u + x_v) * 2pi/3) = 1
    # This holds if (x_u + x_v) * 2pi/3 is a multiple of 2pi.
    # (x_u + x_v) * 1/3 = k, where k is an integer.
    # Since x_i in {1, -1}, x_u + x_v in {2, 0, -2}.
    # The only way for (x_u + x_v)/3 to be an integer is if x_u + x_v = 0.
    # So x_u = -x_v.
    # If f_i = 1, x_i = 1. If f_i = 2, x_i = -1.
    # x_u = -x_v means f_u != f_v.
    # This is a bipartite graph coloring problem.

    colors = [0] * (n + 1)
    for i in range(1, n + 1):
        if colors[i] == 0:
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return

    print("YES")
    print(*(colors[1:]))

solve()
