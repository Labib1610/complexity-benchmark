```python
import sys

# The condition is X^2 - (b_u + b_v + 2)Y^2 = -1.
# This is a Pell-like equation X^2 - DY^2 = -1.
# A solution exists if and only if D is not of the form k^2 and 
# the continued fraction of sqrt(D) has an odd period length.
# Possible values for D = b_u + b_v + 2:
# If b_u=1, b_v=1: D = 1+1+2 = 4. X^2 - 4Y^2 = -1 => X^2 - (2Y)^2 = -1.
#   (X-2Y)(X+2Y) = -1. No integer solution.
# If b_u=1, b_v=2: D = 1+2+2 = 5. X^2 - 5Y^2 = -1.
#   Fundamental solution to X^2 - 5Y^2 = -1 is (2, 1). Valid.
# If b_u=2, b_v=2: D = 2+2+2 = 6. X^2 - 6Y^2 = -1.
#   Fundamental solution to X^2 - 6Y^2 = 1 is (5, 2).
#   For X^2 - 6Y^2 = -1, no solution exists (6 is not a sum of two squares, etc).
# Thus, the condition is satisfied if and only if (b_u, b_v) is (1, 2) or (2, 1).
# This means b_u != b_v.
# The problem reduces to checking if the graph is bipartite.
# If bipartite, we want to minimize the sequence b_1, ..., b_n lexicographically.
# This means we want b_1 = 1. Then b_i is determined by the connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To minimize lexicographically, set the first node of each component to 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
