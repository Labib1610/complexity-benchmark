```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # We need to find an assignment of values in {0, 1} to each node
    # such that for every edge (u, v), val[u] != val[v].
    # This is a bipartite graph check.
    # We want the lexicographically smallest assignment (interpreted as base-3 number).
    # Since we prefer smaller digits, we prefer 0 over 1.
    # We perform BFS. When visiting a new node, we assign the smallest valid value.
    # If unvisited, assign 0. If its neighbor has value x, we must assign 1-x.
    # Since we process nodes in order 0 to n-1, and assign 0 whenever possible,
    # this yields the lexicographically smallest valid assignment.

    color = [-1] * n
    possible = True

    for i in range(n):
        if color[i] == -1:
            # Start BFS for this component
            queue = deque([i])
            color[i] = 0
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break

    if not possible:
        print("NO")
    else:
        print("YES")
        # Convert 0-based colors (0, 1) to 1-based bases (1, 2)
        # 0 -> 1, 1 -> 2
        result = [str(x + 1) for x in color]
        print(" ".join(result))

if __name__ == '__main__':
    solve()
```
