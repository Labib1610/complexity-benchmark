```python
import sys

# Matrix properties:
# S1 = [[0, 1, 0], [0, 0, 1], [1, 0, 0]]
# S2 = [[0, 0, 1], [1, 0, 0], [0, 1, 0]]
# S1^2 = [[0, 0, 1], [1, 0, 0], [0, 1, 0]] = S2
# S2^2 = [[0, 1, 0], [0, 0, 1], [1, 0, 0]] = S1
# S1 * S2 = I (Identity) -> Tr(S1*S2) = 3
# S2 * S1 = I (Identity) -> Tr(S2*S1) = 3
# S1 * S1 = S2 -> Tr(S2) = 0
# S2 * S2 = S1 -> Tr(S1) = 0
# Condition Tr(Mu * Mv) = 3 means:
# If u != v: Mu must be different from Mv (S1*S2 or S2*S1)
# If u == v: Mu * Mu must be I, but S1^2=S2 and S2^2=S1, so Tr(Mu^2) is 0.
# Thus, if u == v, no solution exists.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # 2-coloring problem: Mu != Mv means different bases
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # To minimize the sequence lexicographically, 
            # we prefer basis 1 (color 1) over basis 2 (color 2)
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
