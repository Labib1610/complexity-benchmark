```python
import sys

# The condition for sensor i:
# If b_i = 1, product of neighbors = 2^d_i
# If b_i = 2, product of neighbors = 1
# Let x_i = log2(b_i). Then b_i = 2^x_i, where x_i in {0, 1}.
# The condition becomes:
# If x_i = 0, sum of neighbors' x_j = d_i (mod 2)
# If x_i = 1, sum of neighbors' x_j = 0 (mod 2)
# This is equivalent to: sum_{j in neighbors(i)} x_j = d_i * (1 - x_i) (mod 2)
# sum_{j in neighbors(i)} x_j + d_i * x_i = d_i (mod 2)
# This is a system of linear equations over GF(2).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # System: Ax = b where A_ii = d_i, A_ij = 1 if j in neighbors(i), else 0
    # b_i = d_i % 2
    # We want to minimize the sequence b_1, ..., b_n lexicographically.
    # This means we want x_i = 0 if possible (since b_i = 2^x_i, x_i=0 => b_i=1).
    
    # The system is: sum_{j in adj(i)} x_j + d_i * x_i = d_i (mod 2)
    # This is equivalent to: sum_{j in adj(i)} x_j + x_i * (d_i % 2) = d_i % 2
    
    # Let's use Gaussian elimination or component-wise solving.
    # Since we want the lexicographically smallest, we try to set x_i = 0.
    
    visited = [False] * n
    assignment = [0] * n
    
    for i in range(n):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Solve for this component
            # Equations: sum_{j in adj(u)} x_j + x_u * (d_u % 2) = d_u % 2
            # This is a system of linear equations.
            # We can use Gaussian elimination.
            comp_map = {node: idx for idx, node in enumerate(component)}
            size = len(component)
            matrix = [[0] * (size + 1) for _ in range(size)]
            for r, u in enumerate(component):
                for v in adj[u]:
                    matrix[r][comp_map[v]] = 1
                matrix[r][r] ^= (len(adj[u]) % 2)
                matrix[r][size] = len(adj[u]) % 2
            
            # Gaussian elimination
            pivot_row = 0
            for col in range(size):
                sel = -1
                for r in range(pivot_row, size):
                    if matrix[r][col]:
                        sel = r
                        break
                if sel != -1:
                    matrix[pivot_row], matrix[sel] = matrix[sel], matrix[pivot_row]
                    for r in range(size):
                        if r != pivot_row and matrix[r][col]:
                            for c in range(col, size + 1):
                                matrix[r][c] ^= matrix[pivot_row][c]
                    pivot_row += 1
            
            # Check consistency
            for r in range(pivot_row, size):
                if matrix[r][size]:
                    print("NO")
                    return
            
            # Back substitution to find smallest x
            # Free variables should be 0 to minimize x_i
            res = [0] * size
            is_pivot = [-1] * size
            curr = 0
            for r in range(pivot_row):
                for c in range(size):
                    if matrix[r][c]:
                        is_pivot[c] = r
                        break
            
            for c in range(size):
                if is_pivot[c] != -1:
                    row = is_pivot[c]
                    val = matrix[row][size]
                    for next_c in range(c + 1, size):
                        if matrix[row][next_c]:
                            val ^= res[next_c]
                    res[c] = val
            
            for idx, node in enumerate(component):
                assignment[node] = res[idx] + 1
                
    print("YES")
    print(*(assignment))

solve()
```
