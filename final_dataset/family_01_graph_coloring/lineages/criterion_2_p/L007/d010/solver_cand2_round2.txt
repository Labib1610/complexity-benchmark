```python
import sys

# The Cartesian product G \square H has an Eulerian circuit if and only if
# every vertex in G \square H has an even degree.
# Let G be a graph with n_G vertices and m_G edges.
# Let H be a graph with n_H vertices and m_H edges.
# The degree of a vertex (u, v) in G \square H is deg_G(u) + deg_H(v).
# Topology 1: 2-node linear graph (K2). Vertices: 2, Edges: 1.
# Degrees: 1, 1.
# Topology 2: 3-node ring graph (C3). Vertices: 3, Edges: 3.
# Degrees: 2, 2, 2.
#
# Case 1: G=K2, H=K2. Degrees in G \square H: 1+1=2. All vertices have degree 2.
# Eulerian circuit exists.
# Case 2: G=K2, H=C3. Degrees in G \square H: 1+2=3. All vertices have degree 3.
# No Eulerian circuit (odd degrees).
# Case 3: G=C3, H=C3. Degrees in G \square H: 2+2=4. All vertices have degree 4.
# Eulerian circuit exists.
#
# Robustness condition: G \square H must NOT contain an Eulerian circuit.
# This means we cannot have (G=K2, H=K2) or (G=C3, H=C3).
# Thus, for every link (u, v), we must have b_u != b_v.
# This is a bipartite coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # We want to minimize the sequence b_1, ..., b_n.
    # b_i \in {1, 2}.
    # For each connected component, we can choose the color of the first node
    # to be 1 or 2. To minimize the sequence, we greedily pick 1 if possible.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
