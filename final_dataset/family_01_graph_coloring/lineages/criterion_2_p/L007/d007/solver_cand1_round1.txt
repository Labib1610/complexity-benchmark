```python
import sys

# The condition A_u(A_v(x)) = x^6:
# If f_u=1, f_v=1: (x^2)^2 = x^4 != x^6
# If f_u=1, f_v=2: (x^3)^2 = x^6 (Valid)
# If f_u=2, f_v=1: (x^2)^3 = x^6 (Valid)
# If f_u=2, f_v=2: (x^3)^3 = x^9 != x^6
# Thus, for every link (u, v), f_u != f_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (f_1, f_2, ..., f_n).
# To minimize f_1, we set f_1 = 1. Then propagate constraints.
# If a component is not fully determined, we set the first uncolored node to 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color component
            # To get lexicographically smallest, we prefer 1 over 2
            stack = [(i, 1)]
            assignment[i] = 1
            while stack:
                u, color = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - color
                        stack.append((v, assignment[v]))
                    elif assignment[v] == color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
