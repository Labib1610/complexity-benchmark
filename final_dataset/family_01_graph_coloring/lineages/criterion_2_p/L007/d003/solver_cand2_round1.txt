```python
import sys

# The problem states:
# Configuration 1: (High, Low)
# Configuration 2: (Low, High)
# Stability: At each step, one is High and one is Low.
# If tower u has config f_u and tower v has config f_v:
# Step 1: u is High if f_u=1, Low if f_u=2. v is High if f_v=1, Low if f_v=2.
# Stability requires: (f_u=1 and f_v=2) OR (f_u=2 and f_v=1).
# This is equivalent to f_u != f_v.
# This is a standard 2-coloring problem on a graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # BFS to color the component
            # To get lexicographically smallest, we prefer color 1 over 2
            # when starting a new component.
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
```
