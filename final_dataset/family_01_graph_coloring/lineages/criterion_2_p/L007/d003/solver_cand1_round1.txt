```python
import sys

# The handshake stability predicate:
# A tower with value f_i transmits at times t = f_i, 2f_i, ...
# In the window 0 < t <= 2:
# If f_i = 1, it transmits at t=1 and t=2 (2 pulses).
# If f_i = 2, it transmits at t=2 (1 pulse).
# The total pulses for a pair (u, v) must be 3.
# Possible combinations:
# (f_u=1, f_v=2) -> 2 + 1 = 3 (Stable)
# (f_u=2, f_v=1) -> 1 + 2 = 3 (Stable)
# (f_u=1, f_v=1) -> 2 + 2 = 4 (Unstable)
# (f_u=2, f_v=2) -> 1 + 1 = 2 (Unstable)
# Thus, for every edge (u, v), f_u must be different from f_v.
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # To get the lexicographically smallest assignment, we prefer 1 over 2.
    # We iterate 1 to n and try to assign 1 if possible.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
