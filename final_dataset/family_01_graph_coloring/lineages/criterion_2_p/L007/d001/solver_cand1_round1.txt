```python
import sys

# The problem states that for any critical link (u, v), the configuration values 
# must sum to 3. Since values are 1 or 2, this means f(u) + f(v) = 3.
# This implies f(u) != f(v).
# This is a standard bipartite graph coloring problem where we want to color
# the graph with 1s and 2s such that no two adjacent nodes have the same color.
# To get the lexicographically smallest assignment, we iterate from 1 to n,
# and if a node is not colored, we assign it 1 (the smaller value) and perform
# a BFS/DFS to color its connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer color 1 for the 
            # first node of each component.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_c
                        stack.append((v, next_c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
