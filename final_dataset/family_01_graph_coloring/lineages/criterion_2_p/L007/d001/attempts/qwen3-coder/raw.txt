import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    adj = [[] for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    color = [0] * (n + 1)  # 0: unvisited, 1: color 1, 2: color 2
    
    # Check if bipartite using BFS for each component
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        queue = deque([start])
        color[start] = 1
        
        while queue:
            u = queue.popleft()
            current_color = color[u]
            neighbor_color = 3 - current_color  # 1 -> 2, 2 -> 1
            
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = neighbor_color
                    queue.append(v)
                elif color[v] != neighbor_color:
                    print("NO")
                    return
    
    # If we reach here, a valid assignment exists
    print("YES")
    result = []
    for i in range(1, n + 1):
        if color[i] == 1:
            result.append("1")
        else:
            result.append("2")
    print(" ".join(result))

if __name__ == "__main__":
    main()
