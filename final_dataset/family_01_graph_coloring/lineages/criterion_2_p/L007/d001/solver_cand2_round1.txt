```python
import sys

# The problem asks for an assignment of 1 or 2 to each tower such that
# for every coupled pair (u, v), the product of their multipliers is 2.
# Since the multipliers are {1, 2}, the product is 2 if and only if
# one tower is 1 and the other is 2.
# This is equivalent to a 2-coloring problem on a graph where edges
# represent the constraint that the two nodes must have different colors.
# We want the lexicographically smallest assignment, which means we prefer
# 1 over 2 for each node whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # colors: 0 = unassigned, 1 = multiplier 1, 2 = multiplier 2
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 first
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
