```python
import sys

# The state transitions:
# f=1: t=0:Sync, t=1:Ping, t=2:Ack, t=3:Sync, ...
# f=2: t=0:Sync, t=1:Ack, t=2:Ping, t=3:Sync, ...
#
# Conflict conditions for link (u, v):
# Never both Ping:
# If f_u=1, f_v=1: Ping at t=1, 4, 7... (Conflict)
# If f_u=2, f_v=2: Ping at t=2, 5, 8... (Conflict)
# If f_u=1, f_v=2: Ping at t=1, 4, 7... (u) and t=2, 5, 8... (v). No conflict.
#
# Never both Ack:
# If f_u=1, f_v=1: Ack at t=2, 5, 8... (Conflict)
# If f_u=2, f_v=2: Ack at t=1, 4, 7... (Conflict)
# If f_u=1, f_v=2: Ack at t=2, 5, 8... (u) and t=1, 4, 7... (v). No conflict.
#
# Conclusion: For any edge (u, v), f_u must not equal f_v.
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
