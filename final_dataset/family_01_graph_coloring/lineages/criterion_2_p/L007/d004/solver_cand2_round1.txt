import sys

# The problem states:
# Configuration 1: [x, y, z] -> [y, z, x] (Forward shift)
# Configuration 2: [x, y, z] -> [z, x, y] (Backward shift)
#
# Let f(1, [x, y, z]) = [y, z, x]
# Let f(2, [x, y, z]) = [z, x, y]
#
# A link between u and v is stable if:
# f(f_u, f(f_v, [x, y, z])) = [x, y, z]
#
# If f_u = 1, f_v = 1: f(1, f(1, [x, y, z])) = f(1, [y, z, x]) = [z, x, y] != [x, y, z]
# If f_u = 2, f_v = 2: f(2, f(2, [x, y, z])) = f(2, [z, x, y]) = [y, z, x] != [x, y, z]
# If f_u = 1, f_v = 2: f(1, f(2, [x, y, z])) = f(1, [z, x, y]) = [x, y, z] == [x, y, z]
# If f_u = 2, f_v = 1: f(2, f(1, [x, y, z])) = f(2, [y, z, x]) = [x, y, z] == [x, y, z]
#
# Thus, for every link (u, v), f_u must be different from f_v.
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
