```python
import sys

# The phase factors are c_k = i^{f_k}.
# If f_k = 1, c_k = i.
# If f_k = 2, c_k = i^2 = -1.
# For a link (u, v):
# If f_u = 1, f_v = 1: |i + i| = |2i| = 2.
# If f_u = 2, f_v = 2: |-1 - 1| = |-2| = 2.
# If f_u = 1, f_v = 2: |i - 1| = sqrt(1^2 + (-1)^2) = sqrt(2).
# If f_u = 2, f_v = 1: |-1 + i| = sqrt((-1)^2 + 1^2) = sqrt(2).
#
# Total interference = (count of same-flag links) * 2 + (count of diff-flag links) * sqrt(2).
# Let S be the number of same-flag links, D be the number of diff-flag links.
# S + D = m.
# Total interference = 2S + D * sqrt(2) <= m * sqrt(2).
# 2S + (m - S) * sqrt(2) <= m * sqrt(2)
# S(2 - sqrt(2)) <= m(sqrt(2) - 1)
# S <= m * (sqrt(2) - 1) / (2 - sqrt(2))
# Note that (sqrt(2) - 1) / (2 - sqrt(2)) = (sqrt(2) - 1) / (sqrt(2) * (sqrt(2) - 1)) = 1 / sqrt(2) = sqrt(2) / 2.
# So S <= m * sqrt(2) / 2 = m / sqrt(2).
# Since S must be an integer, S <= floor(m / sqrt(2)).
#
# This is equivalent to saying that the graph must be bipartite.
# If the graph is bipartite, we can color it with 2 colors (1 and 2) such that all links are diff-flag.
# Then S = 0, which satisfies 0 <= m / sqrt(2).
# If the graph is not bipartite, it contains an odd cycle.
# Any assignment will have at least one same-flag link (S >= 1).
# For a single odd cycle of length L, the minimum S is 1.
# The condition S <= m / sqrt(2) is very restrictive.
# Actually, for any graph, if it is bipartite, we can set all links to be diff-flag (S=0).
# If it is not bipartite, any assignment will have S >= 1.
# The condition S <= m / sqrt(2) is only possible if the graph is bipartite.
# Proof: If not bipartite, S >= 1. Is 1 <= m / sqrt(2)? m >= sqrt(2) approx 1.41.
# If m=1, S=1, 1 <= 1/sqrt(2) is false.
# If m=2, S=1, 1 <= 2/sqrt(2) = sqrt(2) is true.
# However, the problem asks for a valid assignment. The only way to satisfy the condition
# for all m is to have S=0, i.e., the graph must be bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
