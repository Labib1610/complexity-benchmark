```python
import sys

# The problem states:
# Primary protocol of u must match the inactive protocol of v.
# If f_i is the primary protocol of tower i, then the inactive protocol is 3 - f_i.
# The condition is: f_u = 3 - f_v, which simplifies to f_u + f_v = 3.
# This means f_u != f_v.
# This is a bipartite graph coloring problem.
# We need to color the graph with 2 colors (1 and 2) such that no two adjacent
# nodes have the same color.
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If a node is not colored, we assign it 1 (the smallest possible) and perform BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try color 1 first
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
