import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The problem is equivalent to 2-coloring a graph.
    # We want the lexicographically smallest assignment (p1, p2, ..., pn).
    # This means we prefer p_i = 1 whenever possible.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a BFS/DFS to color the connected component.
            # To get the lexicographically smallest, we try to assign 1 to the 
            # smallest index node in the component.
            stack = [i]
            color[i] = 1
            component = []
            
            # Collect component
            q = [i]
            component.append(i)
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = -1 # Temporary mark
                        q.append(v)
                        component.append(v)
            
            # Now color the component. We want the smallest index in the component
            # to have color 1.
            # Actually, for any connected component, there are only two ways to color it.
            # We check both and pick the one that is lexicographically smaller.
            
            def get_coloring(start_node, start_color):
                c = {start_node: start_color}
                stack = [start_node]
                while stack:
                    u = stack.pop()
                    for v in adj[u]:
                        if v not in c:
                            c[v] = 3 - c[u]
                            stack.append(v)
                        elif c[v] == c[u]:
                            return None
                return c

            c1 = get_coloring(i, 1)
            c2 = get_coloring(i, 2)
            
            if c1 is None and c2 is None:
                print("NO")
                return
            
            if c1 is None:
                best_c = c2
            elif c2 is None:
                best_c = c1
            else:
                # Compare lexicographically
                # We only need to compare the values at the smallest index
                # But actually, we need to compare the whole sequence.
                # Since we want the smallest overall sequence, we compare
                # the assignments for all nodes in the component.
                is_c1_smaller = False
                for node in sorted(component):
                    if c1[node] < c2[node]:
                        is_c1_smaller = True
                        break
                    elif c1[node] > c2[node]:
                        is_c1_smaller = False
                        break
                best_c = c1 if is_c1_smaller else c2
            
            for node in component:
                color[node] = best_c[node]
                
    print("YES")
    print(*(color[1:]))

solve()
