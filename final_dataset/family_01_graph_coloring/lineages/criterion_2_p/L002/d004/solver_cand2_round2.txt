```python
import sys

# The condition is det([[s_u, 1], [1, s_v]]) = 1
# s_u * s_v - 1 * 1 = 1
# s_u * s_v = 2
# Since s_u, s_v are in {1, 2}, the only way s_u * s_v = 2 is:
# (s_u = 1 and s_v = 2) OR (s_u = 2 and s_v = 1)
# This means s_u != s_v.
# This is a standard 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If a node is not visited, we assign it 1 (the smallest possible value)
# and perform BFS/DFS to color its connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign color 1 to i
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
