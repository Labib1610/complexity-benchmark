```python
import sys

# The condition: x^2 - (t_u + t_v)x + 2 = 0 has integer solutions.
# The discriminant D = (t_u + t_v)^2 - 8 must be a perfect square.
# Possible tiers are 1 or 2.
# If t_u=1, t_v=1: D = (1+1)^2 - 8 = 4 - 8 = -4 (No)
# If t_u=1, t_v=2: D = (1+2)^2 - 8 = 9 - 8 = 1 (Yes, sqrt(1)=1)
# If t_u=2, t_v=2: D = (2+2)^2 - 8 = 16 - 8 = 8 (No)
# Thus, for every obligation (u, v), t_u must be different from t_v.
# This is a bipartite graph coloring problem.
# We need the lexicographically smallest assignment.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If not colored, try color 1 first, then 2.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
