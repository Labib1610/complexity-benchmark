```python
import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# color(u) != color(v). This is equivalent to checking if the graph is bipartite.
# If it is bipartite, we want the lexicographically smallest sequence of colors.
# Since we have only two colors (1 and 2), lexicographically smallest means
# we prefer 1 over 2 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If an account is not colored, we assign it 1 (the smallest possible color).
    # Then we perform BFS/DFS to color its connected component.
    for i in range(1, n + 1):
        if colors[i] == 0:
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs if using DFS, 
    # but here we use iterative stack-based approach.
    solve()
```
