```python
import sys

# The transformations:
# f1(A)=B, f1(B)=A, f1(C)=C
# f2(A)=A, f2(B)=C, f2(C)=B
#
# Check f_v(f_u(C)) != C:
# If u=1, v=1: f1(f1(C)) = f1(C) = C. (Invalid)
# If u=1, v=2: f2(f1(C)) = f2(C) = B. (Valid)
# If u=2, v=1: f1(f2(C)) = f1(B) = A. (Valid)
# If u=2, v=2: f2(f2(C)) = f2(B) = C. (Invalid)
#
# So, u and v must have different tiers. This is a bipartite graph problem.
# We need to check if the graph is bipartite.
# To get the lexicographically smallest assignment, we iterate 1 to n.
# If an account is not visited, start BFS/DFS and assign the smallest possible
# tier (1) to the first node in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    result = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                result[u] = c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(result[1:]))

solve()
```
