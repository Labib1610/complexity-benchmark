```python
import sys

# The condition is that for any pair (u, v), tier[u] XOR tier[v] = 3.
# Since tiers are 1 or 2:
# 1 XOR 2 = 3
# 1 XOR 1 = 0
# 2 XOR 2 = 0
# This means if u and v are connected, they must have different tiers.
# This is a standard bipartite graph coloring problem.
# We want the lexicographically smallest sequence of tiers (1s and 2s).
# For each connected component, we can either:
# 1. Assign the smallest possible tier (1) to the smallest index in the component.
# 2. Assign the other tier (2) to the smallest index in the component.
# To get the lexicographically smallest sequence, we iterate from 1 to N.
# If an employee is not yet assigned, we start a BFS/DFS.
# To make the sequence lexicographically smallest, we should assign 1 to the 
# smallest index in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    tiers = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if tiers[i] == 0:
            # Start BFS/DFS for this component
            # To ensure lexicographical smallest, we assign 1 to the first node
            stack = [(i, 1)]
            tiers[i] = 1
            component = []
            
            while stack:
                u, t = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if tiers[v] == 0:
                        tiers[v] = 3 - t
                        stack.append((v, 3 - t))
                    elif tiers[v] == t:
                        print("NO")
                        return
                        
    print("YES")
    print(*(tiers[1:]))

solve()
```
