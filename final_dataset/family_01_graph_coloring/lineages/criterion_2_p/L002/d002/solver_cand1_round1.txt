```python
import sys

# The condition: (u % v) + (v % u) == 1
# If u=1, v=1: (1%1) + (1%1) = 0 + 0 = 0 != 1
# If u=1, v=2: (1%2) + (2%1) = 1 + 0 = 1 (Valid)
# If u=2, v=1: (2%1) + (1%2) = 0 + 1 = 1 (Valid)
# If u=2, v=2: (2%2) + (2%2) = 0 + 0 = 0 != 1
# Thus, for every interaction (u, v), one must be 1 and the other must be 2.
# This is equivalent to a 2-coloring problem on a graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # colors: 0 = unassigned, 1 = color 1, 2 = color 2
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # To get lexicographically smallest, we prefer color 1 for the 
            # first node in each connected component.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_c
                        stack.append((v, next_c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
