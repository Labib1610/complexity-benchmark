```python
import sys

# The condition: (tier[u] ^ tier[v]) == 3
# Possible tiers are 1 (binary 01) and 2 (binary 10).
# 1 ^ 2 = 3.
# This means for every interaction (u, v), tier[u] must be different from tier[v].
# This is a standard bipartite graph coloring problem.
# We want the lexicographically smallest sequence of tiers (1s and 2s).
# To get the lexicographically smallest sequence, we prefer 1 over 2.
# For each connected component, we can pick the tier of the first node (the one with the smallest index)
# to be 1, then propagate the constraints. If that leads to a contradiction, the graph is not bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # tiers: 0 means unassigned, 1 or 2 are the assigned tiers
    tiers = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if tiers[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we assign 1 to the smallest node in the component
            stack = [(i, 1)]
            tiers[i] = 1
            
            while stack:
                u, t = stack.pop()
                
                for v in adj[u]:
                    if tiers[v] == 0:
                        tiers[v] = 3 - t # If t is 1, 3-1=2. If t is 2, 3-2=1.
                        stack.append((v, tiers[v]))
                    elif tiers[v] == t:
                        print("NO")
                        return
                        
    print("YES")
    print(*(tiers[1:]))

solve()
```
