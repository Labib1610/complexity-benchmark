import sys

# The condition is: state(u) XOR state(v) = 3.
# Possible states are 1 (binary 01) and 2 (binary 10).
# 1 XOR 2 = 3.
# This means for every edge (u, v), state(u) must be different from state(v).
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we want state(i) = 1 whenever possible.
# For each connected component, we have two choices:
# 1. Assign 1 to the smallest index node in the component.
# 2. Assign 2 to the smallest index node in the component.
# We choose the one that results in a lexicographically smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            idx_in_comp = 0
            possible = True
            
            # BFS/DFS to color component
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have two colorings for this component:
            # Option A: color[v] as is (1 or 2)
            # Option B: 3 - color[v]
            # We want the one that puts 1 at the smallest index node in the component.
            # Let 'first' be the smallest index in this component.
            # If color[first] == 1, Option A gives 1 at 'first'.
            # If color[first] == 2, Option B gives 1 at 'first'.
            
            first = min(q)
            if color[first] == 1:
                for node in q:
                    res[node] = color[node]
            else:
                for node in q:
                    res[node] = 3 - color[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
