```python
import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(1000000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # assignment: 0 = unassigned, 1 = Schedule 1, 2 = Schedule 2
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a node is unassigned, we try assigning it to 1 first.
    # This is a standard bipartite check using DFS.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS from i, assigning 1 to it
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, color = stack.pop()
                
                # Neighbors must have the opposite color
                target_color = 3 - color
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = target_color
                        stack.append((v, target_color))
                    elif assignment[v] != target_color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    solve()
```
