```python
import sys

# The resonance condition: p_u^{p_v} + p_v^{p_u} = 3
# If p_u = 1, p_v = 1: 1^1 + 1^1 = 2 != 3
# If p_u = 1, p_v = 2: 1^2 + 2^1 = 1 + 2 = 3 (Valid)
# If p_u = 2, p_v = 1: 2^1 + 1^2 = 2 + 1 = 3 (Valid)
# If p_u = 2, p_v = 2: 2^2 + 2^2 = 4 + 4 = 8 != 3
# Thus, for every link (u, v), p_u must be different from p_v.
# This is equivalent to checking if the graph is bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # To get the lexicographically smallest assignment, we want p_i = 1 whenever possible.
    # We iterate from 1 to n. If a node is unvisited, we start a BFS/DFS.
    # To make it lexicographically smallest, we assign 1 to the first node of each component.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
