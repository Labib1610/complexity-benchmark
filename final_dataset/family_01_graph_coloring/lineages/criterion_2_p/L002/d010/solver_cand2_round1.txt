```python
import sys

# The constraint is that for any two processes u and v sharing a constraint,
# p_u + p_v must be a multiple of 3.
# Since p_i is either 1 or 2, the only way p_u + p_v is a multiple of 3
# is if p_u + p_v = 3.
# This means p_u must be 1 if p_v is 2, and p_u must be 2 if p_v is 1.
# This is equivalent to saying p_u != p_v.
# Thus, the problem is equivalent to 2-coloring a graph.
# We want the lexicographically smallest sequence (p_1, p_2, ..., p_n).
# To achieve this, we greedily assign 1 to the first unvisited node in each
# connected component, then propagate the constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we prefer 1 over 2.
            stack = [(i, 1)]
            assignment[i] = 1
            while stack:
                u, color = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - color
                        stack.append((v, assignment[v]))
                    elif assignment[v] == color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
