We can model this as a bipartite graph problem.  
If the graph is not bipartite, output `NO`.  
Otherwise, we need the lexicographically smallest assignment of `1` and `2`.

We can do a BFS/DFS coloring, but to get the lexicographically smallest sequence, we should try to assign `1` whenever possible, starting from person 1.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n

def bfs(start):
    q = deque([start])
    color[start] = 1
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = 3 - color[u]  # alternate 1<->2
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

for i in range(n):
    if color[i] == -1:
        if not bfs(i):
            print("NO")
            sys.exit()

# Now we have a valid bipartite coloring, but we need lexicographically smallest.
# We can try to flip components if it makes the sequence smaller.
visited = [False] * n
for i in range(n):
    if not visited[i]:
        # collect component
        comp = []
        q = deque([i])
        visited[i] = True
        comp.append(i)
        while q:
            u = q.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    comp.append(v)
                    q.append(v)
        # Try flipping colors in this component
        # If flipping makes the smallest index in comp become 1, do it
        # Actually we want lexicographically smallest overall, so for each component,
        # check if flipping improves the first differing position.
        # Since components are independent, we can decide per component.
        # The smallest index in comp determines the decision.
        min_idx = min(comp)
        if color[min_idx] == 2:
            # flip
            for u in comp:
                color[u] = 3 - color[u]

print("YES")
print(" ".join(str(color[i]) for i in range(n)))
```
