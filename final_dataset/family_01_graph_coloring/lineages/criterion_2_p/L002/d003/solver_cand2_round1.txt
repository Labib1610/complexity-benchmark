```python
import sys

# The condition is |tier(u)^2 - tier(v)^2| = 3.
# Possible tiers are 1 and 2.
# 1^2 = 1, 2^2 = 4.
# |1 - 4| = 3.
# This means for every edge (u, v), tier(u) must be different from tier(v).
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If an account is not visited, we start a BFS/DFS.
# To make it lexicographically smallest, when we start a component,
# we assign the smallest possible value (1) to the smallest index in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
        else:
            print("NO")
            return
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for this component
            # To ensure lexicographical smallest, we want color[i] = 1
            queue = [i]
            color[i] = 1
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
