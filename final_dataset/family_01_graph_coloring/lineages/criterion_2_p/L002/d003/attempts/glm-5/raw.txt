import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
    
    tiers = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if tiers[i] == 0:
            queue = deque([i])
            tiers[i] = 1
            
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if tiers[v] == 0:
                        tiers[v] = 3 - tiers[u]
                        queue.append(v)
                    elif tiers[v] == tiers[u]:
                        print("NO")
                        return
    
    print("YES")
    print(' '.join(map(str, tiers[1:])))

if __name__ == '__main__':
    solve()
