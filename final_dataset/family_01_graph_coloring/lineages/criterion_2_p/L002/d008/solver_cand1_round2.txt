import sys

# The condition p_u * p_v = 2 means:
# If p_u = 1, then p_v = 2.
# If p_u = 2, then p_v = 1.
# This is equivalent to saying p_u != p_v.
# This is a standard bipartite graph coloring problem.
# We want the lexicographically smallest sequence (p_1, p_2, ..., p_n).
# To make it lexicographically smallest, we prefer 1 over 2.
# For each connected component, we can try setting the first node to 1.
# If that fails, we set it to 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try setting color[i] = 1 (phase 1)
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Try setting color[i] = 2 (phase 2)
                color = [0] * (n + 1) # Reset for this component
                # Actually, we need to reset only the current component
                # But since we failed, we must try the other color for the whole component
                # Re-run with color[i] = 2
                component = []
                stack = [(i, 2)]
                color[i] = 2
                component.append(i)
                
                possible = True
                idx_comp = 0
                while idx_comp < len(component):
                    u = component[idx_comp]
                    idx_comp += 1
                    for v in adj[u]:
                        if color[v] == 0:
                            color[v] = 3 - color[u]
                            component.append(v)
                        elif color[v] == color[u]:
                            print("NO")
                            return
                    if not possible:
                        break
            
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
