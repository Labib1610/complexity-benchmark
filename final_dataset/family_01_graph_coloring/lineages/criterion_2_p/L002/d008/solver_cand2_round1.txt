```python
import sys

# The condition E_u^3 + E_v^3 = 9 with E_i in {1, 2}
# 1^3 = 1, 2^3 = 8.
# 1 + 8 = 9.
# So for every constraint (u, v), one must be 1 and the other must be 2.
# This is a bipartite coloring problem.
# We want the lexicographically smallest sequence (E_1, E_2, ..., E_n).
# To make it lexicographically smallest, we prefer E_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    # color 1: state 1, color 2: state 2
    # To get lexicographically smallest, for each component, 
    # we try setting the first node to 1.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if swapping colors gives a lexicographically smaller result
            # The current assignment is color[u].
            # If we swap, color[u] becomes 3 - color[u].
            # We want the smallest value at the smallest index.
            # Find the first node in this component.
            first_node = min(u for u, c in component)
            if color[first_node] == 2:
                for u, c in component:
                    color[u] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
```
