```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    order = [[] for _ in range(n + 1)]
    deg = [0] * (n + 1)
    
    for i in range(1, n + 1):
        for j in adj[i]:
            if j > i:
                deg[i] += 1
                deg[j] += 1
        order[deg[i]].append(i)
    
    for d in range(n + 1):
        for v in order[d]:
            if color[v] != 0:
                continue
            color[v] = 1
            queue = deque([v])
            while queue:
                u = queue.popleft()
                for w in adj[u]:
                    if color[w] == 0:
                        color[w] = 3 - color[u]
                        queue.append(w)
                    elif color[w] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(' '.join(map(str, color[1:])))

solve()
```
