```python
import sys

# The condition is |t_u^3 - t_v^3| = 7.
# Possible tiers are 1 and 2.
# 1^3 = 1, 2^3 = 8.
# |1 - 1| = 0
# |1 - 8| = 7
# |8 - 1| = 7
# |8 - 8| = 0
# Thus, for any reciprocal obligation (u, v), one must be tier 1 and the other tier 2.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If an account is not colored, we assign it 1 (the smallest possible tier).
    # Then we perform BFS/DFS to color its connected component.
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```
