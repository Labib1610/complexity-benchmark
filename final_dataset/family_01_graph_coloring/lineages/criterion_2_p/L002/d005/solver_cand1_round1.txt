```python
import sys

# The condition is that the number of ways to choose t_u members from t_u + t_v 
# is odd. This is equivalent to the binomial coefficient C(t_u + t_v, t_u) being odd.
# By Lucas's Theorem, C(n, k) is odd if and only if (n & k) == k.
# Here, n = t_u + t_v and k = t_u.
# (t_u + t_v) & t_u == t_u.
# Let's test the possible values for t_u, t_v in {1, 2}:
# 1) t_u=1, t_v=1: C(2, 1) = 2 (even) -> Invalid
# 2) t_u=1, t_v=2: C(3, 1) = 3 (odd)  -> Valid
# 3) t_u=2, t_v=1: C(3, 2) = 3 (odd)  -> Valid
# 4) t_u=2, t_v=2: C(4, 2) = 6 (even) -> Invalid
#
# Thus, for any reciprocal obligation (u, v), we must have t_u != t_v.
# This is a standard 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest assignment, we greedily assign 1 
# to the first node of each connected component, then propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
