import sys

# The condition: (3 * t_u) ^ (3 * t_v) == 5
# t_u, t_v in {1, 2}
# If t_u = 1, 3*t_u = 3 (binary 011)
# If t_u = 2, 3*t_u = 6 (binary 110)
# Possible XORs:
# 3 ^ 3 = 0
# 3 ^ 6 = 5
# 6 ^ 3 = 5
# 6 ^ 6 = 0
# So the condition (3 * t_u) ^ (3 * t_v) == 5 is equivalent to t_u != t_v.
# This is a standard 2-coloring problem (bipartite graph).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    # We need to check if the graph is bipartite.
    # The protocol: for i = 1 to n, try assigning 1.
    # If valid, assign 1. Else assign 2.
    # This is equivalent to checking if a valid 2-coloring exists.
    # Since the graph must be bipartite, we can use BFS/DFS to color.
    
    color = [0] * (n + 1)
    
    # The protocol forces a specific assignment.
    # For each connected component, we have two possible colorings (swap 1 and 2).
    # The protocol picks the one that assigns 1 to the smallest index in the component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # The protocol says: try 1 for i. If valid, keep it.
            # Since we already checked bipartiteness, any valid coloring works.
            # The protocol's greedy choice is simply the one where the first node
            # encountered in the component gets 1.
            pass

    print("YES")
    print(*(color[1:]))

solve()
