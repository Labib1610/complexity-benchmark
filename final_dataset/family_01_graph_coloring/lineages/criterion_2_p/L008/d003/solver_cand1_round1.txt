```python
import sys

# The problem requires that for every edge (u, v), the XOR of the polarization 
# axes c[u] and c[v] must be 3.
# The possible values for c[i] are {1, 2}.
# 1 XOR 1 = 0
# 1 XOR 2 = 3
# 2 XOR 1 = 3
# 2 XOR 2 = 0
# Thus, for every edge (u, v), we must have c[u] != c[v].
# This means the graph must be bipartite.
# We want the lexicographically smallest sequence of {1, 2}.
# This means we want as many 1s as possible at the beginning.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If a node is not colored, we try assigning it 1.
    # This is a standard bipartite check using BFS/DFS.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
