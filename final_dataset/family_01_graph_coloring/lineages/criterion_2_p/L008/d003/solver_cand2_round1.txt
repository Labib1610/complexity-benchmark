import sys

# The condition c_u * c_v = 2 with c_i in {1, 2} implies:
# If c_u = 1, then c_v = 2.
# If c_u = 2, then c_v = 1.
# This is a bipartite coloring problem.
# We need to assign 1 or 2 such that for every edge (u, v), c_u != c_v.
# This is possible if and only if the graph is bipartite.
# To get the lexicographically smallest assignment, we want c_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we prefer color 1.
    # For each connected component, we can choose the color of the first node.
    # If we pick color 1 for the first node, the rest are determined.
    # If we pick color 2 for the first node, the rest are determined.
    # We compare the two possibilities for each component and pick the one
    # that results in a lexicographically smaller sequence.
    
    ans = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if swapping colors in this component gives a better result
            # We want the smallest value at the smallest index.
            # Since we iterate 1 to n, the first node in the component is the smallest index.
            # If color[first_node] is 1, we keep it. If it's 2, we check if swapping
            # makes the sequence lexicographically smaller.
            # Actually, for each component, we just need to ensure the first node
            # encountered gets 1 if possible.
            
            # Wait, the lexicographical requirement is global.
            # For each component, we have two choices:
            # Choice A: color[i] = 1, others follow.
            # Choice B: color[i] = 2, others follow.
            # We pick the one that makes the sequence smaller.
            
            comp_nodes = sorted(component)
            # Try Choice A
            res_a = {}
            for node in component:
                res_a[node] = color[node]
            
            # Try Choice B
            res_b = {}
            for node in component:
                res_b[node] = 3 - color[node]
            
            # Compare
            first = comp_nodes[0]
            if res_a[first] < res_b[first]:
                for node in component:
                    ans[node] = res_a[node]
            else:
                for node in component:
                    ans[node] = res_b[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
