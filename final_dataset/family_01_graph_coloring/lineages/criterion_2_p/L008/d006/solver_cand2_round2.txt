```python
import sys

# The interference metric is x^2 + 3y^2 where x, y in {1, 2}.
# Possible metrics:
# x=1, y=1: 1^2 + 3(1^2) = 4 (Not prime)
# x=1, y=2: 1^2 + 3(2^2) = 1 + 12 = 13 (Prime)
# x=2, y=1: 2^2 + 3(1^2) = 4 + 3 = 7 (Prime)
# x=2, y=2: 2^2 + 3(2^2) = 4 + 12 = 16 (Not prime)
#
# So, for a tube from u to v:
# If u=1, v=1: 4 (Invalid)
# If u=1, v=2: 13 (Valid)
# If u=2, v=1: 7 (Valid)
# If u=2, v=2: 16 (Invalid)
#
# This means:
# If u=1, v must be 2.
# If u=2, v must be 1.
# This is equivalent to saying u != v.
# If u=v, the configuration is impossible.
# If u != v, we must have c[u] != c[v].
# This is a 2-coloring problem on the graph formed by the tubes.
# Since we want the lexicographically smallest sequence, we greedily assign 1
# to the first node in each connected component, then propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color component
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
