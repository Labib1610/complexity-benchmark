```python
import sys

# The expression C^T L C evaluates to the sum of (c_u - c_v)^2 for all edges (u, v).
# Since c_i is either 1 or 2, (c_u - c_v)^2 is 0 if c_u == c_v, and 1 if c_u != c_v.
# Thus, C^T L C = m means that for every edge (u, v), c_u must be different from c_v.
# This is equivalent to checking if the graph is bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we perform BFS/DFS on each component.
# For each component, we try assigning 1 to the smallest node index.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from node i, assign color 1 to it
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
