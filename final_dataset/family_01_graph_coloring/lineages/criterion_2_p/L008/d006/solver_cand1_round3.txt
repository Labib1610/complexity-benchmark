import sys

# The condition is S_i + d_i * c_i = 3 * d_i.
# Let x_i = c_i - 1. Since c_i is 1 or 2, x_i is 0 or 1.
# S_i = sum_{j in neighbors(i)} c_j = sum_{j in neighbors(i)} (x_j + 1) = sum x_j + d_i.
# Substituting into the condition:
# (sum_{j in neighbors(i)} x_j + d_i) + d_i * (x_i + 1) = 3 * d_i
# sum_{j in neighbors(i)} x_j + d_i + d_i * x_i + d_i = 3 * d_i
# sum_{j in neighbors(i)} x_j + d_i * x_i = d_i
# This is equivalent to: sum_{j in neighbors(i)} x_j = d_i * (1 - x_i)
# If x_i = 1, then sum_{j in neighbors(i)} x_j = 0.
# If x_i = 0, then sum_{j in neighbors(i)} x_j = d_i.
# This means if x_i = 1, all neighbors must have x_j = 0.
# If x_i = 0, all neighbors must have x_j = 1.
# This is exactly the condition for a bipartite graph.
# If the graph is bipartite, we can color it with 0 and 1.
# To get the lexicographically smallest sequence c_i (where c_i = x_i + 1),
# we want x_i to be 0 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            # If there is a self-loop, d_i includes 2 connections to i.
            # The condition sum_{j in neighbors} x_j = d_i * (1 - x_i)
            # becomes (sum_{j != i} x_j) + 2 * x_i = d_i * (1 - x_i)
            # If x_i = 1, sum + 2 = 0 (impossible since x_j >= 0)
            # If x_i = 0, sum = d_i (impossible since sum includes x_i=0)
            # Actually, a self-loop forces a contradiction.
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            # To make the sequence lexicographically smallest, we prefer c_i = 1 (x_i = 0).
            # We try setting the first node of each component to 0.
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[i] + 1 for i in range(1, n + 1)))

solve()
