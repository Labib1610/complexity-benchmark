```python
import sys

# The footprint of a test between microservices u and v:
# If tier(u) == tier(v), footprint = 1000.
# If tier(u) != tier(v), footprint = 2000.
# Total footprint = sum_{edges (u,v)} footprint(u,v)
# Let E1 be the number of edges where tier(u) == tier(v)
# Let E2 be the number of edges where tier(u) != tier(v)
# Total footprint = 1000 * E1 + 2000 * E2
# We are given E1 + E2 = m
# We want 1000 * E1 + 2000 * E2 = 2000 * m
# 1000 * E1 + 2000 * (m - E1) = 2000 * m
# 1000 * E1 + 2000 * m - 2000 * E1 = 2000 * m
# -1000 * E1 = 0 => E1 = 0
# This means there must be no edges between microservices of the same tier.
# This is equivalent to checking if the graph is bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
