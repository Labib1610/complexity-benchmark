import sys

# The problem asks for an assignment of c_i in {1, 2} such that there exists
# an orientation of edges where the in-degree of node i is d_i * (c_i - 1).
# If c_i = 1, in-degree = 0. If c_i = 2, in-degree = d_i.
# Let x_i = c_i - 1, so x_i in {0, 1}.
# We need in-degree(i) = d_i * x_i.
# For a connected component, let E be the set of edges and V be the set of vertices.
# Sum of in-degrees = sum_{i in V} d_i * x_i = |E|.
# Also, sum of in-degrees = |E|.
# For a component, if we orient edges, the sum of in-degrees is the number of edges.
# A graph has an orientation with in-degrees d_i * x_i if and only if
# for every subset S of vertices, the number of edges with both endpoints in S
# is at most sum_{i in S} d_i * x_i.
# Specifically, for the whole component, sum_{i in V} d_i * x_i = |E|.
# If we set x_i = 0 for all i, sum = 0, which only works if |E| = 0.
# If we set x_i = 1 for all i, sum = sum d_i = 2|E|, which only works if |E| = 0.
# Actually, the condition for an orientation to exist with given in-degrees d_i'
# is sum_{i in V} d_i' = |E| and for any subset S, sum_{i in S} d_i' >= |E(S)|.
# Here d_i' = d_i * x_i.
# For a connected component, if we pick x_i = 1 for some nodes and 0 for others,
# the condition sum_{i in V} d_i * x_i = |E| must hold.
# This is equivalent to sum_{i in V, x_i=1} d_i = |E|.
# Since sum_{i in V} d_i = 2|E|, this means sum_{i in V, x_i=1} d_i = 1/2 sum_{i in V} d_i.
# This is only possible if the component is bipartite and we pick one partition.
# If the component is not bipartite, no such assignment exists.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    deg = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1
        deg[v] += 1
        idx += 2
    
    color = [-1] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            sum0 = sum(deg[u] for u, c in component if c == 0)
            sum1 = sum(deg[u] for u, c in component if c == 1)
            
            # We need sum_{u in V, x_u=1} d_u = |E_comp|
            # sum_{u in V} d_u = 2 * |E_comp|
            # So we need sum_{u in V, x_u=1} d_u = 1/2 * sum_{u in V} d_u
            # This means sum_{u in V, x_u=1} d_u = sum_{u in V, x_u=0} d_u
            
            # Try setting x_u = 1 for c=0 nodes or c=1 nodes
            # Lexicographically smallest: prefer c_i=1 (x_i=0)
            # We want to minimize the sequence c_1, c_2, ...
            
            possible = []
            if sum0 == sum1:
                # Two valid assignments for this component
                # Option 1: x_u = 1 if c=0, x_u = 0 if c=1
                # Option 2: x_u = 0 if c=0, x_u = 1 if c=1
                # c_u = x_u + 1
                opt1 = [(u, 2 if c == 0 else 1) for u, c in component]
                opt2 = [(u, 1 if c == 0 else 2) for u, c in component]
                opt1.sort(); opt2.sort()
                
                # Compare opt1 and opt2
                if [x[1] for x in opt1] < [x[1] for x in opt2]:
                    for u, val in opt1: ans[u] = val
                else:
                    for u, val in opt2: ans[u] = val
            else:
                print("NO")
                return
                
    print("YES")
    print(*(ans[1:]))

solve()
