```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)          # 0 = unvisited, 1 or 2 = final answer
    side = [0] * (n + 1)           # temporary bipartite side (0/1)

    for start in range(1, n + 1):
        if color[start]:
            continue
        # BFS for this component
        q = deque([start])
        side[start] = 0
        comp_nodes = [start]
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if side[v] == 0 and color[v] == 0 and v not in comp_nodes:
                    # not visited in this component yet
                    side[v] = side[u] ^ 1
                    q.append(v)
                    comp_nodes.append(v)
                elif side[v] == 0 and color[v] == 0 and v in comp_nodes:
                    # already in component but side not set yet (should not happen)
                    pass
                else:
                    if side[v] == side[u]:
                        ok = False
                        break
            if not ok:
                break
        if not ok:
            print("NO")
            return

        # Determine which side gets label 1
        min_node = min(comp_nodes)
        # side of min_node
        if side[min_node] == 0:
            map0, map1 = 1, 2
        else:
            map0, map1 = 2, 1

        for v in comp_nodes:
            color[v] = map0 if side[v] == 0 else map1

    # all components processed
    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
