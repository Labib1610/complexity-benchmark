```python
import sys

# The integral is int_{c_u}^{c_v} (3x^2 - 6x) dx = [x^3 - 3x^2]_{c_u}^{c_v}
# Let f(x) = x^3 - 3x^2.
# f(1) = 1 - 3 = -2
# f(2) = 8 - 12 = -4
# The condition is |f(c_v) - f(c_u)| = 2.
# If c_u = 1, c_v = 2: |f(2) - f(1)| = |-4 - (-2)| = |-2| = 2.
# If c_u = 2, c_v = 1: |f(1) - f(2)| = |-2 - (-4)| = |2| = 2.
# If c_u = c_v: |f(c_u) - f(c_u)| = 0 != 2.
# Thus, for every tube (u, v), we must have c_u != c_v.
# This means the graph must be bipartite.
# We need the lexicographically smallest assignment.
# This means we prefer 1 over 2 for each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
