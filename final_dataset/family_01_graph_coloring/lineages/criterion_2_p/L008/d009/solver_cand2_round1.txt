import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    degree = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
        idx += 2
        
    # Let a_i be the number of anode endpoints at lens i.
    # Then c_i = d_i - a_i is the number of cathode endpoints.
    # S_i = 3 * a_i + 8 * c_i = 3 * a_i + 8 * (d_i - a_i) = 8 * d_i - 5 * a_i
    # Q_i = 3^2 * a_i + 8^2 * c_i = 9 * a_i + 64 * (d_i - a_i) = 64 * d_i - 55 * a_i
    # Equation: 10 * S_i - Q_i = d_i * (26 - 5 * c_i)
    # 10 * (8 * d_i - 5 * a_i) - (64 * d_i - 55 * a_i) = d_i * (26 - 5 * c_i)
    # 80 * d_i - 50 * a_i - 64 * d_i + 55 * a_i = 16 * d_i + 5 * a_i
    # 16 * d_i + 5 * a_i = d_i * (26 - 5 * c_i)
    # 16 * d_i + 5 * a_i = 26 * d_i - 5 * d_i * c_i
    # 5 * a_i + 5 * d_i * c_i = 10 * d_i
    # a_i + d_i * c_i = 2 * d_i
    # a_i = d_i * (2 - c_i)
    
    # Since 0 <= a_i <= d_i:
    # If c_i = 1: a_i = d_i * (2 - 1) = d_i. All endpoints must be anodes.
    # If c_i = 2: a_i = d_i * (2 - 2) = 0. All endpoints must be cathodes.
    
    # For a tube (u, v), one end is anode, one is cathode.
    # If u is anode, v is cathode.
    # If c_u = 1, u must be anode. If c_u = 2, u must be cathode.
    # If c_u = 1 and c_v = 1, both must be anodes (impossible for a tube).
    # If c_u = 2 and c_v = 2, both must be cathodes (impossible for a tube).
    # Thus, c_u != c_v for every edge (u, v).
    # This is a bipartite graph check.
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
