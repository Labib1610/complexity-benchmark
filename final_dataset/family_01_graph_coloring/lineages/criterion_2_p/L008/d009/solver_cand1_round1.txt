import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    d = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        d[u] += 1
        d[v] += 1
        idx += 2
        
    # Condition 1: S_i^3 + K_i^3 = d_i^3.
    # Since S_i + K_i = d_i, this implies S_i * K_i = 0.
    # So for each lens, either S_i = 0 or K_i = 0.
    # This means all tubes connected to lens i must be either all sources or all sinks.
    # This is equivalent to orienting the graph such that each node is either a source or a sink.
    # This is only possible if the graph is bipartite.
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    # Condition 2: S_i - d_i(2 - c_i) = k * (d_i + 1)
    # If S_i = d_i (all sources), then d_i - d_i(2 - c_i) = d_i(c_i - 1) = k(d_i + 1)
    # If S_i = 0 (all sinks), then 0 - d_i(2 - c_i) = d_i(c_i - 2) = k(d_i + 1)
    
    # For a bipartite graph, we can partition into sets A and B.
    # If we set all nodes in A as sources (S_i = d_i) and B as sinks (S_i = 0):
    # For u in A: d_u(c_u - 1) = k(d_u + 1) => c_u - 1 must be a multiple of (d_u + 1)/gcd(d_u, d_u+1) = d_u+1.
    # Since c_u is 1 or 2, c_u - 1 is 0 or 1. Only 0 works. So c_u = 1.
    # For v in B: d_v(c_v - 2) = k(d_v + 1) => c_v - 2 must be a multiple of (d_v + 1).
    # Since c_v is 1 or 2, c_v - 2 is -1 or 0. Only 0 works. So c_v = 2.
    
    # This works if we can orient the graph such that all nodes in A are sources and B are sinks.
    # This is always possible in a bipartite graph by directing all edges from A to B.
    
    res = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 1:
            res[i] = 1
        else:
            res[i] = 2
            
    print("YES")
    print(*(res[1:]))

solve()
