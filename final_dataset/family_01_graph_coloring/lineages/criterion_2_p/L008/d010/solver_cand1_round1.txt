import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Local stability index: 3*cu + 3*cv - 2*cu*cv
    # If cu=1, cv=1: 3+3-2 = 4
    # If cu=1, cv=2: 3+6-4 = 5
    # If cu=2, cv=1: 6+3-4 = 5
    # If cu=2, cv=2: 6+6-8 = 4
    # We want sum of indices = 5*m.
    # Let x be number of edges with (1,2) or (2,1).
    # Let y be number of edges with (1,1) or (2,2).
    # x + y = m
    # 5x + 4y = 5m
    # 5x + 4(m-x) = 5m => x = m.
    # This means every edge must connect a 1 and a 2.
    # This is exactly the definition of a bipartite graph.
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # To get lexicographically smallest, we want 1s as much as possible.
    # The graph might have multiple connected components.
    # For each component, we have two choices: (1, 2, 1, 2...) or (2, 1, 2, 1...).
    # We pick the one that makes the sequence lexicographically smallest.
    
    res = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Check which coloring is better for this component
            # color[u] is 1 or 2.
            # Option 1: keep color[u] as is.
            # Option 2: flip color[u] (1->2, 2->1).
            
            # Find the smallest index in the component
            min_node = min(component)
            if color[min_node] == 1:
                for node in component:
                    res[node] = color[node]
            else:
                for node in component:
                    res[node] = 3 - color[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
