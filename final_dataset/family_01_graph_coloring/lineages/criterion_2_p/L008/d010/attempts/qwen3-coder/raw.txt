```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # The local stability index is 3c_u + 3c_v - 2c_u c_v
    # Let's compute this for all combinations:
    # c_u=1, c_v=1: 3+3-2 = 4
    # c_u=1, c_v=2: 3+6-4 = 5
    # c_u=2, c_v=1: 6+3-4 = 5
    # c_u=2, c_v=2: 6+6-8 = 4
    # So we want edges to contribute 5 each, meaning c_u != c_v for all edges.
    # This is equivalent to checking if the graph is bipartite.
    
    # Check if the graph is bipartite
    color = [-1] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            queue = deque()
            queue.append(i)
            color[i] = 0
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    # If bipartite, assign colors. We want lexicographically smallest.
    # color 0 -> 1, color 1 -> 2
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:
            result.append("1")
        else:
            result.append("2")
    
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
