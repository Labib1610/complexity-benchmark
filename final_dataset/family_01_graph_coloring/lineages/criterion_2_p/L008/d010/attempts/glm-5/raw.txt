import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # We need to find assignment c_i in {1, 2} such that
    # sum over edges (u,v) of (3c_u + 3c_v - 2c_u c_v) = 5m
    # Let x_u = c_u - 1, so x_u in {0, 1}
    # c_u = 1 + x_u
    # Term = 3(1+x_u) + 3(1+x_v) - 2(1+x_u)(1+x_v)
    #       = 3 + 3x_u + 3 + 3x_v - 2(1 + x_u + x_v + x_u x_v)
    #       = 6 + 3x_u + 3x_v - 2 - 2x_u - 2x_v - 2x_u x_v
    #       = 4 + x_u + x_v - 2x_u x_v
    # Sum = 4m + sum(x_u + x_v) - 2 sum(x_u x_v)
    # Let S = sum of degrees of vertices with x=1 = sum(d_u * x_u)
    # Sum = 4m + S - 2 * E_11, where E_11 is edges with both ends x=1
    # We need Sum = 5m, so 4m + S - 2E_11 = 5m => S - 2E_11 = m
    # Let E_10 = edges with one end x=1, other x=0
    # S = 2E_11 + E_10 (each E_11 edge contributes 2 to S, each E_10 contributes 1)
    # So 2E_11 + E_10 - 2E_11 = m => E_10 = m
    # This means ALL edges must be between a vertex with x=1 and x=0.
    # So the graph must be bipartite, and we need to choose one part to have x=1.
    # To get lexicographically smallest c (prefer c=1, i.e., x=0),
    # we should choose the smaller part to have x=1.
    
    color = [-1] * (n + 1)
    components = []
    
    for start in range(1, n + 1):
        if color[start] == -1:
            q = deque([start])
            color[start] = 0
            comp = [[], []]
            comp[0].append(start)
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        comp[color[v]].append(v)
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            components.append(comp)
    
    # For each component, decide which part gets x=1
    # To minimize c sequence (prefer 1s), we want to minimize number of 2s
    # So we want to minimize number of x=1
    # Choose smaller part to have x=1
    # If equal size, choose part containing smaller indexed vertex to have x=0
    # (so smaller indices get x=0, i.e., c=1)
    
    result = [0] * (n + 1)
    
    for comp in components:
        part0 = comp[0]
        part1 = comp[1]
        if len(part0) < len(part1):
            # part0 is smaller, so part0 gets x=1 (c=2)
            for v in part0:
                result[v] = 2
            for v in part1:
                result[v] = 1
        elif len(part1) < len(part0):
            # part1 is smaller, so part1 gets x=1 (c=2)
            for v in part0:
                result[v] = 1
            for v in part1:
                result[v] = 2
        else:
            # Equal size, choose part with smaller min vertex to have x=0 (c=1)
            min0 = min(part0)
            min1 = min(part1)
            if min0 < min1:
                # part0 has smaller min, so part0 gets x=0 (c=1)
                for v in part0:
                    result[v] = 1
                for v in part1:
                    result[v] = 2
            else:
                for v in part0:
                    result[v] = 2
                for v in part1:
                    result[v] = 1
    
    print("YES")
    print(' '.join(map(str, result[1:])))

solve()
