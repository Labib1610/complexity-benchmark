```python
import sys

# The problem asks to assign each worker to shift 1 or 2 such that for every pair (u, v),
# the workers are in different shifts. This is equivalent to 2-coloring a graph.
# If the graph is bipartite, a solution exists.
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If a worker is not yet assigned, we assign them to shift 1 and perform BFS/DFS.
# To ensure lexicographical smallest, when we have a choice, we prefer shift 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we always try to assign 1 first
            stack = [i]
            assignment[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
