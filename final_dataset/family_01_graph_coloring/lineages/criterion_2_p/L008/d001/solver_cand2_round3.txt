```python
import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem is equivalent to 2-coloring a graph.
    # Each workstation (u, v) implies shift(u) != shift(v).
    # This is a bipartite graph check.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get the lexicographically smallest assignment,
            # we prefer shift 1 for the smallest index in the component.
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, color = stack.pop()
                
                # Sort neighbors to process them in a deterministic order
                # though for bipartite coloring, the assignment is fixed
                # once the first node is colored.
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - color
                        stack.append((v, 3 - color))
                    elif assignment[v] == color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
