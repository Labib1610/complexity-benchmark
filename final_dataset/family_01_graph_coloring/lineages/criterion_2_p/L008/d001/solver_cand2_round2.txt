import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # We need to 2-color the graph.
    # To get the lexicographically smallest assignment, we want worker 1 to be 1,
    # then worker 2 to be 1 if possible, etc.
    # This is equivalent to finding a 2-coloring where we greedily assign 1 to 
    # the smallest available node in each connected component.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we prefer color 1 for the 
            # smallest node in the component.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Standard iterative DFS to color the component
            q = [i]
            component.append(i)
            while q:
                u = q.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if swapping colors 1 and 2 results in a smaller lexicographical order
            # We only care about the smallest index in this component.
            # If the smallest index i has color 2, we swap all 1s to 2s and 2s to 1s.
            if color[i] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
