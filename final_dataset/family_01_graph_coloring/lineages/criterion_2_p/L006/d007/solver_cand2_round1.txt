import sys

# The condition <psi_u, psi_v> = 4 where psi_i = (s_i, 3-s_i)
# s_i is in {1, 2}.
# If s_i = 1, psi_i = (1, 2). If s_i = 2, psi_i = (2, 1).
# <psi_u, psi_v> = s_u * s_v + (3 - s_u) * (3 - s_v)
# If s_u = 1, s_v = 1: 1*1 + 2*2 = 5
# If s_u = 1, s_v = 2: 1*2 + 2*1 = 4
# If s_u = 2, s_v = 1: 2*1 + 1*2 = 4
# If s_u = 2, s_v = 2: 2*2 + 1*1 = 5
# The condition <psi_u, psi_v> = 4 is equivalent to s_u != s_v.
# This is a bipartite graph coloring problem.
# We need to color the graph with 2 colors (1 and 2).
# To minimize E(S) = sum s_i * 3^(n-i), we want s_1 to be 1, then s_2 to be 1 if possible, etc.
# Since it's a bipartite graph, for each connected component, there are at most 2 ways to color it.
# Let the two colors be A and B. We can assign color A to 1 and B to 2, or vice versa.
# For each component, we pick the assignment that makes the first node (smallest index) 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Find the smallest index in the component
            first_node = min(component)
            # We want res[first_node] to be 1.
            # If color[first_node] is 1, then nodes with color 1 get 1, color 2 get 2.
            # If color[first_node] is 2, then nodes with color 2 get 1, color 1 get 2.
            target = 1
            mapping = {color[first_node]: 1, 3 - color[first_node]: 2}
            
            for node in component:
                res[node] = mapping[color[node]]
                
    print("YES")
    print(*(res[1:]))

solve()
