import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Let S = sum(s_i).
    # Condition 1: If we set s_u = 1 and s_v = 1, new sum < S.
    # (S - s_u - s_v + 1 + 1) < S  =>  2 < s_u + s_v
    # Condition 2: If we set s_u = 2 and s_v = 2, new sum > S.
    # (S - s_u - s_v + 2 + 2) > S  =>  4 > s_u + s_v
    
    # Both conditions imply: 2 < s_u + s_v < 4.
    # Since s_i in {1, 2}, s_u + s_v must be 3.
    # This means one of {s_u, s_v} must be 1 and the other must be 2.
    # This is a bipartite coloring problem.
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    # To get lexicographically smallest, we want s_i = 1 whenever possible.
    # We try to color components. For each component, we have two choices:
    # 1. Start with color 1, then neighbors are 2, etc.
    # 2. Start with color 2, then neighbors are 1, etc.
    # We pick the one that makes the sequence lexicographically smaller.
    
    res = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Try setting the first node of the component to 1
            # If color[i] == 1, then nodes with color 1 get value 1, color 2 get value 2
            # If color[i] == 2, then nodes with color 2 get value 1, color 1 get value 2
            # We want the smallest value at the smallest index i.
            # Since i is the smallest index in this component, we want res[i] = 1.
            # So we map color[i] to 1.
            for node in component:
                if color[node] == color[i]:
                    res[node] = 1
                else:
                    res[node] = 2
                    
    print("YES")
    print(*(res[1:]))

solve()
