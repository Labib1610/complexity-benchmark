import sys

# The problem states that for each reaction (u, v), c_u * c_v = 2.
# Since c_i is in {1, 2}, the only way c_u * c_v = 2 is if one is 1 and the other is 2.
# This is equivalent to a 2-coloring problem on a graph where each edge (u, v)
# implies c_u != c_v.
# If u == v, c_u * c_u = 2 is impossible (1*1=1, 2*2=4), so output NO.
# Otherwise, the graph must be bipartite.
# To get the lexicographically smallest sequence, we want c_i = 1 whenever possible.
# For each connected component, we have two choices for coloring (if bipartite).
# We pick the coloring that makes the smallest index in the component 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_stack = 0
            while idx_stack < len(component):
                u = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if we should flip colors to make the smallest index 1
            # Find the smallest index in this component
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
