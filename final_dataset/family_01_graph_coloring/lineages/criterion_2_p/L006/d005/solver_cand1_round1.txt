```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # P_j(x) = integral from 0 to x of (t - s_u)(t - s_v) dt
    # P_j(x) = integral from 0 to x of (t^2 - (s_u + s_v)t + s_u*s_v) dt
    # P_j(x) = (1/3)x^3 - (1/2)(s_u + s_v)x^2 + (s_u*s_v)x
    # P_j'(x) = x^2 - (s_u + s_v)x + s_u*s_v
    # P_j''(x) = 2x - (s_u + s_v)
    # Inflection point at x = 1.5 means P_j''(1.5) = 0
    # 2(1.5) - (s_u + s_v) = 0 => 3 - (s_u + s_v) = 0 => s_u + s_v = 3
    # Since s_i in {1, 2}, s_u + s_v = 3 implies one is 1 and the other is 2.
    # This is equivalent to s_u != s_v.
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Lexicographically smallest: try setting s_i = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
