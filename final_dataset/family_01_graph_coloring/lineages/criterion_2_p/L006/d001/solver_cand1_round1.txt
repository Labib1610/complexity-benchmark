```python
import sys

# The problem asks for a 2-coloring of a graph where each edge (u, v) 
# requires that u and v have different states (1 and 2).
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest arrangement, we want to assign 
# state 1 to as many early indices as possible.
# For each connected component, we can pick the state of the first node 
# (the one with the smallest index) to be 1, and then propagate the 
# constraints. If that leads to a contradiction, the graph is not bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    state = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is unvisited, it's the start of a new component.
    # We assign it state 1 to keep the sequence lexicographically smallest.
    for i in range(1, n + 1):
        if state[i] == 0:
            stack = [(i, 1)]
            state[i] = 1
            while stack:
                u, s = stack.pop()
                next_s = 2 if s == 1 else 1
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = next_s
                        stack.append((v, next_s))
                    elif state[v] == s:
                        print("NO")
                        return
    
    print("YES")
    print(*(state[1:]))

if __name__ == '__main__':
    solve()
```
