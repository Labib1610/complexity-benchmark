import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If an employee is not assigned, we try assigning them to 1 first.
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [i]
            assignment[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process nodes in a way that respects lexicographical order
            comp_nodes = []
            q = [i]
            visited = {i}
            while q:
                u = q.pop()
                comp_nodes.append(u)
                for v in adj[u]:
                    if v not in visited:
                        visited.add(v)
                        q.append(v)
            
            # For the component, we need to check if it's bipartite.
            # Since we want the lexicographically smallest, we try assigning 
            # the smallest node in the component to 1.
            # Actually, for each component, we have two choices:
            # 1. Smallest node = 1
            # 2. Smallest node = 2
            # We pick the one that results in a lexicographically smaller sequence.
            
            def get_assignment(start_node, start_val):
                local_assign = {start_node: start_val}
                stack = [start_node]
                while stack:
                    u = stack.pop()
                    for v in adj[u]:
                        if v not in local_assign:
                            local_assign[v] = 3 - local_assign[u]
                            stack.append(v)
                        elif local_assign[v] == local_assign[u]:
                            return None
                return local_assign

            res1 = get_assignment(i, 1)
            res2 = get_assignment(i, 2)
            
            if res1 is None and res2 is None:
                print("NO")
                return
            
            if res1 is None:
                best = res2
            elif res2 is None:
                best = res1
            else:
                # Compare lexicographically
                # We only care about the nodes in this component
                nodes = sorted(comp_nodes)
                for node in nodes:
                    if res1[node] < res2[node]:
                        best = res1
                        break
                    elif res1[node] > res2[node]:
                        best = res2
                        break
                else:
                    best = res1
            
            for node, val in best.items():
                assignment[node] = val
                
    print("YES")
    print(*(assignment[1:]))

solve()
