import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unassigned, 1 = shift 1, 2 = shift 2
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n
    # If node is unassigned, try assigning 1 first (greedy)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [i]
            color[i] = 1
            component = []
            
            # Use a stack to perform DFS
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
