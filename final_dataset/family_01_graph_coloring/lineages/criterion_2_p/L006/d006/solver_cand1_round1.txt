import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    d = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        d[u] += 1
        d[v] += 1
        
    # Equilibrium condition: F_i = C_i
    # F_i = sum_{j in interactions of i} s_j^3
    # C_i = d_i * (9 - s_i^3)
    # Let x_i = s_i^3. Since s_i in {1, 2}, x_i in {1, 8}.
    # F_i = sum_{j in neighbors} x_j = d_i * (9 - x_i)
    # sum_{j in neighbors} x_j + d_i * x_i = 9 * d_i
    
    # Let y_i = x_i - 1. Since x_i in {1, 8}, y_i in {0, 7}.
    # sum_{j in neighbors} (y_j + 1) + d_i * (y_i + 1) = 9 * d_i
    # sum_{j in neighbors} y_j + d_i + d_i * y_i + d_i = 9 * d_i
    # sum_{j in neighbors} y_j + d_i * y_i = 7 * d_i
    
    # Since y_i is either 0 or 7, let z_i = y_i / 7. z_i in {0, 1}.
    # sum_{j in neighbors} (7 * z_j) + d_i * (7 * z_i) = 7 * d_i
    # sum_{j in neighbors} z_j + d_i * z_i = d_i
    
    # This is a system of linear equations over GF(2) if we consider the parity,
    # but actually it's a simple constraint:
    # sum_{j in neighbors} z_j = d_i * (1 - z_i)
    
    # If d_i = 0, the equation is 0 = 0, z_i can be 0 (s_i=1).
    # If d_i > 0:
    # If z_i = 1, then sum_{j in neighbors} z_j = 0 => all neighbors z_j = 0.
    # If z_i = 0, then sum_{j in neighbors} z_j = d_i => all neighbors z_j = 1.
    
    # This is a 2-coloring problem on the graph.
    # For each connected component, pick z_i for one node, the rest are determined.
    
    z = [-1] * (n + 1)
    for i in range(1, n + 1):
        if d[i] == 0:
            z[i] = 0
            continue
        if z[i] == -1:
            # Try z[i] = 0 first for lexicographical smallest (s_i=1)
            # If z[i] = 0, all neighbors must be 1.
            # If z[i] = 1, all neighbors must be 0.
            # This is bipartite coloring.
            comp = []
            stack = [(i, 0)]
            z[i] = 0
            possible = True
            while stack:
                u, val = stack.pop()
                comp.append(u)
                target = 1 - val
                for v in adj[u]:
                    if z[v] == -1:
                        z[v] = target
                        stack.append((v, target))
                    elif z[v] != target:
                        possible = False
                        break
                if not possible: break
            
            if not possible:
                print("NO")
                return

    # Verify
    for i in range(1, n + 1):
        if d[i] > 0:
            s = sum(z[v] for v in adj[i])
            if s != d[i] * (1 - z[i]):
                print("NO")
                return
                
    res = []
    for i in range(1, n + 1):
        res.append(1 if z[i] == 0 else 2)
    
    print("YES")
    print(*(res))

solve()
