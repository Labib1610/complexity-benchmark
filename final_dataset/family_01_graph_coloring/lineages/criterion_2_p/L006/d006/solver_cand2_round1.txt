import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # M_u * M_u + M_u * M_u = 2 * M_u^2
            # M_1^2 = 169*I, M_2^2 = 169*I. Never zero.
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    # M1 = [[5, 12], [12, -5]], M2 = [[12, -5], [-5, -12]]
    # M1^2 = 169*I, M2^2 = 169*I
    # M1*M2 + M2*M1 = [[0, -119], [-119, 0]]
    # Let x_i = 0 if s_i=1, x_i = 1 if s_i=2.
    # If s_i=s_j, C = 2*M1^2 = 338*I (if s=1) or 338*I (if s=2).
    # If s_i != s_j, C = M1*M2 + M2*M1 = [[0, -119], [-119, 0]].
    # Sum C = sum_{edges} C_j = 0.
    # The diagonal elements of C_j are 338 if s_i=s_j, 0 if s_i!=s_j.
    # Sum of diagonal elements = 338 * (number of edges with s_i == s_j) = 0.
    # This implies number of edges with s_i == s_j must be 0.
    # This means the graph must be bipartite.
    
    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # For each component, we have two choices:
            # Choice 1: color 0 -> s=1, color 1 -> s=2
            # Choice 2: color 0 -> s=2, color 1 -> s=1
            # We want lexicographically smallest, so we check if we can set
            # the smallest index in the component to 1.
            min_node = min(node for node, c in component)
            if color[min_node] == 0:
                for node, c in component:
                    res[node] = c + 1
            else:
                for node, c in component:
                    res[node] = (1 - c) + 1
                    
    print("YES")
    print(*(res[1:]))

solve()
