```python
import sys

# The problem states that for every monitor (u, v), the sum of states s_u + s_v must be 3.
# Since s_i is either 1 or 2, s_u + s_v = 3 implies that s_u != s_v.
# This is a bipartite coloring problem.
# We need to color the graph with 2 colors (1 and 2) such that no two adjacent nodes have the same color.
# To get the lexicographically smallest sequence, we greedily assign 1 to the smallest available node
# in each connected component, then propagate the constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    state = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer state 1 for the smallest node
            stack = [(i, 1)]
            state[i] = 1
            while stack:
                u, s = stack.pop()
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = 3 - s
                        stack.append((v, state[v]))
                    elif state[v] == s:
                        print("NO")
                        return
    
    print("YES")
    print(*(state[1:]))

solve()
```
