import sys

# The problem states the total interaction amplitude is the product of (c_u^3 + c_v^3) for all m entanglements.
# We need this product to be 9^m.
# Since c_i is either 1 or 2, c_i^3 is either 1 or 8.
# The possible values for c_u^3 + c_v^3 are:
# 1+1 = 2
# 1+8 = 9
# 8+8 = 16
# For the product to be 9^m, every single entanglement (u, v) must satisfy c_u^3 + c_v^3 = 9.
# This means for every edge (u, v), one particle must have energy 1 (c^3=1) and the other must have energy 2 (c^3=8).
# This is equivalent to saying the graph must be bipartite, and we need to 2-color it.
# To get the lexicographically smallest assignment, we want c_i = 1 whenever possible.
# We can use BFS/DFS to 2-color each connected component.
# For each component, we have two choices of coloring (swapping 1s and 2s).
# We choose the one that makes the first node in the component 1 if possible, 
# but we must be careful: lexicographical order means we want c_1=1, then c_2=1, etc.
# Actually, for each connected component, we have two colorings: (A, B) or (B, A).
# We pick the one that results in a smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # Standard bipartite check
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # We have two colorings: color[v] or 3-color[v].
            # We want the lexicographically smallest.
            # Find the smallest index in this component.
            first_node = min(component)
            # If we set color[first_node] = 1, is it better than setting it to 2?
            # We compare the two assignments.
            # Actually, just pick the one that makes the smallest index in the component 1.
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]

    print("YES")
    print(*(color[1:]))

solve()
