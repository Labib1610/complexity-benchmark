```python
import sys

# The condition: the sphere with diameter (c_u, 0, 0) and (c_v, 0, 0)
# contains the point (1.5, 0.4, 0.3).
# The sphere equation: (x - c_u)(x - c_v) + (y - 0)(y - 0) + (z - 0)(z - 0) = 0
# Substituting (1.5, 0.4, 0.3):
# (1.5 - c_u)(1.5 - c_v) + 0.4^2 + 0.3^2 = 0
# (1.5 - c_u)(1.5 - c_v) + 0.16 + 0.09 = 0
# (1.5 - c_u)(1.5 - c_v) + 0.25 = 0
# 2.25 - 1.5(c_u + c_v) + c_u * c_v + 0.25 = 0
# c_u * c_v - 1.5(c_u + c_v) + 2.5 = 0
# Multiply by 4:
# 4 * c_u * c_v - 6(c_u + c_v) + 10 = 0
# (2 * c_u - 3)(2 * c_v - 3) + 1 = 0
# (2 * c_u - 3)(2 * c_v - 3) = -1
# Since c_u, c_v are integers, (2 * c_u - 3) and (2 * c_v - 3) must be integers.
# The only integer factors of -1 are 1 and -1.
# Case 1: 2 * c_u - 3 = 1  => 2 * c_u = 4 => c_u = 2
#         2 * c_v - 3 = -1 => 2 * c_v = 2 => c_v = 1
# Case 2: 2 * c_u - 3 = -1 => c_u = 1
#         2 * c_v - 3 = 1  => c_v = 2
# Thus, for every constraint (u, v), one must be 1 and the other must be 2.
# This is a bipartite coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    # Lexicographically smallest: 1 is smaller than 2.
    # color 1 -> 1, color 2 -> 2
    res = []
    for i in range(1, n + 1):
        res.append(str(color[i]))
    print(" ".join(res))

solve()
```
