```python
import sys

# The condition (s_u & s_v) == 0 for s_i in {1, 2} means:
# 1 & 1 = 1 != 0
# 2 & 2 = 2 != 0
# 1 & 2 = 0
# 2 & 1 = 0
# This is equivalent to saying that if there is an edge between u and v,
# s_u must be different from s_v.
# This is exactly the definition of a bipartite graph.
# We need to color the graph with 2 colors (1 and 2).
# To get the lexicographically smallest sequence, we want s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is not colored, we start a BFS/DFS.
    # To make it lexicographically smallest, we prefer color 1.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
