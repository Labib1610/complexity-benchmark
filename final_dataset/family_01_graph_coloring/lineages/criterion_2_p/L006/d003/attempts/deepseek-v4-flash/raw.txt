import sys
sys.setrecursionlimit(1_000_000)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    ok = True

    # BFS to 2-color the graph
    from collections import deque
    for start in range(n):
        if color[start] != -1:
            continue
        color[start] = 0
        q = deque([start])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                elif color[v] == color[u]:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            break

    if not ok:
        print("NO")
        return

    # Now we have a bipartition. We need to assign states 1 or 2.
    # Constraint: for each edge (u,v), one state must be twice the other.
    # Since states are only 1 and2, the only valid pair is (1,2) or (2,1).
    # So every edge must connect a 1 and a 2.
    # Therefore the bipartition already determines the assignment up to swapping the two parts.
    # Lexicographically smallest: assign the part containing vertex 0 to state 1,
    # the other part to state 2. If vertex 0 is isolated, we can choose its part as 1.
    # But we must check: if the graph is disconnected, each component can be flipped independently.

    # We need lexicographically smallest overall sequence.

    # For each connected component, we have two possible assignments:
    # either the color 0 vertices get 1 and color 1 vertices get 2,
    # or vice versa.

    # We process components in order of their smallest vertex index.

    comp_id = [-1] * n
    comps = []  # list of lists of vertices in each component
    comp_count = 0
    for i in range(n):
        if comp_id[i] == -1:
            # BFS to collect component
            cid = comp_count

            comp_id[i] = cid
            q = deque([i])
            verts = []
            while q:
                u = q.popleft()
                verts.append(u)
                for v in adj[u]:
                    if comp_id[v] == -1:
                        comp_id[v] = cid
                        q.append(v)
            comps.append(verts)

    # For each component, decide orientation.

    # We want lexicographically smallest final array.

    # For a component, the two possible assignments are:
    # Option A: color 0 -> 1, color 1 -> 2
    # Option B: color 0 -> 2, color 1 -> 1
    # We choose the option that makes the smallest vertex in the component have state 1,
    # because lexicographic order prioritizes earlier indices.

    # If the smallest vertex in the component has color 0, then Option A gives it 1,
    # Option B gives it 2. So choose A.

    # If the smallest vertex has color 1, then Option A gives it 2,
    # Option B gives it 1. So choose B.

    ans = [0] * n
    for verts in comps:
        # find smallest index in component
        min_vertex = min(verts)
        base_color = color[min_vertex]
        # We want min_vertex to have state 1.

        # If base_color == 0, then color 0 -> 1, color 1 -> 2
        # If base_color ==  ​1, then color 1 -> 1, color 0 -> 2
