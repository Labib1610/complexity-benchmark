```python
import sys

# The problem states that for each monitor (u, v), one valve must be 1 and the other 2.
# This is because the tiers are {1, 2}, and the only way one is twice the other is 1 and 2.
# This is equivalent to a 2-coloring problem on a graph where each edge (u, v)
# implies that u and v must have different colors (1 and 2).
# We need the lexicographically smallest configuration, so we prefer 1 over 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 to the current node
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
