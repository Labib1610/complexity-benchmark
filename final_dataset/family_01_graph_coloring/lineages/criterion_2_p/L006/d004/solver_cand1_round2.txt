import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # s_u + s_u = 3 => 2*s_u = 3, impossible
            print("NO")
            return
        adj[u].append((v, i))
        adj[v].append((u, i))
        
    # Equation for row j: s_{u_j} + s_{v_j} = 3
    # This means s_{u_j} != s_{v_j}.
    # This is a 2-coloring problem.
    # s_i can be 1 or 2.
    # If s_u = 1, then s_v = 2. If s_u = 2, then s_v = 1.
    
    color = [0] * (n + 1)
    # To get lexicographically smallest, we want s_i = 1 whenever possible.
    # We iterate 1 to n. If not colored, try setting s_i = 1.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v, _ in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if this component is consistent with the constraints
            # The constraints are already satisfied by the bipartite coloring.
            # We already set s_i = 1 for the first node of each component.
            # This is the lexicographically smallest choice.
            
    print("YES")
    print(*(color[1:]))

solve()
