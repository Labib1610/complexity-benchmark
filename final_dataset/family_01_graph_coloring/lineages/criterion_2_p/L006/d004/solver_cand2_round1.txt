```python
import sys

# The condition for the matrix M = [[s_u, 1], [2, s_v]] to be singular is:
# det(M) = s_u * s_v - 2 = 0
# Since s_i \in {1, 2}, the only possible values for s_u * s_v are 1, 2, 4.
# The condition s_u * s_v = 2 is satisfied if and only if one state is 1 and the other is 2.
# This means for every monitor (u, v), s_u != s_v.
# This is a bipartite graph coloring problem.
# We need to assign s_i \in {1, 2} such that for every edge (u, v), s_u != s_v.
# To minimize sum(s_i * 3^(n-i)), we want s_i to be 1 whenever possible, starting from i=1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    states = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if states[i] == 0:
            # To minimize the sum, we greedily set the first node of a component to 1.
            stack = [(i, 1)]
            states[i] = 1
            while stack:
                u, color = stack.pop()
                for v in adj[u]:
                    if states[v] == 0:
                        states[v] = 3 - color
                        stack.append((v, states[v]))
                    elif states[v] == color:
                        print("NO")
                        return
    
    print("YES")
    print(*(states[1:]))

solve()
```
