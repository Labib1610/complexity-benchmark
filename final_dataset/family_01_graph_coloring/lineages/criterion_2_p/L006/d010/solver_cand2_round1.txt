```python
import sys

# The transformation matrices are:
# T1 = [[2, 1], [-3, -2]]
# T2 = [[7, 12], [-4, -7]]
#
# Let's check the condition: T_v(T_u(X)) = X has no non-zero fixed points.
# This is equivalent to saying that the matrix M = T_v * T_u does not have 1 as an eigenvalue.
# det(M - I) != 0.
#
# T1^2 = [[2, 1], [-3, -2]] * [[2, 1], [-3, -2]] = [[4-3, 2-2], [-6+6, -3+4]] = [[1, 0], [0, 1]] = I
# T2^2 = [[7, 12], [-4, -7]] * [[7, 12], [-4, -7]] = [[49-48, 84-84], [-28+28, -48+49]] = [[1, 0], [0, 1]] = I
#
# So if u == v, T_u(T_u(X)) = X, which always has fixed points.
# Thus, if any connection has u == v, it's impossible.
#
# What about T2(T1(X))?
# T2*T1 = [[7, 12], [-4, -7]] * [[2, 1], [-3, -2]] = [[14-36, 7-24], [-8+21, -4+14]] = [[-22, -17], [13, 10]]
# det(T2*T1 - I) = det([[-23, -17], [13, 9]]) = (-23 * 9) - (-17 * 13) = -207 + 221 = 14 != 0.
#
# What about T1(T2(X))?
# T1*T2 = [[2, 1], [-3, -2]] * [[7, 12], [-4, -7]] = [[14-4, 24-7], [-21+8, -36+14]] = [[10, 17], [-13, -22]]
# det(T1*T2 - I) = det([[9, 17], [-13, -23]]) = (9 * -23) - (17 * -13) = -207 + 221 = 14 != 0.
#
# So, the condition "no non-zero fixed points" is satisfied if and only if u != v and the states are different (s_u != s_v).
# If s_u == s_v, then T_u(T_u(X)) = X, which is bad.
# So we need s_u != s_v for all edges (u, v).
# This is exactly the condition for a graph to be bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
