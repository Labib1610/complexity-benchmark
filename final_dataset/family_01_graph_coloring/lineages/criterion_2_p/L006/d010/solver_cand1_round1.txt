```python
import sys

# The condition: Re(c_u * c_v) > 0
# c_1 = 3 + 4i, c_2 = 3 - 4i
# c_1 * c_1 = (3+4i)^2 = 9 - 16 + 24i = -7 + 24i (Re = -7)
# c_2 * c_2 = (3-4i)^2 = 9 - 16 - 24i = -7 - 24i (Re = -7)
# c_1 * c_2 = (3+4i)(3-4i) = 9 + 16 = 25 (Re = 25)
#
# Stability requires:
# If u == v: Re(c_u^2) = -7 > 0 (Impossible)
# If u != v:
#   If s_u == s_v: Re(c_u * c_v) = -7 > 0 (Impossible)
#   If s_u != s_v: Re(c_u * c_v) = 25 > 0 (Valid)
#
# Thus, for every connection (u, v), we must have u != v and s_u != s_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence s_1, ..., s_n.
# This means we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we try to assign 1 to nodes in order 1..n
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
