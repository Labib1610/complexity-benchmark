import sys
sys.setrecursionlimit(1 << 25)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        if u != v:
            adj[v].append(u)

    # s[k] = 0: unassigned, 1: type 1, 2: type 2
    s = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if s[i] == 0:
            # Start BFS/DFS from this node, assign type 1
            s[i] = 1
            queue = [i]
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                for v in adj[u]:
                    if s[v] == 0:
                        # Constraint: Re(c_u * c_v) > 0
                        # If s[u]=1 (3+4i), neighbor must be 1.
                        # If s[u]=2 (3-4i), neighbor must be 2.
                        s[v] = s[u]
                        queue.append(v)
                    elif s[v] != s[u]:
                        # Conflict detected
                        print("NO")
                        return
    
    print("YES")
    print(*(s[1:]))

if __name__ == '__main__':
    solve()
