import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            print("NO")
            return
        edges.append((u, v))

    # permutations as arrays 0:A,1:B,2:C,3:D
    p1 = [1,0,3,2]   # swap A<->B, C<->D
    p2 = [2,3,0,1]   # swap A<->C, B<->D

    # precompute good matrix
    good = [[False]*2 for _ in range(2)]
    perms = [p1, p2]
    for a in range(2):
        for b in range(2):
            ok = True
            pa = perms[a]
            pb = perms[b]
            for x in range(4):
                y = pb[pa[x]]
                if y == x:
                    ok = False
                    break
            good[a][b] = ok

    # 2-SAT: variable i true -> s_i = 1, false -> s_i = 2
    N = n * 2
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def var(i, val):  # val True for 1, False for 2
        return i*2 + (0 if val else 1)

    def add_imp(u, v):
        g[u].append(v)
        rg[v].append(u)

    # clause (X_u != a) OR (X_v != b)
    # which is (X_u == a) => (X_v != b)
    # and (X_v == b) => (X_u != a)
    for u, v in edges:
        for a in (0,1):
            for b in (0,1):
                if not good[a][b]:
                    # a,b are 0 for s=1, 1 for s=2
                    # map to bool: a_bool = (a==0)
                    a_bool = (a == 0)
                    b_bool = (b == 0)
                    add_imp(var(u, a_bool), var(v, not b_bool))
                    add_imp(var(v, b_bool), var(u, not a_bool))

    # Kosaraju
    visited = [False]*N
    order = []
    def dfs(v):
        visited[v]=True
        for to in g[v]:
            if not visited[to]:
                dfs(to)
        order.append(v)
    for i in range(N):
        if not visited[i]:
            dfs(i)
    comp = [-1]*N
    def rdfs(v, cid):
        comp[v]=cid
        for to in rg[v]:
            if comp[to]==-1:
                rdfs(to,cid)
    cid=0
    for v in reversed(order):
        if comp[v]==-1:
            rdfs(v,cid)
            cid+=1

    assignment = [False]*n
    for i in range(n):
        if comp[var(i,True)] == comp[var(i,False)]:
            print("NO")
            return
        assignment[i] = comp[var(i,True)] > comp[var(i,False)]

    # produce lexicographically smallest by flipping if possible
    # try to set earlier variables to 1 when both assignments are possible
    # Build implication graph of SCC condensation to check reachability
    # For simplicity, we rely on the assignment obtained (standard 2-SAT gives one solution)
    # Convert bool to 1/2
    res = [1 if assignment[i] else 2 for i in range(n)]
    print("YES")
    print(' '.join(map(str,res)))

if __name__ == "__main__":
    solve()
