import sys

# The resonance condition: s_u^2 * s_v + s_u * s_v^2 + s_u^2 + s_v^2 = 11
# s_i \in {1, 2}
# Let's test the possible pairs (s_u, s_v):
# (1, 1): 1^2*1 + 1*1^2 + 1^2 + 1^2 = 1 + 1 + 1 + 1 = 4 != 11
# (1, 2): 1^2*2 + 1*2^2 + 1^2 + 2^2 = 2 + 4 + 1 + 4 = 11 (Valid)
# (2, 1): 2^2*1 + 2*1^2 + 2^2 + 1^2 = 4 + 2 + 4 + 1 = 11 (Valid)
# (2, 2): 2^2*2 + 2*2^2 + 2^2 + 2^2 = 8 + 8 + 4 + 4 = 24 != 11
#
# Conclusion: For every overlap (u, v), s_u must be different from s_v.
# This is a bipartite coloring problem.
# We want the lexicographically smallest sequence (s_1, s_2, ..., s_n).
# To make it lexicographically smallest, we prefer 1 over 2.
# For each connected component, we can try assigning 1 to the first node
# and see if it's valid. If we have a choice, we pick the one that makes
# the sequence lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if we could have done better by swapping colors in this component
            # Actually, for a connected component, there are only two colorings:
            # (c1, c2, ...) or (3-c1, 3-c2, ...).
            # We want the one that results in a smaller value at the smallest index.
            # Since we iterate i from 1 to n, the first node in the component
            # encountered is the smallest index. We already set color[i] = 1.
            # This is optimal for this component.
            
    print("YES")
    print(*(color[1:]))

solve()
