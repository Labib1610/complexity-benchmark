```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Check the resonance condition: s_u^2 s_v + s_u s_v^2 + s_u^2 + s_v^2 = 11
    # With s_u, s_v ∈ {1, 2}, let's evaluate all combinations:
    # (1,1): 1*1 + 1*1 + 1 + 1 = 4
    # (1,2): 1*2 + 1*4 + 1 + 4 = 2 + 4 + 1 + 4 = 11 ✓
    # (2,1): 4*1 + 2*1 + 4 + 1 = 4 + 2 + 4 + 1 = 11 ✓
    # (2,2): 4*2 + 2*4 + 4 + 4 = 8 + 8 + 4 + 4 = 24
    
    # So the constraint is satisfied iff (s_u=1 and s_v=2) OR (s_u=2 and s_v=1)
    # This means the graph must be bipartite and each edge must connect a node with value 1 to a node with value 2
    
    # If m == 0, we can assign all nodes to 1 (lexicographically smallest)
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Check if the graph is bipartite
    color = [-1] * n  # -1: unvisited, 0: color A, 1: color B
    queue = deque()
    
    # Process each connected component
    for start in range(n):
        if color[start] == -1:
            queue.append(start)
            color[start] = 0
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # The graph is bipartite. Now we need to determine which color gets 1 and which gets 2.
    # We want the lexicographically smallest assignment.
    # That means we try to assign 1 to as many early indices as possible.
    
    # In a bipartite graph, we have two sets U and V
    # To get lexicographically smallest, we need to decide which set gets labeled 1
    # We'll try both assignments and pick the lexicographically smaller one
    
    # Color 0 -> value 1, Color 1 -> value 2
    assignment1 = [0] * n
    # Color 0 -> value 2, Color 1 -> value 1
    assignment2 = [0] * n
    
    for i in range(n):
        if color[i] == -1:  # Isolated vertex
            assignment1[i] = 1
            assignment2[i] = 1
        elif color[i] == 0:
            assignment1[i] = 1
            assignment2[i] = 2
        else:
            assignment1[i] = 2
            assignment2[i] = 1
    
    # Choose the lexicographically smaller assignment
    result = assignment1
    for i in range(n):
        if assignment1[i] < assignment2[i]:
            result = assignment1
            break
        elif assignment2[i] < assignment1[i]:
            result = assignment2
            break
    
    # Verify the solution
    for u, v in edges:
        s_u, s_v = result[u], result[v]
        if not (s_u**2 * s_v + s_u * s_v**2 + s_u**2 + s_v**2 == 11):
            print("NO")
            return
    
    print("YES")
    print(" ".join(map(str, result)))

solve()
```
