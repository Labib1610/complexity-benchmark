import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    g = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        g[u].append(v)
        g[v].append(u)

    color = [-1] * n          # 0 or 1
    from collections import deque

    for start in range(n):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        comp_vertices = [start]
        while q:
            v = q.popleft()
            for to in g[v]:
                if color[to] == -1:
                    color[to] = color[v] ^ 1
                    q.append(to)
                    comp_vertices.append(to)
                elif color[to] == color[v]:
                    print("NO")
                    return
        # ensure smallest index vertex gets color 0 (mapped to 1)
        min_idx = min(comp_vertices)
        if color[min_idx] == 1:
            # flip whole component
            for v in comp_vertices:
                color[v] ^= 1

    # map 0->1, 1->2
    ans = ['1' if c == 0 else '2' for c in color]
    print("YES")
    print(' '.join(ans))

if __name__ == "__main__":
    solve()
