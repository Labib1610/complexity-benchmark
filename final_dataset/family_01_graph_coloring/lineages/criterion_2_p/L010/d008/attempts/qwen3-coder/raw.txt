import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Check the resonance condition: s_u^3 + s_v^3 = 9
    # If s_u = 1, s_v^3 = 9 - 1 = 8, so s_v = 2
    # If s_u = 2, s_v^3 = 9 - 8 = 1, so s_v = 1
    # This means adjacent nodes must have different values - it's a bipartite graph check
    
    # Color nodes with 0 or 1 (representing modes 1 or 2)
    color = [-1] * (n + 1)
    
    # Check if graph is bipartite
    for start in range(1, n + 1):
        if color[start] == -1:
            queue = deque([start])
            color[start] = 0
            
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # If bipartite, construct the lexicographically smallest solution
    # We want mode 1 (color 0) as much as possible at the beginning
    result = []
    for i in range(1, n + 1):
        # Mode 1 corresponds to color 0, Mode 2 corresponds to color 1
        result.append(str(color[i] + 1))
    
    print("YES")
    print(" ".join(result))

solve()
