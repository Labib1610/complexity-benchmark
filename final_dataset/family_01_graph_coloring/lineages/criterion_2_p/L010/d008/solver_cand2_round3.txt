import sys

# The condition s_u^3 + s_v^3 = 9 with s_i in {1, 2}
# 1^3 = 1, 2^3 = 8
# 1 + 8 = 9, 8 + 1 = 9
# This means if there is an edge between u and v, one must be 1 and the other 2.
# This is exactly the definition of a bipartite graph.
# We need to color the graph with 1 and 2 such that adjacent nodes have different colors.
# To get the lexicographically smallest sequence, we want s_i = 1 whenever possible.
# For each connected component, we have two choices for coloring (if bipartite).
# Let the two sets of the bipartition be A and B.
# We can either assign 1 to A and 2 to B, or 2 to A and 1 to B.
# To make the sequence lexicographically smallest, for each component,
# we look at the smallest index i in the component.
# If we assign s_i = 1, we check if that forces a valid coloring.
# Since it's a bipartite graph, once we fix one node, all others are fixed.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings:
            # 1: color[v] remains as is
            # 2: color[v] becomes 3 - color[v]
            # We want the one that makes the smallest index in the component 1.
            # Find the smallest index in this component
            min_node = min(component)
            
            # If color[min_node] is 1, we prefer this coloring.
            # If color[min_node] is 2, we prefer the flipped coloring.
            flip = (color[min_node] == 2)
            
            for node in component:
                if flip:
                    res[node] = 3 - color[node]
                else:
                    res[node] = color[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
