```python
import sys

# The condition s_u^3 + s_v^3 = 9 for s_i in {1, 2}
# 1^3 = 1, 2^3 = 8.
# 1 + 8 = 9.
# So for every edge (u, v), one must be 1 and the other must be 2.
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == 0:
            comp = []
            stack = [(i, 1)]
            color[i] = 1
            comp.append((i, 1))
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        comp.append((v, 3 - c))
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            components.append(comp)
            
    # Energy E = sum_{i=1}^n (s_i * 3^(n-i))
    # We want to minimize E.
    # For each component, we have two choices:
    # Choice 1: nodes with color 1 get s=1, nodes with color 2 get s=2
    # Choice 2: nodes with color 1 get s=2, nodes with color 2 get s=1
    # Let C1 be the set of nodes with color 1, C2 be the set of nodes with color 2.
    # E = sum_{i in C1} (s_i * 3^(n-i)) + sum_{j in C2} (s_j * 3^(n-j))
    # If we pick Choice 1: E = sum_{i in C1} (1 * 3^(n-i)) + sum_{j in C2} (2 * 3^(n-j))
    # If we pick Choice 2: E = sum_{i in C1} (2 * 3^(n-i)) + sum_{j in C2} (1 * 3^(n-j))
    # Difference = E1 - E2 = sum_{i in C1} (1-2)*3^(n-i) + sum_{j in C2} (2-1)*3^(n-j)
    # Difference = sum_{j in C2} 3^(n-j) - sum_{i in C1} 3^(n-i)
    
    res = [0] * (n + 1)
    for comp in components:
        diff = 0
        for node, c in comp:
            if c == 2:
                diff += 3**(n - node)
            else:
                diff -= 3**(n - node)
        
        if diff <= 0:
            # Choice 1 is better or equal
            for node, c in comp:
                res[node] = c
        else:
            # Choice 2 is better
            for node, c in comp:
                res[node] = 3 - c
                
    print("YES")
    print(*(res[1:]))

solve()
```
