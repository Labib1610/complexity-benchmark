```python
import sys

# The stability score E = s_u^{s_v} + s_v must be 3.
# Possible values for s_u, s_v are {1, 2}.
# If s_u = 1, s_v = 1: E = 1^1 + 1 = 2 (Invalid)
# If s_u = 1, s_v = 2: E = 1^2 + 2 = 3 (Valid)
# If s_u = 2, s_v = 1: E = 2^1 + 1 = 3 (Valid)
# If s_u = 2, s_v = 2: E = 2^2 + 2 = 6 (Invalid)
#
# So for each edge (u, v), we must have (s_u, s_v) in {(1, 2), (2, 1)}.
# This means s_u != s_v.
# This is a bipartite coloring problem.
# We want the lexicographically smallest sequence s_1, s_2, ..., s_n.
# To get the lexicographically smallest sequence, we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try setting color[i] = 1 to keep it lexicographically smallest
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
