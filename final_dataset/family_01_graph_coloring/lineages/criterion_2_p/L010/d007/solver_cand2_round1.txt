```python
import sys

# The resonance condition:
# R(u, v) = integral_0^inf e^{-3t} (e^{s_u t} + e^{s_v t}) dt
# R(u, v) = integral_0^inf (e^{-(3-s_u)t} + e^{-(3-s_v)t}) dt
# Since s_u, s_v in {1, 2}, 3-s_u and 3-s_v are in {1, 2}.
# The integral of e^{-kt} from 0 to inf is 1/k.
# R(u, v) = 1/(3-s_u) + 1/(3-s_v)
# We need R(u, v) = 3/2.
# Let x_u = 3 - s_u. Then x_u in {1, 2}.
# 1/x_u + 1/x_v = 3/2.
# If x_u = 1, 1 + 1/x_v = 3/2 => 1/x_v = 1/2 => x_v = 2.
# If x_u = 2, 1/2 + 1/x_v = 3/2 => 1/x_v = 1 => x_v = 1.
# Thus, x_u != x_v.
# This means s_u != s_v.
# The problem reduces to 2-coloring the graph.
# If the graph is bipartite, we have two possible colorings (swapping 1 and 2).
# We want to minimize W = sum s_i * 2^{n-i}.
# This is equivalent to minimizing the integer formed by s_1 s_2 ... s_n.
# We prefer s_i = 1 over s_i = 2 for smaller i.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # For each component, we have two choices:
            # Choice A: nodes with color 1 get s=1, nodes with color 2 get s=2
            # Choice B: nodes with color 1 get s=2, nodes with color 2 get s=1
            # We want to pick the choice that makes the sequence s_1...s_n lexicographically smallest.
            # Find the first node in the component.
            first_node = min(node for node, c in component)
            c_first = color[first_node]
            
            # If c_first == 1, Choice A gives s[first_node] = 1, Choice B gives s[first_node] = 2.
            # We prefer Choice A.
            # If c_first == 2, Choice A gives s[first_node] = 2, Choice B gives s[first_node] = 1.
            # We prefer Choice B.
            
            if c_first == 2:
                for node, c in component:
                    color[node] = 3 - c

    print("YES")
    print(*(color[1:]))

solve()
```
