```python
import sys

# The constraint is that for any pair (u, v) sharing a workspace,
# s_u^3 + s_v^3 = 9.
# Since s_i \in {1, 2}, s_i^3 \in {1, 8}.
# The only way to get 1 + 8 = 9 is if one arm is 1 and the other is 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest configuration, we want s_1 to be 1.
# If s_1 = 1, then all nodes in its connected component are determined.
# If a component is not connected to 1, we want the smallest node in that
# component to be 1 to keep the sequence lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we process nodes 1 to n.
    # If a node is uncolored, we start a BFS/DFS.
    # To make the sequence lexicographically smallest, we assign 1 to the
    # first node of each component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
