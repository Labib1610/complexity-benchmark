```python
import sys

# The interference index between a and b is a^b + b^a.
# Possible values for a, b in {1, 2}:
# If a=1, b=1: 1^1 + 1^1 = 2
# If a=1, b=2: 1^2 + 2^1 = 1 + 2 = 3
# If a=2, b=1: 2^1 + 1^2 = 2 + 1 = 3
# If a=2, b=2: 2^2 + 2^2 = 4 + 4 = 8
# The constraint is that for any pair (u, v) sharing a workspace,
# the interference index must be 3.
# This means one must be 1 and the other must be 2.
# This is a standard 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unvisited, 1 = multiplier 1, 2 = multiplier 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest configuration, we want 1s as much as possible.
    # We iterate from 1 to n. If unvisited, start BFS/DFS.
    # For lexicographical smallest, we prefer assigning 1 to the smallest index.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
