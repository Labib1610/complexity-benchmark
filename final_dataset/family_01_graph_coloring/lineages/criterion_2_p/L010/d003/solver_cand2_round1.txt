```python
import sys

# The equation x^2 - 3x + 2 = 0 factors to (x-1)(x-2) = 0.
# So, the possible frequency multipliers are 1 and 2.
# A 2x2 matrix [[a, b], [b, a]] is invertible if its determinant is non-zero.
# det = a*a - b*b = (a-b)(a+b).
# Since a, b are in {1, 2}, a+b is always positive.
# Thus, the matrix is invertible if and only if a != b.
# This means for every pair (u, v) that shares a workspace, s_u != s_v.
# This is equivalent to 2-coloring the graph of arms.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # colors: 0 = unassigned, 1 = multiplier 1, 2 = multiplier 2
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # BFS/DFS to color the connected component
            # To get the lexicographically smallest configuration,
            # we prefer assigning 1 whenever possible.
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
```
