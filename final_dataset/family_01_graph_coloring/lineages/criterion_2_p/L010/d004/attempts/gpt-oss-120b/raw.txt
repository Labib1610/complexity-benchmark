```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)          # constraint is symmetric: values must differ

    color = [0] * (n + 1)          # 0 = unvisited, 1 or 2
    for start in range(1, n + 1):
        if color[start]:
            continue
        # assign smallest possible value to the first vertex of this component
        color[start] = 1
        q = deque([start])
        while q:
            u = q.popleft()
            cu = color[u]
            for v in adj[u]:
                if not color[v]:
                    color[v] = 3 - cu
                    q.append(v)
                elif color[v] == cu:
                    print("NO")
                    return

    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
