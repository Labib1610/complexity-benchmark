import sys
from collections import deque

def solve():
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr]); ptr += 1
    m = int(input[ptr]); ptr += 1
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(input[ptr]) - 1; ptr += 1
        v = int(input[ptr]) - 1; ptr += 1
        adj[u].append(v)
    
    # We want lexicographically smallest assignment.
    # Try to assign 1 to each node if possible.
    # If assigning 1 to u causes conflict (neighbor v must be 1 but is assigned 2),
    # we must assign 2 to u.
    # This is a greedy propagation logic.
    
    s = [0] * n
    q = deque()
    
    for i in range(n):
        if s[i] == 0:
            # Try to assign 1
            s[i] = 1
            q.append(i)
            while q:
                u = q.popleft()
                for v in adj[u]:
                    # Constraint: s[u] * 10 + s[v] divisible by 3
                    # If s[u]=1, s[v] must be 2 (12 % 3 == 0)
                    # If s[u]=2, s[v] must be 1 (21 % 3 == 0)
                    req = 2 if s[u] == 1 else 1
                    if s[v] == 0:
                        s[v] = req
                        q.append(v)
                    elif s[v] != req:
                        print("NO")
                        return
    
    # Verify all constraints
    for u in range(n):
        for v in adj[u]:
            code = s[u] * 10 + s[v]
            if code % 3 != 0:
                print("NO")
                return
    
    print("YES")
    print(*(s[i] for i in range(n)))

if __name__ == '__main__':
    solve()
