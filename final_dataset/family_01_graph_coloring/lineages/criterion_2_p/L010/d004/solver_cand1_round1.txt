import sys

# The condition is that for every overlap (u, v), the number formed by s_u s_v must be a multiple of 3.
# Possible codes: 11 (1), 12 (3), 21 (3), 22 (4).
# Multiples of 3 are 12 and 21.
# So, if there is an overlap (u, v):
# If s_u = 1, then s_v must be 2.
# If s_u = 2, then s_v must be 1.
# This means s_v must be the opposite of s_u (s_v = 3 - s_u).
# This is a bipartite graph problem where each connected component can be colored in two ways.
# To get the lexicographically smallest sequence, we iterate from 1 to n.
# If node i is not colored, we try setting s_i = 1 and propagate the constraints.
# If that fails, we try s_i = 2 and propagate. If both fail, output NO.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Try assigning 1 to the first node of the component
            component = []
            stack = [(i, 1)]
            assignment[i] = 1
            component.append(i)
            
            possible = True
            ptr = 0
            while ptr < len(component):
                u = component[ptr]
                ptr += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Reset and try assigning 2
                for node in component:
                    assignment[node] = 0
                
                stack = [(i, 2)]
                assignment[i] = 2
                component = [i]
                possible = True
                ptr = 0
                while ptr < len(component):
                    u = component[ptr]
                    ptr += 1
                    for v in adj[u]:
                        if assignment[v] == 0:
                            assignment[v] = 3 - assignment[u]
                            component.append(v)
                        elif assignment[v] == assignment[u]:
                            possible = False
                            break
                    if not possible:
                        break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(assignment[1:]))

solve()
