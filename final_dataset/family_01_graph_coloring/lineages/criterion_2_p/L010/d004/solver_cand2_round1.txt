```python
import sys

# The diagnostic check:
# Registry starts at 29 (binary 11101).
# Protocol (u, v):
# registry = 29 ^ (s_u * 14) ^ (s_v * 14)
# We need registry == 15 (binary 01111).
# 29 ^ (s_u * 14) ^ (s_v * 14) = 15
# (s_u * 14) ^ (s_v * 14) = 29 ^ 15
# 29 = 16 + 8 + 4 + 1
# 15 = 8 + 4 + 2 + 1
# 29 ^ 15 = 16 + 2 = 18
# So, (s_u * 14) ^ (s_v * 14) = 18
# If s_i = 1: s_i * 14 = 14 (binary 01110)
# If s_i = 2: s_i * 14 = 28 (binary 11100)
# Let x_u = s_u * 14, x_v = s_v * 14
# If s_u=1, s_v=1: 14 ^ 14 = 0 != 18
# If s_u=1, s_v=2: 14 ^ 28 = 01110 ^ 11100 = 10010 = 18 (Matches!)
# If s_u=2, s_v=1: 28 ^ 14 = 18 (Matches!)
# If s_u=2, s_v=2: 28 ^ 28 = 0 != 18
# Thus, s_u != s_v.
# This is a standard 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # To get lexicographically smallest, we prefer 1 over 2.
    # We iterate 1 to n, if not colored, color with 1.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
