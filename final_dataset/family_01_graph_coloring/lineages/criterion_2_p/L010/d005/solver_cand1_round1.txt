```python
import sys

# The routing rules:
# Mode 1: A <-> B, C <-> D
# Mode 2: A <-> C, B <-> D
#
# Let the state be a vector (A, B, C, D).
# Mode 1: (0, 1, 0, 0) -> (1, 0, 0, 0)
# Mode 2: (0, 1, 0, 0) -> (0, 0, 0, 1)
#
# Let's represent the state as a permutation of {A, B, C, D}.
# Initially, pulse is in A.
# After arm u:
# Mode 1: A->B, B->A, C->D, D->C
# Mode 2: A->C, C->A, B->D, D->B
#
# After arm u then arm v:
# If s_u=1, s_v=1: A -> B -> A (Result A)
# If s_u=1, s_v=2: A -> B -> D (Result D)
# If s_u=2, s_v=1: A -> C -> D (Result D)
# If s_u=2, s_v=2: A -> C -> A (Result A)
#
# The condition is that the pulse must emerge from D.
# This happens if (s_u=1 and s_v=2) or (s_u=2 and s_v=1).
# This is equivalent to s_u != s_v.
#
# The problem reduces to:
# For each overlap (u, v), s_u != s_v.
# This is a 2-coloring problem on a graph.
# We need to check if the graph is bipartite.
# If bipartite, we want the lexicographically smallest assignment of {1, 2}.
# Lexicographically smallest means we prefer 1 over 2 for each node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color component
            # To get lexicographically smallest, we try assigning 1 to the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
