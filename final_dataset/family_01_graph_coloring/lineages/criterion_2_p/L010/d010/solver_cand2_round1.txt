import sys

def solve():
    # The condition: number of divisors d(s) = s.
    # For s=1, d(1)=1. (Valid)
    # For s=2, d(2)=2. (Valid)
    # For s=3, d(3)=2 != 3.
    # For s=4, d(4)=3 != 4.
    # For s > 2, d(s) < s.
    # So only s=1 and s=2 are possible.
    
    # The safety check: sum_{k=1}^inf (s_u^k + s_v^k) / 3^k
    # This is a geometric series: sum (s_u/3)^k + sum (s_v/3)^k
    # = (s_u/3) / (1 - s_u/3) + (s_v/3) / (1 - s_v/3)
    # = s_u / (3 - s_u) + s_v / (3 - s_v)
    
    # If s=1: 1 / (3-1) = 1/2
    # If s=2: 2 / (3-2) = 2/1 = 2
    
    # Possible values for the sum:
    # If s_u=1, s_v=1: 1/2 + 1/2 = 1 (Integer, NOT SAFE)
    # If s_u=1, s_v=2: 1/2 + 2 = 2.5 (Not integer, SAFE)
    # If s_u=2, s_v=2: 2 + 2 = 4 (Integer, NOT SAFE)
    
    # Conclusion: For every overlap (u, v), we must have s_u != s_v.
    # This is equivalent to 2-coloring the graph.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    # color 1 maps to s=1, color 2 maps to s=2
    # To get lexicographically smallest, we want 1s as much as possible.
    # The coloring is fixed per connected component.
    # For each component, we can either use (1, 2) or (2, 1).
    # We choose the one that starts with 1 if possible.
    
    res = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Check if we can make the first node in component 1
            # The component is bipartite, color[u] is 1 or 2.
            # If we set color[i] = 1, then all nodes with color[u]=1 get 1, color[u]=2 get 2.
            # If we set color[i] = 2, then all nodes with color[u]=1 get 2, color[u]=2 get 1.
            # We want the smallest sequence, so we compare the two options.
            
            # Actually, just pick the one that makes the smallest value at the smallest index.
            # Since we iterate 1..n, the first node i in the component should be 1 if possible.
            # If color[i] == 1, we keep it. If color[i] == 2, we flip.
            
            flip = (color[i] == 2)
            for node in component:
                if flip:
                    res[node] = 3 - color[node]
                else:
                    res[node] = color[node]
                    
    print(*(res[1:]))

solve()
