import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    comp = [0] * (n + 1)
    comp_id = 0
    comp_flipped = []
    
    for i in range(1, n + 1):
        if color[i] == 0:
            comp_id += 1
            color[i] = 1
            comp[i] = comp_id
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        comp[v] = comp_id
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            comp_flipped.append(False)
    
    # Greedy: process components in order of their first appearing vertex.
    # For each component, decide whether to flip so that the first vertex gets color 1.
    # Since color[i] is initially 1 for the first vertex of each component,
    # we never need to flip to make the first vertex color 1.
    # This yields the lexicographically smallest assignment.
    
    res = [0] * (n + 1)
    for i in range(1, n + 1):
        res[i] = color[i]
    
    out = ["YES"]
    out.append(" ".join(str(res[i]) for i in range(1, n + 1)))
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
