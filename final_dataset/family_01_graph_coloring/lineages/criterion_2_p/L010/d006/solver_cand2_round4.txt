import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest sequence, we want to assign 1 to 
    # the smallest possible index whenever possible.
    # We use a 2-coloring approach.
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a BFS/DFS to color the connected component
            # To ensure lexicographical smallest, we prefer assigning 1 to the 
            # smallest node in the component.
            stack = [i]
            color[i] = 1
            component = []
            
            # Collect component
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if we can flip the colors of this component to make the 
            # first node 1 (already done) or if we need to check lexicographical order.
            # Actually, for each connected component, we have two choices:
            # (color[v]) or (3 - color[v]). We pick the one that makes the 
            # smallest index in the component have the smallest possible value.
            # Since we iterate 1 to n, the first node encountered in a component 
            # is the smallest index. We already set color[i] = 1.
            
    print("YES")
    print(*(color[1:]))

solve()
