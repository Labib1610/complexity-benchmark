```python
import sys

# The system of equations is:
# s_u * x + s_v * y = 3
# s_v * x + s_u * y = 3
# The determinant of the coefficient matrix is (s_u^2 - s_v^2).
# For exactly one solution, the determinant must be non-zero.
# s_u^2 - s_v^2 != 0  =>  s_u != s_v (since s_i in {1, 2}).
# Thus, for every overlap (u, v), we must have s_u != s_v.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest configuration, we want s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unassigned, 1 = mode 1, 2 = mode 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we prefer color 1
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
