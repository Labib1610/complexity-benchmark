```python
import sys

# The problem states that for any pair (u, v) sharing workspace, 
# the product of their multipliers must be 2.
# Multipliers are in {1, 2}.
# If u * v = 2, then one must be 1 and the other must be 2.
# This is a bipartite coloring problem.
# We want the lexicographically smallest sequence (s1, s2, ..., sn).
# To make it lexicographically smallest, we prefer 1 over 2 for s1, then s2, etc.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is not colored, we try assigning 1 first.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
