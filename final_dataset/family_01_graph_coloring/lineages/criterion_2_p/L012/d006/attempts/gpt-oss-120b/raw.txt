```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n          # 0/1 parity inside component
    height = [0] * n          # final heights 1 or 2
    visited = [False] * n

    from collections import deque

    for start in range(n):
        if visited[start]:
            continue
        # BFS this component
        q = deque([start])
        visited[start] = True
        color[start] = 0
        comp_nodes = [start]
        bip = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    color[v] = color[u] ^ 1
                    q.append(v)
                    comp_nodes.append(v)
                else:
                    if color[v] == color[u]:
                        bip = False
        if not bip:
            print("NO")
            return

        # decide orientation to minimize lexicographic order
        # smallest index in component determines orientation
        min_idx = min(comp_nodes)
        # we want height[min_idx] = 1
        # if its parity is 0, map parity 0 -> 1, 1 -> 2
        # else map parity 1 -> 1, 0 -> 2
        if color[min_idx] == 0:
            for v in comp_nodes:
                height[v] = 1 if color[v] == 0 else 2
        else:
            for v in comp_nodes:
                height[v] = 1 if color[v] == 1 else 2

    print("YES")
    print(' '.join(map(str, height)))

if __name__ == "__main__":
    solve()
```
