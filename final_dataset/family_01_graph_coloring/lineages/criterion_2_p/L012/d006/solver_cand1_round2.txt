```python
import sys

# The condition: P(x) = x^2 - (s_u + s_v)x + s_u * s_v = (x - s_u)(x - s_v)
# P(3) = (3 - s_u)(3 - s_v) = 2
# Since s_u, s_v are in {1, 2}:
# If s_u = 1, (3 - 1)(3 - s_v) = 2(3 - s_v) = 2 => 3 - s_v = 1 => s_v = 2
# If s_u = 2, (3 - 2)(3 - s_v) = 1(3 - s_v) = 2 => 3 - s_v = 2 => s_v = 1
# Thus, s_u != s_v. This is a bipartite graph coloring problem.
# We need to color the graph with 1 and 2 such that adjacent nodes have different colors.
# To minimize sum_{i=1}^n s_i * 3^{n-i}, we want s_1 to be 1 if possible, then s_2, etc.
# This is equivalent to lexicographical minimization of the sequence (s_1, s_2, ..., s_n).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try to set color[i] = 1 to minimize lexicographically
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
