import sys

def solve():
    # The equation is (M + D)S = 3d.
    # Let s_i = 1 + x_i, where x_i \in {0, 1}.
    # (M + D)(1 + x) = 3d
    # (M + D)1 + (M + D)x = 3d
    # Since (M + D)1 is the vector where the i-th component is sum_{j} M_{ij} + D_{ii} = 2 * D_{ii} = 2 * d_i.
    # So, 2d + (M + D)x = 3d
    # (M + D)x = d
    # For each row i: sum_{j} M_{ij} * x_j + D_{ii} * x_i = d_i
    # sum_{j} M_{ij} * x_j + d_i * x_i = d_i
    # If x_i = 0, then sum_{j} M_{ij} * x_j = d_i.
    # If x_i = 1, then sum_{j} M_{ij} * x_j + d_i = d_i => sum_{j} M_{ij} * x_j = 0.
    # Since M_{ij} >= 0 and x_j >= 0, sum_{j} M_{ij} * x_j = 0 implies x_j = 0 for all j adjacent to i.
    # Let's rephrase:
    # If x_i = 1, all neighbors j must have x_j = 0.
    # If x_i = 0, the sum of x_j for all neighbors j (with multiplicity) must be d_i.
    # This means for every neighbor j of i, x_j must be 1.
    # So: x_i = 1 => all neighbors j have x_j = 0.
    #     x_i = 0 => all neighbors j have x_j = 1.
    # This is exactly the condition for a bipartite graph where x_i is 0 on one partition and 1 on the other.
    # If the graph is not bipartite, there is no solution.
    # If it is bipartite, we have two possible assignments (swapping 0 and 1).
    # We want to minimize sum s_i * 3^{n-i} = sum (1 + x_i) * 3^{n-i}.
    # This is equivalent to minimizing sum x_i * 3^{n-i}.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # Two possible assignments: x_i = color[i] or x_i = 1 - color[i]
    # We want to minimize sum x_i * 3^{n-i}.
    # Compare sum_{i=1}^n color[i] * 3^{n-i} and sum_{i=1}^n (1 - color[i]) * 3^{n-i}.
    val1 = 0
    val2 = 0
    for i in range(1, n + 1):
        if color[i] == 1:
            val1 += pow(3, n - i)
        else:
            val2 += pow(3, n - i)
            
    res_x = color if val1 <= val2 else [1 - c for c in color]
    print("YES")

solve()
