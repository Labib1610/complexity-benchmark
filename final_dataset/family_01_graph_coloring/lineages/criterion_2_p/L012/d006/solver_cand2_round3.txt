import sys

# The condition s_u ^ s_v = 3 for s_i in {1, 2}
# 1 ^ 2 = 3, 2 ^ 1 = 3.
# This means if there is an edge (u, v), s_u must be different from s_v.
# This is a standard bipartite graph coloring problem.
# We want to minimize sum(s_i * 3^(n-i)).
# Since 3^(n-i) is strictly decreasing with i, we want to make s_1 as small as possible,
# then s_2, and so on.
# For each connected component, we have two choices:
# Choice A: Assign colors based on a BFS/DFS starting from the smallest node in the component.
# Choice B: Swap the colors (1 becomes 2, 2 becomes 1).
# We need to pick the choice for each component that makes the sequence lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # For this component, we have two colorings:
            # Option 1: color[v] as is
            # Option 2: 3 - color[v]
            # We want to pick the one that is lexicographically smaller.
            # Find the first node in the component (smallest index)
            # and ensure its color is 1 if possible.
            
            first_node = min(component)
            # If color[first_node] is 2, we can flip all colors in this component
            # to make color[first_node] = 1.
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                ans[node] = color[node]

    print("YES")
    print(*(ans[1:]))

solve()
