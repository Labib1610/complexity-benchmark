```python
import sys

# The condition: mean of neighbors = 3 - s_i
# Let s_i be the calibration mode (1 or 2).
# Let N(i) be the set of neighbors of i, and d_i = |N(i)|.
# The condition is: (sum_{j in N(i)} s_j) / d_i = 3 - s_i
# sum_{j in N(i)} s_j = d_i * (3 - s_i)
# sum_{j in N(i)} s_j + d_i * s_i = 3 * d_i
# Let x_i = s_i - 1. Then s_i = x_i + 1, where x_i in {0, 1}.
# sum_{j in N(i)} (x_j + 1) + d_i * (x_i + 1) = 3 * d_i
# sum_{j in N(i)} x_j + d_i + d_i * x_i + d_i = 3 * d_i
# sum_{j in N(i)} x_j + d_i * x_i = d_i
# This is a system of linear equations over GF(2) if we consider the parity.
# Actually, the equation is sum_{j in N(i)} x_j + d_i * x_i = d_i.
# If d_i is even, sum_{j in N(i)} x_j = 0 (mod 2).
# If d_i is odd, sum_{j in N(i)} x_j + x_i = 1 (mod 2).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    # x_i in {0, 1}
    # For each connected component, we can solve the system.
    # If d_i is even: sum_{j in N(i)} x_j = 0
    # If d_i is odd: sum_{j in N(i)} x_j + x_i = 1
    
    ans = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i] and adj[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Try setting x_i = 0 then x_i = 1 to find lexicographically smallest
            # Since we want smallest s_i, we want smallest x_i.
            # This is a system of linear equations.
            # We can use BFS/DFS to propagate constraints.
            
            def check(start_node, start_val):
                res = {}
                res[start_node] = start_val
                q = [start_node]
                while q:
                    u = q.pop(0)
                    # Equation: sum_{j in N(u)} x_j + (d_u % 2) * x_u = d_u % 2
                    # sum_{j in N(u), j != known} x_j = (d_u % 2) - (d_u % 2) * x_u - sum_{j in N(u), j known} x_j
                    unknowns = [v for v in adj[u] if v not in res]
                    known_sum = sum(res[v] for v in adj[u] if v in res)
                    rhs = (len(adj[u]) % 2) - (len(adj[u]) % 2) * res[u] - known_sum
                    
                    if len(unknowns) == 0:
                        if rhs % 2 != 0: return None
                    elif len(unknowns) == 1:
                        res[unknowns[0]] = rhs % 2
                        q.append(unknowns[0])
                    else:
                        # Multiple unknowns, pick the smallest index and set to 0
                        target = min(unknowns)
                        res[target] = 0
                        q.append(target)
                return res

            # Try x_i = 0
            sol = check(component[0], 0)
            if sol is None:
                sol = check(component[0], 1)
            
            if sol is None:
                print("NO")
                return
            for node in component:
                ans[node] = sol[node] + 1
        elif not visited[i]:
            ans[i] = 1
            
    print("YES")
    print(*(ans[1:]))

solve()
```
