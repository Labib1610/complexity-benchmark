```python
import sys

# The condition is that for every strut (u, v), the perimeter of the bounding box
# is 2|u - v| + 2.
# The perimeter of the bounding box of (u, h_u) and (v, h_v) is 2|u - v| + 2|h_u - h_v|.
# The condition 2|u - v| + 2|h_u - h_v| = 2|u - v| + 2 simplifies to |h_u - h_v| = 1.
# Since h_i is either 1 or 2, |h_u - h_v| = 1 means h_u != h_v.
# This is a standard 2-coloring problem on a graph.
# We want to minimize the sequence h_1 h_2 ... h_n as a base-3 number.
# This means we want h_1 to be 1 if possible, then h_2 to be 1 if possible, etc.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # To minimize the sequence, we greedily set the first node of each
            # connected component to 1.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
