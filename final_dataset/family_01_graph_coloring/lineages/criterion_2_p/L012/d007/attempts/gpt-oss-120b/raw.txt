import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [ -1 ] * (n + 1)          # parity 0/1 for bipartite check
    height = [0] * (n + 1)            # final heights 1 or 2
    from collections import deque

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        comp = [start]
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                    comp.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
        # decide orientation: smallest index vertex must get height 1
        min_vertex = min(comp)
        # if we map parity 0 -> 1, parity 1 -> 2, then min_vertex gets:
        #   1 if color[min_vertex]==0 else 2
        # we want it to be 1, so if its color is 1 we flip mapping.
        flip = (color[min_vertex] == 1)
        for v in comp:
            parity = color[v] ^ flip   # effective parity after possible flip
            height[v] = 1 if parity == 0 else 2

    print("YES")
    print(' '.join(str(height[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
