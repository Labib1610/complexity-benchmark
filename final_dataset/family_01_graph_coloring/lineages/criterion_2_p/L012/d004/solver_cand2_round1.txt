```python
import sys

# The system of equations:
# s_u * x + s_v * y = s_u * s_v
# s_u^2 * x + s_v^2 * y = s_u^2 * s_v^2
# This is a system of two linear equations in x and y.
# The determinant of the coefficient matrix is:
# D = s_u * s_v^2 - s_v * s_u^2 = s_u * s_v * (s_v - s_u)
# For a unique solution, D must be non-zero.
# Since s_u, s_v are in {1, 2}, D != 0 if and only if s_u != s_v.
# Thus, the condition is simply s_u != s_v for all connected welders.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we greedily assign 1 
# to the smallest index in each connected component, then propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign 1 to i to be lexicographically smallest
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
