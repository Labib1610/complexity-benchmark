We can model this as a bipartite graph problem.  
The condition “product of modes = 2” means one welder must be 1 and the other 2.  
So the graph must be bipartite.  

We need the lexicographically smallest assignment, so we try to assign 1 whenever possible, starting from node 1.  

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n
color[0] = 1  # try to assign 1 to node 1

# BFS to check bipartiteness and assign colors
for start in range(n):
    if color[start] == -1:
        color[start] = 1
        q = deque([start])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 3 - color[u]  # 1 -> 2, 2 -> 1
                    q.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    sys.exit()

# Now we have a valid bipartition, but we need lexicographically smallest.
# We can flip all colors in a connected component if it makes the sequence smaller.
# Since we started with node 1 = 1, we check each component: if flipping gives smaller lexicographic order, flip.

visited = [False] * n
for i in range(n):
    if not visited[i]:
        # collect component
        comp = []
        q = deque([i])
        visited[i] = True
        while q:
            u = q.popleft()
            comp.append(u)
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    q.append(v)
        # check if flipping helps
        # current assignment for this component
        # if we flip, color becomes 3 - color
        flip = False
        for u in comp:
            if color[u] == 2 and (3 - color[u] == 1):
                # flipping makes this node 1
                # but we must check if it improves lexicographically
                pass
        # Actually, we can just try flipping and compare
        # Build current and flipped sequences for this component
        cur = [color[u] for u in comp]
        flipped = [3 - color[u] for u in comp]
        # We need to compare in original index order
        comp_sorted = sorted(comp)
        cur_seq = [color[u] for u in comp_sorted]
        flip_seq = [3 - color[u] for u in comp_sorted]
        if flip_seq < cur_seq:
            for u in comp:
                color[u] = 3 - color[u]
```
