import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem is equivalent to 2-coloring a graph.
    # To get the lexicographically smallest sequence, we want s_1=1, then s_2=1 if possible, etc.
    # For each connected component, we can pick the color of the smallest node.
    # If we pick color 1 for the smallest node in a component, the rest are determined.
    # If we pick color 2, the rest are also determined (swapping 1s and 2s).
    # We want to choose the color for the smallest node in each component such that
    # the resulting sequence is lexicographically smallest.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            # We need to check if the component is bipartite
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # Let the current coloring be C. The other valid coloring is 3-C.
            # We want to compare the sequence generated by C and 3-C.
            # We only care about the first node in the component (the smallest index).
            # If color[i] == 1, the sequence starts with 1. If color[i] == 2, it starts with 2.
            # Since we want the lexicographically smallest, we prefer 1.
            # If color[i] is already 1, we keep it. If color[i] is 2, we flip it.
            if color[i] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
