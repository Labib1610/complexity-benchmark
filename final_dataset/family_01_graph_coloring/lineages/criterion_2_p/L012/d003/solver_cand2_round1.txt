```python
import sys

# The condition is (t_u / t_v) + (t_v / t_u) = 2.5
# Let x = t_u / t_v. Then x + 1/x = 2.5 => x^2 - 2.5x + 1 = 0
# 2x^2 - 5x + 2 = 0 => (2x - 1)(x - 2) = 0
# So x = 2 or x = 0.5.
# This means t_u / t_v = 2 or t_u / t_v = 0.5.
# Since t_u, t_v are in {1, 2}, this implies one must be 1 and the other 2.
# This is a bipartite coloring problem.
# We want the lexicographically smallest sequence (t_1, t_2, ..., t_n).
# This means we prefer 1 over 2 for each node in order.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning 1 to the current component's root to be lexicographically smallest
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
