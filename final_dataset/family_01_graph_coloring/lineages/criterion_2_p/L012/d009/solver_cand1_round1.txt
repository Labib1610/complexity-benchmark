import sys

# The condition: (C & 12) > 0
# C = 2 * (A - I)
# A = (3 * h_u + u) * h_v
# I = u * h_v + 2 * h_v
# C = 2 * ((3 * h_u + u) * h_v - (u * h_v + 2 * h_v))
# C = 2 * (3 * h_u * h_v + u * h_v - u * h_v - 2 * h_v)
# C = 2 * (3 * h_u * h_v - 2 * h_v)
# C = 2 * h_v * (3 * h_u - 2)
#
# Check: (2 * h_v * (3 * h_u - 2)) & 12 > 0
# Let's test h_u, h_v in {1, 2}:
# 1) h_u=1, h_v=1: C = 2 * 1 * (3*1 - 2) = 2 * 1 = 2. (2 & 12) = 2 > 0 (True)
# 2) h_u=1, h_v=2: C = 2 * 2 * (3*1 - 2) = 4 * 1 = 4. (4 & 12) = 4 > 0 (True)
# 3) h_u=2, h_v=1: C = 2 * 1 * (3*2 - 2) = 2 * 4 = 8. (8 & 12) = 8 > 0 (True)
# 4) h_u=2, h_v=2: C = 2 * 2 * (3*2 - 2) = 4 * 4 = 16. (16 & 12) = 0 (False)
#
# The condition is equivalent to: NOT (h_u == 2 AND h_v == 2)
# This is a 2-SAT problem where each clause is (h_u=1 OR h_v=1).
# We want to minimize the sequence h_1 h_2 ... h_n in base 3.
# This is equivalent to setting h_i = 1 whenever possible.
# Since (h_u=1 OR h_v=1) is equivalent to (h_u=2 => h_v=1) and (h_v=2 => h_u=1),
# this is a simple implication graph.
# To minimize the sequence, we greedily try to set h_i = 1.
# If we set h_i = 2, we must set all its neighbors to 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    h = [0] * (n + 1)
    # We want to minimize h_1 h_2 ... h_n.
    # h_i = 1 is better than h_i = 2.
    # If we set h_i = 2, then all neighbors must be 1.
    # If we set h_i = 1, no restriction.
    # This is equivalent to: if any neighbor is 2, current must be 1.
    # If we have a connected component, we can pick one node to be 2,
    # then all its neighbors must be 1, and their neighbors must be 2, etc.
    # This is only possible if the component is bipartite.
    
    ans = [1] * (n + 1)
    visited = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if visited[i] == 0:
            comp = []
            stack = [(i, 1)]
            visited[i] = 1
            while stack:
                u, color = stack.pop()
                comp.append((u, color))
                for v in adj[u]:
                    if visited[v] == 0:
                        visited[v] = 3 - color
                        stack.append((v, 3 - color))
                    elif visited[v] == color:
                        print("NO")
                        return
            
            # Try setting color 1 to 1 and color 2 to 2
            # Or color 1 to 2 and color 2 to 1
            # We want to minimize the sequence.
            c1 = [u for u, c in comp if c == 1]
            c2 = [u for u, c in comp if c == 2]
            
            # Option A: c1=1, c2=2
            # Option B: c1=2, c2=1
            # Compare lexicographically
            def get_seq(c1_val, c2_val):
                res = [0] * (n + 1)
                for u in c1: res[u] = c1_val
                for u in c2: res[u] = c2_val
                return res
            
            s1 = get_seq(1, 2)
            s2 = get_seq(2, 1)
            
            best = s1
            for k in range(1, n + 1):
                if s1[k] < s2[k]:
                    best = s1
                    break
                elif s2[k] < s1[k]:
                    best = s2
                    break
            for k in range(1, n + 1):
                ans[k] = best[k]
                
    print("YES")
    print(*(ans[1:]))

solve()
