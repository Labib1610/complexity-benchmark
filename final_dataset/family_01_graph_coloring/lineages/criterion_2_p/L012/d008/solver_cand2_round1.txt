import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # det = h_u * h_u - 0 = h_u^2. 
            # Requirement: det = 2 - 0 = 2.
            # h_u^2 = 2 has no integer solution for h_u in {1, 2}.
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    # The condition: det = h_u * h_v - (u - v)^2 = 2 - (u - v)^2
    # This simplifies to: h_u * h_v = 2
    # Since h_i \in {1, 2}, the only way h_u * h_v = 2 is if one is 1 and the other is 2.
    # This is a bipartite coloring problem.
    
    color = [0] * (n + 1)
    # To minimize the base-3 number, we want h_i = 1 whenever possible.
    # We iterate 1 to n. If not colored, try color 1.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If a component is not connected to anything, color[i] remains 1.
    # This is optimal because 1 < 2.
    res = []
    for i in range(1, n + 1):
        res.append(str(color[i]))
    
    print("YES")
    print(" ".join(res))

solve()
