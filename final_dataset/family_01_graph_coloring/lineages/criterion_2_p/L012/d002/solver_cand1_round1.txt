```python
import sys

# The condition is that for every link (u, v), multiplier[u] XOR multiplier[v] = 3.
# Multipliers are 1 (binary 01) or 2 (binary 10).
# 1 XOR 2 = 3.
# This means for every link (u, v), multiplier[u] must be different from multiplier[v].
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we want to assign 1 to the smallest
# possible ID whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is not colored, we start a BFS/DFS.
    # To make it lexicographically smallest, we prefer color 1 for the first node
    # of each connected component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
