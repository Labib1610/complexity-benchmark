```python
import sys

# The constraint:
# Bucket starts at 0.
# Node u adds h_u tokens. Bucket = h_u.
# Node v performs h_v actions.
# If h_u = 1, h_v = 1: Bucket 1 -> 1 action: 1-1=0. Final 0. (Invalid)
# If h_u = 1, h_v = 2: Bucket 1 -> 1 action: 1-1=0. 2nd action: 0 empty -> add 1. Final 1. (Valid)
# If h_u = 2, h_v = 1: Bucket 2 -> 1 action: 2-1=1. Final 1. (Valid)
# If h_u = 2, h_v = 2: Bucket 2 -> 1 action: 2-1=1. 2nd action: 1-1=0. Final 0. (Invalid)
#
# Summary:
# (1, 2) is valid.
# (2, 1) is valid.
# (1, 1) is invalid.
# (2, 2) is invalid.
# This is equivalent to h_u != h_v.
# The problem is to 2-color the graph such that for every edge (u, v), h_u != h_v.
# This is possible if and only if the graph is bipartite.
# We want the lexicographically smallest sequence (h_1, ..., h_n).
# We try to set h_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we prefer color 1 for smaller indices
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

solve()
```
