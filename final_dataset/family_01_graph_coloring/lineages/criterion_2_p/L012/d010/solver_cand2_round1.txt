```python
import sys

# The condition |i^h_u + i^h_v| = sqrt(2)
# h_k is in {1, 2}.
# i^1 = i, i^2 = -1.
# Possible pairs (h_u, h_v):
# (1, 1): |i + i| = |2i| = 2 != sqrt(2)
# (1, 2): |i - 1| = sqrt(1^2 + (-1)^2) = sqrt(2) (Valid)
# (2, 1): |-1 + i| = sqrt((-1)^2 + 1^2) = sqrt(2) (Valid)
# (2, 2): |-1 - 1| = |-2| = 2 != sqrt(2)
# Thus, h_u != h_v. This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unassigned, 1 = state 1, 2 = state 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To minimize E = sum h_k * 4^(n-k), we want h_1 to be 1, then h_2 to be 1, etc.
            # Greedy approach: try assigning 1 to the first uncolored node in the component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
