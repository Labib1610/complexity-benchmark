```python
import sys

# The condition: gcd(c_u + 1, c_v + 1) == 1
# c_u, c_v are in {1, 2}
# c_u + 1, c_v + 1 are in {2, 3}
# Possible pairs (c_u, c_v):
# (1, 1) -> gcd(2, 2) = 2 (Invalid)
# (1, 2) -> gcd(2, 3) = 1 (Valid)
# (2, 1) -> gcd(3, 2) = 1 (Valid)
# (2, 2) -> gcd(3, 3) = 3 (Invalid)
#
# This means for every edge (u, v), c_u != c_v.
# This is exactly the definition of a bipartite graph.
# We need to color the graph with 1 and 2 such that adjacent nodes have different colors.
# To minimize F = sum(c_i * 10^(n-i)), we want c_1 to be as small as possible (1),
# then c_2 as small as possible, etc.
# This is a greedy approach on connected components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i. To minimize F, we want color[i] = 1.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if we could have done better by setting color[i] = 2
            # Actually, for each component, we have two choices:
            # 1. color[i] = 1, then all others are fixed.
            # 2. color[i] = 2, then all others are flipped.
            # We compare the two assignments lexicographically.
            
            # Assignment 1:
            res1 = []
            for node, c in component:
                res1.append((node, c))
            res1.sort()
            
            # Assignment 2:
            res2 = []
            for node, c in component:
                res2.append((node, 3 - c))
            res2.sort()
            
            # Pick the one that is lexicographically smaller
            for j in range(len(res1)):
                if res1[j][1] < res2[j][1]:
                    break
                elif res1[j][1] > res2[j][1]:
                    for node, c in res2:
                        color[node] = c
                    break
            else:
                for node, c in res1:
                    color[node] = c
                    
    print("YES")
    print(*(color[1:]))

solve()
```
