```python
import sys

# The interference condition is p_u^2 + p_v^2 + p_u * p_v = 7.
# Possible values for p_u, p_v are {1, 2}.
# If p_u = 1, p_v = 1: 1^2 + 1^2 + 1*1 = 3 != 7
# If p_u = 1, p_v = 2: 1^2 + 2^2 + 1*2 = 1 + 4 + 2 = 7
# If p_u = 2, p_v = 1: 2^2 + 1^2 + 2*1 = 4 + 1 + 2 = 7
# If p_u = 2, p_v = 2: 2^2 + 2^2 + 2*2 = 4 + 4 + 4 = 12 != 7
# Thus, for every edge (u, v), p_u must be different from p_v.
# This is a bipartite graph coloring problem.
# We need to color the graph with 1 and 2 such that no two adjacent nodes have the same color.
# The tie-breaking rule: lexicographically smallest sequence (1 before 2).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we prefer color 1.
    # We iterate 1 to n. If not colored, start BFS/DFS.
    # To make it lexicographically smallest, we want color[i] = 1 if possible.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS component
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
