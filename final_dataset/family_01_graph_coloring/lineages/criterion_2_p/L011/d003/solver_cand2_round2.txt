import sys

# The problem asks for a configuration of 1s and 2s such that for every edge (u, v),
# duration[u] * duration[v] = 2.
# Since durations are 1 or 2, the only way the product is 2 is if one is 1 and the other is 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with two colors (1 and 2).
# To get the lexicographically smallest sequence, for each connected component,
# we have two choices of coloring (swapping 1s and 2s).
# We want to pick the choice that makes the sequence lexicographically smallest.
# For a component, let the set of nodes be V_c. Let the two possible colorings be:
# C1: nodes in set A get 1, nodes in set B get 2.
# C2: nodes in set A get 2, nodes in set B get 1.
# We want to compare C1 and C2 at the smallest index i in V_c.
# If i is in A, C1 gives 1 and C2 gives 2. C1 is smaller.
# If i is in B, C1 gives 2 and C2 gives 1. C2 is smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_node = min(component)
            # If color[min_node] == 1, we prefer the coloring where min_node is 1.
            # If color[min_node] == 2, we prefer the coloring where min_node is 1.
            # In the first case, we keep the current colors.
            # In the second case, we flip the colors (1->2, 2->1).
            
            if color[min_node] == 1:
                for node in component:
                    ans[node] = color[node]
            else:
                for node in component:
                    ans[node] = 3 - color[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
