import sys

# The condition: for every edge (u, v), duration[u] * duration[v] = 2.
# Since durations are 1 or 2, the only way to get a product of 2 is:
# (1, 2) or (2, 1).
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with two colors (say 0 and 1).
# Let color 0 correspond to duration 1 and color 1 correspond to duration 2.
# Or vice versa.
# For each connected component, we have two choices:
# Choice A: nodes with color 0 get 1, nodes with color 1 get 2.
# Choice B: nodes with color 0 get 2, nodes with color 1 get 1.
# To get the lexicographically smallest sequence, for each component,
# we pick the choice that results in a smaller value at the smallest index
# in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    result = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            component.append(i)
            
            is_bipartite = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        component.append(v)
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            start_node = min(component)
            
            # Choice 1: color 0 -> 1, color 1 -> 2
            # Choice 2: color 0 -> 2, color 1 -> 1
            # We want the smallest value at start_node.
            # If color[start_node] == 0:
            #   Choice 1 gives 1, Choice 2 gives 2. Pick Choice 1.
            # If color[start_node] == 1:
            #   Choice 1 gives 2, Choice 2 gives 1. Pick Choice 2.
            
            if color[start_node] == 0:
                for node in component:
                    result[node] = 1 if color[node] == 0 else 2
            else:
                for node in component:
                    result[node] = 2 if color[node] == 0 else 1
                    
    print("YES")
    print(*(result[1:]))

solve()
