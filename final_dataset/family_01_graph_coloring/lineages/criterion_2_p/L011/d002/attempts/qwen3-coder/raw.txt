```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    graph = defaultdict(list)
    
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        index += 2
        edges.append((u, v))
        graph[u].append(v)
        graph[v].append(u)
    
    # Color array: 0 = unvisited, 1 = color A, 2 = color B
    color = [0] * n
    visited = [False] * n
    
    # Check if bipartite for each component
    for start in range(n):
        if visited[start]:
            continue
            
        queue = deque()
        queue.append(start)
        visited[start] = True
        color[start] = 1  # Start with color 1
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            next_color = 3 - current_color  # Switches 1->2, 2->1
            
            for neighbor in graph[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = next_color
                    queue.append(neighbor)
                elif color[neighbor] != next_color:
                    # Conflict found: not bipartite
                    print("NO")
                    return
    
    # If we reach here, the graph is bipartite (2-colorable)
    # Now assign phases: color 1 -> phase 1, color 2 -> phase 2
    # This ensures lex smallest since we prefer 1 over 2
    
    phases = []
    for c in color:
        if c == 1:
            phases.append(1)
        else:
            phases.append(2)
    
    # Verify all edges satisfy the constraint: one is twice the other
    # Since phases are only 1 or 2, one must be 1 and the other 2
    for u, v in edges:
        if not ((phases[u] == 1 and phases[v] == 2) or (phases[u] == 2 and phases[v] == 1)):
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, phases)))

if __name__ == "__main__":
    main()
```
