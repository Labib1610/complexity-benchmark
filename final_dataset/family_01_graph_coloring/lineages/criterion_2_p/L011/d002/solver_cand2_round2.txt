import sys

# The condition: for any two projectors u and v sharing a region,
# power(u)^2 + power(v)^2 = 5.
# Since power levels are 1 or 2, 1^2 + 2^2 = 1 + 4 = 5.
# This means if u and v are connected, they must have different power levels.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we want to assign 1
# to as many nodes as possible, prioritizing smaller indices.
# For each connected component, we can color it with 1s and 2s.
# There are two ways to color a bipartite component:
# 1. Color A with 1s, Color B with 2s.
# 2. Color A with 2s, Color B with 1s.
# We choose the one that makes the sequence lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # Option 1: color[v] as is (1s and 2s)
            # Option 2: 3 - color[v]
            # We want the lexicographically smallest result.
            # Compare the two options by looking at the smallest index in the component.
            
            # Find the smallest index in this component
            comp_min = min(component)
            
            # If color[comp_min] == 1, Option 1 gives 1 at comp_min, Option 2 gives 2.
            # We prefer 1.
            if color[comp_min] == 1:
                for node in component:
                    res[node] = color[node]
            else:
                for node in component:
                    res[node] = 3 - color[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
