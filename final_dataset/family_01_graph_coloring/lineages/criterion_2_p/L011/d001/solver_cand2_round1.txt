```python
import sys

# The problem asks us to assign values 1 or 2 to each node in a graph such that
# for every edge (u, v), the product of values is 2.
# Let x_i be the value of node i. x_i in {1, 2}.
# x_u * x_v = 2.
# If x_u = 1, then x_v = 2.
# If x_u = 2, then x_v = 1.
# This is equivalent to a 2-coloring problem where the two colors are 1 and 2.
# A graph is 2-colorable if and only if it is bipartite.
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If a node is not visited, we assign it 1 (the smaller value) and perform BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the connected component
            # To be lexicographically smallest, we prefer 1 for the first node
            assignment[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    solve()
```
