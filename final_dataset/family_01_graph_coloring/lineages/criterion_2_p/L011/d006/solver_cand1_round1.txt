```python
import sys

# The stability score for a pair (p_u, p_v) where p_u, p_v in {1, 2}:
# If (1, 1): 1^2 + 1^2 = 2. Binary 10. Hamming weight = 1.
# If (1, 2): 1^2 + 2^2 = 5. Binary 101. Hamming weight = 2.
# If (2, 1): 2^2 + 1^2 = 5. Binary 101. Hamming weight = 2.
# If (2, 2): 2^2 + 2^2 = 8. Binary 1000. Hamming weight = 1.
#
# Total sum of scores = sum_{edges} weight(u, v) = 2m.
# Let x_i = 0 if p_i = 1, and x_i = 1 if p_i = 2.
# Weight(u, v) = 1 if x_u == x_v, and 2 if x_u != x_v.
# Total sum = sum_{edges} (1 + [x_u != x_v]) = m + sum_{edges} [x_u != x_v].
# We need m + sum_{edges} [x_u != x_v] = 2m, which means sum_{edges} [x_u != x_v] = m.
# This is only possible if every edge is a cut edge (x_u != x_v).
# This is equivalent to saying the graph is bipartite.
# If the graph is bipartite, we can color it with 0 and 1.
# To get the lexicographically smallest sequence (1s and 2s),
# we want 1s as much as possible.
# For each connected component, we have two choices of coloring (0/1 or 1/0).
# Let the two sets of the bipartition be A and B.
# If we assign 1 to A and 2 to B, the sequence is determined by the indices.
# We want the smallest index to be 1.
# For each component, find the smallest index i. If i is in A, assign 1 to A, 2 to B.
# If i is in B, assign 1 to B, 2 to A.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            component.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        component.append(v)
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Smallest index in component
            min_node = min(component)
            # If min_node is color 0, we want color 0 to be 1 (cycle 1)
            # If min_node is color 1, we want color 1 to be 1 (cycle 1)
            target_color_for_1 = color[min_node]
            for node in component:
                if color[node] == target_color_for_1:
                    assignment[node] = 1
                else:
                    assignment[node] = 2
                    
    print("YES")
    print(*(assignment[1:]))

solve()
```
