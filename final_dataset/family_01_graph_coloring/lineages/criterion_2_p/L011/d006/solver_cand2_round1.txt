```python
import sys

# The synchronization score for a threshold (u, v) depends on R = p_u^2 + p_v^2.
# Possible values for p_u, p_v are {1, 2}.
# Case 1: p_u = 1, p_v = 1 => R = 1^2 + 1^2 = 2.
#   Sequence: 2 -> 1. Operations: 1.
# Case 2: p_u = 1, p_v = 2 (or 2, 1) => R = 1^2 + 2^2 = 5.
#   Sequence: 5 -> 16 -> 8 -> 4 -> 2 -> 1. Operations: 5.
# Case 3: p_u = 2, p_v = 2 => R = 2^2 + 2^2 = 8.
#   Sequence: 8 -> 4 -> 2 -> 1. Operations: 3.
#
# Let x_i be 0 if p_i = 1, and 1 if p_i = 2.
# Score(u, v) = 1 if x_u == x_v == 0
# Score(u, v) = 5 if x_u != x_v
# Score(u, v) = 3 if x_u == x_v == 1
#
# Total sum = sum_{edges} Score(u, v) = 5m
# Let n0 be the number of edges with (0,0), n1 be (0,1) or (1,0), n2 be (1,1).
# n0 + n1 + n2 = m
# 1*n0 + 5*n1 + 3*n2 = 5m
# Substitute n0 = m - n1 - n2:
# (m - n1 - n2) + 5*n1 + 3*n2 = 5m
# 4*n1 + 2*n2 = 4m => 2*n1 + n2 = 2m
# Since n1 + n2 <= m, this implies n1 = 0 and n2 = m.
# This means all edges must have x_u = 1 and x_v = 1.
# This is only possible if the graph has no edges (m=0) or if all nodes
# connected by edges are assigned 2.
# Wait, if n1=0, then all edges must be (1,1). This means every connected
# component with at least one edge must have all its nodes assigned 2.
# Nodes not in any edge can be 1 to be lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            if not adj[i]:
                assignment[i] = 1
                visited[i] = True
            else:
                # Component has edges, all must be 2
                stack = [i]
                visited[i] = True
                while stack:
                    u = stack.pop()
                    assignment[u] = 2
                    for v in adj[u]:
                        if not visited[v]:
                            visited[v] = True
                            stack.append(v)
                            
    # Verify
    total_score = 0
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        r = assignment[u]**2 + assignment[v]**2
        # Calculate Collatz steps
        steps = 0
        curr = r
        while curr != 1:
            if curr % 2 == 0:
                curr //= 2
            else:
                curr = 3 * curr + 1
            steps += 1
        total_score += steps
        idx += 2
        
    if m == 0 or total_score == 5 * m:
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
```
