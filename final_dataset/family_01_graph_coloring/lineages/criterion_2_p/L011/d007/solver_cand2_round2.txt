import sys

# The condition p_u * p_v = 2 with p_i in {1, 2} means:
# If p_u = 1, then p_v = 2.
# If p_u = 2, then p_v = 1.
# This is equivalent to saying p_u != p_v.
# This is a bipartite coloring problem.
# We want the lexicographically smallest sequence (p_1, ..., p_n).
# To make it lexicographically smallest, we prefer 1 over 2.
# For each connected component, we have two possible colorings (swapping 1s and 2s).
# We choose the coloring that results in a smaller value at the first index 
# encountered in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            idx_stack = 0
            while idx_stack < len(component):
                u, c = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            first_node = min(node for node, c in component)
            # We want the color of first_node to be 1 if possible.
            # If color[first_node] is 2, we flip all colors in this component.
            target_color_of_first = color[first_node]
            if target_color_of_first == 2:
                for node, c in component:
                    ans[node] = 3 - c
            else:
                for node, c in component:
                    ans[node] = c
                    
    print("YES")
    print(*(ans[1:]))

solve()
