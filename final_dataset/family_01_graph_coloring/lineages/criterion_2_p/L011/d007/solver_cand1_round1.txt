import sys

# The condition det(M) = 0 means f_u * f_v - 2 = 0.
# Since f_i \in {1, 2}, the only way f_u * f_v = 2 is if one is 1 and the other is 2.
# This is a bipartite graph coloring problem where 1 and 2 are the two colors.
# If the graph is bipartite, we can color each connected component in two ways.
# To minimize Z = sum(f_i * 3^(n-i)), we want smaller values (1) at earlier indices.
# For each connected component, let the nodes be {v_1, v_2, ...} with v_1 being the smallest index.
# We have two choices for the component:
# Choice A: f_{v_1} = 1, then all neighbors of v_1 must be 2, etc.
# Choice B: f_{v_1} = 2, then all neighbors of v_1 must be 1, etc.
# We compare the two choices for each component and pick the one that results in a smaller Z.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Try both colorings
            # Option 1: color[i] = 1 (already set)
            # Option 2: color[i] = 2
            
            # Calculate Z for Option 1
            z1 = 0
            for u, c in component:
                val = c
                z1 += val * (3**(n - u))
            
            # Calculate Z for Option 2
            z2 = 0
            for u, c in component:
                val = 3 - c
                z2 += val * (3**(n - u))
            
            if z1 <= z2:
                for u, c in component:
                    ans[u] = c
            else:
                for u, c in component:
                    ans[u] = 3 - c
                    
    print("YES")
    print(*(ans[1:]))

solve()
