```python
import sys

# The resonance equation is p_u^3 + p_v^3 - p_u * p_v = 7
# Possible values for p_i are {1, 2}.
# If p_u = 1, p_v = 1: 1 + 1 - 1 = 1 != 7
# If p_u = 1, p_v = 2: 1 + 8 - 2 = 7 (Valid)
# If p_u = 2, p_v = 1: 8 + 1 - 2 = 7 (Valid)
# If p_u = 2, p_v = 2: 8 + 8 - 4 = 12 != 7
# Thus, for every requirement (u, v), p_u must be different from p_v.
# This is a standard 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we prefer 1 over 2.
    # We iterate from 1 to n. If a node is uncolored, we assign it 1 and BFS/DFS.
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
