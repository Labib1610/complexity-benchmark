import sys

def solve():
    # The condition R = D:
    # R = 2 * sum_{j=1}^m (p_{u_j} * p_{v_j}) + 9m
    # D = sum_{i=1}^n d_i * (6*p_i - p_i^2)
    # Let x_i = p_i - 1. Since p_i in {1, 2}, x_i in {0, 1}.
    # p_i = x_i + 1
    # p_{u_j} * p_{v_j} = (x_{u_j} + 1)(x_{v_j} + 1) = x_{u_j}x_{v_j} + x_{u_j} + x_{v_j} + 1
    # R = 2 * sum(x_{u_j}x_{v_j} + x_{u_j} + x_{v_j} + 1) + 9m
    #   = 2 * sum(x_{u_j}x_{v_j}) + 2 * sum(x_{u_j} + x_{v_j}) + 2m + 9m
    #   = 2 * sum(x_{u_j}x_{v_j}) + 2 * sum(d_i * x_i) + 11m
    # D = sum(d_i * (6(x_i + 1) - (x_i + 1)^2))
    #   = sum(d_i * (6x_i + 6 - (x_i^2 + 2x_i + 1)))
    #   = sum(d_i * (-x_i^2 + 4x_i + 5))
    # Since x_i in {0, 1}, x_i^2 = x_i.
    #   = sum(d_i * (3x_i + 5)) = 3 * sum(d_i * x_i) + 5 * sum(d_i)
    # Note: sum(d_i) = 2m.
    # D = 3 * sum(d_i * x_i) + 10m
    # R = D => 2 * sum(x_{u_j}x_{v_j}) + 2 * sum(d_i * x_i) + 11m = 3 * sum(d_i * x_i) + 10m
    # 2 * sum(x_{u_j}x_{v_j}) + m = sum(d_i * x_i)
    # sum_{j=1}^m (2 * x_{u_j}x_{v_j} - x_{u_j} - x_{v_j}) + m = 0
    # sum_{j=1}^m (2 * x_{u_j}x_{v_j} - x_{u_j} - x_{v_j} + 1) = 0
    # sum_{j=1}^m (1 - 2x_{u_j}x_{v_j} + x_{u_j} + x_{v_j} - 1) ... wait
    # Let y_i = 2x_i - 1. y_i in {-1, 1}.
    # (y_{u_j} + 1)(y_{v_j} + 1) = 4 * x_{u_j}x_{v_j}
    # The equation is sum_{j=1}^m (2 * x_{u_j}x_{v_j} - x_{u_j} - x_{v_j} + 1) = 0
    # This is equivalent to sum_{j=1}^m (1 - 2x_{u_j}x_{v_j} + x_{u_j} + x_{v_j}) = m
    # Actually, the equation simplifies to: sum_{j=1}^m (x_{u_j} - 1)(x_{v_j} - 1) = 0
    # Let z_i = x_i - 1. z_i in {-1, 0}. This is not quite right.
    # Let's re-evaluate: 2 * sum(x_{u_j}x_{v_j}) - sum(d_i * x_i) + m = 0
    # This is sum_{j=1}^m (2 * x_{u_j}x_{v_j} - x_{u_j} - x_{v_j} + 1) = 0
    # sum_{j=1}^m (2 * x_{u_j}x_{v_j} - x_{u_j} - x_{v_j} + 1) = sum_{j=1}^m (1 - x_{u_j})(1 - x_{v_j}) = 0
    # Since (1 - x_i) is 1 if p_i=1 and 0 if p_i=2, let q_i = 1 - x_i.
    # We need sum_{j=1}^m (q_{u_j} * q_{v_j}) = 0.
    # This means for every requirement (u, v), we cannot have both q_u=1 and q_v=1.
    # q_i = 1 means p_i = 1. q_i = 0 means p_i = 2.
    # So we need to assign p_i in {1, 2} such that no two projectors with p_i=1 are connected.
    # This is the Independent Set problem, but we want the lexicographically smallest p.
    # Lexicographically smallest p means we want as many 1s as possible at the start.
    # This is equivalent to finding the Maximum Independent Set, but we want the greedy approach.
    # Actually, we want to set p_i=1 if possible.
    # For each i from 1 to n: if we can set p_i=1 without violating the condition, do it.
    
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
        idx += 2
    
    p = [0] * (n + 1)
    # To be lexicographically smallest, we want p_i=1 as much as possible.
    # This is equivalent to finding the lexicographically first maximal independent set?
    # No, just check if we can set p_i=1.
    # If we set p_i=1, all neighbors must be 2.
    # If we set p_i=2, neighbors can be 1 or 2.
    # This is 2-SAT.
    
    ans = [0] * (n + 1)
    # For each component, we want to pick 1s greedily.
    # This is bipartite matching/coloring.
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            comp = []
            while stack:
                u, c = stack.pop()
                comp.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # For this component, we have two sets: S0 (color 0) and S1 (color 1).
            # We can set all S0=1, S1=2 OR all S0=2, S1=1.
            # To make p lexicographically smallest, compare the two options.
            def get_seq(c0_val):
                res = [0] * (n + 1)
                for node, c in comp:
                    res[node] = 1 if c == c0_val else 2
                return res
            
            s1 = get_seq(0)
            s2 = get_seq(1)
            
            # Compare s1 and s2
            best = s1
            for node in sorted([x[0] for x in comp]):
                if s1[node] < s2[node]:
                    best = s1
                    break
                elif s2[node] < s1[node]:
                    best = s2
                    break
            for node in range(1, n + 1):
                if best[node] != 0: ans[node] = best[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
