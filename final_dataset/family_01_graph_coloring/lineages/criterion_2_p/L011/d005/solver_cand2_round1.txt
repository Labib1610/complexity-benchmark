import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Flux formula: Phi(p_u, p_v) = 5(p_u + p_v) - 3(p_u * p_v)
    # If p=1, 2:
    # (1, 1) -> 5(2) - 3(1) = 7
    # (1, 2) -> 5(3) - 3(2) = 9
    # (2, 1) -> 5(3) - 3(2) = 9
    # (2, 2) -> 5(4) - 3(4) = 8
    # Let x_i = 0 if p_i = 1, x_i = 1 if p_i = 2.
    # If p_u=1, p_v=1 (x_u=0, x_v=0): Phi = 7
    # If p_u=1, p_v=2 (x_u=0, x_v=1): Phi = 9
    # If p_u=2, p_v=1 (x_u=1, x_v=0): Phi = 9
    # If p_u=2, p_v=2 (x_u=1, x_v=1): Phi = 8
    # Total Flux = sum(Phi) = 9m
    # Let c00, c01, c11 be counts of edges (0,0), (0,1), (1,1).
    # c00 + c01 + c11 = m
    # 7*c00 + 9*c01 + 8*c11 = 9m
    # 7*c00 + 9*c01 + 8*c11 = 9(c00 + c01 + c11)
    # -2*c00 - c11 = 0 => 2*c00 + c11 = 0
    # Since c00, c11 >= 0, this implies c00 = 0 and c11 = 0.
    # This means there are no edges between two 1s and no edges between two 2s.
    # This is exactly the condition for a bipartite graph.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * n
    components = []
    for i in range(n):
        if color[i] == -1:
            comp = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            components.append(comp)
            
    # We want the lexicographically smallest string of 1s and 2s.
    # 1 corresponds to x=0, 2 corresponds to x=1.
    # For each component, we have two choices:
    # Choice A: nodes with color 0 get x=0 (1), nodes with color 1 get x=1 (2)
    # Choice B: nodes with color 0 get x=1 (2), nodes with color 1 get x=0 (1)
    # We want to pick the choice that makes the sequence smallest.
    
    res = [0] * n
    for comp in components:
        # Find the smallest index in this component
        min_idx = min(node[0] for node in comp)
        # Check which assignment puts a 1 at min_idx
        # If color[min_idx] == 0:
        # Choice A: min_idx gets 1. Choice B: min_idx gets 2.
        # Choice A is better.
        # If color[min_idx] == 1:
        # Choice A: min_idx gets 2. Choice B: min_idx gets 1.
        # Choice B is better.
        
        c_min = color[min_idx]
        for node, c in comp:
            if c_min == 0:
                res[node] = 1 if c == 0 else 2
            else:
                res[node] = 2 if c == 0 else 1
                
    print("YES")
    print(*(res))

solve()
