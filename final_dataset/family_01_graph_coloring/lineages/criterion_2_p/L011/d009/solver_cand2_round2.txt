import sys

# The problem states:
# Mixture 1: 1 Alpha, 2 Beta
# Mixture 2: 2 Alpha, 1 Beta
# Requirement (u, v): Alpha(u) + Alpha(v) = 3, Beta(u) + Beta(v) = 3
# Since Alpha(u) + Beta(u) = 3, the Beta condition is redundant if the Alpha condition holds.
# Alpha(u) + Alpha(v) = 3 means if Alpha(u) = 1, then Alpha(v) = 2, and vice versa.
# This is a bipartite coloring problem.
# For each connected component, we can pick the value of one node, and the rest are determined.
# To get the lexicographically smallest sequence, for each component, we try setting the 
# first node (by index) to 1, then check for consistency. If consistent, we check if 
# setting it to 2 is better (it won't be, 1 is always better).
# Actually, for each component, we have two choices:
# Choice A: node i = 1, then propagate.
# Choice B: node i = 2, then propagate.
# We want the smallest value at the smallest index.
# For each component, find the smallest index 'min_idx'.
# We must decide if node 'min_idx' should be 1 or 2.
# If we set node 'min_idx' = 1, we get a valid assignment (if possible).
# If we set node 'min_idx' = 2, we get a valid assignment (if possible).
# Since 1 < 2, we always prefer 1 if it's valid.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # We want the smallest index in this component to be 1
            # Find all nodes in this component first
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_in_comp = 0
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring, check if the smallest index in this component is 1
            # If color[min_idx] is 2, we can flip all colors in this component
            min_idx = min(q)
            if color[min_idx] == 2:
                for node in q:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
