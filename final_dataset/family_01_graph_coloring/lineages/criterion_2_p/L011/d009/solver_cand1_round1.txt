import sys

# The cross-modulation spectrum S_{u,v} = {x * y | x in I_u, y in I_v}.
# If I_u = [a, b] and I_v = [c, d], then S_{u,v} = [min(ac, ad, bc, bd), max(ac, ad, bc, bd)].
# The span is max(S_{u,v}) - min(S_{u,v}).
# Possible intervals are [0, 1] (right endpoint 1) and [1, 2] (right endpoint 2).
# Let x_i be the right endpoint of I_i (x_i in {1, 2}).
# If I_i = [0, 1], then I_i = [x_i-1, x_i].
# If I_i = [1, 2], then I_i = [x_i-1, x_i].
# In both cases, I_i = [x_i-1, x_i].
# S_{u,v} = [(x_u-1)(x_v-1), x_u * x_v].
# Span = x_u * x_v - (x_u-1)(x_v-1) = x_u * x_v - (x_u * x_v - x_u - x_v + 1) = x_u + x_v - 1.
# We require span = 2, so x_u + x_v - 1 = 2 => x_u + x_v = 3.
# Since x_i in {1, 2}, x_u + x_v = 3 means x_u != x_v.
# This is a 2-coloring problem on a graph where edges represent x_u != x_v.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest sequence, try color 1 first
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if starting with 1 was optimal for this component
            # Actually, for lexicographical smallest, we want color[i] = 1 if possible.
            # Since the graph is bipartite, once color[i] is fixed, all others are fixed.
            # The greedy choice is always to set the first uncolored node to 1.
            
    print("YES")
    print(*(color[1:]))

solve()
