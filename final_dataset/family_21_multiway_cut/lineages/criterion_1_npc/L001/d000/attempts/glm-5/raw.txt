```python
import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    B = int(next(it))
    
    generators = set()
    for _ in range(k):
        generators.add(int(next(it)))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w, i))
        adj[v].append((u, w, i))
        edges.append((u, v, w))
    
    # BFS from each generator to find minimum cut edges
    # We need to find edges that separate generators from each other
    
    # For each generator, BFS to find edges that lead to other generators
    # Track which generator reaches each node first
    
    owner = [0] * (n + 1)
    boundary_edges = set()
    
    for gen in generators:
        if owner[gen] != 0:
            continue
        q = deque([gen])
        owner[gen] = gen
        while q:
            u = q.popleft()
            for v, w, idx in adj[u]:
                if owner[v] == 0:
                    owner[v] = gen
                    q.append(v)
                elif owner[v] != gen:
                    # This edge crosses to another generator's region
                    boundary_edges.add(idx)
    
    # Now we need to check if removing boundary_edges is enough
    # and if cost <= B
    
    # Build graph without boundary edges
    adj2 = [[] for _ in range(n + 1)]
    for i, (u, v, w) in enumerate(edges):
        if i not in boundary_edges:
            adj2[u].append(v)
            adj2[v].append(u)
    
    # Check connectivity of generators
    visited = [False] * (n + 1)
    gen_list = list(generators)
    first_gen = gen_list[0]
    q = deque([first_gen])
    visited[first_gen] = True
    while q:
        u = q.popleft()
        for v in adj2[u]:
            if not visited[v]:
                visited[v] = True
                q.append(v)
    
    all_disconnected = True
    for g in generators:
        if not visited[g]:
            continue
        if g != first_gen:
            all_disconnected = False
            break
    
    if all_disconnected:
        # Check if first_gen is the only visited generator
        count = sum(1 for g in generators if visited[g])
        if count > 1:
            all_disconnected = False
    
    # Actually, we need to check that NO two generators are connected
    # So we need to check each component has at most one generator
    
    visited2 = [False] * (n + 1)
    for start in range(1, n + 1):
        if visited2[start]:
            continue
        q = deque([start])
        visited2[start] = True
        gen_count = 0
        while q:
            u = q.popleft()
            if u in generators:
                gen_count += 1
            for v in adj2[u]:
                if not visited2[v]:
                    visited2[v] = True
                    q.append(v)
        if gen_count > 1:
            all_disconnected = False
            break
    
    if all_disconnected:
        total_cost = sum(edges[i][2] for i in boundary_edges)
        if total_cost <= B:
            print("YES")
            print(len(boundary_edges))
            for i in boundary_edges:
                u, v, w = edges[i]
                print(u, v)
            return
    
    print("NO")

if __name__ == "__main__":
    solve()
```
