    t = int(next(it))
    B = int(next(it))

            if dist_s[v] > d + w:
                dist_s[v] = d + w
                heapq.heappush(pq, (dist_s[v], v))

    dist_t = [INF] * (n + 1)

    pq = [(0, t)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist_t[u]:
            continue
        for v, w, _ in adj[u]:
            if dist_t[v] > d + w:
                dist_t[v] = d + w
                heapq.heappush(pq, (dist_t[v], v))
    
    # Check if s and t are already disconnected
    if dist_s[t] == INF:
        print("YES")
        print("0")
        return

    if dist_s[t] > B:
        print("NO")
        return
    
    # Now we need to find a valid cut
    # Strategy: find shortest path, then output edges on it if total <= B
    
    # Find shortest path from s to t
    parent = [(-1, -1)] * (n + 1)  # (prev_node, edge_index)

    dist[s] = 0

    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue

            break

            if dist[v] > d + w:
                dist[v] = d + w
                parent[v] = (u, idx)
                heapq.heappush(pq, (dist[v], v))

        prev, idx = parent[cur]
        path_edges.append(idx)
        cur = prev

        result_edges.append((u, v, w))
        total_cost += w

        print("YES")
        print(len(result_edges))
        for u, v, _ in result_edges:

    # Actually for cut, we want to find edges on some path to delete

    # For larger graphs, compute path-based DP

    # dp[k] = (min_cost, edge_set) for disconnecting s from current frontier using k edges
    # Too complex...
    
    # Use level-based cut: nodes at dist_s[v] < cutoff are in S set
    # Find best cutoff
    best_cost = INF

            if dist_s[s] >= cutoff and dist_s[t] < cutoff:

                best_edges = cut_edges
    
    if best_edges is not None:
        print("YES")
        print(len(best_edges))
        for u, v in best_edges:
            print(u, v)
        return
    
    print("NO")

solve()
