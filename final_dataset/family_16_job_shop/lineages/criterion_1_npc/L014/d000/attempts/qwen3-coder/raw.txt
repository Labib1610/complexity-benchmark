from collections import defaultdict
import sys

def solve():
    j, m = map(int, input().split())
    
    queries = []
    for _ in range(j):
        line = list(map(int, input().split()))
        o = line[0]
        steps = []
        for i in range(o):
            resource = line[1 + 2*i]
            duration = line[1 + 2*i + 1]
            steps.append((resource, duration))
        queries.append(steps)
    
    # Binary search on the answer
    def can_finish_by(max_time):
        # For each query, we need to assign start times to its steps
        # such that:
        # 1. Steps of each query happen in order
        # 2. No resource is used by multiple steps at the same time
        # 3. All steps finish by max_time
        
        # This is a scheduling problem. We can model it as a SAT problem
        # or use a greedy approach with backtracking.
        
        # Let's try a greedy approach with backtracking:
        # For each query, we'll try to schedule its steps as early as possible,
        # respecting the constraints.
        
        # We'll use a recursive function with memoization
        # State: for each query, the earliest time its next step can start
        # and for each resource, the earliest time it's free
        
        # Actually, let's think differently.
        # We can model this as a constraint satisfaction problem.
        # For each step of each query, we have a variable: its start time.
        # Constraints:
        # 1. step i of query q starts after step i-1 of query q finishes
        # 2. if two steps use the same resource, they don't overlap
        # 3. all steps finish by max_time
        
        # Let's use a backtracking approach.
        # For each query, we'll try to assign start times to its steps.
        
        # resource_free_at[r] is the earliest time resource r is free
        resource_free_at = [0] * (m + 1)
        # For each query, the earliest time its next step can start
        query_next_earliest = [0] * j
        
        # Try to schedule queries one by one
        for q_idx in range(j):
            query = queries[q_idx]
            # Try to schedule this query's steps
            step_start_times = []
            current_time = query_next_earliest[q_idx]
            
            valid_schedule = True
            for step_idx, (resource, duration) in enumerate(query):
                # The step must start at or after current_time
                # and at or after when the resource is free
                start_time = max(current_time, resource_free_at[resource])
                
                # Check if this step can finish by max_time
                if start_time + duration > max_time:
                    valid_schedule = False
                    break
                
                step_start_times.append(start_time)
                current_time = start_time + duration
                # Update when the resource will be free
                resource_free_at[resource] = start_time + duration
            
            if not valid_schedule:
                return False
        
        return True
    
    # Binary search on the answer
    left, right = 0, 50 * 1000  # Upper bound: 50 steps * 1000 duration
    result = right
    
    while left <= right:
        mid = (left + right) // 2
        if can_finish_by(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1
    
    print(result)

solve()
