from collections import deque
import sys

def solve():
    j, m = map(int, input().split())
    
    documents = []
    for _ in range(j):
        line = list(map(int, input().split()))
        o = line[0]
        reviews = []
        for i in range(o):
            official = line[1 + 2*i]
            duration = line[1 + 2*i + 1]
            reviews.append((official, duration))
        documents.append(reviews)
    
    # Try all possible orderings of documents using backtracking
    def backtrack(doc_indices, official_end_times, doc_end_times):
        if len(doc_indices) == j:
            return max(doc_end_times)
        
        min_time = float('inf')
        for i in range(j):
            if i in doc_indices:
                continue
                
            # Calculate when this document can finish
            current_doc_end_times = doc_end_times[:]
            current_official_end_times = official_end_times[:]
            doc_finish_time = schedule_document(i, current_official_end_times, current_doc_end_times)
            
            # Recurse
            new_doc_indices = doc_indices + [i]
            min_time = min(min_time, backtrack(new_doc_indices, current_official_end_times, current_doc_end_times))
            
        return min_time
    
    def schedule_document(doc_idx, official_end_times, doc_end_times):
        doc = documents[doc_idx]
        current_time = max(doc_end_times) if doc_end_times else 0
        
        for official, duration in doc:
            # This review can start only after:
            # 1. The previous review of the same document finishes
            # 2. The official is free
            start_time = max(current_time, official_end_times[official-1])
            end_time = start_time + duration
            official_end_times[official-1] = end_time
            current_time = end_time
            
        doc_end_times.append(current_time)
        return current_time
    
    # Since j <= 10, we can try all permutations
    import itertools
    
    min_finish_time = float('inf')
    
    for perm in itertools.permutations(range(j)):
        official_end_times = [0] * m
        doc_end_times = []
        current_time = 0
        
        for doc_idx in perm:
            doc = documents[doc_idx]
            doc_start_time = 0
            
            # For each review in the document
            for official, duration in doc:
                # Earliest start time is max of when the official is free and when previous review finished
                start_time = max(doc_start_time, official_end_times[official-1])
                end_time = start_time + duration
                official_end_times[official-1] = end_time
                doc_start_time = end_time
                
            doc_end_times.append(doc_start_time)
            
        min_finish_time = min(min_finish_time, max(doc_end_times))
        
    print(min_finish_time)

solve()
