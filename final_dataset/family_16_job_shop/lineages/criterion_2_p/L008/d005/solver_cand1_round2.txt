import sys

def solve():
    # The problem asks to minimize the total accumulated cost.
    # Let the sequence of sectors be (p1_i, p2_i).
    # The total cost is sum(p2_i) + sum(grid_draw_i).
    # Since sum(p2_i) is constant regardless of the order, we only need to minimize
    # the sum of grid draws.
    # Let C_i be the battery charge before processing sector i.
    # grid_draw_i = max(0, p1_i - C_i)
    # C_{i+1} = max(0, C_i - p1_i) + p2_i
    #
    # Consider two adjacent sectors i and j.
    # If we process i then j:
    # Cost_i = max(0, p1_i - C) + p2_i
    # C_mid = max(0, C - p1_i) + p2_i
    # Cost_j = max(0, p1_j - C_mid) + p2_j
    # Total = max(0, p1_i - C) + max(0, p1_j - max(0, C - p1_i) - p2_i) + p2_i + p2_j
    #
    # Case 1: p1_i <= p2_i. These sectors increase the battery charge.
    # To minimize grid draws, we should process sectors that increase the charge
    # as early as possible if they help satisfy future requirements.
    # Actually, sorting by p1_i ascending for p1_i <= p2_i and p1_i > p2_i
    # is a standard greedy approach for this type of scheduling.
    # Specifically:
    # 1. Sectors with p1_i <= p2_i: sort by p1_i ascending.
    # 2. Sectors with p1_i > p2_i: sort by p2_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sectors = []
    idx = 1
    total_p2 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        total_p2 += p2
        idx += 2
        
    # Group 1: p1 <= p2 (net gainers)
    # Group 2: p1 > p2 (net losers)
    g1 = [s for s in sectors if s[0] <= s[1]]
    g2 = [s for s in sectors if s[0] > s[1]]
    
    g1.sort(key=lambda x: x[0])
    g2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = g1 + g2
    
    battery = 0
    grid_draw = 0
    for p1, p2 in sequence:
        if battery < p1:
            grid_draw += (p1 - battery)
            battery = 0
        else:
            battery -= p1
        battery += p2
        
    print(total_p2 + grid_draw)

if __name__ == '__main__':
    solve()
