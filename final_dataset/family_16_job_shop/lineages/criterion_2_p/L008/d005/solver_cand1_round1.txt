import sys

def solve():
    # The problem asks to minimize (final_reserve + sum(p1)).
    # Let the sequence be (p1_1, p2_1), (p1_2, p2_2), ..., (p1_j, p2_j).
    # Let R_i be the reserve after step i.
    # R_0 = 0
    # R_i = max(0, R_{i-1} - p1_i) + p2_i
    # We want to minimize R_j + sum(p1_i).
    
    # Let's analyze the structure.
    # If we have two sectors (p1_a, p2_a) and (p1_b, p2_b):
    # Order (a, b):
    # R_a = max(0, -p1_a) + p2_a = p2_a
    # R_b = max(0, p2_a - p1_b) + p2_b
    # Metric = max(0, p2_a - p1_b) + p2_b + p1_a + p1_b
    #        = max(p2_a + p1_a, p1_a + p1_b - p1_b + p2_b) ... wait.
    # Actually, the metric is:
    # If we process sectors with p2 >= p1 first, the reserve grows.
    # If we process sectors with p2 < p1, the reserve shrinks.
    # To minimize the final reserve, we should process sectors that shrink the reserve
    # (p2 < p1) as early as possible, and sectors that grow the reserve (p2 >= p1)
    # as late as possible.
    
    # Specifically, for sectors where p2 < p1, we want to process them in an order
    # that keeps the reserve as low as possible. For sectors where p2 >= p1,
    # the order doesn't affect the final reserve as much, but we should put them
    # after the shrinking ones.
    
    # Actually, the optimal strategy is:
    # 1. Sort sectors with p2 < p1 by p1 descending.
    # 2. Sort sectors with p2 >= p1 by p2 ascending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sectors = []
    sum_p1 = 0
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        sum_p1 += p1
        idx += 2
        
    shrinking = [s for s in sectors if s[1] < s[0]]
    growing = [s for s in sectors if s[1] >= s[0]]
    
    # Sort shrinking by p1 descending
    shrinking.sort(key=lambda x: x[0], reverse=True)
    # Sort growing by p2 ascending
    growing.sort(key=lambda x: x[1])
    
    sequence = shrinking + growing
    
    reserve = 0
    for p1, p2 in sequence:
        reserve -= p1
        if reserve < 0:
            reserve = 0
        reserve += p2
        
    print(reserve + sum_p1)

if __name__ == '__main__':
    solve()
