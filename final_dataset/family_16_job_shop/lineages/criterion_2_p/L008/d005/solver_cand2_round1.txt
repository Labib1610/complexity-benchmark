import sys

def solve():
    # The problem asks to minimize: Sum(p1) + Final_Calibration_Level
    # Let C_i be the calibration level after sector i.
    # C_i = max(C_{i-1} - p1_i + p2_i, p2_i)
    # Let D_i = p2_i - p1_i.
    # C_i = max(C_{i-1} + D_i, p2_i)
    #
    # Let's analyze the final calibration level C_j:
    # C_1 = max(0 + D_1, p2_1) = max(p2_1 - p1_1, p2_1) = p2_1
    # C_2 = max(C_1 + D_2, p2_2) = max(p2_1 + p2_2 - p1_2, p2_2)
    # C_3 = max(C_2 + D_3, p2_3) = max(p2_1 + p2_2 - p1_2 + p2_3 - p1_3, p2_2 + p2_3 - p1_3, p2_3)
    #
    # In general, C_j = max(p2_j, p2_j + p2_{j-1} - p1_j, p2_j + p2_{j-1} + p2_{j-2} - p1_j - p1_{j-1}, ..., Sum(p2_1..p2_j) - Sum(p1_2..p1_j))
    # This simplifies to:
    # C_j = max_{1 <= k <= j} (Sum_{i=k}^j p2_i - Sum_{i=k+1}^j p1_i)
    # C_j = max_{1 <= k <= j} (p2_k + Sum_{i=k+1}^j (p2_i - p1_i))
    #
    # Let S = Sum(p1_i). We want to minimize S + C_j.
    # S + C_j = Sum(p1_i) + max_{1 <= k <= j} (p2_k + Sum_{i=k+1}^j (p2_i - p1_i))
    #         = max_{1 <= k <= j} (Sum(p1_i) + p2_k + Sum_{i=k+1}^j p2_i - Sum_{i=k+1}^j p1_i)
    #         = max_{1 <= k <= j} (Sum(p1_1..p1_k) + p2_k + Sum_{i=k+1}^j p2_i)
    #
    # To minimize this, we want to pick an ordering.
    # Notice that Sum(p1_1..p1_k) + p2_k + Sum_{i=k+1}^j p2_i = Sum(p2_all) + Sum(p1_1..p1_k) - Sum(p2_1..p2_k) + p2_k
    # This doesn't look right. Let's re-evaluate.
    #
    # Actually, the expression is: C_j = max(p2_j, p2_j + p2_{j-1} - p1_j, p2_j + p2_{j-1} + p2_{j-2} - p1_j - p1_{j-1}, ...)
    # Let x_i = p2_i - p1_i.
    # C_j = max(p2_j, p2_j + x_{j-1}, p2_j + x_{j-1} + x_{j-2}, ..., p2_j + x_{j-1} + ... + x_1)
    # To minimize this, we should sort the sectors such that the terms being added are as small as possible.
    # The terms are x_i = p2_i - p1_i. We want to put the smallest x_i values at the end of the sequence (the ones that get added most).
    # Sorting by x_i ascending: x_1 <= x_2 <= ... <= x_j.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sectors = []
    idx = 1
    sum_p1 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        sum_p1 += p1
        idx += 2
    
    # Sort by p2 - p1 ascending
    sectors.sort(key=lambda x: x[1] - x[0])
    
    curr_c = 0
    for p1, p2 in sectors:
        curr_c = max(curr_c - p1 + p2, p2)
        
    print(sum_p1 + curr_c)

solve()
