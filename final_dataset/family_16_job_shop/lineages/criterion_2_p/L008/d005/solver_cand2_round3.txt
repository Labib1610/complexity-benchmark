import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    sectors = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        idx += 2
        
    # Let the sequence be (a1, b1), (a2, b2), ..., (aj, bj).
    # Path score at k: S1(k) + S2(k, j)
    # S1(k) = a1 + ... + ak
    # S2(k, j) = bk + ... + bj
    # Path score(k) = (a1 + ... + ak) + (bk + ... + bj)
    # Let S = sum(a_i) + sum(b_i).
    # Path score(k) = (a1 + ... + ak) + (S - (b1 + ... + b_{k-1}))
    # Path score(k) = S + (a1 - b1) + (a2 - b2) + ... + (ak - bk) + bk
    # Let d_i = a_i - b_i.
    # Path score(k) = S + bk + sum_{i=1}^k d_i
    
    # To minimize the peak score, we want to arrange d_i such that the 
    # prefix sums of d_i are balanced.
    # Sort sectors by d_i = p1 - p2.
    sectors.sort(key=lambda x: x[0] - x[1])
    
    # With the sorted order, we want to check the peak score.
    # The peak score is max_{k=1..j} (S + bk + sum_{i=1}^k d_i)
    # Since we want to minimize this, we can use a greedy approach:
    # Place sectors with large d_i in the middle to keep prefix sums small.
    # Actually, the optimal arrangement is to alternate placing the smallest 
    # and largest remaining d_i values.
    
    from collections import deque
    dq = deque(sectors)
    new_order = []
    while dq:
        new_order.append(dq.popleft())
        if dq:
            new_order.append(dq.pop())
    
    # The sequence is now new_order. Calculate peak score.
    s_total = sum(p[0] for p in new_order) + sum(p[1] for p in new_order)
    
    # Path score(k) = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # Let P1_prefix[k] = sum_{i=1}^k p1_i
    # Let P2_suffix[k] = sum_{i=k}^j p2_i
    
    p1_prefix = [0] * (j + 1)
    for i in range(j):
        p1_prefix[i+1] = p1_prefix[i] + new_order[i][0]
        
    p2_suffix = [0] * (j + 2)
    for i in range(j, 0, -1):
        p2_suffix[i] = p2_suffix[i+1] + new_order[i-1][1]
        
    peak_score = 0
    for k in range(1, j + 1):
        path_score = p1_prefix[k] + p2_suffix[k]
        if path_score > peak_score:
            peak_score = path_score
            
    print(peak_score)

solve()
