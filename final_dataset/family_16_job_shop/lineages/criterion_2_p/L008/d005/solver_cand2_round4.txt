```python
import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    sectors = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        idx += 2
    
    # To minimize total elapsed time:
    # Total time = sum(p2) + sum(shortfalls)
    # Shortfall for sector i = max(0, p1_i - current_cache)
    # If we have a shortfall, current_cache becomes 0.
    # If we don't, current_cache becomes current_cache - p1_i + p2_i.
    # To minimize shortfalls, we want to keep current_cache as high as possible.
    # This is equivalent to processing sectors with p1 <= p2 first (they increase cache),
    # and then sectors with p1 > p2.
    # Among those with p1 <= p2, we should process them in increasing order of p1
    # to unlock them as easily as possible.
    # Among those with p1 > p2, we should process them in decreasing order of p1
    # to minimize the impact of the inevitable shortfall.
    
    pos = []
    neg = []
    for p1, p2 in sectors:
        if p1 <= p2:
            pos.append((p1, p2))
        else:
            neg.append((p1, p2))
            
    pos.sort(key=lambda x: x[0])
    neg.sort(key=lambda x: x[0], reverse=True)
    
    sequence = pos + neg
    
    total_time = 0
    cache = 0
    for p1, p2 in sequence:
        if cache < p1:
            shortfall = p1 - cache
            total_time += shortfall
            cache = 0
        else:
            cache -= p1
        
        total_time += p2
        cache += p2
        
    print(total_time)

if __name__ == '__main__':
    solve()
```
