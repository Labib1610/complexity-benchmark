import sys

def solve():
    # The problem asks to minimize the total cost.
    # Let the sequence be (p1_i, p2_i).
    # Total cost = sum(max(0, p1_i - reserve_{i-1}) + p2_i)
    # reserve_i = max(0, reserve_{i-1} - p1_i) + p2_i
    #
    # Let's analyze the total cost:
    # Total Cost = sum(p2_i) + sum(max(0, p1_i - reserve_{i-1}))
    # reserve_0 = 0
    # reserve_1 = p2_1
    # reserve_2 = max(0, p2_1 - p1_2) + p2_2
    #
    # This is equivalent to:
    # Total Cost = sum(p2_i) + max(0, p1_1, p1_2 - p2_1, p1_3 - p2_1 - p2_2 + p1_2, ...)
    # Actually, a simpler way to view this:
    # The total cost is sum(p2_i) + max(0, p1_1, p1_2 - p2_1, p1_3 - p2_1 - p2_2, ...)
    # This is a classic scheduling problem.
    # If we have two sectors (a1, a2) and (b1, b2):
    # Order (a, b): cost = a2 + b2 + max(0, a1, b1 - a2)
    # Order (b, a): cost = b2 + a2 + max(0, b1, a1 - b2)
    # We prefer (a, b) if max(0, a1, b1 - a2) < max(0, b1, a1 - b2).
    # This is equivalent to sorting by p2 descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    sectors = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        idx += 2
        
    # Sort by p2 descending
    sectors.sort(key=lambda x: x[1], reverse=True)
    
    total_cost = 0
    reserve = 0
    for p1, p2 in sectors:
        shortfall = max(0, p1 - reserve)
        total_cost += shortfall + p2
        reserve = max(0, reserve - p1) + p2
        
    print(total_cost)

if __name__ == '__main__':
    solve()
