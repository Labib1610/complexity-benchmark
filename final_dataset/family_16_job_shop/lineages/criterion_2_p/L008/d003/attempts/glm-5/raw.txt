import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    sectors = []
    for _ in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        sectors.append((p1, p2))
    
    # Sort by Johnson's rule:
    # Sectors where p1 <= p2 come first, sorted by increasing p1.
    # Sectors where p1 > p2 come after, sorted by decreasing p2.
    group1 = []
    group2 = []
    for p1, p2 in sectors:
        if p1 <= p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: -x[1])
    
    ordered = group1 + group2
    
    # Simulate the process
    # Time when surface scan of current sector finishes
    scan_finish_time = 0
    # Time when deep core extraction of previous sector finishes
    prev_extract_finish_time = 0
    
    for p1, p2 in ordered:
        scan_finish_time += p1
        # Extraction starts when both scan is done and previous extraction is done
        extract_start = max(scan_finish_time, prev_extract_finish_time)
        prev_extract_finish_time = extract_start + p2
    
    print(prev_extract_finish_time)

if __name__ == "__main__":
    solve()
