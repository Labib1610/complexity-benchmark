```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    group1 = []  # a <= b
    group2 = []  # a > b
    for _ in range(n):
        a = int(next(it))
        b = int(next(it))
        if a <= b:
            group1.append((a, b))
        else:
            group2.append((a, b))
    group1.sort(key=lambda x: x[0])          # increasing a
    group2.sort(key=lambda x: -x[1])         # decreasing b
    order = group1 + group2

    t1 = 0  # completion time on machine 1 (scans)
    t2 = 0  # completion time on machine 2 (extractions)
    for a, b in order:
        t1 += a
        if t2 < t1:
            t2 = t1
        t2 += b
    sys.stdout.write(str(t2))

if __name__ == "__main__":
    solve()
```
