import sys

def solve():
    # The problem asks to minimize the completion time of the last deep core extraction.
    # Let the sequence of sectors be (s_1, e_1), (s_2, e_2), ..., (s_n, e_n).
    # Let C_i be the completion time of the extraction of sector i.
    # C_0 = 0
    # Scan completion time of sector i: S_i = sum_{k=1}^i s_k
    # Extraction completion time of sector i: C_i = max(S_i, C_{i-1}) + e_i
    # This is a classic scheduling problem known as Johnson's Rule for 2 machines.
    # To minimize the makespan, we sort the jobs based on the following:
    # If s_i < e_i, the job should come earlier.
    # If s_i >= e_i, the job should come later.
    # Specifically, sort by min(s_i, e_i).
    # If s_i < e_i, sort by s_i ascending.
    # If s_i >= e_i, sort by e_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(n):
        s = int(input_data[idx])
        e = int(input_data[idx+1])
        jobs.append((s, e))
        idx += 2
        
    # Johnson's Rule implementation
    group1 = [] # s < e
    group2 = [] # s >= e
    for s, e in jobs:
        if s < e:
            group1.append((s, e))
        else:
            group2.append((s, e))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_jobs = group1 + group2
    
    current_scan_time = 0
    current_extraction_time = 0
    
    for s, e in sorted_jobs:
        current_scan_time += s
        current_extraction_time = max(current_scan_time, current_extraction_time) + e
        
    print(current_extraction_time)

if __name__ == '__main__':
    solve()
