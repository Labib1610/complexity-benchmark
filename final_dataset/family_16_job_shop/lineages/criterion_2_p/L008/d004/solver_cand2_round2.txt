import sys

def solve():
    # Let p_i be the surface scan duration and q_i be the deep core extraction duration.
    # If we have a sequence of sectors (p_1, q_1), (p_2, q_2), ..., (p_j, q_j),
    # the surface scan of sector k ends at S_k = sum_{i=1}^k p_i.
    # The deep core extraction of sector k starts at D_k_start = max(S_k, D_{k-1}_end).
    # The deep core extraction of sector k ends at D_k_end = D_k_start + q_k.
    # This simplifies to D_k_end = max(S_k + q_k, D_{k-1}_end + q_k).
    # Expanding this recurrence:
    # D_j_end = max(S_j + q_j, S_{j-1} + q_{j-1} + q_j, ..., S_1 + q_1 + ... + q_j)
    # D_j_end = max_{1 <= k <= j} (S_k + sum_{i=k}^j q_i)
    # D_j_end = max_{1 <= k <= j} (sum_{i=1}^k p_i + sum_{i=k}^j q_i)
    # D_j_end = sum_{i=1}^j q_i + max_{1 <= k <= j} (sum_{i=1}^k p_i - sum_{i=1}^{k-1} q_i)
    # Let P_k = sum_{i=1}^k p_i and Q_{k-1} = sum_{i=1}^{k-1} q_i (with Q_0 = 0).
    # We want to minimize max_{1 <= k <= j} (P_k + sum_{i=k}^j q_i).
    # This is equivalent to minimizing the maximum of these values.
    # By Smith's Rule / Johnson's Rule logic for scheduling:
    # To minimize the makespan, we should sort sectors such that if we have two
    # adjacent sectors (p1, q1) and (p2, q2), we prefer (p1, q1) before (p2, q2)
    # if min(p1, q2) < min(p2, q1).
    # Specifically, sort by p_i < q_i ascending, then p_i > q_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sectors = []
    idx = 1
    for _ in range(j):
        p = int(input_data[idx])
        q = int(input_data[idx+1])
        sectors.append((p, q))
        idx += 2
        
    # Sort using the Johnson's rule criteria
    # Group 1: p < q, sort by p ascending
    # Group 2: p >= q, sort by q descending
    g1 = [s for s in sectors if s[0] < s[1]]
    g2 = [s for s in sectors if s[0] >= s[1]]
    g1.sort(key=lambda x: x[0])
    g2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_sectors = g1 + g2
    
    current_scan_end = 0
    current_extraction_end = 0
    
    for p, q in sorted_sectors:
        current_scan_end += p
        current_extraction_end = max(current_scan_end, current_extraction_end) + q
        
    print(current_extraction_end)

if __name__ == '__main__':
    solve()
