```python
import sys

def solve():
    # The problem asks to minimize the final beta.
    # Let the sequence be (p1_i, p2_i).
    # alpha_k = sum_{i=1}^k p1_i
    # beta_k = max(beta_{k-1}, alpha_k) + p2_k
    # This is equivalent to beta_n = max_{k=1..n} (alpha_k + sum_{i=k}^n p2_i)
    # Let S = sum_{i=1}^n p2_i.
    # beta_n = max_{k=1..n} (alpha_k + S - sum_{i=1}^{k-1} p2_i)
    # beta_n = S + max_{k=1..n} (alpha_k - sum_{i=1}^{k-1} p2_i)
    # Let q_k = sum_{i=1}^{k-1} p2_i. We want to minimize max(alpha_k - q_k).
    # This is a classic scheduling problem.
    # Consider two adjacent sectors i and j.
    # If we process i then j:
    # beta_i = max(alpha + p1_i, beta_prev) + p2_i
    # beta_j = max(alpha + p1_i + p1_j, beta_i) + p2_j
    #        = max(alpha + p1_i + p1_j, max(alpha + p1_i, beta_prev) + p2_i) + p2_j
    #        = max(alpha + p1_i + p1_j + p2_j, alpha + p1_i + p2_i + p2_j, beta_prev + p2_i + p2_j)
    # Comparing i,j vs j,i:
    # The term beta_prev + p2_i + p2_j is constant.
    # We compare max(p1_i + p1_j + p2_j, p1_i + p2_i + p2_j) vs max(p1_j + p1_i + p2_i, p1_j + p2_j + p2_i)
    # Which simplifies to comparing max(p1_i + p1_j + p2_j, p1_i + p2_i + p2_j) vs max(p1_j + p1_i + p2_i, p1_j + p2_j + p2_i)
    # This is equivalent to sorting by p2 in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sectors = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        idx += 2
    
    # Sort by p2 descending
    sectors.sort(key=lambda x: x[1], reverse=True)
    
    alpha = 0
    beta = 0
    for p1, p2 in sectors:
        alpha += p1
        if alpha > beta:
            beta = alpha
        beta += p2
        
    print(beta)

if __name__ == '__main__':
    solve()
```
