```python
import sys

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n = data[0]
    arr = data[1:]
    jobs1 = []  # p1 <= p2
    jobs2 = []  # p1 > p2
    it = iter(arr)
    for p1, p2 in zip(it, it):
        if p1 <= p2:
            jobs1.append((p1, p2))
        else:
            jobs2.append((p1, p2))
    jobs1.sort(key=lambda x: x[0])          # ascending p1
    jobs2.sort(key=lambda x: x[1], reverse=True)  # descending p2
    order = jobs1 + jobs2

    t1 = 0  # time on Radar Rover (machine 1)
    t2 = 0  # time on Sampling Rover (machine 2)
    for p1, p2 in order:
        t1 += p1
        if t2 < t1:
            t2 = t1
        t2 += p2
    print(t2)

if __name__ == "__main__":
    solve()
```
