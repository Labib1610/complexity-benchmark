```python
import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    commissions = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        commissions.append((p1, p2))
        idx += 2
    
    # Let the sequence be (s_1, c_1), (s_2, c_2), ..., (s_j, c_j).
    # The score for pivot k is: sum_{i=1}^k s_i + sum_{i=k}^j c_i
    # This is equal to: sum_{i=1}^j s_i + sum_{i=k}^j c_i - sum_{i=k+1}^j s_i
    # Actually, it's simpler:
    # Score(k) = (sum_{i=1}^{k-1} s_i) + s_k + c_k + (sum_{i=k+1}^j c_i)
    # Let S = sum(s_i) and C = sum(c_i).
    # Score(k) = (sum_{i=1}^k s_i) + (sum_{i=k}^j c_i)
    # To minimize the maximum score, we should sort the commissions.
    # If we have two commissions (s1, c1) and (s2, c2):
    # Order 1, 2: max(s1+c1+c2, s1+s2+c2)
    # Order 2, 1: max(s2+c2+c1, s2+s1+c1)
    # This is a classic scheduling problem. The optimal order is to sort by
    # min(s_i, c_i). Specifically, if s_i < c_i, put it early. If s_i >= c_i, put it late.
    # More precisely, sort by s_i < c_i ascending, then s_i >= c_i descending.
    
    group1 = []
    group2 = []
    for s, c in commissions:
        if s < c:
            group1.append((s, c))
        else:
            group2.append((s, c))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    seq = group1 + group2
    
    # Calculate prefix sums of s and suffix sums of c
    n = len(seq)
    pref_s = [0] * (n + 1)
    for i in range(n):
        pref_s[i+1] = pref_s[i] + seq[i][0]
        
    suff_c = [0] * (n + 1)
    for i in range(n - 1, -1, -1):
        suff_c[i] = suff_c[i+1] + seq[i][1]
        
    max_score = 0
    for k in range(1, n + 1):
        # Pivot at k (1-indexed)
        # Score = sum(s_1...s_k) + sum(c_k...c_j)
        score = pref_s[k] + suff_c[k-1]
        if score > max_score:
            max_score = score
            
    print(max_score)

if __name__ == '__main__':
    solve()
```
