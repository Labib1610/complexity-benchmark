import sys

def solve():
    # The problem asks to minimize: final_index + sum(p1)
    # Let current index be S.
    # Update rule: S_new = max(S - p1 + p2, p2)
    # Let d = p2 - p1. Then S_new = max(S + d, p2).
    # We want to minimize S_final + sum(p1).
    # Let's analyze the update: S_new = max(S + p2 - p1, p2).
    # This is equivalent to S_new = max(S - p1, 0) + p2.
    # Let f_i(S) = max(S - p1_i, 0) + p2_i.
    # We want to compute f_n(f_{n-1}(...f_1(0)...)) + sum(p1_i).
    # Note that f_i(S) = max(S - p1_i + p2_i, p2_i).
    # Let's re-examine the order.
    # If we have two sectors (p1_a, p2_a) and (p1_b, p2_b):
    # Order a, b: S1 = max(0 - p1_a + p2_a, p2_a) = max(p2_a - p1_a, p2_a) = p2_a
    # S2 = max(S1 - p1_b + p2_b, p2_b) = max(p2_a - p1_b + p2_b, p2_b)
    # Order b, a: S1 = p2_b
    # S2 = max(p2_b - p1_a + p2_a, p2_a)
    # We want to minimize max(p2_a + p2_b - p1_b, p2_b) vs max(p2_b + p2_a - p1_a, p2_a).
    # This is a classic scheduling problem.
    # If p2_a <= p2_b, then max(p2_a + p2_b - p1_b, p2_b) vs max(p2_b + p2_a - p1_a, p2_a).
    # Sorting by p2 ascending is the optimal strategy.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    sectors = []
    sum_p1 = 0
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        sum_p1 += p1
        idx += 2
        
    # Sort by p2 ascending
    sectors.sort(key=lambda x: x[1])
    
    current_index = 0
    for p1, p2 in sectors:
        current_index = max(current_index - p1 + p2, p2)
        
    print(current_index + sum_p1)

if __name__ == '__main__':
    solve()
