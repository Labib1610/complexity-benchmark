```python
import sys

def solve():
    # Let the sectors be (m_i, d_i).
    # If we order them as (m_1, d_1), (m_2, d_2), ..., (m_j, d_j),
    # the completion time at position k is:
    # T_k = (sum_{i=1}^k m_i) + (sum_{i=k}^j d_i)
    # Let S_m = sum_{i=1}^j m_i and S_d = sum_{i=1}^j d_i.
    # T_k = (sum_{i=1}^k m_i) + (S_d - sum_{i=1}^{k-1} d_i)
    # T_k = S_d + m_k + sum_{i=1}^{k-1} (m_i - d_i)
    # We want to minimize max_{k} T_k.
    # This is a classic scheduling problem. To minimize the maximum,
    # we should sort the sectors based on their drilling durations d_i.
    # Specifically, if we have two adjacent sectors (m_i, d_i) and (m_j, d_j),
    # swapping them changes the max of the two T values.
    # The optimal order is to sort by d_i in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    sectors = []
    idx = 1
    for _ in range(j):
        m = int(input_data[idx])
        d = int(input_data[idx+1])
        sectors.append((m, d))
        idx += 2
        
    # Sort by drilling duration d descending
    sectors.sort(key=lambda x: x[1], reverse=True)
    
    # Calculate T_k = (sum_{i=1}^k m_i) + (sum_{i=k}^j d_i)
    # Let prefix_m[k] = sum_{i=0}^{k-1} m_i
    # Let suffix_d[k] = sum_{i=k}^{j-1} d_i
    # T_k = prefix_m[k+1] + suffix_d[k]
    
    prefix_m = [0] * (j + 1)
    for i in range(j):
        prefix_m[i+1] = prefix_m[i] + sectors[i][0]
        
    suffix_d = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suffix_d[i] = suffix_d[i+1] + sectors[i][1]
        
    max_time = 0
    for k in range(1, j + 1):
        current_time = prefix_m[k] + suffix_d[k-1]
        if current_time > max_time:
            max_time = current_time
            
    print(max_time)

if __name__ == '__main__':
    solve()
```
