import sys

def solve():
    # The problem asks to minimize the peak resonance.
    # Let the sequence be (p1_1, p2_1), (p1_2, p2_2), ..., (p1_j, p2_j).
    # The intensity if shifting at k is:
    # I_k = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # I_k = sum_{i=1}^j p2_i + sum_{i=1}^k (p1_i - p2_i)
    # Let S = sum_{i=1}^j p2_i. Then I_k = S + sum_{i=1}^k (p1_i - p2_i).
    # We want to minimize max_{k=1..j} (S + sum_{i=1}^k (p1_i - p2_i)).
    # This is equivalent to minimizing max_{k=1..j} (sum_{i=1}^k (p1_i - p2_i)).
    # Let d_i = p1_i - p2_i. We want to order d_i to minimize the maximum prefix sum.
    # To minimize the maximum prefix sum, we should put negative d_i values first.
    # If we sort d_i in non-decreasing order, the prefix sums will be as small as possible.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    rituals = []
    idx = 1
    total_p2 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        rituals.append(p1 - p2)
        total_p2 += p2
        idx += 2
        
    rituals.sort()
    
    max_prefix_sum = -float('inf')
    current_sum = 0
    for d in rituals:
        current_sum += d
        if current_sum > max_prefix_sum:
            max_prefix_sum = current_sum
            
    print(total_p2 + max_prefix_sum)

if __name__ == '__main__':
    solve()
