```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    items = []
    total_b = 0
    for _ in range(n):
        a = int(next(it))
        b = int(next(it))
        items.append((a, b, a - b))
        total_b += b
    # sort by (a - b) descending
    items.sort(key=lambda x: x[2], reverse=True)

    pref_a = 0
    pref_b = 0
    suffix_b = total_b
    ans = 0
    for a, b, _ in items:
        pref_a += a
        suffix_b -= b
        cur = pref_a + suffix_b
        if cur > ans:
            ans = cur
    # also consider k = 0 (no prefix) -> sum b of all
    # but formula with pref_a=0, suffix_b=total_b already covered before loop if needed
    # and k = n (suffix empty) -> pref_a + 0 covered after last iteration
    print(ans)

if __name__ == "__main__":
    solve()
```
