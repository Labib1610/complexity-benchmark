```python
import sys

def solve():
    # The problem asks to minimize the maximum strain.
    # Let the sequence be (a_1, b_1), (a_2, b_2), ..., (a_j, b_j).
    # Strain at position k is S_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i.
    # S_k = sum_{i=1}^j a_i + sum_{i=k}^j b_i - sum_{i=k+1}^j a_i (Wait, this is wrong).
    # Let A = sum(a_i) and B = sum(b_i).
    # S_k = (sum_{i=1}^k a_i) + (sum_{i=k}^j b_i)
    # S_k = (sum_{i=1}^k a_i) + (sum_{i=1}^j b_i - sum_{i=1}^{k-1} b_i)
    # S_k = B + a_k + sum_{i=1}^{k-1} (a_i - b_i)
    
    # To minimize the maximum strain, we should sort the rituals.
    # Consider two adjacent rituals (a_i, b_i) and (a_{i+1}, b_{i+1}).
    # If we swap them, the strains at other positions remain unchanged.
    # The strains at i and i+1 change.
    # Let P = sum_{m=1}^{i-1} a_m and Q = sum_{m=i+2}^j b_m.
    # Original:
    # S_i = P + a_i + (b_i + b_{i+1} + Q)
    # S_{i+1} = P + a_i + a_{i+1} + (b_{i+1} + Q)
    # Max(S_i, S_{i+1}) = P + Q + max(a_i + b_i + b_{i+1}, a_i + a_{i+1} + b_{i+1})
    # = P + Q + b_{i+1} + max(a_i + b_i, a_i + a_{i+1})
    
    # Swapped:
    # S'_i = P + a_{i+1} + (b_{i+1} + b_i + Q)
    # S'_{i+1} = P + a_{i+1} + a_i + (b_i + Q)
    # Max(S'_i, S'_{i+1}) = P + Q + b_i + max(a_{i+1} + b_{i+1}, a_{i+1} + a_i)
    
    # We prefer the original if:
    # b_{i+1} + max(a_i + b_i, a_i + a_{i+1}) < b_i + max(a_{i+1} + b_{i+1}, a_{i+1} + a_i)
    # This is a standard sorting condition. Sorting by a_i < b_i works.
    # Specifically, sort by a_i < b_i ascending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    rituals = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        rituals.append((a, b))
        idx += 2
        
    # Sort based on the condition derived from the exchange argument
    # Sort by a_i < b_i: those with a_i < b_i come first (sorted by a_i),
    # those with a_i >= b_i come last (sorted by b_i descending).
    rituals.sort(key=lambda x: (x[0] >= x[1], x[0] if x[0] < x[1] else -x[1]))
    
    prefix_a = [0] * (j + 1)
    for i in range(j):
        prefix_a[i+1] = prefix_a[i] + rituals[i][0]
        
    suffix_b = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suffix_b[i] = suffix_b[i+1] + rituals[i][1]
        
    max_strain = 0
    for k in range(j):
        # Focal point at k (0-indexed)
        # Strain = sum(a_0...a_k) + sum(b_k...b_{j-1})
        strain = prefix_a[k+1] + suffix_b[k]
        if strain > max_strain:
            max_strain = strain
            
    print(max_strain)

solve()
```
