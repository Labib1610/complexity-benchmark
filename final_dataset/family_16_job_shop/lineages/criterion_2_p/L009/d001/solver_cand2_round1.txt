```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    effects = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        effects.append((p1, p2))
        idx += 2
    
    # Let the sequence be (a_1, b_1), (a_2, b_2), ..., (a_j, b_j).
    # The delay at transition k is: sum_{i=1}^k a_i + sum_{i=k}^j b_i.
    # This is equal to (sum_{i=1}^j b_i) + sum_{i=1}^k (a_i - b_i) + b_k.
    # To minimize the maximum of these values, we should sort the effects.
    # Consider two adjacent effects (a1, b1) and (a2, b2).
    # If we put (a1, b1) then (a2, b2), the max delay is affected by 
    # the transition points at these two positions.
    # This is a classic scheduling problem equivalent to Johnson's Rule.
    # Sort by min(p1, p2). If p1 < p2, put at the beginning (ascending p1).
    # If p2 <= p1, put at the end (descending p2).
    
    group1 = []
    group2 = []
    for p1, p2 in effects:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    seq = group1 + group2
    
    # Calculate the max delay for the optimal sequence
    # Delay(k) = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # Let S_a[k] be prefix sum of a, S_b[k] be suffix sum of b
    # Delay(k) = S_a[k] + S_b[k]
    
    n = len(seq)
    a = [x[0] for x in seq]
    b = [x[1] for x in seq]
    
    prefix_a = [0] * (n + 1)
    for i in range(n):
        prefix_a[i+1] = prefix_a[i] + a[i]
        
    suffix_b = [0] * (n + 1)
    for i in range(n - 1, -1, -1):
        suffix_b[i] = suffix_b[i+1] + b[i]
        
    max_delay = 0
    for k in range(1, n + 1):
        current_delay = prefix_a[k] + suffix_b[k-1]
        if current_delay > max_delay:
            max_delay = current_delay
            
    print(max_delay)

if __name__ == '__main__':
    solve()
```
