```python
import sys

def solve():
    # Let the effects be (g_i, l_i).
    # For a sequence of effects, the total time for split point k is:
    # T(k) = sum_{i=1}^k g_i + sum_{i=k}^j l_i
    # T(k) = sum_{i=1}^k g_i + sum_{i=k}^j l_i
    # Note that sum_{i=k}^j l_i = (sum_{i=1}^j l_i) - (sum_{i=1}^{k-1} l_i)
    # So T(k) = sum_{i=1}^k g_i - sum_{i=1}^{k-1} l_i + sum_{i=1}^j l_i
    # Let S_L = sum_{i=1}^j l_i.
    # T(k) = S_L + sum_{i=1}^{k-1} (g_i - l_i) + g_k
    # We want to minimize max_k T(k).
    # This is a classic scheduling problem. If we sort effects such that
    # those with g_i < l_i come first (sorted by g_i ascending),
    # and those with g_i >= l_i come later (sorted by l_i descending),
    # we minimize the maximum prefix sum.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    effects = []
    idx = 1
    for _ in range(j):
        g = int(input_data[idx])
        l = int(input_data[idx+1])
        effects.append((g, l))
        idx += 2
        
    # Johnson's rule logic:
    # Group 1: g_i < l_i, sort by g_i ascending
    # Group 2: g_i >= l_i, sort by l_i descending
    group1 = [e for e in effects if e[0] < e[1]]
    group2 = [e for e in effects if e[0] >= e[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate max T(k) for this sequence
    # T(k) = sum_{i=1}^k g_i + sum_{i=k}^j l_i
    # Let G[k] be prefix sum of g, L[k] be suffix sum of l
    # T(k) = G[k] + L[k]
    
    g_vals = [e[0] for e in sequence]
    l_vals = [e[1] for e in sequence]
    
    n = len(sequence)
    prefix_g = [0] * (n + 1)
    for i in range(n):
        prefix_g[i+1] = prefix_g[i] + g_vals[i]
        
    suffix_l = [0] * (n + 1)
    for i in range(n - 1, -1, -1):
        suffix_l[i] = suffix_l[i+1] + l_vals[i]
        
    max_t = 0
    for k in range(1, n + 1):
        # k is 1-indexed split point
        # sum_{i=1}^k g_i is prefix_g[k]
        # sum_{i=k}^j l_i is suffix_l[k-1]
        current_t = prefix_g[k] + suffix_l[k-1]
        if current_t > max_t:
            max_t = current_t
            
    print(max_t)

if __name__ == '__main__':
    solve()
```
