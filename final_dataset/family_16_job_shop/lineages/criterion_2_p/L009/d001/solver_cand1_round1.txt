import sys

def solve():
    # The problem asks to minimize the maximum of:
    # S_k = (sum_{i=1}^k p1_i) + (sum_{i=k}^j p2_i)
    # Let P1_sum = sum_{i=1}^j p1_i and P2_sum = sum_{i=1}^j p2_i.
    # S_k = (sum_{i=1}^k p1_i) + (sum_{i=1}^j p2_i - sum_{i=1}^{k-1} p2_i)
    # S_k = P2_sum + sum_{i=1}^k p1_i - sum_{i=1}^{k-1} p2_i
    # Let d_i = p1_i - p2_i.
    # S_k = P2_sum + p1_k + sum_{i=1}^{k-1} (p1_i - p2_i)
    # S_k = P2_sum + p1_k + sum_{i=1}^{k-1} d_i
    
    # To minimize the maximum S_k, we should sort the effects.
    # If we have two adjacent effects (p1_a, p2_a) and (p1_b, p2_b),
    # swapping them changes the S_k values for k=a and k=b.
    # Standard scheduling theory (Johnson's rule logic) suggests sorting by
    # min(p1, p2). Specifically, effects with p1 < p2 should come first
    # sorted by p1 ascending, and effects with p1 >= p2 should come last
    # sorted by p2 descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    effects = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        effects.append((p1, p2))
        idx += 2
        
    group1 = [e for e in effects if e[0] < e[1]]
    group2 = [e for e in effects if e[0] >= e[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_effects = group1 + group2
    
    # Calculate max S_k
    # S_k = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # Let prefix_p1[k] = sum_{i=1}^k p1_i
    # Let suffix_p2[k] = sum_{i=k}^j p2_i
    
    p1_arr = [e[0] for e in sorted_effects]
    p2_arr = [e[1] for e in sorted_effects]
    
    prefix_p1 = [0] * (j + 1)
    for i in range(j):
        prefix_p1[i+1] = prefix_p1[i] + p1_arr[i]
        
    suffix_p2 = [0] * (j + 2)
    for i in range(j, 0, -1):
        suffix_p2[i] = suffix_p2[i+1] + p2_arr[i-1]
        
    max_time = 0
    for k in range(1, j + 1):
        current_time = prefix_p1[k] + suffix_p2[k]
        if current_time > max_time:
            max_time = current_time
            
    print(max_time)

if __name__ == '__main__':
    solve()
