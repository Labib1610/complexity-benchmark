We can solve this by sorting the effects in decreasing order of their tracer time, then simulating the pipeline.  

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    j = int(data[0])
    effects = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx+1])
        idx += 2
        effects.append((p1, p2))
    
    # Sort by tracer time descending
    effects.sort(key=lambda x: x[1], reverse=True)
    
    solver_time = 0
    tracer_time = 0
    for p1, p2 in effects:
        solver_time += p1
        if tracer_time < solver_time:
            tracer_time = solver_time
        tracer_time += p2
    
    print(tracer_time)

if __name__ == "__main__":
    main()
```
