```python
import sys

def solve():
    # The cost for a pivot at position k is:
    # Sum_{i=1}^k p1_i + Sum_{i=k}^j p2_i
    # Let S1_k = Sum_{i=1}^k p1_i and S2_k = Sum_{i=k}^j p2_i
    # Cost_k = S1_k + S2_k
    # Note that S1_k + S2_k = (Sum_{i=1}^j p1_i) + (Sum_{i=1}^j p2_i) - (Sum_{i=1}^{k-1} p1_i) + (Sum_{i=k+1}^j p2_i)
    # Actually, a simpler way:
    # Cost_k = (Sum_{i=1}^{k-1} p1_i) + p1_k + p2_k + (Sum_{i=k+1}^j p2_i)
    # Let P = Sum_{i=1}^j p1_i and Q = Sum_{i=1}^j p2_i.
    # Cost_k = (Sum_{i=1}^k p1_i) + (Sum_{i=k}^j p2_i)
    # Let's sort the blocks. If we have two adjacent blocks (a1, a2) and (b1, b2):
    # Order (a, b):
    # k=1: a1 + (a2 + b2)
    # k=2: (a1 + b1) + b2
    # Max is max(a1+a2+b2, a1+b1+b2) = a1+b2 + max(a2, b1)
    # Order (b, a):
    # k=1: b1 + (b2 + a2)
    # k=2: (b1 + a1) + a2
    # Max is max(b1+b2+a2, b1+a1+a2) = b1+a2 + max(b2, a1)
    # We want to minimize the max. This is a classic exchange argument.
    # Sorting by p1 < p2 puts blocks with smaller p1 first and larger p2 last.
    # Specifically, sort by p1 - p2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    blocks = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        blocks.append((p1, p2))
        idx += 2
    
    # Sort such that p1 < p2 blocks come first, p1 > p2 blocks come last.
    # The condition derived from the exchange argument is to sort by p1 - p2.
    blocks.sort(key=lambda x: x[0] - x[1])
    
    # Calculate prefix sums of p1 and suffix sums of p2
    p1_pref = [0] * (j + 1)
    for i in range(j):
        p1_pref[i+1] = p1_pref[i] + blocks[i][0]
        
    p2_suff = [0] * (j + 2)
    for i in range(j, 0, -1):
        p2_suff[i] = p2_suff[i+1] + blocks[i-1][1]
        
    ans = float('inf')
    for k in range(1, j + 1):
        # Cost_k = Sum_{i=1}^k p1_i + Sum_{i=k}^j p2_i
        cost = p1_pref[k] + p2_suff[k]
        if cost < ans:
            ans = cost
            
    print(ans)

if __name__ == '__main__':
    solve()
```
