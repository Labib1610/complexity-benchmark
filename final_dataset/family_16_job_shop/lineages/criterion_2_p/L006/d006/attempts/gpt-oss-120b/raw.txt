```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    inc = []   # p2 >= p1
    dec = []   # p2 < p1
    sum_p1 = 0
    for _ in range(n):
        p1 = int(next(it)); p2 = int(next(it))
        sum_p1 += p1
        if p2 >= p1:
            inc.append((p2, p1))
        else:
            dec.append((p1, p2))
    inc.sort(key=lambda x: x[0])          # increasing p2
    dec.sort(key=lambda x: -x[0])         # decreasing p1
    V = 0
    for p2, p1 in inc:
        V = max(p2, V - p1 + p2)
    for p1, p2 in dec:
        V = max(p2, V - p1 + p2)
    print(V + sum_p1)

if __name__ == "__main__":
    solve()
```
