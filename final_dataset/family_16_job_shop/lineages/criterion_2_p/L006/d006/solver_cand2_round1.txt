```python
import sys

def solve():
    # The problem asks to arrange collections (p1_i, p2_i) in rooms 1 to j
    # to minimize max_{k=1..j} (sum_{i=1}^k p1_i + sum_{i=k}^j p2_i).
    # Let R_i = p1_i and B_i = p2_i.
    # The expression is sum_{i=1}^j p1_i + sum_{i=1}^j p2_i - (p1_k - p2_k) is not quite right.
    # Let S1 = sum_{i=1}^j p1_i and S2 = sum_{i=1}^j p2_i.
    # The total viewed for transition k is:
    # (sum_{i=1}^k p1_i) + (sum_{i=k}^j p2_i)
    # = (sum_{i=1}^k p1_i) + (sum_{i=1}^j p2_i - sum_{i=1}^{k-1} p2_i)
    # = S2 + sum_{i=1}^k (p1_i - p2_{i-1}) where p2_0 = 0.
    # Actually, let's re-examine:
    # Total(k) = (p1_1 + ... + p1_k) + (p2_k + ... + p2_j)
    # Total(k) = (p1_1 + ... + p1_k) + (S2 - (p2_1 + ... + p2_{k-1}))
    # Total(k) = S2 + p1_1 + (p1_2 - p2_1) + (p1_3 - p2_2) + ... + (p1_k - p2_{k-1})
    # To minimize the maximum of these, we should sort the collections.
    # Let d_i = p1_i - p2_i.
    # If we sort by d_i, we can minimize the peak.
    # Specifically, sorting by p1_i + p2_i descending or similar?
    # Actually, the optimal strategy is to sort by p1_i + p2_i.
    # Let's test: if we sort such that p1_i + p2_i is constant, it doesn't matter.
    # If we sort by p1_i + p2_i descending, we minimize the maximum.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    collections = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        collections.append((p1, p2))
        idx += 2
    
    # Sort by p1 + p2 descending
    collections.sort(key=lambda x: x[0] + x[1], reverse=True)
    
    p1_arr = [c[0] for c in collections]
    p2_arr = [c[1] for c in collections]
    
    prefix_p1 = [0] * (j + 1)
    for i in range(j):
        prefix_p1[i+1] = prefix_p1[i] + p1_arr[i]
        
    suffix_p2 = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suffix_p2[i] = suffix_p2[i+1] + p2_arr[i]
        
    max_score = 0
    for k in range(1, j + 1):
        # Transition room k (1-indexed)
        # Red: 1 to k, Blue: k to j
        score = prefix_p1[k] + suffix_p2[k-1]
        if score > max_score:
            max_score = score
            
    print(max_score)

solve()
```
