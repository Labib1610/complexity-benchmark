import sys

def solve():
    # The problem asks to minimize the peak operational load.
    # Let the sequence be (p1_1, p2_1), (p1_2, p2_2), ..., (p1_j, p2_j).
    # The load at split point k is:
    # L_k = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # L_k = sum_{i=1}^j p2_i + sum_{i=1}^k (p1_i - p2_i) + p2_k
    # Let S = sum_{i=1}^j p2_i.
    # L_k = S + sum_{i=1}^{k-1} (p1_i - p2_i) + p1_k
    
    # To minimize the maximum of these values, we should sort the displays.
    # Consider two adjacent displays i and i+1.
    # If we swap them, the values of L_k for k != i, i+1 remain unchanged.
    # For k=i, the load changes.
    # This is a classic scheduling problem. The optimal order is to sort by
    # (p1 - p2). Specifically, if p1 < p2, put them at the beginning,
    # if p1 >= p2, put them at the end.
    # Within p1 < p2, sort by p1 ascending.
    # Within p1 >= p2, sort by p2 descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    displays = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        displays.append((p1, p2))
        idx += 2
        
    # Group 1: p1 < p2, sort by p1 ascending
    # Group 2: p1 >= p2, sort by p2 descending
    g1 = [d for d in displays if d[0] < d[1]]
    g2 = [d for d in displays if d[0] >= d[1]]
    
    g1.sort(key=lambda x: x[0])
    g2.sort(key=lambda x: x[1], reverse=True)
    
    order = g1 + g2
    
    # Calculate peak operational load
    # L_k = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # Let prefix_p1[k] = sum_{i=1}^k p1_i
    # Let suffix_p2[k] = sum_{i=k}^j p2_i
    
    n = len(order)
    p1s = [d[0] for d in order]
    p2s = [d[1] for d in order]
    
    prefix_p1 = [0] * (n + 1)
    for i in range(n):
        prefix_p1[i+1] = prefix_p1[i] + p1s[i]
        
    suffix_p2 = [0] * (n + 2)
    for i in range(n, 0, -1):
        suffix_p2[i] = suffix_p2[i+1] + p2s[i-1]
        
    max_load = 0
    for k in range(1, n + 1):
        current_load = prefix_p1[k] + suffix_p2[k]
        if current_load > max_load:
            max_load = current_load
            
    print(max_load)

if __name__ == '__main__':
    solve()
