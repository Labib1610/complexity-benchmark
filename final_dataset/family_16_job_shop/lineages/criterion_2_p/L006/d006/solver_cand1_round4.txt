```python
import sys

def solve():
    # The formula for V' is V' = max(p2, V - p1 + p2).
    # Let S = sum of all p1.
    # The inefficiency rating is S + V_final.
    # Let's analyze the update: V_new = max(p2, V - p1 + p2).
    # This is equivalent to V_new = V - p1 + p2 if V - p1 + p2 >= p2 (i.e., V >= p1),
    # else V_new = p2.
    # Let's rewrite the update as V_new = max(p2, V - p1 + p2).
    # Let f_i(V) = max(p2_i, V - p1_i + p2_i).
    # We want to minimize S + f_n(f_{n-1}(...f_1(0)...)).
    # Note that f_i(V) = V - p1_i + max(p2_i, p1_i).
    # Actually, let's re-examine: V_new = max(p2, V - p1 + p2).
    # Let q_i = p2_i - p1_i. Then V_new = max(p2_i, V + q_i).
    # This is a standard form. If we have two stages (p1_a, p2_a) and (p1_b, p2_b),
    # order a then b: V_ab = max(p2_b, max(p2_a, V + q_a) + q_b) = max(p2_b, p2_a + q_b, V + q_a + q_b).
    # Order b then a: V_ba = max(p2_a, p2_b + q_a, V + q_a + q_b).
    # We prefer a then b if max(p2_b, p2_a + q_b) < max(p2_a, p2_b + q_a).
    # This is equivalent to sorting by p2_i - q_i = p2_i - (p2_i - p1_i) = p1_i.
    # Wait, let's check the condition:
    # p2_b < p2_a + q_a  => p2_b < p2_a + p2_a - p1_a (No)
    # Let's use the exchange argument:
    # Compare max(p2_b, p2_a + p2_b - p1_b) vs max(p2_a, p2_b + p2_a - p1_a).
    # This simplifies to sorting by p1_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    stages = []
    sum_p1 = 0
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        stages.append((p1, p2))
        sum_p1 += p1
        idx += 2
    
    # Sort by p1 ascending
    stages.sort(key=lambda x: x[0])
    
    v = 0
    for p1, p2 in stages:
        v = max(p2, v - p1 + p2)
        
    print(sum_p1 + v)

if __name__ == '__main__':
    solve()
```
