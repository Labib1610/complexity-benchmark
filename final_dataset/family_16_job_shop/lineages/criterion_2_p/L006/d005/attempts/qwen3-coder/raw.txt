```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    apps = []
    total_p1 = 0
    
    index = 1
    for i in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        index += 2
        apps.append((p1, p2))
        total_p1 += p1
    
    # Sort applications by (p2 - p1) in descending order
    # This prioritizes applications that generate more than they consume
    apps.sort(key=lambda x: x[1] - x[0], reverse=True)
    
    battery = 0
    for p1, p2 in apps:
        # Consume p1 units
        if battery >= p1:
            battery -= p1
        else:
            # Deficit is covered by grid, battery goes to 0
            battery = 0
        # Generate p2 units
        battery += p2
    
    print(total_p1 + battery)

if __name__ == "__main__":
    main()
```
