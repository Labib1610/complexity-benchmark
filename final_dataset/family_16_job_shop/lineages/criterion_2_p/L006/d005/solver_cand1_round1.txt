import sys

def solve():
    # The problem asks to minimize sum(p1_i) + D_j.
    # Since sum(p1_i) is constant regardless of the permutation,
    # we only need to minimize D_j.
    # D_i = max(0, D_{i-1} - p1_i) + p2_i
    # Let's analyze the order.
    # If we have two components (p1_a, p2_a) and (p1_b, p2_b):
    # Order (a, b): D_2 = max(0, max(0, 0 - p1_a) + p2_a - p1_b) + p2_b
    #               D_2 = max(0, p2_a - p1_b) + p2_b
    # Order (b, a): D_2 = max(0, p2_b - p1_a) + p2_a
    # We want to choose the order that minimizes D_2.
    # This is a classic scheduling problem.
    # Components with p1 >= p2 should come first (sorted by p1 ascending),
    # and components with p1 < p2 should come last (sorted by p2 descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    components = []
    sum_p1 = 0
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        components.append((p1, p2))
        sum_p1 += p1
        idx += 2
        
    group1 = [] # p1 >= p2
    group2 = [] # p1 < p2
    
    for p1, p2 in components:
        if p1 >= p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    # Sort group1 by p1 ascending
    group1.sort(key=lambda x: x[0])
    # Sort group2 by p2 descending
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    d = 0
    for p1, p2 in order:
        d = max(0, d - p1) + p2
        
    print(sum_p1 + d)

if __name__ == '__main__':
    solve()
