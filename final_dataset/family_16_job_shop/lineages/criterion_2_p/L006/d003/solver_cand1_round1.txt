```python
import sys

def solve():
    # The problem is equivalent to Johnson's Rule for the two-machine flow shop scheduling problem.
    # We want to minimize the makespan (total time).
    # Let p1 be the time on machine 1 (entry) and p2 be the time on machine 2 (landing).
    # Johnson's Rule:
    # 1. List all jobs.
    # 2. Divide jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 3. Sort Set A in ascending order of p1.
    # 4. Sort Set B in descending order of p2.
    # 5. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate makespan
    # time_m1: completion time on machine 1
    # time_m2: completion time on machine 2
    time_m1 = 0
    time_m2 = 0
    
    for p1, p2 in sequence:
        time_m1 += p1
        # Machine 2 can start only after machine 1 finishes current job
        # AND machine 2 finishes previous job
        if time_m2 < time_m1:
            time_m2 = time_m1 + p2
        else:
            time_m2 += p2
            
    print(time_m2)

if __name__ == '__main__':
    solve()
```
