import sys

def solve():
    # The initial temperature is S = sum(p2_i).
    # When processing rod i, the temperature changes from T to T + p1_i, then to T + p1_i - p2_i.
    # The peak temperature for rod i is T_prev + p1_i.
    # Let the sequence of rods be i_1, i_2, ..., i_j.
    # T_0 = S
    # T_1 = T_0 - p2_{i_1} = S - p2_{i_1}
    # T_2 = T_1 - p2_{i_2} = S - p2_{i_1} - p2_{i_2}
    # ...
    # T_k = S - sum_{m=1}^k p2_{i_m}
    # The peak at step k is T_{k-1} + p1_{i_k} = S - sum_{m=1}^{k-1} p2_{i_m} + p1_{i_k}.
    # We want to minimize max_{k=1..j} (S - sum_{m=1}^{k-1} p2_{i_m} + p1_{i_k}).
    # This is equivalent to minimizing max_{k=1..j} (S + p1_{i_k} - sum_{m=1}^{k-1} p2_{i_m}).
    # Let diff_i = p1_i - p2_i.
    # Note that S + p1_{i_k} - sum_{m=1}^{k-1} p2_{i_m} = S + p1_{i_k} - p2_{i_k} + p2_{i_k} - sum_{m=1}^{k-1} p2_{i_m}
    # This looks like we should process rods with p1_i >= p2_i first, then p1_i < p2_i.
    # Actually, for rods with p1_i >= p2_i, we want to process them in increasing order of p2_i.
    # For rods with p1_i < p2_i, we want to process them in decreasing order of p1_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    rods = []
    idx = 1
    s = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        rods.append((p1, p2))
        s += p2
        idx += 2
        
    pos = [r for r in rods if r[0] >= r[1]]
    neg = [r for r in rods if r[0] < r[1]]
    
    # Sort pos by p2 ascending
    pos.sort(key=lambda x: x[1])
    # Sort neg by p1 descending
    neg.sort(key=lambda x: x[0], reverse=True)
    
    sequence = pos + neg
    
    max_temp = 0
    current_temp = s
    for p1, p2 in sequence:
        peak = current_temp + p1
        if peak > max_temp:
            max_temp = peak
        current_temp -= p2
        
    print(max_temp)

if __name__ == '__main__':
    solve()
