```python
import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        u = int(input_data[idx])
        l = int(input_data[idx+1])
        tiles.append((u, l))
        idx += 2
    
    # Let the sequence of tiles be (u_1, l_1), (u_2, l_2), ..., (u_j, l_j).
    # The cost of moving down at tile k is:
    # C_k = sum_{i=1}^k u_i + sum_{i=k}^j l_i
    # C_k = sum_{i=1}^j u_i + sum_{i=1}^j l_i - sum_{i=1}^{k-1} u_i - sum_{i=k+1}^j l_i
    # Let S_U = sum u_i, S_L = sum l_i.
    # C_k = S_U + S_L - (sum_{i=1}^{k-1} u_i + sum_{i=k+1}^j l_i)
    # To minimize the maximum C_k, we want to maximize the minimum of (sum_{i=1}^{k-1} u_i + sum_{i=k+1}^j l_i).
    # Let x_i = u_i - l_i.
    # C_k = sum_{i=1}^j l_i + sum_{i=1}^k (u_i - l_i) = S_L + sum_{i=1}^k x_i.
    # We want to minimize max_{k=1..j} (S_L + sum_{i=1}^k x_i).
    # This is equivalent to minimizing the maximum prefix sum of the sequence x_1, x_2, ..., x_j.
    # To minimize the maximum prefix sum, we should sort x_i in descending order.
    
    x = [u - l for u, l in tiles]
    x.sort(reverse=True)
    
    s_l = sum(l for u, l in tiles)
    
    max_cost = -float('inf')
    current_prefix_sum = 0
    for val in x:
        current_prefix_sum += val
        max_cost = max(max_cost, s_l + current_prefix_sum)
        
    print(max_cost)

if __name__ == '__main__':
    solve()
```
