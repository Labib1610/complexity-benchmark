```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tiles.append((p1, p2))
        idx += 2
    
    # Let the tiles in the column be (L_1, R_1), (L_2, R_2), ..., (L_j, R_j).
    # A path crossing at tile k has cost:
    # sum(L_1...L_k) + sum(R_k...R_j)
    # = sum(L_1...L_j) + sum(R_1...R_j) - sum(L_{k+1}...L_j) - sum(R_1...R_{k-1})
    # Let S_L = sum(L_i) and S_R = sum(R_i).
    # Cost_k = S_L + S_R - (sum_{i=k+1}^j L_i + sum_{i=1}^{k-1} R_i)
    # To minimize the maximum cost, we want to maximize the minimum value of:
    # (sum_{i=k+1}^j L_i + sum_{i=1}^{k-1} R_i)
    # Let f(k) = sum_{i=k+1}^j L_i + sum_{i=1}^{k-1} R_i.
    # We want to arrange tiles to maximize min_{k=1..j} f(k).
    # This is equivalent to sorting tiles such that L_i - R_i is non-decreasing.
    # Specifically, sort by (L_i - R_i).
    
    tiles.sort(key=lambda x: x[0] - x[1])
    
    s_l = sum(t[0] for t in tiles)
    s_r = sum(t[1] for t in tiles)
    
    # f(k) = sum_{i=k+1}^j L_i + sum_{i=1}^{k-1} R_i
    # f(1) = sum_{i=2}^j L_i
    # f(k+1) = f(k) - L_{k+1} + R_k
    
    current_f = sum(t[0] for t in tiles[1:])
    min_f = current_f
    
    for k in range(1, j):
        # Move from k to k+1
        # f(k+1) = f(k) - L_{k+1} + R_k
        current_f = current_f - tiles[k][0] + tiles[k-1][1]
        if current_f < min_f:
            min_f = current_f
            
    print(s_l + s_r - min_f)

solve()
```
