import sys

def solve():
    # The problem asks for the minimum completion time of the last starship.
    # Let the starships be ordered 1, 2, ..., j.
    # Let p1_i and p2_i be the scan and exit times for starship i.
    # The scan for starship i finishes at S_i = sum_{k=1}^i p1_k.
    # The exit for starship i starts at max(S_i, E_{i-1}), where E_{i-1} is the finish time of the previous starship.
    # E_i = max(S_i, E_{i-1}) + p2_i.
    # This is equivalent to E_i = max(S_i + p2_i, E_{i-1} + p2_i).
    # Expanding this: E_j = max(S_1 + p2_1 + p2_2 + ... + p2_j, 
    #                           S_2 + p2_2 + p2_3 + ... + p2_j, 
    #                           ..., 
    #                           S_j + p2_j).
    # S_i = sum_{k=1}^i p1_k.
    # E_j = max_{1 <= i <= j} (sum_{k=1}^i p1_k + sum_{k=i}^j p2_k).
    # Let P1_i = sum_{k=1}^i p1_k and P2_i = sum_{k=1}^i p2_k.
    # E_j = max_{1 <= i <= j} (P1_i + P2_j - P2_{i-1}) where P2_0 = 0.
    # E_j = P2_j + max_{1 <= i <= j} (P1_i - P2_{i-1}).
    
    # To minimize E_j, we need to reorder the starships to minimize 
    # max_{1 <= i <= j} (P1_i - P2_{i-1}).
    # This is a classic scheduling problem. To minimize the maximum completion time,
    # we should sort the starships by their p2 values in descending order.
    # If p2_i > p2_{i+1}, swapping them reduces the maximum.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    ships = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        ships.append((p1, p2))
        idx += 2
    
    # Sort by p2 descending
    ships.sort(key=lambda x: x[1], reverse=True)
    
    p1_sum = 0
    p2_sum = 0
    max_val = -float('inf')
    
    # P1_i - P2_{i-1}
    # i=1: P1_1 - P2_0 = p1_1 - 0
    # i=2: P1_2 - P2_1 = (p1_1 + p1_2) - p2_1
    
    current_p1 = 0
    current_p2 = 0
    for p1, p2 in ships:
        # Before adding current ship, P2_{i-1} is current_p2
        # P1_i is current_p1 + p1
        val = (current_p1 + p1) - current_p2
        if val > max_val:
            max_val = val
        current_p1 += p1
        current_p2 += p2
        
    print(current_p2 + max_val)

if __name__ == '__main__':
    solve()
