import sys

def solve():
    # The problem is a variation of Johnson's Rule for the two-machine flow shop scheduling problem.
    # We want to minimize the makespan (the time the last job finishes on the second machine).
    # Johnson's Rule:
    # 1. List all jobs.
    # 2. Divide jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 3. Sort Set 1 in increasing order of p1.
    # 4. Sort Set 2 in decreasing order of p2.
    # 5. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate the completion time
    # time1: finish time on security scanner
    # time2: finish time on departure bay
    time1 = 0
    time2 = 0
    
    for p1, p2 in sequence:
        time1 += p1
        if time1 > time2:
            time2 = time1 + p2
        else:
            time2 += p2
            
    print(time2)

if __name__ == '__main__':
    solve()
