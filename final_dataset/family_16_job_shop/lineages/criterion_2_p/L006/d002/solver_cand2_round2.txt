import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (inner airlock and outer airlock) and j jobs.
    # Each job must go through the inner airlock first, then the outer airlock.
    # We want to minimize the makespan (the time the last job finishes).
    
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where inner_time < outer_time
    #    Set B: jobs where inner_time >= outer_time
    # 2. Sort Set A in ascending order of inner_time.
    # 3. Sort Set B in descending order of outer_time.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate makespan
    inner_time = 0
    outer_time = 0
    
    for p1, p2 in sequence:
        inner_time += p1
        # The outer airlock can start only after the inner airlock finishes
        # the current job AND the outer airlock finishes the previous job.
        if outer_time < inner_time:
            outer_time = inner_time + p2
        else:
            outer_time += p2
            
    print(outer_time)

if __name__ == '__main__':
    solve()
