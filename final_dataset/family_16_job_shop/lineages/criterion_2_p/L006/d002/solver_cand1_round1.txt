import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    ships = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        ships.append((p1, p2))
        idx += 2
    
    # Let the sequence be s_1, s_2, ..., s_j.
    # Let p1_i be atmospheric entry and p2_i be landing protocol.
    # Total time for switch point k:
    # T(k) = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # T(k) = sum_{i=1}^j p2_i + sum_{i=1}^k (p1_i - p2_i) + p2_k
    # Wait, the formula is: sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # T(k) = (sum_{i=1}^k p1_i) + (sum_{i=1}^j p2_i) - (sum_{i=1}^{k-1} p2_i)
    # T(k) = (sum_{i=1}^j p2_i) + sum_{i=1}^{k-1} (p1_i - p2_i) + p1_k
    
    # To minimize max_k T(k), we should sort the ships.
    # Consider two adjacent ships i and i+1.
    # If we swap them, the only T(k) values that change are at k=i.
    # T(i) = (sum_{1..i-1} p1) + p1_i + (sum_{i..j} p2)
    # T(i+1) = (sum_{1..i-1} p1) + p1_i + p1_{i+1} + (sum_{i+1..j} p2)
    # This is a classic scheduling problem. The optimal order is to sort by
    # p1_i < p2_i first, then p1_i >= p2_i.
    # Specifically, sort by p1_i - p2_i.
    
    ships.sort(key=lambda x: x[0] - x[1])
    
    # Now calculate the max T(k) for this sequence
    p1_sum = [0] * (j + 1)
    p2_sum = [0] * (j + 2)
    
    for i in range(j):
        p1_sum[i+1] = p1_sum[i] + ships[i][0]
    for i in range(j, 0, -1):
        p2_sum[i] = p2_sum[i+1] + ships[i-1][1]
        
    ans = 0
    for k in range(1, j + 1):
        current_t = p1_sum[k] + p2_sum[k]
        if current_t > ans:
            ans = current_t
            
    print(ans)

if __name__ == '__main__':
    solve()
