```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    caravans = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        caravans.append((p1, p2))
        idx += 2
    
    # Johnson's Rule logic:
    # To minimize the makespan of a two-machine flow shop,
    # sort caravans such that those with smaller p1 come first,
    # and those with smaller p2 come last.
    # Specifically:
    # Group 1: p1 < p2, sort by p1 ascending
    # Group 2: p1 >= p2, sort by p2 descending
    
    group1 = []
    group2 = []
    for p1, p2 in caravans:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate completion time
    # current_sand_time: time when the current caravan finishes Sand Sea
    # current_frost_time: time when the current caravan finishes Frost Ridge
    current_sand_time = 0
    current_frost_time = 0
    
    for p1, p2 in sequence:
        current_sand_time += p1
        # The caravan can start Frost Ridge at max(current_sand_time, current_frost_time)
        if current_sand_time > current_frost_time:
            current_frost_time = current_sand_time + p2
        else:
            current_frost_time += p2
            
    print(current_frost_time)

if __name__ == '__main__':
    solve()
```
