import sys

def solve():
    # The problem asks to minimize the completion time of the last vessel
    # given that all vessels must pass through Inner Airlock (I) then Outer Airlock (O)
    # in the same order.
    # Let the order be 1, 2, ..., j.
    # Let I_i and O_i be the times for vessel i.
    # Finish time of vessel i at I: F_I(i) = sum_{k=1}^i I_k
    # Finish time of vessel i at O: F_O(i) = max(F_O(i-1), F_I(i)) + O_i
    # This is a classic scheduling problem (Johnson's Rule variant).
    # To minimize the completion time of the last job in a two-machine flow shop,
    # we sort jobs such that those with smaller I_i come first, and those with
    # smaller O_i come last. Specifically, sort by min(I_i, O_i).
    # Actually, for the two-machine flow shop problem (F2||Cmax), Johnson's rule
    # states:
    # 1. List all I_i and O_i.
    # 2. Find the minimum value.
    # 3. If it's an I_i, put the job at the beginning of the sequence.
    # 4. If it's an O_i, put the job at the end of the sequence.
    # 5. Remove the job and repeat.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Johnson's Rule implementation
    group1 = [] # I_i < O_i
    group2 = [] # I_i >= O_i
    for i in range(j):
        if jobs[i][0] < jobs[i][1]:
            group1.append(jobs[i])
        else:
            group2.append(jobs[i])
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_i_time = 0
    current_o_time = 0
    
    for i_time, o_time in order:
        current_i_time += i_time
        current_o_time = max(current_o_time, current_i_time) + o_time
        
    print(current_o_time)

if __name__ == '__main__':
    solve()
