import sys

def solve():
    # The problem is a variation of the flow-shop scheduling problem (Johnson's Rule).
    # We have j jobs, each with two tasks: transcription (p1) and binding (p2).
    # The order of jobs is the same for both machines.
    # Let the sequence be s_1, s_2, ..., s_j.
    # Let C1(i) be the completion time of transcription for job i.
    # Let C2(i) be the completion time of binding for job i.
    # C1(i) = sum_{k=1}^i p1_k
    # C2(i) = max(C2(i-1), C1(i)) + p2_i
    # This is equivalent to minimizing the makespan of a 2-machine flow shop.
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 2. Sort Set A in increasing order of p1.
    # 3. Sort Set B in decreasing order of p2.
    # 4. The optimal sequence is Set A followed by Set B.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    c1 = 0
    c2 = 0
    for p1, p2 in sequence:
        c1 += p1
        if c1 > c2:
            c2 = c1 + p2
        else:
            c2 += p2
            
    print(c2)

if __name__ == '__main__':
    solve()
