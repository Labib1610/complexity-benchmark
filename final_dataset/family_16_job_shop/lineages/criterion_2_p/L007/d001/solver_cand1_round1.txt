```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have j jobs, each with two tasks: transcription (p1) and binding (p2).
    # The goal is to minimize the makespan (total time until the last job is bound).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 2. Sort Set A in ascending order of p1.
    # 3. Sort Set B in descending order of p2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    # Sort A by p1 ascending
    set_a.sort(key=lambda x: x[0])
    # Sort B by p2 descending
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate completion time
    # current_p1_finish: time when calligrapher finishes current job
    # current_p2_finish: time when bookbinder finishes current job
    current_p1_finish = 0
    current_p2_finish = 0
    
    for p1, p2 in sequence:
        current_p1_finish += p1
        # Bookbinder can start only after calligrapher finishes current job
        # and after bookbinder finishes previous job
        current_p2_finish = max(current_p1_finish, current_p2_finish) + p2
        
    print(current_p2_finish)

if __name__ == '__main__':
    solve()
```
