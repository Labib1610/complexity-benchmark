import sys

def solve():
    # The problem asks to minimize the completion time of the bookbinder.
    # Let p1_i be the transcription time and p2_i be the binding time for manuscript i.
    # If the calligrapher processes manuscripts in order sigma and the bookbinder
    # processes them in order tau, the completion time is:
    # C = max_{k=1..j} (sum_{i=1..k} p1_{sigma(i)} + sum_{i=k..j} p2_{tau(i)})
    # This is a classic scheduling problem. To minimize the maximum completion time,
    # the optimal strategy is to have both artisans process the manuscripts in the
    # same order, specifically the order that minimizes the makespan of a 2-machine
    # flow shop (Johnson's Rule).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: p1_i < p2_i
    #    Set B: p1_i >= p2_i
    # 2. Sort Set A in ascending order of p1_i.
    # 3. Sort Set B in descending order of p2_i.
    # 4. The optimal order is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    order = set_a + set_b
    
    # Calculate completion time
    # Calligrapher finishes i-th job at time T1_i = sum(p1_1...p1_i)
    # Bookbinder finishes i-th job at time T2_i = max(T1_i, T2_{i-1}) + p2_i
    
    t1 = 0
    t2 = 0
    for p1, p2 in order:
        t1 += p1
        if t1 > t2:
            t2 = t1 + p2
        else:
            t2 = t2 + p2
            
    print(t2)

if __name__ == '__main__':
    solve()
