```python
import sys

def solve():
    # The problem is a variation of Johnson's Rule for the two-machine flow shop scheduling problem.
    # We have j jobs, each with two tasks:
    # Task 1 (Linguist): duration p1
    # Task 2 (Preservationist): duration p2
    # The constraint is that the order must be the same for both machines.
    # Johnson's Rule states that to minimize the makespan:
    # 1. List all jobs with their p1 and p2 values.
    # 2. Divide jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 3. Sort Set A in ascending order of p1.
    # 4. Sort Set B in descending order of p2.
    # 5. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = [job for job in jobs if job[0] < job[1]]
    set_b = [job for job in jobs if job[0] >= job[1]]
    
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    order = set_a + set_b
    
    # Calculate completion time
    # linguist_time: time when linguist finishes current job
    # preservationist_time: time when preservationist finishes current job
    linguist_time = 0
    preservationist_time = 0
    
    for p1, p2 in order:
        linguist_time += p1
        # Preservationist can start only after linguist finishes current job
        # and after preservationist finishes previous job
        preservationist_time = max(linguist_time, preservationist_time) + p2
        
    print(preservationist_time)

if __name__ == '__main__':
    solve()
```
