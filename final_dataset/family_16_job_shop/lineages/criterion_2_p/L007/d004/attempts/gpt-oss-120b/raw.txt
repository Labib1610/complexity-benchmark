import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    items = []
    for _ in range(n):
        p1 = int(next(it))
        p2 = int(next(it))
        items.append((p1, p2, p1 - p2))
    # sort by (p1 - p2) descending
    items.sort(key=lambda x: x[2], reverse=True)

    pref = [0] * (n + 1)   # pref[i]: sum of p1 of first i items
    for i in range(1, n + 1):
        pref[i] = pref[i - 1] + items[i - 1][0]

    suf = [0] * (n + 1)    # suf[i]: sum of p2 from i to n-1
    for i in range(n - 1, -1, -1):
        suf[i] = suf[i + 1] + items[i][1]

    ans = 0
    for k in range(1, n + 1):
        # transition at k-th manuscript (1‑based)
        total = pref[k] + suf[k - 1]
        if total > ans:
            ans = total
    print(ans)

if __name__ == "__main__":
    solve()
