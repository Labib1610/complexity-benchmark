import sys
from functools import cmp_to_key

def compare_jobs(job1, job2):
    # job = (p1, p2, original_index)
    # This is Johnson's rule for scheduling
    # Compare p1 + min(p2_first, p1_second) vs p2 + min(p1_first, p2_second)
    # But more simply, we can use the standard comparison:
    # if job1.p1 <= job1.p2, job1 should go before job2 if job1.p2 < job2.p1
    # if job1.p1 > job1.p2, job1 should go after job2 if job2.p1 >= job2.p2
    
    p1_1, p2_1 = job1[0], job1[1]
    p1_2, p2_2 = job2[0], job2[1]
    
    # Johnson's algorithm sorting rule
    if p1_1 <= p2_1:  # job1 is "short" in first phase
        if p1_2 > p2_2:  # job2 is "long" in first phase
            return -1  # job1 comes first
        else:  # both are "short" in first phase, sort by p2 ascending
            if p2_1 < p2_2:
                return -1
            elif p2_1 > p2_2:
                return 1
            else:
                return 0
    else:  # job1 is "long" in first phase
        if p1_2 <= p2_2:  # job2 is "short" in first phase
            return 1  # job2 comes first
        else:  # both are "long" in first phase, sort by p1 ascending
            if p1_1 < p1_2:
                return -1
            elif p1_1 > p1_2:
                return 1
            else:
                return 0

def main():
    j = int(sys.stdin.readline())
    jobs = []
    
    for i in range(j):
        p1, p2 = map(int, sys.stdin.readline().split())
        jobs.append((p1, p2))
    
    # Create jobs with indices for stable sort if needed
    indexed_jobs = [(jobs[i][0], jobs[i][1], i) for i in range(j)]
    
    # Sort using Johnson's rule
    def johnson_compare(x, y):
        p1_x, p2_x = x[0], x[1]
        p1_y, p2_y = y[0], y[1]
        
        # If x is in group 1 (p1 <= p2) and y is in group 2 (p1 > p2), x first
        if p1_x <= p2_x and p1_y > p2_y:
            return -1
        if p1_x > p2_x and p1_y <= p2_y:
            return 1
        
        # If both in group 1, sort by p1 ascending
        if p1_x <= p2_x and p1_y <= p2_y:
            if p1_x < p1_y:
                return -1
            elif p1_x > p1_y:
                return 1
            else:
                return 0
                
        # If both in group 2, sort by p2 descending
        if p1_x > p2_x and p1_y > p2_y:
            if p2_x > p2_y:
                return -1
            elif p2_x < p2_y:
                return 1
            else:
                return 0
        
        return 0
    
    sorted_jobs = sorted(indexed_jobs, key=cmp_to_key(johnson_compare))
    
    # Calculate the finish time
    time_calligrapher = 0
    time_bookbinder = 0
    
    for job in sorted_jobs:
        p1, p2 = job[0], job[1]
        time_calligrapher += p1
        time_bookbinder = max(time_bookbinder, time_calligrapher) + p2
    
    print(time_bookbinder)

if __name__ == "__main__":
    main()
