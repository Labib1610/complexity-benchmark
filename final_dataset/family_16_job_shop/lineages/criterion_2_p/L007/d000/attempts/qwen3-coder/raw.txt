```python
import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    j = int(sys.stdin.readline())
    manuscripts = []
    for i in range(j):
        p1, p2 = map(int, sys.stdin.readline().split())
        manuscripts.append((p1, p2))
    
    # Sort using Johnson's rule: 
    # - If p1 <= p2, sort by p1 ascending (calligrapher time)
    # - If p1 > p2, sort by p2 descending (bookbinder time)
    # But we can simplify this:
    # Split into two groups:
    # Group 1: p1 <= p2, sort by p1 ascending
    # Group 2: p1 > p2, sort by p2 descending
    # Then sequence is Group1 + Group2 (reversed)
    
    group1 = []  # p1 <= p2
    group2 = []  # p1 > p2
    
    for i in range(j):
        p1, p2 = manuscripts[i]
        if p1 <= p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
    
    # Sort group1 by p1 ascending
    group1.sort(key=lambda x: x[0])
    # Sort group2 by p2 descending
    group2.sort(key=lambda x: x[1], reverse=True)
    
    # Final sequence: group1 + group2
    sequence = group1 + group2
    
    # Now simulate the processing
    calligrapher_time = 0
    bookbinder_time = 0
    
    for p1, p2 in sequence:
        calligrapher_time += p1
        bookbinder_time = max(bookbinder_time, calligrapher_time) + p2
    
    print(bookbinder_time)

threading.Thread(target=main).start()
```
