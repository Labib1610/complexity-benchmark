import sys

def solve():
    # Let the performances be (a_i, b_i) where a_i is Main Stage time, b_i is Review Room time.
    # Let the order be p_1, p_2, ..., p_j.
    # Let S_i be the time the i-th performance finishes its Main Stage act.
    # S_i = sum_{k=1}^i a_{p_k}
    # Let R_i be the time the i-th performance finishes its Review Room session.
    # R_0 = 0
    # R_i = max(S_i, R_{i-1}) + b_{p_i}
    # This is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # To minimize the makespan, we sort the jobs:
    # Group 1: jobs where a_i < b_i, sorted by a_i ascending.
    # Group 2: jobs where a_i >= b_i, sorted by b_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        jobs.append((a, b))
        idx += 2
        
    group1 = []
    group2 = []
    for a, b in jobs:
        if a < b:
            group1.append((a, b))
        else:
            group2.append((a, b))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_main_time = 0
    current_review_time = 0
    
    for a, b in order:
        current_main_time += a
        current_review_time = max(current_main_time, current_review_time) + b
        
    print(current_review_time)

if __name__ == '__main__':
    solve()
