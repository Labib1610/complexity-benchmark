```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    performances = []
    for _ in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        performances.append((p1, p2))
    
    # Sort by p2 descending, then by p1 descending (or any order for ties)
    performances.sort(key=lambda x: (-x[1], -x[0]))
    
    total_main_time = 0
    max_finish_time = 0
    
    for p1, p2 in performances:
        total_main_time += p1
        # Critique starts at total_main_time, ends at total_main_time + p2
        finish_time = total_main_time + p2
        if finish_time > max_finish_time:
            max_finish_time = finish_time
    
    print(max_finish_time)

if __name__ == "__main__":
    solve()
```
