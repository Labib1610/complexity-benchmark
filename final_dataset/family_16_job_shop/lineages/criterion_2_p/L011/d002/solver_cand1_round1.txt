import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    plants = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        plants.append((p1, p2))
        idx += 2
        
    # Let the sequence of seeds be (s1, g1), (s2, g2), ..., (sj, gj)
    # The time the i-th seed finishes sowing is T_s(i) = sum_{k=1}^i s_k
    # The time the i-th seed finishes germinating is T_g(i)
    # T_g(1) = T_s(1) + g1
    # T_g(i) = max(T_s(i), T_g(i-1)) + gi
    # This is equivalent to Johnson's Rule for two-machine scheduling (Flow Shop).
    # To minimize the completion time of the last job:
    # Sort jobs with p1 < p2 in ascending order of p1.
    # Sort jobs with p1 >= p2 in descending order of p2.
    
    group1 = []
    group2 = []
    for p1, p2 in plants:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_sowing_time = 0
    current_germination_finish_time = 0
    
    for p1, p2 in order:
        current_sowing_time += p1
        current_germination_finish_time = max(current_sowing_time, current_germination_finish_time) + p2
        
    print(current_germination_finish_time)

if __name__ == '__main__':
    solve()
