```python
import sys

def solve():
    # The problem is equivalent to Johnson's Rule for a two-machine flow shop.
    # We want to minimize the makespan of a two-stage pipeline.
    # Let p1_i be the compression time and p2_i be the encryption time for file i.
    # Johnson's Rule:
    # 1. Divide files into two sets:
    #    Set A: files where p1_i < p2_i
    #    Set B: files where p1_i >= p2_i
    # 2. Sort Set A in ascending order of p1_i.
    # 3. Sort Set B in descending order of p2_i.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    files = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        files.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in files:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate the completion time
    # compressor_time: time when current file finishes compression
    # encryptor_time: time when current file finishes encryption
    compressor_time = 0
    encryptor_time = 0
    
    for p1, p2 in sequence:
        compressor_time += p1
        # Encryptor starts at max(compressor_time, previous_encryptor_finish_time)
        if compressor_time > encryptor_time:
            encryptor_time = compressor_time + p2
        else:
            encryptor_time += p2
            
    print(encryptor_time)

if __name__ == '__main__':
    solve()
```
