import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We want to minimize the makespan of a two-machine flow shop.
    # The optimal sequence is found by:
    # 1. Partitioning jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 2. Sorting Set A in ascending order of p1.
    # 3. Sorting Set B in descending order of p2.
    # 4. Concatenating Set A and Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    # Sort set_a by p1 ascending
    set_a.sort(key=lambda x: x[0])
    # Sort set_b by p2 descending
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    virtual_end = 0
    physical_end = 0
    
    for p1, p2 in sequence:
        virtual_end += p1
        if virtual_end > physical_end:
            physical_end = virtual_end
        physical_end += p2
        
    print(physical_end)

if __name__ == '__main__':
    solve()
