import sys
from typing import List, Tuple

def solve():
    import sys
    import threading
    
    def main():
        import sys
        sys.setrecursionlimit(1 << 25)
        
        j = int(sys.stdin.readline())
        trains = []
        for _ in range(j):
            x, y = map(int, sys.stdin.readline().split())
            trains.append((x, y))
        
        # Sort by descending order of Y_k to minimize the final completion time
        # This is a classic scheduling problem: Smith's rule
        # We want to minimize the maximum completion time
        # Sort by -Y_k (descending Y_k)
        trains.sort(key=lambda t: -t[1])
        
        t = 0  # Structural start time
        u = 0  # Cargo start time
        
        for x, y in trains:
            t += x
            u = max(u + y_prev if 'y_prev' in locals() else 0, t)
            y_prev = y
        
        # The final completion time is u + y_last
        print(u + y_prev)

    threading.Thread(target=main).start()

solve()
