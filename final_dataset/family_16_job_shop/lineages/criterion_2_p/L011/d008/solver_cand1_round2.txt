```python
import sys

def solve():
    # This problem is a classic application of Johnson's Rule for the 
    # two-machine flow shop scheduling problem.
    # We want to minimize the makespan (completion time of the last job).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where X_k < Y_k
    #    Set 2: jobs where X_k >= Y_k
    # 2. Sort Set 1 in ascending order of X_k.
    # 3. Sort Set 2 in descending order of Y_k.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        jobs.append((x, y))
        idx += 2
        
    set1 = []
    set2 = []
    for x, y in jobs:
        if x < y:
            set1.append((x, y))
        else:
            set2.append((x, y))
            
    set1.sort(key=lambda item: item[0])
    set2.sort(key=lambda item: item[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate makespan
    # t_k is structural start time, u_k is cargo start time
    # t_1 = 0
    # t_{k+1} = t_k + X_k
    # u_1 = X_1
    # u_{k+1} = max(u_k + Y_k, t_{k+1} + X_{k+1})
    # Final completion = u_j + Y_j
    
    current_t = 0
    current_u = 0
    
    for x, y in sequence:
        # Structural finish time for this job
        structural_finish = current_t + x
        # Cargo start time is max(previous cargo finish, structural finish)
        cargo_start = max(current_u, structural_finish)
        # Cargo finish time
        current_u = cargo_start + y
        # Update structural start for next job
        current_t = structural_finish
        
    print(current_u)

if __name__ == '__main__':
    solve()
```
