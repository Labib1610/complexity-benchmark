```python
import sys

def solve():
    # The problem asks to arrange j tiles (top_i, bottom_i) to minimize the maximum path cost.
    # A path consists of moving right in the top row until column k, then moving down,
    # then moving right in the bottom row.
    # The cost of a path moving down at column k is:
    # sum(top[0...k]) + sum(bottom[k...j-1])
    # Let S_top = sum(top), S_bot = sum(bottom).
    # Cost(k) = sum(top[0...k]) + sum(bottom[k...j-1])
    #         = sum(top[0...k]) + (S_bot - sum(bottom[0...k-1]))
    # Let T_i = top_i, B_i = bottom_i.
    # Cost(k) = T_0 + ... + T_k + B_k + ... + B_{j-1}
    # Let P_i = T_i - B_i.
    # Cost(k) = sum(T_0...T_{j-1}) + sum(B_k...B_{j-1}) - sum(B_0...B_{k-1}) ... this is not quite right.
    # Let's re-evaluate:
    # Cost(k) = (sum_{i=0}^k T_i) + (sum_{i=k}^{j-1} B_i)
    # Cost(k) = (sum_{i=0}^{j-1} B_i) + sum_{i=0}^k (T_i - B_i)
    # Let D_i = T_i - B_i.
    # Cost(k) = S_bot + sum_{i=0}^k D_i
    # To minimize max_k(Cost(k)), we need to minimize max_k(sum_{i=0}^k D_i).
    # This is equivalent to arranging D_i such that the prefix sums are as small as possible.
    # To minimize the maximum prefix sum, we should sort D_i in descending order.
    # If we sort D_i such that D_0 >= D_1 >= ... >= D_{j-1},
    # the prefix sums will be as small as possible at the end and large at the beginning.
    # Wait, if we want to minimize the maximum prefix sum, we should put the smallest D_i values first.
    # Let's check: if D = [2, -1, -2], prefix sums are [2, 1, -1]. Max is 2.
    # If D = [-2, -1, 2], prefix sums are [-2, -3, -1]. Max is -1.
    # Yes, sorting D in ascending order minimizes the maximum prefix sum.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    s_bot = 0
    idx = 1
    for _ in range(j):
        t = int(input_data[idx])
        b = int(input_data[idx+1])
        tiles.append(t - b)
        s_bot += b
        idx += 2
    
    tiles.sort()
    
    max_prefix_sum = -float('inf')
    current_sum = 0
    for d in tiles:
        current_sum += d
        if current_sum > max_prefix_sum:
            max_prefix_sum = current_sum
            
    print(s_bot + max_prefix_sum)

solve()
```
