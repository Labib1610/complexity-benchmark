```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    performances = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        performances.append((p1, p2))
        idx += 2
    
    # To minimize the completion time of the last critique, we use Johnson's Rule logic.
    # The total time is max(finish_time_i) for all i.
    # Let Main Stage times be A_i and Review Room times be B_i.
    # Sort performances such that those with smaller A_i come first if A_i < B_i,
    # and those with larger B_i come first if A_i >= B_i.
    
    group1 = []
    group2 = []
    for p1, p2 in performances:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_main_time = 0
    current_review_finish_time = 0
    
    for p1, p2 in order:
        current_main_time += p1
        # The critique for this performance can start at max(current_main_time, current_review_finish_time)
        # and lasts for p2.
        start_time = max(current_main_time, current_review_finish_time)
        current_review_finish_time = start_time + p2
        
    print(current_review_finish_time)

if __name__ == '__main__':
    solve()
```
