```python
import sys

def solve():
    # The problem asks to minimize the completion time of the last critique.
    # Let the performances be (a_i, b_i) where a_i is the Main Stage duration
    # and b_i is the Review Room duration.
    # If we fix the order of performances on the Main Stage as p_1, p_2, ..., p_j,
    # the completion time of the i-th performance on the Main Stage is S_i = sum_{k=1}^i a_{p_k}.
    # If we fix the order of critiques in the Review Room as q_1, q_2, ..., q_j,
    # the completion time of the i-th critique C_i is:
    # C_1 = S_{index(q_1)} + b_{q_1}
    # C_i = max(C_{i-1}, S_{index(q_i)}) + b_{q_i}
    
    # This is a variation of Johnson's Rule for two-machine flow shop scheduling.
    # To minimize the makespan, we should sort the performances.
    # Performances with a_i < b_i should be processed in increasing order of a_i.
    # Performances with a_i >= b_i should be processed in decreasing order of b_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    performances = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        performances.append((p1, p2))
        idx += 2
        
    # Johnson's Rule sorting
    group1 = [p for p in performances if p[0] < p[1]]
    group2 = [p for p in performances if p[0] >= p[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate completion time
    # The optimal order for the Main Stage is the same as the Review Room.
    current_main_stage_time = 0
    current_review_room_time = 0
    
    for a, b in order:
        current_main_stage_time += a
        current_review_room_time = max(current_review_room_time, current_main_stage_time) + b
        
    print(current_review_room_time)

if __name__ == '__main__':
    solve()
```
