import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    segments = []
    idx = 1
    for _ in range(n):
        u = int(input_data[idx])
        l = int(input_data[idx+1])
        segments.append((u, l))
        idx += 2
    
    # Let the sequence be (u_1, l_1), (u_2, l_2), ..., (u_n, l_n).
    # The total time for a drop point k is:
    # T_k = sum_{i=1}^k u_i + sum_{i=k}^n l_i
    # T_k = sum_{i=1}^n u_i + sum_{i=k}^n (l_i - u_i)
    # Let S = sum_{i=1}^n u_i.
    # T_k = S + sum_{i=k}^n (l_i - u_i).
    # We want to minimize max_k (T_k).
    # Let d_i = l_i - u_i.
    # T_k = S + sum_{i=k}^n d_i.
    # To minimize the maximum of these partial sums, we should sort the segments
    # such that the sequence of d_i values is non-increasing.
    # If d_1 >= d_2 >= ... >= d_n, then the partial sums sum_{i=k}^n d_i
    # are maximized when k is small.
    # Specifically, if we sort by d_i descending, the sequence of partial sums
    # P_k = sum_{i=k}^n d_i will be such that P_1 is the total sum of all d_i,
    # and P_k decreases as k increases.
    # Thus, the maximum will always be P_1 = sum_{i=1}^n d_i.
    # Wait, let's re-evaluate:
    # T_k = sum_{i=1}^{k-1} u_i + u_k + l_k + sum_{i=k+1}^n l_i
    # T_k = sum_{i=1}^n u_i + sum_{i=k}^n (l_i - u_i)
    # Let D_i = l_i - u_i.
    # T_k = (sum u_i) + (D_k + D_{k+1} + ... + D_n)
    # To minimize max_k (T_k), we want to arrange D_i such that the prefix sums
    # of the reversed sequence are minimized.
    # This is achieved by sorting D_i in non-decreasing order.
    # If D_1 <= D_2 <= ... <= D_n, then the sums S_k = sum_{i=k}^n D_i
    # are maximized at k=1.
    # So we sort segments such that l_i - u_i is non-decreasing.
    
    segments.sort(key=lambda x: x[1] - x[0])
    
    u_sum = sum(s[0] for s in segments)
    
    max_time = 0
    current_d_sum = sum(s[1] - s[0] for s in segments)
    
    # T_k = u_sum + current_d_sum
    # As we iterate k from 1 to n, current_d_sum decreases by (l_k - u_k)
    for i in range(n):
        max_time = max(max_time, u_sum + current_d_sum)
        current_d_sum -= (segments[i][1] - segments[i][0])
        
    print(max_time)

if __name__ == '__main__':
    solve()
