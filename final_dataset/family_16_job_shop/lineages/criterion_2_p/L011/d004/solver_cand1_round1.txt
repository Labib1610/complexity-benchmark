import sys

def solve():
    # The problem asks us to arrange j segments (p1_i, p2_i) to minimize the 
    # maximum path score. A path consists of moving right through some number 
    # of upper chambers, moving down at some column k, and then moving right 
    # through the remaining lower chambers.
    # The score for moving down at column k is:
    # S_k = (sum_{i=1}^k p1_i) + (sum_{i=k}^j p2_i)
    # Let P1 = sum(p1_i) and P2 = sum(p2_i).
    # S_k = (sum_{i=1}^k p1_i) + (P2 - sum_{i=1}^{k-1} p2_i)
    # S_k = P2 + p1_k + sum_{i=1}^{k-1} (p1_i - p2_i)
    # Let d_i = p1_i - p2_i.
    # S_k = P2 + p1_k + sum_{i=1}^{k-1} d_i
    # We want to minimize max_k (S_k).
    # This is equivalent to minimizing max_k (sum_{i=1}^{k-1} d_i + p1_k) + P2.
    # Sorting segments by d_i = p1_i - p2_i in non-decreasing order is optimal.
    # Proof sketch: If we have two adjacent segments i and j with d_i > d_j,
    # swapping them reduces the maximum of the two path scores.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    segments = []
    idx = 1
    total_p2 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        segments.append((p1, p2, p1 - p2))
        total_p2 += p2
        idx += 2
        
    # Sort by d_i = p1 - p2
    segments.sort(key=lambda x: x[2])
    
    max_score = 0
    current_prefix_d = 0
    for i in range(j):
        p1, p2, d = segments[i]
        # S_{i+1} = total_p2 + p1 + current_prefix_d
        current_score = total_p2 + p1 + current_prefix_d
        if current_score > max_score:
            max_score = current_score
        current_prefix_d += d
        
    print(max_score)

if __name__ == '__main__':
    solve()
