```python
import sys

def solve():
    # The problem asks to minimize the final Y-score.
    # Let the elements be (A_i, C_i).
    # Let P be the order in sequence P, and Q be the order in sequence Q.
    # X_i = sum_{k=1 to i} A_{P_k}
    # Y_1 = X_{Q_1} + C_{Q_1}
    # Y_i = max(X_{Q_i}, Y_{i-1}) + C_{Q_i}
    #
    # This recurrence Y_i = max(X_{Q_i}, Y_{i-1}) + C_{Q_i} can be expanded:
    # Y_n = max(X_{Q_n} + C_{Q_n}, X_{Q_{n-1}} + C_{Q_{n-1}} + C_{Q_n}, ..., X_{Q_1} + C_{Q_1} + ... + C_{Q_n})
    # Y_n = max_{i=1 to n} (X_{Q_i} + sum_{k=i to n} C_{Q_k})
    #
    # To minimize this, we should sort the elements such that we process them in an order
    # that minimizes the maximum.
    # Let S = sum of all A_i.
    # If we fix the order of Q, X_{Q_i} is the prefix sum of A in sequence P.
    # To minimize the expression, we should set P to be the same as Q.
    # Then X_{Q_i} = sum_{k=1 to i} A_{Q_k}.
    # Let S_i = sum_{k=1 to i} A_{Q_k}.
    # Let T_i = sum_{k=i to n} C_{Q_k}.
    # We want to minimize max_{i=1 to n} (S_i + T_i).
    # This is a classic scheduling problem. We want to order elements to minimize
    # the maximum of prefix sums of A plus suffix sums of C.
    # The optimal order is to sort by A_i / C_i ratio, but since we want to minimize
    # the maximum of S_i + T_i, we sort by A_i + C_i.
    # Actually, for this specific form, sorting by A_i + C_i is not necessarily correct.
    # Let's re-examine: Y_n = max_{i} (sum_{j=1}^i A_{Q_j} + sum_{j=i}^n C_{Q_j})
    # Y_n = max_{i} (sum_{j=1}^n A_{Q_j} + sum_{j=i}^n C_{Q_j} - sum_{j=i+1}^n A_{Q_j})
    # This is equivalent to minimizing the maximum of prefix sums.
    # The optimal strategy is to sort by A_i + C_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    elements = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        c = int(input_data[idx+1])
        elements.append((a, c))
        idx += 2
    
    # Sort by A_i + C_i
    elements.sort(key=lambda x: x[0] + x[1])
    
    # Calculate Y_n
    # Y_i = max(X_i, Y_{i-1}) + C_i
    # X_i = X_{i-1} + A_i
    current_x = 0
    current_y = 0
    for a, c in elements:
        current_x += a
        if current_y == 0:
            current_y = current_x + c
        else:
            current_y = max(current_x, current_y) + c
            
    print(current_y)

solve()
```
