import sys

def solve():
    # The total time for a drop point k is:
    # Sum_{i=1 to k} U_i + Sum_{i=k to j} L_i
    # = Sum_{i=1 to j} U_i + Sum_{i=k to j} (L_i - U_i)
    # Let S = Sum_{i=1 to j} U_i.
    # Let D_i = L_i - U_i.
    # Total time at k = S + Sum_{i=k to j} D_i.
    # We want to order the segments to minimize max_{k=1 to j} (S + Sum_{i=k to j} D_i).
    # Since S is constant, we want to minimize max_{k=1 to j} (Sum_{i=k to j} D_i).
    
    # Let T_k = Sum_{i=k to j} D_i.
    # T_j = D_j
    # T_{j-1} = D_{j-1} + D_j
    # ...
    # T_1 = D_1 + D_2 + ... + D_j
    
    # To minimize the maximum of these prefix sums of the reversed sequence D,
    # we should sort D_i such that the sequence of partial sums is as "flat" as possible.
    # This is a classic problem: to minimize the maximum prefix sum of a sequence,
    # we should sort the elements in descending order.
    # Let's verify: if we have D_1, D_2, ..., D_j, the partial sums are
    # P_1 = D_1
    # P_2 = D_1 + D_2
    # ...
    # If we sort D_i in descending order, the partial sums grow as slowly as possible
    # relative to the total sum. Actually, for the sequence of partial sums 
    # T_k = D_k + ... + D_j, if we sort D_i in ascending order, the partial sums
    # are minimized.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    segments = []
    idx = 1
    sum_u = 0
    for _ in range(j):
        u = int(input_data[idx])
        l = int(input_data[idx+1])
        segments.append(l - u)
        sum_u += u
        idx += 2
        
    # We want to minimize max_{k=1..j} (Sum_{i=k..j} D_i)
    # Let the chosen order be d_1, d_2, ..., d_j.
    # We want to minimize max(d_j, d_{j-1}+d_j, ..., d_1+...+d_j).
    # This is equivalent to minimizing the maximum prefix sum of the reversed sequence.
    # To minimize the maximum prefix sum of a sequence, sort it in descending order.
    # So we sort D in descending order: d_1 >= d_2 >= ... >= d_j.
    # Then the sequence of partial sums (from the end) is:
    # T_j = d_j
    # T_{j-1} = d_{j-1} + d_j
    # ...
    # T_1 = d_1 + ... + d_j
    # This is minimized when d_i are sorted in ascending order.
    
    segments.sort()
    
    max_val = -float('inf')
    current_sum = 0
    # We need max_{k=1..j} (Sum_{i=k..j} D_i)
    # With segments sorted ascending: d_1 <= d_2 <= ... <= d_j
    # Sum_{i=k..j} D_i = D_k + D_{k+1} + ... + D_j
    # This is maximized at k=1.
    # Wait, if we sort ascending, the max is always the total sum.
    # If we sort descending, the max is the largest element.
    # Let's re-evaluate:
    # We want to minimize max(T_1, T_2, ..., T_j) where T_k = sum_{i=k}^j D_i.
    # If we sort D_i ascending: D_1 <= D_2 <= ... <= D_j.
    # T_j = D_j
    # T_{j-1} = D_{j-1} + D_j
    # ...
    # T_1 = D_1 + ... + D_j
    # The max is T_1.
    # If we sort D_i descending: D_1 >= D_2 >= ... >= D_j.
    # T_j = D_j
    # T_{j-1} = D_{j-1} + D_j
    # ...
    # T_1 = D_1 + ... + D_j
    # The max is T_1.
    # Actually, the max is always the total sum if all D_i are positive.
    # If some are negative, we want to put the most negative ones at the beginning.
    # So sort D_i in ascending order.
    
    current_sum = 0
    max_t = -float('inf')
    # T_k = sum_{i=k}^j D_i.
    # To minimize max(T_k), we want the largest D_i to be at the end (k=j).
    # So sort D_i ascending.
    segments.sort()
    
    # Calculate max_{k=1..j} T_k
    # T_j = D_j
    # T_{j-1} = D_{j-1} + D_j
    # ...
    # T_1 = D_1 + ... + D_j
    
    t = [0] * j
    curr = 0
    for i in range(j - 1, -1, -1):
        curr += segments[i]
        t[i] = curr
        
    print(sum_u + max(t))

solve()
