import sys

def solve():
    # The problem asks to minimize C = S_j + B_j.
    # S_1 = A_1
    # S_i = max(S_{i-1} + B_{i-1}, sum_{k=1}^i A_k)
    # Let T_i = sum_{k=1}^i A_k.
    # S_1 = T_1
    # S_2 = max(S_1 + B_1, T_2) = max(T_1 + B_1, T_2)
    # S_3 = max(S_2 + B_2, T_3) = max(T_1 + B_1 + B_2, T_2 + B_2, T_3)
    # By induction, S_j = max_{1 <= i <= j} (T_i + sum_{k=i}^{j-1} B_k)
    # C = S_j + B_j = max_{1 <= i <= j} (T_i + sum_{k=i}^j B_k)
    # Let A_k be the Main Stage duration and B_k be the Review Room duration.
    # T_i = sum_{k=1}^i A_k.
    # C = max_{1 <= i <= j} (sum_{k=1}^i A_k + sum_{k=i}^j B_k)
    # C = max_{1 <= i <= j} (sum_{k=1}^j A_k + sum_{k=i}^j B_k - sum_{k=i}^j A_k)
    # C = sum_{k=1}^j A_k + max_{1 <= i <= j} (sum_{k=i}^j (B_k - A_k))
    
    # Let D_k = B_k - A_k.
    # We want to minimize max_{1 <= i <= j} (sum_{k=i}^j D_k).
    # This is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # To minimize the makespan, sort performances such that:
    # If A_i < B_i, they go in increasing order of A_i.
    # If A_i >= B_i, they go in decreasing order of B_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    performances = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        performances.append((a, b))
        idx += 2
        
    group1 = []
    group2 = []
    for a, b in performances:
        if a < b:
            group1.append((a, b))
        else:
            group2.append((a, b))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    lineup = group1 + group2
    
    current_s = 0
    current_a_sum = 0
    for a, b in lineup:
        current_a_sum += a
        current_s = max(current_s + b, current_a_sum + b)
        
    print(current_s)

if __name__ == '__main__':
    solve()
