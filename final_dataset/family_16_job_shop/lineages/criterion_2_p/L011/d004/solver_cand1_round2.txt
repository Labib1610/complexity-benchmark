```python
import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    segments = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        segments.append((p1, p2))
        idx += 2
        
    # Let S1 be the sum of all p1, S2 be the sum of all p2.
    # For a given arrangement, the cost of moving down at column k is:
    # (sum of p1 from 1 to k) + (sum of p2 from k to j)
    # = (sum of p1 from 1 to k) + (sum of p2 from 1 to j) - (sum of p2 from 1 to k-1)
    # = S2 + (sum of p1 from 1 to k) - (sum of p2 from 1 to k-1)
    # Let d_i = p1_i - p2_i.
    # Cost at k = S2 + sum_{i=1}^k d_i + p2_k
    # Actually, simpler: Cost at k = (sum_{i=1}^k p1_i) + (sum_{i=k}^j p2_i)
    # Let P1_k = sum_{i=1}^k p1_i, P2_k = sum_{i=k}^j p2_i.
    # We want to minimize max_k (P1_k + P2_k).
    # This is equivalent to sorting segments such that the sequence of partial sums
    # is as "flat" as possible.
    # Sorting by p1 - p2 is a standard approach for this type of scheduling problem.
    
    segments.sort(key=lambda x: x[0] - x[1])
    
    p1_sum = 0
    p2_sum = sum(s[1] for s in segments)
    
    max_cost = 0
    # At each step k (1-indexed), we move down at column k.
    # The path cost is (sum of p1 up to k) + (sum of p2 from k to j)
    # = p1_sum_k + p2_sum_k
    
    current_p1 = 0
    current_p2_sum = p2_sum
    
    for p1, p2 in segments:
        current_p1 += p1
        # Cost if we move down at this column
        cost = current_p1 + current_p2_sum
        if cost > max_cost:
            max_cost = cost
        current_p2_sum -= p2
        
    print(max_cost)

if __name__ == '__main__':
    solve()
```
