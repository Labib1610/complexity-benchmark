```python
import sys

def solve():
    # The problem asks to minimize Y_j = max(X_1, Y_0) + B_1, ..., Y_j = max(X_j, Y_{j-1}) + B_j.
    # Let S_k = sum_{i=1}^k A_i.
    # Y_j = max(S_1 + B_1 + B_2 + ... + B_j, S_2 + B_2 + ... + B_j, ..., S_j + B_j)
    # Y_j = max_{k=1 to j} (S_k + sum_{i=k}^j B_i)
    # Let T = sum_{i=1}^j B_i.
    # Y_j = max_{k=1 to j} (S_k + T - sum_{i=1}^{k-1} B_i)
    # Y_j = T + max_{k=1 to j} (S_k - sum_{i=1}^{k-1} B_i)
    # Let P_k = sum_{i=1}^{k-1} B_i (with P_1 = 0).
    # We want to minimize max_{k=1 to j} (S_k - P_k).
    # This is a classic scheduling problem.
    # If we have two adjacent tasks (a1, b1) and (a2, b2):
    # Order 1: (a1, b1), (a2, b2) -> max(S+a1-0, S+a1+a2-b1)
    # Order 2: (a2, b2), (a1, b1) -> max(S+a2-0, S+a2+a1-b2)
    # We prefer Order 1 if max(a1, a1+a2-b1) < max(a2, a1+a2-b2).
    # This simplifies to sorting by b_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    pairs = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        pairs.append((a, b))
        idx += 2
    
    # Sort by b descending
    pairs.sort(key=lambda x: x[1], reverse=True)
    
    current_x = 0
    current_y = 0
    for a, b in pairs:
        current_x += a
        current_y = max(current_x, current_y) + b
        
    print(current_y)

if __name__ == '__main__':
    solve()
```
