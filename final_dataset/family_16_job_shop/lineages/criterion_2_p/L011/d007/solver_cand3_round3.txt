```python
import sys

def solve():
    # The problem is equivalent to Johnson's Rule for the two-machine flow shop scheduling problem.
    # We want to minimize the makespan (the time the last thread finishes the Beta Mutex).
    # Johnson's Rule:
    # 1. Divide threads into two sets:
    #    Set 1: threads where Alpha time < Beta time
    #    Set 2: threads where Alpha time >= Beta time
    # 2. Sort Set 1 in ascending order of Alpha time.
    # 3. Sort Set 2 in descending order of Beta time.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    threads = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        threads.append((a, b))
        idx += 2
        
    set1 = []
    set2 = []
    
    for t in threads:
        if t[0] < t[1]:
            set1.append(t)
        else:
            set2.append(t)
            
    # Sort set1 by Alpha time ascending
    set1.sort(key=lambda x: x[0])
    # Sort set2 by Beta time descending
    set2.sort(key=lambda x: x[1], reverse=True)
    
    order = set1 + set2
    
    # Calculate completion time
    # alpha_finish[i] = sum of alpha times of threads 0 to i
    # beta_finish[i] = max(alpha_finish[i], beta_finish[i-1]) + beta_time[i]
    
    current_alpha_finish = 0
    current_beta_finish = 0
    
    for a, b in order:
        current_alpha_finish += a
        current_beta_finish = max(current_alpha_finish, current_beta_finish) + b
        
    print(current_beta_finish)

if __name__ == '__main__':
    solve()
```
