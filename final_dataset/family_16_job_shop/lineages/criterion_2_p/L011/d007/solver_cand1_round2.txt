```python
import sys

def solve():
    # The problem asks to minimize the final Y after applying all steps.
    # Let the sequence of steps be (A_i, B_i).
    # X_k = sum_{i=1}^k A_i
    # Y_k = max(X_k, Y_{k-1}) + B_k
    # This is a classic scheduling problem.
    # If we have two adjacent steps (A1, B1) and (A2, B2):
    # Order 1, 2:
    # Y_1 = max(X + A1, Y) + B1
    # Y_2 = max(X + A1 + A2, Y_1) + B2
    #     = max(X + A1 + A2, max(X + A1, Y) + B1) + B2
    #     = max(X + A1 + A2 + B2, X + A1 + B1 + B2, Y + B1 + B2)
    # Order 2, 1:
    # Y_2' = max(X + A2 + A1 + B1, X + A2 + B2 + B1, Y + B2 + B1)
    # Comparing the two, we want to minimize the max.
    # The terms X + A1 + A2 + B2 and X + A2 + A1 + B1 are common.
    # We compare max(X + A1 + B1 + B2, Y + B1 + B2) vs max(X + A2 + B2 + B1, Y + B2 + B1).
    # This simplifies to comparing B2 vs B1.
    # If B1 < B2, we should put the step with smaller B first.
    # If B1 == B2, the order doesn't matter for the final Y.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    steps = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        steps.append((a, b))
        idx += 2
        
    # Sort by B ascending
    steps.sort(key=lambda x: x[1])
    
    x_curr = 0
    y_curr = 0
    for a, b in steps:
        x_curr += a
        y_curr = max(x_curr, y_curr) + b
        
    print(y_curr)

if __name__ == '__main__':
    solve()
```
