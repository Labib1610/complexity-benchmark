```python
import sys

def solve():
    # The problem asks to arrange j tiles (p1_i, p2_i) to minimize the maximum path cost.
    # A path moving down at column k has cost:
    # sum(p1_1...p1_k) + sum(p2_k...p2_j)
    # Let S1 = sum(p1_i) and S2 = sum(p2_i).
    # The cost at column k is:
    # (sum_{i=1}^k p1_i) + (sum_{i=k}^j p2_i)
    # = (sum_{i=1}^k p1_i) + (sum_{i=1}^j p2_i - sum_{i=1}^{k-1} p2_i)
    # = S2 + sum_{i=1}^k (p1_i - p2_{i-1}) where p2_0 = 0.
    # Let d_i = p1_i - p2_i.
    # The cost at column k is S2 + p1_k + sum_{i=1}^{k-1} (p1_i - p2_i)
    # = S2 + p1_k + sum_{i=1}^{k-1} d_i.
    # To minimize the maximum of these values, we should sort the tiles.
    # Let the tiles be (a_i, b_i). We want to minimize max_k (sum_{i=1}^k a_i + sum_{i=k}^j b_i).
    # This is equivalent to minimizing max_k (sum_{i=1}^j b_i + sum_{i=1}^k (a_i - b_i)).
    # Let d_i = a_i - b_i. We want to minimize max_k (S2 + sum_{i=1}^k d_i).
    # This is minimized by sorting d_i in non-decreasing order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    s2 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tiles.append((p1, p2))
        s2 += p2
        idx += 2
    
    # Sort by d_i = p1 - p2
    tiles.sort(key=lambda x: x[0] - x[1])
    
    max_cost = 0
    current_sum_d = 0
    for i in range(j):
        # Cost at column i+1 is S2 + sum_{k=1}^{i+1} (p1_k - p2_k) + p2_{i+1}
        # Wait, the formula is: sum_{k=1}^i p1_k + sum_{k=i}^j p2_k
        # = sum_{k=1}^i p1_k + p2_i + sum_{k=i+1}^j p2_k
        # = sum_{k=1}^i (p1_k - p2_k) + sum_{k=1}^j p2_k + p1_i (if we define p1_0=0)
        # Actually, simpler:
        # Path cost at column k (1-indexed):
        # C_k = (sum_{i=1}^k p1_i) + (sum_{i=k}^j p2_i)
        # C_k = (sum_{i=1}^k (p1_i - p2_i)) + sum_{i=1}^j p2_i
        
        current_sum_d += (tiles[i][0] - tiles[i][1])
        max_cost = max(max_cost, s2 + current_sum_d)
        
    print(max_cost)

if __name__ == '__main__':
    solve()
```
