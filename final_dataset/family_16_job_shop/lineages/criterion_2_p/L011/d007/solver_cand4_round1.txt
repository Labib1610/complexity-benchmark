import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    artifacts = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        artifacts.append((p1, p2))
        idx += 2
    
    # Let L_i be the left thickness and R_i be the right thickness of artifact at position i.
    # Total resistance for shift k is:
    # S_k = sum_{i=1}^k L_i + sum_{i=k}^j R_i
    # S_k = sum_{i=1}^j R_i + sum_{i=1}^k (L_i - R_i)
    # Let D_i = L_i - R_i.
    # S_k = sum(R) + sum_{i=1}^k D_i
    # We want to minimize max_{k=1..j} (sum(R) + sum_{i=1}^k D_i)
    # This is equivalent to minimizing max_{k=1..j} (sum_{i=1}^k D_i)
    # Let P_k = sum_{i=1}^k D_i be the prefix sums of D_i.
    # We want to minimize max(P_1, P_2, ..., P_j).
    
    # To minimize the maximum prefix sum, we should sort the artifacts based on D_i.
    # If we sort such that D_i are in non-decreasing order, the prefix sums will
    # grow as slowly as possible.
    
    artifacts.sort(key=lambda x: x[0] - x[1])
    
    sum_r = sum(a[1] for a in artifacts)
    
    max_resistance = -float('inf')
    current_prefix_sum = 0
    
    # We need to find min over all permutations of max_{k=1..j} (sum(R) + P_k)
    # With D_i sorted non-decreasingly, the prefix sums P_k are minimized.
    # Specifically, if D_1 <= D_2 <= ... <= D_j, then the sequence of prefix sums
    # P_k = sum_{i=1}^k D_i is the optimal way to keep the maximum prefix sum small.
    
    ans = -float('inf')
    current_p = 0
    for i in range(j):
        current_p += (artifacts[i][0] - artifacts[i][1])
        if current_p > ans:
            ans = current_p
            
    print(sum_r + ans)

if __name__ == '__main__':
    solve()
