```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    reactions = []
    idx = 1
    total_p2 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        reactions.append((p1, p2))
        total_p2 += p2
        idx += 2
    
    # Let the initial pressure be S = sum(p2_i).
    # For a sequence of reactions (p1_i, p2_i), the peak pressure at step k is:
    # Peak_k = S - sum_{i=1}^{k-1} p2_i + p1_k
    # We want to minimize max(Peak_k).
    # Peak_k = S + p1_k - sum_{i=1}^{k-1} p2_i
    # Let diff_i = p1_i - p2_i.
    # Peak_k = S + p1_k - (sum_{all} p2_i - sum_{i=k}^{j} p2_i)
    # Peak_k = S - sum_{all} p2_i + p1_k + sum_{i=k}^{j} p2_i
    # Since S = sum_{all} p2_i, this simplifies to:
    # Peak_k = p1_k + sum_{i=k}^{j} p2_i
    
    # To minimize max(p1_k + sum_{i=k}^{j} p2_i), we use the exchange argument.
    # Consider two adjacent reactions i and i+1.
    # Order (i, i+1): max(p1_i + p2_i + p2_{i+1} + sum_{rest}, p1_{i+1} + p2_{i+1} + sum_{rest})
    # Order (i+1, i): max(p1_{i+1} + p2_{i+1} + p2_i + sum_{rest}, p1_i + p2_i + sum_{rest})
    # We prefer (i, i+1) if max(p1_i + p2_i + p2_{i+1}, p1_{i+1} + p2_{i+1}) < max(p1_{i+1} + p2_{i+1} + p2_i, p1_i + p2_i)
    # This simplifies to sorting by p1_i in descending order.
    
    reactions.sort(key=lambda x: x[0], reverse=True)
    
    max_peak = 0
    current_sum_p2 = total_p2
    for p1, p2 in reactions:
        peak = p1 + current_sum_p2
        if peak > max_peak:
            max_peak = peak
        current_sum_p2 -= p2
        
    print(max_peak)

if __name__ == '__main__':
    solve()
```
