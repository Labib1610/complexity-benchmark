import sys

def solve():
    # The problem asks to minimize Y_j = max(X_1 + b_1 + b_2 + ... + b_j, 
    #                                       X_2 + b_2 + ... + b_j, 
    #                                       ..., 
    #                                       X_j + b_j)
    # where X_k = a_1 + a_2 + ... + a_k.
    # This is equivalent to minimizing max_{k=1..j} (X_k + sum_{i=k}^j b_i).
    # Let S_a = sum(a_i) and S_b = sum(b_i).
    # The expression is max_{k=1..j} (X_k + S_b - sum_{i=1}^{k-1} b_i).
    # This is equivalent to minimizing max_{k=1..j} (X_k - sum_{i=1}^{k-1} b_i) + S_b.
    # Let P_k = X_k - sum_{i=1}^{k-1} b_i.
    # We want to order items to minimize the maximum prefix sum of (a_i - b_{i-1}) where b_0 = 0.
    # This is a classic scheduling problem. If we have two items (a1, b1) and (a2, b2):
    # Order 1, 2: max(a1, a1+a2-b1) + b1+b2 = max(a1+b1+b2, a1+a2+b2)
    # Order 2, 1: max(a2, a2+a1-b2) + b2+b1 = max(a2+b2+b1, a2+a1+b1)
    # We prefer 1, 2 if max(a1+b1+b2, a1+a2+b2) < max(a2+b2+b1, a2+a1+b1).
    # This simplifies to comparing b1 and b2.
    # If b1 < b2, we should put the item with the smaller b first.
    # If b1 == b2, the order doesn't matter for the max.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    items = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        items.append((a, b))
        idx += 2
        
    # Sort by secondary value b ascending
    items.sort(key=lambda x: x[1])
    
    x_curr = 0
    y_curr = 0
    for a, b in items:
        x_curr += a
        y_curr = max(x_curr, y_curr) + b
        
    print(y_curr)

if __name__ == '__main__':
    solve()
