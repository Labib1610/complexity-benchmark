```python
import sys

def solve():
    # The problem asks to minimize C = sum(A_k) + D_j.
    # Since sum(A_k) is constant regardless of the order, we only need to minimize D_j.
    # The recurrence is D_i = max(D_{i-1} - A_i, 0) + B_i.
    # This is equivalent to Johnson's Rule for the two-machine flow shop scheduling problem.
    # To minimize the makespan (which is sum(A_k) + D_j), we sort the performances:
    # 1. Performances with A_i < B_i are placed in increasing order of A_i.
    # 2. Performances with A_i >= B_i are placed in decreasing order of B_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    performances = []
    idx = 1
    total_a = 0
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        performances.append((a, b))
        total_a += a
        idx += 2
        
    group1 = []
    group2 = []
    
    for a, b in performances:
        if a < b:
            group1.append((a, b))
        else:
            group2.append((a, b))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    d = 0
    for a, b in order:
        d = max(d - a, 0) + b
        
    print(total_a + d)

if __name__ == '__main__':
    solve()
```
