```python
import sys

def solve():
    # The bottleneck score at index k is:
    # S_k = sum_{i=1}^k A_i + sum_{i=k}^j B_i
    # S_k = sum_{i=1}^j B_i + sum_{i=1}^k (A_i - B_i)
    # Let D_i = A_i - B_i.
    # S_k = (sum_{i=1}^j B_i) + (sum_{i=1}^k D_i)
    # To minimize max(S_k), we need to minimize the maximum prefix sum of D_i.
    # This is a classic scheduling problem.
    # If we sort modules such that those with D_i < 0 come first (sorted by D_i ascending),
    # and those with D_i >= 0 come later (sorted by D_i ascending),
    # we minimize the maximum prefix sum.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    modules = []
    idx = 1
    total_b = 0
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        modules.append((a, b, a - b))
        total_b += b
        idx += 2
        
    # Sort: negative D_i first (ascending), then positive D_i (ascending)
    # Actually, the standard result for minimizing max prefix sum is:
    # Sort by D_i ascending.
    modules.sort(key=lambda x: x[2])
    
    max_s = -float('inf')
    current_prefix_d = 0
    for i in range(j):
        current_prefix_d += modules[i][2]
        max_s = max(max_s, total_b + current_prefix_d)
        
    print(max_s)

if __name__ == '__main__':
    solve()
```
