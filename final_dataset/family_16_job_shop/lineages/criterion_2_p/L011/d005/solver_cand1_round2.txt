import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # To minimize the makespan T = max_{1<=k<=j} (sum_{i=1}^k X_i + sum_{i=k}^j Y_i),
    # we sort the batches based on the following criteria:
    # 1. If min(X_i, Y_i) == X_i, the batch should come earlier.
    # 2. If min(X_i, Y_i) == Y_i, the batch should come later.
    # Specifically, sort by min(X_i, Y_i). If X_i < Y_i, sort by X_i ascending.
    # If Y_i <= X_i, sort by Y_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    batches = []
    idx = 1
    for _ in range(j):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        batches.append((x, y))
        idx += 2
        
    # Johnson's Rule sorting
    # Group 1: X_i < Y_i, sorted by X_i ascending
    # Group 2: X_i >= Y_i, sorted by Y_i descending
    group1 = []
    group2 = []
    for x, y in batches:
        if x < y:
            group1.append((x, y))
        else:
            group2.append((x, y))
            
    group1.sort(key=lambda b: b[0])
    group2.sort(key=lambda b: b[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate T using the formula:
    # T = max_{1<=k<=j} (sum_{i=1}^k X_i + sum_{i=k}^j Y_i)
    # Let P_k = sum_{i=1}^k X_i and Q_k = sum_{i=k}^j Y_i
    # T = max_{k} (P_k + Q_k)
    
    # P_k = P_{k-1} + X_k
    # Q_k = Q_{k+1} + Y_k
    
    n = len(sequence)
    x_vals = [b[0] for b in sequence]
    y_vals = [b[1] for b in sequence]
    
    p = [0] * (n + 1)
    for i in range(n):
        p[i+1] = p[i] + x_vals[i]
        
    q = [0] * (n + 2)
    for i in range(n, 0, -1):
        q[i] = q[i+1] + y_vals[i-1]
        
    max_t = 0
    for k in range(1, n + 1):
        current_t = p[k] + q[k]
        if current_t > max_t:
            max_t = current_t
            
    print(max_t)

if __name__ == '__main__':
    solve()
