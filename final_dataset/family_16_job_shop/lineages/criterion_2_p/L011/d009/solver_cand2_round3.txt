```python
import sys

def solve():
    # The score for a sequence d_1, ..., d_j is:
    # max_{1 <= k <= j} (sum_{i=1}^k top(d_i) + sum_{i=k}^j bottom(d_i))
    # Let S_top = sum_{i=1}^j top(d_i) and S_bot = sum_{i=1}^j bottom(d_i).
    # Let T_i = top(d_i) and B_i = bottom(d_i).
    # The expression is: sum_{i=1}^k T_i + sum_{i=k}^j B_i
    # = sum_{i=1}^k T_i + B_k + sum_{i=k+1}^j B_i
    # = sum_{i=1}^k T_i + sum_{i=k}^j B_i
    # Let P_k = sum_{i=1}^k T_i + sum_{i=k}^j B_i.
    # Note that P_{k+1} - P_k = T_{k+1} - B_k.
    # We want to minimize max(P_1, P_2, ..., P_j).
    # This is equivalent to minimizing the peak of a sequence where
    # P_1 = T_1 + sum_{i=1}^j B_i
    # P_{k+1} = P_k + T_{k+1} - B_k
    # Let D_i = T_i - B_i.
    # P_k = sum_{i=1}^j B_i + sum_{i=1}^k T_i - sum_{i=1}^{k-1} B_i
    # P_k = sum_{i=1}^j B_i + sum_{i=1}^k (T_i - B_i) + B_k
    # Actually, simpler: P_k = (sum_{i=1}^j B_i) + sum_{i=1}^k (T_i - B_i) + B_k
    # Let S = sum B_i. P_k = S + sum_{i=1}^k (T_i - B_i) + B_k.
    # This is a classic problem: arrange items to minimize the maximum prefix sum.
    # To minimize the maximum prefix sum, we should sort the items by D_i = T_i - B_i.
    # If we sort such that D_1 <= D_2 <= ... <= D_j, the maximum is minimized.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        t = int(input_data[idx])
        b = int(input_data[idx+1])
        tiles.append((t, b))
        idx += 2
        
    # Sort by T_i - B_i
    tiles.sort(key=lambda x: x[0] - x[1])
    
    # Calculate P_k = sum_{i=1}^k T_i + sum_{i=k}^j B_i
    # P_k = (sum_{i=1}^j B_i) + sum_{i=1}^k (T_i - B_i) + B_k
    
    sum_b = sum(tile[1] for tile in tiles)
    
    current_diff_sum = 0
    max_score = -float('inf')
    
    for i in range(j):
        t, b = tiles[i]
        current_diff_sum += (t - b)
        score = sum_b + current_diff_sum + b
        if score > max_score:
            max_score = score
            
    print(max_score)

if __name__ == '__main__':
    solve()
```
