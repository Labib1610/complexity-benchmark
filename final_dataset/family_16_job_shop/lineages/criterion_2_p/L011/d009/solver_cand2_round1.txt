```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    blocks = []
    idx = 1
    sum_p2 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        blocks.append((p1, p2))
        sum_p2 += p2
        idx += 2
    
    # Let V_0 = sum(p2).
    # For a sequence of blocks (p1_i, p2_i):
    # P_1 = V_0 + p1_1
    # P_2 = V_0 + p1_1 - p2_1 + p1_2
    # P_3 = V_0 + p1_1 - p2_1 + p1_2 - p2_2 + p1_3
    # ...
    # P_k = V_0 + sum_{i=1}^{k-1} (p1_i - p2_i) + p1_k
    #
    # Let diff_i = p1_i - p2_i.
    # P_k = V_0 + sum_{i=1}^{k-1} diff_i + p1_k
    #
    # To minimize max(P_k), we use the standard scheduling result:
    # If diff_i >= 0, process in increasing order of p1_i.
    # If diff_i < 0, process in decreasing order of p2_i.
    # Blocks with diff_i >= 0 should come before blocks with diff_i < 0.
    
    pos = []
    neg = []
    for p1, p2 in blocks:
        if p1 - p2 >= 0:
            pos.append((p1, p2))
        else:
            neg.append((p1, p2))
            
    # Sort pos: increasing p1
    pos.sort(key=lambda x: x[0])
    # Sort neg: decreasing p2 (which is equivalent to increasing p1-p2)
    neg.sort(key=lambda x: x[1], reverse=True)
    
    sequence = pos + neg
    
    max_peak = 0
    current_v = sum_p2
    for p1, p2 in sequence:
        peak = current_v + p1
        if peak > max_peak:
            max_peak = peak
        current_v = current_v + p1 - p2
        
    print(max_peak)

if __name__ == '__main__':
    solve()
```
