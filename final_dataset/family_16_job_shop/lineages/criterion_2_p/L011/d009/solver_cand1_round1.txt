```python
import sys

def solve():
    # The problem asks to minimize S_final + D.
    # Let the amplifiers be (p1_i, p2_i).
    # S_0 = 0
    # S_i = max(S_{i-1} - p1_i, 0) + p2_i
    # We want to minimize S_j + sum(p1_i).
    # Let f_i(S) = max(S - p1_i, 0) + p2_i.
    # We want to minimize f_j(f_{j-1}(...f_1(0)...)) + sum(p1_i).
    # Note that f_i(S) = max(S - p1_i + p2_i, p2_i).
    # This is a composition of functions of the form max(S + a, b).
    # It is known that for functions of the form f(S) = max(S + a, b),
    # the composition is also of the form max(S + A, B).
    # Specifically, if f(S) = max(S + a1, b1) and g(S) = max(S + a2, b2),
    # then g(f(S)) = max(max(S + a1, b1) + a2, b2)
    #              = max(S + a1 + a2, b1 + a2, b2).
    # So for a sequence of amplifiers (p1_i, p2_i), let a_i = p2_i - p1_i.
    # The composition is max(S + sum(a_i), max_{k=1..j} (p2_k + sum_{i=k+1}^j a_i)).
    # Since S_0 = 0, S_j = max(sum(a_i), max_{k=1..j} (p2_k + sum_{i=k+1}^j a_i)).
    # We want to minimize S_j + sum(p1_i).
    # Note sum(a_i) = sum(p2_i) - sum(p1_i).
    # So S_j + sum(p1_i) = max(sum(p2_i), max_{k=1..j} (p2_k + sum_{i=k+1}^j (p2_i - p1_i) + sum(p1_i))).
    # This simplifies to max(sum(p2_i), max_{k=1..j} (p2_k + sum_{i=k+1}^j p2_i + sum_{i=1}^k p1_i)).
    # To minimize this, we should sort the amplifiers.
    # The optimal order is to sort by p1_i in ascending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    amps = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        amps.append((p1, p2))
        idx += 2
    
    # Sort by p1 ascending
    amps.sort()
    
    s = 0
    d_sum = 0
    for p1, p2 in amps:
        s = max(s - p1, 0) + p2
        d_sum += p1
        
    print(s + d_sum)

if __name__ == '__main__':
    solve()
```
