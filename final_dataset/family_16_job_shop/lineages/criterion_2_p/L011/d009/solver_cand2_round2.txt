import sys

def solve():
    # The problem asks to minimize the maximum path cost.
    # Let the dominoes be (a_i, b_i) for i=1 to j.
    # A path consists of moving right until some column k, then moving down,
    # then moving right until column j.
    # The cost of a path switching at column k is:
    # sum(a_1...a_k) + sum(b_k...b_j)
    # = sum(a_1...a_j) + sum(b_1...b_j) - sum(a_{k+1}...a_j) - sum(b_1...b_{k-1})
    # Let S_a = sum(a_i) and S_b = sum(b_i).
    # The cost for a switch at column k is:
    # S_a + S_b - (sum_{i=k+1}^j a_i + sum_{i=1}^{k-1} b_i)
    # To minimize the maximum cost, we need to maximize the minimum value of
    # (sum_{i=k+1}^j a_i + sum_{i=1}^{k-1} b_i) over all k=1..j.
    # Let f(k) = sum_{i=1}^{k-1} b_i + sum_{i=k+1}^j a_i.
    # We want to arrange dominoes to maximize min_k f(k).
    # Note f(k+1) - f(k) = b_k - a_{k+1}.
    # This is a known problem: to maximize the minimum prefix/suffix sum,
    # we should sort the dominoes based on the difference (a_i - b_i).
    # Specifically, sorting by (a_i - b_i) ascending is optimal.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    dominoes = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        dominoes.append((a, b))
        idx += 2
        
    # Sort by a_i - b_i
    dominoes.sort(key=lambda x: x[0] - x[1])
    
    a_vals = [d[0] for d in dominoes]
    b_vals = [d[1] for d in dominoes]
    
    sum_a = sum(a_vals)
    sum_b = sum(b_vals)
    
    # f(k) = sum_{i=0}^{k-2} b_i + sum_{i=k}^{j-1} a_i (using 0-indexing for k=1..j)
    # Let prefix_b[k] = sum_{i=0}^{k-1} b_i, suffix_a[k] = sum_{i=k}^{j-1} a_i
    # f(k) = prefix_b[k-1] + suffix_a[k]
    
    prefix_b = [0] * (j + 1)
    for i in range(j):
        prefix_b[i+1] = prefix_b[i] + b_vals[i]
        
    suffix_a = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suffix_a[i] = suffix_a[i+1] + a_vals[i]
        
    max_cost = 0
    for k in range(1, j + 1):
        # Path switching at column k (1-indexed)
        # Cost = sum(a_1..a_k) + sum(b_k..b_j)
        current_cost = (suffix_a[0] - suffix_a[k]) + (prefix_b[j] - prefix_b[k-1])
        if current_cost > max_cost:
            max_cost = current_cost
            
    print(max_cost)

solve()
