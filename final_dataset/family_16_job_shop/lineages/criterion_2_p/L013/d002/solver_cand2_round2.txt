import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    trains = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        trains.append((p1, p2))
        idx += 2
    
    # Let p1_i be the time in the forest and p2_i be the time in the glacier.
    # If we order trains in sequence 1, 2, ..., j:
    # Let S_i be the start time of train i in the forest.
    # S_1 = 0
    # S_i = S_{i-1} + p1_{i-1}
    # Finish time of train i in forest: F1_i = S_i + p1_i
    # Start time of train i in glacier: G_i = max(F1_i, G_{i-1} + p2_{i-1})
    # Finish time of train i in glacier: F2_i = G_i + p2_i
    
    # This is a classic scheduling problem (Johnson's Rule variant).
    # To minimize the completion time of the last job on two machines:
    # Sort trains with p1 < p2 in ascending order of p1.
    # Sort trains with p1 >= p2 in descending order of p2.
    
    group1 = []
    group2 = []
    for p1, p2 in trains:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_forest_finish = 0
    current_glacier_finish = 0
    
    for p1, p2 in order:
        current_forest_finish += p1
        # The train enters the glacier at max(current_forest_finish, current_glacier_finish)
        # current_glacier_finish is the time the previous train finished the glacier
        current_glacier_finish = max(current_forest_finish, current_glacier_finish) + p2
        
    print(current_glacier_finish)

if __name__ == '__main__':
    solve()
