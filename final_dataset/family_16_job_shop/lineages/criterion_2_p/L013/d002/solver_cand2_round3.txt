```python
import sys

def solve():
    # This problem is a variation of Johnson's Rule for two-machine flow shop scheduling.
    # We want to minimize the makespan (the completion time of the last job).
    # Johnson's Rule states that to minimize the makespan:
    # 1. List all jobs.
    # 2. Find the minimum duration among all valley (A) and glacier (B) times.
    # 3. If the minimum is in A, place that job as early as possible.
    # 4. If the minimum is in B, place that job as late as possible.
    # 5. Remove the job and repeat.
    
    # Implementation:
    # Sort jobs such that:
    # - Jobs with A < B are sorted by A ascending.
    # - Jobs with A >= B are sorted by B descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(n):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        jobs.append((a, b))
        idx += 2
        
    group1 = []
    group2 = []
    
    for a, b in jobs:
        if a < b:
            group1.append((a, b))
        else:
            group2.append((a, b))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_jobs = group1 + group2
    
    # Calculate makespan
    # valley_time: time when the current engine finishes the valley ascent
    # glacier_time: time when the current engine finishes the glacier crossing
    valley_time = 0
    glacier_time = 0
    
    for a, b in sorted_jobs:
        valley_time += a
        # The engine can start the glacier crossing at max(valley_time, glacier_time)
        if valley_time > glacier_time:
            glacier_time = valley_time + b
        else:
            glacier_time += b
            
    print(glacier_time)

if __name__ == '__main__':
    solve()
```
