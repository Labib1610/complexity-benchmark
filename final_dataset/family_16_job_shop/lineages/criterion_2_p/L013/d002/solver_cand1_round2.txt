import sys

def solve():
    # The problem is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # We want to minimize the completion time of the last job on the second machine.
    # Let p1_i be the time on machine 1 and p2_i be the time on machine 2.
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where p1_i < p2_i
    #    Set 2: jobs where p1_i >= p2_i
    # 2. Sort Set 1 in ascending order of p1_i.
    # 3. Sort Set 2 in descending order of p2_i.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time B_j
    # A_k = A_{k-1} + p1_k
    # B_k = max(A_k, B_{k-1}) + p2_k
    
    current_a = 0
    current_b = 0
    
    for p1, p2 in sequence:
        current_a += p1
        if current_a > current_b:
            current_b = current_a + p2
        else:
            current_b = current_b + p2
            
    print(current_b)

if __name__ == '__main__':
    solve()
