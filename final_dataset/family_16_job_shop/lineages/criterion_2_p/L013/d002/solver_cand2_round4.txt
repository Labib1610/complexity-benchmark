import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vehicles = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        vehicles.append((p1, p2))
        idx += 2
    
    # Johnson's Rule logic:
    # To minimize the completion time of a two-stage pipeline,
    # sort vehicles such that those with smaller p1 come first,
    # and those with smaller p2 come last.
    # Specifically, sort by min(p1, p2).
    # If p1 < p2, sort by p1 ascending.
    # If p1 >= p2, sort by p2 descending.
    
    group1 = []
    group2 = []
    for p1, p2 in vehicles:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_vehicles = group1 + group2
    
    # Calculate completion time
    # valley_end[k] = valley_end[k-1] + p1_k
    # ridge_end[k] = max(valley_end[k], ridge_end[k-1]) + p2_k
    
    current_valley_end = 0
    current_ridge_end = 0
    
    for p1, p2 in sorted_vehicles:
        current_valley_end += p1
        current_ridge_end = max(current_valley_end, current_ridge_end) + p2
        
    print(current_ridge_end)

if __name__ == '__main__':
    solve()
