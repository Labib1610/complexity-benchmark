```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    trains = []
    for _ in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        trains.append((p1, p2))
    
    # Sort by p2 descending (Johnson's rule for 2-machine flow shop)
    trains.sort(key=lambda x: -x[1])
    
    # Simulate the flow shop
    # A_k: time train k finishes on machine 1 (lower segment)
    # B_k: time train k finishes on machine 2 (upper segment)
    A = 0
    B = 0
    for p1, p2 in trains:
        A += p1
        B = max(A, B) + p2
    
    print(B)

if __name__ == "__main__":
    solve()
```
