```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    first, second = [], []
    for _ in range(n):
        p1 = int(next(it)); p2 = int(next(it))
        if p1 <= p2:
            first.append((p1, p2))
        else:
            second.append((p1, p2))
    # Johnson's rule
    first.sort(key=lambda x: x[0])          # increasing p1
    second.sort(key=lambda x: -x[1])        # decreasing p2
    seq = first + second

    a_prev = 0  # completion on lower
    b_prev = 0  # completion on upper
    for p1, p2 in seq:
        a_curr = a_prev + p1
        b_curr = max(a_curr, b_prev) + p2
        a_prev, b_prev = a_curr, b_curr

    sys.stdout.write(str(b_prev))

if __name__ == "__main__":
    solve()
```
