import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    trains = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        trains.append((p1, p2))
        idx += 2
    
    # Johnson's Rule logic:
    # We want to minimize the completion time of a two-stage flow shop.
    # Trains with p1 < p2 should be processed in increasing order of p1.
    # Trains with p1 >= p2 should be processed in decreasing order of p2.
    
    group1 = []
    group2 = []
    for p1, p2 in trains:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate B_j based on the optimal sequence
    a_k = 0
    b_k = 0
    for p1, p2 in sequence:
        a_k += p1
        b_k = max(a_k, b_k) + p2
        
    print(b_k)

if __name__ == '__main__':
    solve()
