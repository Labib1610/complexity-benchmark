import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    trains = []
    idx = 1
    for _ in range(n):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        trains.append((a, b))
        idx += 2
        
    # Let A_i be the time in Zone Alpha and B_i be the time in Zone Beta.
    # Let S_i be the time the i-th train enters Zone Alpha.
    # S_0 = 0, S_i = S_{i-1} + A_{i-1}.
    # Let E_i be the time the i-th train exits Zone Beta.
    # E_i = max(E_{i-1}, S_i + A_i) + B_i.
    # This is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # To minimize the completion time (makespan), we sort trains:
    # Group 1: A_i < B_i, sorted by A_i ascending.
    # Group 2: A_i >= B_i, sorted by B_i descending.
    
    group1 = []
    group2 = []
    for a, b in trains:
        if a < b:
            group1.append((a, b))
        else:
            group2.append((a, b))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_alpha_exit = 0
    current_beta_exit = 0
    
    for a, b in order:
        current_alpha_exit += a
        # The train enters Beta at max(current_alpha_exit, current_beta_exit)
        # It exits at that time + b
        current_beta_exit = max(current_alpha_exit, current_beta_exit) + b
        
    print(current_beta_exit)

if __name__ == '__main__':
    solve()
