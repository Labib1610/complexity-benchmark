```python
import sys
import threading

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    nodes = []
    index = 1
    for _ in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        nodes.append((p1, p2))
        index += 2
    
    # Sort nodes based on p1 - p2 in ascending order
    # This heuristic works because we want to minimize the maximum of prefix p1 sum + suffix p2 sum
    # Nodes with smaller (p1 - p2) should come earlier to reduce the peak
    nodes.sort(key=lambda x: x[0] - x[1])
    
    # Calculate prefix sums of p1
    prefix_p1 = [0] * (j + 1)
    for i in range(j):
        prefix_p1[i+1] = prefix_p1[i] + nodes[i][0]
    
    # Calculate suffix sums of p2
    suffix_p2 = [0] * (j + 1)
    for i in range(j-1, -1, -1):
        suffix_p2[i] = suffix_p2[i+1] + nodes[i][1]
    
    # Find the minimum peak stress over all possible pivot indices
    min_stress = float('inf')
    for k in range(1, j+1):
        stress = prefix_p1[k] + suffix_p2[k-1]
        if stress < min_stress:
            min_stress = stress
    
    print(min_stress)

# Use threading to increase recursion limit if needed, but not necessary here
threading.Thread(target=main).start()
```
