import sys

def solve():
    # The problem asks to minimize sum(p1) + D_final.
    # Since sum(p1) is constant regardless of the order, we only need to minimize D_final.
    # D_new = max(D_curr - p1, 0) + p2
    # This is a classic scheduling problem.
    # Let the trains be (p1_i, p2_i).
    # If we have two trains (a1, a2) and (b1, b2):
    # Order (a, b): D = max(max(0 - a1, 0) + a2 - b1, 0) + b2 = max(a2 - b1, 0) + b2
    # Order (b, a): D = max(max(0 - b1, 0) + b2 - a1, 0) + a2 = max(b2 - a1, 0) + a2
    # We want to choose the order that results in a smaller D.
    # This is equivalent to Johnson's rule logic.
    # Split trains into two groups:
    # Group 1: p1 >= p2 (trains that reduce D more than they add)
    # Group 2: p1 < p2 (trains that add more to D than they reduce)
    # Sort Group 1 by p1 descending.
    # Sort Group 2 by p2 ascending.
    # Sequence: Group 1 then Group 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    trains = []
    sum_p1 = 0
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        trains.append((p1, p2))
        sum_p1 += p1
        idx += 2
        
    group1 = []
    group2 = []
    for p1, p2 in trains:
        if p1 >= p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    # Sort group 1 by p1 descending
    group1.sort(key=lambda x: x[0], reverse=True)
    # Sort group 2 by p2 ascending
    group2.sort(key=lambda x: x[1])
    
    sequence = group1 + group2
    
    d = 0
    for p1, p2 in sequence:
        d = max(d - p1, 0) + p2
        
    print(sum_p1 + d)

if __name__ == '__main__':
    solve()
