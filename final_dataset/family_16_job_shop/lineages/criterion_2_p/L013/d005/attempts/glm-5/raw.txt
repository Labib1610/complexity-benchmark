```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    packets = []
    for _ in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        packets.append((p1, p2))
    
    # Sort by p1 in descending order
    packets.sort(key=lambda x: -x[0])
    
    # Simulate the process
    # arrival_time is the sum of p1 values of packets processed so far
    # finish_time is when the current packet finishes processing
    arrival_time = 0
    finish_time = 0
    
    for p1, p2 in packets:
        arrival_time += p1
        # Processing starts at max(arrival_time, finish_time)
        start_time = arrival_time if arrival_time > finish_time else finish_time
        finish_time = start_time + p2
    
    print(finish_time)

if __name__ == "__main__":
    solve()
```
