```python
import sys
import threading

def main():
    import sys
    j = int(sys.stdin.readline())
    packets = []
    for _ in range(j):
        p1, p2 = map(int, sys.stdin.readline().split())
        packets.append((p1, p2))
    
    # Sort packets based on p1 - p2 in ascending order
    # This is the optimal strategy to minimize the total completion time
    packets.sort(key=lambda x: x[0] - x[1])
    
    current_time = 0
    completion_time = 0
    
    for i in range(j):
        p1, p2 = packets[i]
        # Arrival time of this packet is current_time + p1
        arrival_time = current_time + p1
        # Processing starts at max(arrival_time, completion_time)
        # where completion_time is the time when the previous packet finished
        start_time = max(arrival_time, completion_time)
        # This packet finishes at start_time + p2
        completion_time = start_time + p2
        # Update current_time for the next packet's arrival calculation
        current_time += p1
    
    print(completion_time)

# Use threading to increase recursion limit for large inputs
threading.Thread(target=main).start()
```
