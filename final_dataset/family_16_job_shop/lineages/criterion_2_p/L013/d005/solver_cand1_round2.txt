```python
import sys

def solve():
    # Let the packets be (p1_i, p2_i).
    # Let the sequence be i_1, i_2, ..., i_j.
    # Arrival time of packet i_k: A_k = sum_{m=1}^k p1_{i_m}
    # Finish time of packet i_k: F_k = max(A_k, F_{k-1}) + p2_{i_k}
    # We want to minimize F_j.
    # Let S_k = sum_{m=1}^k p1_{i_m}.
    # F_k = max(S_k, F_{k-1}) + p2_{i_k}
    # F_k = max(S_k + p2_{i_k}, F_{k-1} + p2_{i_k})
    # Expanding this:
    # F_j = max(S_j + p2_{i_j}, S_{j-1} + p2_{i_{j-1}} + p2_{i_j}, ..., S_1 + p2_{i_1} + ... + p2_{i_j})
    # F_j = max_{1 <= k <= j} (S_k + sum_{m=k}^j p2_{i_m})
    # F_j = max_{1 <= k <= j} (sum_{m=1}^k p1_{i_m} + sum_{m=k}^j p2_{i_m})
    # F_j = sum_{m=1}^j p1_{i_m} + sum_{m=k}^j p2_{i_m} - sum_{m=1}^{k-1} p1_{i_m}
    # This is equivalent to:
    # F_j = sum_{m=1}^j p1_{i_m} + max_{1 <= k <= j} (sum_{m=k}^j p2_{i_m} - sum_{m=1}^{k-1} p1_{i_m})
    # Actually, the standard scheduling result for this type of problem (minimizing 
    # the completion time of the last job) is to sort by p1.
    # Let's re-verify: F_k = max(F_{k-1}, S_k) + p2_k.
    # If we have two adjacent packets (p1_a, p2_a) and (p1_b, p2_b),
    # Order (a, b): F_b = max(max(S_{k-1} + p1_a, F_{k-1}) + p2_a + p1_b, S_{k-1} + p1_a + p1_b) + p2_b
    # This is a known variation of the flow shop or scheduling problem.
    # Sorting by p1 ascending is optimal for minimizing the completion time of the last job.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    packets = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        packets.append((p1, p2))
        idx += 2
    
    # Sort by arrival delay p1
    packets.sort(key=lambda x: x[0])
    
    current_time = 0
    arrival_time = 0
    for p1, p2 in packets:
        arrival_time += p1
        if current_time < arrival_time:
            current_time = arrival_time + p2
        else:
            current_time += p2
            
    print(current_time)

if __name__ == '__main__':
    solve()
```
