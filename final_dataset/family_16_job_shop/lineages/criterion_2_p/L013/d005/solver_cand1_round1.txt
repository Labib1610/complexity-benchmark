```python
import sys

def solve():
    # Let the sequence of nodes be (a1, b1), (a2, b2), ..., (aj, bj).
    # Pivot cost at k: C_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # C_k = sum_{i=1}^j b_i + sum_{i=1}^k (a_i - b_i)
    # Let S = sum_{i=1}^j b_i. Then C_k = S + sum_{i=1}^k (a_i - b_i).
    # We want to minimize max_{k=1..j} (S + sum_{i=1}^k (a_i - b_i)).
    # Let d_i = a_i - b_i. We want to minimize max_{k=1..j} (S + sum_{i=1}^k d_i).
    # This is equivalent to minimizing the maximum prefix sum of the sequence d_1, ..., d_j.
    # To minimize the maximum prefix sum, we should sort the d_i values.
    # If we sort d_i such that d_1 <= d_2 <= ... <= d_j, the prefix sums are
    # P_k = sum_{i=1}^k d_i.
    # Since d_i are sorted, the prefix sums are convex-like.
    # Actually, the optimal strategy to minimize the maximum prefix sum is to 
    # put the smallest d_i values first.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    nodes = []
    idx = 1
    total_b = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        nodes.append(p1 - p2)
        total_b += p2
        idx += 2
    
    nodes.sort()
    
    max_stress = -float('inf')
    current_prefix_sum = 0
    for d in nodes:
        current_prefix_sum += d
        stress = total_b + current_prefix_sum
        if stress > max_stress:
            max_stress = stress
            
    print(max_stress)

if __name__ == '__main__':
    solve()
```
