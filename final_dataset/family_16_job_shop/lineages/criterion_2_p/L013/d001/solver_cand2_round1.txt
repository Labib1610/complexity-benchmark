import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    trains = []
    idx = 1
    for _ in range(j):
        l = int(input_data[idx])
        u = int(input_data[idx+1])
        trains.append((l, u))
        idx += 2
    
    # This is a variation of Johnson's Rule for scheduling.
    # We want to minimize the completion time of the last job on a two-machine flow shop.
    # The optimal sequence is to sort trains such that:
    # If L < U, they go first (sorted by L ascending).
    # If L >= U, they go last (sorted by U descending).
    
    group1 = []
    group2 = []
    for l, u in trains:
        if l < u:
            group1.append((l, u))
        else:
            group2.append((l, u))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate completion time
    # current_lower_time: time when the current train finishes the Lower Track
    # current_upper_time: time when the current train finishes the Upper Track
    current_lower_time = 0
    current_upper_time = 0
    
    for l, u in sequence:
        current_lower_time += l
        # The train can start the Upper Track at max(current_lower_time, current_upper_time)
        if current_lower_time > current_upper_time:
            current_upper_time = current_lower_time + u
        else:
            current_upper_time += u
            
    print(current_upper_time)

if __name__ == '__main__':
    solve()
