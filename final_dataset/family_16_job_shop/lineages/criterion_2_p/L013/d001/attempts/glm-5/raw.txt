```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    trains = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        idx += 2
        trains.append((p1, p2))
        
    # Sort by Johnson's rule: 
    # p1 <= p2 in ascending order of p1, 
    # p1 > p2 in descending order of p2
    trains.sort(key=lambda x: (x[0] <= x[1], x[0] if x[0] <= x[1] else -x[1]))
    
    t1 = 0
    t2 = 0
    for p1, p2 in trains:
        t1 += p1
        if t1 > t2:
            t2 = t1 + p2
        else:
            t2 += p2
            
    print(t2)

if __name__ == '__main__':
    solve()
```
