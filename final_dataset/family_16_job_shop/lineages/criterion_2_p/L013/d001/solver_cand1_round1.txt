```python
import sys

def solve():
    # The problem is a classic application of Johnson's Rule for the 
    # two-machine flow shop scheduling problem.
    # We want to minimize the makespan (the time the last task finishes).
    # Johnson's Rule:
    # 1. Divide tasks into two sets:
    #    Set A: tasks where p1 < p2
    #    Set B: tasks where p1 >= p2
    # 2. Sort Set A in ascending order of p1.
    # 3. Sort Set B in descending order of p2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    tasks = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tasks.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in tasks:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate makespan
    # time_p: time when preparation bay finishes current task
    # time_f: time when finishing bay finishes current task
    time_p = 0
    time_f = 0
    
    for p1, p2 in sequence:
        time_p += p1
        # Finishing bay can start only after preparation is done
        # and after the finishing bay is free from the previous task
        time_f = max(time_p, time_f) + p2
        
    print(time_f)

if __name__ == '__main__':
    solve()
```
