import sys

def solve():
    # The problem asks to minimize the baseline stress + sum(discharging durations).
    # Let p1_i be charging duration and p2_i be discharging duration for node i.
    # Let the sequence be S_1, S_2, ..., S_j.
    # Transient peak at step k is: (sum_{i=1}^{k-1} (p1_i - p2_i)) + p1_k.
    # Baseline stress = max_{k=1..j} (sum_{i=1}^{k-1} (p1_i - p2_i) + p1_k).
    # Total peak stress = max_{k=1..j} (sum_{i=1}^{k-1} (p1_i - p2_i) + p1_k) + sum_{i=1}^j p2_i.
    # Let D_i = p1_i - p2_i.
    # Total peak stress = max_{k=1..j} (sum_{i=1}^{k-1} D_i + p1_k) + sum_{i=1}^j p2_i.
    # Note that sum_{i=1}^{k-1} D_i + p1_k = sum_{i=1}^{k-1} (p1_i - p2_i) + p1_k
    # = sum_{i=1}^{k} p1_i - sum_{i=1}^{k-1} p2_i.
    # This is equivalent to: (sum_{i=1}^j p1_i) - (sum_{i=k+1}^j p1_i) - (sum_{i=1}^{k-1} p2_i).
    # Total peak stress = max_{k=1..j} (sum_{i=1}^k p1_i - sum_{i=1}^{k-1} p2_i) + sum_{i=1}^j p2_i.
    # Let S_p1 = sum_{i=1}^j p1_i and S_p2 = sum_{i=1}^j p2_i.
    # Total peak stress = max_{k=1..j} (sum_{i=1}^k p1_i + sum_{i=k}^j p2_i) - p2_k + S_p2 - p2_k? No.
    # Let's re-evaluate:
    # Peak_k = (sum_{i=1}^{k-1} p1_i - sum_{i=1}^{k-1} p2_i) + p1_k
    # Total = max_k (sum_{i=1}^{k-1} p1_i - sum_{i=1}^{k-1} p2_i + p1_k) + sum_{i=1}^j p2_i
    # Total = max_k (sum_{i=1}^k p1_i - sum_{i=1}^{k-1} p2_i) + sum_{i=1}^j p2_i
    # Total = max_k (sum_{i=1}^k p1_i + sum_{i=k}^j p2_i)
    # We want to minimize max_k (sum_{i=1}^k p1_i + sum_{i=k}^j p2_i).
    # This is a classic scheduling problem. We sort nodes by p1_i < p2_i.
    # Nodes with p1_i < p2_i should come first, sorted by p1_i ascending.
    # Nodes with p1_i >= p2_i should come last, sorted by p2_i descending.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    nodes = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        nodes.append((p1, p2))
        idx += 2
    
    group1 = [n for n in nodes if n[0] < n[1]]
    group2 = [n for n in nodes if n[0] >= n[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    max_val = 0
    current_p1_sum = 0
    # We need max_k (sum_{i=1}^k p1_i + sum_{i=k}^j p2_i)
    # Let S_p2_total = sum_{i=1}^j p2_i
    # sum_{i=k}^j p2_i = S_p2_total - sum_{i=1}^{k-1} p2_i
    # So we want max_k (sum_{i=1}^k p1_i - sum_{i=1}^{k-1} p2_i) + S_p2_total
    
    s_p2_total = sum(n[1] for n in nodes)
    current_p1 = 0
    current_p2 = 0
    ans = 0
    for i in range(j):
        current_p1 += sequence[i][0]
        # Peak at step i+1 is current_p1 - current_p2
        ans = max(ans, current_p1 - current_p2)
        current_p2 += sequence[i][1]
        
    print(ans + s_p2_total)

solve()
