We can solve this by sorting the jobs using Johnson’s rule for two-machine flow shop scheduling.  
The rule:  
- Jobs with `p1 <= p2` are sorted by increasing `p1`.  
- Jobs with `p1 > p2` are sorted by decreasing `p2`.  
- Concatenate the two groups.

Then simulate the schedule to get the total completion time.

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    j = int(data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx+1])
        idx += 2
        jobs.append((p1, p2))
    
    first = []
    second = []
    for p1, p2 in jobs:
        if p1 <= p2:
            first.append((p1, p2))
        else:
            second.append((p1, p2))
    
    first.sort(key=lambda x: x[0])
    second.sort(key=lambda x: x[1], reverse=True)
    
    sequence = first + second
    
    t1 = 0
    t2 = 0
    for p1, p2 in sequence:
        t1 += p1
        if t2 < t1:
            t2 = t1
        t2 += p2
    
    print(t2)

if __name__ == "__main__":
    main()
```
