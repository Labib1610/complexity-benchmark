We can solve this using Johnson’s rule for two-machine flow shop scheduling.  
The idea is to sort jobs:  
- If `p1 < p2`, put them in the front, sorted by `p1` ascending.  
- If `p1 >= p2`, put them in the back, sorted by `p2` descending.  

Then simulate the schedule to get the completion time.

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    j = int(data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx+1])
        idx += 2
        jobs.append((p1, p2))
    
    first = []
    second = []
    for p1, p2 in jobs:
        if p1 < p2:
            first.append((p1, p2))
        else:
            second.append((p1, p2))
    
    first.sort(key=lambda x: x[0])
    second.sort(key=lambda x: -x[1])
    
    sequence = first + second
    
    mill_time = 0
    polish_time = 0
    for p1, p2 in sequence:
        mill_time += p1
        if polish_time < mill_time:
            polish_time = mill_time
        polish_time += p2
    
    print(polish_time)

if __name__ == "__main__":
    main()
```
