```python
import sys

def solve():
    # This is a classic application of Johnson's Rule for the two-machine flow shop problem.
    # To minimize the makespan (the time the last part finishes on the second machine),
    # we divide the parts into two sets:
    # Set 1: Parts where p1 < p2. Sort these by p1 in ascending order.
    # Set 2: Parts where p1 >= p2. Sort these by p2 in descending order.
    # The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    parts = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        parts.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for p in parts:
        if p[0] < p[1]:
            set1.append(p)
        else:
            set2.append(p)
            
    # Sort set1 by p1 ascending
    set1.sort(key=lambda x: x[0])
    # Sort set2 by p2 descending
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    # milling_time: time when current part finishes on Milling Machine
    # polishing_time: time when current part finishes on Polishing Machine
    milling_time = 0
    polishing_time = 0
    
    for p1, p2 in sequence:
        milling_time += p1
        # Polishing machine starts at max(milling_time, previous_polishing_finish)
        if milling_time > polishing_time:
            polishing_time = milling_time + p2
        else:
            polishing_time += p2
            
    print(polishing_time)

if __name__ == '__main__':
    solve()
```
