```python
import sys

def solve():
    # The grid score for a given arrangement is the maximum of:
    # (sum of all top cells) + (bottom cell of column k) + (sum of bottom cells from k+1 to j)
    # for all k in 1..j.
    # Let T_i be the top cell of column i, B_i be the bottom cell of column i.
    # Let S_T = sum(T_i) and S_B = sum(B_i).
    # The path cost if we move down at column k is:
    # sum(T_1...T_k) + sum(B_k...B_j)
    # = sum(T_1...T_k) + sum(B_k...B_j)
    # = sum(T_1...T_j) - sum(T_{k+1}...T_j) + sum(B_k...B_j)
    # = S_T + sum(B_k...B_j) - sum(T_{k+1}...T_j)
    # Let D_i = B_i - T_i.
    # The cost is S_T + B_k + sum_{i=k+1}^j (B_i - T_i) = S_T + B_k + sum_{i=k+1}^j D_i.
    # To minimize the maximum of these values, we should sort the columns.
    # Let's re-examine:
    # Path cost at column k: (sum_{i=1}^k T_i) + (sum_{i=k}^j B_i)
    # = (sum_{i=1}^j T_i) + (sum_{i=k}^j B_i) - (sum_{i=k+1}^j T_i)
    # = S_T + B_k + sum_{i=k+1}^j (B_i - T_i)
    # Let D_i = B_i - T_i. We want to minimize max_k (S_T + B_k + sum_{i=k+1}^j D_i).
    # This is equivalent to minimizing max_k (B_k + sum_{i=k+1}^j D_i).
    # If we sort columns by D_i in descending order, we minimize the maximum.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    cols = []
    idx = 1
    sum_t = 0
    for _ in range(j):
        t = int(input_data[idx])
        b = int(input_data[idx+1])
        cols.append((t, b, b - t))
        sum_t += t
        idx += 2
    
    # Sort by D_i = B_i - T_i descending
    cols.sort(key=lambda x: x[2], reverse=True)
    
    # Calculate suffix sums of D_i
    d_vals = [c[2] for c in cols]
    suffix_d = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suffix_d[i] = suffix_d[i+1] + d_vals[i]
        
    max_path = 0
    for k in range(j):
        # Path cost moving down at column k (0-indexed)
        # sum(T_0...T_k) + sum(B_k...B_{j-1})
        # = sum(T_0...T_{j-1}) + sum(B_k...B_{j-1}) - sum(T_{k+1}...T_{j-1})
        # = sum_t + B_k + sum_{i=k+1}^{j-1} (B_i - T_i)
        # = sum_t + cols[k][1] + suffix_d[k+1]
        current_path = sum_t + cols[k][1] + suffix_d[k+1]
        if k == 0 or current_path > max_path:
            max_path = current_path
            
    print(max_path)

solve()
```
