```python
import sys

def solve():
    # The cost function is S_k = sum_{m=1}^k p1_{v_m} + sum_{m=k}^j p2_{v_m}
    # S_k = sum_{m=1}^j p2_{v_m} + sum_{m=1}^k p1_{v_m} - sum_{m=1}^{k-1} p2_{v_m}
    # Let P1 = sum(p1_i) and P2 = sum(p2_i).
    # S_k = P2 + sum_{m=1}^k (p1_{v_m} - p2_{v_m}) + p2_{v_k}
    # Actually, let's rewrite:
    # S_k = (sum_{m=1}^k p1_{v_m}) + (sum_{m=k}^j p2_{v_m})
    # S_k = (sum_{m=1}^k p1_{v_m}) + (sum_{m=1}^j p2_{v_m}) - (sum_{m=1}^{k-1} p2_{v_m})
    # Let P2_total = sum_{i=1}^j p2_i.
    # S_k = P2_total + p1_{v_k} + sum_{m=1}^{k-1} (p1_{v_m} - p2_{v_m})
    
    # To minimize the maximum of these, we should sort the elements.
    # Consider two adjacent elements i and j.
    # If we have ... i, j ...
    # S_k(i) = ... + p1_i + (p2_i + p2_j + ...)
    # S_{k+1}(j) = ... + p1_i + p1_j + (p2_j + ...)
    # If we swap them to ... j, i ...
    # S_k(j) = ... + p1_j + (p2_j + p2_i + ...)
    # S_{k+1}(i) = ... + p1_j + p1_i + (p2_i + ...)
    # The max of the two positions is:
    # Case 1: max(p1_i + p2_i + p2_j, p1_i + p1_j + p2_j) = p1_i + p2_j + max(p2_i, p1_j)
    # Case 2: max(p1_j + p2_j + p2_i, p1_j + p1_i + p2_i) = p1_j + p2_i + max(p2_j, p1_i)
    # We prefer i before j if p1_i + p2_j + max(p2_i, p1_j) < p1_j + p2_i + max(p2_j, p1_i)
    # This is a standard sorting criterion: sort by min(p1, p2).
    # Specifically, if p1_i < p2_i and p1_j < p2_j, sort by p1 ascending.
    # If p1_i >= p2_i and p1_j >= p2_j, sort by p2 descending.
    # If p1_i < p2_i and p1_j >= p2_j, i comes first.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    elements = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        elements.append((p1, p2))
        idx += 2
        
    # Sort elements:
    # Group 1: p1 < p2, sort by p1 ascending
    # Group 2: p1 >= p2, sort by p2 descending
    g1 = [e for e in elements if e[0] < e[1]]
    g2 = [e for e in elements if e[0] >= e[1]]
    g1.sort(key=lambda x: x[0])
    g2.sort(key=lambda x: x[1], reverse=True)
    seq = g1 + g2
    
    p2_total = sum(e[1] for e in seq)
    current_p1_sum = 0
    current_p2_sum = 0
    max_s = 0
    
    for p1, p2 in seq:
        current_p1_sum += p1
        s_k = current_p1_sum + p2_total - current_p2_sum
        if s_k > max_s:
            max_s = s_k
        current_p2_sum += p2
        
    print(max_s)

solve()
```
