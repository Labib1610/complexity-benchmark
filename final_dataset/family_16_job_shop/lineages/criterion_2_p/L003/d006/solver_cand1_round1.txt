```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tiles.append((p1, p2))
        idx += 2
    
    # Let the tiles be (a_i, b_i).
    # A path moving down at column k has cost:
    # sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # = sum_{i=1}^j a_i + sum_{i=k}^j b_i - sum_{i=1}^{k-1} a_i
    # Let S_a = sum(a_i), S_b = sum(b_i).
    # Cost(k) = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # We want to minimize max_k Cost(k).
    # Note that Cost(k+1) - Cost(k) = a_{k+1} - b_k.
    # To minimize the maximum, we want the sequence of costs to be as flat as possible.
    # This is a classic problem: we want to order tiles such that the prefix sums
    # of (a_i - b_{i-1}) are minimized.
    # Actually, the optimal strategy is to sort tiles such that those with
    # a_i < b_i come first (sorted by a_i ascending), and those with a_i >= b_i
    # come last (sorted by b_i descending).
    
    group1 = []
    group2 = []
    for p1, p2 in tiles:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    ordered_tiles = group1 + group2
    
    # Calculate max path cost for this arrangement
    # Cost(k) = prefix_a[k] + suffix_b[k]
    # where prefix_a[k] = sum(a_0...a_{k-1}), suffix_b[k] = sum(b_{k-1}...b_{j-1})
    
    n = len(ordered_tiles)
    a = [t[0] for t in ordered_tiles]
    b = [t[1] for t in ordered_tiles]
    
    prefix_a = [0] * (n + 1)
    for i in range(n):
        prefix_a[i+1] = prefix_a[i] + a[i]
        
    suffix_b = [0] * (n + 1)
    for i in range(n - 1, -1, -1):
        suffix_b[i] = suffix_b[i+1] + b[i]
        
    max_cost = 0
    for k in range(1, n + 1):
        current_cost = prefix_a[k] + suffix_b[k-1]
        if current_cost > max_cost:
            max_cost = current_cost
            
    print(max_cost)

if __name__ == '__main__':
    solve()
```
