```python
import sys

def solve():
    # The problem asks to minimize the completion time of the last dataset.
    # Let the datasets be (a_i, b_i) where a_i is p1 and b_i is p2.
    # If we process datasets in order 1, 2, ..., j:
    # Let E_i be the time the i-th dataset finishes in the Extractor.
    # E_i = sum_{k=1}^i a_k
    # Let T_i be the time the i-th dataset finishes in the Transformer.
    # T_i = max(T_{i-1}, E_i) + b_i, with T_0 = 0.
    # This is equivalent to Johnson's Rule for a two-machine flow shop problem.
    # To minimize the makespan, we sort datasets:
    # 1. Datasets with a_i < b_i are sorted by increasing a_i.
    # 2. Datasets with a_i >= b_i are sorted by decreasing b_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    datasets = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        datasets.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    for p1, p2 in datasets:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_extractor_time = 0
    current_transformer_time = 0
    
    for p1, p2 in order:
        current_extractor_time += p1
        current_transformer_time = max(current_transformer_time, current_extractor_time) + p2
        
    print(current_transformer_time)

if __name__ == '__main__':
    solve()
```
