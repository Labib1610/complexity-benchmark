```python
import sys

def solve():
    # The problem asks to minimize the maximum path cost.
    # Let the tiles be (a_i, b_i).
    # A path moving down at column k has cost:
    # sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # Let S_a[k] = sum_{i=1}^k a_i and S_b[k] = sum_{i=k}^j b_i.
    # The cost is S_a[k] + S_b[k].
    # Note that S_a[k] + S_b[k] = sum_{i=1}^j a_i + sum_{i=k}^j b_i - sum_{i=k}^j a_i
    # = sum_{i=1}^j a_i + sum_{i=k}^j (b_i - a_i).
    # Let D_i = b_i - a_i. The cost is (sum a_i) + sum_{i=k}^j D_i.
    # To minimize the maximum of these costs, we want to arrange D_i such that
    # the partial sums of D_i are as balanced as possible.
    # Specifically, we want to minimize max_k (sum_{i=k}^j D_i).
    # This is equivalent to minimizing the maximum suffix sum of the sequence D.
    # By rearranging D_i, the minimum possible maximum suffix sum is achieved
    # by sorting D_i in non-decreasing order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    tiles = []
    idx = 1
    sum_a = 0
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        tiles.append(b - a)
        sum_a += a
        idx += 2
        
    # Sort D_i = b_i - a_i in non-decreasing order.
    tiles.sort()
    
    # The costs are C_k = sum_a + sum_{i=k}^j D_i (using 1-based indexing for k).
    # With sorted D_i, the suffix sums are:
    # Suffix_j = D_j
    # Suffix_{j-1} = D_j + D_{j-1}
    # ...
    # Suffix_1 = sum(D)
    # The maximum of these is the maximum suffix sum.
    
    current_suffix_sum = 0
    max_suffix_sum = 0
    for i in range(j - 1, -1, -1):
        current_suffix_sum += tiles[i]
        if current_suffix_sum > max_suffix_sum:
            max_suffix_sum = current_suffix_sum
            
    print(sum_a + max_suffix_sum)

if __name__ == '__main__':
    solve()
```
