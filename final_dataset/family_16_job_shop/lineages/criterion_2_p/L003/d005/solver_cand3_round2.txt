import sys

def solve():
    # This problem is a variation of Johnson's Rule for the two-machine flow shop scheduling problem.
    # The goal is to minimize the makespan (the completion time of the last job).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    # Sort set1 by p1 ascending
    set1.sort(key=lambda x: x[0])
    # Sort set2 by p2 descending
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    current_a = 0
    current_b = 0
    
    for p1, p2 in sequence:
        current_a += p1
        if current_a > current_b:
            current_b = current_a + p2
        else:
            current_b = current_b + p2
            
    print(current_b)

if __name__ == '__main__':
    solve()
