We can solve this by sorting the jobs using a known rule for two-machine flow shop scheduling:  

- If `p1 < p2`, sort by `p1` ascending.  
- If `p1 > p2`, sort by `p2` descending.  
- If equal, either order works.  

This is Johnson’s rule. Then simulate the schedule to compute the completion time.

```python
import sys

def solve():
    input = sys.stdin.readline
    j = int(input())
    jobs = []
    for _ in range(j):
        p1, p2 = map(int, input().split())
        jobs.append((p1, p2))
    
    # Johnson's rule
    first = []
    second = []
    for p1, p2 in jobs:
        if p1 < p2:
            first.append((p1, p2))
        else:
            second.append((p1, p2))
    
    first.sort(key=lambda x: x[0])
    second.sort(key=lambda x: -x[1])
    sequence = first + second
    
    time_main = 0
    time_test = 0
    for p1, p2 in sequence:
        time_main += p1
        if time_test < time_main:
            time_test = time_main
        time_test += p2
    
    print(time_test)

if __name__ == "__main__":
    solve()
```
