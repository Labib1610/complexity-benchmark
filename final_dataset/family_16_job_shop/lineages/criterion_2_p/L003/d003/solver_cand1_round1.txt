import sys

def solve():
    # The stress level at position k is:
    # S_k = (sum of p1 from 1 to k) + (sum of p2 from k to j)
    # Let P1_sum = sum of all p1, P2_sum = sum of all p2.
    # Let G_i be the gain of stage at position i, D_i be the drop of stage at position i.
    # S_k = (G_1 + ... + G_k) + (D_k + ... + D_j)
    # S_k = (G_1 + ... + G_k) + (P2_sum - (D_1 + ... + D_{k-1}))
    # S_k = P2_sum + (G_1 - D_1) + (G_2 - D_2) + ... + (G_{k-1} - D_{k-1}) + G_k
    # Let x_i = G_i - D_i.
    # S_k = P2_sum + G_k + sum_{i=1}^{k-1} x_i
    # To minimize the maximum S_k, we should sort the stages.
    # If we have a sequence, the stress levels are:
    # S_1 = P2_sum + G_1
    # S_2 = P2_sum + G_2 + x_1
    # S_3 = P2_sum + G_3 + x_1 + x_2
    # ...
    # S_k = P2_sum + G_k + sum_{i=1}^{k-1} x_i
    # This is equivalent to minimizing the maximum of prefix sums of a modified sequence.
    # The optimal strategy is to sort stages by x_i = G_i - D_i in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    stages = []
    idx = 1
    p2_sum = 0
    for _ in range(j):
        g = int(input_data[idx])
        d = int(input_data[idx+1])
        stages.append((g, d))
        p2_sum += d
        idx += 2
        
    # Sort by G_i - D_i descending
    stages.sort(key=lambda x: x[0] - x[1], reverse=True)
    
    max_stress = 0
    current_prefix_x = 0
    for i in range(j):
        g, d = stages[i]
        stress = p2_sum + current_prefix_x + g
        if stress > max_stress:
            max_stress = stress
        current_prefix_x += (g - d)
        
    print(max_stress)

if __name__ == '__main__':
    solve()
