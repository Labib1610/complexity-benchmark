import sys

def solve():
    # The problem is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # We want to minimize the makespan.
    # Let the chapters be (p1_i, p2_i).
    # Sort chapters into two groups:
    # Group 1: p1_i < p2_i, sorted by p1_i ascending.
    # Group 2: p1_i >= p2_i, sorted by p2_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    chapters = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        chapters.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in chapters:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate makespan
    # Translation finishes chapter i at time T1_i = sum(p1_1...p1_i)
    # Proofreading finishes chapter i at time T2_i = max(T1_i, T2_{i-1}) + p2_i
    
    t1 = 0
    t2 = 0
    for p1, p2 in sequence:
        t1 += p1
        if t1 > t2:
            t2 = t1 + p2
        else:
            t2 = t2 + p2
            
    print(t2)

if __name__ == '__main__':
    solve()
