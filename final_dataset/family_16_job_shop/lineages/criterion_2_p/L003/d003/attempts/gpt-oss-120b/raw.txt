```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    first = []   # p1 <= p2
    second = []  # p1 > p2
    for _ in range(n):
        p1 = int(next(it))
        p2 = int(next(it))
        if p1 <= p2:
            first.append((p1, p2))
        else:
            second.append((p1, p2))
    # Johnson's rule
    first.sort(key=lambda x: x[0])          # increasing p1
    second.sort(key=lambda x: -x[1])        # decreasing p2
    seq = first + second

    time_a = 0
    time_b = 0
    for p1, p2 in seq:
        time_a += p1
        start_b = time_a if time_a > time_b else time_b
        time_b = start_b + p2
    print(time_b)

if __name__ == "__main__":
    solve()
```
