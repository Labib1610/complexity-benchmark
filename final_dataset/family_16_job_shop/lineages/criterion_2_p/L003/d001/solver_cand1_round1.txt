import sys

def solve():
    # This is a variation of Johnson's Rule for the two-machine flow shop problem.
    # We want to minimize the makespan (Cmax).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    order = set1 + set2
    
    # Calculate completion time
    # fabrication_time: time when current job finishes fabrication
    # assembly_time: time when current job finishes assembly
    fabrication_time = 0
    assembly_time = 0
    
    for p1, p2 in order:
        fabrication_time += p1
        # Assembly starts at max(fabrication_time, previous_assembly_time)
        if fabrication_time > assembly_time:
            assembly_time = fabrication_time + p2
        else:
            assembly_time += p2
            
    print(assembly_time)

if __name__ == '__main__':
    solve()
