```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx]); p2 = int(data[idx+1])
        jobs.append((p1, p2))
        idx += 2
    
    # Johnson's rule for two-machine flow shop
    # Group 1: p1 <= p2, sort by p1 ascending
    # Group 2: p1 > p2, sort by p2 descending
    group1 = []
    group2 = []
    for p1, p2 in jobs:
        if p1 <= p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    fab_time = 0
    completion = 0
    for p1, p2 in order:
        fab_time += p1
        # assembly starts at max(fab_time, previous completion)
        if fab_time > completion:
            completion = fab_time + p2
        else:
            completion = completion + p2
    
    print(completion)

solve()
```
