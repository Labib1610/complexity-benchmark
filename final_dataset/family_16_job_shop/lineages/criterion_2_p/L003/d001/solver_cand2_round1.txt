```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    plays = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        plays.append((p1, p2))
        idx += 2
    
    # Let S_i be the stage time and L_i be the lobby time for play i.
    # If we order plays as 1, 2, ..., j:
    # Stage finish times: T_i = sum_{k=1}^i S_k
    # Lobby finish times: F_i = max(F_{i-1}, T_i) + L_i
    # This is equivalent to Johnson's rule logic for minimizing makespan.
    # To minimize the final F_j, we should sort plays such that those with
    # smaller S_i come first, and those with larger L_i come later.
    # Specifically, sort by min(S_i, L_i). If S_i < L_i, put it early.
    # If S_i >= L_i, put it late.
    
    group1 = []
    group2 = []
    for p in plays:
        if p[0] < p[1]:
            group1.append(p)
        else:
            group2.append(p)
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_plays = group1 + group2
    
    current_stage_time = 0
    current_lobby_finish = 0
    
    for s, l in sorted_plays:
        current_stage_time += s
        # The lobby discussion starts at max(current_stage_time, current_lobby_finish)
        current_lobby_finish = max(current_stage_time, current_lobby_finish) + l
        
    print(current_lobby_finish)

if __name__ == '__main__':
    solve()
```
