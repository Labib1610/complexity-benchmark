```python
import sys

def solve():
    # This problem is a variation of Johnson's Rule for the two-machine flow shop scheduling problem.
    # The goal is to minimize the makespan (the time the last job finishes on the second machine).
    # Johnson's Rule states:
    # 1. List all jobs.
    # 2. Divide jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 3. Sort Set A in ascending order of p1.
    # 4. Sort Set B in descending order of p2.
    # 5. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate the completion time
    # mixer_time: time when the mixer finishes the current job
    # froster_time: time when the froster finishes the current job
    mixer_time = 0
    froster_time = 0
    
    for p1, p2 in sequence:
        mixer_time += p1
        # The froster can start only when the mixer finishes the current job
        # AND the froster is free from the previous job.
        if froster_time < mixer_time:
            froster_time = mixer_time + p2
        else:
            froster_time += p2
            
    print(froster_time)

if __name__ == '__main__':
    solve()
```
