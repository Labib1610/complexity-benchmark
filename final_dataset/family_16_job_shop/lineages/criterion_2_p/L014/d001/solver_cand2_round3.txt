```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (oven and icing station).
    # To minimize the total completion time (makespan), we sort the jobs:
    # 1. Jobs where p1 < p2 should be processed in increasing order of p1.
    # 2. Jobs where p1 >= p2 should be processed in decreasing order of p2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    orders = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        orders.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in orders:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    # Sort group1 by p1 ascending
    group1.sort(key=lambda x: x[0])
    # Sort group2 by p2 descending
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    current_x = 0
    current_y = 0
    
    for p1, p2 in sequence:
        current_x += p1
        current_y = max(current_x, current_y) + p2
        
    print(current_y)

if __name__ == '__main__':
    solve()
```
