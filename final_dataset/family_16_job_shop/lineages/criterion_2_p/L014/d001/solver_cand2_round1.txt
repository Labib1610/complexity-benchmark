```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    blocks = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        blocks.append((p1, p2))
        idx += 2
    
    # The score of a grid is max(sum(top[0...i]) + sum(bottom[i...j-1])) for all i in 0..j-1.
    # Let S1 = sum(p1_i) and S2 = sum(p2_i).
    # The path cost if we go down at column i (0-indexed) is:
    # sum(p1[0...i]) + sum(p2[i...j-1])
    # = sum(p1[0...i]) + (S2 - sum(p2[0...i-1]))
    # Let diff_i = p1_i - p2_i.
    # Then the cost at column i is S2 + sum_{k=0}^i (p1_k - p2_k) = S2 + sum_{k=0}^i diff_k.
    # To minimize the maximum of these values, we should sort the blocks such that
    # the prefix sums of (p1 - p2) are as small as possible.
    # This is achieved by sorting blocks by (p1 - p2) in ascending order.
    
    blocks.sort(key=lambda x: x[0] - x[1])
    
    s2 = sum(b[1] for b in blocks)
    
    current_diff_sum = 0
    max_score = -float('inf')
    
    for p1, p2 in blocks:
        current_diff_sum += (p1 - p2)
        max_score = max(max_score, s2 + current_diff_sum)
        
    print(max_score)

if __name__ == '__main__':
    solve()
```
