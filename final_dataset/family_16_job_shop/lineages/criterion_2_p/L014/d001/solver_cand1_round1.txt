```python
import sys

def solve():
    # The problem asks us to order j items (p1_i, p2_i) to minimize:
    # max_{1 <= k <= j} (sum_{i=1}^k p1_i + sum_{i=k}^j p2_i)
    # Let S1 = sum(p1_i) and S2 = sum(p2_i).
    # The expression is: sum_{i=1}^k p1_i + p2_k + sum_{i=k+1}^j p2_i
    # = sum_{i=1}^k p1_i + p2_k + (S2 - sum_{i=1}^k p2_i)
    # = S2 + sum_{i=1}^k (p1_i - p2_i) + p2_k
    # Let d_i = p1_i - p2_i.
    # We want to minimize max_{1 <= k <= j} (S2 + sum_{i=1}^k d_i + p2_k).
    # This is equivalent to minimizing max_{1 <= k <= j} (sum_{i=1}^k d_i + p2_k).
    
    # Let the chosen order be a permutation of indices.
    # We want to minimize max_{k} (prefix_sum_d[k] + p2_k).
    # This is a classic scheduling problem. If we have two adjacent items i and j,
    # swapping them changes the max penalty.
    # Comparing (d_i + p2_i, d_i + d_j + p2_j) vs (d_j + p2_j, d_j + d_i + p2_i).
    # The optimal order is to sort by p2_i in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    orders = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        orders.append((p1, p2))
        idx += 2
        
    # Sort by p2 descending
    orders.sort(key=lambda x: x[1], reverse=True)
    
    s2 = sum(x[1] for x in orders)
    
    current_d_sum = 0
    max_penalty = 0
    
    for i in range(j):
        p1, p2 = orders[i]
        current_d_sum += (p1 - p2)
        penalty = s2 + current_d_sum + p2
        if i == 0 or penalty > max_penalty:
            max_penalty = penalty
            
    print(max_penalty)

if __name__ == '__main__':
    solve()
```
