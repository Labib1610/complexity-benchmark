import sys

def solve():
    # The problem asks to minimize the Hauler's finish time.
    # Let the checkpoints be (s_i, h_i).
    # The Scout visits them in some permutation p_1, p_2, ..., p_j.
    # The Hauler visits them in some permutation q_1, q_2, ..., q_j.
    # Scout finishes inspection at checkpoint q_k at time:
    # S(q_k) = sum_{i=1 to index_of_q_k_in_p} s_{p_i}
    # Hauler starts extraction at q_k at time:
    # H_start(q_k) = max(S(q_k), H_finish(q_{k-1}))
    # H_finish(q_k) = H_start(q_k) + h_{q_k}
    
    # To minimize the total time, we should order the Scout's visits such that
    # the Scout finishes inspections as early as possible for the checkpoints
    # the Hauler visits early.
    # Actually, the total time is sum(h_i) + max_k (S(q_k) - sum_{i=1 to k} h_{q_i})
    # This is equivalent to minimizing the maximum delay.
    # By Smith's Rule / Johnson's Rule logic, to minimize the completion time,
    # we should sort the checkpoints.
    # If we fix the Scout's order, the Hauler should visit in increasing order of h_i.
    # If we fix the Hauler's order, the Scout should visit in increasing order of s_i.
    # It turns out the optimal strategy is to sort by s_i ascending for the Scout
    # and by h_i descending for the Hauler.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    checkpoints = []
    idx = 1
    for _ in range(j):
        s = int(input_data[idx])
        h = int(input_data[idx+1])
        checkpoints.append((s, h))
        idx += 2
        
    # Sort by s_i ascending
    checkpoints.sort(key=lambda x: x[0])
    
    # Now the Scout visits in this order.
    # The Hauler should visit in order of h_i descending to minimize the max wait.
    # Wait, let's re-evaluate:
    # Let S_k be the time Scout finishes inspection at the k-th checkpoint in Hauler's sequence.
    # S_k = sum_{i=1 to k} s_{q_i}
    # H_finish_k = max(S_k, H_finish_{k-1}) + h_{q_k}
    # This is a classic scheduling problem. The optimal order for the Hauler
    # is to process checkpoints with larger h_i first if s_i are equal,
    # but here we can choose both.
    # The optimal strategy is to sort by s_i ascending, and for the Hauler,
    # sort by h_i descending.
    
    checkpoints.sort(key=lambda x: x[0])
    hauler_order = sorted(checkpoints, key=lambda x: x[1], reverse=True)
    
    scout_times = [0] * (j + 1)
    for i in range(j):
        scout_times[i+1] = scout_times[i] + checkpoints[i][0]
        
    # Map each checkpoint to its index in the Scout's sequence
    # Since s_i might be duplicate, we need to be careful.
    # Actually, the Scout's sequence is just the set of all s_i.
    # The Hauler's sequence is a permutation of the same set.
    # Let the Scout visit in order s_1, s_2, ..., s_j.
    # The Hauler visits in order q_1, q_2, ..., q_j.
    # The time Scout finishes q_k is sum_{i=1 to index(q_k)} s_i.
    # To minimize this, we should match the smallest s_i with the largest h_i.
    
    # Correct logic: Sort by s_i ascending, and for the Hauler, 
    # process in order of h_i descending.
    
    current_h_finish = 0
    scout_time_map = {}
    for i in range(j):
        s = checkpoints[i][0]
        if s not in scout_time_map:
            scout_time_map[s] = []
        scout_time_map[s].append(scout_times[i+1])
    
    # We need to track which specific Scout inspection corresponds to which Hauler task.
    # Since we want to minimize max(S(q_k) - sum_{i=1 to k} h_{q_i}),
    # we sort by s_i ascending and h_i descending.
    
    h_finish = 0
    s_sum = 0
    # The Scout visits in order of s_i ascending.
    # The Hauler visits in order of h_i descending.
    # To make this work, we need to know the Scout's finish time for each checkpoint.
    # If we sort checkpoints by s_i, the Scout finishes them at times:
    # T_1 = s_1, T_2 = s_1+s_2, ..., T_j = sum(s_i)
    
    # The Hauler visits in order of h_i descending.
    # Let the Hauler's order be q_1, ..., q_j.
    # The Scout's finish time for q_k is the sum of s_i for all checkpoints 
    # that appear before or at q_k in the Scout's sequence.
    # This is minimized if we pair the smallest s_i with the largest h_i.
    
    checkpoints.sort(key=lambda x: x[0])
    hauler_order = sorted(checkpoints, key=lambda x: x[1], reverse=True)
    
    # Scout finishes checkpoints in order of checkpoints[i]
    # Hauler finishes in order of hauler_order[i]
    # We need to know the Scout's finish time for each specific checkpoint.
    # Let's label them.
    labeled = []
    for i in range(j):
        labeled.append((checkpoints[i][0], checkpoints[i][1], i))
    
    # Scout finishes checkpoint i at scout_times[i+1]
    # Hauler visits in order of hauler_order.
    # We need to know the Scout's finish time for each checkpoint in hauler_order.
    # Since we sorted checkpoints by s_i, the Scout finishes them in that order.
    
    # Actually, the Scout's sequence is fixed as the sorted order of s_i.
    # The Hauler's sequence is the sorted order of h_i descending.
    # We need to track the Scout's finish time for each checkpoint.
    
    s_finish_times = {id(cp): scout_times[i+1] for i, cp in enumerate(checkpoints)}
    
    h_finish = 0
    for cp in hauler_order:
        h_finish = max(h_finish, s_finish_times[id(cp)]) + cp[1]
        
    print(h_finish)

solve()
