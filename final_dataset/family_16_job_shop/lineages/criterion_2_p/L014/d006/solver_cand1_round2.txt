```python
import sys

def solve():
    # The problem asks to minimize max_{k=1..j} (sum_{i=1}^k p1_i + sum_{i=k}^j p2_i)
    # Let S1 = sum_{i=1}^j p1_i and S2 = sum_{i=1}^j p2_i.
    # The expression is: sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # = sum_{i=1}^k p1_i + (sum_{i=1}^j p2_i - sum_{i=1}^{k-1} p2_i)
    # = S2 + sum_{i=1}^k p1_i - sum_{i=1}^{k-1} p2_i
    # Let d_i = p1_i - p2_i.
    # The expression is S2 + p1_1 + sum_{i=2}^k (p1_i - p2_{i-1}).
    # This is a variation of Johnson's rule.
    # To minimize the maximum, we should sort the checkpoints.
    # If p1_i < p2_i, they should come earlier. If p1_i > p2_i, they should come later.
    # Specifically, sort by p1_i < p2_i ascending, then p1_i >= p2_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    checkpoints = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        checkpoints.append((p1, p2))
        idx += 2
        
    # Johnson's rule sorting:
    # Group 1: p1 < p2, sort by p1 ascending
    # Group 2: p1 >= p2, sort by p2 descending
    group1 = []
    group2 = []
    for p1, p2 in checkpoints:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate the completion time for the sorted sequence
    # Time = max_{k=1..j} (prefix_sum_p1[k] + suffix_sum_p2[k])
    # prefix_sum_p1[k] = p1_1 + ... + p1_k
    # suffix_sum_p2[k] = p2_k + ... + p2_j
    
    current_p1_sum = 0
    p1_sums = []
    for p1, p2 in sequence:
        current_p1_sum += p1
        p1_sums.append(current_p1_sum)
        
    current_p2_sum = 0
    p2_sums = [0] * j
    for i in range(j - 1, -1, -1):
        current_p2_sum += sequence[i][1]
        p2_sums[i] = current_p2_sum
        
    max_time = 0
    for k in range(j):
        time = p1_sums[k] + p2_sums[k]
        if time > max_time:
            max_time = time
            
    print(max_time)

if __name__ == '__main__':
    solve()
```
