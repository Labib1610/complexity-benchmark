import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    checkpoints = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        checkpoints.append((p1, p2))
        idx += 2
    
    # Let the sequence be (a1, b1), (a2, b2), ..., (aj, bj).
    # The mission time for transfer point k is:
    # T_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # T_k = sum_{i=1}^j b_i + sum_{i=1}^k (a_i - b_i) + b_k
    # Let S = sum_{i=1}^j b_i.
    # T_k = S + sum_{i=1}^{k-1} (a_i - b_i) + a_k
    
    # This is a variation of Johnson's Rule for scheduling.
    # To minimize the maximum of T_k, we sort checkpoints based on:
    # If a_i < b_i, put them first sorted by a_i ascending.
    # If a_i >= b_i, put them last sorted by b_i descending.
    
    group1 = []
    group2 = []
    for p1, p2 in checkpoints:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate worst-case mission time for this sequence
    # T_k = (sum_{i=1}^k a_i) + (sum_{i=k}^j b_i)
    # Let prefix_a[k] = sum_{i=1}^k a_i
    # Let suffix_b[k] = sum_{i=k}^j b_i
    
    n = len(sequence)
    prefix_a = [0] * (n + 1)
    for i in range(n):
        prefix_a[i+1] = prefix_a[i] + sequence[i][0]
        
    suffix_b = [0] * (n + 2)
    for i in range(n, 0, -1):
        suffix_b[i] = suffix_b[i+1] + sequence[i-1][1]
        
    max_mission_time = 0
    for k in range(1, n + 1):
        current_time = prefix_a[k] + suffix_b[k]
        if current_time > max_mission_time:
            max_mission_time = current_time
            
    print(max_mission_time)

if __name__ == '__main__':
    solve()
