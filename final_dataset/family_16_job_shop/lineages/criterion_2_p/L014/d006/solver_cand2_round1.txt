```python
import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    checkpoints = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        checkpoints.append((p1, p2))
        idx += 2
    
    # Let the Scout visit checkpoints in some order (s_1, s_2, ..., s_j)
    # Let the Hauler visit checkpoints in some order (h_1, h_2, ..., h_j)
    # Scout finishes inspection of s_i at time T_i = sum_{k=1}^i p1_{s_k}
    # Hauler starts extraction at h_i at time S_i = max(S_{i-1} + p2_{h_{i-1}}, T_{pos(h_i)})
    # where pos(h_i) is the index of checkpoint h_i in the Scout's sequence.
    # This simplifies to: Hauler finishes at max_{i=1..j} (sum_{k=1}^i p1_{s_k} + sum_{k=i}^j p2_{h_k})
    # To minimize this, we should sort the Scout's sequence by p1 and the Hauler's by p2.
    # Specifically, if we sort checkpoints such that p1 is increasing, 
    # and we want to minimize max(T_i + sum_{k=i}^j p2_{h_k}),
    # the optimal strategy is to sort Scout by p1 ascending and Hauler by p2 descending.
    
    # Sort by p1 ascending
    checkpoints.sort(key=lambda x: x[0])
    p1_list = [c[0] for c in checkpoints]
    # Sort by p2 descending
    p2_list = sorted([c[1] for c in checkpoints], reverse=True)
    
    # T_i = sum_{k=0}^{i-1} p1_k
    # We want to minimize max_{i=0..j-1} (T_{i+1} + sum_{k=i}^{j-1} p2_k)
    
    current_p1_sum = 0
    total_p2_sum = sum(p2_list)
    
    max_time = 0
    current_p2_sum = total_p2_sum
    
    for i in range(j):
        current_p1_sum += p1_list[i]
        # At step i (0-indexed), Scout finishes at current_p1_sum
        # Hauler needs to finish all remaining extractions starting from this point
        time_at_i = current_p1_sum + current_p2_sum
        if time_at_i > max_time:
            max_time = time_at_i
        current_p2_sum -= p2_list[i]
        
    print(max_time)

if __name__ == '__main__':
    solve()
```
