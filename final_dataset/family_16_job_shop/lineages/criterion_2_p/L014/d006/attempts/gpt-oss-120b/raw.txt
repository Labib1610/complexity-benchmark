```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    items = [(int(next(it)), int(next(it))) for _ in range(n)]
    # sort by decreasing p2 (Johnson's rule for two-machine flow shop)
    items.sort(key=lambda x: -x[1])

    total_p2 = sum(p2 for _, p2 in items)
    pref1 = 0
    pref2 = 0
    ans = 0
    for p1, p2 in items:
        # before adding current p2 to pref2, suffix includes current p2
        candidate = pref1 + p1 + (total_p2 - pref2)
        if candidate > ans:
            ans = candidate
        pref1 += p1
        pref2 += p2
    print(ans)

if __name__ == "__main__":
    solve()
```
