import sys

def solve():
    # Read j
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tokens = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tokens.append((p1, p2))
        idx += 2

    # Thread X processes tokens in the order we provide.
    # Let the order be t_1, t_2, ..., t_j.
    # Token i finishes state 0->1 at time S_i = sum_{k=1}^i p1_k.
    # Thread Y processes tokens in state 1. Since Thread X finishes tokens
    # in order, Thread Y will always receive tokens in the same order.
    # Thread Y starts processing token i at time max(S_i, finish_time_Y_{i-1}).
    # finish_time_Y_i = max(S_i, finish_time_Y_{i-1}) + p2_i.
    
    # To minimize the final finish time, we should sort tokens.
    # This is a classic scheduling problem (Johnson's rule variant).
    # We want to minimize max(S_i + p2_i + delay_i).
    # Sort by p1 ascending for those where p1 < p2, and p2 descending for p1 >= p2.
    
    group1 = []
    group2 = []
    for p1, p2 in tokens:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_tokens = group1 + group2
    
    current_s = 0
    finish_y = 0
    for p1, p2 in sorted_tokens:
        current_s += p1
        finish_y = max(current_s, finish_y) + p2
        
    print(finish_y)

if __name__ == '__main__':
    solve()
