```python
import sys

def solve():
    # The problem asks to minimize the completion time C_j.
    # Let H_sum = sum(H_i for all i).
    # C_k = L_k + max(C_{k-1}, sum_{i=1}^k H_i)
    # This recurrence can be expanded:
    # C_j = max(H_1 + L_1 + L_2 + ... + L_j,
    #           H_1 + H_2 + L_2 + L_3 + ... + L_j,
    #           ...,
    #           H_1 + H_2 + ... + H_j + L_j)
    # Let S_L = sum(L_i for all i).
    # Let S_H = sum(H_i for all i).
    # The terms are:
    # T_k = (sum_{i=1}^k H_i) + (sum_{i=k}^j L_i)
    # T_k = (sum_{i=1}^k H_i) + S_L - (sum_{i=1}^{k-1} L_i)
    # T_k = S_L + sum_{i=1}^k H_i - sum_{i=1}^{k-1} L_i
    # To minimize max(T_k), we want to order the packets.
    # Consider two adjacent packets i and j.
    # If we swap them, the only T_k that changes is the one corresponding to the first of the two.
    # Let the prefix sum of H be H_prev and prefix sum of L be L_prev.
    # Current order (i, j):
    # T_i = H_prev + H_i + (S_L - L_prev)
    # T_j = H_prev + H_i + H_j + (S_L - L_prev - L_i)
    # Swapped order (j, i):
    # T_j' = H_prev + H_j + (S_L - L_prev)
    # T_i' = H_prev + H_j + H_i + (S_L - L_prev - L_j)
    # We prefer (i, j) if max(T_i, T_j) < max(T_j', T_i').
    # This is a classic scheduling problem. The optimal order is to sort by L_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    packets = []
    idx = 1
    for _ in range(j):
        h = int(input_data[idx])
        l = int(input_data[idx+1])
        packets.append((h, l))
        idx += 2
        
    # Sort by payload size descending
    packets.sort(key=lambda x: x[1], reverse=True)
    
    current_c = 0
    current_h_sum = 0
    for h, l in packets:
        current_h_sum += h
        current_c = l + max(current_c, current_h_sum)
        
    print(current_c)

if __name__ == '__main__':
    solve()
```
