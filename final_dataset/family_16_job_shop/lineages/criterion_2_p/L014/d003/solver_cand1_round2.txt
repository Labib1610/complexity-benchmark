import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We want to minimize the makespan of a two-machine flow shop.
    # The optimal sequence is found by:
    # 1. Partitioning jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 2. Sorting Set 1 in ascending order of p1.
    # 3. Sorting Set 2 in descending order of p2.
    # 4. Concatenating Set 1 and Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    t_up = 0
    t_down = 0
    
    for p1, p2 in sequence:
        t_up += p1
        t_down = max(t_up, t_down) + p2
        
    print(t_down)

if __name__ == '__main__':
    solve()
