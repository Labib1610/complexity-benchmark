```python
import sys

def solve():
    # The problem is a variation of the Job Shop Scheduling problem, 
    # specifically a 2-machine flow shop problem (Johnson's Rule).
    # We have j tasks, each with p1 (Node 1) and p2 (Node 2).
    # We want to minimize the makespan (completion time of the last task on Node 2).
    # Johnson's Rule:
    # 1. List all tasks.
    # 2. Divide tasks into two sets:
    #    Set A: tasks where p1 < p2
    #    Set B: tasks where p1 >= p2
    # 3. Sort Set A in ascending order of p1.
    # 4. Sort Set B in descending order of p2.
    # 5. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    tasks = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tasks.append((p1, p2))
        idx += 2
        
    set_a = [t for t in tasks if t[0] < t[1]]
    set_b = [t for t in tasks if t[0] >= t[1]]
    
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate makespan
    # node1_finish_time[i] is the time Node 1 finishes task i
    # node2_finish_time[i] is the time Node 2 finishes task i
    
    node1_time = 0
    node2_time = 0
    
    for p1, p2 in sequence:
        node1_time += p1
        # Node 2 can start task i only after Node 1 finishes task i
        # AND Node 2 finishes task i-1
        node2_time = max(node1_time, node2_time) + p2
        
    print(node2_time)

if __name__ == '__main__':
    solve()
```
