```python
import sys

def solve():
    # The problem asks to minimize the maximum of:
    # sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # This is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # The optimal sequence is to sort blocks such that:
    # Blocks with p1 < p2 come first, sorted by increasing p1.
    # Blocks with p1 >= p2 come last, sorted by decreasing p2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    blocks = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        blocks.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for b in blocks:
        if b[0] < b[1]:
            group1.append(b)
        else:
            group2.append(b)
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate completion time
    # Let S_k be the time the k-th block finishes substitution
    # Let T_k be the time the k-th block finishes transposition
    # S_k = S_{k-1} + p1_k
    # T_k = max(S_k, T_{k-1}) + p2_k
    
    s_time = 0
    t_time = 0
    for p1, p2 in order:
        s_time += p1
        t_time = max(s_time, t_time) + p2
        
    print(t_time)

if __name__ == '__main__':
    solve()
```
