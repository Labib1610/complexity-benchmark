```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    cols = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        cols.append((p1, p2))
        idx += 2
    
    # We want to minimize max_{1 <= k <= j} (sum_{i=1}^k A_i + sum_{i=k}^j B_i)
    # Let S_A = sum(A_i) and S_B = sum(B_i).
    # P(k) = sum_{i=1}^k A_i + sum_{i=k}^j B_i
    # P(k) = sum_{i=1}^k A_i + (S_B - sum_{i=1}^{k-1} B_i)
    # P(k) = S_B + sum_{i=1}^{k-1} (A_i - B_i) + A_k
    # Let D_i = A_i - B_i.
    # P(k) = S_B + sum_{i=1}^{k-1} D_i + A_k
    # To minimize the maximum, we should sort the columns.
    # If we have two adjacent columns (A1, B1) and (A2, B2),
    # swapping them changes the max of P(k) for k=i and k=i+1.
    # Standard exchange argument: sort by A_i - B_i.
    # Specifically, sort by A_i - B_i ascending.
    
    cols.sort(key=lambda x: x[0] - x[1])
    
    s_b = sum(c[1] for c in cols)
    
    # P(k) = sum_{i=1}^k A_i + sum_{i=k}^j B_i
    # Let prefix_a[k] = sum_{i=1}^k A_i
    # Let suffix_b[k] = sum_{i=k}^j B_i
    
    current_a = 0
    current_b_sum = s_b
    ans = 0
    
    for i in range(j):
        a, b = cols[i]
        current_a += a
        # P(i+1) = current_a + current_b_sum
        val = current_a + current_b_sum
        if val > ans:
            ans = val
        current_b_sum -= b
        
    print(ans)

if __name__ == '__main__':
    solve()
```
