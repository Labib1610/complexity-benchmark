```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    items = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        items.append((p1, p2))
        idx += 2
    
    # C(k) = sum_{i=1}^k A_i + sum_{i=k}^j B_i
    # C(k) = sum_{i=1}^j B_i + sum_{i=1}^k (A_i - B_i)
    # Let D_i = A_i - B_i.
    # C(k) = (sum_{i=1}^j B_i) + sum_{i=1}^k D_i
    # To minimize max(C(k)), we need to minimize the prefix sums of D_i.
    # This is a classic problem: to minimize the maximum prefix sum of a sequence,
    # we should sort the sequence D_i in non-decreasing order.
    
    sum_b = sum(item[1] for item in items)
    diffs = sorted([item[0] - item[1] for item in items])
    
    max_score = -float('inf')
    current_prefix_sum = 0
    
    # The score is max_{k=1..j} (sum_b + prefix_sum_k)
    # Since sum_b is constant, we just need to find the max prefix sum of diffs.
    # Note: k ranges from 1 to j.
    
    ans = -float('inf')
    for d in diffs:
        current_prefix_sum += d
        if sum_b + current_prefix_sum > ans:
            ans = sum_b + current_prefix_sum
            
    print(ans)

if __name__ == '__main__':
    solve()
```
