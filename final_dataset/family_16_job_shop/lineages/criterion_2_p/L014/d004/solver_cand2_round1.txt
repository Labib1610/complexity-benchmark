import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    artifacts = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        o = int(input_data[idx+1])
        artifacts.append((a, o))
        idx += 2
    
    # Let the sequence be (a_1, o_1), (a_2, o_2), ..., (a_j, o_j).
    # Score for pivot k: S_k = sum_{i=1}^k a_i + sum_{i=k}^j o_i
    # S_k = sum_{i=1}^j a_i + sum_{i=k}^j o_i - sum_{i=k+1}^j a_i
    # Actually, S_k = (sum_{i=1}^{k-1} a_i) + (a_k + o_k) + (sum_{i=k+1}^j o_i)
    # Let A = sum(a_i) and O = sum(o_i).
    # S_k = sum_{i=1}^k a_i + sum_{i=k}^j o_i
    # S_k = sum_{i=1}^k a_i + (O - sum_{i=1}^{k-1} o_i)
    # S_k = O + a_k + sum_{i=1}^{k-1} (a_i - o_i)
    
    # To minimize max(S_k), we should sort artifacts based on (a_i - o_i).
    # Let d_i = a_i - o_i.
    # S_k = O + a_k + sum_{i=1}^{k-1} d_i
    # If we sort by d_i ascending, the prefix sums of d_i grow as slowly as possible.
    
    artifacts.sort(key=lambda x: x[0] - x[1])
    
    a_vals = [x[0] for x in artifacts]
    o_vals = [x[1] for x in artifacts]
    
    total_o = sum(o_vals)
    
    # S_k = sum_{i=1}^k a_i + sum_{i=k}^j o_i
    # Let prefix_a[k] = sum_{i=1}^k a_i
    # Let suffix_o[k] = sum_{i=k}^j o_i
    
    prefix_a = [0] * (j + 1)
    for i in range(j):
        prefix_a[i+1] = prefix_a[i] + a_vals[i]
        
    suffix_o = [0] * (j + 2)
    for i in range(j, 0, -1):
        suffix_o[i] = suffix_o[i+1] + o_vals[i-1]
        
    max_score = 0
    for k in range(1, j + 1):
        score = prefix_a[k] + suffix_o[k]
        if score > max_score:
            max_score = score
            
    print(max_score)

if __name__ == '__main__':
    solve()
