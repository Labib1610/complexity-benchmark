```python
import sys

def solve():
    # The problem asks us to arrange j columns (p1_i, p2_i) to minimize the maximum path sum.
    # A path consists of moving right and down. If we choose to move down at column k,
    # the path sum is (sum of p1 from 1 to k) + (sum of p2 from k to j).
    # Let S1 = sum(p1_i) and S2 = sum(p2_i).
    # The path sum moving down at column k is:
    # sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # = sum_{i=1}^j p1_i + sum_{i=k}^j p2_i - sum_{i=k+1}^j p1_i (Wait, this is wrong)
    # Let's re-evaluate:
    # Path sum at k = (p1_1 + ... + p1_k) + (p2_k + ... + p2_j)
    # Let P1_sum = sum(p1_i) and P2_sum = sum(p2_i).
    # Path sum at k = (sum_{i=1}^j p1_i) + (sum_{i=k}^j p2_i) - (sum_{i=k}^j p1_i)
    # Path sum at k = P1_sum + sum_{i=k}^j (p2_i - p1_i)
    # Let d_i = p2_i - p1_i.
    # Path sum at k = P1_sum + sum_{i=k}^j d_i.
    # To minimize the maximum of these sums, we need to arrange d_i such that
    # the prefix sums of d_i (from the right) are minimized.
    # Let D_k = sum_{i=k}^j d_i. We want to minimize max(D_1, D_2, ..., D_j).
    # This is equivalent to arranging d_i such that the partial sums from the right
    # are as small as possible.
    # This is a classic problem: to minimize the maximum prefix sum, sort the values.
    # Specifically, if we want to minimize max_{k} (sum_{i=k}^j d_i), we should
    # place the largest d_i values at the end (right side).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    cols = []
    idx = 1
    p1_total = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        cols.append(p2 - p1)
        p1_total += p1
        idx += 2
    
    # We want to minimize max_{k=1..j} (sum_{i=k}^j d_i)
    # Let the sorted d_i be d'_1 <= d'_2 <= ... <= d'_j.
    # To minimize the maximum suffix sum, we should arrange them in increasing order:
    # d'_1, d'_2, ..., d'_j.
    # Then the suffix sums are:
    # S_j = d'_j
    # S_{j-1} = d'_j + d'_{j-1}
    # ...
    # S_1 = d'_j + ... + d'_1
    # The maximum is S_1. This doesn't seem right.
    # Actually, to minimize the maximum suffix sum, we should put the largest values
    # at the beginning (left) and smallest at the end (right).
    # Let's re-check: if we have d_i, we want to arrange them as d_1, d_2, ..., d_j
    # such that max(d_j, d_j+d_{j-1}, ..., d_j+...+d_1) is minimized.
    # This is minimized by sorting d_i in descending order: d_1 >= d_2 >= ... >= d_j.
    
    cols.sort(reverse=True)
    
    max_val = -float('inf')
    current_suffix_sum = 0
    for i in range(j - 1, -1, -1):
        current_suffix_sum += cols[i]
        if current_suffix_sum > max_val:
            max_val = current_suffix_sum
            
    print(p1_total + max_val)

solve()
```
