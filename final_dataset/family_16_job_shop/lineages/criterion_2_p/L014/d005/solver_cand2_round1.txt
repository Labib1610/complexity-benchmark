```python
import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    checkpoints = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        checkpoints.append((p1, p2))
        idx += 2
    
    # Let the sequence be (a1, b1), (a2, b2), ..., (aj, bj)
    # Scout finishes inspection at checkpoint i at time S_i = sum_{k=1}^i a_k
    # Hauler finishes extraction at checkpoint i at time H_i
    # H_i = max(H_{i-1}, S_i) + b_i
    # This is a classic scheduling problem (Johnson's Rule variant).
    # To minimize the completion time, we sort checkpoints:
    # If a_i < b_i, it should come earlier.
    # If a_i >= b_i, it should come later.
    # Specifically, sort by min(a_i, b_i). If a_i < b_i, sort ascending by a_i.
    # If a_i >= b_i, sort descending by b_i.
    
    group1 = []
    group2 = []
    for p1, p2 in checkpoints:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    current_scout_time = 0
    current_hauler_time = 0
    
    for p1, p2 in sequence:
        current_scout_time += p1
        if current_hauler_time < current_scout_time:
            current_hauler_time = current_scout_time + p2
        else:
            current_hauler_time += p2
            
    print(current_hauler_time)

if __name__ == '__main__':
    solve()
```
