import sys

def solve():
    # This is a classic Johnson's Rule problem for a 2-machine flow shop.
    # To minimize the makespan, we sort the jobs based on the following criteria:
    # 1. Jobs with p1 < p2 should be processed in increasing order of p1.
    # 2. Jobs with p1 >= p2 should be processed in decreasing order of p2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    # Sort group1 by p1 ascending
    group1.sort(key=lambda x: x[0])
    # Sort group2 by p2 descending
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate makespan
    time_layer1 = 0
    time_layer2 = 0
    
    for p1, p2 in order:
        time_layer1 += p1
        if time_layer1 > time_layer2:
            time_layer2 = time_layer1 + p2
        else:
            time_layer2 += p2
            
    print(time_layer2)

if __name__ == '__main__':
    solve()
