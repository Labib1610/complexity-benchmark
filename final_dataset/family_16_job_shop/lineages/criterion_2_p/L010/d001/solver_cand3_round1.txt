```python
import sys

def solve():
    # The problem is a variation of Johnson's Rule for two-machine scheduling.
    # We have j jobs (wagons), each with two operations:
    # Machine 1 (Bridge) with time b_i
    # Machine 2 (Tunnel) with time t_i
    # We want to minimize the makespan (completion time of the last job on Machine 2).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where b_i < t_i
    #    Set 2: jobs where b_i >= t_i
    # 2. Sort Set 1 in ascending order of b_i.
    # 3. Sort Set 2 in descending order of t_i.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    wagons = []
    idx = 1
    for _ in range(j):
        b = int(input_data[idx])
        t = int(input_data[idx+1])
        wagons.append((b, t))
        idx += 2
        
    set1 = []
    set2 = []
    for b, t in wagons:
        if b < t:
            set1.append((b, t))
        else:
            set2.append((b, t))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    # bridge_finish_time[i] = sum(b_1...b_i)
    # tunnel_finish_time[i] = max(bridge_finish_time[i], tunnel_finish_time[i-1]) + t_i
    
    current_bridge_time = 0
    current_tunnel_time = 0
    
    for b, t in sequence:
        current_bridge_time += b
        current_tunnel_time = max(current_bridge_time, current_tunnel_time) + t
        
    print(current_tunnel_time)

if __name__ == '__main__':
    solve()
```
