import sys

def solve():
    # The problem is a variation of Johnson's Rule for two-machine scheduling.
    # We have j jobs, each with a bridge time (p1) and a tunnel time (p2).
    # The bridge is machine 1, the tunnel is machine 2.
    # We want to minimize the completion time of the last job on machine 2.
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 2. Sort Set A in ascending order of p1.
    # 3. Sort Set B in descending order of p2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate completion time
    # bridge_finish[i] = time when wagon i finishes bridge
    # tunnel_finish[i] = time when wagon i finishes tunnel
    # tunnel_finish[i] = max(bridge_finish[i], tunnel_finish[i-1]) + p2_i
    
    current_bridge_time = 0
    current_tunnel_time = 0
    
    for p1, p2 in sequence:
        current_bridge_time += p1
        current_tunnel_time = max(current_bridge_time, current_tunnel_time) + p2
        
    print(current_tunnel_time)

if __name__ == '__main__':
    solve()
