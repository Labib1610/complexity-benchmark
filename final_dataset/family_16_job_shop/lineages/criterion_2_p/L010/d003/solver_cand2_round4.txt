```python
import sys

def solve():
    # The problem asks to minimize Y_j + d_2(L_j).
    # Let X_1 = 0.
    # X_{k+1} = X_k + d_1(L_k) => X_k = sum_{i=1}^{k-1} d_1(L_i).
    # Y_k >= X_k + d_1(L_k) = sum_{i=1}^{k} d_1(L_i).
    # Y_{k+1} >= Y_k + d_2(L_k).
    # This implies Y_k >= max_{1 <= i <= k} (sum_{m=1}^{i} d_1(L_m) + sum_{m=i}^{k-1} d_2(L_m)).
    # Specifically, Y_j >= sum_{i=1}^{j} d_1(L_i) + sum_{i=1}^{j-1} d_2(L_i) is not quite right.
    # Let S1_k = sum_{i=1}^k d_1(L_i) and S2_k = sum_{i=1}^k d_2(L_i).
    # The condition Y_k >= S1_k and Y_{k+1} >= Y_k + d_2(L_k) implies:
    # Y_j >= S1_1 + d_2(L_1) + d_2(L_2) + ... + d_2(L_{j-1})
    # Y_j >= S1_2 + d_2(L_2) + ... + d_2(L_{j-1})
    # ...
    # Y_j >= S1_j
    # So Y_j = max_{1 <= i <= j} (S1_i + sum_{m=i}^{j-1} d_2(L_m)).
    # We want to minimize Y_j + d_2(L_j) = max_{1 <= i <= j} (S1_i + sum_{m=i}^{j} d_2(L_m)).
    # Let S2_0 = 0. Then sum_{m=i}^{j} d_2(L_m) = S2_j - S2_{i-1}.
    # Y_j + d_2(L_j) = max_{1 <= i <= j} (S1_i - S2_{i-1} + S2_j).
    # This is equivalent to Johnson's Rule for two-machine scheduling (minimizing makespan).
    # Sort lenses: if d1 < d2, put in increasing order of d1; if d1 >= d2, put in decreasing order of d2.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    lenses = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        lenses.append((d1, d2))
        idx += 2
    
    group1 = [l for l in lenses if l[0] < l[1]]
    group2 = [l for l in lenses if l[0] >= l[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_x = 0
    current_y = 0
    for d1, d2 in order:
        current_x += d1
        current_y = max(current_y, current_x) + d2
        
    print(current_y)

if __name__ == '__main__':
    solve()
```
