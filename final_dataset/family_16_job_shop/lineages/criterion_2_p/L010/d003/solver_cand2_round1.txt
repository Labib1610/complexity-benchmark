```python
import sys

def solve():
    # The problem asks to minimize Y_j + d_2(M_j).
    # Let X_1 = 0.
    # X_{k+1} = X_k + d_1(M_k) => X_k = sum_{i=1}^{k-1} d_1(M_i)
    # Y_k >= X_k + d_1(M_k)
    # Y_{k+1} >= Y_k + d_2(M_k)
    # To minimize Y_j + d_2(M_j), we set Y_k to be as small as possible:
    # Y_1 = d_1(M_1)
    # Y_k = max(Y_{k-1} + d_2(M_{k-1}), X_k + d_1(M_k))
    # Let S_k = sum_{i=1}^k d_1(M_i). Then X_k = S_{k-1}.
    # Y_k = max(Y_{k-1} + d_2(M_{k-1}), S_k)
    # This is equivalent to Y_j = max_{1 <= k <= j} (S_k + sum_{i=k}^{j-1} d_2(M_i))
    # Final boundary = Y_j + d_2(M_j) = max_{1 <= k <= j} (S_k + sum_{i=k}^j d_2(M_i))
    # Let D1_i = d_1(M_i), D2_i = d_2(M_i).
    # Boundary = max_{1 <= k <= j} (sum_{i=1}^k D1_i + sum_{i=k}^j D2_i)
    # Boundary = sum_{i=1}^j D2_i + max_{1 <= k <= j} (sum_{i=1}^k D1_i - sum_{i=1}^{k-1} D2_i)
    # Let f(k) = D1_k - D2_k.
    # We want to minimize the max value. This is a variation of Johnson's rule.
    # Sort minerals such that those with D1 < D2 come first (sorted by D1 ascending),
    # and those with D1 >= D2 come last (sorted by D2 descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    minerals = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        minerals.append((d1, d2))
        idx += 2
        
    group1 = [m for m in minerals if m[0] < m[1]]
    group2 = [m for m in minerals if m[0] >= m[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_x = 0
    current_y = 0
    for d1, d2 in order:
        current_x += d1
        current_y = max(current_y + d2, current_x)
        
    print(current_y)

solve()
```
