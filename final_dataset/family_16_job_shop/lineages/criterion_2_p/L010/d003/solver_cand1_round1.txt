```python
import sys

def solve():
    # The problem asks to minimize V_N = max(V_{k-1}, U_k) + b_{p_k}.
    # Let S_k = sum_{i=1}^k a_{p_i}.
    # V_k = max(V_{k-1}, S_k) + b_{p_k}.
    # This is a classic scheduling problem.
    # Consider two adjacent items i and j.
    # If we process i then j:
    # V_i = max(V_{prev}, S_{prev} + a_i) + b_i
    # V_j = max(V_i, S_{prev} + a_i + a_j) + b_j
    #     = max(max(V_{prev}, S_{prev} + a_i) + b_i, S_{prev} + a_i + a_j) + b_j
    #     = max(V_{prev} + b_i + b_j, S_{prev} + a_i + b_i + b_j, S_{prev} + a_i + a_j + b_j)
    #
    # Comparing (i, j) vs (j, i):
    # We want to minimize max(S_{prev} + a_i + b_i + b_j, S_{prev} + a_i + a_j + b_j)
    # vs max(S_{prev} + a_j + b_j + b_i, S_{prev} + a_j + a_i + b_i)
    #
    # This simplifies to comparing max(a_i + b_i + b_j, a_i + a_j + b_j)
    # vs max(a_j + b_j + b_i, a_j + a_i + b_i)
    #
    # Let's use the standard exchange argument for this type of scheduling:
    # Sort by b_i descending. If b_i == b_j, the order doesn't matter for the
    # max term involving a_i, a_j.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    orders = []
    idx = 1
    for _ in range(n):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        orders.append((a, b))
        idx += 2
        
    # Sort by b descending
    orders.sort(key=lambda x: x[1], reverse=True)
    
    u = 0
    v = 0
    for a, b in orders:
        u += a
        v = max(v, u) + b
        
    print(v)

if __name__ == '__main__':
    solve()
```
