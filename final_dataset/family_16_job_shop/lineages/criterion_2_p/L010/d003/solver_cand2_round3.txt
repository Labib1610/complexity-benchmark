```python
import sys

def solve():
    # The problem asks to minimize the final value of Y.
    # Let X_k be the value of X after step k, and Y_k be the value of Y after step k.
    # X_k = sum_{i=1}^k m1_i
    # Y_k = max(Y_{k-1}, X_k) + m2_k
    # Y_0 = 0
    #
    # Let S = sum_{i=1}^j m1_i.
    # Y_j = max(X_1 + m2_1 + m2_2 + ... + m2_j,
    #           X_2 + m2_2 + ... + m2_j,
    #           ...,
    #           X_j + m2_j)
    # Y_j = max_{k=1..j} (X_k + sum_{i=k}^j m2_i)
    #
    # Let M2_sum = sum_{i=1}^j m2_i.
    # Let P2_k = sum_{i=1}^{k-1} m2_i (with P2_1 = 0).
    # Then sum_{i=k}^j m2_i = M2_sum - P2_k.
    # Y_j = max_{k=1..j} (X_k - P2_k) + M2_sum.
    #
    # To minimize Y_j, we need to minimize max_{k=1..j} (X_k - P2_k).
    # X_k = sum_{i=1}^k m1_i.
    # We want to order the packets to minimize the maximum value of (sum_{i=1}^k m1_i - sum_{i=1}^{k-1} m2_i).
    # Let f_k = X_k - P2_k = m1_1 + (m1_2 - m2_1) + (m1_3 - m2_2) + ... + (m1_k - m2_{k-1}).
    # This looks like we should sort the packets.
    # Consider two adjacent packets i and j.
    # If we swap them, the only term that changes is the one at position i.
    # The condition for optimal ordering is to sort by m2 in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    packets = []
    idx = 1
    for _ in range(j):
        m1 = int(input_data[idx])
        m2 = int(input_data[idx+1])
        packets.append((m1, m2))
        idx += 2
    
    # Sort by m2 descending
    packets.sort(key=lambda x: x[1], reverse=True)
    
    x = 0
    y = 0
    for m1, m2 in packets:
        x += m1
        if y < x:
            y = x
        y += m2
        
    print(y)

if __name__ == '__main__':
    solve()
```
