import sys

def solve():
    # The problem asks to minimize Y_j + W_B(B_j).
    # Let the sequence be B_1, B_2, ..., B_j.
    # X_1 = 0
    # X_{k+1} = X_k + W_R(B_k) => X_k = sum_{i=1}^{k-1} W_R(B_i)
    # Y_k >= X_k + W_R(B_k)
    # Y_{k+1} >= Y_k + W_B(B_k)
    #
    # To minimize Y_j + W_B(B_j), we want to keep Y_k as small as possible.
    # Y_1 = X_1 + W_R(B_1) = W_R(B_1)
    # Y_2 = max(Y_1 + W_B(B_1), X_2 + W_R(B_2))
    #     = max(W_R(B_1) + W_B(B_1), W_R(B_1) + W_R(B_2))
    # Y_k = max(Y_{k-1} + W_B(B_{k-1}), X_k + W_R(B_k))
    #
    # Let S_R(k) = sum_{i=1}^k W_R(B_i). Then X_k = S_R(k-1).
    # Y_k = max(Y_{k-1} + W_B(B_{k-1}), S_R(k-1) + W_R(B_k))
    #
    # This recurrence is equivalent to:
    # Y_k = max_{1 <= i <= k} (S_R(i-1) + W_R(B_i) + sum_{m=i}^{k-1} W_B(B_m))
    # The target is Y_j + W_B(B_j) = max_{1 <= i <= j} (S_R(i-1) + W_R(B_i) + sum_{m=i}^{j} W_B(B_m))
    #
    # Let S_B(k) = sum_{i=1}^k W_B(B_i), with S_B(0) = 0.
    # Target = max_{1 <= i <= j} (S_R(i-1) + W_R(B_i) + S_B(j) - S_B(i-1))
    # Target = S_B(j) + max_{1 <= i <= j} (S_R(i-1) - S_B(i-1) + W_R(B_i))
    #
    # To minimize this, we need to order the blocks.
    # Consider two adjacent blocks B_i and B_{i+1}.
    # If we swap them, the only terms in the max that change are for i and i+1.
    # The condition for optimal ordering is Smith's Rule variant:
    # Sort by W_R + W_B descending? No, let's check:
    # For any two blocks (R1, B1) and (R2, B2), we want to minimize:
    # max(R1 + B1 + B2, R1 + R2 + B2)
    # This is R1 + B2 + max(B1, R2).
    # To minimize this, we want to minimize max(B1, R2).
    # This is a classic scheduling problem. Sort by W_R descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    blocks = []
    idx = 1
    for _ in range(j):
        wr = int(input_data[idx])
        wb = int(input_data[idx+1])
        blocks.append((wr, wb))
        idx += 2
    
    # Sort by W_R descending
    blocks.sort(key=lambda x: x[0], reverse=True)
    
    current_x = 0
    current_y = 0
    for i in range(j):
        wr, wb = blocks[i]
        current_y = max(current_y + (0 if i == 0 else blocks[i-1][1]), current_x + wr)
        current_x += wr
        
    print(current_y + blocks[-1][1])

solve()
