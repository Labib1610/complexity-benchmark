```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    cores = []
    idx = 1
    total_y = 0
    for _ in range(j):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        cores.append((x, y))
        total_y += y
        idx += 2
        
    # Let S = sum(y_i).
    # Baseline at step k (1-indexed) is:
    # B_1 = S
    # B_k = S + sum_{i=1}^{k-1} (x_i - y_i)
    # Peak at step k is P_k = B_k + x_k = S + sum_{i=1}^{k-1} (x_i - y_i) + x_k
    # P_k = S + sum_{i=1}^{k} x_i - sum_{i=1}^{k-1} y_i
    # P_k = S + x_k + sum_{i=1}^{k-1} (x_i - y_i)
    
    # We want to minimize max(P_k).
    # Let d_i = x_i - y_i.
    # P_k = S + x_k + sum_{i=1}^{k-1} d_i
    
    # Sort cores by d_i.
    # If we have two cores i and j, and we process i then j:
    # P_i = S + x_i
    # P_j = S + d_i + x_j
    # Max is max(S + x_i, S + d_i + x_j)
    # If we process j then i:
    # P_j = S + x_j
    # P_i = S + d_j + x_i
    # Max is max(S + x_j, S + d_j + x_i)
    
    # This is a classic scheduling problem. Sort by d_i = x_i - y_i.
    # If d_i < d_j, we should process i before j.
    
    cores.sort(key=lambda c: c[0] - c[1])
    
    max_peak = 0
    current_baseline = total_y
    for x, y in cores:
        peak = current_baseline + x
        if peak > max_peak:
            max_peak = peak
        current_baseline += (x - y)
        
    print(max_peak)

if __name__ == '__main__':
    solve()
```
