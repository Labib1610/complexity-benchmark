```python
import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    cols = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        cols.append((d1, d2))
        idx += 2
    
    # The grid score for a given arrangement is:
    # max_{1 <= k <= j} (sum_{i=1}^k d1_i + sum_{i=k}^j d2_i)
    # Let S1_k = sum_{i=1}^k d1_i and S2_k = sum_{i=k}^j d2_i
    # Total sum T = sum_{i=1}^j (d1_i + d2_i)
    # Note that sum_{i=1}^k d1_i + sum_{i=k}^j d2_i = S1_{k-1} + d1_k + d2_k + S2_{k+1}
    # This is equal to (sum_{i=1}^j d1_i) + (sum_{i=k}^j d2_i)
    # Actually, the path cost for a split at column k is:
    # (sum_{i=1}^k d1_i) + (sum_{i=k}^j d2_i)
    # Let A_i = d1_i, B_i = d2_i.
    # We want to minimize max_k (sum_{i=1}^k A_i + sum_{i=k}^j B_i)
    # This is equivalent to minimizing max_k (sum_{i=1}^j B_i + sum_{i=1}^k (A_i - B_i))
    # Let C_i = A_i - B_i. We want to minimize (sum B_i + max_k (sum_{i=1}^k C_i))
    # To minimize the maximum prefix sum of C_i, we should sort C_i in non-decreasing order.
    
    c = [col[0] - col[1] for col in cols]
    c.sort()
    
    sum_b = sum(col[1] for col in cols)
    
    # The path cost for a split at k (1-indexed) is:
    # sum_{i=1}^j B_i + sum_{i=1}^k C_i
    # Since we sorted C_i, the prefix sums are minimized.
    # The maximum prefix sum will be the sum of all positive C_i values.
    # If all C_i are negative, the max prefix sum is the first one (or 0 if we consider empty).
    # Actually, the path must include at least one column, so k ranges from 1 to j.
    
    current_prefix_sum = 0
    max_prefix_sum = -float('inf')
    
    for val in c:
        current_prefix_sum += val
        if current_prefix_sum > max_prefix_sum:
            max_prefix_sum = current_prefix_sum
            
    print(sum_b + max_prefix_sum)

if __name__ == '__main__':
    solve()
```
