```python
import sys

def solve():
    # The state machine updates:
    # A_k = A_{k-1} + d1_k
    # B_k = max(A_k, B_{k-1}) + d2_k
    # Let A_0 = 0, B_0 = 0.
    # A_k = sum_{i=1}^k d1_i
    # B_k = max(A_k + d2_k, B_{k-1} + d2_k)
    # Expanding B_k:
    # B_k = max(A_k + d2_k, A_{k-1} + d2_{k-1} + d2_k, ..., A_1 + d2_1 + ... + d2_k)
    # B_k = max_{1 <= i <= k} (A_i + sum_{m=i}^k d2_m)
    # Let S_k = sum_{i=1}^k d1_i.
    # B_k = max_{1 <= i <= k} (S_i + sum_{m=i}^k d2_m)
    # B_k = max_{1 <= i <= k} (S_{i-1} + d1_i + sum_{m=i}^k d2_m)
    # This is equivalent to minimizing the final B_j.
    # Let the sequence be p_1, p_2, ..., p_j.
    # B_j = max(S_1 + d2_1 + ... + d2_j, S_2 + d2_2 + ... + d2_j, ..., S_j + d2_j)
    # B_j = max_{1 <= i <= j} (S_i + sum_{m=i}^j d2_m)
    # Let D = sum_{m=1}^j d2_m.
    # B_j = max_{1 <= i <= j} (S_i - sum_{m=1}^{i-1} d2_m) + D
    # To minimize B_j, we need to minimize max_{1 <= i <= j} (S_i - sum_{m=1}^{i-1} d2_m).
    # This is a classic scheduling problem.
    # If we have two adjacent instructions (d1_a, d2_a) and (d1_b, d2_b):
    # Order (a, b): max(S + d1_a - 0, S + d1_a + d1_b - d2_a) + D'
    # Order (b, a): max(S + d1_b - 0, S + d1_b + d1_a - d2_b) + D'
    # We prefer (a, b) if max(d1_a, d1_a + d1_b - d2_a) < max(d1_b, d1_b + d1_a - d2_b)
    # This simplifies to comparing d2_a and d2_b.
    # Specifically, sort by d2 in descending order.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    instructions = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        instructions.append((d1, d2))
        idx += 2
    
    # Sort by d2 descending
    instructions.sort(key=lambda x: x[1], reverse=True)
    
    a = 0
    b = 0
    for d1, d2 in instructions:
        a += d1
        b = max(a, b) + d2
        
    print(b)

if __name__ == '__main__':
    solve()
```
