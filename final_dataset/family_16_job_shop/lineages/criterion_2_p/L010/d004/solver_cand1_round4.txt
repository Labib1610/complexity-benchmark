```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    cols = []
    idx = 1
    for _ in range(j):
        t = int(input_data[idx])
        b = int(input_data[idx+1])
        cols.append((t, b))
        idx += 2
    
    # S(k) = sum_{m=1}^k t_m + sum_{m=k}^j b_m
    # S(k) = sum_{m=1}^j b_m + sum_{m=1}^k (t_m - b_m)
    # Let d_m = t_m - b_m.
    # S(k) = sum(B) + sum_{m=1}^k d_m
    # To minimize max(S(k)), we need to minimize the maximum prefix sum of d_m.
    # This is a classic problem: to minimize the maximum prefix sum of a sequence,
    # sort the sequence such that negative values come first, then positive values.
    
    sum_b = sum(c[1] for c in cols)
    diffs = [c[0] - c[1] for c in cols]
    
    neg = sorted([d for d in diffs if d < 0])
    pos = sorted([d for d in diffs if d >= 0])
    
    sorted_diffs = neg + pos
    
    max_prefix_sum = -float('inf')
    current_sum = 0
    for d in sorted_diffs:
        current_sum += d
        if current_sum > max_prefix_sum:
            max_prefix_sum = current_sum
            
    print(sum_b + max_prefix_sum)

if __name__ == '__main__':
    solve()
```
