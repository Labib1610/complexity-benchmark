```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    modules = []
    idx = 1
    sum_d1 = 0
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        modules.append((d1, d2))
        sum_d1 += d1
        idx += 2
    
    # The final buffer W_final after a sequence of modules (d1_i, d2_i)
    # can be expressed as:
    # W_0 = 0
    # W_i = max(0, W_{i-1} - d1_i) + d2_i
    # This is equivalent to:
    # W_final = max(0, sum(d2_i) - sum(d1_i), 
    #               d2_n - d1_n, 
    #               d2_n + d2_{n-1} - d1_n - d1_{n-1}, ...)
    # To minimize W_final, we use the Johnson's rule-like approach.
    # Modules with d1 >= d2 should come first (sorted by d1 ascending),
    # modules with d1 < d2 should come last (sorted by d2 descending).
    
    group1 = []
    group2 = []
    for d1, d2 in modules:
        if d1 >= d2:
            group1.append((d1, d2))
        else:
            group2.append((d1, d2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    w = 0
    for d1, d2 in sequence:
        w = max(0, w - d1) + d2
        
    print(sum_d1 + w)

if __name__ == '__main__':
    solve()
```
