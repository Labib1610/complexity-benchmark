```python
import sys

def solve():
    # The problem asks to minimize W_j + sum(x_i).
    # W_j is defined as:
    # W_1 = y_1
    # W_k = max(W_{k-1} - x_k, 0) + y_k
    #
    # Let S = sum(x_i).
    # We want to minimize W_j + S.
    # This is a classic scheduling problem.
    # Let's consider two adjacent packets (x_i, y_i) and (x_j, y_j).
    # If we process i then j:
    # W_i = y_i
    # W_j = max(y_i - x_j, 0) + y_j
    # If we process j then i:
    # W_j = y_j
    # W_i = max(y_j - x_i, 0) + y_i
    #
    # This specific recurrence is equivalent to minimizing the final value
    # in a system where you have a resource that gets consumed by x and replenished by y.
    # The optimal order is:
    # 1. Packets with y_i < x_i sorted by y_i ascending.
    # 2. Packets with y_i >= x_i sorted by x_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    packets = []
    idx = 1
    sum_x = 0
    for _ in range(j):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        packets.append((x, y))
        sum_x += x
        idx += 2
        
    group1 = [] # y < x
    group2 = [] # y >= x
    
    for x, y in packets:
        if y < x:
            group1.append((x, y))
        else:
            group2.append((x, y))
            
    # Sort group1 by y ascending
    group1.sort(key=lambda p: p[1])
    # Sort group2 by x descending
    group2.sort(key=lambda p: p[0], reverse=True)
    
    sequence = group1 + group2
    
    w = 0
    for i in range(j):
        x, y = sequence[i]
        if i == 0:
            w = y
        else:
            w = max(w - x, 0) + y
            
    print(w + sum_x)

if __name__ == '__main__':
    solve()
```
