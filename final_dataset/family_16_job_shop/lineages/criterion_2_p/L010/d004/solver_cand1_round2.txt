```python
import sys

def solve():
    # The cost for a fixed arrangement T_1, ..., T_j is:
    # max_{1 <= k <= j} (sum_{i=1}^k top(T_i) + sum_{i=k}^j bottom(T_i))
    # Let S_top = sum_{i=1}^j top(T_i) and S_bot = sum_{i=1}^j bottom(T_i).
    # The cost is max_{1 <= k <= j} (sum_{i=1}^k top(T_i) + sum_{i=k}^j bottom(T_i))
    # Note that sum_{i=k}^j bottom(T_i) = S_bot - sum_{i=1}^{k-1} bottom(T_i).
    # So cost = S_bot + max_{1 <= k <= j} (sum_{i=1}^k top(T_i) - sum_{i=1}^{k-1} bottom(T_i))
    # Let diff_i = top(T_i) - bottom(T_i).
    # Then sum_{i=1}^k top(T_i) - sum_{i=1}^{k-1} bottom(T_i) = top(T_k) + sum_{i=1}^{k-1} (top(T_i) - bottom(T_i))
    # = top(T_k) + sum_{i=1}^{k-1} diff_i.
    # To minimize the max, we should sort the tiles.
    # If we have two tiles (t1, b1) and (t2, b2), swapping them changes the max.
    # It is a known result for this type of problem that sorting by (t_i - b_i)
    # in descending order minimizes the maximum prefix sum.
    # Specifically, sort such that t_i - b_i is descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        t = int(input_data[idx])
        b = int(input_data[idx+1])
        tiles.append((t, b))
        idx += 2
        
    # Sort by (t - b) descending
    tiles.sort(key=lambda x: x[0] - x[1], reverse=True)
    
    # Calculate the grid score for this arrangement
    # Score = max_{1 <= k <= j} (sum_{i=1}^k top_i + sum_{i=k}^j bot_i)
    # Let P_top[k] be prefix sum of tops, P_bot[k] be prefix sum of bottoms.
    # Score = max_{1 <= k <= j} (P_top[k] + (P_bot[j] - P_bot[k-1]))
    
    p_top = [0] * (j + 1)
    p_bot = [0] * (j + 1)
    for i in range(j):
        p_top[i+1] = p_top[i] + tiles[i][0]
        p_bot[i+1] = p_bot[i] + tiles[i][1]
        
    ans = -1
    for k in range(1, j + 1):
        current_score = p_top[k] + (p_bot[j] - p_bot[k-1])
        if ans == -1 or current_score > ans:
            ans = current_score
            
    print(ans)

if __name__ == '__main__':
    solve()
```
