```python
import sys

def solve():
    # Let the sequence of brackets be (d1_i, d2_i) for i = 1 to j.
    # Let P_i be the peak of the Primary strut of the i-th bracket.
    # P_i = sum_{k=1}^i d1_k.
    # Let S_i be the peak of the Secondary strut of the i-th bracket.
    # Let B_i be the base of the Secondary strut of the i-th bracket.
    # B_1 = P_1 = d1_1.
    # S_1 = B_1 + d2_1 = d1_1 + d2_1.
    # For i > 1:
    # B_i = max(S_{i-1}, P_i)
    # S_i = B_i + d2_i = max(S_{i-1}, P_i) + d2_i
    #
    # Let's expand S_i:
    # S_1 = P_1 + d2_1
    # S_2 = max(P_1 + d2_1, P_2) + d2_2 = max(P_1 + d2_1 + d2_2, P_2 + d2_2)
    # S_3 = max(P_1 + d2_1 + d2_2 + d2_3, P_2 + d2_2 + d2_3, P_3 + d2_3)
    # In general, S_j = max_{1 <= i <= j} (P_i + sum_{k=i}^j d2_k)
    # S_j = max_{1 <= i <= j} (sum_{k=1}^i d1_k + sum_{k=i}^j d2_k)
    # S_j = sum_{k=1}^j d2_k + max_{1 <= i <= j} (sum_{k=1}^i d1_k - sum_{k=1}^{i-1} d2_k)
    # Let D_i = d1_i - d2_i.
    # Then sum_{k=1}^i d1_k - sum_{k=1}^{i-1} d2_k = sum_{k=1}^{i-1} (d1_k - d2_k) + d1_i
    # = sum_{k=1}^{i-1} D_k + d1_i.
    # We want to minimize S_j.
    # S_j = sum_{k=1}^j d2_k + max_{1 <= i <= j} (d1_i + sum_{k=1}^{i-1} D_k)
    #
    # This is a classic scheduling problem. To minimize the maximum of prefix sums,
    # we should sort the brackets.
    # If we have two adjacent brackets (d1_a, d2_a) and (d1_b, d2_b),
    # the contribution to the max is max(d1_a, d1_a + D_a + d1_b - d1_a) = max(d1_a, D_a + d1_b).
    # We want to order them such that max(d1_a, d1_a + d1_b - d2_a) is minimized.
    # This is equivalent to sorting by d2_i in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    brackets = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        brackets.append((d1, d2))
        idx += 2
    
    # Sort by d2 descending
    brackets.sort(key=lambda x: x[1], reverse=True)
    
    current_p = 0
    current_s = 0
    for d1, d2 in brackets:
        current_p += d1
        current_s = max(current_s, current_p) + d2
        
    print(current_s)

if __name__ == '__main__':
    solve()
```
