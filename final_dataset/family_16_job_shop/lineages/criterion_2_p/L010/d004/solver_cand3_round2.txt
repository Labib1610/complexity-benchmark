```python
import sys

def solve():
    # The constraints are:
    # X_{k+1} >= X_k + d1_k
    # Y_k >= X_k + d1_k
    # Y_{k+1} >= Y_k + d2_k
    # We want to minimize Y_j + d2_j.
    # Let X_1 = 0.
    # Then X_k = sum_{i=1}^{k-1} d1_i.
    # Y_k >= X_k + d1_k = sum_{i=1}^{k-1} d1_i + d1_k = sum_{i=1}^k d1_i.
    # Also Y_{k+1} >= Y_k + d2_k.
    # This implies Y_k >= Y_1 + sum_{i=1}^{k-1} d2_i.
    # Combining: Y_k >= max(sum_{i=1}^k d1_i, Y_1 + sum_{i=1}^{k-1} d2_i).
    # For k=j: Y_j >= max(sum_{i=1}^j d1_i, Y_1 + sum_{i=1}^{j-1} d2_j).
    # Since Y_1 >= d1_1, the smallest Y_1 is d1_1.
    # Then Y_j >= max(sum_{i=1}^j d1_i, d1_1 + sum_{i=1}^{j-1} d2_i).
    # We want to minimize Y_j + d2_j.
    # Let S1 = sum(d1_i) and S2 = sum(d2_i).
    # Y_j >= max(S1, d1_1 + S2 - d2_j).
    # Final boundary = Y_j + d2_j >= max(S1 + d2_j, d1_1 + S2).
    # To minimize this, we need to minimize S1 + d2_j subject to the order.
    # Actually, S1 and S2 are constant regardless of order.
    # The only variable is d2_j (the last bracket's d2).
    # To minimize max(S1 + d2_j, d1_1 + S2), we should pick the bracket with the smallest d2_j.
    # Wait, d1_1 is also dependent on the order.
    # Let the sequence be B_1, ..., B_j.
    # Y_j >= sum_{i=1}^j d1_i
    # Y_j >= d1_1 + sum_{i=1}^{j-1} d2_i
    # Final = Y_j + d2_j >= max(sum_{i=1}^j d1_i + d2_j, d1_1 + sum_{i=1}^j d2_i)
    # Let S1 = sum(d1_i), S2 = sum(d2_i).
    # Final >= max(S1 + d2_j, d1_1 + S2).
    # To minimize this, we want to minimize d2_j and d1_1.
    # If we pick a bracket to be first (d1_1) and one to be last (d2_j),
    # we should pick the one with the smallest d1 for the first position
    # and the one with the smallest d2 for the last position.
    # If j=1, result is d1_1 + d2_1.
    # If j>1, we can pick any bracket for first and any for last.
    # To minimize max(S1 + d2_j, d1_1 + S2), we should pick the bracket with
    # the smallest d2 to be last, and the bracket with the smallest d1 to be first.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    brackets = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        brackets.append((d1, d2))
        idx += 2
        
    if j == 1:
        print(brackets[0][0] + brackets[0][1])
        return
        
    s1 = sum(b[0] for b in brackets)
    s2 = sum(b[1] for b in brackets)
    
    min_d1 = min(b[0] for b in brackets)
    min_d2 = min(b[1] for b in brackets)
    
    print(max(s1 + min_d2, min_d1 + s2))

solve()
```
