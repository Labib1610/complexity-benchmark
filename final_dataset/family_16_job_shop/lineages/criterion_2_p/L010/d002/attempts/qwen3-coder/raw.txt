import sys
from functools import cmp_to_key

def compare_orders(a, b):
    # a[0] is d1, a[1] is d2
    # Johnson's rule comparison
    if a[1] <= a[0] and b[1] <= b[0]:
        # Both are A-type jobs, sort by d2 increasing
        if a[1] != b[1]:
            return a[1] - b[1]
        else:
            return a[0] - b[0]
    elif a[1] > a[0] and b[1] > b[0]:
        # Both are B-type jobs, sort by d1 decreasing
        if a[0] != b[0]:
            return b[0] - a[0]
        else:
            return a[1] - b[1]
    elif a[1] <= a[0] and b[1] > b[0]:
        # a is A-type, b is B-type, a comes first
        return -1
    else:
        # a is B-type, b is A-type, b comes first
        return 1

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    orders = []
    index = 1
    for i in range(j):
        d1 = int(data[index])
        d2 = int(data[index+1])
        orders.append((d1, d2))
        index += 2
    
    # Sort according to Johnson's rule
    orders.sort(key=cmp_to_key(compare_orders))
    
    # Calculate the minimum target value
    X = 0  # Starting primary timestamp
    Y = 0  # Starting secondary timestamp
    
    for d1, d2 in orders:
        X += d1
        Y = max(Y, X) + d2
    
    print(Y)

if __name__ == "__main__":
    main()
