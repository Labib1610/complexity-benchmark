```python
import sys
import threading

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    brackets = []
    index = 1
    for _ in range(j):
        d1 = int(data[index])
        d2 = int(data[index+1])
        brackets.append((d1, d2))
        index += 2
    
    # Sort brackets by d1 ascending, and if equal, by d2 descending
    brackets.sort(key=lambda x: (x[0], -x[1]))
    
    # Now simulate placing brackets in this order
    alpha_pos = 0
    beta_pos = 0
    for d1, d2 in brackets:
        # Place alpha footprint
        alpha_start = alpha_pos
        alpha_end = alpha_start + d1
        alpha_pos = alpha_end
        
        # Place beta footprint: must start >= alpha_end
        beta_start = max(beta_pos, alpha_end)
        beta_end = beta_start + d2
        beta_pos = beta_end
    
    print(beta_pos)

# Use threading to increase recursion limit for large inputs
threading.Thread(target=main).start()
```
