```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    pairs = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        pairs.append((d1, d2))
        idx += 2
    
    # S(i) = sum_{k=1}^i d1_k + sum_{k=i}^j d2_k
    # S(i) = sum_{k=1}^j d2_k + sum_{k=1}^i (d1_k - d2_k)
    # Let D_k = d1_k - d2_k.
    # S(i) = sum(d2) + sum_{k=1}^i D_k
    # To minimize max(S(i)), we need to arrange D_k such that the prefix sums
    # of D_k are as small as possible.
    # This is achieved by sorting D_k in ascending order.
    
    sum_d2 = sum(p[1] for p in pairs)
    diffs = sorted([p[0] - p[1] for p in pairs])
    
    max_s = -float('inf')
    current_s = sum_d2
    
    # S(i) = sum_d2 + prefix_sum_of_diffs[i]
    # i ranges from 1 to j
    for d in diffs:
        current_s += d
        if current_s > max_s:
            max_s = current_s
            
    print(max_s)

if __name__ == '__main__':
    solve()
```
