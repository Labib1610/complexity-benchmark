import sys

def solve():
    # The problem asks to minimize the peak saturation.
    # Let the sequence of crystals be (u_1, l_1), (u_2, l_2), ..., (u_j, l_j).
    # The path dropping at k has total resonance:
    # S_k = sum_{i=1}^k u_i + sum_{i=k}^j l_i
    # S_k = sum_{i=1}^j u_i + sum_{i=k}^j (l_i - u_i)
    # Let D_i = l_i - u_i.
    # S_k = sum_{i=1}^j u_i + sum_{i=k}^j D_i
    # Let U = sum_{i=1}^j u_i.
    # S_k = U + sum_{i=k}^j D_i.
    # We want to minimize max_{k=1..j} (U + sum_{i=k}^j D_i).
    # Since U is constant for any permutation, we minimize max_{k=1..j} (sum_{i=k}^j D_i).
    # Let P_k = sum_{i=k}^j D_i. We want to minimize max(P_1, P_2, ..., P_j).
    # Note P_j = D_j, P_{j-1} = D_{j-1} + D_j, ..., P_1 = D_1 + ... + D_j.
    # This is equivalent to minimizing the maximum suffix sum of the sequence D_1, ..., D_j.
    # To minimize the maximum suffix sum, we should sort D_i in non-decreasing order.
    # If D_1 <= D_2 <= ... <= D_j, then the suffix sums are:
    # P_j = D_j
    # P_{j-1} = D_{j-1} + D_j
    # ...
    # P_1 = D_1 + ... + D_j
    # The maximum is P_1 = sum_{i=1}^j D_i.
    # Wait, if we sort D_i in non-decreasing order, the maximum suffix sum is indeed the sum of all D_i.
    # Is there a better way? If we sort D_i in non-increasing order:
    # D_1 >= D_2 >= ... >= D_j.
    # P_j = D_j
    # P_{j-1} = D_{j-1} + D_j
    # ...
    # P_1 = sum D_i.
    # Actually, the maximum suffix sum is minimized when we sort D_i in non-increasing order.
    # Let's check Example 1: (3,2), (2,4), (1,3).
    # D values: 2-3=-1, 4-2=2, 3-1=2. D = {-1, 2, 2}.
    # If sorted non-increasing: 2, 2, -1.
    # Suffix sums: P_3 = -1, P_2 = 2-1=1, P_1 = 2+2-1=3. Max = 3.
    # Total resonance = U + max(P_k) = (3+2+1) + 3 = 6 + 3 = 9.
    # Wait, the example says 10. Let's re-read.
    # Example 1: (3,2), (2,4), (1,3). U = 6.
    # Sequence: (1,3), (2,4), (3,2). D = {2, 2, -1}.
    # S_1 = 1 + (3+4+2) = 10.
    # S_2 = 1+2 + (4+2) = 9.
    # S_3 = 1+2+3 + 2 = 8.
    # Max is 10.
    # My formula S_k = U + sum_{i=k}^j D_i is correct.
    # To minimize max(S_k), we need to minimize max_{k} (sum_{i=k}^j D_i).
    # This is a known problem: to minimize the maximum suffix sum, sort D_i in non-increasing order.
    # Let's re-check Example 1 with D = {2, 2, -1}.
    # Suffix sums: P_3 = -1, P_2 = 1, P_1 = 3. Max = 3.
    # U + 3 = 6 + 3 = 9. Why is the example 10?
    # Ah, the example sequence is (1,3), (2,4), (3,2).
    # D_1 = 3-1=2, D_2 = 4-2=2, D_3 = 2-3=-1.
    # S_1 = U + (D_1+D_2+D_3) = 6 + 3 = 9.
    # Wait, the example says 10. Let me re-calculate.
    # Path 1: drop at 1. Upper: 1. Lower: 3, 4, 2. Sum = 1+3+4+2 = 10.
    # Path 2: drop at 2. Upper: 1, 2. Lower: 4, 2. Sum = 1+2+4+2 = 9.
    # Path 3: drop at 3. Upper: 1, 2, 3. Lower: 2. Sum = 1+2+3+2 = 8.
    # Max is 10. My D_i calculation was D_i = l_i - u_i.
    # S_k = sum_{i=1}^k u_i + sum_{i=k}^j l_i = sum_{i=1}^j u_i + sum_{i=k}^j (l_i - u_i).
    # Yes, the formula is correct. The sequence (1,3), (2,4), (3,2) gives D = {2, 2, -1}.
    # S_1 = 6 + (2+2-1) = 9.
    # Wait, 1+3+4+2 = 10. 1+2+4+2 = 9. 1+2+3+2 = 8.
    # My manual sum was wrong. 1+3+4+2 = 10. Correct.
    # The sequence (1,3), (2,4), (3,2) has D = {2, 2, -1}.
    # S_1 = 6 + (2+2-1) = 9. Wait, 1+3+4+2 = 10.
    # 1+3+4+2 = 10. 1+2+4+2 = 9. 1+2+3+2 = 8.
    # The sum is 10. My formula S_1 = U + D_1 + D_2 + D_3 = 6 + 3 = 9.
    # Where is the error?
    # S_1 = u_1 + l_1 + l_2 + l_3 = 1 + 3 + 4 + 2 = 10.
    # U = 1+2+3 = 6.
    # D_1 = l_1 - u_1 = 3 - 1 = 2.
    # D_2 = l_2 - u_2 = 4 - 2 = 2.
    # D_3 = l_3 - u_3 = 2 - 3 = -1.
    # U + D_1 + D_2 + D_3 = 6 + 2 + 2 - 1 = 9.
    # The formula S_k = U + sum_{i=k}^j D_i is for k=1: U + D_1 + D_2 + D_3 = sum u_i + sum (l_i - u_i) = sum l_i.
    # But S_1 = u_1 + l_1 + l_2 + l_3.
    # The formula should be S_k = sum_{i=1}^{k-1} u_i + l_k + sum_{i=k+1}^j l_i = sum_{i=1}^j l_i + sum_{i=1}^{k-1} (u_i - l_i).
    # Let E_i = u_i - l_i.
    # S_k = sum_{i=1}^j l_i + sum_{i=1}^{k-1} E_i.
    # To minimize max S_k, we need to minimize max_{k} (sum_{i=1}^{k-1} E_i).
    # This is the maximum prefix sum of E_i.
    # To minimize the maximum prefix sum, sort E_i in non-decreasing order.
    
    input_data = sys.stdin.read().split()
    if not input_data: return
    j = int(input_data[0])
    E = []
    sum_l = 0
    for i in range(j):
        u = int(input_data[2*i + 1])
        l = int(input_data[2*i + 2])
        E.append(u - l)
        sum_l += l
    
    E.sort()
    
    max_prefix_sum = 0
    current_prefix_sum = 0
    for val in E:
        current_prefix_sum += val
        if current_prefix_sum > max_prefix_sum:
            max_prefix_sum = current_prefix_sum
            
    print(sum_l + max_prefix_sum)

solve()
