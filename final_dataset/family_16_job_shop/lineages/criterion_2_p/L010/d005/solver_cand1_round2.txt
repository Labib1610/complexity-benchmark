```python
import sys

def solve():
    # The problem asks to minimize the rightmost coordinate of the Beta block of the final bracket.
    # Let the sequence of brackets be (a_1, b_1), (a_2, b_2), ..., (a_j, b_j).
    # Let A_k be the start coordinate of the Alpha block of bracket k.
    # Let B_k be the start coordinate of the Beta block of bracket k.
    # Constraints:
    # 1. A_k + a_k <= B_k (Internal link)
    # 2. A_k + a_k <= A_{k+1} (Alpha daisy-chain)
    # 3. B_k + b_k <= B_{k+1} (Beta daisy-chain)
    #
    # From (2), A_k = sum_{i=1}^{k-1} a_i (assuming A_1 = 0).
    # From (3), B_k = max(B_{k-1} + b_{k-1}, A_k + a_k).
    # We want to minimize B_j.
    # B_1 = a_1
    # B_2 = max(B_1 + b_1, A_2 + a_2) = max(a_1 + b_1, a_1 + a_2)
    # B_3 = max(B_2 + b_2, A_3 + a_3) = max(a_1 + b_1 + b_2, a_1 + a_2 + b_2, a_1 + a_2 + a_3)
    # In general, B_j = max_{1 <= k <= j} (sum_{i=1}^{k} a_i + sum_{i=k}^{j} b_i - b_k)
    # Actually, the recurrence is B_k = max(B_{k-1} + b_{k-1}, A_k + a_k).
    # Let S_A(k) = sum_{i=1}^k a_i. Then A_k = S_A(k-1).
    # B_k = max(B_{k-1} + b_{k-1}, S_A(k-1) + a_k) = max(B_{k-1} + b_{k-1}, S_A(k)).
    # Expanding this:
    # B_j = max(S_A(1) + sum_{i=1}^{j-1} b_i, S_A(2) + sum_{i=2}^{j-1} b_i, ..., S_A(j-1) + b_{j-1}, S_A(j))
    # This is equivalent to:
    # B_j = max_{1 <= k <= j} (S_A(k) + sum_{i=k}^{j-1} b_i)
    # Let S_B(k) = sum_{i=1}^k b_i, with S_B(0) = 0.
    # B_j = max_{1 <= k <= j} (S_A(k) + S_B(j-1) - S_B(k-1))
    # B_j = S_B(j-1) + max_{1 <= k <= j} (S_A(k) - S_B(k-1))
    # To minimize this, we need to choose the order of brackets.
    # This is a classic scheduling problem. To minimize the max of partial sums,
    # we should sort by the difference or use the Smith's rule logic.
    # Actually, for this specific form, sorting by a_i - b_i is optimal.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    brackets = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        brackets.append((a, b))
        idx += 2
    
    # Sort by a_i - b_i
    brackets.sort(key=lambda x: x[0] - x[1])
    
    s_a = 0
    s_b = 0
    # B_k = max(B_{k-1} + b_{k-1}, S_A(k))
    # Let current_b be the end of the Beta block of the previous bracket.
    # current_b = B_{k-1} + b_{k-1}
    # B_k = max(current_b, S_A(k))
    
    ans = 0
    current_b = 0
    for a, b in brackets:
        s_a += a
        ans = max(current_b + b, s_a + b)
        current_b = ans
        
    print(ans)

solve()
```
