```python
import sys

def solve():
    # The problem asks to minimize the final value of B.
    # Let A_i and B_i be the parameters of packet i.
    # After processing a sequence of packets, the final B is:
    # B_final = max(sum(A_1..A_k) + B_k) for all k in 1..j
    # This is a classic scheduling problem.
    # If we have two adjacent packets (a1, b1) and (a2, b2):
    # Order 1, 2: B_final = max(A+a1+b1, A+a1+a2+b2)
    # Order 2, 1: B_final = max(A+a2+b2, A+a2+a1+b1)
    # We prefer 1, 2 if max(a1+b1, a1+a2+b2) < max(a2+b2, a2+a1+b1)
    # This simplifies to comparing b1 and b2.
    # Specifically, if b1 < b2, we should put packet 1 before packet 2.
    # If b1 == b2, the order doesn't matter.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    packets = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        packets.append((d1, d2))
        idx += 2
        
    # Sort by Beta parameter (d2) ascending
    packets.sort(key=lambda x: x[1])
    
    current_a = 0
    current_b = 0
    
    for d1, d2 in packets:
        current_a += d1
        current_b = max(current_a, current_b) + d2
        
    print(current_b)

if __name__ == '__main__':
    solve()
```
