```python
import sys

def solve():
    # The split cost at k is:
    # C(k) = sum_{i=1}^k d1(B_i) + sum_{i=k}^j d2(B_i)
    # C(k) = sum_{i=1}^k d1(B_i) + d2(B_k) + sum_{i=k+1}^j d2(B_i)
    # Let S1(k) = sum_{i=1}^k d1(B_i) and S2(k) = sum_{i=k}^j d2(B_i)
    # C(k) = S1(k) + S2(k)
    # Note that S2(k) = d2(B_k) + S2(k+1)
    # C(k) = S1(k-1) + d1(B_k) + d2(B_k) + S2(k+1)
    # Let T = sum_{i=1}^j d2(B_i). Then S2(k) = T - sum_{i=1}^{k-1} d2(B_i)
    # C(k) = sum_{i=1}^k d1(B_i) + T - sum_{i=1}^{k-1} d2(B_i)
    # C(k) = T + d1(B_k) + sum_{i=1}^{k-1} (d1(B_i) - d2(B_i))
    
    # To minimize the maximum C(k), we should sort the brackets.
    # Consider two adjacent brackets i and i+1.
    # If we swap them, the only C(k) that changes is at k=i.
    # C(i) = T + d1(B_i) + sum_{j=1}^{i-1} (d1(B_j) - d2(B_j))
    # The term sum_{j=1}^{i-1} (d1(B_j) - d2(B_j)) is constant for a fixed position i.
    # To minimize the max, we should sort by d1 - d2.
    # Specifically, if d1_i - d2_i < d1_{i+1} - d2_{i+1}, we prefer i before i+1.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    brackets = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        brackets.append((d1, d2))
        idx += 2
        
    # Sort by d1 - d2
    brackets.sort(key=lambda x: x[0] - x[1])
    
    # Calculate prefix sums of d1 and suffix sums of d2
    # C(k) = sum_{i=1}^k d1_i + sum_{i=k}^j d2_i
    # Let P1[k] = sum_{i=0}^{k-1} d1_i
    # Let S2[k] = sum_{i=k}^{j-1} d2_i
    # C(k+1) = P1[k+1] + S2[k]
    
    p1 = [0] * (j + 1)
    for i in range(j):
        p1[i+1] = p1[i] + brackets[i][0]
        
    s2 = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        s2[i] = s2[i+1] + brackets[i][1]
        
    max_c = 0
    for k in range(1, j + 1):
        current_c = p1[k] + s2[k-1]
        if current_c > max_c:
            max_c = current_c
            
    print(max_c)

if __name__ == '__main__':
    solve()
```
