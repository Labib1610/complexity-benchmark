```python
import sys

def solve():
    # The total energy for a split point k is:
    # E(k) = sum_{i=1}^k d1(S_i) + sum_{i=k}^j d2(S_i)
    # E(k) = sum_{i=1}^j d2(S_i) + sum_{i=1}^k (d1(S_i) - d2(S_i)) + d2(S_k)
    # Let D_i = d1(S_i) - d2(S_i).
    # E(k) = sum(d2) + sum_{i=1}^k D_i + d2(S_k)
    # To minimize the maximum E(k), we should sort the seals.
    # Let's consider the difference D_i = d1_i - d2_i.
    # If we sort by D_i in ascending order, we minimize the peak.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    seals = []
    idx = 1
    sum_d2 = 0
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        seals.append((d1, d2))
        sum_d2 += d2
        idx += 2
        
    # Sort by d1 - d2 ascending
    seals.sort(key=lambda x: x[0] - x[1])
    
    # Now calculate the peak resonance for this order
    # E(k) = sum_d2 + prefix_sum_D[k] + d2(S_k)
    # where prefix_sum_D[k] = sum_{i=0}^{k-1} (d1_i - d2_i)
    # Note: k is 1-indexed in the problem, so k=1 corresponds to index 0
    
    current_prefix_D = 0
    max_energy = -float('inf')
    
    for i in range(j):
        d1, d2 = seals[i]
        current_prefix_D += (d1 - d2)
        energy = sum_d2 + current_prefix_D + d2
        if energy > max_energy:
            max_energy = energy
            
    print(max_energy)

if __name__ == '__main__':
    solve()
```
