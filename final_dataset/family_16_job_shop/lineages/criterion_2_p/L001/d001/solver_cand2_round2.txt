```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # To minimize the makespan of a two-machine flow shop, we sort tasks:
    # 1. Tasks with p1 < p2 are sorted in ascending order of p1.
    # 2. Tasks with p1 >= p2 are sorted in descending order of p2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    tasks = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tasks.append((p1, p2))
        idx += 2
        
    # Johnson's Rule sorting
    # Group 1: p1 < p2, sort by p1 ascending
    # Group 2: p1 >= p2, sort by p2 descending
    group1 = []
    group2 = []
    for p1, p2 in tasks:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate completion time
    fetcher_time = 0
    executor_time = 0
    
    for p1, p2 in sequence:
        fetcher_time += p1
        # Executor can start only after fetcher finishes current task
        # and after executor finishes previous task
        if executor_time < fetcher_time:
            executor_time = fetcher_time + p2
        else:
            executor_time += p2
            
    print(executor_time)

if __name__ == '__main__':
    solve()
```
