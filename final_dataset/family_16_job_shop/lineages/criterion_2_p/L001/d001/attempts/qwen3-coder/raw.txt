```python
import sys
from functools import cmp_to_key

def compare_dossiers(a, b):
    # a[0] is initial review time, a[1] is final sign-off time
    # For two dossiers, we want to decide which should come first
    # Compare the cost of doing a then b vs b then a
    # If we do dossier 1 then dossier 2:
    # Time = max(p1_1 + p2_1, p1_1 + p1_2 + p2_2)
    # If we do dossier 2 then dossier 1:
    # Time = max(p1_2 + p2_2, p1_2 + p1_1 + p2_1)
    
    # To minimize total time, we can use Johnson's rule:
    # If min(p1_a, p2_b) < min(p1_b, p2_a), then a should come before b
    
    if min(a[0], b[1]) < min(b[0], a[1]):
        return -1
    elif min(a[0], b[1]) > min(b[0], a[1]):
        return 1
    else:
        return 0

def main():
    j = int(sys.stdin.readline().strip())
    dossiers = []
    
    for _ in range(j):
        p1, p2 = map(int, sys.stdin.readline().strip().split())
        dossiers.append((p1, p2))
    
    # Sort according to Johnson's rule
    dossiers.sort(key=cmp_to_key(compare_dossiers))
    
    # Simulate the process
    time_initial_review = 0  # Time when initial reviewer finishes current dossier
    time_final_signoff = 0   # Time when final sign-off desk finishes current dossier
    
    for p1, p2 in dossiers:
        # Initial reviewer starts this dossier at time_initial_review
        # and finishes at time_initial_review + p1
        time_initial_review += p1
        
        # This dossier arrives at the final sign-off desk at time_initial_review
        # The final sign-off desk can start working on it at 
        # max(time_final_signoff, time_initial_review)
        # and finishes at that time + p2
        time_final_signoff = max(time_final_signoff, time_initial_review) + p2
    
    print(time_final_signoff)

if __name__ == "__main__":
    main()
```
