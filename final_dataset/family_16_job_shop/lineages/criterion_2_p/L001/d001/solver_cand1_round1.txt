```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (initial review and final sign-off).
    # We want to minimize the makespan.
    # Johnson's Rule:
    # 1. Divide the jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    order = set1 + set2
    
    # Calculate makespan
    time_p1 = 0
    time_p2 = 0
    
    for p1, p2 in order:
        time_p1 += p1
        # The second machine can start only after the first machine finishes the job
        # AND the second machine finishes the previous job.
        if time_p2 < time_p1:
            time_p2 = time_p1 + p2
        else:
            time_p2 += p2
            
    print(time_p2)

if __name__ == '__main__':
    solve()
```
