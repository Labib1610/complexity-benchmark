```python
import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    dossiers = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        dossiers.append((p1, p2))
        idx += 2
    
    # Sort by min(p1, p2)
    dossiers.sort(key=lambda x: min(x[0], x[1]))
    
    total_p1 = 0
    total_p2 = sum(p2 for p1, p2 in dossiers)
    
    ans = 0
    for p1, p2 in dossiers:
        total_p1 += p1
        total_p2 -= p2
        val = total_p1 + total_p2 + p2
        if val > ans:
            ans = val
    
    print(ans)

solve()
```
