```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    ships = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        ships.append((p1, p2))
        idx += 2
    
    # This is a variation of Johnson's Rule for two-machine flow shop scheduling.
    # To minimize the makespan (total time), we sort the jobs:
    # Group 1: p1 < p2, sort by p1 ascending.
    # Group 2: p1 >= p2, sort by p2 descending.
    
    group1 = []
    group2 = []
    for p1, p2 in ships:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate completion time
    # current_p1_end: time when the current ship finishes the asteroid field
    # current_p2_end: time when the current ship finishes the ion storm
    current_p1_end = 0
    current_p2_end = 0
    
    for p1, p2 in order:
        current_p1_end += p1
        # The ship enters the ion storm at max(current_p1_end, current_p2_end)
        # and finishes at that time + p2
        current_p2_end = max(current_p1_end, current_p2_end) + p2
        
    print(current_p2_end)

if __name__ == '__main__':
    solve()
```
