import sys

def solve():
    # Let the sequence of vessels be v_1, v_2, ..., v_j.
    # Let (b_i, s_i) be the bow and stern parameters of vessel v_i.
    # Let B_i be the time of the Bow Ping of vessel v_i.
    # Let S_i be the time of the Stern Ping of vessel v_i.
    # B_1 = b_1
    # B_i = B_{i-1} + b_i for i > 1
    # So B_i = sum_{k=1}^i b_k
    # S_1 = B_1 + s_1 = b_1 + s_1
    # S_i = max(B_i, S_{i-1}) + s_i
    #
    # Let T_i = S_i - B_i.
    # S_i = max(B_i, S_{i-1}) + s_i
    # S_i - B_i = max(0, S_{i-1} - B_i) + s_i
    # S_i - B_i = max(0, S_{i-1} - (B_{i-1} + b_i)) + s_i
    # T_i = max(0, T_{i-1} - b_i) + s_i
    #
    # We want to minimize S_j = B_j + T_j.
    # B_j = sum_{k=1}^j b_k is constant regardless of order.
    # We need to minimize T_j.
    # T_1 = s_1
    # T_i = max(s_i, T_{i-1} - b_i + s_i)
    #
    # This is equivalent to Johnson's Rule for the two-machine flow shop problem
    # where we want to minimize the makespan.
    # Machine 1: time b_i, Machine 2: time s_i.
    # The optimal order is to sort vessels with b_i < s_i in increasing order of b_i,
    # and vessels with b_i >= s_i in decreasing order of s_i.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        b = int(input_data[idx])
        s = int(input_data[idx+1])
        vessels.append((b, s))
        idx += 2
    
    group1 = [v for v in vessels if v[0] < v[1]]
    group2 = [v for v in vessels if v[0] >= v[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_b = 0
    current_s = 0
    for b, s in order:
        current_b += b
        current_s = max(current_b, current_s) + s
        
    print(current_s)

if __name__ == '__main__':
    solve()
