```python
import sys

def solve():
    # The problem asks to minimize the sequence rating.
    # Let the sites be (s_i, l_i).
    # For a fixed sequence, the rating is max_{k=1..j} (sum_{i=1}^k s_i + sum_{i=k}^j l_i).
    # This can be rewritten as:
    # sum_{i=1}^j l_i + max_{k=1..j} (sum_{i=1}^k s_i - sum_{i=1}^{k-1} l_i)
    # Let x_i = s_i - l_i.
    # The expression is sum(l_i) + max_{k=1..j} (sum_{i=1}^k x_i + l_k).
    # Actually, let's re-evaluate:
    # Resistance at k = (s_1 + ... + s_k) + (l_k + ... + l_j)
    # = (s_1 + ... + s_k) + (l_1 + ... + l_j) - (l_1 + ... + l_{k-1})
    # = sum(l) + (s_1 - 0) + (s_2 - l_1) + (s_3 - l_2) + ... + (s_k - l_{k-1})
    # Let x_i = s_i - l_i.
    # Resistance at k = sum(l) + s_1 + sum_{i=2}^k x_{i-1}
    # To minimize the maximum of these, we should sort the sites.
    # If we sort by s_i + l_i, it doesn't necessarily work.
    # Let's look at the condition: we want to minimize max_k (prefix_s_k + suffix_l_k).
    # This is a classic scheduling problem.
    # If we have two adjacent sites (s1, l1) and (s2, l2):
    # Order 1, 2: max(s1 + l1 + l2, s1 + s2 + l2)
    # Order 2, 1: max(s2 + l2 + l1, s2 + s1 + l1)
    # We prefer 1, 2 if max(s1+l1+l2, s1+s2+l2) < max(s2+l2+l1, s2+s1+l1)
    # This simplifies to: if s1 < l2, put 1 first. If s2 < l1, put 2 first.
    # This is equivalent to sorting by s_i < l_i.
    # Specifically, sites with s_i < l_i should come first (sorted by s_i ascending),
    # and sites with s_i >= l_i should come last (sorted by l_i descending).

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sites = []
    idx = 1
    for _ in range(j):
        s = int(input_data[idx])
        l = int(input_data[idx+1])
        sites.append((s, l))
        idx += 2
    
    group1 = [x for x in sites if x[0] < x[1]]
    group2 = [x for x in sites if x[0] >= x[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate sequence rating
    s_sum = 0
    l_sum = sum(x[1] for x in sequence)
    
    max_res = 0
    current_l_sum = l_sum
    for s, l in sequence:
        s_sum += s
        res = s_sum + current_l_sum
        if res > max_res:
            max_res = res
        current_l_sum -= l
        
    print(max_res)

solve()
```
