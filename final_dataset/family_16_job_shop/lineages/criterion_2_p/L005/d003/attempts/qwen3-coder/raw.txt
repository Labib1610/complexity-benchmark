```python
import sys
from functools import cmp_to_key

def compare(vessel1, vessel2):
    # Compare two vessels to determine their order
    # We want to sort such that the total time is minimized
    # For two vessels A and B with parameters (a1, a2) and (b1, b2)
    # Compare a1 + max(a2, b1) + b2 vs b1 + max(b2, a1) + a2
    # Simplifying: a1 + max(a2, b1) + b2 vs b1 + max(b2, a1) + a2
    # Rearranging: a1 + max(a2, b1) + b2 - a2 - b1 - max(b2, a1) vs 0
    # Which is equivalent to comparing a1 - max(b2, a1) + max(a2, b1) - b1 vs b2 - max(a2, b1) + max(b2, a1) - a1
    # Simpler approach: compare a1 - b2 vs b1 - a2
    
    a1, a2 = vessel1
    b1, b2 = vessel2
    
    # If a1 - b2 < b1 - a2, then vessel1 should come before vessel2
    if a1 - b2 < b1 - a2:
        return -1
    elif a1 - b2 > b1 - a2:
        return 1
    else:
        return 0

def calculate_time(vessels):
    if not vessels:
        return 0
        
    time_bow = 0
    time_stern = 0
    
    for bow_param, stern_param in vessels:
        time_bow += bow_param
        time_stern = max(time_bow, time_stern) + stern_param
    
    return time_stern

def main():
    j = int(sys.stdin.readline().strip())
    vessels = []
    
    for _ in range(j):
        t1, t2 = map(int, sys.stdin.readline().strip().split())
        vessels.append((t1, t2))
    
    # Sort vessels based on the comparison function
    sorted_vessels = sorted(vessels, key=cmp_to_key(compare))
    
    # Calculate the minimum time
    result = calculate_time(sorted_vessels)
    print(result)

if __name__ == "__main__":
    main()
```
