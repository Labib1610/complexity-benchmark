```python
import sys

def solve():
    # The problem asks to minimize the final Secondary Milestone.
    # Let the sequence of vessels be (p_i, s_i).
    # Let P_k = sum_{i=1}^k p_i.
    # Let S_k = max(P_k, S_{k-1}) + s_k, with S_0 = 0.
    # This is equivalent to S_j = max_{1 <= k <= j} (P_k + sum_{i=k}^j s_i).
    # S_j = max_{1 <= k <= j} (sum_{i=1}^k p_i + sum_{i=k}^j s_i)
    # S_j = sum_{i=1}^j s_i + max_{1 <= k <= j} (sum_{i=1}^k p_i - sum_{i=1}^{k-1} s_i)
    # Let P_0 = 0, S_0 = 0.
    # S_j = sum_{i=1}^j s_i + max_{1 <= k <= j} (P_k - sum_{i=1}^{k-1} s_i)
    # To minimize this, we want to order the vessels.
    # This is a classic scheduling problem (Johnson's rule variant).
    # The optimal order is to sort by s_i. Specifically, if we have two adjacent
    # vessels (p1, s1) and (p2, s2), the order (p1, s1), (p2, s2) is better if:
    # max(p1+s1, p1+p2+s2) < max(p2+s2, p2+p1+s1)
    # This simplifies to sorting by s_i in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        p = int(input_data[idx])
        s = int(input_data[idx+1])
        vessels.append((p, s))
        idx += 2
    
    # Sort by secondary duration descending
    vessels.sort(key=lambda x: x[1], reverse=True)
    
    current_p = 0
    current_s = 0
    for p, s in vessels:
        current_p += p
        current_s = max(current_p, current_s) + s
        
    print(current_s)

if __name__ == '__main__':
    solve()
```
