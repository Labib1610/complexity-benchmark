import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        vessels.append((t1, t2))
        idx += 2
    
    # Let the sequence be p_1, p_2, ..., p_j.
    # Let U_i be the Upstream time and D_i be the Downstream time for vessel i.
    # The time vessel i finishes Upstream is S_i = sum_{k=1}^i U_{p_k}.
    # The time vessel i finishes Downstream is F_i = max(S_i, F_{i-1}) + D_{p_i}.
    # This is a classic scheduling problem (Johnson's Rule variant).
    # To minimize the completion time of the last job, we sort jobs:
    # Jobs with U_i < D_i come first (sorted by increasing U_i).
    # Jobs with U_i >= D_i come last (sorted by decreasing D_i).
    
    group1 = []
    group2 = []
    for t1, t2 in vessels:
        if t1 < t2:
            group1.append((t1, t2))
        else:
            group2.append((t1, t2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    current_upstream_finish = 0
    current_downstream_finish = 0
    
    for t1, t2 in sequence:
        current_upstream_finish += t1
        current_downstream_finish = max(current_upstream_finish, current_downstream_finish) + t2
        
    print(current_downstream_finish)

if __name__ == '__main__':
    solve()
