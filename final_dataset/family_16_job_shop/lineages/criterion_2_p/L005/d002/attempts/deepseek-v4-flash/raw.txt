We can solve this by sorting the vessels in decreasing order of their Beta phase duration, then simulating the process.

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    j = int(data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        t1 = int(data[idx])
        t2 = int(data[idx+1])
        vessels.append((t1, t2))
        idx += 2

    # Sort by Beta duration descending
    vessels.sort(key=lambda x: x[1], reverse=True)

    alpha_end = 0
    beta_end = 0
    for t1, t2 in vessels:
        alpha_end += t1
        if beta_end > alpha_end:
            beta_end += t2
        else:
            beta_end = alpha_end + t2

    print(beta_end)

if __name__ == "__main__":
    main()
```
