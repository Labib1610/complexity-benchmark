import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        vessels.append((t1, t2))
        idx += 2
    
    # Johnson's Rule logic:
    # To minimize the completion time of the last job in a two-stage flow shop,
    # we sort jobs based on the minimum of their Alpha and Beta durations.
    # Jobs with smaller Alpha go first (ascending order of Alpha),
    # jobs with smaller Beta go last (descending order of Beta).
    
    group1 = []
    group2 = []
    
    for t1, t2 in vessels:
        if t1 < t2:
            group1.append((t1, t2))
        else:
            group2.append((t1, t2))
            
    # Sort group1 by t1 ascending
    group1.sort(key=lambda x: x[0])
    # Sort group2 by t2 descending
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate completion time
    # alpha_sum tracks the time when the current vessel finishes its Alpha phase
    # beta_finish tracks the time when the current vessel finishes its Beta phase
    alpha_sum = 0
    beta_finish = 0
    
    for t1, t2 in order:
        alpha_sum += t1
        # Beta phase starts at max(alpha_sum, previous_beta_finish)
        # Beta phase ends at start_time + t2
        if alpha_sum > beta_finish:
            beta_finish = alpha_sum + t2
        else:
            beta_finish = beta_finish + t2
            
    print(beta_finish)

if __name__ == '__main__':
    solve()
