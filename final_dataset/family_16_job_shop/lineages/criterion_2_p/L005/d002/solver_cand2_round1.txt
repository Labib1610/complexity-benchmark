import sys

def solve():
    # The problem asks to minimize X_j = max(E_k, X_{k-1}) + t2_k.
    # This is a variation of Johnson's Rule for the two-machine flow shop scheduling problem.
    # The goal is to minimize the makespan on two machines (entry gate and exit gate).
    # Johnson's Rule:
    # 1. List all jobs.
    # 2. Divide jobs into two sets:
    #    Set A: jobs where t1 < t2
    #    Set B: jobs where t1 >= t2
    # 3. Sort Set A in ascending order of t1.
    # 4. Sort Set B in descending order of t2.
    # 5. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    ships = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        ships.append((t1, t2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for t1, t2 in ships:
        if t1 < t2:
            set_a.append((t1, t2))
        else:
            set_b.append((t1, t2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    e_k = 0
    x_k = 0
    
    for t1, t2 in sequence:
        e_k += t1
        x_k = max(e_k, x_k) + t2
        
    print(x_k)

if __name__ == '__main__':
    solve()
