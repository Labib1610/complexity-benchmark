import sys

def solve():
    # The problem asks to minimize the completion time of the last packet.
    # Let the sequence of packets be p_1, p_2, ..., p_j.
    # Let t1_i be the source transmission time and t2_i be the relay transmission time.
    # The relay node starts receiving packet i at time sum(t1_1...t1_{i-1})
    # and finishes receiving it at time sum(t1_1...t1_i).
    # Let S_i = sum(t1_1...t1_i).
    # The relay node starts transmitting packet i at time max(S_i, finish_time_{i-1}).
    # The relay node finishes transmitting packet i at time max(S_i, finish_time_{i-1}) + t2_i.
    
    # This is a variation of Johnson's Rule for two-machine flow shop scheduling.
    # To minimize the makespan, we sort the packets:
    # 1. Packets with t1 < t2 should come first, sorted by t1 ascending.
    # 2. Packets with t1 >= t2 should come last, sorted by t2 descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    packets = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        packets.append((t1, t2))
        idx += 2
        
    group1 = []
    group2 = []
    for p in packets:
        if p[0] < p[1]:
            group1.append(p)
        else:
            group2.append(p)
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_packets = group1 + group2
    
    current_s = 0
    finish_time = 0
    
    for t1, t2 in sorted_packets:
        current_s += t1
        if finish_time < current_s:
            finish_time = current_s + t2
        else:
            finish_time += t2
            
    print(finish_time)

if __name__ == '__main__':
    solve()
