```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    jobs = []
    for _ in range(n):
        t1 = int(next(it)); t2 = int(next(it))
        if t1 <= t2:
            # early group, sort by increasing t1
            jobs.append((0, t1, -t2, t1, t2))
        else:
            # late group, sort by decreasing t2
            jobs.append((1, -t2, t1, t1, t2))
    # sort by the constructed keys
    jobs.sort()
    time_a = 0
    time_b = 0
    for _, _, _, t1, t2 in jobs:
        time_a += t1
        if time_b < time_a:
            time_b = time_a
        time_b += t2
    sys.stdout.write(str(time_b))

if __name__ == "__main__":
    solve()
```
