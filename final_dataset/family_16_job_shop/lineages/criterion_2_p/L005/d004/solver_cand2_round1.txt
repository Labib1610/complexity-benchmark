```python
import sys

def solve():
    # The problem asks to arrange j inducers (h_i, c_i) to minimize the peak stress.
    # Let the sequence be (h_1, c_1), (h_2, c_2), ..., (h_j, c_j).
    # The stress at position k is S_k = sum_{i=1}^k h_i + sum_{i=k}^j c_i.
    # S_k = sum_{i=1}^j c_i + sum_{i=1}^k (h_i - c_i).
    # Let d_i = h_i - c_i.
    # S_k = (sum_{i=1}^j c_i) + (sum_{i=1}^k d_i).
    # To minimize the maximum of S_k, we need to minimize the maximum prefix sum of d_i.
    # This is a classic scheduling problem:
    # If we have items with d_i, we want to order them such that the maximum prefix sum is minimized.
    # Split items into two groups:
    # 1. d_i >= 0: Sort by increasing d_i.
    # 2. d_i < 0: Sort by decreasing d_i.
    # Actually, the standard result for minimizing the maximum prefix sum is:
    # Put all d_i >= 0 first (sorted by d_i ascending), then all d_i < 0 (sorted by d_i descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    items = []
    sum_c = 0
    idx = 1
    for _ in range(j):
        h = int(input_data[idx])
        c = int(input_data[idx+1])
        items.append((h - c, h, c))
        sum_c += c
        idx += 2
        
    pos = [x for x in items if x[0] >= 0]
    neg = [x for x in items if x[0] < 0]
    
    pos.sort(key=lambda x: x[0])
    neg.sort(key=lambda x: x[0], reverse=True)
    
    sequence = pos + neg
    
    max_stress = 0
    current_prefix_d = 0
    for d, h, c in sequence:
        current_prefix_d += d
        max_stress = max(max_stress, sum_c + current_prefix_d)
        
    print(max_stress)

if __name__ == '__main__':
    solve()
```
