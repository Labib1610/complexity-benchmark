```python
import sys

def solve():
    # The total time for a chosen gallery k is:
    # Sum_{i=1 to k} t1_i + Sum_{i=k to j} t2_i
    # = Sum_{i=1 to j} t1_i + Sum_{i=k to j} (t2_i - t1_i)
    # Let S = Sum_{i=1 to j} t1_i.
    # The critic chooses k to maximize: S + Sum_{i=k to j} (t2_i - t1_i).
    # Let d_i = t2_i - t1_i.
    # We want to order the galleries to minimize:
    # max_{k=1 to j} (S + Sum_{i=k to j} d_i)
    # Let P_k = Sum_{i=k to j} d_i. We want to minimize max(P_1, P_2, ..., P_j).
    # Note that P_k = d_k + P_{k+1}, with P_{j+1} = 0.
    # This is a classic scheduling problem. To minimize the maximum prefix sum of 
    # a sequence (in reverse), we sort the elements d_i.
    # Specifically, if we have a sequence of d_i, the maximum prefix sum is minimized
    # by sorting d_i in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    galleries = []
    idx = 1
    sum_t1 = 0
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        galleries.append(t2 - t1)
        sum_t1 += t1
        idx += 2
    
    # Sort d_i in descending order to minimize the maximum partial sum
    galleries.sort(reverse=True)
    
    # The sequence of P_k values:
    # P_j = d_j
    # P_{j-1} = d_{j-1} + d_j
    # ...
    # P_1 = d_1 + d_2 + ... + d_j
    # We want to minimize max(P_1, ..., P_j).
    # With d_i sorted descending, the sequence P_k is increasing as k decreases.
    # The maximum is P_1 = Sum(d_i).
    # Wait, let's re-check:
    # If we sort d_i descending: d_1 >= d_2 >= ... >= d_j
    # P_j = d_j
    # P_{j-1} = d_{j-1} + d_j
    # ...
    # P_1 = d_1 + d_2 + ... + d_j
    # The maximum is P_1.
    # Actually, the problem is to minimize the maximum of {Sum_{i=k to j} d_i}.
    # If we sort d_i in ascending order: d_1 <= d_2 <= ... <= d_j
    # P_j = d_j
    # P_{j-1} = d_{j-1} + d_j
    # ...
    # P_1 = Sum_{i=1 to j} d_i
    # This doesn't seem right. Let's re-evaluate.
    # We want to minimize max_{k} (Sum_{i=k to j} d_i).
    # This is equivalent to minimizing the maximum suffix sum.
    # To minimize the maximum suffix sum, we should sort d_i in ascending order.
    
    galleries.sort()
    
    max_val = -float('inf')
    current_suffix_sum = 0
    for i in range(j - 1, -1, -1):
        current_suffix_sum += galleries[i]
        if current_suffix_sum > max_val:
            max_val = current_suffix_sum
            
    print(sum_t1 + max_val)

solve()
```
