```python
import sys

def solve():
    # The problem asks to minimize the maximum path cost.
    # A path moving down at column k has cost:
    # sum(top[0...k]) + sum(bottom[k...j-1])
    # Let S_top = sum(top), S_bot = sum(bottom).
    # Cost(k) = sum(top[0...k]) + S_bot - sum(bottom[0...k-1])
    # Cost(k) = S_bot + sum(top[0...k] - bottom[0...k-1])
    # Let d_i = top_i - bottom_i.
    # Cost(k) = S_bot + top_0 + sum_{i=1 to k} (top_i - bottom_{i-1})
    # This is equivalent to:
    # Cost(k) = sum(top[0...k]) + sum(bottom[k...j-1])
    # Let T = sum(top), B = sum(bottom).
    # Cost(k) = T - sum(top[k+1...j-1]) + B - sum(bottom[0...k-1])
    # Let x_i = top_i, y_i = bottom_i.
    # Cost(k) = sum_{i=0}^k x_i + sum_{i=k}^{j-1} y_i
    # To minimize the maximum of these, we want to balance the prefix sums.
    # This is a classic problem: we want to order tiles (x_i, y_i) such that
    # max_k (sum_{i=0}^k x_i + sum_{i=k}^{j-1} y_i) is minimized.
    # Let S = sum(x_i) + sum(y_i).
    # Cost(k) = sum_{i=0}^k x_i + (S - sum_{i=0}^{k-1} y_i - x_k)
    # Cost(k) = S + sum_{i=0}^{k-1} (x_i - y_i)
    # To minimize the maximum, we should sort tiles by (x_i - y_i).
    # If we sort such that x_i - y_i is increasing, the prefix sums of (x_i - y_i)
    # will be as small as possible for the early k and large for late k.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        tiles.append((t1, t2))
        idx += 2
        
    # Sort by x_i - y_i
    tiles.sort(key=lambda p: p[0] - p[1])
    
    # Calculate costs
    # Cost(k) = sum_{i=0}^k x_i + sum_{i=k}^{j-1} y_i
    # Let P_x[k] be prefix sum of x, P_y[k] be prefix sum of y
    # Cost(k) = P_x[k] + (Total_y - P_y[k-1])
    
    total_y = sum(t[1] for t in tiles)
    
    max_cost = 0
    current_x = 0
    current_y = 0
    
    # k goes from 0 to j-1
    for i in range(j):
        current_x += tiles[i][0]
        # Cost if we go down at column i
        # Path: top[0...i] then bottom[i...j-1]
        cost = current_x + (total_y - current_y)
        if cost > max_cost:
            max_cost = cost
        current_y += tiles[i][1]
        
    print(max_cost)

solve()
```
