```python
import sys

def solve():
    # The recurrence is:
    # A_k = A_{k-1} + X_k
    # B_k = max(A_k, B_{k-1}) + Y_k
    #
    # Let's analyze the effect of swapping two adjacent elements (X_i, Y_i) and (X_j, Y_j).
    # Suppose we have a prefix with A_{k-1} and B_{k-1}.
    # Order (i, j):
    # A_i = A_{k-1} + X_i
    # B_i = max(A_i, B_{k-1}) + Y_i
    # A_j = A_i + X_j = A_{k-1} + X_i + X_j
    # B_j = max(A_j, B_i) + Y_j = max(A_{k-1} + X_i + X_j, max(A_{k-1} + X_i, B_{k-1}) + Y_i) + Y_j
    #
    # This is a classic scheduling problem. The optimal order is to sort by Y_i in descending order.
    # If Y_i > Y_j, we prefer (i, j).
    # Let's verify:
    # B_j = max(A_{k-1} + X_i + X_j + Y_j, B_{k-1} + Y_i + Y_j, A_{k-1} + X_i + Y_i + Y_j)
    # B_j = max(A_{k-1} + X_i + X_j + Y_j, B_{k-1} + Y_i + Y_j, A_{k-1} + X_i + Y_i + Y_j)
    #
    # Comparing (i, j) vs (j, i):
    # We want to minimize max(A_{k-1} + X_i + X_j + Y_j, B_{k-1} + Y_i + Y_j, A_{k-1} + X_i + Y_i + Y_j)
    # vs max(A_{k-1} + X_j + X_i + Y_i, B_{k-1} + Y_j + Y_i, A_{k-1} + X_j + Y_j + Y_i)
    #
    # The term B_{k-1} + Y_i + Y_j is common.
    # We compare max(X_i + X_j + Y_j, X_i + Y_i + Y_j) vs max(X_j + X_i + Y_i, X_j + Y_j + Y_i)
    # This simplifies to comparing max(X_i + Y_j, Y_i + Y_j) + X_j vs max(X_j + Y_i, Y_j + Y_i) + X_i
    # Actually, the standard result for this specific recurrence B_k = max(A_k, B_{k-1}) + Y_k
    # is to sort by Y_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    elements = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        elements.append((t1, t2))
        idx += 2
        
    # Sort by Y descending
    elements.sort(key=lambda x: x[1], reverse=True)
    
    a = 0
    b = 0
    for x, y in elements:
        a += x
        b = max(a, b) + y
        
    print(b)

if __name__ == '__main__':
    solve()
```
