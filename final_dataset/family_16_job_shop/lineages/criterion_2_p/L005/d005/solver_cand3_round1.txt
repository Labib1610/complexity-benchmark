```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    dominoes = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        dominoes.append((t1, t2))
        idx += 2
    
    # Let the sequence be (t1_1, t2_1), (t1_2, t2_2), ..., (t1_j, t2_j).
    # A path consists of staying on the top row for some dominoes 1..k,
    # moving down at domino k, and staying on the bottom row for k..j.
    # The score for a fixed k is:
    # sum(t1_1...t1_k) + sum(t2_k...t2_j)
    # = sum(t1_1...t1_j) + sum(t2_k...t2_j) - sum(t1_k...t1_j) + t1_k
    # = sum(t1_all) + sum(t2_all) - sum(t2_1...t2_{k-1}) - sum(t1_{k+1}...t1_j)
    # Let S1 = sum(t1_i), S2 = sum(t2_i).
    # Score(k) = S1 + S2 - sum_{i=1}^{k-1} t2_i - sum_{i=k+1}^{j} t1_i
    # To minimize the maximum score, we want to minimize the value of
    # max_k (S1 + S2 - sum_{i=1}^{k-1} t2_i - sum_{i=k+1}^{j} t1_i).
    # This is equivalent to maximizing min_k (sum_{i=1}^{k-1} t2_i + sum_{i=k+1}^{j} t1_i).
    # Let A_i = t1_i and B_i = t2_i.
    # We want to order the dominoes to maximize min_k (sum_{i=1}^{k-1} B_i + sum_{i=k+1}^{j} A_i).
    # Let f(k) = sum_{i=1}^{k-1} B_i + sum_{i=k+1}^{j} A_i.
    # Note that f(k+1) - f(k) = B_k - A_k.
    # To make the minimum of f(k) as large as possible, we should sort the dominoes
    # such that the sequence of B_i - A_i is non-decreasing.
    # If B_i - A_i is sorted, the minimum of f(k) will be at the boundaries.
    
    dominoes.sort(key=lambda x: x[1] - x[0])
    
    s1 = sum(d[0] for d in dominoes)
    s2 = sum(d[1] for d in dominoes)
    
    # Calculate scores for all k from 0 to j-1
    # k is the index where we move down (0-indexed)
    # Score(k) = sum(t1[0...k]) + sum(t2[k...j-1])
    
    current_t1 = 0
    current_t2 = s2
    min_max_score = float('inf')
    
    for i in range(j):
        current_t1 += dominoes[i][0]
        score = current_t1 + current_t2
        if score < min_max_score:
            min_max_score = score
        current_t2 -= dominoes[i][1]
        
    print(min_max_score)

solve()
```
