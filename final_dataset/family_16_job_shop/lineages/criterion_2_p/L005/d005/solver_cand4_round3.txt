```python
import sys

def solve():
    # The problem asks to minimize max_{1 <= k <= j} (sum_{i=1}^k A_i + sum_{i=k}^j B_i).
    # This is a classic scheduling problem known as Johnson's Rule.
    # We divide the signals into two groups:
    # Group 1: A_i < B_i (sort by A_i ascending)
    # Group 2: A_i >= B_i (sort by B_i descending)
    # This ordering minimizes the makespan of a two-machine flow shop.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    signals = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        signals.append((a, b))
        idx += 2
        
    group1 = []
    group2 = []
    for s in signals:
        if s[0] < s[1]:
            group1.append(s)
        else:
            group2.append(s)
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate the makespan for the optimal order
    # Let alpha_time be the completion time of the current signal on Filter Alpha
    # Let beta_time be the completion time of the current signal on Filter Beta
    alpha_time = 0
    beta_time = 0
    
    for a, b in order:
        alpha_time += a
        # The signal can start on Filter Beta at max(alpha_time, previous_beta_time)
        if alpha_time > beta_time:
            beta_time = alpha_time + b
        else:
            beta_time = beta_time + b
            
    print(beta_time)

if __name__ == '__main__':
    solve()
```
