```python
import sys

def solve():
    # The problem asks us to arrange vessels (t1_i, t2_i) in a sequence
    # such that the maximum duration of any valid path is minimized.
    # A path consists of traversing upper decks of vessels 1 to k,
    # then lower decks of vessels k to j.
    # Total duration for a chosen k (1 <= k <= j) is:
    # D(k) = sum_{i=1}^k t1_i + sum_{i=k}^j t2_i
    # Let S1_k = sum_{i=1}^k t1_i and S2_k = sum_{i=k}^j t2_i.
    # D(k) = S1_k + S2_k.
    # Let T1_total = sum_{i=1}^j t1_i and T2_total = sum_{i=1}^j t2_i.
    # Note that S2_k = T2_total - sum_{i=1}^{k-1} t2_i.
    # D(k) = sum_{i=1}^k t1_i + T2_total - sum_{i=1}^{k-1} t2_i
    # D(k) = T2_total + t1_k + sum_{i=1}^{k-1} (t1_i - t2_i)
    # Let d_i = t1_i - t2_i.
    # D(k) = T2_total + t1_k + sum_{i=1}^{k-1} d_i.
    # We want to minimize max_{k=1..j} D(k).
    # This is equivalent to minimizing max_{k=1..j} (t1_k + sum_{i=1}^{k-1} d_i).
    # This is a classic scheduling problem. To minimize the maximum prefix sum
    # of a sequence, we should sort the items based on their d_i values.
    # Specifically, if we have two adjacent items i and j with (t1_i, t2_i) and (t1_j, t2_j),
    # swapping them changes the max of the two path sums.
    # The optimal order is to sort by d_i = t1_i - t2_i in ascending order.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        vessels.append((t1, t2))
        idx += 2
    
    # Sort by t1 - t2 ascending
    vessels.sort(key=lambda x: x[0] - x[1])
    
    t2_total = sum(v[1] for v in vessels)
    
    max_duration = 0
    current_prefix_t1 = 0
    current_prefix_t2 = 0
    
    # D(k) = sum_{i=1}^k t1_i + (T2_total - sum_{i=1}^{k-1} t2_i)
    for i in range(j):
        t1, t2 = vessels[i]
        current_prefix_t1 += t1
        # D(k) where k is 1-indexed (i+1)
        # D(i+1) = current_prefix_t1 + (t2_total - current_prefix_t2)
        duration = current_prefix_t1 + (t2_total - current_prefix_t2)
        if duration > max_duration:
            max_duration = duration
        current_prefix_t2 += t2
        
    print(max_duration)

solve()
```
