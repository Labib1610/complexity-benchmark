```python
import sys

def solve():
    # The expression is: sum_{i=1}^k x_{P[i]} + sum_{i=k}^j y_{P[i]}
    # This can be rewritten as:
    # sum_{i=1}^j y_{P[i]} + sum_{i=1}^k (x_{P[i]} - y_{P[i]})
    # Let d_i = x_i - y_i.
    # Then E(P) = sum(y_i) + max_{1 <= k <= j} (sum_{i=1}^k d_{P[i]})
    # To minimize E(P), we need to minimize the maximum prefix sum of d_{P[i]}.
    # This is a classic problem: to minimize the maximum prefix sum of a sequence,
    # we should sort the elements such that negative values come first,
    # followed by positive values.
    # Specifically, if we have d_i values, we sort them in ascending order.
    # Let the sorted values be d'_1 <= d'_2 <= ... <= d'_j.
    # The prefix sums will be S_k = sum_{i=1}^k d'_i.
    # The maximum prefix sum will be max(0, max_{1 <= k <= j} S_k).
    # Since we must pick at least one element (k >= 1), the max is over S_1, ..., S_j.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    pairs = []
    idx = 1
    sum_y = 0
    for _ in range(j):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        pairs.append(x - y)
        sum_y += y
        idx += 2
        
    pairs.sort()
    
    max_prefix_sum = -float('inf')
    current_sum = 0
    for d in pairs:
        current_sum += d
        if current_sum > max_prefix_sum:
            max_prefix_sum = current_sum
            
    print(sum_y + max_prefix_sum)

if __name__ == '__main__':
    solve()
```
