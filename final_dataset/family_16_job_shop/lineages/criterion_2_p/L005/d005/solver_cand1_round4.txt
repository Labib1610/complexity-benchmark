```python
import sys

def solve():
    # Let the sequence be (a1, b1), (a2, b2), ..., (aj, bj).
    # Cost at split k: C_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # C_k = sum_{i=1}^j b_i + sum_{i=1}^k (a_i - b_i)
    # Let S = sum_{i=1}^j b_i.
    # C_k = S + sum_{i=1}^k (a_i - b_i).
    # Let d_i = a_i - b_i.
    # C_k = S + prefix_sum_d[k].
    # We want to minimize max_{k=1..j} (S + prefix_sum_d[k]).
    # This is equivalent to minimizing max_{k=1..j} (prefix_sum_d[k]).
    # To minimize the maximum prefix sum of d_i, we should sort d_i in descending order.
    # If we sort d_i such that d_1 >= d_2 >= ... >= d_j,
    # then the prefix sums are maximized at the end if all d_i > 0,
    # or we want to keep the prefix sums as small as possible.
    # Actually, if we sort d_i in descending order, the prefix sums are:
    # P_1 = d_1
    # P_2 = d_1 + d_2
    # ...
    # P_j = d_1 + ... + d_j
    # If we sort d_i in ascending order: d_1 <= d_2 <= ... <= d_j
    # P_1 = d_1
    # P_2 = d_1 + d_2
    # ...
    # P_j = d_1 + ... + d_j
    # The maximum prefix sum is always P_j = sum(d_i) = sum(a_i) - sum(b_i).
    # Wait, the cost is C_k = S + P_k.
    # C_k = sum(b_i) + sum_{i=1}^k (a_i - b_i).
    # Let's re-examine:
    # C_k = (a_1 + ... + a_k) + (b_k + ... + b_j)
    # C_k = (a_1 + ... + a_k) + (b_k + b_{k+1} + ... + b_j)
    # C_k = (a_1 + ... + a_k) + (b_k + (S - (b_1 + ... + b_{k-1})))
    # This is not quite right.
    # Let S_b = sum_{i=1}^j b_i.
    # C_k = sum_{i=1}^k a_i + S_b - sum_{i=1}^{k-1} b_i
    # C_k = S_b + a_k + sum_{i=1}^{k-1} (a_i - b_i)
    # To minimize the maximum of these, we should sort by (a_i - b_i).
    # Specifically, sorting by (a_i - b_i) in descending order is optimal.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    elements = []
    idx = 1
    sum_b = 0
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        elements.append((t1, t2))
        sum_b += t2
        idx += 2
    
    # Sort by (a_i - b_i) descending
    elements.sort(key=lambda x: x[0] - x[1], reverse=True)
    
    max_cost = -float('inf')
    current_a_sum = 0
    current_b_sum = 0
    
    # C_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # Let's precalculate suffix sums of b
    suffix_b = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suffix_b[i] = suffix_b[i+1] + elements[i][1]
        
    for k in range(1, j + 1):
        current_a_sum += elements[k-1][0]
        cost = current_a_sum + suffix_b[k-1]
        if cost > max_cost:
            max_cost = cost
            
    print(max_cost)

solve()
```
