import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    vessels = []
    for _ in range(j):
        t1 = int(next(it))
        t2 = int(next(it))
        vessels.append((t1, t2))
    
    # Sort by min(t1, t2)
    vessels.sort(key=lambda x: min(x[0], x[1]))
    
    # Split into two groups based on comparison
    group1 = []  # t1 <= t2
    group2 = []  # t1 > t2
    for t1, t2 in vessels:
        if t1 <= t2:
            group1.append((t1, t2))
        else:
            group2.append((t1, t2))
    
    # Group1: order as is (sorted by t1)
    # Group2: reverse order (sorted by t2)
    group2.reverse()
    
    ordered = group1 + group2
    
    # Compute prefix sums of t1
    prefix_t1 = [0] * (j + 1)
    for i in range(j):
        prefix_t1[i + 1] = prefix_t1[i] + ordered[i][0]
    
    # Compute suffix sums of t2
    suffix_t2 = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suffix_t2[i] = suffix_t2[i + 1] + ordered[i][1]
    
    # Find maximum chain duration
    max_chain = 0
    for k in range(1, j + 1):
        chain = prefix_t1[k] + suffix_t2[k - 1]
        if chain > max_chain:
            max_chain = chain
    
    print(max_chain)

solve()
