```python
import sys

def solve():
    # The problem asks to order segments (t1_i, t2_i) to minimize the maximum
    # strain over all possible deflection points k.
    # Let the chosen order be (a_1, b_1), (a_2, b_2), ..., (a_j, b_j).
    # The strain for a deflection at k is:
    # S_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # S_k = sum_{i=1}^j b_i + sum_{i=1}^k (a_i - b_i)
    # Let d_i = a_i - b_i. Then S_k = sum(b) + sum_{i=1}^k d_i.
    # To minimize max(S_k), we need to minimize max_{k} (sum_{i=1}^k d_i).
    # This is a classic problem: to minimize the maximum prefix sum,
    # we should sort the d_i values such that negative values come first.
    # If d_i < 0, they decrease the prefix sum, so they should be at the beginning.
    # If d_i >= 0, they increase the prefix sum, so they should be at the end.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    segments = []
    idx = 1
    sum_b = 0
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        segments.append((t1, t2))
        sum_b += t2
        idx += 2
        
    # d_i = t1_i - t2_i
    # Sort segments by d_i ascending.
    segments.sort(key=lambda x: x[0] - x[1])
    
    max_strain = -float('inf')
    current_prefix_d = 0
    
    for i in range(j):
        d_i = segments[i][0] - segments[i][1]
        current_prefix_d += d_i
        max_strain = max(max_strain, sum_b + current_prefix_d)
        
    print(max_strain)

if __name__ == '__main__':
    solve()
```
