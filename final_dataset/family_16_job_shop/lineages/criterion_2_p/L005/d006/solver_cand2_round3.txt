```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (Cleaving and Folding).
    # To minimize the makespan, we sort the jobs:
    # 1. Jobs with t1 < t2 should be processed in increasing order of t1.
    # 2. Jobs with t1 >= t2 should be processed in decreasing order of t2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        jobs.append((t1, t2))
        idx += 2
        
    # Johnson's Rule implementation
    group1 = [] # t1 < t2
    group2 = [] # t1 >= t2
    
    for job in jobs:
        if job[0] < job[1]:
            group1.append(job)
        else:
            group2.append(job)
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate the completion time
    cleave_time = 0
    fold_time = 0
    
    for t1, t2 in sequence:
        cleave_time += t1
        # Folding starts when cleaving is done AND previous folding is done
        fold_time = max(cleave_time, fold_time) + t2
        
    print(fold_time)

if __name__ == '__main__':
    solve()
```
