import sys

def solve():
    # The cost for a sequence pi_1, ..., pi_j with transfer point k is:
    # Sum_{i=1}^k t1_i + Sum_{i=k}^j t2_i
    # = Sum_{i=1}^j t1_i + Sum_{i=k}^j (t2_i - t1_i)
    # Let d_i = t2_i - t1_i.
    # Cost = Sum(t1) + Sum_{i=k}^j d_i
    # To minimize the maximum cost over all k, we want to arrange the zones
    # such that the maximum prefix sum of d_i (starting from the end) is minimized.
    # This is a variation of Johnson's rule.
    # We split zones into two groups:
    # Group 1: t1_i < t2_i (d_i > 0). These should be at the beginning.
    # Group 2: t1_i >= t2_i (d_i <= 0). These should be at the end.
    # Within Group 1, sort by t1_i ascending.
    # Within Group 2, sort by t2_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    zones = []
    idx = 1
    sum_t1 = 0
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        zones.append((t1, t2))
        sum_t1 += t1
        idx += 2
        
    group1 = []
    group2 = []
    for t1, t2 in zones:
        if t1 < t2:
            group1.append((t1, t2))
        else:
            group2.append((t1, t2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate max_{k} (Sum_{i=k}^j d_i)
    # Let S[k] = Sum_{i=k}^j d_i
    # S[j] = d_j
    # S[k] = d_k + S[k+1]
    d = [z[1] - z[0] for z in sequence]
    
    max_suffix_sum = -float('inf')
    current_suffix_sum = 0
    for i in range(j - 1, -1, -1):
        current_suffix_sum += d[i]
        if current_suffix_sum > max_suffix_sum:
            max_suffix_sum = current_suffix_sum
            
    print(sum_t1 + max_suffix_sum)

if __name__ == '__main__':
    solve()
