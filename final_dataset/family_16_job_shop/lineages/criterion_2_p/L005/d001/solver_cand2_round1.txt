```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vehicles = []
    idx = 1
    for _ in range(j):
        s = int(input_data[idx])
        h = int(input_data[idx+1])
        vehicles.append((s, h))
        idx += 2
    
    # Let S_i be straightway time and H_i be handling time for vehicle i.
    # If we have a sequence of vehicles, the finish time is:
    # F_1 = S_1 + H_1
    # F_i = max(F_{i-1}, sum_{k=1}^i S_k) + H_i
    # This is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # To minimize the makespan, sort vehicles:
    # 1. Vehicles with S_i < H_i in increasing order of S_i.
    # 2. Vehicles with S_i >= H_i in decreasing order of H_i.
    
    group1 = []
    group2 = []
    for s, h in vehicles:
        if s < h:
            group1.append((s, h))
        else:
            group2.append((s, h))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    current_straight_finish = 0
    current_handling_finish = 0
    
    for s, h in sequence:
        current_straight_finish += s
        current_handling_finish = max(current_straight_finish, current_handling_finish) + h
        
    print(current_handling_finish)

if __name__ == '__main__':
    solve()
```
