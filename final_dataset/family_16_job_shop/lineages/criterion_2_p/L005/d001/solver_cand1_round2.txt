import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We want to minimize the makespan of a 2-stage flow shop.
    # Johnson's Rule:
    # 1. Divide the jobs into two sets:
    #    Set A: jobs where t1 < t2
    #    Set B: jobs where t1 >= t2
    # 2. Sort Set A in ascending order of t1.
    # 3. Sort Set B in descending order of t2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        jobs.append((t1, t2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for t1, t2 in jobs:
        if t1 < t2:
            set_a.append((t1, t2))
        else:
            set_b.append((t1, t2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate the completion time
    # time1: time when stage 1 finishes for the current ship
    # time2: time when stage 2 finishes for the current ship
    time1 = 0
    time2 = 0
    
    for t1, t2 in sequence:
        time1 += t1
        # Stage 2 can start only after stage 1 finishes for this ship
        # AND after stage 2 finishes for the previous ship
        if time1 > time2:
            time2 = time1 + t2
        else:
            time2 = time2 + t2
            
    print(time2)

if __name__ == '__main__':
    solve()
