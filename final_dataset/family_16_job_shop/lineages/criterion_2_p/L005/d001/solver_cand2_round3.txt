import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (inspection bay and scanning tunnel).
    # We want to minimize the makespan (total time to finish all jobs).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 2. Sort Set A in ascending order of p1.
    # 3. Sort Set B in descending order of p2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate completion time
    # time_p1: time when current job finishes inspection
    # time_p2: time when current job finishes scanning
    time_p1 = 0
    time_p2 = 0
    
    for p1, p2 in sequence:
        time_p1 += p1
        # The scanning tunnel can start only after inspection is done
        # AND the scanning tunnel is free from the previous job.
        time_p2 = max(time_p1, time_p2) + p2
        
    print(time_p2)

if __name__ == '__main__':
    solve()
