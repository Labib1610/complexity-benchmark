```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    modules = []
    idx = 1
    total_t1 = 0
    total_t2 = 0
    
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        modules.append((t1, t2))
        total_t1 += t1
        total_t2 += t2
        idx += 2
        
    # Let the sequence of modules be m_1, m_2, ..., m_j.
    # Interference at slot k:
    # I_k = sum_{i=1}^{k-1} t1_i + (t1_k + t2_k) + sum_{i=k+1}^{j} t2_i
    # I_k = sum_{i=1}^{j} t2_i + sum_{i=1}^{k} (t1_i - t2_i)
    # Let D_i = t1_i - t2_i.
    # I_k = total_t2 + sum_{i=1}^{k} D_i
    # To minimize max(I_k), we need to arrange D_i such that the prefix sums
    # of D_i are as small as possible.
    # This is achieved by sorting D_i in ascending order.
    
    modules.sort(key=lambda x: x[0] - x[1])
    
    max_interference = 0
    current_sum_d = 0
    for t1, t2 in modules:
        current_sum_d += (t1 - t2)
        interference = total_t2 + current_sum_d
        if interference > max_interference:
            max_interference = interference
            
    print(max_interference)

if __name__ == '__main__':
    solve()
```
