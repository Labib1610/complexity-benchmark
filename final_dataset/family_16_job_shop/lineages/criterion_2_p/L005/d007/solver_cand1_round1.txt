import sys

def solve():
    # The problem asks to minimize the maximum peak pressure.
    # Let S be the sum of all Beta coefficients.
    # The running pressure starts at S.
    # For a vessel i with (t1_i, t2_i):
    # Peak_i = Current_Pressure + t1_i
    # New_Pressure = Current_Pressure + t1_i - t2_i
    #
    # Let the sequence be p_1, p_2, ..., p_j.
    # Peak_1 = S + t1_{p1}
    # Peak_2 = (S + t1_{p1} - t2_{p1}) + t1_{p2}
    # Peak_k = S + sum_{i=1}^{k-1} (t1_{pi} - t2_{pi}) + t1_{pk}
    #
    # We want to minimize max(Peak_k) for k=1..j.
    # This is equivalent to minimizing max(S + t1_{pk} - sum_{i=1}^{k-1} (t2_{pi} - t1_{pi})).
    #
    # This is a classic scheduling problem.
    # If we have two adjacent vessels i and j:
    # Order (i, j): peaks are S + t1_i and S + t1_i - t2_i + t1_j
    # Order (j, i): peaks are S + t1_j and S + t1_j - t2_j + t1_i
    # We prefer (i, j) if max(t1_i, t1_i - t2_i + t1_j) < max(t1_j, t1_j - t2_j + t1_i).
    #
    # Standard greedy approach:
    # 1. Vessels with t1 < t2 should come first, sorted by t1 ascending.
    # 2. Vessels with t1 >= t2 should come last, sorted by t2 descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    total_beta = 0
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        vessels.append((t1, t2))
        total_beta += t2
        idx += 2
        
    group1 = [] # t1 < t2
    group2 = [] # t1 >= t2
    
    for v in vessels:
        if v[0] < v[1]:
            group1.append(v)
        else:
            group2.append(v)
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    max_peak = 0
    current_pressure = total_beta
    for t1, t2 in sequence:
        peak = current_pressure + t1
        if peak > max_peak:
            max_peak = peak
        current_pressure = current_pressure + t1 - t2
        
    print(max_peak)

if __name__ == '__main__':
    solve()
