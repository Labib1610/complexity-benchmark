import sys

def solve():
    # The inspection cost for a sequence of dams (t1_1, t2_1), ..., (t1_j, t2_j)
    # with inspection station at k is:
    # Cost(k) = sum_{i=1}^k t1_i + sum_{i=k}^j t2_i
    # Cost(k) = sum_{i=1}^j t2_i + sum_{i=1}^k (t1_i - t2_i) + t2_k
    # Actually, let's rewrite:
    # Cost(k) = (sum_{i=1}^k t1_i) + (sum_{i=k}^j t2_i)
    # Let S1_k = sum_{i=1}^k t1_i and S2_k = sum_{i=k}^j t2_i
    # Cost(k) = S1_k + S2_k
    # Let T = sum_{i=1}^j t2_i. Then S2_k = T - sum_{i=1}^{k-1} t2_i.
    # Cost(k) = sum_{i=1}^k t1_i + T - sum_{i=1}^{k-1} t2_i
    # Cost(k) = T + t1_k + sum_{i=1}^{k-1} (t1_i - t2_i)
    
    # To minimize the maximum Cost(k), we want to arrange the dams such that
    # the partial sums of (t1_i - t2_i) are kept as small as possible.
    # Let d_i = t1_i - t2_i.
    # Cost(k) = T + t1_k + sum_{i=1}^{k-1} d_i
    # If we sort the dams such that d_i are in non-decreasing order,
    # we minimize the growth of the partial sums.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    dams = []
    idx = 1
    total_t2 = 0
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        dams.append((t1, t2, t1 - t2))
        total_t2 += t2
        idx += 2
        
    # Sort by d_i = t1 - t2
    dams.sort(key=lambda x: x[2])
    
    # Calculate vulnerability score: max_{k=1..j} (sum_{i=1}^k t1_i + sum_{i=k}^j t2_i)
    # = max_{k=1..j} (total_t2 + sum_{i=1}^k t1_i - sum_{i=1}^{k-1} t2_i)
    
    max_cost = 0
    current_s1 = 0
    current_s2_prefix = 0
    
    for i in range(j):
        t1, t2, d = dams[i]
        current_s1 += t1
        # Cost at position k=i+1 is current_s1 + (total_t2 - current_s2_prefix)
        cost = current_s1 + (total_t2 - current_s2_prefix)
        if cost > max_cost:
            max_cost = cost
        current_s2_prefix += t2
        
    print(max_cost)

if __name__ == '__main__':
    solve()
