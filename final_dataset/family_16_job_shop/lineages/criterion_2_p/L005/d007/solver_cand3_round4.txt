import sys

def solve():
    # The problem asks to minimize the peak exposure.
    # Let S be the sum of all outflows (t2).
    # Let the sequence of accounts be (in1, out1), (in2, out2), ..., (inj, outj).
    # Initial balance = S.
    # Step 1: balance = S + in1. Exposure1 = S + in1. New balance = S + in1 - out1.
    # Step 2: balance = (S + in1 - out1) + in2. Exposure2 = S + in1 - out1 + in2.
    # Step k: Exposure_k = S + sum_{i=1}^k (in_i - out_i) + out_k.
    # Let diff_i = in_i - out_i.
    # Exposure_k = S + sum_{i=1}^{k-1} diff_i + in_k.
    
    # To minimize the maximum of these values, we use a greedy approach.
    # Accounts with in_i < out_i (diff_i < 0) should be processed first.
    # Among these, sort by outflow (t2) descending.
    # Accounts with in_i >= out_i (diff_i >= 0) should be processed last.
    # Among these, sort by inflow (t1) ascending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    accounts = []
    idx = 1
    total_outflow = 0
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        accounts.append((t1, t2))
        total_outflow += t2
        idx += 2
        
    neg = []
    pos = []
    for t1, t2 in accounts:
        if t1 < t2:
            neg.append((t1, t2))
        else:
            pos.append((t1, t2))
            
    # Sort neg: t2 descending
    neg.sort(key=lambda x: x[1], reverse=True)
    # Sort pos: t1 ascending
    pos.sort(key=lambda x: x[0])
    
    sequence = neg + pos
    
    max_exposure = 0
    current_balance = total_outflow
    
    for t1, t2 in sequence:
        current_balance += t1
        if current_balance > max_exposure:
            max_exposure = current_balance
        current_balance -= t2
        
    print(max_exposure)

if __name__ == '__main__':
    solve()
