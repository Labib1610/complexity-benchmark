```python
import sys

def solve():
    # The interference spike at position k is:
    # S = sum(t2_all) + sum(t1_1...k) - sum(t2_1...k-1)
    # Let T1_k = sum(t1_1...k) and T2_k = sum(t2_1...k)
    # S_k = T2_j + T1_k - T2_{k-1}
    # S_k = T2_j + (T1_k - T2_k) + t2_k
    # Since T2_j is constant, we want to minimize the maximum of (T1_k - T2_{k-1})
    # Let d_i = t1_i - t2_i.
    # S_k = T2_j + sum_{i=1}^k t1_i - sum_{i=1}^{k-1} t2_i
    # S_k = T2_j + t1_1 + (t1_2 - t2_1) + (t1_3 - t2_2) + ... + (t1_k - t2_{k-1})
    
    # This is a classic scheduling problem.
    # If we have two adjacent processors i and j:
    # Sequence (i, j):
    # Spike at i: T2_j + t1_i
    # Spike at j: T2_j + t1_i + t1_j - t2_i
    # Sequence (j, i):
    # Spike at j: T2_j + t1_j
    # Spike at i: T2_j + t1_j + t1_i - t2_j
    
    # We want to minimize max(t1_i, t1_i + t1_j - t2_i) vs max(t1_j, t1_j + t1_i - t2_j)
    # This is minimized by sorting:
    # If t1_i < t2_i and t1_j < t2_j, sort by t1 ascending.
    # If t1_i >= t2_i and t1_j >= t2_j, sort by t2 descending.
    # Generally, processors with t1 < t2 come first (sorted by t1),
    # then processors with t1 >= t2 (sorted by t2 descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    procs = []
    idx = 1
    sum_t2 = 0
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        procs.append((t1, t2))
        sum_t2 += t2
        idx += 2
        
    group1 = [p for p in procs if p[0] < p[1]]
    group2 = [p for p in procs if p[0] >= p[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    max_spike = 0
    current_t1_sum = 0
    current_t2_sum = 0
    
    for t1, t2 in sequence:
        current_t1_sum += t1
        spike = sum_t2 + current_t1_sum - current_t2_sum
        if spike > max_spike:
            max_spike = spike
        current_t2_sum += t2
        
    print(max_spike)

if __name__ == '__main__':
    solve()
```
