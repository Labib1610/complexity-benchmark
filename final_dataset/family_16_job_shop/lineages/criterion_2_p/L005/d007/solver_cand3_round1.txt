```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    modules = []
    idx = 1
    total_b = 0
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        modules.append((a, b))
        total_b += b
        idx += 2
    
    # Let S be the current System Load (initially sum of all B).
    # Let P be the current Active Power (initially 0).
    # For a module (A_i, B_i):
    # 1. P becomes P + A_i
    # 2. Stress = P + S
    # 3. S becomes S - B_i
    #
    # We want to minimize max(P_i + S_i) for all i.
    # P_i = sum_{k=1 to i} A_k
    # S_i = (sum_{all} B) - sum_{k=1 to i-1} B_k
    # Stress_i = (sum_{k=1 to i} A_k) + (sum_{all} B) - (sum_{k=1 to i-1} B_k)
    # Stress_i = (sum_{all} B) + A_i + sum_{k=1 to i-1} (A_k - B_k)
    #
    # To minimize the maximum, we should sort by (A_i - B_i).
    # If A_i - B_i < 0, we want these modules first.
    # If A_i - B_i >= 0, we want these modules later.
    # Specifically, sort by (A_i - B_i) ascending.
    
    modules.sort(key=lambda x: x[0] - x[1])
    
    max_stress = 0
    current_p = 0
    current_s = total_b
    
    for a, b in modules:
        current_p += a
        stress = current_p + current_s
        if stress > max_stress:
            max_stress = stress
        current_s -= b
        
    print(max_stress)

if __name__ == '__main__':
    solve()
```
