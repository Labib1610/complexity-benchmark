import sys

def solve():
    # The problem asks to minimize the maximum spike value.
    # Let B = sum(t2_i for all i).
    # For a sequence of vessels (t1_1, t2_1), (t1_2, t2_2), ..., (t1_j, t2_j):
    # Spike value at position k:
    # S_k = B + t2_k + sum_{i=1}^k (t1_i - t2_i)
    # S_k = B + t1_k + sum_{i=1}^{k-1} (t1_i - t2_i)
    # Let d_i = t1_i - t2_i.
    # S_k = B + t1_k + sum_{i=1}^{k-1} d_i
    # We want to minimize max(S_1, S_2, ..., S_j).
    
    # This is a variation of Johnson's rule or scheduling problems.
    # Let's split the vessels into two groups:
    # Group 1: t1_i < t2_i (d_i < 0)
    # Group 2: t1_i >= t2_i (d_i >= 0)
    # Within Group 1, sort by t1_i ascending.
    # Within Group 2, sort by t2_i descending.
    # This ordering minimizes the maximum value of the prefix sums.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    vessels = []
    idx = 1
    b_sum = 0
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        vessels.append((t1, t2))
        b_sum += t2
        idx += 2
        
    group1 = [v for v in vessels if v[0] < v[1]]
    group2 = [v for v in vessels if v[0] >= v[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    max_spike = -float('inf')
    current_prefix_d = 0
    for t1, t2 in sequence:
        spike = b_sum + t1 + current_prefix_d
        if spike > max_spike:
            max_spike = spike
        current_prefix_d += (t1 - t2)
        
    print(max_spike)

if __name__ == '__main__':
    solve()
