```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    artifacts = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        artifacts.append((t1, t2))
        idx += 2
    
    # The instability at step i is:
    # A_i = sum_{k=1}^i t1_k
    # C_i = (sum_{all} t2) - sum_{k=1}^{i-1} t2_k
    # Instability_i = A_i + C_i = sum_{k=1}^i t1_k + sum_{all} t2 - sum_{k=1}^{i-1} t2_k
    # Instability_i = (sum_{all} t2) + t1_i + sum_{k=1}^{i-1} (t1_k - t2_k)
    
    # To minimize the maximum of these values, we use the exchange argument.
    # Consider two adjacent artifacts i and i+1.
    # If we swap them, the instability values for other positions remain the same.
    # Let S = sum_{k=1}^{i-1} (t1_k - t2_k) + sum_{all} t2.
    # Order (i, i+1):
    # Instability_i = S + t1_i
    # Instability_{i+1} = S + (t1_i - t2_i) + t1_{i+1}
    # Max is max(S + t1_i, S + t1_i - t2_i + t1_{i+1})
    # Order (i+1, i):
    # Instability_{i+1} = S + t1_{i+1}
    # Instability_i = S + (t1_{i+1} - t2_{i+1}) + t1_i
    # Max is max(S + t1_{i+1}, S + t1_{i+1} - t2_{i+1} + t1_i)
    
    # We want to sort such that t2_i > t2_{i+1} (descending order of t2).
    # If t2_i = t2_{i+1}, the order doesn't matter.
    
    artifacts.sort(key=lambda x: x[1], reverse=True)
    
    total_c = sum(a[1] for a in artifacts)
    current_a = 0
    current_c = total_c
    max_instability = 0
    
    for t1, t2 in artifacts:
        current_a += t1
        instability = current_a + current_c
        if instability > max_instability:
            max_instability = instability
        current_c -= t2
        
    print(max_instability)

if __name__ == '__main__':
    solve()
```
