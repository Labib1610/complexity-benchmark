```python
import sys

def solve():
    # The problem asks to minimize the maximum of:
    # (sum of p1[0...k]) + (sum of p2[k...j-1])
    # This is equivalent to:
    # (sum of all p1) + (sum of all p2) - (sum of p1[k+1...j-1]) + (sum of p2[0...k-1])
    # Let S1 = sum(p1), S2 = sum(p2).
    # The expression is S1 + S2 - (sum of p1[k+1...j-1]) + (sum of p2[0...k-1])
    # This is a variation of Johnson's Rule.
    # To minimize the maximum bottleneck, we sort vessels:
    # If p1 < p2, put them at the beginning (sorted by p1 ascending).
    # If p1 >= p2, put them at the end (sorted by p2 descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        vessels.append((p1, p2))
        idx += 2
        
    group1 = [v for v in vessels if v[0] < v[1]]
    group2 = [v for v in vessels if v[0] >= v[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate the completion time for this order
    # Time at which Federal officer finishes vessel i:
    # F_i = max(F_{i-1}, L_i) + p2_i
    # where L_i is the time Local officer finishes vessel i.
    # L_i = L_{i-1} + p1_i
    
    current_local_time = 0
    current_federal_time = 0
    
    for p1, p2 in order:
        current_local_time += p1
        current_federal_time = max(current_federal_time, current_local_time) + p2
        
    print(current_federal_time)

if __name__ == '__main__':
    solve()
```
