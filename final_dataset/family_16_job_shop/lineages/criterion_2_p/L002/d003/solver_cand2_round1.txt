```python
import sys

def solve():
    # The problem is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # We want to minimize the makespan.
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: p1 < p2 (Local Desk time < Federal Desk time)
    #    Set 2: p1 >= p2 (Local Desk time >= Federal Desk time)
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    # Sort set1 by p1 ascending
    set1.sort(key=lambda x: x[0])
    # Sort set2 by p2 descending
    set2.sort(key=lambda x: x[1], reverse=True)
    
    schedule = set1 + set2
    
    local_finish_time = 0
    federal_finish_time = 0
    
    for p1, p2 in schedule:
        local_finish_time += p1
        # Federal desk starts when local desk finishes OR when federal desk is free
        federal_finish_time = max(local_finish_time, federal_finish_time) + p2
        
    print(federal_finish_time)

if __name__ == '__main__':
    solve()
```
