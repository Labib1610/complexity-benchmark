import sys

def solve():
    # The problem asks to minimize the completion time of the Federal officer.
    # Let the vessels be (L_i, F_i).
    # If we have a sequence of vessels, the Local officer finishes vessel i at time:
    # T_L(i) = sum_{k=1}^i L_k
    # The Federal officer finishes vessel i at time:
    # T_F(i) = max(T_F(i-1), T_L(i)) + F_i
    # This is a classic scheduling problem known as Johnson's Rule for 2 machines.
    # To minimize the total completion time (makespan), we sort the jobs:
    # 1. Jobs with L_i < F_i are sorted by increasing L_i.
    # 2. Jobs with L_i >= F_i are sorted by decreasing F_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        l = int(input_data[idx])
        f = int(input_data[idx+1])
        vessels.append((l, f))
        idx += 2
        
    group1 = []
    group2 = []
    
    for l, f in vessels:
        if l < f:
            group1.append((l, f))
        else:
            group2.append((l, f))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_l_time = 0
    current_f_time = 0
    
    for l, f in order:
        current_l_time += l
        current_f_time = max(current_f_time, current_l_time) + f
        
    print(current_f_time)

if __name__ == '__main__':
    solve()
