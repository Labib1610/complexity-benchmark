```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        vessels.append((p1, p2))
        idx += 2
    
    # This is a variation of Johnson's Rule for two-machine scheduling.
    # To minimize the makespan, we sort vessels based on their p1 and p2 values.
    # Group 1: p1 < p2, sort by p1 ascending.
    # Group 2: p1 >= p2, sort by p2 descending.
    
    group1 = []
    group2 = []
    for p1, p2 in vessels:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    local_finish_time = 0
    federal_finish_time = 0
    
    for p1, p2 in order:
        local_finish_time += p1
        # Federal desk starts when local desk finishes OR when federal desk is free
        federal_finish_time = max(local_finish_time, federal_finish_time) + p2
        
    print(federal_finish_time)

if __name__ == '__main__':
    solve()
```
