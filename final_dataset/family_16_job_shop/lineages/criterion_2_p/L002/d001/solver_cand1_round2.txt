import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    shipments = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        shipments.append((p1, p2))
        idx += 2
    
    # Let the sequence of shipments be s_1, s_2, ..., s_j.
    # Let p1_i and p2_i be the entry and exit times for shipment s_i.
    # Entry checkpoint finish times:
    # E_1 = p1_1
    # E_2 = E_1 + p1_2
    # ...
    # E_i = sum_{k=1}^i p1_k
    #
    # Exit checkpoint finish times:
    # F_1 = E_1 + p2_1
    # F_2 = max(F_1, E_2) + p2_2
    # F_i = max(F_{i-1}, E_i) + p2_i
    #
    # This is a classic scheduling problem. To minimize the completion time
    # of the last job, we use Johnson's Rule logic.
    # Sort shipments into two groups:
    # Group 1: p1_i < p2_i, sorted by p1_i ascending.
    # Group 2: p1_i >= p2_i, sorted by p2_i descending.
    
    group1 = []
    group2 = []
    for p1, p2 in shipments:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    current_entry_time = 0
    current_exit_time = 0
    
    for p1, p2 in sequence:
        current_entry_time += p1
        current_exit_time = max(current_exit_time, current_entry_time) + p2
        
    print(current_exit_time)

if __name__ == '__main__':
    solve()
