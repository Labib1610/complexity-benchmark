```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    containers = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx]); p2 = int(data[idx+1])
        idx += 2
        containers.append((p1, p2))
    
    # Sort by loading time descending (Johnson's rule for two-machine flow shop)
    containers.sort(key=lambda x: -x[1])
    
    total_inspection = 0
    finish_loading = 0
    for p1, p2 in containers:
        total_inspection += p1
        # Loading starts after inspection of this container finishes
        start_loading = total_inspection
        if start_loading > finish_loading:
            finish_loading = start_loading + p2
        else:
            finish_loading += p2
    
    print(finish_loading)

solve()
```
