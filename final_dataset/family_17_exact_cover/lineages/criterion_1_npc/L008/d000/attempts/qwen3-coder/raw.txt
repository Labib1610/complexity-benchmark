from collections import defaultdict
import sys

def solve():
    q, t = map(int, sys.stdin.readline().split())
    n = 3 * q
    cells = []
    for _ in range(t):
        a, b, c = map(int, sys.stdin.readline().split())
        cells.append((a, b, c))
    
    # Build a graph where each node is an operator and edges represent possible pairs in a cell
    # We'll use a backtracking approach with optimizations
    
    # First, let's see if it's even possible by checking degrees
    # Each operator must appear in exactly one cell
    degree = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]
    
    for i, (a, b, c) in enumerate(cells):
        degree[a] += 1
        degree[b] += 1
        degree[c] += 1
        adj[a].append(i)
        adj[b].append(i)
        adj[c].append(i)
    
    # Check if every operator appears at least once
    for i in range(1, n + 1):
        if degree[i] == 0:
            print("NO")
            return
    
    # Try to find a matching that covers all nodes exactly once
    # This is essentially a exact cover problem - we can model it as finding a perfect matching
    # in a hypergraph where each hyperedge is a cell
    
    # Use backtracking with pruning
    selected = [False] * t
    result = []
    used = [False] * (n + 1)
    
    def backtrack(cell_index):
        if len(result) == q:
            return True
            
        if cell_index >= t:
            return False
            
        # Pruning: if remaining cells cannot cover remaining operators, return False
        remaining_operators = 0
        for i in range(1, n + 1):
            if not used[i]:
                remaining_operators += 1
                
        if remaining_operators > (q - len(result)) * 3:
            return False
            
        # Try including current cell
        a, b, c = cells[cell_index]
        if not used[a] and not used[b] and not used[c]:
            selected[cell_index] = True
            used[a] = used[b] = used[c] = True
            result.append(cell_index)
            
            if backtrack(cell_index + 1):
                return True
                
            # Backtrack
            selected[cell_index] = False
            used[a] = used[b] = used[c] = False
            result.pop()
        
        # Try not including current cell (only if it's still possible)
        # Pruning: if we skip too many, we won't have enough cells
        if q - len(result) <= t - cell_index - 1:
            if backtrack(cell_index + 1):
                return True
                
        return False
    
    # Sort cells by how "constrained" they are (try more constrained first)
    # Cells with operators of higher degree should be tried later
    # Let's sort by sum of degrees of operators in the cell (ascending)
    cell_degrees = []
    for i, (a, b, c) in enumerate(cells):
        cell_degrees.append((degree[a] + degree[b] + degree[c], i))
    
    cell_indices = sorted(range(t), key=lambda x: cell_degrees[x][0])
    
    # More sophisticated approach using recursive backtracking with better ordering
    def solve_recursive(pos, count):
        if count == q:
            return True
        if pos == t:
            return False
            
        # Pruning
        remaining_ops = sum(1 for i in range(1, n+1) if not used[i])
        if remaining_ops > (q - count) * 3:
            return False
        if (q - count) > (t - pos):
            return False
            
        idx = cell_indices[pos]
        a, b, c = cells[idx]
        
        # Try taking this cell
        if not used[a] and not used[b] and not used[c]:
            used[a] = used[b] = used[c] = True
            result.append(idx)
            if solve_recursive(pos + 1, count + 1):
                return True
            used[a] = used[b] = used[c] = False
            result.pop()
            
        # Try not taking this cell
        if solve_recursive(pos + 1, count):
            return True
            
        return False
    
    if solve_recursive(0, 0):
        print("YES")
        for idx in result:
            a, b, c = cells[idx]
            print(a, b, c)
    else:
        print("NO")

solve()
