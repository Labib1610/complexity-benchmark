We can model this as an exact cover problem: we have `3q` items (modules) and `t` possible triples (configurations). We need to pick `q` triples that cover all `3q` items exactly once.

Since `q ≤ 200` and `t ≤ 20000`, a naive backtracking might be too slow, but we can use **Algorithm X** (Dancing Links) or a simpler recursive backtracking with pruning.

Here’s a clean Python solution using backtracking with bitmasks for efficiency:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    q = int(input_data[0])
    t = int(input_data[1])
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx+1]) - 1
        c = int(input_data[idx+2]) - 1
        idx += 3
        triples.append((a, b, c))
    
    n = 3 * q
    # For each module, list of triple indices that contain it
    cover = [[] for _ in range(n)]
    for i, (a, b, c) in enumerate(triples):
        cover[a].append(i)
        cover[b].append(i)
        cover[c].append(i)
    
    used = [False] * t
    chosen = []
    
    # Recursive backtracking
    def backtrack(remaining):
        if remaining == 0:
            return True
        # Find module with fewest available triples
        best = -1
        best_count = 10**9
        for mod in range(n):
            if not any(mod in triples[i] for i in chosen):
                cnt = 0
                for i in cover[mod]:
                    if not used[i]:
                        cnt += 1
                if cnt < best_count:
                    best_count = cnt
                    best = mod
                    if cnt == 0:
                        break
        if best_count == 0:
            return False
        
        # Try each candidate triple for this module
        for i in cover[best]:
            if used[i]:
                continue
            a, b, c = triples[i]
            # Check if any of a,b,c already used
            if any(any(mod in triples[j] for j in chosen) for mod in (a,b,c)):
                continue
            used[i] = True
            chosen.append(i)
            if backtrack(remaining - 1):
                return True
            chosen.pop()
            used[i] = False
        return False
    
    if backtrack(q):
        print("YES")
        for i in chosen:
            a, b, c = triples[i]
            print(a+1, b+1, c+1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
