We can model this as an exact cover problem: we have 3q artifacts and a set of triples (cases). We need to pick q triples that cover all artifacts exactly once.

Since q ≤ 200, the total number of artifacts is at most 600. The number of cases t can be up to 20000. A backtracking with pruning can work if we order artifacts by how few cases contain them.

Here’s the Python solution:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    q = int(next(it))
    t = int(next(it))
    n = 3 * q

    cases = []
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        c = int(next(it)) - 1
        cases.append((a, b, c))

    # For each artifact, list of case indices that contain it
    artifact_to_cases = [[] for _ in range(n)]
    for idx, (a, b, c) in enumerate(cases):
        artifact_to_cases[a].append(idx)
        artifact_to_cases[b].append(idx)
        artifact_to_cases[c].append(idx)

    # Order artifacts by how many cases contain them (fewest first)
    order = sorted(range(n), key=lambda x: len(artifact_to_cases[x]))

    used_case = [False] * t
    chosen = []
    covered = [False] * n

    def backtrack(pos):
        if pos == n:
            return len(chosen) == q
        artifact = order[pos]
        if covered[artifact]:
            return backtrack(pos + 1)
        # Try cases containing this artifact
        for case_idx in artifact_to_cases[artifact]:
            if used_case[case_idx]:
                continue
            a, b, c = cases[case_idx]
            if covered[a] or covered[b] or covered[c]:
                continue
            # Use this case
            used_case[case_idx] = True
            covered[a] = covered[b] = covered[c] = True
            chosen.append((a, b, c))
            if backtrack(pos + 1):
                return True
            # Backtrack
            chosen.pop()
            covered[a] = covered[b] = covered[c] = False
            used_case[case_idx] = False
        return False

    if backtrack(0):
        print("YES")
        for a, b, c in chosen:
            print(a + 1, b + 1, c + 1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
