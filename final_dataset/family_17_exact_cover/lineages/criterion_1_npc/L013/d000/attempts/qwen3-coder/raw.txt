from collections import defaultdict, deque
import sys

def find_matching(q, t, triples):
    # Create a graph for matching
    # Each component is a node, each triple is a hyperedge
    # We want to find a perfect matching that covers all 3q components
    
    if t == 0:
        if q == 0:
            print("YES")
            return
        else:
            print("NO")
            return
    
    # Build adjacency list for components to triples
    comp_to_triples = defaultdict(list)
    for i, (a, b, c) in enumerate(triples):
        comp_to_triples[a].append(i)
        comp_to_triples[b].append(i)
        comp_to_triples[c].append(i)
    
    # Try to find a maximum matching using backtracking with pruning
    used_triples = [False] * t
    selected = []
    component_used = [False] * (3*q + 1)
    
    def backtrack(idx):
        if idx == q:
            return True
            
        # Find the first unused component
        first_unused = -1
        for i in range(1, 3*q + 1):
            if not component_used[i]:
                first_unused = i
                break
        
        if first_unused == -1:
            return False
            
        # Try all triples containing this component
        for triple_idx in comp_to_triples[first_unused]:
            if used_triples[triple_idx]:
                continue
                
            a, b, c = triples[triple_idx]
            
            # Check if all components in this triple are available
            if component_used[a] or component_used[b] or component_used[c]:
                continue
            
            # Use this triple
            used_triples[triple_idx] = True
            component_used[a] = True
            component_used[b] = True
            component_used[c] = True
            selected.append((a, b, c))
            
            if backtrack(idx + 1):
                return True
                
            # Backtrack
            used_triples[triple_idx] = False
            component_used[a] = False
            component_used[b] = False
            component_used[c] = False
            selected.pop()
        
        return False
    
    # Also try a greedy approach first for better performance
    def greedy():
        temp_used = [False] * (3*q + 1)
        temp_selected = []
        temp_triple_used = [False] * t
        
        # Sort triples by some heuristic - prioritize those with least options
        triple_degrees = []
        for i, (a, b, c) in enumerate(triples):
            deg = len(comp_to_triples[a]) + len(comp_to_triples[b]) + len(comp_to_triples[c])
            triple_degrees.append((deg, i))
        
        triple_degrees.sort()
        
        for _, triple_idx in triple_degrees:
            if temp_triple_used[triple_idx]:
                continue
                
            a, b, c = triples[triple_idx]
            if temp_used[a] or temp_used[b] or temp_used[c]:
                continue
            
            temp_triple_used[triple_idx] = True
            temp_used[a] = True
            temp_used[b] = True
            temp_used[c] = True
            temp_selected.append((a, b, c))
        
        count = sum(temp_used[1:])
        if count == 3*q:
            return temp_selected
        return None
    
    # Try greedy first
    greedy_result = greedy()
    if greedy_result and len(greedy_result) == q:
        print("YES")
        for a, b, c in greedy_result:
            print(a, b, c)
        return
    
    # If greedy doesn't work, try backtracking
    if backtrack(0):
        print("YES")
        for a, b, c in selected:
            print(a, b, c)
    else:
        print("NO")

# Read input
line = input().split()
q, t = int(line[0]), int(line[1])

triples = []
for _ in range(t):
    a, b, c = map(int, input().split())
    triples.append((a, b, c))

find_matching(q, t, triples)
