 We need to solve a bipartite matching problem: mentors (1..q) on the left, apprentices (q+1..2q) on the right.  
We need a perfect matching covering all vertices.  
Given constraints (q up to 100k, edges up to 400k), we can use the Hopcroft–Karp algorithm for maximum bipartite matching, which runs in O(E√V).

Here’s the implementation:

```python
import sys
from collections import deque

def hopcroft_karp(n, m, adj):
    # n: number of left vertices (mentors), m: number of right vertices (apprentices)
    # adj[u] = list of v (0-indexed right side)
    INF = 10**9
    pairU = [-1] * n
    pairV = [-1] * m
    dist = [0] * n

    def bfs():
        q = deque()
        for u in range(n):
            if pairU[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        found = False
        while q:
            u = q.popleft()
            for v in adj[u]:
                pu = pairV[v]
                if pu != -1 and dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    q.append(pu)
                elif pu == -1:
                    found = True
        return found

    def dfs(u):
        for v in adj[u]:
            pu = pairV[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pairU[u] = v
                pairV[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(n):
            if pairU[u] == -1 and dfs(u):
                matching += 1
    return matching, pairU

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    adj = [[] for _ in range(q)]
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - q - 1  # shift to 0-indexed right side
        adj[a].append(b)
    
    matching, pairU = hopcroft_karp(q, q, adj)
    if matching != q:
        sys.stdout.write("NO\n")
    else:
        out = ["YES"]
        for u in range(q):
            out.append(f"{u+1} {pairU[u]+q+1}")
        sys.stdout.write("\n".join(out) + "\n")

if __name__ == "__main__":
    main()
```
