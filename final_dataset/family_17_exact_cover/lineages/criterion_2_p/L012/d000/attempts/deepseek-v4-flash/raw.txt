We need to find a perfect matching between controllers (1..q) and platforms (q+1..2q) using the given edges.  
This is a bipartite matching problem, and since q can be up to 100,000 and t up to 400,000, we need an efficient algorithm — Hopcroft–Karp is suitable.

We’ll implement Hopcroft–Karp to find maximum matching, and if the size equals q, we output YES and the matching edges.

```python
import sys
from collections import deque

input = sys.stdin.readline

def hopcroft_karp(n, m, adj):
    # adj: list of lists, 0-indexed for left side (controllers)
    # right side: platforms, 0-indexed
    INF = 10**9
    pair_u = [-1] * n
    pair_v = [-1] * m
    dist = [0] * n

    def bfs():
        q = deque()
        for u in range(n):
            if pair_u[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        found = False
        while q:
            u = q.popleft()
            for v in adj[u]:
                pu = pair_v[v]
                if pu != -1 and dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    q.append(pu)
                elif pu == -1:
                    found = True
        return found

    def dfs(u):
        for v in adj[u]:
            pu = pair_v[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(n):
            if pair_u[u] == -1 and dfs(u):
                matching += 1
    return matching, pair_u

def main():
    q, t = map(int, input().split())
    adj = [[] for _ in range(q)]
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a-1].append(b-q-1)  # platform index 0..q-1

    matching, pair_u = hopcroft_karp(q, q, adj)
    if matching != q:
        print("NO")
    else:
        print("YES")
        for u in range(q):
            v = pair_u[u]
            print(u+1, v+q+1)

if __name__ == "__main__":
    main()
```
