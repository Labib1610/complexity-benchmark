import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    edges = []
    idx = 2
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx+1])
        idx += 2
        edges.append((a, b - q - 1))
    
    size = 1
    while size < q:
        size <<= 1
    
    # Build segment tree over platforms [0, q-1]
    # Each node stores list of (controller, edge_index)
    tree = [[] for _ in range(2 * size)]
    
    for i, (a, b) in enumerate(edges):
        if 0 <= b < q:
            pos = b + size
            tree[pos].append((a, i))
            pos >>= 1
            while pos:
                tree[pos].append((a, i))
                pos >>= 1
    
    # Hopcroft-Karp on the segment tree
    pair_u = [-1] * (q + 1)  # controller -> edge index
    pair_v = [-1] * (2 * size)  # segment tree node -> edge index
    dist = [0] * (2 * size)
    used_edge = [False] * t
    INF = 10 ** 9
    
    from collections import deque
    
    def bfs():
        queue = deque()
        for v in range(size, size + q):
            if pair_v[v] == -1:
                dist[v] = 0
                queue.append(v)
            else:
                dist[v] = INF
        found = False
        while queue:
            v = queue.popleft()
            for a, eid in tree[v]:
                if used_edge[eid]:
                    continue
                u = pair_u[a]
                if u == -1:
                    found = True
                elif dist[u] == INF:
                    dist[u] = dist[v] + 1
                    queue.append(u)
        return found
    
    def dfs(v):
        for a, eid in tree[v]:
            if used_edge[eid]:
                continue
            u = pair_u[a]
            if u == -1 or (dist[u] == dist[v] + 1 and dfs(u)):
                pair_v[v] = eid
                pair_u[a] = eid
                return True
        dist[v] = INF
        return False
    
    matching = 0
    while bfs():
        for v in range(size, size + q):
            if pair_v[v] == -1:
                if dfs(v):
                    matching += 1
    
    if matching == q:
        out = ["YES"]
        for a in range(1, q + 1):
            eid = pair_u[a]
            ca, cb = edges[eid]
            out.append(f"{ca} {cb + q + 1}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

solve()
