from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Controllers: 1 to q
    # Platforms: q+1 to 2*q
    adj = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a].append(b)
        adj[b].append(a)
        edges.append((a, b))
    
    # Try to find a perfect matching using Hopcroft-Karp algorithm
    # But we'll use a simpler approach with DFS-based augmenting paths
    
    # pair_u[u] stores the matched platform for controller u (0 if unmatched)
    # pair_v[v] stores the matched controller for platform v (0 if unmatched)
    pair_u = [0] * (q + 1)  # 1-indexed for controllers
    pair_v = [0] * (q + 1)  # 1-indexed for platforms (v-q gives actual platform id)
    
    def dfs(u, visited):
        for v in adj[u]:
            if visited[v - q]:  # Convert platform id to index
                continue
            visited[v - q] = True
            
            # If v is unmatched or we can find an augmenting path from v's match
            if pair_v[v - q] == 0 or dfs(pair_v[v - q], visited):
                pair_u[u] = v
                pair_v[v - q] = u
                return True
        return False
    
    # Keep finding augmenting paths until no more can be found
    matching = 0
    while True:
        visited = [False] * (q + 1)
        found_augmenting_path = False
        
        # Try to find an augmenting path starting from each unmatched controller
        for u in range(1, q + 1):
            if pair_u[u] == 0:  # Unmatched controller
                if dfs(u, visited):
                    found_augmenting_path = True
                    
        if not found_augmenting_path:
            break
            
        matching += 1
    
    # Check if we have a perfect matching
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, pair_u[u])
    else:
        print("NO")

solve()
