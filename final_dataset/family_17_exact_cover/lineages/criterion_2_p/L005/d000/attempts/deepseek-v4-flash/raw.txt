We need to find a perfect matching in a bipartite graph where the left side is precursors 1…q and the right side is q+1…2q, using only given edgesWORDS.

The problem is a standard maximum bipartite matching for a perfect matching check. We can solve using Hopcroft–Karp algorithm for efficiency.

We’ll output the matching if one exists with size q.  

Here’s the implementation:  

```python
import sys
from collections import deque

input = sys.stdin.readline

def hopcroft_karp(n, m, adj):
    # adj is list of lists, vertices on left are 0..n-1, right are 0..m-1
    INF = 10**9
    pairU = [-1] * n
    pairV = [-1] * m
    dist = [0] * n

    def bfs():
        q = deque()
        for u in range(n):
            if pairU[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        found = False
        while q:
            u = q.popleft()
            for v in adj[u]:
                if pairV[v] == -1:
                    found = True
                else:
                    pu = pairV[v]
                    if dist[pu] == INF:
                        dist[pu] = dist[u] + 1
                        q.append(pu)
        return found

    def dfs(u):
        for v in adj[u]:
            if pairV[v] == -1 or (dist[pairV[v]] == dist[u] + 1 and dfs(pairV[v])):
                pairU[u] = v
                pairV[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(n):
            if pairU[u] == -1 and dfs(u):
                matching += 1
    return matching, pairU, pairV

def main():
    q, t = map(int, input().split())
    adj = [[] for _ in range(q)]
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a - 1].append(b - q - 1)  # right side index 0..q-1

    matching, pairU, pairV = hopcroft_karp(q, q, adj)

    if matching == q:
        print("YES")
        for u in range(q):
            v = pairU[u]
            print(f"{u+1} {v+q+1}")
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
