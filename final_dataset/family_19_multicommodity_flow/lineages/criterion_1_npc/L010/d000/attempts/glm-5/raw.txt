import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    streams = []
    for _ in range(k):
        s = int(next(it))
        t = int(next(it))
        d = int(next(it))
        streams.append((s, t, d))
    
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u, v, cap))
    
    # Build flow network
    # Nodes: 0 is super source, 1..n are switches, n+1..n+k are stream sources, n+k+1..n+2k are stream sinks
    # Total nodes: 1 + n + 2k + 1 = n + 2k + 2
    # Let's renumber: super source = 0, switches = 1..n, stream sources = n+1..n+k, stream sinks = n+k+1..n+2k, super sink = n+2k+1
    
    super_source = 0
    super_sink = n + 2*k + 1
    total_nodes = n + 2*k + 2
    
    adj = [[] for _ in range(total_nodes)]
    edge_list = []
    
    def add_edge(u, v, cap, edge_id):
        # forward edge
        edge_list.append([u, v, cap, 0, edge_id])
        adj[u].append(len(edge_list) - 1)
        # backward edge
        edge_list.append([v, u, 0, 0, -1])
        adj[v].append(len(edge_list) - 1)
    
    # Super source to stream sources
    total_demand = 0
    for i, (s, t, d) in enumerate(streams):
        stream_source = n + 1 + i
        stream_sink = n + 1 + k + i
        add_edge(super_source, stream_source, d, -1)
        add_edge(stream_source, s, d, -1)
        add_edge(stream_sink, super_sink, d, -1)
        total_demand += d
    
    # Original edges with capacity cap
    for idx, (u, v, cap) in enumerate(edges):
        add_edge(u, v, cap, idx)
    
    # Dinic's algorithm
    level = [0] * total_nodes
    ptr = [0] * total_nodes
    
    def bfs():
        for i in range(total_nodes):
            level[i] = -1
        level[super_source] = 0
        q = deque([super_source])
        while q:
            u = q.popleft()
            for ei in adj[u]:
                v, cap, flow = edge_list[ei][1], edge_list[ei][2], edge_list[ei][3]
                if cap - flow > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[super_sink] != -1
    
    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == super_sink:
            return pushed
        for i in range(ptr[u], len(adj[u])):
            ei = adj[u][i]
            v, cap, flow, edge_id = edge_list[ei][1], edge_list[ei][2], edge_list[ei][3], edge_list[ei][4]
            if level[v] != level[u] + 1 or cap - flow == 0:
                ptr[u] += 1
                continue
            tr = dfs(v, min(pushed, cap - flow))
            if tr == 0:
                ptr[u] += 1
                continue
            edge_list[ei][3] += tr
            edge_list[ei ^ 1][3] -= tr
            return tr
        return 0
    
    max_flow = 0
    while bfs():
        ptr = [0] * total_nodes
        while True:
            pushed = dfs(super_source, float('inf'))
            if pushed == 0:
                break
            max_flow += pushed
    
    if max_flow != total_demand:
        print("NO")
        return
    
    print("YES")
    
    # Extract flow for each stream on each original edge
    # For stream i, we need to find flow on edges from stream_source to switches and between switches
    # But the problem asks for flow on each of the m original edges for each stream
    
    # We need to decompose the flow
    # The flow on original edge (u, v) with edge_id idx is stored in edge_list
    # We need to find how much of that flow belongs to each stream
    
    # Build residual graph for flow decomposition
    # For each stream, we trace paths from its source to its sink
    
    # First, collect all edges with their current flow
    # We need to track which stream's flow uses which edge
    
    # Let's use a different approach: for each stream, compute the flow on each edge
    # by finding paths in the residual network specific to that stream
    
    # Actually, let's reconstruct: the flow on edge (u,v) with edge_id idx is edge_list[ei][3]
    # We need to decompose this into k components
    
    # Build adjacency for flow decomposition
    # For each original edge (u,v), we have flow f on it
    # We need to find k paths for each stream
    
    # Let's use a greedy approach: for each stream, find paths and subtract flow
    
    # Build graph with only edges that have positive flow
    flow_edges = []
    for ei in range(len(edge_list)):
        if ei % 2 == 0:  # forward edge
            u, v, cap, flow, edge_id = edge_list[ei]
            if flow > 0 and edge_id >= 0:  # original edge
                flow_edges.append((u, v, flow, edge_id))
    
    # For each stream, we need to route d_i units from s_i to t_i
    # We'll use the flow on edges and decompose
    
    # Build adjacency list for flow decomposition
    flow_adj = [[] for _ in range(n + 1)]
    edge_flow = {}  # (u, v) -> list of (flow, edge_id)
    for u, v, f, eid in flow_edges:
        flow_adj[u].append((v, f, eid))
        if (u, v) not in edge_flow:
            edge_flow[(u, v)] = []
        edge_flow[(u, v)].append((f, eid))
    
    # Result: for each stream, flow on each original edge
    result = [[0] * m for _ in range(k)]
    
    # For each stream, find paths using DFS
    for stream_idx in range(k):
        s, t, d = streams[stream_idx]
        remaining = d
        
        # Build a working copy of flow_adj
        working_adj = {}
        for u in range(1, n + 1):
            working_adj[u] = []
            for v, f, eid in flow_adj[u]:
                working_adj[u].append([v, f, eid])  # mutable
        
        while remaining > 0:
            # Find a path from s to t
            parent = {}
            stack = [(s, None, None, float('inf'))]  # (node, prev_node, edge_id, min_flow)
            found = False
            path = []
            
            # DFS to find path
            dfs_stack = [(s, None, None, float('inf'))]
            visited = set()
            
            while dfs_stack:
                u, prev_u, prev_eid, min_f = dfs_stack.pop()
                if u in visited:
                    continue
                visited.add(u)
                parent[u] = (prev_u, prev_eid, min_f)
                
                if u == t:
                    found = True
                    break
                
                for v, f, eid in working_adj[u]:
                    if v not in visited and f > 0:
                        new_min = min(min_f, f)
                        dfs_stack.append((v, u, eid, new_min))
            
            if not found:
                break
            
            # Reconstruct path and find min flow
            path_edges = []
            curr = t
            path_flow = float('inf')
            while curr != s:
                prev_u, prev_eid, _ = parent[curr]
                if prev_u is None:
                    break
                # Find the edge in working_adj
                for i, (v, f, eid) in enumerate(working_adj[prev_u]):
                    if v == curr and f > 0:
                        path_edges.append((prev_u, curr, eid, i))
                        path_flow = min(path_flow, f)
                        break
                curr = prev_u
            
            if path_flow == float('inf') or path_flow == 0:
                break
            
            path_flow = min(path_flow, remaining)
            
            # Update flow and result
            for u, v, eid, idx in path_edges:
                working_adj[u][idx][1] -= path_flow
                result[stream_idx][eid] += path_flow
            
            remaining -= path_flow
    
    for row in result:
        print(' '.join(map(str, row)))

if __name__ == "__main__":
    solve()
