from collections import defaultdict, deque
import sys

def solve():
    n, m, k = map(int, input().split())
    
    compounds = []
    for i in range(k):
        s, t, d = map(int, input().split())
        compounds.append((s, t, d))
    
    edges = []
    edge_index = {}
    adj = [[] for _ in range(n + 1)]
    
    for i in range(m):
        u, v, cap = map(int, input().split())
        edges.append((u, v, cap))
        edge_index[(u, v)] = i
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # For each compound, find flow paths using max flow
    # We'll use a simplified approach since we need to route exact demands
    
    # Build a graph for each compound
    def bfs_flow(source, sink, demand, used_flow):
        # BFS to find a path and push flow
        parent = [-1] * (n + 1)
        parent_edge = [-1] * (n + 1)
        visited = [False] * (n + 1)
        
        queue = deque([source])
        visited[source] = True
        
        while queue:
            node = queue.popleft()
            
            if node == sink:
                # Reconstruct path and find bottleneck
                path_edges = []
                curr = sink
                min_cap = demand
                
                while curr != source:
                    eid = parent_edge[curr]
                    u, v, _ = edges[eid]
                    cap_left = edges[eid][2] - used_flow[eid]
                    min_cap = min(min_cap, cap_left)
                    path_edges.append(eid)
                    curr = u if (u, v) == (u, curr) else v
                    curr = u
                
                if min_cap > 0:
                    for eid in path_edges:
                        used_flow[eid] += min_cap
                    return min_cap
                else:
                    return 0
            
            for neighbor, eid in adj[node]:
                if not visited[neighbor]:
                    u, v, cap = edges[eid]
                    # Check direction
                    if u == node:  # Forward edge
                        if used_flow[eid] < cap:
                            visited[neighbor] = True
                            parent[neighbor] = node
                            parent_edge[neighbor] = eid
                            queue.append(neighbor)
                    else:  # Backward edge - but we don't use backward edges in this simple approach
                        pass
        
        return 0
    
    # Try to route each compound
    total_flow = [0] * m
    compound_flows = []
    
    for idx, (source, sink, demand) in enumerate(compounds):
        # For each compound, we need to route 'demand' units from source to sink
        # We'll use a simple path finding approach
        
        # Build residual graph for this compound
        local_flow = [0] * m
        remaining_demand = demand
        
        # Use DFS to find paths
        def dfs(node, target, flow, visited_edges, path):
            if node == target:
                return flow
            
            for neighbor, eid in adj[node]:
                if eid in visited_edges:
                    continue
                    
                u, v, cap = edges[eid]
                if u == node:  # Forward direction
                    available = cap - total_flow[eid] - local_flow[eid]
                    if available > 0:
                        bottleneck = min(flow, available)
                        visited_edges.add(eid)
                        path.append(eid)
                        res = dfs(v, target, bottleneck, visited_edges, path)
                        if res > 0:
                            return res
                        path.pop()
                        visited_edges.remove(eid)
                # Not handling reverse edges as this is a simpler approach
                        
            return 0
        
        compound_flow = [0] * m
        while remaining_demand > 0:
            visited_edges = set()
            path = []
            sent = dfs(source, sink, remaining_demand, visited_edges, path)
            if sent == 0:
                print("NO")
                return
            
            for eid in path:
                compound_flow[eid] += sent
                local_flow[eid] += sent
                
            remaining_demand -= sent
        
        compound_flows.append(compound_flow)
        for i in range(m):
            total_flow[i] += compound_flow[i]
    
    # Check if any edge exceeds capacity
    for i in range(m):
        if total_flow[i] > edges[i][2]:
            print("NO")
            return
    
    print("YES")
    for flow in compound_flows:
        print(' '.join(map(str, flow)))

def main():
    solve()

if __name__ == "__main__":
    main()
