We can model this as a flow problem with lower bounds 0 and upper bounds given by the link limits.  
Since \( k = 1 \), we just need to find a feasible flow of value \( d \) from \( s \) to \( t \).

We can solve it using max flow with lower bounds, or more simply, since lower bounds are all 0, we can just run a standard max flow from \( s \) to \( t \) and check if the max flow is at least \( d \).  
If yes, we can output the flow on each edge.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

class MaxFlow:
    def __init__(self, n):
        self.n = n
        self.adj = [[] for _ in range(n)]
    
    def add_edge(self, u, v, cap):
        self.adj[u].append([v, cap, len(self.adj[v])])
        self.adj[v].append([u, 0, len(self.adj[u]) - 1])
    
    def bfs(self, s, t, level):
        for i in range(self.n):
            level[i] = -1
        q = deque([s])
        level[s] = 0
        while q:
            u = q.popleft()
            for v, cap, _ in self.adj[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[t] >= 0
    
    def dfs(self, u, t, f, level, it):
        if u == t:
            return f
        while it[u] < len(self.adj[u]):
            v, cap, rev = self.adj[u][it[u]]
            if cap > 0 and level[v] == level[u] + 1:
                ret = self.dfs(v, t, min(f, cap), level, it)
                if ret > 0:
                    self.adj[u][it[u]][1] -= ret
                    self.adj[v][rev][1] += ret
                    return ret
            it[u] += 1
        return 0
    
    def max_flow(self, s, t):
        flow = 0
        level = [-1] * self.n
        INF = 10**18
        while self.bfs(s, t, level):
            it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF, level, it)
                if f == 0:
                    break
                flow += f
        return flow

def main():
    n, m, k = map(int, input().split())
    s, t, d = map(int, input().split())
    s -= 1
    t -= 1
    
    mf = MaxFlow(n)
    edges = []
    for _ in range(m):
        u, v, L = map(int, input().split())
        u -= 1
        v -= 1
        mf.add_edge(u, v, L)
        edges.append((u, v, L))
    
    flow = mf.max_flow(s, t)
    if flow < d:
        print("NO")
        return
    
    # Extract flow on each edge
    result = []
    for u, v, L in edges:
        # find the forward edge in adjacency
        for i, (to, cap, rev) in enumerate(mf.adj[u]):
            if to == v and mf.adj[v][rev][1] > 0:
                # flow = original cap - current cap
                result.append(L - cap)
                break
        else:
            result.append(0)
    
    print("YES")
    print(" ".join(map(str, result)))

if __name__ == "__main__":
    main()
```
