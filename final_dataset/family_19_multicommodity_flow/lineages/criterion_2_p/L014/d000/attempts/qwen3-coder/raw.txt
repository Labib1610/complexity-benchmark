from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    index = 0
    n = int(data[index])
    index += 1
    m = int(data[index])
    index += 1
    _ = int(data[index])
    index += 1
    
    s = int(data[index])
    index += 1
    t = int(data[index])
    index += 1
    d = int(data[index])
    index += 1
    
    edges = []
    capacity = []
    graph = defaultdict(list)
    
    for i in range(m):
        u = int(data[index])
        index += 1
        v = int(data[index])
        index += 1
        cap = int(data[index])
        index += 1
        edges.append((u, v))
        capacity.append(cap)
        graph[u].append(i)
        graph[v].append(i)
    
    # We will use Ford-Fulkerson with BFS (Edmonds-Karp) to find max flow
    # Build residual graph
    res_capacity = capacity[:]
    res_graph = [[] for _ in range(n+1)]
    edge_index_map = {}
    
    for i in range(m):
        u, v = edges[i]
        res_graph[u].append((v, i, True))   # forward edge
        res_graph[v].append((u, i, False)) # backward edge
    
    def bfs():
        parent = [-1] * (n + 1)
        parent_edge = [-1] * (n + 1)
        visited = [False] * (n + 1)
        queue = deque()
        queue.append(s)
        visited[s] = True
        
        while queue:
            node = queue.popleft()
            for neighbor, idx, is_forward in res_graph[node]:
                if not visited[neighbor]:
                    # Check if we can push flow
                    if is_forward:
                        if res_capacity[idx] > 0:
                            visited[neighbor] = True
                            parent[neighbor] = node
                            parent_edge[neighbor] = idx
                            if neighbor == t:
                                return parent, parent_edge
                            queue.append(neighbor)
                    else:
                        # Backward edge - check original flow
                        if capacity[idx] - res_capacity[idx] > 0:
                            visited[neighbor] = True
                            parent[neighbor] = node
                            parent_edge[neighbor] = idx
                            if neighbor == t:
                                return parent, parent_edge
                            queue.append(neighbor)
        return None, None
    
    max_flow = 0
    flow_on_edge = [0] * m
    
    while max_flow < d:
        parent, parent_edge = bfs()
        if parent is None:
            break
            
        # Find minimum capacity along the path
        path_flow = float('inf')
        curr = t
        while curr != s:
            edge_idx = parent_edge[curr]
            prev = parent[curr]
            # Determine if this is forward or backward edge
            found_direction = None
            for neighbor, idx, is_forward in res_graph[prev]:
                if neighbor == curr and idx == edge_idx:
                    found_direction = is_forward
                    break
                    
            if found_direction:
                path_flow = min(path_flow, res_capacity[edge_idx])
            else:
                # This is a backward edge
                path_flow = min(path_flow, capacity[edge_idx] - res_capacity[edge_idx])
            curr = parent[curr]
            
        if path_flow == 0:
            break
            
        path_flow = int(path_flow)
        if max_flow + path_flow > d:
            path_flow = d - max_flow
            
        max_flow += path_flow
        
        # Update residual capacities and flow_on_edge
        curr = t
        while curr != s:
            edge_idx = parent_edge[curr]
            prev = parent[curr]
            # Determine direction again
            found_direction = None
            for neighbor, idx, is_forward in res_graph[prev]:
                if neighbor == curr and idx == edge_idx:
                    found_direction = is_forward
                    break
                    
            if found_direction:
                res_capacity[edge_idx] -= path_flow
                flow_on_edge[edge_idx] += path_flow
            else:
                res_capacity[edge_idx] += path_flow
                flow_on_edge[edge_idx] -= path_flow
            curr = parent[curr]
    
    if max_flow >= d:
        print("YES")
        print(" ".join(map(str, flow_on_edge)))
    else:
        print("NO")

solve()
