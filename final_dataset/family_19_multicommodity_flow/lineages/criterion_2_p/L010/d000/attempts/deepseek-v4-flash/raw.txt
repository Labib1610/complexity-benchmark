We can model this as a flow problem with a source and a sink, and check whether we can push exactly `d` units of flow from `s` to `t` within edge capacities scooping.

We’ll use a standard max-flow algorithm (e.g., Edmonds–Karp or Dinic) to compute the maximum flow from `s` to `t`. If the max flow is at least `d`, we trace back to assign flow values to the input edgesizarrely.

I’ll implement Dinic for efficiency.

from collections import deque
import sys

class Dinic:
    def __init__(self, n):
        self.n = n
        self.graph = [[] for _ in range(n)]
        self.edges = []  # (to, rev, cap)

    def add_edge(self, fr, to, cap):
        forward = [to, len(self.graph[to]), cap]
        backward = [fr, len(self.graph[fr]), 0]
        self.graph[fr].append(forward)
        self.graph[to].append(backward)
        self.edges.append((forward, backward))

    def bfs_level(self, s, t, level):
        for i in range(self.n):
            level[i] = -1
        q = deque([s])
        level[s] = 0
        while q:
            v = q.popleft()
            for e in self.graph[v]:
                if e[2] > 0 and level[e[0]] < 0:
                    level[e[0]] = level[v] + 1
                    q.append(e[0])
        return level[t] >= 0

    def dfs_flow(self, v, t, f, it, level):
        if v == t:
            return f
        while it[v] < len(self.graph[v]):
            e = self.graph[v][it[v]]
            if e[2] > 0 and level[v] + 1 == level[e[0]]:
                ret = self.dfs_flow(e[0], t, min(f, e[2]), it, level)
                if ret > 0:
                    e[2] -= ret
                    self.graph[e[0]][e[1]][2] += ret
                    return ret
            it[v] += 1
        return 0

    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        level = [-1]*self.n
        while self.bfs_level(s, t, level):
            it = [0]*self.n
            while True:
                f = self.dfs_flow(s, t, INF, it, level)
                if f == 0:
                    break
                flow += f
        return flow
