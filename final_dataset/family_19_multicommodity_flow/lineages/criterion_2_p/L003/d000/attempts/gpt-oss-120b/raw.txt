import sys
sys.setrecursionlimit(1 << 25)

class Edge:
    __slots__ = ('to', 'rev', 'cap', 'flow', 'idx')
    def __init__(self, to, rev, cap, idx=-1):
        self.to = to
        self.rev = rev
        self.cap = cap
        self.flow = 0
        self.idx = idx  # original edge index, -1 for added reverse edges

class Dinic:
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]

    def add_edge(self, fr, to, cap, idx=-1):
        fwd = Edge(to, len(self.g[to]), cap, idx)
        rev = Edge(fr, len(self.g[fr]), 0, -1)
        self.g[fr].append(fwd)
        self.g[to].append(rev)

    def bfs(self, s, t):
        self.level = [-1]*self.n
        q = [s]
        self.level[s] = 0
        for v in q:
            for e in self.g[v]:
                if e.cap - e.flow > 0 and self.level[e.to] < 0:
                    self.level[e.to] = self.level[v] + 1
                    q.append(e.to)
        return self.level[t] >= 0

    def dfs(self, v, t, f):
        if v == t:
            return f
        for i in range(self.it[v], len(self.g[v])):
            e = self.g[v][i]
            if e.cap - e.flow > 0 and self.level[v] < self.level[e.to]:
                ret = self.dfs(e.to, t, min(f, e.cap - e.flow))
                if ret:
                    e.flow += ret
                    self.g[e.to][e.rev].flow -= ret
                    return ret
            self.it[v] += 1
        return 0

    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0]*self.n
            while True:
                f = self.dfs(s, t, INF)
                if not f:
                    break
                flow += f
        return flow

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); _ = int(next(it))  # the third value is always 1, ignored
    s = int(next(it))-1; t = int(next(it))-1; d = int(next(it))
    dinic = Dinic(n)
    edge_refs = [None]*m
    for idx in range(m):
        u = int(next(it))-1
        v = int(next(it))-1
        cap = int(next(it))
        dinic.add_edge(u, v, cap, idx)
        # store reference to the forward edge just added
        edge_refs[idx] = dinic.g[u][-1]

    flow = dinic.max_flow(s, t)
    if flow != d:
        print("NO")
        return
    # retrieve flows in original order
    ans = [str(edge_refs[i].flow) for i in range(m)]
    print("YES")
    print(' '.join(ans))

if __name__ == "__main__":
    solve()
