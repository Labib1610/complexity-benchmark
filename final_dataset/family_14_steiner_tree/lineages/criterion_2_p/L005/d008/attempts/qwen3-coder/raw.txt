import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    if t <= 1:
        print(0)
        return
        
    focus = list(map(int, sys.stdin.readline().split()))
    focus_set = set(focus)
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    # Union-Find data structure
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
    
    # Sort edges by weight
    edges.sort()
    
    # Kruskal's algorithm: we want to connect all focus vertices with minimum cost
    # We can stop as soon as all focus vertices are in the same component
    cost = 0
    focus_count = len(focus)
    connected_focus = 1  # Start with one focus vertex as the root
    
    # Initially, each focus vertex is its own component
    focus_parent = {v: v for v in focus}
    
    def find_focus(v):
        if focus_parent[v] != v:
            focus_parent[v] = find_focus(focus_parent[v])
        return focus_parent[v]
    
    def union_focus(u, v):
        pu, pv = find_focus(u), find_focus(v)
        if pu != pv:
            focus_parent[pv] = pu
            return True
        return False
    
    # We'll run Kruskal but only track connections between focus vertices
    for w, u, v in edges:
        # First, union in the main graph
        if find(u) != find(v):
            union(u, v)
            cost += w
            # Check if this edge connects two focus vertices or connects a focus vertex
            if u in focus_set and v in focus_set:
                if union_focus(u, v):
                    connected_focus += 1
            elif u in focus_set or v in focus_set:
                # One is focus, the other might connect to focus through the tree
                # We need to check if this connects to any existing focus component
                pass
                    
        # Optimization: if all focus vertices are connected, we can stop
        if connected_focus == focus_count:
            break
            
    # Better approach: since the graph is a tree (m = n-1), we can find the minimal subtree
    # that connects all focus vertices. This is the Steiner tree problem on a tree.
    
    # Reset Union-Find for a cleaner approach
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    # Sort edges by weight for Kruskal
    edges.sort()
    
    total_cost = 0
    components_with_focus = set(focus)
    
    # Use Union-Find to track which components contain focus vertices
    # When we merge two components, if both had focus vertices, we needed this edge
    # Keep track of how many focus vertices are in each component
    
    focus_count_in_component = {v: 1 for v in focus}
    
    for w, u, v in edges:
        # Find roots of u and v
        root_u = find(u)
        root_v = find(v)
        
        if root_u != root_v:
            # Check if both components have focus vertices
            count_u = focus_count_in_component.get(root_u, 0)
            count_v = focus_count_in_component.get(root_v, 0)
            
            # Merge the components
            if rank[root_u] < rank[root_v]:
                root_u, root_v = root_v, root_u
                
            parent[root_v] = root_u
            if rank[root_u] == rank[root_v]:
                rank[root_u] += 1
                
            # Update focus count for the new component
            new_count = count_u + count_v
            if new_count > 0:
                focus_count_in_component[root_u] = new_count
                
            # If both components had focus vertices, we need this edge
            if count_u > 0 and count_v > 0:
                total_cost += w
    
    print(total_cost)

threading.Thread(target=main).start()
