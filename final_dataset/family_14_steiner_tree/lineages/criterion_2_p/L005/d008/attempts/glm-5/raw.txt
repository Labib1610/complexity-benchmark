```python
import sys
import heapq

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    focus = set()
    for _ in range(t):
        focus.add(int(next(it)))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    INF = float('inf')
    dist = [INF] * (n + 1)
    in_mst = [False] * (n + 1)
    pq = []
    
    for node in focus:
        dist[node] = 0
        heapq.heappush(pq, (0, node))
    
    total_cost = 0
    nodes_in_mst = 0
    
    while pq and nodes_in_mst < n:
        d, u = heapq.heappop(pq)
        if in_mst[u]:
            continue
        in_mst[u] = True
        total_cost += d
        nodes_in_mst += 1
        
        for v, w in adj[u]:
            if not in_mst[v] and w < dist[v]:
                dist[v] = w
                heapq.heappush(pq, (w, v))
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
