import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    essential = set()
    idx = 3
    for _ in range(t):
        essential.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the smallest subtree containing all essential nodes.
    # This is equivalent to summing the weights of all edges that lie on 
    # the path between at least one pair of essential nodes.
    # An edge (u, v) with weight w is part of the Steiner tree if and only if
    # the subtree formed by removing the edge contains at least one essential node
    # on both sides.
    
    is_essential = [False] * (n + 1)
    for node in essential:
        is_essential[node] = True
        
    count_essential = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if is_essential[u]:
            count_essential[u] += 1
        
        if parent[u] != 0:
            p = parent[u]
            count_essential[p] += count_essential[u]
            # If this edge separates essential nodes, it must be included
            if count_essential[u] > 0 and count_essential[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
