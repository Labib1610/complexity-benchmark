We can solve this by pruning the tree: remove all non-essential leaves iteratively, then sum the weights of the remaining edges.  

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m, t = map(int, input().split())
essential = set(map(int, input().split()))

adj = [[] for _ in range(n + 1)]
for _ in range(m):
    u, v, w = map(int, input().split())
    adj[u].append((v, w))
    adj[v].append((u, w))

degree = [0] * (n + 1)
for i in range(1, n + 1):
    degree[i] = len(adj[i])

# Remove non-essential leaves
removed = [False] * (n + 1)
q = deque()
for i in range(1, n + 1):
    if i not in essential and degree[i] == 1:
        q.append(i)

while q:
    u = q.popleft()
    removed[u] = True
    for v, w in adj[u]:
        if not removed[v]:
            degree[v] -= 1
            if v not in essential and degree[v] == 1:
                q.append(v)

# Sum weights of edges connecting remaining nodes
total = 0
for u in range(1, n + 1):
    if removed[u]:
        continue
    for v, w in adj[u]:
        if u < v and not removed[v]:
            total += w

print(total)
```
