import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1000000)
    
    n, m, t = map(int, sys.stdin.readline().split())
    priority = set(map(int, sys.stdin.readline().split()))
    
    # Since it's a tree, we can represent it with adjacency list
    tree = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        tree[u].append((v, w))
        tree[v].append((u, w))
        edges.append((u, v, w))
    
    # If there's only one priority node, cost is 0
    if t <= 1:
        print(0)
        return
    
    # Find the subtree that spans all priority nodes
    # We can do this by finding all edges that are on paths between priority nodes
    
    # First, find all nodes in the subtree spanning the priority nodes
    # We'll use a multi-source BFS to find the minimal subtree
    
    # But there's a simpler approach for a tree:
    # The minimal connected subgraph containing all priority nodes is the union of paths between them
    # In a tree, this is equivalent to the minimal subtree containing all priority nodes
    
    # Let's find the minimal subtree containing all priority nodes
    # by doing a DFS and keeping only edges that lead to priority nodes
    
    visited = [False] * (n + 1)
    included_edges = []
    
    def dfs(u, parent):
        visited[u] = True
        has_priority = u in priority
        children_info = []
        
        for v, w in tree[u]:
            if v == parent:
                continue
            if not visited[v]:
                child_has_priority, child_edges = dfs(v, u)
                if child_has_priority:
                    has_priority = True
                    included_edges.extend(child_edges)
                    included_edges.append((u, v, w))
        
        return has_priority, []
    
    # Start DFS from any node
    dfs(1, -1)
    
    # Sum the weights of included edges
    total_cost = sum(w for u, v, w in included_edges)
    print(total_cost)

if __name__ == "__main__":
    main()
