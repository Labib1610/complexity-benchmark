import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    landmarks = set()
    idx = 3
    for _ in range(t):
        landmarks.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the sum of weights of all edges that lie on the path
    # between at least one pair of landmark nodes.
    # An edge (u, v) with weight w lies on a path between two landmarks if and only if
    # the subtree formed by removing (u, v) contains at least one landmark on both sides.
    
    is_landmark = [False] * (n + 1)
    for l in landmarks:
        is_landmark[l] = True
        
    subtree_landmarks = [0] * (n + 1)
    total_weight = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to count landmarks in subtrees
    for u in reversed(order):
        if is_landmark[u]:
            subtree_landmarks[u] += 1
        if parent[u] != -1:
            subtree_landmarks[parent[u]] += subtree_landmarks[u]
            
    # An edge is part of the union of paths if both sides have >= 1 landmark
    ans = 0
    for i in range(2, n + 1):
        count = subtree_landmarks[i]
        if count > 0 and count < t:
            ans += edge_to_parent_weight[i]
            
    print(ans)

if __name__ == '__main__':
    solve()
