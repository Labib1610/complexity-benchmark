import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); t = int(next(it))
    essential = [0] * (n + 1)
    for _ in range(t):
        essential[int(next(it))] = 1
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w, len(edges)))
        adj[v].append((u, w, len(edges)))
        edges.append((u, v, w))

    if t <= 1:
        print(0)
        return

    parent = [0] * (n + 1)
    pw = [0] * (n + 1)          # weight to parent
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        v = stack.pop()
        order.append(v)
        for to, w, _ in adj[v]:
            if to == parent[v]:
                continue
            parent[to] = v
            pw[to] = w
            stack.append(to)

    # post-order processing
    cnt = [0] * (n + 1)
    for v in reversed(order):
        cnt[v] = essential[v]
        for to, _, _ in adj[v]:
            if parent[to] == v:
                cnt[v] += cnt[to]

    total_ess = t
    ans = 0
    for v in range(2, n + 1):
        if cnt[v] > 0 and total_ess - cnt[v] > 0:
            ans += pw[v]

    print(ans)

if __name__ == "__main__":
    solve()
