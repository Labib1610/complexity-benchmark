```python
import sys

# The problem asks for the sum of weights of all edges that lie on the unique 
# paths between any two critical modules in a tree.
# An edge is part of the path between two critical modules if and only if 
# removing that edge splits the tree into two components, each containing 
# at least one critical module.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many critical nodes are in the subtree of each node.
    # Since it's a tree, we can use DFS. To avoid recursion depth issues, use iterative DFS.
    
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    critical_count = [0] * (n + 1)
    for node in critical_nodes:
        critical_count[node] = 1
        
    total_energy = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If the subtree at u contains some but not all critical nodes,
            # the edge (u, p) must be included.
            if 0 < critical_count[u] < t:
                total_energy += edge_weight[u]
            critical_count[p] += critical_count[u]
            
    print(total_energy)

if __name__ == '__main__':
    # Increase recursion depth just in case, though iterative is used
    sys.setrecursionlimit(300000)
    solve()
```
