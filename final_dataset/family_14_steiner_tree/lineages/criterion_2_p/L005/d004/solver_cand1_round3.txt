import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    mandatory = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        mandatory[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the smallest subtree containing all mandatory nodes.
    # This is equivalent to the sum of weights of all edges that lie on a path
    # between any two mandatory nodes.
    # An edge is part of this subtree if and only if the subtree rooted at one
    # of its endpoints contains at least one mandatory node and the other
    # subtree also contains at least one mandatory node.
    
    has_mandatory = [False] * (n + 1)
    for i in range(1, n + 1):
        if mandatory[i]:
            has_mandatory[i] = True
            
    total_weight = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order
    for u in reversed(order):
        if has_mandatory[u]:
            if parent[u] != 0:
                has_mandatory[parent[u]] = True
                total_weight += edge_to_parent_weight[u]
                
    print(total_weight)

if __name__ == '__main__':
    solve()
