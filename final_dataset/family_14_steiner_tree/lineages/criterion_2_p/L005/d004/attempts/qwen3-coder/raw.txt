import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1000000)
    
    n, m, t = map(int, sys.stdin.readline().split())
    critical = set(map(int, sys.stdin.readline().split()))
    
    # Build the tree
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # If there's only one or zero critical modules, no conduits needed
    if t <= 1:
        print(0)
        return
    
    # Find the subtree that spans all critical nodes
    # We can do this by finding all nodes that are on paths between critical nodes
    
    # First, find one critical node to start BFS
    start = next(iter(critical))
    
    # BFS to find distances and parents
    parent = [-1] * (n + 1)
    dist = [-1] * (n + 1)
    dist[start] = 0
    queue = deque([start])
    
    while queue:
        u = queue.popleft()
        for v, w in adj[u]:
            if dist[v] == -1:
                dist[v] = dist[u] + 1
                parent[v] = u
                queue.append(v)
    
    # Find the farthest critical node from start
    max_dist = -1
    farthest = start
    for node in critical:
        if dist[node] > max_dist:
            max_dist = dist[node]
            farthest = node
    
    # BFS from farthest critical node to find the other end of the diameter
    dist2 = [-1] * (n + 1)
    parent2 = [-1] * (n + 1)
    dist2[farthest] = 0
    queue = deque([farthest])
    
    while queue:
        u = queue.popleft()
        for v, w in adj[u]:
            if dist2[v] == -1:
                dist2[v] = dist2[u] + 1
                parent2[v] = u
                queue.append(v)
    
    # Find the critical node farthest from 'farthest'
    max_dist2 = -1
    farthest2 = farthest
    for node in critical:
        if dist2[node] > max_dist2:
            max_dist2 = dist2[node]
            farthest2 = node
    
    # Now we have the diameter of the critical nodes
    # The path between farthest and farthest2 covers all critical nodes in the subtree
    # We need to activate all edges on paths between critical nodes
    
    # Mark all nodes on the path from farthest2 to farthest
    path_nodes = set()
    current = farthest2
    while current != -1:
        path_nodes.add(current)
        current = parent2[current]
    
    # Now we do a DFS from each critical node to mark all paths to other critical nodes
    # But since it's a tree and we want the minimal subtree connecting all critical nodes,
    # we can find the Steiner tree
    
    # A simpler approach: since it's a tree, the minimal subtree connecting all critical nodes
    # is the union of paths between all pairs, but we can find it more efficiently
    
    # We'll find the minimal subtree that connects all critical nodes
    # by marking all nodes that are on paths between critical nodes
    
    # Let's use a different approach:
    # Since it's a tree, we can find the minimal connected subtree containing all critical nodes
    # by finding the union of paths between a root critical node and all other critical nodes
    
    # Mark which nodes are visited in our subtree
    visited = [False] * (n + 1)
    
    # Function to mark path between two nodes
    def mark_path(a, b):
        # Find path from a to b
        # First, get path from a to root, and b to root
        path_a = []
        x = a
        while x != -1:
            path_a.append(x)
            x = parent[x]
        
        path_b = []
        x = b
        while x != -1:
            path_b.append(x)
            x = parent[x]
        
        # Convert to sets to find LCA
        set_a = set(path_a)
        lca = -1
        for x in path_b:
            if x in set_a:
                lca = x
                break
        
        # Mark path from a to LCA
        x = a
        while x != lca:
            visited[x] = True
            x = parent[x]
        
        # Mark path from b to LCA
        x = b
        while x != lca:
            visited[x] = True
            x = parent[x]
        
        # Mark LCA
        visited[lca] = True
    
    # Mark all nodes on paths between critical nodes
    # Take first critical node as reference
    ref = next(iter(critical))
    for node in critical:
        if node != ref:
            mark_path(ref, node)
    
    # Also mark the reference node
    visited[ref] = True
    
    # Now sum up weights of edges in the subtree
    total_energy = 0
    
    # Check each edge
    for u, v, w in edges:
        # If both endpoints are marked, this edge is in our subtree
        if visited[u] and visited[v]:
            total_energy += w
    
    print(total_energy)

if __name__ == "__main__":
    main()
