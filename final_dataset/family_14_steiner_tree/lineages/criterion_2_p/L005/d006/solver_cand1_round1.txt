```python
import sys

# The problem asks for the minimum cost to connect all primary sensors in a tree.
# Since m = n - 1 and the graph is connected, the input is a tree.
# We need to find the smallest subtree that contains all primary sensors.
# The cost of this subtree is the sum of the weights of all edges that lie on
# any path between two primary sensors.
# An edge (u, v) with weight w is part of the minimal subtree if and only if
# the removal of this edge splits the tree into two components, each containing
# at least one primary sensor.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary_sensors = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many primary sensors are in the subtree rooted at each node
    # We use an iterative DFS to avoid recursion depth issues
    count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in primary_sensors:
            count[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            count[p] += count[u]
            # If the subtree at u contains at least one primary sensor
            # and the rest of the tree contains at least one, include the edge
            if 0 < count[u] < t:
                total_cost += edge_weight[u]
                
    print(total_cost)

solve()
```
