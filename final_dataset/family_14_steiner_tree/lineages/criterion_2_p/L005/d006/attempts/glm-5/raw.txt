```python
import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    t = int(input_data[idx]); idx += 1

    primary = set()
    for _ in range(t):
        primary.add(int(input_data[idx]))
        idx += 1

    edges = []
    for _ in range(m):
        u = int(input_data[idx]); idx += 1
        v = int(input_data[idx]); idx += 1
        w = int(input_data[idx]); idx += 1
        edges.append((w, u, v))

    if t <= 1:
        print(0)
        return

    edges.sort()

    parent = list(range(n + 1))
    size = [1] * (n + 1)
    primary_count = [0] * (n + 1)
    for p in primary:
        primary_count[p] = 1

    total_cost = 0

    for w, u, v in edges:
        ru = u
        while parent[ru] != ru:
            parent[ru] = parent[parent[ru]]
            ru = parent[ru]

        rv = v
        while parent[rv] != rv:
            parent[rv] = parent[parent[rv]]
            rv = parent[rv]

        if ru == rv:
            continue

        if size[ru] < size[rv]:
            ru, rv = rv, ru

        parent[rv] = ru
        size[ru] += size[rv]
        primary_count[ru] += primary_count[rv]

        total_cost += w

        if primary_count[ru] == t:
            print(total_cost)
            return

    print(total_cost)

if __name__ == "__main__":
    solve()
```
