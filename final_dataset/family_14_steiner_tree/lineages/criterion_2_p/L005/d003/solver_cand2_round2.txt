```python
import sys

# The problem states that the n-1 constraints form a tree (since it's an integrated system
# with n-1 edges connecting n nodes). We need to find the minimum cost to connect all
# target variables. This is equivalent to finding the sum of weights of all edges
# that lie on the unique paths between any two target variables.
# An edge in a tree is part of the path between two target variables if and only if
# the edge splits the tree into two components, each containing at least one target variable.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many target variables are in the subtree of each node.
    # Since it's a tree, we can root it at an arbitrary node (e.g., 1).
    # For an edge (u, v) where v is a child of u, the edge is needed if the number
    # of targets in v's subtree is > 0 and < t.
    
    target_count = [0] * (n + 1)
    for node in targets:
        target_count[node] = 1
        
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != -1:
            p = parent[u]
            target_count[p] += target_count[u]
            # If this subtree contains some but not all targets, the edge is required
            if 0 < target_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    # Increase recursion depth for deep trees
    sys.setrecursionlimit(300000)
    solve()
```
