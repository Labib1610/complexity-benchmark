```python
import sys

# The problem states that the proposed transmission lines form a tree (n nodes, n-1 edges, no redundancy).
# We need to find the minimum weight of the subtree that connects all priority nodes.
# In a tree, the unique path between any two nodes is fixed.
# The union of all paths between all pairs of priority nodes forms the minimal subtree.
# An edge is part of this subtree if and only if it separates the tree into two components,
# each of which contains at least one priority node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many priority nodes are in the subtree rooted at each node
    # when the tree is rooted at node 1.
    subtree_priority_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in priority_nodes:
            subtree_priority_count[node] += 1
            
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_priority_count[p] += subtree_priority_count[u]
            # If this subtree contains some but not all priority nodes,
            # the edge to the parent must be included.
            if 0 < subtree_priority_count[u] < t:
                total_weight += parent_edge_weight[u]
                
    print(total_weight)

solve()
```
