```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes (priority substations)
# in a tree structure. This is equivalent to finding the weight of the Steiner Tree
# in a tree, which is simply the sum of weights of all edges that lie on the unique
# path between any two priority nodes.
# An edge is part of the Steiner Tree if and only if the subtree formed by removing
# that edge contains at least one priority node on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_nodes = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many priority nodes are in the subtree of each edge.
    # Since it's a tree, we can root it at node 1 and perform a DFS.
    
    priority_count = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [(1, 0)]
    visited[1] = True
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != p:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append((v, u))
                
    for node in order:
        if node in priority_nodes:
            priority_count[node] = 1
            
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        priority_count[p] += priority_count[u]
        
        # If the subtree at u contains some but not all priority nodes,
        # the edge (u, p) must be included.
        if 0 < priority_count[u] < t:
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
