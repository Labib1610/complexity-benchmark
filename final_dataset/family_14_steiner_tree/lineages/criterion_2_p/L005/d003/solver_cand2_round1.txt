```python
import sys

# The problem states that we have n variables and n-1 constraints, 
# which form a tree structure (since it reduces to 1 degree of freedom).
# We need to find the minimum cost to connect all target variables.
# This is equivalent to finding the sum of weights of all edges on the 
# unique paths between all target variables in the tree.
# This is equivalent to the sum of weights of all edges that have at least 
# one target variable in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # subtree_targets[u] stores the number of target variables in the subtree of u
    subtree_targets = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            subtree_targets[u] += 1
        if parent[u] != -1:
            subtree_targets[parent[u]] += subtree_targets[u]
            
    total_cost = 0
    # For each edge (u, v) where v is child of u, 
    # if 0 < subtree_targets[v] < t, the edge must be included.
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if parent[v] == u:
                if 0 < subtree_targets[v] < t:
                    total_cost += w
            elif parent[u] == v:
                if 0 < subtree_targets[u] < t:
                    total_cost += w
                    
    print(total_cost // 2)

solve()
```
