```python
import sys

# The problem asks for a subset of edges S such that the sum of squared differences
# along edges in S bounds the squared difference between any two focus indices.
# This is equivalent to saying that in the graph formed by S, all focus indices
# must belong to the same connected component.
# Since the original graph is a tree (m = n - 1), the subset S must contain
# all edges on the unique paths between all pairs of focus indices.
# This is equivalent to finding the minimal subtree that connects all focus indices.
# This is the Steiner Tree problem on a tree, which is simply the union of all
# paths between the focus indices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    focus_indices = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the minimal subtree connecting all focus indices:
    # 1. Root the tree at an arbitrary focus index.
    # 2. A node is part of the minimal subtree if its subtree contains at least one focus index
    #    AND the path from it to the root contains at least one focus index (or it is one).
    #    Actually, simpler: an edge is in the minimal subtree if and only if
    #    the two components it splits the tree into both contain at least one focus index.
    
    root = next(iter(focus_indices))
    
    # Count focus nodes in each subtree
    subtree_focus_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in focus_indices:
            subtree_focus_count[u] += 1
        if parent[u] != 0:
            subtree_focus_count[parent[u]] += subtree_focus_count[u]
            
    total_cost = 0
    for i in range(1, n + 1):
        if i == root:
            continue
        # If the subtree at i contains some focus nodes, and the rest of the tree
        # also contains some focus nodes, the edge to parent must be included.
        if 0 < subtree_focus_count[i] < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
