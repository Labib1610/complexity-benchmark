```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    essential_sites = set()
    idx = 3
    for _ in range(t):
        essential_sites.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # is_essential_subtree[u] will be true if the subtree rooted at u 
    # contains at least one essential site.
    # count_essential[u] will be the number of essential sites in the subtree.
    count_essential = [0] * (n + 1)
    
    # We need to traverse the tree to calculate which edges are part of the 
    # path between any two essential sites. An edge (u, v) where v is a child 
    # of u is part of the path if and only if the subtree at v contains 
    # at least one essential site AND the rest of the tree (outside v's subtree) 
    # contains at least one essential site.
    
    total_essential = t
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_cost = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_cost[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in essential_sites:
            count_essential[u] += 1
        
        if parent[u] != 0:
            count_essential[parent[u]] += count_essential[u]
            
            # Check if edge (parent[u], u) is essential
            # It is essential if 0 < count_essential[u] < total_essential
            if 0 < count_essential[u] < total_essential:
                total_cost += edge_to_parent_cost[u]
                
    print(total_cost)

solve()
```
