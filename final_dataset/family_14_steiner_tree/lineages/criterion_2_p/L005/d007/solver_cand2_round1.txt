```python
import sys

# The problem asks for the sum of weights of edges in a tree (n nodes, n-1 edges)
# such that removing the edge splits the tree into two components, each containing
# at least one control variable.
# This is equivalent to saying that an edge is critical if and only if it lies
# on the unique path between any two control variables.
# More generally, an edge is critical if and only if it separates the set of
# control variables into two non-empty sets.
# This happens if and only if the subtree formed by removing the edge contains
# at least one control variable, and the other component also contains at least
# one control variable.
# Let 'count[u]' be the number of control variables in the subtree rooted at u.
# An edge (u, parent(u)) is critical if 0 < count[u] < t.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    controls = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # DFS to compute subtree control counts
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    control_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if i in controls:
            control_count[i] = 1
            
    total_critical_weight = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        control_count[p] += control_count[u]
        if 0 < control_count[u] < t:
            total_critical_weight += edge_to_parent_w[u]
            
    print(total_critical_weight)

if __name__ == '__main__':
    solve()
```
