```python
import sys

# The problem states that the network is a tree (N clocks, N-1 procedures, 
# uniquely connected). We need to find the minimum cost to connect all 
# master clocks. In a tree, the set of edges required to connect a subset 
# of nodes is exactly the union of all paths between any two nodes in that 
# subset. This is equivalent to the Steiner Tree problem, but since the 
# graph is a tree, the Steiner Tree is simply the minimal subtree containing 
# all master clocks. This subtree consists of all edges that lie on the 
# unique path between any two master clocks.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    K = int(input_data[2])
    
    if K <= 1:
        print(0)
        return
    
    masters = set(map(int, input_data[3:3+K]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + K
    for _ in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the minimal subtree containing all master clocks:
    # 1. Root the tree at an arbitrary master clock.
    # 2. An edge is part of the minimal subtree if and only if the subtree 
    #    defined by that edge (away from the root) contains at least one master clock.
    
    root = next(iter(masters))
    
    # has_master[u] will be True if the subtree rooted at u contains a master clock
    has_master = [False] * (N + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_to_parent_cost = [0] * (N + 1)
    order = []
    stack = [root]
    visited = [False] * (N + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_cost[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in masters:
            has_master[u] = True
        
        if has_master[u] and parent[u] != 0:
            has_master[parent[u]] = True
            total_cost += edge_to_parent_cost[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
