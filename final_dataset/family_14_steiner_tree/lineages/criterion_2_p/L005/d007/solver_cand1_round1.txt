```python
import sys

# The problem asks for the minimum cost to connect all focus indices in a graph
# where edges represent vectors e_u - e_v.
# The vector e_a - e_b is in the span of a set of vectors {e_u_i - e_v_i}
# if and only if there is a path between a and b in the graph formed by these edges.
# Since we need e_a - e_b for all pairs of focus indices, all focus indices
# must belong to the same connected component in the chosen subgraph.
# We are given m = n - 1 edges, which form a tree.
# We need to find the minimum cost to connect all focus indices.
# This is equivalent to finding the sum of weights of all edges that lie on
# the unique paths between the focus indices in the tree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    focus_indices = set(map(int, input_data[3:3+t]))
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many focus nodes are in the subtree of each node
    subtree_focus_count = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in focus_indices:
            subtree_focus_count[u] += 1
        if parent[u] != 0:
            subtree_focus_count[parent[u]] += subtree_focus_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If an edge splits the focus nodes into two non-empty sets,
        # it must be included in the spanning set.
        count = subtree_focus_count[i]
        if count > 0 and count < t:
            total_cost += edge_weight_to_parent[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
