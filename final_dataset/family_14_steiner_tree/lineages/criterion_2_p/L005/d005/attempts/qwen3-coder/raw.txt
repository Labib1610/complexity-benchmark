import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal_with_critical_nodes(n, edges, critical_nodes):
    if len(critical_nodes) <= 1:
        return 0
    
    # Create a mapping for critical nodes to compressed indices
    critical_set = set(critical_nodes)
    
    # First, build the full MST to get distances between all nodes
    parent = list(range(n))
    rank = [0] * n
    edges.sort()
    
    # Build MST using Kruskal's algorithm
    mst_edges = []
    for w, u, v in edges:
        if union(parent, rank, u, v):
            mst_edges.append((w, u, v))
    
    # Now create a graph with only the MST edges
    adj = defaultdict(list)
    for w, u, v in mst_edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Function to find distance between two nodes in tree
    def bfs_distances(start, adj, nodeset):
        distances = {}
        queue = [(start, 0)]
        visited = set()
        while queue:
            node, dist = queue.pop(0)
            if node in visited:
                continue
            visited.add(node)
            if node in nodeset:
                distances[node] = dist
            for neighbor, weight in adj[node]:
                if neighbor not in visited:
                    queue.append((neighbor, dist + weight))
        return distances
    
    # Find the minimum spanning tree connecting only critical nodes
    # This is essentially finding a Steiner tree, but we can do it by 
    # creating a complete graph on critical nodes with shortest path distances
    
    # Calculate shortest paths between all pairs of critical nodes
    critical_list = list(critical_nodes)
    distance_matrix = {}
    
    for i, start_node in enumerate(critical_list):
        distances = bfs_distances(start_node, adj, critical_set)
        distance_matrix[start_node] = distances
    
    # Create edges for complete graph on critical nodes
    steiner_edges = []
    for i in range(len(critical_list)):
        for j in range(i+1, len(critical_list)):
            u = critical_list[i]
            v = critical_list[j]
            if v in distance_matrix[u]:
                steiner_edges.append((distance_matrix[u][v], u, v))
    
    # Run Kruskal on this complete graph to get minimum cost to connect all critical nodes
    steiner_parent = list(range(n))  # Using same size for simplicity
    steiner_rank = [0] * n
    steiner_edges.sort()
    
    total_cost = 0
    connected_components = len(critical_list)
    
    for w, u, v in steiner_edges:
        if find(steiner_parent, u) != find(steiner_parent, v):
            union(steiner_parent, steiner_rank, u, v)
            total_cost += w
            connected_components -= 1
            if connected_components == 1:
                break
                
    return total_cost

# Read input
line1 = list(map(int, sys.stdin.readline().split()))
n, m, t = line1[0], line1[1], line1[2]

critical_sectors = list(map(int, sys.stdin.readline().split()))
# Convert to 0-indexed
critical_sectors = [x - 1 for x in critical_sectors]

edges = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    # Convert to 0-indexed
    edges.append((w, u-1, v-1))

result = kruskal_with_critical_nodes(n, edges, critical_sectors)
print(result)
