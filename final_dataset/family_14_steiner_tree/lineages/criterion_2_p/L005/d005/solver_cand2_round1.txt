```python
import sys

# The problem states that the incidence matrix columns are linearly independent.
# Since there are n variables and m = n - 1 dependencies, and the incidence matrix
# has n rows and n-1 columns, linear independence implies that the graph
# formed by the dependencies is a tree (or a forest if n > m+1, but here m=n-1).
# A graph with n nodes and n-1 edges that is linearly independent in its 
# incidence matrix is a tree.
# Thus, the graph is a tree.
# To unify a set of primary variables in a tree, we must select all edges
# that lie on the unique paths between any two primary variables.
# This is equivalent to selecting all edges in the minimal subtree that
# connects all primary variables (the Steiner Tree in a tree).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary_vars = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the edges in the minimal subtree connecting all primary variables:
    # 1. Root the tree at an arbitrary primary variable.
    # 2. For each node, determine if its subtree contains any primary variable.
    # 3. An edge (u, v) where v is a child of u is in the minimal subtree
    #    if and only if the subtree at v contains at least one primary variable
    #    AND the rest of the tree (outside v's subtree) contains at least one.
    
    root = next(iter(primary_vars))
    has_primary = [False] * (n + 1)
    for p in primary_vars:
        has_primary[p] = True
        
    total_primary = t
    subtree_has_primary = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    ans = 0
    for u in reversed(order):
        if has_primary[u]:
            subtree_has_primary[u] += 1
        if parent[u] != 0:
            subtree_has_primary[parent[u]] += subtree_has_primary[u]
            # If the subtree at u has primary nodes and the rest of the tree has primary nodes
            if 0 < subtree_has_primary[u] < total_primary:
                ans += edge_weight[u]
                
    print(ans)

solve()
```
