```python
import sys
from sys import setrecursionlimit

def main():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    terminals = set()
    for _ in range(t):
        terminals.add(int(data[idx]) - 1)
        idx += 1
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(data[idx]) - 1; idx += 1
        v = int(data[idx]) - 1; idx += 1
        w = int(data[idx]); idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    parent = [0] * n
    parent_weight = [0] * n
    depth = [0] * n
    order = [0] * n
    parent[0] = -1
    stack = [0]
    visited = [False] * n
    visited[0] = True
    cnt = 0
    
    while stack:
        u = stack.pop()
        order[cnt] = u
        cnt += 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_weight[v] = w
                depth[v] = depth[u] + 1
                stack.append(v)
    
    is_terminal = [False] * n
    for x in terminals:
        is_terminal[x] = True
    
    marked = [False] * n
    for x in terminals:
        marked[x] = True
    
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 0:
            continue
        if marked[u]:
            marked[parent[u]] = True
    
    total = 0
    for i in range(n - 1, 0, -1):
        u = order[i]
        if marked[u]:
            total += parent_weight[u]
    
    print(total)

main()
```
