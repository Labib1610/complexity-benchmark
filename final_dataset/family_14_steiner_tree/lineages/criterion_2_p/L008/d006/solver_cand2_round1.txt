```python
import sys

# The problem states that the graph is a tree (R nodes, R-1 edges, connected).
# A subset of nodes is "coherent" if it is connected in the original tree.
# We need to find a connected subgraph that contains all P mandatory nodes
# such that the sum of weights of edges in the subgraph is minimized.
# The minimum connected subgraph containing a set of nodes is the Steiner Tree.
# In a tree, the Steiner Tree is simply the union of all paths between any two
# mandatory nodes. This is equivalent to the minimal subtree containing all
# mandatory nodes.
# The edges in this minimal subtree are exactly those edges that lie on the
# unique path between at least one pair of mandatory nodes.
# An edge (u, v) with weight w is in this subtree if and only if the removal
# of (u, v) splits the tree into two components, each containing at least one
# mandatory node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandatory = set()
    idx = 3
    for _ in range(P):
        mandatory.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    edges = []
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count mandatory nodes in each subtree formed by removing an edge.
    # Root the tree at node 1.
    subtree_mandatory = [0] * (R + 1)
    parent = [0] * (R + 1)
    edge_to_parent_weight = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in mandatory:
            subtree_mandatory[u] += 1
        if parent[u] != 0:
            subtree_mandatory[parent[u]] += subtree_mandatory[u]
            
    total_cost = 0
    for i in range(2, R + 1):
        # Edge is between i and parent[i]
        count = subtree_mandatory[i]
        if count > 0 and count < P:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

solve()
```
