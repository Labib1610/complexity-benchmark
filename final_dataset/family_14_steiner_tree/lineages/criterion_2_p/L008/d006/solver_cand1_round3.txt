```python
import sys

# The problem states that there are R components and S = R - 1 integrations.
# This means the graph is a tree.
# We need to select a subset of components that includes all P mandated components
# and is cohesive (connected).
# In a tree, the smallest connected subgraph containing a set of nodes P is the
# unique minimal subtree that spans all nodes in P.
# Any valid cohesive subset containing P must contain this minimal subtree.
# To minimize the total weight of internalized integrations, we should select
# exactly the minimal subtree that connects all P mandated components.
# The cost is the sum of weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set()
    idx = 3
    for _ in range(P):
        mandated.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # We need to find the sum of weights of edges that lie on the paths
    # between any two mandated nodes.
    # An edge is part of the minimal subtree if and only if the subtree
    # formed by removing the edge splits the mandated nodes into two non-empty sets.
    
    # Count how many mandated nodes are in each subtree if we root the tree at 1.
    subtree_mandated_count = [0] * (R + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    edge_weight_to_parent = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in mandated:
            subtree_mandated_count[u] += 1
        if parent[u] != 0:
            subtree_mandated_count[parent[u]] += subtree_mandated_count[u]
            
    total_cost = 0
    for i in range(2, R + 1):
        # If the subtree at i contains some but not all mandated nodes,
        # the edge connecting i to its parent must be in the minimal subtree.
        count = subtree_mandated_count[i]
        if count > 0 and count < P:
            total_cost += edge_weight_to_parent[i]
            
    print(total_cost)

solve()
```
