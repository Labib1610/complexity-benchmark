```python
import sys

# The problem states that the graph is fully joined (connected) and the number of 
# relationships S is the minimum required, which means S = R - 1.
# Thus, the graph is a tree.
# A schema is a subset of nodes that includes all P core nodes and is connected.
# In a tree, the unique minimal connected subgraph containing a set of nodes P 
# is the union of all simple paths between any two nodes in P.
# This is equivalent to the Steiner Tree problem on a tree, which is simply 
# the sum of weights of all edges that lie on the path between any two core nodes.
# An edge lies on the path between two core nodes if and only if removing that 
# edge splits the tree into two components, each containing at least one core node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    core_nodes = set(map(int, input_data[3:3+P]))
    
    adj = [[] for _ in range(R + 1)]
    idx = 3 + P
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # If only one core node, no edges are needed (latency 0)
    if P <= 1:
        print(0)
        return

    # Count how many core nodes are in each subtree
    core_count = [0] * (R + 1)
    for node in core_nodes:
        core_count[node] = 1
        
    total_latency = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    # Process in reverse topological order to propagate core counts
    for i in range(R - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            p, w = parent[u]
            core_count[p] += core_count[u]
            # If this edge separates the tree into two parts both containing core nodes
            if core_count[u] > 0 and core_count[u] < P:
                total_latency += w
                
    print(total_latency)

solve()
```
