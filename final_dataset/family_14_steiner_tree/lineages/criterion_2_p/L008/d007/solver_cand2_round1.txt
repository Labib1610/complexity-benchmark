```python
import sys

# The problem asks for the minimum weight of a connected subgraph that includes 
# all P mandated components. Since the graph is a tree (R nodes, R-1 edges), 
# the unique path between any two nodes is the only way to connect them.
# The union of all paths between any two mandated components forms the 
# minimal connected subgraph containing all mandated components.
# This is equivalent to the Steiner Tree problem on a tree, which is simply 
# the sum of weights of all edges that lie on the path between at least two 
# mandated components. An edge lies on such a path if and only if the 
# subtree formed by removing that edge contains at least one mandated 
# component on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set()
    idx = 3
    for _ in range(P):
        mandated.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # count_mandated[u] will store the number of mandated nodes in the subtree of u
    count_mandated = [0] * (R + 1)
    total_power = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    edge_to_parent_weight = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in mandated:
            count_mandated[u] = 1
            
    # Process in reverse topological order to propagate counts up
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        count_mandated[p] += count_mandated[u]
        # If the edge to parent separates mandated nodes, it must be included
        if count_mandated[u] > 0 and count_mandated[u] < P:
            total_power += edge_to_parent_weight[u]
            
    print(total_power)

solve()
```
