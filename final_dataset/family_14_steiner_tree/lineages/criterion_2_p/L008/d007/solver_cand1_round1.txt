```python
import sys

# The problem states the network is connected and S = R - 1, so it's a tree.
# We need to select a connected subgraph containing all P mandated nodes.
# The cost is the sum of weights of edges where both endpoints are in the subgraph.
# Let the set of mandated nodes be M. The smallest connected subgraph containing M
# is the minimal Steiner Tree, which is the union of all paths between nodes in M.
# Any edge in this minimal Steiner Tree must be included.
# If we include an edge (u, v) with weight w, both u and v are in the cluster.
# The cost is the sum of weights of all edges in the minimal Steiner Tree.
# Any edge not in the minimal Steiner Tree, if included, would increase the cost
# without being necessary to connect the mandated nodes.
# Thus, the minimum cost is the sum of weights of all edges in the minimal Steiner Tree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set(map(int, input_data[3:3+P]))
    
    adj = [[] for _ in range(R + 1)]
    idx = 3 + P
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the minimal Steiner Tree, we can prune the tree.
    # Any leaf node that is not in 'mandated' can be removed.
    # We repeat this until all leaves are in 'mandated'.
    
    degree = [0] * (R + 1)
    for i in range(1, R + 1):
        degree[i] = len(adj[i])
        
    queue = []
    for i in range(1, R + 1):
        if degree[i] == 1 and i not in mandated:
            queue.append(i)
            
    removed = [False] * (R + 1)
    total_cost = 0
    
    # Calculate total weight of all edges first
    all_edges_weight = 0
    for u in range(1, R + 1):
        for v, w in adj[u]:
            if u < v:
                all_edges_weight += w
                
    # Prune edges
    current_removed_weight = 0
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        removed[u] = True
        for v, w in adj[u]:
            if not removed[v]:
                current_removed_weight += w
                degree[v] -= 1
                if degree[v] == 1 and v not in mandated:
                    queue.append(v)
                    
    print(all_edges_weight - current_removed_weight)

solve()
```
