import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    R = int(data[idx]); idx += 1
    S = int(data[idx]); idx += 1
    P = int(data[idx]); idx += 1
    
    mandated = [False] * (R + 1)
    for _ in range(P):
        v = int(data[idx]); idx += 1
        mandated[v] = True
    
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Iterative DFS to build parent, depth, and edge_to_parent
    parent = [0] * (R + 1)
    depth = [0] * (R + 1)
    edge_to_parent = [0] * (R + 1)
    order = []
    visited = [False] * (R + 1)
    stack = [1]
    visited[1] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                depth[v] = depth[u] + 1
                edge_to_parent[v] = w
                stack.append(v)
    
    # Process in reverse order (bottom-up)
    # dp[u] = (cost, has_mandated)
    dp_cost = [0] * (R + 1)
    dp_has = [False] * (R + 1)
    
    for u in reversed(order):
        if mandated[u]:
            dp_has[u] = True
        for v, w in adj[u]:
            if v == parent[u]:
                continue
            if dp_has[v]:
                dp_cost[u] += dp_cost[v] + w
                dp_has[u] = True
    
    # Top-down pass to compute cost if we cut edge to parent
    # cut_cost[u] = cost of the subtree containing u if we cut edge (u, parent[u])
    # We need to find the minimum cut_cost among all nodes
    # cut_cost[root] = dp_cost[root]
    
    cut_cost = [0] * (R + 1)
    cut_cost[1] = dp_cost[1]
    
    for u in order:
        if u == 1:
            continue
        p = parent[u]
        w = edge_to_parent[u]
        # If we cut edge (u, p), the cost is:
        # dp_cost[u] (subtree of u) + (cost from p's side excluding u's subtree)
        # The cost from p's side = cut_cost[p] - (dp_cost[u] + w) if dp_has[u] else cut_cost[p]
        if dp_has[u]:
            cost_from_p = cut_cost[p] - dp_cost[u] - w
        else:
            cost_from_p = cut_cost[p]
        cut_cost[u] = dp_cost[u] + cost_from_p
    
    ans = min(cut_cost[1:])
    print(ans)

solve()
