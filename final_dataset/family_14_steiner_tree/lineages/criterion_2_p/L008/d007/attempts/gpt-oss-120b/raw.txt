```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    R = int(next(it))
    S = int(next(it))
    P = int(next(it))
    required = [0] * (R + 1)
    for _ in range(P):
        required[int(next(it))] = 1

    adj = [[] for _ in range(R + 1)]
    edges = []
    for _ in range(S):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    parent = [0] * (R + 1)
    pw = [0] * (R + 1)          # weight to parent
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        node = stack.pop()
        order.append(node)
        for nb, w in adj[node]:
            if nb == parent[node]:
                continue
            parent[nb] = node
            pw[nb] = w
            stack.append(nb)

    cnt = [0] * (R + 1)
    for v in range(1, R + 1):
        cnt[v] = required[v]

    ans = 0
    for v in reversed(order):
        if v == 1:
            continue
        if 0 < cnt[v] < P:
            ans += pw[v]
        cnt[parent[v]] += cnt[v]

    print(ans)

if __name__ == "__main__":
    solve()
```
