```python
import sys

# The problem asks for the minimum cost to form a connected cluster containing 
# all P mandated components. Since the graph is a tree (R nodes, R-1 edges), 
# the set of nodes forming the smallest connected subgraph containing all P 
# mandated nodes is unique: it is the union of all paths between any two 
# mandated nodes. Any additional nodes included in the cluster would only 
# increase the cost (since all edge weights are positive).
# Thus, the cluster must be exactly the minimal Steiner Tree for the P nodes,
# which in a tree is the unique subtree spanning all P nodes.
# The cost is the sum of weights of all edges that lie on the path between 
# any two mandated nodes. An edge is part of this subtree if and only if 
# removing it splits the tree into two components, each containing at least 
# one of the P mandated nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set()
    idx = 3
    for _ in range(P):
        mandated.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # count[u] will store how many mandated nodes are in the subtree rooted at u
    count = [0] * (R + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order (leaves to root)
    edge_weights = {}
    for u in range(1, R + 1):
        if u in mandated:
            count[u] = 1
            
    for u in reversed(order):
        if u == 1: continue
        p = parent[u]
        # Find weight of edge (u, p)
        w = 0
        for v, weight in adj[u]:
            if v == p:
                w = weight
                break
        
        if count[u] > 0 and count[u] < P:
            total_cost += w
        count[p] += count[u]
        
    print(total_cost)

solve()
```
