```python
import sys

# The problem asks for the minimum cost to connect a set of P mandated modules.
# Since the graph is a tree (S = R - 1), the set of modules that must be 
# included to form a contiguous block containing all P mandated modules is 
# exactly the set of all nodes on the paths between any two mandated modules.
# This is the minimal Steiner Tree in a tree graph.
# The cost is the sum of weights of all edges that lie on the paths between 
# any two mandated modules.
# An edge (u, v) with weight w is part of the Steiner Tree if and only if 
# the subtree formed by removing the edge (u, v) contains at least one 
# mandated module on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set()
    idx = 3
    for _ in range(P):
        mandated.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # count_mandated[u] will store the number of mandated nodes in the subtree of u
    count_mandated = [0] * (R + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    for u in order:
        if u in mandated:
            count_mandated[u] = 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if parent[u] != -1:
            p = parent[u]
            # Find weight of edge (u, p)
            w_edge = 0
            for v, w in adj[u]:
                if v == p:
                    w_edge = w
                    break
            
            # If the subtree at u contains some but not all mandated nodes,
            # the edge (u, p) must be part of the Steiner Tree.
            if 0 < count_mandated[u] < P:
                total_cost += w_edge
            
            count_mandated[p] += count_mandated[u]
            
    print(total_cost)

solve()
```
