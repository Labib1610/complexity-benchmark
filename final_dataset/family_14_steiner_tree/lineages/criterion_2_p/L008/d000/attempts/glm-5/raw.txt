import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    high_priority = [int(next(it)) for _ in range(t)]
    is_priority = [False] * (n + 1)
    for node in high_priority:
        is_priority[node] = True
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    setrecursionlimit(1 << 25)
    
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0 and v != parent[1]:
                parent[v] = u
                parent_weight[v] = w
                stack.append(v)
    
    cnt = [0] * (n + 1)
    for u in reversed(order):
        if is_priority[u]:
            cnt[u] += 1
        p = parent[u]
        if p != -1:
            cnt[p] += cnt[u]
    
    total_cost = 0
    for u in range(2, n + 1):
        if cnt[u] > 0:
            total_cost += parent_weight[u]
    
    print(total_cost)

if __name__ == "__main__":
    solve()
