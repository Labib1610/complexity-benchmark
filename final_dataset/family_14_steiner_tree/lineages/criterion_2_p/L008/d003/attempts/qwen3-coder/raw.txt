```python
import sys
from collections import defaultdict, deque

sys.setrecursionlimit(1 << 25)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    R = int(data[idx])
    S = int(data[idx+1])
    P = int(data[idx+2])
    idx += 3
    
    if P <= 1:
        print(0)
        return
    
    compromised = list(map(int, data[idx:idx+P]))
    idx += P
    
    # Build the tree
    tree = defaultdict(list)
    edges = []
    for _ in range(S):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        idx += 3
        tree[u].append((v, w))
        tree[v].append((u, w))
        edges.append((u, v, w))
    
    # Find the minimal subtree that connects all compromised nodes
    # This is the Steiner tree problem on a tree, which can be solved efficiently
    
    # Since the graph is a tree, we can find the minimal connected subtree containing all compromised nodes
    # by finding the union of paths between all pairs of compromised nodes
    # However, there's a simpler way: find the "subtree" induced by the compromised nodes
    
    # Strategy:
    # 1. Mark all nodes as not visited
    # 2. Do a BFS/DFS from one compromised node to find distances to all other nodes
    # 3. Among all compromised nodes, find the one that minimizes the maximum distance to others
    # Actually, better approach:
    # Since it's a tree, the minimal cost to connect all compromised nodes is the sum of edge weights
    # in the minimal subtree that spans all compromised nodes.
    
    # Efficient approach:
    # Root the tree at some node (say node 1)
    # Compute for each node whether its subtree contains any compromised node
    # Then, include an edge in the result if it lies on the path between two compromised nodes
    
    # Even simpler: since it's a tree, we can do the following:
    # 1. Pick any compromised node as root
    # 2. Do DFS to compute parent and depth
    # 3. For each compromised node, mark the path from it to root
    # 4. The total cost is the sum of weights of marked edges
    
    # Let's implement this properly
    
    # Convert compromised list to set for fast lookup
    comp_set = set(compromised)
    
    # If only one compromised node, cost is 0
    if len(comp_set) <= 1:
        print(0)
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Root the tree at an arbitrary node, say compromised[0]
    root = compromised[0]
    
    # DFS to compute parent relationships and weights to parent
    parent = {node: None for node in range(1, R+1)}
    pw = {node: 0 for node in range(1, R+1)}  # parent weight
    depth = {node: 0 for node in range(1, R+1)}
    
    visited = set()
    stack = [root]
    while stack:
        u = stack.pop()
        if u in visited:
            continue
        visited.add(u)
        for v, w in adj[u]:
            if v not in visited and parent[u] != v:  # Not parent
                parent[v] = u
                pw[v] = w
                depth[v] = depth[u] + 1
                stack.append(v)
    
    # Now, for each compromised node, we need to mark the path from it to root
    # But we should avoid double counting. A better way is to find the union of paths
    # from all compromised nodes to the root and count each edge once.
    
    # We'll use a different approach:
    # For each node, compute if its subtree contains any compromised node
    # An edge should be included if it separates compromised nodes
    
    # Re-rooting approach:
    # Compute for each node how many compromised nodes are in its subtree
    in_subtree = {node: 0 for node in range(1, R+1)}
    
    # First DFS to compute in_subtree counts
    def dfs_count(node, par):
        count = 1 if node in comp_set else 0
        for nei, w in adj[node]:
            if nei == par:
                continue
            count += dfs_count(nei, node)
        in_subtree[node] = count
        return count
    
    dfs_count(root, -1)
    
    # Now traverse the tree again to collect edges that must be included
    total_cost = 0
    
    def dfs_collect(node, par):
        nonlocal total_cost
        for nei, w in adj[node]:
            if nei == par:
                continue
            # Edge from node to nei
            # How many compromised nodes in nei's subtree?
            c_nei = in_subtree[nei]
            # How many outside?
            c_out = len(comp_set) - c_nei
            # If both > 0, then this edge must be included
            if c_nei > 0 and c_out > 0:
                total_cost += w
            dfs_collect(nei, node)
    
    dfs_collect(root, -1)
    print(total_cost)

if __name__ == "__main__":
    main()
```
