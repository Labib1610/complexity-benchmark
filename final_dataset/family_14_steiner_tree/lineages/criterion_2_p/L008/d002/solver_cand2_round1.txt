```python
import sys

# The problem states the network is a tree (N rooms, N-1 segments, fully connected).
# In a tree, there is exactly one unique path between any two nodes.
# To connect a set of high-priority rooms, we must include all edges that lie
# on the paths between any pair of these high-priority rooms.
# This is equivalent to finding the Steiner Tree in a tree, which is simply
# the union of all paths between the high-priority nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    n = int(input_data[idx])
    m = int(input_data[idx+1])
    k = int(input_data[idx+2])
    idx += 3
    
    high_priority = set()
    for _ in range(k):
        high_priority.add(int(input_data[idx]))
        idx += 1
        
    if k <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Root the tree at an arbitrary high-priority node
    root = next(iter(high_priority))
    
    # parent[u] = (p, weight_to_p)
    parent = [None] * (n + 1)
    is_high = [False] * (n + 1)
    for node in high_priority:
        is_high[node] = True
        
    # DFS to build parent pointers and subtree high-priority counts
    order = []
    stack = [root]
    parent[root] = (0, 0)
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] is None:
                parent[v] = (u, w)
                stack.append(v)
                
    # subtree_has_high[u] is true if the subtree at u contains at least one high-priority node
    subtree_has_high = [False] * (n + 1)
    for i in range(1, n + 1):
        if is_high[i]:
            subtree_has_high[i] = True
            
    total_cost = 0
    # Process nodes in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == root:
            continue
        if subtree_has_high[u]:
            p, w = parent[u]
            total_cost += w
            subtree_has_high[p] = True
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
