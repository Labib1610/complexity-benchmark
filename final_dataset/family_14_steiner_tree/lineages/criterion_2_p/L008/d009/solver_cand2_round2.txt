```python
import sys

# The problem asks for the minimum cost to connect a set of P required nodes
# in a tree such that the number of edges is one less than the number of nodes.
# This is equivalent to finding the Steiner Tree of the P nodes in the given tree.
# In a tree, the Steiner Tree is simply the minimal subtree containing all P nodes.
# The cost is the sum of weights of all edges that lie on the paths between any
# two of the P nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    required_nodes = set()
    idx = 3
    for _ in range(P):
        required_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the sum of weights of all edges that are part of the 
    # minimal subtree connecting all required nodes.
    # An edge (u, v) with weight w is in the minimal subtree if and only if
    # the subtree rooted at v (when the edge is cut) contains at least one 
    # required node AND the rest of the tree contains at least one required node.
    
    is_required = [False] * (R + 1)
    for node in required_nodes:
        is_required[node] = True
        
    subtree_has_required = [False] * (R + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    edge_to_parent_weight = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order
    for u in reversed(order):
        if is_required[u]:
            subtree_has_required[u] = True
        
        if subtree_has_required[u] and parent[u] != 0:
            total_cost += edge_to_parent_weight[u]
            subtree_has_required[parent[u]] = True
            
    print(total_cost)

solve()
```
