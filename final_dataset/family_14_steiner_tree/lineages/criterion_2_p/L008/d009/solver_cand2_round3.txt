```python
import sys

# The problem asks for the minimum penalty to activate a set of zones that includes
# all P mandated zones and is "synchronous". A set of zones is synchronous if
# the subgraph induced by these zones is connected.
# Since the original graph is a tree, the smallest connected subgraph containing
# a set of nodes is the minimal subtree spanning those nodes.
# Any set of activated zones that is synchronous and contains the P mandated zones
# must contain the minimal subtree spanning those P zones.
# If we activate exactly the nodes in the minimal subtree spanning the P zones,
# the cost is the sum of weights of all edges in that subtree.
# Any additional node activated would only increase the cost because it would
# require adding an edge to the connected set, and all edge weights are positive.
# Thus, the minimum cost is the sum of weights of all edges in the minimal
# subtree spanning the P mandated zones.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set()
    idx = 3
    for _ in range(P):
        mandated.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # Find the minimal subtree spanning the mandated nodes.
    # An edge is in the minimal subtree if and only if it separates the tree
    # into two components, both of which contain at least one mandated node.
    
    # Root the tree at an arbitrary mandated node (e.g., the first one)
    root = next(iter(mandated))
    
    # Count mandated nodes in each subtree
    count = [0] * (R + 1)
    total_penalty = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    order = []
    stack = [root]
    visited = [False] * (R + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    for u in reversed(order):
        if u in mandated:
            count[u] += 1
        if parent[u] != 0:
            p, w = parent[u]
            count[p] += count[u]
            # If this edge separates mandated nodes, it must be in the subtree
            if count[u] > 0 and count[u] < P:
                total_penalty += w
                
    print(total_penalty)

solve()
```
