```python
import sys

# The problem asks for the minimum cost to form a connected subgraph containing 
# all P mandated assets. Since the graph is a tree (R nodes, R-1 edges), 
# the unique path between any two mandated assets must be included in the 
# subgraph. The union of all paths between all pairs of mandated assets 
# forms the minimal subtree containing all mandated assets.
# The cost is the sum of weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set()
    idx = 3
    for _ in range(P):
        mandated.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # To find the sum of weights of the minimal subtree:
    # Use DFS to find which edges are part of the minimal subtree.
    # An edge (u, v) with weight w is in the subtree if and only if 
    # the subtree rooted at v (when considering u as parent) contains 
    # at least one mandated asset AND the rest of the tree contains 
    # at least one mandated asset.
    
    has_mandated = [False] * (R + 1)
    for m in mandated:
        has_mandated[m] = True
        
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    edge_to_parent_w = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    # Process in reverse topological order to propagate mandated counts
    subtree_has_mandated = [has_mandated[i] for i in range(R + 1)]
    for i in range(R - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if subtree_has_mandated[u]:
            total_cost += edge_to_parent_w[u]
            subtree_has_mandated[p] = True
            
    print(total_cost)

solve()
```
