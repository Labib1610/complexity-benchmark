```python
import sys

# The problem states that the network is a tree (R colonies, R-1 threads, connected).
# We need to select a subset of nodes that includes all P priority colonies and is contiguous.
# A contiguous subset of a tree is a subtree.
# The smallest subtree containing a set of nodes is the Steiner Tree of those nodes.
# In a tree, the Steiner Tree is unique and consists of all paths between any two priority nodes.
# The total nutrient upkeep is the sum of weights of all edges that lie on the path between
# any two priority nodes.
# An edge is part of the Steiner Tree if and only if it separates the tree into two components,
# both of which contain at least one priority node.

def solve():
    # Increase recursion depth for deep trees
    sys.setrecursionlimit(300000)
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    priority_nodes = set()
    idx = 3
    for _ in range(P):
        priority_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count_p[u] will store the number of priority nodes in the subtree rooted at u
    count_p = [0] * (R + 1)
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    edge_to_parent_w = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    for u in order:
        if u in priority_nodes:
            count_p[u] = 1
            
    # Process in reverse topological order (bottom-up)
    for i in range(R - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        p = parent[u]
        count_p[p] += count_p[u]
        # If the subtree at u contains at least one priority node and 
        # the rest of the tree contains at least one priority node,
        # then the edge (u, p) must be part of the Steiner Tree.
        if count_p[u] > 0 and count_p[u] < P:
            total_weight += edge_to_parent_w[u]
            
    print(total_weight)

if __name__ == '__main__':
    solve()
```
