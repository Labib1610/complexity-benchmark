```python
import sys
from collections import deque

# Set recursion limit for DFS
sys.setrecursionlimit(1 << 25)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        R = int(next(iterator))
        S = int(next(iterator))
        P = int(next(iterator))
    except StopIteration:
        return

    mandated = [0] * (R + 1)
    try:
        for _ in range(P):
            u = int(next(iterator))
            mandated[u] = 1
    except StopIteration:
        pass

    adj = [[] for _ in range(R + 1)]
    edges = []
    for _ in range(S):
        u = int(next(iterator))
        v = int(next(iterator))
        w = int(next(iterator))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    # Root the tree. Find a mandated node to be the root if possible.
    root = 1
    for i in range(1, R + 1):
        if mandated[i]:
            root = i
            break

    parent = [0] * (R + 1)
    p_edge = [0] * (R + 1) # weight of edge to parent
    children = [[] for _ in range(R + 1)]
    
    # BFS to build the rooted tree structure
    q = deque([root])
    parent[root] = -1
    order = []
    
    while q:
        u = q.popleft()
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                p_edge[v] = w
                children[u].append(v)
                q.append(v)

    # DP arrays
    # dp0[u]: min cost for subtree u, u is NOT activated
    # dp1[u]: min cost for subtree u, u IS activated
    dp0 = [0] * (R + 1)
    dp1 = [0] * (R + 1)
    
    # Process in reverse order of BFS (bottom-up)
    for u in reversed(order):
        if not children[u]:
            # Leaf node
            if mandated[u]:
                dp0[u] = float('inf')
                dp1[u] = 0
            else:
                dp0[u] = 0
                dp1[u] = 0
        else:
            # Internal node
            # Calculate dp1[u]: u is activated.
            # Children can be activated or not.
            # If child v is activated, add edge cost p_edge[v].
            cost_if_u_active = 0
            for v in children[u]:
                # Cost to include v in the activated set (v active)
                cost_v_active = dp1[v] + p_edge[v]
                # Cost to exclude v from the activated set (v inactive)
                cost_v_inactive = dp0[v]
                
                # If v is mandated, it must be active
                if mandated[v]:
                    cost_v_inactive = float('inf')
                
                cost_if_u_active += min(cost_v_active, cost_v_inactive)
            dp1[u] = cost_if_u_active
            
            # Calculate dp0[u]: u is NOT activated.
            # If u is mandated, this state is impossible.
            if mandated[u]:
                dp0[u] = float('inf')
            else:
                # u is not activated. To maintain connectivity of the activated set,
                # at most one child subtree can contain activated nodes.
                # If multiple children had activated nodes, they would be disconnected
                # because u is not active.
                
                # Base cost: all children subtrees have no activated nodes.
                # This is valid only if no child is mandated.
                base_cost = 0
                possible = True
                for v in children[u]:
                    if mandated[v]:
                        possible = False
                        break
                    base_cost += dp0[v]
                
                min_cost = float('inf')
                if possible:
                    min_cost = base_cost
                
                # Try activating exactly one child subtree v.
                # The cost is: (cost to activate v) + (cost to keep others inactive)
                for v in children[u]:
                    # Cost to activate v
                    cost_activate_v = dp1[v] + p_edge[v]
                    
                    # Cost to keep others inactive
                    cost_others_inactive = 0
                    valid = True
                    for other_v in children[u]:
                        if other_v == v:
                            continue
                        if mandated[other_v]:
                            valid = False
                            break
                        cost_others_inactive += dp0[other_v]
                    
                    if valid:
                        current_total = cost_activate_v + cost_others_inactive
                        if current_total < min_cost:
                            min_cost = current_total
                dp0[u] = min_cost

    # The root must be activated if it's mandated.
    # If not mandated, we can choose to activate it or not.
    # However, the problem implies we must activate a set containing the mandated nodes.
    # If root is not mandated, we can choose dp1[root] or dp0[root].
    # But dp0[root] means root is not activated.
    # If root is not activated, and it has children, only one child can be active.
    # If multiple mandated nodes exist in different subtrees of root, and root is not mandated,
    # dp0[root] would be inf, so dp1[root] would be chosen.
    # If only one child has mandated nodes, dp0[root] might be valid.
    # But we need to activate the mandated nodes. The set of activated nodes is defined by the DP.
    # If we pick dp0[root], the activated set is the one in the single active child.
    # If we pick dp1[root], the activated set includes root.
    # Both are valid sets of activated nodes containing the mandated ones.
    # We want the minimum cost.
    ans = min(dp1[root], dp0[root])
    print(ans)

if __name__ == '__main__':
    solve()
```
