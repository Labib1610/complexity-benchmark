```python
import sys

# The problem states S = R - 1 and the graph is a tree.
# A subset of vertices is internally convertible if and only if it forms a connected subgraph.
# To include all P anchor currencies in a connected subgraph, we must include all vertices
# on the unique paths between any two anchor currencies.
# This is equivalent to the minimal subtree containing all P anchor currencies.
# The total administrative overhead is the sum of weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    anchors = set()
    idx = 3
    for _ in range(P):
        anchors.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the minimal subtree containing all anchors:
    # 1. Root the tree at an arbitrary anchor (e.g., the first one).
    # 2. A node is in the minimal subtree if it is an anchor or if it has
    #    at least two branches (or one branch and itself) leading to anchors.
    # 3. More simply: an edge (u, v) where v is a child of u is in the subtree
    #    if and only if the subtree rooted at v contains at least one anchor.
    
    root = next(iter(anchors))
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    edge_to_parent_w = [0] * (R + 1)
    order = []
    stack = [root]
    visited = [False] * (R + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    has_anchor = [False] * (R + 1)
    for a in anchors:
        has_anchor[a] = True
        
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for i in range(R - 1, 0, -1):
        u = order[i]
        if has_anchor[u]:
            p = parent[u]
            if p != 0:
                total_weight += edge_to_parent_w[u]
                has_anchor[p] = True
                
    print(total_weight)

solve()
```
