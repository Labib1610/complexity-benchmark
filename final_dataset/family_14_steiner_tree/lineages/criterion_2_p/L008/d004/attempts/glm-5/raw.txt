import sys
sys.setrecursionlimit(1 << 25)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    
    try:
        R = int(next(iterator))
        S = int(next(iterator))
        P = int(next(iterator))
    except StopIteration:
        return

    injected = []
    for _ in range(P):
        injected.append(int(next(iterator)))
    
    adj = [[] for _ in range(R + 1)]
    edges = []
    for _ in range(S):
        u = int(next(iterator))
        v = int(next(iterator))
        w = int(next(iterator))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    # If only one signal, it's already converged, cost 0
    if P <= 1:
        print(0)
        return

    # Arbitrary root for LCA, let's use 1
    root = 1
    parent = [0] * (R + 1)
    depth = [0] * (R + 1)
    cost_from_parent = [0] * (R + 1)
    
    # BFS/DFS to setup tree structure
    stack = [root]
    parent[root] = -1
    depth[root] = 0
    
    # Using stack for DFS
    # To handle large R, iterative is better
    # We need to process nodes, let's use a simple stack approach
    # But we need to make sure we don't overflow recursion, so iterative
    stack = [root]
    order = []
    parent[root] = 0 # 0 indicates no parent
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                depth[v] = depth[u] + 1
                cost_from_parent[v] = w
                stack.append(v)
                
    # Binary Lifting Table
    LOG = (R).bit_length()
    up = [[0] * (R + 1) for _ in range(LOG)]
    up[0] = parent[:] # up[0][v] is parent of v
    
    for k in range(1, LOG):
        for v in range(1, R + 1):
            up[k][v] = up[k-1][up[k-1][v]]

    def lca(u, v):
        if depth[u] < depth[v]:
            u, v = v, u
        
        diff = depth[u] - depth[v]
        for k in range(LOG):
            if (diff >> k) & 1:
                u = up[k][u]
        
        if u == v:
            return u
        
        for k in range(LOG - 1, -1, -1):
            if up[k][u] != up[k][v]:
                u = up[k][u]
                v = up[k][v]
        
        return up[0][u]

    # Find LCA of all injected nodes
    current_lca = injected[0]
    for i in range(1, P):
        current_lca = lca(current_lca, injected[i])
    
    # Mark paths from each injected node to the LCA
    # We use a difference array technique on the tree edges
    # cnt[v] counts how many paths use the edge (parent[v], v)
    cnt = [0] * (R + 1)
    
    for node in injected:
        # Increment path from node to current_lca
        # This is equivalent to +1 at node, -1 at current_lca
        # When we sum up the tree, the sum at node v represents the count for edge (parent[v], v)
        # But we must be careful: the path stops at LCA.
        # Standard trick: cnt[node]++, cnt[lca]--
        cnt[node] += 1
        cnt[current_lca] -= 1
        
    # Propagate counts up the tree
    # Process nodes in reverse order of visitation (bottom-up)
    # The 'order' list from DFS has parent before children.
    # So we iterate in reverse.
    for u in reversed(order):
        if u == root:
            continue
        p = parent[u]
        cnt[p] += cnt[u]
        
    total_cost = 0
    # For each edge (parent[v], v), if cnt[v] == P, we must include it
    # The cost is cost_from_parent[v]
    for v in range(2, R + 1):
        if cnt[v] == P:
            total_cost += cost_from_parent[v]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
