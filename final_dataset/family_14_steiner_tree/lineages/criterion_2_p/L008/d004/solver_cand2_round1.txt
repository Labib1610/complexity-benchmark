import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    non_negotiable = set()
    idx = 3
    for _ in range(P):
        non_negotiable.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # We need to find the minimal subtree that connects all non-negotiable nodes.
    # This is equivalent to the sum of weights of all edges that lie on the path
    # between any two non-negotiable nodes.
    # An edge is part of this subtree if and only if the subtree formed by 
    # removing the edge splits the non-negotiable nodes into two non-empty sets.
    
    is_non_negotiable = [False] * (R + 1)
    for node in non_negotiable:
        is_non_negotiable[node] = True
        
    count_non_negotiable = [0] * (R + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process nodes in reverse topological order to count non-negotiable nodes in subtrees
    for u in reversed(order):
        count_non_negotiable[u] += (1 if is_non_negotiable[u] else 0)
        if parent[u] != -1:
            count_non_negotiable[parent[u]] += count_non_negotiable[u]
            
    # Calculate total cost
    # An edge (u, parent[u]) is included if count_non_negotiable[u] > 0 
    # and count_non_negotiable[u] < P
    for u in range(2, R + 1):
        c = count_non_negotiable[u]
        if 0 < c < P:
            # Find weight of edge (u, parent[u])
            for v, w in adj[u]:
                if v == parent[u]:
                    total_cost += w
                    break
                    
    print(total_cost)

solve()
