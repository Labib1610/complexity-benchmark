```python
import sys

# The problem describes a tree structure where activating a set of nodes
# causes all nodes on the paths between them to be activated.
# This means the final active state is the smallest subtree containing all
# mandated nodes.
# The cost is the sum of weights of all edges (u, v) where both u and v
# are in this subtree.
# Let T be the set of nodes in the smallest subtree containing all mandated nodes.
# An edge (u, v) with weight w is saturated if both u and v are in T.
# This is equivalent to saying the edge (u, v) is NOT saturated if at least
# one of the components formed by removing (u, v) contains no mandated nodes.
# Total cost = (Sum of all weights) - (Sum of weights of edges that are NOT saturated).
# An edge (u, v) is NOT saturated if removing it splits the tree into two
# components, one of which contains zero mandated nodes.
# Let P be the set of mandated nodes.
# For an edge e, let size(e) be the number of mandated nodes in one of the
# components formed by removing e. The edge is NOT saturated if size(e) == 0
# or size(e) == P_total.
# Since the edge is part of the tree, size(e) == 0 means the component
# contains no mandated nodes.
# Thus, we sum the weights of all edges such that the component on one side
# has 0 mandated nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set()
    idx = 3
    for _ in range(P):
        mandated.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    total_weight = 0
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        total_weight += w
        idx += 3
        
    # Count mandated nodes in subtree
    subtree_mandated = [0] * (R + 1)
    parent = [0] * (R + 1)
    edge_to_parent_weight = [0] * (R + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in mandated:
            subtree_mandated[u] += 1
        if parent[u] != -1:
            subtree_mandated[parent[u]] += subtree_mandated[u]
            
    # An edge (u, parent[u]) is NOT saturated if subtree_mandated[u] == 0
    # or subtree_mandated[u] == P.
    not_saturated_weight = 0
    for i in range(2, R + 1):
        m = subtree_mandated[i]
        if m == 0 or m == P:
            not_saturated_weight += edge_to_parent_weight[i]
            
    print(total_weight - not_saturated_weight)

solve()
```
