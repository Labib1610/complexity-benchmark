```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    R = int(data[idx]); idx += 1
    S = int(data[idx]); idx += 1
    P = int(data[idx]); idx += 1
    
    injected = [False] * (R + 1)
    for _ in range(P):
        node = int(data[idx]); idx += 1
        injected[node] = True
    
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    parent = [0] * (R + 1)
    parent_weight = [0] * (R + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                parent_weight[v] = w
                stack.append(v)
    
    dp = [0] * (R + 1)
    cnt = [0] * (R + 1)
    
    for u in reversed(order):
        if injected[u]:
            cnt[u] = 1
        for v, w in adj[u]:
            if parent[v] == u:
                if cnt[v] > 0:
                    dp[u] += dp[v] + w
                    cnt[u] += cnt[v]
                else:
                    dp[u] += dp[v]
    
    print(dp[1])

solve()
```
