```python
import sys

# The problem asks for the minimum cost to connect all critical variables in a tree.
# The generators x_u - x_v form a tree structure because there are n-1 generators
# and they are minimally sufficient to generate the ideal of the main diagonal.
# The condition x_{c_x} - x_{c_y} \in I means that in the graph where edges are
# the chosen generators, all critical variables must be in the same connected component.
# This is equivalent to finding the minimum weight subtree that spans all critical nodes.
# This is the Steiner Tree problem on a tree, which is simply the sum of weights
# of all edges that lie on the unique paths between any two critical nodes.
# An edge (u, v) is part of the Steiner tree if and only if the removal of this edge
# splits the tree into two components, each containing at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_critical_count[u] stores how many critical nodes are in the subtree
    # rooted at u (when rooted at 1).
    subtree_critical_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in critical_nodes:
        subtree_critical_count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == 1:
            continue
        if subtree_critical_count[u] > 0 and subtree_critical_count[u] < t:
            total_cost += parent_edge_weight[u]
        subtree_critical_count[parent[u]] += subtree_critical_count[u]
        
    print(total_cost)

solve()
```
