```python
import sys

# The problem states that the graph is a tree (n nodes, n-1 edges, connected).
# We need to find the minimum cost to connect a subset of t core concepts.
# In a tree, the set of edges that connect a subset of nodes is unique:
# it is the union of all paths between any two core nodes.
# This is equivalent to the edges on the minimal subtree containing all core nodes.
# We can solve this by rooting the tree at an arbitrary node (e.g., the first core node)
# and calculating the sum of weights of all edges that have at least one core node
# in their subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Use iterative DFS to avoid recursion depth issues
    root = next(iter(core_nodes))
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    has_core_in_subtree = [False] * (n + 1)
    for node in core_nodes:
        has_core_in_subtree[node] = True
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        if has_core_in_subtree[u]:
            total_cost += edge_to_parent_weight[u]
            has_core_in_subtree[parent[u]] = True
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
