```python
import sys

# The problem states the station is a tree (n modules, n-1 corridors).
# We need to find the minimum weight of a subtree that contains all t vital modules.
# This is equivalent to the sum of weights of all edges that lie on the paths
# between any two vital modules.
# An edge is part of this set if and only if the subtree formed by removing
# that edge splits the vital modules into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    vital_modules = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many vital modules are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    vital_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in order:
        if u in vital_modules:
            vital_count[u] = 1
            
    total_power = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            vital_count[p] += vital_count[u]
            # If this edge separates vital modules, it must be pressurized
            if 0 < vital_count[u] < t:
                total_power += edge_weight_to_parent[u]
                
    print(total_power)

solve()
```
