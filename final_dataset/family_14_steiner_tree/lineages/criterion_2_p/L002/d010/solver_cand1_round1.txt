```python
import sys

# The problem asks for the minimum cost to connect all critical variables in a graph
# where each edge (u, v) with weight w represents the equation x_u - x_v = 0.
# Since m = n - 1 and the full set of equations has full row rank, the graph
# formed by all m edges is a tree.
# We need to select a subset of edges such that all critical variables are in the
# same connected component.
# In a tree, the unique path between any two nodes is the only way to connect them.
# To connect a set of critical nodes in a tree, we must select all edges that lie
# on the paths between these nodes.
# This is equivalent to finding the minimal subtree containing all critical nodes.
# The cost is the sum of weights of all edges that are part of the minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_vars = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Root the tree at an arbitrary critical node
    root = next(iter(critical_vars))
    
    # We need to identify edges that are on the path between any two critical nodes.
    # An edge (u, v) is in the minimal subtree if and only if the subtree rooted at
    # the child node (say v) contains at least one critical node and the rest of
    # the tree (the part not in v's subtree) also contains at least one critical node.
    
    has_critical = [False] * (n + 1)
    for node in critical_vars:
        has_critical[node] = True
        
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to propagate critical node counts
    for u in reversed(order):
        if u == root:
            continue
        p = parent[u]
        if has_critical[u]:
            total_cost += edge_weight[u]
            has_critical[p] = True
            
    print(total_cost)

solve()
```
