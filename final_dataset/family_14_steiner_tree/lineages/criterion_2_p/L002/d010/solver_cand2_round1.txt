```python
import sys

# The problem states we have n variables and m = n-1 equations.
# Each equation x_u - x_v = 0 implies x_u = x_v.
# This is equivalent to an edge (u, v) in a graph with n nodes.
# Activating a set of equations means selecting a subset of edges.
# The solution space V is defined by the connected components of the graph formed by selected edges.
# If two variables are in the same component, they must be equal.
# The dimension of the solution space V is the number of connected components.
# The projection pi(V) onto t critical variables has dimension 1 if and only if
# all t critical variables belong to the same connected component.
# We want to select a subset of edges such that all t critical variables are in the same component,
# minimizing the sum of weights of the selected edges.
# Since the full set of m = n-1 edges forms a tree (as they are linearly independent and n-1),
# we are looking for a subtree that connects all t critical variables.
# The minimum cost to connect a set of nodes in a tree is the sum of weights of all edges
# that lie on the unique paths between any two critical variables.
# This is equivalent to the sum of weights of all edges e such that removing e
# splits the tree into two components, each containing at least one critical variable.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = set()
    idx = 3
    for _ in range(t):
        critical.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_critical[u] = number of critical variables in the subtree rooted at u
    subtree_critical = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in critical:
            subtree_critical[u] += 1
        if parent[u] != 0:
            subtree_critical[parent[u]] += subtree_critical[u]
            # If this edge separates the tree into two parts both containing critical nodes
            if 0 < subtree_critical[u] < t:
                total_cost += edge_weight[u]
                
    print(total_cost)

solve()
```
