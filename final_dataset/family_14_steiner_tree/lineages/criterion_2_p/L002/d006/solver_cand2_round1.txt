```python
import sys

# The problem asks for the minimum cost to lock a subset of edges such that
# if any single unlocked edge (u, v) faults, the baseline sensors remain
# connected in the graph of locked edges.
#
# Let the tree be T. If we remove an edge e, T splits into two components,
# T1 and T2. If any baseline sensor is in T1 and another is in T2, then
# the fault in e will cause a discrepancy between the baseline sensors.
# Thus, for every edge e, if it separates the set of baseline sensors into
# two non-empty sets, e MUST be locked.
#
# If an edge e does not separate the baseline sensors (i.e., all baseline
# sensors are in T1 or all are in T2), then a fault in e does not affect
# the relative readings of the baseline sensors.
#
# Therefore, we must lock all edges that lie on the unique path between any
# two baseline sensors. This is equivalent to the union of all paths between
# all pairs of baseline sensors, which is the minimal subtree containing all
# baseline sensors (the Steiner Tree of the baseline sensors in the given tree).
#
# Since the graph is already a tree, the Steiner Tree is simply the unique
# path connecting all baseline sensors. Any edge on this path must be locked.
# Any edge not on this path does not need to be locked.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    baseline = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Find the minimal subtree containing all baseline sensors.
    # We can do this by rooting the tree at baseline[0] and marking
    # nodes that are in the subtree of a baseline sensor.
    
    is_baseline = [False] * (n + 1)
    for b in baseline:
        is_baseline[b] = True
        
    has_baseline = [False] * (n + 1)
    parent_edge_cost = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [baseline[0]]
    visited = [False] * (n + 1)
    visited[baseline[0]] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_cost[v] = w
                stack.append(v)
                
    # Post-order traversal to mark nodes
    total_cost = 0
    for u in reversed(order):
        if is_baseline[u]:
            has_baseline[u] = True
        if has_baseline[u] and u != baseline[0]:
            total_cost += parent_edge_cost[u]
            has_baseline[parent[u]] = True
            
    print(total_cost)

solve()
```
