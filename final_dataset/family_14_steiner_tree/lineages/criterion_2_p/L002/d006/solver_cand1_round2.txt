```python
import sys

# The problem asks for the minimum cost to select a subset of edges S such that
# there exists a root r from which all target vertices are reachable via S.
# In a tree, the set of edges forming paths from r to all target vertices is
# exactly the union of all paths from r to each target vertex.
# This union is the minimal subtree containing r and all target vertices.
# The cost of this subtree is the sum of weights of all edges on the paths
# from r to each target vertex.
# Let T be the set of target vertices. We want to find r that minimizes
# the sum of weights of edges in the minimal subtree containing {r} U T.
# Let f(r) be the sum of weights of edges in the minimal subtree containing {r} U T.
# If r is a target vertex, f(r) is the sum of weights of all edges in the
# minimal subtree containing all target vertices (the Steiner Tree of T).
# If r is not a target vertex, let r' be the vertex in the Steiner Tree of T
# closest to r. Then the minimal subtree containing {r} U T is the Steiner Tree
# of T plus the path from r' to r.
# Thus, f(r) = Cost(Steiner Tree of T) + dist(r, Steiner Tree of T).
# To minimize f(r), we should pick r to be any vertex in the Steiner Tree of T,
# which makes dist(r, Steiner Tree of T) = 0.
# Therefore, the answer is simply the sum of weights of all edges that lie on
# any path between two target vertices.
# An edge lies on a path between two target vertices if and only if removing
# the edge splits the tree into two components, both of which contain at least
# one target vertex.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count targets in subtree
    subtree_targets = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            subtree_targets[u] += 1
        if parent[u] != 0:
            subtree_targets[parent[u]] += subtree_targets[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the edge to parent splits the tree such that both sides have targets
        if subtree_targets[i] > 0 and subtree_targets[i] < t:
            total_cost += edge_to_parent_w[i]
            
    print(total_cost)

solve()
```
