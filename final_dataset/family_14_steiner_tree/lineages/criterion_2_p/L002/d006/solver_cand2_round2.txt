```python
import sys

# The problem asks for the minimum cost to connect all focal nodes in a tree.
# The constraints form a tree structure (n nodes, n-1 edges).
# We need to select a subset of edges such that all focal nodes are in the same connected component.
# This is equivalent to finding the minimum cost to connect a set of nodes in a tree.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of nodes {f1, f2, ..., ft}, we must include all edges that lie on the paths
# between these nodes.
# An edge (u, v) with weight w is part of the required set if and only if the removal of
# this edge splits the tree into two components, each containing at least one focal node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    focal_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # count_focal[u] will store the number of focal nodes in the subtree rooted at u
    count_focal = [0] * (n + 1)
    
    # DFS to compute subtree focal counts
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in order:
        if u in focal_nodes:
            count_focal[u] = 1
            
    total_cost = 0
    # Process in reverse topological order
    for u in reversed(order):
        if parent[u] != 0:
            p = parent[u]
            count_focal[p] += count_focal[u]
            
            # Find weight of edge (u, p)
            # Since we need the weight, we can store it in the adjacency list or a map
            # Let's find the weight of edge (u, p)
            for v, w in adj[u]:
                if v == p:
                    # If this edge separates the tree into two parts, both having focal nodes
                    if count_focal[u] > 0 and count_focal[u] < t:
                        total_cost += w
                    break
                    
    print(total_cost)

solve()
```
