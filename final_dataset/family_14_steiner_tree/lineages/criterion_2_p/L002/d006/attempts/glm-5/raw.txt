```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return
    
    focal = set()
    for _ in range(t):
        try:
            focal.add(int(next(iterator)))
        except StopIteration:
            return
            
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            w = int(next(iterator))
            adj[u].append((v, w))
            adj[v].append((u, w))
        except StopIteration:
            return
            
    # Prim's algorithm for MST
    # Start from an arbitrary focal node
    start_node = next(iter(focal))
    
    in_mst = [False] * (n + 1)
    min_cost = [float('inf')] * (n + 1)
    min_cost[start_node] = 0
    total_cost = 0
    
    # Priority queue stores (weight, node)
    pq = [(0, start_node)]
    
    while pq:
        w, u = heapq.heappop(pq)
        
        if in_mst[u]:
            continue
        in_mst[u] = True
        total_cost += w
        
        for v, weight in adj[u]:
            if not in_mst[v] and weight < min_cost[v]:
                min_cost[v] = weight
                heapq.heappush(pq, (weight, v))
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
