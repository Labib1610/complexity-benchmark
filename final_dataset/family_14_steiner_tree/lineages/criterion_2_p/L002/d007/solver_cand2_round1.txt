```python
import sys

# The condition that (x - y_u)^2 + (x - y_v)^2 has a real root is equivalent to
# 2x^2 - 2(y_u + y_v)x + (y_u^2 + y_v^2) = 0 having a real root.
# The discriminant is D = 4(y_u + y_v)^2 - 8(y_u^2 + y_v^2) = 4(y_u^2 + 2y_u y_v + y_v^2 - 2y_u^2 - 2y_v^2)
# D = 4(-y_u^2 + 2y_u y_v - y_v^2) = -4(y_u - y_v)^2.
# For D >= 0, we must have (y_u - y_v)^2 <= 0, which implies y_u = y_v.
# Thus, activating a constraint (u, v) forces y_u = y_v.
# The focal polynomial Q(x) = sum_{k in T} (x - y_k)^2 has a real root if and only if
# its discriminant D_Q >= 0.
# Q(x) = |T|x^2 - 2(sum y_k)x + sum y_k^2.
# D_Q = 4(sum y_k)^2 - 4|T|(sum y_k^2) >= 0
# (sum y_k)^2 >= |T| * sum y_k^2.
# By Cauchy-Schwarz, (sum y_k)^2 <= |T| * sum y_k^2.
# Equality holds if and only if y_k are all equal for all k in T.
# So we need to force y_i = y_j for all i, j in T.
# This is equivalent to connecting all nodes in T in the graph formed by the constraints.
# This is a Steiner Tree problem on a tree, which is simply the sum of weights of edges
# on the unique paths between all nodes in T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, t = map(int, input_data[:3])
    focal = list(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u, v, w = map(int, input_data[idx:idx+3])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3

    if t <= 1:
        print(0)
        return

    # Find the minimal subtree containing all focal nodes.
    # We can use DFS to mark nodes that are part of the minimal subtree.
    is_focal = [False] * (n + 1)
    for f in focal:
        is_focal[f] = True
    
    subtree_nodes = [False] * (n + 1)
    
    def dfs(u, p):
        count = 1 if is_focal[u] else 0
        total_w = 0
        for v, w in adj[u]:
            if v == p:
                continue
            c, sub_w = dfs(v, u)
            if c > 0:
                total_w += sub_w + w
                count += c
        return count, total_w

    # Root the tree at the first focal node
    _, result = dfs(focal[0], -1)
    print(result)

solve()
```
