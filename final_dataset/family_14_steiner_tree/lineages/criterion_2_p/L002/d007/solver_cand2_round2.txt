```python
import sys

# The problem asks for the minimum cost to connect all target positions into a single
# connected component. Since the graph is a tree (n nodes, n-1 edges, connected),
# the set of edges that connects a subset of nodes is the union of paths between
# them in the tree. Specifically, we need to find the minimum cost to connect
# all target nodes. This is equivalent to finding the Steiner Tree for the
# target nodes in the given tree. In a tree, the Steiner Tree is simply the
# minimal subtree containing all target nodes.
# The edges in this subtree are exactly those edges that lie on the path between
# at least one pair of target nodes. An edge (u, v) is in the Steiner Tree if
# and only if the two components formed by removing (u, v) both contain at
# least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_targets[u] will store the number of target nodes in the subtree
    # rooted at u (when rooted at 1).
    subtree_targets = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in targets:
        subtree_targets[u] = 1
        
    total_cost = 0
    # Process in reverse topological order to propagate target counts
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if subtree_targets[u] > 0 and subtree_targets[u] < t:
            total_cost += parent_edge_weight[u]
        subtree_targets[p] += subtree_targets[u]
        
    print(total_cost)

solve()
```
