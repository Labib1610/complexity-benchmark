```python
import sys

# The problem asks for the minimum cost to ensure that for all x in the null space of B,
# x_i = x_j for all target indices i, j.
# A vector x is in the null space of B if for every selected row vector (u, v) with cost w,
# x_u - x_v = 0, i.e., x_u = x_v.
# This means x_i = x_j if and only if there is a path between i and j in the graph
# formed by the selected edges (u, v).
# To make all target indices have the same value, all target indices must be in the
# same connected component in the graph formed by the selected edges.
# Since we have m = n - 1 edges and the full graph is connected (null space is span of 1s),
# the graph is a tree.
# We need to select a subset of edges such that all target indices are in the same
# connected component. This is equivalent to finding the minimum cost to connect
# all target nodes in the tree.
# This is the Steiner Tree problem on a tree, which is equivalent to finding the
# sum of weights of all edges that lie on the unique path between any two target nodes.
# An edge lies on the path between two target nodes if and only if removing that edge
# splits the tree into two components, each containing at least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_targets[u] = number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    
    # DFS to count targets in subtrees
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            subtree_targets[u] += 1
        if parent[u] != 0:
            subtree_targets[parent[u]] += subtree_targets[u]
            
    total_cost = 0
    for u in range(2, n + 1):
        # If the edge (u, parent[u]) separates the tree into two parts,
        # and both parts contain at least one target, the edge must be included.
        count = subtree_targets[u]
        if count > 0 and count < t:
            total_cost += edge_weight[u]
            
    print(total_cost)

solve()
```
