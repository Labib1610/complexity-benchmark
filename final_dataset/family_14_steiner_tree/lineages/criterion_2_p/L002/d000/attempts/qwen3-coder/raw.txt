```python
import sys
from collections import defaultdict
import heapq

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    vital = set(map(int, sys.stdin.readline().split()))
    
    # If there's only one vital module, no power is needed
    if t <= 1:
        print(0)
        return
    
    # Build the graph
    graph = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
        edges.append((w, u, v))
    
    # If there are no edges, this shouldn't happen per constraints, but just in case
    if not edges:
        print(0)
        return
    
    # Find the Steiner tree: minimal cost to connect all vital nodes
    # We use a modified Kruskal's algorithm on the subtree that spans the vital nodes
    
    # First, we need to find the "important" edges that form the Steiner tree
    # A classic approach is to build a complete graph on vital nodes where edge weights 
    # are shortest path distances, then find MST of that. But that's too slow for large inputs.
    
    # Better approach: Use the fact that we're dealing with a tree.
    # In a tree, the minimal subgraph connecting a set of nodes is also a tree,
    # and its edges are exactly those edges in the original tree that lie on paths
    # between pairs of vital nodes.
    
    # Efficient way:
    # 1. Root the tree at some node
    # 2. For each vital node, mark the path to root
    # 3. Count how many times each edge is used across all vital nodes' paths to root
    # 4. An edge should be included if it lies on the path between at least two vital nodes
    
    # Actually, let's implement a simpler and correct approach:
    # Since the input is a tree, we can:
    # 1. Build the tree
    # 2. Find the minimal subtree that connects all vital nodes (the Steiner tree on a tree)
    # This is done by:
    # - Rooting the tree
    # - Finding the lowest common ancestor (LCA) structure
    # - Marking all nodes that are on paths between vital nodes
    
    # Simpler yet: 
    # Run a multi-source BFS from all vital nodes to compute for each node
    # how many vital nodes are in its subtree.
    # Then, an edge {u,v} is in the Steiner tree if removing it separates the vital nodes into two non-empty groups.
    # That happens when the number of vital nodes in one side of the edge is > 0 and < t.
    
    # Let's build adjacency list for tree processing
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Compute subtree sizes for a rooted tree
    # Let's root at node 1
    parent = [-1] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    # BFS to build parent relationships
    queue = [1]
    visited[1] = True
    bfs_order = []
    
    while queue:
        u = queue.pop()
        bfs_order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                queue.append(v)
    
    # Now count how many vital nodes are in each subtree
    vital_count = [0] * (n + 1)
    for v in vital:
        vital_count[v] = 1
    
    # Propagate counts up the tree
    for u in reversed(bfs_order):
        if parent[u] != -1:
            vital_count[parent[u]] += vital_count[u]
    
    # Now sum up the weights of edges that must be included
    total_power = 0
    for u in range(1, n + 1):
        if parent[u] != -1:
            # Edge from u to parent[u] with weight parent_edge_weight[u]
            # This edge is needed if 0 < vital_count[u] < t
            # Because removing it separates the tree into two parts:
            # one containing 'vital_count[u]' vital nodes, another containing 't - vital_count[u]'
            if 0 < vital_count[u] < t:
                total_power += parent_edge_weight[u]
    
    print(total_power)

if __name__ == "__main__":
    main()
```
