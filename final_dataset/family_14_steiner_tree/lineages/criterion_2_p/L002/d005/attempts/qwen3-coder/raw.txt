import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal(n, edges, must_connect):
    if len(must_connect) <= 1:
        return 0
    
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # First, connect all must_connect elements with 0 cost edges
    components = {}
    for node in must_connect:
        components[node] = node
    
    cost = 0
    new_edges = []
    
    # Process edges that connect two must_connect elements
    for u, v, w in edges:
        if u in must_connect and v in must_connect:
            # Try to connect in the main union-find
            if union(parent, rank, u, v):
                # Also track in components
                pu, pv = find(parent, u), find(parent, v)
                # Connect components with zero cost
                root_u = components.get(u, u)
                root_v = components.get(v, v)
                if root_u != root_v:
                    # Merge components
                    for node in must_connect:
                        if components.get(node, node) == root_v:
                            components[node] = root_u
        else:
            new_edges.append((w, u, v))
    
    # Check if all must_connect elements are connected
    roots = set()
    for node in must_connect:
        roots.add(find(parent, node))
    
    if len(roots) == 1:
        return 0
    
    # Now run Kruskal on the remaining edges
    new_edges.sort()
    
    for w, u, v in new_edges:
        if union(parent, rank, u, v):
            cost += w
            # Check if all must_connect elements are now connected
            roots = set()
            for node in must_connect:
                roots.add(find(parent, node))
            if len(roots) == 1:
                return cost
    
    return cost
