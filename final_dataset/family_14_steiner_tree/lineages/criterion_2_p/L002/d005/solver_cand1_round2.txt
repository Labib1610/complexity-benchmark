```python
import sys

# The problem asks for the minimum cost to connect a set of target elements
# into the same orbit using a subset of given transpositions.
# A transposition (u, v) connects u and v in the same orbit.
# If we select a set of transpositions, two elements are in the same orbit
# if there is a path between them in the graph where edges are the selected transpositions.
# Thus, the problem is equivalent to finding the minimum cost to connect all
# target elements in a graph where edges are the given transpositions.
# This is the Steiner Tree problem, but since we only need to connect a set of
# target nodes in a graph where the edges are given by the transpositions,
# and the graph is a tree (m = n-1 and it generates S_n, so it must be a tree),
# the minimum cost is simply the sum of weights of all edges on the unique
# paths between the target nodes in the tree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Root the tree at targets[0]
    root = targets[0]
    parent = [0] * (n + 1)
    edge_to_parent = [0] * (n + 1)
    depth = [0] * (n + 1)
    
    stack = [(root, 0, 0, 0)]
    while stack:
        u, p, d, w = stack.pop()
        parent[u] = p
        depth[u] = d
        edge_to_parent[u] = w
        for v, weight in adj[u]:
            if v != p:
                stack.append((v, u, d + 1, weight))
                
    # Mark nodes that are on the path between any two targets
    # A node is on the path if it is an ancestor of some target
    # and a descendant of the LCA of all targets.
    # Actually, simpler: mark all targets, then mark all nodes on the path
    # to the root. The union of these paths forms the Steiner Tree.
    
    in_steiner = [False] * (n + 1)
    for target in targets:
        curr = target
        while curr != 0 and not in_steiner[curr]:
            in_steiner[curr] = True
            curr = parent[curr]
            
    total_cost = 0
    for i in range(1, n + 1):
        if in_steiner[i] and i != root:
            total_cost += edge_to_parent[i]
            
    print(total_cost)

solve()
```
