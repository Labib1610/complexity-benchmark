import sys
from collections import defaultdict
import heapq

def find_mst_with_pseudo_critical_edges(n, edges, critical_nodes):
    if len(critical_nodes) <= 1:
        return 0
    
    # Convert critical_nodes to a set for faster lookup
    critical_set = set(critical_nodes)
    
    # Build adjacency list
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Find a spanning tree that connects all critical nodes
    # We'll use a modified Prim's algorithm
    
    # Start with any critical node
    start_node = critical_nodes[0]
    visited = set()
    min_heap = [(0, start_node, -1)]  # (weight, node, parent)
    total_weight = 0
    connected_critical = set()
    
    parent = {}
    
    while min_heap and len(connected_critical) < len(critical_set):
        weight, node, par = heapq.heappop(min_heap)
        
        if node in visited:
            continue
            
        visited.add(node)
        parent[node] = par
        
        if node in critical_set:
            connected_critical.add(node)
            total_weight += weight
        
        # Add neighbors to heap
        for neighbor, edge_weight in graph[node]:
            if neighbor not in visited:
                heapq.heappush(min_heap, (edge_weight, neighbor, node))
    
    return total_weight

def find_mst_subgraph(n, edges, critical_nodes):
    if len(critical_nodes) <= 1:
        return 0
    
    critical_set = set(critical_nodes)
    
    # Use Union-Find for Kruskal's algorithm
    class UnionFind:
        def __init__(self, size):
            self.parent = list(range(size))
            self.rank = [0] * size
        
        def find(self, x):
            if self.parent[x] != x:
                self.parent[x] = self.find(self.parent[x])
            return self.parent[x]
        
        def union(self, x, y):
            px, py = self.find(x), self.find(y)
            if px == py:
                return False
            if self.rank[px] < self.rank[py]:
                px, py = py, px
            self.parent[py] = px
            if self.rank[px] == self.rank[py]:
                self.rank[px] += 1
            return True
    
    # Sort edges by weight
    edges.sort(key=lambda x: x[2])
    
    uf = UnionFind(n + 1)
    total_weight = 0
    connected_critical = set()
    
    for u, v, w in edges:
        if uf.union(u, v):
            total_weight += w
            if u in critical_set:
                connected_critical.add(u)
            if v in critical_set:
                connected_critical.add(v)
            
            if len(connected_critical) == len(critical_set):
                break
    
    return total_weight

# Read input
line1 = list(map(int, sys.stdin.readline().split()))
n, m, t = line1[0], line1[1], line1[2]

edges = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    edges.append((u, v, w))

# Solve using Kruskal's algorithm variant
result = find_mst_subgraph(n, edges, critical_nodes)
print(result)
