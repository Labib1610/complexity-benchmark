```python
import sys

# The problem describes a tree structure (n modules, n-1 corridors).
# We need to find the minimum power to connect all vital modules.
# In a tree, there is exactly one simple path between any two nodes.
# To connect a set of vital modules, we must include all edges that lie
# on the paths between any two vital modules.
# This is equivalent to the union of paths between all pairs of vital modules.
# An edge is part of this union if and only if the subtree formed by removing
# that edge contains at least one vital module on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    vital_modules = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to count how many vital modules are in the subtree of each node.
    # Since it's a tree, we can root it at 1.
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    vital_count = [0] * (n + 1)
    for v in vital_modules:
        vital_count[v] = 1
        
    total_power = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if vital_count[u] > 0 and vital_count[u] < t:
            total_power += edge_weight_to_parent[u]
        vital_count[p] += vital_count[u]
        
    print(total_power)

solve()
```
