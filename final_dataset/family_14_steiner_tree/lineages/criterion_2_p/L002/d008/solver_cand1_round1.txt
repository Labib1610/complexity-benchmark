```python
import sys

# The problem asks for the minimum cost to force x_i = x_j for all target indices i, j.
# A condition (u, v) with cost w implies x_u = x_v.
# This is equivalent to finding a Minimum Spanning Tree in a graph where nodes are 1..n
# and edges are the given conditions.
# If we contract all target indices into a single super-node, the condition x_i = x_j
# for all target indices i, j is satisfied if all target indices are in the same
# connected component.
# We can use Kruskal's algorithm or Prim's algorithm.
# Since m = n - 1, the graph is a tree.
# We want to find the minimum cost to connect all target nodes.
# This is the Steiner Tree problem, but since the graph is a tree, it simplifies.
# In a tree, there is a unique path between any two nodes.
# To connect a set of nodes in a tree, we must include all edges that lie on the
# unique paths between any pair of target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Root the tree at an arbitrary target node
    root = next(iter(targets))
    
    # has_target[u] is true if the subtree at u contains at least one target node
    has_target = [False] * (n + 1)
    for node in targets:
        has_target[node] = True
        
    total_cost = 0
    
    # DFS to find edges that are part of the minimal subtree connecting all targets
    # An edge (u, v) is part of the Steiner tree if and only if both the subtree
    # at v and the rest of the tree contain at least one target node.
    
    stack = [(root, -1)]
    order = []
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != p:
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        if u == root:
            continue
        if has_target[u]:
            total_cost += edge_to_parent_w[u]
            has_target[parent[u]] = True
            
    print(total_cost)

solve()
```
