```python
import sys

# The problem asks for the minimum cost to ensure that for any x in the kernel of L,
# x_i = x_j for all i, j in the target set.
# A row vector (1 at u, -1 at v) in L implies x_u - x_v = 0, i.e., x_u = x_v.
# This means the components u and v are forced to be equal.
# We have a set of target dimensions {t_1, t_2, ..., t_k}.
# We need to force t_1 = t_2 = ... = t_k.
# This is equivalent to finding a minimum cost set of edges (u, v) such that
# all target nodes are in the same connected component in the graph formed by these edges.
# This is a classic Steiner Tree problem, but since we only need to connect a set of
# nodes in a graph where edges have weights, and we can use any nodes as intermediate
# points, this is equivalent to finding the Minimum Spanning Tree of the graph
# restricted to the components containing the target nodes.
# Actually, it's simpler: we want to connect all target nodes together.
# This is the Minimum Spanning Tree problem on the graph where we want to connect
# a specific subset of vertices. This is equivalent to the Steiner Tree problem,
# but since the graph is a tree (m = n-1 edges, n nodes), the path between any two
# nodes is unique.
# To connect all target nodes in a tree, we must include all edges that lie on the
# unique paths between any two target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find all edges that are part of the minimal subtree connecting all targets.
    # In a tree, an edge is part of the minimal subtree if and only if both components
    # created by removing the edge contain at least one target node.
    
    target_count = [0] * (n + 1)
    for node in targets:
        target_count[node] = 1
        
    total_targets = t
    total_cost = 0
    
    # Use DFS to count targets in subtrees
    visited = [False] * (n + 1)
    
    def dfs(u, p):
        nonlocal total_cost
        visited[u] = True
        count = target_count[u]
        for v, w in adj[u]:
            if v != p:
                sub_count = dfs(v, u)
                if sub_count > 0 and sub_count < total_targets:
                    total_cost += w
                count += sub_count
        return count

    dfs(1, -1)
    print(total_cost)

solve()
```
