```python
import sys

# The problem asks for the minimum cost to connect a set of t critical nodes
# in a tree. Since the input is a tree (n nodes, n-1 edges, connected),
# the minimum cost to connect a subset of nodes is the sum of the weights
# of all edges that lie on the unique paths between any two nodes in the subset.
# This is equivalent to the sum of weights of all edges that, when removed,
# split the tree into two components, each containing at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_critical[u] stores the number of critical nodes in the subtree of u
    subtree_critical = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(1, 0, 0)] # node, parent, weight_to_parent
    order = []
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    
    while stack:
        u, p, w = stack.pop()
        order.append(u)
        parent[u] = p
        weight_to_parent[u] = w
        for v, weight in adj[u]:
            if v != p:
                stack.append((v, u, weight))
                
    for u in order:
        if u in critical_nodes:
            subtree_critical[u] = 1
            
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        subtree_critical[p] += subtree_critical[u]
        # If this edge separates the tree into two parts both containing critical nodes
        if 0 < subtree_critical[u] < t:
            total_cost += weight_to_parent[u]
            
    print(total_cost)

solve()
```
