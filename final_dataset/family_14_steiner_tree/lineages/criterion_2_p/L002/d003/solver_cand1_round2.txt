```python
import sys

# The problem asks for the minimum weight of a subtree that connects all t critical nodes.
# In a tree, the unique minimal subtree connecting a set of nodes is the union of all 
# paths between any two nodes in the set. This is equivalent to the Steiner Tree 
# problem on a tree, which is simply the sum of weights of all edges that lie on 
# at least one path between two critical nodes.
# An edge lies on a path between two critical nodes if and only if the edge 
# splits the tree into two components, each containing at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_critical_count[u] stores the number of critical nodes in the subtree rooted at u
    subtree_critical_count = [0] * (n + 1)
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    edge_weights = {} # (min(u,v), max(u,v)) -> w
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if u < v:
                edge_weights[(u, v)] = w
                
    for u in reversed(order):
        if u in critical_nodes:
            subtree_critical_count[u] += 1
        
        p = parent[u]
        if p != -1:
            subtree_critical_count[p] += subtree_critical_count[u]
            # If this edge separates the tree into two parts both containing critical nodes
            if subtree_critical_count[u] > 0 and subtree_critical_count[u] < t:
                w = edge_weights[tuple(sorted((u, p)))]
                total_weight += w
                
    print(total_weight)

solve()
```
