```python
import sys

# The problem asks for the minimum weight of a subtree that connects all t critical nodes.
# Since the graph is a tree (m = n - 1), the unique path between any two nodes is fixed.
# The union of all paths between any two critical nodes forms the minimal subtree.
# This is equivalent to the sum of weights of all edges that lie on at least one path
# between two critical nodes.
# An edge (u, v) with weight w lies on a path between two critical nodes if and only if
# the removal of (u, v) splits the tree into two components, each containing at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many critical nodes are in the subtree rooted at each node
    # We use an iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    critical_count = [0] * (n + 1)
    for node in range(1, n + 1):
        if node in critical_nodes:
            critical_count[node] = 1
            
    total_power = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        critical_count[p] += critical_count[u]
        
        # If the subtree at u contains some but not all critical nodes,
        # the edge to its parent must be included.
        if 0 < critical_count[u] < t:
            total_power += edge_weight_to_parent[u]
            
    print(total_power)

if __name__ == '__main__':
    solve()
```
