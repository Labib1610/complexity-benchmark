```python
import sys

# The problem asks for the minimum total weight of edges required to connect 
# all critical modules in a tree. This is equivalent to finding the sum of 
# weights of all edges that lie on the unique paths between any two critical 
# modules. An edge is part of the required set if and only if the subtree 
# formed by removing that edge contains at least one critical module on both 
# sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    # count_critical[u] stores number of critical nodes in subtree of u
    count_critical = [0] * (n + 1)
    for node in critical_nodes:
        count_critical[node] = 1
        
    total_power = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            count_critical[p] += count_critical[u]
            # If the subtree at u contains some but not all critical nodes,
            # the edge (u, p) must be pressurized.
            if 0 < count_critical[u] < t:
                total_power += edge_weight[u]
                
    print(total_power)

if __name__ == '__main__':
    solve()
```
