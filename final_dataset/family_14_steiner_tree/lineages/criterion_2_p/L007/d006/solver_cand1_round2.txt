```python
import sys

# The problem asks for the minimum cost to ensure that for all target variables
# x_i, x_j, the constraint x_i = x_j is satisfied in the null space.
# The equations x_u - x_v = 0 can be viewed as edges in a graph.
# The null space of a set of equations is 1-dimensional if and only if the
# graph is connected.
# We need to select a subset of edges such that all target variables belong
# to the same connected component.
# This is equivalent to finding the minimum cost to connect all target nodes
# in the given tree (since m = n-1 and the set is minimally complete, the
# graph is a tree).
# This is the Steiner Tree problem on a tree, which is simply the sum of
# weights of all edges on the unique paths between the target nodes.
# In a tree, the union of paths between a set of nodes forms a subtree.
# An edge is part of this subtree if and only if removing it splits the
# tree into two components, each containing at least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_targets[u] = number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            subtree_targets[u] += 1
        for v, w in adj[u]:
            if v != parent[u]:
                subtree_targets[u] += subtree_targets[v]
                # If this edge separates the tree into two parts both containing targets
                if 0 < subtree_targets[v] < t:
                    total_cost += w
                    
    print(total_cost)

solve()
```
