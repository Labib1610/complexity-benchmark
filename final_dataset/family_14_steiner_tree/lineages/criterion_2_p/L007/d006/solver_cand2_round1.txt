```python
import sys

# The problem asks for the minimum cost to connect all elements in T 
# such that they belong to the same connected component in the graph 
# formed by the selected transpositions.
# Since the full set of m = n-1 transpositions forms a tree (as it generates S_n),
# the graph is a tree.
# To make all elements in T belong to the same component, we need to select
# all edges on the unique paths between any two nodes in T.
# This is equivalent to finding the minimal subtree that contains all nodes in T.
# The cost is the sum of weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Root the tree at target_nodes[0]
    root = target_nodes[0]
    
    # parent[u] = (p, weight_to_p)
    parent = [None] * (n + 1)
    is_target = [False] * (n + 1)
    for node in target_nodes:
        is_target[node] = True
        
    order = []
    stack = [root]
    parent[root] = (0, 0)
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] is None:
                parent[v] = (u, w)
                stack.append(v)
                
    # subtree_has_target[u] is true if the subtree at u contains any node from T
    subtree_has_target = [False] * (n + 1)
    for node in target_nodes:
        subtree_has_target[node] = True
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == root:
            continue
        p, w = parent[u]
        if subtree_has_target[u]:
            total_cost += w
            subtree_has_target[p] = True
            
    print(total_cost)

solve()
```
