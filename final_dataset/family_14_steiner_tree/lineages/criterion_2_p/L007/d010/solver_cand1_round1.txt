```python
import sys

# The problem asks for the minimum cost to connect all nodes in a target set T
# using a given set of m = n-1 edges. Since m = n-1 and the graph is guaranteed
# to span the space, the graph is a tree.
# To span e_a - e_b for all a, b in T, we need to ensure all nodes in T
# are in the same connected component in the subgraph formed by the chosen edges.
# Since the original graph is a tree, there is only one path between any two nodes.
# To connect all nodes in T, we must include all edges that lie on the unique
# paths between nodes in T.
# An edge (u, v) with weight w is part of the required set if and only if
# the removal of (u, v) splits the tree into two components, each containing
# at least one node from T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_targets[u] stores the number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Calculate subtree_targets in reverse topological order
    for u in reversed(order):
        if u in target_nodes:
            subtree_targets[u] += 1
        if parent[u] != -1:
            subtree_targets[parent[u]] += subtree_targets[u]
            
    # For each edge (u, parent[u]), it is needed if 0 < subtree_targets[u] < t
    for u in range(2, n + 1):
        count = subtree_targets[u]
        if 0 < count < t:
            # Find weight of edge (u, parent[u])
            for v, w in adj[u]:
                if v == parent[u]:
                    total_cost += w
                    break
                    
    print(total_cost)

solve()
```
