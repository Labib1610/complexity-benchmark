```python
import sys

# The problem asks for the minimum cost to connect all nodes in the target set T
# such that any pair (a, b) in T can be toggled.
# Since m = n - 1 and the graph is guaranteed to be connected (implied by the 
# ability to achieve any pair), the graph is a tree.
# Toggling a pair (a, b) is equivalent to toggling all edges on the unique 
# path between a and b in the tree.
# To be able to toggle any pair (a, b) in T, we need to be able to toggle 
# any edge in the tree that lies on the path between some pair in T.
# Specifically, an edge e is part of the path between some pair (a, b) in T 
# if and only if the removal of e splits the tree into two components, 
# each containing at least one node from T.
# We need to include all such edges in our subset of mutagens.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t < 2:
        print(0)
        return

    # subtree_t[u] stores the number of target nodes in the subtree rooted at u
    subtree_t = [0] * (n + 1)
    
    # DFS to compute subtree_t
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in target_nodes:
            subtree_t[u] += 1
        if parent[u] != 0:
            subtree_t[parent[u]] += subtree_t[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # Edge between i and parent[i] is needed if both components 
        # have at least one target node.
        count = subtree_t[i]
        if count > 0 and count < t:
            total_cost += edge_weight_to_parent[i]
            
    print(total_cost)

solve()
```
