```python
import sys

# The problem states the network is a tree (n racks, n-1 conduits, connected).
# We need to find the minimum cost to connect a subset of designated racks.
# In a tree, the set of edges required to connect a subset of nodes is the union
# of all paths between any two nodes in the subset. This is equivalent to the
# edges on the minimal subtree spanning the designated nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    designated = set(map(int, input_data[3:3+k]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + k
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if k <= 1:
        print(0)
        return

    # We need to find the sum of weights of all edges that lie on a path
    # between any two designated nodes.
    # An edge (u, v) with weight w is part of the minimal subtree if and only if
    # the subtree formed by removing (u, v) contains at least one designated node
    # in both components.
    
    # Root the tree at an arbitrary designated node
    root = next(iter(designated))
    
    # Count designated nodes in each subtree
    subtree_designated_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in designated:
        subtree_designated_count[u] = 1
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == root:
            continue
        
        count = subtree_designated_count[u]
        if count > 0 and count < k:
            total_cost += parent_edge_weight[u]
            
        subtree_designated_count[parent[u]] += count
        
    print(total_cost)

solve()
```
