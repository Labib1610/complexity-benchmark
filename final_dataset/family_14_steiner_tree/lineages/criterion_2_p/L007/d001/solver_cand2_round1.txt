import sys

def solve():
    # Read n, m, t
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    leased_racks = set()
    idx = 3
    for _ in range(t):
        leased_racks.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Since m = n - 1 and the graph is connected, it is a tree.
    # The problem asks for the minimum cost to connect all leased racks.
    # In a tree, there is a unique path between any two nodes.
    # The union of all paths between all pairs of leased racks forms a subtree.
    # We need to sum the weights of all edges that belong to this subtree.
    
    # Root the tree at an arbitrary leased rack (e.g., the first one)
    root = next(iter(leased_racks))
    
    # parent_edge[v] = weight of edge between v and parent[v]
    parent_edge = [0] * (n + 1)
    parent = [0] * (n + 1)
    has_leased_descendant = [False] * (n + 1)
    
    for r in leased_racks:
        has_leased_descendant[r] = True
        
    stack = [(root, 0, 0)]
    order = []
    while stack:
        u, p, w = stack.pop()
        parent[u] = p
        parent_edge[u] = w
        order.append(u)
        for v, weight in adj[u]:
            if v != p:
                stack.append((v, u, weight))
                
    # Post-order traversal to mark nodes that are part of the Steiner tree
    for u in reversed(order):
        if has_leased_descendant[u] and parent[u] != 0:
            has_leased_descendant[parent[u]] = True
            
    total_cost = 0
    for i in range(1, n + 1):
        if has_leased_descendant[i] and parent[i] != 0:
            total_cost += parent_edge[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
