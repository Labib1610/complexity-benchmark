```python
import sys

# The problem states m = n - 1 and the graph is connected (implied by "at least one valid configuration exists").
# Since m = n - 1 and the graph is connected, the graph is a tree.
# In a tree, there is exactly one path between any two nodes.
# To connect a subset of nodes in a tree, we must include all edges that lie on the paths
# between any pair of the subset of nodes.
# This is equivalent to the union of all paths between the leased nodes.
# Since it's a tree, this is the minimal subtree containing all leased nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    leased = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Root the tree at an arbitrary leased node
    root = next(iter(leased))
    
    # subtree_leased[u] = number of leased nodes in subtree of u
    subtree_leased = [0] * (n + 1)
    
    # To avoid recursion depth issues, use iterative DFS
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in leased:
        subtree_leased[u] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == root:
            continue
        if subtree_leased[u] > 0 and subtree_leased[u] < t:
            total_cost += edge_to_parent_weight[u]
        subtree_leased[parent[u]] += subtree_leased[u]
        
    print(total_cost)

solve()
```
