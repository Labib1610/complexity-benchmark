import sys
from collections import defaultdict
import heapq

def main():
    import sys
    sys.setrecursionlimit(1000000)
    
    n, m, t = map(int, sys.stdin.readline().split())
    leased = set(map(int, sys.stdin.readline().split()))
    
    # If there's only one leased rack, cost is 0
    if t <= 1:
        print(0)
        return
    
    # Build the graph
    graph = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
        edges.append((w, u, v))
    
    # Find the minimum spanning tree of the leased nodes (Steiner tree approximation)
    # We'll use the fact that the optimal solution is the MST of the subgraph 
    # induced by the leased nodes and their connecting paths
    
    # First, find the subgraph that connects all leased nodes
    # We can do this by finding the minimum spanning tree of the entire graph
    # and then finding the minimum cost to connect all leased nodes
    
    # Kruskal's algorithm to find MST
    edges.sort()
    parent = list(range(n + 1))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # We need to find the minimum cost to connect all leased nodes
    # This is essentially finding the minimum Steiner tree, which is NP-hard
    # But we can use the following approach:
    # 1. Build the MST of the entire graph
    # 2. In this MST, find the subgraph that connects all leased nodes
    # 3. The cost of this subgraph is our answer
    
    # Actually, let's think differently:
    # The problem is to find the minimum cost to connect all leased nodes
    # This is the minimum Steiner tree problem, but we can solve it efficiently
    # by building an auxiliary graph where nodes are the leased nodes
    # and edge weights are the shortest paths between them in the original graph
    
    # But there's a simpler approach for trees:
    # Since the original graph is a tree, we can find the minimal subtree
    # that connects all leased nodes. This is called the Steiner tree on a tree.
    
    # Algorithm for Steiner tree on a tree:
    # 1. Find all leaves that are not leased nodes
    # 2. Remove them iteratively until we're left with only nodes needed
    #    to connect the leased nodes
    
    # Let's build the adjacency list representation
    adj = defaultdict(list)
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # But we need to work with the original tree structure
    # Let's use a different approach:
    # Build the actual tree and then find the minimal subtree
    
    # Build adjacency list for the tree
    tree = defaultdict(list)
    for u, v, w in edges:
        tree[u].append((v, w))
        tree[v].append((u, w))
    
    # Find the minimal subtree that connects all leased nodes
    # This is the Steiner tree problem on a tree
    
    # Approach:
    # 1. Start with all leased nodes marked
    # 2. Do a DFS to find the minimal subtree that connects them all
    # 3. The cost is the sum of edge weights in this subtree
    
    # We can do this by:
    # 1. Finding the subtree that contains all leased nodes
    # 2. The cost is the sum of all edges in this subtree
    
    # More efficient approach:
    # Use the fact that in a tree, the minimal Steiner tree can be found by:
    # 1. Root the tree at any node
    # 2. For each node, compute if it's needed (is leased or has leased descendants)
    # 3. Count edges between needed nodes
    
    # Let's root the tree at node 1
    # First, we need to build the tree structure
    
    # Build adjacency list
    adj_list = defaultdict(list)
    for u, v, w in edges:
        adj_list[u].append((v, w))
        adj_list[v].append((u, w))
    
    # DFS to find the minimal Steiner tree
    # We'll do a post-order traversal
    
    # Convert to 0-indexed for easier handling
    visited = [False] * (n + 1)
    total_cost = 0
    
    def dfs(node, parent):
        nonlocal total_cost
        visited[node] = True
        
        # Count leased nodes in subtree
        is_leased = 1 if node in leased else 0
        needed = is_leased
        
        for neighbor, weight in adj_list[node]:
            if neighbor == parent:
                continue
            if visited[neighbor]:
                continue
                
            subtree_needed = dfs(neighbor, node)
            needed += subtree_needed
            
            # If the subtree has leased nodes, we need this edge
            if subtree_needed > 0 and subtree_needed < t:
                total_cost += weight
                
        return needed
    
    # Find a node to start DFS (any node is fine, let's use the first leased node)
    start_node = next(iter(leased))
    dfs(start_node, -1)
    
    print(total_cost)

if __name__ == "__main__":
    main()
