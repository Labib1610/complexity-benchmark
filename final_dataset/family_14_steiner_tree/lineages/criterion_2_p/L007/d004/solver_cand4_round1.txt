```python
import sys

# The problem describes a tree structure where we need to connect a subset of nodes (portfolio).
# The cost to connect a set of nodes in a tree is the sum of weights of all edges 
# on the unique paths between them.
# Since the regulatory fine is 0, we simply need to find the sum of weights of all edges 
# that lie on the paths between any two nodes in the portfolio.
# An edge is part of the minimal subtree connecting the portfolio if and only if 
# the edge splits the tree into two components, both of which contain at least one 
# portfolio node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    portfolio = set()
    idx = 3
    for _ in range(t):
        portfolio.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_portfolio_count[u] stores how many portfolio nodes are in the subtree rooted at u
    subtree_portfolio_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    for u in reversed(order):
        if u in portfolio:
            subtree_portfolio_count[u] += 1
        if parent[u] != 0:
            p, w = parent[u]
            subtree_portfolio_count[p] += subtree_portfolio_count[u]
            # If this edge splits the tree such that both sides have portfolio nodes
            if 0 < subtree_portfolio_count[u] < t:
                total_cost += w
                
    print(total_cost)

solve()
```
