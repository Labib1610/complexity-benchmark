import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    portfolio = set()
    idx = 3
    for _ in range(t):
        portfolio.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the smallest subtree that contains all nodes in the portfolio.
    # The cost is the sum of weights of all edges in this subtree.
    # We can use DFS to mark nodes that are part of the subtree.
    
    has_portfolio = [False] * (n + 1)
    for node in portfolio:
        has_portfolio[node] = True
        
    total_cost = 0
    
    # visited array to keep track of nodes in the subtree
    in_subtree = [False] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    # We need to identify all edges that lie on paths between any two nodes in the portfolio.
    # An edge (u, v) with weight w is in the subtree if and only if the subtree
    # rooted at v (when considering u as parent) contains at least one portfolio node
    # AND the rest of the tree contains at least one portfolio node.
    
    subtree_has_portfolio = [False] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if has_portfolio[u]:
            subtree_has_portfolio[u] = True
        if subtree_has_portfolio[u] and parent[u] != 0:
            subtree_has_portfolio[parent[u]] = True
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

solve()
