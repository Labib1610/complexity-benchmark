```python
import sys

# The problem describes a tree (m = n - 1).
# We need to keep only the links that are part of the unique paths between any two rented racks.
# If t = 1, no paths exist between pairs, so all links are removed.
# If t > 1, the links that must remain are those that lie on the path between at least two rented racks.
# This is equivalent to the edges in the minimal subtree connecting all t rented racks.
# Any edge not in this minimal subtree can be removed.
# The procedure says: "pick the one with the highest activation expense".
# Since we are removing all edges not in the minimal subtree, the order doesn't matter for the final set.
# The final total expense is the sum of weights of edges in the minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    rented = set()
    idx = 3
    for _ in range(t):
        rented.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w, i))
        adj[v].append((u, w, i))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Find the minimal subtree containing all rented nodes.
    # We can prune the tree: any leaf node that is not rented can be removed.
    # Repeat until all leaves are rented.
    
    degree = [0] * (n + 1)
    for i in range(1, n + 1):
        degree[i] = len(adj[i])
        
    queue = []
    for i in range(1, n + 1):
        if degree[i] == 1 and i not in rented:
            queue.append(i)
            
    removed_edge = [False] * m
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        for v, w, edge_idx in adj[u]:
            if not removed_edge[edge_idx]:
                removed_edge[edge_idx] = True
                degree[v] -= 1
                if degree[v] == 1 and v not in rented:
                    queue.append(v)
                    
    total_expense = 0
    for i in range(m):
        if not removed_edge[i]:
            total_expense += edges[i][2]
            
    print(total_expense)

solve()
```
