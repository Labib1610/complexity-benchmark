import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    rented = set(map(int, sys.stdin.readline().split()))
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    # Sort edges by weight
    edges.sort()
    
    # Union-Find data structure
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    # Kruskal's algorithm variant
    total_cost = 0
    components = t  # Initially, each rented node is its own component
    
    for w, u, v in edges:
        # If both nodes are not rented and not connected to any rented node, skip
        # But we need to track which components contain rented nodes
        
        # Actually, let's think differently:
        # We want to connect all nodes that contain or are connected to rented nodes
        # So we run Kruskal and whenever we merge two components, 
        # at least one of which contains a rented node, we count the cost
        
        pu, pv = find(u), find(v)
        if pu != pv:
            # Check if either component contains a rented node
            # For efficiency, we'll mark components that have rented nodes
            # Let's preprocess: initially mark components with rented nodes
            
            # Actually simpler approach:
            # Run Kruskal's algorithm on the whole tree to find MST
            # But we only care about connecting the rented nodes
            # This is exactly the Steiner tree problem, but since the graph is a tree,
            # the solution is to find the minimal subtree that connects all rented nodes
            
            # In a tree, the minimal Steiner tree is found by:
            # 1. Consider only nodes that are rented or lie on paths between rented nodes
            # 2. The edges needed are those in the paths between all pairs of rented nodes
            # 3. But we can't double-count edges
            
            # Better approach:
            # Find the subtree that spans all rented nodes
            # This is equivalent to finding the union of paths between all pairs
            # But in a tree, it's the same as finding the minimal connected subgraph containing all rented nodes
            
            # Even better:
            # Root the tree at some node, find the deepest node among rented ones
            # The minimal subtree is formed by all paths from root to each rented node
            # But that overcounts
            
            # Correct approach:
            # Build the "Steiner tree" for the rented nodes in the tree
            # Algorithm:
            # 1. Root the tree at node 1 (arbitrary)
            # 2. For each rented node, mark its path to root
            # 3. The union of these paths forms the minimal subtree
            # 4. Its total edge weight is our answer
            
            # But we don't have the tree structure yet - edges are given separately
            
            # Simpler idea:
            # Use Union-Find. Process edges in order of increasing weight.
            # When we unite two components, and at least one contains a rented node,
            # we pay for this edge. If both components already had rented nodes,
            # we still pay, because this edge connects them.
            
            # But we need to track whether a component contains a rented node
            
            # Initialize: each rented node marks its component
            # But Union-Find doesn't naturally support this...
            
            # Alternative:
            # Since the graph is a tree, the Steiner tree is unique
            # We can build it by:
            # 1. Building the tree structure
            # 2. Finding the minimal subtree spanning all rented nodes
            
            # Let's build adjacency list representation
            # Then find the subtree
            
            # Too complex. Let's re-think.
            
            # Key insight: In a tree, to connect k nodes, we need exactly k-1 edges
            # in their minimal subtree. But which edges?
            
            # The edges we need are those in the paths between the rented nodes,
            # but without duplication. This forms a subtree.
            
            # Efficient way:
            # 1. Root the tree
            # 2. Find all ancestors for each rented node
            # 3. Find the minimal connected subgraph containing all rented nodes
            
            # Even simpler for implementation:
            # Use the fact that in the MST of the full graph, 
            # the edges we need are those in the minimal subtree spanning the rented nodes
            
            # But since the original graph is a tree, we just need to find 
            # the minimal subtree containing all rented nodes
            
            # Algorithm:
            # 1. Build the tree
            # 2. Find all nodes needed (rented + internal nodes in paths)
            # 3. Sum up the edges in this subtree
            
            # Implementation:
            # Let's use Union-Find with tracking of whether a component has a rented node
            
            # Initialize has_rented for each node
            has_rented = [False] * (n + 1)
            for r in rented:
                has_rented[r] = True
                
            # Now, when we unite two components, we update has_rented
            # If at least one component has a rented node, the new component does too
            # If both had rented nodes, we add the edge cost
            
            pu, pv = find(u), find(v)
            if pu != pv:
                # Check if we should unite and count this edge
                # We unite if at least one component has a rented node
                if has_rented[pu] or has_rented[pv]:
                    total_cost += w
                    # Update the parent's has_rented status
                    union(u, v)
                    new_root = find(u)  # After union, one becomes parent
                    has_rented[new_root] = has_rented[pu] or has_rented[pv]
                    
    print(total_cost)

threading.Thread(target=main).start()
