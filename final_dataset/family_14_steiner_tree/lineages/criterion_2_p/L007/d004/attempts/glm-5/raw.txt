import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return
    
    rented = [0] * (n + 1)
    rented_indices = []
    for _ in range(t):
        idx = int(next(iterator))
        rented_indices.append(idx)
        rented[idx] = 1
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        w = int(next(iterator))
        adj[u].append((v, i))
        adj[v].append((u, i))
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    parent = [-1] * (n + 1)
    parent_edge_idx = [-1] * (n + 1)
    depth = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    queue = deque()
    root = 1
    visited[root] = True
    queue.append(root)
    
    while queue:
        u = queue.popleft()
        for v, edge_idx in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_idx[v] = edge_idx
                depth[v] = depth[u] + 1
                queue.append(v)
    
    count = [0] * (n + 1)
    for node in rented_indices:
        count[node] = 1
    
    total_cost = 0
    
    for node in range(1, n + 1):
        if parent[node] != -1:
            p = parent[node]
            if depth[node] < depth[p]:
                count[node] += count[p]
            elif depth[node] > depth[p]:
                count[p] += count[node]
    
    for v in range(1, n + 1):
        if parent[v] != -1:
            if count[v] > 0:
                edge_idx = parent_edge_idx[v]
                u, w, wt = edges[edge_idx]
                total_cost += wt
    
    print(total_cost)

if __name__ == "__main__":
    solve()
