```python
import sys

# The problem states m = n - 1 and the graph is connected (implied by "valid selection exists").
# Since m = n - 1 and the graph is connected, the infrastructure is a tree.
# We need to connect all rented racks. In a tree, there is a unique path between any two nodes.
# To connect a set of nodes in a tree, we must include all edges that lie on the paths
# between any two rented nodes.
# Since the relay activation fee is 0, we simply need to sum the weights of all edges
# that are part of the minimal subtree connecting all rented nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    rented = set()
    idx = 3
    for _ in range(t):
        rented.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Use post-order traversal to determine if a subtree contains any rented nodes.
    # An edge (u, v) where v is child of u is part of the Steiner tree if and only if
    # the subtree rooted at v contains at least one rented node AND the rest of the
    # tree (outside v's subtree) contains at least one rented node.
    
    has_rented = [False] * (n + 1)
    for r in rented:
        has_rented[r] = True
        
    total_rented = t
    edge_sum = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order
    subtree_rented = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    
    for u in reversed(order):
        if has_rented[u]:
            subtree_rented[u] += 1
        
        for v, w in adj[u]:
            if v == parent[u]:
                edge_weight_to_parent[u] = w
            else:
                subtree_rented[u] += subtree_rented[v]
                
    ans = 0
    for i in range(2, n + 1):
        # Edge between i and parent[i] is needed if subtree at i has rented nodes
        # and the rest of the tree has rented nodes.
        if subtree_rented[i] > 0 and (total_rented - subtree_rented[i]) > 0:
            ans += edge_weight_to_parent[i]
            
    print(ans)

solve()
```
