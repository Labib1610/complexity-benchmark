```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    rented = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        rented[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the Steiner Tree in a tree.
    # In a tree, the Steiner Tree is the union of all paths between any two rented nodes.
    # This is equivalent to the minimal subtree containing all rented nodes.
    # We can prune all branches that do not contain any rented nodes.
    
    has_rented = [False] * (n + 1)
    
    # First pass: identify nodes that are part of the minimal subtree
    # A node is in the subtree if it is rented or if it lies on a path between two rented nodes.
    # This is equivalent to: node u is in the subtree if the subtree rooted at u 
    # contains at least one rented node AND the part of the tree outside u's subtree 
    # contains at least one rented node.
    
    subtree_has_rented = [False] * (n + 1)
    
    def dfs1(u, p):
        res = rented[u]
        for v, w in adj[u]:
            if v != p:
                if dfs1(v, u):
                    res = True
        subtree_has_rented[u] = res
        return res

    dfs1(1, -1)
    
    total_cost = 0
    
    # Second pass: calculate cost
    # A link (u, v) with weight w is included if and only if 
    # one side of the link contains at least one rented node 
    # and the other side also contains at least one rented node.
    
    def dfs2(u, p):
        nonlocal total_cost
        for v, w in adj[u]:
            if v != p:
                # Check if the branch at v contains a rented node
                if subtree_has_rented[v]:
                    # Check if the rest of the tree (excluding branch v) contains a rented node
                    # The rest of the tree contains a rented node if (total_rented - subtree_rented[v]) > 0
                    # We can pre-calculate total rented count
                    if (t - count_rented[v]) > 0:
                        total_cost += w
                    dfs2(v, u)

    count_rented = [0] * (n + 1)
    def dfs_count(u, p):
        c = 1 if rented[u] else 0
        for v, w in adj[u]:
            if v != p:
                c += dfs_count(v, u)
        count_rented[u] = c
        return c

    dfs_count(1, -1)
    dfs2(1, -1)
    
    print(total_cost)

solve()
```
