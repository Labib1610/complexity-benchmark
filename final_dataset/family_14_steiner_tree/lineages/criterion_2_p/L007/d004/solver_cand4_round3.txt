```python
import sys

# The problem asks for the minimum cost to connect all guild outposts.
# Since the structure is a tree, the set of edges that connects all guild outposts
# is the unique path in the tree that spans all guild outposts.
# This is equivalent to the union of all paths between any two guild outposts.
# In a tree, this is the minimal subtree containing all guild outposts.
# The cost is the sum of the weights of all edges in this minimal subtree.
# The transit tax is 0, so we only sum the weights of the edges.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    guild_outposts = set()
    idx = 3
    for _ in range(t):
        guild_outposts.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the minimal subtree containing all guild outposts.
    # An edge is in the minimal subtree if and only if both components
    # formed by removing the edge contain at least one guild outpost.
    
    # Count guild outposts in each subtree using DFS
    subtree_guild_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in guild_outposts:
            subtree_guild_count[u] += 1
        if parent[u] != 0:
            subtree_guild_count[parent[u]] += subtree_guild_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        count = subtree_guild_count[i]
        # If the subtree at i contains some but not all guild outposts,
        # the edge connecting i to its parent must be part of the minimal subtree.
        if count > 0 and count < t:
            total_cost += edge_weight_to_parent[i]
            
    print(total_cost)

solve()
```
