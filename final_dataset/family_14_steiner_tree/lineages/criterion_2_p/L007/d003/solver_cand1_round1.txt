```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    client_racks = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need the Steiner Tree for the given t nodes.
    # Since the graph is a tree, the Steiner Tree is the union of all paths
    # between the client nodes. This is equivalent to the minimal subtree
    # containing all client nodes.
    
    is_client = [False] * (n + 1)
    for r in client_racks:
        is_client[r] = True
        
    subtree_has_client = [False] * (n + 1)
    
    # DFS to identify edges that are part of the minimal subtree
    # An edge (u, v) is in the subtree if the subtree rooted at v contains
    # at least one client node AND the rest of the tree contains at least one.
    
    total_cost = 0
    
    def dfs(u, p):
        nonlocal total_cost
        has_client = is_client[u]
        for v, w in adj[u]:
            if v == p:
                continue
            child_has_client = dfs(v, u)
            if child_has_client:
                total_cost += w
                has_client = True
        return has_client

    # Start DFS from any client node
    start_node = next(iter(client_racks))
    dfs(start_node, -1)
    
    print(total_cost)

solve()
```
