```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    rented = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        rented[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count[u] will store how many rented nodes are in the subtree of u
    count = [0] * (n + 1)
    total_expense = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order
    for u in reversed(order):
        if rented[u]:
            count[u] += 1
        
        if parent[u] != 0:
            p = parent[u]
            count[p] += count[u]
            # If the subtree at u contains at least one rented node
            # and the rest of the tree contains at least one rented node,
            # the edge to parent must be powered on.
            if count[u] > 0 and count[u] < t:
                total_expense += edge_weight[u]
                
    print(total_expense)

if __name__ == '__main__':
    solve()
```
