```python
import sys

# The problem asks for the minimum weight of a subset of edges (restrictions)
# such that all nodes in the target set T are in the same connected component.
# Since the graph of restrictions is a tree (n nodes, n-1 edges, connected),
# the condition "x_a = x_b for all a, b in T" is equivalent to saying that
# all nodes in T must be in the same connected component of the subgraph
# formed by the selected edges.
# To minimize the total weight of edges to connect all nodes in T, we need
# to find the Steiner Tree for the set T in the given tree.
# In a tree, the Steiner Tree for a set of nodes T is the unique minimal
# subtree containing all nodes in T. This subtree consists of all edges
# that lie on the simple path between any two nodes in T.
# An edge (u, v) with weight w is part of the Steiner Tree if and only if
# the removal of this edge splits the tree into two components, each of
# which contains at least one node from T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = set(map(int, input_data[3:3+t]))
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # count_t[u] will store the number of target nodes in the subtree rooted at u
    count_t = [0] * (n + 1)
    for node in target_nodes:
        count_t[node] = 1
        
    # DFS to compute count_t
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    total_weight = 0
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        count_t[p] += count_t[u]
        # If the subtree at u contains some but not all target nodes,
        # the edge (u, p) must be included in the Steiner Tree.
        if 0 < count_t[u] < t:
            total_weight += edge_to_parent_w[u]
            
    print(total_weight)

solve()
```
