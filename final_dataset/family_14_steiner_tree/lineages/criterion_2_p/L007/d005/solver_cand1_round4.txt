```python
import sys

# The problem asks for the minimum cost to connect a subset of t nodes in a tree.
# Since the graph is a tree (n nodes, n-1 edges, connected), there is a unique
# path between any two nodes. The set of edges required to connect all t nodes
# is the union of all paths between every pair of the t nodes.
# This is equivalent to the edges in the minimal subtree containing all t nodes.
# An edge (u, v) with weight w is part of this subtree if and only if the
# removal of this edge splits the tree into two components, each containing
# at least one of the t nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    rented = set()
    idx = 3
    for _ in range(t):
        rented.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # count[u] will store the number of rented nodes in the subtree rooted at u
    count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    parent = [0] * (n + 1)
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in rented:
            count[u] += 1
            
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            count[p] += count[u]
            # If this subtree contains some but not all rented nodes,
            # the edge to the parent must be included.
            if 0 < count[u] < t:
                total_cost += parent_edge_weight[u]
                
    print(total_cost)

solve()
```
