```python
import sys

# The problem asks for the minimum cost to connect all nodes in the set T
# using the given edges (modules).
# The condition "there exists a sequence of modules to swap row a and b
# while keeping other rows in their original positions" is equivalent to
# saying that all nodes in T must belong to the same connected component
# in the graph formed by the selected modules.
# Since the graph of all m modules is a tree (n nodes, n-1 edges),
# the subset of modules that connects all nodes in T is the unique path
# in the tree that spans all nodes in T.
# This is equivalent to the Steiner Tree problem on a tree, which is
# simply the sum of weights of all edges that lie on the paths between
# nodes in T. An edge is part of this set if and only if the subtree
# formed by removing that edge contains at least one node from T on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_t[u] will store the number of target nodes in the subtree rooted at u
    subtree_t = [0] * (n + 1)
    
    # Root the tree at an arbitrary node, say 1
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for node in target_nodes:
        subtree_t[node] = 1
        
    total_weight = 0
    # Process in reverse topological order to compute subtree counts
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        # Find weight of edge (u, p)
        w_edge = 0
        for v, w in adj[u]:
            if v == p:
                w_edge = w
                break
        
        # If the subtree at u contains some but not all target nodes,
        # the edge (u, p) must be included.
        if 0 < subtree_t[u] < t:
            total_weight += w_edge
        
        subtree_t[p] += subtree_t[u]
        
    print(total_weight)

solve()
```
