```python
import sys

# The problem states m = n - 1 and the graph is connected, so it's a tree.
# We need to find the minimum cost to connect a subset of t nodes.
# This is equivalent to finding the sum of weights of all edges that lie on
# the unique paths between any two leased nodes.
# An edge is part of the solution if and only if the subtree formed by
# removing that edge contains at least one leased node on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    leased = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        leased[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    subtree_leased_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if leased[i]:
            subtree_leased_count[i] = 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_leased_count[p] += subtree_leased_count[u]
            # If this subtree contains some but not all leased nodes,
            # the edge to the parent must be included.
            if 0 < subtree_leased_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
