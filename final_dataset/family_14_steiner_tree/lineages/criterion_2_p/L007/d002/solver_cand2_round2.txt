```python
import sys

# The problem states that the graph is a tree (n nodes, n-1 edges, connected).
# To connect a subset of nodes in a tree, the minimum cost is the sum of the weights
# of all edges that lie on the paths between any two nodes in the subset.
# An edge is part of the minimal Steiner tree if and only if it separates the
# set of rented nodes into two non-empty sets.
# Equivalently, an edge is part of the solution if and only if the subtree
# formed by removing that edge contains at least one rented node on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    rented = set()
    idx = 3
    for _ in range(t):
        rented.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # count_rented[u] will store the number of rented nodes in the subtree rooted at u
    count_rented = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order to compute subtree counts
    for u in reversed(order):
        if u in rented:
            count_rented[u] += 1
        if parent[u] != -1:
            count_rented[parent[u]] += count_rented[u]
            
    # Calculate cost: an edge (u, parent[u]) is included if 0 < count_rented[u] < t
    for u in range(2, n + 1):
        c = count_rented[u]
        if 0 < c < t:
            # Find weight of edge (u, parent[u])
            for v, w in adj[u]:
                if v == parent[u]:
                    total_cost += w
                    break
                    
    print(total_cost)

solve()
```
