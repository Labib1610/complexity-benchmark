```python
import sys

# The problem asks for the minimum cost to connect all nodes in a target set T
# using a given set of edges (the catalog). Since the catalog forms a tree
# (n nodes, n-1 edges, connected), we are looking for the cost of the minimal
# subtree that contains all nodes in T.
# This is equivalent to the sum of weights of all edges that lie on the paths
# between nodes in T. An edge belongs to this subtree if and only if removing
# it splits the tree into two components, both of which contain at least one
# node from T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many target nodes are in each subtree formed by removing an edge.
    # We can root the tree at an arbitrary node (e.g., 1).
    # For each edge (u, v) where v is child of u, the subtree at v contains
    # some number of target nodes 'count[v]'.
    # If 0 < count[v] < t, then this edge is part of the minimal subtree.
    
    count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    parent = [0] * (n + 1)
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in target_nodes:
        count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            if count[u] > 0 and count[u] < t:
                total_cost += parent_edge_weight[u]
            count[p] += count[u]
            
    print(total_cost)

solve()
```
