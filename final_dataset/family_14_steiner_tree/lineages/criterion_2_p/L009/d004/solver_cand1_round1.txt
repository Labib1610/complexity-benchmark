```python
import sys

# The problem asks for the minimum cost to connect all critical parameters in a tree.
# Since the input is a tree (N nodes, N-1 edges), we can root the tree arbitrarily.
# A set of edges connects a subset of nodes if and only if the edges form a subtree
# that contains all those nodes. The minimum cost to connect a set of nodes in a tree
# is the sum of weights of all edges that lie on the paths between any two nodes
# in the set. This is equivalent to the sum of weights of all edges that have
# at least one critical node in each of the two components formed by removing the edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    critical = [False] * (N + 1)
    idx = 3
    for _ in range(F):
        critical[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(N + 1)]
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if F <= 1:
        print(0)
        return

    # subtree_critical[u] stores the number of critical nodes in the subtree of u
    subtree_critical = [0] * (N + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_to_parent_w = [0] * (N + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if critical[u]:
            subtree_critical[u] += 1
        if parent[u] != -1:
            subtree_critical[parent[u]] += subtree_critical[u]
            # If an edge separates the tree into two parts, and both parts
            # contain at least one critical node, the edge must be included.
            if 0 < subtree_critical[u] < F:
                total_cost += edge_to_parent_w[u]
                
    print(total_cost)

solve()
```
