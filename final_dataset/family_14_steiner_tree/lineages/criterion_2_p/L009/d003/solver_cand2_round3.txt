```python
import sys

# The problem states the network is a tree (N outposts, N-1 links, fully integrated).
# We need to find the minimum cost to connect all F classified outposts.
# In a tree, the unique path between any two nodes is fixed.
# The set of edges that connect all classified outposts is the union of all paths
# between every pair of classified outposts.
# This is equivalent to the edges in the minimal subtree containing all F nodes.
# An edge (u, v) with weight w is part of this subtree if and only if
# the subtree formed by removing (u, v) has at least one classified outpost
# in both components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    classified = set()
    idx = 3
    for _ in range(F):
        classified.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(N + 1)]
    edges = []
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # count[u] will store the number of classified outposts in the subtree rooted at u
    count = [0] * (N + 1)
    
    # DFS to compute counts
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in classified:
            count[u] += 1
        if parent[u] != 0:
            count[parent[u]] += count[u]
            
    total_cost = 0
    for u in range(1, N + 1):
        if parent[u] != 0:
            # Edge is between u and parent[u]
            # It is part of the minimal subtree if count[u] > 0 and count[u] < F
            if count[u] > 0 and count[u] < F:
                # Find weight of edge (u, parent[u])
                for v, w in adj[u]:
                    if v == parent[u]:
                        total_cost += w
                        break
                        
    print(total_cost)

solve()
```
