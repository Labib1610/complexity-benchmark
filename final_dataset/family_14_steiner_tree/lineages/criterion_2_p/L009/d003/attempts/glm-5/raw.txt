```python
import sys
import heapq

def solve():
    input = sys.stdin.read().split()
    ptr = 0
    N = int(input[ptr]); ptr += 1
    E = int(input[ptr]); ptr += 1
    F = int(input[ptr]); ptr += 1
    
    classified = set()
    for _ in range(F):
        classified.add(int(input[ptr]) - 1)
        ptr += 1
        
    adj = [[] for _ in range(N)]
    for _ in range(E):
        u = int(input[ptr]) - 1; ptr += 1
        v = int(input[ptr]) - 1; ptr += 1
        w = int(input[ptr]); ptr += 1
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Prim's algorithm starting from all classified nodes
    in_mst = [False] * N
    min_edge = [float('inf')] * N
    heap = []
    
    for u in classified:
        min_edge[u] = 0
        heapq.heappush(heap, (0, u))
    
    mst_weight = 0
    nodes_in_mst = 0
    
    while heap:
        w, u = heapq.heappop(heap)
        if in_mst[u]:
            continue
        in_mst[u] = True
        mst_weight += w
        nodes_in_mst += 1
        
        for v, weight in adj[u]:
            if not in_mst[v] and weight < min_edge[v]:
                min_edge[v] = weight
                heapq.heappush(heap, (weight, v))
                
    print(mst_weight)

if __name__ == "__main__":
    solve()
```
