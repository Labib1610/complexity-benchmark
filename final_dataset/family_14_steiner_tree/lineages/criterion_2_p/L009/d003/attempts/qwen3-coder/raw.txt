    # 1. Root the tree at some node
    # 2. Find all ancestors and do a traversal to identify necessary nodes
    
    parent = [-1] * (N+1)
    depth = [0] * (N+1)
    visited = [False] * (N+1)
    
    def dfs(u, p, d):
        parent[u] = p
        depth[u] = d
        visited[u] = True

            v = parent[v]
        # Now move both up until they meet

            v = parent[v]
        return u

    # Since we're working on a tree, the optimal solution is to take the minimal subtree
    # that spans all classified nodes. The cost is the sum of weights of edges in this subtree.
    
    # Algorithm:
    # 1. Mark all classified nodes
    # 2. Do a DFS to compute for each node whether its subtree contains any classified node
    # 3. An edge should be included if it connects two components that both contain classified nodes
    
    marked = [0] * (N+1)  # Number of classified nodes in subtree
    for c in classified:
        marked[c] = 1
        
    result = 0
    visited_dfs = [False] * (N+1)

        visited_dfs[u] = True

                count += child_count
