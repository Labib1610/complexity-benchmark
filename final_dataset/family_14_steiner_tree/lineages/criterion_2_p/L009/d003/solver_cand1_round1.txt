```python
import sys

# The problem states E = N - 1, meaning the network is a tree.
# We need to find the minimum cost to connect all F high-priority terminals.
# In a tree, the unique path between any two nodes is fixed.
# The set of edges that must be included is the union of all paths between
# any two high-priority terminals. This is equivalent to the edges in the
# minimal subtree containing all F terminals.
# An edge (u, v) with weight w is part of this subtree if and only if
# the removal of this edge splits the tree into two components, each
# containing at least one high-priority terminal.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    terminals = set(map(int, input_data[3:3+F]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    edges = []
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if F <= 1:
        print(0)
        return

    # Count how many terminals are in the subtree rooted at each node
    # when rooted at node 1.
    terminal_count = [0] * (N + 1)
    parent_edge_weight = [0] * (N + 1)
    parent = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in terminals:
            terminal_count[u] += 1
        if parent[u] != 0:
            terminal_count[parent[u]] += terminal_count[u]
            
    total_cost = 0
    for i in range(2, N + 1):
        # If the subtree at i contains some terminals and the rest of the tree
        # contains some terminals, the edge to parent must be included.
        count = terminal_count[i]
        if count > 0 and count < F:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
