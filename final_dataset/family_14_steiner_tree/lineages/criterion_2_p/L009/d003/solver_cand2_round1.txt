```python
import sys

# The problem states that the graph is a tree because removing any single 
# pipeline disconnects the graph (N stations, N-1 edges).
# We need to find the minimum cost to connect all F primary reservoirs.
# In a tree, the unique path between any two nodes is fixed.
# The set of edges that connect a subset of nodes in a tree is the union 
# of all edges on the paths between these nodes.
# This is equivalent to the Steiner Tree problem, but since the graph is a tree,
# the Steiner Tree is simply the minimal subtree containing all F nodes.
# We can solve this by identifying all edges that lie on a path between 
# any two primary reservoirs.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    reservoirs = set(map(int, input_data[3:3+F]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if F <= 1:
        print(0)
        return

    # We need to find all edges that are part of the minimal subtree.
    # An edge (u, v) with weight w is in the minimal subtree if and only if
    # the subtree rooted at u (when the edge is removed) contains at least 
    # one reservoir, and the subtree rooted at v contains at least one reservoir.
    
    is_reservoir = [False] * (N + 1)
    for r in reservoirs:
        is_reservoir[r] = True
        
    count_reservoirs = [0] * (N + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        count_reservoirs[u] += (1 if is_reservoir[u] else 0)
        if parent[u] != 0:
            count_reservoirs[parent[u]] += count_reservoirs[u]
            # If the subtree at u contains some but not all reservoirs,
            # the edge to parent must be included.
            if 0 < count_reservoirs[u] < F:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
