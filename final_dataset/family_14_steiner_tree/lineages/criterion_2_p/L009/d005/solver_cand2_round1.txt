```python
import sys

# The problem describes a tree structure (N nodes, N-1 edges).
# A closed walk visiting a set of F nodes in a tree must traverse every edge
# in the minimal subtree containing all F nodes exactly twice.
# The total energy is 2 * (sum of weights of edges in the minimal subtree).
# The problem asks for half of this value, which is the sum of weights
# of edges in the minimal subtree containing all F active sites.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    active_sites = set()
    idx = 3
    for _ in range(F):
        active_sites.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(N + 1)]
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if F <= 1:
        print(0)
        return

    # To find the sum of weights of the minimal subtree:
    # An edge (u, v) with weight w is in the minimal subtree if and only if
    # the subtree rooted at v (when considering u as parent) contains at least
    # one active site AND the rest of the tree contains at least one active site.
    
    subtree_has_active = [False] * (N + 1)
    total_active = F
    
    # Root the tree at node 1
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Post-order traversal to mark subtrees
    ans = 0
    for u in reversed(order):
        if u in active_sites:
            subtree_has_active[u] = True
        
        for v, w in adj[u]:
            if v == parent[u]:
                continue
            if subtree_has_active[v]:
                subtree_has_active[u] = True
                # If the subtree at v contains active sites, and the rest of the
                # tree also contains active sites, this edge is part of the minimal subtree.
                # The rest of the tree has active sites if total_active > count_in_v
                # We need to count active sites in subtree v.
                pass
                
    # Easier approach: count active sites in each subtree
    active_count = [0] * (N + 1)
    for u in order:
        if u in active_sites:
            active_count[u] = 1
            
    for u in reversed(order):
        if parent[u] != 0:
            active_count[parent[u]] += active_count[u]
            
    total_weight = 0
    for i in range(2, N + 1):
        # Edge between i and parent[i] is in the minimal subtree if
        # 0 < active_count[i] < total_active
        if 0 < active_count[i] < total_active:
            total_weight += edge_to_parent_weight[i]
            
    print(total_weight)

solve()
```
