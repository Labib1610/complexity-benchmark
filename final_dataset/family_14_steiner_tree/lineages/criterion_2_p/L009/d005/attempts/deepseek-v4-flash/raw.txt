We are given a tree (since E = N − 1) and need to connect all target nodes with minimum total edge cost. This is essentially finding the minimal subtree that connects all target nodes — the **Steiner tree** on a tree, which is just the union of all paths between target nodes.

A known efficient approach:

1. Mark target nodes.
2. Do a DFS to compute for each node whether its subtree contains any target node.
3. The minimal cost is the sum of all edges that lie on paths between target nodes.  
   This can be computed by summing weights of edges where both sides contain at least one target node.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    N, E, F = map(int, input().split())
    targets = set(map(int, input().split()))
    
    adj = [[] for _ in range(N)]
    edges = []
    for _ in range(E):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # DFS to count targets in subtree
    parent = [-1] * N
    order = [0]
    stack = [0]
    parent[0] = -2
    while stack:
        u = stack.pop()
        order.append(u)
        for v, _ in adj[u]:
            if parent[v] == -1:
                parent[v] = u
                stack.append(v)
    
    has_target = [False] * N
    for t in targets:
        has_target[t - 1] = True
    
    # Process in reverse order (post-order)
    total = 0
    for u in reversed(order):
        if u == 0:
            continue
        p = parent[u]
        # find weight of edge (u, p)
        w = None
        for v, wt in adj[u]:
            if v == p:
                w = wt
                break
        if has_target[u]:
            total += w
            has_target[p] = True
    
    print(total)

if __name__ == "__main__":
    solve()
```
