```python
import sys

# The problem states that we have N positions and E = N-1 operations.
# The operations form a tree structure because E = N-1 and the set is 
# "minimally comprehensive" (meaning it connects all positions).
# We need to select a subset of edges such that all critical positions 
# are in the same connected component.
# This is equivalent to finding the minimum weight set of edges that 
# connects all nodes in the set of critical positions.
# This is the Steiner Tree problem on a tree, which simplifies to 
# finding the sum of weights of all edges that lie on the unique paths 
# between any two critical nodes.
# In a tree, the union of paths between a set of nodes forms a subtree 
# that connects all these nodes. The edges in this subtree are exactly 
# those edges whose removal splits the tree into two components, 
# each containing at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+F]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    edges = []
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_size[u] will store the number of critical nodes in the subtree rooted at u
    subtree_critical_count = [0] * (N + 1)
    
    # DFS to count critical nodes in subtrees
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in critical_nodes:
        subtree_critical_count[node] = 1
        
    total_weight = 0
    # Process in reverse topological order
    for i in range(N - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        p = parent[u]
        if subtree_critical_count[u] > 0 and subtree_critical_count[u] < F:
            total_weight += edge_to_parent_weight[u]
        subtree_critical_count[p] += subtree_critical_count[u]
        
    print(total_weight)

solve()
```
