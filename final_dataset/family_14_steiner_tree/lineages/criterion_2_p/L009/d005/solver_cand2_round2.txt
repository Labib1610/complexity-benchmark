```python
import sys

# The problem states E = N - 1 and the structure is a tree.
# We need to find the minimum cost to connect all F flagship brands.
# This is equivalent to finding the sum of weights of all edges in the 
# minimal subtree that contains all F flagship brands.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    flagships = set(map(int, input_data[3:3+F]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if F <= 1:
        print(0)
        return

    # To find the minimal subtree containing all flagships:
    # 1. Root the tree at an arbitrary flagship.
    # 2. A node is part of the minimal subtree if it is a flagship 
    #    or if it has at least two branches leading to flagships.
    # 3. Alternatively, use post-order traversal:
    #    A subtree rooted at u needs to be connected if it contains any flagship.
    #    If it contains a flagship, the edge to its parent must be included.
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [next(iter(flagships))]
    visited = [False] * (N + 1)
    visited[stack[0]] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    has_flagship = [False] * (N + 1)
    for f in flagships:
        has_flagship[f] = True
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if has_flagship[u]:
            if u != order[0]: # Not the root
                total_cost += edge_to_parent_weight[u]
                has_flagship[parent[u]] = True
                
    print(total_cost)

solve()
```
