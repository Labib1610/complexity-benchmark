```python
import sys

# The problem states E = N - 1, which means the graph is a tree.
# To connect a subset of F target registers in a tree, we need to include
# all edges that lie on the paths between any two target registers.
# This is equivalent to finding the Steiner Tree in a tree, which is simply
# the union of all paths between the target registers.
# In a tree, this is the minimal subtree containing all target registers.
# An edge (u, v) with weight w is part of this subtree if and only if
# the removal of this edge splits the tree into two components, each of which
# contains at least one target register.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+F]))
    
    adj = [[] for _ in range(N + 1)]
    edges = []
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many target registers are in each subtree formed by removing an edge.
    # We can root the tree at an arbitrary node (e.g., 1).
    # For each node u, count how many targets are in its subtree.
    
    target_count = [0] * (N + 1)
    for t in targets:
        target_count[t] = 1
        
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to propagate target counts
    total_cost = 0
    for u in reversed(order):
        if u == 1:
            continue
        
        # If the subtree at u contains some targets but not all,
        # the edge connecting u to its parent must be included.
        count = target_count[u]
        if 0 < count < F:
            total_cost += edge_to_parent_weight[u]
            
        # Propagate count to parent
        if parent[u] != 0:
            target_count[parent[u]] += count
            
    print(total_cost)

solve()
```
