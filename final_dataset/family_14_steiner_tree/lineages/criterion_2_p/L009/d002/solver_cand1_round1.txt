```python
import sys

# The problem states the network is a tree (N stations, N-1 edges, connected).
# We need to connect a subset of F hubs. The minimum cost to connect a set of nodes
# in a tree is the sum of the weights of all edges on the unique paths between them.
# This is equivalent to the sum of weights of all edges that have at least one hub
# in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    hubs = set()
    idx = 3
    for _ in range(F):
        hubs.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(N + 1)]
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if F <= 1:
        print(0)
        return

    # Use DFS to count how many hubs are in the subtree of each node
    # Since N is up to 200,000, use iterative DFS to avoid recursion depth issues
    hub_count = [0] * (N + 1)
    parent = [0] * (N + 1)
    edge_weight_to_parent = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for node in order:
        if node in hubs:
            hub_count[node] = 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(N - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            hub_count[p] += hub_count[u]
            # If this edge separates some hubs from others, it must be reinforced
            if hub_count[u] > 0 and hub_count[u] < F:
                total_cost += edge_weight_to_parent[u]
                
    print(total_cost)

solve()
```
