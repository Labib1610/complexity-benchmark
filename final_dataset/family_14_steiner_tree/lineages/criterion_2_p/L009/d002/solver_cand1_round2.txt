```python
import sys

# The problem asks for the minimum cost to connect a set of F freight hubs in a tree.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of nodes S, we must include all edges that lie on the unique
# paths between any pair of nodes in S.
# This is equivalent to the Steiner Tree problem on a tree, which is simply
# the union of all paths between the nodes in S.
# The edges that must be included are those that have at least one hub in
# one of the two components formed by removing the edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    hubs = set(map(int, input_data[3:3+F]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    edges = []
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if F <= 1:
        print(0)
        return

    # subtree_hubs[u] stores the number of hubs in the subtree rooted at u
    subtree_hubs = [0] * (N + 1)
    
    # DFS to count hubs in subtrees
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in order:
        if u in hubs:
            subtree_hubs[u] += 1
            
    # Process in reverse topological order to propagate hub counts
    for i in range(N - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            subtree_hubs[parent[u]] += subtree_hubs[u]
            
    total_cost = 0
    for u, v, w in edges:
        # Edge (u, v) splits the tree into two components.
        # One component has subtree_hubs[child] hubs, the other has F - subtree_hubs[child].
        # If both are > 0, the edge must be included.
        child = u if parent[v] == u else v
        count = subtree_hubs[child]
        if count > 0 and count < F:
            total_cost += w
            
    print(total_cost)

solve()
```
