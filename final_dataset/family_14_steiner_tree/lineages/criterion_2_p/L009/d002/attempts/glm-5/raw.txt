```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    N = int(data[idx]); idx += 1
    E = int(data[idx]); idx += 1
    F = int(data[idx]); idx += 1

    critical = [False] * (N + 1)
    for _ in range(F):
        c = int(data[idx]); idx += 1
        critical[c] = True

    adj = [[] for _ in range(N + 1)]
    for _ in range(E):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))

    if F <= 1:
        print(0)
        return

    parent = [0] * (N + 1)
    parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                parent_weight[v] = w
                stack.append(v)

    cnt = [0] * (N + 1)
    for i in range(N, 0, -1):
        u = order[i - 1]
        if critical[u]:
            cnt[u] += 1
        p = parent[u]
        if p != -1:
            cnt[p] += cnt[u]

    ans = 0
    for u in range(2, N + 1):
        if cnt[u] > 0 and cnt[u] < F:
            ans += parent_weight[u]

    print(ans)

solve()
```
