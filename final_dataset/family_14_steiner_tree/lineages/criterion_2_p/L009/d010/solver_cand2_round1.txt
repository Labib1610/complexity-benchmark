import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    critical_nodes = []
    idx = 3
    for _ in range(F):
        critical_nodes.append(int(input_data[idx]))
        idx += 1
        
    if F <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    # The condition "f(a) = f(b) for all a, b in Y" means that in the graph
    # formed by the selected comparators, all nodes in Y must be in the same
    # connected component.
    # We can contract all nodes in Y into a single super-node and find the
    # Minimum Spanning Tree of the resulting graph.
    
    parent = list(range(N + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # Contract all critical nodes into one
    first_critical = critical_nodes[0]
    for i in range(1, F):
        union(first_critical, critical_nodes[i])
        
    edges.sort()
    
    total_cost = 0
    edges_count = 0
    for w, u, v in edges:
        if union(u, v):
            total_cost += w
            edges_count += 1
            
    print(total_cost)

if __name__ == '__main__':
    solve()
