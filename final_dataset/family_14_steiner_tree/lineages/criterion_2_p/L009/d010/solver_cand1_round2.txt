```python
import sys

# The problem describes a graph where each update (u, v) with weight w
# corresponds to an edge in a graph. The matrix M is the Laplacian matrix
# of the graph formed by the selected edges.
# The null space of the Laplacian matrix is spanned by the indicator vectors
# of the connected components of the graph.
# The condition "x_a = x_b for all target indices a and b" means that
# all target indices must belong to the same connected component.
# We are given E = N - 1 edges, and the full graph is connected.
# We want to select a subset of edges such that all target indices are
# in the same connected component, minimizing the total cost.
# This is equivalent to finding the minimum cost to connect all target
# nodes in a graph where we can choose to either include an edge (cost w)
# or exclude it (cost 0).
# This is the Steiner Tree problem, but since the graph is a tree (E = N-1),
# the only way to connect a set of nodes is to include all edges on the
# unique paths between them.
# The union of all paths between target nodes forms a subtree.
# We need to sum the weights of all edges that lie on the path between
# any two target nodes. An edge lies on such a path if and only if
# removing it splits the tree into two components, each containing at
# least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+F]))
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many targets are in the subtree of each node
    target_count = [0] * (N + 1)
    for t in targets:
        target_count[t] = 1
        
    total_cost = 0
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_weight = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if parent[u] != 0:
            target_count[parent[u]] += target_count[u]
            if 0 < target_count[u] < F:
                total_cost += edge_weight[u]
                
    print(total_cost)

solve()
```
