```python
import sys

# The problem states the network is a tree (N nodes, N-1 edges, connected).
# We need to find the minimum weight subtree that connects all F injection sites.
# This is equivalent to the Steiner Tree problem on a tree, which is simply
# the sum of weights of all edges that lie on the unique path between any two
# injection sites. An edge belongs to this set if and only if the subtree
# formed by removing that edge contains at least one injection site in both
# resulting components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    injection_sites = set()
    idx = 3
    for _ in range(F):
        injection_sites.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(N + 1)]
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many injection sites are in the subtree rooted at u
    # when the tree is rooted at node 1.
    subtree_injection_count = [0] * (N + 1)
    parent_edge_weight = [0] * (N + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [1]
    parent = [0] * (N + 1)
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in injection_sites:
            subtree_injection_count[u] += 1
            
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for i in range(N - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_injection_count[p] += subtree_injection_count[u]
            # If the subtree at u contains some but not all injection sites,
            # the edge (u, p) must be part of the Steiner Tree.
            if 0 < subtree_injection_count[u] < F:
                total_weight += parent_edge_weight[u]
                
    print(total_weight)

if __name__ == '__main__':
    solve()
```
