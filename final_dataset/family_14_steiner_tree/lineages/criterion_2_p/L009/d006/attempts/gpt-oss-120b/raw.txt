```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    N = int(next(it))
    E = int(next(it))  # equals N-1
    F = int(next(it))
    vital = [0] * (N + 1)
    for _ in range(F):
        vital[int(next(it))] = 1

    adj = [[] for _ in range(N + 1)]
    edges = []
    for _ in range(E):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    if F <= 1:
        print(0)
        return

    parent = [0] * (N + 1)
    pw = [0] * (N + 1)          # weight to parent
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        node = stack.pop()
        order.append(node)
        for nb, w in adj[node]:
            if nb == parent[node]:
                continue
            parent[nb] = node
            pw[nb] = w
            stack.append(nb)

    cnt = [0] * (N + 1)
    for i in range(1, N + 1):
        cnt[i] = vital[i]

    total = 0
    for node in reversed(order):
        if parent[node] == -1:
            continue
        sub = cnt[node]
        if 0 < sub < F:
            total += pw[node]
        cnt[parent[node]] += sub

    print(total)

if __name__ == "__main__":
    solve()
```
