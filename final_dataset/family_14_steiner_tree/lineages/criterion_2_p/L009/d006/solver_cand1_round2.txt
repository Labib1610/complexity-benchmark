import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    vital_chambers = set(map(int, input_data[3:3+F]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the smallest subtree that contains all vital chambers.
    # This is equivalent to the sum of weights of all edges that lie on the 
    # unique path between any two vital chambers.
    # An edge is part of this subtree if and only if the subtree formed by 
    # removing that edge has vital chambers in both components.
    
    is_vital = [False] * (N + 1)
    for v in vital_chambers:
        is_vital[v] = True
        
    vital_count_subtree = [0] * (N + 1)
    total_effort = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order to count vital nodes in subtrees
    edge_weights = {}
    for u in reversed(order):
        if is_vital[u]:
            vital_count_subtree[u] += 1
        
        if parent[u] != -1:
            p = parent[u]
            vital_count_subtree[p] += vital_count_subtree[u]
            
            # Find weight of edge (u, p)
            for v, w in adj[u]:
                if v == p:
                    # If this edge separates the tree into two parts, 
                    # both containing at least one vital node, it must be included.
                    if 0 < vital_count_subtree[u] < F:
                        total_effort += w
                    break
                    
    print(total_effort)

solve()
