```python
import sys

# The problem asks for the minimum weight of a subset of edges in a tree 
# such that all F critical nodes are connected.
# Since the graph is a tree (E = N - 1 and it's connected), there is only 
# one unique path between any two nodes.
# The union of all paths between any two critical nodes forms a subtree.
# The edges that must be included are exactly those that lie on the path 
# between at least one pair of critical nodes.
# An edge (u, v) with weight w is part of this subtree if and only if 
# removing it splits the tree into two components, each containing at least 
# one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+F]))
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many critical nodes are in the subtree rooted at each node
    # when rooted at an arbitrary node (e.g., 1).
    critical_count = [0] * (N + 1)
    for node in critical_nodes:
        critical_count[node] = 1
        
    parent = [0] * (N + 1)
    edge_weight_to_parent = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    total_weight = 0
    # Process in reverse topological order to propagate counts
    for i in range(N - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            critical_count[p] += critical_count[u]
            # If the subtree at u contains some but not all critical nodes,
            # the edge to parent must be included.
            if 0 < critical_count[u] < F:
                total_weight += edge_weight_to_parent[u]
                
    print(total_weight)

solve()
```
