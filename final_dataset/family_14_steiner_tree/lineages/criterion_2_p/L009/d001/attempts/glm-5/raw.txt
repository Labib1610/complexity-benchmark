```python
import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    N = int(input_data[idx]); idx += 1
    E = int(input_data[idx]); idx += 1
    F = int(input_data[idx]); idx += 1

    is_hub = [False] * (N + 1)
    for _ in range(F):
        h = int(input_data[idx]); idx += 1
        is_hub[h] = True

    adj = [[] for _ in range(N + 1)]
    for _ in range(E):
        u = int(input_data[idx]); idx += 1
        v = int(input_data[idx]); idx += 1
        w = int(input_data[idx]); idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))

    if F <= 1:
        print(0)
        return

    setrecursionlimit(1 << 25)
    parent = [0] * (N + 1)
    parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    parent[1] = -1

    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                parent_weight[v] = w
                stack.append(v)

    dp = [0] * (N + 1)
    cnt = [0] * (N + 1)

    for u in reversed(order):
        c = 1 if is_hub[u] else 0
        best = 0
        for v, w in adj[u]:
            if parent[v] == u:
                c += cnt[v]
                if cnt[v] > 0:
                    if best == 0 or w < best:
                        best = w
                dp[u] += dp[v]
        cnt[u] = c
        if c < F:
            dp[u] += best

    print(dp[1])

solve()
```
