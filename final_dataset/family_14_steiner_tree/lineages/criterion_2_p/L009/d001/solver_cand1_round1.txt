```python
import sys

# The problem states E = N - 1, which means the graph is a tree.
# To connect a subset of nodes (freight hubs) in a tree, we must include
# all edges that lie on the unique paths between any two freight hubs.
# This is equivalent to the Steiner Tree problem on a tree, which is simply
# the sum of weights of all edges that have at least one freight hub in
# each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    hubs = set()
    idx = 3
    for _ in range(F):
        hubs.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(N + 1)]
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if F <= 1:
        print(0)
        return

    # subtree_hubs[u] stores the number of freight hubs in the subtree rooted at u
    subtree_hubs = [0] * (N + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u in hubs:
            subtree_hubs[u] += 1
        
        for v, w in adj[u]:
            if v != parent[u]:
                subtree_hubs[u] += subtree_hubs[v]
                # If this edge separates some hubs from others, it must be included
                if subtree_hubs[v] > 0 and subtree_hubs[v] < F:
                    total_cost += w
                    
    print(total_cost)

solve()
```
