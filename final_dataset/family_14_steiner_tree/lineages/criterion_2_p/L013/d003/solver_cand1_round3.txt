```python
import sys

# The problem states that the network of translators forms a tree (n nodes, n-1 edges, 
# and for any two languages there is exactly one path).
# We need to find the minimum cost to connect a subset of t priority languages.
# This is equivalent to finding the weight of the minimal subtree that contains 
# all t priority languages.
# This subtree consists of all edges that lie on the simple path between any two 
# priority languages.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_languages = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to identify which edges are part of the minimal subtree.
    # An edge is in the subtree if and only if both components formed by 
    # removing the edge contain at least one priority language.
    
    # Root the tree at an arbitrary priority language
    root = next(iter(priority_languages))
    
    # subtree_priority_count[u] stores the number of priority languages in the subtree of u
    subtree_priority_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(root, -1, 0)] # node, parent, weight_to_parent
    order = []
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    
    dfs_stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while dfs_stack:
        u = dfs_stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                weight_to_parent[v] = w
                dfs_stack.append(v)
                
    for u in reversed(order):
        if u in priority_languages:
            subtree_priority_count[u] += 1
        if parent[u] != 0:
            subtree_priority_count[parent[u]] += subtree_priority_count[u]
            # If this edge separates the tree into two parts both containing 
            # priority languages, it must be included.
            if 0 < subtree_priority_count[u] < t:
                total_cost += weight_to_parent[u]
                
    print(total_cost)

solve()
```
