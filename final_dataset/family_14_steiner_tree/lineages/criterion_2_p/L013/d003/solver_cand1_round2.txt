import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    priority_nodes = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that lie on the path between any two priority nodes.
    # This is equivalent to finding the minimal subtree containing all priority nodes.
    # An edge is in this subtree if and only if the subtree formed by the edge
    # splits the set of priority nodes into two non-empty sets.
    
    is_priority = [False] * (n + 1)
    for p in priority_nodes:
        is_priority[p] = True
        
    count_priority = [0] * (n + 1)
    
    # DFS to count priority nodes in subtree
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_priority[u]:
            count_priority[u] += 1
        if parent[u] != 0:
            count_priority[parent[u]] += count_priority[u]
            
    total_fee = 0
    for i in range(2, n + 1):
        # If the subtree at i contains some but not all priority nodes,
        # the edge connecting i to its parent is part of the minimal subtree.
        if 0 < count_priority[i] < t:
            total_fee += edge_weight[i]
            
    print(total_fee)

solve()
