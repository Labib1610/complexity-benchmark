```python
import sys

# The problem states that the pool of translators is non-redundant and 
# connects n languages with m = n-1 translators. This means the graph 
# is a tree.
# We need to find the minimum cost to connect all t priority languages.
# In a tree, the set of edges required to connect a subset of nodes is 
# exactly the set of edges that lie on the paths between any two nodes 
# in that subset. This is equivalent to the union of all paths between 
# the priority nodes, which forms a subtree.
# We can solve this by rooting the tree arbitrarily (e.g., at node 1) 
# and checking for each edge whether it lies on a path between two 
# priority nodes. An edge (u, v) (where v is a child of u) is part of 
# the required subtree if and only if the subtree rooted at v contains 
# at least one priority node AND the rest of the tree (outside the 
# subtree rooted at v) also contains at least one priority node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_nodes = set()
    idx = 3
    for _ in range(t):
        priority_nodes.add(int(input_data[idx]))
        idx += 1
        
    if t <= 1:
        print(0)
        return
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # subtree_priority_count[u] stores how many priority nodes are in u's subtree
    subtree_priority_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in priority_nodes:
            subtree_priority_count[u] += 1
        if parent[u] != 0:
            subtree_priority_count[parent[u]] += subtree_priority_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # Edge connects i and parent[i].
        # It is needed if the subtree at i has some priority nodes,
        # and the rest of the tree also has some priority nodes.
        count_in_subtree = subtree_priority_count[i]
        if count_in_subtree > 0 and count_in_subtree < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

solve()
```
