```python
import sys

# The problem states the network is minimally connected (a tree).
# We need to find the minimum cost to connect all t priority languages.
# In a tree, the set of edges required to connect a subset of nodes is
# exactly the set of edges that lie on the paths between any two nodes
# in the subset. This is equivalent to the union of all paths between
# the priority nodes, which forms the minimal subtree containing all
# priority nodes.
# We can solve this by rooting the tree at an arbitrary node (e.g., node 1)
# and calculating for each edge whether it is part of the minimal subtree.
# An edge (u, v) where v is a child of u is part of the subtree if and only
# if the subtree rooted at v contains at least one priority node AND
# the rest of the tree (outside the subtree of v) contains at least one
# priority node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # DFS to compute subtree priority counts
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    subtree_priority_count = [0] * (n + 1)
    for node in priority_nodes:
        subtree_priority_count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_priority_count[p] += subtree_priority_count[u]
            # If the subtree at u has some priority nodes, and the rest of the
            # tree also has some, this edge must be included.
            if 0 < subtree_priority_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
