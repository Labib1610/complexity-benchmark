We can model the languages and translators as a tree.  
The problem reduces to finding the minimum total edge weight needed to connect all target nodes in the tree.  

This is equivalent to finding the **minimum Steiner tree** for the given set of target nodes in a tree, which can be solved by:

1. Marking all target nodes.
2. Removing all subtrees that contain no target nodes.
3. Summing the weights of the remaining edges.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    targets = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # DFS to mark which nodes are needed
    needed = [False] * (n + 1)
    for x in targets:
        needed[x] = True
    
    parent = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
    
    # Post-order: propagate needed status
    for u in reversed(order):
        for v, w in adj[u]:
            if v != parent[u]:
                if needed[v]:
                    needed[u] = True
    
    total = 0
    for u in range(2, n + 1):
        if needed[u]:
            total += parent_edge_weight[u]
    
    print(total)

if __name__ == "__main__":
    solve()
```
