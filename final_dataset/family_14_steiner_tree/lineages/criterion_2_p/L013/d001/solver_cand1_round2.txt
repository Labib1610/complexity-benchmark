```python
import sys

# The problem states that the n workstations and n-1 links form a single 
# infrastructure, which means the graph is a tree.
# We need to find the minimum cost to connect all critical workstations.
# In a tree, there is a unique path between any two nodes.
# To connect a set of critical nodes, we must include all edges that lie 
# on the paths between any pair of critical nodes.
# This is equivalent to finding the smallest subtree that contains all 
# critical nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical = set()
    idx = 3
    for _ in range(t):
        critical.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to identify all edges that are part of the minimal subtree.
    # An edge is part of the minimal subtree if and only if both components 
    # created by removing that edge contain at least one critical node.
    
    # Root the tree at an arbitrary critical node (e.g., the first one)
    root = next(iter(critical))
    
    # count_crit[u] will store the number of critical nodes in the subtree of u
    count_crit = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in critical:
        count_crit[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == root:
            continue
        
        # If the subtree at u contains at least one critical node and 
        # the rest of the tree contains at least one critical node, 
        # the edge to the parent must be included.
        if 0 < count_crit[u] < t:
            total_cost += parent_edge_weight[u]
            
        count_crit[parent[u]] += count_crit[u]
        
    print(total_cost)

solve()
```
