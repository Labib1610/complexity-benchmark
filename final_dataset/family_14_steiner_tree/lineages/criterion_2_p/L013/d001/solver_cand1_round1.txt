```python
import sys

# The problem states that the facility is physically interconnected and 
# the number of links m is exactly n-1. This means the network is a tree.
# Since it's a tree, there is exactly one unique path between any two nodes.
# To connect all critical workstations, we must activate all links that lie 
# on the paths between any two critical workstations.
# This is equivalent to finding the smallest subtree that contains all 
# critical nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find all edges that are part of the unique path between 
    # any two critical nodes. In a tree, an edge (u, v) is part of the 
    # path between some critical nodes if and only if the subtree 
    # formed by removing the edge contains at least one critical node 
    # on both sides.
    
    # Root the tree at an arbitrary critical node (e.g., the first one)
    root = next(iter(critical_nodes))
    
    # count_critical[u] = number of critical nodes in the subtree rooted at u
    count_critical = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for node in critical_nodes:
        count_critical[node] = 1
        
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if count_critical[u] > 0 and count_critical[u] < t:
            total_cost += edge_weight_to_parent[u]
        count_critical[p] += count_critical[u]
        
    print(total_cost)

solve()
```
