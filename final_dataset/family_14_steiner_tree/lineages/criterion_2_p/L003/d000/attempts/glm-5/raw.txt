import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    is_priority = [False] * (n + 1)
    for _ in range(t):
        p = int(data[idx]); idx += 1
        is_priority[p] = True
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                parent_weight[v] = w
                stack.append(v)
    
    count = [0] * (n + 1)
    for u in reversed(order):
        if is_priority[u]:
            count[u] += 1
        p = parent[u]
        if p != -1:
            count[p] += count[u]
    
    total_cost = 0
    for u in range(2, n + 1):
        if count[u] > 0:
            total_cost += parent_weight[u]
    
    print(total_cost)

solve()
