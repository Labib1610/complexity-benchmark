```python
import sys

# The problem states that the m = n - 1 links form a tree (since they connect n nodes).
# We need to find the sum of weights of all edges that lie on the unique path 
# between any two priority modules.
# An edge is part of such a path if and only if removing it splits the tree 
# into two components, both of which contain at least one priority module.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_modules = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many priority modules are in the subtree of each node
    # We root the tree at node 1
    subtree_priority_count = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [(1, 0)]
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        parent[u] = p
        for v, w in adj[u]:
            if v != p:
                edge_to_parent_weight[v] = w
                stack.append((v, u))
                
    for u in order:
        if u in priority_modules:
            subtree_priority_count[u] += 1
            
    # Process in reverse topological order to propagate counts up
    total_cost = 0
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        count = subtree_priority_count[u]
        
        # If the subtree at u contains some priority modules but not all,
        # the edge (u, parent[u]) is indispensable.
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[u]
            
        subtree_priority_count[p] += count
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
