```python
import sys

# The problem states the network is a tree (n servers, n-1 connections, connected).
# We need to find the minimum total length of fiber to connect all critical databases.
# In a tree, the path between any two nodes is unique.
# The set of edges required to connect a subset of nodes is the union of all paths
# between every pair of critical nodes. This is equivalent to the edges in the
# minimal subtree containing all critical nodes.
# An edge is part of this subtree if and only if the subtree formed by removing
# that edge splits the critical nodes into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many critical nodes are in the subtree of each node.
    # Since it's a tree, we can use DFS.
    # To avoid recursion depth issues, use iterative DFS.
    
    is_critical = [0] * (n + 1)
    for node in critical_nodes:
        is_critical[node] = 1
        
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    critical_count = [0] * (n + 1)
    for i in range(1, n + 1):
        critical_count[i] = is_critical[i]
        
    total_fiber = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            critical_count[p] += critical_count[u]
            # If this subtree contains some but not all critical nodes,
            # the edge to the parent must be included.
            if 0 < critical_count[u] < t:
                total_fiber += edge_weight_to_parent[u]
                
    print(total_fiber)

if __name__ == '__main__':
    # Increase recursion depth just in case, though iterative is used
    sys.setrecursionlimit(300000)
    solve()
```
