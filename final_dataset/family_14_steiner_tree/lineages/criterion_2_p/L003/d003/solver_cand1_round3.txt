```python
import sys

# The problem states that there are n nodes and m = n - 1 links, 
# and the graph is fully connected. This means the graph is a tree.
# To connect all priority nodes in a tree, we must include all edges 
# that lie on the unique simple paths between any two priority nodes.
# This is equivalent to finding the minimal subtree that contains all 
# priority nodes (the Steiner Tree in a tree).
# The edges to include are exactly those that have at least one priority 
# node in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_nodes = set()
    idx = 3
    for _ in range(t):
        priority_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many priority nodes are in the subtree of each node
    # when rooted at node 1.
    subtree_priority_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in priority_nodes:
            subtree_priority_count[node] += 1
            
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_priority_count[p] += subtree_priority_count[u]
            # If this edge separates some priority nodes from others, it must be included
            if 0 < subtree_priority_count[u] < t:
                total_cost += parent_edge_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
