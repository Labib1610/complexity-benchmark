```python
import sys

# The problem states that m = n - 1 and the graph is a tree.
# If the graph is a tree, there is exactly one path between any two nodes.
# To connect all priority reaction chambers, we must include all edges that lie
# on the unique paths between any two priority reaction chambers.
# This is equivalent to finding the smallest subtree that contains all priority nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Root the tree at the first priority node
    root = next(iter(priority_nodes))
    
    # We need to identify which edges are part of the minimal subtree.
    # An edge (u, v) is in the subtree if and only if the subtree rooted at the
    # child node contains at least one priority node and the rest of the tree
    # (the part not in the child's subtree) also contains at least one priority node.
    
    has_priority = [False] * (n + 1)
    for p in priority_nodes:
        has_priority[p] = True
        
    total_priority = t
    subtree_priority_count = [0] * (n + 1)
    edge_costs = {}
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [root]
    parent[root] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_costs[v] = w
                stack.append(v)
                
    ans = 0
    for u in reversed(order):
        subtree_priority_count[u] += (1 if has_priority[u] else 0)
        if u != root:
            p = parent[u]
            subtree_priority_count[p] += subtree_priority_count[u]
            # If this subtree contains some but not all priority nodes,
            # the edge connecting it to its parent must be included.
            if 0 < subtree_priority_count[u] < total_priority:
                ans += edge_costs[u]
                
    print(ans)

solve()
```
