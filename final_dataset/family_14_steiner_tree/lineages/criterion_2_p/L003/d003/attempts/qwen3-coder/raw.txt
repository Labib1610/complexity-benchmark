```python
import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    critical = set(map(int, sys.stdin.readline().split()))
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    # Since m = n-1 and the graph is connected, it's a tree
    # We need to find the minimum cost to connect all critical nodes
    # This is essentially finding the Minimum Spanning Tree of the critical nodes
    # In a tree, to connect k nodes, we need exactly k-1 edges
    # We can use Kruskal's algorithm but only consider edges that connect different components containing critical nodes
    
    # However, since the input graph is already a tree, we can think differently:
    # We need to find the minimal subtree that spans all critical nodes
    # This is the Steiner tree problem on a tree, which can be solved efficiently
    
    # Build the tree
    from collections import defaultdict
    adj = defaultdict(list)
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # If there's only one critical node, cost is 0
    if t <= 1:
        print(0)
        return
    
    # Find the minimal subtree that connects all critical nodes
    # For a tree, we can do this by:
    # 1. Find all leaves that are not critical
    # 2. Remove them iteratively until only the minimal subtree remains
    # 3. Sum the weights of the remaining edges
    
    # But a simpler approach for a tree:
    # The minimal cost is the total weight of edges in the subtree that connects all critical nodes
    # We can find this by running a modified DFS
    
    # Let's use the approach of finding the subtree that connects all critical nodes
    # We'll mark all nodes that are critical or are on paths between critical nodes
    
    # First, let's find all pairwise paths between critical nodes and mark edges used
    # But that's complex. Better approach:
    
    # Since it's a tree, we can find the "Steiner tree" by:
    # 1. Root the tree at any node
    # 2. For each node, compute if its subtree contains any critical node
    # 3. An edge should be included if both its endpoints' subtrees contain critical nodes
    
    # Let's root the tree at node 1
    parent = [-1] * (n + 1)
    children = defaultdict(list)
    edge_weight = {}
    
    def dfs_root(v, par):
        parent[v] = par
        for u, w in adj[v]:
            if u != par:
                children[v].append(u)
                edge_weight[(v, u)] = w
                edge_weight[(u, v)] = w
                dfs_root(u, v)
    
    dfs_root(1, -1)
    
    # For each node, compute if its subtree contains any critical node
    contains_critical = [False] * (n + 1)
    
    def dfs_critical(v):
        if v in critical:
            contains_critical[v] = True
        for u in children[v]:
            dfs_critical(u)
            if contains_critical[u]:
                contains_critical[v] = True
    
    dfs_critical(1)
    
    # Now, count the cost of edges where both endpoints' subtrees contain critical nodes
    total_cost = 0
    
    def dfs_count(v):
        nonlocal total_cost
        for u in children[v]:
            if contains_critical[u]:  # This subtree has critical nodes
                # Check if removing this edge would disconnect critical nodes
                # Since u's subtree contains critical nodes, and if v (or v's other subtrees) also contain critical nodes
                # Then this edge is needed
                # But we need to be more precise
                # An edge (v,u) is needed if:
                # 1. u's subtree contains at least one critical node
                # 2. The rest of the tree (including v's parent and v's other children) contains at least one critical node
                total_cost += edge_weight[(v, u)]
            dfs_count(u)
    
    # A better way: 
    # An edge (parent[v], v) is in the Steiner tree if and only if:
    # 1. v's subtree contains at least one critical node
    # 2. The rest of the tree (V - v's subtree) contains at least one critical node
    
    # Let's count total critical nodes
    total_critical = len(critical)
    
    # For each node, count critical nodes in its subtree
    crit_count = [0] * (n + 1)
    
    def dfs_count_crit(v):
        if v in critical:
            crit_count[v] = 1
        for u in children[v]:
            dfs_count_crit(u)
            crit_count[v] += crit_count[u]
    
    dfs_count_crit(1)
    
    # Now, for each edge (parent[v], v), check if it should be included
    total_cost = 0
    
    def dfs_edges(v):
        nonlocal total_cost
        for u in children[v]:
            # Edge (v, u)
            # crit_count[u] = number of critical nodes in u's subtree
            # total_critical - crit_count[u] = number of critical nodes outside u's subtree
            if crit_count[u] > 0 and (total_critical - crit_count[u]) > 0:
                total_cost += edge_weight[(v, u)]
            dfs_edges(u)
    
    dfs_edges(1)
    
    print(total_cost)

threading.Thread(target=main).start()
```
