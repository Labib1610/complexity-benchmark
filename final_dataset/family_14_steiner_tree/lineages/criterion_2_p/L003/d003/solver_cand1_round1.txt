```python
import sys

# The problem states m = n - 1 and the graph is connected.
# This means the graph is a tree.
# We need to find the minimum cost to connect all critical processors.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of nodes in a tree, we must include all edges that lie
# on the paths between any pair of critical nodes.
# An edge is part of the required set if and only if it splits the tree
# into two components, each containing at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many critical nodes are in the subtree of each node
    # when rooted at 1.
    subtree_critical = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in critical_nodes:
        subtree_critical[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if subtree_critical[u] > 0 and subtree_critical[u] < t:
            total_cost += parent_edge_weight[u]
        subtree_critical[p] += subtree_critical[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
