```python
import sys

# The problem states that the schematic is a tree (n nodes, n-1 edges, connected).
# We need to find the minimum cost to connect all priority chambers.
# In a tree, the set of edges that connect a subset of nodes is the union of 
# all paths between any two nodes in that subset.
# This is equivalent to the edges in the minimal subtree containing all priority chambers.
# An edge is part of this subtree if and only if the two components it separates 
# both contain at least one priority chamber.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_chambers = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many priority chambers are in the subtree rooted at each node
    # when the tree is rooted at node 1.
    subtree_priority_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in priority_chambers:
            subtree_priority_count[node] += 1
            
    total_cost = 0
    # Process in reverse topological order to propagate counts up
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        subtree_priority_count[p] += subtree_priority_count[u]
        
        # If the subtree at u contains some but not all priority chambers,
        # the edge connecting u to its parent must be included.
        if 0 < subtree_priority_count[u] < t:
            total_cost += parent_edge_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    # Increase recursion depth for deep trees
    sys.setrecursionlimit(300000)
    solve()
```
