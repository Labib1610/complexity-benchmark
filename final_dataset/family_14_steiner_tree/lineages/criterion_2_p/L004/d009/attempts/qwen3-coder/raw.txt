import sys
from collections import defaultdict

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def main():

        idx += 1

        idx += 1
        w = int(data[idx])

    parent = list(range(n + 1))

            u_root = find(parent, u)

            priority_in_v = v_is_priority or any(x in priority and find(parent, x) == v_root for x in list(priority))
            
            union(parent, rank, u, v)
            
            new_root = find(parent, u)
            # Check if we just connected two components that had priority nodes
            if priority_in_u and priority_in_v:
                total_cost += w
            elif priority_in_u or priority_in_v:
                total_cost += w
        # Early termination if all priority nodes are connected
        # This is tricky to check, so we'll just go through all edges
    
    # Alternative approach: build MST and then find cost to connect priority nodes
    # Let's use a different method:
    # Build the MST, then find the minimum cost to connect all priority nodes
    
    # Actually, let's think differently:
    # We want to connect all priority nodes with minimum cost
    # This is a Steiner tree problem, but since m=n-1 and the graph is a tree,

    # In a tree, to connect a subset of nodes, we need the edges in the subtree
    # that spans all those nodes
    
    # Build adjacency list representation of the tree
    adj = defaultdict(list)
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))

    if t <= 1:

    # Find the minimal subtree that connects all priority nodes
    # This is the Steiner tree for a tree, which can be found by:
    # 1. Finding all nodes that are on paths between priority nodes
    # 2. Taking all edges in the subtree induced by these nodes
    
    # Mark all priority nodes
    is_priority = [False] * (n + 1)
    for p in priority:
        is_priority[p] = True
    
    # DFS to find the minimal subtree
    visited = [False] * (n + 1)
    subtree_nodes = set()
    
    def dfs(u, parent_node):
        visited[u] = True
        has_priority_descendant = is_priority[u]
        for v, w in adj[u]:
            if v == parent_node:
                continue
            if not visited[v]:
                child_has_priority = dfs(v, u)
                if child_has_priority:
                    has_priority_descendant = True
                    subtree_nodes.add(u)
                    subtree_nodes.add(v)
        if has_priority_descendant:
            subtree_nodes.add(u)
        return has_priority_descendant
    
    # Start DFS from any node, but to be safe, start from a priority node if exists

    dfs(start_node, -1)
    
    # Now compute the total cost of edges in the subtree
    visited_edges = set()
    total_cost = 0

            if v == parent_node:
                continue
            edge_key = tuple(sorted((u, v)))
            if edge_key not in visited_edges and v in subtree_nodes:
                visited_edges.add(edge_key)
                total_cost += w
                compute_cost(v, u)
    
    if subtree_nodes:
        start_subtree = next(iter(subtree_nodes))
        compute_cost(start_subtree, -1)
    
    print(total_cost)
