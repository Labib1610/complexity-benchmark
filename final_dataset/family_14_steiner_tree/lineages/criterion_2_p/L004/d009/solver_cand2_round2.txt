```python
import sys

# The problem asks for the minimum cost to connect a set of t critical nodes in a tree.
# Since the graph is a tree (n nodes, n-1 edges, connected), the unique path between
# any two nodes is the only way to connect them.
# To connect all t critical nodes, we must include all edges that lie on the paths
# between any two critical nodes.
# This is equivalent to finding the smallest subtree that contains all t critical nodes.
# An edge (u, v) with weight w is part of this subtree if and only if the removal
# of this edge splits the tree into two components, each containing at least one
# critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # count[u] will store the number of critical nodes in the subtree rooted at u
    count = [0] * (n + 1)
    
    # DFS to compute counts
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in critical_nodes:
            count[u] += 1
        if parent[u] != 0:
            count[parent[u]] += count[u]
            
    total_cost = 0
    for u, v, w in edges:
        # The edge connects u and v. One is the parent of the other.
        child = u if parent[u] == v else v
        # If the subtree at 'child' contains some but not all critical nodes,
        # the edge must be part of the minimal connecting subtree.
        c = count[child]
        if c > 0 and c < t:
            total_cost += w
            
    print(total_cost)

solve()
```
