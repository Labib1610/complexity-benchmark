```python
import sys

# The problem asks for the minimum cost to connect a set of target nodes in a graph
# where the edges are the given transpositions. Since the graph is a tree (m = n-1),
# the minimum cost to connect a subset of nodes is the sum of the weights of all
# edges that lie on the unique paths between the target nodes.
# An edge lies on the path between some pair of target nodes if and only if
# removing that edge splits the tree into two components, each containing at least
# one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_targets[u] will store the number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in targets:
        subtree_targets[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            p = parent[u]
            subtree_targets[p] += subtree_targets[u]
            # If this edge separates the tree into two parts both containing targets
            if 0 < subtree_targets[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
