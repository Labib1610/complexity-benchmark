```python
import sys

# The problem describes a graph with n nodes and n-1 edges (columns).
# Since the rank is n-1, the graph is a tree.
# Removing an edge i splits the tree into two connected components.
# The vector z^(i) is the indicator vector of one of these components.
# z^(i) is constant across target indices if and only if all target indices
# belong to the same component.
# Thus, column i is target-splitting if and only if the two components
# formed by removing edge i both contain at least one target index.
# This is equivalent to saying that the edge i lies on the unique path
# between some two target indices.
# More simply, an edge is target-splitting if and only if it separates
# the set of target indices into two non-empty sets.
# This happens if and only if the subtree formed by removing the edge
# contains some, but not all, of the target indices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w, i))
        adj[v].append((u, w, i))
        edges.append((u, v, w))
        idx += 3
        
    if t < 2:
        print(0)
        return

    # subtree_targets[u] stores the number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    edge_splitting = [False] * m
    
    # DFS to count targets in subtrees
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w, i in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, i)
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            subtree_targets[u] += 1
        if parent[u] != 0:
            p, edge_idx = parent[u]
            subtree_targets[p] += subtree_targets[u]
            # If the subtree at u contains some but not all targets, it's splitting
            if 0 < subtree_targets[u] < t:
                edge_splitting[edge_idx] = True
                
    total_weight = 0
    for i in range(m):
        if edge_splitting[i]:
            total_weight += edges[i][2]
            
    print(total_weight)

solve()
```
