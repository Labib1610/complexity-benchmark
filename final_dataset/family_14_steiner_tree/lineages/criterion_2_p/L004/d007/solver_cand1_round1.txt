```python
import sys

# The problem asks for the minimum cost to connect a set of t critical elements
# in a graph where edges are transpositions.
# A set of transpositions generates a subgroup H. The orbit of an element x
# under H is the set of all elements y such that there is a path between x and y
# using the selected transpositions.
# For all t critical elements to be in the same orbit, they must all belong
# to the same connected component in the graph formed by the selected edges.
# This is equivalent to finding the Minimum Spanning Tree (MST) of the
# subgraph that connects all t critical elements.
# This is a classic Steiner Tree problem, but since the graph is a tree
# (m = n-1 and the generators form S_n, meaning the graph is connected),
# the Steiner Tree is simply the unique path in the tree connecting all
# critical nodes.
# The cost is the sum of weights of all edges that lie on the paths between
# any two critical nodes. An edge belongs to this set if and only if the
# removal of that edge splits the tree into two components, both of which
# contain at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = set()
    idx = 3
    for _ in range(t):
        critical.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many critical nodes are in the subtree of each node
    # when rooted at an arbitrary node (e.g., 1).
    subtree_critical = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in critical:
            subtree_critical[u] += 1
        if parent[u] != 0:
            subtree_critical[parent[u]] += subtree_critical[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the edge to parent splits the tree such that both sides
        # have at least one critical node, include it.
        count = subtree_critical[i]
        if count > 0 and count < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
