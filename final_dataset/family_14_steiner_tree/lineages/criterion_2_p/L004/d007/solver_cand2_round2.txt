```python
import sys

# The problem asks for the minimum cost to ensure that all t critical indices
# are in the same connected component in the graph formed by the chosen edges.
# This is equivalent to finding a Steiner Tree that connects all t critical nodes.
# Since the graph is a tree (n nodes, n-1 edges), the Steiner Tree is the unique
# path in the tree that connects all t critical nodes.
# The cost is the sum of weights of all edges on the unique paths between
# any two critical nodes. This is equivalent to the sum of weights of all edges
# that have at least one critical node in each of the two components formed
# by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = [False] * (n + 1)
    for i in range(t):
        critical[int(input_data[3 + i])] = True
        
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_critical[u] stores the number of critical nodes in the subtree rooted at u
    subtree_critical = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    for u in reversed(order):
        if critical[u]:
            subtree_critical[u] += 1
        if parent[u] != 0:
            p, w = parent[u]
            subtree_critical[p] += subtree_critical[u]
            # If this edge separates some critical nodes from others, it must be included
            if subtree_critical[u] > 0 and subtree_critical[u] < t:
                total_cost += w
                
    print(total_cost)

solve()
```
