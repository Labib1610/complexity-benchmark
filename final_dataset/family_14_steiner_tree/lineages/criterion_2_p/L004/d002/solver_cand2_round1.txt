```python
import sys

# The problem states that the system is fully traversable and has no redundancy (m = n - 1).
# This means the system is a tree.
# We need to find the minimum weight of the subtree that connects all t detected isotopes.
# This is equivalent to the sum of weights of all edges that lie on the paths between
# any two of the t detected isotopes.
# An edge belongs to this subtree if and only if the two components it creates
# both contain at least one of the t detected isotopes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    detected = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many detected isotopes are in each subtree
    # Root the tree at 1
    count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in detected:
        count[node] = 1
        
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if count[u] > 0 and count[u] < t:
            total_weight += parent_edge_weight[u]
        count[p] += count[u]
        
    print(total_weight)

solve()
```
