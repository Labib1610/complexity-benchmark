```python
import sys

# The problem states the plant is a tree (n units, n-1 pipes, connected).
# We need to find the minimum weight of the subtree that connects all flagged units.
# In a tree, the unique path between any two nodes is fixed.
# The union of all paths between all pairs of flagged units forms the minimal subtree.
# This is equivalent to the sum of weights of all edges that lie on at least one path
# between two flagged units.
# An edge (u, v) with weight w lies on a path between two flagged units if and only if
# the removal of (u, v) splits the tree into two components, each containing at least one flagged unit.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    flagged = set()
    idx = 3
    for _ in range(t):
        flagged.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many flagged units are in each subtree
    # Root the tree at 1
    subtree_flagged = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in flagged:
            subtree_flagged[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to propagate counts
    total_energy = 0
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_flagged[p] += subtree_flagged[u]
            # If this edge separates the tree into two parts both containing flagged units
            if subtree_flagged[u] > 0 and subtree_flagged[u] < t:
                total_energy += edge_to_parent_weight[u]
                
    print(total_energy)

solve()
```
