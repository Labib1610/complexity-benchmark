import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    flagged = [False] * (n + 1)
    for _ in range(t):
        u = int(next(it))
        flagged[u] = True
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    depth = [0] * (n + 1)
    order = [1]
    parent[1] = 0
    depth[1] = 0
    
    for u in order:
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                depth[v] = depth[u] + 1
                order.append(v)
    
    cnt = [0] * (n + 1)
    for i in range(n, 0, -1):
        u = order[i - 1]
        if flagged[u]:
            cnt[u] += 1
        p = parent[u]
        if p != 0:
            cnt[p] += cnt[u]
    
    total_energy = 0
    for u in range(2, n + 1):
        if cnt[u] > 0 and cnt[u] < t:
            total_energy += parent_weight[u]
    
    print(total_energy)

if __name__ == "__main__":
    solve()
