```python
import sys

# The problem states that the m = n - 1 modules form a tree structure.
# We need to find the minimum cost to connect all critical nodes.
# This is equivalent to finding the sum of weights of all edges that lie on
# the unique paths between any two critical nodes.
# An edge in a tree is part of the path between two critical nodes if and only if
# the removal of that edge splits the tree into two components, each containing
# at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many critical nodes are in the subtree of each node.
    # Since n is up to 200,000, we use an iterative DFS to avoid recursion depth issues.
    
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    critical_count = [0] * (n + 1)
    for node in critical_nodes:
        critical_count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If the subtree at u contains some but not all critical nodes,
            # the edge (u, p) must be included.
            if 0 < critical_count[u] < t:
                total_cost += edge_weight_to_parent[u]
            critical_count[p] += critical_count[u]
            
    print(total_cost)

solve()
```
