We can solve this by finding the minimal subtree that connects all critical nodes.  
Since the graph is a tree, the minimal connected subgraph containing all critical nodes is the union of paths between them.  
We can compute this by marking critical nodes and doing a DFS to find the smallest subtree that contains them.

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    critical = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t == 1:
        print(0)
        return
    
    # DFS to find minimal subtree cost
    visited = [False] * (n + 1)
    
    def dfs(u, parent):
        visited[u] = True
        total_cost = 0
        has_critical = u in critical
        
        for v, w in adj[u]:
            if v == parent:
                continue
            child_cost, child_has = dfs(v, u)
            if child_has:
                total_cost += child_cost + w
                has_critical = True
        
        return total_cost, has_critical
    
    # Start from any critical node
    start = next(iter(critical))
    total_cost, _ = dfs(start, -1)
    print(total_cost)

if __name__ == "__main__":
    solve()
```
