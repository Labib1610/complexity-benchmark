import sys

def solve():
    # The problem states the facility layout uses the absolute minimum number of 
    # conveyor belts such that every workstation can be reached from any other.
    # This means the graph is a tree (n nodes, n-1 edges).
    # We need to find the minimum weight subtree that connects all t identified nodes.
    # This is the Steiner Tree problem on a tree, which is simply the sum of weights
    # of all edges that lie on the unique path between any two identified nodes.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Since it's a tree, we can use DFS/BFS to find edges that are part of the 
    # minimal subtree. An edge is part of the minimal subtree if and only if
    # both components formed by removing the edge contain at least one target node.
    
    # Count how many targets are in the subtree rooted at u
    target_count = [0] * (n + 1)
    
    def dfs_count(u, p):
        count = 1 if u in targets else 0
        for v, w in adj[u]:
            if v != p:
                count += dfs_count(v, u)
        target_count[u] = count
        return count

    dfs_count(1, -1)
    
    total_weight = 0
    # For each edge (u, v) with weight w, if the subtree at v contains k targets
    # and the total targets is t, the edge is part of the Steiner tree if 
    # 0 < target_count[v] < t.
    def dfs_sum(u, p):
        nonlocal total_weight
        for v, w in adj[u]:
            if v != p:
                if 0 < target_count[v] < t:
                    total_weight += w
                dfs_sum(v, u)
                
    dfs_sum(1, -1)
    print(total_weight)

solve()
