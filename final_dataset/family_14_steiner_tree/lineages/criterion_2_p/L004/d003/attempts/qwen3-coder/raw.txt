import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    critical = []
    for _ in range(t):
        critical.append(int(data[idx]))
        idx += 1
    
    edges = []
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((w, u, v))
    
    if t <= 1:
        print(0)
        return
    
    # Sort edges by weight
    edges.sort()
    
    # Initialize DSU
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # First, connect all critical nodes
    critical_set = set(critical)
    cost = 0
    connected_critical = 1  # Start with first critical node
    
    # We'll use the first critical node as the root
    root = critical[0]
    
    for w, u, v in edges:
        # Check if this edge connects something to our critical component
        pu = find(parent, u)
        pv = find(parent, v)
        root_component = find(parent, root)
        
        # If both u and v are already in the critical component, skip
        if pu == root_component and pv == root_component:
            continue
            
        # If one is in critical component and the other isn't
        if pu == root_component and pv != root_component:
            # If v is critical, we're connecting two critical components
            if v in critical_set:
                connected_critical += 1
            union(parent, rank, u, v)
            cost += w
        elif pv == root_component and pu != root_component:
            # If u is critical, we're connecting two critical components
            if u in critical_set:
                connected_critical += 1
            union(parent, rank, u, v)
            cost += w
        # If neither is in critical component, but they connect to make a path to critical
        elif (u in critical_set and v not in critical_set) or (v in critical_set and u not in critical_set):
            if u in critical_set:
                connected_critical += 1
            if v in critical_set:
                connected_critical += 1
            union(parent, rank, u, v)
            cost += w
        # If one or both are in critical component now
        elif find(parent, u) == root_component or find(parent, v) == root_component:
            # Check if we just connected a new critical node
            new_critical_connected = False
            if u in critical_set and find(parent, u) == root_component and pu != root_component:
                connected_critical += 1
                new_critical_connected = True
            if v in critical_set and find(parent, v) == root_component and pv != root_component:
                if not (u in critical_set and new_critical_connected):  # Avoid double counting
                    connected_critical += 1
            union(parent, rank, u, v)
            cost += w
        else:
            # Neither is connected to critical component yet, just union normally
            # But check if this creates a connection to critical component
            if u in critical_set or v in critical_set:
                if u in critical_set:
                    connected_critical += 1
                if v in critical_set:
                    connected_critical += 1
                union(parent, rank, u, v)
                cost += w
            elif union(parent, rank, u, v):  # Only if they were separate components
                # Check if this union connected to critical through either side
                pass  # Will be handled when we encounter an edge connecting to this component
        
        # Early termination
        if connected_critical == t:
            break
    
    print(cost)

if __name__ == "__main__":
    main()
