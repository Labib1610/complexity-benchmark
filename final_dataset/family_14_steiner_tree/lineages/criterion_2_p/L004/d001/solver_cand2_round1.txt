```python
import sys

# The problem states the graph is a tree (n nodes, n-1 links).
# We need to find the minimum weight of a subtree that connects all flagged nodes.
# In a tree, the unique path between any two nodes is fixed.
# The union of all paths between all pairs of flagged nodes forms the minimal subtree.
# This is equivalent to the sum of weights of all edges that lie on at least one path
# between two flagged nodes.
# An edge (u, v) with weight w lies on a path between two flagged nodes if and only if
# the removal of the edge splits the tree into two components, each containing at least one flagged node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    flagged = set()
    idx = 3
    for _ in range(t):
        flagged.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many flagged nodes are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    subtree_flagged = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in flagged:
        subtree_flagged[u] = 1
        
    total_power = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        subtree_flagged[p] += subtree_flagged[u]
        
        # If the subtree at u contains some flagged nodes and the rest of the tree
        # also contains some flagged nodes, this edge must be included.
        if 0 < subtree_flagged[u] < t:
            total_power += edge_to_parent_weight[u]
            
    print(total_power)

solve()
```
