```python
import sys

# The problem asks for the minimum weight of a subtree that connects all t flagged nodes.
# In a tree, the minimum weight subtree connecting a set of nodes is the union of all 
# paths between these nodes. This is equivalent to the sum of weights of all edges 
# that lie on at least one path between two flagged nodes.
# An edge (u, v) with weight w lies on a path between two flagged nodes if and only if
# the removal of this edge splits the tree into two components, each containing at 
# least one flagged node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    flagged = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        flagged[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many flagged nodes are in the subtree rooted at each node
    # We use iterative DFS to avoid recursion depth issues
    subtree_flagged = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if flagged[u]:
            subtree_flagged[u] += 1
        if parent[u] != -1:
            subtree_flagged[parent[u]] += subtree_flagged[u]
            
    total_energy = 0
    for i in range(2, n + 1):
        # If the subtree at i contains some flagged nodes but not all,
        # then the edge connecting i to its parent must be part of the Steiner tree.
        count = subtree_flagged[i]
        if count > 0 and count < t:
            total_energy += edge_weight_to_parent[i]
            
    print(total_energy)

if __name__ == '__main__':
    solve()
```
