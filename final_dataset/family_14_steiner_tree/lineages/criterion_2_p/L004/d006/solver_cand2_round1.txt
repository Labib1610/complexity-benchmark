```python
import sys

# The problem asks for the minimum cost to connect all t critical propositions
# in a graph where edges represent axioms with weights.
# Since m = n - 1, the graph is a tree.
# We need to find the minimum weight subset of edges such that all t critical
# nodes are in the same connected component.
# This is equivalent to finding the Steiner Tree for the set of t nodes.
# In a tree, the Steiner Tree is the unique path connecting all t nodes.
# The edges in the Steiner Tree are exactly those edges that lie on the
# simple path between any two critical nodes.
# An edge (u, v) with weight w is part of the Steiner Tree if and only if
# the removal of this edge splits the tree into two components, each of which
# contains at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_critical_count[u] stores the number of critical nodes in the subtree rooted at u
    subtree_critical_count = [0] * (n + 1)
    
    # DFS to count critical nodes in subtrees
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in critical_nodes:
            subtree_critical_count[u] += 1
        if parent[u] != 0:
            subtree_critical_count[parent[u]] += subtree_critical_count[u]
            
    total_cost = 0
    for u, v, w in edges:
        # The edge connects u and v. One is the parent of the other.
        child = u if parent[u] == v else v
        count = subtree_critical_count[child]
        # If the edge separates the tree such that both sides have critical nodes
        if count > 0 and count < t:
            total_cost += w
            
    print(total_cost)

solve()
```
