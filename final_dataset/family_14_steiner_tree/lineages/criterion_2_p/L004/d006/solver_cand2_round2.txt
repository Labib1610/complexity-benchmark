```python
import sys

# The problem states that the m axioms form a tree structure (n propositions, m=n-1 axioms, 
# unique path between any two nodes). We need to find the minimum cost to connect 
# a subset of t critical nodes. This is equivalent to finding the weight of the 
# Steiner Tree for the given set of t nodes in the tree.
# In a tree, the Steiner Tree is simply the union of all paths between the 
# critical nodes. This is the unique minimal subtree containing all critical nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the weight of the minimal subtree containing all critical nodes:
    # 1. Root the tree at an arbitrary critical node.
    # 2. A node is part of the minimal subtree if it is a critical node 
    #    or if it has at least two branches leading to critical nodes.
    # 3. Alternatively, use post-order traversal: 
    #    A subtree rooted at 'u' needs to be included if it contains at least 
    #    one critical node. If it contains a critical node, the edge to its 
    #    parent must be included if the subtree contains at least one 
    #    critical node and the rest of the tree contains at least one.
    
    root = next(iter(critical_nodes))
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = {root: 0}
    edge_to_parent_weight = {root: 0}
    order = []
    stack = [root]
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    has_critical = {i: (i in critical_nodes) for i in range(1, n + 1)}
    
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == root:
            continue
        p = parent[u]
        if has_critical[u]:
            total_cost += edge_to_parent_weight[u]
            has_critical[p] = True
            
    print(total_cost)

solve()
```
