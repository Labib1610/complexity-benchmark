```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_facilities = set()
    idx = 3
    for _ in range(t):
        priority_facilities.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the sum of weights of all edges that lie on the path
    # between any two priority nodes. An edge is part of this set if and only
    # if the subtree formed by removing the edge contains at least one 
    # priority node on both sides.
    
    is_priority = [False] * (n + 1)
    for p in priority_facilities:
        is_priority[p] = True
        
    total_priority_count = t
    subtree_priority_count = [0] * (n + 1)
    ans = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in reversed(order):
        subtree_priority_count[u] += (1 if is_priority[u] else 0)
        if parent[u] != 0:
            subtree_priority_count[parent[u]] += subtree_priority_count[u]
            # If this edge separates the tree into two parts, both containing
            # at least one priority node, then this edge must be included.
            count = subtree_priority_count[u]
            if count > 0 and count < total_priority_count:
                ans += edge_weight_to_parent[u]
                
    print(ans)

solve()
```
