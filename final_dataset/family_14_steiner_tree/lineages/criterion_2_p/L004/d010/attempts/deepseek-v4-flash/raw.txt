We can solve this by treating the network as a tree (since there are exactly n−1 edges and it’s connected).  
We need to find the minimal subtree that connects all priority nodes.  

The approach:  
- Root the tree at any node (say 1).  
- Mark all priority nodes.  
- Do a DFS to determine which edges are needed: an edge is needed if there is at least one priority node in both sides of the edge.  
- Sum the weights of all needed edges.  

Here’s the implementation:  

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    priority = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    parent = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    # DFS to build parent relationships
    stack = [1]
    visited[1] = True
    order = []
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
    
    # Count priority nodes in each subtree
    subtree_priority_count = [0] * (n + 1)
    for node in priority:
        subtree_priority_count[node] = 1
    
    total_cost = 0
    # Process nodes in reverse order of discovery (post-order)
    for u in reversed(order):
        if u != 1:
            p = parent[u]
            subtree_priority_count[p] += subtree_priority_count[u]
            # If this subtree has at least one priority node and the rest of the tree also has at least one,
            # then this edge is needed.
            if subtree_priority_count[u] > 0 and (t - subtree_priority_count[u]) > 0:
                total_cost += parent_edge_weight[u]
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
