```python
import sys

# The problem states that the matrix A is symmetric and the modifications
# form a structure such that (I+A)^(n-1) has positive entries everywhere.
# Since m = n-1, the graph formed by the modifications is a tree.
# M = I + A is the adjacency matrix of the graph plus the identity matrix.
# M^k_{i,j} > 0 if and only if there is a path of length <= k between i and j.
# Since the graph is a tree and we have n-1 edges, the diameter is at most n-1.
# Thus, M^(n-1)_{i,j} > 0 if and only if there is a path between i and j.
# We need all pairs (i, j) in the target set to be connected in the subgraph
# formed by the chosen modifications.
# This is the Steiner Tree problem on a tree, which is simply the sum of
# weights of all edges on the unique paths between all nodes in the target set.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(input_data[i + 3]) - 1 for i in range(t)]
    
    adj = [[] for _ in range(n)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        w = int(input_data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # To connect all target nodes in a tree, we need the union of all paths
    # between them. This is the set of edges that lie on at least one path
    # between two target nodes.
    # An edge is in the Steiner tree iff it splits the tree into two components,
    # both of which contain at least one target node.
    
    target_set = [False] * n
    for node in targets:
        target_set[node] = True
        
    subtree_targets = [0] * n
    
    # DFS to count targets in each subtree
    parent = [-1] * n
    edge_weight = [0] * n
    order = []
    stack = [0]
    visited = [False] * n
    visited[0] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        subtree_targets[u] += (1 if target_set[u] else 0)
        if parent[u] != -1:
            subtree_targets[parent[u]] += subtree_targets[u]
            
    total_weight = 0
    for i in range(1, n):
        # If the edge (i, parent[i]) separates the tree such that both sides
        # have at least one target, it must be included.
        count = subtree_targets[i]
        if count > 0 and count < t:
            total_weight += edge_weight[i]
            
    print(total_weight)

solve()
```
