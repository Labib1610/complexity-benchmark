```python
import sys

# The problem asks for the minimum weight subset of edges (e_u - e_v)
# such that for all target indices a, b, the vector e_a - e_b is in the span.
# A vector e_a - e_b is in the span of a set of edges if and only if
# a and b are in the same connected component in the graph formed by the edges.
# Since we have m = n - 1 edges and the span of all edges has dimension n - 1,
# the graph is a tree.
# We need all target indices to be in the same connected component.
# This means we need to select a set of edges that connects all target nodes.
# In a tree, there is a unique path between any two nodes.
# To connect a set of nodes in a tree, we must select all edges on the paths
# between them. The union of all paths between nodes in a set S is the
# minimal subtree containing S (the Steiner Tree).
# Since the graph is already a tree, the minimal subtree is unique.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Identify nodes that are part of the minimal subtree connecting targets.
    # A node is in the subtree if it lies on a path between two targets.
    # This is equivalent to saying the subtree rooted at the node contains
    # at least one target, and the rest of the tree contains at least one target.
    
    has_target = [False] * (n + 1)
    for node in targets:
        has_target[node] = True
        
    subtree_has_target = [False] * (n + 1)
    edge_weights = []
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    total_weight = 0
    for u in reversed(order):
        if has_target[u]:
            subtree_has_target[u] = True
        if parent[u] != 0:
            p, w = parent[u]
            if subtree_has_target[u]:
                subtree_has_target[p] = True
                total_weight += w
                
    print(total_weight)

solve()
```
