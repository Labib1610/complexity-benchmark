```python
import sys

# The problem states m = n - 1, which means the graph is a tree.
# If the graph is a tree, there is only one unique path between any two nodes.
# To connect all critical facilities, we must include all edges that lie on the 
# unique paths between any pair of critical facilities.
# This is equivalent to finding the smallest subtree that contains all critical nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Since it's a tree, we can root it at the first critical node.
    # We need to count edges that have at least one critical node in both 
    # components formed by removing the edge.
    
    root = next(iter(critical))
    subtree_critical_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(root, -1, 0)] # node, parent, edge_weight_to_parent
    order = []
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    
    dfs_stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while dfs_stack:
        u = dfs_stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                weight_to_parent[v] = w
                dfs_stack.append(v)
                
    for u in reversed(order):
        if u in critical:
            subtree_critical_count[u] += 1
        if parent[u] != 0:
            subtree_critical_count[parent[u]] += subtree_critical_count[u]
            
    for i in range(1, n + 1):
        if parent[i] != 0:
            # If the edge to parent separates the tree into two parts,
            # and both parts contain at least one critical node, include the edge.
            count = subtree_critical_count[i]
            if count > 0 and count < t:
                total_cost += weight_to_parent[i]
                
    print(total_cost)

solve()
```
