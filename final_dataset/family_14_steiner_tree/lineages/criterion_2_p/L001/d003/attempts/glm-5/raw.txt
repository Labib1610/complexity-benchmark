import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    it = iter(data)
    
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    is_core = [False] * (n + 1)
    for _ in range(t):
        c = int(next(it))
        is_core[c] = True
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    parent = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    depth = [0] * (n + 1)
    
    stack = [1]
    parent[1] = 0
    order = [1]
    
    while stack:
        u = stack.pop()
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                parent_edge_weight[v] = w
                depth[v] = depth[u] + 1
                stack.append(v)
                order.append(v)
    
    max_edge_to_root = [0] * (n + 1)
    core_count = [0] * (n + 1)
    
    for u in order:
        if is_core[u]:
            core_count[u] = 1
    
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        core_count[p] += core_count[u]
        if core_count[u] > 0 and core_count[u] < t:
            max_edge_to_root[p] = max(max_edge_to_root[p], 
                                       max(max_edge_to_root[u], parent_edge_weight[u]))
    
    total_cost = 0
    for u in range(2, n + 1):
        if core_count[u] > 0 and core_count[u] < t:
            total_cost += parent_edge_weight[u]
    
    print(total_cost)

solve()
