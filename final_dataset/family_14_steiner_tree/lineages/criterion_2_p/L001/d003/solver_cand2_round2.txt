```python
import sys

# The problem states the network is a tree (n nodes, n-1 edges, connected).
# To connect all core factions, we must include all edges that lie on the 
# unique paths between any two core factions.
# This is equivalent to finding the minimal subtree that contains all core nodes.
# An edge belongs to this subtree if and only if the subtree formed by 
# removing that edge splits the core nodes into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    core_factions = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many core nodes are in the subtree of each node.
    # Since n is up to 200,000, we use iterative DFS to avoid recursion depth issues.
    
    core_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in core_factions:
            core_count[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        # If the subtree at u contains some core nodes but not all,
        # the edge to its parent must be included.
        if 0 < core_count[u] < t:
            total_cost += edge_to_parent_weight[u]
            
        # Propagate core count to parent
        p = parent[u]
        core_count[p] += core_count[u]
        
    print(total_cost)

solve()
```
