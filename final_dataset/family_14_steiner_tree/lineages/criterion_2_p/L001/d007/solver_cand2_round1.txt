```python
import sys

# The problem asks for a subset of nodes S such that the induced subgraph
# has a perfect matching (or rather, a matching that covers all nodes except one).
# In a tree, a subset of nodes induces a forest. A forest has a perfect matching
# (or near-perfect) if and only if each connected component has a perfect matching
# (or near-perfect).
# Actually, the condition "exactly one activated module is assigned zero energized channels,
# and every other activated module is assigned exactly one" is equivalent to saying
# the energized channels form a perfect matching on the activated modules,
# except for one module which is left unmatched.
# This means the activated modules must form a set where the induced subgraph
# has a matching covering all but one vertex.
# In a tree, this is equivalent to saying the activated nodes form a connected
# component (a subtree) and the number of nodes is odd.
# Wait, let's re-read: "Exactly one activated module is assigned zero energized channels,
# every other activated module is assigned exactly one."
# This is exactly the definition of a matching that covers all vertices except one.
# In a tree, a set of vertices has such a matching if and only if the induced
# subgraph is a tree with an odd number of vertices.
# We need to find a subtree containing all primary nodes with an odd number of vertices
# that minimizes the sum of weights of edges in the subtree.
# Let S be the set of primary nodes. The minimal subtree containing S is the
# Steiner Tree of S. Let this be T_S.
# If |T_S| is odd, the cost is the sum of weights of edges in T_S.
# If |T_S| is even, we must add one more node to the subtree to make it odd.
# We should pick a node v adjacent to T_S such that the edge (u, v) has minimal weight.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, t = map(int, input_data[:3])
    primary = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u, v, w = map(int, input_data[idx:idx+3])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3

    # Find Steiner Tree nodes
    is_primary = [False] * (n + 1)
    for p in primary:
        is_primary[p] = True
    
    has_primary = [False] * (n + 1)
    def dfs_primary(u, p):
        res = is_primary[u]
        for v, w in adj[u]:
            if v != p:
                if dfs_primary(v, u):
                    res = True
        has_primary[u] = res
        return res
    
    dfs_primary(next(iter(primary)), -1)
    
    # Calculate Steiner Tree cost and size
    steiner_edges_sum = 0
    steiner_nodes_count = 0
    for u in range(1, n + 1):
        if has_primary[u]:
            steiner_nodes_count += 1
            for v, w in adj[u]:
                if v > u and has_primary[v]:
                    steiner_edges_sum += w
                    
    if steiner_nodes_count % 2 != 0:
        print(steiner_edges_sum)
    else:
        # Need to add one node to make count odd.
        # Find min edge connecting a node in Steiner Tree to one outside.
        min_w = float('inf')
        for u in range(1, n + 1):
            if has_primary[u]:
                for v, w in adj[u]:
                    if not has_primary[v]:
                        if w < min_w:
                            min_w = w
        print(steiner_edges_sum + min_w)

solve()
```
