```python
import sys
from sys import setrecursionlimit

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    core = list(map(int, input().split()))
    is_core = [False] * (n + 1)
    for c in core:
        is_core[c] = True
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0 and v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                stack.append(v)
    
    dp = [0] * (n + 1)
    cnt_core = [0] * (n + 1)
    
    for u in reversed(order):
        if is_core[u]:
            cnt_core[u] += 1
        for v, w in adj[u]:
            if v != parent[u]:
                cnt_core[u] += cnt_core[v]
                dp[u] += dp[v]
                if cnt_core[v] > 0:
                    dp[u] += w
    
    ans = dp[1]
    print(ans)

if __name__ == "__main__":
    setrecursionlimit(1 << 25)
    solve()
```
