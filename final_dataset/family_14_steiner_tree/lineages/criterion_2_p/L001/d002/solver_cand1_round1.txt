```python
import sys

# The problem states m = n - 1, which means the graph is a tree.
# We need to find the minimum cost to connect all primary locations.
# In a tree, the unique path between any two nodes is fixed.
# The union of all paths between all pairs of primary locations forms a subtree.
# Any edge in this subtree must be included if it lies on a path between two primary nodes.
# An edge lies on a path between two primary nodes if and only if the edge
# splits the tree into two components, each containing at least one primary node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    primary_nodes = set()
    idx = 3
    for _ in range(t):
        primary_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many primary nodes are in the subtree rooted at u
    primary_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in primary_nodes:
            primary_count[u] += 1
        if parent[u] != 0:
            primary_count[parent[u]] += primary_count[u]
            # If this edge separates the tree into two parts, both having primary nodes,
            # then this edge must be part of the secure zone.
            if primary_count[u] > 0 and primary_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
