```python
import sys

# The problem states that the given m = n-1 links form a tree (since they connect all n locations).
# We need to find the minimum cost to connect all primary locations.
# In a tree, the path between any two nodes is unique.
# To connect a set of primary nodes, we must include all edges that lie on the unique paths
# between any two primary nodes.
# This is equivalent to finding the smallest subtree that contains all primary nodes.
# An edge (u, v) with weight w is part of this subtree if and only if the subtree
# formed by removing (u, v) has primary nodes in both components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary_nodes = set()
    idx = 3
    for _ in range(t):
        primary_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many primary nodes are in the subtree of each node.
    # Since it's a tree, we can root it at an arbitrary node (e.g., 1).
    # For an edge (u, v) where v is a child of u, the edge is included if
    # the number of primary nodes in v's subtree is > 0 and < t.
    
    primary_count = [0] * (n + 1)
    for node in primary_nodes:
        primary_count[node] = 1
        
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            p = parent[u]
            primary_count[p] += primary_count[u]
            # If the subtree at u contains some but not all primary nodes,
            # the edge (u, parent[u]) must be included.
            if 0 < primary_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
