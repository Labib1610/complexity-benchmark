import sys

# The problem states the network is a tree (n nodes, n-1 edges).
# Let A be the set of active nodes, k = |A|.
# Isolation score of active node i is 3 - d_i, where d_i is the number of active neighbors.
# Sum of isolation scores = sum_{i in A} (3 - d_i) = 3k - sum_{i in A} d_i.
# Since each edge between two active nodes is counted twice in sum d_i,
# sum d_i = 2 * (number of edges between active nodes).
# Let E_A be the number of edges between active nodes.
# Sum of scores = 3k - 2 * E_A.
# Condition: 3k - 2 * E_A = k + 2  =>  2k - 2 * E_A = 2  =>  k - E_A = 1.
# In a forest, k - E_A is the number of connected components formed by the active nodes.
# So the condition is equivalent to: the active nodes must form exactly one connected component (a subtree).

# We need to find a subtree that contains all mandatory nodes and minimizes the sum of weights of edges in the subtree.
# This is the Steiner Tree problem on a tree, which is simply the minimal subtree containing all mandatory nodes.
# The minimal subtree containing a set of nodes S is the union of all paths between nodes in S.
# In a tree, this is unique and consists of all edges that lie on a path between at least two nodes in S.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    mandatory = set()
    idx = 3
    for _ in range(t):
        mandatory.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t == 0:
        print(0)
        return
    if t == 1:
        print(0)
        return

    # To find the minimal subtree containing all mandatory nodes:
    # 1. Root the tree arbitrarily (e.g., at node 1).
    # 2. For each node, determine if its subtree contains any mandatory nodes.
    # 3. An edge (u, v) where v is a child of u is in the minimal subtree if and only if
    #    the subtree at v contains at least one mandatory node AND the rest of the tree
    #    (outside v's subtree) contains at least one mandatory node.
    
    has_mandatory = [False] * (n + 1)
    for node in mandatory:
        has_mandatory[node] = True
        
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    subtree_has_mandatory = [False] * (n + 1)
    for u in reversed(order):
        if has_mandatory[u]:
            subtree_has_mandatory[u] = True
        for v, w in adj[u]:
            if parent[v] and parent[v][0] == u:
                if subtree_has_mandatory[v]:
                    subtree_has_mandatory[u] = True
                    
    total_mandatory = t
    ans = 0
    for i in range(2, n + 1):
        u, w = parent[i]
        # Edge (u, i) is in the minimal subtree if:
        # subtree at i has mandatory nodes AND (total mandatory - subtree at i has mandatory) > 0
        if subtree_has_mandatory[i] and (total_mandatory - (subtree_has_mandatory[i] if has_mandatory[i] else 0) > 0):
            # Wait, the condition is simpler:
            # Edge (u, i) is in the minimal subtree if subtree at i has at least one mandatory node
            # AND the part of the tree "above" i has at least one mandatory node.
            count_in_subtree = 0
            # We need a count, not just a boolean
            pass

    # Re-calculating with counts
    count_mandatory = [0] * (n + 1)
    for node in mandatory:
        count_mandatory[node] = 1
    for u in reversed(order):
        for v, w in adj[u]:
            if parent[v] and parent[v][0] == u:
                count_mandatory[u] += count_mandatory[v]
                
    ans = 0
    for i in range(2, n + 1):
        u, w = parent[i]
        if count_mandatory[i] > 0 and (total_mandatory - count_mandatory[i]) > 0:
            ans += w
            
    print(ans)

solve()
