```python
import sys

# The problem asks for the minimum cost to activate a set of nodes such that
# the sum of (2 - k_i) for all active nodes i equals 2, where k_i is the number
# of active neighbors of i.
# Let S be the set of active nodes. The condition is sum_{i in S} (2 - k_i) = 2.
# sum_{i in S} 2 - sum_{i in S} k_i = 2.
# 2 * |S| - 2 * |Edges within S| = 2.
# |S| - |Edges within S| = 1.
# This is the formula for the number of connected components in a forest.
# So, the set of active nodes must form a single connected component (a tree).
# We need to find a subtree that contains all mandatory nodes and minimizes the
# sum of weights of edges within the subtree.
# This is the Steiner Tree problem on a tree, which is simply the weight of the
# minimal subtree connecting all mandatory nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    mandatory = set()
    idx = 3
    for _ in range(t):
        mandatory.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Root the tree at an arbitrary mandatory node
    root = next(iter(mandatory))
    
    # is_mandatory_subtree[u] is True if the subtree rooted at u contains a mandatory node
    is_mandatory_subtree = [False] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in mandatory:
            is_mandatory_subtree[u] = True
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == root:
            continue
        if is_mandatory_subtree[u]:
            total_cost += edge_to_parent_weight[u]
            is_mandatory_subtree[parent[u]] = True
            
    print(total_cost)

solve()
```
