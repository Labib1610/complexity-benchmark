import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return

    mandatory = set()
    for _ in range(t):
        mandatory.add(int(next(iterator)))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        w = int(next(iterator))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # Check if mandatory stations are already connected in a way that satisfies B(x)=2
    # We need to check if the mandatory stations form a single connected component
    # and if the sum of (2 - k_i) over mandatory stations equals 2.
    # If mandatory stations are not connected, we must connect them.
    
    # We can use a modified Prim's algorithm.
    # We start with all mandatory stations in the MST.
    # We add edges to connect components until we have a single component.
    # The cost is the sum of weights of edges added.
    # However, we need to ensure that the final B(x) = 2.
    # If we have a single connected component of active stations, B(x) = 2.
    # So we just need to find the minimum cost to connect all mandatory stations.
    
    # But wait, the problem says "The network is constructed incrementally: starting with a single initial station, each subsequent station was added by connecting it to exactly one previously existing station via a single channel."
    # This means the graph is a tree.
    # So we just need to find the minimum Steiner tree connecting the mandatory stations.
    # Since the graph is a tree, the Steiner tree is just the minimal subtree connecting the mandatory stations.
    # We can find this by finding the union of all paths between mandatory stations.
    # Or more simply, we can use a modified Prim's algorithm starting with all mandatory stations.
    
    # Let's implement a modified Prim's algorithm.
    # We maintain a set of visited nodes (initially all mandatory).
    # We use a priority queue to store edges connecting visited nodes to unvisited nodes.
    # We extract the minimum weight edge and add the new node to the visited set.
    # We repeat until all mandatory stations are connected.
    # But we need to ensure that we connect all mandatory stations, not just any nodes.
    # Actually, since the graph is a tree, connecting all mandatory stations will result in a tree that includes some non-mandatory stations if necessary.
    
    # Let's implement this.
    
    in_mst = [False] * (n + 1)
    pq = []
    
    # Start with all mandatory stations
    for node in mandatory:
        in_mst[node] = True
        for neighbor, weight in adj[node]:
            if not in_mst[neighbor]:
                heapq.heappush(pq, (weight, node, neighbor))
    
    total_cost = 0
    num_components = len(mandatory)
    
    # We need to connect all mandatory stations.
    # The number of edges needed is num_components - 1.
    # But we might need to add non-mandatory stations to connect them.
    
    while pq and num_components > 1:
        weight, u, v = heapq.heappop(pq)
        if in_mst[v]:
            continue
        in_mst[v] = True
        total_cost += weight
        num_components -= 1
        for neighbor, w in adj[v]:
            if not in_mst[neighbor]:
                heapq.heappush(pq, (w, v, neighbor))
    
    print(total_cost)

if __name__ == "__main__":
    solve()
