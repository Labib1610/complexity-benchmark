import sys

# The problem states that the constraints form a minimal, complete equivalence system,
# which means the graph is a tree.
# The structural index I(x) = sum(x_i) - sum(x_u * x_v) for edges (u, v).
# For a tree, sum(x_i) - sum(x_u * x_v) is exactly the number of connected components
# formed by the nodes with x_i = 1.
# We need I(x) = 1, meaning the nodes with x_i = 1 must form a single connected component.
# Since all mandatory variables must have x_i = 1, the set of nodes with x_i = 1
# must be a connected subtree that contains all mandatory variables.
# To minimize the cost C(x) = sum(w_j for edges where both endpoints are 1),
# we want to pick a subtree that includes all mandatory variables and minimizes the sum of weights of edges in that subtree.
# Wait, the cost is the sum of weights of edges where BOTH endpoints are 1.
# If we pick a connected subtree S of nodes with x_i = 1, the edges with both endpoints in S
# are exactly the edges of the subtree S.
# So we need to find a subtree S that contains all mandatory variables and minimizes the sum of weights of its edges.
# The smallest such subtree is the Steiner Tree of the mandatory variables.
# In a tree, the Steiner Tree is the unique minimal subtree connecting all mandatory variables.
# This is simply the union of all paths between any two mandatory variables.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    mandatory = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the sum of weights of the Steiner Tree:
    # A node is in the Steiner Tree if it lies on a path between two mandatory nodes.
    # An edge is in the Steiner Tree if removing it splits the tree into two components,
    # both of which contain at least one mandatory node.
    
    subtree_size = [0] * (n + 1)
    mandatory_count = [0] * (n + 1)
    
    for node in mandatory:
        mandatory_count[node] = 1
        
    # DFS to count mandatory nodes in subtrees
    order = []
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    total_cost = 0
    for u in reversed(order):
        if u == 1: continue
        if mandatory_count[u] > 0 and mandatory_count[u] < t:
            total_cost += edge_to_parent_w[u]
        mandatory_count[parent[u]] += mandatory_count[u]
        
    print(total_cost)

solve()
