```python
import sys

# The problem asks for the minimum cost to activate a set of nodes such that
# the total net stability is 2.
# Each activated node contributes 2 stability.
# Each edge between two activated nodes consumes 1 stability from each endpoint (total 2).
# Let A be the set of activated nodes, E_A be the set of edges with both endpoints in A.
# Total stability = 2 * |A| - 2 * |E_A| = 2.
# This simplifies to |A| - |E_A| = 1.
# In a tree, |A| - |E_A| is the number of connected components formed by A.
# So we need the activated nodes to form exactly one connected component.
# Since all critical nodes must be activated, the connected component must contain
# all critical nodes.
# The minimum cost to connect a set of nodes in a tree is the sum of weights
# of all edges on the unique paths between them (the Steiner Tree).
# However, we can activate additional nodes to connect them.
# Let dp[u][0] be the min cost to connect all critical nodes in u's subtree,
# with u not activated.
# Let dp[u][1] be the min cost to connect all critical nodes in u's subtree,
# with u activated.
# If u is activated, it must be part of the connected component.
# If u is not activated, all critical nodes in its subtree must have been
# connected already (so they must form a valid component, but this is only
# possible if no critical nodes exist in the subtree, or if we already
# satisfied the condition).
# Actually, the condition |A| - |E_A| = 1 means the activated nodes form a tree.
# The cost is the sum of weights of edges in that tree.
# This is exactly the Steiner Tree problem on a tree, which is the sum of
# weights of all edges that lie on the path between any two critical nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = [False] * (n + 1)
    for i in range(t):
        critical[int(input_data[3 + i])] = True
        
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # The Steiner Tree in a tree is the union of paths between all critical nodes.
    # An edge is in the Steiner Tree if and only if it separates the tree into
    # two components, both of which contain at least one critical node.
    
    subtree_has_critical = [False] * (n + 1)
    total_critical = t
    
    ans = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    subtree_critical_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if critical[i]:
            subtree_critical_count[i] = 1
            
    for u in reversed(order):
        if parent[u] != -1:
            subtree_critical_count[parent[u]] += subtree_critical_count[u]
            
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if parent[v] == u:
                # Edge (u, v) is in the Steiner Tree if the subtree at v
                # contains some but not all critical nodes.
                count = subtree_critical_count[v]
                if count > 0 and count < total_critical:
                    ans += w
                    
    print(ans)

solve()
```
