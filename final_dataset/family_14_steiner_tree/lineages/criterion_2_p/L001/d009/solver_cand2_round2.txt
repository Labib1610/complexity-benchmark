```python
import sys

# The problem states that the network is a tree (n stations, n-1 channels, connected).
# A valid assignment has (active stations) - (active channels) = 1.
# In a tree, any set of active stations induces a forest.
# If the active stations form a forest with k connected components,
# then (active stations) - (active channels) = k.
# Thus, the condition "k = 1" means the active stations must form a single connected component (a subtree).
# We need to find a subtree that contains all mandatory stations and minimizes the sum of weights of edges in that subtree.
# This is the Steiner Tree problem on a tree, which is equivalent to finding the smallest subtree containing all mandatory nodes.
# The smallest subtree containing a set of nodes S is the union of all paths between nodes in S.
# In a tree, this is simply the unique minimal subtree connecting all nodes in S.
# The cost is the sum of weights of all edges that lie on at least one path between two mandatory nodes.
# An edge is in this subtree if and only if the subtree formed by removing the edge splits the mandatory nodes into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    mandatory = [False] * (n + 1)
    for i in range(t):
        mandatory[int(input_data[3 + i])] = True
        
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count mandatory nodes in subtree
    subtree_mandatory = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    total_cost = 0
    for u in reversed(order):
        if mandatory[u]:
            subtree_mandatory[u] += 1
        
        if parent[u] != -1:
            if subtree_mandatory[u] > 0 and subtree_mandatory[u] < t:
                total_cost += edge_to_parent_w[u]
            subtree_mandatory[parent[u]] += subtree_mandatory[u]
            
    print(total_cost)

solve()
```
