```python
import sys

# The condition x^T (2I - A) x = 2 for a tree (m = n-1) simplifies to:
# 2 * sum(x_i) - sum_{i~j} (x_i * x_j) * 2 = 2
# sum(x_i) - sum_{i~j} (x_i * x_j) = 1
# This is equivalent to saying the active nodes form a single connected component.
# The cost function simplifies to sum_{i~j, x_i=1, x_j=1} w_{ij}.
# We need to find a connected subgraph containing all mandatory nodes that minimizes
# the sum of weights of edges in the subgraph.
# This is the Steiner Tree problem on a tree, which is simply the sum of weights
# of all edges on the unique paths between all mandatory nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    mandatory = [int(x) for x in input_data[3:3+t]]
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t == 0:
        print(0)
        return

    # Root the tree at the first mandatory node
    root = mandatory[0]
    
    # is_mandatory[u] is true if the subtree at u contains a mandatory node
    is_mandatory = [False] * (n + 1)
    for node in mandatory:
        is_mandatory[node] = True
        
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    # Post-order traversal to mark subtrees
    total_cost = 0
    for u in reversed(order):
        if u == root:
            continue
        if is_mandatory[u]:
            total_cost += edge_to_parent_w[u]
            is_mandatory[parent[u]] = True
            
    print(total_cost)

solve()
```
