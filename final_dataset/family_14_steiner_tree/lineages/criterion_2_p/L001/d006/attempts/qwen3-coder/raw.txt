import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    primary = set()
    for _ in range(t):
        primary.add(int(data[idx]))
        idx += 1
    
    edges = []
    adj = defaultdict(list)
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((u, v, w))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Since the graph is a tree (m = n-1 and connected), we can use BFS/DFS to find distances
    # We need to find the minimum cost to activate a subtree that includes all primary nodes
    # and satisfies the stability condition: sum of (2 - degree_in_activated_subgraph) = 2
    
    # Let S be the set of activated nodes including all primary nodes.
    # Sum over u in S of (2 - degree_S(u)) = 2
    # This equals 2*|S| - sum_u degree_S(u) = 2
    # But sum_u degree_S(u) = 2*(number of edges in subgraph induced by S)
    # So 2*|S| - 2*|E_S| = 2 => |S| - |E_S| = 1
    # Since the subgraph is connected (it's a tree), |E_S| = |S| - 1
    # So |S| - (|S|-1) = 1, which always holds.
    # Therefore, any connected subtree containing all primary nodes satisfies the stability condition.
    
    # So the problem reduces to finding the minimum cost to activate a connected subtree
    # that includes all primary nodes. The cost is the sum of weights of edges fully within this subtree.
    
    # This is equivalent to finding the minimum Steiner tree in a tree, which can be solved optimally
    # by taking the union of paths between all pairs of primary nodes.
    
    # In a tree, the minimum Steiner tree is the union of all paths between primary nodes.
    # The cost is the sum of edge weights in this union.
    
    # Efficient way:
    # Root the tree at some node, compute for each node whether it needs to be included
    # based on whether it lies on a path between any two primary nodes.
    
    # Better approach:
    # Build the subtree induced by the primary nodes and their connecting paths.
    # In a tree, this is equivalent to finding the minimal connected subtree containing all primary nodes.
    
    # Algorithm:
    # 1. Root the tree at an arbitrary node (say 1)
    # 2. Compute parent and depth for each node via BFS/DFS
    # 3. For each pair of primary nodes, mark all edges on the path between them
    # 4. The cost is the sum of weights of marked edges
    
    # However, since there could be many primary nodes, pairwise path marking is inefficient.
    
    # Optimal approach:
    # Use the fact that in a tree, the minimal connected subtree containing a set of nodes
    # is obtained by including all nodes on paths between any two of them.
    # This is equivalent to taking the union of paths between all pairs, but more efficiently:
    # Find the "Steiner tree" by:
    # 1. Rooting the tree
    # 2. Finding the lowest common ancestor (LCA) structure
    # 3. Marking edges that lie on paths between primary nodes
    
    # Even simpler for trees:
    # The minimal subtree containing all primary nodes is the union of paths from root to each primary node,
    # minus overlapping parts. But easier is:
    # Just run a multi-source BFS from all primary nodes to see which edges are needed.
    # Wait, that doesn't work because we want the minimal connected component containing all primary nodes.
    
    # Correct approach:
    # The minimal connected subtree containing all primary nodes can be found by:
    # 1. Pick one primary node as root
    # 2. Do BFS from this root, but only traverse until all other primary nodes are reached
    # 3. The edges traversed form the minimal subtree
    
    # Actually, even simpler:
    # In a tree, the minimal connected subtree containing a set of nodes S is the union of paths
    # between consecutive nodes in the BFS order of S.
    # Or just: take all edges that lie on any path between any two primary nodes.
    
    # Efficient method:
    # 1. Root the tree at node 1
    # 2. Compute parent and depth for each node
    # 3. Implement LCA to find paths
    # 4. For each edge, check if it's on any path between two primary nodes
    
    # Let's implement a clean version:
    
    # Root the tree at node 1
    root = 1
    parent = [-1] * (n+1)
    depth = [0] * (n+1)
    visited = [False] * (n+1)
    tree_adj = defaultdict(list)
    
    for u, v, w in edges:
        tree_adj[u].append((v, w))
        tree_adj[v].append((u, w))
    
    # BFS to build rooted tree
    queue = deque([root])
    visited[root] = True
    while queue:
        u = queue.popleft()
        for v, w in tree_adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                depth[v] = depth[u] + 1
                queue.append(v)
    
    # Function to find path edges between two nodes
    def get_path_edges(a, b):
        path_edges = set()
        # Bring a and b to same depth
        pa = a
        pb = b
        if depth[pa] > depth[pb]:
            while depth[pa] > depth[pb]:
                # Find parent edge
                p_pa = parent[pa]
                edge_id = tuple(sorted((pa, p_pa)))
                path_edges.add(edge_id)
                pa = p_pa
        elif depth[pb] > depth[pa]:
            while depth[pb] > depth[pa]:
                p_pb = parent[pb]
                edge_id = tuple(sorted((pb, p_pb)))
                path_edges.add(edge_id)
                pb = p_pb
        # Now move both up until they meet
        while pa != pb:
            p_pa = parent[pa]
            p_pb = parent[pb]
            edge_id_a = tuple(sorted((pa, p_pa)))
            edge_id_b = tuple(sorted((pb, p_pb)))
            path_edges.add(edge_id_a)
            path_edges.add(edge_id_b)
            pa = p_pa
            pb = p_pb
        return path_edges
    
    # Map edges to IDs for tracking
    edge_map = {}
    edge_list = []
    for i, (u, v, w) in enumerate(edges):
        key = tuple(sorted((u, v)))
        edge_map[key] = w
        edge_list.append(key)
    
    # Get all primary nodes as list
    primary_list = list(primary)
    
    # If only one primary node, cost is 0
    if len(primary_list) == 1:
        print(0)
        return
    
    # Collect all edges used in paths between primary nodes
    used_edges = set()
    for i in range(len(primary_list)):
        for j in range(i+1, len(primary_list)):
            a = primary_list[i]
            b = primary_list[j]
            path_e = get_path_edges(a, b)
            used_edges.update(path_e)
    
    # Calculate total cost
    total_cost = 0
    for edge_key in used_edges:
        total_cost += edge_map[edge_key]
    
    print(total_cost)

if __name__ == "__main__":
    main()
