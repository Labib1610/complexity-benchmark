We can solve this by finding the minimal subtree that connects all key patent holders.  
Since the graph is a tree, the minimal connected subgraph containing all key nodes is the **Steiner tree**, which in a tree is simply the union of all paths between key nodes.  

We can compute this by:

1. Marking key nodes.
2. Doing a DFS/BFS from any key node to find the farthest key node (diameter-like approach), but here we just need the total weight of the minimal subtree.
3. A simpler method:  
   - Root the tree at any node.  
   - For each edge, if removing it separates the key nodes into two non-empty groups, that edge must be included.  
   - Sum weights of such edges.

Implementation:

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    keys = set(map(int, input().split()))
    adj = [[] for _ in range(n+1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    # DFS to count key nodes in subtree
    parent = [0]*(n+1)
    parent_edge_weight = [0]*(n+1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
    
    key_count = [0]*(n+1)
    for u in order[::-1]:
        if u in keys:
            key_count[u] += 1
        if parent[u] != -1:
            key_count[parent[u]] += key_count[u]
    
    total = 0
    for u in range(2, n+1):
        if key_count[u] > 0 and key_count[u] < t:
            total += parent_edge_weight[u]
    
    print(total)

if __name__ == "__main__":
    solve()
```
