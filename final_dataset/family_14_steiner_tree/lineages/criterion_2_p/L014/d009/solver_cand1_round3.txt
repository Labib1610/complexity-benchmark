import sys

# The problem asks for the sum of weights of edges that are part of the unique path
# between any two key patent holders in the tree formed by the m = n-1 agreements.
# An edge is "vital" if its removal disconnects at least one pair of key patent holders.
# This is equivalent to saying the edge lies on the path between at least two key patent holders.
# In a tree, an edge (u, v) lies on the path between two key patent holders if and only if
# the two components formed by removing (u, v) both contain at least one key patent holder.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    key_holders = set()
    idx = 3
    for _ in range(t):
        key_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_keys[u] stores the number of key patent holders in the subtree rooted at u
    subtree_keys = [0] * (n + 1)
    
    # DFS to count key holders in each subtree
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in key_holders:
            subtree_keys[u] += 1
        if parent[u] != 0:
            subtree_keys[parent[u]] += subtree_keys[u]
            
    total_vital_weight = 0
    # An edge (u, v) where v is child of u is vital if 0 < subtree_keys[v] < t
    for v in range(2, n + 1):
        u = parent[v]
        # Find weight of edge (u, v)
        # Since it's a tree, we can just look at the edge list or store it during DFS
        # To be efficient, let's store the weight in the child node
        pass
        
    # Re-run to get weights
    edge_weight = [0] * (n + 1)
    for u in order:
        for v, w in adj[u]:
            if v != parent[u]:
                edge_weight[v] = w
                
    for v in range(2, n + 1):
        count = subtree_keys[v]
        if count > 0 and count < t:
            total_vital_weight += edge_weight[v]
            
    print(total_vital_weight)

solve()
