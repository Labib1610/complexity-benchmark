```python
import sys

# The problem asks us to find the sum of weights of edges that remain after 
# repeatedly removing edges connected to "marginal" nodes.
# A node is marginal if it is not a key patent holder and has degree 1.
# This is equivalent to pruning all nodes that are not part of the minimal 
# subtree connecting all key patent holders.
# The core agreements are exactly the edges in the minimal subtree 
# (Steiner Tree) that connects all key patent holders.
# Since the graph is a tree, the minimal subtree is the union of all paths 
# between any two key patent holders.
# An edge is in the core if and only if removing it splits the tree into 
# two components, both of which contain at least one key patent holder.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_holders = set()
    idx = 3
    for _ in range(t):
        key_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t == 0:
        print(0)
        return
    if t == 1:
        print(0)
        return

    # Count how many key holders are in the subtree of each node
    # We can root the tree at an arbitrary key holder
    root = next(iter(key_holders))
    
    subtree_key_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in key_holders:
            subtree_key_count[u] += 1
        if parent[u] != 0:
            subtree_key_count[parent[u]] += subtree_key_count[u]
            
    total_weight = 0
    for i in range(1, n + 1):
        if parent[i] != 0:
            # Edge between i and parent[i] is in the core if 
            # both components have at least one key holder.
            # One component has subtree_key_count[i] key holders.
            # The other has t - subtree_key_count[i].
            if subtree_key_count[i] > 0 and (t - subtree_key_count[i]) > 0:
                total_weight += parent_edge_weight[i]
                
    print(total_weight)

solve()
```
