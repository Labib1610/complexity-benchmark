import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(1000000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n, m, t = map(int, input_data[:3])
    key_holders = set(map(int, input_data[3:3+t]))
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t < 2:
        print(0)
        return

    # Kruskal's to find the MST
    edges.sort(key=lambda x: x[2])
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i: return i
        parent[i] = find(parent[i])
        return parent[i]
    
    adj = [[] for _ in range(n + 1)]
    mst_weight = 0
    for u, v, w in edges:
        root_u, root_v = find(u), find(v)
        if root_u != root_v:
            parent[root_u] = root_v
            adj[u].append((v, w))
            adj[v].append((u, w))
            mst_weight += w
            
    # Identify critical edges: edges on the path between any two key holders in the MST
    # An edge is critical if it separates the tree into two components, 
    # each containing at least one key holder.
    is_key = [False] * (n + 1)
    for k in key_holders:
        is_key[k] = True
        
    subtree_keys = [0] * (n + 1)
    critical_weight = 0
    
    # DFS to count keys in subtrees
    order = []
    stack = [(1, 0)]
    parent_node = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != p:
                parent_node[v] = u
                edge_to_parent_w[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        subtree_keys[u] += (1 if is_key[u] else 0)
        if parent_node[u] != 0:
            subtree_keys[parent_node[u]] += subtree_keys[u]
            # If this edge splits the tree into two parts with keys, it's critical
            if subtree_keys[u] > 0 and subtree_keys[u] < t:
                critical_weight += edge_to_parent_w[u]
                
    print(critical_weight)

solve()
