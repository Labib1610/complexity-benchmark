```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    specials = [0] * (n + 1)
    for _ in range(t):
        specials[int(next(it))] = 1
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    # tree traversal to get parent and order
    parent = [0] * (n + 1)
    pw = [0] * (n + 1)          # weight to parent
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v == parent[u]:
                continue
            parent[v] = u
            pw[v] = w
            stack.append(v)

    sub_special = [0] * (n + 1)
    total_special = t
    ans = 0
    for u in reversed(order):
        cnt = specials[u]
        for v, w in adj[u]:
            if parent[v] == u:          # v is child
                cnt += sub_special[v]
        sub_special[u] = cnt
        if parent[u] != -1:              # not root, evaluate edge to parent
            if cnt > 0 and total_special - cnt > 0:
                ans += pw[u]

    print(ans)

if __name__ == "__main__":
    solve()
```
