```python
import sys

# The problem states the market is a "minimally integrated market", 
# which means the graph of corporations and agreements is a tree.
# An agreement (edge) is "essential" if removing it splits the tree into 
# two components, each containing at least one key patent holder.
# This is equivalent to saying that the edge lies on the unique path 
# between at least two key patent holders.
# The union of all paths between all pairs of key patent holders forms 
# a subtree (the Steiner Tree of the key patent holders).
# Any edge in this Steiner Tree is essential.
# Any edge not in this Steiner Tree is not essential.
# We must include all essential edges. The cost is the sum of weights 
# of all edges in the Steiner Tree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_holders = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # To find the Steiner Tree, we can use a post-order traversal.
    # A node is part of the Steiner Tree if its subtree contains at least one key holder
    # AND the rest of the tree contains at least one key holder.
    
    has_key = [False] * (n + 1)
    for k in key_holders:
        has_key[k] = True
        
    total_essential_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    subtree_has_key = [has_key[i] for i in range(n + 1)]
    
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            p = parent[u]
            if subtree_has_key[u]:
                # If this subtree has a key holder, and the rest of the tree 
                # (total_keys - subtree_keys) has a key holder, the edge is essential.
                # Actually, simpler: if subtree_has_key[u] is true and 
                # (total_keys - count_in_subtree) > 0, it's essential.
                # Since it's a tree, if subtree_has_key[u] is true and 
                # there is at least one key holder outside this subtree, it's essential.
                subtree_has_key[p] = True
                
    # Now count edges where the subtree contains some but not all key holders
    subtree_key_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if has_key[i]:
            subtree_key_count[i] = 1
            
    for i in range(n - 1, 0, -1):
        u = order[i]
        if parent[u] != 0:
            subtree_key_count[parent[u]] += subtree_key_count[u]
            
    total_keys = t
    ans = 0
    for i in range(2, n + 1):
        # Edge is essential if it separates the tree into two parts, 
        # both having >= 1 key holder.
        count = subtree_key_count[i]
        if count > 0 and count < total_keys:
            ans += edge_weight[i]
            
    print(ans)

solve()
```
