```python
import sys

# The problem states that there are n corporations and m = n - 1 agreements.
# The condition "for any subset of k corporations, there are never more than k-1 
# agreements" implies the graph is a forest (a collection of trees).
# Since m = n - 1, the graph is a single tree.
# We need to find the minimum cost to connect all t patent holders.
# This is equivalent to finding the Steiner Tree in a tree, which is simply
# the sum of weights of all edges that lie on the paths between any two patent holders.
# An edge belongs to the Steiner Tree if and only if the subtree formed by 
# removing that edge contains at least one patent holder on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    patent_holders = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many patent holders are in the subtree of each node
    # Since it's a tree, we can root it at 1 and use DFS
    patent_count = [0] * (n + 1)
    for p in patent_holders:
        patent_count[p] = 1
        
    total_fee = 0
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p, w = parent[u]
        if patent_count[u] > 0 and patent_count[u] < t:
            total_fee += w
        patent_count[p] += patent_count[u]
        
    print(total_fee)

solve()
```
