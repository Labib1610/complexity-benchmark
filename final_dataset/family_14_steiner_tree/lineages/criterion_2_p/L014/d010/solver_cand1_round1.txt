```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_holders = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t < 2:
        print(0)
        return

    # subtree_keys[u] = number of key patent holders in subtree of u
    subtree_keys = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in key_holders:
            subtree_keys[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if parent[u] != 0:
            subtree_keys[parent[u]] += subtree_keys[u]
            
    total_keys = t
    ans = 0
    
    # An edge (u, v) splits the tree into two components with k and (total_keys - k) keys.
    # The edge is contested if both components have at least one key holder.
    # This happens if 0 < k < total_keys.
    for u, v, w in edges:
        # Identify which node is the child in the rooted tree
        child = u if parent[u] == v else v
        k = subtree_keys[child]
        if k > 0 and k < total_keys:
            ans += w
            
    print(ans)

if __name__ == '__main__':
    solve()
```
