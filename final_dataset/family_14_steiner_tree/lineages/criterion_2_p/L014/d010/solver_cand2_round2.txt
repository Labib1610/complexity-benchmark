```python
import sys

# The problem asks for the minimum weight of a subset of edges such that
# for any partition of the vertices into two sets S0 and S1, if there is
# at least one key patent holder in S0 and at least one in S1, then there
# is an edge in the subset connecting a vertex in S0 to a vertex in S1.
# This is equivalent to saying that the chosen edges must connect all
# key patent holders into a single connected component.
# If there are fewer than 2 key patent holders, the condition is vacuously
# true, so the minimum cost is 0.
# Otherwise, we need to find the Steiner Tree for the set of key patent
# holders. Since the graph is a tree, the Steiner Tree is the unique
# minimal subtree connecting all key patent holders.
# This is simply the sum of weights of all edges that lie on the paths
# between any two key patent holders. An edge lies on such a path if and
# only if removing it splits the tree into two components, each containing
# at least one key patent holder.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    key_holders = set()
    idx = 3
    for _ in range(t):
        key_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many key holders are in each subtree
    subtree_keys = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in key_holders:
            subtree_keys[u] += 1
        if parent[u] != -1:
            subtree_keys[parent[u]] += subtree_keys[u]
            
    total_weight = 0
    for u in range(1, n + 1):
        if parent[u] != -1:
            # Edge between u and parent[u]
            # It is part of the Steiner tree if both components have key holders
            count = subtree_keys[u]
            if count > 0 and count < t:
                # Find weight of edge (u, parent[u])
                for v, w in adj[u]:
                    if v == parent[u]:
                        total_weight += w
                        break
                        
    print(total_weight)

solve()
```
