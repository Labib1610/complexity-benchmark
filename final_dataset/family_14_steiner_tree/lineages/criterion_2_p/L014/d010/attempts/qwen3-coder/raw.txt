```python
import sys
from collections import deque, defaultdict
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    patent_holders = set(map(int, sys.stdin.readline().split()))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u, v, w))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # If there's only one patent holder, no edge can be contested
    if t <= 1:
        print(0)
        return
    
    # Compute distances from each patent holder using BFS
    # We'll do BFS from all patent holders simultaneously
    # We need to know for each node, its distance to each patent holder
    # But that's too much memory. Instead, we compute distances from each patent holder separately
    
    # Actually, better approach:
    # For each edge (u,v), we need to check if it's contested
    # That means we need to see if there exists at least one patent holder closer to u
    # and at least one patent holder closer to v
    
    # So for each patent holder, we run BFS to get distances to all nodes
    # Then for each edge, we check the distances
    
    # Let's store distance from each patent holder to each node
    dist_from_patent = {}
    
    for p in patent_holders:
        dist = [-1] * (n + 1)
        queue = deque()
        dist[p] = 0
        queue.append(p)
        while queue:
            node = queue.popleft()
            for neighbor, weight in adj[node]:
                if dist[neighbor] == -1:
                    dist[neighbor] = dist[node] + 1
                    queue.append(neighbor)
        dist_from_patent[p] = dist
    
    total_cost = 0
    
    for u, v, w in edges:
        votes_u = 0
        votes_v = 0
        for p in patent_holders:
            du = dist_from_patent[p][u]
            dv = dist_from_patent[p][v]
            if du < dv:
                votes_u += 1
            elif dv < du:
                votes_v += 1
            # If du == dv, this can't happen because the graph is a tree
        if votes_u >= 1 and votes_v >= 1:
            total_cost += w
    
    print(total_cost)

threading.Thread(target=main).start()
```
