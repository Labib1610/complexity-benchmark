```python
import sys

# The problem states that the graph is a tree (n corporations, m = n-1 agreements, 
# connected). An agreement is essential if removing it splits the tree into two 
# components, each containing at least one key patent holder.
# Let T be the tree. Removing an edge (u, v) splits T into two components.
# Let S be the set of key patent holders. Let count(u) be the number of key 
# patent holders in the subtree rooted at u (when the tree is rooted arbitrarily).
# An edge (u, v) where v is a child of u is essential if and only if 
# 0 < count(v) < t, where t is the total number of key patent holders.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_holders = set()
    idx = 3
    for _ in range(t):
        key_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # DFS to find subtree counts of key holders
    subtree_key_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in key_holders:
            subtree_key_count[u] += 1
        if parent[u] != 0:
            subtree_key_count[parent[u]] += subtree_key_count[u]
            
    total_weight = 0
    for i in range(2, n + 1):
        # Edge is between i and parent[i]
        # Removing it creates two components with key counts:
        # subtree_key_count[i] and (t - subtree_key_count[i])
        # Essential if both > 0
        c = subtree_key_count[i]
        if c > 0 and (t - c) > 0:
            total_weight += parent_edge_weight[i]
            
    print(total_weight)

if __name__ == '__main__':
    solve()
```
