import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    key_holders = set()
    idx = 3
    for _ in range(t):
        key_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need the minimal subtree containing all key_holders.
    # This is the union of all paths between any two key_holders.
    # We can find this by pruning all leaves that are not key_holders.
    
    is_key = [False] * (n + 1)
    for k in key_holders:
        is_key[k] = True
        
    degree = [0] * (n + 1)
    for i in range(1, n + 1):
        degree[i] = len(adj[i])
        
    queue = []
    for i in range(1, n + 1):
        if degree[i] == 1 and not is_key[i]:
            queue.append(i)
            
    removed = [False] * (n + 1)
    total_weight = 0
    
    # Calculate total weight of all edges
    # We will subtract weights of edges that are not part of the minimal subtree
    current_total = 0
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if u < v:
                current_total += w
                
    # Prune
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        removed[u] = True
        for v, w in adj[u]:
            if not removed[v]:
                total_weight += w
                degree[v] -= 1
                if degree[v] == 1 and not is_key[v]:
                    queue.append(v)
                    
    print(current_total - total_weight)

solve()
