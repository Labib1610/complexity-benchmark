import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def main():
    n, m, t = map(int, sys.stdin.readline().split())
    patents = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    # Sort edges by weight
    edges.sort()
    
    # Initialize DSU
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # Connect all patent holders
    cost = 0
    patents_set = set(patents)
    connected_patents = 1  # First patent is already "connected" to itself
    
    for w, u, v in edges:
        if find(parent, u) != find(parent, v):
            # Check if this edge connects to a new patent holder
            u_is_patent = u in patents_set
            v_is_patent = v in patents_set
            u_root = find(parent, u)
            v_root = find(parent, v)
            
            u_has_patent = u_root in patents_set or any(find(parent, p) == u_root for p in patents_set)
            v_has_patent = v_root in patents_set or any(find(parent, p) == v_root for p in patents_set)
            
            # If we're connecting two components and at least one has a patent
            if u_has_patent or v_has_patent:
                if union(parent, rank, u, v):
                    cost += w
                    # Update connected patents count
                    if u_is_patent and not any(find(parent, p) == find(parent, u) for p in patents if p != u):
                        connected_patents += 1
                    if v_is_patent and not any(find(parent, p) == find(parent, v) for p in patents if p != v):
                        connected_patents += 1
                        
                    # After union, check how many patents are now connected
                    patent_roots = set()
                    for p in patents:
                        patent_roots.add(find(parent, p))
                    
                    if len(patent_roots) == 1:
                        print(cost)
                        return
            else:
                # Neither component has a patent, just connect them
                union(parent, rank, u, v)

    # Alternative approach: Use Kruskal's algorithm to find MST of the induced subgraph
    # that connects all patent holders
    
    # Reset DSU
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    cost = 0
    components_with_patents = set(patents)
    num_components = len(patents)
    
    for w, u, v in edges:
        root_u = find(parent, u)
        root_v = find(parent, v)
        
        if root_u != root_v:
            # Check if either component contains a patent holder
            u_has_patent = root_u in components_with_patents
            v_has_patent = root_v in components_with_patents
            
            if u_has_patent or v_has_patent:
                union(parent, rank, u, v)
                cost += w
                
                # Update which component has patents
                new_root = find(parent, u)
                if root_u in components_with_patents:
                    components_with_patents.discard(root_u)
                if root_v in components_with_patents:
                    components_with_patents.discard(root_v)
                if u_has_patent or v_has_patent:
                    components_with_patents.add(new_root)
                
                # Count how many separate patent-containing components remain
                if len(components_with_patents) == 1:
                    print(cost)
                    return
                    
    print(cost)

if __name__ == "__main__":
    main()
