```python
import sys

# The problem states that the m proposals form a tree structure (since m = n-1).
# We are given a tree where some nodes are "patent holders".
# We need to find the minimum weight of edges that connect all patent holders.
# This is equivalent to finding the Steiner Tree in a tree, which is simply
# the sum of weights of all edges that lie on the unique path between any two
# patent holders.
# An edge in a tree is part of the path between two patent holders if and only if
# the removal of that edge splits the tree into two components, both of which
# contain at least one patent holder.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    patent_holders = set()
    idx = 3
    for _ in range(t):
        patent_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # subtree_patents[u] stores the number of patent holders in the subtree of u
    subtree_patents = [0] * (n + 1)
    for p in patent_holders:
        subtree_patents[p] = 1
        
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if parent[u] != 0:
            # If this subtree contains some but not all patent holders,
            # the edge to the parent must be included.
            if 0 < subtree_patents[u] < t:
                total_weight += edge_weight[u]
            subtree_patents[parent[u]] += subtree_patents[u]
            
    print(total_weight)

if __name__ == '__main__':
    solve()
```
