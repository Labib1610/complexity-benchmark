```python
import sys

# The problem states that the graph of proposed agreements has m = n - 1 edges
# and satisfies the condition that any subset of k vertices has at most k-1 edges.
# This means the graph is a forest (a collection of trees).
# We need to connect all t key patent holders into a single component.
# Since the graph is a forest, there is a unique path between any two nodes
# in the same tree. If two key patent holders are in different trees, they
# cannot be connected. However, the problem implies a solution exists.
# Actually, the problem asks for the minimum cost to connect all t nodes.
# Since it's a forest, we can only connect nodes that are already in the same tree.
# If all t nodes are in the same tree, the minimum cost is the sum of weights
# of the edges on the unique path connecting all t nodes.
# This is equivalent to the sum of weights of all edges in the minimal subtree
# containing all t nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n, m, t = map(int, input_data[:3])
    key_holders = list(map(int, input_data[3:3+t]))
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Find the minimal subtree containing all key_holders.
    # This is the union of paths between key_holders.
    # We can use DFS to mark nodes that are part of the minimal subtree.
    
    is_key = [False] * (n + 1)
    for k in key_holders:
        is_key[k] = True
        
    # Root the tree at the first key holder
    root = key_holders[0]
    parent = [0] * (n + 1)
    edge_to_parent = [0] * (n + 1)
    depth = [0] * (n + 1)
    has_key_in_subtree = [False] * (n + 1)
    
    stack = [(root, 0, 0, 0)]
    order = []
    while stack:
        u, p, d, w = stack.pop()
        parent[u] = p
        depth[u] = d
        edge_to_parent[u] = w
        has_key_in_subtree[u] = is_key[u]
        order.append(u)
        for v, weight in adj[u]:
            if v != p:
                stack.append((v, u, d + 1, weight))
                
    for u in reversed(order):
        if has_key_in_subtree[u] and parent[u] != 0:
            has_key_in_subtree[parent[u]] = True
            
    total_weight = 0
    for i in range(1, n + 1):
        if has_key_in_subtree[i] and parent[i] != 0:
            total_weight += edge_to_parent[i]
            
    print(total_weight)

solve()
```
