```python
import sys

# The problem states m = n - 1, meaning the graph is a tree.
# We need to find the minimum cost to connect all patent holders.
# In a tree, the path between any two nodes is unique.
# The set of edges required to connect a subset of nodes in a tree
# is the union of all paths between them.
# This is equivalent to the sum of weights of all edges that have
# at least one patent holder in each of the two components formed
# by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    patent_holders = set()
    idx = 3
    for _ in range(t):
        patent_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count[u] will store the number of patent holders in the subtree of u
    count = [0] * (n + 1)
    for p in patent_holders:
        count[p] = 1
        
    total_fee = 0
    
    # DFS to compute subtree counts and sum weights
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if parent[u] != 0:
            count[parent[u]] += count[u]
            # If this edge separates some patent holders from others, it must be included
            if count[u] > 0 and count[u] < t:
                total_fee += edge_to_parent_weight[u]
                
    print(total_fee)

solve()
```
