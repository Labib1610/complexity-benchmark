```python
import sys

# The problem states that the m agreements form a tree structure (n nodes, m=n-1 edges, 
# no cycles, connected). We need to find the minimum weight path connecting all t 
# key patent holders in this tree.
# The minimum weight path connecting a set of nodes in a tree is the sum of weights 
# of all edges that lie on the unique paths between any two of the t nodes.
# This is equivalent to the sum of weights of all edges that have at least one 
# key patent holder in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_holders = set()
    idx = 3
    for _ in range(t):
        key_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many key holders are in the subtree of each node
    subtree_key_count = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in key_holders:
            subtree_key_count[node] += 1
            
    total_fee = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_key_count[p] += subtree_key_count[u]
            # If this edge separates some key holders from others, it must be included
            if 0 < subtree_key_count[u] < t:
                total_fee += edge_to_parent_weight[u]
                
    print(total_fee)

solve()
```
