```python
import sys
from sys import stdin, stdout

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    terminals = []
    for _ in range(t):
        terminals.append(int(data[idx]) - 1)
        idx += 1
    
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(data[idx]) - 1; idx += 1
        v = int(data[idx]) - 1; idx += 1
        w = int(data[idx]); idx += 1
        adj[u].append((v, len(edges)))
        adj[v].append((u, len(edges)))
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    LOG = 1
    while (1 << LOG) < n:
        LOG += 1
    
    parent = [[-1] * n for _ in range(LOG)]
    depth = [0] * n
    parent_edge = [-1] * n
    
    stack = [0]
    parent[0][0] = -1
    depth[0] = 0
    visited = [False] * n
    visited[0] = True
    
    while stack:
        u = stack.pop()
        for v, eid in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[0][v] = u
                parent_edge[v] = eid
                depth[v] = depth[u] + 1
                stack.append(v)
    
    for k in range(1, LOG):
        for v in range(n):
            if parent[k-1][v] != -1:
                parent[k][v] = parent[k-1][parent[k-1][v]]
    
    def lca(u, v):
        if depth[u] < depth[v]:
            u, v = v, u
        diff = depth[u] - depth[v]
        for k in range(LOG):
            if (diff >> k) & 1:
                u = parent[k][u]
        if u == v:
            return u
        for k in range(LOG - 1, -1, -1):
            if parent[k][u] != parent[k][v]:
                u = parent[k][u]
                v = parent[k][v]
        return parent[0][u]
    
    terminals.sort(key=lambda x: depth[x])
    
    cnt = [0] * n
    for i in range(t - 1):
        u = terminals[i]
        v = terminals[i + 1]
        w = lca(u, v)
        cnt[u] += 1
        cnt[v] += 1
        cnt[w] -= 2
    
    stack = list(range(n))
    stack.sort(key=lambda x: -depth[x])
    
    for u in stack:
        if u == 0:
            continue
        p = parent[0][u]
        if p != -1:
            cnt[p] += cnt[u]
    
    ans = 0
    for u in range(1, n):
        if cnt[u] > 0:
            ans += edges[parent_edge[u]][2]
    
    print(ans)

solve()
```
