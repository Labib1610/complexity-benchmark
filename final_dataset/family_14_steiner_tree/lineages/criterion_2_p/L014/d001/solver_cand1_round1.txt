```python
import sys

# The problem asks for the minimum cost to connect a subset of t required sites
# in a graph where m = n - 1. Since m = n - 1 and the graph is connected,
# the graph is a tree.
# In a tree, there is a unique path between any two nodes.
# To connect a set of required nodes, we must include all edges that lie on
# the paths between any two required nodes.
# This is equivalent to finding the Steiner Tree in a tree, which is simply
# the union of all paths between the required nodes.
# This is also the minimal subtree containing all required nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    required = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that are part of the minimal subtree.
    # An edge is part of the minimal subtree if and only if the two components
    # it creates both contain at least one required node.
    
    # Count required nodes in each subtree using DFS
    subtree_required_count = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in required:
            subtree_required_count[node] = 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_required_count[p] += subtree_required_count[u]
            # If the subtree at u contains some but not all required nodes,
            # the edge (u, p) must be part of the minimal subtree.
            if 0 < subtree_required_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
