```python
import sys

# The problem states that the conduits form a tree (n outposts, n-1 conduits, 
# connected). We need to find the minimum cost to connect all primary hubs.
# In a tree, there is exactly one path between any two nodes. To connect a 
# set of nodes in a tree, we must include all edges that lie on the paths 
# between any two primary hubs. This is equivalent to the union of all paths 
# between the primary hubs.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    hubs = set()
    idx = 3
    for _ in range(t):
        hubs.add(int(input_data[idx]))
        idx += 1
        
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that are part of the minimal subtree 
    # connecting all hubs. An edge is in this subtree if and only if 
    # the two components it splits the tree into both contain at least 
    # one primary hub.
    
    # First, root the tree at an arbitrary hub (e.g., the first one in the set)
    root = next(iter(hubs))
    
    # count_hubs[u] will store the number of primary hubs in the subtree rooted at u
    count_hubs = [0] * (n + 1)
    parent_edge_cost = [0] * (n + 1)
    
    # DFS to calculate hub counts in subtrees
    stack = [(root, -1)]
    order = []
    while stack:
        u, p = stack.pop()
        order.append((u, p))
        for v, w in adj[u]:
            if v != p:
                parent_edge_cost[v] = w
                stack.append((v, u))
                
    for u, p in reversed(order):
        if u in hubs:
            count_hubs[u] += 1
        if p != -1:
            count_hubs[p] += count_hubs[u]
            
    total_cost = 0
    # An edge (u, parent(u)) is part of the Steiner tree if 
    # 0 < count_hubs[u] < t
    for i in range(1, n + 1):
        if i == root:
            continue
        if 0 < count_hubs[i] < t:
            total_cost += parent_edge_cost[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
