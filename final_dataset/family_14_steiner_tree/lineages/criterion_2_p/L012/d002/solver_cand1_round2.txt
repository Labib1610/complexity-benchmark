```python
import sys

# The problem states that there are n outposts and exactly n-1 links.
# This means the network is a tree.
# We need to connect all primary command hubs.
# In a tree, there is exactly one unique path between any two nodes.
# To connect a set of nodes in a tree, we must activate all edges that lie
# on the paths between any two primary command hubs.
# This is equivalent to finding the smallest subtree that contains all primary hubs.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    hubs = set()
    idx = 3
    for _ in range(t):
        hubs.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to identify edges that are part of the minimal subtree.
    # An edge is part of the minimal subtree if and only if both components
    # formed by removing that edge contain at least one primary hub.
    
    # Root the tree at an arbitrary hub (hubs[0])
    root = next(iter(hubs))
    
    # subtree_hubs[u] = number of primary hubs in the subtree rooted at u
    subtree_hubs = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in hubs:
            subtree_hubs[u] += 1
        if parent[u] != 0:
            subtree_hubs[parent[u]] += subtree_hubs[u]
            # If the subtree at u contains at least one hub and the rest of the
            # tree contains at least one hub, the edge (u, parent[u]) is needed.
            if subtree_hubs[u] > 0 and subtree_hubs[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
