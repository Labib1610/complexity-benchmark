```python
import sys

# The problem states that the blueprint forms a tree (n nodes, n-1 edges, connected).
# We need to connect all t Root Authorities.
# In a tree, there is a unique path between any two nodes.
# To connect a set of nodes in a tree, we must activate all edges that lie on the
# unique paths between any pair of these nodes.
# This is equivalent to activating all edges that lie on the unique path between
# the "lowest" common ancestor and the nodes, or more simply, all edges that
# belong to the minimal subtree containing all Root Authorities.
# An edge belongs to this minimal subtree if and only if the two components
# formed by removing that edge both contain at least one Root Authority.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    roots = set()
    idx = 3
    for _ in range(t):
        roots.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many roots are in the subtree of each node.
    # Since it's a tree, we can use DFS.
    # To avoid recursion depth issues, use iterative DFS.
    
    root_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in roots:
        root_count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If the subtree at u contains some roots but not all,
            # the edge (u, p) must be part of the minimal subtree.
            if root_count[u] > 0 and root_count[u] < t:
                total_cost += edge_to_parent_weight[u]
            root_count[p] += root_count[u]
            
    print(total_cost)

solve()
```
