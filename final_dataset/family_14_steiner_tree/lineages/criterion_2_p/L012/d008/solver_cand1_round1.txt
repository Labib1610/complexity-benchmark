```python
import sys

# The problem states that there are n enclaves and m = n - 1 bridges.
# The redundancy constraint (for any group of k enclaves, the number of bridges
# between them is at most k - 1) implies that the graph is a forest (a collection of trees).
# Since m = n - 1, the graph is a single tree.
# We need to find the minimum cost to connect all t Root Authorities.
# This is equivalent to finding the sum of weights of all edges that lie on the
# unique paths between any two Root Authorities.
# An edge is part of the path between two Root Authorities if and only if
# removing that edge splits the tree into two components, each containing at least one Root Authority.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    roots = set()
    idx = 3
    for _ in range(t):
        roots.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # subtree_roots[u] stores the number of Root Authorities in the subtree of u
    subtree_roots = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(1, 0, 0)] # node, parent, weight_to_parent
    order = []
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    
    while stack:
        u, p, w = stack.pop()
        order.append(u)
        parent[u] = p
        weight_to_parent[u] = w
        if u in roots:
            subtree_roots[u] = 1
        for v, weight in adj[u]:
            if v != p:
                stack.append((v, u, weight))
                
    # Process in reverse topological order
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        subtree_roots[p] += subtree_roots[u]
        # If this edge separates the tree into two parts, both having at least one root
        if subtree_roots[u] > 0 and subtree_roots[u] < t:
            total_cost += weight_to_parent[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
