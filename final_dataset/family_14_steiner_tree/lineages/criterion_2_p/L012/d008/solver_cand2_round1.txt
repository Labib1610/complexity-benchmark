```python
import sys

# The problem asks for the minimum cost to select a subset of edges (e_u - e_v)
# such that for any two target dimensions a and b, e_a - e_b is in the span.
# The vectors e_u - e_v correspond to edges in a graph with n vertices.
# The span of a set of edges {e_u - e_v} contains e_a - e_b if and only if
# a and b are in the same connected component of the graph formed by these edges.
# Thus, we need all target dimensions to be in the same connected component.
# Since we are given m = n - 1 edges and they span the (n-1)-dimensional subspace,
# the graph is a tree.
# We need to select a subset of edges such that all target dimensions are in the
# same connected component. This is equivalent to finding the minimum cost
# to connect all target nodes in the tree.
# In a tree, the unique path between any two nodes is fixed. To connect a set
# of nodes, we must include all edges that lie on the paths between them.
# This is the Steiner Tree problem on a tree, which is simply the union of all
# paths between the target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that lie on the path between any two target nodes.
    # An edge (u, v) is part of the Steiner tree if and only if the subtree
    # formed by removing the edge contains at least one target node in both components.
    
    # Root the tree at an arbitrary target node
    root = next(iter(targets))
    
    # count_targets[u] = number of target nodes in the subtree rooted at u
    count_targets = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            count_targets[u] += 1
        if parent[u] != 0:
            count_targets[parent[u]] += count_targets[u]
            # If this edge separates the tree into two parts, both containing
            # at least one target, then this edge must be included.
            if 0 < count_targets[u] < t:
                total_cost += edge_weight[u]
                
    print(total_cost)

solve()
```
