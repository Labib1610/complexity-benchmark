```python
import sys

# The problem states the molecule is a tree (n nodes, n-1 edges, connected).
# We need to find the minimum weight of edges to keep all t tagged nodes connected.
# This is equivalent to finding the sum of weights of all edges that lie on the 
# unique paths between any two tagged nodes.
# An edge (u, v) with weight w lies on the path between two tagged nodes if and only if
# the removal of (u, v) splits the tree into two components, each containing at least one tagged node.
# Let T be the total number of tagged nodes.
# If we root the tree at an arbitrary node, for an edge (u, v) where v is the child of u,
# let count[v] be the number of tagged nodes in the subtree rooted at v.
# The edge (u, v) is part of the required fragment if 0 < count[v] < T.
# If this condition holds, the edge must be included.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    tagged = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        tagged[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    count = [0] * (n + 1)
    for i in range(1, n + 1):
        if tagged[i]:
            count[i] = 1
            
    edge_weight = {}
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if v != parent[u]:
                edge_weight[v] = w
                
    total_stabilizer = 0
    # Process in reverse topological order
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != -1:
            p = parent[u]
            count[p] += count[u]
            if 0 < count[u] < t:
                total_stabilizer += edge_weight[u]
                
    print(total_stabilizer)

solve()
```
