import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    targets = set()
    for _ in range(t):
        targets.add(int(data[idx]) - 1)  # Convert to 0-indexed
        idx += 1
    
    edges = []
    for _ in range(m):
        u = int(data[idx]) - 1  # Convert to 0-indexed
        idx += 1
        v = int(data[idx]) - 1  # Convert to 0-indexed
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((w, u, v))
    
    if t <= 1:
        print(0)
        return
    
    # Sort edges by weight
    edges.sort()
    
    # We want to connect all target nodes in a spanning tree with minimal cost
    # This is essentially finding the Minimum Spanning Tree that connects all targets
    # We can use a modified Kruskal's algorithm
    
    parent = list(range(n))
    rank = [0] * n
    targets_connected = set()
    total_cost = 0
    edges_used = 0
    
    for w, u, v in edges:
        # If both u and v are targets and they are not connected yet, we must connect them
        pu = find(parent, u)
        pv = find(parent, v)
        
        if pu != pv:
            # Check if connecting this edge helps connect target components
            # We connect it if at least one of the endpoints is a target or 
            # if it connects two different components where at least one has a target
            
            # Actually, let's think differently:
            # We want the final graph to have all targets in the same component
            # So we run Kruskal but only until all targets are connected
            # That means we stop when we have connected all targets into one component
            
            # Let's track how many separate target components there are
            # Initially each target is its own component
            # When we unite two components, if both had targets, we decrease the count
            
            # Simpler approach:
            # Run Kruskal's algorithm until all targets are in the same component
            # But tracking if the current union operation connects two components that contain targets
            
            # Even simpler: just run Kruskal and check at each step if all targets are connected
            # But that's inefficient
            
            # Best approach:
            # Use Union-Find and keep track of which components contain targets
            # When we unite two components, we update the target information
            # Stop when all targets are in the same component
            
            # Let's implement this properly:
            
            # Keep track of which components contain targets
            # For each component root, store the set of targets in that component
            # But this could be memory intensive
            
            # Better idea:
            # Just run standard Kruskal and stop when all targets are connected
            # We can check if all targets are in the same component by checking if find(target) is the same for all
            
            union(parent, rank, u, v)
            total_cost += w
            edges_used += 1
            
            # Check if all targets are now connected
            # Get the root of the first target
            first_root = find(parent, next(iter(targets)))
            all_connected = True
            for target in targets:
                if find(parent, target) != first_root:
                    all_connected = False
                    break
            
            if all_connected:
                break
    
    print(total_cost)

if __name__ == "__main__":
    main()
