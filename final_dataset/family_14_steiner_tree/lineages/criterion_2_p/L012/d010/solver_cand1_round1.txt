```python
import sys

# The problem describes a graph where each operation (u, v) with cost w is an edge.
# The matrix L is the Laplacian matrix of the graph formed by the chosen edges.
# The condition Lx = 0 implies x_a = x_b for all a, b in the primary set T
# if and only if all nodes in T belong to the same connected component in the graph.
# Since we are given m = n - 1 edges and the full graph is connected,
# the graph is a tree.
# To make all nodes in T belong to the same connected component, we must select
# a subset of edges such that all nodes in T are connected.
# This is equivalent to finding the Steiner Tree for the set T in the given tree.
# In a tree, the Steiner Tree is the unique minimal subtree connecting all nodes in T.
# The cost is the sum of weights of all edges on the paths between nodes in T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary_indices = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Identify edges that are part of the Steiner Tree.
    # An edge is in the Steiner Tree if and only if the subtree on one side
    # contains at least one node from T and the subtree on the other side
    # contains at least one node from T.
    
    has_primary = [False] * (n + 1)
    for p in primary_indices:
        has_primary[p] = True
        
    total_cost = 0
    
    # Root the tree at node 1 and perform DFS to count primary nodes in subtrees
    subtree_primary_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        subtree_primary_count[u] += (1 if has_primary[u] else 0)
        if parent[u] != 0:
            subtree_primary_count[parent[u]] += subtree_primary_count[u]
            
    for i in range(2, n + 1):
        count = subtree_primary_count[i]
        if count > 0 and count < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
