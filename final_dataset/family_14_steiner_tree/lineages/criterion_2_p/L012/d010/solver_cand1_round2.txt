```python
import sys

# The problem describes a graph where each operation (u, v) with cost w
# corresponds to an edge between nodes u and v.
# The matrix M is the Laplacian matrix of the graph formed by the chosen edges.
# The condition that x in the null space of M satisfies x_{p_1} = x_{p_2} = ... = x_{p_t}
# is equivalent to saying that all nodes p_1, p_2, ..., p_t must be in the same
# connected component in the graph formed by the chosen edges.
# Since we want to minimize the total cost to connect these t nodes,
# and we are given m = n - 1 edges (a tree structure), we need to find the
# minimum cost to connect the set of required nodes {p_1, ..., p_t}.
# This is the Steiner Tree problem, but since the graph is a tree,
# the unique path between any two nodes is fixed.
# The set of edges that must be included are those that lie on the paths
# between any pair of the required nodes.
# This is equivalent to the union of paths between all pairs of required nodes,
# which is the minimal subtree containing all required nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    required = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Root the tree at the first required node
    root = required[0]
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    depth = [0] * (n + 1)
    
    stack = [(root, 0, 0, 0)]
    while stack:
        u, p, d, w = stack.pop()
        parent[u] = p
        depth[u] = d
        edge_weight[u] = w
        for v, weight in adj[u]:
            if v != p:
                stack.append((v, u, d + 1, weight))
                
    # Mark nodes that are part of the minimal subtree
    is_required = [False] * (n + 1)
    for r in required:
        is_required[r] = True
        
    # Use a bottom-up approach to mark nodes
    # A node is in the subtree if it is required or has a descendant that is required
    in_subtree = [False] * (n + 1)
    
    # Sort nodes by depth descending to process leaves first
    nodes_by_depth = sorted(range(1, n + 1), key=lambda x: depth[x], reverse=True)
    
    total_cost = 0
    for u in nodes_by_depth:
        if is_required[u]:
            in_subtree[u] = True
        if in_subtree[u] and parent[u] != 0:
            in_subtree[parent[u]] = True
            total_cost += edge_weight[u]
            
    print(total_cost)

solve()
```
