```python
import sys

# The problem asks for the minimum cost to connect all t primary elements
# in the same connected component using a subset of the given n-1 transpositions.
# Since there are n-1 transpositions and they generate S_n, the graph formed
# by these transpositions is a tree.
# To make all t primary elements belong to the same connected component,
# we need to select the edges (transpositions) that form the minimal subtree
# connecting all t primary elements.
# This is equivalent to finding the Steiner Tree for the set of t primary elements
# in the given tree. In a tree, the Steiner Tree is the union of all paths
# between the primary elements, which is the minimal subtree containing all of them.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary_elements = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the minimal subtree containing all primary elements:
    # 1. Root the tree at an arbitrary primary element.
    # 2. A node is part of the minimal subtree if it is a primary element
    #    or if it has at least two branches containing primary elements.
    # Alternatively:
    # A node is in the minimal subtree if it lies on a path between two primary elements.
    # An edge is in the minimal subtree if removing it splits the primary elements
    # into two non-empty sets.
    
    # Count primary elements in each subtree
    has_primary = [0] * (n + 1)
    for p in primary_elements:
        has_primary[p] = 1
        
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order to propagate counts
    edge_cost = {}
    for u in reversed(order):
        if parent[u] != -1:
            p = parent[u]
            # Find weight of edge (u, p)
            w = 0
            for v, weight in adj[u]:
                if v == p:
                    w = weight
                    break
            
            if has_primary[u] > 0 and has_primary[u] < t:
                total_cost += w
            
            has_primary[p] += has_primary[u]
            
    print(total_cost)

solve()
```
