```python
import sys

# The problem asks for the minimum cost to connect all t core entities.
# Since the graph is a tree (n nodes, n-1 edges, connected), there is a unique
# path between any two nodes. To connect a set of nodes in a tree, we must
# include all edges that lie on the paths between any pair of these nodes.
# This is equivalent to the sum of weights of all edges that, when removed,
# split the tree into two components such that both components contain at least
# one core entity.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    core_entities = set()
    idx = 3
    for _ in range(t):
        core_entities.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We use DFS to count how many core entities are in the subtree of each node.
    # An edge (u, v) where v is a child of u is part of the Steiner tree if
    # the number of core entities in v's subtree is > 0 and < t.
    
    core_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in core_entities:
            core_count[node] = 1
            
    for node in reversed(order):
        if node == 1:
            continue
        p = parent[node]
        core_count[p] += core_count[node]
        if 0 < core_count[node] < t:
            total_cost += edge_to_parent_weight[node]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
