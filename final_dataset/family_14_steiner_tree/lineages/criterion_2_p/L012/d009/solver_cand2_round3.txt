```python
import sys

# The problem states that the m = n - 1 regulations form a tree structure.
# We need to find the minimum cost to connect all benchmark commodities.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of nodes in a tree, we must include all edges that lie
# on the paths between any two benchmark nodes.
# This is equivalent to finding the smallest subtree that contains all
# benchmark nodes.
# An edge (u, v) with weight w is part of this subtree if and only if
# the removal of this edge splits the tree into two components, each
# containing at least one benchmark node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    benchmarks = set()
    idx = 3
    for _ in range(t):
        benchmarks.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many benchmark nodes are in the subtree rooted at u
    # when the tree is rooted at an arbitrary node (e.g., 1).
    count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    parent = [0] * (n + 1)
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in benchmarks:
            count[u] += 1
        if parent[u] != 0:
            count[parent[u]] += count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the subtree at i contains some but not all benchmarks,
        # the edge connecting i to its parent must be included.
        if count[i] > 0 and count[i] < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
