```python
import sys

# The problem asks for the minimum cost to connect all Primary Registers
# in a tree structure. Since the locks form a tree (n nodes, n-1 edges),
# the path between any two nodes is unique.
# To make all t Primary Registers share the same phase state, we must
# engage all locks that lie on the paths connecting these registers.
# This is equivalent to finding the union of all paths between the
# Primary Registers in the tree.
# The edges that must be included are those that separate the tree into
# components such that no component contains more than one Primary Register.
# Equivalently, an edge must be included if and only if removing it
# splits the tree into two components, both of which contain at least one
# Primary Register.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary_registers = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count[u] will store the number of primary registers in the subtree
    # rooted at u (when rooted at 1).
    count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in primary_registers:
            count[u] += 1
            
    # Process in reverse topological order to propagate counts
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            p = parent[u]
            count[p] += count[u]
            # If the edge (u, p) separates the tree into two parts
            # both containing at least one primary register, it must be included.
            if count[u] > 0 and count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
