```python
import sys

# The problem asks for the minimum cost to connect all benchmark commodities.
# Since the regulations form a tree, the set of edges that must be included
# are those that lie on the unique path between any two benchmark commodities.
# This is equivalent to the union of all paths between all pairs of benchmark
# commodities, which is the minimal subtree containing all benchmark nodes.
# We can solve this by rooting the tree arbitrarily (e.g., at node 1) and
# checking for each edge whether it lies on a path between two benchmark nodes.
# An edge (u, v) where v is a child of u lies on such a path if and only if
# the subtree rooted at v contains at least one benchmark node AND the
# rest of the tree (outside the subtree of v) also contains at least one
# benchmark node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    benchmarks = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Root the tree at 1
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Count benchmark nodes in subtree
    subtree_benchmarks = [0] * (n + 1)
    for node in benchmarks:
        subtree_benchmarks[node] = 1
        
    for u in reversed(order):
        if parent[u] != 0:
            subtree_benchmarks[parent[u]] += subtree_benchmarks[u]
            
    total_cost = 0
    # An edge (u, v) where v is child of u is needed if:
    # 0 < subtree_benchmarks[v] < t
    for i in range(2, n + 1):
        count = subtree_benchmarks[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

solve()
```
