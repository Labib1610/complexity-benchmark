```python
import sys

# The problem states that the network is a tree (n nodes, n-1 edges, connected).
# We need to find the minimum cost to connect all t Auditing Offices.
# In a tree, the unique path between any two nodes is fixed.
# The union of all paths between all pairs of Auditing Offices is the minimal
# subtree that contains all Auditing Offices.
# We need to sum the weights of all edges that lie on at least one path
# between two Auditing Offices.
# An edge (u, v) with weight w is part of this subtree if and only if
# removing it splits the tree into two components, each containing at least
# one Auditing Office.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    auditing_offices = set()
    idx = 3
    for _ in range(t):
        auditing_offices.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count[u] will store the number of Auditing Offices in the subtree rooted at u
    count = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in auditing_offices:
            count[u] += 1
            
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        p = parent[u]
        count[p] += count[u]
        # If the subtree at u contains some but not all auditing offices,
        # the edge connecting u to its parent must be included.
        if count[u] > 0 and count[u] < t:
            total_cost += edge_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
