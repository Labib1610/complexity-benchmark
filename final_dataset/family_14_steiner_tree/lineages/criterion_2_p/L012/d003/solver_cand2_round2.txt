```python
import sys

# The problem asks for the minimum weight of a subtree that connects all t Auditing Offices.
# Since the graph is a tree (n nodes, n-1 edges, connected), the unique path between any two
# nodes is the only way to connect them.
# The union of all paths between all pairs of Auditing Offices forms the minimal subtree
# that connects all Auditing Offices.
# This is equivalent to the sum of weights of all edges that have at least one Auditing Office
# in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    auditing_offices = set()
    idx = 3
    for _ in range(t):
        auditing_offices.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use iterative DFS to avoid recursion depth issues
    # count[u] will store the number of auditing offices in the subtree rooted at u
    count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in auditing_offices:
            count[u] = 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if parent[u] != -1:
            p = parent[u]
            count[p] += count[u]
            # If the edge to parent separates some auditing offices from others,
            # it must be included.
            if count[u] > 0 and count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
