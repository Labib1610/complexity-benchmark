```python
import sys

# The problem states that the network has n nodes and n-1 edges.
# It also states that all t Auditing Offices are in a single connected component
# which contains no cycles (a tree).
# Since m = n - 1 and the component containing the Auditing Offices is a tree,
# the entire graph is actually a forest where one component is a tree containing
# all Auditing Offices, and other components might contain cycles.
# However, the problem asks for the minimum cost to connect all Auditing Offices.
# Since the Auditing Offices are all in one tree component, we only need to
# consider the edges in that specific tree component.
# Any edge not in the tree component containing the Auditing Offices
# cannot help connect any Auditing Offices.
# Thus, we need to find the minimum Steiner Tree in the tree component
# containing all Auditing Offices.
# In a tree, the Steiner Tree is simply the unique minimal subtree
# connecting all specified nodes. This is the union of all paths between
# the Auditing Offices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    auditing_offices = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Find the tree component containing the Auditing Offices.
    # We can use DFS to find which nodes are part of the minimal subtree.
    # A node is in the minimal subtree if it is an Auditing Office
    # or if it lies on a path between two Auditing Offices.
    # This is equivalent to saying a node is in the subtree if the subtree
    # rooted at this node (in the tree component) contains at least one
    # Auditing Office and the part of the tree "above" it also contains
    # at least one Auditing Office.
    
    # First, pick an arbitrary Auditing Office as root.
    root = next(iter(auditing_offices))
    
    # has_office[u] will be true if the subtree at u contains an Auditing Office.
    has_office = [False] * (n + 1)
    for office in auditing_offices:
        has_office[office] = True
        
    total_cost = 0
    
    # We need to traverse the tree. Since it's a tree, we can use DFS.
    # To avoid recursion depth issues, use iterative DFS.
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == root:
            continue
        if has_office[u]:
            total_cost += edge_to_parent_weight[u]
            has_office[parent[u]] = True
            
    print(total_cost)

solve()
```
