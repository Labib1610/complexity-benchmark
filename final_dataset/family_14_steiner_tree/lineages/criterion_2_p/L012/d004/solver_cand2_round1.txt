```python
import sys

# The problem describes a tree structure where variables are nodes and assertions are edges.
# We need to find the minimum cost to connect all t target variables.
# This is equivalent to finding the weight of the minimal subtree that contains all t target nodes.
# In a tree, the minimal subtree containing a set of nodes is the union of all paths 
# between these nodes. This is equivalent to the sum of weights of all edges that 
# have at least one target node in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many target nodes are in each subtree.
    # Root the tree at an arbitrary target node (e.g., the first one).
    root = next(iter(targets))
    
    target_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(root, -1)]
    order = []
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        parent[u] = p
        for v, w in adj[u]:
            if v != p:
                edge_to_parent_weight[v] = w
                stack.append((v, u))
                
    # Process in reverse topological order to propagate target counts
    for u in reversed(order):
        if u in targets:
            target_count[u] += 1
        
        p = parent[u]
        if p != -1:
            target_count[p] += target_count[u]
            # If this edge separates some targets from others, it must be included
            if target_count[u] > 0 and target_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    # Increase recursion depth just in case, though iterative is used
    sys.setrecursionlimit(300000)
    solve()
```
