import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    critical = [int(next(it)) for _ in range(t)]
    is_critical = [False] * (n + 1)
    for x in critical:
        is_critical[x] = True

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))

    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = []
    stack = [(1, 0, 0)]
    while stack:
        node, par, pw = stack.pop()
        parent[node] = par
        parent_weight[node] = pw
        order.append(node)
        for nb, w in adj[node]:
            if nb != par:
                stack.append((nb, node, w))

    dp = [0] * (n + 1)
    for u in reversed(order):
        if is_critical[u]:
            dp[u] = parent_weight[u]
        else:
            child_sum = 0
            for nb, w in adj[u]:
                if nb != parent[u]:
                    child_sum += dp[nb]
            dp[u] = min(parent_weight[u], child_sum)

    print(dp[1])

if __name__ == "__main__":
    solve()
