```python
import sys

# The problem asks for the minimum cost to select a subset of edges in a tree
# such that all target variables are in the same connected component.
# Since the input structure is a tree (n variables, n-1 edges, no redundant terms),
# the condition that all target variables are equal means they must all be
# connected in the subgraph formed by the selected edges.
# To minimize the cost, we need to find the minimum weight set of edges that
# connects all target variables. This is equivalent to finding the Steiner Tree
# for the set of target variables in the given tree.
# In a tree, the Steiner Tree is simply the union of all paths between the
# target variables. This is the unique minimal subtree containing all target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the edges that lie on the paths between any two target nodes.
    # An edge is part of the Steiner Tree if and only if both components formed
    # by removing that edge contain at least one target node.
    
    # Count how many target nodes are in the subtree rooted at each node.
    # We can root the tree at an arbitrary target node.
    root = next(iter(targets))
    
    subtree_targets = [0] * (n + 1)
    edge_costs = {} # (u, v) -> weight
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_costs[v] = w
                stack.append(v)
                
    for u in order:
        if u in targets:
            subtree_targets[u] += 1
            
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == root:
            continue
        p = parent[u]
        subtree_targets[p] += subtree_targets[u]
        # If the subtree at u contains some but not all targets, the edge (u, p) is needed
        if 0 < subtree_targets[u] < t:
            total_cost += edge_costs[u]
            
    print(total_cost)

solve()
```
