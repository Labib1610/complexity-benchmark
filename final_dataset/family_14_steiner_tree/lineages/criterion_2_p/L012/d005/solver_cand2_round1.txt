```python
import sys

# The problem states m = n - 1 and the layout is a tree.
# We need to find the minimum cost to connect all t primary stations.
# In a tree, the set of edges required to connect a subset of nodes is 
# exactly the union of all paths between any two nodes in that subset.
# This is equivalent to the Steiner Tree problem on a tree, which is 
# simply the sum of weights of all edges that lie on the path between 
# any two primary stations.
# An edge lies on the path between two primary stations if and only if 
# removing that edge splits the tree into two components, each containing 
# at least one primary station.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary_stations = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many primary stations are in each subtree
    subtree_primary_count = [0] * (n + 1)
    
    # Since it's a tree, we can use DFS or BFS to compute counts
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in order:
        if u in primary_stations:
            subtree_primary_count[u] += 1
            
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        # Find weight of edge (u, p)
        weight = 0
        for v, w in adj[u]:
            if v == p:
                weight = w
                break
        
        # If this subtree contains some but not all primary stations,
        # the edge to the parent must be included.
        if 0 < subtree_primary_count[u] < t:
            total_cost += weight
        
        subtree_primary_count[p] += subtree_primary_count[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
