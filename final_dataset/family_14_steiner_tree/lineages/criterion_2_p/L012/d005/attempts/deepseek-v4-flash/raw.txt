We have a tree with \( n \) nodes and \( n-1 \) edges, each edge has a cost.  
We need to select a subset of edges such that all critical nodes are in the same connected component.  
The minimum cost is the cost of the minimal subtree connecting all critical nodes.

This is a classic **Steiner tree** problem on a tree, which can be solved by:

1. Mark critical nodes.
2. Remove all subtrees that do not contain any critical node.
3. The remaining tree is the minimal subtree connecting all critical nodes.
4. Sum the edge costs in that remaining tree.

We can do this efficiently with DFS.

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    critical = set(map(int, input().split()))
    
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # DFS to find which edges are needed
    # We'll root at any critical node, say first one
    root = next(iter(critical))
    
    parent = [-1] * n
    parent_edge_cost = [0] * n
    stack = [(root, -1)]
    order = []
    while stack:
        u, p = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v == p:
                continue
            parent[v] = u
            parent_edge_cost[v] = w
            stack.append((v, u))
    
    # Post-order: determine if subtree contains critical node
    has_critical = [False] * n
    for u in critical:
        has_critical[u] = True
    
    total_cost = 0
    for u in reversed(order):
        if u == root:
            continue
        if has_critical[u]:
            total_cost += parent_edge_cost[u]
            # propagate up
            p = parent[u]
            has_critical[p] = True
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
