```python
import sys

# The problem asks for the minimum cost to select a subset of edges in a tree
# such that all critical nodes are in the same connected component.
# If we select a subset of edges, the condition F(x) = 0 implies that for every
# selected edge (u, v), x_u = x_v. This means all nodes in the same connected
# component of the selected edges must have equal values.
# To ensure all t critical nodes have equal values, they must all belong to the
# same connected component in the graph formed by the selected edges.
# Since the original graph is a tree (n nodes, n-1 edges, connected),
# the condition is equivalent to saying that the unique path between any two
# critical nodes must be included in the selected subset of edges.
# The union of all paths between any two critical nodes in a tree is the
# minimal subtree that contains all critical nodes (the Steiner Tree).
# The cost is the sum of the weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # To find the sum of weights of the minimal subtree containing all critical nodes:
    # A node is in the minimal subtree if and only if it lies on a path between
    # two critical nodes.
    # Alternatively, an edge is in the minimal subtree if and only if removing it
    # splits the tree into two components, each containing at least one critical node.
    
    # Count critical nodes in each subtree
    subtree_critical_count = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for node in critical_nodes:
        subtree_critical_count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_critical_count[p] += subtree_critical_count[u]
            # If this subtree contains some but not all critical nodes,
            # the edge to the parent must be part of the Steiner tree.
            if 0 < subtree_critical_count[u] < t:
                total_cost += edge_weight_to_parent[u]
                
    print(total_cost)

solve()
```
