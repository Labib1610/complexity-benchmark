```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))   # = n-1
    t = int(next(it))
    targets = set()
    for _ in range(t):
        targets.add(int(next(it)))
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    if t <= 1:
        print(0)
        return

    total_targets = t
    cnt = [0] * (n + 1)
    ans = 0

    # iterative DFS post-order
    stack = [(1, 0, 0)]  # node, parent, state 0=pre,1=post
    while stack:
        node, parent, state = stack.pop()
        if state == 0:
            stack.append((node, parent, 1))
            for nb, w in adj[node]:
                if nb == parent:
                    continue
                stack.append((nb, node, 0))
        else:
            c = 1 if node in targets else 0
            for nb, w in adj[node]:
                if nb == parent:
                    continue
                c_child = cnt[nb]
                if 0 < c_child < total_targets:
                    ans += w
                c += c_child
            cnt[node] = c

    print(ans)

if __name__ == "__main__":
    solve()
```
