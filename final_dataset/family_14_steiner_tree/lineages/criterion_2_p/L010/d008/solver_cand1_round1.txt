```python
import sys

# The problem asks for the minimum cost to connect all target dimensions in a graph
# where each edge (u, v) with weight w represents the vector e_u - e_v.
# The condition "e_ca - e_cb is in the span" is equivalent to saying that
# in the graph where edges are the selected vectors, there is a path between
# target dimension ca and target dimension cb.
# Since we need this for all pairs of target dimensions, all target dimensions
# must belong to the same connected component.
# We are given m = n - 1 edges, and the graph is guaranteed to span all e_i - e_j.
# This means the graph is a tree.
# In a tree, there is a unique path between any two nodes.
# To connect all target nodes, we must select all edges that lie on the unique
# paths between any two target nodes.
# This is equivalent to selecting all edges that are part of the minimal subtree
# containing all target nodes (the Steiner Tree in a tree).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Find the minimal subtree containing all targets.
    # An edge is in the minimal subtree if and only if both components
    # formed by removing the edge contain at least one target node.
    
    # Root the tree at an arbitrary target node.
    root = next(iter(targets))
    
    # subtree_targets[u] = number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            subtree_targets[u] += 1
        if parent[u] != 0:
            subtree_targets[parent[u]] += subtree_targets[u]
            # If the subtree at u contains some targets but not all,
            # the edge (parent[u], u) must be part of the minimal subtree.
            if 0 < subtree_targets[u] < t:
                total_cost += edge_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
