```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes in a tree.
# This is equivalent to finding the weight of the Steiner Tree in a tree,
# which is simply the sum of weights of all edges that lie on the unique
# path between any two spore-containing nodes.
# An edge is part of the Steiner Tree if and only if the subtree formed by
# removing that edge contains at least one spore-containing node on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    spores = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to count how many spores are in the subtree of each node.
    # Since n is up to 200,000, we use an iterative DFS to avoid recursion limits.
    has_spore = [0] * (n + 1)
    for s in spores:
        has_spore[s] = 1
        
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to propagate spore counts
    total_cost = 0
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if has_spore[u] > 0 and has_spore[u] < t:
            total_cost += edge_to_parent_weight[u]
        has_spore[p] += has_spore[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
