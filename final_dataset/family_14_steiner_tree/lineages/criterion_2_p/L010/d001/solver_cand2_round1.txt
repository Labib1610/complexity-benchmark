```python
import sys

# The problem states that the m threads form the minimal number of threads 
# to connect n vesicles, which means the graph is a tree.
# We need to find the minimum cost to connect all t fruiting body vesicles.
# In a tree, there is a unique path between any two nodes.
# The set of edges required to connect a subset of nodes in a tree is the 
# union of all paths between these nodes. This is equivalent to the 
# minimal subtree containing all the target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    fruiting_bodies = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the minimal subtree, we can prune the tree.
    # Any leaf node that is not a fruiting body can be removed iteratively.
    # Alternatively, use DFS to mark nodes that are part of the minimal subtree.
    
    is_fruiting = [False] * (n + 1)
    for f in fruiting_bodies:
        is_fruiting[f] = True
        
    # subtree_has_fruiting[u] will be true if the subtree rooted at u contains a fruiting body
    subtree_has_fruiting = [False] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_cost = [0] * (n + 1)
    order = []
    stack = [fruiting_bodies[0]]
    visited = [False] * (n + 1)
    visited[fruiting_bodies[0]] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_cost[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_fruiting[u]:
            subtree_has_fruiting[u] = True
        if subtree_has_fruiting[u] and parent[u] != 0:
            subtree_has_fruiting[parent[u]] = True
            total_cost += edge_to_parent_cost[u]
            
    print(total_cost)

solve()
```
