```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes in a tree.
# Since the graph is a tree (n nodes, n-1 edges, fully contiguous), 
# the minimum cost to connect a set of nodes is the sum of the weights of 
# all edges that lie on the paths between any two nodes in the set.
# An edge is part of the required set if and only if it splits the tree into 
# two components, both of which contain at least one spore-bearing junction.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    spores = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # spore_count[u] will store the number of spore-bearing junctions in the subtree of u
    spore_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u in spores:
            spore_count[u] += 1
        
        for v, w in adj[u]:
            if v == parent[u]:
                continue
            
            # If the subtree at v contains some spores but not all, 
            # the edge (u, v) must be included.
            if 0 < spore_count[v] < t:
                total_cost += w
            
            spore_count[u] += spore_count[v]
            
    print(total_cost)

solve()
```
