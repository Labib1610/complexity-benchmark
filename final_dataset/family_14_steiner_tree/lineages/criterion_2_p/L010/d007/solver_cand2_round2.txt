```python
import sys

# The problem describes a tree structure (n nodes, n-1 edges).
# We need to connect all target variables together.
# The minimum set of edges to connect a subset of nodes in a tree
# is the union of all paths between any two target nodes.
# This is equivalent to the sum of weights of all edges that lie on
# at least one path between two target nodes.
# An edge (u, v) lies on a path between two target nodes if and only if
# the removal of that edge splits the tree into two components,
# each containing at least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use DFS to count how many target nodes are in each subtree
    # Since n is up to 200,000, use iterative DFS to avoid recursion depth issues
    target_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in targets:
            target_count[u] = 1
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    total_penalty = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != -1:
            p = parent[u]
            target_count[p] += target_count[u]
            # If this subtree contains some but not all targets, the edge must be enforced
            if 0 < target_count[u] < t:
                total_penalty += edge_weight[u]
                
    print(total_penalty)

if __name__ == '__main__':
    solve()
```
