```python
import sys

# The problem asks for the minimum cost to connect all critical nodes in a tree.
# Since the graph is a tree (n nodes, n-1 edges, connected), the path between
# any two nodes is unique. To connect a set of nodes in a tree, we must
# activate all edges that lie on the paths between any pair of these nodes.
# This is equivalent to finding the smallest subtree containing all critical nodes.
# The cost is the sum of weights of all edges that have at least one critical node
# in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        critical[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use iterative DFS to avoid recursion depth issues
    # count[u] will store the number of critical nodes in the subtree of u
    count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for i in range(1, n + 1):
        if critical[i]:
            count[i] = 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        if count[u] > 0 and count[u] < t:
            total_cost += edge_to_parent_weight[u]
        count[p] += count[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
