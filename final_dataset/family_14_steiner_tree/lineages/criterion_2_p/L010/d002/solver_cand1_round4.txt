```python
import sys

# The problem states the graph is a tree (n nodes, n-1 edges, contiguous).
# To connect a subset of nodes in a tree, we must include all edges that lie
# on the unique paths between any two nodes in the subset.
# This is equivalent to finding the minimal subtree containing all reproductive nodes.
# An edge (u, v) with weight w is part of this subtree if and only if the
# subtree rooted at one side of the edge contains at least one reproductive node,
# and the subtree rooted at the other side also contains at least one.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    reproductive_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use iterative DFS to avoid recursion depth issues
    # count[u] will store the number of reproductive nodes in the subtree of u
    count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in reproductive_nodes:
        count[node] = 1
        
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if count[u] > 0:
            # If this subtree contains reproductive nodes, and it's not the whole tree,
            # the edge to the parent must be included if there are reproductive nodes
            # outside this subtree.
            if count[u] < t:
                total_weight += edge_to_parent_weight[u]
            count[p] += count[u]
            
    print(total_weight)

if __name__ == '__main__':
    solve()
```
