import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    spores = set()
    idx = 3
    for _ in range(t):
        spores.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the smallest subtree containing all spore nodes.
    # This is equivalent to summing the weights of all edges that lie on 
    # the path between any two spore nodes.
    # An edge (u, v) with weight w is part of the minimal subtree if and only if
    # there is at least one spore in the subtree of u (when rooted at v) 
    # AND at least one spore in the subtree of v (when rooted at u).
    
    has_spore = [False] * (n + 1)
    for s in spores:
        has_spore[s] = True
        
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process nodes in reverse topological order (bottom-up)
    subtree_has_spore = [False] * (n + 1)
    for i in range(1, n + 1):
        if has_spore[i]:
            subtree_has_spore[i] = True
            
    # Store edge weights to parents
    edge_to_parent_w = [0] * (n + 1)
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if parent[v] == u:
                edge_to_parent_w[v] = w
            elif parent[u] == v:
                edge_to_parent_w[u] = w
                
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        if subtree_has_spore[u]:
            total_cost += edge_to_parent_w[u]
            subtree_has_spore[p] = True
            
    print(total_cost)

solve()
