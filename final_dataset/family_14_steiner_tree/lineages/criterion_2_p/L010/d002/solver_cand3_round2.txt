```python
import sys

# The problem states that the facility is a tree (n workstations, n-1 belts, integrated).
# To connect a subset of nodes in a tree, we must include all edges that lie on the 
# unique paths between any two nodes in the subset.
# This is equivalent to finding the Steiner Tree in a tree, which is simply the 
# union of all paths between the quality control stations.
# In a tree, this is the minimal subtree containing all quality control stations.
# An edge is part of this subtree if and only if the two components it separates 
# both contain at least one quality control station.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    qc_stations = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many QC stations are in each subtree if we root at 1
    qc_count = [0] * (n + 1)
    for node in qc_stations:
        qc_count[node] = 1
        
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            qc_count[p] += qc_count[u]
            # If the subtree at u contains at least one QC station 
            # and the rest of the tree contains at least one QC station,
            # the edge (u, p) must be included.
            if qc_count[u] > 0 and qc_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
