```python
import sys

# The problem states that the organism is fully contiguous and m = n - 1.
# This means the structure is a tree.
# We need to find the minimum cost to connect all spore-containing vesicles.
# In a tree, the unique path between any two nodes is fixed.
# The set of edges that must be activated is the union of all paths between
# any two spore-containing vesicles.
# This is equivalent to the edges in the minimal subtree containing all spore nodes.
# An edge (u, v) with weight w is in this subtree if and only if the removal
# of this edge splits the tree into two components, each containing at least one
# spore-containing vesicle.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    spores = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many spores are in the subtree rooted at each node
    # We use iterative DFS to avoid recursion depth issues
    spore_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in spores:
            spore_count[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If the subtree at u contains at least one spore and 
            # the rest of the tree contains at least one spore,
            # the edge (u, p) must be included.
            if 0 < spore_count[u] < t:
                total_cost += edge_to_parent_weight[u]
            spore_count[p] += spore_count[u]
            
    print(total_cost)

solve()
```
