```python
import sys

# The problem asks for the minimum weight of a subset of edges in a tree
# such that all critical nodes are in the same connected component.
# This is equivalent to finding the weight of the minimal subtree that
# contains all critical nodes (the Steiner Tree in a tree).
# In a tree, the minimal subtree containing a set of nodes is the union
# of all paths between any two nodes in the set.
# This is equivalent to the sum of weights of all edges that lie on
# at least one path between two critical nodes.
# An edge (u, v) with weight w lies on a path between two critical nodes
# if and only if removing that edge splits the tree into two components,
# each containing at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        critical[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many critical nodes are in each subtree
    subtree_critical_count = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    for i in range(1, n + 1):
        if critical[i]:
            subtree_critical_count[i] = 1
            
    total_weight = 0
    # Process in reverse topological order
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        # Find weight of edge (u, p)
        w_up = 0
        for v, w in adj[u]:
            if v == p:
                w_up = w
                break
        
        # If this edge separates the tree into two parts, both containing
        # at least one critical node, then this edge must be included.
        count = subtree_critical_count[u]
        if count > 0 and count < t:
            total_weight += w_up
            
        subtree_critical_count[p] += count
        
    print(total_weight)

solve()
```
