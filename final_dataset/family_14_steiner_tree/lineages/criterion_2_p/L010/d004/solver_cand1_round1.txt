```python
import sys

# The problem asks for the minimum total weight of edges in a subtree that 
# connects all t quality control stations to a chosen hub node.
# Since the graph is a tree (n nodes, n-1 edges), the set of edges required 
# to connect a set of nodes S to a hub H is the union of paths from each 
# node in S to H.
# The union of paths from all nodes in S to a hub H is exactly the minimal 
# subtree containing all nodes in S and H.
# The total weight of this subtree is the sum of weights of all edges 
# that lie on at least one path from a node in S to H.
# An edge e splits the tree into two components of sizes S_e and (n - S_e).
# If the edge e is on the path from some node in S to H, then H must be in 
# one component and at least one node from S must be in the other.
# Specifically, an edge e is part of the minimal subtree connecting S and H 
# if and only if there is at least one node from S in both components 
# created by removing e.
# Let 'count' be the number of quality control stations in a subtree.
# An edge e is included if 0 < count < t.
# The total weight is the sum of weights of all edges e such that 
# 0 < count(e) < t.
# This sum is constant regardless of the choice of H, provided H is 
# chosen such that all nodes in S can reach it. Since it's a tree, 
# any H works.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    qc_stations = set()
    idx = 3
    for _ in range(t):
        qc_stations.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # DFS to calculate the number of QC stations in each subtree
    # Using iterative DFS to avoid recursion depth issues
    qc_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in qc_stations:
            qc_count[u] += 1
        if parent[u] != -1:
            qc_count[parent[u]] += qc_count[u]
            
    total_weight = 0
    for i in range(2, n + 1):
        # Edge connects i to parent[i]
        # count of QC stations in subtree at i is qc_count[i]
        # If 0 < qc_count[i] < t, the edge is necessary
        if 0 < qc_count[i] < t:
            total_weight += edge_weight[i]
            
    print(total_weight)

solve()
```
