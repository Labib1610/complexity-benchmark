We can solve this by finding the minimal subtree that connects all the quality control stations, then computing the sum of edge weights in that subtree.  

The minimal subtree connecting a set of nodes in a tree is the union of all paths between those nodes.  
We can find it by:

1. Marking the quality control stations.
2. Doing a DFS from any quality control station to count how many quality control stations are in each subtree.
3. An edge is needed if both sides contain at least one quality control station.
4. Sum the weights of all such edges.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    qc = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # DFS to count QC stations in each subtree
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    order = []
    stack = [next(iter(qc))]
    parent[next(iter(qc))] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                weight_to_parent[v] = w
                stack.append(v)
    
    count = [0] * (n + 1)
    for u in qc:
        count[u] = 1
    
    total = 0
    for u in reversed(order):
        if parent[u] != -1:
            if count[u] > 0 and (t - count[u]) > 0:
                total += weight_to_parent[u]
            count[parent[u]] += count[u]
    
    print(total)

if __name__ == "__main__":
    solve()
```
