```python
import sys

# The problem asks for the minimum cost to connect all target variables in a graph
# where each generator (u, v) with weight w represents an edge between u and v.
# The condition "x_ca - x_cb is in the linear span of the working set" is 
# equivalent to saying that there is a path between ca and cb in the graph 
# formed by the selected edges.
# To ensure all target variables are connected, we need to find the Minimum 
# Spanning Tree (MST) of the subgraph that connects all target variables.
# This is equivalent to finding the Steiner Tree problem, but since the 
# generators form a basis for the (n-1) dimensional space, the graph is a tree.
# In a tree, there is only one path between any two nodes.
# Thus, the set of edges required is the union of all paths between target variables.
# This is equivalent to the edges on the unique path in the tree connecting all 
# target variables.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Since the generators form a basis for the (n-1) dimensional space,
    # the graph is a tree.
    # We need the sum of weights of all edges that lie on paths between any two targets.
    # An edge (u, v) with weight w is part of the solution if and only if
    # removing it splits the tree into two components, each containing at least one target.
    
    is_target = [False] * (n + 1)
    for x in targets:
        is_target[x] = True
        
    target_count_subtree = [0] * (n + 1)
    edge_weights = {}
    
    # DFS to count targets in subtrees
    stack = [(1, -1)]
    order = []
    parent = {1: -1}
    edge_to_parent_w = {}
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != p:
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        count = 1 if is_target[u] else 0
        for v, w in adj[u]:
            if v != parent.get(u, -1):
                count += target_count_subtree[v]
        target_count_subtree[u] = count
        
    total_cost = 0
    for u in range(2, n + 1):
        c = target_count_subtree[u]
        if c > 0 and c < t:
            total_cost += edge_to_parent_w[u]
            
    print(total_cost)

solve()
```
