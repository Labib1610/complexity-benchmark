```python
import sys

# The problem asks for the minimum cost to select a subset of edges (operators)
# such that in the resulting graph, all target nodes c_1, c_2, ..., c_t 
# belong to the same connected component.
# An operator M_{u,v} forces x_u = x_v in any invariant state.
# If we select a set of edges, the invariant states are those where x_u = x_v
# for all u, v in the same connected component.
# The condition sum_{i=1}^t x_{c_i} = t * x_{c_1} is satisfied if and only if
# x_{c_1} = x_{c_2} = ... = x_{c_t}.
# This is equivalent to saying that all c_i must be in the same connected component.
# We need to find the minimum cost to connect all nodes in {c_1, ..., c_t}.
# This is the Steiner Tree problem, but since we only have n-1 edges (a tree),
# the graph is already a tree. The minimum cost to connect a set of nodes in a tree
# is the sum of the weights of all edges on the unique paths between them.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = [int(x) for x in input_data[3:3+t]]
    edges = []
    idx = 3 + t
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # In a tree, the union of paths between a set of nodes is the set of edges
    # that lie on at least one path between two target nodes.
    # An edge (u, v) is part of the Steiner tree if and only if the two components
    # formed by removing (u, v) both contain at least one target node.
    
    target_set = [False] * (n + 1)
    for c in targets:
        target_set[c] = True
        
    subtree_targets = [0] * (n + 1)
    
    # DFS to count targets in subtrees
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_targets = t
    ans = 0
    for u in reversed(order):
        subtree_targets[u] += (1 if target_set[u] else 0)
        if u != 1:
            p = parent[u]
            if subtree_targets[u] > 0 and subtree_targets[u] < total_targets:
                ans += edge_to_parent_weight[u]
            subtree_targets[p] += subtree_targets[u]
            
    print(ans)

solve()
```
