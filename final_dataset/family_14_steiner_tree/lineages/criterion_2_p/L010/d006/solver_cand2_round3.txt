```python
import sys

# The problem asks for the minimum total calibration time to connect all core modules.
# Since the graph is a tree (N modules, N-1 buses), there is exactly one unique path
# between any two nodes. To connect a set of core modules, we must calibrate all
# buses that lie on the paths between any two core modules.
# This is equivalent to finding the smallest subtree that contains all core modules.
# We can solve this by rooting the tree arbitrarily (e.g., at node 1) and using
# a post-order traversal (or DFS) to determine if a subtree contains any core modules.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    K = int(input_data[2])
    
    if K <= 1:
        print(0)
        return
    
    core_modules = set(map(int, input_data[3:3+K]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + K
    for _ in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # has_core[u] will be True if the subtree rooted at u contains at least one core module
    has_core = [False] * (N + 1)
    for c in core_modules:
        has_core[c] = True
        
    total_weight = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if has_core[u] and parent[u] != 0:
            total_weight += edge_to_parent_weight[u]
            has_core[parent[u]] = True
            
    print(total_weight)

if __name__ == '__main__':
    solve()
```
