```python
import sys

# The problem asks for the sum of weights of all edges that lie on the unique 
# path between any two quality-control stations. Since the graph is a tree, 
# an edge is part of the required set if and only if it separates the tree 
# into two components, both of which contain at least one quality-control station.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    qc_stations = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # qc_count[u] will store the number of quality-control stations in the 
    # subtree rooted at u (using DFS traversal).
    qc_count = [0] * (n + 1)
    for node in qc_stations:
        qc_count[node] = 1
        
    total_weight = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(1, 0, 0)] # node, parent, weight_from_parent
    order = []
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    
    while stack:
        u, p, w = stack.pop()
        order.append(u)
        parent[u] = p
        weight_to_parent[u] = w
        for v, weight in adj[u]:
            if v != p:
                stack.append((v, u, weight))
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if parent[u] != 0:
            qc_count[parent[u]] += qc_count[u]
            # If the edge (parent[u], u) separates the tree such that both 
            # sides have at least one QC station, it must be included.
            if qc_count[u] > 0 and qc_count[u] < t:
                total_weight += weight_to_parent[u]
                
    print(total_weight)

if __name__ == '__main__':
    solve()
```
