```python
import sys

# The problem states m = n - 1 and the graph is connected, so it is a tree.
# We need to find a node (precursor) such that the sum of weights of all edges
# on the paths from the precursor to all t critical nodes is minimized.
# This is equivalent to finding the Steiner Tree for the set of t nodes,
# but since the graph is a tree, the Steiner Tree is simply the union of all
# paths connecting the t nodes.
# Let S be the set of critical nodes. The edges that must be activated are
# exactly those edges that lie on the unique path between any two nodes in S.
# An edge e splits the tree into two components of sizes c1 and c2.
# If both components contain at least one node from S, the edge must be activated.
# If one component contains no nodes from S, the edge does not need to be activated.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count[u] will store the number of critical nodes in the subtree rooted at u
    count = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for node in critical_nodes:
        count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order to compute subtree counts
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        # Find weight of edge (u, p)
        w_edge = 0
        for v, w in adj[u]:
            if v == p:
                w_edge = w
                break
        
        # If the subtree at u contains some but not all critical nodes,
        # the edge (u, p) must be activated.
        if 0 < count[u] < t:
            total_cost += w_edge
            
        count[p] += count[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
