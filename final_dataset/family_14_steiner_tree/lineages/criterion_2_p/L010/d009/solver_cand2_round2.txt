```python
import sys

# The problem describes a graph where each operation (u, v) with cost w
# corresponds to an edge between nodes u and v.
# The matrix M is the negative of the Laplacian matrix of the graph formed
# by the selected edges.
# The null space of the Laplacian of a graph consists of vectors that are
# constant on each connected component.
# The condition "for every vector x in the null space, x_{c_a} = x_{c_b} for
# all target indices c_a, c_b" is equivalent to saying that all target
# indices must belong to the same connected component in the graph formed
# by the selected edges.
# We want to find the minimum cost to connect all target indices.
# This is the Steiner Tree problem, but since the graph of all m operations
# is a tree (n nodes, n-1 edges), the Steiner Tree is simply the unique
# path in the tree connecting all target indices.
# The cost is the sum of weights of all edges on the unique paths between
# any two target indices. This is equivalent to the sum of weights of all
# edges that have at least one target index in each of the two components
# formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # subtree_targets[u] stores the number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    
    # DFS to count targets in subtrees
    stack = [(1, 0)]
    order = []
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        parent[u] = p
        for v, w in adj[u]:
            if v != p:
                edge_to_parent_w[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        if u in targets:
            subtree_targets[u] += 1
        if parent[u] != 0:
            subtree_targets[parent[u]] += subtree_targets[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If an edge splits the tree such that both sides have at least one target,
        # the edge must be included in the Steiner Tree.
        count = subtree_targets[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_w[i]
            
    print(total_cost)

solve()
```
