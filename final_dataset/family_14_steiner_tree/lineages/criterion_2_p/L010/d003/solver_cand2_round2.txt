```python
import sys

# The problem states the graph is a tree (n nodes, n-1 edges, connected).
# We need to find the minimum cost to connect all t inspection labs.
# This is equivalent to finding the sum of weights of all edges that lie on
# the unique path between any two inspection labs.
# An edge is part of the required set if and only if it separates the tree
# into two components, each containing at least one inspection lab.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    labs = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many labs are in the subtree of each node
    # Use iterative DFS to avoid recursion depth issues
    lab_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in labs:
            lab_count[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        # If the subtree at u contains some labs but not all,
        # the edge connecting u to its parent must be included.
        if 0 < lab_count[u] < t:
            total_cost += edge_to_parent_weight[u]
            
        # Propagate lab count to parent
        p = parent[u]
        lab_count[p] += lab_count[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
