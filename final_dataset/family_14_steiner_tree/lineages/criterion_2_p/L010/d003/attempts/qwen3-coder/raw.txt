import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    terminals = set(map(int, sys.stdin.readline().split()))
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    # Union-Find data structure
    parent = list(range(n + 1))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # Sort edges by weight
    edges.sort()
    
    # Kruskal's algorithm to build MST
    mst_edges = []
    for w, u, v in edges:
        if union(u, v):
            mst_edges.append((w, u, v))
        if len(mst_edges) == n - 1:
            break
    
    # Build adjacency list for MST
    adj = [[] for _ in range(n + 1)]
    for w, u, v in mst_edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Find nodes that are terminals or have terminal descendants
    # Use a modified DFS to count terminals in each subtree
    visited = [False] * (n + 1)
    result = 0
    
    def dfs(node, parent_node):
        nonlocal result
        visited[node] = True
        terminal_count = 0
        if node in terminals:
            terminal_count += 1
            
        for neighbor, weight in adj[node]:
            if neighbor == parent_node:
                continue
            if not visited[neighbor]:
                child_terminal_count = dfs(neighbor, node)
                if child_terminal_count > 0 and (t - child_terminal_count) > 0:
                    result += weight
                terminal_count += child_terminal_count
                
        return terminal_count
    
    # Start DFS from any node (e.g., node 1)
    dfs(1, -1)
    print(result)

threading.Thread(target=main).start()
