```python
import sys

# The problem states the network is a tree (n nodes, n-1 edges, fully integrated).
# We need to find the sum of weights of all edges that lie on the unique path
# between any two quality control stations.
# An edge is part of the path between two quality control stations if and only if
# the removal of that edge splits the tree into two components, each containing
# at least one quality control station.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    qc_stations = set()
    idx = 3
    for _ in range(t):
        qc_stations.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many QC stations are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    qc_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in qc_stations:
        qc_count[u] = 1
        
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        qc_count[p] += qc_count[u]
        
        # If the subtree at u contains at least one QC station and 
        # the rest of the tree contains at least one QC station,
        # then the edge (u, p) must be included.
        if qc_count[u] > 0 and qc_count[u] < t:
            total_weight += edge_weight_to_parent[u]
            
    print(total_weight)

solve()
```
