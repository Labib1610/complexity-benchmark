```python
import sys

# The problem asks for the minimum cost to connect a set of active chambers
# such that they can exchange energy. Since the graph is a tree (n nodes, n-1 edges),
# the set of edges required to connect a subset of nodes is the union of all
# paths between any two nodes in that subset.
# In a tree, the union of paths between a set of nodes S is the set of all edges
# that lie on at least one path between two nodes in S.
# An edge (u, v) lies on a path between two nodes in S if and only if
# removing (u, v) splits the tree into two components, each containing at least
# one node from S.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    active_nodes = set()
    idx = 3
    for _ in range(t):
        active_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_active[u] stores the number of active nodes in the subtree rooted at u
    subtree_active = [0] * (n + 1)
    
    # DFS to compute subtree_active
    # Since n is up to 200,000, use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in active_nodes:
            subtree_active[u] += 1
        if parent[u] != 0:
            subtree_active[parent[u]] += subtree_active[u]
            
    total_cost = 0
    for u, v, w in edges:
        # The edge connects u and v. One is the parent of the other.
        child = u if parent[u] == v else v
        count = subtree_active[child]
        # If the subtree at 'child' contains some active nodes, but not all,
        # then this edge is part of the minimal Steiner tree.
        if 0 < count < t:
            total_cost += w
            
    print(total_cost)

solve()
```
