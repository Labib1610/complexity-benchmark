```python
import sys

# The problem asks for the minimum cost to connect all t critical cores.
# Since the graph is a tree (n nodes, n-1 edges, connected), there is a unique
# path between any two nodes. To connect all t critical cores, we must activate
# all edges that lie on the paths between any pair of critical cores.
# This is equivalent to finding the smallest subtree that contains all t critical cores.
# The edges in this subtree are exactly those that, when removed, split the tree
# into components such that at least one component contains some critical cores
# and at least one other component contains the remaining critical cores.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_cores = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many critical cores are in the subtree of each node.
    # Let count[u] be the number of critical cores in the subtree rooted at u.
    # An edge (u, v) where v is a child of u is part of the minimal subtree
    # if and only if 0 < count[v] < t.
    
    count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in critical_cores:
            count[u] += 1
        if parent[u] != 0:
            count[parent[u]] += count[u]
            # If the subtree at u contains some but not all critical cores,
            # the edge connecting u to its parent must be activated.
            if 0 < count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
