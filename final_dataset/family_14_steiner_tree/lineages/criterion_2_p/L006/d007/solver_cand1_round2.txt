```python
import sys

# The problem asks for the minimum weight of a subset of edges such that all nodes in T
# are in the same connected component. Since the graph is a tree (n nodes, n-1 edges, 
# and all nodes are connected), the set of edges that connect all nodes in T is 
# exactly the set of edges that lie on the unique paths between any two nodes in T.
# This is equivalent to the Steiner Tree problem on a tree, which is simply the 
# union of all paths between nodes in T.
# In a tree, the union of paths between a set of nodes T is the set of all edges 
# that have at least one node from T in each of the two components formed by 
# removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many critical nodes are in each subtree
    subtree_critical_count = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for node in critical_nodes:
        subtree_critical_count[node] = 1
        
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_critical_count[p] += subtree_critical_count[u]
            
    # An edge (u, parent[u]) is part of the Steiner tree if 
    # 0 < subtree_critical_count[u] < t
    for i in range(1, n + 1):
        if i == 1: continue
        count = subtree_critical_count[i]
        if count > 0 and count < t:
            # Find weight of edge (i, parent[i])
            for v, w in adj[i]:
                if v == parent[i]:
                    total_weight += w
                    break
                    
    print(total_weight)

solve()
```
