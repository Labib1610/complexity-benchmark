```python
import sys

# The problem states that the ledger is a tree (n nodes, n-1 edges, and the property holds for the whole tree).
# The condition "if the t flagged accounts are not all assigned to the same jurisdiction, 
# then your collection C must contain at least one procedure that involves one account 
# assigned to Jurisdiction A and one account assigned to Jurisdiction B" is equivalent to 
# saying that the chosen edges must form a cut that separates the flagged accounts 
# into at least two non-empty sets.
# In a tree, the set of edges that separates a set of nodes S into at least two 
# components is the set of edges on the paths between nodes in S.
# Specifically, if we consider the minimal subtree containing all t flagged nodes, 
# any edge in this subtree, if removed, splits the flagged nodes into two non-empty sets.
# To satisfy the condition for *any* assignment where the t nodes are not all in the same 
# jurisdiction, we must include all edges in the minimal subtree that connects all t flagged nodes.
# If t=1, the condition is vacuously true (the flagged account cannot be split), so cost is 0.
# If t > 1, we need the sum of weights of all edges in the minimal subtree connecting the t nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    flagged = set()
    idx = 3
    for _ in range(t):
        flagged.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Find the minimal subtree containing all flagged nodes.
    # An edge is in the minimal subtree if and only if both components formed by 
    # removing the edge contain at least one flagged node.
    
    subtree_nodes = [0] * (n + 1)
    
    # Count flagged nodes in each subtree
    def dfs_count(u, p):
        count = 1 if u in flagged else 0
        for v, w in adj[u]:
            if v != p:
                count += dfs_count(v, u)
        subtree_nodes[u] = count
        return count
    
    dfs_count(1, -1)
    
    total_cost = 0
    # An edge (u, v) is in the minimal subtree if 0 < subtree_nodes[v] < t
    # (assuming v is the child of u)
    def dfs_sum(u, p):
        nonlocal total_cost
        for v, w in adj[u]:
            if v != p:
                if 0 < subtree_nodes[v] < t:
                    total_cost += w
                dfs_sum(v, u)
                
    dfs_sum(1, -1)
    print(total_cost)

solve()
```
