```python
import sys

def solve():
    # The problem asks for the minimum weight subset of edges such that
    # for any partition of the primary anchors into two non-empty sets,
    # there is an edge connecting the two sets.
    # This is equivalent to saying that all primary anchors must be in the
    # same connected component in the chosen subgraph.
    # If there is only one primary anchor, the condition is vacuously true
    # for any partition (since you cannot split one anchor into two non-empty sets),
    # so the minimum weight is 0.
    # If there are t > 1 primary anchors, we must connect all t anchors
    # into a single connected component.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    anchors = [int(x) for x in input_data[3:3+t]]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    if t <= 1:
        print(0)
        return
    
    # Kruskal's algorithm to connect all anchors.
    # We can treat all anchors as being part of the same "super-node"
    # or simply contract them. A simpler way:
    # Use DSU to connect nodes.
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # To connect all anchors, we can union all anchors together first.
    for i in range(t - 1):
        union(anchors[i], anchors[i+1])
        
    edges.sort()
    
    total_weight = 0
    for w, u, v in edges:
        if union(u, v):
            total_weight += w
            
    print(total_weight)

solve()
```
