import sys

def solve():
    # Read n, m, t
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_sensors = set()
    idx = 3
    for _ in range(t):
        priority_sensors.add(int(input_data[idx]))
        idx += 1
        
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The problem states the graph is a tree (m = n - 1 and connected).
    # We need to find the minimum cost to connect all priority sensors.
    # In a tree, the set of edges required to connect a subset of nodes
    # is exactly the set of edges on the unique paths between all pairs
    # of nodes in that subset. This is equivalent to the union of paths
    # from a root to all nodes in the subset.
    
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Root the tree at the first priority sensor
    root = next(iter(priority_sensors))
    
    # parent_edge[v] = (weight of edge to parent, parent_node)
    parent_edge = {}
    stack = [(root, -1, 0)]
    visited = {root}
    
    # To track if a subtree contains any priority sensor
    has_priority = {i: (i in priority_sensors) for i in range(1, n + 1)}
    
    # DFS to build parent pointers and identify edges to keep
    order = []
    stack = [root]
    parent = {root: None}
    edge_to_parent = {root: 0}
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v not in parent:
                parent[v] = u
                edge_to_parent[v] = w
                stack.append(v)
                
    # Post-order traversal to mark subtrees
    total_cost = 0
    for u in reversed(order):
        if u == root:
            continue
        if has_priority[u]:
            total_cost += edge_to_parent[u]
            if parent[u] is not None:
                has_priority[parent[u]] = True
                
    print(total_cost)

solve()
