```python
import sys

# The problem states that the m = n - 1 operations form a minimal basis,
# which means the graph of sensors and operations is a tree.
# We need to find the minimum cost to connect all t priority sensors.
# In a tree, the set of edges required to connect a subset of nodes is
# exactly the union of all paths between every pair of nodes in that subset.
# This is equivalent to the edges in the minimal Steiner Tree for the
# priority nodes, which in a tree is simply the minimal subtree containing
# all priority nodes.
# An edge (u, v) with weight w is part of this subtree if and only if
# the removal of this edge splits the tree into two components, each
# containing at least one priority node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_sensors = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_priority_count[u] stores how many priority nodes are in the
    # subtree rooted at u (when rooted at node 1).
    subtree_priority_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order
    edge_weights = {}
    for u in reversed(order):
        if u == 1:
            continue
        
        # Find weight of edge (parent[u], u)
        w = 0
        for v, weight in adj[u]:
            if v == parent[u]:
                w = weight
                break
        
        count = (1 if u in priority_sensors else 0) + subtree_priority_count[u]
        subtree_priority_count[parent[u]] += count
        
        # If this edge separates the tree into two parts both containing
        # at least one priority node, it must be included.
        if 0 < count < t:
            total_cost += w
            
    print(total_cost)

solve()
```
