import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    priority = list(map(int, sys.stdin.readline().split()))
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u-1, v-1))  # Convert to 0-indexed
    
    if t <= 1:
        print(0)
        return
    
    # Union-Find data structure
    class UnionFind:
        def __init__(self, size):
            self.parent = list(range(size))
            self.rank = [0] * size
        
        def find(self, x):
            if self.parent[x] != x:
                self.parent[x] = self.find(self.parent[x])
            return self.parent[x]
        
        def union(self, x, y):
            px, py = self.find(x), self.find(y)
            if px == py:
                return False
            if self.rank[px] < self.rank[py]:
                px, py = py, px
            self.parent[py] = px
            if self.rank[px] == self.rank[py]:
                self.rank[px] += 1
            return True
    
    # Sort edges by weight
    edges.sort()
    
    # Create a mapping for priority nodes
    priority_set = set(p - 1 for p in priority)  # Convert to 0-indexed
    node_map = {node: i for i, node in enumerate(sorted(priority_set))}
    reverse_map = {i: node for node, i in node_map.items()}
    k = len(priority_set)
    
    if k <= 1:
        print(0)
        return
    
    # We need to connect all priority nodes with minimum cost
    # This is a Minimum Spanning Tree problem on the induced subgraph
    # But we can't directly build the subgraph, so we use the original graph
    # and only consider connections between priority nodes
    
    uf = UnionFind(k)
    total_cost = 0
    components = k
    
    for w, u, v in edges:
        # Check if this edge connects two priority nodes (in the mapped space)
        u_in_priority = u in priority_set
        v_in_priority = v in priority_set
        
        if u_in_priority or v_in_priority:
            # Map the nodes to their indices in the priority set
            u_mapped = node_map.get(u, -1)
            v_mapped = node_map.get(v, -1)
            
            # If both ends are priority nodes, try to connect them
            if u_in_priority and v_in_priority:
                if uf.union(u_mapped, v_mapped):
                    total_cost += w
                    components -= 1
                    if components == 1:
                        break
    
    print(total_cost)

# Use threading to avoid recursion limit issues
threading.Thread(target=main).start()
