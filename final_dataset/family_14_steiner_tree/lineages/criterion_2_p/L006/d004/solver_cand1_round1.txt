```python
import sys

# The problem describes a graph where each vector e_u - e_v is an edge between nodes u and v.
# The condition that e_a - e_b can be expressed as a linear combination of the acquired
# vectors means that in the graph formed by the acquired edges, there must be a path
# between a and b for all priority indices a and b.
# Since the catalog forms a tree (n nodes, n-1 edges, minimal basis), the subset of
# edges that connects all priority indices is the unique path in the tree connecting
# all priority nodes. This is equivalent to the Steiner Tree problem on a tree,
# which is simply the union of all paths between the priority nodes.
# In a tree, this is the set of all edges that lie on the unique path between any
# two priority nodes. An edge belongs to this set if and only if removing it
# splits the tree into two components, each containing at least one priority node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_indices = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w, _))
        adj[v].append((u, w, _))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_priority_count[u] stores how many priority nodes are in the subtree rooted at u
    subtree_priority_count = [0] * (n + 1)
    
    # DFS to calculate subtree_priority_count
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w, _ in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in priority_indices:
            subtree_priority_count[u] += 1
        if parent[u] != 0:
            subtree_priority_count[parent[u]] += subtree_priority_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If an edge (i, parent[i]) separates the tree into two components,
        # and both components contain at least one priority node, the edge must be included.
        count = subtree_priority_count[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
