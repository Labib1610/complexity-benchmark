```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); t = int(next(it))
    specials = [0] * (n + 1)
    for _ in range(t):
        specials[int(next(it))] = 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))

    parent = [0] * (n + 1)
    pw = [0] * (n + 1)          # weight to parent
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v == parent[u]:
                continue
            parent[v] = u
            pw[v] = w
            stack.append(v)

    cnt = [0] * (n + 1)
    for u in reversed(order):
        cnt[u] = specials[u]
        for v, _ in adj[u]:
            if parent[v] == u:
                cnt[u] += cnt[v]

    total_special = t
    ans = 0
    for v in range(2, n + 1):
        if cnt[v] > 0 and total_special - cnt[v] > 0:
            ans += pw[v]

    print(ans)

if __name__ == "__main__":
    solve()
```
