```python
import sys

# The condition pi_C(e_a) = pi_C(e_b) for all a, b in T means that 
# all indices in T must belong to the same connected component in the graph 
# formed by the selected edges (transformations).
# If we select a set of edges C, the invariant subspace V_C is spanned by 
# vectors that are constant on each connected component of the graph (V, C).
# Specifically, if a and b are in the same connected component, then 
# the projection of e_a and e_b onto V_C will be the same.
# Thus, we need to select a subset of edges such that all indices in T 
# are in the same connected component.
# This is equivalent to finding the minimum weight set of edges that 
# connects all vertices in T. This is the Steiner Tree problem.
# Since the graph is a tree (m = n-1 and the problem implies connectivity),
# the Steiner Tree is simply the unique path in the tree connecting all 
# vertices in T.
# The cost is the sum of weights of all edges on the unique paths 
# between all pairs of nodes in T. This is equivalent to the sum of weights 
# of all edges e such that the removal of e splits the tree into two 
# components, each containing at least one node from T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_t[u] stores the number of target nodes in the subtree rooted at u
    subtree_t = [0] * (n + 1)
    
    # DFS to count target nodes in subtrees
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            subtree_t[u] += 1
        if parent[u] != 0:
            subtree_t[parent[u]] += subtree_t[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the edge (i, parent[i]) separates the tree into two parts,
        # and both parts contain at least one target node, the edge must be included.
        count = subtree_t[i]
        if count > 0 and count < t:
            total_cost += edge_weight[i]
            
    print(total_cost)

solve()
```
