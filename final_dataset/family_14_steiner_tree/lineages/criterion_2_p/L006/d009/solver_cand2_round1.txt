```python
import sys

# The problem states that the graph is connected (since the scalar field condition 
# implies the graph is a tree, as m = n-1).
# We need to find the minimum weight of edges that connect all nodes in T.
# In a tree, the unique path between any two nodes in T must be included.
# The union of all paths between nodes in T forms a subtree.
# The total weight is the sum of weights of all edges that lie on at least one 
# path between two nodes in T.
# An edge (u, v) with weight w is in this subtree if and only if the removal 
# of this edge splits the tree into two components, each containing at least 
# one node from T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_t[u] stores the number of critical nodes in the subtree rooted at u
    subtree_t = [0] * (n + 1)
    total_weight = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    for u in reversed(order):
        if u in critical_nodes:
            subtree_t[u] += 1
        if parent[u] != 0:
            p, w = parent[u]
            subtree_t[p] += subtree_t[u]
            # If the edge (p, u) separates the tree into two parts, 
            # both containing at least one critical node, add its weight.
            if subtree_t[u] > 0 and subtree_t[u] < t:
                total_weight += w
                
    print(total_weight)

solve()
```
