import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    if t <= 1:
        print(0)
        return
    
    # The problem asks for the minimum cost to connect all target sensors.
    # This is equivalent to finding the Steiner Tree in a graph.
    # However, the problem states m = n - 1 and the graph is connected.
    # This means the graph is a tree.
    # In a tree, there is exactly one path between any two nodes.
    # To deduce the relative difference between all target sensors, we must
    # include all edges that lie on the paths between any two target sensors.
    # This is the union of all paths between all pairs of target sensors.
    
    adj = [[] for _ in range(n + 1)]
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Root the tree at the first target sensor
    root = next(iter(targets))
    
    # parent[u] = (p, weight_to_p)
    parent = [None] * (n + 1)
    is_target = [False] * (n + 1)
    for node in targets:
        is_target[node] = True
        
    stack = [(root, 0, 0)]
    visited = [False] * (n + 1)
    visited[root] = True
    
    order = []
    while stack:
        u, p, w = stack.pop()
        parent[u] = (p, w)
        order.append(u)
        for v, weight in adj[u]:
            if not visited[v]:
                visited[v] = True
                stack.append((v, u, weight))
                
    # Mark nodes that are part of the Steiner Tree
    # A node is in the Steiner Tree if it is a target or has a descendant that is a target
    in_steiner = [False] * (n + 1)
    for node in targets:
        in_steiner[node] = True
        
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if in_steiner[u] and parent[u][0] != 0:
            in_steiner[parent[u][0]] = True
            
    total_cost = 0
    for i in range(1, n + 1):
        if in_steiner[i] and parent[i][0] != 0:
            total_cost += parent[i][1]
            
    print(total_cost)

solve()
