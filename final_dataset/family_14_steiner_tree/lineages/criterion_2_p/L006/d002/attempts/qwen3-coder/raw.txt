import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    priority = set(map(int, sys.stdin.readline().split()))
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    # Union-Find data structure
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    # Sort edges by weight
    edges.sort()
    
    # Kruskal's algorithm variant: we only care about connecting priority nodes
    cost = 0
    connected = len(priority)
    
    for w, u, v in edges:
        if find(u) != find(v):
            # Check if this edge helps connect priority nodes
            u_is_priority = u in priority
            v_is_priority = v in priority
            u_root = find(u)
            v_root = find(v)
            
            # Count how many priority nodes are in each component
            # For efficiency, we'll use a different approach:
            # We simulate the process and stop when all priority nodes are connected
            
            union(u, v)
            cost += w
            
            # Check if we've connected all priority nodes
            # All priority nodes have the same root
            roots = set(find(x) for x in priority)
            if len(roots) == 1:
                break
    
    print(cost)

threading.Thread(target=main).start()
