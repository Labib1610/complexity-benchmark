```python
import sys

# The problem asks for the minimum cost to connect a set of t priority sensors
# in a tree structure. Since the input is already a tree (n nodes, n-1 edges),
# the subset of edges that connects all priority sensors is unique: it is the
# union of all paths between any two priority sensors.
# This is equivalent to the minimal subtree containing all priority nodes.
# We can solve this by rooting the tree arbitrarily (e.g., at the first priority node)
# and pruning all branches that do not contain any priority nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_sensors = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Root the tree at the first priority sensor
    root = next(iter(priority_sensors))
    
    # We need to find all edges that lie on a path between any two priority nodes.
    # An edge (u, v) is part of the minimal subtree if and only if the subtree
    # rooted at the child node contains at least one priority node, AND the
    # rest of the tree (outside that subtree) also contains at least one priority node.
    
    has_priority = [False] * (n + 1)
    for p in priority_sensors:
        has_priority[p] = True
        
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to propagate priority status
    for u in reversed(order):
        if u == root:
            continue
        if has_priority[u]:
            total_cost += edge_to_parent_weight[u]
            has_priority[parent[u]] = True
            
    print(total_cost)

solve()
```
