```python
import sys

# The problem states that the graph is a tree (n nodes, n-1 edges, no redundancies).
# We need to find the minimum cost to connect all t target sensors.
# In a tree, the set of edges required to connect a subset of nodes is the union
# of all paths between any two nodes in that subset.
# This is equivalent to the edges in the minimal subtree containing all target nodes.
# An edge (u, v) with weight w is part of this minimal subtree if and only if
# the removal of this edge splits the tree into two components, both of which
# contain at least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_targets[u] will store the number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in order:
        if u in targets:
            subtree_targets[u] = 1
            
    # Process in reverse topological order to propagate counts up
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            subtree_targets[parent[u]] += subtree_targets[u]
            
    total_cost = 0
    for u, v, w in edges:
        # The edge connects u and v. One is the parent of the other.
        child = u if parent[u] == v else v
        count = subtree_targets[child]
        # If the subtree at 'child' contains some targets (but not all),
        # then this edge is necessary.
        if 0 < count < t:
            total_cost += w
            
    print(total_cost)

solve()
```
