```python
import sys

# The problem asks for the minimum cost to select a subset of edges in a tree
# such that for every cut separating the target set T, at least one edge is selected.
# This is equivalent to finding the minimum weight set of edges that separates
# all components of T in the tree.
# If we contract all edges with weight 0 (if any), or more generally,
# consider the tree structure, the condition "for every S such that S intersect T is not empty
# and T \ S is not empty, there exists an edge in C with exactly one element in S"
# is equivalent to saying that the chosen edges must disconnect all nodes in T
# from each other in the original tree.
# This is the "Multiway Cut" problem on a tree, which is solvable by DP or
# by considering the tree as a set of paths between nodes in T.
# Specifically, for a tree, the minimum cost to separate all nodes in T is the
# sum of the minimum edge weights on the paths between all pairs of nodes in T
# such that no two nodes in T are in the same connected component.
# This is equivalent to: for every edge e, if removing e splits the tree into
# two components T1 and T2, and both T1 and T2 contain at least one node from T,
# then e must be cut. The minimum cost is the sum of weights of all such edges.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count[u] will store the number of target nodes in the subtree rooted at u
    count = [0] * (n + 1)
    for node in target_nodes:
        count[node] = 1
        
    total_cost = 0
    
    # DFS to compute subtree counts and identify edges to cut
    # Using iterative DFS to avoid recursion depth issues
    stack = [(1, 0, 0)] # node, parent, weight_to_parent
    order = []
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    
    dfs_stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    while dfs_stack:
        u = dfs_stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                weight_to_parent[v] = w
                dfs_stack.append(v)
                
    for u in reversed(order):
        if parent[u] != 0:
            count[parent[u]] += count[u]
            # If this edge splits T into two non-empty sets
            if count[u] > 0 and count[u] < t:
                total_cost += weight_to_parent[u]
                
    print(total_cost)

solve()
```
