```python
import sys
import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    is_priority = [False] * (n + 1)
    for _ in range(t):
        node = int(next(it))
        is_priority[node] = True
        
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        edges.append((w, u, v))
        
    # We need to find the minimum cost set of edges C such that
    # for every S intersecting T but not containing T, there is an edge crossing S.
    # This is equivalent to ensuring that for every partition of T into non-empty A, B,
    # there is an edge crossing between the component containing A and the component containing B.
    # This is equivalent to the edges connecting all priority nodes into a single connected component.
    
    # Sort edges by weight
    edges.sort()
    
    parent = list(range(n + 1))
    comp_priority_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if is_priority[i]:
            comp_priority_count[i] = 1
            
    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i
        
    total_cost = 0
    # Kruskal's algorithm
    for w, u, v in edges:
        root_u = find(u)
        root_v = find(v)
        if root_u != root_v:
            # Merge smaller into larger (union by size is optional, path compression is enough)
            # We just need to track priority counts
            if comp_priority_count[root_u] < comp_priority_count[root_v]:
                root_u, root_v = root_v, root_u
            parent[root_v] = root_u
            comp_priority_count[root_u] += comp_priority_count[root_v]
            total_cost += w
            if comp_priority_count[root_u] == t:
                break
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
