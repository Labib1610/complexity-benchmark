```python
import sys

# The problem asks for the minimum cost to connect all nodes in T.
# Since m = n - 1 and the graph is a tree (given all m synchronizers connect T),
# we are looking for the minimum weight subtree that spans all nodes in T.
# This is equivalent to the Steiner Tree problem on a tree, which is simply
# the sum of weights of all edges that lie on the unique paths between nodes in T.
# An edge is part of the solution if and only if removing it splits the tree
# into two components, both of which contain at least one node from T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_t[u] stores the number of critical nodes in the subtree rooted at u
    subtree_t = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in order:
        if u in critical_nodes:
            subtree_t[u] += 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        # If the subtree at u contains some but not all critical nodes,
        # the edge to its parent must be included.
        if 0 < subtree_t[u] < t:
            total_cost += edge_weight_to_parent[u]
            
        # Propagate count to parent
        subtree_t[parent[u]] += subtree_t[u]
        
    print(total_cost)

solve()
```
