```python
import sys

# The problem asks for the minimum cost to connect all nodes in T.
# Since m = n - 1 and the graph is connected, the graph is a tree.
# We need to find the minimum weight subset of edges such that all nodes in T
# are in the same connected component.
# This is equivalent to finding the Steiner Tree in a tree, which is simply
# the sum of weights of all edges that lie on the unique paths between nodes in T.
# An edge is part of the Steiner Tree if and only if the subtree formed by
# removing the edge contains at least one node from T on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    t_indices = set(map(int, input_data[3:3+t]))
    edges = []
    idx = 3 + t
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Root the tree at an arbitrary node from T, say t_indices[0]
    root = next(iter(t_indices))
    
    # count_t[u] = number of nodes from T in the subtree rooted at u
    count_t = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in t_indices:
            count_t[u] += 1
        if parent[u] != 0:
            count_t[parent[u]] += count_t[u]
            
    total_weight = 0
    for i in range(1, n + 1):
        if i == root:
            continue
        # If the subtree at i contains some nodes from T, but not all,
        # then the edge connecting i to its parent must be included.
        if count_t[i] > 0 and count_t[i] < t:
            total_weight += parent_edge_weight[i]
            
    print(total_weight)

solve()
```
