```python
import sys

# The problem asks for the minimum weight of a subset of edges in a tree 
# such that all critical nodes are in the same connected component.
# Since the graph is a tree (N nodes, N-1 edges, connected), 
# the subset of edges that connects all critical nodes is the unique 
# minimal subtree containing all critical nodes (the Steiner Tree).
# In a tree, the Steiner Tree is simply the union of all paths between 
# any two critical nodes.
# An edge is part of this Steiner Tree if and only if removing it 
# splits the tree into two components, each containing at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+T]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + T
    for _ in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if T <= 1:
        print(0)
        return

    # count_critical[u] will store the number of critical nodes in the subtree rooted at u
    count_critical = [0] * (N + 1)
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u in critical_nodes:
            count_critical[u] += 1
        
        if parent[u] != 0:
            p, w = parent[u]
            count_critical[p] += count_critical[u]
            # If this edge separates critical nodes, it must be included
            if count_critical[u] > 0 and count_critical[u] < T:
                total_weight += w
                
    print(total_weight)

solve()
```
