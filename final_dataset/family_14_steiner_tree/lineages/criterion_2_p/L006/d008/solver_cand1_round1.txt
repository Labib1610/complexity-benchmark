```python
import sys

# The problem asks for the minimum cost to connect all nodes in T in a tree.
# Since the graph is a tree (n nodes, n-1 edges, irredundant), the condition
# "for every partition of the n variables into two sets A and B, if both A and B
# contain at least one index from T, there must be an edge between A and B"
# is equivalent to saying that all nodes in T must be in the same connected component
# in the subgraph formed by the chosen edges.
# In a tree, the unique path between any two nodes in T must be included.
# Thus, we need to find the union of all paths between all pairs of nodes in T.
# This is equivalent to the sum of weights of all edges that lie on at least one
# path between two nodes in T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Root the tree at an arbitrary critical node
    root = next(iter(critical_nodes))
    
    # subtree_t[u] = number of critical nodes in subtree of u
    subtree_t = [0] * (n + 1)
    parent_edge_w = [0] * (n + 1)
    
    stack = [(root, -1, 0)]
    order = []
    while stack:
        u, p, w = stack.pop()
        order.append((u, p, w))
        parent_edge_w[u] = w
        for v, weight in adj[u]:
            if v != p:
                stack.append((v, u, weight))
                
    for u, p, w in reversed(order):
        if u in critical_nodes:
            subtree_t[u] += 1
        if p != -1:
            subtree_t[p] += subtree_t[u]
            
    total_cost = 0
    for u in range(1, n + 1):
        if u == root:
            continue
        # If the edge to parent separates the tree into two parts,
        # and both parts contain at least one critical node,
        # then this edge must be included.
        # Part 1: subtree_t[u]
        # Part 2: t - subtree_t[u]
        if subtree_t[u] > 0 and (t - subtree_t[u]) > 0:
            total_cost += parent_edge_w[u]
            
    print(total_cost)

solve()
```
