```python
import sys

# The problem asks for the minimum cost to connect all nodes in T in a graph
# where edges are vectors e_u - e_v.
# The condition "e_a - e_b is in the span of C" is equivalent to saying
# that in the graph formed by edges in C, all nodes in T are in the same
# connected component.
# Since m = n - 1 and the span of all m vectors is n - 1, the graph
# formed by all m vectors is a tree (a spanning tree of the n nodes).
# We need to find a subset of edges such that all nodes in T are connected.
# This is the Steiner Tree problem, but since the graph is a tree,
# the unique path between any two nodes in T must be included.
# The union of all paths between nodes in T in a tree forms a subtree.
# We need to sum the weights of all edges that lie on the path between
# any two nodes in T.
# An edge (u, v) with weight w is part of the required set if and only if
# removing it splits the tree into two components, each containing at least
# one node from T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_t[u] stores the number of target nodes in the subtree rooted at u
    subtree_t = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in target_nodes:
        subtree_t[u] = 1
        
    total_cost = 0
    # Process in reverse topological order
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            p = parent[u]
            subtree_t[p] += subtree_t[u]
            # If this edge separates the tree into two parts both containing T nodes
            if subtree_t[u] > 0 and subtree_t[u] < t:
                total_cost += edge_weight[u]
                
    print(total_cost)

solve()
```
