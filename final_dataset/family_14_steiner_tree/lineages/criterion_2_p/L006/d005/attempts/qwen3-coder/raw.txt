import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def main():
    n, m, t = map(int, sys.stdin.readline().split())
    priority = set(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    # Sort edges by weight
    edges.sort()
    
    # We need to connect all priority nodes
    # We'll use a modified Kruskal's algorithm
    # But we only care about connecting priority nodes
    
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    total_cost = 0
    connected_priority = set()
    
    # First, add one priority node to start
    start_node = next(iter(priority))
    connected_priority.add(start_node)
    
    # For remaining priority nodes, we need to connect them
    remaining_priority = priority - connected_priority
    
    # We'll use a priority queue to find the cheapest way to connect
    # a new priority node to the current component
    
    # Initially, we consider all edges that touch our starting node
    pq = []
    in_pq = set()
    
    def add_edges_for_node(node):
        for i in range(len(edges)):
            w, u, v = edges[i]
            if u == node or v == node:
                if (w, u, v) not in in_pq and (w, v, u) not in in_pq:
                    heapq.heappush(pq, (w, u, v))
                    in_pq.add((w, u, v))
    
    add_edges_for_node(start_node)
    
    while remaining_priority:
        # Find the cheapest edge that connects to a new priority node
        found = False
        while pq:
            w, u, v = heapq.heappop(pq)
            in_pq.discard((w, u, v))
            
            # Check if this edge connects us to a new priority node
            other = None
            if u in connected_priority and v not in connected_priority:
                other = v
            elif v in connected_priority and u not in connected_priority:
                other = u
                
            if other is not None and other in priority:
                # Connect this edge
                total_cost += w
                union(parent, rank, u, v)
                connected_priority.add(other)
                remaining_priority.discard(other)
                
                # Add edges for the newly added node
                add_edges_for_node(other)
                found = True
                break
            elif other is not None:
                # Connect but it's not a priority node
                total_cost += w
                union(parent, rank, u, v)
                connected_priority.add(other)
                # Add edges for the newly added node
                add_edges_for_node(other)
                found = True
                break
        
        if not found:
            # If we couldn't find an edge, we need to reconsider
            # This shouldn't happen given the problem constraints
            break
            
    print(total_cost)

if __name__ == "__main__":
    main()
