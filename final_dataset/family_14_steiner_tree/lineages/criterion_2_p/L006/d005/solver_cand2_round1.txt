```python
import sys

# The problem states that the m = n-1 constraints form a tree structure
# because they are minimally restrictive and connect all n variables.
# We need to ensure all variables in the priority set are in the same
# connected component. This is equivalent to finding the minimum weight
# of the edges on the unique paths between all nodes in the priority set.
# In a tree, the union of paths between a set of nodes forms a subtree.
# The edges that must be included are exactly those that lie on the paths
# between any two nodes in the priority set.
# An edge (u, v) with weight w must be included if and only if removing
# it splits the tree into two components, both of which contain at least
# one node from the priority set.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_nodes = set()
    idx = 3
    for _ in range(t):
        priority_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # count_priority[u] will store the number of priority nodes in the
    # subtree rooted at u (when rooted at an arbitrary node, say 1).
    count_priority = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in priority_nodes:
            count_priority[u] += 1
        if parent[u] != 0:
            count_priority[parent[u]] += count_priority[u]
            # If the edge (u, parent[u]) separates the tree such that
            # both sides have at least one priority node, it must be included.
            if count_priority[u] > 0 and count_priority[u] < t:
                total_cost += edge_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
