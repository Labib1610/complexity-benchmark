```python
import sys

# The problem asks for the minimum cost to connect all nodes in a priority set
# in a tree structure. Since m = n - 1 and the graph is connected, it is a tree.
# To ensure x_i = x_j for all i, j in the priority set, we must select a set of
# edges such that all nodes in the priority set belong to the same connected component.
# In a tree, the unique path between any two nodes in the priority set must be
# included. The union of all paths between nodes in the priority set forms a
# subtree. The cost is the sum of weights of all edges in this subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_nodes = set()
    idx = 3
    for _ in range(t):
        priority_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the smallest subtree containing all priority nodes.
    # This is equivalent to finding all edges that lie on a path between
    # any two priority nodes.
    # An edge (u, v) with weight w is in the subtree if and only if
    # the two components formed by removing (u, v) both contain at least
    # one priority node.
    
    # Root the tree at an arbitrary priority node
    root = next(iter(priority_nodes))
    
    subtree_priority_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in priority_nodes:
            subtree_priority_count[node] += 1
            
    for node in reversed(order):
        if node != root:
            p = parent[node]
            subtree_priority_count[p] += subtree_priority_count[node]
            # If this edge separates the tree into two parts both containing
            # priority nodes, it must be included.
            if 0 < subtree_priority_count[node] < t:
                total_cost += edge_to_parent_weight[node]
                
    print(total_cost)

solve()
```
