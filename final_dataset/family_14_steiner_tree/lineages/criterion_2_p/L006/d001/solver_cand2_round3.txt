```python
import sys

# The problem states m = n - 1 and the graph is a tree.
# We need to find the minimum cost to connect all t critical stations.
# In a tree, the minimum set of edges connecting a subset of nodes is the
# union of all paths between any two critical nodes.
# This is equivalent to the sum of weights of all edges that have at least
# one critical node in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # count_crit[u] will store the number of critical nodes in the subtree of u
    count_crit = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order
    for u in reversed(order):
        if u in critical_nodes:
            count_crit[u] += 1
        
        for v, w in adj[u]:
            if v == parent[u]:
                # This is the edge (u, parent[u]) with weight w
                # If 0 < count_crit[u] < t, this edge must be included
                if 0 < count_crit[u] < t:
                    total_cost += w
            else:
                count_crit[u] += count_crit[v]
                
    print(total_cost)

solve()
```
