```python
import sys

# The problem states that m = n - 1 and the graph is connected.
# This means the graph is a tree.
# To connect all critical facilities, we must include all edges that lie on
# the unique path between any two critical facilities.
# In a tree, the set of edges that connect a subset of nodes is exactly the
# set of edges that lie on the paths between these nodes.
# An edge (u, v) with weight w is part of the required set if and only if
# removing it splits the tree into two components, each containing at least
# one critical facility.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count_critical[u] will store the number of critical nodes in the subtree of u
    count_critical = [0] * (n + 1)
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order to compute subtree counts
    for u in reversed(order):
        if u in critical_nodes:
            count_critical[u] += 1
        if parent[u] != -1:
            count_critical[parent[u]] += count_critical[u]
            
    # An edge (u, parent[u]) is needed if count_critical[u] > 0 
    # and count_critical[u] < t
    ans = 0
    for u in range(2, n + 1):
        c = count_critical[u]
        if 0 < c < t:
            # Find weight of edge (u, parent[u])
            for v, w in adj[u]:
                if v == parent[u]:
                    ans += w
                    break
                    
    print(ans)

if __name__ == '__main__':
    solve()
```
