We need the minimum cost to connect all target sensors, possibly using non-target sensors as intermediates.  

Since the graph is a tree (m = n - 1), we can find the minimum Steiner tree connecting the target sensors.  
This is equivalent to finding the minimal subtree that contains all target nodes.  

The minimal subtree cost = sum of all edges in the tree minus the cost of edges that are not needed (i.e., edges not on any path between target nodes).  

We can find the needed edges by marking all nodes that are on paths between target nodes, using a DFS from the targets.

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    targets = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    total_cost = 0
    
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        total_cost += w
    
    # Find nodes that are part of minimal subtree connecting targets
    parent = [0] * (n + 1)
    parent_edge_cost = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    # DFS from any target
    start = next(iter(targets))
    stack = [start]
    visited[start] = True
    order = []
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_cost[v] = w
                stack.append(v)
    
    needed = [False] * (n + 1)
    for node in targets:
        needed[node] = True
    
    # Propagate needed status up the tree
    for u in reversed(order):
        if needed[u] and parent[u] != 0:
            needed[parent[u]] = True
    
    # Sum cost of edges in minimal subtree
    min_cost = 0
    for u in range(1, n + 1):
        if needed[u] and parent[u] != 0 and needed[parent[u]]:
            min_cost += parent_edge_cost[u]
    
    print(min_cost)

if __name__ == "__main__":
    solve()
```
