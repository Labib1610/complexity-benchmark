```python
import sys

# The problem states m = n - 1, so the graph is a tree.
# A subset of reservoirs is fully mixable if it is connected.
# We start with n reservoirs (a tree).
# We can seal a reservoir if the remaining set is still connected.
# This means we can only remove leaves of the current tree that are not critical.
# We repeat this process: remove a non-critical leaf, then check if the new leaf is non-critical.
# Effectively, we are pruning all nodes that are not on the unique simple path 
# between any two critical nodes (the Steiner Tree of the critical nodes).
# The remaining nodes form the minimal subtree containing all critical nodes.
# The cost is the sum of weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = set()
    idx = 3
    for _ in range(t):
        critical.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t == 0:
        print(0)
        return
    if t == 1:
        print(0)
        return

    # To find the sum of weights of the minimal subtree containing all critical nodes:
    # This is the sum of weights of all edges that lie on a path between any two critical nodes.
    # An edge (u, v) with weight w is in the minimal subtree if and only if 
    # the two components formed by removing (u, v) both contain at least one critical node.
    
    # Root the tree at an arbitrary critical node
    root = next(iter(critical))
    
    subtree_has_critical = [False] * (n + 1)
    for c in critical:
        subtree_has_critical[c] = True
        
    total_cost = 0
    
    # DFS to compute subtree_has_critical
    stack = [(root, -1)]
    order = []
    parent = {root: -1}
    edge_to_parent_weight = {root: 0}
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != p:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        if u == root:
            continue
        p = parent[u]
        if subtree_has_critical[u]:
            total_cost += edge_to_parent_weight[u]
            subtree_has_critical[p] = True
            
    print(total_cost)

solve()
```
