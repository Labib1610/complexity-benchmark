```python
import sys

# The problem states that the initial set of n slots is fully mixable and m = n - 1.
# This implies the initial graph is a tree.
# A subset of slots is fully mixable if and only if the subgraph induced by those slots is connected.
# We start with a tree and want to seal slots (remove nodes) such that the remaining graph
# is still connected and contains all t critical slots.
# Removing a node (and its incident edges) from a tree keeps the remaining graph connected
# if and only if the removed node is a leaf.
# We can repeatedly remove leaves as long as they are not critical.
# This is equivalent to finding the minimal subtree that contains all t critical nodes.
# The total maintenance budget is the sum of weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_slots = set()
    idx = 3
    for _ in range(t):
        critical_slots.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t == 0:
        print(0)
        return
    if t == 1:
        print(0)
        return

    # The minimal subtree containing all critical nodes is the union of all paths
    # between any two critical nodes.
    # We can find this by pruning all nodes that are not critical and not on a path
    # between two critical nodes.
    # A node can be pruned if it is not critical and has degree 1.
    
    degree = [0] * (n + 1)
    for i in range(1, n + 1):
        degree[i] = len(adj[i])
        
    is_critical = [False] * (n + 1)
    for s in critical_slots:
        is_critical[s] = True
        
    queue = []
    for i in range(1, n + 1):
        if degree[i] == 1 and not is_critical[i]:
            queue.append(i)
            
    removed = [False] * (n + 1)
    total_weight = sum(e[2] for e in edges)
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        removed[u] = True
        for v, w in adj[u]:
            if not removed[v]:
                total_weight -= w
                degree[v] -= 1
                if degree[v] == 1 and not is_critical[v]:
                    queue.append(v)
                    
    print(total_weight)

solve()
```
