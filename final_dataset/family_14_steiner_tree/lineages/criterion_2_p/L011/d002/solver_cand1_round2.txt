```python
import sys

# The problem states that the potential links form a tree (n nodes, n-1 edges, 
# and every facility can reach any other through a unique sequence).
# To connect a subset of core facilities in a tree, we must include all edges 
# that lie on the paths between any two core facilities.
# This is equivalent to finding the smallest subtree that contains all core nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core_facilities = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the sum of weights of all edges that are part of the 
    # minimal subtree connecting all core facilities.
    # An edge (u, v) with weight w is in the subtree if and only if 
    # there is at least one core node in the component containing u 
    # (when the edge is removed) and at least one core node in the 
    # component containing v.
    
    # Let's root the tree at an arbitrary core node (e.g., the first one).
    root = next(iter(core_facilities))
    
    # count_core[u] = number of core nodes in the subtree rooted at u
    count_core = [0] * (n + 1)
    total_weight = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(root, -1, 0)] # node, parent, weight_from_parent
    order = []
    parent_info = {} # node: (parent, weight)
    
    dfs_stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while dfs_stack:
        u = dfs_stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent_info[v] = (u, w)
                dfs_stack.append(v)
                
    for u in order:
        if u in core_facilities:
            count_core[u] = 1
            
    # Process in reverse topological order to propagate core counts
    for u in reversed(order):
        if u == root:
            continue
        p, w = parent_info[u]
        count_core[p] += count_core[u]
        # If the subtree at u contains some but not all core nodes, 
        # the edge (p, u) must be included.
        if 0 < count_core[u] < t:
            total_weight += w
            
    print(total_weight)

solve()
```
