```python
import sys

# The problem states that the graph of all proposed connections is a tree 
# (n buildings, n-1 connections, exactly one path between any two).
# We need to find the minimum cost to connect all core facilities.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of core facilities, we must include all edges that lie 
# on the unique paths between any pair of core facilities.
# This is equivalent to finding the minimal subtree that contains all core facilities.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core_facilities = set()
    idx = 3
    for _ in range(t):
        core_facilities.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to identify all edges that are part of the minimal subtree 
    # connecting all core facilities. An edge (u, v) with weight w is in 
    # the minimal subtree if and only if both components formed by removing 
    # (u, v) contain at least one core facility.
    
    # Root the tree at an arbitrary core facility (e.g., the first one)
    root = next(iter(core_facilities))
    
    # subtree_has_core[u] will store if the subtree rooted at u contains a core facility
    subtree_has_core = [False] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u in core_facilities:
            subtree_has_core[u] = True
        
        if u != root:
            if subtree_has_core[u]:
                total_cost += edge_to_parent_weight[u]
                subtree_has_core[parent[u]] = True
                
    print(total_cost)

solve()
```
