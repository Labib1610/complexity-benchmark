```python
import sys

# The problem states m = n - 1 and the graph is connected (it's a tree).
# We need to find the minimum cost to connect all reserve currencies.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of nodes in a tree, we must include all edges that lie
# on the paths between any pair of reserve nodes.
# This is equivalent to finding the union of all paths between reserve nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    reserves = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Root the tree at the first reserve currency
    root = next(iter(reserves))
    
    # parent[u] = (p, weight_to_p)
    parent = [None] * (n + 1)
    is_reserve_subtree = [False] * (n + 1)
    
    # DFS to build parent pointers and identify subtrees containing reserves
    stack = [(root, 0, 0)]
    order = []
    while stack:
        u, p, w = stack.pop()
        parent[u] = (p, w)
        order.append(u)
        for v, weight in adj[u]:
            if v != p:
                stack.append((v, u, weight))
                
    for r in reserves:
        is_reserve_subtree[r] = True
        
    # Process in reverse topological order (leaves to root)
    total_cost = 0
    for u in reversed(order):
        if u == root:
            continue
        p, w = parent[u]
        if is_reserve_subtree[u]:
            total_cost += w
            is_reserve_subtree[p] = True
            
    print(total_cost)

solve()
```
