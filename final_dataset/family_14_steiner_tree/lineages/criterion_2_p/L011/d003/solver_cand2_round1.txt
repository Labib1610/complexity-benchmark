```python
import sys

# The problem asks for the minimum cost to connect all t observable variables
# in a graph where we have n nodes and n-1 edges (a tree).
# Since the graph is a tree, there is exactly one path between any two nodes.
# To make all t observable variables equal, we must enforce all constraints
# (edges) that lie on the paths connecting these t variables.
# This is equivalent to finding the union of all paths between the t nodes.
# In a tree, the union of paths between a set of nodes S is the set of edges
# that belong to the minimal subtree containing S (the Steiner Tree).
# Since the graph is already a tree, the Steiner Tree is simply the unique
# subtree that connects all nodes in S.
# An edge belongs to this subtree if and only if removing it splits the tree
# into two components, each containing at least one node from S.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    observables = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many observable nodes are in the subtree of each node
    # when rooted at 1.
    subtree_observables = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in observables:
            subtree_observables[u] += 1
        if parent[u] != 0:
            subtree_observables[parent[u]] += subtree_observables[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the subtree at i contains some but not all observable nodes,
        # the edge connecting i to its parent must be included.
        count = subtree_observables[i]
        if count > 0 and count < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
