import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    anchors = set()
    idx = 3
    for _ in range(t):
        anchors.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the minimal subtree containing all anchor zones.
    # This is equivalent to the sum of weights of all edges that lie on 
    # the path between any two anchor zones.
    # An edge (u, v) with weight w is part of the minimal subtree if and only if
    # the subtree formed by removing this edge has anchor zones in both components.
    
    is_anchor = [False] * (n + 1)
    for a in anchors:
        is_anchor[a] = True
        
    subtree_anchors = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        subtree_anchors[u] += (1 if is_anchor[u] else 0)
        if parent[u] != 0:
            subtree_anchors[parent[u]] += subtree_anchors[u]
            # If this edge separates some anchors from others, it must be included
            if 0 < subtree_anchors[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
