```python
import sys

# The problem states that the system of n modules and m=n-1 synergies is fully cohesive.
# A graph with n nodes and n-1 edges that is cohesive must be a tree.
# A subset of a tree is cohesive if and only if it is connected (a subtree).
# We need to find a subtree that contains all t critical modules such that the sum of
# weights of edges within the subtree is minimized.
# Wait, the problem asks for the minimum total cost of triggered synergies.
# If we select a subset of modules K, the cost is the sum of weights of edges (u, v)
# where u, v are in K.
# Let E be the set of all edges. Let S be the set of edges in the subtree K.
# The cost is sum_{e in S} weight(e).
# To minimize this, we want the smallest subtree containing all critical modules.
# The smallest subtree containing a set of nodes is the Steiner Tree.
# In a tree, the Steiner Tree is unique: it is the union of all paths between
# any two critical modules.
# The cost is the sum of weights of all edges that lie on the path between
# any two critical modules.
# An edge e splits the tree into two components of sizes s_e and n-s_e.
# An edge e is in the Steiner Tree if and only if both components contain at least one critical module.
# Let T be the total number of critical modules.
# An edge e is in the Steiner Tree if the number of critical modules in one component
# is > 0 and < T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count critical nodes in subtree
    subtree_critical = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in critical_nodes:
            subtree_critical[u] += 1
        if parent[u] != 0:
            subtree_critical[parent[u]] += subtree_critical[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        count = subtree_critical[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

solve()
```
