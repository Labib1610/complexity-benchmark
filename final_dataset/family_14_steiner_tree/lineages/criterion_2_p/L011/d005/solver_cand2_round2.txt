```python
import sys

# The problem asks us to find a subset of nodes (the enclave) such that:
# 1. All primary nodes are in the enclave.
# 2. The subgraph induced by the enclave is connected.
# 3. The sum of weights of all edges in the induced subgraph is minimized.
# Since the original graph is a tree, the induced subgraph of a set of nodes
# is connected if and only if the set of nodes forms a subtree.
# The smallest subtree containing all primary nodes is the Steiner Tree,
# which in a tree is simply the union of all paths between primary nodes.
# Let S be the set of nodes in this minimal Steiner Tree.
# Any node v not in S, if added to the enclave, would add all edges connecting v
# to the existing enclave. Since the graph is a tree, if we add a node v
# adjacent to the current enclave, we add exactly one edge.
# To minimize the cost, we should only include nodes in the enclave if they
# are part of the minimal Steiner Tree or if adding them doesn't increase the cost.
# However, the condition says: "The chosen interfaces will be EXACTLY those
# potential interfaces that have BOTH of their endpoints within the enclave."
# If we have a Steiner Tree, adding a leaf node v adjacent to the tree
# adds one edge (u, v) with weight w. This increases the cost by w.
# Thus, we should only include nodes that are part of the minimal Steiner Tree.
# The cost is the sum of all edges (u, v) where both u and v are in the Steiner Tree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    primary_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Identify nodes in the minimal Steiner Tree
    # A node is in the Steiner Tree if it lies on a path between two primary nodes.
    # This is equivalent to saying the subtree rooted at v contains at least one
    # primary node and the rest of the tree contains at least one primary node.
    
    has_primary = [False] * (n + 1)
    for p in primary_nodes:
        has_primary[p] = True
        
    subtree_has_primary = [False] * (n + 1)
    
    # DFS to find which edges are part of the Steiner Tree
    # An edge (u, v) is in the Steiner Tree if the subtree at v (when rooted at u)
    # contains at least one primary node AND the rest of the tree contains at least one.
    
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    subtree_has_primary = [False] * (n + 1)
    for i in range(1, n + 1):
        if has_primary[i]:
            subtree_has_primary[i] = True
            
    for u in reversed(order):
        if u != 1 and subtree_has_primary[u]:
            subtree_has_primary[parent[u]] = True
            
    total_cost = 0
    for i in range(2, n + 1):
        # Edge (i, parent[i]) is in the Steiner Tree if both i and parent[i]
        # are part of the Steiner Tree.
        # A node u is in the Steiner Tree if it's on a path between primary nodes.
        # This is true if the subtree at u has a primary node AND the rest of the tree has one.
        if subtree_has_primary[i]:
            # Check if the "rest of the tree" has a primary node
            # This is true if (total_primary - subtree_primary_count) > 0
            # We can precalculate subtree_primary_count
            pass

    # Simpler approach:
    # An edge is in the Steiner Tree if it separates the set of primary nodes
    # into two non-empty sets.
    subtree_primary_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if has_primary[i]:
            subtree_primary_count[i] = 1
            
    for u in reversed(order):
        if u != 1:
            subtree_primary_count[parent[u]] += subtree_primary_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        count = subtree_primary_count[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

solve()
```
