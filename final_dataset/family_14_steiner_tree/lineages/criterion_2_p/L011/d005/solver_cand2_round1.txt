```python
import sys

# The problem states that the full set of n variables is indecomposable with m = n-1 protocols.
# This means the graph formed by the protocols is a tree.
# A subset of variables is indecomposable if and only if the subgraph induced by these variables
# is connected.
# We need to find a connected subgraph (a subtree) that contains all t critical variables
# and minimizes the sum of weights of the edges in that subtree.
# The minimum weight connected subgraph containing a set of nodes is the Steiner Tree.
# For a tree, the Steiner Tree is simply the unique minimal subtree connecting all critical nodes.
# This is the union of all paths between any two critical nodes.
# The cost is the sum of weights of all edges that lie on at least one path between two critical nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = set()
    idx = 3
    for _ in range(t):
        critical.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # To find the sum of weights of edges in the minimal subtree:
    # An edge (u, v) with weight w is in the minimal subtree if and only if
    # removing it splits the tree into two components, each containing at least one critical node.
    
    # Root the tree at an arbitrary critical node (e.g., the first one)
    root = next(iter(critical))
    
    subtree_has_critical = [False] * (n + 1)
    for c in critical:
        subtree_has_critical[c] = True
        
    total_cost = 0
    
    # Use DFS to compute subtree_has_critical
    stack = [(root, -1)]
    order = []
    parent = {root: -1}
    edge_to_parent_weight = {root: 0}
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != p:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        if u == root:
            continue
        p = parent[u]
        if subtree_has_critical[u]:
            total_cost += edge_to_parent_weight[u]
            subtree_has_critical[p] = True
            
    print(total_cost)

solve()
```
