```python
import sys

# The problem asks for the minimum weight subset of edges (e_u - e_v) such that
# the dimension of the intersection of the span of these edges with the target
# subspace V_T is t-1.
# The vectors e_u - e_v represent edges in a graph with n vertices.
# The span of a set of edges is the set of vectors sum(c_i * (e_{u_i} - e_{v_i})).
# This is equivalent to the cycle space/cut space properties.
# Specifically, the dimension of the intersection of the span of edges with
# the subspace spanned by target indices T is |T| - (number of connected components
# formed by the edges when restricted to T).
# We want this dimension to be t-1, which means the edges must connect all t
# target vertices into a single connected component within the graph.
# Since we have m = n-1 edges and the graph is connected (implied by span dimension n-1),
# we are looking for a spanning tree.
# The condition "dimension of intersection is t-1" means that in the graph
# formed by the chosen edges, all t target vertices must belong to the same
# connected component.
# This is equivalent to finding the Minimum Spanning Tree of the graph where
# we contract all target vertices into a single super-node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        
        # Contract all target vertices into a single node (labeled 0)
        u_mapped = 0 if u in targets else u
        v_mapped = 0 if v in targets else v
        
        if u_mapped != v_mapped:
            edges.append((w, u_mapped, v_mapped))
            
    edges.sort()
    
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    total_weight = 0
    edges_count = 0
    for w, u, v in edges:
        if union(u, v):
            total_weight += w
            edges_count += 1
            
    print(total_weight)

solve()
```
