```python
import sys

# The problem states that the m = n - 1 options form a tree.
# We need to select a subset of nodes (the active zone) such that:
# 1. All required sites are in the active zone.
# 2. The edges (u, v) where both u and v are in the active zone must connect all nodes in the active zone.
# Since the original graph is a tree, the only way for a subset of nodes to be connected
# by the edges induced by that subset is if the subset forms a connected subtree of the original tree.
# Thus, the active zone must be a connected subtree containing all required sites.
# The cost is the sum of weights of all edges (u, v) where both u and v are in the active zone.
# Let S be the set of nodes in the active zone. We want to minimize sum_{e=(u,v), u,v in S} weight(e).
# This is equivalent to finding a subtree S containing all required sites that minimizes the sum of weights of edges in S.
# Since all weights are positive, the minimum cost subtree containing all required sites is the Steiner Tree,
# which in a tree is simply the unique minimal subtree connecting all required sites.
# The edges in this minimal subtree are exactly the edges that lie on the paths between any two required sites.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    required = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the sum of weights of all edges that lie on the path between any two required nodes.
    # An edge (u, v) with weight w is part of the minimal subtree if and only if
    # removing it splits the tree into two components, each containing at least one required node.
    
    subtree_required_count = [0] * (n + 1)
    edge_weights = {}
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weights[v] = w
                stack.append(v)
                
    for node in order:
        if node in required:
            subtree_required_count[node] = 1
            
    total_cost = 0
    for node in reversed(order):
        if node == 1:
            continue
        p = parent[node]
        if subtree_required_count[node] > 0 and subtree_required_count[node] < t:
            total_cost += edge_weights[node]
        subtree_required_count[p] += subtree_required_count[node]
        
    print(total_cost)

solve()
```
