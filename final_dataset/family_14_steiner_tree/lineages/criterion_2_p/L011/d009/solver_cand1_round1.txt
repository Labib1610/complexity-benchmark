```python
import sys

# The problem states that the system is monolithic if all remaining variables
# must share a single common value. This means the graph of remaining variables
# and remaining equality conditions must be connected.
# We start with a tree (n nodes, n-1 edges).
# We can eliminate a variable if it is not a mandated variable and its
# elimination does not disconnect the graph.
# In a tree, a node can be removed if it is a leaf.
# If we remove a leaf, the remaining graph is still a tree (connected).
# We can keep removing leaves as long as they are not mandated variables.
# This is equivalent to finding the minimal subtree that contains all mandated variables.
# Any edge not in this minimal subtree can be removed.
# The total cost of the remaining edges is the sum of weights of edges in the
# minimal subtree connecting all mandated variables.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    mandated = set()
    idx = 3
    for _ in range(t):
        mandated.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    total_weight = 0
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        total_weight += w
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Find the minimal subtree containing all mandated variables.
    # This is the union of paths between all mandated variables.
    # We can use pruning: repeatedly remove leaves that are not mandated.
    
    degree = [0] * (n + 1)
    for i in range(1, n + 1):
        degree[i] = len(adj[i])
        
    queue = []
    for i in range(1, n + 1):
        if degree[i] == 1 and i not in mandated:
            queue.append(i)
            
    removed_weight = 0
    is_removed = [False] * (n + 1)
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        is_removed[u] = True
        for v, w in adj[u]:
            if not is_removed[v]:
                removed_weight += w
                degree[v] -= 1
                if degree[v] == 1 and v not in mandated:
                    queue.append(v)
                    
    print(total_weight - removed_weight)

solve()
```
