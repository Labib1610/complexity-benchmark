```python
import sys

# The problem states that the chip is a tree (n nodes, n-1 edges).
# A subset of nodes is "hydrostatically locked" if it is connected.
# We can seal a node if it is not an extraction chamber and its removal
# does not disconnect the remaining nodes.
# In a tree, removing a node disconnects the tree if and only if it is not a leaf.
# Thus, we can only seal leaves that are not extraction chambers.
# Once a leaf is sealed, its parent might become a new leaf.
# This is equivalent to saying we can prune any node that is not an extraction chamber
# as long as it is a leaf in the current tree.
# Effectively, we can remove any node that is not an extraction chamber
# provided it is not on the unique path between any two extraction chambers.
# The remaining nodes will be exactly the minimal subtree containing all extraction chambers.
# The total capacity of the remaining tubes is the sum of weights of all edges
# that are part of the minimal subtree connecting all extraction chambers.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    extraction_chambers = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t == 0:
        print(0)
        return
    if t == 1:
        print(0)
        return

    # Identify nodes that are part of the minimal subtree connecting all extraction chambers.
    # A node is in the minimal subtree if it lies on a path between two extraction chambers.
    # We can use DFS to find the subtree.
    
    is_extraction = [False] * (n + 1)
    for x in extraction_chambers:
        is_extraction[x] = True
        
    has_extraction_in_subtree = [False] * (n + 1)
    
    # Root the tree at an arbitrary extraction chamber
    root = next(iter(extraction_chambers))
    
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_extraction[u]:
            has_extraction_in_subtree[u] = True
        
        if has_extraction_in_subtree[u] and parent[u] != 0:
            total_weight += edge_to_parent_weight[u]
            has_extraction_in_subtree[parent[u]] = True
            
    print(total_weight)

solve()
```
