```python
import sys

# The problem states that the initial set of n oscillators is fully phase-locked.
# Since m = n - 1 and the graph is connected (implied by the phase-locked property),
# the network is a tree.
# A subset S is phase-locked if all oscillators in S are in the same connected component
# of the subgraph induced by S.
# We cannot isolate reference oscillators.
# We can isolate an oscillator if the remaining set S remains phase-locked.
# This means we can only remove leaves of the current tree that are not reference oscillators.
# If we remove a leaf, the edge connected to it is removed.
# We want to minimize the sum of weights of remaining edges.
# This is equivalent to removing as many edges as possible.
# An edge can be removed if and only if one of its endpoints can be isolated.
# Since we can only isolate non-reference oscillators, we can prune any subtree
# that does not contain any reference oscillators.
# The remaining edges will be exactly those that lie on the unique paths between
# any two reference oscillators.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    refs = set()
    idx = 3
    for _ in range(t):
        refs.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t == 0:
        print(0)
        return
    if t == 1:
        print(0)
        return

    # We need to find the sum of weights of all edges that lie on the paths
    # between any two reference oscillators.
    # An edge (u, v) with weight w is on such a path if and only if
    # removing it splits the tree into two components, both of which contain
    # at least one reference oscillator.
    
    # Count reference oscillators in each subtree
    ref_count = [0] * (n + 1)
    for r in refs:
        ref_count[r] = 1
        
    total_weight = 0
    
    # Root the tree at an arbitrary reference oscillator
    root = next(iter(refs))
    
    stack = [(root, -1)]
    order = []
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != p:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        if parent[u] != -1:
            ref_count[parent[u]] += ref_count[u]
            
    for i in range(1, n + 1):
        if i == root:
            continue
        # If the subtree at i contains some but not all reference oscillators,
        # the edge to its parent must be kept.
        if 0 < ref_count[i] < t:
            total_weight += edge_to_parent_weight[i]
            
    print(total_weight)

solve()
```
