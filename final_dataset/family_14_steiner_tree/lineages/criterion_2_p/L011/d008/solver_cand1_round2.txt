```python
import sys

# The problem states that a subset is phase-locked if it is impossible to 
# partition it into two non-empty sets with no edges between them.
# This is equivalent to saying the subset must be connected.
# Since m = n - 1, the graph is a tree.
# A subset of a tree is connected if and only if it is a subtree.
# We start with the full tree (which is connected).
# We can isolate an oscillator (remove a node) if the remaining nodes 
# stay connected. This means we can only remove leaves of the current tree.
# We cannot remove any of the t reference oscillators.
# Thus, we can remove any node that is not a reference oscillator, 
# provided it is a leaf in the current tree.
# By repeatedly removing non-reference leaves, we eventually reach the 
# minimal subtree that contains all t reference oscillators (the Steiner Tree).
# Any node not in this Steiner Tree can be removed.
# The remaining links are exactly the edges of the Steiner Tree.
# The cost is the sum of weights of edges in the Steiner Tree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    refs = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t == 0:
        print(0)
        return
    if t == 1:
        print(0)
        return

    # Steiner Tree in a tree is the unique path between all reference nodes.
    # We can find this by pruning all nodes that are not part of any path 
    # between two reference nodes.
    # A node is in the Steiner Tree if it is a reference node or if it 
    # lies on a path between two reference nodes.
    # Equivalently, a node is NOT in the Steiner Tree if it is in a 
    # subtree that contains no reference nodes.
    
    is_ref = [False] * (n + 1)
    for r in refs:
        is_ref[r] = True
        
    degree = [0] * (n + 1)
    for i in range(1, n + 1):
        degree[i] = len(adj[i])
        
    queue = []
    for i in range(1, n + 1):
        if degree[i] == 1 and not is_ref[i]:
            queue.append(i)
            
    removed = [False] * (n + 1)
    total_weight = sum(e[2] for e in edges)
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        removed[u] = True
        for v, w in adj[u]:
            if not removed[v]:
                total_weight -= w
                degree[v] -= 1
                if degree[v] == 1 and not is_ref[v]:
                    queue.append(v)
                    
    print(total_weight)

solve()
```
