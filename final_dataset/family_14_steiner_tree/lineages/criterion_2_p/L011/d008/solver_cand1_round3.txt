```python
import sys

# The problem describes a tree structure (n variables, n-1 constraints).
# A set of variables is "coupled" if they form a connected component.
# We start with a tree of n nodes. We must keep t target variables.
# We can remove a variable (leaf node) if the remaining set is still connected.
# This means we can only remove nodes that are leaves in the current tree
# and are not target variables.
# Effectively, we are pruning the tree until only the minimal subtree
# containing all t target variables remains.
# The cost of the remaining system is the sum of weights of the edges
# in the minimal subtree connecting all t target variables.
# This is the Steiner Tree problem on a tree, which is simply the sum of
# weights of all edges on the unique paths between the target variables.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Identify edges that are part of the minimal subtree connecting all targets.
    # An edge is in the minimal subtree if and only if both components
    # formed by removing the edge contain at least one target variable.
    
    subtree_weight = 0
    
    # Use DFS to count targets in subtrees
    target_count = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    for u in order:
        if u in targets:
            target_count[u] = 1
            
    for u in reversed(order):
        if u != 1:
            p = parent[u]
            target_count[p] += target_count[u]
            if target_count[u] > 0 and target_count[u] < t:
                subtree_weight += edge_to_parent_w[u]
                
    print(subtree_weight)

solve()
```
