import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n = data[0]
    m = data[1]
    t = data[2]
    idx = 3
    targets = data[idx:idx + t]
    idx += t
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = data[idx]
        v = data[idx + 1]
        w = data[idx + 2]
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t == 1:
        print(0)
        return
    
    target_count = [0] * (n + 1)
    for node in targets:
        target_count[node] = 1
    
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    q = [1]
    parent[1] = -1
    q_idx = 0
    while q_idx < len(q):
        u = q[q_idx]
        q_idx += 1
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                q.append(v)
    
    for u in reversed(q):
        p = parent[u]
        if p != -1:
            target_count[p] += target_count[u]
    
    ans = 0
    for u in range(2, n + 1):
        cnt = target_count[u]
        if 1 <= cnt <= t - 1:
            ans += parent_weight[u]
    
    print(ans)
