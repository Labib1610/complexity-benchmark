```python
import sys

# The problem asks for the minimum cost of remaining constraints after eliminating
# variables such that the remaining set is always coupled and target variables are kept.
# Since the initial graph is a tree (n nodes, n-1 edges, coupled), the condition
# "remaining variables must be coupled" means the remaining set of nodes must
# form a connected subgraph.
# If we eliminate a leaf node that is not a target variable, the remaining graph
# stays connected. We can keep eliminating leaves until all remaining leaves
# are target variables.
# The final set of variables will be the minimal subtree containing all target variables.
# Any edge not in this minimal subtree must be eliminated.
# The cost of the remaining system is the sum of weights of all edges in the
# minimal subtree connecting the target variables.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t == 0:
        print(0)
        return
    if t == 1:
        print(0)
        return

    # We need the sum of weights of edges in the minimal subtree containing all targets.
    # An edge is in the minimal subtree if and only if the two components it creates
    # both contain at least one target variable.
    
    # Root the tree at an arbitrary target node
    root = next(iter(targets))
    
    subtree_has_target = [False] * (n + 1)
    for target in targets:
        subtree_has_target[target] = True
        
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to propagate target existence
    for i in range(n - 1, 0, -1):
        u = order[i]
        if subtree_has_target[u]:
            p = parent[u]
            subtree_has_target[p] = True
            total_weight += edge_to_parent_weight[u]
            
    print(total_weight)

solve()
```
