```python
import sys

# The problem states m = n - 1, so the framework is a tree.
# A set of joints is cohesive if it is connected.
# Removing a joint is allowed if the remaining joints stay connected.
# In a tree, removing a node keeps the graph connected if and only if the node is a leaf.
# We cannot remove anchor joints.
# Thus, we can repeatedly remove any leaf that is not an anchor joint.
# This process will eventually leave only the minimal subtree containing all anchor joints.
# The remaining struts are those that form the minimal subtree connecting all anchor joints.
# The total mass of the remaining struts is the sum of weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    anchors = set()
    idx = 3
    for _ in range(t):
        anchors.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    total_weight = 0
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        total_weight += w
        idx += 3
        
    # The minimal subtree containing all anchor joints consists of all edges
    # that lie on the simple path between any two anchor joints.
    # An edge is in this subtree if and only if removing it splits the tree
    # into two components, both of which contain at least one anchor joint.
    
    is_anchor_subtree = [False] * (n + 1)
    
    # Count anchors in each subtree
    subtree_anchors = [0] * (n + 1)
    def dfs_count(u, p):
        count = 1 if u in anchors else 0
        for v, w in adj[u]:
            if v != p:
                count += dfs_count(v, u)
        subtree_anchors[u] = count
        return count
    
    dfs_count(1, -1)
    
    remaining_weight = 0
    # An edge (u, v) is in the minimal subtree if the number of anchors in the
    # component containing v (when edge (u, v) is cut) is > 0 and < t.
    # Let v be the child of u. The component at v has subtree_anchors[v] anchors.
    # If 0 < subtree_anchors[v] < t, the edge (u, v) is part of the minimal subtree.
    
    def dfs_sum(u, p):
        nonlocal remaining_weight
        for v, w in adj[u]:
            if v != p:
                if 0 < subtree_anchors[v] < t:
                    remaining_weight += w
                dfs_sum(v, u)
                
    dfs_sum(1, -1)
    print(remaining_weight)

solve()
```
