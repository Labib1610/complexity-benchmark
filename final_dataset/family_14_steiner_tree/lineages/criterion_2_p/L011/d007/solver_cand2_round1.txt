```python
import sys

# The problem states that the system has n components and m = n - 1 interactions.
# This means the system is a tree.
# A subset S is energy-cohesive if for every proper subset A, E(S) > E(A) + E(S \ A).
# In a tree, E(S) is the sum of weights of edges where both endpoints are in S.
# If S is a connected subgraph (a subtree), then E(S) is the sum of weights of all edges in S.
# If S is disconnected, E(S) = sum of E(connected components of S).
# The condition E(S) > E(A) + E(S \ A) for all A implies that S must be connected.
# If S is connected, E(S) is the sum of weights of edges in the subtree induced by S.
# We need to find a connected subtree S that contains all t mandatory components
# and minimizes the sum of weights of edges in S.
# The smallest connected subtree containing a set of nodes is the Steiner Tree.
# In a tree, the Steiner Tree is unique and consists of all edges on the paths
# between any two mandatory nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    mandatory = set()
    idx = 3
    for _ in range(t):
        mandatory.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # To find the sum of weights of the Steiner Tree:
    # 1. Root the tree at an arbitrary mandatory node.
    # 2. A node is in the Steiner Tree if it is a mandatory node or 
    #    if it has at least two branches in its subtree that contain mandatory nodes.
    # 3. An edge (u, v) where v is a child of u is in the Steiner Tree if 
    #    the subtree at v contains at least one mandatory node.
    
    has_mandatory = [False] * (n + 1)
    for node in mandatory:
        has_mandatory[node] = True
        
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [next(iter(mandatory))]
    visited = [False] * (n + 1)
    visited[stack[0]] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if has_mandatory[u]:
            if parent[u] != 0:
                has_mandatory[parent[u]] = True
                total_weight += edge_to_parent_weight[u]
                
    print(total_weight)

solve()
```
