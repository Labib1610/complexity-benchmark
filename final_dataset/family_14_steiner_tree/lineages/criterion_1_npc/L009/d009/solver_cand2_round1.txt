```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority = [int(x) for x in input_data[3:3+t]]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "for every pair of priority coordinates x and y, 
    # b_x + b_y is in the row space of M" is equivalent to saying that 
    # all priority coordinates must belong to the same connected component 
    # in the graph formed by the chosen edges.
    # If we pick a set of edges, the row space of the resulting matrix M 
    # contains b_u + b_v for every edge (u, v) in the set.
    # By transitivity, if we have a path between u and v, b_u + b_v is in the row space.
    # Thus, we need to connect all priority nodes in a single component.
    # This is a Steiner Tree problem on a graph where we want to connect 
    # the set of priority nodes with minimum weight.
    
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm.
    # dp[mask][i] = min cost to connect the subset of priority nodes 'mask' 
    # such that node 'i' is in the connected component.
    
    inf = float('inf')
    adj = [[inf] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        adj[i][i] = 0
    for u, v, w in edges:
        if w < adj[u][v]:
            adj[u][v] = adj[v][u] = w
            
    # Floyd-Warshall for all-pairs shortest paths
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if adj[i][k] + adj[k][j] < adj[i][j]:
                    adj[i][j] = adj[i][k] + adj[k][j]
                    
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][priority[i]] = 0
        
    for mask in range(1, 1 << t):
        # Submask splitting
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relax with edges
        for i in range(1, n + 1):
            if dp[mask][i] == inf: continue
            for j in range(1, n + 1):
                if adj[i][j] != inf:
                    dp[mask][j] = min(dp[mask][j], dp[mask][i] + adj[i][j])
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
