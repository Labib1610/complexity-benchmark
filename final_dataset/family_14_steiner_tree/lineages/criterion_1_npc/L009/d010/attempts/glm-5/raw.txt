import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    critical = [int(next(it)) for _ in range(t)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    # We need all critical controls to be in the same connected component.
    # This is a Steiner tree problem on the graph with terminals = critical controls.
    # Since t <= 10, we can use the Dreyfus-Wagner DP.
    # Nodes are 1-indexed; we'll map to 0-indexed internally.
    
    INF = float('inf')
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Dreyfus-Wagner algorithm for Steiner tree
    # dp[S][v] = minimum cost of a Steiner tree connecting terminals in S and node v
    # where v is the root of the tree (v may or may not be a terminal).
    
    # We'll use dictionaries or arrays. Since n <= 100 and t <= 10, 2^t = 1024.
    # dp[S][v] for S subset of terminals, v in 1..n
    
    # Initialize
    dp = [[INF] * (n + 1) for _ in range(1 << t)]
    for i, c in enumerate(critical):
        dp[1 << i][c] = 0
    
    # Precompute all subsets
    # First, for each subset S, compute dp[S] by considering subsets S' and S\S'
    # and combining via shortest paths.
    
    # We need shortest path distances between all pairs. Since n <= 100, we can run Dijkstra from each node.
    # But actually, the Dreyfus-Wagner algorithm can be implemented by:
    # 1. For each S, for each v, dp[S][v] = min over S' subset S, S' nonempty, S' != S of dp[S'][v] + dp[S\S'][v]
    # 2. Then, relax dp[S] using shortest paths: dp[S][v] = min(dp[S][v], min over u of dp[S][u] + dist(u, v))
    
    # Step 2 is essentially running a shortest path from all nodes with initial distances dp[S][*].
    # This can be done with Dijkstra (multi-source).
    
    import heapq
    
    for S in range(1, 1 << t):
        # Step 1: combine subsets
        # We iterate over proper nonempty subsets S' of S.
        # To avoid redundant work, we can iterate over S' = (S-1) & S, then S' = (S'-1) & S, etc.
        # But a simpler way: iterate over submask.
        sub = (S - 1) & S
        while sub > 0:
            # sub is a proper nonempty subset of S
            # We need to combine dp[sub] and dp[S ^ sub]
            for v in range(1, n + 1):
                if dp[sub][v] < INF and dp[S ^ sub][v] < INF:
                    val = dp[sub][v] + dp[S ^ sub][v]
                    if val < dp[S][v]:
                        dp[S][v] = val
            sub = (sub - 1) & S
        
        # Step 2: relax with shortest paths (multi-source Dijkstra)
        # Initial distances are dp[S][v]
        dist = dp[S][:]
        pq = []
        for v in range(1, n + 1):
            if dist[v] < INF:
                heapq.heappush(pq, (dist[v], v))
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue
            for v, w in adj[u]:
                nd = d + w
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(pq, (nd, v))
        dp[S] = dist
    
    full = (1 << t) - 1
    ans = min(dp[full][v] for v in range(1, n + 1))
    print(ans)

solve()
