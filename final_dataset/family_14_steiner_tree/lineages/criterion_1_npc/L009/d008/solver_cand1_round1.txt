```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_coords = [int(x) for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to connect all priority coordinates in a single connected component
    # using the available vectors b_u + b_v.
    # This is equivalent to finding a Minimum Spanning Tree in a graph where
    # nodes are 1..n and edges are (u, v) with weight w.
    # However, we only care about the connectivity of the priority coordinates.
    # This is a Steiner Tree problem, but since we can use any node as a Steiner point,
    # it's equivalent to finding the MST of the graph where we contract all 
    # priority nodes into one, or more simply, finding the MST of the graph
    # and then considering the cost to connect the priority nodes.
    # Actually, the condition "for every pair of distinct priority coordinates x and y, 
    # b_x + b_y is in the subspace" means all priority coordinates must belong 
    # to the same connected component in the graph formed by the chosen edges.
    
    # This is exactly the Steiner Tree problem where we want to connect a set of 
    # terminals (priority_coords) with minimum weight.
    # Since n is small (100), we can use the Dreyfus-Wagner algorithm or 
    # simply realize that for a small number of terminals (t <= 10), 
    # we can use the property that the optimal Steiner tree is a tree 
    # connecting the terminals.
    
    # Given the constraints and the nature of the problem (b_u + b_v), 
    # this is equivalent to finding the MST of the graph where we want to 
    # connect the set of priority nodes.
    
    # Let's use the standard approach for Steiner Tree with small t:
    # dp[mask][i] = min cost to connect the subset of terminals represented by mask,
    # with i being a node in the tree.
    
    dist = [[float('inf')] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
            
    # Floyd-Warshall to get all-pairs shortest paths
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    dp = [[float('inf')] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][priority_coords[i]] = 0
        
    for mask in range(1, 1 << t):
        for i in range(1, n + 1):
            submask = (mask - 1) & mask
            while submask > 0:
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
                submask = (submask - 1) & mask
        
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[mask][j] + dist[j][i])
                
    print(min(dp[(1 << t) - 1]))

solve()
```
