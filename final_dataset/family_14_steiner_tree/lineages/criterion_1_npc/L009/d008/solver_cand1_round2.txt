import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority = [int(x) for x in input_data[3:3+t]]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The condition "for every pair of priority coordinates x and y, b_x + b_y is in V"
    # is equivalent to saying that all priority coordinates belong to the same connected 
    # component in the graph formed by the chosen edges.
    # Let the chosen edges form a graph G' = (V, E').
    # If we pick a set of edges, the subspace spanned by {b_u + b_v | (u, v) in E'}
    # contains b_x + b_y if and only if x and y are in the same connected component of G'.
    # We need all priority coordinates to be in the same connected component.
    # We can introduce a virtual node 'root' and connect it to all priority nodes.
    # However, the standard way to solve this is to use Kruskal's or Prim's on a graph
    # where we want to connect all priority nodes.
    # This is the Steiner Tree problem, but since we only have n <= 100, 
    # we can use the property that we want to connect all priority nodes.
    # Actually, this is equivalent to finding a Minimum Spanning Tree in a graph
    # where we contract all priority nodes into one, but that's not quite right.
    # Correct approach:
    # We want to connect all priority nodes {p_1, ..., p_t}.
    # This is exactly the Steiner Tree problem. For t <= 10, we can use the 
    # Dreyfus-Wagner algorithm or simply notice that since n is small, 
    # we can use dynamic programming on subsets of priority nodes.
    
    # dp[mask][i] = min cost to connect the subset of priority nodes represented by mask,
    # with node i being part of the connected component.
    
    inf = float('inf')
    dist = [[inf] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
            
    # Floyd-Warshall to get all-pairs shortest paths
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][priority[i]] = 0
        
    for mask in range(1, 1 << t):
        for i in range(1, n + 1):
            submask = (mask - 1) & mask
            while submask > 0:
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
                submask = (submask - 1) & mask
        
        for i in range(1, n + 1):
            if dp[mask][i] == inf: continue
            for j in range(1, n + 1):
                if dist[i][j] != inf:
                    dp[mask][j] = min(dp[mask][j], dp[mask][i] + dist[i][j])
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
