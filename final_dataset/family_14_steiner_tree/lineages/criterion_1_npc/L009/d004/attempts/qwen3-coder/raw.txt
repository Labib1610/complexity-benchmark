import sys
import heapq
from collections import defaultdict

def dijkstra(graph, start, end):
    pq = [(0, start)]
    dist = {start: 0}
    while pq:
        d, u = heapq.heappop(pq)
        if u == end:
            return d
        if d > dist.get(u, float('inf')):
            continue
        for v, w in graph[u]:
            alt = d + w
            if alt < dist.get(v, float('inf')):
                dist[v] = alt
                heapq.heappush(pq, (alt, v))
    return float('inf')

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    critical = list(map(int, data[idx:idx+t]))
    idx += t
    
    edges = []
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        edges.append((u, v, w))
        idx += 3
    
    if t <= 1:
        print(0)
        return
    
    # Build graph
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Compute shortest paths between all pairs of critical nodes
    # We'll use a simplified version since t <= 10
    # Create a complete graph on critical nodes with edge weights as shortest path
    critical_set = set(critical)
    k = len(critical)
    
    # Distance matrix for critical nodes
    dist_matrix = [[float('inf')] * k for _ in range(k)]
    for i in range(k):
        dist_matrix[i][i] = 0
    
    for i in range(k):
        start_node = critical[i]
        # Run Dijkstra from start_node
        pq = [(0, start_node)]
        dist_local = {start_node: 0}
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist_local.get(u, float('inf')):
                continue
            for v, w in graph[u]:
                alt = d + w
                if alt < dist_local.get(v, float('inf')):
                    dist_local[v] = alt
                    heapq.heappush(pq, (alt, v))
        
        for j in range(k):
            target = critical[j]
            if target in dist_local:
                dist_matrix[i][j] = dist_local[target]
    
    # Now we have a complete graph on critical nodes with shortest path distances
    # Find Minimum Spanning Tree of this complete graph
    # Since k <= 10, we can use Prim's algorithm efficiently
    
    # Nodes are 0 to k-1 corresponding to indices in critical list
    in_mst = [False] * k
    min_edge = [float('inf')] * k
    min_edge[0] = 0
    total_cost = 0
    
    for _ in range(k):
        # Find the vertex with minimum edge weight not yet included in MST
        u = -1
        for v in range(k):
            if not in_mst[v]:
                if u == -1 or min_edge[v] < min_edge[u]:
                    u = v
        
        if u == -1:
            break
            
        in_mst[u] = True
        total_cost += min_edge[u]
        
        # Update min_edge values for adjacent vertices
        for v in range(k):
            if not in_mst[v] and dist_matrix[u][v] < min_edge[v]:
                min_edge[v] = dist_matrix[u][v]
    
    print(total_cost)

if __name__ == "__main__":
    main()
