```python
import sys

def solve():
    # Read n, m, t
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    # The problem asks for the minimum weight to make all elements in T 
    # belong to the same orbit. Since we are using transpositions, 
    # the orbit of an element x is the connected component containing x 
    # in the graph where edges are the chosen transpositions.
    # Thus, we need all elements in T to be in the same connected component.
    
    # Kruskal's algorithm approach:
    # We want to connect all nodes in T. This is equivalent to finding the 
    # Minimum Spanning Tree of the graph where we want to connect the set T.
    # We can add a dummy node 'S' and connect it to all nodes in T with weight 0.
    # Then we find the MST of the graph including S.
    
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # Connect all targets to a virtual node 0
    for i in range(1, t):
        union(targets[0], targets[i])
        
    edges.sort()
    
    total_weight = 0
    for w, u, v in edges:
        if union(u, v):
            total_weight += w
            
    print(total_weight)

solve()
```
