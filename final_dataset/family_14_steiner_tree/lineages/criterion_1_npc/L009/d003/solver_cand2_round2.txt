```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    hubs = [int(x) for x in input_data[3:3+t]]
    
    adj = [[float('inf')] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        adj[i][i] = 0
        
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        if w < adj[u][v]:
            adj[u][v] = adj[v][u] = w
        idx += 3
        
    # Floyd-Warshall to get all-pairs shortest paths
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if adj[i][j] > adj[i][k] + adj[k][j]:
                    adj[i][j] = adj[i][k] + adj[k][j]
                    
    if t <= 1:
        print(0)
        return

    # Steiner Tree problem for small t (t <= 10)
    # dp[mask][i] is min cost to connect subset of hubs represented by mask,
    # with i being a node in the tree.
    dp = [[float('inf')] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][hubs[i]] = 0
        
    for mask in range(1, 1 << t):
        # Combine sub-masks
        for submask in range((mask - 1) & mask, 0, -1):
            submask &= mask
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
        
        # Relax edges using Dijkstra
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != float('inf'):
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v in range(1, n + 1):
                if adj[u][v] != float('inf'):
                    if dp[mask][v] > d + adj[u][v]:
                        dp[mask][v] = d + adj[u][v]
                        heapq.heappush(pq, (dp[mask][v], v))
                        
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
