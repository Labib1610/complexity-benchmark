```python
import sys
import heapq

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))

    if t == 1:
        print(0)
        return

    full = (1 << t) - 1
    INF = 10**18
    dp = [[INF] * n for _ in range(1 << t)]

    # initialize single terminals
    for idx, node in enumerate(terminals):
        dp[1 << idx][node] = 0

    # function to run multi-source Dijkstra for a given mask
    def dijkstra(mask: int):
        dist = dp[mask]
        heap = [(dist[v], v) for v in range(n) if dist[v] < INF]
        heapq.heapify(heap)
        visited = [False] * n
        while heap:
            d, u = heapq.heappop(heap)
            if visited[u]:
                continue
            visited[u] = True
            for v, w in adj[u]:
                nd = d + w
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(heap, (nd, v))

    # process masks in increasing order of bits
    for mask in range(1, full + 1):
        # skip singletons (already initialized)
        if mask & (mask - 1):
            # combine submasks
            sub = (mask - 1) & mask
            while sub:
                other = mask ^ sub
                if other:
                    for v in range(n):
                        val = dp[sub][v] + dp[other][v]
                        if val < dp[mask][v]:
                            dp[mask][v] = val
                sub = (sub - 1) & mask
        # relax through edges
        dijkstra(mask)

    answer = min(dp[full])
    print(answer)

if __name__ == "__main__":
    solve()
```
