```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    s_indices = [int(x) - 1 for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # The condition (A^n)_{i,j} > 0 means there is a path of length at most n
    # between i and j in the graph defined by the edges in A.
    # Since A is symmetric and A_{i,i}=1, the graph has self-loops.
    # A path of length at most n exists if and only if i and j are in the same
    # connected component.
    # Thus, we need all nodes in S to be in the same connected component.
    # This is the Steiner Tree problem, but since we only need to connect
    # a set of nodes S, and the graph is undirected, this is equivalent to
    # finding a Minimum Spanning Tree that connects all nodes in S.
    
    # Dreyfus-Wagner algorithm for Steiner Tree
    # dp[mask][i] = min cost to connect nodes in mask such that i is in the component
    
    inf = float('inf')
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][s_indices[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dp[mask][i] using submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Dijkstra to update dp[mask][i] using edges
        pq = []
        for i in range(n):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
