```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem asks for the ability to map any target state to any other.
    # This is equivalent to saying all target states must belong to the same
    # connected component in the graph formed by the purchased edges.
    # We need to find a minimum cost set of edges such that all target states
    # are in the same connected component.
    # This is the Steiner Tree problem, but since we only care about connectivity
    # between the set of target states, we can use the property that we want
    # to connect all nodes in 'targets' using some auxiliary nodes.
    # This is equivalent to finding the Minimum Spanning Tree of a graph
    # where we contract all target nodes into a single super-node, but that's
    # not quite right. Actually, this is exactly the Steiner Tree problem
    # where the terminal set is 'targets'.
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm
    # or a simpler approach if applicable.
    
    # Steiner Tree DP: dp[mask][i] is the min cost to connect the subset of
    # targets represented by 'mask', with 'i' being a node in the tree.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    for mask in range(1, 1 << t):
        # Update dp[mask][i] using submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Dijkstra to update dp[mask][i]
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
