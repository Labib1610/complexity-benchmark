import sys
from heapq import heappush, heappop

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    critical = [int(next(it)) - 1 for _ in range(t)]
    
    # Build adjacency list for the graph
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # If t <= 1, no pairs to satisfy
    if t <= 1:
        print(0)
        return
    
    # Compute shortest paths between all pairs of critical nodes
    # Using Dijkstra from each critical node
    INF = float('inf')
    dist = [[INF] * n for _ in range(t)]
    
    for idx, start in enumerate(critical):
        d = dist[idx]
        d[start] = 0
        pq = [(0, start)]
        while pq:
            cur_d, u = heappop(pq)
            if cur_d > d[u]:
                continue
            for v, w in adj[u]:
                nd = cur_d + w
                if nd < d[v]:
                    d[v] = nd
                    heappush(pq, (nd, v))
    
    # Build complete graph on critical nodes with edge weights = shortest path
    # Then find MST of this complete graph
    # Use Prim's algorithm
    total_cost = 0
    in_mst = [False] * t
    min_edge = [INF] * t
    min_edge[0] = 0
    
    for _ in range(t):
        # Find minimum edge to add
        u = -1
        for v in range(t):
            if not in_mst[v] and (u == -1 or min_edge[v] < min_edge[u]):
                u = v
        
        in_mst[u] = True
        total_cost += min_edge[u]
        
        # Update min_edge for vertices not in MST
        for v in range(t):
            if not in_mst[v]:
                # Edge weight is shortest path from u to v
                w = dist[u][critical[v]]
                if w < min_edge[v]:
                    min_edge[v] = w
    
    print(total_cost)

if __name__ == "__main__":
    solve()
