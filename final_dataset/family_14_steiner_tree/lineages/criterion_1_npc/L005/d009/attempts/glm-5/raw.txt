import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    core = [int(data[idx+i]) for i in range(t)]
    idx += t
    
    # Build edge list
    edges = []
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    # We need to ensure: for every partition of core into (A, B) both nonempty,
    # there's no autonomous subset containing A but not B.
    # 
    # Key insight: Consider the graph G of purchased edges. An autonomous subset
    # is a union of connected components. The condition requires that no union
    # of components contains some but not all core nodes.
    #
    # This means: for every partition of core into (A, B) both nonempty,
    # A and B must be in the same connected component of G.
    # (If they were in different components, the union of components containing A
    #  would be autonomous and contain A but not B.)
    #
    # So all core nodes must be in the same connected component.
    # But that's not sufficient! We also need that no non-core component
    # can "merge" with a subset of core.
    #
    # Actually, let's think more carefully. The condition is:
    # For every partition (A, B) of core with both nonempty,
    # there is NO autonomous subset S with A ⊆ S and B ∩ S = ∅.
    #
    # An autonomous subset is a union of connected components.
    # So the condition is: for every partition (A,B), there's no union of
    # components that contains all of A and none of B.
    #
    # This is equivalent to: A and B are in the same component.
    # Because if some node of A and some node of B are in different components,
    # we could take the union of components containing A-nodes (which excludes B-nodes
    # in different components)... wait, that's not quite right either.
    #
    # Let me reconsider. If all core nodes are in the same component C,
    # then any autonomous subset either contains C (hence all core) or doesn't (hence no core).
    # So the condition IS just: all core nodes in same component.
    #
    # But wait, example 2: edges (1,2) and (2,3) connect cores 1,2,3.
    # Node 4 is isolated. Autonomous subsets: {}, {4}, {1,2,3}, {1,2,3,4}.
    # All contain all or none of core. Correct!
    #
    # So the answer is: minimum cost to connect all core nodes.
    # This is Steiner tree on core nodes, but we can use non-core nodes as Steiner points.
    #
    # With t <= 10 and n <= 100, we can do Steiner tree DP.
    
    # Steiner tree DP
    # dp[mask][v] = min cost of a tree connecting the core nodes in mask,
    #              where v is in the tree (v is a "root" / representative node)
    # mask is a subset of core nodes (bitmask of size t)
    
    # First, compute all-pairs shortest paths
    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            if dist[i][k] == INF:
                continue
            for j in range(1, n + 1):
                if dist[k][j] == INF:
                    continue
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    # Steiner tree DP
    # dp[mask][v]: minimum cost tree connecting core nodes in mask, with v in the tree
    full_mask = (1 << t) - 1
    dp = [[INF] * (n + 1) for _ in range(1 << t)]
    
    # Base case: single core node
    for i in range(t):
        mask = 1 << i
        v = core[i]
        dp[mask][v] = 0
    
    # Iterate over masks in increasing order of popcount
    for mask in range(1, 1 << t):
        # First, try merging submasks
        sub = (mask - 1) & mask
        while sub > 0:
            other = mask ^ sub
            for v in range(1, n + 1):
                if dp[sub][v] < INF and dp[other][v] < INF:
                    val = dp[sub][v] + dp[other][v]
                    if val < dp[mask][v]:
                        dp[mask][v] = val
            sub = (sub - 1) & mask
        
        # Then, relax using shortest paths (Steiner node optimization)
        # dp[mask][v] = min over u of (dp[mask][u] + dist[u][v])
        # This is like running Dijkstra from all nodes with finite dp[mask]
        # But with Floyd-Warshall already done, we can just do:
        for v in range(1, n + 1):
            best = dp[mask][v]
            for u in range(1, n + 1):
                if dp[mask][u] < INF and dist[u][v] < INF:
                    val = dp[mask][u] + dist[u][v]
                    if val < best:
                        best = val
            dp[mask][v] = best
    
    result = min(dp[full_mask][v] for v in range(1, n + 1))
    print(result)

solve()
