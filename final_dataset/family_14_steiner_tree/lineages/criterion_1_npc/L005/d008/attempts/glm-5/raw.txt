```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    idx = 0
    n = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    t = int(input_data[idx]); idx += 1
    
    regulated = []
    for _ in range(t):
        regulated.append(int(input_data[idx]) - 1)
        idx += 1
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(m):
        u = int(input_data[idx]) - 1; idx += 1
        v = int(input_data[idx]) - 1; idx += 1
        w = int(input_data[idx]); idx += 1
        edges.append((u, v, w))
    
    # For each regulated vertex, compute shortest path to all other vertices
    INF = float('inf')
    dist = [[INF] * n for _ in range(t)]
    
    for i in range(t):
        src = regulated[i]
        dist[i][src] = 0
        # Bellman-Ford
        for _ in range(n - 1):
            updated = False
            for u, v, w in edges:
                if dist[i][u] + w < dist[i][v]:
                    dist[i][v] = dist[i][u] + w
                    updated = True
                if dist[i][v] + w < dist[i][u]:
                    dist[i][u] = dist[i][v] + w
                    updated = True
            if not updated:
                break
    
    # Build complete graph on regulated vertices
    # Edge (i, j) has cost = shortest path from regulated[i] to regulated[j]
    # We need minimum spanning tree
    num_r = t
    mst_edges = []
    for i in range(num_r):
        for j in range(i + 1, num_r):
            cost = dist[i][regulated[j]]
            mst_edges.append((cost, i, j))
    
    mst_edges.sort()
    
    parent = list(range(num_r))
    
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        parent[px] = py
        return True
    
    total = 0
    edges_used = 0
    for cost, i, j in mst_edges:
        if union(i, j):
            total += cost
            edges_used += 1
            if edges_used == num_r - 1:
                break
    
    print(total)

solve()
```
