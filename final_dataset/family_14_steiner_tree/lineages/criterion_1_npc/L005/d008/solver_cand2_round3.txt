import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    regulated = [int(input_data[i + 3]) - 1 for i in range(t)]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        w = int(input_data[idx + 2])
        edges.append((u, v, w))
        idx += 3
        
    if t < 2:
        print(0)
        return

    # The condition "can reach state where i and j are odd" is equivalent to
    # saying i and j are in the same connected component in the graph formed
    # by the purchased edges.
    # We need all regulated ledgers to be in the same connected component.
    # This is a Steiner Tree problem where we need to connect a set of terminals.
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm.
    
    # Map regulated ledgers to 0..t-1
    reg_map = {node: i for i, node in enumerate(regulated)}
    
    # dp[mask][i] = min cost to connect terminals in mask, with i being in the component
    inf = float('inf')
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][regulated[i]] = 0
        
    # Precompute shortest paths for Dijkstra
    adj = [[] for _ in range(n)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    import heapq
    def dijkstra(start_node):
        dist = [inf] * n
        dist[start_node] = 0
        pq = [(0, start_node)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]: continue
            for v, w in adj[u]:
                if dist[v] > d + w:
                    dist[v] = d + w
                    heapq.heappush(pq, (dist[v], v))
        return dist

    for mask in range(1, 1 << t):
        # Submask split
        sub = (mask - 1) & mask
        while sub > 0:
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[sub][i] + dp[mask ^ sub][i])
            sub = (sub - 1) & mask
        
        # Relax with edges
        pq = []
        for i in range(n):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]: continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    print(min(dp[(1 << t) - 1]))

solve()
