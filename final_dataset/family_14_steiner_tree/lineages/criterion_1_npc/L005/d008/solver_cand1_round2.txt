```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem asks for the minimum cost to connect all target variables
    # in a graph where edges are the generators (x_u - x_v).
    # If we have a path between u and v, say (x_u - x_a) + (x_a - x_b) + ... + (x_k - x_v),
    # then (x_u - x_v) is in the closure.
    # Thus, we need all target variables to be in the same connected component.
    # This is a Steiner Tree problem, but since we only need to connect the 
    # target variables, and we can use any intermediate nodes (1 to n),
    # this is equivalent to finding the Minimum Spanning Tree of the 
    # Steiner Tree problem where the set of terminals is 'targets'.
    
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm
    # or simply realize that for t <= 10, we can use the Steiner Tree DP.
    
    # dp[mask][i] = min cost to connect the subset of targets represented by mask,
    # with node i being part of the tree.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    for mask in range(1, 1 << t):
        # Update dp[mask][i] using submasks
        for submask in range((mask - 1) & mask, 0, -1):
            submask &= mask
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
        
        # Dijkstra to update dp[mask][i] for all i
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
