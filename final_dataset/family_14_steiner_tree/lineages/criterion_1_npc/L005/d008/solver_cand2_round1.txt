```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = [int(x) - 1 for x in input_data[3:3+t]]
    is_critical = [False] * n
    for c in critical:
        is_critical[c] = True
        
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # We can represent each flag as a node in a graph.
    # A script (u, v) allows us to toggle u and v.
    # This is equivalent to adding an edge between u and v.
    # If we have a path between u and v, we can toggle u and v by toggling
    # all edges along the path.
    # We want to be able to toggle any pair of critical flags (c_i, c_j)
    # such that the path between them only uses non-critical flags as intermediates.
    # Actually, the problem is equivalent to:
    # We can treat all non-critical flags as a single "super-node" or
    # simply allow them to be toggled to facilitate connections.
    # Let's map each non-critical flag to a special node 'S'.
    # Any script (u, v) where u is critical and v is non-critical becomes (u, S).
    # Any script (u, v) where both are non-critical becomes (S, S) (useless).
    # Any script (u, v) where both are critical is (u, v).
    
    # We need to connect all critical flags to the 'S' node such that
    # any pair of critical flags can be toggled.
    # This is equivalent to forming a connected component of critical flags
    # that can reach 'S'.
    
    # Let's use Kruskal's or Prim's.
    # Nodes: 0 to t-1 (the critical flags), and node t (the 'S' node).
    # For each edge (u, v, w):
    # If u, v are both critical: edge between u and v with weight w.
    # If u is critical, v is non-critical: edge between u and t with weight w.
    # If u, v are both non-critical: ignore.
    
    adj = []
    crit_map = {c: i for i, c in enumerate(critical)}
    
    for u, v, w in edges:
        u_is_c = is_critical[u]
        v_is_c = is_critical[v]
        
        if u_is_c and v_is_c:
            adj.append((w, crit_map[u], crit_map[v]))
        elif u_is_c and not v_is_c:
            adj.append((w, crit_map[u], t))
        elif not u_is_c and v_is_c:
            adj.append((w, t, crit_map[v]))
            
    # We need to connect all critical nodes (0 to t-1) to the 'S' node (t).
    # This is a Minimum Spanning Tree problem on t+1 nodes.
    
    adj.sort()
    parent = list(range(t + 1))
    def find(i):
        if parent[i] == i: return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    mst_weight = 0
    edges_count = 0
    for w, u, v in adj:
        if union(u, v):
            mst_weight += w
            edges_count += 1
            
    print(mst_weight)

solve()
```
