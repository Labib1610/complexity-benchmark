import sys
from itertools import combinations

def find_min_cost_to_connect_focal_variables(n, m, t, focal_vars, constraints):
    if t <= 1:
        return 0
    
    # We need to find the minimum cost to connect all focal variables
    # This is essentially finding the minimum spanning tree in a graph where
    # nodes are the focal variables and edge weights are the minimum cost
    # to connect two components containing these variables
    
    # Use Union-Find to track connected components
    class UnionFind:
        def __init__(self, size):
            self.parent = list(range(size))
            self.rank = [0] * size
            
        def find(self, x):
            if self.parent[x] != x:
                self.parent[x] = self.find(self.parent[x])
            return self.parent[x]
        
        def union(self, x, y):
            px, py = self.find(x), self.find(y)
            if px == py:
                return False
            if self.rank[px] < self.rank[py]:
                px, py = py, px
            self.parent[py] = px
            if self.rank[px] == self.rank[py]:
                self.rank[px] += 1
            return True
    
    # Since t is small (≤ 10), we can use a different approach
    # Find shortest paths between all pairs of focal variables
    # Then find minimum spanning tree in the complete graph of focal variables
    
    # Initialize distance matrix for focal variables
    INF = float('inf')
    focal_indices = {var: i for i, var in enumerate(focal_vars)}
    dist = [[INF] * t for _ in range(t)]
    
    # Floyd-Warshall to find shortest paths between all nodes
    # Distance matrix for all variables
    full_dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        full_dist[i][i] = 0
    
    # Add constraints to the graph
    for u, v, w in constraints:
        full_dist[u][v] = min(full_dist[u][v], w)
        full_dist[v][u] = min(full_dist[v][u], w)
    
    # Floyd-Warshall algorithm
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if full_dist[i][k] + full_dist[k][j] < full_dist[i][j]:
                    full_dist[i][j] = full_dist[i][k] + full_dist[k][j]
    
    # Build distance matrix for focal variables only
    for i in range(t):
        for j in range(t):
            u, v = focal_vars[i], focal_vars[j]
            dist[i][j] = full_dist[u][v]
    
    # Now find MST of the focal variables using Prim's algorithm
    # Since t is small, this is efficient
    in_mst = [False] * t
    min_edge = [INF] * t
    min_edge[0] = 0
    total_cost = 0
    
    for _ in range(t):
        # Find the vertex with minimum edge weight not yet in MST
        u = -1
        for v in range(t):
            if not in_mst[v] and (u == -1 or min_edge[v] < min_edge[u]):
                u = v
        
        # Add the vertex to MST
        in_mst[u] = True
        total_cost += min_edge[u]
        
        # Update min_edge values for adjacent vertices
        for v in range(t):
            if not in_mst[v] and dist[u][v] < min_edge[v]:
                min_edge[v] = dist[u][v]
    
    return total_cost

# Read input
line = sys.stdin.readline().split()
n, m, t = int(line[0]), int(line[1]), int(line[2])

if t <= 1:
    print(0)
else:
    focal_vars = list(map(int, sys.stdin.readline().split()))
    
    constraints = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        constraints.append((u, v, w))
    
    result = find_min_cost_to_connect_focal_variables(n, m, t, focal_vars, constraints)
    print(result)
