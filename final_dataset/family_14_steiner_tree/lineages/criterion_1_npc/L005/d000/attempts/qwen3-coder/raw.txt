import sys
import heapq
from collections import defaultdict

def dijkstra(graph, start, nodes):
    dist = {node: float('inf') for node in nodes}
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
            
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
                
    return dist

def solve():
    n, m, t = map(int, input().split())
    delegated = list(map(int, input().split()))
    
    if t <= 1:
        print(0)
        return
    
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, input().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Compute shortest paths between all pairs of delegated languages
    # We'll use a modified Floyd-Warshall approach but only for delegated nodes
    # First, find shortest paths from each delegated node to all other nodes
    distances = {}
    for lang in delegated:
        distances[lang] = dijkstra(graph, lang, range(1, n+1))
    
    # Build a complete graph of delegated languages with edge weights as shortest paths
    # Then find MST of this graph
    edges = []
    for i in range(t):
        for j in range(i+1, t):
            u, v = delegated[i], delegated[j]
            weight = distances[u][v]
            edges.append((weight, u, v))
    
    # Sort edges by weight
    edges.sort()
    
    # Union-Find for Kruskal's algorithm
    parent = {lang: lang for lang in delegated}
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # Kruskal's algorithm
    total_cost = 0
    for weight, u, v in edges:
        if union(u, v):
            total_cost += weight
    
    print(total_cost)

solve()
