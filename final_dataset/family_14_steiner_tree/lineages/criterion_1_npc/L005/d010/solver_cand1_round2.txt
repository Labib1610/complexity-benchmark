```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = [int(input_data[i + 3]) - 1 for i in range(t)]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        w = int(input_data[idx + 2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The condition "all critical oscillators must have the same frequency"
    # is equivalent to saying that in the graph formed by the purchased links,
    # all critical oscillators must belong to the same connected component.
    # This is a Steiner Tree problem where the set of terminals is the set of critical oscillators.
    # Since t <= 10 and n <= 100, we can use the Dreyfus-Wagner algorithm.
    
    inf = float('inf')
    dist = [[inf] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
        
    # Floyd-Warshall for all-pairs shortest paths
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    # dp[mask][i] is the min cost to connect the subset of critical nodes represented by mask,
    # with node i being in the resulting component.
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][critical[i]] = 0
        
    for mask in range(1, 1 << t):
        # Submask splitting
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relax with edges
        for i in range(n):
            if dp[mask][i] == inf: continue
            for j in range(n):
                if dist[i][j] != inf:
                    dp[mask][j] = min(dp[mask][j], dp[mask][i] + dist[i][j])
                    
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
