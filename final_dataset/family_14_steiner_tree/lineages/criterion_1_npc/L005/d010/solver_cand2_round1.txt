```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    key_features = [int(x) - 1 for x in input_data[3:3+t]]
    
    rules = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        rules.append((u, v, w))
        idx += 3
        
    # A classification splits key features if there exist key features k1, k2
    # such that label(k1) != label(k2).
    # We want to ensure that for any such classification, at least one rule (u, v)
    # is violated (label(u) != label(v)).
    # This is equivalent to: if we pick a subset of rules, we want to ensure that
    # if label(u) == label(v) for all picked rules, then label(k1) == label(k2)
    # for all key features k1, k2.
    # Let the picked rules define an equivalence relation. If u ~ v, then label(u) == label(v).
    # The condition is: if all key features are in the same equivalence class,
    # then the classification is valid.
    # We want to force all key features to be in the same equivalence class.
    # This is a Steiner Tree-like problem on the graph of features.
    # We want to connect all key features into a single component using edges (rules).
    # This is exactly the Minimum Spanning Tree problem if we consider the key features
    # as terminals.
    
    # Since n is small (100), we can use the Dreyfus-Wagner algorithm or simply
    # realize that for a fixed partition of key features, we want to connect them.
    # Actually, the condition "Any classification that splits key features must violate a rule"
    # means that if we don't violate any rules, all key features must have the same label.
    # This means all key features must be in the same connected component of the graph
    # formed by the chosen rules.
    # Thus, we need to find a minimum cost set of edges such that all key features
    # are in the same connected component.
    
    # This is the Steiner Tree problem where terminals are the key features.
    # Since t <= 10, we can use the standard DP for Steiner Tree.
    
    adj = [[] for _ in range(n)]
    for u, v, w in rules:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # dp[mask][i] = min cost to connect the subset of key features 'mask'
    # such that feature 'i' is in the component.
    inf = float('inf')
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][key_features[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update via edges
        for i in range(n):
            for submask in range((mask - 1) & mask, 0, -1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
        
        # Dijkstra
        import heapq
        pq = []
        for i in range(n):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]: continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
