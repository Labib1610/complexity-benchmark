We can solve this by computing the shortest paths between all pairs of hubs, then finding the minimum Steiner tree connecting the critical facilities. Since \( t \leq 10 \), we can use a bitmask DP over subsets of critical facilities.

```python
import sys
import heapq

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    critical = list(map(int, input().split()))
    critical = [x - 1 for x in critical]  # 0-based

    # Build graph
    graph = [[] for _ in range(n)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        graph[u].append((v, w))
        graph[v].append((u, w))

    # If only one critical facility, cost is 0
    if t == 1:
        print(0)
        return

    # Compute all-pairs shortest paths using Dijkstra from each critical node
    INF = 10**18
    dist = [[INF] * n for _ in range(t)]
    for i, src in enumerate(critical):
        d = dist[i]
        d[src] = 0
        pq = [(0, src)]
        while pq:
            du, u = heapq.heappop(pq)
            if du != d[u]:
                continue
            for v, w in graph[u]:
                nd = du + w
                if nd < d[v]:
                    d[v] = nd
                    heapq.heappush(pq, (nd, v))

    # DP over subsets of critical nodes
    # dp[mask][i] = min cost to connect nodes in mask, ending at critical node i
    full_mask = (1 << t) - 1
    dp = [[INF] * t for _ in range(1 << t)]
    for i in range(t):
        dp[1 << i][i] = 0

    # For each subset, try to extend by adding a new critical node
    for mask in range(1 << t):
        for i in range(t):
            if dp[mask][i] == INF:
                continue
            # Try to add another critical node j
            for j in range(t):
                if mask & (1 << j):
                    continue
                new_mask = mask | (1 << j)
                # Connect i to j directly via shortest path
                cost = dp[mask][i] + dist[i][critical[j]]
                if cost < dp[new_mask][j]:
                    dp[new_mask][j] = cost
            # Also try to merge two subsets (Steiner tree style)
            # We can split mask into two non-empty subsets
            sub = (mask - 1) & mask
            while sub:
                other = mask ^ sub
                if sub < other:  # avoid duplicate work
                    for i in range(t):
                        if dp[sub][i] == INF:
                            continue
                        for j in range(t):
                            if dp[other][j] == INF:
                                continue
                            # Merge at node i or j
                            cost = dp[sub][i] + dp[other][j] + dist[i][critical[j]]
                            if cost < dp[mask][i]:
                                dp[mask][i] = cost
                            cost = dp[sub][i] + dp[other][j] + dist[j][critical[i]]
                            if cost < dp[mask][j]:
                                dp[mask][j] = cost
                sub = (sub - 1) & mask

    ans = min(dp[full_mask])
    print(ans)

if __name__ == "__main__":
    solve()
```
