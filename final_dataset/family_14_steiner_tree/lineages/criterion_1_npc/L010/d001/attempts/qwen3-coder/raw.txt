def find_mst(n, edges, required):
    # Union-Find data structure
    parent = list(range(n + 1))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # Sort edges by weight
    edges.sort(key=lambda x: x[2])
    
    # Kruskal's algorithm
    total_cost = 0
    required_set = set(required)
    connected_required = set()
    
    for u, v, w in edges:
        if union(u, v):
            total_cost += w
            # Check if either endpoint is required
            if u in required_set:
                connected_required.add(u)
            if v in required_set:
                connected_required.add(v)
            
            # If all required nodes are connected, we're done
            if len(connected_required) == len(required_set):
                # Check if all required nodes are in the same component
                root = find(required[0])
                if all(find(node) == root for node in required):
                    break
    
    return total_cost
