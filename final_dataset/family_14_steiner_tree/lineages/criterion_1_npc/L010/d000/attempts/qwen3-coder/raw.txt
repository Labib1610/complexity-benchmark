import sys
import heapq
from collections import defaultdict

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal(n, edges, required):
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # Sort edges by weight
    edges.sort(key=lambda x: x[2])
    
    cost = 0
    connected = len(required)
    
    for u, v, w in edges:
        if union(parent, rank, u, v):
            cost += w
            connected -= 1
            if connected == 1:
                break
                
    return cost

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    required = list(map(int, sys.stdin.readline().split()))
    
    # If there's only one required site, no connection needed
    if t <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u, v, w))
    
    # Use Kruskal's algorithm on the required nodes only
    result = kruskal(n, edges, required)
    print(result)

solve()
