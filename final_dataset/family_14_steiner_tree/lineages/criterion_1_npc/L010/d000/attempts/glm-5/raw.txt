```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    required = set()
    for _ in range(t):
        required.add(int(next(it)))
    
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        edges.append((w, u, v))
    
    if t <= 1:
        print(0)
        return
    
    edges.sort()
    
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    
    def union(x, y):
        rx, ry = find(x), find(y)
        if rx == ry:
            return False
        if rank[rx] < rank[ry]:
            parent[rx] = ry
        elif rank[rx] > rank[ry]:
            parent[ry] = rx
        else:
            parent[ry] = rx
            rank[rx] += 1
        return True
    
    total_cost = 0
    required_components = t
    
    for w, u, v in edges:
        if find(u) != find(v):
            root_u = find(u)
            root_v = find(v)
            count_u = 1 if root_u in required else 0
            count_v = 1 if root_v in required else 0
            
            if count_u + count_v < 2:
                union(u, v)
                total_cost += w
            else:
                union(u, v)
                total_cost += w
                required_components -= 1
                if required_components == 1:
                    break
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
