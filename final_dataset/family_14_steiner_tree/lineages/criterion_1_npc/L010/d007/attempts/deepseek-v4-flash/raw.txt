 We can model this as a minimum spanning tree problem on a graph where the priority points must be connected.  

The key insight:  
- Transpositions generate a group where two points are in the same orbit if they are connected in the graph formed by selected transpositions.  
- So we need to connect all priority points (possibly through other points) with minimum total cost.  
- This is exactly the **Steiner tree** problem, but since t ≤ 10, we can solve it with dynamic programming over subsets of priority points.

We’ll use the standard Dreyfus-Wagner algorithm for Steiner tree in graphs.

```python
import sys
import heapq

input = sys.stdin.readline

n, m, t = map(int, input().split())
priority = list(map(int, input().split()))
priority_set = set(priority)

# Build graph
graph = [[] for _ in range(n)]
for _ in range(m):
    u, v, w = map(int, input().split())
    u -= 1
    v -= 1
    graph[u].append((v, w))
    graph[v].append((u, w))

# If only one priority point, cost is 0
if t == 1:
    print(0)
    sys.exit()

# Map priority points to indices 0..t-1
idx = {p: i for i, p in enumerate(priority)}

INF = 10**18

# dp[mask][v] = min cost to connect priority points in mask, ending at vertex v
dp = [[INF] * n for _ in range(1 << t)]

# Initialize: for each priority point, cost 0 to be at that vertex
for i, p in enumerate(priority):
    dp[1 << i][p] = 0

# Dijkstra for each mask to compute shortest paths from that mask's terminals
for mask in range(1 << t):
    # Dijkstra
    dist = dp[mask][:]
    pq = [(dist[i], i) for i in range(n) if dist[i] < INF]
    heapq.heapify(pq)
    while pq:
        d, u = heapq.heappop(pq)
        if d != dist[u]:
            continue
        for v, w in graph[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(pq, (nd, v))
    dp[mask] = dist

# Combine subsets
for mask in range(1 << t):
    # Try splitting mask into two non-empty subsets
    sub = (mask - 1) & mask
    while sub:
        other = mask ^ sub
        if sub < other:  # avoid duplicate work
            for v in range(n):
                if dp[sub][v] + dp[other][v] < dp[mask][v]:
                    dp[mask][v] = dp[sub][v] + dp[other][v]
        sub = (sub - 1) & mask
    # After combining, run Dijkstra again to propagate
    dist = dp[mask][:]
    pq = [(dist[i], i) for i in range(n) if dist[i] < INF]
    heapq.heapify(pq)
    while pq:
        d, u = heapq.heappop(pq)
        if d != dist[u]:
            continue
        for v, w in graph[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(pq, (nd, v))
    dp[mask] = dist

# Answer is min over all vertices for full mask
full_mask = (1 << t) - 1
print(min(dp[full_mask]))
```
