import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "for every pair of target dimensions a and b, e_a - e_b is in the span"
    # is equivalent to saying that all target dimensions must belong to the same connected
    # component in the graph formed by the acquired edges.
    # We can treat the target dimensions as nodes that must be connected.
    # We can use Kruskal's or Prim's, but since we can use any intermediate nodes,
    # this is a Steiner Tree problem. However, with n <= 100 and t <= 10,
    # we can use the Dreyfus-Wagner algorithm or simply realize that we are looking for
    # a Minimum Spanning Tree in a graph where we contract all target nodes into one
    # or use the fact that we need to connect the set of target nodes.
    # Actually, the problem is equivalent to finding a Minimum Spanning Tree in a graph
    # where we want to connect all target nodes. This is exactly the Steiner Tree problem.
    # Given t is very small (t <= 10), we can use the Dreyfus-Wagner algorithm.
    
    # DP state: dp[mask][i] is the min cost to connect the subset of targets represented
    # by mask, with node i being part of the connected component.
    
    inf = float('inf')
    dist = [[inf] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
            
    # Floyd-Warshall to get all-pairs shortest paths
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    for mask in range(1, 1 << t):
        for i in range(1, n + 1):
            submask = (mask - 1) & mask
            while submask > 0:
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
                submask = (submask - 1) & mask
        
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[mask][j] + dist[j][i])
                
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
