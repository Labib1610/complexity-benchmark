```python
import sys
from itertools import combinations

def find_min_cost(n, m, t, S, edges):
    if t <= 1:
        return 0
    
    # Build graph with weights
    graph = [[float('inf')] * (n + 1) for _ in range(n + 1)]
    for u, v, w in edges:
        # Since we can use e_u - e_v or e_v - e_u, treat as undirected
        if w < graph[u][v]:
            graph[u][v] = w
            graph[v][u] = w
    
    # Floyd-Warshall to get all-pairs shortest paths
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if graph[i][k] + graph[k][j] < graph[i][j]:
                    graph[i][j] = graph[i][k] + graph[k][j]
    
    # For the target set S, we need to connect all pairs
    # This is equivalent to finding a minimum spanning tree on the complete graph
    # where edge weights are the shortest path distances between nodes in S
    S = list(S)
    t = len(S)
    
    # If t <= 1, cost is 0
    if t <= 1:
        return 0
    
    # Generate all possible edges between nodes in S with their shortest path weights
    edges_S = []
    for i in range(t):
        for j in range(i + 1, t):
            u = S[i]
            v = S[j]
            edges_S.append((graph[u][v], u, v))
    
    # Sort edges by weight
    edges_S.sort()
    
    # Union-Find for Kruskal's algorithm
    parent = list(range(n + 1))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # Kruskal's algorithm to find MST
    total_cost = 0
    edges_used = 0
    
    for w, u, v in edges_S:
        if union(u, v):
            total_cost += w
            edges_used += 1
            if edges_used == t - 1:
                break
    
    return total_cost

# Read input
line = sys.stdin.readline().split()
n, m, t = int(line[0]), int(line[1]), int(line[2])

if t > 0:
    S = set(map(int, sys.stdin.readline().split()))
else:
    S = set()

edges = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    edges.append((u, v, w))

# Solve and output
result = find_min_cost(n, m, t, S, edges)
print(result)
```
