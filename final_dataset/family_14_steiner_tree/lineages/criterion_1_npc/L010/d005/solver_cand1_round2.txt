```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(input_data[i + 3]) for i in range(t)]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem asks for the minimum cost to make all target variables equal.
    # This is equivalent to finding a Steiner Tree that connects all target variables.
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm.
    
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # dists[i][j] is the shortest path between i and j
    inf = float('inf')
    dists = [[inf] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dists[i][i] = 0
        pq = [(0, i)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dists[i][u]:
                continue
            for v, w in adj[u]:
                if dists[i][v] > d + w:
                    dists[i][v] = d + w
                    heapq.heappush(pq, (dists[i][v], v))
                    
    # dp[mask][i] is the min cost to connect the subset of targets represented by mask,
    # with i being a node in the tree.
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        for node in range(1, n + 1):
            dp[1 << i][node] = dists[targets[i]][node]
            
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relax edges
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
