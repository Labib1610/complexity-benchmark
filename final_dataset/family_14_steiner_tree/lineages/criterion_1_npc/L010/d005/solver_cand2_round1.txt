```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core_branches = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem asks for the minimum cost to make all core branches belong to the same connected component.
    # This is equivalent to the Steiner Tree problem where the set of terminals is the set of core branches.
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm.
    
    # dist[mask][i] is the min cost to connect the subset of core branches represented by mask,
    # with i being a node in the tree.
    
    inf = float('inf')
    num_masks = 1 << t
    dist = [[inf] * (n + 1) for _ in range(num_masks)]
    
    for i in range(t):
        dist[1 << i][core_branches[i]] = 0
        
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    for mask in range(1, num_masks):
        # Step 1: Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dist[mask][i] = min(dist[mask][i], dist[submask][i] + dist[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Step 2: Dijkstra to update distances for the current mask
        pq = []
        for i in range(1, n + 1):
            if dist[mask][i] != inf:
                heapq.heappush(pq, (dist[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[mask][u]:
                continue
            for v, w in adj[u]:
                if dist[mask][v] > d + w:
                    dist[mask][v] = d + w
                    heapq.heappush(pq, (dist[mask][v], v))
                    
    ans = min(dist[num_masks - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
