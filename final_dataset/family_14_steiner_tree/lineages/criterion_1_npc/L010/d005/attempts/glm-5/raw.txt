```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    if t <= 1:
        print(0)
        return
        
    edges.sort()
    
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    target_comp_count = t
    
    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i
        
    def union(i, j):
        nonlocal target_comp_count
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            if rank[root_i] < rank[root_j]:
                parent[root_i] = root_j
            elif rank[root_i] > rank[root_j]:
                parent[root_j] = root_i
            else:
                parent[root_j] = root_i
                rank[root_i] += 1
            return True
        return False

    total_cost = 0
    for w, u, v in edges:
        if find(u) != find(v):
            root_u = find(u)
            root_v = find(v)
            if union(u, v):
                total_cost += w
                if (root_u in targets or root_v in targets) and root_u != root_v:
                    target_comp_count -= 1
                    if target_comp_count == 1:
                        print(total_cost)
                        return

if __name__ == '__main__':
    solve()
```
