We can model this problem as finding the minimum cost to connect all the target indices together in a graph.

The available constraints form an undirected weighted graph. Activating constraints adds edges, and we want all target vertices to be in the same connected component.  

This is exactly the **Steiner tree problem** on a graph with a small number of terminals (t ≤ 10). Since t is small, we can solve it with dynamic programming over subsets of terminals.

Here’s the solution:

```python
import sys
import heapq

input = sys.stdin.readline

def solve():
    n, m, t = map(int, input().split())
    targets = list(map(int, input().split()))
    if t == 1:
        print(0)
        return

    INF = 10**18
    dist = [[INF]*n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    # Floyd-Warshall to get all-pairs shortest paths
    for k in range(n):
        for i in range(n):
            dik = dist[i][k]
            if dik == INF:
                continue
            for j in range(n):
                nd = dik + dist[k][j]
                if nd < dist[i][j]:
                    dist[i][j] = nd

    # Map target vertices to 0..t-1
    idx = {v: i for i, v in enumerate(targets)}

    # dp[mask][v] = min cost to connect terminals in mask with Steiner point v
    dp = [[INF]*n for _ in range(1<<t)]
    for i in range(t):
        for v in range(n):
            dp[1<<i][v] = dist[targets[i]][v]

    for mask in range(1, 1<<t):
        # split mask into two non-empty parts
        sub = (mask-1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid symmetric pairs
                for v in range(n):
                    dp[mask][v] = min(dp[mask][v], dp[sub][v] + dp[other][v])
            sub = (sub-1) & mask

        # relax via Dijkstra for current mask
        pq = [(dp[mask][v], v) for v in range(n) if dp[mask][v] < INF]
        heapq.heapify(pq)
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v in range(n):
                nd = d + dist[u][v]
                if nd < dp[mask][v]:
                    dp[mask][v] = nd
                    heapq.heappush(pq, (nd, v))

    ans = min(dp[(1<<t)-1][v] for v in range(n))
    print(ans)

if __name__ == "__main__":
    solve()
```
