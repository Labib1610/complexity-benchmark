We can model this as a minimum spanning tree problem on a graph where the target elements must be connected.  

The key observation:  
- We only need to connect the target elements into one component.  
- We can use any intermediate vertices, but we only care about the cheapest way to connect the target set.  
- This is exactly the **Steiner tree** problem, but since `t ≤ 10`, we can solve it with **bitmask DP** over subsets of targets.

Here’s the implementation:

```python
import sys
import heapq

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    targets = list(map(int, input().split()))
    targets = [x - 1 for x in targets]  # 0-index

    if t == 1:
        print(0)
        return

    # Build adjacency list
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append((v, w))
        adj[v].append((u, w))

    # Compute all-pairs shortest paths (Dijkstra from each target)
    INF = 10**18
    dist = [[INF] * n for _ in range(t)]
    for i, s in enumerate(targets):
        d = dist[i]
        d[s] = 0
        pq = [(0, s)]
        while pq:
            du, u = heapq.heappop(pq)
            if du != d[u]:
                continue
            for v, w in adj[u]:
                nd = du + w
                if nd < d[v]:
                    d[v] = nd
                    heapq.heappush(pq, (nd, v))

    # DP over subsets of targets
    # dp[mask][i] = min cost to connect targets in mask, ending at target i
    full_mask = 1 << t
    dp = [[INF] * t for _ in range(full_mask)]

    # Initialize: single target
    for i in range(t):
        dp[1 << i][i] = 0

    # Iterate over masks
    for mask in range(1, full_mask):
        # Try merging two submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid duplicate work
                for i in range(t):
                    if dp[sub][i] == INF:
                        continue
                    for j in range(t):
                        if dp[other][j] == INF:
                            continue
                        # connect i and j via shortest path
                        cost = dp[sub][i] + dp[other][j] + dist[i][targets[j]]
                        if cost < dp[mask][i]:
                            dp[mask][i] = cost
                        if cost < dp[mask][j]:
                            dp[mask][j] = cost
            sub = (sub - 1) & mask

        # Also try extending a single target by connecting to another target
        for i in range(t):
            if dp[mask][i] == INF:
                continue
            for j in range(t):
                if mask & (1 << j):
                    continue
                nmask = mask | (1 << j)
                cost = dp[mask][i] + dist[i][targets[j]]
                if cost < dp[nmask][j]:
                    dp[nmask][j] = cost

    ans = min(dp[full_mask - 1])
    print(ans)

if __name__ == "__main__":
    solve()
```
