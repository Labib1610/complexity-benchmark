```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical = [int(x) for x in input_data[3:3+t]]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem is equivalent to finding a Steiner Tree in a graph where
    # nodes are substances and edges are procedures.
    # Since t is small (up to 10), we use the Dreyfus-Wagner algorithm.
    
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # dp[mask][i] is the min cost to connect the subset of critical substances
    # represented by mask, with substance i being in the resulting vessel.
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][critical[i]] = 0
        
    for mask in range(1, 1 << t):
        # Submask updates
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                if dp[submask][i] != inf and dp[mask ^ submask][i] != inf:
                    dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Dijkstra updates
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
