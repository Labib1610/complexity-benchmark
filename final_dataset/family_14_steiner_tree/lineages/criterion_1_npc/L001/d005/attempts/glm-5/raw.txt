```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    targets = [int(next(it)) for _ in range(t)]
    if t <= 1:
        print(0)
        return
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        edges.append((u, v, w))
    edges.sort(key=lambda x: x[2])
    parent = list(range(n + 1))
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    def union(a, b):
        ra = find(a)
        rb = find(b)
        if ra == rb:
            return False
        parent[ra] = rb
        return True
    total = 0
    for u, v, w in edges:
        if union(u, v):
            total += w
            if find(targets[0]) == find(targets[1]):
                all_same = True
                for c in targets[2:]:
                    if find(c) != find(targets[0]):
                        all_same = False
                        break
                if all_same:
                    break
    print(total)

solve()
```
