import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core_banks = [int(input_data[i + 3]) for i in range(t)]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Steiner Tree problem on a graph with n <= 100 and t <= 10.
    # dp[mask][i] is the min cost to connect the subset of core banks represented by mask,
    # with i being a node in the tree.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][core_banks[i]] = 0
        
    # Dijkstra-like approach for Steiner Tree
    for mask in range(1, 1 << t):
        # Combine submasks
        for submask in range((mask - 1) & mask, 0, -1):
            submask = submask & mask
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
        
        # Relax edges using Dijkstra
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                pq.append((dp[mask][i], i))
        
        pq.sort()
        
        # Simple Dijkstra
        dist = list(dp[mask])
        visited = [False] * (n + 1)
        
        adj = [[] for _ in range(n + 1)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
            
        import heapq
        heap = [(d, i) for d, i in pq if d != inf]
        heapq.heapify(heap)
        
        while heap:
            d, u = heapq.heappop(heap)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > dp[mask][u] + w:
                    dp[mask][v] = dp[mask][u] + w
                    heapq.heappush(heap, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
