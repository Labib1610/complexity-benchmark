import sys
import heapq
from collections import defaultdict

def dijkstra(graph, start, end_set):
    if start in end_set:
        return 0
    
    dist = {}
    dist[start] = 0
    pq = [(0, start)]
    visited = set()
    
    while pq:
        d, u = heapq.heappop(pq)
        
        if u in visited:
            continue
            
        visited.add(u)
        
        if u in end_set:
            return d
            
        for v, w in graph[u]:
            if v not in visited:
                new_dist = d + w
                if v not in dist or new_dist < dist[v]:
                    dist[v] = new_dist
                    heapq.heappush(pq, (new_dist, v))
    
    return float('inf')

def solve():
    n, m, t = map(int, input().split())
    mandatory = list(map(int, input().split()))
    
    if t <= 1:
        print(0)
        return
    
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, input().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Create a complete graph of mandatory facilities with shortest path distances
    mandatory_graph = []
    
    for i in range(t):
        for j in range(i+1, t):
            start = mandatory[i]
            end_set = {mandatory[j]}
            
            distance = dijkstra(graph, start, end_set)
            mandatory_graph.append((distance, i, j))
    
    # Kruskal's algorithm for MST
    mandatory_graph.sort()
    
    parent = list(range(t))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    total_cost = 0
    edges_used = 0
    
    for cost, u, v in mandatory_graph:
        if union(u, v):
            total_cost += cost
            edges_used += 1
            if edges_used == t - 1:
                break
    
    print(total_cost)

solve()
