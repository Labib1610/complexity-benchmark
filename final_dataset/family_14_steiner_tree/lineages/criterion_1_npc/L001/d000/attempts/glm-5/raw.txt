import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    mandatory = []
    for _ in range(t):
        mandatory.append(int(data[idx]) - 1)
        idx += 1
    
    INF = float('inf')
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(data[idx]) - 1; idx += 1
        v = int(data[idx]) - 1; idx += 1
        w = int(data[idx]); idx += 1
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    if t <= 1:
        print(0)
        return
    
    # Dreyfus-Wagner algorithm for Steiner tree
    # dp[mask][v] = min cost of Steiner tree connecting mandatory facilities in mask,
    #               where v is a terminal of the tree (v is in the tree)
    full = 1 << t
    dp = [[INF] * n for _ in range(full)]
    
    for i in range(t):
        dp[1 << i][mandatory[i]] = 0
    
    for mask in range(1, full):
        # Combine submasks
        sub = (mask - 1) & mask
        while sub > 0:
            if sub < mask - sub:  # avoid duplicate work, process each pair once
                other = mask ^ sub
                dp_mask = dp[mask]
                dp_sub = dp[sub]
                dp_other = dp[other]
                for v in range(n):
                    val = dp_sub[v] + dp_other[v]
                    if val < dp_mask[v]:
                        dp_mask[v] = val
            sub = (sub - 1) & mask
        
        # Shortest path relaxation: for each v, dp[mask][v] = min over u of (dp[mask][u] + dist[u][v])
        # This is equivalent to: dp[mask][v] = min(dp[mask][v], min_u(dp[mask][u] + dist[u][v]))
        # We can do this by taking the element-wise min after adding dist
        dp_mask = dp[mask]
        best = [INF] * n
        for u in range(n):
            du = dp_mask[u]
            if du == INF:
                continue
            du_dist = dist[u]
            for v in range(n):
                val = du + du_dist[v]
                if val < best[v]:
                    best[v] = val
        for v in range(n):
            if best[v] < dp_mask[v]:
                dp_mask[v] = best[v]
    
    result = min(dp[full - 1])
    print(result)

solve()
