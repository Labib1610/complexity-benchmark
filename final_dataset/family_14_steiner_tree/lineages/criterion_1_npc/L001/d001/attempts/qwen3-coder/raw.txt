```python
import sys
import heapq
from collections import defaultdict

def dijkstra(graph, start, end_set):
    # graph is a dict of dicts: graph[u][v] = weight
    # Find shortest distance from start to any node in end_set
    dist = {}
    pq = [(0, start)]
    dist[start] = 0
    
    while pq:
        d, u = heapq.heappop(pq)
        if u in end_set:
            return d
        if d > dist.get(u, float('inf')):
            continue
            
        for v in graph[u]:
            alt = d + graph[u][v]
            if alt < dist.get(v, float('inf')):
                dist[v] = alt
                heapq.heappush(pq, (alt, v))
    
    return float('inf')

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    critical = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
        
    # Build graph
    graph = defaultdict(dict)
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        # Since synchronizers are bidirectional, we add both directions
        graph[u][v] = min(graph[u].get(v, float('inf')), w)
        graph[v][u] = min(graph[v].get(u, float('inf')), w)
    
    # We need to connect all critical registers
    # This is essentially finding the minimum spanning tree of the critical nodes
    # But we can use a different approach: 
    # Create a complete graph where edge weights are shortest paths between critical nodes
    # Then find MST of this complete graph
    
    # Compute pairwise distances between critical nodes
    k = len(critical)
    dist_matrix = [[float('inf')] * k for _ in range(k)]
    
    for i in range(k):
        for j in range(i+1, k):
            # Find shortest path between critical[i] and critical[j]
            # Using Dijkstra
            dist = dijkstra(graph, critical[i], {critical[j]})
            dist_matrix[i][j] = dist
            dist_matrix[j][i] = dist
    
    # Now we have a complete graph with distances between critical nodes
    # Find MST using Prim's algorithm
    in_mst = [False] * k
    min_edge = [float('inf')] * k
    min_edge[0] = 0
    total_cost = 0
    
    for _ in range(k):
        # Find vertex with minimum edge weight not yet in MST
        u = -1
        for v in range(k):
            if not in_mst[v] and (u == -1 or min_edge[v] < min_edge[u]):
                u = v
                
        in_mst[u] = True
        total_cost += min_edge[u]
        
        # Update minimum edge weights to vertices adjacent to u
        for v in range(k):
            if not in_mst[v] and dist_matrix[u][v] < min_edge[v]:
                min_edge[v] = dist_matrix[u][v]
                
    print(total_cost)

solve()
```
