We can model this as a **Steiner tree** problem, but since the number of terminals is small (t ≤ 10), we can solve it using **bitmask DP over subsets of terminals** combined with shortest paths.

Here’s the plan:

1. Run **Dijkstra** from each terminal to get the shortest distance from every terminal to every node.
2. Use **DP over subsets**:
   - `dp[mask][v]` = minimum cost to connect the terminals in `mask` and end at node `v`.
   - Initialize `dp[1 << i][terminal[i]] = 0`.
   - Combine subsets: for each mask, for each node, try merging two submasks.
   - Also relax using shortest paths: from `dp[mask][u]` to `dp[mask][v]` via `dist[u][v]`.
3. The answer is `min(dp[(1<<t)-1][v])` over all `v`.

```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]
    
    graph = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # If only one terminal, cost is 0
    if t == 1:
        print(0)
        return
    
    # Dijkstra from each terminal
    INF = 10**18
    dist = [[INF] * n for _ in range(t)]
    for i, src in enumerate(terminals):
        dist[i][src] = 0
        pq = [(0, src)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[i][u]:
                continue
            for v, w in graph[u]:
                nd = d + w
                if nd < dist[i][v]:
                    dist[i][v] = nd
                    heapq.heappush(pq, (nd, v))
    
    # DP over subsets
    full_mask = (1 << t) - 1
    dp = [[INF] * n for _ in range(1 << t)]
    for i in range(t):
        dp[1 << i][terminals[i]] = 0
    
    # Precompute all-pairs shortest distances between nodes using Floyd-Warshall or repeated Dijkstra
    # Since n <= 100, Floyd-Warshall is fine
    all_dist = [[INF] * n for _ in range(n)]
    for u in range(n):
        all_dist[u][u] = 0
        for v, w in graph[u]:
            all_dist[u][v] = min(all_dist[u][v], w)
    for k in range(n):
        for i in range(n):
            if all_dist[i][k] == INF:
                continue
            for j in range(n):
                if all_dist[k][j] == INF:
                    continue
                if all_dist[i][j] > all_dist[i][k] + all_dist[k][j]:
                    all_dist[i][j] = all_dist[i][k] + all_dist[k][j]
    
    # DP transitions
    for mask in range(1 << t):
        # Combine submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for u in range(n):
                    if dp[sub][u] == INF:
                        continue
                    for v in range(n):
                        if dp[other][v] == INF:
                            continue
                        if all_dist[u][v] == INF:
                            continue
                        cost = dp[sub][u] + dp[other][v] + all_dist[u][v]
                        if cost < dp[mask][u]:
                            dp[mask][u] = cost
                        if cost < dp[mask][v]:
                            dp[mask][v] = cost
            sub = (sub - 1) & mask
        
        # Relax via shortest paths
        for u in range(n):
            if dp[mask][u] == INF:
                continue
            for v in range(n):
                if all_dist[u][v] == INF:
                    continue
                nd = dp[mask][u] + all_dist[u][v]
                if nd < dp[mask][v]:
                    dp[mask][v] = nd
    
    ans = min(dp[full_mask])
    print(ans)

if __name__ == "__main__":
    solve()
```
