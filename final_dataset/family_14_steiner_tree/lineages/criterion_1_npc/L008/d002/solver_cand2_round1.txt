import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    terminals = [int(x) for x in input_data[3:3+t]]
    edges_data = input_data[3+t:]
    
    adj = [[] for _ in range(n + 1)]
    for i in range(0, len(edges_data), 3):
        u = int(edges_data[i])
        v = int(edges_data[i+1])
        w = int(edges_data[i+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Steiner Tree problem with small number of terminals (t <= 10)
    # DP state: dp[mask][v] is min cost to connect subset of terminals 'mask'
    # such that they are connected to node 'v'.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][terminals[i]] = 0
        
    for mask in range(1, 1 << t):
        # 1. Combine submasks
        for submask in range((mask - 1) & mask, 0, -1):
            submask2 = mask ^ submask
            for v in range(1, n + 1):
                if dp[submask][v] + dp[submask2][v] < dp[mask][v]:
                    dp[mask][v] = dp[submask][submask][v] + dp[submask2][v]
        
        # 2. Dijkstra to update costs for current mask
        pq = []
        for v in range(1, n + 1):
            if dp[mask][v] != inf:
                heapq.heappush(pq, (dp[mask][v], v))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][v] for v in range(1, n + 1))
    print(ans)

if __name__ == '__main__':
    solve()
