import sys
import heapq
from collections import defaultdict

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal_with_terminals(n, edges, terminals):
    # Sort edges by weight
    edges.sort(key=lambda x: x[2])
    
    # Try all possible ways to connect terminals
    # We'll use a modified Kruskal's algorithm
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # First, connect all terminals in the cheapest way
    total_cost = 0
    connected_terminals = [terminals[0]]
    unconnected_terminals = terminals[1:]
    
    while unconnected_terminals:
        min_cost = float('inf')
        best_edge = None
        connecting_terminal = None
        
        # Find the cheapest edge that connects a connected terminal to an unconnected one
        for u, v, w in edges:
            u_root = find(parent, u)
            v_root = find(parent, v)
            
            # Check if this edge connects a connected terminal to an unconnected one
            u_is_terminal = u in connected_terminals
            v_is_terminal = v in connected_terminals
            u_is_unconnected = u in unconnected_terminals
            v_is_unconnected = v in unconnected_terminals
            
            if (u_is_terminal and v_is_unconnected) or (v_is_terminal and u_is_unconnected):
                if w < min_cost:
                    min_cost = w
                    best_edge = (u, v, w)
                    connecting_terminal = u if v_is_unconnected else v
                    
        if best_edge:
            u, v, w = best_edge
            union(parent, rank, u, v)
            total_cost += w
            # Add the newly connected terminal
            if u in unconnected_terminals:
                unconnected_terminals.remove(u)
                connected_terminals.append(u)
            if v in unconnected_terminals:
                unconnected_terminals.remove(v)
                connected_terminals.append(v)
                
    return total_cost

def prim_for_terminals(n, edges, terminals):
    if len(terminals) <= 1:
        return 0
        
    # Build adjacency list
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Use Prim's algorithm to find minimum spanning tree that connects all terminals
    # We'll modify it to stop once all terminals are connected
    
    # Start with the first terminal
    visited = set()
    visited.add(terminals[0])
    
    # Priority queue: (weight, node)
    pq = []
    for v, w in graph[terminals[0]]:
        heapq.heappush(pq, (w, v))
    
    total_cost = 0
    terminals_set = set(terminals)
    terminals_visited = {terminals[0]}
    
    while pq and len(terminals_visited) < len(terminals):
        weight, node = heapq.heappop(pq)
        
        if node in visited:
            continue
            
        visited.add(node)
        if node in terminals_set:
            terminals_visited.add(node)
            
        total_cost += weight
        
        # Add edges from this node
        for v, w in graph[node]:
            if v not in visited:
                heapq.heappush(pq, (w, v))
    
    return total_cost

def solve():
    # Read input
    n, m, t = map(int, sys.stdin.readline().split())
    terminals = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u, v, w))
    
    # Use Prim's algorithm for minimum spanning tree connecting all terminals
    result = prim_for_terminals(n, edges, terminals)
    print(result)

solve()
