import sys
from itertools import combinations
from collections import defaultdict
import heapq

def dijkstra(graph, start, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    
    return dist

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    critical_sites = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u, v, w))
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Compute shortest paths between all pairs of critical sites
    # We'll use a dictionary to store these distances
    critical_distances = {}
    
    for site in critical_sites:
        dist = dijkstra(graph, site, n)
        critical_distances[site] = {}
        for other_site in critical_sites:
            if site != other_site:
                critical_distances[site][other_site] = dist[other_site]
    
    # Now we have a complete graph on critical sites with shortest path weights
    # We need to find the minimum spanning tree of this complete graph
    # But since t is small (<= 10), we can try all possible subsets of edges
    
    # Create all edges between critical sites with their shortest path weights
    critical_edges = []
    for i in range(t):
        for j in range(i+1, t):
            u, v = critical_sites[i], critical_sites[j]
            weight = critical_distances[u][v]
            critical_edges.append((weight, u, v))
    
    # Sort edges by weight for Kruskal's algorithm
    critical_edges.sort()
    
    # Union-Find data structure for Kruskal's algorithm
    parent = {site: site for site in critical_sites}
    rank = {site: 0 for site in critical_sites}
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    # Kruskal's algorithm to find MST
    total_cost = 0
    edges_used = 0
    
    for weight, u, v in critical_edges:
        if find(u) != find(v):
            union(u, v)
            total_cost += weight
            edges_used += 1
            if edges_used == t - 1:
                break
    
    print(total_cost)

solve()
