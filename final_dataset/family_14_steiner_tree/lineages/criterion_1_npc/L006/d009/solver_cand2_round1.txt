import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_vars = set()
    idx = 3
    for _ in range(t):
        critical_vars.add(int(input_data[idx]))
        idx += 1
        
    if t <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem asks for the minimum cost to connect all critical variables
    # in a single connected component using the available transition vectors.
    # This is equivalent to finding the Minimum Spanning Tree (MST) of the 
    # graph where nodes are variables and edges are transition vectors.
    # Since we can use non-critical variables as intermediaries, we treat all
    # variables as nodes in the graph.
    
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # Kruskal's algorithm
    edges.sort(key=lambda x: x[2])
    
    total_cost = 0
    edges_count = 0
    
    # We need to connect all critical variables.
    # This is equivalent to finding the MST of the graph and then 
    # pruning edges that are not part of the path between any two critical variables.
    # Actually, the problem is simpler: we need the critical variables to be in the 
    # same connected component. The minimum cost to connect a subset of nodes 
    # in a graph is the Steiner Tree problem, but here we can use all nodes.
    # The minimum cost to connect a set of nodes S is the MST of the graph 
    # restricted to the nodes that can reach S.
    # Since we can use any node, we just need to find the MST of the graph 
    # and then prune it to the minimal subtree containing all critical variables.
    
    # Standard Kruskal's to find MST
    mst_edges = []
    for u, v, w in edges:
        if union(u, v):
            mst_edges.append((u, v, w))
            
    # Now prune the MST to only include nodes/edges necessary to connect critical variables.
    # Build adjacency list for the MST
    adj = [[] for _ in range(n + 1)]
    for u, v, w in mst_edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Use DFS to find which edges are on the path between any two critical variables.
    # An edge is kept if the subtree below it contains at least one critical variable
    # AND the rest of the tree contains at least one critical variable.
    
    critical_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if i in critical_vars:
            critical_count[i] = 1
            
    def get_counts(u, p):
        for v, w in adj[u]:
            if v != p:
                critical_count[u] += get_counts(v, u)
        return critical_count[u]
    
    # Root the tree at an arbitrary critical variable
    root = next(iter(critical_vars))
    get_counts(root, -1)
    
    ans = 0
    for u, v, w in mst_edges:
        # Edge connects two components. Check if both components have critical variables.
        # In the rooted tree, one node is the child of the other.
        child = u if critical_count[u] < critical_count[v] else v
        if 0 < critical_count[child] < t:
            ans += w
            
    print(ans)

solve()
