```python
import sys
from heapq import heappush, heappop

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    targets = [int(next(it)) - 1 for _ in range(t)]
    
    # Build adjacency list for the graph
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Compute MST cost using Prim's algorithm
    # Start from first target
    in_mst = [False] * n
    min_cost = [float('inf')] * n
    
    # Start from first target
    start = targets[0]
    min_cost[start] = 0
    heap = [(0, start)]
    mst_cost = 0
    nodes_in_mst = 0
    
    while heap and nodes_in_mst < n:
        cost, u = heappop(heap)
        if in_mst[u]:
            continue
        in_mst[u] = True
        mst_cost += cost
        nodes_in_mst += 1
        
        for v, w in adj[u]:
            if not in_mst[v] and w < min_cost[v]:
                min_cost[v] = w
                heappush(heap, (w, v))
    
    print(mst_cost)

if __name__ == "__main__":
    solve()
```
