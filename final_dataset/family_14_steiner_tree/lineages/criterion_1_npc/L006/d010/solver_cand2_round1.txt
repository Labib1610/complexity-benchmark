```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_indices = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "x_a - x_b in I for all a, b in T" is equivalent to saying
    # that all nodes in T belong to the same connected component in the graph
    # formed by the selected edges.
    # This is a Steiner Tree problem where the terminals are the set T.
    # Since |T| <= 10, we can use the Dreyfus-Wagner algorithm.
    
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Precompute all-pairs shortest paths using Dijkstra
    dist = [[float('inf')] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
        pq = [(0, i)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[i][u]:
                continue
            for v, w in adj[u]:
                if dist[i][v] > d + w:
                    dist[i][v] = d + w
                    heapq.heappush(pq, (dist[i][v], v))
                    
    # dp[mask][i] is the min cost to connect the subset of terminals represented by mask,
    # with i being a node in the resulting tree.
    dp = [[float('inf')] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        for node in range(1, n + 1):
            dp[1 << i][node] = dist[target_indices[i]][node]
            
    for mask in range(1, 1 << t):
        # Split mask
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relax with edges
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != float('inf'):
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
