import sys
import heapq
from collections import defaultdict
import itertools

def dijkstra(graph, start, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
            
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
                
    return dist

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    primaries = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
        
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
        
    # Compute shortest paths from each primary currency
    distances = {}
    for p in primaries:
        distances[p] = dijkstra(graph, p, n)
        
    # Build complete graph of primary currencies with shortest path weights
    edges = []
    for i in range(t):
        for j in range(i + 1, t):
            u, v = primaries[i], primaries[j]
            weight = distances[u][v]
            edges.append((weight, u, v))
            
    # Kruskal's algorithm to find MST of primary currencies
    edges.sort()
    
    parent = {p: p for p in primaries}
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    total_cost = 0
    for weight, u, v in edges:
        if union(u, v):
            total_cost += weight
            
    print(total_cost)

solve()
