import sys
from itertools import combinations
import heapq

def dijkstra(graph, start, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    return dist

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    critical = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u, v, w))
    
    # For each subset of edges, check if it connects all critical nodes
    min_cost = float('inf')
    
    # Since t <= 10, we can try all 2^m subsets for small m
    # But let's think differently - we want the subgraph induced by selected edges 
    # to have all critical nodes in one connected component
    
    # Better approach: Try all possible ways to connect critical nodes
    # Minimum spanning tree variant on critical nodes
    
    # Create a complete graph on critical nodes where edge weight is 
    # shortest path distance in the original graph
    # Then find MST of this complete graph
    
    # Build adjacency list for original graph
    graph = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Compute shortest paths between all pairs of critical nodes
    # We'll compute shortest path from each critical node to all others
    distances = {}
    for i in critical:
        distances[i] = dijkstra(graph, i, n)
    
    # Now create a complete graph on critical nodes
    # Edge weight between i and j is distances[i][j]
    # Find minimum spanning tree in this graph
    
    # But we need to be careful - we want S_ij > 0, which means 
    # there must be a path (of any length) between i and j in the selected subgraph
    # In terms of graph connectivity, all critical nodes must be connected
    
    # So the problem reduces to: find minimum cost subset of edges such that
    # all critical nodes are connected in the subgraph
    
    # This is exactly the Steiner Tree problem on graphs, but since t<=10,
    # we can solve it via enumeration or dynamic programming
    
    # Let's use DP with bitmasks
    # dp[mask][i] = minimum cost to connect nodes in mask, rooted at node i
    
    INF = float('inf')
    
    # Special case: if there's only one critical node, cost is 0
    if t == 1:
        print(0)
        return
        
    # For small t, we can enumerate all possible subsets of edges
    # and check connectivity
    
    # Actually, let's model this properly:
    # We want to find minimum cost set of edges such that 
    # the subgraph induced by these edges connects all critical vertices
    
    # This is the Steiner tree problem. For small number of terminals (t<=10),
    # we can solve it with Held-Karp-like DP
    
    # But let's think even simpler: 
    # Since t is very small, we can:
    # 1. Precompute all-pairs shortest paths between critical nodes
    # 2. Then find MST on the complete graph formed by critical nodes
    
    # However, MST doesn't give us the right answer because we care about 
    # connectivity in the original graph, not metric completion
    
    # Correct approach:
    # 1. Precompute shortest paths between all pairs of critical nodes
    # 2. Create a complete graph where weight(i,j) = shortest_path(i,j)  
    # 3. Find minimum spanning tree in this graph
    # 4. Sum up the weights
    
    # But wait, that's still not right because the actual solution might use
    # intermediate non-critical nodes cleverly
    
    # Let's go back to the DP approach for Steiner tree:
    
    # State: dp[mask][last] = minimum cost to connect all nodes in mask, ending at 'last'
    # But actually, for undirected graph, it's just connectivity, so we don't need 'last'
    
    # Even simpler DP:
    # dp[mask] = minimum cost to make all nodes in mask connected
    # Transition: split mask into two parts and connect them optimally
    
    # But we also need to account for connecting through non-critical nodes
    
    # Best approach: 
    # Since t <= 10, we can solve Steiner tree exactly:
    # 1. All subsets of critical nodes: 2^t possibilities
    # 2. For each subset, compute minimum cost to connect them
    # 3. Use subset DP
    
    # dp[mask] = minimum cost to connect all nodes represented by mask
    # To transition: for each mask, try all ways to partition it into two non-empty subsets
    # Cost = dp[submask1] + dp[submask2] + cost to connect them
    
    # But how to compute "cost to connect two subsets"?
    # It's the minimum edge weight connecting any node from submask1 to any from submask2
    
    # Actually, let's precompute pairwise distances between critical nodes
    # Then solve the Steiner tree problem on this metric space
    
    # Precompute distances between critical nodes
    crit_dist = [[INF] * t for _ in range(t)]
    for i in range(t):
        for j in range(t):
            if i != j:
                crit_dist[i][j] = distances[critical[i]][critical[j]]
    
    # Now solve Steiner tree on metric space given by crit_dist
    # Using DP: dp[mask] = minimum cost to connect all nodes in mask
    
    dp = [INF] * (1 << t)
    
    # Base cases: single nodes cost 0 to "connect"
    for i in range(t):
        dp[1 << i] = 0
    
    # Fill DP table
    for mask in range(1, 1 << t):
        if bin(mask).count('1') <= 1:
            continue
            
        # Try all proper subsets
        submask = mask
        while submask:
            submask = (submask - 1) & mask
            if submask == 0 or submask == mask:
                continue
                
            complement = mask ^ submask
            if complement == 0:
                continue
                
            # Find minimum cost edge between submask and complement
            min_edge = INF
            for i in range(t):
                if not (submask & (1 << i)):
                    continue
                for j in range(t):
                    if not (complement & (1 << j)):
                        continue
                    min_edge = min(min_edge, crit_dist[i][j])
            
            if min_edge < INF:
                dp[mask] = min(dp[mask], dp[submask] + dp[complement] + min_edge)
    
    print(dp[(1 << t) - 1])

solve()
