import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    primary = [int(x) - 1 for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return
        
    INF = float('inf')
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
        
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
            
    for k in range(n):
        for i in range(n):
            if dist[i][k] == INF:
                continue
            for j in range(n):
                if dist[k][j] == INF:
                    continue
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    K = t
    dp = [[INF] * n for _ in range(1 << K)]
    
    for i in range(K):
        u = primary[i]
        dp[1 << i][u] = 0
        
    for mask in range(1, 1 << K):
        submask = (mask - 1) & mask
        while submask > 0:
            complement = mask ^ submask
            for v in range(n):
                if dp[submask][v] < INF:
                    for u in range(n):
                        if dp[complement][u] < INF:
                            cost = dp[submask][v] + dp[complement][u] + dist[v][u]
                            if cost < dp[mask][u]:
                                dp[mask][u] = cost
            submask = (submask - 1) & mask
            
    print(min(dp[(1 << K) - 1]))

if __name__ == '__main__':
    solve()
