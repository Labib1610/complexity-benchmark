```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    terminals = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Steiner Tree problem on a small graph (n <= 100)
    # dp[mask][i] is the min cost to connect terminals in mask, with i as root
    # mask is a bitmask of length t
    
    num_masks = 1 << t
    dp = [[float('inf')] * (n + 1) for _ in range(num_masks)]
    
    for i in range(t):
        dp[1 << i][terminals[i]] = 0
        
    for mask in range(1, num_masks):
        # Update dp[mask][i] using submasks
        for i in range(1, n + 1):
            submask = (mask - 1) & mask
            while submask > 0:
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
                submask = (submask - 1) & mask
        
        # Dijkstra to update dp[mask][i] using edges
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != float('inf'):
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[num_masks - 1][i] for i in range(1, n + 1))
    print(ans)

if __name__ == '__main__':
    solve()
```
