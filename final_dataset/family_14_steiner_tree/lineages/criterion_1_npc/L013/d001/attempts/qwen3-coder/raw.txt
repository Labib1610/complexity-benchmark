```python
import sys
import heapq
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    core_controllers = list(map(int, data[idx:idx+t]))
    idx += t
    
    # Convert to 0-based indexing
    core_controllers = [x - 1 for x in core_controllers]
    
    # Build graph
    graph = [[] for _ in range(n)]
    edges = []
    
    for _ in range(m):
        u = int(data[idx]) - 1
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        w = int(data[idx])
        idx += 1
        graph[u].append((v, w))
        graph[v].append((u, w))
        edges.append((w, u, v))
    
    # If there's only one core controller, no connections needed
    if t <= 1:
        print(0)
        return
    
    # Function to compute minimum distance from a node to all others using Dijkstra
    def dijkstra(start):
        dist = [float('inf')] * n
        dist[start] = 0
        pq = [(0, start)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue
            for v, w in graph[u]:
                if dist[u] + w < dist[v]:
                    dist[v] = dist[u] + w
                    heapq.heappush(pq, (dist[v], v))
        return dist
    
    # Compute shortest paths from each core controller
    distances = []
    for core in core_controllers:
        distances.append(dijkstra(core))
    
    # Create a complete graph among core controllers with shortest path weights
    # Then find MST of this complete graph
    # We'll use Kruskal's algorithm
    
    # Generate all edges between core controllers with their shortest path weights
    core_edges = []
    for i in range(t):
        for j in range(i+1, t):
            u = core_controllers[i]
            v = core_controllers[j]
            weight = distances[i][v]  # Shortest path from u to v
            core_edges.append((weight, i, j))
    
    # Sort edges by weight
    core_edges.sort()
    
    # Union-Find for Kruskal's algorithm
    parent = list(range(t))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # Kruskal's algorithm to find MST
    total_power = 0
    edges_used = 0
    
    for weight, u, v in core_edges:
        if union(u, v):
            total_power += weight
            edges_used += 1
            if edges_used == t - 1:
                break
    
    print(total_power)

if __name__ == "__main__":
    main()
```
