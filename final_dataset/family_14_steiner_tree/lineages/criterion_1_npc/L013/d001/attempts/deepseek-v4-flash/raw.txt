We can model this as a **Steiner tree** problem, but since the number of terminals is small (t ≤ 10), we can solve it with **bitmask DP over subsets of terminals** using shortest paths between all pairs of nodes.

We’ll:

1. Run Floyd–Warshall to get shortest distances between all nodes.
2. Use DP where `dp[mask][v]` = minimum cost to connect the terminals in `mask` with a tree rooted at node `v`.
3. Combine subsets and use shortest paths to merge.

Here’s the implementation:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]

    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            if dist[i][k] == INF:
                continue
            for j in range(n):
                if dist[k][j] == INF:
                    continue
                nd = dist[i][k] + dist[k][j]
                if nd < dist[i][j]:
                    dist[i][j] = nd

    # DP over subsets of terminals
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]

    # Base: single terminal
    for i in range(t):
        term = terminals[i]
        for v in range(n):
            dp[1 << i][v] = dist[term][v]

    # Combine subsets
    for mask in range(1, full_mask):
        # Try splitting mask into two non-empty subsets
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid duplicate work
                for v in range(n):
                    dp[mask][v] = min(dp[mask][v], dp[sub][v] + dp[other][v])
            sub = (sub - 1) & mask

        # Relax via shortest paths
        for v in range(n):
            if dp[mask][v] == INF:
                continue
            for u in range(n):
                if dist[v][u] == INF:
                    continue
                nd = dp[mask][v] + dist[v][u]
                if nd < dp[mask][u]:
                    dp[mask][u] = nd

    ans = min(dp[full_mask - 1][v] for v in range(n))
    print(ans)

if __name__ == "__main__":
    solve()
```
