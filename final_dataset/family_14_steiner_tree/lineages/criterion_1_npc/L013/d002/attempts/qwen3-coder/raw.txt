import sys
import heapq
from collections import defaultdict

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1

def kruskal(n, edges, terminals):
    if len(terminals) <= 1:
        return 0
    
    # Sort edges by weight
    edges.sort(key=lambda x: x[2])
    
    # Initialize DSU
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # Connect terminals incrementally
    connected = 1
    total_cost = 0
    
    for u, v, w in edges:
        if find(parent, u) != find(parent, v):
            union(parent, rank, u, v)
            # Check if this union connects new terminals
            terminal_groups = set()
            for t in terminals:
                terminal_groups.add(find(parent, t))
            if len(terminal_groups) < connected:
                connected = len(terminal_groups)
                total_cost += w
            elif len(terminal_groups) == 1:
                break
    
    return total_cost

def main():
    input_lines = sys.stdin.read().splitlines()
    line_idx = 0
    
    # Read n, m, t
    n, m, t = map(int, input_lines[line_idx].split())
    line_idx += 1
    
    # Read primary workstations
    terminals = list(map(int, input_lines[line_idx].split()))
    line_idx += 1
    
    # Read edges
    edges = []
    for _ in range(m):
        u, v, w = map(int, input_lines[line_idx].split())
        edges.append((u, v, w))
        line_idx += 1
    
    # Use Kruskal's algorithm to find MST that connects all terminals
    # This is essentially the Steiner tree problem, but we can solve it
    # by finding the MST of the subgraph induced by the terminals and
    # the nodes needed to connect them
    
    # For small number of terminals (t <= 10), we can try all possible
    # combinations, but a simpler approach is to run Kruskal and stop
    # when all terminals are connected
    
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # Sort edges by weight
    edges.sort(key=lambda x: x[2])
    
    total_cost = 0
    connected_terminals = set()
    connected_terminals.add(terminals[0])
    
    while len(connected_terminals) < len(terminals):
        for i, (u, v, w) in enumerate(edges):
            # Check if this edge connects a new terminal
            root_u = find(parent, u)
            root_v = find(parent, v)
            
            # If they're already connected, skip
            if root_u == root_v:
                continue
                
            # Check if this connects a new terminal
            new_terminals = set()
            if u in terminals and u not in connected_terminals:
                new_terminals.add(u)
            if v in terminals and v not in connected_terminals:
                new_terminals.add(v)
                
            # If one endpoint is already connected to terminals
            u_connected = False
            v_connected = False
            for t in terminals:
                if find(parent, t) == root_u:
                    u_connected = True
                if find(parent, t) == root_v:
                    v_connected = True
                    
            if u_connected and not v_connected and v in terminals:
                new_terminals.add(v)
            if v_connected and not u_connected and u in terminals:
                new_terminals.add(u)
                
            if new_terminals:
                union(parent, rank, u, v)
                connected_terminals.update(new_terminals)
                total_cost += w
                # Mark this edge as used
                edges[i] = (u, v, float('inf'))
                break
        else:
            # If no edge was found, all terminals must be connected
            break
    
    print(total_cost)

if __name__ == "__main__":
    main()
