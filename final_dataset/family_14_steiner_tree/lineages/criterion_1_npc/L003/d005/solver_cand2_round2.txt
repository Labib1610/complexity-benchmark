```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) - 1 for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "impossible to divide into two groups A and B with targets in both"
    # is equivalent to saying that all target corporations must be in the same connected component.
    # We want to find the minimum cost to make all target corporations connected.
    # This is a Steiner Tree problem, but with n=100 and t=10, we can use dynamic programming.
    # Let dp[mask][i] be the min cost to connect the subset of targets represented by 'mask',
    # with node 'i' being in the connected component.
    
    inf = float('inf')
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dp[mask][i] using edges
        for _ in range(n):
            changed = False
            for u, v, w in edges:
                if dp[mask][u] + w < dp[mask][v]:
                    dp[mask][v] = dp[mask][u] + w
                    changed = True
                if dp[mask][v] + w < dp[mask][u]:
                    dp[mask][u] = dp[mask][v] + w
                    changed = True
            if not changed:
                break
        
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                if dp[submask][i] + dp[mask ^ submask][i] < dp[mask][i]:
                    dp[mask][i] = dp[submask][i] + dp[mask ^ submask][i]
            submask = (submask - 1) & mask
            
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
