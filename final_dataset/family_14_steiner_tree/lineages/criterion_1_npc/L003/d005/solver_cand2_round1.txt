```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) - 1 for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem asks for the minimum cost to make all target nodes belong to the same connected component.
    # This is equivalent to the Steiner Tree problem on the graph.
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm.
    
    # dist[mask][i] is the min cost to connect the subset of targets represented by mask,
    # with i being one of the nodes in the resulting tree.
    inf = float('inf')
    dist = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dist[1 << i][targets[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dist[mask][i] using edges (Dijkstra)
        pq = []
        for i in range(n):
            if dist[mask][i] != inf:
                heapq.heappush(pq, (dist[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[mask][u]:
                continue
            for u_edge, v_edge, w in edges:
                if u_edge == u:
                    if dist[mask][v_edge] > d + w:
                        dist[mask][v_edge] = d + w
                        heapq.heappush(pq, (dist[mask][v_edge], v_edge))
                elif v_edge == u:
                    if dist[mask][u_edge] > d + w:
                        dist[mask][u_edge] = d + w
                        heapq.heappush(pq, (dist[mask][u_edge], u_edge))
        
        # Update dist[mask][i] using submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dist[mask][i] = min(dist[mask][i], dist[submask][i] + dist[mask ^ submask][i])
            submask = (submask - 1) & mask
            
    ans = min(dist[(1 << t) - 1])
    print(ans)

solve()
```
