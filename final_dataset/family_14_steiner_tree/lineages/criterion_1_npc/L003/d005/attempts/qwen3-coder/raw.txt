import sys
from itertools import combinations

def find_min_cost_to_connect(n, m, target_set, candidates):
    # If there's only one target, no connections needed
    if len(target_set) <= 1:
        return 0
    
    # Create a graph with edges representing candidate vectors
    # We'll use Union-Find or MST approach to find minimum cost to connect all targets
    target_list = list(target_set)
    target_indices = set(target_list)
    
    # Use a modified Kruskal's algorithm to find minimum spanning tree for target nodes
    # But we need to find minimum cost to connect all targets through intermediate nodes
    
    # Build a complete graph of target nodes where edge weights are shortest paths
    # between them using candidate vectors as edges
    
    # First, let's build adjacency information
    adj = {}
    for i in range(1, n+1):
        adj[i] = []
        
    for u, v, w in candidates:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Floyd-Warshall to find shortest paths between all pairs
    INF = float('inf')
    dist = [[INF] * (n+1) for _ in range(n+1)]
    
    # Initialize distances
    for i in range(1, n+1):
        dist[i][i] = 0
        
    for u, v, w in candidates:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
    
    # Floyd-Warshall
    for k in range(1, n+1):
        for i in range(1, n+1):
            for j in range(1, n+1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    # Now we have shortest distances between all pairs
    # We want to connect all targets with minimum cost
    # This is a Steiner Tree problem, but for small t (t <= 10), we can try all combinations
    
    # For t <= 10, we can solve this optimally by trying all possible ways to connect
    # But let's implement a simpler approach: find MST on the complete graph of targets
    # where edge weight is the shortest path distance
    
    # Extract target-to-target distances
    target_dist = {}
    for i in range(len(target_list)):
        for j in range(i+1, len(target_list)):
            u, v = target_list[i], target_list[j]
            target_dist[(u, v)] = dist[u][v]
            target_dist[(v, u)] = dist[v][u]
    
    # Now find MST of target nodes using these distances
    # Prim's algorithm
    if len(target_list) <= 1:
        return 0
    
    # Use Kruskal's algorithm for MST
    edges = []
    for i in range(len(target_list)):
        for j in range(i+1, len(target_list)):
            u, v = target_list[i], target_list[j]
            edges.append((dist[u][v], u, v))
    
    edges.sort()
    
    # Union-Find data structure
    parent = {}
    rank = {}
    
    def find(x):
        if x not in parent:
            parent[x] = x
            rank[x] = 0
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    total_cost = 0
    edges_used = 0
    
    for cost, u, v in edges:
        if union(u, v):
            total_cost += cost
            edges_used += 1
            if edges_used == len(target_list) - 1:
                break
    
    return total_cost

# Read input
line = sys.stdin.readline().strip()
n, m, t = map(int, line.split())
target_set = set(map(int, sys.stdin.readline().strip().split()))

candidates = []
for _ in range(m):
    line = sys.stdin.readline().strip()
    u, v, w = map(int, line.split())
    candidates.append((u, v, w))

result = find_min_cost_to_connect(n, m, target_set, candidates)
print(result)
