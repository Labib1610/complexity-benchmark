import sys
from itertools import combinations
from collections import defaultdict
import heapq

def dijkstra(graph, start, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    
    return dist

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    focal = list(map(int, data[idx:idx+t]))
    idx += t
    
    graph = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Compute shortest paths from each focal node
    dists = {}
    for f in focal:
        dists[f] = dijkstra(graph, f, n)
    
    # For each subset of focal nodes, compute the minimum cost to connect them
    # We'll use a DP approach on subsets of focal nodes
    
    # But since t <= 10, we can consider all possible ways to connect them
    # The optimal way is to make them all connected, which means making the subgraph induced by them connected
    
    # Let's find the minimum spanning tree cost among the focal nodes using the pairwise distances
    
    if t <= 1:
        print(0)
        return
    
    # Create a complete graph among focal nodes with weights as shortest path distances
    edges = []
    for i in range(t):
        for j in range(i+1, t):
            u = focal[i]
            v = focal[j]
            w = dists[u][v]  # shortest path distance
            edges.append((w, u, v))
    
    # Now find MST of these focal nodes
    parent = list(range(n + 1))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        parent[px] = py
        return True
    
    edges.sort()
    total_cost = 0
    edge_count = 0
    
    for w, u, v in edges:
        if union(u, v):
            total_cost += w
            edge_count += 1
            if edge_count == t - 1:
                break
                
    print(total_cost)

if __name__ == "__main__":
    main()
