 We can solve this by building a graph and then finding the minimum spanning tree that connects all primary facilities.  
Since non-primary facilities can be used as intermediate nodes, we can first compute the shortest paths between all pairs of primary facilities, then run a minimum spanning tree on the primary nodes only.

Here’s the Python solution:

```python
import sys
import heapq

input = sys.stdin.readline

def solve():
    n, m, t = map(int, input().split())
    primaries = list(map(int, input().split()))
    
    # Build adjacency list
    graph = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    # Dijkstra from each primary to get shortest distances to all nodes
    INF = 10**18
    dist_matrix = {}
    for p in primaries:
        dist = [INF] * (n + 1)
        dist[p] = 0
        pq = [(0, p)]
        while pq:
            d, u = heapq.heappop(pq)
            if d != dist[u]:
                continue
            for v, w in graph[u]:
                nd = d + w
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(pq, (nd, v))
        dist_matrix[p] = dist
    
    # Build complete graph among primaries with shortest distances
    edges = []
    for i in range(t):
        for j in range(i + 1, t):
            u = primaries[i]
            v = primaries[j]
            w = dist_matrix[u][v]
            edges.append((w, u, v))
    
    # Kruskal's MST on primaries
    edges.sort()
    parent = {p: p for p in primaries}
    rank = {p: 0 for p in primaries}
    
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    
    def union(x, y):
        rx, ry = find(x), find(y)
        if rx == ry:
            return False
        if rank[rx] < rank[ry]:
            parent[rx] = ry
        elif rank[rx] > rank[ry]:
            parent[ry] = rx
        else:
            parent[ry] = rx
            rank[rx] += 1
        return True
    
    total = 0
    used = 0
    for w, u, v in edges:
        if union(u, v):
            total += w
            used += 1
            if used == t - 1:
                break
    
    print(total)

if __name__ == "__main__":
    solve()
```
