```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_keys = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to connect all nodes in T into a single connected component
    # using any nodes 1..n as intermediate nodes.
    # This is a Steiner Tree problem. Since t is small (up to 10),
    # we can use the Dreyfus-Wagner algorithm or a simpler approach.
    # Given the constraints and the nature of XOR, this is equivalent to
    # finding a Minimum Spanning Tree in a graph where we want to connect
    # all nodes in T.
    
    # Since we can use any node as a Steiner point, we first compute
    # all-pairs shortest paths between nodes in T.
    # Actually, since we want to connect all nodes in T, we can use
    # the Steiner Tree approach. With t <= 10, we can use DP.
    
    dist = [[float('inf')] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
        pq = [(0, i)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[i][u]:
                continue
            for v, w in adj[u]:
                if dist[i][v] > d + w:
                    dist[i][v] = d + w
                    heapq.heappush(pq, (dist[i][v], v))
                    
    # dp[mask][i] is the min cost to connect the subset of T represented by mask,
    # with i being a node in the tree.
    dp = [[float('inf')] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][target_keys[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dp[mask][i] using shortest paths
        for i in range(1, n + 1):
            for submask in range((mask - 1) & mask, 0, -1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
        
        # Update dp[mask][i] using edges
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != float('inf'):
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
