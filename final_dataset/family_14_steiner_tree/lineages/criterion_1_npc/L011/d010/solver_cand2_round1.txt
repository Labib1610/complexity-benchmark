import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t_count = int(input_data[2])
    
    target_set = set()
    idx = 3
    for _ in range(t_count):
        target_set.add(int(input_data[idx]))
        idx += 1
        
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem reduces to finding a Steiner Tree-like structure.
    # The conditions imply that S must be a connected component in the graph
    # formed by edges in C, and |C| = |S| - 1 means C is a spanning tree of S.
    # The partitioning condition: for every k in S \ T, removing k from the tree
    # must leave components that each contain at least one target node.
    # This is equivalent to saying that no node in S \ T is a "leaf" in the 
    # Steiner tree that doesn't contain a target node, and more specifically,
    # every node in S \ T must have degree >= 2 in the tree.
    # This means all nodes in S \ T must be internal nodes of the Steiner tree.
    
    # This is exactly the Steiner Tree problem where we must include all nodes
    # in T, and we can include any subset of nodes from {1..n} \ T as long as
    # they are not leaves.
    
    # Given the constraints (n=100, t=10), we can use the Dreyfus-Wagner algorithm.
    # dp[mask][i] = min cost to connect the subset of targets in 'mask' 
    # such that node 'i' is in the connected component.
    
    inf = float('inf')
    dist = [[inf] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
    for u, v, w in edges:
        dist[u][v] = dist[v][u] = min(dist[u][v], w)
        
    # Floyd-Warshall
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    targets = list(target_set)
    num_t = len(targets)
    dp = [[inf] * (n + 1) for _ in range(1 << num_t)]
    
    for i in range(num_t):
        dp[1 << i][targets[i]] = 0
        
    for mask in range(1, 1 << num_t):
        for i in range(1, n + 1):
            submask = (mask - 1) & mask
            while submask > 0:
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
                submask = (submask - 1) & mask
        
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[mask][j] + dist[j][i])
                
    ans = min(dp[(1 << num_t) - 1])
    print(ans)

solve()
