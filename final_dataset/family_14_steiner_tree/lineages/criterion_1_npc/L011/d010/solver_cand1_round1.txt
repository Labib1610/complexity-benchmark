```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t_count = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t_count]]
    edges = []
    idx = 3 + t_count
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u - 1, v - 1, w))
        idx += 3
        
    # The problem describes a Steiner Tree problem variant.
    # Condition 1 & 2 imply that the selected edges form a tree connecting all targets.
    # Condition 3 (Criticality) implies that every non-target node in the tree 
    # must have a degree of at least 3 (if it had degree 2, removing it would 
    # split the tree into two components, and if one component didn't contain 
    # a target, it would violate the condition).
    # Actually, for a Steiner Tree, the condition that every non-target node 
    # has degree >= 3 is the standard condition for a "full" Steiner tree.
    # The problem asks for the minimum cost to satisfy these conditions.
    # This is equivalent to finding the minimum cost Steiner Tree.
    
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm.
    
    inf = float('inf')
    dist = [[inf] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
        
    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    # dp[mask][i] is the min cost to connect the subset of targets represented by mask,
    # with i being a node in the tree.
    dp = [[inf] * n for _ in range(1 << t_count)]
    
    for i in range(t_count):
        dp[1 << i][targets[i] - 1] = 0
        
    for mask in range(1, 1 << t_count):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relax with edges
        for i in range(n):
            for j in range(n):
                if dp[mask][j] + dist[j][i] < dp[mask][i]:
                    dp[mask][i] = dp[mask][j] + dist[j][i]
                    
    print(min(dp[(1 << t_count) - 1]))

solve()
```
