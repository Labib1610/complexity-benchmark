```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The problem asks for a Steiner Tree where every non-target node has degree >= 2.
    # This is equivalent to finding a tree that spans all target nodes such that
    # no leaf node is a non-target node.
    # This is the Steiner Tree problem with the constraint that all leaves are targets.
    # Since n is small (100), we can use the Dreyfus-Wagner algorithm or a variation.
    # However, the constraint "non-target nodes must have degree >= 2" means
    # we are looking for a Steiner Tree where all leaves are in the target set.
    
    # Let dp[mask][i] be the min cost to connect the subset of targets 'mask'
    # such that the tree is rooted at i.
    # To handle the degree constraint:
    # If i is a target, it can be a leaf.
    # If i is not a target, it cannot be a leaf (degree >= 2).
    
    # Actually, the constraint "every non-target node has degree >= 2"
    # means that if we have a Steiner Tree, we can prune all non-target leaves
    # repeatedly until no non-target leaves remain.
    # The resulting tree will have all leaves as target nodes.
    # So the problem is simply the Steiner Tree problem.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    for mask in range(1, 1 << t):
        # Combine sub-masks
        for submask in range((mask - 1) & mask, 0, -1):
            submask2 = mask ^ submask
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[submask2][i])
        
        # Dijkstra to update costs for current mask
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
