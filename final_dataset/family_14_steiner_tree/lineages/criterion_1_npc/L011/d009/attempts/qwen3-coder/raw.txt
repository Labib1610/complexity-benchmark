import sys
from itertools import combinations

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def is_connected(edges, nodes):
    if len(nodes) <= 1:
        return True
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    for u, v, _ in edges:
        if u in nodes and v in nodes:
            union(parent, rank, u, v)
    components = len(set(find(parent, x) for x in nodes))
    return components == 1

def kruskal(edges, nodes):
    if not nodes:
        return 0
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    cost = 0
    edges_sorted = sorted(edges, key=lambda x: x[2])
    count = 0
    for u, v, w in edges_sorted:
        if u in nodes and v in nodes and union(parent, rank, u, v):
            cost += w
            count += 1
            if count == len(nodes) - 1:
                break
    return cost if count == len(nodes) - 1 else float('inf')

input = sys.stdin.read
data = input().split()

idx = 0
n = int(data[idx])
idx += 1
m = int(data[idx])
idx += 1
t = int(data[idx])
idx += 1

T = set()
for _ in range(t):
    T.add(int(data[idx]))
    idx += 1

edges = []
for _ in range(m):
    u = int(data[idx])
    idx += 1
    v = int(data[idx])
    idx += 1
    w = int(data[idx])
    idx += 1
    edges.append((u, v, w))

min_cost = float('inf')

# Try all subsets of transpositions
for r in range(len(edges) + 1):
    for C in combinations(edges, r):
        # Determine active domain S
        moved_elements = set()
        for u, v, _ in C:
            moved_elements.add(u)
            moved_elements.add(v)
        S = T.union(moved_elements)
        
        # Check generator count
        if len(C) != len(S) - 1:
            continue
            
        # Check transitive action
        if not is_connected(C, S):
            continue
            
        # Check orbit requirement
        valid = True
        for k in S:
            if k in T:
                continue
            # C_k: transpositions that do NOT move k
            C_k = [(u, v, w) for u, v, w in C if u != k and v != k]
            # Active domain for C_k is S \ {k}
            S_minus_k = S - {k}
            if not S_minus_k:
                continue
                
            # Find orbits of S \ {k} under group generated by C_k
            parent_orbit = list(range(n + 1))
            rank_orbit = [0] * (n + 1)
            for u, v, _ in C_k:
                if u in S_minus_k and v in S_minus_k:
                    union(parent_orbit, rank_orbit, u, v)
            
            orbits = {}
            for x in S_minus_k:
                root = find(parent_orbit, x)
                if root not in orbits:
                    orbits[root] = []
                orbits[root].append(x)
            
            orbit_list = list(orbits.values())
            if len(orbit_list) < 2:
                valid = False
                break
                
            for orbit in orbit_list:
                if not any(x in T for x in orbit):
                    valid = False
                    break
                    
            if not valid:
                break
                
        if valid:
            total_cost = sum(w for _, _, w in C)
            min_cost = min(min_cost, total_cost)

print(min_cost)
