```python
import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t_count = int(input_data[2])
    
    target_set = set(map(int, input_data[3:3+t_count]))
    
    edges = []
    idx = 3 + t_count
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    min_total_cost = float('inf')
    
    # The active domain S must contain T.
    # S is T union some subset of {1..n} \ T.
    # Since |T| <= 10, we can iterate over all possible subsets S.
    # However, S is determined by the edges chosen.
    # Actually, the condition "Generator Count: |C| = |S| - 1" and "Transitive Action"
    # implies that the edges in C must form a spanning tree on S.
    
    # Let's iterate over all possible subsets S such that T <= S <= {1..n}.
    # For a fixed S, we need to find a spanning tree of S with minimum cost
    # such that for every k in S \ T, removing edges incident to k results in
    # components each containing at least one element of T.
    
    # This is equivalent to saying that k is a cut-vertex in the tree,
    # and every component of S \ {k} must contain at least one element of T.
    
    # Iterate over all subsets S containing T
    elements_not_in_t = [i for i in range(1, n + 1) if i not in target_set]
    
    for i in range(len(elements_not_in_t) + 1):
        for subset in itertools.combinations(elements_not_in_t, i):
            S = set(target_set) | set(subset)
            
            # Filter edges that only connect nodes in S
            valid_edges = [e for e in edges if e[0] in S and e[1] in S]
            
            # Find MST of S using valid_edges
            parent = {node: node for node in S}
            def find(i):
                if parent[i] == i: return i
                parent[i] = find(parent[i])
                return parent[i]
            
            def union(i, j):
                root_i = find(i)
                root_j = find(j)
                if root_i != root_j:
                    parent[root_i] = root_j
                    return True
                return False
            
            valid_edges.sort(key=lambda x: x[2])
            mst_edges = []
            cost = 0
            for u, v, w in valid_edges:
                if union(u, v):
                    mst_edges.append((u, v))
                    cost += w
            
            if len(mst_edges) != len(S) - 1:
                continue
                
            # Check Orbit Requirement
            possible = True
            for k in S:
                if k in target_set: continue
                
                # Build adjacency list for the tree
                adj = {node: [] for node in S}
                for u, v in mst_edges:
                    adj[u].append(v)
                    adj[v].append(u)
                
                # Remove k
                # Components are formed by neighbors of k
                visited = {k}
                for neighbor in adj[k]:
                    comp = []
                    stack = [neighbor]
                    visited.add(neighbor)
                    while stack:
                        curr = stack.pop()
                        comp.append(curr)
                        for nxt in adj[curr]:
                            if nxt not in visited:
                                visited.add(nxt)
                                stack.append(nxt)
                    
                    # Check if comp contains at least one element of T
                    if not any(node in target_set for node in comp):
                        possible = False
                        break
                if not possible: break
            
            if possible:
                min_total_cost = min(min_total_cost, cost)
                
    print(min_total_cost)

solve()
```
