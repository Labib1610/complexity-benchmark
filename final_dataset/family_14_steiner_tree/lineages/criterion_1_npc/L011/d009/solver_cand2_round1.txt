```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core_set = set()
    idx = 3
    for _ in range(t):
        core_set.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t == 1:
        print(0)
        return

    # The problem describes a tree structure where all nodes in S are connected.
    # The "No Dead Weight" rule: nodes in S \ T must have degree >= 2.
    # This is equivalent to saying that in the tree of S, all leaves must be in T.
    # This is a Steiner Tree problem variant where all leaves must be in T.
    # Since the graph is small (n=100), we can use the Dreyfus-Wagner algorithm.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    core_list = list(core_set)
    for i in range(t):
        dp[1 << i][core_list[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dp[mask][v] using edges
        pq = []
        for v in range(1, n + 1):
            if dp[mask][v] != inf:
                heapq.heappush(pq, (dp[mask][v], v))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]: continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
        
        # Combine masks
        submask = (mask - 1) & mask
        while submask > 0:
            for v in range(1, n + 1):
                dp[mask][v] = min(dp[mask][v], dp[submask][v] + dp[mask ^ submask][v])
            submask = (submask - 1) & mask
            
    ans = min(dp[(1 << t) - 1][v] for v in range(1, n + 1))
    print(ans)

solve()
```
