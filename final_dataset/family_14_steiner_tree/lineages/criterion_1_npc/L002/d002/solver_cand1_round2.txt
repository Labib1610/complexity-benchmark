```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    essential = []
    idx = 3
    for _ in range(t):
        essential.append(int(input_data[idx]))
        idx += 1
        
    if t <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # Steiner Tree problem on a graph with n <= 100 and t <= 10.
    # Since n is small, we can use the Dreyfus-Wagner algorithm.
    # dp[mask][i] is the min cost to connect the subset of essential formats 
    # represented by 'mask' such that they are all connected to node 'i'.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][essential[i]] = 0
        
    # Precompute all-pairs shortest paths using Dijkstra
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    dist = [[inf] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
        pq = [(0, i)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[i][u]:
                continue
            for v, w in adj[u]:
                if dist[i][v] > d + w:
                    dist[i][v] = d + w
                    heapq.heappush(pq, (dist[i][v], v))
                    
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relax with edges
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dp[mask][j] != inf:
                    dp[mask][i] = min(dp[mask][i], dp[mask][j] + dist[j][i])
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
