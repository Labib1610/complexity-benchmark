```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    primaries = [int(x) for x in input_data[3:3+t]]
    
    # Floyd-Warshall to find all-pairs shortest paths
    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
        
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
            
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            if dist[i][k] == INF:
                continue
            for j in range(1, n + 1):
                if dist[k][j] == INF:
                    continue
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]

    # Steiner Tree DP on the primary nodes
    # dp[mask][v] = min cost to connect primaries in `mask` ending at node v
    dp = [[INF] * (n + 1) for _ in range(1 << t)]
    for i in range(t):
        dp[1 << i][primaries[i]] = 0
        
    for mask in range(1, 1 << t):
        # Subset combination
        sub = (mask - 1) & mask
        while sub > 0:
            other = mask ^ sub
            for v in range(1, n + 1):
                if dp[sub][v] + dp[other][v] < dp[mask][v]:
                    dp[mask][v] = dp[sub][v] + dp[other][v]
            sub = (sub - 1) & mask
            
        # Dijkstra-like relaxation (since edge weights are positive, we can just sort and relax)
        # To find the optimal paths for the current mask
        nodes = list(range(1, n + 1))
        nodes.sort(key=lambda x: dp[mask][x])
        
        # We can optimize by only processing nodes that are reachable
        for u_idx in range(len(nodes)):
            u = nodes[u_idx]
            if dp[mask][u] == INF:
                break
            for v in range(1, n + 1):
                if dist[u][v] == INF:
                    continue
                if dp[mask][u] + dist[u][v] < dp[mask][v]:
                    dp[mask][v] = dp[mask][u] + dist[u][v]
                    # Since we are doing a simple sweep, if we improve a node, 
                    # it might need to be re-evaluated. 
                    # A safer approach for positive weights is to just repeat the sweep until no changes.
                    # Given n <= 100, we can just do a standard Bellman-Ford style sweep.
                    
    # Bellman-Ford style relaxation to guarantee shortest paths for each mask
    # Since n is small (100), doing n-1 sweeps is fast enough and strictly correct
    for mask in range(1, 1 << t):
        for _ in range(n - 1):
            updated = False
            for u in range(1, n + 1):
                if dp[mask][u] == INF:
                    continue
                for v in range(1, n + 1):
                    if dist[u][v] == INF:
                        continue
                    if dp[mask][u] + dist[u][v] < dp[mask][v]:
                        dp[mask][v] = dp[mask][u] + dist[u][v]
                        updated = True
            if not updated:
                break

    full_mask = (1 << t) - 1
    ans = min(dp[full_mask][v] for v in range(1, n + 1))
    print(ans)

if __name__ == '__main__':
    solve()
```
