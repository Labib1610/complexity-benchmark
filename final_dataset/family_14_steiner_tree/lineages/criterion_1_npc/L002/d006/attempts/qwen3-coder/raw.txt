import sys
from itertools import combinations

def find_min_cost_to_span(n, m, t, essential_indices, candidates):
    if t <= 1:
        return 0
    
    # Create a graph where each essential index is a node
    # and each candidate covector is an edge with weight
    # We want to find the minimum cost to connect all essential indices
    # This is a Minimum Spanning Tree problem on the essential indices
    
    # However, we need to consider that we can use paths through non-essential indices
    # So we're looking for the minimum cost to ensure all differences ε_a - ε_b 
    # (for essential a,b) are in the span
    
    # The key insight is that if we can connect all essential indices in the graph
    # formed by the candidate covectors, then we can generate all required covectors
    
    # But actually, we need to think differently:
    # The space W is spanned by {ε_a - ε_b : a,b ∈ essential_indices, a≠b}
    # This is equivalent to the space spanned by {ε_{e_i} - ε_{e_1} : i = 2, ..., t}
    # where e_1, ..., e_t are the essential indices
    
    # So we need to be able to generate each ε_{e_i} - ε_{e_1} for i = 2, ..., t
    # This means we need paths in our graph connecting each e_i to e_1
    
    # Actually, let's think of it as: we need the subgraph induced by edges we pick
    # to connect all the essential vertices in the same component
    
    # This is a Steiner Tree problem variant, but since t ≤ 10,
    # we can try all possible subsets of edges and check if they connect all essential indices
    
    # Better approach: since t is small, we can:
    # 1. Try all 2^m subsets of candidates
    # 2. For each subset, check if it connects all essential indices
    # 3. Among those that do, find the one with minimum cost
    
    # Even better: use Union-Find to check connectivity
    
    def find(parent, x):
        if parent[x] != x:
            parent[x] = find(parent, parent[x])
        return parent[x]
    
    def union(parent, rank, x, y):
        px, py = find(parent, x), find(parent, y)
        if px == py:
            return
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
    
    min_cost = float('inf')
    
    # Try all subsets of candidates
    for mask in range(1 << m):
        cost = 0
        parent = list(range(n + 1))
        rank = [0] * (n + 1)
        
        for i in range(m):
            if mask & (1 << i):
                u, v, w = candidates[i]
                cost += w
                union(parent, rank, u, v)
        
        # Check if all essential indices are connected
        root = find(parent, essential_indices[0])
        connected = True
        for idx in essential_indices:
            if find(parent, idx) != root:
                connected = False
                break
        
        if connected:
            min_cost = min(min_cost, cost)
    
    return min_cost

# Read input
line = sys.stdin.readline().split()
n, m, t = int(line[0]), int(line[1]), int(line[2])

essential_indices = list(map(int, sys.stdin.readline().split()))

candidates = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    candidates.append((u, v, w))

result = find_min_cost_to_span(n, m, t, essential_indices, candidates)
print(result)
