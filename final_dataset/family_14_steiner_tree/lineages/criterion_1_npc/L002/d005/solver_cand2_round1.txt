```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(input_data[i + 3]) for i in range(t)]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The condition x_a = x_b for all target dimensions a, b is equivalent to
    # x_a - x_b = 0 for all pairs (a, b) in the target set.
    # This is satisfied if and only if all target dimensions are in the same
    # connected component in the graph formed by the activated constraint vectors.
    # We want to merge all target nodes into a single component with minimum cost.
    
    # This is a Steiner Tree problem variant, but since we only have 10 target nodes,
    # we can use the Dreyfus-Wagner algorithm or simply realize that we want to
    # partition the target nodes into sets that are connected, and eventually
    # all connected.
    # Actually, this is equivalent to finding a Minimum Spanning Tree in a graph
    # where we contract all target nodes into a single node.
    # Wait, no: we need to connect all target nodes together.
    # This is exactly the Steiner Tree problem on the graph with terminals = targets.
    
    # Since t is small (up to 10), we can use dynamic programming.
    # dp[mask][i] = min cost to connect the subset of targets represented by mask,
    # with node i being part of the connected component.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    for mask in range(1, 1 << t):
        # Submask combination
        sub = (mask - 1) & mask
        while sub > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[sub][i] + dp[mask ^ sub][i])
            sub = (sub - 1) & mask
            
        # Dijkstra to update dp[mask]
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                pq.append((dp[mask][i], i))
        
        import heapq
        heapq.heapify(pq)
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
