```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    primary = [int(x) for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The problem asks for the minimum cost to make all primary elements belong to the same orbit.
    # A set of transpositions forms a group G. Two elements u and v are in the same orbit
    # if there is a path between them in the graph where edges are the transpositions.
    # This is equivalent to finding a Steiner Tree in the graph of n nodes
    # that connects all t primary elements.
    
    # Steiner Tree problem is NP-hard, but t is small (t <= 10).
    # We can use the Dreyfus-Wagner algorithm.
    
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # dist[i][j] is the shortest path between i and j
    dist = [[float('inf')] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
        pq = [(0, i)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[i][u]:
                continue
            for v, w in adj[u]:
                if dist[i][v] > d + w:
                    dist[i][v] = d + w
                    heapq.heappush(pq, (dist[i][v], v))
                    
    # dp[mask][i] is the min cost to connect the subset of primary elements
    # represented by mask, with i being a node in the tree.
    dp = [[float('inf')] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][primary[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dp[mask][i] using submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Update dp[mask][i] using shortest paths
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != float('inf'):
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v in range(1, n + 1):
                if dp[mask][v] > d + dist[u][v]:
                    dp[mask][v] = d + dist[u][v]
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
