import sys
import heapq
from collections import defaultdict

def dijkstra(graph, start, nodes):
    dist = {node: float('inf') for node in nodes}
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, weight in graph[u]:
            if dist[u] + weight < dist[v]:
                dist[v] = dist[u] + weight
                heapq.heappush(pq, (dist[v], v))
                
    return dist

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    critical = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
        
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
        
    all_nodes = set(range(1, n+1))
    
    # Compute shortest paths from each critical node to all other nodes
    distances = {}
    for node in critical:
        distances[node] = dijkstra(graph, node, all_nodes)
    
    # Build a complete graph of critical nodes with edge weights as shortest path distances
    # Then find the minimum spanning tree of this complete graph
    
    # Generate all pairwise distances between critical nodes
    edges = []
    for i in range(t):
        for j in range(i+1, t):
            u = critical[i]
            v = critical[j]
            weight = distances[u][v]
            edges.append((weight, u, v))
            
    # Sort edges by weight
    edges.sort()
    
    # Union-Find for Kruskal's algorithm
    parent = {node: node for node in critical}
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
        
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
        
    # Kruskal's algorithm to find MST
    total_cost = 0
    edges_used = 0
    
    for weight, u, v in edges:
        if union(u, v):
            total_cost += weight
            edges_used += 1
            if edges_used == t - 1:
                break
                
    print(total_cost)

solve()
