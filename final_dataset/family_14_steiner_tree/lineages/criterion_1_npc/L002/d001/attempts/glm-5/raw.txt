import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    critical = [int(next(it)) for _ in range(t)]
    
    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    for k in range(1, n + 1):
        dk = dist[k]
        for i in range(1, n + 1):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(1, n + 1):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    if t == 1:
        print(0)
        return
    
    full = (1 << t) - 1
    dp = [INF] * (1 << t)
    dp[0] = 0
    
    for mask in range(1 << t):
        if dp[mask] == INF:
            continue
        for i in range(t):
            if not (mask & (1 << i)):
                for j in range(i + 1, t):
                    if not (mask & (1 << j)):
                        cost = dist[critical[i]][critical[j]]
                        if cost == INF:
                            continue
                        new_mask = mask | (1 << i) | (1 << j)
                        if dp[mask] + cost < dp[new_mask]:
                            dp[new_mask] = dp[mask] + cost
    
    print(dp[full])

solve()
