import sys
import heapq
from collections import defaultdict

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal(n, edges, must_connect):
    if len(must_connect) <= 1:
        return 0
    
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # First, add all edges with weight 0 between must_connect nodes that are already connected
    initial_edges = []
    for i in range(len(must_connect)):
        for j in range(i + 1, len(must_connect)):
            u, v = must_connect[i], must_connect[j]
            initial_edges.append((0, u, v))
    
    # Combine initial edges with actual edges
    all_edges = initial_edges + edges
    all_edges.sort()
    
    total_cost = 0
    connected_components = n
    
    for w, u, v in all_edges:
        if union(parent, rank, u, v):
            total_cost += w
            connected_components -= 1
            # Check if all must_connect nodes are in the same component
            roots = set(find(parent, x) for x in must_connect)
            if len(roots) == 1:
                return total_cost
    
    return total_cost
