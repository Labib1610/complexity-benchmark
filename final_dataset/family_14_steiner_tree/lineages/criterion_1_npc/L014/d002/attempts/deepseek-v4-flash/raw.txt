We can solve this by treating it as a **Steiner tree** problem, but since the number of critical vaults is small (≤ 10), we can use **bitmask DP** over the critical nodes.

We’ll:
1. Compute shortest paths between all pairs of nodes (Floyd–Warshall).
2. Use DP over subsets of critical nodes, where `dp[mask][v]` = minimum cost to connect the critical nodes in `mask` with a tree rooted at node `v`.
3. Combine subsets to get the final answer.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    critical = list(map(int, input().split()))
    critical = [x - 1 for x in critical]  # 0-indexed

    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]

    # DP
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]

    # Initialize: single critical node
    for i in range(t):
        dp[1 << i][critical[i]] = 0

    # For each mask
    for mask in range(full_mask):
        # Combine submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for v in range(n):
                    dp[mask][v] = min(dp[mask][v], dp[sub][v] + dp[other][v])
            sub = (sub - 1) & mask

        # Relax via shortest paths
        for v in range(n):
            if dp[mask][v] < INF:
                for u in range(n):
                    if dist[v][u] < INF:
                        if dp[mask][v] + dist[v][u] < dp[mask][u]:
                            dp[mask][u] = dp[mask][v] + dist[v][u]

    ans = min(dp[full_mask - 1][v] for v in range(n))
    print(ans)

if __name__ == "__main__":
    solve()
```
