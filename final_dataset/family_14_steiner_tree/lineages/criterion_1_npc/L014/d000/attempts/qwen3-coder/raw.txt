import sys
import heapq
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    critical = list(map(int, data[idx:idx+t]))
    idx += t
    critical_set = set(critical)
    
    # Build graph
    graph = [[] for _ in range(n+1)]
    edges = []
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        graph[u].append((v, w))
        graph[v].append((u, w))
        edges.append((w, u, v))
    
    # If there's only one critical node, cost is 0
    if t <= 1:
        print(0)
        return
    
    # Compute shortest paths from each critical node
    dist = {}
    for c in critical:
        dist[c] = [float('inf')] * (n+1)
        dist[c][c] = 0
        pq = [(0, c)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[c][u]:
                continue
            for v, w in graph[u]:
                if dist[c][v] > d + w:
                    dist[c][v] = d + w
                    heapq.heappush(pq, (dist[c][v], v))
    
    # Create a complete graph among critical nodes with shortest path weights
    # Then find MST of this complete graph
    # We'll use Kruskal's algorithm
    
    # Generate all edges between critical nodes with their shortest path weights
    critical_edges = []
    for i in range(t):
        for j in range(i+1, t):
            u = critical[i]
            v = critical[j]
            weight = dist[u][v]
            critical_edges.append((weight, u, v))
    
    # Sort edges by weight
    critical_edges.sort()
    
    # Union-Find for Kruskal's
    parent = list(range(n+1))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # Kruskal's algorithm
    total_cost = 0
    edges_used = 0
    for w, u, v in critical_edges:
        if union(u, v):
            total_cost += w
            edges_used += 1
            if edges_used == t - 1:
                break
    
    print(total_cost)

if __name__ == "__main__":
    main()
