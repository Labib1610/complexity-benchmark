```python
import sys

def solve():
    data = sys.stdin.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    critical = []
    for _ in range(t):
        critical.append(int(data[idx]) - 1)
        idx += 1
    
    edges = []
    for _ in range(m):
        u = int(data[idx]) - 1; idx += 1
        v = int(data[idx]) - 1; idx += 1
        w = int(data[idx]); idx += 1
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    # Floyd-Warshall to find shortest paths between all pairs
    INF = float('inf')
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    for k in range(n):
        for i in range(n):
            if dist[i][k] == INF:
                continue
            for j in range(n):
                if dist[k][j] == INF:
                    continue
                nd = dist[i][k] + dist[k][j]
                if nd < dist[i][j]:
                    dist[i][j] = nd
    
    # Steiner tree DP on critical vertices
    # dp[mask] = minimum cost to connect the set of critical vertices in 'mask'
    # using any intermediate vertices
    full = (1 << t) - 1
    dp = [INF] * (1 << t)
    
    # Base cases: single critical vertex, cost 0
    for i in range(t):
        dp[1 << i] = 0
    
    # For each vertex r, compute cost to connect subsets of critical vertices
    # using r as a Steiner point
    for r in range(n):
        # For each mask, compute the minimum cost to connect the critical vertices in mask
        # using r as a Steiner point: sum of shortest paths from r to each critical vertex
        # But we need to consider all possible partitions
        
        # First, compute cost[mask] = min over all ways to connect mask using r
        # This is the Steiner tree problem restricted to using r
        
        # We process masks in increasing order of popcount
        # cost[mask] = min over submasks of cost[sub] + cost[mask ^ sub]
        # where cost[{i}] = dist[r][critical[i]]
        
        # But we can also use the global dp to update
        for mask in range(1, 1 << t):
            # Try all ways to split mask into two non-empty parts
            # and connect them both through r
            sub = (mask - 1) & mask
            while sub > 0:
                complement = mask ^ sub
                if sub < complement:  # avoid duplicate work
                    val = dp[sub] + dp[complement]
                    # But this doesn't use r... 
                    # Actually we need to add dist[r][...] somehow
                    pass
                sub = (sub - 1) & mask
        
        # Alternative approach: for each r, compute the cost to connect each mask
        # where the tree is rooted at r
        # cost_r[mask] = min over submasks: cost_r[sub] + cost_r[mask^sub]
        # with cost_r[{i}] = dist[r][critical[i]]
        
        cost_r = [INF] * (1 << t)
        for i in range(t):
            cost_r[1 << i] = dist[r][critical[i]]
        
        for mask in range(1, 1 << t):
            if bin(mask).count('1') == 1:
                continue
            sub = (mask - 1) & mask
            while sub > 0:
                complement = mask ^ sub
                val = cost_r[sub] + cost_r[complement]
                if val < cost_r[mask]:
                    cost_r[mask] = val
                sub = (sub - 1) & mask
        
        # Update global dp
        for mask in range(1, 1 << t):
            if cost_r[mask] < dp[mask]:
                dp[mask] = cost_r[mask]
    
    print(dp[full])

solve()
```
