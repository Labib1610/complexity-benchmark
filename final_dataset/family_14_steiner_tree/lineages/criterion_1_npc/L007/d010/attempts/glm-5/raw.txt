import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    priority = [int(next(it)) for _ in range(t)]
    
    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    for k in range(1, n + 1):
        dk = dist[k]
        for i in range(1, n + 1):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(1, n + 1):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    if t <= 1:
        print(0)
        return
    
    # Steiner tree on priority terminals using Dreyfus-Wagner DP.
    K = t
    term = priority
    # dp[mask][v] = min cost of Steiner tree connecting terminals in mask, rooted at v
    dp = [[INF] * (n + 1) for _ in range(1 << K)]
    for i in range(K):
        for v in range(1, n + 1):
            dp[1 << i][v] = dist[term[i]][v]
    
    for mask in range(1, 1 << K):
        if (mask & (mask - 1)) == 0:
            continue
        for v in range(1, n + 1):
            sub = (mask - 1) & mask
            best = INF
            while sub > 0:
                val = dp[sub][v] + dp[mask ^ sub][v]
                if val < best:
                    best = val
                sub = (sub - 1) & mask
            dp[mask][v] = best
        
        # relax using shortest-path-like propagation
        for v in range(1, n + 1):
            dv = dist[v]
            cur = dp[mask][v]
            for u in range(1, n + 1):
                nd = cur + dv[u]
                if nd < dp[mask][u]:
                    dp[mask][u] = nd
    
    full = (1 << K) - 1
    ans = min(dp[full][v] for v in range(1, n + 1))
    print(ans)

solve()
