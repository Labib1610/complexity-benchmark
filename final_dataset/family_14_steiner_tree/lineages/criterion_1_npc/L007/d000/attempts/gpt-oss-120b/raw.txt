```python
import sys
import heapq

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    req = [int(next(it)) - 1 for _ in range(t)]

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))

    INF = 10**18
    full_mask = (1 << t) - 1
    dp = [[INF] * n for _ in range(1 << t)]

    for i, node in enumerate(req):
        dp[1 << i][node] = 0

    def dijkstra(mask: int):
        dist = dp[mask]
        pq = [(dist[v], v) for v in range(n) if dist[v] != INF]
        heapq.heapify(pq)
        while pq:
            d, u = heapq.heappop(pq)
            if d != dist[u]:
                continue
            for v, w in adj[u]:
                nd = d + w
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(pq, (nd, v))

    for mask in range(1, 1 << t):
        # combine smaller submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            # only need to consider each partition once; sub iterates all non‑empty proper submasks
            for v in range(n):
                val = dp[sub][v] + dp[other][v]
                if val < dp[mask][v]:
                    dp[mask][v] = val
            sub = (sub - 1) & mask
        # propagate through the graph
        dijkstra(mask)

    answer = min(dp[full_mask])
    print(answer)

if __name__ == "__main__":
    solve()
```
