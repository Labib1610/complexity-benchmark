```python
import sys
import heapq
from collections import defaultdict
import itertools

def dijkstra(graph, start, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
            
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
                
    return dist

def main():
    input_lines = []
    for line in sys.stdin:
        input_lines.append(line.strip())
    
    line_idx = 0
    
    n, m, t = map(int, input_lines[line_idx].split())
    line_idx += 1
    
    required = list(map(int, input_lines[line_idx].split()))
    line_idx += 1
    
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, input_lines[line_idx].split())
        line_idx += 1
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    # Compute shortest paths between all required nodes
    required_dist = {}
    
    for req in required:
        dist = dijkstra(graph, req, n)
        required_dist[req] = {r: dist[r] for r in required}
    
    # Create a complete graph of required nodes with shortest path weights
    # Then find MST of this graph
    
    # Use Prim's algorithm on the complete graph of required nodes
    req_set = set(required)
    start = required[0]
    included = {start}
    edge_candidates = []
    
    # Add edges from start node to all other required nodes
    for req in required[1:]:
        heapq.heappush(edge_candidates, (required_dist[start][req], req))
    
    total_cost = 0
    
    while len(included) < t:
        # Find minimum edge to add
        while edge_candidates:
            cost, node = heapq.heappop(edge_candidates)
            if node not in included:
                break
        else:
            # Should not happen as graph is connected
            break
            
        total_cost += cost
        included.add(node)
        
        # Add new edges from this node to all other required nodes not yet included
        for req in required:
            if req not in included:
                heapq.heappush(edge_candidates, (required_dist[node][req], req))
    
    print(total_cost)

if __name__ == "__main__":
    main()
```
