```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "for every pair of target indices a and b, e_a - e_b is in the span"
    # is equivalent to saying that all target indices must belong to the same connected
    # component in the graph formed by the selected edges.
    # This is a Minimum Spanning Tree problem on a graph where we want to connect
    # all target nodes. We can contract all target nodes into a single super-node
    # or use Kruskal's algorithm with a Disjoint Set Union.
    
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # To connect all targets, we can treat all targets as already connected
    # by initializing the DSU with all targets having the same parent.
    first_target = targets[0]
    for i in range(1, t):
        union(first_target, targets[i])
        
    edges.sort(key=lambda x: x[2])
    
    total_penalty = 0
    for u, v, w in edges:
        if union(u, v):
            total_penalty += w
            
    print(total_penalty)

solve()
```
