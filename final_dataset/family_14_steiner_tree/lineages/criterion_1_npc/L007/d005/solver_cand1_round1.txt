```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(input_data[i + 3]) - 1 for i in range(t)]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        w = int(input_data[idx + 2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "all target coordinates of y are equal" means y_i = y_j for all i, j in targets.
    # This is equivalent to y_i - y_j = 0 for all i, j in targets.
    # Let the target indices be T = {t_1, t_2, ..., t_t}.
    # We need the subspace orthogonal to the active set to be contained in the subspace
    # defined by y_{t_1} = y_{t_2} = ... = y_{t_t}.
    # This is equivalent to saying that the span of the active set must contain
    # the vectors {e_{t_1} - e_{t_2}, e_{t_2} - e_{t_3}, ..., e_{t_{t-1}} - e_{t_t}}.
    # This is a connectivity problem: we need to connect all nodes in T in the graph
    # where edges are the available test vectors.
    # This is exactly the Steiner Tree problem where the terminals are the target indices.
    
    # Since t is small (t <= 10), we can use the Dreyfus-Wagner algorithm.
    
    inf = float('inf')
    dist = [[inf] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
            
    # Floyd-Warshall for all-pairs shortest paths
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    # dp[mask][i] is the min cost to connect the subset of targets represented by mask,
    # with i being a node in the resulting tree.
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
        
        # Relax edges
        for i in range(n):
            for j in range(n):
                if dp[mask][j] + dist[j][i] < dp[mask][i]:
                    dp[mask][i] = dp[mask][j] + dist[j][i]
                    
    print(min(dp[(1 << t) - 1]))

solve()
```
