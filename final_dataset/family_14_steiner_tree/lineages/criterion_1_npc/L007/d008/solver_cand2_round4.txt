```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(input_data[i + 3]) - 1 for i in range(t)]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        w = int(input_data[idx + 2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The problem asks for the minimum cost to make all target variables belong 
    # to the same connected component in the graph formed by the chosen edges.
    # This is equivalent to finding a Steiner Tree for the set of target variables.
    # Since n is small (up to 100), we can use the Dreyfus-Wagner algorithm.
    
    # dist[i][j] is the shortest path between i and j
    dist = [[float('inf')] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
        
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][j] > dist[i][k] + dist[k][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    # dp[mask][i] is the min cost to connect the subset of targets represented by mask,
    # with i being a node in the resulting tree.
    dp = [[float('inf')] * n for _ in range(1 << t)]
    
    for i in range(t):
        for node in range(n):
            dp[1 << i][node] = dist[targets[i]][node]
            
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for node in range(n):
                dp[mask][node] = min(dp[mask][node], dp[submask][node] + dp[mask ^ submask][node])
            submask = (submask - 1) & mask
            
        # Relax edges
        pq = []
        for node in range(n):
            if dp[mask][node] != float('inf'):
                heapq.heappush(pq, (dp[mask][node], node))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w, _ in edges:
                # This is a bit inefficient, let's use the precomputed dist
                pass
            # Actually, standard Dijkstra:
            for v in range(n):
                if dist[u][v] != float('inf'):
                    if dp[mask][v] > dp[mask][u] + dist[u][v]:
                        dp[mask][v] = dp[mask][u] + dist[u][v]
                        heapq.heappush(pq, (dp[mask][v], v))
                        
    print(min(dp[(1 << t) - 1]))

solve()
```
