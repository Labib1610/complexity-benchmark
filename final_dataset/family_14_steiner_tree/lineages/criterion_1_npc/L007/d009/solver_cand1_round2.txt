```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_indices = [int(x) for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t < 2:
        print(0)
        return

    # The condition "indifferent to all activated transpositions implies 
    # indifference to (a, b) for all a, b in priority_indices" is equivalent 
    # to saying that all priority indices must belong to the same connected 
    # component in the graph formed by the activated transpositions.
    
    # Kruskal's algorithm approach:
    # We want to connect all nodes in priority_indices with minimum weight.
    # This is a Steiner Tree problem, but since t is very small (t <= 10),
    # we can use the Dreyfus-Wagner algorithm or simply realize that for 
    # small t, we can use a variation of Prim's or Kruskal's.
    # Actually, for t <= 10, we can use dynamic programming on subsets.
    
    # dp[mask][i] = min cost to connect the subset of priority indices 
    # represented by 'mask', with node 'i' being in the component.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][priority_indices[i]] = 0
        
    for mask in range(1, 1 << t):
        # Relax edges
        for _ in range(n):
            changed = False
            for u, v, w in edges:
                for i in range(1, n + 1):
                    if dp[mask][u] + w < dp[mask][v]:
                        dp[mask][v] = dp[mask][u] + w
                        changed = True
                    if dp[mask][v] + w < dp[mask][u]:
                        dp[mask][u] = dp[mask][v] + w
                        changed = True
            if not changed:
                break
        
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
