```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The target subspace is spanned by {e_i - e_j | i, j in critical}.
    # This is equivalent to saying that all nodes in 'critical' must be in the same
    # connected component in the graph formed by the chosen edges.
    # This is a Steiner Tree problem variant, but since we only need to connect
    # the set of 'critical' nodes, we can use the property that we want to 
    # partition the nodes into components such that all critical nodes are in one.
    # Actually, this is equivalent to finding a Minimum Spanning Tree in a graph
    # where we contract all critical nodes into a single super-node.
    
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # Contract all critical nodes into one
    first_crit = critical[0]
    for i in range(1, t):
        union(first_crit, critical[i])
        
    # Kruskal's algorithm on the remaining components
    edges.sort(key=lambda x: x[2])
    
    total_cost = 0
    for u, v, w in edges:
        if union(u, v):
            total_cost += w
            
    print(total_cost)

solve()
```
