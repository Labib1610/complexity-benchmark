```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    priority = [int(next(it)) for _ in range(t)]
    priority_set = set(priority)
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Dijkstra from each priority language
    dist = {}
    for p in priority:
        d = [float('inf')] * (n + 1)
        d[p] = 0
        pq = [(0, p)]
        while pq:
            du, u = heapq.heappop(pq)
            if du > d[u]:
                continue
            for v, w in adj[u]:
                if d[v] > du + w:
                    d[v] = du + w
                    heapq.heappush(pq, (d[v], v))
        dist[p] = d
    
    # Try each language as source
    best = float('inf')
    for s in range(1, n + 1):
        total = 0
        for p in priority:
            total += dist[p][s]
        best = min(best, total)
    
    print(best)

solve()
```
