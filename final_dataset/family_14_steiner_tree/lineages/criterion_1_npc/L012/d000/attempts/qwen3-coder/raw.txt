import sys
import heapq
from collections import defaultdict
import math

def dijkstra(graph, start, nodes):
    dist = {node: math.inf for node in nodes}
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
            
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
                
    return dist

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    primary = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
        
    edges = []
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u, v, w))
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Compute shortest paths between all pairs of primary languages
    # Using a modified Floyd-Warshall approach but only for primary languages
    primary_set = set(primary)
    cost = {}
    
    for p in primary:
        distances = dijkstra(graph, p, range(1, n+1))
        for q in primary:
            if p != q:
                cost[(p, q)] = distances[q]
    
    # Now we have a complete graph on primary languages
    # Find minimum spanning tree of this complete graph
    # Using Prim's algorithm
    
    in_mst = set()
    min_cost = {p: math.inf for p in primary}
    min_cost[primary[0]] = 0
    total_cost = 0
    
    for _ in range(t):
        # Find the vertex with minimum edge weight to tree
        u = None
        for v in primary:
            if v not in in_mst and (u is None or min_cost[v] < min_cost[u]):
                u = v
                
        in_mst.add(u)
        total_cost += min_cost[u]
        
        # Update min_cost for remaining vertices
        for v in primary:
            if v not in in_mst:
                if (u, v) in cost:
                    min_cost[v] = min(min_cost[v], cost[(u, v)])
                    
    print(total_cost)

solve()
