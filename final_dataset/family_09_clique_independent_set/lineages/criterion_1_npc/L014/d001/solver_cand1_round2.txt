import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a meet-in-the-middle approach or 
    # a backtracking algorithm with pruning.
    
    def find_independent_set(candidates, current_set, k_needed):
        if k_needed == 0:
            return current_set
        if not candidates or len(candidates) < k_needed:
            return None
        
        # Heuristic: pick the node with the most neighbors in candidates to prune
        v = candidates[0]
        
        # Option 1: Include v
        new_candidates = [c for c in candidates if c != v and not (adj[v] & (1 << c))]
        res = find_independent_set(new_candidates, current_set + [v + 1], k_needed - 1)
        if res:
            return res
        
        # Option 2: Exclude v
        res = find_independent_set(candidates[1:], current_set, k_needed)
        return res

    # For n=46, simple backtracking might be slow, but the constraints 
    # and the nature of the problem allow for optimized search.
    # Using Bron-Kerbosch style logic or simple bitmask recursion.
    
    def solve_recursive(candidates_mask, current_set, k_needed):
        if k_needed == 0:
            return current_set
        if bin(candidates_mask).count('1') < k_needed:
            return None
        
        # Pick node with highest index
        v = (candidates_mask & -candidates_mask).bit_length() - 1
        
        # Include v
        res = solve_recursive(candidates_mask & ~adj[v] & ~(1 << v), current_set + [v + 1], k_needed - 1)
        if res:
            return res
        
        # Exclude v
        res = solve_recursive(candidates_mask & ~(1 << v), current_set, k_needed)
        return res

    result = solve_recursive((1 << n) - 1, [], k)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
