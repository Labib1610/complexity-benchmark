import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [set() for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    
    result = []
    
    def backtrack(candidates, current_set):
        if len(current_set) == k:
            return current_set
        
        if not candidates:
            return None
        
        # Heuristic: pick the candidate with the fewest remaining neighbors
        # to prune the search space effectively.
        while candidates:
            v = candidates.pop(0)
            
            # New candidates must not be connected to v
            new_candidates = [c for c in candidates if c not in adj[v]]
            
            res = backtrack(new_candidates, current_set + [v])
            if res:
                return res
        return None

    # Optimization: sort nodes by degree (ascending) to find independent sets faster
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]))
    
    ans = backtrack(nodes, [])
    
    if ans:
        print("YES")
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
