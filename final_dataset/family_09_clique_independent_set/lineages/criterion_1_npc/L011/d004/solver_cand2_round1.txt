import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find a subset of size k such that no two elements are connected.
    # This is equivalent to finding an independent set of size k.
    # Since n <= 46, we use a meet-in-the-middle approach.
    
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the two halves
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            subset = []
            valid = True
            mask = 0
            for j in range(size):
                if (i >> j) & 1:
                    u = nodes[j]
                    if (mask & adj_list[u]):
                        valid = False
                        break
                    mask |= (1 << u)
                    subset.append(u)
            if valid:
                res[len(subset)].append((mask, subset))
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    for k1 in range(max(0, k - n2), min(k, n1) + 1):
        k2 = k - k1
        if k2 < 0 or k2 > n2:
            continue
            
        for mask1, sub1 in sets1[k1]:
            # Filter sets2 that are compatible with mask1
            # A set in sets2 is compatible if no node in it is connected to any node in mask1
            forbidden = 0
            for u in sub1:
                forbidden |= adj[u]
            
            for mask2, sub2 in sets2[k2]:
                if not (mask2 & forbidden):
                    print("YES")
                    result = [x + 1 for x in sub1 + sub2]
                    print(*(sorted(result)))
                    return

    print("NO")

solve()
