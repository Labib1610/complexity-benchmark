import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # The condition x^T L x = sum(D_ii * x_i)
    # x^T (D - A) x = sum(D_ii * x_i)
    # x^T D x - x^T A x = sum(D_ii * x_i)
    # Since x is binary, x_i^2 = x_i, so x^T D x = sum(D_ii * x_i)
    # The equation simplifies to: x^T A x = 0
    # x^T A x = sum_{i,j} A_ij * x_i * x_j = 2 * (number of edges within the panel)
    # So the condition is: number of edges within the panel must be 0.
    # This means the panel must be an independent set of size k.
    
    # Since n <= 46, we can use meet-in-the-middle to find an independent set of size k.
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        res = {} # size -> list of bitmasks
        for i in range(1 << size):
            mask = 0
            valid = True
            count = 0
            for j in range(size):
                if (i >> j) & 1:
                    if (mask & adj_list[nodes[j]]) != 0:
                        valid = False
                        break
                    mask |= (1 << nodes[j])
                    count += 1
            if valid:
                if count not in res: res[count] = []
                res[count].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    for c1 in sets1:
        c2 = k - c1
        if c2 in sets2:
            for m1 in sets1[c1]:
                for m2 in sets2[c2]:
                    # Check if there are edges between m1 and m2
                    possible = True
                    for i in range(n1):
                        if (m1 >> i) & 1:
                            if (adj[i] & m2) != 0:
                                possible = False
                                break
                    if possible:
                        print("YES")
                        res = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                res.append(str(i + 1))
                        print(" ".join(res))
                        return
    print("NO")

solve()
