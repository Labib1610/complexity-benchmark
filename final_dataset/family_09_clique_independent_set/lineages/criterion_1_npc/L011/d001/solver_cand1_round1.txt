import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1] |= (1 << (v-1))
        adj[v-1] |= (1 << (u-1))

    # The condition "at least one member of every interaction must be in the control group"
    # is equivalent to saying "no two members of the focus panel can have an interaction".
    # This means the focus panel must be an Independent Set of size k.
    
    # Since n <= 46, we use a meet-in-the-middle approach to find an independent set of size k.
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        res = {}
        for i in range(1 << size):
            is_independent = True
            count = 0
            mask = 0
            for j in range(size):
                if (i >> j) & 1:
                    count += 1
                    node = nodes[j]
                    if (mask & adj_list[node]):
                        is_independent = False
                        break
                    mask |= (1 << node)
            if is_independent:
                if count not in res:
                    res[count] = []
                res[count].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    for c1 in sets1:
        c2 = k - c1
        if c2 in sets2:
            for m1 in sets1[c1]:
                # Check if m1 is compatible with any m2 in sets2[c2]
                # m1 is compatible with m2 if no node in m1 is connected to any node in m2
                forbidden_mask = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        forbidden_mask |= adj[i]
                
                for m2 in sets2[c2]:
                    if not (m2 & forbidden_mask):
                        # Found it
                        res = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                res.append(i + 1)
                        print("YES")
                        print(*(res))
                        return

    print("NO")

solve()
