import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * (n + 1)
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    # The problem asks for an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    
    def find_independent_set(current_mask, count, start_node):
        if count == k:
            res = []
            for i in range(n):
                if (current_mask >> i) & 1:
                    res.append(i + 1)
            return res
        
        if start_node > n:
            return None
        
        # Remaining nodes to pick
        remaining = k - count
        if n - start_node + 1 < remaining:
            return None
            
        for i in range(start_node, n + 1):
            # Check if node i is compatible with current set
            if not (adj[i] & current_mask):
                res = find_independent_set(current_mask | (1 << (i - 1)), count + 1, i + 1)
                if res is not None:
                    return res
        return None

    result = find_independent_set(0, 0, 1)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
