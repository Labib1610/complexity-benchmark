from itertools import combinations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    edges = []
    idx = 3
    for _ in range(m):
        u = int(data[idx]) - 1  # Convert to 0-indexed
        v = int(data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
    
    # If there are no edges, any k nodes work
    if m == 0:
        print("YES")
        print(' '.join(str(i+1) for i in range(k)))
        return
    
    # We need to find a subset of k nodes that covers all edges
    # This is equivalent to finding a subset of (n-k) nodes that forms a vertex cover
    # Then the complement would be the independent set of size k
    
    # Try all combinations of k nodes
    # Since n <= 46, we can't check all C(n,k)
    # But we can check if there exists an independent set of size k
    # Which is equivalent to checking if the complement (of size n-k) is a vertex cover
    
    # However, let's try a different approach: meet in the middle
    # Split the nodes into two halves and enumerate independent sets in each half
    
    mid = n // 2
    left_nodes = list(range(mid))
    right_nodes = list(range(mid, n))
    
    # Generate all independent subsets in the left part
    left_independent = {}
    for mask in range(1 << mid):
        subset = [i for i in range(mid) if mask & (1 << i)]
        # Check if this subset is independent
        valid = True
        for u, v in edges:
            if u < mid and v < mid:
                if (mask & (1 << u)) and (mask & (1 << v)):
                    valid = False
                    break
        if valid:
            size = bin(mask).count('1')
            if size not in left_independent:
                left_independent[size] = []
            left_independent[size].append(mask)
    
    # Generate all independent subsets in the right part
    right_independent = {}
    right_edges = [(u-mid, v-mid) for u,v in edges if u >= mid and v >= mid]
    cross_edges = [(u, v-mid) for u,v in edges if u < mid and v >= mid]
    cross_edges.extend([(v, u-mid) for u,v in edges if v < mid and u >= mid])
    
    for mask in range(1 << (n - mid)):
        subset = [i for i in range(n - mid) if mask & (1 << i)]
        # Check if this subset is independent among right nodes
        valid = True
        for u, v in right_edges:
            if (mask & (1 << u)) and (mask & (1 << v)):
                valid = False
                break
        if valid:
            size = bin(mask).count('1')
            if size not in right_independent:
                right_independent[size] = []
            right_independent[size].append(mask)
    
    # Now check for combinations
    target = k
    for left_size in left_independent:
        right_size_needed = target - left_size
        if right_size_needed in right_independent:
            for left_mask in left_independent[left_size]:
                for right_mask in right_independent[right_size_needed]:
                    # Check cross edges
                    valid = True
                    for u, v in cross_edges:
                        if (left_mask & (1 << u)) and (right_mask & (1 << v)):
                            valid = False
                            break
                    if valid:
                        # Found a solution
                        result = []
                        for i in range(mid):
                            if left_mask & (1 << i):
                                result.append(i + 1)
                        for i in range(n - mid):
                            if right_mask & (1 << i):
                                result.append(i + mid + 1)
                        print("YES")
                        print(' '.join(map(str, result)))
                        return
    
    # Special case when k=1
    if k == 1:
        # Just pick any node that is not part of all edges
        # Actually, we just need to check if there's a node that doesn't touch all edges
        # Or even simpler, just check if we can remove n-1 nodes such that no edge remains
        # That means we're looking for a vertex cover of size n-1
        # The complement would be an independent set of size 1
        for node in range(n):
            # Check if removing all other nodes leaves no edges
            node_in_edge = False
            for u, v in edges:
                if u != node and v != node:
                    node_in_edge = True
                    break
            if not node_in_edge:
                print("YES")
                print(node + 1)
                return
                
    # For small n, brute force might work
    if n <= 20:
        for combo in combinations(range(n), k):
            # Check if this combination is independent
            valid = True
            for u, v in edges:
                if u in combo and v in combo:
                    valid = False
                    break
            if valid:
                print("YES")
                print(' '.join(str(x+1) for x in combo))
                return
        print("NO")
        return
    
    print("NO")

solve()
