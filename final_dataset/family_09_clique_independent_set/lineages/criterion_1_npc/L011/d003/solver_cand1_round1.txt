import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
        
    # A panel of size k is valid if and only if every edge has at least one endpoint
    # in the control group (the set of n-k members not in the panel).
    # This is equivalent to saying the panel must be an Independent Set of size k.
    # If the panel is an independent set, then for every edge (u, v), 
    # at least one of u or v is NOT in the panel (i.e., in the control group).
    
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a meet-in-the-middle approach or a simple backtracking
    # with pruning. Given the constraints and the nature of the problem, 
    # Bron-Kerbosch or simple backtracking is efficient enough.
    
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    def get_independent_set(candidates, current_set):
        if len(current_set) == k:
            return current_set
        if len(current_set) + bin(candidates).count('1') < k:
            return None
        
        if candidates == 0:
            return None
            
        # Heuristic: pick vertex with most neighbors in candidates
        v = (candidates & -candidates).bit_length() - 1
        
        # Option 1: Include v in independent set
        res = get_independent_set(candidates & ~adj[v] & ~(1 << v), current_set | {v})
        if res is not None:
            return res
            
        # Option 2: Exclude v
        res = get_independent_set(candidates & ~(1 << v), current_set)
        return res

    result = get_independent_set((1 << n) - 1, set())
    
    if result is not None:
        print("YES")
        print(*(sorted([x + 1 for x in result])))
    else:
        print("NO")

solve()
