import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
        
    # The condition: "no recorded event is left without a qualified auditor in the oversight committee"
    # means that for every edge (u, v), at least one of u or v must be in the oversight committee.
    # This is equivalent to saying that the focus panel (size k) must be an Independent Set.
    # We need to find an Independent Set of size exactly k.
    
    # Since n <= 46, we can use a meet-in-the-middle approach to find an IS of size k.
    # Split into two halves: n1 = n // 2, n2 = n - n1
    n1 = n // 2
    n2 = n - n1
    
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    def get_is(nodes):
        # nodes is a list of vertex indices
        size = len(nodes)
        # map nodes to 0..size-1
        mapping = {node: i for i, node in enumerate(nodes)}
        sub_adj = [0] * size
        for i in range(size):
            for j in range(i + 1, size):
                if (adj[nodes[i]] >> nodes[j]) & 1:
                    sub_adj[i] |= (1 << j)
                    sub_adj[j] |= (1 << i)
        
        # res[count] = list of bitmasks (relative to nodes)
        res = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            is_independent = True
            for j in range(size):
                if (i >> j) & 1:
                    if sub_adj[j] & i:
                        is_independent = False
                        break
            if is_independent:
                res[bin(i).count('1')].append(i)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    is1 = get_is(nodes1)
    is2 = get_is(nodes2)
    
    # Precompute cross-edges between nodes1 and nodes2
    cross_adj = [0] * n1
    for i in range(n1):
        for j in range(n2):
            if (adj[i] >> (n1 + j)) & 1:
                cross_adj[i] |= (1 << j)
                
    for c1 in range(min(k + 1, n1 + 1)):
        c2 = k - c1
        if c2 < 0 or c2 > n2:
            continue
        
        for mask1 in is1[c1]:
            # Find which nodes in nodes2 are forbidden by mask1
            forbidden = 0
            for i in range(n1):
                if (mask1 >> i) & 1:
                    forbidden |= cross_adj[i]
            
            for mask2 in is2[c2]:
                if not (mask2 & forbidden):
                    # Found it
                    res = []
                    for i in range(n1):
                        if (mask1 >> i) & 1:
                            res.append(i + 1)
                    for i in range(n2):
                        if (mask2 >> i) & 1:
                            res.append(n1 + i + 1)
                    print("YES")
                    print(*(res))
                    return
    print("NO")

solve()
