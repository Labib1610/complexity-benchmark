import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
        
    # The condition: control group (size n-k) must contain at least one 
    # participant from every edge.
    # This is equivalent to: the focus panel (size k) must NOT contain 
    # both endpoints of any edge.
    # In other words, the focus panel must be an Independent Set of size k.
    
    # Since n <= 46, we use a meet-in-the-middle approach to find an 
    # independent set of size k.
    
    n1 = n // 2
    n2 = n - n1
    
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    def get_independent_sets(nodes):
        size = len(nodes)
        res = {} # size -> list of bitmasks
        for i in range(1 << size):
            mask = 0
            count = 0
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    u = nodes[j]
                    if (mask & adj[u]):
                        valid = False
                        break
                    mask |= (1 << u)
                    count += 1
            if valid:
                if count not in res: res[count] = []
                res[count].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1)
    sets2 = get_independent_sets(nodes2)
    
    for s1 in sets1:
        s2 = k - s1
        if s2 in sets2:
            for m1 in sets1[s1]:
                for m2 in sets2[s2]:
                    # Check if m1 and m2 are independent
                    # m1 is independent, m2 is independent
                    # Need to check edges between m1 and m2
                    possible = True
                    for i in range(n1):
                        if (m1 >> i) & 1:
                            if (adj[i] & m2):
                                possible = False
                                break
                    if possible:
                        print("YES")
                        ans = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                ans.append(str(i + 1))
                        print(" ".join(ans))
                        return
    print("NO")

solve()
