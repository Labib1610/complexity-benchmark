import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * (n + 1)
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    
    def find_independent_set(candidates, current_set, k_needed):
        if k_needed == 0:
            return current_set
        if not candidates or len(candidates) < k_needed:
            return None
        
        # Heuristic: pick the node with the most constraints (highest degree)
        # or simply pick the first one.
        v = candidates[0]
        
        # Option 1: Include v in the set
        new_candidates = [c for c in candidates[1:] if not (adj[v] & (1 << (c - 1)))]
        res = find_independent_set(new_candidates, current_set + [v], k_needed - 1)
        if res is not None:
            return res
        
        # Option 2: Exclude v from the set
        res = find_independent_set(candidates[1:], current_set, k_needed)
        if res is not None:
            return res
            
        return None

    result = find_independent_set(list(range(1, n + 1)), [], k)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
