import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
        
    # Score = sum(c!) for each edge, where c is number of endpoints in panel.
    # c=0: score 0! = 1
    # c=1: score 1! = 1
    # c=2: score 2! = 2
    # Total score = sum_{edges} (1 if c=0 or c=1 else 2)
    # Total score = sum_{edges} (1 + (1 if c=2 else 0))
    # Total score = m + (number of edges with both endpoints in panel)
    # We want Total score = m, so we need (number of edges with both endpoints in panel) = 0.
    # This means the panel must be an independent set of size k.
    
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use meet-in-the-middle or backtracking with pruning.
    # Given the constraints and the nature of independent set, simple backtracking
    # with bitmasking is efficient enough for n=46.
    
    def find_is(idx, current_mask, count):
        if count == k:
            return current_mask
        if idx == n or (count + (n - idx) < k):
            return None
        
        # Try including idx
        if not (current_mask & (1 << idx)) and not (adj[idx] & current_mask):
            res = find_is(idx + 1, current_mask | (1 << idx), count + 1)
            if res is not None:
                return res
        
        # Try excluding idx
        res = find_is(idx + 1, current_mask, count)
        return res

    # Optimization: sort vertices by degree
    nodes = sorted(range(n), key=lambda x: bin(adj[x]).count('1'))
    
    # Rebuild adj for sorted nodes
    new_adj = [0] * n
    mapping = {old: new for new, old in enumerate(nodes)}
    for u, v in edges:
        u_new, v_new = mapping[u], mapping[v]
        new_adj[u_new] |= (1 << v_new)
        new_adj[v_new] |= (1 << u_new)
    
    def backtrack(idx, current_mask, count):
        if count == k:
            return current_mask
        if idx == n or (count + (n - idx) < k):
            return None
        
        # Try including
        if not (new_adj[idx] & current_mask):
            res = backtrack(idx + 1, current_mask | (1 << idx), count + 1)
            if res is not None:
                return res
        
        # Try excluding
        return backtrack(idx + 1, current_mask, count)

    result_mask = backtrack(0, 0, 0)
    
    if result_mask is not None:
        print("YES")
        res_nodes = []
        for i in range(n):
            if (result_mask >> i) & 1:
                res_nodes.append(nodes[i] + 1)
        print(*(sorted(res_nodes)))
    else:
        print("NO")

solve()
