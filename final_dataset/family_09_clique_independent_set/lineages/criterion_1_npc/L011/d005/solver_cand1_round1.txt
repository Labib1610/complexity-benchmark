import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
        idx += 2
        
    w = [len(adj[i]) for i in range(n + 1)]
    
    # ECL = 5m + 2 * sum(W_i for i in P) + 3 * sum(W_j for j in C)
    # Since sum(W_i for i in P) + sum(W_j for j in C) = 2m,
    # sum(W_j for j in C) = 2m - sum(W_i for i in P)
    # ECL = 5m + 2 * sum(W_i for i in P) + 3 * (2m - sum(W_i for i in P))
    # ECL = 5m + 6m - sum(W_i for i in P) = 11m - sum(W_i for i in P)
    
    # ACL = 10 * (edges between P and C) + 5 * (edges within P) + 11 * (edges within C)
    # Let E_P be edges within P, E_C be edges within C, E_PC be edges between P and C.
    # E_P + E_C + E_PC = m
    # sum(W_i for i in P) = 2 * E_P + E_PC
    # ACL = 10 * E_PC + 5 * E_P + 11 * (m - E_P - E_PC)
    # ACL = 10 * E_PC + 5 * E_P + 11m - 11 * E_P - 11 * E_PC
    # ACL = 11m - 6 * E_P - E_PC
    
    # Condition: ECL = ACL
    # 11m - sum(W_i for i in P) = 11m - 6 * E_P - E_PC
    # sum(W_i for i in P) = 6 * E_P + E_PC
    # 2 * E_P + E_PC = 6 * E_P + E_PC
    # 4 * E_P = 0 => E_P = 0
    
    # So the condition is simply that there are no edges between any two members in the panel.
    # This means the panel must be an independent set of size k.
    
    if k > n:
        print("NO")
        return

    # Meet-in-the-middle for independent set of size k
    # n <= 46 is too large for simple recursion, but we need an independent set of size k.
    # Actually, the condition E_P = 0 is necessary and sufficient.
    
    def get_independent_sets(nodes):
        size = len(nodes)
        res = {}
        for i in range(1 << size):
            subset = [nodes[j] for j in range(size) if (i & (1 << j))]
            is_independent = True
            for u in subset:
                for v in subset:
                    if v in adj[u]:
                        is_independent = False
                        break
                if not is_independent: break
            if is_independent:
                count = len(subset)
                if count not in res: res[count] = []
                res[count].append(set(subset))
        return res

    mid = n // 2
    left_nodes = list(range(1, mid + 1))
    right_nodes = list(range(mid + 1, n + 1))
    
    left_sets = get_independent_sets(left_nodes)
    right_sets = get_independent_sets(right_nodes)
    
    for k1 in range(k + 1):
        k2 = k - k1
        if k1 in left_sets and k2 in right_sets:
            for s1 in left_sets[k1]:
                for s2 in right_sets[k2]:
                    # Check if s1 union s2 is independent
                    valid = True
                    for u in s1:
                        for v in s2:
                            if v in adj[u]:
                                valid = False
                                break
                        if not valid: break
                    if valid:
                        print("YES")
                        print(*(list(s1) + list(s2)))
                        return
    print("NO")

solve()
