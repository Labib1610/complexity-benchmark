import sys
from collections import defaultdict

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = defaultdict(set)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Greedy approach: repeatedly pick vertex with minimum degree in remaining graph
    # This is a standard heuristic for finding large independent sets
    remaining = set(range(1, n + 1))
    result = []
    
    while remaining and len(result) < k:
        # Find vertex with minimum degree in the remaining graph
        best_vertex = None
        best_degree = float('inf')
        
        for v in remaining:
            deg = len(adj[v] & remaining)
            if deg < best_degree:
                best_degree = deg
                best_vertex = v
        
        # Add this vertex to our independent set
        result.append(best_vertex)
        
        # Remove this vertex and all its neighbors from remaining
        neighbors_to_remove = adj[best_vertex] & remaining
        remaining.remove(best_vertex)
        remaining -= neighbors_to_remove
    
    if len(result) == k:
        print("YES")
        print(' '.join(map(str, result)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
