```python
import sys
from itertools import combinations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    # Read interfering pairs
    interfering = set()
    idx = 3
    for _ in range(m):
        u = int(data[idx]) - 1  # Convert to 0-based indexing
        v = int(data[idx+1]) - 1
        if u > v:
            u, v = v, u
        interfering.add((u, v))
        idx += 2
    
    # If k is 1, just pick any marker
    if k == 1:
        print("YES")
        print(1)
        return
    
    # Generate all combinations of k markers
    markers = list(range(n))
    for combo in combinations(markers, k):
        valid = True
        # Check all pairs in this combination
        for i in range(len(combo)):
            for j in range(i+1, len(combo)):
                u = combo[i]
                v = combo[j]
                if u > v:
                    u, v = v, u
                if (u, v) in interfering:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            # Found a valid combination
            print("YES")
            print(' '.join(str(x + 1) for x in combo))  # Convert back to 1-based indexing
            return
    
    print("NO")

if __name__ == "__main__":
    main()
```
