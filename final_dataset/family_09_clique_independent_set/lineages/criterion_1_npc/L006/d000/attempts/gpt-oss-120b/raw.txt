import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    ALL = (1 << n) - 1
    target = k
    answer_mask = 0
    found = False

    # order vertices by degree (small to large) for better branching
    order = sorted(range(n), key=lambda x: bin(adj[x]).count("1"))
    # map original index to position in ordered list
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i
    # rebuild adjacency in ordered space
    adj2 = [0] * n
    for i in range(n):
        v = order[i]
        mask = 0
        for u in range(n):
            if (adj[v] >> u) & 1:
                mask |= 1 << pos[u]
        adj2[i] = mask

    # initial candidate set: all vertices in ordered space
    candidates0 = (1 << n) - 1

    # popcount helper
    popcnt = int.bit_count

    def dfs(cur_mask: int, cand_mask: int, cur_size: int):
        nonlocal answer_mask, found
        if found:
            return
        if cur_size == target:
            answer_mask = cur_mask
            found = True
            return
        # prune if even taking all remaining cannot reach target
        if cur_size + popcnt(cand_mask) < target:
            return
        # choose a vertex: pick the one with max degree in candidates (simple heuristic)
        # get first vertex
        v = (cand_mask & -cand_mask).bit_length() - 1
        # branch 1: include v
        new_cand = cand_mask & ~adj2[v] & ~(1 << v)
        dfs(cur_mask | (1 << v), new_cand, cur_size + 1)
        if found:
            return
        # branch 2: exclude v
        dfs(cur_mask, cand_mask & ~(1 << v), cur_size)

    dfs(0, candidates0, 0)

    if not found:
        print("NO")
    else:
        # translate answer_mask back to original vertex numbers (1‑based)
        res = []
        for i in range(n):
            if (answer_mask >> i) & 1:
                res.append(order[i] + 1)
        print("YES")
        print(" ".join(map(str, res)))

if __name__ == "__main__":
    solve()
