import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Max Independent Set / Clique problem is NP-hard, but n <= 46 allows
    # for a meet-in-the-middle approach or a backtracking search with pruning.
    # Given the constraints and time limit, a backtracking search with bitmasking
    # is efficient enough for n=46.
    
    def find_clique(candidates, current_set, needed):
        if needed == 0:
            return current_set
        if not candidates or len(candidates) < needed:
            return None
        
        # Heuristic: pick the node with the most constraints to prune faster
        # Or simply pick the first one
        v = (candidates & -candidates).bit_length() - 1
        
        # Option 1: Include v
        res = find_clique(candidates & ~adj[v] & ~(1 << v), current_set | (1 << v), needed - 1)
        if res is not None:
            return res
        
        # Option 2: Exclude v
        return find_clique(candidates & ~(1 << v), current_set, needed)

    # The problem asks for an independent set of size k.
    # This is equivalent to finding a clique of size k in the complement graph.
    # Let's use Bron-Kerbosch or simple backtracking.
    
    def backtrack(candidates, current_size, current_list):
        if current_size == k:
            return current_list
        if not candidates or current_size + bin(candidates).count('1') < k:
            return None
        
        # Pick vertex with highest degree in remaining candidates to prune
        v = (candidates & -candidates).bit_length() - 1
        
        # Try including v
        new_candidates = candidates & ~adj[v] & ~(1 << v)
        res = backtrack(new_candidates, current_size + 1, current_list + [v + 1])
        if res:
            return res
        
        # Try excluding v
        return backtrack(candidates & ~(1 << v), current_size, current_list)

    result = backtrack((1 << n) - 1, 0, [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
