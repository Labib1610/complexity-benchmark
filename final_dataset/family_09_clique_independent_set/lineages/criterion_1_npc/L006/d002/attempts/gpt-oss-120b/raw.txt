import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    ALL = (1 << n) - 1

    # vertices ordered by degree (ascending) for better pruning
    order = sorted(range(n), key=lambda x: bin(adj[x]).count("1"))
    # map original index to position in ordered list
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i

    # rebuild adjacency in ordered space
    adj2 = [0] * n
    for i in range(n):
        v = order[i]
        mask = 0
        for j in range(n):
            if adj[v] >> order[j] & 1:
                mask |= 1 << j
        adj2[i] = mask

    # initial candidate set: all vertices
    cand = (1 << n) - 1
    result_mask = 0
    found = False

    def dfs(cur_mask: int, cur_cand: int, cur_size: int):
        nonlocal result_mask, found
        if found:
            return
        # prune if even taking all candidates we cannot reach k
        if cur_size + cur_cand.bit_count() < k:
            return
        if cur_size == k:
            result_mask = cur_mask
            found = True
            return
        # choose a vertex with smallest degree among candidates
        # simple heuristic: pick first set bit
        v = (cur_cand & -cur_cand).bit_length() - 1
        # branch: include v
        dfs(cur_mask | (1 << v),
            cur_cand & ~adj2[v] & ~(1 << v),
            cur_size + 1)
        if found:
            return
        # branch: exclude v
        dfs(cur_mask, cur_cand & ~(1 << v), cur_size)

    dfs(0, cand, 0)

    if not found:
        print("NO")
    else:
        # translate back to original numbering
        ans = []
        for i in range(n):
            if result_mask >> i & 1:
                ans.append(order[i] + 1)
        print("YES")
        print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
