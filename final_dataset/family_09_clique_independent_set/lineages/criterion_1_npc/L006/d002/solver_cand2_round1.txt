import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [set() for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n is small enough for typical competitive programming constraints 
    # (implied by the 5s time limit), we use a backtracking approach with pruning.
    
    selected = []
    
    def backtrack(candidate_idx, current_k):
        if current_k == k:
            return True
        
        if candidate_idx > n:
            return False
        
        # Pruning: if remaining candidates are not enough to reach k
        if (n - candidate_idx + 1) < (k - current_k):
            return False
        
        # Try including candidate_idx
        can_add = True
        for s in selected:
            if candidate_idx in adj[s]:
                can_add = False
                break
        
        if can_add:
            selected.append(candidate_idx)
            if backtrack(candidate_idx + 1, current_k + 1):
                return True
            selected.pop()
            
        # Try excluding candidate_idx
        if backtrack(candidate_idx + 1, current_k):
            return True
            
        return False

    # Heuristic: sort nodes by degree to prune faster
    # But for simplicity and correctness, standard backtracking is fine.
    if backtrack(1, 0):
        print("YES")
        print(*(selected))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
