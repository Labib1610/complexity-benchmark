import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find a subset of size k such that no two elements in the subset
    # have an edge between them. This is the Maximum Independent Set problem.
    # Since n <= 46, we can use a backtracking approach with pruning.
    
    def get_independent_set(candidates, current_set):
        if len(current_set) == k:
            return current_set
        
        if not candidates:
            return None
        
        # Heuristic: pick the vertex with the highest degree in the subgraph
        # or simply pick the first one.
        v = candidates[0]
        
        # Option 1: Include v in the set
        new_candidates = [c for c in candidates[1:] if not (adj[v] & (1 << c))]
        res = get_independent_set(new_candidates, current_set + [v + 1])
        if res:
            return res
        
        # Option 2: Exclude v from the set
        res = get_independent_set(candidates[1:], current_set)
        if res:
            return res
            
        return None

    # Optimization: If k is large, it's easier to find a clique in the complement graph.
    # But n=46 is small enough for a well-pruned search.
    result = get_independent_set(list(range(n)), [])
    
    if result:
        print("YES")
        print(*(sorted(result)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
