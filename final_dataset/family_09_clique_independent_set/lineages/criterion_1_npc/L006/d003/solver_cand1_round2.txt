import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # Meet-in-the-middle for Maximum Independent Set of size k
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the two halves
    # sets1[count] = list of bitmasks of size 'count'
    sets1 = [[] for _ in range(n1 + 1)]
    for i in range(1 << n1):
        valid = True
        count = 0
        for j in range(n1):
            if (i >> j) & 1:
                count += 1
                if adj[j] & i:
                    valid = False
                    break
        if valid:
            sets1[count].append(i)
            
    sets2 = [[] for _ in range(n2 + 1)]
    for i in range(1 << n2):
        valid = True
        count = 0
        mask = i << n1
        for j in range(n2):
            if (i >> j) & 1:
                count += 1
                if adj[j + n1] & mask:
                    valid = False
                    break
        if valid:
            sets2[count].append(i)

    # Check compatibility between halves
    # For each mask in sets1, check which nodes in the second half are compatible
    adj_in_second = [0] * len(sets1[0]) # Placeholder
    
    for c1 in range(n1 + 1):
        if c1 > k: continue
        c2 = k - c1
        if c2 < 0 or c2 > n2: continue
        
        for m1 in sets1[c1]:
            # Find nodes in second half compatible with m1
            allowed_mask = (1 << n2) - 1
            for j in range(n1):
                if (m1 >> j) & 1:
                    allowed_mask &= ~(adj[j] >> n1)
            
            for m2 in sets2[c2]:
                if (m2 & allowed_mask) == m2:
                    # Found it
                    res = []
                    for j in range(n1):
                        if (m1 >> j) & 1: res.append(j + 1)
                    for j in range(n2):
                        if (m2 >> j) & 1: res.append(j + n1 + 1)
                    print("YES")
                    print(*(res))
                    return

    print("NO")

solve()
