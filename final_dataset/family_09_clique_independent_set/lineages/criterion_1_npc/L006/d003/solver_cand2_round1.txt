import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    k = int(input_data[1])
    a = [int(x) for x in input_data[2:]]
    
    # We need to select k elements such that their sum is divisible by k.
    # This is a variation of the subset sum problem.
    # dp[i][j][rem] = max sum of j elements from first i elements with sum % k == rem.
    # Since we want the lexicographically largest, we should process elements
    # and keep track of the best subset.
    
    # Given the constraints usually associated with this type of problem,
    # we use DP to find the maximum sum for each remainder.
    # dp[count][remainder] = max_sum
    
    dp = [[-1] * k for _ in range(k + 1)]
    dp[0][0] = 0
    
    # To reconstruct the lexicographically largest, we store the choices.
    # However, if n is small, we can use bitmask or simple DP.
    # If n is large, we need a different approach.
    # Assuming standard competitive programming constraints:
    
    # Let's use a list of sets to keep track of possible sums for each (count, remainder)
    # To get lexicographically largest, we sort the input array descending.
    a.sort(reverse=True)
    
    # dp[count][remainder] = max sum
    dp = [[-1] * k for _ in range(k + 1)]
    dp[0][0] = 0
    
    # To track the actual elements, we can store the path or use a bitset.
    # Given the "lexicographically largest" requirement, we want the largest elements first.
    # If we sort a descending, we want to pick elements as early as possible.
    
    # Standard DP for subset sum:
    # dp[i][j] = max sum using j elements with remainder i
    dp = [[-1] * k for _ in range(k + 1)]
    dp[0][0] = 0
    
    # To reconstruct, we store which elements were used.
    # Since we want lexicographically largest, we sort a descending.
    # If we pick a[i], we check if it leads to a valid state.
    
    # Actually, the simplest way to get the lexicographically largest:
    # Try to include a[i] if it's possible to complete the subset of size k
    # with remainder 0 using the remaining elements.
    
    a.sort(reverse=True)
    
    def can_reach(idx, count, rem):
        if count == 0:
            return rem == 0
        if idx == n or count < 0:
            return False
        # This is still subset sum, use memoization
        return memo.get((idx, count, rem), False)

    memo = {}
    
    # Precompute reachability
    # dp[idx][count][rem]
    possible = [[[False] * k for _ in range(k + 1)] for _ in range(n + 1)]
    possible[n][0][0] = True
    for i in range(n - 1, -1, -1):
        for j in range(k + 1):
            for r in range(k):
                # Don't include a[i]
                if possible[i+1][j][r]:
                    possible[i][j][r] = True
                # Include a[i]
                if j > 0:
                    prev_r = (r - a[i]) % k
                    if possible[i+1][j-1][prev_r]:
                        possible[i][j][r] = True
                        
    res = []
    curr_rem = 0
    curr_k = k
    for i in range(n):
        if curr_k > 0:
            # Can we include a[i]?
            target_rem = (curr_rem - a[i]) % k
            if possible[i+1][curr_k-1][target_rem]:
                res.append(a[i])
                curr_rem = target_rem
                curr_k -= 1
                
    if len(res) == k and sum(res) % k == 0:
        print(*(sorted(res, reverse=True)))
    else:
        print("-1")

solve()
