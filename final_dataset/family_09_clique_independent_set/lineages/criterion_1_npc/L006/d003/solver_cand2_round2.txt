import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Max Independent Set / Clique problem variant
    # Since n <= 46, we use a meet-in-the-middle approach or optimized backtracking
    # Given the constraints and time limit, a Bron-Kerbosch style or simple backtracking
    # with bitmask pruning is efficient enough.
    
    n1 = n // 2
    n2 = n - n1
    
    # adj1: adjacency for first n1 nodes, adj2: for remaining n2 nodes
    # We want to find a subset of size k.
    
    def get_independent_sets(nodes, adj_list):
        # Returns a dictionary mapping size -> list of bitmasks
        res = {i: [] for i in range(len(nodes) + 1)}
        res[0].append(0)
        
        for i, node in enumerate(nodes):
            mask = 1 << i
            new_res = {size: list(masks) for size, masks in res.items()}
            for size in range(len(nodes)):
                for m in res[size]:
                    if not (m & adj_list[i]):
                        new_res[size + 1].append(m | mask)
            res = new_res
        return res

    # Split nodes into two halves
    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    adj1 = [adj[i] & ((1 << n1) - 1) for i in nodes1]
    adj2 = [(adj[i] >> n1) for i in nodes2]
    
    # Cross adjacency
    cross = [(adj[i] >> n1) for i in nodes1]
    
    sets1 = get_independent_sets(nodes1, adj1)
    sets2 = get_independent_sets(nodes2, adj2)
    
    for s1 in range(max(0, k - n2), min(k, n1) + 1):
        s2 = k - s1
        for m1 in sets1[s1]:
            # Find nodes in second half compatible with m1
            forbidden = 0
            for i in range(n1):
                if (m1 >> i) & 1:
                    forbidden |= cross[i]
            
            for m2 in sets2[s2]:
                if not (m2 & forbidden):
                    # Found it
                    res = []
                    for i in range(n1):
                        if (m1 >> i) & 1:
                            res.append(i + 1)
                    for i in range(n2):
                        if (m2 >> i) & 1:
                            res.append(i + n1 + 1)
                    print("YES")
                    print(*(res))
                    return
    print("NO")

solve()
