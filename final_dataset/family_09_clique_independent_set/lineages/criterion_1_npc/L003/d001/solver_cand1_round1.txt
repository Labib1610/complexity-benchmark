import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # The problem is to find an independent set of size k in the graph.
    # Since n <= 46, we use a meet-in-the-middle approach.
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, adj_list):
        # nodes is a list of indices
        size = len(nodes)
        res = {} # count -> list of bitmasks
        for i in range(1 << size):
            mask = 0
            valid = True
            count = 0
            for j in range(size):
                if (i >> j) & 1:
                    if adj_list[nodes[j]] & mask:
                        valid = False
                        break
                    mask |= (1 << nodes[j])
                    count += 1
            if valid:
                if count not in res:
                    res[count] = []
                res[count].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Precompute adjacency for sets2 to check compatibility
    # adj_mask2[mask] is the bitmask of nodes in 0..n1-1 that are connected to any node in mask
    adj_mask2 = {}
    for count, masks in sets2.items():
        for m2 in masks:
            forbidden = 0
            for i in range(n2):
                if (m2 >> (i + n1)) & 1:
                    forbidden |= adj[i + n1]
            adj_mask2[m2] = forbidden & ((1 << n1) - 1)

    for c1 in range(k + 1):
        c2 = k - c1
        if c1 in sets1 and c2 in sets2:
            for m1 in sets1[c1]:
                # Find m2 such that (adj[m1] & m2) == 0 and (adj_mask2[m2] & m1) == 0
                # Which is equivalent to m2 not having any bits in adj[m1]
                # and m1 not having any bits in adj_mask2[m2]
                for m2 in sets2[c2]:
                    if not (adj[m1] & m2) and not (adj_mask2[m2] & m1):
                        print("YES")
                        res = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                res.append(str(i + 1))
                        print(" ".join(res))
                        return

    print("NO")

solve()
