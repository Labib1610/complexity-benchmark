import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # The problem is equivalent to finding an independent set of size k in the graph.
    # Since n <= 46, we use a meet-in-the-middle approach.
    n1 = n // 2
    n2 = n - n1
    
    def get_subsets(nodes, offset):
        # nodes is a list of indices
        # returns a dict: count of set bits -> list of bitmasks
        size = len(nodes)
        res = {}
        for i in range(1 << size):
            mask = 0
            bits = 0
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    u = nodes[j]
                    if (mask & adj[u]):
                        valid = False
                        break
                    mask |= (1 << u)
                    bits += 1
            if valid:
                if bits not in res:
                    res[bits] = []
                res[bits].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    subsets1 = get_subsets(nodes1, 0)
    subsets2 = get_subsets(nodes2, n1)
    
    # Precompute adjacency for second half to check against first half
    adj2 = [adj[i] & ((1 << n1) - 1) for i in range(n1, n)]
    
    for k1 in range(min(k, n1) + 1):
        k2 = k - k1
        if k2 in subsets2:
            for m1 in subsets1.get(k1, []):
                # Check if m1 is compatible with any m2 in subsets2[k2]
                # m1 is compatible with m2 if (m1 & adj[v]) == 0 for all v in m2
                # This is equivalent to: m1 & (OR of adj[v] for all v in m2) == 0
                # Precompute OR of adj[v] for each m2
                for m2 in subsets2[k2]:
                    # Check compatibility
                    conflict = False
                    for i in range(n2):
                        if (m2 >> (i + n1)) & 1:
                            if (m1 & adj[i + n1]):
                                conflict = True
                                break
                    if not conflict:
                        # Found it
                        res = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                res.append(i + 1)
                        print("YES")
                        print(*(res))
                        return

    print("NO")

solve()
