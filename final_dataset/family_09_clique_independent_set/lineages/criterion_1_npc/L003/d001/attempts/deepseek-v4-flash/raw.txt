We need to choose exactly `k` vertices such that no edge has both endpoints chosen.  
That means the chosen set must be an **independent set** of size exactly `k` in the given graph.

Since `n ≤ 46`, we can use a **meet-in-the-middle** approach:

- Split vertices into two halves.
- For each half, enumerate all independent sets and store their bitmasks and sizes.
- For each independent set in the first half, find a compatible independent set in the second half (no edges between the halves) such that total size = `k`.

We’ll implement this efficiently.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Split into two halves
    n1 = n // 2
    n2 = n - n1
    left = list(range(n1))
    right = list(range(n1, n))

    # Precompute adjacency masks for each half
    adj_left = [0] * n1
    adj_right = [0] * n2
    for i in range(n1):
        u = left[i]
        adj_left[i] = adj[u] & ((1 << n1) - 1)
    for i in range(n2):
        u = right[i]
        adj_right[i] = (adj[u] >> n1) & ((1 << n2) - 1)

    # Enumerate independent sets in left half
    left_sets = {}  # size -> list of (mask, cross_mask)
    for mask in range(1 << n1):
        ok = True
        for i in range(n1):
            if mask & (1 << i):
                if mask & adj_left[i]:
                    ok = False
                    break
        if ok:
            size = bin(mask).count("1")
            cross = 0
            for i in range(n1):
                if mask & (1 << i):
                    cross |= (adj[left[i]] >> n1)
            left_sets.setdefault(size, []).append((mask, cross))

    # Enumerate independent sets in right half
    right_sets = {}  # size -> list of mask
    for mask in range(1 << n2):
        ok = True
        for i in range(n2):
            if mask & (1 << i):
                if mask & adj_right[i]:
                    ok = False
                    break
        if ok:
            size = bin(mask).count("1")
            right_sets.setdefault(size, []).append(mask)

    # Try to combine
    for size_l, lst_l in left_sets.items():
        need = k - size_l
        if need < 0 or need not in right_sets:
            continue
        for mask_l, cross in lst_l:
            for mask_r in right_sets[need]:
                if cross & mask_r == 0:
                    # Found solution
                    chosen = []
                    for i in range(n1):
                        if mask_l & (1 << i):
                            chosen.append(left[i] + 1)
                    for i in range(n2):
                        if mask_r & (1 << i):
                            chosen.append(right[i] + 1)
                    print("YES")
                    print(" ".join(map(str, chosen)))
                    return

    print("NO")

if __name__ == "__main__":
    solve()
```
