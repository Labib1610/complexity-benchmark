from itertools import combinations
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    gates = []
    idx = 3
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        gates.append((u, v))
        idx += 2
    
    # Try all combinations of k pins to set HIGH
    for combo in combinations(range(1, n+1), k):
        high_set = set(combo)
        valid = True
        for u, v in gates:
            # A gate outputs LOW only if both inputs are HIGH
            if u in high_set and v in high_set:
                valid = False
                break
        if valid:
            print("YES")
            print(' '.join(map(str, combo)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
