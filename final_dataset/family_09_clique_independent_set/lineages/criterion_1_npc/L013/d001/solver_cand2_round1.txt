import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * (n + 1)
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    
    def find_is(candidates, current_set):
        if len(current_set) == k:
            return current_set
        
        if not candidates:
            return None
        
        # Heuristic: pick the node with the fewest candidates to branch on
        # Or simply pick the first one
        v = -1
        for i in range(n):
            if (candidates >> i) & 1:
                v = i
                break
        
        if v == -1:
            return None
        
        # Option 1: Include v
        new_candidates = candidates & ~adj[v+1] & ~(1 << v)
        res = find_is(new_candidates, current_set + [v + 1])
        if res:
            return res
            
        # Option 2: Exclude v
        res = find_is(candidates & ~(1 << v), current_set)
        return res

    result = find_is((1 << n) - 1, [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
