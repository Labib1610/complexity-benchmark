import sys

sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    ALL = (1 << n) - 1
    solution = []

    # order vertices by degree descending for better pruning
    order = sorted(range(n), key=lambda x: bin(adj[x]).count("1"), reverse=True)
    pos_in_order = [0] * n
    for i, v in enumerate(order):
        pos_in_order[v] = i

    # remap adjacency to ordered indices
    adj2 = [0] * n
    for i in range(n):
        mask = 0
        for j in range(n):
            if adj[i] >> j & 1:
                mask |= 1 << pos_in_order[j]
        adj2[pos_in_order[i]] = mask

    # recursive search
    def dfs(idx: int, chosen_mask: int, chosen_cnt: int, cand_mask: int) -> bool:
        # pruning
        if chosen_cnt == k:
            # found
            nonlocal solution
            solution = [order[i] + 1 for i in range(n) if (chosen_mask >> i) & 1]
            return True
        if idx >= n:
            return False
        # maximum possible vertices we can still take
        remaining = bin(cand_mask).count("1")
        if chosen_cnt + remaining < k:
            return False

        # pick next vertex from candidates (smallest index)
        # find first set bit in cand_mask
        v = (cand_mask & -cand_mask).bit_length() - 1

        # branch: include v
        new_cand = cand_mask & ~((1 << v) | adj2[v])
        if dfs(idx + 1, chosen_mask | (1 << v), chosen_cnt + 1, new_cand):
            return True

        # branch: exclude v
        cand_mask_without_v = cand_mask & ~(1 << v)
        if dfs(idx + 1, chosen_mask, chosen_cnt, cand_mask_without_v):
            return True

        return False

    initial_cand = (1 << n) - 1
    if dfs(0, 0, 0, initial_cand):
        print("YES")
        print(" ".join(map(str, solution)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
