import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # The problem is to find an independent set of size k in a graph with n <= 46.
    # We use a meet-in-the-middle approach or a simple backtracking with pruning.
    # Given n=46, a simple backtracking with bitmasking is efficient enough.
    
    def find_is(candidates, current_set, size_needed):
        if size_needed == 0:
            return current_set
        if not candidates or len(candidates) < size_needed:
            return None
        
        # Heuristic: pick the vertex with the smallest degree in the subgraph
        v = candidates[0]
        
        # Option 1: Include v
        new_candidates = [c for c in candidates[1:] if not (adj[v] & (1 << c))]
        res = find_is(new_candidates, current_set + [v + 1], size_needed - 1)
        if res:
            return res
        
        # Option 2: Exclude v
        res = find_is(candidates[1:], current_set, size_needed)
        return res

    # Sort vertices by degree to improve pruning
    nodes = sorted(range(n), key=lambda x: bin(adj[x]).count('1'))
    
    result = find_is(nodes, [], k)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
