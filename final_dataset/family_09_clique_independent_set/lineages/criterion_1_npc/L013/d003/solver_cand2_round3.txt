import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [set() for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # The problem is to find an Independent Set of size k.
    # Since n is small (max 46 for human diploid chromosomes), 
    # we can use a backtracking approach with pruning.
    
    selected = []
    
    def backtrack(start_node, current_k):
        if current_k == k:
            return True
        
        # Heuristic: remaining nodes must be at least needed
        if (n - start_node + 1) < (k - current_k):
            return False
            
        for i in range(start_node, n + 1):
            # Check if i can be added
            can_add = True
            for s in selected:
                if i in adj[s]:
                    can_add = False
                    break
            
            if can_add:
                selected.append(i)
                if backtrack(i + 1, current_k + 1):
                    return True
                selected.pop()
        return False

    # Optimization: sort nodes by degree (smallest first)
    # This is a common heuristic for Maximum Independent Set
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]))
    
    # Re-map adjacency for sorted nodes
    new_adj = [set() for _ in range(n + 1)]
    mapping = {old: new for new, old in enumerate(nodes, 1)}
    rev_mapping = {new: old for new, old in enumerate(nodes, 1)}
    
    for u in range(1, n + 1):
        for v in adj[u]:
            new_adj[mapping[u]].add(mapping[v])
            
    final_selected = []
    def backtrack_optimized(idx, count):
        if count == k:
            return True
        if idx > n or (n - idx + 1) < (k - count):
            return False
        
        # Try including idx
        can_add = True
        for s in final_selected:
            if idx in new_adj[s]:
                can_add = False
                break
        
        if can_add:
            final_selected.append(idx)
            if backtrack_optimized(idx + 1, count + 1):
                return True
            final_selected.pop()
            
        # Try excluding idx
        if backtrack_optimized(idx + 1, count):
            return True
            
        return False

    if backtrack_optimized(1, 0):
        print("YES")
        print(*(sorted([rev_mapping[x] for x in final_selected])))
    else:
        print("NO")

solve()
