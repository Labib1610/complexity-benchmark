import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # The problem is to find an Independent Set of size k.
    # Since n <= 46, we use a meet-in-the-middle approach.
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, adj_list):
        # returns a dict: size -> list of bitmasks
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        
        def backtrack(idx, current_mask, current_size):
            if idx == size:
                res[current_size].append(current_mask)
                return
            
            # Option 1: Don't include nodes[idx]
            backtrack(idx + 1, current_mask, current_size)
            
            # Option 2: Include nodes[idx] if possible
            if not (adj_list[nodes[idx]] & current_mask):
                backtrack(idx + 1, current_mask | (1 << nodes[idx]), current_size + 1)
        
        backtrack(0, 0, 0)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Precompute adjacency for sets2 to check compatibility
    adj2 = [adj[i] for i in nodes2]
    
    for s1 in range(k + 1):
        s2 = k - s1
        if s1 < len(sets1) and s2 < len(sets2):
            for mask1 in sets1[s1]:
                # Find nodes in nodes2 that are not connected to any node in mask1
                forbidden = 0
                for i in range(n1):
                    if (mask1 >> i) & 1:
                        forbidden |= adj[i]
                
                for mask2 in sets2[s2]:
                    if not (mask2 & forbidden):
                        # Found it
                        res = []
                        for i in range(n1):
                            if (mask1 >> i) & 1:
                                res.append(i + 1)
                        for i in range(n2):
                            if (mask2 >> (i + n1)) & 1:
                                res.append(i + n1 + 1)
                        print("YES")
                        print(*(res))
                        return
    print("NO")

solve()
