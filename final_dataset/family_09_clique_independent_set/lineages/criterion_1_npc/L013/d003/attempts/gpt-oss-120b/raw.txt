```python
import sys

sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    all_mask = (1 << n) - 1

    # memoization for dead states (mask, need) -> False
    from functools import lru_cache

    @lru_cache(maxsize=None)
    def dfs(mask: int, need: int):
        if need == 0:
            return 0  # empty set works
        if need < 0:
            return None
        cnt = mask.bit_count()
        if cnt < need:
            return None

        # choose a vertex with maximum degree in mask to improve pruning
        # simple heuristic: pick first set bit
        v = (mask & -mask).bit_length() - 1

        # try include v
        new_mask = mask & ~ (adj[v] | (1 << v))
        if new_mask.bit_count() >= need - 1:
            res = dfs(new_mask, need - 1)
            if res is not None:
                return res | (1 << v)

        # try exclude v
        new_mask = mask & ~ (1 << v)
        if new_mask.bit_count() >= need:
            res = dfs(new_mask, need)
            if res is not None:
                return res

        return None

    ans_mask = dfs(all_mask, k)
    if ans_mask is None:
        print("NO")
    else:
        res = [str(i + 1) for i in range(n) if (ans_mask >> i) & 1]
        print("YES")
        print(" ".join(res))

if __name__ == "__main__":
    solve()
```
