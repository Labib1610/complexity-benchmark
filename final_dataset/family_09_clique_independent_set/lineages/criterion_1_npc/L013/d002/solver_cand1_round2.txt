import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1] |= (1 << (v-1))
        adj[v-1] |= (1 << (u-1))

    n1 = n // 2
    n2 = n - n1

    # Map: count -> list of bitmasks
    left_subsets = [[] for _ in range(n1 + 1)]
    for i in range(1 << n1):
        valid = True
        count = 0
        for j in range(n1):
            if (i >> j) & 1:
                count += 1
                if adj[j] & i:
                    valid = False
                    break
        if valid:
            left_subsets[count].append(i)

    right_subsets = [[] for _ in range(n2 + 1)]
    for i in range(1 << n2):
        valid = True
        count = 0
        mask = i << n1
        for j in range(n2):
            if (i >> j) & 1:
                count += 1
                if adj[j + n1] & mask:
                    valid = False
                    break
        if valid:
            right_subsets[count].append(i << n1)

    # Precompute adjacency for right side relative to left
    # adj_right_to_left[i_right] = mask of left nodes connected to i_right
    adj_right_to_left = [0] * (1 << n2)
    for i in range(n2):
        for j in range(n1):
            if (adj[i + n1] >> j) & 1:
                adj_right_to_left[1 << i] |= (1 << j)
    
    for i in range(1, 1 << n2):
        lowbit = i & -i
        adj_right_to_left[i] = adj_right_to_left[i ^ lowbit] | adj_right_to_left[lowbit]

    for c1 in range(n1 + 1):
        c2 = k - c1
        if 0 <= c2 <= n2:
            for left_mask in left_subsets[c1]:
                for right_mask in right_subsets[c2]:
                    if not (left_mask & adj_right_to_left[(right_mask >> n1)]):
                        res = []
                        for i in range(n):
                            if ((left_mask | right_mask) >> i) & 1:
                                res.append(i + 1)
                        print("YES")
                        print(*(res))
                        return

    print("NO")

solve()
