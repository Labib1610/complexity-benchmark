import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
        
    # The problem asks to select k junctions as privacy zones such that
    # every edge has at least one endpoint NOT in the privacy zones.
    # This is equivalent to saying the set of privacy zones must be an
    # Independent Set of the graph.
    # We need to find an Independent Set of size exactly k.
    
    # Since n <= 46, we can use a meet-in-the-middle approach or a 
    # backtracking search with pruning.
    
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    def get_independent_sets(nodes):
        # Returns a dictionary mapping size to a list of bitmasks
        res = {i: [] for i in range(len(nodes) + 1)}
        res[0].append(0)
        
        for i, node in enumerate(nodes):
            new_res = {size: list(masks) for size, masks in res.items()}
            for size, masks in res.items():
                for mask in masks:
                    if not (mask & adj[node]):
                        new_mask = mask | (1 << node)
                        new_res[size + 1].append(new_mask)
            res = new_res
        return res

    n1 = n // 2
    n2 = n - n1
    
    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1)
    sets2 = get_independent_sets(nodes2)
    
    # Precompute adjacency for sets2 relative to sets1
    # For each mask in sets2, check compatibility with masks in sets1
    # A mask1 and mask2 are compatible if no edge exists between them
    # mask1 is in nodes1, mask2 is in nodes2.
    # Edge exists if (mask1 & adj_in_nodes1_for_node_in_mask2) or (mask2 & adj_in_nodes2_for_node_in_mask1)
    
    adj1 = [adj[i] & ((1 << n1) - 1) for i in range(n1)]
    adj2 = [adj[i] >> n1 for i in range(n1, n)]
    
    for s1 in range(k + 1):
        s2 = k - s1
        if s1 in sets1 and s2 in sets2:
            for m1 in sets1[s1]:
                # Find nodes in m1 that have neighbors in nodes2
                forbidden_in_nodes2 = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        forbidden_in_nodes2 |= adj2[i]
                
                for m2 in sets2[s2]:
                    if not (m2 & forbidden_in_nodes2):
                        # Found it
                        res = []
                        for i in range(n1):
                            if (m1 >> i) & 1: res.append(i + 1)
                        for i in range(n2):
                            if (m2 >> i) & 1: res.append(i + n1 + 1)
                        print("YES")
                        print(*(res))
                        return

    print("NO")

solve()
