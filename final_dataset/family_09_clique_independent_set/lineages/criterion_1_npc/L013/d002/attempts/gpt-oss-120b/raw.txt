```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    ALL = (1 << n) - 1

    # memoization could help but not necessary for n<=46
    from functools import lru_cache

    @lru_cache(maxsize=None)
    def popcnt(x: int) -> int:
        return x.bit_count()

    # recursive search
    def dfs(chosen_mask: int, candidates: int) -> int:
        """return mask of chosen vertices of size k, or -1 if impossible"""
        chosen_cnt = popcnt(chosen_mask)
        if chosen_cnt == k:
            return chosen_mask
        if candidates == 0:
            return -1
        # prune if even taking all candidates we cannot reach k
        if chosen_cnt + popcnt(candidates) < k:
            return -1

        # select a vertex (heuristic: vertex with maximum degree among candidates)
        # simple: take first bit
        v = (candidates & -candidates).bit_length() - 1

        # Branch 1: include v
        new_candidates = candidates & ~adj[v] & ~(1 << v)
        res = dfs(chosen_mask | (1 << v), new_candidates)
        if res != -1:
            return res

        # Branch 2: exclude v
        candidates_without_v = candidates & ~(1 << v)
        return dfs(chosen_mask, candidates_without_v)

    result_mask = dfs(0, ALL)
    if result_mask == -1:
        print("NO")
    else:
        ans = [str(i + 1) for i in range(n) if (result_mask >> i) & 1]
        print("YES")
        print(" ".join(ans))

if __name__ == "__main__":
    solve()
```
