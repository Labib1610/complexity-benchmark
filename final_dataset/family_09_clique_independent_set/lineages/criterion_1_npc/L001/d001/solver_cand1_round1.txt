import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * (n + 1)
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    
    def find_set(candidates, current_set, target):
        if len(current_set) == target:
            return current_set
        
        if not candidates:
            return None
        
        # Heuristic: pick the candidate with the fewest neighbors in the remaining set
        # or simply pick the first one.
        while candidates:
            # Pruning: if remaining candidates + current_set < target, return None
            if len(current_set) + len(candidates) < target:
                return None
            
            v = candidates[0]
            new_candidates = [c for c in candidates[1:] if not (adj[v] & (1 << (c - 1)))]
            
            res = find_set(new_candidates, current_set + [v], target)
            if res:
                return res
            
            candidates = candidates[1:]
            
        return None

    result = find_set(list(range(1, n + 1)), [], k)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
