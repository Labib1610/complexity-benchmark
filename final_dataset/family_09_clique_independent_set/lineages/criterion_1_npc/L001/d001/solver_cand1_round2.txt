import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle approach for Maximum Independent Set
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, adj_list):
        # returns a dict: size -> list of bitmasks
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        
        def backtrack(idx, current_mask, current_size):
            if idx == size:
                res[current_size].append(current_mask)
                return
            
            # Option 1: Don't include nodes[idx]
            backtrack(idx + 1, current_mask, current_size)
            
            # Option 2: Include nodes[idx] if compatible
            if not (adj_list[nodes[idx]] & current_mask):
                backtrack(idx + 1, current_mask | (1 << nodes[idx]), current_size + 1)
        
        backtrack(0, 0, 0)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Precompute compatibility for sets2
    # mask2 is compatible with mask1 if no node in mask2 conflicts with any in mask1
    for s1 in range(n1 + 1):
        for m1 in sets1[s1]:
            # Find nodes in the second half that don't conflict with m1
            allowed_mask2 = 0
            for i in range(n2):
                node = nodes2[i]
                if not (adj[node] & m1):
                    allowed_mask2 |= (1 << node)
            
            needed = k - s1
            if 0 <= needed <= n2:
                for m2 in sets2[needed]:
                    if (m2 & allowed_mask2) == m2:
                        # Found it
                        res = []
                        for i in range(n):
                            if (m1 | m2) & (1 << i):
                                res.append(i + 1)
                        print("YES")
                        print(*(res))
                        return
    print("NO")

solve()
