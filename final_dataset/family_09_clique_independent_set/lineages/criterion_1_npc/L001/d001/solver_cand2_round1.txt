import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we use a meet-in-the-middle approach or a simple backtracking
    # with pruning (Bron-Kerbosch style or bitmask search).
    
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the two halves
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            subset = []
            valid = True
            mask = 0
            for j in range(size):
                if (i >> j) & 1:
                    u = nodes[j]
                    if adj_list[u] & mask:
                        valid = False
                        break
                    mask |= (1 << u)
                    subset.append(u)
            if valid:
                res[len(subset)].append((mask, subset))
        return res

    sets1 = get_independent_sets(list(range(n1)), adj)
    sets2 = get_independent_sets(list(range(n1, n)), adj)
    
    for s1 in range(k + 1):
        s2 = k - s1
        if s1 < len(sets1) and s2 < len(sets2):
            for mask1, list1 in sets1[s1]:
                for mask2, list2 in sets2[s2]:
                    # Check if any conflict between the two sets
                    conflict = False
                    for u in list1:
                        if adj[u] & mask2:
                            conflict = True
                            break
                    if not conflict:
                        print("YES")
                        print(*(x + 1 for x in (list1 + list2)))
                        return

    print("NO")

solve()
