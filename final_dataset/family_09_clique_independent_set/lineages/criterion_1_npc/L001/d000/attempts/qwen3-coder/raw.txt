from itertools import combinations
import sys

def main():
    n, m, k = map(int, sys.stdin.readline().split())
    
    # Create adjacency list representation of the graph
    # where edges represent reactive pairs
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].add(v)
        adj[v].add(u)
    
    # Generate all possible combinations of k substances
    substances = list(range(1, n + 1))
    
    # For small n, we can check all combinations
    # But for larger n, we need a more efficient approach
    
    # Since n <= 46, we can't check all combinations for large k
    # We'll use a meet-in-the-middle approach for larger cases
    
    if n <= 25:
        # Brute force for smaller n
        for combo in combinations(substances, k):
            # Check if this combination is stable
            valid = True
            for i in range(k):
                for j in range(i + 1, k):
                    if combo[j] in adj[combo[i]]:
                        valid = False
                        break
                if not valid:
                    break
            if valid:
                print("YES")
                print(*combo)
                return
        
        print("NO")
        return
    else:
        # Meet-in-the-middle approach for larger n
        # Split substances into two halves
        mid = n // 2
        left_substances = list(range(1, mid + 1))
        right_substances = list(range(mid + 1, n + 1))
        
        # Generate all valid subsets of left half
        left_valid = {}
        for size in range(min(k + 1, len(left_substances) + 1)):
            for combo in combinations(left_substances, size):
                # Check if combo is valid (no internal reactions)
                valid = True
                for i in range(len(combo)):
                    for j in range(i + 1, len(combo)):
                        if combo[j] in adj[combo[i]]:
                            valid = False
                            break
                    if not valid:
                        break
                if valid:
                    left_valid[combo] = size
        
        # Generate all valid subsets of right half
        right_valid = {}
        for size in range(min(k + 1, len(right_substances) + 1)):
            for combo in combinations(right_substances, size):
                # Check if combo is valid (no internal reactions)
                valid = True
                for i in range(len(combo)):
                    for j in range(i + 1, len(combo)):
                        if combo[j] in adj[combo[i]]:
                            valid = False
                            break
                    if not valid:
                        break
                if valid:
                    right_valid[combo] = size
        
        # Check all combinations of left and right subsets
        for left_combo, left_size in left_valid.items():
            for right_combo, right_size in right_valid.items():
                if left_size + right_size == k:
                    # Check if there are no reactions between left and right
                    valid = True
                    for u in left_combo:
                        for v in right_combo:
                            if v in adj[u]:
                                valid = False
                                break
                        if not valid:
                            break
                    if valid:
                        result = list(left_combo) + list(right_combo)
                        print("YES")
                        print(*result)
                        return
        
        print("NO")

if __name__ == "__main__":
    main()
