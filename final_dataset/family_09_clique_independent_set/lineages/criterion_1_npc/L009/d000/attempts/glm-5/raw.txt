import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    if k == 0:
        print("YES")
        print()
        return
    
    if k == 1:
        print("YES")
        print(1)
        return
    
    half = n // 2
    left_size = half
    right_size = n - half
    
    left_nodes = list(range(1, left_size + 1))
    right_nodes = list(range(left_size + 1, n + 1))
    
    left_mask = {}
    for i, v in enumerate(left_nodes):
        left_mask[v] = 1 << i
    
    right_mask = {}
    for i, v in enumerate(right_nodes):
        right_mask[v] = 1 << i
    
    left_adj = [0] * left_size
    for i, v in enumerate(left_nodes):
        for u in left_nodes:
            if adj[v] & (1 << u):
                left_adj[i] |= left_mask[u]
    
    right_adj = [0] * right_size
    for i, v in enumerate(right_nodes):
        for u in right_nodes:
            if adj[v] & (1 << u):
                right_adj[i] |= right_mask[u]
    
    cross_adj = [0] * left_size
    for i, v in enumerate(left_nodes):
        for u in right_nodes:
            if adj[v] & (1 << u):
                cross_adj[i] |= right_mask[u]
    
    left_solutions = {}
    left_solutions[0] = 0
    
    for i in range(left_size):
        new_solutions = {}
        for mask, cnt in left_solutions.items():
            if cnt + 1 <= k and (mask & left_adj[i]) == 0:
                new_mask = mask | (1 << i)
                if new_mask not in left_solutions and new_mask not in new_solutions:
                    new_solutions[new_mask] = cnt + 1
        left_solutions.update(new_solutions)
    
    right_solutions = {}
    right_solutions[0] = 0
    
    for i in range(right_size):
        new_solutions = {}
        for mask, cnt in right_solutions.items():
            if cnt + 1 <= k and (mask & right_adj[i]) == 0:
                new_mask = mask | (1 << i)
                if new_mask not in right_solutions and new_mask not in new_solutions:
                    new_solutions[new_mask] = cnt + 1
        right_solutions.update(new_solutions)
    
    right_by_count = {}
    for mask, cnt in right_solutions.items():
        if cnt not in right_by_count:
            right_by_count[cnt] = []
        right_by_count[cnt].append(mask)
    
    for cnt in right_by_count:
        right_by_count[cnt].sort()
    
    for lmask, lcnt in left_solutions.items():
        rcnt = k - lcnt
        if rcnt < 0 or rcnt > right_size:
            continue
        
        forbidden = 0
        for i in range(left_size):
            if lmask & (1 << i):
                forbidden |= cross_adj[i]
        
        needed = (~forbidden) & ((1 << right_size) - 1)
        
        if rcnt not in right_by_count:
            continue
        
        masks = right_by_count[rcnt]
        lo, hi = 0, len(masks)
        while lo < hi:
            mid = (lo + hi) // 2
            if masks[mid] < needed:
                lo = mid + 1
            else:
                hi = mid
        
        if lo < len(masks) and (masks[lo] & forbidden) == 0:
            rmask = masks[lo]
            result = []
            for i in range(left_size):
                if lmask & (1 << i):
                    result.append(left_nodes[i])
            for i in range(right_size):
                if rmask & (1 << i):
                    result.append(right_nodes[i])
            print("YES")
            print(' '.join(map(str, result)))
            return
    
    print("NO")

solve()
