import sys
sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    full_mask = (1 << n) - 1
    solution = None

    # order vertices by degree descending for better pruning
    order = sorted(range(n), key=lambda x: bin(adj[x]).count('1'), reverse=True)
    # map original index to position in ordered list
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i

    # rebuild adjacency in ordered space
    adj2 = [0] * n
    for i in range(n):
        v = order[i]
        mask = 0
        for u in range(n):
            if adj[v] >> u & 1:
                mask |= 1 << pos[u]
        adj2[i] = mask

    # initial candidate set: all vertices
    candidates = (1 << n) - 1

    # store current chosen vertices as mask in ordered space
    chosen = 0

    # simple recursive backtracking
    def dfs(chosen_mask: int, cand_mask: int, need: int) -> bool:
        nonlocal solution
        if need == 0:
            solution = chosen_mask
            return True
        # prune
        if cand_mask.bit_count() < need:
            return False
        # choose a vertex with max degree heuristic
        # pick first vertex in cand_mask
        v = (cand_mask & -cand_mask).bit_length() - 1
        # branch: include v
        new_chosen = chosen_mask | (1 << v)
        new_cand = cand_mask & ~adj2[v] & ~(1 << v)
        if dfs(new_chosen, new_cand, need - 1):
            return True
        # branch: exclude v
        cand_mask &= ~(1 << v)
        if dfs(chosen_mask, cand_mask, need):
            return True
        return False

    dfs(chosen, candidates, k)

    if solution is None:
        print("NO")
    else:
        # translate back to original numbering
        res = []
        for i in range(n):
            if solution >> i & 1:
                res.append(order[i] + 1)
        print("YES")
        print(' '.join(map(str, res)))

if __name__ == "__main__":
    solve()
