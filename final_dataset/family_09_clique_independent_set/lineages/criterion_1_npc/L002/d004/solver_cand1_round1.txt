import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2
        
    # Let x_i be 1 if node i is positive, 0 otherwise.
    # Local load L_ij = x_i + x_j.
    # Equilibrium: sum(L_ij^2) = sum(L_ij)
    # sum((x_i + x_j)^2) = sum(x_i + x_j)
    # sum(x_i^2 + 2*x_i*x_j + x_j^2) = sum(x_i + x_j)
    # Since x_i is 0 or 1, x_i^2 = x_i.
    # sum(x_i + x_j + 2*x_i*x_j) = sum(x_i + x_j)
    # sum(2*x_i*x_j) = 0
    # Since x_i, x_j >= 0, this means for every edge (i, j), x_i * x_j must be 0.
    # This is equivalent to saying no two positive nodes can be connected by an edge.
    # We need to find an independent set of size k.
    
    if n <= 24:
        for combo in itertools.combinations(range(n), k):
            is_independent = True
            combo_set = set(combo)
            for u, v in adj:
                if u in combo_set and v in combo_set:
                    is_independent = False
                    break
            if is_independent:
                print("YES")
                print(*(c + 1 for c in combo))
                return
        print("NO")
    else:
        # Meet-in-the-middle for larger n
        n1 = n // 2
        n2 = n - n1
        
        adj1 = [(u, v) for u, v in adj if u < n1 and v < n1]
        adj2 = [(u-n1, v-n1) for u, v in adj if u >= n1 and v >= n1]
        cross = [(u, v-n1) for u, v in adj if u < n1 and v >= n1]
        
        def get_independent_sets(nodes_count, edges):
            res = {}
            for i in range(1 << nodes_count):
                subset = [j for j in range(nodes_count) if (i & (1 << j))]
                valid = True
                for u, v in edges:
                    if (i & (1 << u)) and (i & (1 << v)):
                        valid = False
                        break
                if valid:
                    size = len(subset)
                    if size not in res: res[size] = []
                    res[size].append((i, set(subset)))
            return res

        sets1 = get_independent_sets(n1, adj1)
        sets2 = get_independent_sets(n2, adj2)
        
        for k1 in range(k + 1):
            k2 = k - k1
            if k1 in sets1 and k2 in sets2:
                for mask1, nodes1 in sets1[k1]:
                    for mask2, nodes2 in sets2[k2]:
                        valid = True
                        for u, v in cross:
                            if (mask1 & (1 << u)) and (mask2 & (1 << v)):
                                valid = False
                                break
                        if valid:
                            print("YES")
                            res = [i + 1 for i in nodes1] + [i + n1 + 1 for i in nodes2]
                            print(*(res))
                            return
        print("NO")

solve()
