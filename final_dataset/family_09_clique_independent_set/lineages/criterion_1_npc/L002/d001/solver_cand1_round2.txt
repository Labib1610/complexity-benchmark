import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    
    def find_set(candidates, current_set, target):
        if len(current_set) == target:
            return current_set
        
        if not candidates:
            return None
        
        # Heuristic: pick the node with the fewest candidates to branch on
        # Or simply pick the first one.
        # To optimize, we can use Bron-Kerbosch style branching.
        
        # Simple backtracking
        while candidates:
            # Pick the smallest index node
            v = (candidates & -candidates).bit_length() - 1
            
            res = find_set(candidates & ~adj[v] & ~(1 << v), current_set + [v + 1], target)
            if res:
                return res
            
            candidates &= ~(1 << v)
            
            # Pruning: if remaining candidates are not enough to reach target
            if bin(candidates).count('1') < (target - len(current_set)):
                break
        return None

    # Initial candidates: all nodes
    initial_candidates = (1 << n) - 1
    result = find_set(initial_candidates, [], k)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
