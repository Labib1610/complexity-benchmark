from itertools import combinations

def solve():
    n, m, k = map(int, input().split())
    
    # Read transformer connections
    transformers = []
    for _ in range(m):
        u, v = map(int, input().split())
        transformers.append((u, v))
    
    # Try all combinations of k servers
    for servers in combinations(range(1, n + 1), k):
        server_set = set(servers)
        valid = True
        
        # Check if any transformer connects two selected servers
        for u, v in transformers:
            if u in server_set and v in server_set:
                valid = False
                break
        
        if valid:
            print("YES")
            print(*servers)
            return
    
    print("NO")

solve()
