import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    links = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        links.append((u, v))
        idx += 2
        
    # The rule: for each link (u, v), the number of activated traits 
    # must not exceed the number of dormant traits.
    # Let A be the set of activated traits.
    # For a link (u, v):
    # If both u, v in A: activated = 2, dormant = 0. 2 <= 0 is False.
    # If one in A, one not: activated = 1, dormant = 1. 1 <= 1 is True.
    # If neither in A: activated = 0, dormant = 2. 0 <= 2 is True.
    # Thus, the rule is equivalent to: no two activated traits can be connected by a link.
    # This means the set of activated traits must be an Independent Set of size k.
    
    # Since n <= 46, we can use a backtracking approach with pruning or 
    # bitmasking if n were smaller. For n=46, finding an independent set of size k
    # is NP-hard, but the constraints and time limit suggest a search.
    
    adj = [0] * n
    for u, v in links:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    def find_independent_set(k, current_mask, candidates):
        if k == 0:
            return [i for i in range(n) if (current_mask & (1 << i))]
        
        if not candidates:
            return None
        
        # Heuristic: pick the node with the fewest candidates to branch on
        v = candidates[0]
        
        # Option 1: Include v in the independent set
        new_candidates = [c for c in candidates if c != v and not (adj[v] & (1 << c))]
        res = find_independent_set(k - 1, current_mask | (1 << v), new_candidates)
        if res:
            return res
            
        # Option 2: Exclude v
        new_candidates = candidates[1:]
        res = find_independent_set(k, current_mask, new_candidates)
        return res

    # For n=46, simple backtracking might be slow, but let's try with ordering
    # Sort nodes by degree descending to prune faster
    nodes = sorted(range(n), key=lambda x: bin(adj[x]).count('1'), reverse=True)
    
    result = find_independent_set(k, 0, nodes)
    
    if result:
        print("YES")
        print(*(sorted([x + 1 for x in result])))
    else:
        print("NO")

solve()
