import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # The problem asks for an independent set of size k.
    # Since n <= 46, we use a meet-in-the-middle approach.
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        # Iterate through all subsets of nodes
        for i in range(1 << size):
            subset = []
            mask = 0
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    u = nodes[j]
                    if adj_list[u] & mask:
                        valid = False
                        break
                    subset.append(u)
                    mask |= (1 << u)
            if valid:
                res[len(subset)].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Precompute adjacency for nodes2 relative to nodes1
    adj_in_n1 = [adj[i] & ((1 << n1) - 1) for i in range(n1, n)]
    
    for k1 in range(min(k, n1) + 1):
        k2 = k - k1
        if 0 <= k2 <= n2:
            for mask1 in sets1[k1]:
                # Find valid masks in sets2
                # A mask2 is valid if no node in mask2 is adjacent to any node in mask1
                forbidden_in_n2 = 0
                for i in range(n1):
                    if (mask1 >> i) & 1:
                        forbidden_in_n2 |= (adj[i] >> n1)
                
                for mask2 in sets2[k2]:
                    if not (mask2 & forbidden_in_n2):
                        # Found it
                        print("YES")
                        res = []
                        for i in range(n1):
                            if (mask1 >> i) & 1:
                                res.append(i + 1)
                        for i in range(n2):
                            if (mask2 >> (i + n1)) & 1:
                                res.append(i + n1 + 1)
                        print(*(res))
                        return
    print("NO")

solve()
