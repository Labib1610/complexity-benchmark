import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle for Maximum Independent Set variant
    # n <= 46 is too large for 2^(n/2), but we need exactly k.
    # Since n is up to 46, we split into n1 = 23, n2 = 23.
    n1 = n // 2
    n2 = n - n1
    
    # Precompute valid subsets for each half
    def get_subsets(start, size):
        subsets = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            nodes = [start + j for j in range(size) if (i & (1 << j))]
            is_independent = True
            for idx1 in range(len(nodes)):
                for idx2 in range(idx1 + 1, len(nodes)):
                    if (adj[nodes[idx1]] >> nodes[idx2]) & 1:
                        is_independent = False
                        break
                if not is_independent: break
            if is_independent:
                mask = 0
                for node in nodes:
                    mask |= adj[node]
                subsets[len(nodes)].append((i, mask))
        return subsets

    subsets1 = get_subsets(0, n1)
    subsets2 = get_subsets(n1, n2)
    
    for k1 in range(k + 1):
        k2 = k - k1
        if k1 < len(subsets1) and k2 < len(subsets2):
            for mask1_bits, mask1_adj in subsets1[k1]:
                for mask2_bits, mask2_adj in subsets2[k2]:
                    # Check if any node in set 1 is connected to a node in set 2
                    # mask1_adj contains all neighbors of nodes in set 1
                    # We need to ensure no node in set 2 is in mask1_adj
                    if not (mask1_adj & (mask2_bits << n1)):
                        # Also check if any node in set 2 is connected to a node in set 1
                        if not (mask2_adj & mask1_bits):
                            print("YES")
                            res = []
                            for i in range(n1):
                                if (mask1_bits >> i) & 1:
                                    res.append(i + 1)
                            for i in range(n2):
                                if (mask2_bits >> i) & 1:
                                    res.append(i + n1 + 1)
                            print(*(res))
                            return
    print("NO")

solve()
