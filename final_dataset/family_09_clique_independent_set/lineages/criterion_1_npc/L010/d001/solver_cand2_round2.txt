import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle approach for Maximum Independent Set
    # n <= 46 is small enough for a optimized search
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the two halves
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        # sets_by_size[s] stores a list of bitmasks of size s
        sets_by_size = [[] for _ in range(size + 1)]
        sets_by_size[0].append(0)
        
        for i in range(size):
            u = nodes[i]
            u_mask = (1 << i)
            u_adj = 0
            for j in range(size):
                if (adj_list[u] >> nodes[j]) & 1:
                    u_adj |= (1 << j)
            
            for s in range(i, -1, -1):
                for mask in sets_by_size[s]:
                    if not (mask & u_adj):
                        sets_by_size[s+1].append(mask | u_mask)
        return sets_by_size

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Precompute adjacency between halves
    adj_cross = [0] * n1
    for i in range(n1):
        for j in range(n2):
            if (adj[i] >> (n1 + j)) & 1:
                adj_cross[i] |= (1 << j)
                
    for s1 in range(k + 1):
        s2 = k - s1
        if s1 < len(sets1) and s2 < len(sets2):
            for mask1 in sets1[s1]:
                # Find nodes in second half not connected to any node in mask1
                allowed_mask2 = (1 << n2) - 1
                for i in range(n1):
                    if (mask1 >> i) & 1:
                        allowed_mask2 &= ~adj_cross[i]
                
                for mask2 in sets2[s2]:
                    if (mask2 & allowed_mask2) == mask2:
                        # Found it
                        res = []
                        for i in range(n1):
                            if (mask1 >> i) & 1:
                                res.append(i + 1)
                        for i in range(n2):
                            if (mask2 >> i) & 1:
                                res.append(n1 + i + 1)
                        print("YES")
                        print(*(res))
                        return
    print("NO")

solve()
