import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle approach for Maximum Independent Set
    # Split n into two halves: n1 = n // 2, n2 = n - n1
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the second half
    # subsets2[size] = list of bitmasks of size 'size'
    subsets2 = [[] for _ in range(n2 + 1)]
    for i in range(1 << n2):
        is_independent = True
        mask = 0
        for j in range(n2):
            if (i >> j) & 1:
                if (adj[n1 + j] >> n1) & i:
                    is_independent = False
                    break
                mask |= adj[n1 + j]
        if is_independent:
            count = bin(i).count('1')
            subsets2[count].append((i << n1, mask))
            
    # Search for independent set of size k
    for i in range(1 << n1):
        is_independent = True
        mask1 = 0
        for j in range(n1):
            if (i >> j) & 1:
                if (adj[j] & i):
                    is_independent = False
                    break
                mask1 |= adj[j]
        
        if is_independent:
            c1 = bin(i).count('1')
            if c1 > k:
                continue
            
            needed = k - c1
            if needed > n2:
                continue
            
            # Check if any subset in subsets2[needed] is compatible with mask1
            for mask2_val, mask2_adj in subsets2[needed]:
                if not (mask1 & mask2_val) and not (mask2_adj & i):
                    # Found it
                    res = []
                    for j in range(n1):
                        if (i >> j) & 1:
                            res.append(j + 1)
                    for j in range(n2):
                        if (mask2_val >> (n1 + j)) & 1:
                            res.append(n1 + j + 1)
                    print("YES")
                    print(*(res))
                    return

    print("NO")

solve()
