```python
import sys

def solve():
    # The condition max(0, x_u + x_v - 1)^2 = 0 means x_u + x_v <= 1.
    # This is equivalent to saying that for every bound (u, v), 
    # we cannot have both u and v active.
    # This is exactly the Independent Set problem: find an independent set of size k.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle for Independent Set
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, adj_list):
        # nodes is a list of original indices
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            subset = []
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    # Check if any previously added node in subset is adjacent
                    for prev in subset:
                        if (adj_list[nodes[j]] >> nodes[prev]) & 1:
                            valid = False
                            break
                    if not valid: break
                    subset.append(j)
            if valid:
                res[len(subset)].append(i)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Precompute adjacency for sets2 relative to sets1
    # adj_mask[i] is a bitmask of nodes in nodes2 that are adjacent to node i in nodes1
    adj_mask = []
    for i in range(n1):
        mask = 0
        for j in range(n2):
            if (adj[i] >> (n1 + j)) & 1:
                mask |= (1 << j)
        adj_mask.append(mask)
        
    for k1 in range(min(k + 1, n1 + 1)):
        k2 = k - k1
        if k2 < 0 or k2 > n2:
            continue
        
        for mask1 in sets1[k1]:
            # Calculate forbidden mask in nodes2
            forbidden = 0
            for i in range(n1):
                if (mask1 >> i) & 1:
                    forbidden |= adj_mask[i]
            
            for mask2 in sets2[k2]:
                if not (mask2 & forbidden):
                    # Found it
                    res = []
                    for i in range(n1):
                        if (mask1 >> i) & 1: res.append(i + 1)
                    for i in range(n2):
                        if (mask2 >> i) & 1: res.append(n1 + i + 1)
                    print("YES")
                    print(*(res))
                    return
    print("NO")

solve()
```
