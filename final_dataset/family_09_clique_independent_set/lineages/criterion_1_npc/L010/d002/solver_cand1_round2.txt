import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle for Maximum Independent Set / k-Independent Set
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for both halves
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        sets = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            mask = 0
            count = 0
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    if adj_list[nodes[j]] & mask:
                        valid = False
                        break
                    mask |= (1 << nodes[j])
                    count += 1
            if valid:
                sets[count].append(mask)
        return sets

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Check compatibility between sets
    for c1 in range(n1 + 1):
        c2 = k - c1
        if 0 <= c2 <= n2:
            for m1 in sets1[c1]:
                # Find nodes in second half compatible with m1
                forbidden = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        forbidden |= adj[i]
                
                for m2 in sets2[c2]:
                    if not (m2 & forbidden):
                        # Found it
                        res = []
                        for i in range(n1):
                            if (m1 >> i) & 1:
                                res.append(i + 1)
                        for i in range(n2):
                            if (m2 >> (n1 + i)) & 1:
                                res.append(n1 + i + 1)
                        print("YES")
                        print(*(res))
                        return
    
    print("NO")

solve()
