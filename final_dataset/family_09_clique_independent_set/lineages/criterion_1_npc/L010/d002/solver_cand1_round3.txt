import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # Meet-in-the-middle: n <= 46, split into n1 = 23, n2 = n - 23
    n1 = n // 2
    n2 = n - n1
    
    # Precompute valid subsets for each half
    # valid1[count] = list of bitmasks of size 'count' that are independent
    valid1 = [[] for _ in range(n1 + 1)]
    for i in range(1 << n1):
        cnt = bin(i).count('1')
        is_valid = True
        for j in range(n1):
            if (i >> j) & 1:
                if (adj[j] & i) != 0:
                    is_valid = False
                    break
        if is_valid:
            valid1[cnt].append(i)
            
    valid2 = [[] for _ in range(n2 + 1)]
    for i in range(1 << n2):
        cnt = bin(i).count('1')
        is_valid = True
        for j in range(n2):
            if (i >> j) & 1:
                # Check internal constraints
                if (adj[j + n1] >> n1) & i:
                    is_valid = False
                    break
        if is_valid:
            # Precompute which nodes in the first half are blocked by this subset
            mask_blocked = 0
            for j in range(n2):
                if (i >> j) & 1:
                    mask_blocked |= adj[j + n1] & ((1 << n1) - 1)
            valid2[cnt].append((i, mask_blocked))

    for c1 in range(n1 + 1):
        c2 = k - c1
        if 0 <= c2 <= n2:
            for m1 in valid1[c1]:
                for m2, m2_blocked in valid2[c2]:
                    if not (m1 & m2_blocked):
                        print("YES")
                        res = []
                        for i in range(n1):
                            if (m1 >> i) & 1:
                                res.append(i + 1)
                        for i in range(n2):
                            if (m2 >> i) & 1:
                                res.append(i + n1 + 1)
                        print(*(res))
                        return
    print("NO")

solve()
