import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    full = (1 << n) - 1
    
    if k == 0:
        print("YES")
        print()
        return
    
    if k == 1:
        print("YES")
        print(1)
        return
    
    if k == n:
        if m == 0:
            print("YES")
            print(' '.join(str(i) for i in range(1, n + 1)))
        else:
            print("NO")
        return
    
    half = n // 2
    left_size = half
    right_size = n - half
    
    left_nodes = list(range(1, left_size + 1))
    right_nodes = list(range(left_size + 1, n + 1))
    
    left_internal = [0] * (1 << left_size)
    left_pop = [0] * (1 << left_size)
    
    for S in range(1 << left_size):
        cnt = 0
        edges = 0
        for i in range(left_size):
            if S & (1 << i):
                cnt += 1
                edges += bin(adj[left_nodes[i]] & S).count('1')
        left_internal[S] = edges
        left_pop[S] = cnt
    
    right_internal = [0] * (1 << right_size)
    right_pop = [0] * (1 << right_size)
    
    for S in range(1 << right_size):
        cnt = 0
        edges = 0
        for i in range(right_size):
            if S & (1 << i):
                cnt += 1
                edges += bin(adj[right_nodes[i]] & S).count('1')
        right_internal[S] = edges
        right_pop[S] = cnt
    
    right_mask = [0] * (1 << right_size)
    for S in range(1 << right_size):
        cross = 0
        for i in range(right_size):
            if S & (1 << i):
                cross |= adj[right_nodes[i]] & full
        right_mask[S] = cross
    
    left_to_right = [0] * (1 << left_size)
    for S in range(1 << left_size):
        cross = 0
        for i in range(left_size):
            if S & (1 << i):
                cross |= adj[left_nodes[i]] & full
        left_to_right[S] = cross
    
    for S in range(1 << left_size):
        if left_pop[S] > k:
            continue
        if left_internal[S] > 0:
            continue
        need = k - left_pop[S]
        if need < 0 or need > right_size:
            continue
        
        cross_mask = left_to_right[S]
        cross_mask_shifted = cross_mask >> (left_size + 1)
        
        for T in range(1 << right_size):
            if right_pop[T] != need:
                continue
            if right_internal[T] > 0:
                continue
            if (right_mask[T] & cross_mask_shifted) != 0:
                continue
            
            ans = []
            for i in range(left_size):
                if S & (1 << i):
                    ans.append(left_nodes[i])
            for i in range(right_size):
                if T & (1 << i):
                    ans.append(right_nodes[i])
            
            print("YES")
            print(' '.join(map(str, ans)))
            return
    
    print("NO")

solve()
