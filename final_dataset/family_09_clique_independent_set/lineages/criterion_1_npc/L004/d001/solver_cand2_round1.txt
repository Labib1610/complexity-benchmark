import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [set() for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # The problem asks for an independent set of size k.
    # Since n is small (n*(n-1)/2 <= 1035 implies n <= 46),
    # we can use a backtracking approach with pruning.
    
    committee = []
    
    def backtrack(start_node, current_committee):
        if len(current_committee) == k:
            return current_committee
        
        for i in range(start_node, n + 1):
            # Check if i can be added to the current committee
            is_independent = True
            for member in current_committee:
                if i in adj[member]:
                    is_independent = False
                    break
            
            if is_independent:
                # Pruning: if remaining nodes are not enough to reach k
                if len(current_committee) + (n - i + 1) < k:
                    continue
                
                res = backtrack(i + 1, current_committee + [i])
                if res:
                    return res
        return None

    result = backtrack(1, [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
