```python
import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # We need to choose k variables to be 1 such that no two chosen variables
    # have an edge between them. This is equivalent to finding an independent set
    # of size k in the graph.
    
    # Since n <= 46, we can use a meet-in-the-middle approach or randomized search.
    # Given the constraints and the nature of the problem (Independent Set),
    # a randomized approach with local search or backtracking is often effective.
    
    nodes = list(range(n))
    
    # Try randomized greedy/local search
    for _ in range(200000):
        random.shuffle(nodes)
        chosen = []
        is_chosen = [False] * n
        
        for node in nodes:
            if len(chosen) == k:
                break
            
            can_add = True
            for neighbor in adj[node]:
                if is_chosen[neighbor]:
                    can_add = False
                    break
            
            if can_add:
                chosen.append(node)
                is_chosen[node] = True
        
        if len(chosen) == k:
            print("YES")
            print(*(sorted([x + 1 for x in chosen])))
            return

    print("NO")

solve()
```
