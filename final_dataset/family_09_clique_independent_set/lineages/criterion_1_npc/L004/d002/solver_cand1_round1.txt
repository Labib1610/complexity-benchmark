import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    adj = [0] * (n + 1)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))

    n1 = n // 2
    n2 = n - n1
    
    # Precompute bitmasks for the first half
    # map: (count_of_ones, sum_of_adj_masks) -> bitmask
    # Since we need sum of adj to be 0, we need the chosen variables 
    # to have no edges between them.
    
    def get_subsets(n_size, offset):
        subsets = {}
        for i in range(1 << n_size):
            cnt = bin(i).count('1')
            if cnt > k: continue
            
            mask = 0
            valid = True
            nodes = []
            for j in range(n_size):
                if (i >> j) & 1:
                    nodes.append(j + offset)
            
            # Check internal edges
            current_adj_sum = 0
            for idx, u in enumerate(nodes):
                for v in nodes[idx+1:]:
                    if (adj[u] >> (v - 1)) & 1:
                        valid = False
                        break
                if not valid: break
                current_adj_sum |= adj[u]
            
            if valid:
                subsets[(cnt, current_adj_sum)] = i << offset
        return subsets

    # Meet-in-the-middle
    # We need:
    # 1. cnt1 + cnt2 = k
    # 2. (mask1 & mask2) == 0 (no edges between set1 and set2)
    # 3. (adj_sum1 & mask2) == 0 and (adj_sum2 & mask1) == 0
    
    left = get_subsets(n1, 0)
    right = get_subsets(n2, n1)
    
    # To optimize, iterate over left and check right
    for (c1, adj1), m1 in left.items():
        c2 = k - c1
        if c2 < 0 or c2 > n2: continue
        
        for (c2_val, adj2), m2 in right.items():
            if c2_val == c2:
                # Check cross edges
                # No edges between m1 and m2: (adj1 & m2) == 0 and (adj2 & m1) == 0
                if (adj1 & m2) == 0 and (adj2 & m1) == 0:
                    res = []
                    combined = m1 | m2
                    for i in range(n):
                        if (combined >> i) & 1:
                            res.append(i + 1)
                    print("YES")
                    print(*(res))
                    return

    print("NO")

solve()
