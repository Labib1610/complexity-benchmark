import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    edges = []
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Total Pathway Count = sum of degrees of selected genes
    # System Resonance = sum over all edges (2^c - 1) where c is number of selected endpoints
    # Let S be the set of selected genes.
    # Total Pathway Count = sum_{v in S} deg(v)
    # System Resonance = sum_{(u,v) in E} (2^{I(u in S) + I(v in S)} - 1)
    # If both u, v in S: resonance = 2^2 - 1 = 3
    # If one in S: resonance = 2^1 - 1 = 1
    # If none in S: resonance = 2^0 - 1 = 0
    # Let E_S be edges with both endpoints in S, E_O be edges with exactly one endpoint in S.
    # Resonance = 3 * |E_S| + 1 * |E_O|
    # Total Pathway Count = sum_{v in S} deg(v) = 2 * |E_S| + 1 * |E_O|
    # Condition: 3 * |E_S| + |E_O| = 2 * |E_S| + |E_O|
    # This simplifies to |E_S| = 0.
    # So we need to select k genes such that no two selected genes share an edge.
    # This is equivalent to finding an Independent Set of size k.

    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Meet-in-the-middle for Independent Set of size k
    n1 = n // 2
    n2 = n - n1
    
    def get_is(nodes):
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            subset = []
            mask = 0
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    if mask & (1 << nodes[j]):
                        valid = False
                        break
                    mask |= (1 << nodes[j])
                    subset.append(nodes[j])
            if valid:
                res[len(subset)].append((mask, subset))
        return res

    left_nodes = list(range(n1))
    right_nodes = list(range(n1, n))
    
    left_is = get_is(left_nodes)
    right_is = get_is(right_nodes)
    
    # Precompute adjacency for right side
    adj_right = [adj[i] & ((1 << n) - (1 << n1)) for i in range(n1, n)]
    
    for k1 in range(k + 1):
        k2 = k - k1
        if k1 < len(left_is) and k2 < len(right_is):
            for mask1, sub1 in left_is[k1]:
                # Filter right candidates: must not be adjacent to any in sub1
                forbidden = 0
                for u in sub1:
                    forbidden |= adj[u]
                
                for mask2, sub2 in right_is[k2]:
                    if not (mask2 & forbidden):
                        print("YES")
                        print(*(sorted([x + 1 for x in sub1 + sub2])))
                        return

    print("NO")

solve()
