We need to select exactly k vertices in a graph of n vertices (documents) and m edges (shared blocks), such that no edge has both endpoints selected.  

That means the selected set must be an **independent set** of size exactly k.  

So the problem reduces to:  
- Given a graph with n ≤ 46, find an independent set of size exactly k.  
- If found, print YES and the vertices.  
- Otherwise, print NO.  

Since n is small (≤ 46), we can use a **meet-in-the-middle** approach:  
- Split vertices into two halves.  
- For each half, enumerate all independent sets and store their bitmasks and sizes.  
- Then combine: for each independent set in the first half, find an independent set in the second half that is compatible (no edges between the halves) and whose size sums to k.  

We’ll implement this efficiently.  

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    # Split into two halves
    n1 = n // 2
    n2 = n - n1
    
    # Enumerate independent sets in first half
    # mask1: bit i corresponds to vertex i
    list1 = []  # (mask, size)
    for mask in range(1 << n1):
        ok = True
        for i in range(n1):
            if mask & (1 << i):
                # check conflict with other selected in this half
                if adj[i] & mask & ((1 << n1) - 1):
                    ok = False
                    break
        if ok:
            list1.append((mask, bin(mask).count('1')))
    
    # Enumerate independent sets in second half
    # mask2: bit j corresponds to vertex n1 + j
    list2 = []  # (mask, size)
    for mask in range(1 << n2):
        ok = True
        for j in range(n2):
            if mask & (1 << j):
                v = n1 + j
                # check conflict with other selected in this half
                if adj[v] & (mask << n1) & (((1 << n2) - 1) << n1):
                    ok = False
                    break
        if ok:
            list2.append((mask, bin(mask).count('1')))
    
    # For each mask in list1, we need a compatible mask in list2
    # Compatibility: no edges between selected vertices of the two halves
    # Precompute for each mask1 the forbidden mask2 bits
    forbidden = {}
    for mask1, _ in list1:
        forb = 0
        for i in range(n1):
            if mask1 & (1 << i):
                # neighbors in second half
                forb |= (adj[i] >> n1)
        forbidden[mask1] = forb
    
    # Group list2 by size
    by_size = {}
    for mask2, sz in list2:
        by_size.setdefault(sz, []).append(mask2)
    
    # Search
    for mask1, sz1 in list1:
        need = k - sz1
        if need < 0 or need > n2:
            continue
        if need not in by_size:
            continue
        forb = forbidden[mask1]
        for mask2 in by_size[need]:
            if mask2 & forb == 0:
                # Found
                selected = []
                for i in range(n1):
                    if mask1 & (1 << i):
                        selected.append(i + 1)
                for j in range(n2):
                    if mask2 & (1 << j):
                        selected.append(n1 + j + 1)
                print("YES")
                print(" ".join(map(str, selected)))
                return
    
    print("NO")

if __name__ == "__main__":
    solve()
```
