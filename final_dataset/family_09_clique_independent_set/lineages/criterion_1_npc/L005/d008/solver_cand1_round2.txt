import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2
        
    # The condition "Boundary Size == Total Reference Count" is equivalent to:
    # No shared data block has both of its referencing documents selected.
    # This is exactly the Independent Set problem on the graph where 
    # documents are vertices and shared blocks are edges.
    # We need to find an independent set of size exactly k.
    
    # Since n <= 46, we use meet-in-the-middle.
    n1 = n // 2
    n2 = n - n1
    
    def get_subsets(nodes, edges):
        # nodes: list of node indices
        # edges: list of (u, v) where both u, v in nodes
        size = len(nodes)
        res = {} # count -> list of bitmasks
        for i in range(1 << size):
            # Check if independent set
            is_independent = True
            for u, v in edges:
                if (i & (1 << nodes.index(u))) and (i & (1 << nodes.index(v))):
                    is_independent = False
                    break
            if is_independent:
                cnt = bin(i).count('1')
                if cnt not in res:
                    res[cnt] = []
                res[cnt].append(i)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    edges1 = [(u, v) for u, v in adj if u < n1 and v < n1]
    edges2 = [(u, v) for u, v in adj if u >= n1 and v >= n1]
    cross_edges = [(u, v) for u, v in adj if (u < n1 and v >= n1) or (v < n1 and u >= n1)]
    
    subsets1 = get_subsets(nodes1, edges1)
    subsets2 = get_subsets(nodes2, edges2)
    
    for c1 in subsets1:
        c2 = k - c1
        if c2 in subsets2:
            for mask1 in subsets1[c1]:
                for mask2 in subsets2[c2]:
                    # Check cross edges
                    valid = True
                    for u, v in cross_edges:
                        u_in = (mask1 & (1 << u)) if u < n1 else (mask2 & (1 << (u - n1)))
                        v_in = (mask1 & (1 << v)) if v < n1 else (mask2 & (1 << (v - n1)))
                        if u_in and v_in:
                            valid = False
                            break
                    if valid:
                        print("YES")
                        res = []
                        for i in range(n1):
                            if mask1 & (1 << i): res.append(i + 1)
                        for i in range(n2):
                            if mask2 & (1 << i): res.append(i + n1 + 1)
                        print(*(res))
                        return
    print("NO")

solve()
