import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # A block (u, v) is in the Standard Union if at least one of u or v is selected.
    # A block (u, v) is in the Symmetric Union if exactly one of u or v is selected.
    # The condition "Symmetric Union == Standard Union" means that for every block (u, v),
    # it is NOT the case that both u and v are selected.
    # This is equivalent to saying that the selected set of documents must be an
    # Independent Set in the graph where edges are the shared data blocks.

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    # We need to find an independent set of size k.
    # Since n <= 46, we can use a meet-in-the-middle approach or backtracking.
    # Given the constraints and the nature of the problem, a randomized search
    # or simple backtracking with pruning is efficient enough.
    
    nodes = list(range(n))
    
    def get_independent_set(k):
        # Try randomized greedy search
        for _ in range(200000):
            random.shuffle(nodes)
            res = []
            forbidden = [False] * n
            for node in nodes:
                if not forbidden[node]:
                    res.append(node)
                    if len(res) == k:
                        return sorted(res)
                    forbidden[node] = True
                    for neighbor in adj[node]:
                        forbidden[neighbor] = True
        return None

    # For n=46, simple backtracking with bitmask or pruning
    def backtrack(idx, current_set, forbidden_mask):
        if len(current_set) == k:
            return current_set
        if idx == n or len(current_set) + (n - idx) < k:
            return None
        
        # Try including idx
        if not (forbidden_mask & (1 << idx)):
            new_forbidden = forbidden_mask | (1 << idx)
            for neighbor in adj[idx]:
                new_forbidden |= (1 << neighbor)
            res = backtrack(idx + 1, current_set + [idx], new_forbidden)
            if res: return res
            
        # Try excluding idx
        res = backtrack(idx + 1, current_set, forbidden_mask)
        if res: return res
        return None

    # The backtracking is efficient for finding independent sets.
    result = backtrack(0, [], 0)
    
    if result:
        print("YES")
        print(*(x + 1 for x in result))
    else:
        print("NO")

solve()
