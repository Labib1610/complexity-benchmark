import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n)]
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
        idx += 2
        
    # Let S be the set of selected items.
    # Let x_i = 1 if i in S, else 0.
    # T = sum_{i in S} deg(i) = sum_{i=0}^{n-1} x_i * deg(i)
    # Score = sum_{(u,v) in E} 5^(x_u + x_v)
    # Score = sum_{(u,v) in E} (1 if x_u+x_v=0 else 5 if x_u+x_v=1 else 25)
    # Score = sum_{(u,v) in E} (1 + 4*x_u + 4*x_v - 24*x_u*x_v)
    # Score = m + 4 * sum_{(u,v) in E} (x_u + x_v) - 24 * sum_{(u,v) in E} (x_u * x_v)
    # Score = m + 4 * T - 24 * (number of edges with both endpoints in S)
    # We want Score = m + 4T, which implies 24 * (number of edges in S) = 0.
    # So we need to select k items such that no two selected items are connected.
    # This is equivalent to finding an independent set of size k.
    
    if k > n:
        print("NO")
        return

    # Meet-in-the-middle for Independent Set of size k
    n1 = n // 2
    n2 = n - n1
    
    def get_subsets(nodes):
        size = len(nodes)
        res = {} # {count: [masks]}
        for i in range(1 << size):
            mask = 0
            cnt = 0
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    mask |= (1 << nodes[j])
                    cnt += 1
            for u in nodes:
                if (mask >> u) & 1:
                    for v in adj[u]:
                        if (mask >> v) & 1:
                            valid = False
                            break
                if not valid: break
            if valid:
                if cnt not in res: res[cnt] = []
                res[cnt].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sub1 = get_subsets(nodes1)
    sub2 = get_subsets(nodes2)
    
    for c1 in sub1:
        c2 = k - c1
        if c2 in sub2:
            for m1 in sub1[c1]:
                for m2 in sub2[c2]:
                    # Check if any edge between m1 and m2
                    valid = True
                    for u in range(n1):
                        if (m1 >> u) & 1:
                            for v in adj[u]:
                                if v >= n1 and ((m2 >> v) & 1):
                                    valid = False
                                    break
                        if not valid: break
                    if valid:
                        print("YES")
                        res = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                res.append(str(i + 1))
                        print(" ".join(res))
                        return
    print("NO")

solve()
