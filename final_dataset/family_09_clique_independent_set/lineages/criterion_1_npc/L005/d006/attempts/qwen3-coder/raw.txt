from itertools import combinations

def solve():
    n, m, k = map(int, input().split())
    partnerships = []
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u, v = map(int, input().split())
        partnerships.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # Try all combinations of k items
    for selected in combinations(range(1, n + 1), k):
        selected_set = set(selected)
        
        # Calculate T (sum of partnership counts for selected items)
        T = 0
        for item in selected:
            T += len(adj[item])
        
        # Calculate the sum of configuration scores
        score_sum = 0
        for u, v in partnerships:
            c = 0
            if u in selected_set:
                c += 1
            if v in selected_set:
                c += 1
            score_sum += 5 ** c
        
        # Check if the condition is satisfied
        target = m + 4 * T
        if score_sum == target:
            print("YES")
            print(*selected)
            return
    
    print("NO")

solve()
