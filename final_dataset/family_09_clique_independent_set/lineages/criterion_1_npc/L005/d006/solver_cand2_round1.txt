import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The condition sum(y_j^2) = sum(y_j) for y_j in {0, 1, 2}
    # y_j^2 = y_j is true if y_j is 0 or 1.
    # y_j^2 = 4, y_j = 2 if y_j is 2.
    # So sum(y_j^2) - sum(y_j) = sum(y_j^2 - y_j) = 0.
    # y_j^2 - y_j is 0 if y_j is 0 or 1, and 2 if y_j is 2.
    # Thus, the condition is equivalent to: no y_j can be 2.
    # y_j = x_u + x_v. y_j = 2 iff x_u = 1 and x_v = 1.
    # So we need to choose k indices such that no edge (u, v) has both u and v chosen.
    # This is exactly the Independent Set problem on the graph of edges.

    if k > n:
        print("NO")
        return

    # Meet-in-the-middle for Independent Set
    n1 = n // 2
    n2 = n - n1
    
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    def get_subsets(nodes, offset):
        size = len(nodes)
        res = {}
        for i in range(1 << size):
            mask = 0
            cnt = 0
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    u = nodes[j]
                    if (mask & adj[u]):
                        valid = False
                        break
                    mask |= (1 << u)
                    cnt += 1
            if valid:
                if cnt not in res:
                    res[cnt] = []
                res[cnt].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    subsets1 = get_subsets(nodes1, 0)
    subsets2 = get_subsets(nodes2, n1)

    for c1, masks1 in subsets1.items():
        c2 = k - c1
        if c2 in subsets2:
            for m1 in masks1:
                for m2 in subsets2[c2]:
                    # Check if any edge between m1 and m2
                    valid = True
                    for u in range(n1):
                        if (m1 >> u) & 1:
                            if (adj[u] & m2):
                                valid = False
                                break
                    if valid:
                        print("YES")
                        res = []
                        combined = m1 | m2
                        for i in range(n):
                            if (combined >> i) & 1:
                                res.append(str(i + 1))
                        print(" ".join(res))
                        return
    print("NO")

solve()
