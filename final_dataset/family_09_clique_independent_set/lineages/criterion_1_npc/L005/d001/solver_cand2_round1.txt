import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1] |= (1 << (v-1))
        adj[v-1] |= (1 << (u-1))

    # Split into two halves for meet-in-the-middle
    n1 = n // 2
    n2 = n - n1
    
    # Precompute valid subsets for each half
    # valid1[count][mask] = True if mask is independent
    valid1 = [[] for _ in range(n1 + 1)]
    for i in range(1 << n1):
        is_independent = True
        count = 0
        for j in range(n1):
            if (i >> j) & 1:
                count += 1
                if adj[j] & i:
                    is_independent = False
                    break
        if is_independent:
            valid1[count].append((i, [j + 1 for j in range(n1) if (i >> j) & 1]))

    # Precompute valid subsets for second half
    # We need to store them such that we can check compatibility with first half
    # For each mask in second half, store its forbidden mask in first half
    valid2 = [[] for _ in range(n2 + 1)]
    for i in range(1 << n2):
        is_independent = True
        count = 0
        forbidden_in_first = 0
        for j in range(n2):
            if (i >> j) & 1:
                count += 1
                if adj[n1 + j] & (i << n1):
                    is_independent = False
                    break
                # Add constraints imposed by this node on the first half
                forbidden_in_first |= (adj[n1 + j] & ((1 << n1) - 1))
        
        if is_independent:
            valid2[count].append((i, forbidden_in_first, [n1 + j + 1 for j in range(n2) if (i >> j) & 1]))

    # Search for k
    for c1 in range(n1 + 1):
        c2 = k - c1
        if 0 <= c2 <= n2:
            # Optimization: use a dictionary or array to quickly find compatible masks
            # Since n1 is at most 23, we can use an array of size 2^n1
            # But we only need to check if there exists a mask in valid1[c1] 
            # that is not forbidden by mask in valid2[c2]
            
            # To speed up, pre-process valid1[c1]
            # For each mask in valid1[c1], mark it as available
            available = [None] * (1 << n1)
            for mask, members in valid1[c1]:
                available[mask] = members
            
            for mask2, forbidden, members2 in valid2[c2]:
                # We need a mask1 in valid1[c1] such that (mask1 & forbidden) == 0
                # This is equivalent to mask1 being a subset of (~forbidden & ((1 << n1) - 1))
                target = (~forbidden) & ((1 << n1) - 1)
                
                # Iterate through all submasks of target
                sub = target
                while True:
                    if available[sub] is not None:
                        print("YES")
                        print(*(available[sub] + members2))
                        return
                    if sub == 0:
                        break
                    sub = (sub - 1) & target
                    
    print("NO")

solve()
