import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u - 1, v - 1))

    # A delegation D of size k is harmonious if for every edge (u, v),
    # at least one of u or v is NOT in D.
    # This is equivalent to saying that the set of employees NOT in D
    # (let's call this set R, where |R| = n - k) must be a Vertex Cover
    # of the graph.
    # We need to find a Vertex Cover of size exactly n - k.
    
    target_vc_size = n - k
    
    # Meet-in-the-middle for Vertex Cover
    # Split vertices into two sets: 0 to n//2 - 1 and n//2 to n - 1
    n1 = n // 2
    n2 = n - n1
    
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    # Precompute masks for edges within n1, within n2, and between n1 and n2
    # Actually, just use the standard recursive search with pruning
    # Since n <= 46, we can use a simple backtracking with bitmask
    
    def get_vc(idx, current_vc_mask, count):
        if count > target_vc_size:
            return None
        if idx == n:
            if count == target_vc_size:
                return current_vc_mask
            return None
        
        # Try including idx in VC
        res = get_vc(idx + 1, current_vc_mask | (1 << idx), count + 1)
        if res is not None:
            return res
        
        # Try excluding idx from VC
        # If we exclude idx, all its neighbors must be in the VC
        neighbors = adj[idx]
        new_mask = current_vc_mask | (neighbors & ~current_vc_mask)
        new_count = bin(new_mask).count('1')
        if new_count <= target_vc_size:
            res = get_vc(idx + 1, new_mask, new_count)
            if res is not None:
                return res
        return None

    # The search above is slightly wrong for standard VC.
    # Correct approach:
    def find_vc(idx, current_vc_mask, count):
        if count > target_vc_size:
            return None
        
        # Find first uncovered edge
        found_edge = -1
        for i in range(idx, n):
            # Check if any edge (i, j) is not covered
            # An edge (i, j) is covered if i in VC or j in VC
            # So we need to check if there exists j > i such that
            # (i, j) is an edge and neither i nor j is in VC
            if not (current_vc_mask & (1 << i)):
                uncovered = adj[i] & ~current_vc_mask
                if uncovered:
                    found_edge = i
                    break
        
        if found_edge == -1:
            # All edges covered, fill remaining with arbitrary nodes
            if count <= target_vc_size:
                res = current_vc_mask
                for i in range(n):
                    if count < target_vc_size and not (res & (1 << i)):
                        res |= (1 << i)
                        count += 1
                return res
            return None
        
        # Branching: either u or v must be in VC
        u = found_edge
        v = (adj[u] & ~current_vc_mask).bit_length() - 1
        
        # Try adding u
        res = find_vc(u + 1, current_vc_mask | (1 << u), count + 1)
        if res is not None: return res
        
        # Try adding v
        res = find_vc(u + 1, current_vc_mask | (1 << v), count + 1)
        if res is not None: return res
        
        return None

    vc_mask = find_vc(0, 0, 0)
    
    if vc_mask is None:
        print("NO")
    else:
        print("YES")
        delegation = []
        for i in range(n):
            if not (vc_mask & (1 << i)):
                delegation.append(str(i + 1))
        print(" ".join(delegation))

solve()
