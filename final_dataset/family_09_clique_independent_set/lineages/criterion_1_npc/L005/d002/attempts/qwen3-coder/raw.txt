from itertools import combinations
import sys

def main():
    n, m, k = map(int, sys.stdin.readline().split())
    
    # Read friction pairs
    frictions = set()
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        if u > v:
            u, v = v, u
        frictions.add((u, v))
    
    # Special case: if k=1, just pick any employee
    if k == 1:
        print("YES")
        print(1)
        return
    
    # For small n, try all combinations
    if n <= 25:
        for combo in combinations(range(1, n+1), k):
            valid = True
            for u, v in frictions:
                if u in combo and v in combo:
                    valid = False
                    break
            if valid:
                print("YES")
                print(' '.join(map(str, combo)))
                return
        print("NO")
        return
    
    # For larger n, use meet-in-the-middle approach
    # Split employees into two halves
    mid = n // 2
    left_set = list(range(1, mid + 1))
    right_set = list(range(mid + 1, n + 1))
    
    # Generate all valid subsets for left half
    left_valid = {}
    for i in range(min(k + 1, len(left_set) + 1)):
        for left_combo in combinations(left_set, i):
            # Check if this subset is internally valid
            valid = True
            for u, v in frictions:
                if u in left_combo and v in left_combo:
                    valid = False
                    break
            if valid:
                left_valid[left_combo] = True
    
    # Generate all valid subsets for right half
    right_valid = {}
    for i in range(min(k + 1, len(right_set) + 1)):
        for right_combo in combinations(right_set, i):
            # Check if this subset is internally valid
            valid = True
            for u, v in frictions:
                if u in right_combo and v in right_combo:
                    valid = False
                    break
            if valid:
                right_valid[right_combo] = True
    
    # Now check all combinations of left_size and right_size that sum to k
    for left_size in range(k + 1):
        right_size = k - left_size
        if left_size > len(left_set) or right_size > len(right_set):
            continue
            
        # Get all valid left subsets of size left_size
        left_subsets = []
        for combo in combinations(left_set, left_size):
            valid = True
            for u, v in frictions:
                if u in combo and v in combo:
                    valid = False
                    break
            if valid:
                left_subsets.append(set(combo))
        
        # Get all valid right subsets of size right_size
        right_subsets = []
        for combo in combinations(right_set, right_size):
            valid = True
            for u, v in frictions:
                if u in combo and v in combo:
                    valid = False
                    break
            if valid:
                right_subsets.append(set(combo))
        
        # Check all pairs
        for left_subset in left_subsets:
            for right_subset in right_subsets:
                # Check if there's no friction between left and right
                valid = True
                for u, v in frictions:
                    if (u in left_subset and v in right_subset) or (u in right_subset and v in left_subset):
                        valid = False
                        break
                if valid:
                    result = list(left_subset) + list(right_subset)
                    print("YES")
                    print(' '.join(map(str, result)))
                    return
    
    print("NO")

if __name__ == "__main__":
    main()
