import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # x^T A x = sum(x_i^2) + 2 * sum_{i<j} A_{i,j} x_i x_j
    # Since x_i in {0, 1}, x_i^2 = x_i.
    # x^T A x = k + 2 * sum_{i<j} A_{i,j} x_i x_j
    # We want x^T A x = k, so we need sum_{i<j} A_{i,j} x_i x_j = 0.
    # This means for any two indices i, j where x_i=1 and x_j=1, A_{i,j} must be 0.
    # This is equivalent to finding an independent set of size k in the graph
    # where edges are the pairs (u, v) given in the input.

    n1 = n // 2
    n2 = n - n1
    
    # Meet-in-the-middle for Maximum Independent Set
    # adj1: adjacency for first n1 nodes, adj2: for remaining n2 nodes
    adj1 = [adj[i] & ((1 << n1) - 1) for i in range(n1)]
    adj2 = [(adj[i + n1] >> n1) for i in range(n2)]
    cross = [(adj[i] >> n1) for i in range(n1)]

    def get_independent_sets(nodes_adj, num_nodes):
        # returns dict: size -> list of bitmasks
        res = [[] for _ in range(num_nodes + 1)]
        res[0].append(0)
        for i in range(num_nodes):
            new_res = [list(res[j]) for j in range(num_nodes + 1)]
            for j in range(num_nodes):
                for mask in res[j]:
                    if not (mask & (1 << i)) and not (nodes_adj[i] & mask):
                        new_res[j+1].append(mask | (1 << i))
            res = new_res
        return res

    sets1 = get_independent_sets(adj1, n1)
    sets2 = get_independent_sets(adj2, n2)

    for k1 in range(k + 1):
        k2 = k - k1
        if k1 > n1 or k2 > n2:
            continue
        for m1 in sets1[k1]:
            # Check which nodes in second half are compatible with m1
            forbidden = 0
            for i in range(n1):
                if (m1 >> i) & 1:
                    forbidden |= cross[i]
            
            for m2 in sets2[k2]:
                if not (m2 & forbidden):
                    print("YES")
                    ans = []
                    for i in range(n1):
                        if (m1 >> i) & 1: ans.append(i + 1)
                    for i in range(n2):
                        if (m2 >> i) & 1: ans.append(i + n1 + 1)
                    print(*(sorted(ans)))
                    return

    print("NO")

solve()
