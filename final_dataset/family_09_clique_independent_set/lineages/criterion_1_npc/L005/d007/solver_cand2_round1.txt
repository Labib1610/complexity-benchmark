import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The condition: sum of reference counts == archive footprint
    # Let S be the set of selected documents.
    # Sum of reference counts = sum_{d in S} (number of edges connected to d)
    # Archive footprint = number of edges (u, v) such that u in S or v in S
    # Let E_S be the set of edges (u, v) such that u in S or v in S.
    # Let E_in be the set of edges (u, v) such that u in S AND v in S.
    # Sum of reference counts = sum_{d in S} deg(d) = |E_S| + |E_in|
    # Archive footprint = |E_S|
    # Condition: |E_S| + |E_in| = |E_S|  => |E_in| = 0
    # This means no two selected documents can share a data block.
    # We need to find an independent set of size k in the graph where edges are shared blocks.

    if k > n:
        print("NO")
        return

    # Meet-in-the-middle for finding an independent set of size k
    # n <= 46 is too large for simple backtracking, but we need an independent set.
    # Actually, the condition |E_in| = 0 is exactly the definition of an independent set.
    
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    def get_independent_sets(nodes):
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        res[0].append(0)
        
        for i in range(size):
            u = nodes[i]
            mask_u = 1 << i
            for s in range(i, -1, -1):
                for mask in res[s]:
                    if not (mask & (1 << i)) and not (adj[u] & sum_mask(mask, nodes)):
                        new_mask = mask | (1 << i)
                        res[s+1].append(new_mask)
        return res

    def sum_mask(mask, nodes):
        m = 0
        for i in range(len(nodes)):
            if (mask >> i) & 1:
                m |= (1 << nodes[i])
        return m

    # Standard backtracking with pruning for independent set
    def find_is(idx, current_mask, count):
        if count == k:
            return [i for i in range(n) if (current_mask >> i) & 1]
        if idx == n or count + (n - idx) < k:
            return None
        
        # Try including idx
        if not (adj[idx] & current_mask):
            res = find_is(idx + 1, current_mask | (1 << idx), count + 1)
            if res: return res
        
        # Try excluding idx
        res = find_is(idx + 1, current_mask, count)
        if res: return res
        return None

    result = find_is(0, 0, 0)
    if result:
        print("YES")
        print(*(x + 1 for x in result))
    else:
        print("NO")

solve()
