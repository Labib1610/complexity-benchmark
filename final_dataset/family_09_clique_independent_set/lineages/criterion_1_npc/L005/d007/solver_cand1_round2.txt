import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    deg = [len(adj[i]) for i in range(n)]
    
    # S = sum of degrees of active nodes
    # B = number of edges (u, v) such that one is active and one is inactive
    # Let A be the set of active nodes, I be the set of inactive nodes.
    # S = sum_{u in A} deg(u)
    # B = sum_{u in A} (number of neighbors of u in I)
    #   = sum_{u in A} (deg(u) - number of neighbors of u in A)
    #   = sum_{u in A} deg(u) - sum_{u in A} (number of neighbors of u in A)
    #   = S - 2 * (number of edges with both endpoints in A)
    # Condition S = B becomes S = S - 2 * (edges in A)
    # This implies 2 * (edges in A) = 0, so edges in A must be 0.
    # The problem reduces to finding an independent set of size k.
    
    if k > n:
        print("NO")
        return

    # Meet-in-the-middle for Independent Set
    n1 = n // 2
    n2 = n - n1
    
    adj_mask = [0] * n
    for i in range(n):
        for neighbor in adj[i]:
            adj_mask[i] |= (1 << neighbor)
            
    def get_independent_sets(nodes):
        size = len(nodes)
        res = {} # count -> list of masks
        for i in range(1 << size):
            mask = 0
            valid = True
            cnt = 0
            for j in range(size):
                if (i >> j) & 1:
                    node = nodes[j]
                    if mask & (1 << node):
                        valid = False
                        break
                    mask |= (1 << node)
                    mask |= adj_mask[node]
                    cnt += 1
            if valid:
                if cnt not in res: res[cnt] = []
                res[cnt].append(mask & ((1 << n) - 1))
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1)
    sets2 = get_independent_sets(nodes2)
    
    # We need to pick k1 from sets1 and k2 from sets2 such that k1+k2 = k
    # and the combined mask is independent.
    # mask1 is the set of nodes in A1 plus their neighbors.
    # mask2 is the set of nodes in A2 plus their neighbors.
    # Condition: A1 is independent, A2 is independent, 
    # and no node in A1 is adjacent to a node in A2.
    
    # Re-calculate: mask should be just the set of nodes in A
    def get_is_masks(nodes):
        size = len(nodes)
        res = {}
        for i in range(1 << size):
            mask = 0
            valid = True
            cnt = 0
            for j in range(size):
                if (i >> j) & 1:
                    node = nodes[j]
                    if mask & adj_mask[node]:
                        valid = False
                        break
                    mask |= (1 << node)
                    cnt += 1
            if valid:
                if cnt not in res: res[cnt] = []
                res[cnt].append((mask, i))
        return res

    is1 = get_is_masks(nodes1)
    is2 = get_is_masks(nodes2)
    
    for k1 in range(k + 1):
        k2 = k - k1
        if k1 in is1 and k2 in is2:
            for m1, raw1 in is1[k1]:
                for m2, raw2 in is2[k2]:
                    # Check if any node in A1 is adjacent to A2
                    valid = True
                    for i in range(n1):
                        if (raw1 >> i) & 1:
                            if adj_mask[i] & m2:
                                valid = False
                                break
                    if valid:
                        print("YES")
                        res = []
                        for i in range(n1):
                            if (raw1 >> i) & 1: res.append(i + 1)
                        for i in range(n2):
                            if (raw2 >> i) & 1: res.append(i + n1 + 1)
                        print(*(res))
                        return
    print("NO")

solve()
