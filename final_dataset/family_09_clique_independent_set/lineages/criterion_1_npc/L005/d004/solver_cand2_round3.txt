import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # Meet-in-the-middle for Maximum Independent Set of size k
    n1 = n // 2
    n2 = n - n1
    
    # Precompute masks for first half
    # subset_masks[count] = list of (mask, adj_mask_within_first_half)
    subsets1 = [[] for _ in range(n1 + 1)]
    for i in range(1 << n1):
        valid = True
        count = 0
        adj_mask = 0
        for j in range(n1):
            if (i >> j) & 1:
                if adj[j] & i:
                    valid = False
                    break
                count += 1
                adj_mask |= (adj[j] & ((1 << n1) - 1))
        if valid:
            subsets1[count].append((i, adj_mask))

    # Precompute masks for second half
    subsets2 = [[] for _ in range(n2 + 1)]
    for i in range(1 << n2):
        valid = True
        count = 0
        adj_mask = 0
        for j in range(n2):
            if (i >> j) & 1:
                if adj[j + n1] & (i << n1):
                    valid = False
                    break
                count += 1
                adj_mask |= (adj[j + n1] >> n1)
        if valid:
            subsets2[count].append((i, adj_mask))

    # Cross-half conflicts
    cross_adj = [0] * n2
    for i in range(n2):
        cross_adj[i] = (adj[i + n1] & ((1 << n1) - 1))

    for c1 in range(n1 + 1):
        c2 = k - c1
        if 0 <= c2 <= n2:
            for m1, am1 in subsets1[c1]:
                # Precompute which nodes in second half are forbidden by m1
                forbidden_in_2 = 0
                for j in range(n1):
                    if (m1 >> j) & 1:
                        forbidden_in_2 |= (adj[j] >> n1)
                
                for m2, am2 in subsets2[c2]:
                    if not (m2 & forbidden_in_2):
                        # Check if m1 conflicts with m2
                        # m1 conflicts with m2 if any bit in m2 has a neighbor in m1
                        conflict = False
                        for j in range(n2):
                            if (m2 >> j) & 1:
                                if cross_adj[j] & m1:
                                    conflict = True
                                    break
                        if not conflict:
                            res = []
                            for j in range(n1):
                                if (m1 >> j) & 1: res.append(j + 1)
                            for j in range(n2):
                                if (m2 >> j) & 1: res.append(j + n1 + 1)
                            print("YES")
                            print(*(res))
                            return
    print("NO")

solve()
