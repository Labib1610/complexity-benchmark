import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # We need to find an independent set of size k.
    # Since n <= 46, we use a meet-in-the-middle approach.
    n1 = n // 2
    n2 = n - n1
    
    # Precompute masks for the two halves
    # adj1[i] is the mask of neighbors of i in the first half
    # adj2[i] is the mask of neighbors of i in the second half
    
    def get_independent_sets(nodes):
        # nodes is a list of indices
        size = len(nodes)
        res = {} # count -> list of masks
        for i in range(1 << size):
            mask = 0
            valid = True
            cnt = 0
            for j in range(size):
                if (i >> j) & 1:
                    cnt += 1
                    u = nodes[j]
                    if mask & adj[u]:
                        valid = False
                        break
                    mask |= (1 << u)
            if valid:
                if cnt not in res: res[cnt] = []
                res[cnt].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1)
    sets2 = get_independent_sets(nodes2)
    
    # For each mask in sets1, check if there's a compatible mask in sets2
    # A mask m1 and m2 are compatible if (m1 & adj_of_m2) == 0
    # which is equivalent to (m2 & adj_of_m1) == 0
    
    for c1 in range(k + 1):
        c2 = k - c1
        if c1 in sets1 and c2 in sets2:
            # Precompute adjacency masks for each mask in sets1
            # Actually, just check compatibility
            for m1 in sets1[c1]:
                # Calculate forbidden mask in the second half
                forbidden = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        forbidden |= (adj[i] >> n1)
                
                for m2 in sets2[c2]:
                    if not (m2 & forbidden):
                        # Found it
                        print("YES")
                        ans = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                ans.append(str(i + 1))
                        print(" ".join(ans))
                        return

    print("NO")

solve()
