import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    ALL = (1 << n) - 1

    # order vertices by degree (small to large) to improve pruning
    order = sorted(range(n), key=lambda x: bin(adj[x]).count("1"))
    # map original index to position in ordered list
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i

    # rebuild adjacency in ordered space
    adj2 = [0] * n
    for i in range(n):
        v = order[i]
        mask = 0
        for u in range(n):
            if adj[v] >> u & 1:
                mask |= 1 << pos[u]
        adj2[i] = mask

    # recursive search
    answer = None

    def dfs(cur_mask: int, cand_mask: int, cur_list):
        nonlocal answer
        if answer is not None:
            return
        cur_size = len(cur_list)
        if cur_size == k:
            answer = cur_list[:]
            return
        # pruning: not enough candidates left
        if cur_size + cand_mask.bit_count() < k:
            return
        # choose a vertex with smallest degree in candidates (heuristic)
        # pick first set bit
        v = (cand_mask & -cand_mask).bit_length() - 1

        # branch 1: include v
        new_cand = cand_mask & ~adj2[v] & ~(1 << v)
        dfs(cur_mask | (1 << v), new_cand, cur_list + [order[v] + 1])
        if answer is not None:
            return
        # branch 2: exclude v
        dfs(cur_mask, cand_mask & ~(1 << v), cur_list)

    dfs(0, (1 << n) - 1, [])

    if answer is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, answer)))

if __name__ == "__main__":
    solve()
