import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    forbidden = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        forbidden[u] |= (1 << (v - 1))
        forbidden[v] |= (1 << (u - 1))
    
    # Convert to 0-indexed masks
    mask = [0] * n
    for i in range(1, n + 1):
        mask[i-1] = forbidden[i]
    
    half = n // 2
    left_size = half
    right_size = n - half
    
    # Precompute popcount for numbers up to max(2^left_size, 2^right_size)
    max_pop = max(1 << left_size, 1 << right_size)
    popcount = [0] * max_pop
    for i in range(1, max_pop):
        popcount[i] = popcount[i & (i - 1)] + 1
    
    # Generate left subsets
    left_list = []
    left_masks = []
    for s in range(1 << left_size):
        # Check if s is independent
        ok = True
        bits = s
        while bits:
            lb = bits & -bits
            i = (lb.bit_length() - 1)
            if (mask[i] & s) != 0:
                ok = False
                break
            bits ^= lb
        if ok:
            left_list.append(s)
            left_masks.append(popcount[s])
    
    # Generate right subsets
    right_list = []
    right_masks = []
    for s in range(1 << right_size):
        ok = True
        bits = s
        while bits:
            lb = bits & -bits
            i = (lb.bit_length() - 1) + half
            shifted = s << half
            if (mask[i] & shifted) != 0:
                ok = False
                break
            bits ^= lb
        if ok:
            right_list.append(s)
            right_masks.append(popcount[s])
    
    # Build map from (size, forbidden_mask) to list of subsets
    right_map = {}
    for idx, s in enumerate(right_list):
        sz = right_masks[idx]
        # Compute forbidden mask for left side
        # For each bit j in s, it forbids all left neighbors of node (half + j)
        forb = 0
        bits = s
        while bits:
            lb = bits & -bits
            j = (lb.bit_length() - 1)
            node = half + j
            forb |= (mask[node] & ((1 << half) - 1))
            bits ^= lb
        key = (sz, forb)
        if key not in right_map:
            right_map[key] = []
        right_map[key].append(s)
    
    # Search for valid combination
    for left_idx, left in enumerate(left_list):
        left_sz = left_masks[left_idx]
        # Compute forbidden mask for right side
        forb_right = 0
        bits = left
        while bits:
            lb = bits & -bits
            i = (lb.bit_length() - 1)
            forb_right |= (mask[i] >> half)
            bits ^= lb
        
        need = k - left_sz
        if need < 0 or need > right_size:
            continue
        
        key = (need, forb_right)
        if key in right_map:
            right = right_map[key][0]
            # Found solution
            result = []
            for i in range(left_size):
                if (left >> i) & 1:
                    result.append(i + 1)
            for j in range(right_size):
                if (right >> j) & 1:
                    result.append(half + j + 1)
            print("YES")
            print(' '.join(map(str, result)))
            return
    
    print("NO")

if __name__ == "__main__":
    solve()
