```python
import sys

# The problem states that any closed loop has an even number of devices.
# This means the graph is bipartite.
# A bipartite graph can be partitioned into two sets, say A and B,
# such that all edges connect a vertex in A to a vertex in B.
# An interference-free collection is an independent set.
# In a bipartite graph, the maximum independent set size is n - (size of minimum vertex cover).
# By Konig's theorem, in bipartite graphs, min vertex cover = max matching.
# However, we don't need the maximum independent set. We need to know if we can
# pick exactly k vertices.
# Since the graph is bipartite, each connected component can be colored with 2 colors.
# Let a component have c1 vertices of color 1 and c2 vertices of color 2.
# We can pick either all c1 vertices or all c2 vertices to be in our independent set.
# This is a variation of the subset sum problem (or knapsack).
# For each component i, we have two choices: pick c1_i vertices or c2_i vertices.
# We want to know if there exists a selection such that sum = k.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_nodes = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp_nodes[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append((len(comp_nodes[0]), len(comp_nodes[1]), comp_nodes[0], comp_nodes[1]))
            
    # dp[i][j] = can we get sum j using first i components?
    # To reconstruct, store which choice was made.
    # dp[j] stores the choice made for each component to reach sum j.
    # Since n is 10^5, we use a bitset-like approach or simple DP.
    
    dp = {0: []}
    for c1, c2, nodes1, nodes2 in components:
        new_dp = {}
        for s, path in dp.items():
            if s + c1 <= k:
                new_dp[s + c1] = path + [(nodes1, 0)]
            if s + c2 <= k:
                new_dp[s + c2] = path + [(nodes2, 1)]
        dp = new_dp
        
    if k in dp:
        print("YES")
        res = []
        for nodes, _ in dp[k]:
            res.extend(nodes)
        print(*(res))
    else:
        print("NO")

solve()
```
