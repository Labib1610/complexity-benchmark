import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_nodes = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp_nodes[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append(comp_nodes)
            
    # dp[i][j] = can we pick j nodes from first i components?
    # Since we need to reconstruct, we store which choice was made.
    # dp[i][j] = (choice_for_comp_i, prev_j)
    # choice_for_comp_i: 0 means pick comp_nodes[0], 1 means pick comp_nodes[1]
    
    dp = {0: None}
    for i, (c0, c1) in enumerate(components):
        new_dp = {}
        s0, s1 = len(c0), len(c1)
        for j, _ in dp.items():
            if j + s0 <= k:
                new_dp[j + s0] = (0, j)
            if j + s1 <= k:
                new_dp[j + s1] = (1, j)
        dp = new_dp
        
    if k not in dp:
        print("NO")
        return
    
    print("YES")
    res = []
    curr_k = k
    for i in range(len(components) - 1, -1, -1):
        choice, prev_k = dp[curr_k]
        res.extend(components[i][choice])
        curr_k = prev_k
        
    print(*(res))

solve()
