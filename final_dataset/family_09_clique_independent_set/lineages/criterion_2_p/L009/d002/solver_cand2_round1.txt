```python
import sys

# The problem states that for any link (u, v), dist(u, C) and dist(v, C) differ by 1.
# This property implies that the graph is bipartite.
# In a bipartite graph, the vertices can be partitioned into two sets, L and R,
# such that all edges connect a vertex in L to a vertex in R.
# An interference-free set is an independent set.
# In a bipartite graph, the two partitions L and R are independent sets.
# Any subset of L is an independent set, and any subset of R is an independent set.
# Let |L| = n1 and |R| = n2.
# We can form an independent set of size k if k <= n1 or k <= n2.
# If k > n1 and k > n2, it is impossible to pick k independent vertices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    partition0 = []
    partition1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            partition0.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            partition0.append(v)
                        else:
                            partition1.append(v)
                        stack.append((v, color[v]))
    
    if len(partition0) >= k:
        print("YES")
        print(*(partition0[:k]))
    elif len(partition1) >= k:
        print("YES")
        print(*(partition1[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
