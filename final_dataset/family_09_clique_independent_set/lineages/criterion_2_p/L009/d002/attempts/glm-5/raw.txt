```python
import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    sets = [[], []]
    
    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            stack = [start]
            while stack:
                u = stack.pop()
                sets[color[u] - 1].append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
    
    if len(sets[0]) >= k:
        print("YES")
        print(' '.join(map(str, sets[0][:k])))
    elif len(sets[1]) >= k:
        print("YES")
        print(' '.join(map(str, sets[1][:k])))
    else:
        print("NO")

solve()
```
