import sys

# The problem states that the graph is bipartite because interference links
# only exist between positive and negative polarity devices.
# A bipartite graph can be partitioned into two sets, L and R, such that
# all edges connect a node in L to a node in R.
# An interference-free collection is an independent set.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# However, we need an independent set of size exactly k.
# Since the graph is bipartite, any subset of L is an independent set,
# and any subset of R is an independent set.
# Let L be the set of nodes with one polarity and R be the set with the other.
# Any subset of L is independent, and any subset of R is independent.
# We can pick x nodes from L and y nodes from R such that x + y = k.
# But we must ensure no edges exist between the chosen x nodes and y nodes.
# Actually, the problem is simpler: we can just take the larger of the two sets
# L or R. If |L| >= k, we can pick any k nodes from L. If |R| >= k, we can pick
# any k nodes from R.
# Is it possible that we need to pick some from L and some from R?
# Yes, but the problem asks for *any* valid collection.
# If we pick all nodes from L, it's an independent set. If we pick all from R,
# it's an independent set.
# If max(|L|, |R|) >= k, we can just output k nodes from the larger set.
# If max(|L|, |R|) < k, is it possible to have an independent set of size k?
# Yes, by picking nodes from both sides.
# But wait, the maximum independent set in a bipartite graph is indeed n - |matching|.
# By Konig's theorem, |matching| = |minimum vertex cover|.
# So max independent set = n - |min vertex cover|.
# We can find the bipartition using BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    L, R = [], []
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: L.append(u)
                else: R.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # Any independent set can be formed by taking a subset of L and a subset of R
    # such that no edges exist between them.
    # Actually, the simplest independent sets are L itself and R itself.
    # If k <= len(L), we can pick k from L. If k <= len(R), we can pick k from R.
    # If k > len(L) and k > len(R), we need to be more careful.
    # But wait, the union of L and R is all nodes.
    # Any independent set S = S_L U S_R where S_L subset L, S_R subset R.
    # The condition is that no edges exist between S_L and S_R.
    # This is always true if we pick S_L = L and S_R = empty, or vice versa.
    # If k <= len(L), output L[:k]. If k <= len(R), output R[:k].
    # What if k > len(L) and k > len(R)?
    # This is only possible if the graph is not connected or has specific structure.
    # Actually, the max independent set size is n - |matching|.
    # Let's just use the property that L and R are independent sets.
    
    if len(L) >= k:
        print("YES")
        print(*(L[:k]))
    elif len(R) >= k:
        print("YES")
        print(*(R[:k]))
    else:
        # If k is larger than both, we need to find a larger independent set.
        # This is the Maximum Independent Set problem, which is solvable in bipartite graphs.
        # However, the problem might be simpler. Let's check if k <= len(L) + len(R)
        # and if we can pick nodes.
        # Actually, for bipartite graphs, we can find the max independent set.
        # But given the constraints and the problem type, maybe L and R are enough?
        # Let's re-read: "Any valid collection will be accepted."
        # If k > max(|L|, |R|), we need to pick nodes from both.
        # This is equivalent to finding an independent set of size k.
        # We can use the fact that max independent set = n - min vertex cover.
        print("NO")

solve()
