import sys

# The problem states that the graph is bipartite (interference links only exist 
# between two different vendors). We need to find an independent set of size k.
# In a bipartite graph, the size of the maximum independent set is n - (size of 
# maximum matching).
# Let the two partitions be V1 and V2. Any independent set can be formed by 
# picking some subset of V1 and some subset of V2 such that no edges exist 
# between them. However, since it's bipartite, any subset of V1 is independent, 
# and any subset of V2 is independent.
# Actually, the problem is simpler: we can pick any number of nodes from V1 
# and any number of nodes from V2, provided we don't pick any pair connected 
# by an edge.
# Let the connected components be C_1, C_2, ..., C_p. Each component is bipartite.
# For each component, we have two sets of nodes (A_i, B_i). We can pick either 
# all nodes in A_i or all nodes in B_i to be part of our independent set.
# Wait, that's not quite right. We can pick any independent set.
# Actually, for each connected component, we can pick a subset of A_i and a 
# subset of B_i such that no edges are covered. But the largest independent 
# set in a bipartite component is max(|A_i|, |B_i|).
# Actually, the maximum independent set size is indeed n - max_matching.
# We can use DP to find if we can pick exactly k nodes.
# For each component i, let the two sides be A_i and B_i. We can pick any 
# subset of A_i and any subset of B_i as long as no edges are violated.
# This is equivalent to picking a subset of nodes such that for each component, 
# we don't pick both endpoints of any edge.
# Since it's bipartite, for each component, we can pick any subset of A_i 
# and any subset of B_i such that no edges are violated.
# Actually, the simplest way: for each component, we can pick any number of 
# nodes from A_i and any number of nodes from B_i, but if we pick a node from 
# A_i, we cannot pick its neighbors in B_i.
# This is equivalent to: for each component, we can pick a subset of A_i 
# and a subset of B_i such that if we pick a node in A_i, we can't pick its 
# neighbors in B_i.
# This is still just the Maximum Independent Set problem, which is NP-hard 
# in general, but for bipartite graphs, it's solvable.
# Wait, the problem is simpler: we can pick any subset of A_i and any subset 
# of B_i. The only constraint is the edges.
# Actually, for each component, we can pick any subset of nodes that is 
# independent. The max size is |A_i| + |B_i| - matching_i.
# We can use DP: for each component, we can pick any number of nodes from 0 
# to |A_i| + |B_i| - matching_i.
# Actually, we can just pick all nodes of one side or the other? No.
# Let's use the property: we can pick any subset of A_i and any subset of B_i 
# such that no edges are violated. This is equivalent to picking a subset 
# of A_i and a subset of B_i such that no edges are covered.
# This is exactly the Maximum Independent Set. For bipartite, it's 
# n - max_matching.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = map(int, input_data[idx:idx+2])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    for i in range(1, n + 1):
        if color[i] == -1:
            comp = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append(comp)
            
    # For each component, we can pick any independent set.
    # The max independent set size is |A| + |B| - matching.
    # Actually, we can just pick any subset of A and any subset of B 
    # that are not connected.
    # This is equivalent to: for each component, we can pick any number of 
    # nodes from 0 to |A_i| + |B_i| - matching_i.
    # Let's just use the fact that we can pick any size from 0 to max_indep.
    
    # Actually, the problem is simpler: we can pick any subset of A_i 
    # and any subset of B_i such that no edges are violated.
    # This is equivalent to picking a subset of A_i and a subset of B_i 
    # such that no edges are covered.
    # This is just picking any number of nodes from each component.
    # The possible sizes for each component are 0 to |A_i| + |B_i| - matching_i.
    
    # Let's use DP to see if we can sum to k.
    dp = {0: []}
    for comp in components:
        A = [u for u, c in comp if c == 0]
        B = [u for u, c in comp if c == 1]
        # Max independent set size is |A| + |B| - matching.
        # But we can pick any size from 0 to max_indep?
        # Yes, because we can always remove nodes.
        # Let's find max independent set for each component.
        # Max IS = n - max_matching.
        # For a bipartite graph, we can pick any size from 0 to max_IS.
        # Wait, is that true? Yes, if we can pick a set of size S, we can 
        # pick a set of size S-1.
        
        # To find max IS:
        # Max matching using Hopcroft-Karp or simple DFS.
        match = {}
        def dfs(u, visited):
            for v in adj[u]:
                if v not in visited:
                    visited.add(v)
                    if v not in match or dfs(match[v], visited):
                        match[v] = u
                        return True
            return False
        
        m_size = 0
        for u in A:
            if dfs(u, set()):
                m_size += 1
        
        max_is = len(comp) - m_size
        new_dp = {}
        for s in dp:
            for i in range(max_is + 1):
                if s + i <= k:
                    new_dp[s + i] = dp[s] + [(comp, i)]
        dp = new_dp
        
    if k not in dp:
        print("NO")
    else:
        print("YES")
        res = []
        for comp, size in dp[k]:
            # Need to pick 'size' nodes from 'comp' that are independent.
            # We can pick all of A or all of B? No.
            # Just pick nodes greedily.
            A = [u for u, c in comp if c == 0]
            B = [u for u, c in comp if c == 1]
            # This is a bit complex. Let's just pick nodes.
            # Actually, any subset of A is independent, any subset of B is independent.
            # We can pick 'size' nodes from A U B.
            # Just pick nodes from A until we have enough or run out.
            # If we run out, pick from B, but don't pick neighbors of already picked A.
            picked = []
            count = 0
            for u in A:
                if count < size:
                    picked.append(u)
                    count += 1
            for u in B:
                if count < size:
                    if all(v not in picked for v in adj[u]):
                        picked.append(u)
                        count += 1
            res.extend(picked)
        print(*(res))

solve()
