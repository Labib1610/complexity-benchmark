import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_nodes = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            comp_nodes[0].append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        comp_nodes[1 - c].append(v)
                        stack.append((v, 1 - c))
            components.append(comp_nodes)

    # dp[i][j] = can we pick j nodes from first i components?
    # Since we need to reconstruct, store which choice was made.
    # dp[i][j] = (choice_for_comp_i, prev_j)
    # choice_for_comp_i: 0 (pick comp_nodes[0]), 1 (pick comp_nodes[1])
    
    num_comps = len(components)
    dp = [{} for _ in range(num_comps + 1)]
    dp[0][0] = None
    
    for i in range(num_comps):
        c0 = len(components[i][0])
        c1 = len(components[i][1])
        for j in dp[i]:
            if j + c0 <= k:
                dp[i+1][j + c0] = (0, j)
            if j + c1 <= k:
                dp[i+1][j + c1] = (1, j)
                
    if k not in dp[num_comps]:
        print("NO")
        return
    
    print("YES")
    res = []
    curr_k = k
    for i in range(num_comps, 0, -1):
        choice, prev_k = dp[i][curr_k]
        res.extend(components[i-1][choice])
        curr_k = prev_k
        
    print(*(res))

solve()
