```python
import sys

# The problem states that the graph is bipartite because all cycles are even.
# In a bipartite graph, we can color nodes with 0 and 1 such that no two 
# adjacent nodes have the same color.
# Let the two sets be S0 and S1 with sizes |S0| and |S1|.
# We need to pick k nodes such that no two are connected.
# This means we can pick any number of nodes from S0 and any number from S1,
# provided we don't pick two connected nodes.
# Actually, in a bipartite graph, any subset of S0 is an independent set,
# and any subset of S1 is an independent set.
# Wait, the condition is: we need to pick k nodes such that no two are connected.
# This is equivalent to picking a subset of S0 and a subset of S1.
# Since no edges exist within S0 and no edges exist within S1,
# any subset of S0 is independent, and any subset of S1 is independent.
# However, we can pick nodes from both S0 and S1 as long as we don't pick
# any pair (u, v) where u in S0, v in S1, and (u, v) is an edge.
# Actually, the problem is simpler: we have connected components.
# Each component is bipartite. Let a component have parts A_i and B_i.
# We can pick any subset of A_i or any subset of B_i.
# But we can't pick nodes from both A_i and B_i if they are connected.
# Actually, for each connected component, we can either pick a subset of A_i
# or a subset of B_i.
# Let the component have parts A_i and B_i. We can pick x_i nodes from A_i
# (0 <= x_i <= |A_i|) OR y_i nodes from B_i (0 <= y_i <= |B_i|).
# This is a variation of the subset sum problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    for i in range(1, n + 1):
        if color[i] == -1:
            s0, s1 = [], []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: s0.append(u)
                else: s1.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append((s0, s1))
            
    # dp[i][j] = can we pick j nodes using first i components
    # Since we need to reconstruct, store choices.
    # Each component i offers choices: pick any size from 0 to |s0| or 0 to |s1|.
    # This is equivalent to: for each component, we have a set of possible counts:
    # {0, 1, ..., |s0|} U {0, 1, ..., |s1|}.
    
    dp = {0: []}
    for i, (s0, s1) in enumerate(components):
        new_dp = {}
        n0, n1 = len(s0), len(s1)
        for current_k, path in dp.items():
            for count in range(n0 + 1):
                if current_k + count <= k:
                    new_dp[current_k + count] = path + [(0, count)]
            for count in range(n1 + 1):
                if current_k + count <= k:
                    new_dp[current_k + count] = path + [(1, count)]
        dp = new_dp
        
    if k not in dp:
        print("NO")
    else:
        print("YES")
        res = []
        for i, (choice, count) in enumerate(dp[k]):
            if count == 0: continue
            if choice == 0:
                res.extend(components[i][0][:count])
            else:
                res.extend(components[i][1][:count])
        print(*(res))

solve()
```
