```python
import sys

# The problem states that any cycle has an even number of links.
# This means the graph is bipartite.
# A bipartite graph can be decomposed into connected components,
# each of which is bipartite (can be colored with 2 colors, say 0 and 1).
# For each connected component, we have two sets of nodes: S0 and S1.
# We can pick either all nodes from S0 or all nodes from S1 to be in our set.
# Let n_i = |S0_i| and m_i = |S1_i| for component i.
# We need to choose one set from each component such that the sum of sizes is k.
# This is a variation of the subset sum problem (knapsack).
# Since the total number of modules is 100,000, we can use bitsets or 
# dynamic programming with optimization.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp0 = []
            comp1 = []
            stack = [(i, 0)]
            color[i] = 0
            comp0.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0: comp0.append(v)
                        else: comp1.append(v)
                        stack.append((v, 1 - c))
            components.append((comp0, comp1))
            
    # dp[s] stores which choice was made for each component to reach sum s
    # dp[s] = (component_index, choice_0_or_1)
    # To optimize, use bitset-like approach or just a dictionary
    dp = {0: []}
    
    for i, (c0, c1) in enumerate(components):
        new_dp = {}
        s0, s1 = len(c0), len(c1)
        for s, path in dp.items():
            if s + s0 <= k:
                new_dp[s + s0] = path + [(i, 0)]
            if s + s1 <= k:
                new_dp[s + s1] = path + [(i, 1)]
        dp = new_dp
        
    if k not in dp:
        print("NO")
    else:
        print("YES")
        res = []
        choices = dp[k]
        chosen_comp = [None] * len(components)
        for comp_idx, choice in choices:
            chosen_comp[comp_idx] = choice
            
        output = []
        for i, (c0, c1) in enumerate(components):
            if chosen_comp[i] == 0:
                output.extend(c0)
            else:
                output.extend(c1)
        print(*(output))

solve()
```
