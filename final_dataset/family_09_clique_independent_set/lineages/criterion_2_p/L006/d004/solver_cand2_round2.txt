```python
import sys

# The problem states the graph is bipartite.
# We need to find an independent set of size k.
# In a bipartite graph, the maximum independent set size is n - (size of minimum vertex cover).
# By Konig's theorem, min vertex cover = max matching.
# So max independent set size = n - max matching.
# However, we need to find if there exists an independent set of size exactly k.
# In a bipartite graph, for each connected component, we can pick either the set of nodes
# in partition A or partition B as an independent set.
# Let a_i and b_i be the sizes of the two partitions in component i.
# We can pick any subset of nodes from the component that is independent.
# Actually, for any bipartite graph, we can pick any number of nodes from 0 to max_independent_set_size?
# No, that's not true. But we can pick any number of nodes from 0 to max_independent_set_size
# if we consider the structure of bipartite components.
# Actually, for each component, we can pick any number of nodes from 0 to max(a_i, b_i)
# if we only pick from one side? No.
# The correct approach: For each connected component, we have two partitions A_i and B_i.
# We can pick any subset of A_i (size |A_i|) or any subset of B_i (size |B_i|).
# Actually, we can pick any number of nodes from 0 to |A_i| by picking a subset of A_i,
# and any number from 0 to |B_i| by picking a subset of B_i.
# Since we can pick any subset of A_i or B_i, we can pick any number of nodes
# from 0 to max(|A_i|, |B_i|) for each component.
# Wait, is it possible to pick more? Yes, by mixing.
# But the problem asks for *any* valid selection.
# The simplest valid selection is to pick nodes from the larger partition of each component.
# Let S_i = max(|A_i|, |B_i|). The total max independent set size is sum(S_i).
# Can we pick any k <= sum(S_i)? Yes, because we can pick any subset of size x <= |A_i|
# from A_i, or size y <= |B_i| from B_i.
# So we just need to check if k <= sum(max(|A_i|, |B_i|)).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_a = []
            comp_b = []
            stack = [(i, 0)]
            color[i] = 0
            comp_a.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0: comp_a.append(v)
                        else: comp_b.append(v)
                        stack.append((v, 1 - c))
            components.append((comp_a, comp_b))
            
    total_max = sum(max(len(a), len(b)) for a, b in components)
    
    if k > total_max:
        print("NO")
        return
    
    print("YES")
    res = []
    rem = k
    for a, b in components:
        if rem <= 0: break
        if len(a) >= len(b):
            take = min(rem, len(a))
            res.extend(a[:take])
            rem -= take
        else:
            take = min(rem, len(b))
            res.extend(b[:take])
            rem -= take
            
    print(*(res))

solve()
```
