```python
import sys

# The problem states the graph is bipartite (guaranteed free-rotation property).
# We need to find an independent set of size k.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# Let the two partitions be A and B.
# Any independent set can be formed by taking some subset of A and some subset of B.
# Specifically, if we take all of A, we can't take any neighbors of A in B.
# The problem asks for *any* independent set of size k.
# A simple way to find a large independent set is to take the larger partition
# of the bipartite graph. Let the partitions be V0 and V1.
# Any subset of V0 is an independent set, and any subset of V1 is an independent set.
# However, we can potentially pick more than max(|V0|, |V1|) by picking
# vertices from both sides.
# Actually, for any bipartite graph, we can find the maximum independent set
# using Konig's theorem: |MIS| = n - |Max Matching|.
# But we just need *any* independent set of size k.
# Since the graph is bipartite, we can color it with 2 colors (0 and 1).
# Let V0 be the set of nodes with color 0, V1 be the set of nodes with color 1.
# Any subset of V0 is independent, any subset of V1 is independent.
# To get size k, we can just pick k nodes from the larger of V0 or V1 if k <= max(|V0|, |V1|).
# Wait, is it possible to have an independent set of size k even if k > max(|V0|, |V1|)?
# Yes, e.g., a graph with no edges, |V0|=1, |V1|=1, k=2.
# The maximum independent set in a bipartite graph can be found by:
# 1. Find max matching M.
# 2. Let U be the set of unmatched vertices in V0.
# 3. Let Z be the set of vertices reachable from U via alternating paths.
# 4. MIS = (V0 \ Z) U (V1 ∩ Z).
# This MIS has size |V0| - |V0 ∩ Z| + |V1 ∩ Z| = |V0| + (|V1 ∩ Z| - |V0 ∩ Z|).
# Actually, a simpler approach:
# Any independent set of size k exists if k <= |MIS|.
# We can find the MIS and then just take any k vertices from it.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    v0, v1 = [], []
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: v0.append(u)
                else: v1.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # MIS construction:
    # Max matching using Hopcroft-Karp or simple DFS
    match = [-1] * (n + 1)
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    for u in v0:
        dfs(u, [False] * (n + 1))
    
    # Z = reachable from unmatched V0 via alternating paths
    matched_v0 = {match[v] for v in v1 if match[v] != -1}
    unmatched_v0 = [u for u in v0 if u not in matched_v0]
    
    z = [False] * (n + 1)
    stack = unmatched_v0
    for u in stack: z[u] = True
    while stack:
        u = stack.pop()
        if color[u] == 0:
            for v in adj[u]:
                if not z[v] and match[v] != u:
                    z[v] = True
                    stack.append(v)
        else:
            if match[u] != -1 and not z[match[u]]:
                z[match[u]] = True
                stack.append(match[u])
                
    mis = [i for i in v0 if not z[i]] + [i for i in v1 if z[i]]
    
    if len(mis) >= k:
        print("YES")
        print(*(mis[:k]))
    else:
        print("NO")

solve()
```
