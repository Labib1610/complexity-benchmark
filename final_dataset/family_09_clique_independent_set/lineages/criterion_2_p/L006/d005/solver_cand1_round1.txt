```python
import sys

# The problem states that all cycles are even, which means the graph is bipartite.
# We can decompose the graph into connected components.
# For each component, we have two sets of vertices (bipartition sets A and B).
# We can choose either all vertices in A or all vertices in B as an independent set.
# Let |A_i| and |B_i| be the sizes of the two sets in component i.
# We want to choose a subset of components and for each, pick either A_i or B_i
# such that the sum of sizes is exactly k.
# This is a variation of the subset sum problem (knapsack).
# Since we have many components, we can use bitsets or dynamic programming.
# Given n=100,000, we use the property that we have many items with values |A_i| and |B_i|.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_a = []
            comp_b = []
            stack = [(i, 0)]
            color[i] = 0
            comp_a.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0: comp_a.append(v)
                        else: comp_b.append(v)
                        stack.append((v, 1 - c))
            components.append((comp_a, comp_b))

    # dp[s] stores which choice was made to reach sum s
    # dp[s] = (comp_idx, choice_type)
    # To optimize, use bitset-like approach or grouping
    dp = {0: []}
    
    for i, (a, b) in enumerate(components):
        new_dp = dp.copy()
        len_a, len_b = len(a), len(b)
        for s, path in dp.items():
            if s + len_a <= k and (s + len_a) not in new_dp:
                new_dp[s + len_a] = path + [(i, 0)]
            if s + len_b <= k and (s + len_b) not in new_dp:
                new_dp[s + len_b] = path + [(i, 1)]
        dp = new_dp
        if k in dp:
            break
            
    if k not in dp:
        print("NO")
    else:
        print("YES")
        res = []
        for comp_idx, choice in dp[k]:
            if choice == 0:
                res.extend(components[comp_idx][0])
            else:
                res.extend(components[comp_idx][1])
        print(*(res))

solve()
```
