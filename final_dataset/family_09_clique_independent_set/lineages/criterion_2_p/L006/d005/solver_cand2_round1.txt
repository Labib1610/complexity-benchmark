```python
import sys

# The problem states that the network can be oriented such that every station 
# is either a pure source or a pure sink. This is the definition of a bipartite graph.
# In a bipartite graph, we can partition the vertices into two sets, L and R, 
# such that all edges connect a vertex in L to a vertex in R.
# The maximum independent set in a bipartite graph is n - (size of maximum matching).
# However, we need to find an independent set of size exactly k.
# Since the graph is bipartite, we can find the two partitions using BFS/DFS.
# Let the partitions be V1 and V2.
# Any subset of V1 is an independent set, and any subset of V2 is an independent set.
# The maximum independent set size is n - |matching|.
# We can find the maximum independent set by taking all vertices in one partition 
# that are not in the minimum vertex cover.
# Actually, for a bipartite graph, the size of the maximum independent set is 
# n - |maximum matching|. Any size from 0 up to the max independent set size 
# is not necessarily achievable, but here we just need to check if k <= max_independent_set.
# Wait, the problem asks for *exactly* k. In a bipartite graph, if we have 
# an independent set of size S, we can always find independent sets of size 
# S-1, S-2, ..., 0 by removing vertices.
# So we just need to find the maximum independent set size.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    v1, v2 = [], []
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: v1.append(u)
                else: v2.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # Max Independent Set in bipartite graph = n - Max Matching
    # Hopcroft-Karp or simple DFS matching
    match = [-1] * (n + 1)
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    matching_size = 0
    for u in v1:
        visited = [False] * (n + 1)
        if dfs(u, visited):
            matching_size += 1
            
    # Max Independent Set size
    max_is_size = n - matching_size
    
    if k > max_is_size:
        print("NO")
        return
    
    # To find an independent set of size k:
    # Min Vertex Cover = set of vertices that cover all edges.
    # Max Independent Set = V \ Min Vertex Cover.
    # Konig's theorem: Min Vertex Cover size = Max Matching size.
    # Construction: Let U be the set of unmatched vertices in V1.
    # Let Z be the set of vertices reachable from U via alternating paths.
    # Min Vertex Cover = (V1 \ Z) U (V2 \cap Z).
    
    z = [False] * (n + 1)
    def dfs_z(u):
        z[u] = True
        for v in adj[u]:
            if color[u] == 0: # u in V1, edge to v in V2
                if v != match[u] and not z[v]:
                    dfs_z(v)
            else: # u in V2, edge to v in V1
                if v == match[u] and not z[v]:
                    dfs_z(v)
                    
    unmatched_v1 = [u for u in v1 if match[u] == -1]
    for u in unmatched_v1:
        if not z[u]: dfs_z(u)
        
    mvc = []
    for u in v1:
        if not z[u]: mvc.append(u)
    for u in v2:
        if z[u]: mvc.append(u)
        
    mis = [i for i in range(1, n + 1) if i not in set(mvc)]
    
    print("YES")
    print(*(mis[:k]))

solve()
```
