```python
import sys

# The problem states the graph is bipartite.
# We need to find an independent set of size k.
# In a bipartite graph, the size of the maximum independent set is n - (size of maximum matching).
# Let M be the size of the maximum matching. The maximum independent set size is n - M.
# However, we need to find if there exists an independent set of size k.
# Since it's bipartite, we can find the maximum independent set by taking all vertices
# in one partition of the complement of the minimum vertex cover.
# Actually, for any bipartite graph, we can find the maximum independent set by:
# 1. Finding a maximum matching.
# 2. Constructing the minimum vertex cover using Konig's theorem.
# 3. The complement of the minimum vertex cover is a maximum independent set.
# Any subset of an independent set is also an independent set.
# So if k <= max_independent_set_size, we can pick k vertices from the max independent set.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Bipartite coloring
    color = [-1] * (n + 1)
    partition = [[], []]
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                partition[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # Max matching using Hopcroft-Karp or simple DFS
    match = [-1] * (n + 1)
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    matching_size = 0
    for u in partition[0]:
        visited = [False] * (n + 1)
        if dfs(u, visited):
            matching_size += 1
            
    # Konig's theorem: Min Vertex Cover size = Max Matching size
    # Max Independent Set size = n - Max Matching size
    if k > (n - matching_size):
        print("NO")
        return

    # Find Max Independent Set
    # Let U be partition[0], V be partition[1]
    # Min Vertex Cover S = (U \ Z) U (V \cap Z) where Z is reachable from unmatched U
    unmatched_u = [u for u in partition[0] if not any(match[v] == u for v in adj[u])]
    
    reachable = [False] * (n + 1)
    stack = unmatched_u
    for u in stack: reachable[u] = True
    while stack:
        u = stack.pop()
        if color[u] == 0:
            for v in adj[u]:
                if match[v] != u and not reachable[v]:
                    reachable[v] = True
                    stack.append(v)
        else:
            v = match[u]
            if v != -1 and not reachable[v]:
                reachable[v] = True
                stack.append(v)
                
    # Independent set: (U \cap reachable) U (V \setminus reachable)
    indep_set = []
    for u in partition[0]:
        if reachable[u]: indep_set.append(u)
    for v in partition[1]:
        if not reachable[v]: indep_set.append(v)
        
    print("YES")
    print(*(indep_set[:k]))

solve()
```
