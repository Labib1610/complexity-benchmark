import sys

# The problem states the assembly is fully functional, meaning it is bipartite.
# We need to find an independent set of size k.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# However, we need exactly k. Since it's bipartite, we can partition the graph
# into two sets, A and B, such that all edges are between A and B.
# Any subset of A is an independent set, and any subset of B is an independent set.
# We can pick any number of nodes from A and any number from B as long as we don't
# pick two connected nodes. But wait, the problem is simpler:
# Any bipartite graph has two independent sets: the two partitions.
# Let the partitions be V1 and V2. Any subset of V1 is independent, and any subset of V2 is independent.
# Actually, we can pick any subset of V1 and any subset of V2.
# The maximum independent set size is |V1| + |V2| - |matching|? No, that's for vertex cover.
# The maximum independent set size is indeed n - |matching|.
# But we just need to know if we can pick k.
# Since it's bipartite, we can pick any subset of V1 and any subset of V2.
# Wait, that's not true. We can pick any subset of V1, and any subset of V2.
# The union of any subset of V1 and any subset of V2 is an independent set.
# So the maximum independent set size is |V1| + |V2| = n.
# Wait, the condition is "no two gears selected can share an engagement point".
# This is exactly the definition of an independent set.
# In a bipartite graph, the set of all nodes in V1 is an independent set.
# The set of all nodes in V2 is an independent set.
# Any subset of V1 is an independent set. Any subset of V2 is an independent set.
# Can we pick k nodes? Yes, if k <= |V1| + |V2|? No, that's only if there are no edges.
# The maximum independent set size is n - |max_matching|.
# Let's use the property that we can pick any subset of V1 and any subset of V2.
# Actually, we can pick any subset of V1 and any subset of V2 such that no edges exist between them.
# But all edges are between V1 and V2. So we can pick any subset of V1 and any subset of V2.
# The only constraint is we cannot pick u from V1 and v from V2 if there is an edge (u, v).
# This is the Maximum Independent Set problem, which is NP-hard in general,
# but for bipartite graphs, it equals n - |max_matching|.
# We need to find if there exists an independent set of size k.
# Actually, in a bipartite graph, we can always find an independent set of size at least |V1| and at least |V2|.
# The size of the maximum independent set is n - |matching|.
# We can greedily pick nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    v1, v2 = [], []
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: v1.append(u)
                else: v2.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # Max independent set in bipartite graph:
    # Use Hopcroft-Karp or simple DFS for matching
    match = [-1] * (n + 1)
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    matching_size = 0
    for u in v1:
        visited = [False] * (n + 1)
        if dfs(u, visited):
            matching_size += 1
            
    if k > n - matching_size:
        print("NO")
    else:
        print("YES")
        # Vertex cover is the set of nodes in matching.
        # Independent set is V \ Vertex Cover.
        # This is a standard construction.
        is_in_vc = [False] * (n + 1)
        # To find min vertex cover in bipartite:
        # 1. Find max matching M.
        # 2. Let U be the set of unmatched vertices in V1.
        # 3. Let Z be the set of vertices connected to U by alternating paths.
        # 4. Min VC = (V1 \ Z) U (V2 ∩ Z).
        # Independent set = (V1 ∩ Z) U (V2 \ Z).
        
        z = [False] * (n + 1)
        matched_in_v1 = {match[v] for v in v2 if match[v] != -1}
        q = [u for u in v1 if u not in matched_in_v1]
        for u in q: z[u] = True
        
        while q:
            u = q.pop()
            for v in adj[u]:
                if not z[v] and v != match[u]:
                    z[v] = True
                    if match[v] != -1:
                        z[match[v]] = True
                        q.append(match[v])
        
        indep_set = []
        for u in v1:
            if z[u]: indep_set.append(u)
        for v in v2:
            if not z[v]: indep_set.append(v)
            
        print(*(indep_set[:k]))

solve()
