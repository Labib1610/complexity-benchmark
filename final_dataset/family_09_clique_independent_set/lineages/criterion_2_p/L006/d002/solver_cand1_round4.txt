import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_nodes = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp_nodes[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append(comp_nodes)
            
    # dp[i][j] = can we pick j nodes from first i components?
    # Since we need to pick exactly k, and each component offers two sets (size A, size B)
    # This is a variation of the subset sum problem.
    # dp[j] stores the possible ways to pick j nodes.
    # To reconstruct, we store which choice was made.
    
    # dp[j] = (comp_idx, choice_idx)
    dp = {0: None}
    
    for i, (c0, c1) in enumerate(components):
        new_dp = {}
        s0, s1 = len(c0), len(c1)
        for j in dp:
            if j + s0 <= k:
                new_dp[j + s0] = (i, 0)
            if j + s1 <= k:
                new_dp[j + s1] = (i, 1)
        dp = new_dp
        
    if k not in dp:
        print("NO")
        return
    
    print("YES")
    res = []
    curr_k = k
    while curr_k > 0:
        comp_idx, choice = dp[curr_k]
        c0, c1 = components[comp_idx]
        if choice == 0:
            res.extend(c0)
            curr_k -= len(c0)
        else:
            res.extend(c1)
            curr_k -= len(c1)
        # Find previous state
        # This is tricky, so let's store the full path or re-calculate
        # Actually, let's store the path in dp
        
    # Re-implementing with path tracking
    dp = {0: []}
    for i, (c0, c1) in enumerate(components):
        new_dp = {}
        s0, s1 = len(c0), len(c1)
        for j, path in dp.items():
            if j + s0 <= k:
                new_dp[j + s0] = path + [(c0)]
            if j + s1 <= k:
                new_dp[j + s1] = path + [(c1)]
        dp = new_dp
        
    if k not in dp:
        print("NO")
    else:
        ans = []
        for group in dp[k]:
            ans.extend(group)
        print(*(ans))

solve()
