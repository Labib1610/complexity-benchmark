```python
import sys
from collections import defaultdict, deque

def read_input():
    n, m, k = map(int, sys.stdin.readline().split())
    edges = []
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    return n, m, k, edges, adj

def bipartite_partition(n, adj):
    color = [-1] * (n + 1)
    sets = [[], []]
    for i in range(1, n + 1):
        if color[i] == -1:
            queue = deque([i])
            color[i] = 0
            sets[0].append(i)
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        sets[color[v]].append(v)
                        queue.append(v)
                    elif color[v] == color[u]:
                        return None
    return sets

def main():
    n, m, k, edges, adj = read_input()
    
    if m == 0:
        if k <= n:
            print("YES")
            print(' '.join(map(str, range(1, min(k+1, n+1)))))
        else:
            print("NO")
        return
    
    bipartition = bipartite_partition(n, adj)
    if bipartition is None:
        print("NO")
        return
        
    set_a, set_b = bipartition
    size_a, size_b = len(set_a), len(set_b)
    
    if k > size_a and k > size_b:
        print("NO")
        return
    
    if k <= size_a:
        print("YES")
        print(' '.join(map(str, set_a[:k])))
    elif k <= size_b:
        print("YES")
        print(' '.join(map(str, set_b[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
