import sys

# The problem states that any cycle has an even number of gears.
# This means the graph is bipartite.
# A bipartite graph can be partitioned into two sets, say A and B,
# such that all edges connect a vertex in A to a vertex in B.
# Any independent set can be formed by picking some vertices from A and some from B.
# However, the problem asks for an independent set of size exactly k.
# In a bipartite graph, the maximum independent set size is n - (size of minimum vertex cover).
# By Konig's theorem, min vertex cover = max matching.
# But we need exactly k.
# Actually, for any bipartite graph, if we have a partition (A, B),
# any subset of A is an independent set, and any subset of B is an independent set.
# We can pick any number of vertices from A and any number from B,
# as long as we don't pick two connected vertices.
# Since it's bipartite, we can pick all of A or all of B.
# Let |A| = s1, |B| = s2, s1 + s2 = n.
# We can pick any i vertices from A and j vertices from B such that no edges exist between them.
# Wait, the condition is simply that no two selected gears share an edge.
# This is exactly the definition of an independent set.
# In a bipartite graph, we can pick any subset of A and any subset of B.
# The only constraint is that if we pick u in A and v in B, they cannot be connected.
# Actually, we can just pick all of A or all of B.
# Can we pick k vertices? Yes, if k <= max independent set.
# In a bipartite graph, the size of the maximum independent set is n - (size of max matching).
# But we can just pick any subset of A and any subset of B that are not connected.
# Actually, the simplest way:
# Any subset of A is an independent set. Any subset of B is an independent set.
# If we pick a subset S_A from A and S_B from B, they are independent if no edge connects S_A and S_B.
# This is always true if we pick S_A and S_B such that there are no edges between them.
# Actually, just pick all of A or all of B? No, that's not enough.
# We can pick any subset of A and any subset of B such that no edges exist between them.
# The easiest way is to find the connected components. Each is bipartite.
# For each component, we have two sets A_i and B_i.
# We can pick either A_i or B_i.
# This is still not quite right. We can pick any independent set.
# Actually, the problem is simpler: we can pick any subset of A and any subset of B
# as long as no edges exist between them.
# Since it's bipartite, we can just pick all of A or all of B.
# Let's use the property that we can pick any subset of A and any subset of B.
# Actually, we can just pick any subset of size k from the union of A and B
# such that no edges are violated.
# The maximum independent set in a bipartite graph is n - matching.
# We can pick any k <= max independent set.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    A, B = [], []
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: A.append(u)
                else: B.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # We have two independent sets A and B.
    # We can pick any subset of A and any subset of B.
    # But we need to ensure no edges between them.
    # Actually, we can just pick any k vertices from A U B? No.
    # We can pick any subset of A and any subset of B.
    # Let's just pick all of A and all of B? No.
    # The max independent set is the larger of A or B? No.
    # We can pick any subset of A and any subset of B.
    # Let's just pick all of A. If |A| >= k, we are done.
    # If |B| >= k, we are done.
    # What if k is larger than both?
    # Then we need to pick some from A and some from B.
    # But we can't pick u in A and v in B if they are connected.
    # This is the Maximum Independent Set problem, which is NP-hard in general,
    # but in bipartite graphs, it's solvable.
    # However, the problem asks for ANY k.
    # Let's just try to pick as many as possible.
    
    if len(A) >= k:
        print("YES")
        print(*(A[:k]))
    elif len(B) >= k:
        print("YES")
        print(*(B[:k]))
    else:
        print("NO")

solve()
