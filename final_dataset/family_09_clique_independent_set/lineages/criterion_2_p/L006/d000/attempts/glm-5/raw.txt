```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    comp = []
    for start in range(1, n + 1):
        if color[start] == 0:
            stack = [start]
            color[start] = 1
            side0 = []
            side1 = [start]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                        if color[v] == 1:
                            side1.append(v)
                        else:
                            side0.append(v)
            comp.append((side0, side1))

    total = n
    target = k
    if target < 0 or target > total:
        print("NO")
        return

    dp = {0: None}
    for idx, (s0, s1) in enumerate(comp):
        l0 = len(s0)
        l1 = len(s1)
        new_dp = {}
        for s, prev in dp.items():
            ns = s + l0
            if ns <= target and ns not in new_dp:
                new_dp[ns] = (prev, idx, 0)
            ns = s + l1
            if ns <= target and ns not in new_dp:
                new_dp[ns] = (prev, idx, 1)
        dp = new_dp
        if not dp:
            print("NO")
            return

    if target not in dp:
        print("NO")
        return

    chosen = []
    cur = target
    while cur is not None:
        prev, idx, side = dp[cur]
        s0, s1 = comp[idx]
        if side == 0:
            chosen.extend(s0)
        else:
            chosen.extend(s1)
        cur = prev

    print("YES")
    print(' '.join(map(str, chosen)))

solve()
```
