```python
import sys

# The problem states the mechanism is "fully operational", which implies 
# the graph of wheels is bipartite. If it were not bipartite, there would 
# be an odd cycle, which would cause the mechanism to bind.
# We need to find an independent set of size k.
# In a bipartite graph, we can partition vertices into two sets, L and R.
# Any subset of L is an independent set, and any subset of R is an independent set.
# Let |L| = n1 and |R| = n2. We can pick any number of vertices from L 
# up to n1, and any number from R up to n2.
# Since we need exactly k, we can pick x from L and y from R such that x+y=k,
# where 0 <= x <= n1 and 0 <= y <= n2.
# This is possible if and only if k <= n1 + n2 = n, which is always true.
# Wait, the constraint is that we cannot pick two connected vertices.
# Since it's bipartite, we can pick any subset of L and any subset of R.
# The maximum independent set size is n. We can always pick k vertices 
# as long as we don't pick both ends of an edge.
# Actually, in a bipartite graph, the maximum independent set size is 
# n - (size of maximum matching). But we don't need the maximum independent set,
# we just need *any* independent set of size k.
# Since the graph is bipartite, we can color it with 2 colors (0 and 1).
# Let S0 be the set of vertices with color 0, S1 be the set of vertices with color 1.
# Any subset of S0 is independent, and any subset of S1 is independent.
# We can pick x vertices from S0 and y vertices from S1 such that x+y=k.
# This is possible if we can find x, y such that 0 <= x <= |S0| and 0 <= y <= |S1|.
# Since x+y=k, we need to check if there exist x, y satisfying these.
# The range of possible values for x+y is [0, |S0| + |S1|] = [0, n].
# Since k <= n, we can always find such x and y.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    s0 = []
    s1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [i]
            color[i] = 0
            s0.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        if color[v] == 0:
                            s0.append(v)
                        else:
                            s1.append(v)
                        stack.append(v)
    
    # We need to pick k vertices. We have |s0| and |s1|.
    # We need x from s0, y from s1 such that x+y=k, 0<=x<=|s0|, 0<=y<=|s1|.
    # Let x = min(k, |s0|). Then y = k - x.
    # Is y <= |s1|? 
    # k - x <= |s1|  => k - |s1| <= x.
    # Since x = min(k, |s0|), if k <= |s0|, x=k, y=0 <= |s1|.
    # If k > |s0|, x=|s0|, y=k-|s0|. Since k <= |s0|+|s1|, k-|s0| <= |s1|.
    # So this always works.
    
    x = min(k, len(s0))
    y = k - x
    
    res = s0[:x] + s1[:y]
    
    print("YES")
    print(*(res))

solve()
```
