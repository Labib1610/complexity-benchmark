import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # Bipartition the graph
    color = [0] * (n + 1)
    sets = [[], []]
    valid = True
    
    for i in range(1, n + 1):
        if color[i] == 0:
            queue = deque([i])
            color[i] = 1
            while queue:
                u = queue.popleft()
                sets[color[u] - 1].append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
    
    # Choose the larger set if k allows, otherwise the smaller
    if len(sets[0]) >= k:
        result = sets[0][:k]
    elif len(sets[1]) >= k:
        result = sets[1][:k]
    else:
        print("NO")
        return
    
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    solve()
