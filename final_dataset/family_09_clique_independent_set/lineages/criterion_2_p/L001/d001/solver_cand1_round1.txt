import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    camp0 = []
    camp1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0:
                    camp0.append(u)
                else:
                    camp1.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # In a bipartite graph, an independent set is a set of vertices with no edges.
    # Since all edges are between camp0 and camp1, any subset of camp0 is an independent set,
    # and any subset of camp1 is an independent set.
    # We want to pick x from camp0 and y from camp1 such that x + y = k,
    # and no edges exist between the chosen x and y.
    # This is equivalent to: for every chosen u in camp0, none of its neighbors are in the chosen camp1.
    # Let S0 be the chosen subset of camp0, S1 be the chosen subset of camp1.
    # Condition: No edges between S0 and S1.
    # This means S1 must be a subset of (All_Camp1 \ Neighbors(S0)).
    # To maximize the size, for a fixed S0, we should pick all available nodes in camp1.
    # Let N(S0) be the union of neighbors of all nodes in S0.
    # We need |S0| + |camp1 \ N(S0)| >= k.
    # Actually, we can just pick S0 = camp0, then we can pick any k - |camp0| nodes from camp1 \ N(camp0).
    # Or pick S1 = camp1, then pick any k - |camp1| nodes from camp0 \ N(camp1).
    
    n0 = len(camp0)
    n1 = len(camp1)
    
    # Neighbors of all nodes in camp0
    neigh_of_camp0 = set()
    for u in camp0:
        for v in adj[u]:
            neigh_of_camp0.add(v)
    
    # Available in camp1 if we take all of camp0
    avail_in_camp1 = [v for v in camp1 if v not in neigh_of_camp0]
    if n0 + len(avail_in_camp1) >= k:
        res = camp0[:min(n0, k)]
        rem = k - len(res)
        res.extend(avail_in_camp1[:rem])
        print("YES")
        print(*(res))
        return

    # Neighbors of all nodes in camp1
    neigh_of_camp1 = set()
    for u in camp1:
        for v in adj[u]:
            neigh_of_camp1.add(v)
            
    # Available in camp0 if we take all of camp1
    avail_in_camp0 = [u for u in camp0 if u not in neigh_of_camp1]
    if n1 + len(avail_in_camp0) >= k:
        res = camp1[:min(n1, k)]
        rem = k - len(res)
        res.extend(avail_in_camp0[:rem])
        print("YES")
        print(*(res))
        return

    print("NO")

solve()
