```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    components = []
    
    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            queue = deque([start])
            comp = [start]
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                        comp.append(v)
            components.append(comp)
    
    total = 0
    for comp in components:
        total += max(
            sum(1 for u in comp if color[u] == 1),
            sum(1 for u in comp if color[u] == 2)
        )
    
    if k > total:
        print("NO")
        return
    
    print("YES")
    
    remaining = k
    result = []
    for comp in components:
        if remaining <= 0:
            break
        cnt1 = sum(1 for u in comp if color[u] == 1)
        cnt2 = len(comp) - cnt1
        if cnt1 >= cnt2:
            take = min(cnt1, remaining)
            for u in comp:
                if color[u] == 1:
                    result.append(u)
                    take -= 1
                    if take == 0:
                        break
            remaining -= min(cnt1, remaining)
        else:
            take = min(cnt2, remaining)
            for u in comp:
                if color[u] == 2:
                    result.append(u)
                    take -= 1
                    if take == 0:
                        break
            remaining -= min(cnt2, remaining)
    
    print(' '.join(map(str, result[:k])))

solve()
```
