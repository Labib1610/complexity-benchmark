```python
import sys

# The problem states that destructive interference occurs only if the sum of phase parameters is 0.
# This means interference only occurs between antennas with different phase parameters (+1 and -1).
# This implies the graph of interference is bipartite.
# We need to find a subset of size k that is an independent set.
# In a bipartite graph, any independent set is a subset of one of the two partitions.
# Let the two partitions be V1 and V2. Any subset of V1 is an independent set,
# and any subset of V2 is an independent set.
# We need to check if we can pick k antennas from V1 or V2.
# Wait, the problem says "a way to power on exactly k antennas such that the selected group is entirely free of destructive interference".
# Since the graph is bipartite, the maximum independent set is the union of the larger partition
# and any subset of the other partition that doesn't have edges to the chosen ones.
# Actually, in a bipartite graph, any subset of V1 is independent, and any subset of V2 is independent.
# If we pick a subset of V1, it is independent. If we pick a subset of V2, it is independent.
# Can we pick some from V1 and some from V2? Only if there are no edges between them.
# But the problem says the graph is bipartite. If we pick a set S, it is independent iff
# no two vertices in S have an edge.
# This is equivalent to saying S is a subset of V1 or S is a subset of V2? No.
# Actually, any independent set in a bipartite graph is a subset of V1 union V2.
# But we can pick vertices from both sides as long as no edge connects them.
# However, the problem guarantees a valid assignment of +1/-1 exists.
# Let V+ be the set of antennas with +1, V- be the set with -1.
# Edges only exist between V+ and V-.
# Any subset of V+ is independent. Any subset of V- is independent.
# Can we pick some from V+ and some from V-? Only if there are no edges between them.
# But the graph is defined by the interference pairs.
# If we pick a subset S, it is safe if no edge (u, v) has both u, v in S.
# This is exactly the definition of an independent set.
# In a bipartite graph, the maximum independent set size is n - (size of minimum vertex cover).
# By Konig's theorem, min vertex cover = max matching.
# But we just need to find *any* independent set of size k.
# Since V+ and V- are independent sets, if |V+| >= k, we can pick k from V+.
# If |V-| >= k, we can pick k from V-.
# Is it possible to pick k that is not a subset of V+ or V-?
# Yes, if there are no edges between the chosen subset of V+ and the chosen subset of V-.
# But the problem is simpler: we just need to find if there exists an independent set of size k.
# Since the graph is bipartite, we can just use the two partitions V+ and V-.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = {}
    for i in range(1, n + 1):
        if i not in color:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if v not in color:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    v0 = [i for i in color if color[i] == 0]
    v1 = [i for i in color if color[i] == 1]
    
    if len(v0) >= k:
        print("YES")
        print(*(v0[:k]))
    elif len(v1) >= k:
        print("YES")
        print(*(v1[:k]))
    else:
        print("NO")

solve()
```
