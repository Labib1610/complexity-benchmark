import sys
from collections import defaultdict, deque

def solve():
    n, m, k = map(int, input().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    degree = [0] * (n + 1)
    
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
    
    # Since the graph has no odd cycles, it's bipartite
    # We can color it with BFS
    color = [-1] * (n + 1)  # -1: unvisited, 0: color A, 1: color B
    visited = [False] * (n + 1)
    
    set_a = []
    set_b = []
    
    # Process each connected component
    for start in range(1, n + 1):
        if visited[start]:
            continue
            
        queue = deque([start])
        color[start] = 0
        visited[start] = True
        
        local_a = [start]
        local_b = []
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            
            for neighbor in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = 1 - current_color
                    if color[neighbor] == 0:
                        local_a.append(neighbor)
                    else:
                        local_b.append(neighbor)
                    queue.append(neighbor)
                elif color[neighbor] == current_color:
                    # This should not happen as graph is bipartite
                    print("NO")
                    return
        
        set_a.extend(local_a)
        set_b.extend(local_b)
    
    # Check if we can select exactly k antennas
    size_a = len(set_a)
    size_b = len(set_b)
    
    if k <= size_a:
        print("YES")
        print(' '.join(map(str, set_a[:k])))
    elif k <= size_b:
        print("YES")
        print(' '.join(map(str, set_b[:k])))
    elif k > size_a + size_b:
        print("NO")
    else:
        # We need to take some from A and some from B
        # But we can't take antennas from different partitions that are connected
        # Since A and B are independent sets, we can only take from one partition
        print("NO")

solve()
