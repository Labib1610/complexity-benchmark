```python
import sys

# The problem describes a bipartite graph because every volume is shared 
# between a producer and a consumer, and no two producers or two consumers 
# share a volume. We need to select k vertices such that no two are connected.
# This is equivalent to finding an independent set of size k.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# However, we need to select exactly k vertices.
# Since it's bipartite, we can partition the graph into two sets, L and R.
# Any subset of L is an independent set, and any subset of R is an independent set.
# We can pick x vertices from L and y vertices from R such that x + y = k.
# But we must ensure no edges exist between the chosen vertices.
# Actually, the problem is simpler: we can pick any subset of L or any subset of R.
# Wait, the condition is "no two selected microservices share a volume".
# This means the selected set must be an independent set.
# In a bipartite graph, we can always pick all vertices from one side of the partition.
# Let the two partitions be V1 and V2. Any subset of V1 is an independent set,
# and any subset of V2 is an independent set.
# We can pick any number of vertices from V1 and any number from V2, 
# provided no edges exist between them.
# Actually, the simplest way is to pick as many as possible from V1 and V2.
# Let |V1| = n1, |V2| = n2. We can pick any subset of V1 (size up to n1) 
# and any subset of V2 (size up to n2).
# But we cannot pick a vertex from V1 and a vertex from V2 if they are connected.
# This is the Maximum Independent Set problem, which is generally hard,
# but for bipartite graphs, it's solvable.
# However, the problem asks for *any* k vertices.
# If we pick a subset S1 from V1 and S2 from V2, they are independent if 
# there are no edges between S1 and S2.
# This is satisfied if we pick all of V1 or all of V2.
# Actually, we can just pick any subset of V1 and any subset of V2 such that 
# no edges exist between them.
# The easiest way: pick all of V1, then add vertices from V2 that have no edges to V1.
# But that's not necessary. We can just pick any subset of V1 or V2.
# If k <= max(|V1|, |V2|), we can just pick k vertices from the larger partition.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = map(int, input_data[idx:idx+2])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    v1, v2 = [], []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: v1.append(u)
                else: v2.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    if k <= len(v1):
        print("YES")
        print(*(v1[:k]))
    elif k <= len(v2):
        print("YES")
        print(*(v2[:k]))
    else:
        print("NO")

solve()
```
