import sys

# The problem states that for every interference link (u, v), the sum of their 
# polarities is 0. This means polarity(u) + polarity(v) = 0, so polarity(u) = -polarity(v).
# This implies the graph is bipartite. The two sets of the bipartition are 
# nodes with polarity +1 and nodes with polarity -1.
# Any interference-free set can contain nodes from both sets, but no two nodes 
# in the set can be connected by an edge.
# Actually, the condition "sum of polarities is 0" means the graph is bipartite.
# Let the two partitions be A and B. Any subset of A is interference-free, 
# and any subset of B is interference-free.
# The problem asks for a set of size k. Since the graph is bipartite, 
# we can pick any subset of A or any subset of B.
# Wait, the problem asks for *any* interference-free set of size k.
# In a bipartite graph, the maximum independent set is not necessarily just A or B.
# However, the problem guarantees the polarity property, which means the graph is bipartite.
# Let the bipartition be (V1, V2). Any independent set is a subset of V1 union V2.
# Actually, any subset of V1 is independent, and any subset of V2 is independent.
# Can we pick nodes from both? Yes, if there are no edges between them.
# But in a bipartite graph, the largest independent set is the size of the maximum 
# independent set. By Konig's theorem, |Max Independent Set| = |V| - |Min Vertex Cover|.
# Actually, for any bipartite graph, we can just find the bipartition (V1, V2).
# Any independent set is a subset of V1 union V2.
# The problem is simpler: we just need to find if there exists an independent set of size k.
# In a bipartite graph, the size of the maximum independent set is |V| - |Maximum Matching|.
# We can greedily pick nodes. Actually, just taking the larger of the two partitions 
# V1 or V2 gives an independent set of size max(|V1|, |V2|).
# Is it possible to have an independent set larger than max(|V1|, |V2|)?
# Yes, e.g., a path 1-2-3-4. V1={1,3}, V2={2,4}. |V1|=2, |V2|=2. 
# Independent set {1, 4} has size 2. {1, 3} has size 2. {2, 4} has size 2.
# Actually, for any bipartite graph, the maximum independent set size is |V| - |Max Matching|.
# We can find the maximum matching using Hopcroft-Karp or simple DFS/BFS.
# Given the constraints, a simple DFS-based matching is sufficient.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    v1, v2 = [], []
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                if c == 1: v1.append(u)
                else: v2.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
    
    # Bipartite matching to find max independent set
    match = [0] * (n + 1)
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] == 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    matching_size = 0
    for u in v1:
        visited = [False] * (n + 1)
        if dfs(u, visited):
            matching_size += 1
            
    # Max independent set size is n - matching_size
    if k > n - matching_size:
        print("NO")
        return
    
    # To find the independent set:
    # Min vertex cover U = {nodes in V1 not visited in DFS} U {nodes in V2 visited in DFS}
    # Max independent set = V \ U
    visited_v1 = [False] * (n + 1)
    def dfs_cover(u):
        visited_v1[u] = True
        for v in adj[u]:
            if v != match[u] and not visited_v1[v]:
                visited_v1[v] = True
                if match[v] != 0 and not visited_v1[match[v]]:
                    dfs_cover(match[v])
    
    for u in v1:
        if match[u] == 0:
            dfs_cover(u)
            
    indep_set = []
    for u in v1:
        if not visited_v1[u]: indep_set.append(u)
    for u in v2:
        if visited_v1[u]: indep_set.append(u)
        
    print("YES")
    print(*(indep_set[:k]))

solve()
