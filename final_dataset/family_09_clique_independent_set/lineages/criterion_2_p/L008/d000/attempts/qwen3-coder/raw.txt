```python
import sys
from collections import defaultdict, deque

def read_input():
    n, m, k = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
    return n, m, k, edges

def bfs_coloring(n, edges):
    # Build adjacency list
    adj = defaultdict(list)
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Color nodes with BFS
    color = [-1] * (n + 1)  # 0 or 1, -1 means unvisited
    for start in range(1, n + 1):
        if color[start] == -1:
            queue = deque([start])
            color[start] = 0
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        # This should not happen according to problem statement
                        raise ValueError("Graph is not bipartite")
    return color

def solve():
    n, m, k, edges = read_input()
    
    # If no edges, any k nodes will do
    if m == 0:
        print("YES")
        print(' '.join(map(str, range(1, k + 1))))
        return
    
    # Color the graph
    color = bfs_coloring(n, edges)
    
    # Separate nodes by color
    set0 = [i for i in range(1, n + 1) if color[i] == 0]
    set1 = [i for i in range(1, n + 1) if color[i] == 1]
    
    # Check which set we can take from
    if k <= len(set0):
        print("YES")
        print(' '.join(map(str, set0[:k])))
    elif k <= len(set1):
        print("YES")
        print(' '.join(map(str, set1[:k])))
    else:
        print("NO")

solve()
```
