```python
import sys

# The problem states the graph is bipartite (guaranteed by the opposite-state property).
# A mutually disconnected collection is an independent set.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# However, we don't need the maximum independent set; we just need any independent set of size k.
# In a bipartite graph, the two sets of the bipartition (let's call them A and B)
# are both independent sets. Any subset of A is an independent set, and any subset of B is an independent set.
# Let |A| = size_A and |B| = size_B.
# If k <= size_A, we can pick k elements from A.
# If k <= size_B, we can pick k elements from B.
# If k > size_A and k > size_B, is it possible to have an independent set of size k?
# Actually, the maximum independent set in a bipartite graph is indeed the larger of the two
# partitions if the graph is connected. If the graph has multiple connected components,
# we can pick elements from the partitions of each component.
# Let each component i have partitions A_i and B_i.
# We want to pick a total of k elements such that no two are connected.
# This is equivalent to picking a subset of vertices S such that for each component,
# we pick either a subset of A_i or a subset of B_i.
# Actually, we can just pick any subset of A_i or B_i from each component.
# Let S_A = sum(|A_i|) and S_B = sum(|B_i|).
# The maximum independent set size is max(S_A, S_B).
# If k <= max(S_A, S_B), we can definitely find k elements.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    set_a = []
    set_b = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            set_a.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            set_a.append(v)
                        else:
                            set_b.append(v)
                        stack.append((v, color[v]))
    
    if k <= len(set_a):
        print("YES")
        print(*(set_a[:k]))
    elif k <= len(set_b):
        print("YES")
        print(*(set_b[:k]))
    else:
        print("NO")

solve()
```
