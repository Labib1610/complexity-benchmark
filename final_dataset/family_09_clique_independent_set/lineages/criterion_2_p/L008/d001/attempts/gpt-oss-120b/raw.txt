```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    # bipartition via BFS coloring
    color = [-1] * (n + 1)          # 0 = left, 1 = right
    for start in range(1, n + 1):
        if color[start] == -1:
            color[start] = 0
            q = deque([start])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        q.append(v)

    left_nodes = [i for i in range(1, n + 1) if color[i] == 0]
    # Hopcroft–Karp
    INF = 10 ** 9
    pairU = [0] * (n + 1)   # only meaningful for left vertices
    pairV = [0] * (n + 1)   # only for right vertices
    dist = [0] * (n + 1)

    def bfs() -> bool:
        q = deque()
        for u in left_nodes:
            if pairU[u] == 0:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while q:
            u = q.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if color[v] != 1:
                        continue
                    if pairV[v] == 0:
                        if dist[0] == INF:
                            dist[0] = dist[u] + 1
                    elif dist[pairV[v]] == INF:
                        dist[pairV[v]] = dist[u] + 1
                        q.append(pairV[v])
        return dist[0] != INF

    def dfs(u: int) -> bool:
        if u == 0:
            return True
        for v in adj[u]:
            if color[v] != 1:
                continue
            pu = pairV[v]
            if pu == 0 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pairU[u] = v
                pairV[v] = u
                return True
        dist[u] = INF
        return False

    while bfs():
        for u in left_nodes:
            if pairU[u] == 0:
                dfs(u)

    # Build minimum vertex cover from alternating BFS
    visited_left = [False] * (n + 1)
    visited_right = [False] * (n + 1)
    q = deque()
    for u in left_nodes:
        if pairU[u] == 0:
            q.append(u)
            visited_left[u] = True
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] != 1:
                continue
            if not visited_right[v] and pairU[u] != v:  # edge not in matching
                visited_right[v] = True
                if pairV[v] and not visited_left[pairV[v]]:
                    visited_left[pairV[v]] = True
                    q.append(pairV[v])

    # Minimum vertex cover = (Left \ visited_left) ∪ (Right ∩ visited_right)
    # Independent set = (Left ∩ visited_left) ∪ (Right \ visited_right)
    indep = []
    for u in range(1, n + 1):
        if color[u] == 0:
            if visited_left[u]:
                indep.append(u)
        else:
            if not visited_right[u]:
                indep.append(u)

    if len(indep) < k:
        print("NO")
        return
    print("YES")
    print(' '.join(map(str, indep[:k])))

if __name__ == "__main__":
    solve()
```
