import sys
from collections import defaultdict, deque

def solve():
    n, m, k = map(int, input().split())
    
    # Build adjacency list
    adj = defaultdict(set)
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
    
    # If no edges, we can pick any k nodes
    if m == 0:
        print("YES")
        print(' '.join(map(str, range(1, k + 1))))
        return
    
    # Find connected components and bipartition each component
    visited = [False] * (n + 1)
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if not visited[i]:
            # BFS to find component and check bipartiteness
            queue = deque([i])
            visited[i] = True
            color[i] = 0
            part0 = [i]
            part1 = []
            
            is_bipartite = True
            while queue and is_bipartite:
                u = queue.popleft()
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        color[v] = 1 - color[u]
                        if color[v] == 0:
                            part0.append(v)
                        else:
                            part1.append(v)
                        queue.append(v)
                    elif color[v] == color[u]:
                        is_bipartite = False
                        break
            
            if not is_bipartite:
                # This shouldn't happen according to problem statement
                print("NO")
                return
            
            components.append((part0, part1))
    
    # For each component, we can choose the larger part
    total_in_larger = 0
    choices = []
    
    for part0, part1 in components:
        if len(part0) >= len(part1):
            total_in_larger += len(part0)
            choices.append((part0, len(part0)))
        else:
            total_in_larger += len(part1)
            choices.append((part1, len(part1)))
    
    # If we can get exactly k, great
    if total_in_larger >= k:
        # Greedily select from largest parts until we have k nodes
        choices.sort(key=lambda x: x[1], reverse=True)
        selected = []
        count = 0
        
        for part, size in choices:
            if count + size <= k:
                selected.extend(part)
                count += size
            else:
                # We need to be more careful - take only what we need
                needed = k - count
                selected.extend(part[:needed])
                count = k
                break
        
        if count == k:
            print("YES")
            print(' '.join(map(str, selected)))
            return
    
    print("NO")

solve()
