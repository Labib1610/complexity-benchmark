def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]

        idx += 2

    pq = [(degree[i], i) for i in range(1, n + 1)]

                # We don't need to update degrees in pq, just check removed

    else:
