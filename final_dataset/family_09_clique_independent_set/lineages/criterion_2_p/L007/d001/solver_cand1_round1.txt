```python
import sys

# The problem states that the graph is bipartite because all closed loops have an even number of links.
# An interference-free cluster is an independent set.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# However, we need to find an independent set of size exactly k.
# Since the graph is bipartite, we can partition the vertices into two sets, L and R,
# such that all edges are between L and R.
# Any subset of L is an independent set, and any subset of R is an independent set.
# Let |L| = n1 and |R| = n2. We can pick any size s1 from L and s2 from R such that s1 + s2 = k.
# But this is only true if we don't pick any pair (u, v) where u in L and v in R are connected.
# Actually, the problem is simpler: we can pick any subset of L and any subset of R
# as long as no edges exist between them.
# Since the graph is bipartite, we can find the two partitions L and R.
# Let the connected components be (L_i, R_i).
# We can pick any number of vertices from each component's L_i or R_i.
# This is equivalent to the subset sum problem, but we have a specific structure.
# For each component i, we can pick either all vertices in L_i (size |L_i|) or all in R_i (size |R_i|).
# Wait, that's not quite right. We can pick any subset of L_i and any subset of R_i
# as long as we don't pick any connected pair.
# Actually, for each component, we can pick any subset of L_i OR any subset of R_i.
# To maximize the independent set, we pick the larger of |L_i| or |R_i|.
# To get exactly k, we use dynamic programming.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_l = []
            comp_r = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: comp_l.append(u)
                else: comp_r.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append((comp_l, comp_r))

    # dp[i][j] = which set (0 for L, 1 for R) was used to reach sum j
    # We need to pick a subset of L_i or R_i for each component.
    # Actually, we can pick any number of vertices from L_i or R_i.
    # But if we pick any from L_i, we can't pick any from R_i that are connected.
    # Since it's a bipartite component, if we pick a subset of L_i, we can't pick any neighbors in R_i.
    # The largest independent set in a component is max(|L_i|, |R_i|).
    # We can pick any size s from 0 to |L_i| using L_i, or 0 to |R_i| using R_i.
    
    dp = {0: []}
    for l_nodes, r_nodes in components:
        new_dp = {}
        len_l, len_r = len(l_nodes), len(r_nodes)
        for s, path in dp.items():
            # Option 1: pick some from L_i (0 to len_l)
            for i in range(len_l + 1):
                if s + i <= k:
                    new_dp[s + i] = path + [('L', l_nodes[:i])]
            # Option 2: pick some from R_i (0 to len_r)
            for i in range(len_r + 1):
                if s + i <= k:
                    new_dp[s + i] = path + [('R', r_nodes[:i])]
        dp = new_dp
        
    if k in dp:
        print("YES")
        res = []
        for type, nodes in dp[k]:
            res.extend(nodes)
        print(*(res))
    else:
        print("NO")

solve()
```
