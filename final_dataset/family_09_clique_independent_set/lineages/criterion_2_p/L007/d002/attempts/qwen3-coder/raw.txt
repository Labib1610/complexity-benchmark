```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    index = 3
    edges = []
    adj = defaultdict(list)
    
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # Since every cycle has even length, the graph is bipartite
    # We can color it with two colors
    color = [-1] * n
    visited = [False] * n
    
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 0
        visited[start] = True
        c0 = 1
        c1 = 0
        
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    color[v] = 1 - color[u]
                    if color[v] == 0:
                        c0 += 1
                    else:
                        c1 += 1
                    queue.append(v)
                elif color[v] == color[u]:
                    # This should not happen due to the problem constraint
                    return -1, -1
        return c0, c1
    
    total_c0 = 0
    total_c1 = 0
    
    for i in range(n):
        if not visited[i]:
            c0, c1 = bfs(i)
            if c0 == -1:
                # This case should not occur due to problem constraints
                print("NO")
                return
            total_c0 += c0
            total_c1 += c1
    
    # Now we have two independent sets: color 0 and color 1
    # We want to pick exactly k nodes with no edges between them
    # So we pick from the larger set if possible
    
    if total_c0 >= k:
        selected = []
        count = 0
        for i in range(n):
            if color[i] == 0:
                selected.append(i+1)
                count += 1
                if count == k:
                    break
        print("YES")
        print(' '.join(map(str, selected)))
    elif total_c1 >= k:
        selected = []
        count = 0
        for i in range(n):
            if color[i] == 1:
                selected.append(i+1)
                count += 1
                if count == k:
                    break
        print("YES")
        print(' '.join(map(str, selected)))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
