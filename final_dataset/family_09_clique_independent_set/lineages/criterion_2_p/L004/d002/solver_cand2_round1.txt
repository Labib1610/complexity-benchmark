```python
import sys

# The problem states that the conflict graph contains no odd-length cycles.
# This means the graph is bipartite.
# A bipartite graph can be partitioned into two sets, say A and B, such that
# all edges only connect a node in A to a node in B.
# We can find this partition using BFS or DFS.
# Once we have the two sets A and B, we can pick any number of nodes from A
# and any number of nodes from B to form a safe batch, as long as the total
# count is k.
# Let |A| = sizeA and |B| = sizeB. We need to find if there exist x, y such that
# x + y = k, where 0 <= x <= sizeA and 0 <= y <= sizeB.
# This is equivalent to checking if max(0, k - sizeB) <= x <= min(k, sizeA).
# If such an x exists, we can pick x nodes from A and (k-x) nodes from B.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    set_a = []
    set_b = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            set_a.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            set_a.append(v)
                        else:
                            set_b.append(v)
                        stack.append((v, color[v]))
    
    size_a = len(set_a)
    size_b = len(set_b)
    
    # We need x + y = k, 0 <= x <= size_a, 0 <= y <= size_b
    # y = k - x => 0 <= k - x <= size_b => k - size_b <= x <= k
    # So max(0, k - size_b) <= x <= min(k, size_a)
    
    low = max(0, k - size_b)
    high = min(k, size_a)
    
    if low <= high:
        x = low
        y = k - x
        result = set_a[:x] + set_b[:y]
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
