import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    if k == 0:
        print("YES")
        print()
        return
    
    if k > n:
        print("NO")
        return
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    components = []
    
    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            q = deque([start])
            side0 = []
            side1 = [start]
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        if color[v] == 1:
                            side1.append(v)
                        else:
                            side0.append(v)
                        q.append(v)
            components.append((side0, side1))
    
    total = sum(max(len(a), len(b)) for a, b in components)
    if total < k:
        print("NO")
        return
    
    dp = {0: None}
    for idx, (a, b) in enumerate(components):
        new_dp = {}
        for s, prev in dp.items():
            if prev is not None:
                continue
            for choice, size in [(0, len(a)), (1, len(b))]:
                ns = s + size
                if ns <= k and ns not in new_dp:
                    new_dp[ns] = (idx, choice, s)
        dp.update(new_dp)
        if k in dp and dp[k] is not None:
            break
    
    if k not in dp or dp[k] is None:
        print("NO")
        return
    
    choices = {}
    cur = k
    while dp[cur] is not None:
        idx, choice, prev = dp[cur]
        choices[idx] = choice
        cur = prev
    
    result = []
    for idx, (a, b) in enumerate(components):
        if idx in choices:
            if choices[idx] == 0:
                result.extend(a)
            else:
                result.extend(b)
        else:
            big = a if len(a) >= len(b) else b
            result.extend(big)
    
    result = result[:k]
    print("YES")
    print(' '.join(map(str, result)))

solve()
