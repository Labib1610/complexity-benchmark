```python
import sys

# The problem states that the conflict graph is bipartite (no odd-length cycles).
# A safe batch is an independent set. In a bipartite graph, we can find the 
# maximum independent set by partitioning the graph into two sets (bipartition).
# For each connected component, we have two sets of vertices, say A_i and B_i.
# We can pick either A_i or B_i to be in our independent set.
# Let |A_i| = a_i and |B_i| = b_i. We want to choose a subset of components 
# such that the sum of sizes of chosen sets equals k.
# This is a variation of the subset sum problem (knapsack).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_a = []
            comp_b = []
            stack = [(i, 0)]
            color[i] = 0
            comp_a.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0: comp_a.append(v)
                        else: comp_b.append(v)
                        stack.append((v, color[v]))
            components.append((comp_a, comp_b))

    # dp[i][j] stores which choice was made for component i to get sum j
    # dp[i][j] = (choice_index, prev_sum)
    # choice_index: 0 for comp_a, 1 for comp_b
    dp = {0: []}
    
    for i, (a, b) in enumerate(components):
        new_dp = {}
        len_a, len_b = len(a), len(b)
        for current_sum, path in dp.items():
            if current_sum + len_a <= k:
                new_dp[current_sum + len_a] = path + [(a, 0)]
            if current_sum + len_b <= k:
                new_dp[current_sum + len_b] = path + [(b, 1)]
        dp = new_dp
        
    if k in dp:
        print("YES")
        result = []
        for nodes, _ in dp[k]:
            result.extend(nodes)
        print(*(result))
    else:
        print("NO")

solve()
```
