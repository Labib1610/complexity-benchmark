```python
import sys

# The problem states the interference graph has no odd-length cycles.
# This means the graph is bipartite.
# In a bipartite graph, we can partition the vertices into two sets,
# say A and B, such that all edges connect a vertex in A to a vertex in B.
# A secure cohort is an independent set.
# In a bipartite graph, the two partitions A and B are independent sets.
# If we have multiple connected components, each component has its own
# bipartite partition (A_i, B_i).
# We want to pick a subset of these partitions such that the total number
# of vertices is exactly k.
# This is a variation of the subset sum problem (or knapsack).
# Since we have many components, we can use dynamic programming.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_nodes = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp_nodes[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append((len(comp_nodes[0]), len(comp_nodes[1]), comp_nodes[0], comp_nodes[1]))

    # dp[i][j] = which choice was made for component i to get sum j
    # dp[i][j] = (choice_index, previous_sum)
    # choice_index: 0 for picking comp_nodes[0], 1 for picking comp_nodes[1]
    dp = {0: []}
    
    for i, (s0, s1, nodes0, nodes1) in enumerate(components):
        new_dp = {}
        for current_sum, path in dp.items():
            if current_sum + s0 <= k:
                new_dp[current_sum + s0] = path + [(0, nodes0)]
            if current_sum + s1 <= k:
                new_dp[current_sum + s1] = path + [(1, nodes1)]
        dp = new_dp
        
    if k in dp:
        print("YES")
        result = []
        for _, nodes in dp[k]:
            result.extend(nodes)
        print(*(result))
    else:
        print("NO")

solve()
```
