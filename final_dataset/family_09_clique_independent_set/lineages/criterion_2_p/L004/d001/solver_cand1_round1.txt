```python
import sys

# The problem states the conflict graph has no odd-length cycles.
# This means the graph is bipartite.
# A bipartite graph can be partitioned into two sets (colors 0 and 1)
# such that no two nodes in the same set have an edge between them.
# Let the sizes of the two sets be S0 and S1.
# We can pick any number of nodes from set 0 (up to S0) and any number
# from set 1 (up to S1).
# Thus, we can form a safe batch of size k if and only if there exist
# a, b such that a + b = k, 0 <= a <= S0, and 0 <= b <= S1.
# This is equivalent to checking if max(0, k - S1) <= a <= min(k, S0).
# Such an 'a' exists if and only if max(0, k - S1) <= min(k, S0).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    sets = [[], []]
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                sets[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    s0 = len(sets[0])
    s1 = len(sets[1])
    
    # We need a + b = k, 0 <= a <= s0, 0 <= b <= s1
    # b = k - a => 0 <= k - a <= s1 => k - s1 <= a <= k
    # So max(0, k - s1) <= a <= min(k, s0)
    
    low = max(0, k - s1)
    high = min(k, s0)
    
    if low <= high:
        a = low
        b = k - a
        res = sets[0][:a] + sets[1][:b]
        print("YES")
        print(*(res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
