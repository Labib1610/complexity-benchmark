import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * (n + 1)
    tiers = [[], []]
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [i]
            color[i] = 0
            while stack:
                u = stack.pop()
                tiers[color[u]].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        stack.append(v)
    
    if len(tiers[0]) >= k:
        print("YES")
        print(' '.join(map(str, tiers[0][:k])))
    elif len(tiers[1]) >= k:
        print("YES")
        print(' '.join(map(str, tiers[1][:k])))
    else:
        print("NO")

solve()
