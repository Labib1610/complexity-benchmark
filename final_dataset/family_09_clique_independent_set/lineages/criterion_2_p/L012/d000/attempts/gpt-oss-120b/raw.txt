```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)
    side = [[], []]          # side[0] = vertices colored 0, side[1] = colored 1

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # start new component, assign color 0
        q = deque([start])
        color[start] = 0
        while q:
            u = q.popleft()
            side[color[u]].append(u)
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                # else: graph is guaranteed bipartite, no need to check conflict

    # choose the larger side
    if len(side[0]) >= k:
        chosen = side[0][:k]
    elif len(side[1]) >= k:
        chosen = side[1][:k]
    else:
        print("NO")
        return

    print("YES")
    print(' '.join(map(str, chosen)))

if __name__ == "__main__":
    solve()
```
