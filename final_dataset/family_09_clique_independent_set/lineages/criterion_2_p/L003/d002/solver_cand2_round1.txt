```python
import sys

def solve():
    # The problem states that a conflict exists between u and v if and only if
    # popcount(config[u] ^ config[v]) is odd.
    # This is equivalent to saying that the parity of the popcount of the 
    # configuration integers defines a bipartite graph.
    # Let p(x) be the parity of the popcount of config[x].
    # If p(u) == p(v), then popcount(config[u] ^ config[v]) is even, so no conflict.
    # If p(u) != p(v), then popcount(config[u] ^ config[v]) is odd, so a conflict is possible.
    # Thus, the conflict graph is bipartite.
    # We can partition the relays into two sets, A and B, such that all conflicts
    # are between a relay in A and a relay in B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    sets = [[], []]
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                sets[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        # This should not happen based on problem constraints
                        print("NO")
                        return
    
    # We have two independent sets: sets[0] and sets[1].
    # We need to pick k nodes such that no two are connected.
    # Since the graph is bipartite, any subset of sets[0] is an independent set,
    # and any subset of sets[1] is an independent set.
    # However, we can pick nodes from both sets as long as we don't pick
    # any pair (u, v) where u in sets[0], v in sets[1] and (u, v) is an edge.
    # Actually, the problem is simpler: we need an independent set of size k.
    # In a bipartite graph, the maximum independent set is n - (size of min vertex cover).
    # But we just need *any* independent set of size k.
    # The largest independent set is the union of the larger partition and 
    # potentially some nodes from the other partition.
    # Wait, the problem is simpler: any subset of an independent set is an independent set.
    # The two sets 'sets[0]' and 'sets[1]' are independent sets.
    # Let S0 = sets[0], S1 = sets[1].
    # Any subset of S0 is independent. Any subset of S1 is independent.
    # We can pick k nodes from S0 if len(S0) >= k, or from S1 if len(S1) >= k.
    
    if len(sets[0]) >= k:
        print("YES")
        print(*(sets[0][:k]))
    elif len(sets[1]) >= k:
        print("YES")
        print(*(sets[1][:k]))
    else:
        print("NO")

solve()
```
