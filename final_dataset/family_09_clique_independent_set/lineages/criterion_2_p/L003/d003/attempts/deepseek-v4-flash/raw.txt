We can model the nodes as vertices in a graph where edges represent interference.  
The bias voltage rule means the graph is bipartite: each connected component can be split into two independent sets (no edges inside them).  

We need to pick exactly `k` nodes with no edges between them — that means we can only pick nodes from one side of each bipartite component.  

So the problem reduces to:  
- For each connected component, find the sizes of its two color classes.  
- We can choose either all nodes from one side or all from the other (or none).  
- We need to see if we can sum some choices to exactly `k`.  

This is a subset-sum problem over component sizes, which we can solve with bitset DP.  

```python
import sys
from collections import deque

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    visited = [-1] * n
    comps = []
    for i in range(n):
        if visited[i] == -1:
            # BFS to color the component
            q = deque([i])
            visited[i] = 0
            cnt = [0, 0]
            cnt[0] += 1
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if visited[v] == -1:
                        visited[v] = visited[u] ^ 1
                        cnt[visited[v]] += 1
                        q.append(v)
            comps.append((cnt[0], cnt[1]))

    # DP: possible sums of sizes
    dp = 1  # bitmask of reachable sums
    for a, b in comps:
        dp = (dp << a) | (dp << b)

    if (dp >> k) & 1:
        print("YES")
        # Reconstruct solution
        chosen = []
        # We'll backtrack from k
        target = k
        for a, b in reversed(comps):
            # Check if target can be reached by taking a or b

                # take side a
                # find nodes of that side
                chosen.extend(find_nodes(a, b, visited, comps, target, a))
                target -= a

                # take side b
                chosen.extend(find_nodes(a, b, visited, comps, target, b))
                target -= b
            dp >>= max(a, b)  # not exactly right, but we'll handle differently

        # Let's do it properly below.
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
