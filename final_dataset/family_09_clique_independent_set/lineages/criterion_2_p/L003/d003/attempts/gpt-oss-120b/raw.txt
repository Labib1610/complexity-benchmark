```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [ -1 ] * (n+1)
    comps = []          # list of (list0, list1)
    for start in range(1, n+1):
        if color[start] != -1:
            continue
        q = deque([start])
        color[start] = 0
        side = [[], []]
        side[0].append(start)
        while q:
            u = q.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    side[color[v]].append(v)
                    q.append(v)
                elif color[v] == cu:
                    # should not happen per problem statement
                    pass
        comps.append((side[0], side[1]))

    total_max = sum(max(len(a), len(b)) for a, b in comps)
    if total_max < k:
        print("NO")
        return

    need = k
    answer = []
    for a, b in comps:
        if need == 0:
            break
        if len(a) >= len(b):
            take_side = a
        else:
            take_side = b
        take = min(need, len(take_side))
        answer.extend(take_side[:take])
        need -= take

    # if still need (should not happen), take from remaining sides
    if need:
        for a, b in comps:
            # try the other side if not used fully
            other = b if len(a) >= len(b) else a
            # already taken some from take_side, skip those already in answer
            remaining = [v for v in other if v not in answer]
            take = min(need, len(remaining))
            answer.extend(remaining[:take])
            need -= take
            if need == 0:
                break

    print("YES")
    print(' '.join(map(str, answer[:k])))

if __name__ == "__main__":
    solve()
```
