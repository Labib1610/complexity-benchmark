```python
import sys

# The problem states that conflicts only occur between orthogonal orientations.
# This implies the conflict graph is a collection of bipartite graphs.
# Specifically, if we have orientations N, S, E, W:
# N and S are collinear (no conflict), E and W are collinear (no conflict).
# Conflicts only occur between {N, S} and {E, W}.
# This means the graph is bipartite.
# We can find the connected components, and for each component, find the two
# partitions (size A_i and B_i). We need to pick a total of k nodes.
# This is a variation of the subset sum problem (or knapsack).
# Since we need to pick exactly k, we can use DP.
# Given n=100,000, we need an efficient way to solve the subset sum.
# Since the sum of sizes is n, we can use the optimization for subset sum
# with many items: group items by size.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp = []
            stack = [(i, 0)]
            color[i] = 0
            c0, c1 = 0, 0
            nodes0, nodes1 = [], []
            while stack:
                u, c = stack.pop()
                if c == 0:
                    c0 += 1
                    nodes0.append(u)
                else:
                    c1 += 1
                    nodes1.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append((c0, c1, nodes0, nodes1))

    # dp[i] stores which component index and which partition (0 or 1) 
    # was used to reach sum i.
    # To reconstruct, we store (prev_sum, partition_choice)
    dp = {0: (None, None)}
    
    for i, (c0, c1, n0, n1) in enumerate(components):
        new_dp = dp.copy()
        for s in dp:
            if s + c0 <= k and (s + c0) not in new_dp:
                new_dp[s + c0] = (s, 0)
            if s + c1 <= k and (s + c1) not in new_dp:
                new_dp[s + c1] = (s, 1)
        dp = new_dp

    if k not in dp:
        print("NO")
        return

    print("YES")
    res = []
    curr = k
    for i in range(len(components) - 1, -1, -1):
        prev_s, choice = dp[curr]
        c0, c1, n0, n1 = components[i]
        if choice == 0:
            res.extend(n0)
            curr -= c0
        else:
            res.extend(n1)
            curr -= c1
    print(*(res))

solve()
```
