import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            comp[0].append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        comp[1 - c].append(v)
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            components.append(comp)
            
    # dp[i][j] = can we pick j nodes from first i components
    # Since we need to pick exactly k, and each component is bipartite,
    # we can pick either comp[0] or comp[1] from each component.
    # Let s0 = len(comp[0]), s1 = len(comp[1]).
    # We need to choose x_i in {s0_i, s1_i} such that sum(x_i) = k.
    
    dp = {0: []}
    for i, comp in enumerate(components):
        s0, s1 = len(comp[0]), len(comp[1])
        new_dp = {}
        for current_sum, path in dp.items():
            if current_sum + s0 <= k:
                new_dp[current_sum + s0] = path + [(i, 0)]
            if current_sum + s1 <= k:
                new_dp[current_sum + s1] = path + [(i, 1)]
        dp = new_dp
        
    if k not in dp:
        print("NO")
    else:
        print("YES")
        result = []
        for comp_idx, side in dp[k]:
            result.extend(components[comp_idx][side])
        print(*(result))

solve()
