import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    k = int(input_data[idx]); idx += 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx]); idx += 1
        v = int(input_data[idx]); idx += 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    comp = []
    comp_sizes = []
    comp_sides = []

    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            queue = deque([start])
            cnt = [0, 0]
            cnt[0] = 1
            members = [start]
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        cnt[color[v] - 1] += 1
                        members.append(v)
                        queue.append(v)
            comp.append(members)
            comp_sizes.append(cnt)
            comp_sides.append([set(), set()])
            for u in members:
                comp_sides[-1][color[u] - 1].add(u)

    C = len(comp)
    total = sum(s[0] + s[1] for s in comp_sizes)
    if k > total:
        print("NO")
        return

    INF = float('inf')
    dp = [[False] * (k + 1) for _ in range(C + 1)]
    dp[0][0] = True
    for i in range(1, C + 1):
        a, b = comp_sizes[i - 1]
        for j in range(k + 1):
            if dp[i - 1][j]:
                if j + a <= k:
                    dp[i][j + a] = True
                if j + b <= k:
                    dp[i][j + b] = True

    if not dp[C][k]:
        print("NO")
        return

    j = k
    chosen_side = [0] * C
    for i in range(C, 0, -1):
        a, b = comp_sizes[i - 1]
        if j >= a and dp[i - 1][j - a]:
            chosen_side[i - 1] = 0
            j -= a
        else:
            chosen_side[i - 1] = 1
            j -= b

    result = []
    for i in range(C):
        side = chosen_side[i]
        result.extend(comp_sides[i][side])

    print("YES")
    print(' '.join(map(str, result)))

solve()
