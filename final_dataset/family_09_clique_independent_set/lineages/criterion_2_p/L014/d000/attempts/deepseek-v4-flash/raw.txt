We can model the network as a bipartite graph. The structural property guarantees that the graph is bipartite, and the two color classes are independent sets.  

We can pick the larger color class if it has at least `k` vertices; otherwise, we can pick from the smaller one if it has at least `k` vertices. If neither has `k` vertices, the answer is `NO`.

```python
import sys
from collections import deque

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * n
    part = [[], []]
    
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            part[0].append(i)
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        part[color[v]].append(v)
                        q.append(v)
    
    if len(part[0]) >= k:
        chosen = part[0][:k]
    elif len(part[1]) >= k:
        chosen = part[1][:k]
    else:
        print("NO")
        return
    
    print("YES")
    print(" ".join(str(x + 1) for x in chosen))

if __name__ == "__main__":
    solve()
```
