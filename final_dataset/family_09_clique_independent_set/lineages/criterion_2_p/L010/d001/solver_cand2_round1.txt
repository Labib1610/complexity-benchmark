```python
import sys

# The problem states that for every interference link (u, v), the sum of their 
# secret codes is odd. This implies that one component must have an odd code 
# and the other an even code. Thus, the graph is bipartite.
# In a bipartite graph, the two sets of the bipartition are independent sets.
# Let the two sets be S1 and S2. Any subset of S1 is safe, and any subset of S2 is safe.
# We need to find a subset of size k. If |S1| >= k, we can pick k elements from S1.
# If |S2| >= k, we can pick k elements from S2.
# If both |S1| < k and |S2| < k, then no safe subset of size k exists.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    s1 = []
    s2 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [i]
            color[i] = 0
            s1.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        if color[v] == 0:
                            s1.append(v)
                        else:
                            s2.append(v)
                        stack.append(v)
    
    if len(s1) >= k:
        print("YES")
        print(*(s1[:k]))
    elif len(s2) >= k:
        print("YES")
        print(*(s2[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
