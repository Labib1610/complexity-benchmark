import sys

# The problem states that for every edge (u, v), there exist values x_u, x_v such that x_u + x_v = 0.
# This implies x_u = -x_v. If we have a connected component, picking a node and assigning it
# a value x forces the values of all its neighbors to be -x, and their neighbors to be x, and so on.
# This means the graph is bipartite.
# In a bipartite graph, each connected component can be partitioned into two independent sets,
# say S_i and T_i. Any conflict-free configuration must pick a subset of nodes from S_i U T_i
# such that no two are connected.
# Since the graph is bipartite, we can pick either all nodes in S_i or all nodes in T_i,
# or any subset of S_i U T_i that doesn't contain an edge.
# Actually, the problem is simpler: we have a bipartite graph. We want to pick k nodes.
# For each connected component, we have two sets S_i and T_i. We can pick any subset of S_i
# and any subset of T_i, but we cannot pick a node from S_i and its neighbor in T_i.
# Wait, the condition is just that we cannot pick two connected nodes.
# This is the Maximum Independent Set problem, which is NP-hard in general,
# but here we need to pick exactly k nodes.
# Actually, for each component, we can pick either S_i or T_i.
# Let |S_i| = s_i and |T_i| = t_i. We can pick any number of nodes from S_i and T_i
# as long as we don't pick an edge. But we can just pick all of S_i or all of T_i.
# Actually, we can pick any subset of S_i and any subset of T_i? No, that's wrong.
# We can pick any subset of S_i and any subset of T_i such that no edge is covered.
# But if we pick a subset of S_i and a subset of T_i, we just need to ensure no edge exists.
# The simplest way is to pick either S_i or T_i.
# Can we pick more? Yes, we can pick any independent set.
# But the problem is solvable if we can pick k nodes.
# Since it's bipartite, we can use DP to see if we can pick k nodes.
# For each component, we can pick any subset of S_i or T_i.
# Actually, we can pick any subset of S_i and any subset of T_i that are independent.
# The maximum number of nodes we can pick from a component is max(|S_i|, |T_i|).
# Wait, the problem is simpler: we can pick any subset of S_i and any subset of T_i
# as long as no edge is violated. This is equivalent to picking a subset of S_i and
# a subset of T_i such that no edge (u, v) has both u and v selected.
# This is exactly the definition of an independent set.
# However, in a bipartite graph, the size of the maximum independent set is n - matching.
# But we just need to find *any* k nodes.
# We can pick all nodes in S_i or all nodes in T_i.
# Let's use DP: for each component, we can contribute any number of nodes from 0 to max(|S_i|, |T_i|).
# Actually, we can contribute any number of nodes from 0 to |S_i| + |T_i|? No.
# We can contribute any number of nodes from 0 to |S_i| by picking a subset of S_i,
# and 0 to |T_i| by picking a subset of T_i.
# The total number of nodes we can pick is the sum of sizes of independent sets.
# For each component, we can pick any number of nodes from 0 to |S_i| (by picking a subset of S_i)
# or 0 to |T_i| (by picking a subset of T_i).
# So for each component, we can pick any number of nodes in [0, |S_i|] OR [0, |T_i|].

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = map(int, input_data[idx:idx+2])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = {}
    components = []
    for i in range(1, n + 1):
        if i not in color:
            s, t = [], []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: s.append(u)
                else: t.append(u)
                for v in adj[u]:
                    if v not in color:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append((s, t))
            
    # dp[i][j] = can we pick j nodes using first i components
    # To reconstruct, store which choice was made
    dp = {0: []}
    for s, t in components:
        new_dp = {}
        for j, path in dp.items():
            # Option 1: pick subset of s (size 0 to len(s))
            for count in range(len(s) + 1):
                new_dp[j + count] = path + [s[:count]]
            # Option 2: pick subset of t (size 0 to len(t))
            for count in range(len(t) + 1):
                new_dp[j + count] = path + [t[:count]]
        dp = new_dp
        
    if k in dp:
        print("YES")
        res = []
        for group in dp[k]:
            res.extend(group)
        print(*(res))
    else:
        print("NO")

solve()
