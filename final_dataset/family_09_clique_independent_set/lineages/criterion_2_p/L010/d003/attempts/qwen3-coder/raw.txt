```python
import sys
from collections import defaultdict, deque

def read_input():
    n, m, k = map(int, sys.stdin.readline().split())
    edges = []
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    return n, m, k, edges, adj

def bipartite_check(n, adj):
    color = [0] * (n + 1)  # 0: unvisited, 1: color A, -1: color B
    partition = {1: [], -1: []}
    
    for start in range(1, n + 1):
        if color[start] == 0:
            queue = deque([start])
            color[start] = 1
            partition[1].append(start)
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in adj[node]:
                    if color[neighbor] == 0:
                        color[neighbor] = -current_color
                        partition[-current_color].append(neighbor)
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        return None, None  # Not bipartite
    
    return partition[1], partition[-1]

def solve():
    n, m, k, edges, adj = read_input()
    
    if m == 0:
        print("YES")
        print(' '.join(map(str, range(1, k + 1))))
        return
    
    # Check if the graph is bipartite
    set_a, set_b = bipartite_check(n, adj)
    
    if set_a is None:
        # This shouldn't happen according to the problem constraint
        print("NO")
        return
    
    # Try to find a stable group of size k
    if k <= len(set_a):
        print("YES")
        print(' '.join(map(str, set_a[:k])))
    elif k <= len(set_b):
        print("YES")
        print(' '.join(map(str, set_b[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
