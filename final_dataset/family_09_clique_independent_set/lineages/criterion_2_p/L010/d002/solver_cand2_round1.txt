```python
import sys

# The problem states that the graph is such that we can assign directions to edges
# such that no component has both incoming and outgoing edges.
# This means the graph is a collection of disjoint components, where each component
# is a complete bipartite graph (or a collection of edges forming a structure
# where each connected component is a complete bipartite graph).
# Actually, the condition "no component has both incoming and outgoing links"
# implies that the graph is a collection of disjoint complete bipartite graphs.
# In a complete bipartite graph with partitions of size A and B, the maximum
# independent set is max(A, B).
# We can find the connected components, and for each component, identify the
# two partitions. Since it's a complete bipartite graph, we can use 2-coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append(comp)

    # For each component, we have two sets of nodes: those with color 0 and color 1.
    # Since it's a complete bipartite graph, any node in color 0 is connected to
    # all nodes in color 1. Thus, we can pick either all nodes in color 0 OR
    # all nodes in color 1 as an independent set.
    
    # We want to pick a total of k nodes.
    # Let S_i be the set of sizes of the two partitions in component i: (c0_i, c1_i).
    # We need to choose one from each pair such that the sum is k.
    # This is a variation of the subset sum problem.
    
    dp = {0: []}
    for i, comp in enumerate(components):
        c0 = [node for node, c in comp if c == 0]
        c1 = [node for node, c in comp if c == 1]
        new_dp = {}
        for s, nodes in dp.items():
            if s + len(c0) <= k:
                new_dp[s + len(c0)] = nodes + c0
            if s + len(c1) <= k:
                new_dp[s + len(c1)] = nodes + c1
        dp = new_dp
        
    if k in dp:
        print("YES")
        print(*(dp[k]))
    else:
        print("NO")

solve()
```
