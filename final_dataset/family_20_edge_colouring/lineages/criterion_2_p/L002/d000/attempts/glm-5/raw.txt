import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    nL = int(next(it))
    nR = int(next(it))
    m = int(next(it))
    
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
    
    diff = [0] * (nL + 1)
    degL = [0] * nL
    degR = [0] * nR
    
    for u, v in edges:
        degL[u] += 1
        degR[v] += 1
    
    max_deg = 0
    if nL > 0:
        max_deg = max(degL)
    if nR > 0:
        max_deg = max(max_deg, max(degR))
    
    capacity = [0] * max_deg
    
    ans = [0] * m
    
    for i, (u, v) in enumerate(edges):
        if u < v:
            x, y = u, v
            is_left = True
        else:
            x, y = v, u
            is_left = False
        
        cap = 0
        while (capacity[cap] >> x) & 1:
            cap += 1
        
        ans[i] = cap + 1
        capacity[cap] |= (1 << x)
    
    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
