import sys
from collections import defaultdict
import heapq

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    nL = int(data[idx])
    idx += 1
    nR = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    
    edges = []
    adj = [defaultdict(int) for _ in range(nL + nR + 1)]
    
    for i in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        edges.append((u, v + nL))
        adj[u][v + nL] += 1
        adj[v + nL][u] += 1
    
    # Compute maximum degree
    max_degree = 0
    for i in range(1, nL + nR + 1):
        max_degree = max(max_degree, len(adj[i]))
    
    if max_degree == 0:
        print("YES")
        return
    
    # Create adjacency list for the graph (each edge is a node)
    edge_adj = [[] for _ in range(m)]
    edge_map = {}
    
    for i in range(m):
        u, v = edges[i]
        edge_map[(u, v)] = i
    
    # For each vertex, connect all its incident edges
    for node in range(1, nL + nR + 1):
        neighbors = list(adj[node].keys())
        deg = len(neighbors)
        # We need to connect edges that share this node
        # But we have to be careful about multiple edges
        edge_indices = []
        temp_map = {}
        for i in range(len(edges)):
            u, v = edges[i]
            if u == node or v == node:
                cnt = adj[node][u if u != node else v]
                for _ in range(cnt):
                    edge_indices.append(i)
        
        # Now connect every pair among these edge indices
        for i in range(len(edge_indices)):
            for j in range(i+1, len(edge_indices)):
                e1 = edge_indices[i]
                e2 = edge_indices[j]
                edge_adj[e1].append(e2)
                edge_adj[e2].append(e1)
    
    # Color the line graph with Delta colors
    result = [0] * m
    available = [True] * (max_degree + 1)
    
    for i in range(m):
        # Mark colors used by neighbors
        for neighbor in edge_adj[i]:
            if result[neighbor] != 0:
                available[result[neighbor]] = False
        
        # Find first available color
        for color in range(1, max_degree + 1):
            if available[color]:
                result[i] = color
                break
                
        # Reset availability
        for neighbor in edge_adj[i]:
            if result[neighbor] != 0:
                available[result[neighbor]] = True
    
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
