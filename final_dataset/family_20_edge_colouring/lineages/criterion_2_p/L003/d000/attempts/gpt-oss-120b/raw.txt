import sys
from collections import deque, defaultdict

def solve() -> None:
    sys.setrecursionlimit(1 << 25)
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    nL = int(next(it))
    nR = int(next(it))
    m = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    degL = [0] * nL
    degR = [0] * nR
    for u, v in edges:
        degL[u] += 1
        degR[v] += 1
    Delta = 0
    for d in degL:
        if d > Delta:
            Delta = d
    for d in degR:
        if d > Delta:
            Delta = d
    if Delta == 0:
        print("YES")
        print()
        return

    # for each vertex store color -> edge id (at most one per color)
    left_col = [dict() for _ in range(nL)]
    right_col = [dict() for _ in range(nR)]
    ans = [0] * m

    for eid, (u, v) in enumerate(edges):
        # find free colors on u and v
        used_u = left_col[u]
        used_v = right_col[v]
        free_u = None
        free_v = None
        for c in range(1, Delta + 1):
            if free_u is None and c not in used_u:
                free_u = c
            if free_v is None and c not in used_v:
                free_v = c
            if free_u is not None and free_v is not None:
                break
        if free_u == free_v:
            c = free_u
            left_col[u][c] = eid
            right_col[v][c] = eid
            ans[eid] = c
            continue

        # need to find alternating path between colors free_u (cu) and free_v (cv)
        cu, cv = free_u, free_v
        # BFS over vertices, alternating colors cu and cv
        # state: vertex side (0 left,1 right), vertex index
        prev = {}  # key -> (prev_key, edge_id, set_color)
        dq = deque()
        start_key = (0, u)  # left side
        dq.append(start_key)
        prev[start_key] = None
        found = None
        while dq and found is None:
            side, x = dq.popleft()
            if side == 0:  # left, look for edge of color cu
                if cu in left_col[x]:
                    eid2 = left_col[x][cu]
                    _, y = edges[eid2]
                    nxt = (1, y)
                    if nxt not in prev:
                        prev[nxt] = ( (side, x), eid2, cu )
                        dq.append(nxt)
                else:
                    # cu free on this left vertex, but we started from u which already free, ignore
                    pass
            else:  # right, look for edge of color cv
                if cv in right_col[x]:
                    eid2 = right_col[x][cv]
                    u2, _ = edges[eid2]
                    nxt = (0, u2)
                    if nxt not in prev:
                        prev[nxt] = ( (side, x), eid2, cv )
                        dq.append(nxt)
                else:
                    # cv free on this right vertex -> we reached target
                    found = (side, x)
        # reconstruct path and flip colors
        # start from found vertex back to start
        cur = found
        while cur != start_key:
            p, e_id, col = prev[cur]
            # flip color of edge e_id: it currently has col, change to the other color
            if col == cu:
                newcol = cv
            else:
                newcol = cu
            u_e, v_e = edges[e_id]
            # remove old mapping
            del left_col[u_e][col]
            del right_col[v_e][col]
            # add new mapping
            left_col[u_e][newcol] = e_id
            right_col[v_e][newcol] = e_id
            cur = p
        # now assign cu to current edge
        left_col[u][cu] = eid
        right_col[v][cu] = eid
        ans[eid] = cu

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
