    idx = 3

    degR = [0]*(nR+1)

        edges.append((u,v))

        degR[v] += 1
    maxL = max(degL) if nL >= 1 else 0
    maxR = max(degR) if nR >= 1 else 0
    delta = max(maxL, maxR)

    # A simple greedy with augmenting paths works but is complex.
    #
    # Alternative: Use the fact that we can do a simple greedy if we process edges in a good order.
    # Actually, for bipartite graphs, a simple greedy edge coloring (pick any available color)
    # can use up to 2*delta - 1 colors in worst case for general graphs, but for bipartite

    # Euler tour / splitting into matchings approach, or the augmenting path approach.

    #
    # Let's implement the augmenting path based algorithm (Misra & Gries style for bipartite,
    # or the simpler version for bipartite graphs).
    #

    # Repeatedly find a perfect matching in a delta-regular bipartite graph.
    # But our graph may not be regular.
    #
    # Let's use the approach from "A simple algorithm for edge coloring bipartite multigraphs"
    # which processes edges one by one and uses color swapping via alternating paths.
    #
    # Implementation:
    # Maintain for each vertex (both sides) a mapping color -> edge index (or -1 if free).
    # Also maintain a "free color" for each vertex (any color not currently used).

    #   Find a color free at u (fu) and a color free at v (fv).
    #   If fu == fv: assign fu.
    #   Else: assign fu to edge, then try to recolor the edge at v that has color fu
    #         by swapping along an alternating path from v using colors fu and fv.
    #
    # This is the standard algorithm. Let's implement it.

    # For left vertices: color_of_L[u][c] = edge index or -1
    # For right vertices: color_of_R[v][c] = edge index or -1
    # But delta can be up to 400000 and nL up to 100000, so we can't allocate 2D arrays.
    # Use dictionaries or lists of lists only for vertices with degree > 0.
    #

    # For each vertex, store a dict: color -> edge_index
    # And a variable: any_free_color (we can compute by scanning, but that's slow).
    #
    # Better: maintain a "free color" stack/list for each vertex.
    # Initially, free colors for vertex with degree d are {1, 2, ..., delta} minus used ones.

    # Let's think differently. Since delta <= 400000 and sum of degrees = 2m <= 800000,

    # nL, nR up to 100000, m up to 400000, delta up to 400000.

    # color_of_R[v]: dict mapping color -> edge_index
    # free_L[u]: a color not used by u (we'll maintain a "candidate" that we check)

    color_of_R = [dict() for _ in range(nR+1)]

    free_R = [1]*(nR+1)

        d = color_of_L[u]
        c = free_L[u]
        while c in d:
            c += 1
        free_L[u] = c
        return c

        d = color_of_R[v]
        c = free_R[v]
        while c in d:
            c += 1
        free_R[v] = c
        return c

        fu = get_free_L(u)
        fv = get_free_R(v)

            color_of_L[u][fu] = i
            color_of_R[v][fu] = i
            # update free candidates

            if free_R[v] == fu:
                free_R[v] = fu + 1

            # We need to find an alternating path from v using colors fu and fv.
            # The path: v -> (edge with color fu at v) -> u' -> (edge with color fv at u') -> ...

            # The edge at v with color fu (if any) needs to be recolored to fv.
            # Then at the other endpoint of that edge, fu becomes free but fv might be used.

            # Let's do the swap along the path.
            # We'll collect the path of edges and then recolor.

            # The path starts at v. At v, color fu is used by some edge e1=(u1,v).
            # We want to change e1's color from fu to fv.
            # At u1, fv might be used by some edge e2=(u1,v2). Change e2 from fv to fu.
            # At v2, fu might be used by some edge e3=(u3,v2). Change e3 from fu to fv.

            # We alternate: at right vertices, we look for color fu; at left vertices, color fv.

            path_edges = []  # list of (edge_index, old_color, new_color)
            curr_v = v
            curr_color_to_find = fu  # at right vertex, find this color
            curr_new_color = fv

            # The edge (u,v) will get color fu.
            # We need to find at v an edge with color fu, move it to fv.
            # Then at its left endpoint, find edge with color fv, move to fu.
            # etc.

            # Actually, let's think of it as: we're building a fan or path.
            # Standard algorithm: we have edge (u,v), free colors fu at u, fv at v, fu != fv.
            # We color (u,v) with fu. Now v has two edges with color fu (the new one and old one).
            # We need to recolor the old one. We try to recolor it to fv.
            # If fv is free at the other endpoint of the old edge, great.
            # Otherwise, we continue the alternating path.

            # Let's implement iteratively:
            # We have a "conflict" at vertex v: color fu is now used twice (by edge i and old edge).
            # We want to move the old edge to fv.
            # old_edge = color_of_R[v][fu]  (but we haven't assigned yet)

            color[i] = fu
            old_edge_at_v = color_of_R[v].get(fu, -1)
            color_of_L[u][fu] = i
            color_of_R[v][fu] = i

            if old_edge_at_v == -1:
                # No conflict, fu was free at v. But we said fu != fv and fu was free at u,

                # If fu is free at v, then we can just use fu. But we computed fv = get_free_R(v)
                # which is the smallest free color at v. If fu < fv, then fu is used at v.

            # We want to recolor old_edge_at_v to fv.
