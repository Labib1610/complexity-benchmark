We can model this as a bipartite graph edge-coloring problem.  
The minimum number of colors needed equals the maximum degree Δ.  
We can find such a coloring using a standard algorithm for bipartite edge coloring.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

nP, nS, m = map(int, input().split())
edges = []
degP = [0] * (nP + 1)
degS = [0] * (nS + 1)

for _ in range(m):
    u, v = map(int, input().split())
    edges.append((u, v))
    degP[u] += 1
    degS[v] += 1

Delta = max(max(degP), max(degS))

# Build adjacency lists with edge indices
adjP = [[] for _ in range(nP + 1)]
adjS = [[] for _ in range(nS + 1)]

for idx, (u, v) in enumerate(edges):
    adjP[u].append((v, idx))
    adjS[v].append((u, idx))

# Color assignment
color = [0] * m

# For each primary vertex, assign colors to its incident edges
for u in range(1, nP + 1):
    used = [False] * (Delta + 1)
    for v, idx in adjP[u]:
        if color[idx] != 0:
            used[color[idx]] = True
    # Find free colors for each edge
    for v, idx in adjP[u]:
        if color[idx] == 0:
            # find smallest free color
            c = 1
            while c <= Delta and used[c]:
                c += 1
            color[idx] = c
            used[c] = True

# Now fix conflicts on secondary side using Kempe chain argument
# We'll do a standard bipartite edge coloring algorithm:
# For each edge, if conflict, swap colors along alternating path.

# Build adjacency with colors for secondary side
adjS_colored = [[] for _ in range(nS + 1)]
for idx, (u, v) in enumerate(edges):
    adjS_colored[v].append((u, idx, color[idx]))

# Rebuild primary adjacency with colors
adjP_colored = [[] for _ in range(nP + 1)]
for idx, (u, v) in enumerate(edges):
    adjP_colored[u].append((v, idx, color[idx]))

# Fix conflicts
for v in range(1, nS + 1):
    # Check for duplicate colors at this secondary
    color_to_edge = {}
    for u, idx, c in adjS_colored[v]:
        if c in color_to_edge:
            # conflict between idx and color_to_edge[c]
            idx1 = idx
            idx2 = color_to_edge[c]
            # find a free color for idx2 at this vertex
            used = set()
            for _, _, cc in adjS_colored[v]:
                used.add(cc)
            free_c = 1
            while free_c in used:
                free_c += 1
            # Kempe chain: swap colors between idx1 and idx2 along alternating path
            # Start from idx2, color free_c
            # We'll implement iterative path recoloring
            stack = [(idx2, free_c)]
            visited_edges = set()
            while stack:
                eidx, newc = stack.pop()
                if eidx in visited_edges:
                    continue
                visited_edges.add(eidx)
                oldc = color[eidx]
                color[eidx] = newc
                u, vv = edges[eidx]
                # find the other edge at vv with color newc (if any)
                for _, e2, c2 in adjS_colored[vv]:
                    if e2 != eidx and c2 == newc:
                        stack.append((e2, oldc))
                        break
                # also check primary side
                for _, e2, c2 in adjP_colored[u]:
                    if e2 != eidx and c2 == newc:
                        stack.append((e2, oldc))
                        break
            # After swap, update adjacency color lists
            adjS_colored[v] = [(u2, i2, color[i2]) for u2, i2, _ in adjS_colored[v]]
            adjP_colored = [[] for _ in range(nP + 1)]
            for i2, (uu, vv) in enumerate(edges):
                adjP_colored[uu].append((vv, i2, color[i2]))
            adjS_colored = [[] for _ in range(nS + 1)]
            for i2, (uu, vv) in enumerate(edges):
                adjS_colored[vv].append((uu, i2, color[i2]))
        else:
            color_to_edge[c] = idx

# Output
print("YES")
print(" ".join(map(str, color)))
```
