import sys

def main():
    data = sys.stdin.buffer.read().split()
    idx = 0
    nL = int(data[idx]); idx += 1
    nR = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1

    edges = []
    degL = [0] * (nL + 1)
    degR = [0] * (nR + 1)

    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        edges.append((u, v))
        degL[u] += 1
        degR[v] += 1

    delta = 0
    for i in range(1, nL + 1):
        if degL[i] > delta:
            delta = degL[i]
    for i in range(1, nR + 1):
        if degR[i] > delta:
            delta = degR[i]

    if delta == 0:
        out = ["YES"]
        if m > 0:
            out.append("")
        sys.stdout.write("\n".join(out))
        return

    # Build bipartite graph: left ports 0..nL-1, right ports nL..nL+nR-1
    N = nL + nR
    head = [-1] * N
    to = [0] * m
    nxt = [0] * m
    edge_id = [0] * m

    for i, (u, v) in enumerate(edges):
        lu = u - 1
        rv = nL + v - 1
        to[i] = rv
        nxt[i] = head[lu]
        head[lu] = i
        edge_id[i] = i

    color = [0] * m
    used = [0] * (delta + 1)

    # Vizing-style greedy coloring with adjacency scan
    for s in range(N):
        e = head[s]
        while e != -1:
            t = to[e]
            # scan neighbors of t to collect used colors
            f = head[t]
            while f != -1:
                used[color[f]] = 1
                f = nxt[f]
            # find first free color
            c = 1
            while used[c]:
                c += 1
            color[e] = c
            # clear used array
            f = head[t]
            while f != -1:
                used[color[f]] = 0
                f = nxt[f]
            e = nxt[e]

    out = ["YES"]
    out.append(" ".join(str(color[i]) for i in range(m)))
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
