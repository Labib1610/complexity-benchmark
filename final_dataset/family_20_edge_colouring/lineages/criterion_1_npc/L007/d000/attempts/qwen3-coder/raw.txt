```python
import sys
from collections import defaultdict

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # Read edges
    edges = []
    adj = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
    
    # Each node has exactly 3 edges
    # We need to color the edges with 3 colors such that edges incident to 
    # the same node have different colors
    
    # This is equivalent to edge coloring of a 3-regular graph with 3 colors
    # Since each node has degree 3, we can always do this if the graph is simple
    # and each component is either a cycle or a path with even length or...
    # Actually, every 3-regular graph can be edge-colored with 3 colors iff it's class 1
    # But checking if a 3-regular graph is class 1 is NP-hard in general
    # However, since n is small, we can try a greedy approach
    
    # For each node, we need to assign 3 different colors to its 3 edges
    # Let's try a greedy algorithm:
    # For each edge, try to assign it the smallest color that doesn't conflict
    
    color = [0] * m  # color[i] is the color of edge i (1, 2, or 3)
    
    # For each node, keep track of which colors are used on its edges
    node_colors = [set() for _ in range(n)]
    
    # Try to color edges one by one
    for i in range(m):
        u, v = edges[i]
        
        # Find a color that is not used by either u or v
        used_colors = node_colors[u] | node_colors[v]
        c = 1
        while c in used_colors:
            c += 1
        
        if c > 3:
            print("NO")
            return
            
        color[i] = c
        node_colors[u].add(c)
        node_colors[v].add(c)
    
    # Check if we succeeded
    # Each node should have exactly 3 edges with 3 different colors
    for i in range(n):
        if len(node_colors[i]) != 3:
            print("NO")
            return
    
    print("YES")
    print(' '.join(str(c) for c in color))

solve()
```
