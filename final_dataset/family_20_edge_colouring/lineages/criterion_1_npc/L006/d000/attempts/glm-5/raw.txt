import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    color = [0] * m
    
    # Eulerian circuit on a 3-regular graph
    # Since n is even and the graph is 3-regular, it's Eulerian.
    # We find an Eulerian circuit and alternate colors 1,2,1,2,...
    # The third color is used for bridges.
    
    # Using iterative Hierholzer's algorithm
    ptr = [0] * (n + 1)
    used = [False] * m
    stack = [1]
    circuit = []
    
    while stack:
        v = stack[-1]
        while ptr[v] < len(adj[v]):
            u, eid = adj[v][ptr[v]]
            ptr[v] += 1
            if not used[eid]:
                used[eid] = True
                stack.append(u)
                break
        else:
            circuit.append(v)
            stack.pop()
    
    # circuit now has the Eulerian circuit in reverse
    # Assign colors 1,2,1,2,... along the circuit
    # Each edge appears once in the circuit
    
    # We need to map circuit vertices to edges
    # The circuit is a sequence of vertices v0, v1, ..., vk where v0 = vk
    # Edge i connects circuit[i] and circuit[i+1]
    
    # But we need to find which original edge index this corresponds to
    # Let's use a different approach: track edges as we build the circuit
    
    # Rebuild with edge tracking
    ptr2 = [0] * (n + 1)
    used2 = [False] * m
    stack2 = [(1, -1)]  # (vertex, edge_used_to_get_here)
    edge_order = []
    
    while stack2:
        v, came_from = stack2[-1]
        found = False
        while ptr2[v] < len(adj[v]):
            u, eid = adj[v][ptr2[v]]
            ptr2[v] += 1
            if not used2[eid]:
                used2[eid] = True
                stack2.append((u, eid))
                found = True
                break
        if not found:
            if came_from != -1:
                edge_order.append(came_from)
            stack2.pop()
    
    # edge_order now has edges in the order of the Eulerian circuit
    # Assign alternating colors
    for idx, eid in enumerate(edge_order):
        color[eid] = 1 if idx % 2 == 0 else 2
    
    # Now check if all junctions have 3 distinct colors
    # At each junction, 3 edges meet. In the Eulerian circuit, 
    # the junction is visited and 3 edges are traversed.
    # If the junction is visited once (3 edges in sequence in circuit),
    # colors are 1,2,1 or 2,1,2 - two edges share color -> need bridge
    # If visited twice (one edge is a "return"), need to check
    
    # Actually, let's just check and fix
    # For each vertex, check if its 3 edges have distinct colors
    # If not, one edge needs to be color 3
    
    # Find edges that need to be color 3
    # At each vertex, if two edges have the same color, 
    # one of them must be changed to 3
    
    # Build vertex -> edges mapping with positions in circuit
    edge_pos = [0] * m
    for pos, eid in enumerate(edge_order):
        edge_pos[eid] = pos
    
    # For each vertex, look at its 3 edges and their positions in circuit
    needs_change = [False] * m
    
    for v in range(1, n + 1):
        edge_list = [eid for (u, eid) in adj[v]]
        # Sort by position in circuit
        edge_list.sort(key=lambda e: edge_pos[e])
        # These edges appear in this order in the circuit
        # Colors alternate, so positions 0,1 -> colors 1,2 or 2,1
        # positions 0,1,2 -> colors 1,2,1 or 2,1,2
        # If 3 edges are consecutive (vertex visited once), 
        # first and third have same color
        # If vertex visited twice, one edge is separate
        
        c = [color[e] for e in edge_list]
        if c[0] == c[1]:
            needs_change[edge_list[0]] = True
        elif c[1] == c[2]:
            needs_change[edge_list[1]] = True
        elif c[0] == c[2]:
            needs_change[edge_list[0]] = True
    
    for i in range(m):
        if needs_change[i]:
            color[i] = 3
    
    # Verify
    for v in range(1, n + 1):
        cols = set()
        for (u, eid) in adj[v]:
            cols.add(color[eid])
        if len(cols) < 3:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color)))

solve()
