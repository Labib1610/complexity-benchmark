import sys
from collections import defaultdict

def solve():
    n, m = map(int, input().split())
    
    # Read edges
    edges = []
    adj = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Each vertex must have degree 3
    for u in range(1, n + 1):
        if len(adj[u]) != 3:
            print("NO")
            return
    
    # Try to 3-color the edges (assign gases 1,2,3 to edges)
    # This is equivalent to edge-coloring a 3-regular graph with 3 colors
    # which is possible iff the graph is bipartite or has a perfect matching structure
    
    # We'll use a greedy approach with backtracking
    
    color = [0] * m  # 0 means uncolored, 1,2,3 are the colors
    
    def is_valid(edge_idx, c):
        u, v = edges[edge_idx]
        # Check if any adjacent edge at u or v already has color c
        for neighbor, idx in adj[u]:
            if color[idx] == c:
                return False
        for neighbor, idx in adj[v]:
            if color[idx] == c:
                return False
        return True
    
    def backtrack(edge_idx):
        if edge_idx == m:
            return True
            
        u, v = edges[edge_idx]
        for c in [1, 2, 3]:
            if is_valid(edge_idx, c):
                color[edge_idx] = c
                if backtrack(edge_idx + 1):
                    return True
                color[edge_idx] = 0
        return False
    
    # Since this is a 3-regular graph (cubic graph) with n even,
    # and m = 3n/2, it's a cubic graph which can be edge-colored with 3 colors
    # if and only if it's class 1 (not class 2).
    # All bipartite cubic graphs are class 1.
    # But we need to check if a valid coloring exists.
    
    # Let's try a more systematic approach
    # Build the line graph and try to 3-color it
    
    if backtrack(0):
        print("YES")
        print(' '.join(map(str, color)))
    else:
        print("NO")

solve()
