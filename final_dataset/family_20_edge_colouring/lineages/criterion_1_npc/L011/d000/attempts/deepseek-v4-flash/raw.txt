We can model this as a **3-edge-coloring** problem on a **3-regular graph** (each vertex has degree 3).  

By **Kőnig’s line coloring theorem**, every bipartite graph has an edge coloring with Δ colors, but here the graph is not necessarily bipartite. However, for **3-regular graphs**, a 3-edge-coloring exists if and only if the graph is **bridgeless** (by the **Four Color Theorem** equivalent for cubic graphs).  

We can solve it by:
1. Checking if the graph is **bridgeless** (no bridges).
2. If yes, find a 3-edge-coloring using a constructive algorithm (e.g., finding a perfect matching, removing it, then coloring the remaining 2-regular graph with 2 colors).

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(1000000)

def find_bridges(n, adj, edges):
    visited = [False] * (n + 1)
    tin = [-1] * (n + 1)
    low = [-1] * (n + 1)
    timer = 0
    bridges = set()

    def dfs(v, p_edge):
        nonlocal timer
        visited[v] = True
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in adj[v]:
            if eid == p_edge:
                continue
            if visited[to]:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    bridges.add(eid)
    dfs(1, -1)
    return bridges

def solve():
    input_data = sys.stdin.read().strip().split()
    n, m = int(input_data[0]), int(input_data[1])
    edges = []
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for eid in range(m):
        u = int(input_data[idx]); v = int(input_data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].append((v, eid))
        adj[v].append((u, eid))

    # Check for bridges
    bridges = find_bridges(n, adj, edges)
    if bridges:
        print("NO")
        return

    # Find a perfect matching using greedy + augmenting path (since graph is cubic and bridgeless)
    match = [-1] * (n + 1)
    used = [False] * (n + 1)

    def augment(u, seen):
        for v, eid in adj[u]:
            if seen[v]:
                continue
            seen[v] = True
            if match[v] == -1 or augment(match[v], seen):
                match[v] = u
                match[u] = v
                return True
        return False

    for u in range(1, n + 1):
        if match[u] == -1:
            seen = [False] * (n + 1)
            augment(u, seen)

    # Color matching edges with color 1
    color = [0] * m
    for u in range(1, n + 1):
        if match[u] != -1 and u < match[u]:
            for v, eid in adj[u]:
                if v == match[u]:
                    color[eid] = 1
                    break

    # Remove matching edges, remaining graph is 2-regular (cycles)
    remaining_adj = [[] for _ in range(n + 1)]
    for eid, (u, v) in enumerate(edges):
        if color[eid] == 0:
            remaining_adj[u].append((v, eid))
            remaining_adj[v].append((u, eid))

    # Color cycles with 2 and 3
    visited = [False] * (n + 1)
    for start in range(1, n + 1):
        if not visited[start] and remaining_adj[start]:
            # walk the cycle
            cur = start
            prev = -1
            while True:
                visited[cur] = True
                nxt, eid = remaining_adj[cur][0] if remaining_adj[cur][0][0] != prev else remaining_adj[cur][1]
                if visited[nxt] and nxt != start:
                    break
                # assign alternating colors
                if color[eid] == 0:
                    # determine color based on position
                    # We'll just alternate 2 and 3
                    if len(visited) % 2 == 0:
                        color[eid] = 2
                    else:
                        color[eid] = 3
                prev = cur
                cur = nxt
                if cur == start:
                    break

    # Ensure all edges colored
    for i in range(m):
        if color[i] == 0:
            color[i] = 2  # fallback

    print("YES")
    print(" ".join(map(str, color)))

if __name__ == "__main__":
    solve()
```
