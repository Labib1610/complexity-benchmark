import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    adj = [[] for _ in range(n + 1)]
    for i in range(m):
        u = int(input_data[2 + 2 * i])
        v = int(input_data[2 + 2 * i + 1])
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
    
    color = [0] * m
    visited = [False] * (n + 1)
    
    for start in range(1, n + 1):
        if visited[start]:
            continue
        stack = [(start, 0, -1)]
        visited[start] = True
        while stack:
            node, parity, parent_edge = stack.pop()
            for eid in adj[node]:
                if eid == parent_edge:
                    continue
                u, v = edges[eid]
                neighbor = v if u == node else u
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[eid] = 1 + (parity ^ 1)
                    stack.append((neighbor, parity ^ 1, eid))
                else:
                    if color[eid] == 0:
                        color[eid] = 1 + parity
    
    for u in range(1, n + 1):
        cnt = [0, 0, 0, 0]
        for eid in adj[u]:
            cnt[color[eid]] += 1
        if cnt[1] > 1 or cnt[2] > 1 or cnt[3] > 1:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color)))

solve()
