import sys
sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    edges = []
    g = [[] for _ in range(n)]
    for idx in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
        g[u].append((v, idx))
        g[v].append((u, idx))

    # ---------- bridge detection ----------
    tin = [-1] * n
    low = [0] * n
    timer = 0
    is_bridge = [False] * m

    def dfs(v, p_e):
        nonlocal timer
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in g[v]:
            if eid == p_e:
                continue
            if tin[to] != -1:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    is_bridge[eid] = True

    dfs(0, -1)
    if any(b for b in is_bridge):
        print("NO")
        return

    # ---------- Edmonds' blossom for maximum matching ----------
    # implementation adapted from https://github.com/cheran-senthil/PyRival/blob/master/pyrival/graph/matching.py
    class Blossom:
        def __init__(self, n):
            self.n = n
            self.adj = [[] for _ in range(n)]

        def add_edge(self, u, v, idx):
            self.adj[u].append((v, idx))
            self.adj[v].append((u, idx))

        def max_matching(self):
            n = self.n
            match = [-1] * n
            base = list(range(n))
            p = [-1] * n
            q = []
            in_blossom = [False] * n

            def lca(a, b):
                used = [False] * n
                while True:
                    a = base[a]
                    used[a] = True
                    if match[a] == -1:
                        break
                    a = p[match[a]]
                while True:
                    b = base[b]
                    if used[b]:
                        return b
                    b = p[match[b]]

            def mark_path(v, b, child):
                while base[v] != b:
                    in_blossom[base[v]] = in_blossom[base[match[v]]] = True
                    p[v] = child
                    child = match[v]
                    v = p[match[v]]

            def find_augmenting_path(root):
                nonlocal q, p, base, in_blossom
                p = [-1] * n
                base = list(range(n))
                q = []
                q_head = 0
                q.append(root)
                while q_head < len(q):
                    v = q[q_head]
                    q_head += 1
                    for to, _ in self.adj[v]:
                        if base[v] == base[to] or match[v] == to:
                            continue
                        if to == root or (match[to] != -1 and p[match[to]] != -1):
                            curbase = lca(v, to)
                            in_blossom = [False] * n
                            mark_path(v, curbase, to)
                            mark_path(to, curbase, v)
                            for i in range(n):
                                if in_blossom[base[i]]:
                                    base[i] = curbase
                                    if not in_blossom[i]:
                                        q.append(i)
                        elif p[to] == -1:
                            p[to] = v
                            if match[to] == -1:
                                return to
                            q.append(match[to])
                return -1

            def augment_path(finish):
                v = finish
                while v != -1:
                    pv = p[v]
                    ppv = match[pv] if pv != -1 else -1
                    match[v] = pv
                    match[pv] = v
                    v = ppv

            for v in range(n):
                if match[v] == -1:
                    finish = find_augmenting_path(v)
                    if finish != -1:
                        augment_path(finish)
            return match

    # Build blossom graph (same as original)
    blossom = Blossom(n)
    for idx, (u, v) in enumerate(edges):
        blossom.add_edge(u, v, idx)

    match = blossom.max_matching()
    # verify perfect matching
    if any(x == -1 for x in match):
        print("NO")
        return

    # ---------- construct coloring ----------
    color = [0] * m
    # color matching edges with 1
    for idx, (u, v) in enumerate(edges):
        if match[u] == v:
            color[idx] = 1

    # remaining edges form cycles
    adj2 = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        if color[idx] == 0:
            adj2[u].append((v, idx))
            adj2[v].append((u, idx))

    visited = [False] * n
    for start in range(n):
        if visited[start]:
            continue
        # traverse a cycle
        cur = start
        prev_edge = -1
        cycle_edges = []
        while True:
            visited[cur] = True
            # pick the unused edge
            for to, eid in adj2[cur]:
                if eid != prev_edge:
                    nxt = to
                    e_used = eid
                    break
            cycle_edges.append(e_used)
            prev_edge = e_used
            cur = nxt
            if cur == start:
                break

        # assign colors 2 and 3 alternating
        for i, eid in enumerate(cycle_edges):
            color[eid] = 2 if i % 2 == 0 else 3

    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
