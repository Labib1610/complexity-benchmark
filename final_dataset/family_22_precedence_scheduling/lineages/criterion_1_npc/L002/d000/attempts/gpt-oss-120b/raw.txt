import sys
from collections import deque, defaultdict

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    indeg = [0]*n
    for _ in range(m):
        a = int(next(it))-1
        b = int(next(it))-1
        adj[a].append(b)
        indeg[b] += 1

    # topological order
    q = deque([i for i in range(n) if indeg[i]==0])
    topo = []
    while q:
        v = q.popleft()
        topo.append(v)
        for to in adj[v]:
            indeg[to]-=1
            if indeg[to]==0:
                q.append(to)
    if len(topo)!=n:
        print("NO")
        return

    # earliest possible stage
    earliest = [1]*n
    for v in topo:
        for to in adj[v]:
            if earliest[to] < earliest[v]+1:
                earliest[to] = earliest[v]+1
    if max(earliest) > T:
        print("NO")
        return

    # schedule greedily by earliest
    stage_of = [0]*n
    ready = []
    indeg = [0]*n
    for u in range(n):
        indeg[u] = len([p for p in range(n) if u in adj[p]])  # recompute indeg
    # better: use original indeg copy
    indeg = [0]*n
    for u in range(n):
        for v in adj[u]:
            indeg[v]+=1
    bucket = defaultdict(deque)
    for i in range(n):
        if indeg[i]==0:
            bucket[earliest[i]].append(i)

    for cur_stage in range(1, T+1):
        cnt = 0
        # take up to M from current and later earliest buckets
        while cnt < M:
            # find smallest stage >= cur_stage with available nodes
            found = None
            for st in range(cur_stage, T+1):
                if bucket[st]:
                    found = st
                    break
            if found is None:
                break
            v = bucket[found].popleft()
            stage_of[v] = cur_stage
            cnt += 1
            for to in adj[v]:
                indeg[to]-=1
                if indeg[to]==0:
                    bucket[earliest[to]].append(to)
        # continue to next stage

    if any(s==0 for s in stage_of):
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, stage_of)))

if __name__ == "__main__":
    solve()
