import sys
import collections
import heapq

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    try:
        n = int(next(it))
        M = int(next(it))
        T = int(next(it))
        m = int(next(it))
    except StopIteration:
        return
    
    out_edges = [[] for _ in range(n)]
    in_degree = [0] * n
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        u -= 1
        v -= 1
        out_edges[u].append(v)
        in_degree[v] += 1
    
    latest_room = [T] * n
    q = collections.deque()
    
    for i in range(n):
        if in_degree[i] == 0:
            q.append(i)
    
    topo_order = []
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in out_edges[u]:
            if latest_room[u] <= latest_room[v]:
                latest_room[v] = latest_room[u] - 1
            in_degree[v] -= 1
            if in_degree[v] == 0:
                q.append(v)
    
    if len(topo_order) != n:
        sys.stdout.write("NO")
        return
    
    earliest_room = [1] * n
    q = collections.deque()
    rev_in_degree = [0] * n
    rev_edges = [[] for _ in range(n)]
    
    for u in range(n):
        for v in out_edges[u]:
            rev_edges[v].append(u)
            rev_in_degree[u] += 1
    
    for i in range(n):
        if rev_in_degree[i] == 0:
            q.append(i)
    
    while q:
        v = q.popleft()
        for u in rev_edges[v]:
            if earliest_room[u] >= earliest_room[v]:
                earliest_room[v] = earliest_room[u] + 1
            rev_in_degree[u] -= 1
            if rev_in_degree[u] == 0:
                q.append(u)
    
    for i in range(n):
        if earliest_room[i] > latest_room[i] or not (1 <= earliest_room[i] <= T) or not (1 <= latest_room[i] <= T):
            sys.stdout.write("NO")
            return
    
    flex = [(earliest_room[i], latest_room[i], i) for i in range(n)]
    flex.sort(key=lambda x: (x[1], x[0]))
    
    final_pos = [0] * n
    room_remaining = [M] * (T + 1)
    
    for e, l, u in flex:
        final_pos[u] = e
    
    for i in range(1, T + 1):
        target_room = i
        for idx in range(n):
            if final_pos[idx] == target_room:
                if room_remaining[target_room] > 0:
                    room_remaining[target_room] -= 1
                else:
                    if target_room == T:
                        sys.stdout.write("NO")
                        return
                    # Group with next room, try to flatten
                    found_next = False
                    for next_room in range(target_room + 1, T + 1):
                        if room_remaining[next_room] > 0 and next_room <= latest_room[idx]:
                            room_remaining[next_room] -= 1
                            final_pos[idx] = next_room
                            found_next = True
                            break
                    if not found_next:
                        sys.stdout.write("NO")
                        return
    
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, (final_pos[i] for i in range(n)))))

if __name__ == "__main__":
    solve()
