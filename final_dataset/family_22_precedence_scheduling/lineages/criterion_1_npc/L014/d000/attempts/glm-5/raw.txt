import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        indeg[b] += 1
    
    # earliest[i] = earliest minute job i can start (1-indexed)
    earliest = [1] * n
    q = deque()
    for i in range(n):
        if indeg[i] == 0:
            q.append(i)
    
    topo_order = []
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in adj[u]:
            if earliest[v] < earliest[u] + 1:
                earliest[v] = earliest[u] + 1
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    
    if len(topo_order) != n:
        print("NO")
        return
    
    # Check if any earliest exceeds T
    for i in range(n):
        if earliest[i] > T:
            print("NO")
            return
    
    # Group jobs by earliest minute
    groups = [[] for _ in range(T + 1)]
    for i in range(n):
        groups[earliest[i]].append(i)
    
    # Assign minutes greedily
    minute = [0] * n
    jobs_by_earliest = [deque(g) for g in groups]
    pending = deque()
    
    for t in range(1, T + 1):
        # Add jobs that become available at time t
        pending.extend(jobs_by_earliest[t])
        
        # Schedule up to M jobs
        for _ in range(M):
            if pending:
                job = pending.popleft()
                minute[job] = t
            else:
                break
    
    # Check if all jobs scheduled
    if any(m == 0 for m in minute):
        print("NO")
        return
    
    print("YES")
    print(' '.join(map(str, minute)))

if __name__ == "__main__":
    solve()
