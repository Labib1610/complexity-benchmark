from collections import defaultdict, deque

def solve():
    n, M, T, m = map(int, input().split())
    
    # Build graph and in-degree count
    graph = defaultdict(list)
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        a, b = map(int, input().split())
        graph[a].append(b)
        in_degree[b] += 1
    
    # Topological sort with level tracking
    queue = deque()
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            queue.append(i)
    
    # Level array to store the earliest shift each component can be scheduled
    level = [0] * (n + 1)
    topo_order = []
    
    while queue:
        node = queue.popleft()
        topo_order.append(node)
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            level[neighbor] = max(level[neighbor], level[node] + 1)
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    
    # Check if all components are processed (no cycles)
    if len(topo_order) != n:
        print("NO")
        return
    
    # Check if schedule fits within T shifts
    max_level = max(level[i] for i in range(1, n + 1))
    if max_level >= T:
        print("NO")
        return
    
    # Count how many components per level
    count_per_level = [0] * (max_level + 1)
    for i in range(1, n + 1):
        count_per_level[level[i]] += 1
    
    # Check if any level exceeds M workbenches
    if any(count > M for count in count_per_level):
        print("NO")
        return
    
    # Generate schedule
    schedule = [0] * (n + 1)
    for i in range(1, n + 1):
        schedule[i] = level[i] + 1  # Shifts are 1-indexed
    
    print("YES")
    print(*schedule[1:])

solve()
