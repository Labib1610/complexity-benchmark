from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    M = int(data[1])
    T = int(data[2])
    m = int(data[3])
    
    edges = []
    index = 4
    for _ in range(m):
        a = int(data[index]) - 1
        b = int(data[index+1]) - 1
        edges.append((a, b))
        index += 2
    
    # Build graph
    graph = defaultdict(list)
    indegree = [0] * n
    for a, b in edges:
        graph[a].append(b)
        indegree[b] += 1
    
    # Topological sort with level assignment
    queue = deque()
    for i in range(n):
        if indegree[i] == 0:
            queue.append(i)
    
    levels = [-1] * n
    level = 0
    count = 0
    
    while queue:
        next_queue = deque()
        # Process current level
        if len(queue) > M * T:
            print("NO")
            return
            
        # Assign as many as possible to current level (up to M reactions)
        reactions_in_level = []
        for _ in range(min(len(queue), M)):
            node = queue.popleft()
            levels[node] = level
            count += 1
            reactions_in_level.append(node)
        
        # Process neighbors for next level
        for node in reactions_in_level:
            for neighbor in graph[node]:
                indegree[neighbor] -= 1
                if indegree[neighbor] == 0:
                    next_queue.append(neighbor)
        
        queue = next_queue
        level += 1
        
        if level > T:
            print("NO")
            return
    
    if count != n:
        print("NO")
        return
    
    # Convert 0-based levels to 1-based cycles
    result = [str(level + 1) for level in levels]
    print("YES")
    print(" ".join(result))

solve()
