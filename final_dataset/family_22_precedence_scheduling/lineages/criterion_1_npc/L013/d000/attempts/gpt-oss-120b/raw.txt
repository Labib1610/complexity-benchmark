import sys
sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    prereq = [[] for _ in range(n)]
    indeg = [0]*n
    for _ in range(m):
        a = int(next(it))-1
        b = int(next(it))-1
        prereq[a].append(b)
        indeg[b] += 1

    # quick impossibility checks
    if n > M*T:
        print("NO")
        return

    # topological order for pruning
    from collections import deque
    q = deque([i for i in range(n) if indeg[i]==0])
    topo = []
    indeg2 = indeg[:]
    while q:
        u = q.popleft()
        topo.append(u)
        for v in prereq[u]:
            indeg2[v] -= 1
            if indeg2[v]==0:
                q.append(v)
    if len(topo)!=n:
        print("NO")
        return

    # adjacency reverse for checking ready
    rev = [[] for _ in range(n)]
    for u in range(n):
        for v in prereq[u]:
            rev[v].append(u)

    assign = [-1]*n
    term_cnt = [0]*(T+1)  # 1-indexed

    # order nodes by number of successors (heuristic)
    order = sorted(range(n), key=lambda x: -len(prereq[x]))

    # precompute for each node the set of predecessors
    pred_mask = [0]*n
    for i in range(n):
        mask = 0
        stack = rev[i][:]
        while stack:
            p = stack.pop()
            if not (mask>>p)&1:
                mask |= 1<<p
                stack.extend(rev[p])
        pred_mask[i] = mask

    full_mask = (1<<n)-1
    memo = {}

    def dfs(done_mask):
        if done_mask == full_mask:
            return True
        if done_mask in memo:
            return False
        # find ready tasks
        ready = []
        for v in order:
            if not (done_mask>>v)&1:
                if (pred_mask[v] & ~done_mask) == 0:
                    ready.append(v)
        # try to place some ready tasks in some term
        # generate combinations limited by M and term capacity
        # simple: try each ready task individually (branching may be large)
        for v in ready:
            for t in range(1, T+1):
                if term_cnt[t] < M:
                    # check that all predecessors are in earlier terms
                    ok = True
                    for p in rev[v]:
                        if assign[p] >= t:
                            ok = False
                            break
                    if not ok:
                        continue
                    assign[v] = t
                    term_cnt[t] += 1
                    if dfs(done_mask | (1<<v)):
                        return True
                    assign[v] = -1
                    term_cnt[t] -= 1
        memo[done_mask] = False
        return False

    if dfs(0):
        print("YES")
        print(' '.join(str(assign[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
