from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    M = int(data[idx+1])
    T = int(data[idx+2])
    m = int(data[idx+3])
    idx += 4
    
    # Build graph
    graph = defaultdict(list)
    indegree = [0] * (n + 1)
    
    for _ in range(m):
        a = int(data[idx])
        b = int(data[idx+1])
        graph[a].append(b)
        indegree[b] += 1
        idx += 2
    
    # Topological sort with level assignment
    queue = deque()
    term = [0] * (n + 1)
    
    # Find all nodes with indegree 0
    for i in range(1, n + 1):
        if indegree[i] == 0:
            queue.append(i)
    
    # Process nodes level by level
    current_term = 1
    count_in_term = 0
    next_queue = deque()
    
    while queue:
        # If we've exceeded the number of terms, it's impossible
        if current_term > T:
            print("NO")
            return
            
        count_in_term = 0
        next_queue = deque()
        
        # Process all nodes available for this term
        while queue and count_in_term < M:
            node = queue.popleft()
            term[node] = current_term
            count_in_term += 1
            
            # Update neighbors
            for neighbor in graph[node]:
                indegree[neighbor] -= 1
                if indegree[neighbor] == 0:
                    next_queue.append(neighbor)
        
        # If there are still nodes that could be processed but we've filled this term
        # We move the remaining nodes to the next term
        while queue:
            next_queue.append(queue.popleft())
            
        queue = next_queue
        current_term += 1
    
    # Check if all nodes were assigned
    for i in range(1, n + 1):
        if term[i] == 0:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, term[1:])))

if __name__ == "__main__":
    main()
