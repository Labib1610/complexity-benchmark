import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    rev_adj = [[] for _ in range(n)]
    
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        rev_adj[b].append(a)
        indeg[b] += 1
    
    # Compute longest path to each node (minimum term required)
    topo_order = []
    q = deque()
    for i in range(n):
        if indeg[i] == 0:
            q.append(i)
    
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    
    if len(topo_order) != n:
        print("NO")
        return
    
    # Compute minimum term for each module (longest path from any source)
    min_term = [1] * n
    for u in topo_order:
        for v in adj[u]:
            if min_term[v] < min_term[u] + 1:
                min_term[v] = min_term[u] + 1
    
    # Check if any module requires term > T
    for i in range(n):
        if min_term[i] > T:
            print("NO")
            return
    
    # Greedy assignment: assign modules to earliest possible term respecting capacity
    term = [0] * n
    capacity = [0] * (T + 2)  # 1-indexed
    
    # Process in topological order
    for u in topo_order:
        # Find earliest term >= min_term[u] with capacity < M
        t = min_term[u]
        while t <= T and capacity[t] >= M:
            t += 1
        if t > T:
            print("NO")
            return
        term[u] = t
        capacity[t] += 1
    
    print("YES")
    print(' '.join(str(x) for x in term))

if __name__ == "__main__":
    solve()
