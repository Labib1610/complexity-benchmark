```python
n = int(input())
nums = list(map(int, input().split()))
T = int(input())

if T == 0:
    print("YES")
    print()
    exit()

def generate_subsets(arr, original_start):
    n_sub = len(arr)
    subsets = []
    for mask in range(0, 1 << n_sub):
        s = 0
        indices = []
        for i in range(n_sub):
            if mask & (1 << i):
                s += arr[i]
                indices.append(original_start + i + 1)
        subsets.append((s, indices))
    return subsets

mid = n // 2
left_part = nums[:mid]
right_part = nums[mid:]

left_subsets = generate_subsets(left_part, 0)
right_subsets = generate_subsets(right_part, mid)

right_dict = {}
for s, indices in right_subsets:
    if s not in right_dict:
        right_dict[s] = indices

for s_l, indices_l in left_subsets:
    remaining = T - s_l
    if remaining in right_dict:
        indices_r = right_dict[remaining]
        total_indices = indices_l + indices_r
        print("YES")
        if not total_indices:
            print()
        else:
            print(' '.join(map(str, total_indices)))
        exit()

print("NO")
```
