import sys

def main():
    n, k, C = map(int, sys.stdin.readline().split())
    weights = list(map(int, sys.stdin.readline().split()))
    sum_weights = sum(weights)
    if sum_weights > k * C:
        print("NO")
        return
    if k >= n:
        print("YES")
        print(' '.join(map(str, range(1, n + 1))))
        return

    sorted_items = sorted([(weights[i], i) for i in range(n)], key=lambda x: (-x[0], x[1]))
    current_loads = [0] * k
    assignment = [0] * n

    def can_assign(index):
        if index == n:
            return True
        weight, original_index = sorted_items[index]
        for i in range(k):
            if current_loads[i] + weight > C:
                continue
            duplicate = False
            for j in range(i):
                if current_loads[j] == current_loads[i]:
                    duplicate = True
                    break
            if duplicate:
                continue
            current_loads[i] += weight
            assignment[original_index] = i + 1
            if can_assign(index + 1):
                return True
            current_loads[i] -= weight
        return False

    if can_assign(0):
        print("YES")
        print(' '.join(str(assignment[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    main()
