def main():
    import sys
    input = sys.stdin.read().split()
    n = int(input[0])
    d = list(map(int, input[1:n+1]))
    total_sum = sum(d)
    if total_sum % 2 != 0:
        print("NO")
        return
    target = total_sum // 2
    mid = n // 2
    first_part = d[:mid]
    second_part = d[mid:]
    
    # Generate all subsets for the first part
    m = len(first_part)
    first_subsets = []
    for mask in range(1 << m):
        current_sum = 0
        indices = []
        for i in range(m):
            if mask & (1 << i):
                current_sum += first_part[i]
                indices.append(i + 1)  # 1-based index
        first_subsets.append((current_sum, indices))
    
    # Generate subset sums for the second part and store in a dictionary
    k = len(second_part)
    second_dict = {}
    for mask in range(1 << k):
        current_sum = 0
        indices = []
        for j in range(k):
            if mask & (1 << j):
                current_sum += second_part[j]
                original_index = mid + j + 1  # 1-based index
                indices.append(original_index)
        if current_sum not in second_dict:
            second_dict[current_sum] = indices
    
    # Check for a valid combination
    for s1, indices1 in first_subsets:
        required = target - s1
        if required in second_dict:
            indices2 = second_dict[required]
            total_indices = indices1 + indices2
            print("YES")
            print(' '.join(map(str, sorted(total_indices))))
            return
    print("NO")

if __name__ == "__main__":
    main()
