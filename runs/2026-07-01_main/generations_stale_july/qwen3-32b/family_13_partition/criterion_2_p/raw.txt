```python
n = int(input())
d = list(map(int, input().split()))
total = sum(d)
if total % 2 != 0:
    print("NO")
else:
    target = total // 2
    dp = [False] * (target + 1)
    dp[0] = True
    for num in d:
        for s in range(target, num - 1, -1):
            if dp[s - num]:
                dp[s] = True
    if not dp[target]:
        print("NO")
    else:
        current_target = target
        indices = []
        for i in reversed(range(n)):
            if current_target >= d[i] and dp[current_target - d[i]]:
                indices.append(i + 1)
                current_target -= d[i]
        print("YES")
        print(' '.join(map(str, indices)))
```
