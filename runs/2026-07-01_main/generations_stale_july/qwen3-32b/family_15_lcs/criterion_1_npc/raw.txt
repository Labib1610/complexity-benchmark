```python
k = int(input())
strings = [input().strip() for _ in range(k)]

def is_subsequence(cand, s):
    if not cand:
        return True
    i = 0
    for char in s:
        if i < len(cand) and cand[i] == char:
            i += 1
    return i == len(cand)

s_ref = min(strings, key=lambda x: len(x))
n = len(s_ref)
candidates = set()

for mask in range(1 << n):
    cand = []
    for i in range(n):
        if mask & (1 << i):
            cand.append(s_ref[i])
    candidates.add(''.join(cand))

max_len = 0
best_cand = ''

for cand in candidates:
    all_valid = True
    for s in strings:
        if not is_subsequence(cand, s):
            all_valid = False
            break
    if all_valid:
        if len(cand) > max_len:
            max_len = len(cand)
            best_cand = cand
        elif len(cand) == max_len and max_len > 0:
            pass  # Keep the first found

print(max_len)
print(best_cand)
```
