```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    C = int(data[1])
    weights = list(map(int, data[2:2+n]))
    for i, w in enumerate(weights, start=1):
        if w > C:
            print(i)
            return

if __name__ == "__main__":
    solve()
```
