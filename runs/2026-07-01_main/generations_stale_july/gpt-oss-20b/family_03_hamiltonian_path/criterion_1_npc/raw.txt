```python
import sys
import threading

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    for lst in adj:
        lst.sort()
    full_mask = (1 << n) - 1

    from functools import lru_cache

    @lru_cache(maxsize=None)
    def can(mask: int, last: int) -> bool:
        if mask == full_mask:
            return True
        for u in adj[last]:
            if not (mask >> u) & 1:
                if can(mask | (1 << u), u):
                    return True
        return False

    for start in range(n):
        if can(1 << start, start):
            path = [start]
            mask = 1 << start
            last = start
            while mask != full_mask:
                for u in adj[last]:
                    if not (mask >> u) & 1:
                        if can(mask | (1 << u), u):
                            path.append(u)
                            mask |= 1 << u
                            last = u
                            break
            print("YES")
            print(' '.join(str(x + 1) for x in path))
            return
    print("NO")

def main():
    solve()

threading.Thread(target=main).start()
```
