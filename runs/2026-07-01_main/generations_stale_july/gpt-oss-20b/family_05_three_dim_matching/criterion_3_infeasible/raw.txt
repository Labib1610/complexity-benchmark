```python
import sys

def solve() -> None:
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    e = int(next(it))
    has_task = [False] * (k + 1)
    for _ in range(e):
        x = int(next(it))
        # y = int(next(it))  # task value not needed
        next(it)
        has_task[x] = True
    for worker in range(1, k + 1):
        if not has_task[worker]:
            print(worker)
            return

if __name__ == "__main__":
    solve()
```
