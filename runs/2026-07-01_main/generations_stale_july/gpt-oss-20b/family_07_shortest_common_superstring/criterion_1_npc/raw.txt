import sys

def solve() -> None:
    data = sys.stdin.read().splitlines()
    if not data:
        return
    k = int(data[0].strip())
    fragments = [line.strip() for line in data[1:k+1]]

    # Remove duplicates
    fragments = list(dict.fromkeys(fragments))

    # Remove strings that are substrings of others
    n = len(fragments)
    removed = [False] * n
    for i in range(n):
        if removed[i]:
            continue
        for j in range(n):
            if i != j and not removed[j]:
                if fragments[i] in fragments[j]:
                    removed[i] = True
                    break
    fragments = [fragments[i] for i in range(n) if not removed[i]]
    n = len(fragments)

    if n == 0:
        print("")
        return
    if n == 1:
        print(fragments[0])
        return

    # Precompute overlaps
    overlap = [[0]*n for _ in range(n)]
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            a, b = fragments[i], fragments[j]
            max_ol = min(len(a), len(b))
            for l in range(max_ol, 0, -1):
                if a[-l:] == b[:l]:
                    overlap[i][j] = l
                    break

    full_mask = (1 << n) - 1
    dp = [[None]*n for _ in range(1 << n)]

    for i in range(n):
        dp[1 << i][i] = fragments[i]

    for mask in range(1 << n):
        for last in range(n):
            cur = dp[mask][last]
            if cur is None:
                continue
            for nxt in range(n):
                if mask & (1 << nxt):
                    continue
                newmask = mask | (1 << nxt)
                cand = cur + fragments[nxt][overlap[last][nxt]:]
                prev = dp[newmask][nxt]
                if prev is None or len(cand) < len(prev) or (len(cand) == len(prev) and cand < prev):
                    dp[newmask][nxt] = cand

    best = None
    for i in range(n):
        s = dp[full_mask][i]
        if s is None:
            continue
        if best is None or len(s) < len(best) or (len(s) == len(best) and s < best):
            best = s

    print(best)

if __name__ == "__main__":
    solve()
