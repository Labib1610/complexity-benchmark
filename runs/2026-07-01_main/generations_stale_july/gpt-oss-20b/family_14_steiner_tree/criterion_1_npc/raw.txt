import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    required = [int(next(it)) - 1 for _ in range(t)]  # zero-indexed

    INF = 10**18
    dist = [[INF]*n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w

    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            ik = di[k]
            if ik == INF:
                continue
            for j in range(n):
                nj = ik + dk[j]
                if nj < di[j]:
                    di[j] = nj

    full_mask = (1 << t) - 1
    dp = [[INF]*n for _ in range(1 << t)]
    for i, node in enumerate(required):
        dp[1 << i][node] = 0

    for mask in range(1, full_mask + 1):
        # combine submasks
        sub = mask
        while sub:
            sub = (sub - 1) & mask
            other = mask ^ sub
            if other == 0:
                continue
            for v in range(n):
                val = dp[sub][v] + dp[other][v]
                if val < dp[mask][v]:
                    dp[mask][v] = val
        # relax via distances
        for v in range(n):
            dv = dp[mask][v]
            if dv == INF:
                continue
            for u in range(n):
                nd = dv + dist[v][u]
                if nd < dp[mask][u]:
                    dp[mask][u] = nd

    ans = min(dp[full_mask])
    print(ans)

if __name__ == "__main__":
    solve()
