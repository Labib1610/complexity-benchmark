import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        clauses.append([l1, l2, l3])

    assign = [-1] * n
    trail = []

    def propagate() -> bool:
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                satisfied = False
                unassigned_lit = None
                unassigned_count = 0
                for lit in clause:
                    var = abs(lit) - 1
                    val = assign[var]
                    if val == -1:
                        unassigned_count += 1
                        unassigned_lit = lit
                    else:
                        if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                            satisfied = True
                            break
                if satisfied:
                    continue
                if unassigned_count == 0:
                    return False
                if unassigned_count == 1:
                    lit = unassigned_lit
                    var = abs(lit) - 1
                    val_needed = 1 if lit > 0 else 0
                    if assign[var] == -1:
                        assign[var] = val_needed
                        trail.append(var)
                        changed = True
                    else:
                        if assign[var] != val_needed:
                            return False
        return True

    def dpll() -> bool:
        if not propagate():
            return False
        # find first unassigned variable
        try:
            var = assign.index(-1)
        except ValueError:
            return True  # all assigned
        for val in (0, 1):
            trail_len = len(trail)
            assign[var] = val
            trail.append(var)
            if dpll():
                return True
            # revert assignments made after this level
            while len(trail) > trail_len:
                v = trail.pop()
                assign[v] = -1
        return False

    if dpll():
        print("YES")
        print(' '.join(str(v) for v in assign))
    else:
        print("NO")

if __name__ == "__main__":
    main()
