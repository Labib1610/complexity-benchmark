import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    d = list(map(int, data[1:1+n]))
    total = sum(d)
    if total % 2:
        print("NO")
        return
    target = total // 2

    # pref[i] : bitset of sums achievable using first i items (0..i-1)
    pref = [0] * (n + 1)
    pref[0] = 1  # only sum 0 reachable
    for i in range(n):
        pref[i+1] = pref[i] | (pref[i] << d[i])

    if ((pref[n] >> target) & 1) == 0:
        print("NO")
        return

    # reconstruct
    res = []
    cur = target
    for i in range(n-1, -1, -1):
        if cur >= d[i] and ((pref[i] >> (cur - d[i])) & 1):
            res.append(i + 1)  # 1-based index
            cur -= d[i]
        if cur == 0:
            break

    print("YES")
    if res:
        print(' '.join(map(str, res)))
    else:
        print()

if __name__ == "__main__":
    solve()
