```python
import sys
import threading

def solve():
    import sys

    sys.setrecursionlimit(10000)
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    m = int(next(it))
    job_ops = []
    total_job_time = [0]*j
    total_machine_time = [0]*m
    for ji in range(j):
        o = int(next(it))
        ops = []
        for _ in range(o):
            machine = int(next(it)) - 1
            duration = int(next(it))
            ops.append((machine, duration))
            total_job_time[ji] += duration
            total_machine_time[machine] += duration
        job_ops.append(ops)

    job_next = [0]*j
    job_available = [0]*j
    machine_available = [0]*m
    rem_job_time = total_job_time[:]
    rem_machine_time = total_machine_time[:]
    best = [10**18]

    # Memoization dictionary: key -> best makespan seen
    memo = {}

    def dfs():
        # Check if all jobs finished
        if all(job_next[ji] == len(job_ops[ji]) for ji in range(j)):
            makespan = max(job_available)
            if makespan < best[0]:
                best[0] = makespan
            return

        # Lower bound calculation
        current_makespan = max(job_available)
        lb = current_makespan
        for ji in range(j):
            lb = max(lb, job_available[ji] + rem_job_time[ji])
        for mi in range(m):
            lb = max(lb, machine_available[mi] + rem_machine_time[mi])
        if lb >= best[0]:
            return

        # Memoization key
        key = (tuple(job_next), tuple(machine_available))
        if key in memo and memo[key] <= current_makespan:
            return
        memo[key] = current_makespan

        # Prepare list of ready jobs
        ready_jobs = [ji for ji in range(j) if job_next[ji] < len(job_ops[ji])]
        # Sort by heuristic: larger remaining job time first
        ready_jobs.sort(key=lambda ji: job_available[ji] + rem_job_time[ji], reverse=True)

        for ji in ready_jobs:
            machine, duration = job_ops[ji][job_next[ji]]
            start = job_available[ji]
            if machine_available[machine] > start:
                start = machine_available[machine]
            finish = start + duration

            # Save old values
            old_job_avail = job_available[ji]
            old_machine_avail = machine_available[machine]
            old_rem_job = rem_job_time[ji]
            old_rem_machine = rem_machine_time[machine]

            # Update
            job_available[ji] = finish
            machine_available[machine] = finish
            rem_job_time[ji] -= duration
            rem_machine_time[machine] -= duration
            job_next[ji] += 1

            dfs()

            # Backtrack
            job_next[ji] -= 1
            rem_job_time[ji] = old_rem_job
            rem_machine_time[machine] = old_rem_machine
            job_available[ji] = old_job_avail
            machine_available[machine] = old_machine_avail

    dfs()
    print(best[0])

def main():
    solve()

if __name__ == "__main__":
    threading.Thread(target=main).start()
```
