import sys
from collections import deque

def hopcroft_karp(adj, n_left, n_right):
    pair_u = [0] * (n_left + 1)
    pair_v = [0] * (n_right + 1)
    dist = [0] * (n_left + 1)

    INF = 10**9

    def bfs():
        q = deque()
        for u in range(1, n_left + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        d = INF
        while q:
            u = q.popleft()
            if dist[u] < d:
                for v in adj[u]:
                    if pair_v[v] == 0:
                        d = dist[u] + 1
                    elif dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        q.append(pair_v[v])
        return d != INF

    def dfs(u):
        for v in adj[u]:
            if pair_v[v] == 0 or (dist[pair_v[v]] == dist[u] + 1 and dfs(pair_v[v])):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(1, n_left + 1):
            if pair_u[u] == 0 and dfs(u):
                matching += 1
    return matching

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    nA = int(next(it))
    nB = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(nA + 1)]
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
    max_match = hopcroft_karp(adj, nA, nB)
    result = nA + nB - max_match
    sys.stdout.write(str(result))

if __name__ == "__main__":
    solve()
