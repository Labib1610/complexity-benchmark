```python
import sys

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    u, s = data[0], data[1]
    seen = [False] * (u + 1)
    idx = 2
    for _ in range(s):
        c = data[idx]
        idx += 1
        for _ in range(c):
            skill = data[idx]
            idx += 1
            if 1 <= skill <= u:
                seen[skill] = True
    for skill_id in range(1, u + 1):
        if not seen[skill_id]:
            print(skill_id)
            return

if __name__ == "__main__":
    solve()
```
