import sys
from collections import deque

def hopcroft_karp(adj, n_left, n_right):
    pairU = [0] * (n_left + 1)
    pairV = [0] * (n_right + 1)
    dist = [0] * (n_left + 1)
    INF = 10**9

    def bfs():
        q = deque()
        for u in range(1, n_left + 1):
            if pairU[u] == 0:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        found = False
        while q:
            u = q.popleft()
            for v in adj[u]:
                pu = pairV[v]
                if pu == 0:
                    found = True
                elif dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    q.append(pu)
        return found

    def dfs(u):
        for v in adj[u]:
            pu = pairV[v]
            if pu == 0 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pairU[u] = v
                pairV[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(1, n_left + 1):
            if pairU[u] == 0 and dfs(u):
                matching += 1
    return matching

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    nA = int(next(it))
    nB = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(nA + 1)]
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
    result = hopcroft_karp(adj, nA, nB)
    print(result)

if __name__ == "__main__":
    main()
