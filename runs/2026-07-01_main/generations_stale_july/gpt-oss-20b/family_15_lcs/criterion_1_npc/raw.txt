import sys

def is_subseq(sub, t):
    it = iter(t)
    return all(c in it for c in sub)

def main():
    data = sys.stdin.read().splitlines()
    if not data:
        return
    k = int(data[0].strip())
    strs = [line.strip() for line in data[1:1+k]]
    # Find shortest string to generate subsequences
    idx = min(range(k), key=lambda i: len(strs[i]))
    s = strs[idx]
    n = len(s)
    # Precompute buckets of masks by popcount
    buckets = [[] for _ in range(n+1)]
    for mask in range(1, 1 << n):
        l = mask.bit_count()
        buckets[l].append(mask)
    best_sub = ""
    best_len = 0
    # Iterate lengths from n down to 1
    for l in range(n, 0, -1):
        for mask in buckets[l]:
            # Build subsequence string
            sub_chars = []
            m = mask
            pos = 0
            while m:
                if m & 1:
                    sub_chars.append(s[pos])
                pos += 1
                m >>= 1
            sub = ''.join(sub_chars)
            # Check against all other strings
            ok = True
            for j in range(k):
                if j == idx:
                    continue
                if not is_subseq(sub, strs[j]):
                    ok = False
                    break
            if ok:
                best_sub = sub
                best_len = l
                print(best_len)
                print(best_sub)
                return
    # If no subsequence found
    print(0)
    print()

if __name__ == "__main__":
    main()
