import sys

def main():
    data = sys.stdin.read().splitlines()
    if not data:
        return
    s = data[0].strip()
    t = data[1].strip() if len(data) > 1 else ''
    n, m = len(s), len(t)

    # DP for LCS length
    dp = [[0] * (m + 1) for _ in range(n + 1)]
    for i in range(n - 1, -1, -1):
        si = s[i]
        dpi = dp[i]
        dpi1 = dp[i + 1]
        for j in range(m - 1, -1, -1):
            if si == t[j]:
                dpi[j] = 1 + dp[i + 1][j + 1]
            else:
                dpi[j] = dpi1[j] if dpi1[j] >= dpi[j + 1] else dpi[j + 1]

    lcs_len = dp[0][0]

    # Precompute next occurrence arrays
    next_s = [[-1] * 26 for _ in range(n + 1)]
    next_t = [[-1] * 26 for _ in range(m + 1)]
    for c in range(26):
        next_s[n][c] = -1
        next_t[m][c] = -1
    for i in range(n - 1, -1, -1):
        next_s[i] = next_s[i + 1][:]
        next_s[i][ord(s[i]) - 97] = i
    for j in range(m - 1, -1, -1):
        next_t[j] = next_t[j + 1][:]
        next_t[j][ord(t[j]) - 97] = j

    # Reconstruct lexicographically smallest LCS
    i = j = 0
    res = []
    cur_len = lcs_len
    while cur_len > 0:
        for c in range(26):
            i1 = next_s[i][c]
            j1 = next_t[j][c]
            if i1 != -1 and j1 != -1 and dp[i1 + 1][j1 + 1] == cur_len - 1:
                res.append(chr(c + 97))
                i = i1 + 1
                j = j1 + 1
                cur_len -= 1
                break

    print(lcs_len)
    print(''.join(res))

if __name__ == "__main__":
    main()
