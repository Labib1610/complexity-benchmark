import sys
from collections import deque

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = next(it)
        v = next(it)
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)  # 0 or 1
    res = [0] * (n + 1)

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS to color component
        q = deque([start])
        color[start] = 0
        comp_nodes = [start]
        min_idx = start
        bipartite = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                    comp_nodes.append(v)
                    if v < min_idx:
                        min_idx = v
                elif color[v] == color[u]:
                    bipartite = False
                    break
            if not bipartite:
                break
        if not bipartite:
            print("NO")
            return
        # Decide orientation
        if color[min_idx] == 0:
            # 0 -> 1, 1 -> 2
            for node in comp_nodes:
                res[node] = 1 if color[node] == 0 else 2
        else:
            # 1 -> 1, 0 -> 2
            for node in comp_nodes:
                res[node] = 1 if color[node] == 1 else 2

    print("YES")
    print(' '.join(str(res[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
