```python
def solve():
    n, m = map(int, input().split())
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        edges.append((u, v, w))
    
    edges.sort(key=lambda x: x[2])
    
    parent = list(range(n + 1))
    
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    total_cost = 0
    num_edges = 0
    
    for u, v, w in edges:
        if union(u, v):
            total_cost += w
            num_edges += 1
        if num_edges == n - 1:
            break
            
    print(total_cost)

solve()
```
