def solve():
    n, m, t = map(int, input().split())
    required_sites = list(map(int, input().split()))
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        edges.append((u, v, w))

    parent = list(range(n + 1))

    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]

    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False

    required_sites_set = set(required_sites)
    
    relevant_edges = []
    for u, v, w in edges:
        if u in required_sites_set or v in required_sites_set:
            relevant_edges.append((u, v, w))
    
    
    relevant_edges.sort(key=lambda x: x[2])
    
    cost = 0
    num_edges = 0
    
    for u, v, w in relevant_edges:
        if union(u, v):
            cost += w
            num_edges += 1
