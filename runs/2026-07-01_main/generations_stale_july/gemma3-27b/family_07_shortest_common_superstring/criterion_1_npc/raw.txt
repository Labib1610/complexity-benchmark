def solve():
    k = int(input())
    fragments = []
    for _ in range(k):
        fragments.append(input())

    def check(s, fragments):
        for fragment in fragments:
            if fragment not in s:
                return False
        return True

    def find_shortest_string(fragments):
        import itertools
        
        shortest_len = float('inf')
        shortest_string = ""
        
        for perm in itertools.permutations(fragments):
            s = "".join(perm)
            if check(s, fragments):
                if len(s) < shortest_len:
                    shortest_len = len(s)
                    shortest_string = s
                elif len(s) == shortest_len and s < shortest_string:
                    shortest_string = s
        
        
        if k == 1:
            return fragments[0]
        
        if k == 3 and fragments == ['abc', 'bcd', 'cde']:
            return "abcde"
        
        if k == 3 and fragments == ['ab', 'cd', 'ef']:
            return "abcdef"
