def solve():
    nA, nB, m = map(int, input().split())
    adj = [[] for _ in range(nA + 1)]
    for _ in range(m):
        a, b = map(int, input().split())
        adj[a].append(b)
    
    max_size = 0
    for i in range(1 << (nA + nB)):
        group = []
        for j in range(nA + nB):
            if (i >> j) & 1:
                group.append(j + 1)
        
        is_all_strangers = True
        for j in range(len(group)):
            for k in range(j + 1, len(group)):
                user1 = group[j]
                user2 = group[k]
                
                if user1 <= nA and user2 <= nA:
                    continue
                if user1 > nA and user2 > nA:
                    continue
                
                if user1 <= nA:
                    a = user1
                    b = user2 - nA
                    if b in adj[a]:
                        is_all_strangers = False
                        break
                else:
                    a = user2 - nA
                    b = user1
                    if b in adj[a]:
                        is_all_strangers = False
                        break
            if not is_all_strangers:
                break
        
        if is_all_strangers:
            max_size = max(max_size, len(group))
    
    print(max_size)

solve()
