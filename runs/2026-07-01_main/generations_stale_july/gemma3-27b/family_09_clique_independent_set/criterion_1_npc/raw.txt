def solve():
    n, m, k = map(int, input().split())
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
    
    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    def is_tight_knit(group):
        for i in range(len(group)):
            for j in range(i + 1, len(group)):
                if group[j] not in adj[group[i]]:
                    return False
        return True
    
    import itertools
    
    for group in itertools.combinations(range(1, n + 1), k):
        if is_tight_knit(group):
            print("YES")
            print(*group)
            return
    
    print("NO")

solve()
