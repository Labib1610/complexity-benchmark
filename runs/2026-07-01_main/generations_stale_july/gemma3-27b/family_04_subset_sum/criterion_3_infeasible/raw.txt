```python
def solve():
    n = int(input())
    costs = list(map(int, input().split()))
    T = int(input())
    
    if T >= 0:
        print("NO")
        return
    
    dp = {}
    
    def find_subset(index, current_sum):
        if index == n:
            if current_sum == T:
                return []
            else:
                return None
        
        if (index, current_sum) in dp:
            return dp[(index, current_sum)]
        
        # Option 1: Don't include the current item
        result1 = find_subset(index + 1, current_sum)
        
        # Option 2: Include the current item
        result2 = find_subset(index + 1, current_sum + costs[index])
        if result2 is not None:
            result2 = [index + 1] + result2
        
        if result1 is not None:
            dp[(index, current_sum)] = result1
            return result1
        elif result2 is not None:
            dp[(index, current_sum)] = result2
            return result2
        else:
            dp[(index, current_sum)] = None
            return None
    
    result = find_subset(0, 0)
    
    if result is None:
        print("NO")
    else:
        print("YES")
        print(*result)

solve()
```
