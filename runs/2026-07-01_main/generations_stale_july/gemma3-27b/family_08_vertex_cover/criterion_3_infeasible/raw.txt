def solve():
    n, m, k = map(int, input().split())
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
    
    independent_edges = []
    used_nodes = set()
    
    for u, v in edges:
        if u not in used_nodes and v not in used_nodes:
            independent_edges.append((u, v))
            used_nodes.add(u)
            used_nodes.add(v)
            if len(independent_edges) == k + 1:
                break
    
    if len(independent_edges) < k + 1:
        
        remaining_edges = []
        for u, v in edges:
            if (u,v) not in independent_edges and (v,u) not in independent_edges:
                remaining_edges.append((u,v))
