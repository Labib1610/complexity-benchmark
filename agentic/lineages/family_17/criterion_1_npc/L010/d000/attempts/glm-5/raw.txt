import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    n = 3 * q
    adj = [[] for _ in range(n + 1)]
    for _ in range(t):
        a = int(next(it)); b = int(next(it)); c = int(next(it))
        adj[a].append((b, c))
        adj[b].append((a, c))
        adj[c].append((a, b))
    for i in range(1, n + 1):
        adj[i].sort()
    match = [0] * (n + 1)
    used = [False] * (n + 1)
    for i in range(1, n + 1):
        if match[i] == 0:
            found = False
            for b, c in adj[i]:
                if match[b] == 0 and match[c] == 0:
                    match[i] = b
                    match[b] = i
                    match[c] = i
                    used[i] = True
                    found = True
                    break
            if not found:
                print("NO")
                return
    for i in range(1, n + 1):
        if match[i] == 0:
            print("NO")
            return
    print("YES")
    seen = [False] * (n + 1)
    for i in range(1, n + 1):
        if not seen[i]:
            j = match[i]
            k = match[j]
            if k == i:
                k = j
            a, b, c = i, j, k
            if a > b:
                a, b = b, a
            if b > c:
                b, c = c, b
            if a > b:
                a, b = b, a
            print(a, b, c)
            seen[a] = seen[b] = seen[c] = True

solve()
