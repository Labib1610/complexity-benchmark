We can model this as an exact cover problem.  
Since the number of people is at most 600 (3q ≤ 600) and t ≤ 20000, we can use backtracking with pruning.

Here’s the Python solution:  

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    q = int(input_data[0])
    t = int(input_data[1])
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triples.append((a, b, c))
    
    n = 3 * q
    # adjacency: for each person, list of triple indices
    person_to_triples = [[] for _ in range(n + 1)]
    for i, (a, b, c) in enumerate(triples):
        person_to_triples[a].append(i)
        person_to_triples[b].append(i)
        person_to_triples[c].append(i)
    
    used = [False] * t
    assigned = [False] * (n + 1)
    chosen = []
    
    # Pre-sort triples by smallest person for better pruning
    triples_sorted = sorted(range(t), key=lambda i: (triples[i][0], triples[i][1], triples[i][2]))
    
    def backtrack():
        if len(chosen) == q:
            return True
        # find first unassigned person
        first = None
        for p in range(1, n + 1):
            if not assigned[p]:
                first = p
                break
        if first is None:
            return False
        
        # try triples containing this person
        for idx in person_to_triples[first]:
            if used[idx]:
                continue
            a, b, c = triples[idx]
            if assigned[a] or assigned[b] or assigned[c]:
                continue
            # assign
            used[idx] = True
            assigned[a] = assigned[b] = assigned[c] = True
            chosen.append((a, b, c))
            if backtrack():
                return True
            # backtrack
            chosen.pop()
            assigned[a] = assigned[b] = assigned[c] = False
            used[idx] = False
        return False
    
    if backtrack():
        print("YES")
        for a, b, c in chosen:
            print(a, b, c)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
