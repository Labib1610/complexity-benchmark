from collections import defaultdict
import sys

def solve():
    q, t = map(int, input().split())
    
    triples = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        triples.append((a, b, c))
    
    # If q == 0, we need empty solution
    if q == 0:
        print("YES")
        return
    
    # If t == 0 but q > 0, impossible
    if t == 0:
        print("NO")
        return
    
    # Try all combinations of q triples from t available
    # Since q <= 200 and t <= 20000, brute force won't work
    # We need a more intelligent approach - this is essentially an exact cover problem
    
    # Use backtracking with pruning
    def backtrack(selected, used_operators, index):
        # If we've selected q triples and covered all 3*q operators
        if len(selected) == q:
            if len(used_operators) == 3 * q:
                return selected[:]
            else:
                return None
        
        # Pruning: if even selecting all remaining triples won't give us enough operators
        remaining_triples = t - index
        needed_more_triples = q - len(selected)
        if remaining_triples < needed_more_triples:
            return None
            
        # Try each triple starting from index
        for i in range(index, t):
            a, b, c = triples[i]
            
            # If any operator in this triple is already used, skip
            if a in used_operators or b in used_operators or c in used_operators:
                continue
            
            # Add this triple
            selected.append(i)
            used_operators.add(a)
            used_operators.add(b)
            used_operators.add(c)
            
            # Recurse
            result = backtrack(selected, used_operators, i + 1)
            if result is not None:
                return result
            
            # Backtrack
            selected.pop()
            used_operators.remove(a)
            used_operators.remove(b)
            used_operators.remove(c)
        
        return None
    
    # Optimization: precompute which triples contain each operator
    operator_to_triples = defaultdict(list)
    for i, (a, b, c) in enumerate(triples):
        operator_to_triples[a].append(i)
        operator_to_triples[b].append(i)
        operator_to_triples[c].append(i)
    
    # Sort triples by the minimum operator they contain to help with pruning
    # This might help find solutions faster
    triple_indices = list(range(t))
    triple_indices.sort(key=lambda i: min(triples[i]))
    
    # Reorder triples based on sorted indices
    sorted_triples = [triples[i] for i in triple_indices]
    triples = sorted_triples
    
    result = backtrack([], set(), 0)
    
    if result is None:
        print("NO")
    else:
        print("YES")
        for idx in result:
            a, b, c = triples[idx]
            print(a, b, c)

solve()
