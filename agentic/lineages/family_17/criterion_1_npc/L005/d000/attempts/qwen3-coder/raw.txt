from collections import defaultdict
import sys

def find_maximum_matching(graph, n):
    """
    Find maximum matching in a bipartite graph using Hopcroft-Karp algorithm approach,
    but implemented here with a simpler DFS-based approach for clarity.
    """
    match = [-1] * (n + 1)
    
    def dfs(u, visited):
        for v in graph[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            if match[v] == -1 or dfs(match[v], visited):
                match[v] = u
                return True
        return False
    
    matching = 0
    for u in range(1, n + 1):
        visited = [False] * (n + 1)
        if dfs(u, visited):
            matching += 1
    
    return matching, match

def solve():
    q, t = map(int, input().split())
    
    # Read all schematics
    schematics = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        schematics.append((a, b, c))
    
    # If we need more schematics than available, impossible
    if t < q:
        print("NO")
        return
    
    # Try all combinations of q schematics
    # This is a hitting set problem - we need to select q triplets such that
    # all 3*q elements are covered exactly once
    
    # We can model this as an exact cover problem and solve with backtracking
    def backtrack(selected, used_elements, index):
        # If we've selected q schematics and used all 3*q elements
        if len(selected) == q:
            if len(used_elements) == 3 * q:
                return selected[:]
            else:
                return None
        
        # If we can't possibly select enough schematics
        if index >= t:
            return None
            
        # Pruning: if even selecting all remaining schematics won't give us enough
        if len(selected) + (t - index) < q:
            return None
            
        # Try including current schematic
        a, b, c = schematics[index]
        if a not in used_elements and b not in used_elements and c not in used_elements:
            selected.append(index)
            used_elements.add(a)
            used_elements.add(b)
            used_elements.add(c)
            
            result = backtrack(selected, used_elements, index + 1)
            if result is not None:
                return result
                
            selected.pop()
            used_elements.remove(a)
            used_elements.remove(b)
            used_elements.remove(c)
        
        # Try not including current schematic
        return backtrack(selected, used_elements, index + 1)
    
    result = backtrack([], set(), 0)
    
    if result is None:
        print("NO")
    else:
        print("YES")
        for idx in result:
            a, b, c = schematics[idx]
            print(a, b, c)

solve()
