from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    records = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        records.append((a, b, c))
    
    if q == 0:
        print("YES")
        return
    
    # Build graph for matching
    # Left side: people (1 to 3*q)
    # Right side: records (0 to t-1)
    adj = defaultdict(list)
    
    for i, (a, b, c) in enumerate(records):
        adj[a].append(i)
        adj[b].append(i)
        adj[c].append(i)
    
    # Try to find a perfect matching using backtracking with pruning
    # We need exactly q records, each person appears in exactly one record
    
    def backtrack(record_idx, used_records, covered_people):
        if len(used_records) == q:
            return used_records[:]
        
        if record_idx >= t:
            return None
        
        # Try including current record
        a, b, c = records[record_idx]
        if a not in covered_people and b not in covered_people and c not in covered_people:
            used_records.append(record_idx)
            covered_people.add(a)
            covered_people.add(b)
            covered_people.add(c)
            
            result = backtrack(record_idx + 1, used_records, covered_people)
            if result is not None:
                return result
            
            used_records.pop()
            covered_people.remove(a)
            covered_people.remove(b)
            covered_people.remove(c)
        
        # Try not including current record
        return backtrack(record_idx + 1, used_records, covered_people)
    
    # Optimization: if any person appears in 0 records, impossible
    person_count = defaultdict(int)
    for a, b, c in records:
        person_count[a] += 1
        person_count[b] += 1
        person_count[c] += 1
    
    for person in range(1, 3*q + 1):
        if person not in person_count:
            print("NO")
            return
    
    result = backtrack(0, [], set())
    
    if result is not None:
        print("YES")
        for idx in result:
            a, b, c = records[idx]
            print(a, b, c)
    else:
        print("NO")

solve()
