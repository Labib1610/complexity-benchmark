from collections import defaultdict
import sys

def find_matching(q, t, cases):
    # Create a graph for matching
    # Each artifact is on the left side, each case is on the right side
    # We want to find a perfect matching that covers all artifacts
    
    if t == 0:
        if q == 0:
            return True, []
        else:
            return False, []
    
    # Build adjacency list for artifacts to cases
    adj = defaultdict(list)
    case_list = []
    
    for i, (a, b, c) in enumerate(cases):
        case_list.append((a, b, c))
        adj[a].append(i)
        adj[b].append(i)
        adj[c].append(i)
    
    # Try to find a maximum matching using backtracking with pruning
    matched_cases = [False] * len(cases)
    artifact_match = [None] * (3*q + 1)  # artifact -> case index
    result = []
    
    def dfs(artifact):
        if artifact > 3*q:
            return True
            
        if artifact_match[artifact] is not None:
            return dfs(artifact + 1)
            
        # Try each case that contains this artifact
        for case_idx in adj[artifact]:
            if matched_cases[case_idx]:
                continue
                
            a, b, c = case_list[case_idx]
            
            # Check if all artifacts in this case are unmatched
            if (artifact_match[a] is None and 
                artifact_match[b] is None and 
                artifact_match[c] is None):
                
                # Match this case
                matched_cases[case_idx] = True
                artifact_match[a] = case_idx
                artifact_match[b] = case_idx
                artifact_match[c] = case_idx
                result.append((a, b, c))
                
                if dfs(artifact + 1):
                    return True
                    
                # Backtrack
                matched_cases[case_idx] = False
                artifact_match[a] = None
                artifact_match[b] = None
                artifact_match[c] = None
                result.pop()
        
        return False
    
    # Start DFS from artifact 1
    if dfs(1):
        return True, result
    else:
        return False, []

def solve():
    line = input().split()
    q, t = int(line[0]), int(line[1])
    
    cases = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        cases.append((a, b, c))
    
    possible, arrangement = find_matching(q, t, cases)
    
    if possible:
        print("YES")
        for a, b, c in arrangement:
            print(a, b, c)
    else:
        print("NO")

solve()
