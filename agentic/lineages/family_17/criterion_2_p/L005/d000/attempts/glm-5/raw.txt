def hopcroft_karp(q, adj):
    INF = float('inf')
    pairU = [-1] * q   # matchL
    pairV = [-1] * q   # matchR
    dist = [0] * q
    
    # Greedy initialization
    for u in range(q):
        for v in adj[u]:
            if pairV[v] == -1:
                pairV[v] = u
                pairU[u] = v
                break
    
    result = sum(1 for x in pairU if x != -1)
    if result == q:
        return pairU
    
    queue = deque()
    while True:
        # BFS
        found_free = False
        queue.clear()
        for u in range(q):
            if pairU[u] == -1:
                dist[u] = 0
                queue.append(u)

        while queue:
