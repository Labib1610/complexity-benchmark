import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    n = 2 * q

    head = list(range(n + 2))
    nxt = [0] * (n + 2)
    prv = [0] * (n + 2)
    for i in range(n + 1):
        nxt[i] = i + 1
        prv[i + 1] = i

    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx + 1]); idx += 2
        adj[a].append(b)
        adj[b].append(a)

    match = [0] * (n + 1)
    used = [False] * (n + 1)
    stack = []

    def remove_node(x):
        nxt[prv[x]] = nxt[x]
        prv[nxt[x]] = prv[x]

    def restore_node(x):
        nxt[prv[x]] = x
        prv[nxt[x]] = x

    def find_aug(x):
        stack.clear()
        stack.append(x)
        while stack:
            v = stack.pop()
            if v == 0:
                continue
            if used[v]:
                continue
            used[v] = True
            remove_node(v)
            w = head[0]
            while w <= n:
                found = False
                for u in adj[v]:
                    if u == w:
                        found = True
                        break
                if found:
                    w = nxt[w]
                else:
                    nxt_w = nxt[w]
                    m = match[w]
                    if m == 0:
                        match[v] = w
                        match[w] = v
                        return True
                    else:
                        remove_node(w)
                        match[w] = v
                        match[v] = w
                        match[m] = 0
                        stack.append(m)
                        w = nxt_w
        return False

    for i in range(1, n + 1):
        if match[i] == 0:
            if not find_aug(i):
                print("NO")
                return

    print("YES")
    out = []
    for i in range(1, q + 1):
        out.append(f"{i} {match[i]}")
    sys.stdout.write("\n".join(out) + "\n")

solve()
