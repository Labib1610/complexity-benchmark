from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Controllers: 1 to q
    # Platforms: q+1 to 2*q
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Try to find a maximum matching using BFS (Hopcroft-Karp like approach)
    # But we can use a simpler approach: try to find a maximum matching with augmenting paths
    
    # match_platform[p] = controller that platform p is matched to (0 if unmatched)
    # match_controller[c] = platform that controller c is matched to (0 if unmatched)
    match_platform = [0] * (2*q + 1)
    match_controller = [0] * (q + 1)
    
    def bfs():
        # Find shortest augmenting paths
        dist = [0] * (q + 1)
        queue = deque()
        
        for c in range(1, q + 1):
            if match_controller[c] == 0:
                dist[c] = 0
                queue.append(c)
            else:
                dist[c] = float('inf')
        
        dist[0] = float('inf')
        
        while queue:
            c = queue.popleft()
            
            if dist[c] < dist[0]:
                for p in graph[c]:
                    next_c = match_platform[p]
                    if dist[next_c] == float('inf'):
                        dist[next_c] = dist[c] + 1
                        queue.append(next_c)
        
        return dist[0] != float('inf')
    
    def dfs(c):
        if c == 0:
            return True
            
        for p in graph[c]:
            next_c = match_platform[p]
            if dist[next_c] == dist[c] + 1:
                if dfs(next_c):
                    match_platform[p] = c
                    match_controller[c] = p
                    return True
        
        dist[c] = float('inf')
        return False
    
    # Main matching algorithm
    matching = 0
    while bfs():
        for c in range(1, q + 1):
            if match_controller[c] == 0:
                if dfs(c):
                    matching += 1
    
    if matching == q:
        print("YES")
        for c in range(1, q + 1):
            print(c, match_controller[c])
    else:
        print("NO")

solve()
