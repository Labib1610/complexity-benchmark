We need to find a perfect matching in a bipartite graph where left side is controllers (1..q) and right side is platforms (q+1..2q).  
We can use Hopcroft–Karp for maximum bipartite matching, which is efficient for up to 100k nodes and 400k edges.

```python
import sys
from collections import deque

input = sys.stdin.readline

def hopcroft_karp(n, m, adj):
    # n = number of left nodes, m = number of right nodes
    INF = 10**9
    pairU = [-1] * n
    pairV = [-1] * m
    dist = [0] * n

    def bfs():
        q = deque()
        for u in range(n):
            if pairU[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        found = False
        while q:
            u = q.popleft()
            for v in adj[u]:
                pu = pairV[v]
                if pu != -1 and dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    q.append(pu)
                elif pu == -1:
                    found = True
        return found

    def dfs(u):
        for v in adj[u]:
            pu = pairV[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pairU[u] = v
                pairV[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(n):
            if pairU[u] == -1 and dfs(u):
                matching += 1
    return matching, pairU

def main():
    q, t = map(int, input().split())
    adj = [[] for _ in range(q)]
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a-1].append(b-q-1)  # right node index 0..q-1

    matching, pairU = hopcroft_karp(q, q, adj)

    if matching != q:
        print("NO")
    else:
        print("YES")
        for u in range(q):
            v = pairU[u]
            print(u+1, v+q+1)

if __name__ == "__main__":
    main()
```
