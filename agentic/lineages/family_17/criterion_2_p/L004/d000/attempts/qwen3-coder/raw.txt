from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    if t == 0:
        if q == 0:
            print("YES")
        else:
            print("NO")
        return
    
    # Build bipartite graph
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Try to find a perfect matching using BFS-based augmenting paths
    # This is essentially finding maximum matching in bipartite graph
    
    # partner[b] = a means controller a is matched to platform b
    partner = {}
    # matched[a] = b means platform b is matched to controller a
    matched = {}
    
    def bfs():
        # Find all unmatched controllers
        queue = deque()
        parent = {}
        visited = set()
        
        for controller in range(1, q + 1):
            if controller not in matched:
                queue.append(controller)
                parent[controller] = None
                visited.add(controller)
        
        found_path = False
        sink = None
        
        while queue and not found_path:
            node = queue.popleft()
            
            if node <= q:  # It's a controller
                # Look for unmatched platforms
                for platform in graph[node]:
                    if platform not in visited:
                        parent[platform] = node
                        visited.add(platform)
                        if platform not in partner:
                            # Found an augmenting path
                            found_path = True
                            sink = platform
                            break
                        else:
                            # Continue to the matched controller
                            next_controller = partner[platform]
                            parent[next_controller] = platform
                            visited.add(next_controller)
                            queue.append(next_controller)
            else:  # It's a platform
                # This case is handled in the controller section
                pass
        
        if not found_path:
            return False
        
        # Augment the path
        current = sink
        while parent[current] is not None:
            if current > q:  # Platform
                controller = parent[current]
                # Remove old matching if exists
                if controller in matched:
                    old_platform = matched[controller]
                    del partner[old_platform]
                # Add new matching
                partner[current] = controller
                matched[controller] = current
            current = parent[current]
        
        return True
    
    # Keep finding augmenting paths until no more exist
    while bfs():
        pass
    
    # Check if we have a perfect matching
    if len(matched) == q:
        print("YES")
        for controller in range(1, q + 1):
            platform = matched[controller]
            print(controller, platform)
    else:
        print("NO")

solve()
