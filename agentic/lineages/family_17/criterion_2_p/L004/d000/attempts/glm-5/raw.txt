import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        q = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return

    edges = []
    adj = [[] for _ in range(2 * q + 1)]
    
    for i in range(t):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            edges.append((u, v))
            adj[u].append((v, i))
            adj[v].append((u, i))
        except StopIteration:
            break

    # We need to find a perfect matching in a bipartite graph.
    # Using Hopcroft-Karp algorithm.
    
    pair_U = [-1] * (q + 1)
    pair_V = [-1] * (2 * q + 1)
    dist = [0] * (q + 1)
    
    def bfs():
        queue = deque()
        for u in range(1, q + 1):
            if pair_U[u] == -1:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        
        found_augmenting_path = False
        while queue:
            u = queue.popleft()
            for v, edge_idx in adj[u]:
                if pair_V[v] != -1:
                    if dist[pair_V[v]] == float('inf'):
                        dist[pair_V[v]] = dist[u] + 1
                        queue.append(pair_V[v])
                else:
                    found_augmenting_path = True
        
        return found_augmenting_path

    def dfs(u):
        for v, edge_idx in adj[u]:
            if pair_V[v] == -1 or (dist[pair_V[v]] == dist[u] + 1 and dfs(pair_V[v])):
                pair_U[u] = v
                pair_V[v] = u
                return True
        dist[u] = float('inf')
        return False

    matching_count = 0
    while bfs():
        for u in range(1, q + 1):
            if pair_U[u] == -1:
                if dfs(u):
                    matching_count += 1
    
    if matching_count != q:
        print("NO")
        return

    print("YES")
    # Output the edges corresponding to the matching
    # We iterate through controllers 1 to q and print the matched edge
    # Note: The problem asks for q lines, each containing the two IDs.
    # The order of lines is not specified, but typically any order is fine.
    # We need to find the original edge index to ensure we print the correct link if multiple exist.
    
    # Reconstructing the edge index from adj list
    for u in range(1, q + 1):
        v = pair_U[u]
        # Find the edge index for (u, v)
        # Since adj[u] stores (neighbor, edge_index), we search for v
        found_idx = -1
        for neighbor, idx in adj[u]:
            if neighbor == v:
                found_idx = idx
                break
        # Print original edge
        orig_u, orig_v = edges[found_idx]
        print(f"{orig_u} {orig_v}")

if __name__ == '__main__':
    solve()
