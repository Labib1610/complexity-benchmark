import sys
from collections import deque

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        q = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list for controllers (1 to q)
    # Platforms are mapped to 1..q (platform q+1..2q becomes 1..q)
    adj = [[] for _ in range(q + 1)]
    
    # Store edges to output them later if needed
    edges = []
    
    for _ in range(t):
        u = int(next(iterator))
        v = int(next(iterator))
        edges.append((u, v))
        adj[u].append(v - q)
    
    # Hopcroft-Karp algorithm for Maximum Bipartite Matching
    # match_u[i] = platform matched to controller i (0 if none)
    # match_v[j] = controller matched to platform j (0 if none)
    match_u = [0] * (q + 1)
    match_v = [0] * (q + 1)
    dist = [0] * (q + 1)
    
    def bfs():
        dq = deque()
        for u in range(1, q + 1):
            if match_u[u] == 0:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = -1
        
        found_augmenting_path = False
        while dq:
            u = dq.popleft()
            for v in adj[u]:
                mu = match_v[v]
                if mu != 0:
                    if dist[mu] == -1:
                        dist[mu] = dist[u] + 1
                        dq.append(mu)
                else:
                    found_augmenting_path = True
        return found_augmenting_path
    
    def dfs(u):
        for v in adj[u]:
            mu = match_v[v]
            if mu == 0 or (dist[mu] == dist[u] + 1 and dfs(mu)):
                match_u[u] = v
                match_v[v] = u
                return True
        dist[u] = -1
        return False
    
    matching_size = 0
    while bfs():
        for u in range(1, q + 1):
            if match_u[u] == 0:
                if dfs(u):
                    matching_size += 1
    
    if matching_size != q:
        print("NO")
        return
    
    print("YES")
    
    # Reconstruct the output using the edges list to preserve original platform numbering
    # match_u[u] stores the mapped platform index (1..q), convert back to original (q+1..2q)
    # We iterate through edges to find the one matching our matching set.
    # To handle multiple edges between same pair correctly (print one instance), we track used.
    
    used_edges = set()
    results = []
    
    # Create a quick lookup for edges: (controller, mapped_platform) -> list of original platform ids
    # Since we just need to print one valid pair, we can iterate edges and check against match_u
    
    for u, v_orig in edges:
        v_mapped = v_orig - q
        if match_u[u] == v_mapped:
            # Check if this specific pair (u, v_orig) is already printed
            # Since match_u[u] is unique, we just need to ensure we print each controller once
            # But simpler: iterate controllers 1..q and find the corresponding edge
            pass
            
    # Simpler approach: iterate controllers 1..q and find the edge in the list
    # To ensure we output the correct original platform number (in case of parallel edges)
    # We can build a map from (u, v_mapped) to the first occurrence of v_orig in edges
    
    edge_map = {}
    for u, v_orig in edges:
        v_mapped = v_orig - q
        key = (u, v_mapped)
        if key not in edge_map:
            edge_map[key] = v_orig
            
    for u in range(1, q + 1):
        v_mapped = match_u[u]
        v_orig = edge_map[(u, v_mapped)]
        print(f"{u} {v_orig}")

if __name__ == '__main__':
    solve()
