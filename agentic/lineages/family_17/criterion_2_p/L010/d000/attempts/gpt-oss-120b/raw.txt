```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    adj = [[] for _ in range(q + 1)]
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        # a is controller (1..q), b is platform (q+1..2q)
        adj[a].append(b - q)   # store platform index 1..q

    NIL = 0
    INF = 10**9
    pairU = [NIL] * (q + 1)   # left -> right
    pairV = [NIL] * (q + 1)   # right -> left
    dist = [0] * (q + 1)

    def bfs() -> bool:
        dq = deque()
        for u in range(1, q + 1):
            if pairU[u] == NIL:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        dist[NIL] = INF
        while dq:
            u = dq.popleft()
            if dist[u] < dist[NIL]:
                for v in adj[u]:
                    if pairV[v] == NIL:
                        if dist[NIL] == INF:
                            dist[NIL] = dist[u] + 1
                    elif dist[pairV[v]] == INF:
                        dist[pairV[v]] = dist[u] + 1
                        dq.append(pairV[v])
        return dist[NIL] != INF

    def dfs(u: int) -> bool:
        if u != NIL:
            for v in adj[u]:
                if pairV[v] == NIL or (dist[pairV[v]] == dist[u] + 1 and dfs(pairV[v])):
                    pairU[u] = v
                    pairV[v] = u
                    return True
            dist[u] = INF
            return False
        return True

    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if pairU[u] == NIL and dfs(u):
                matching += 1

    out = []
    if matching == q:
        out.append("YES")
        for u in range(1, q + 1):
            out.append(f"{u} {pairU[u] + q}")
    else:
        out.append("NO")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```
