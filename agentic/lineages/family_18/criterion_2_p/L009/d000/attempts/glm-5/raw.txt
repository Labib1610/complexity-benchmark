setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(input_data[idx]); idx+=1
    m = int(input_data[idx]); idx+=1
    B = int(input_data[idx]); idx+=1
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(input_data[idx]); idx+=1
        v = int(input_data[idx]); idx+=1
        adj[u].append(v)
        adj[v].append(u)
    
    # Root at node 1, compute subtree sizes, then DFS preorder with largest subtree last
    parent = [0]*(n+1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v in adj[u]:
            if parent[v]==0:
                parent[v]=u
                stack.append(v)
    
    size = [1]*(n+1)
    for u in reversed(order):
        if parent[u]>0:
            size[parent[u]] += size[u]
    
    # Build children lists sorted by size ascending (so largest processed last => placed last)
    children = [[] for _ in range(n+1)]
    for u in range(2, n+1):
        children[parent[u]].append(u)
    for u in range(1, n+1):
        children[u].sort(key=lambda x: size[x])  # ascending, largest last
    
    pos = [0]*(n+1)
    counter = [1]
    # iterative DFS preorder
    stack = [(1, False)]
    while stack:
        u, processed = stack.pop()
        if not processed:
            pos[u] = counter[0]
            counter[0]+=1
            stack.append((u, True))
            # push children in reverse so first child processed first
            for c in reversed(children[u]):
                stack.append((c, False))
    
    total = 0
    for u in range(2, n+1):
        total += abs(pos[u]-pos[parent[u]])
    
    if total <= B:
        print("YES")
        print(' '.join(str(pos[i]) for i in range(1, n+1)))
    else:
        print("NO")

    for _ in range(m):
        u = int(data[idx]); idx+=1
        v = int(data[idx]); idx+=1
        adj[u].append(v)
        adj[v].append(u)
    
    # Build tree rooted at 1
    parent = [0]*(n+1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
    
    size = [1]*(n+1)
    for u in reversed(order):
        p = parent[u]
        if p > 0:
            size[p] += size[u]
    
    children = [[] for _ in range(n+1)]
    for u in range(2, n+1):
        children[parent[u]].append(u)
    
    # For each node, compute:
    # h[u] = min over p of (p + f_u(p))  [for R placement]
    # hp[u] = min over p of (size[u] - p + 1 + f_u(p))  [for L placement]
    # Also store h_arg, hp_arg (the w = L_size achieving the min)
    # And store the DP table for reconstruction
    
    h = [0]*(n+1)
    hp = [0]*(n+1)
    h_arg = [0]*(n+1)  # L_size achieving h
    hp_arg = [0]*(n+1)  # L_size achieving hp
    
    # For reconstruction: store children sorted, sizes, h/hp values, and the DP
    node_info = [None]*(n+1)
    
    for u in reversed(order):
        ch = children[u]
        if not ch:
            h[u] = 1  # p=1, f=0, h = 1+0 = 1
            hp[u] = 1  # size=1, p=1, hp = 1-1+1+0 = 1
            h_arg[u] = 0
            hp_arg[u] = 0
            node_info[u] = (ch, [], [], [], None)
            continue
        
        # Sort children by size DESCENDING
        ch_sorted = sorted(ch, key=lambda x: -size[x])
        k = len(ch_sorted)
        sizes = [size[c] for c in ch_sorted]
        hvals = [h[c] for c in ch_sorted]
        hpvals = [hp[c] for c in ch_sorted]
        
        # DP: state (L_size, L_count), process children descending
        # dp[(L_size, L_count)] = min cost
        # Cost increase for L: hp[j] + sizes[j] * L_count (before adding)
        # Cost increase for R: h[j] + (prefix - L_size) (R_size before adding)
        
        dp = {(0, 0): 0}
        prefix = 0  # total size processed so far
        for i in range(k):
            si = sizes[i]
            hi = hvals[i]
            hpi = hpvals[i]
            new_dp = {}
            for (ls, lc), val in dp.items():
                # Add to L: new L_size = ls + si, new L_count = lc + 1
                # cost increase = hpi + si * lc
                nls = ls + si
                nlc = lc + 1
                nval = val + hpi + si * lc
                key = (nls, nlc)
                if key not in new_dp or nval < new_dp[key]:
                    new_dp[key] = nval
                # Add to R: L_size, L_count unchanged
                # R_size before = prefix - ls
                # cost increase = hi + (prefix - ls)
                nval2 = val + hi + (prefix - ls)
                key2 = (ls, lc)
                if key2 not in new_dp or nval2 < new_dp[key2]:
                    new_dp[key2] = nval2
            dp = new_dp
            prefix += si
        
        # Compute h, hp
        # f_u(p) = dp[(p-1, lc)] for any lc (but we need min over lc for each p-1)
        # Actually, f_u(p) = min over lc of dp[(p-1, lc)]
        # h[u] = min over p of (p + f_u(p)) = min over (ls, lc) of (1 + ls + dp[(ls, lc)])
        # hp[u] = min over p of (size[u] - p + 1 + f_u(p)) = min over (ls, lc) of (size[u] - ls + dp[(ls, lc)])
        
        best_h = float('inf')
        best_hp = float('inf')
        bh = 0
        bhp = 0
        for (ls, lc), val in dp.items():
            vh = 1 + ls + val
            vhp = size[u] - ls + val
            if vh < best_h:
                best_h = vh
                bh = ls
            if vhp < best_hp:
                best_hp = vhp
                bhp = ls
        
        h[u] = best_h
        hp[u] = best_hp
        h_arg[u] = bh
        hp_arg[u] = bhp
        node_info[u] = (ch_sorted, sizes, hvals, hpvals, dp)
    
    # Root answer
    root_dp = node_info[1][4]
    best_val = min(root_dp.values())
    
    if best_val > B:
        print("NO")
        return
    
    # Find best (ls, lc) for root
    best_key = min(root_dp, key=lambda k: root_dp[k])
    best_ls = best_key[0]
    
    # Reconstruct positions
    pos = [0]*(n+1)
    
    # For each node, given block_start and target L_size (which determines relative position),
    # reconstruct the partition and place children.
    # We need to recompute the DP with backpointers, or trace through.
    
    # Stack: (node, block_start, target_ls)
    stack = [(1, 1, best_ls)]
    while stack:
        u, block_start, target_ls = stack.pop()
        pos[u] = block_start + target_ls  # relative position = 1 + target_ls, global = block_start + target_ls
        
        ch_sorted, sizes, hvals, hpvals, dp = node_info[u]
        if not ch_sorted:
            continue
        
        k = len(ch_sorted)
        
        # Redo DP with backpointers to find the partition
        # dp_steps[i] = dict (ls, lc) -> (cost, (prev_ls, prev_lc, assignment))
        dp_steps = [None]*(k+1)
        dp_steps[0] = {(0, 0): (0, None)}
        prefix = 0
        for i in range(k):
            si = sizes[i]
            hi = hvals[i]
            hpi = hpvals[i]
            new_dp = {}
            for (ls, lc), (val, _) in dp_steps[i].items():
                # To L
                nls = ls + si
                nlc = lc + 1
                nval = val + hpi + si * lc
                key = (nls, nlc)
                if key not in new_dp or nval < new_dp[key][0]:
                    new_dp[key] = (nval, (ls, lc, 'L'))
                # To R
                nval2 = val + hi + (prefix - ls)
                key2 = (ls, lc)
                if key2 not in new_dp or nval2 < new_dp[key2][0]:
                    new_dp[key2] = (nval2, (ls, lc, 'R'))
            dp_steps[i+1] = new_dp
            prefix += si
        
        # Find the target (ls, lc) in final DP
        # We need target_ls, but lc can be anything. Find the lc that achieves min cost for target_ls.
        best_lc = None
        best_cost = float('inf')
        for (ls, lc), (val, _) in dp_steps[k].items():
            if ls == target_ls and val < best_cost:
                best_cost = val
                best_lc = lc
        
        # Trace back
        assignments = [None]*k
        cur_ls, cur_lc = target_ls, best_lc
        for i in range(k, 0, -1):
            _, bp = dp_steps[i][(cur_ls, cur_lc)]
            prev_ls, prev_lc, assign = bp
            assignments[i-1] = assign
            cur_ls, cur_lc = prev_ls, prev_lc
        
        # Place children
        # L children: sorted descending (already), placed right to left (largest leftmost = furthest from u)
        # Wait, L children sorted descending: largest first. In the layout, largest is leftmost (furthest from u),
        # smallest is rightmost (closest to u).
        # L children fill relative positions 1..target_ls (block_start..block_start+target_ls-1)
        # Largest L child: block at relative 1..s1, next at s1+1..s1+s2, etc.
        
        # R children: sorted descending, largest first = closest to u (leftmost in R).
        # R children fill relative positions target_ls+2..size[u] (block_start+target_ls+1..block_start+size[u]-1)
        
        l_children = []
        r_children = []
        for i in range(k):
            if assignments[i] == 'L':
                l_children.append((ch_sorted[i], sizes[i]))
            else:
                r_children.append((ch_sorted[i], sizes[i]))
        
        # L children: placed left to right in descending size order (largest leftmost)
        cur_pos = block_start
        for c, cs in l_children:
            # c's block at [cur_pos, cur_pos+cs-1]
            # c is in L, so its relative position = 1 + hp_arg[c]
            c_ls = hp_arg[c]
            stack.append((c, cur_pos, c_ls))
            cur_pos += cs
        
        # R children: placed left to right in descending size order (largest closest to u)
        cur_pos = block_start + target_ls + 1
        for c, cs in r_children:
            c_ls = h_arg[c]
            stack.append((c, cur_pos, c_ls))
            cur_pos += cs
    
    print("YES")
    print(' '.join(str(pos[i]) for i in range(1, n+1)))
