```python
import sys
import itertools

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    # brute force all permutations
    nodes = list(range(n))
    for perm in itertools.permutations(nodes):
        # perm[i] = position (0-indexed) of node i? we need inverse mapping
        # Build position array: pos[node] = slot (1..n)
        pos = [0] * n
        for slot, node in enumerate(perm, start=1):
            pos[node] = slot
        total = 0
        for u, v in edges:
            total += abs(pos[u] - pos[v])
            if total > B:
                break
        if total <= B:
            # output
            out = ['YES', ' '.join(str(pos[i]) for i in range(n))]
            sys.stdout.write('\n'.join(out))
            return
    sys.stdout.write('NO')

if __name__ == "__main__":
    solve()
```
