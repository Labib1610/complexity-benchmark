```python
from itertools import permutations
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    edges = []
    idx = 3
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Try all permutations of slot assignments (from 0 to n-1)
    for perm in permutations(range(n)):
        total_length = 0
        # For each required communication, compute the distance
        for u, v in edges:
            pos_u = perm[u]
            pos_v = perm[v]
            total_length += abs(pos_u - pos_v)
            # Early exit if we already exceed the budget
            if total_length > B:
                break
        if total_length <= B:
            # Found a valid assignment
            # Convert from 0-based to 1-based indexing
            assignment = [str(x + 1) for x in perm]
            print("YES")
            print(" ".join(assignment))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
```
