```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])
    
    deg = [0] * (n + 1)
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        deg[u] += 1
        deg[v] += 1
        
    modules = list(range(1, n + 1))
    # Sort modules by degree in descending order to minimize total wire length
    modules.sort(key=lambda x: -deg[x])
    
    # Assign slots sequentially from 1 to n
    assignment = [0] * (n + 1)
    for i in range(n):
        assignment[modules[i]] = i + 1
        
    # Calculate total wire length
    total_wire = 0
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        total_wire += abs(assignment[u] - assignment[v])
        
    if total_wire <= B:
        print("YES")
        print(" ".join(map(str, assignment[1:])))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
