import sys
from itertools import permutations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    edges = []
    index = 3
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Try all permutations (only for small n)
    if n > 10:
        # For larger n, we need a heuristic or optimization
        # But according to constraints, n can be up to 1000, so this approach won't work
        # Let's implement a greedy approach for larger n
        pass
    
    # Since n can be up to 1000, trying all permutations is infeasible
    # We need a smarter approach
    
    # Let's model this as a quadratic assignment problem variant
    # But for now, let's try a greedy placement
    
    # A simple greedy approach:
    # Try to place nodes that have the most connections closer together
    
    # Count degree of each node
    degree = [0] * n
    for u, v in edges:
        degree[u] += 1
        degree[v] += 1
    
    # Create a list of (degree, node_index) and sort by degree descending
    nodes_with_degree = [(degree[i], i) for i in range(n)]
    nodes_with_degree.sort(reverse=True)
    
    # Initialize positions array
    pos = [-1] * n
    
    # Place the highest degree node in the middle
    if n > 0:
        mid = n // 2
        node_idx = nodes_with_degree[0][1]
        pos[node_idx] = mid
        
        # Place other nodes greedily
        available_positions = list(range(n))
        available_positions.remove(mid)
        
        # For each remaining node, place it in the position that minimizes the total wire
        unplaced_nodes = [x[1] for x in nodes_with_degree[1:]]
        
        # To make it efficient, let's just place them sequentially near the already placed ones
        # This is a very simple heuristic
        
        # Let's try a different approach: use BFS-like placement
        # Start with the highest degree node
        placed = [node_idx]
        unplaced = set(range(n))
        unplaced.remove(node_idx)
        
        # While there are unplaced nodes
        while unplaced:
            # Find a node in unplaced that is connected to a placed node
            found = False
            for u in unplaced:
                neighbors_in_placed = []
                for v in placed:
                    if (u, v) in edges or (v, u) in edges:
                        neighbors_in_placed.append(v)
                if neighbors_in_placed:
                    # Place u next to one of its neighbors
                    # Find the neighbor with position assigned
                    for v in neighbors_in_placed:
                        v_pos = pos[v]
                        # Try to place u in v_pos - 1 or v_pos + 1
                        candidates = []
                        if v_pos > 0 and pos[v_pos - 1] == -1:
                            candidates.append(v_pos - 1)
                        if v_pos < n - 1 and pos[v_pos + 1] == -1:
                            candidates.append(v_pos + 1)
                        if candidates:
                            # Choose the first available
                            pos[u] = candidates[0]
                            placed.append(u)
                            unplaced.remove(u)
                            found = True
                            break
                    if found:
                        break
            if not found:
                # Place any remaining node
                u = unplaced.pop()
                # Find first available position
                for i in range(n):
                    if pos[i] == -1:
                        pos[u] = i
                        placed.append(u)
                        break
        
        # If there are still unplaced (shouldn't happen), place them
        for i in range(n):
            if pos[i] == -1:
                # Find first available slot
                for j in range(n):
                    if j not in pos:
                        pos[i] = j
                        break
                        
        # Calculate total wire needed
        total_wire = 0
        for u, v in edges:
            total_wire += abs(pos[u] - pos[v])
            
        if total_wire <= B:
            print("YES")
            # Convert to 1-based indexing
            result = [p + 1 for p in pos]
            print(' '.join(map(str, result)))
            return
            
    # If greedy didn't work, try a more systematic approach for small n
    if n <= 8:
        # Try all permutations
        for perm in permutations(range(n)):
            total = 0
            pos_map = {i: perm[i] for i in range(n)}
            for u, v in edges:
                total += abs(pos_map[u] - pos_map[v])
                if total > B:
                    break
            if total <= B:
                print("YES")
                result = [perm[i] + 1 for i in range(n)]
                print(' '.join(map(str, result)))
                return
        print("NO")
        return
    
    # For larger n, the above greedy might not be sufficient
    # Let's try a better greedy approach
    
    # Reinitialize
    pos = [-1] * n
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Find node with maximum degree
    max_degree = -1
    center_node = 0
    for i in range(n):
        if len(adj[i]) > max_degree:
            max_degree = len(adj[i])
            center_node = i
    
    # Place center node in the middle
    mid = n // 2
    pos[center_node] = mid
    
    # BFS to place nodes
    from collections import deque
    queue = deque()
    visited = [False] * n
    queue.append(center_node)
    visited[center_node] = True
    
    available_slots = set(range(n))
    available_slots.remove(mid)
    
    while queue:
        node = queue.popleft()
        node_pos = pos[node]
        # Get unvisited neighbors
        for neighbor in adj[node]:
            if not visited[neighbor]:
                visited[neighbor] = True
                queue.append(neighbor)
                # Place neighbor in closest available slot to node_pos
                # Find closest available slot
                best_slot = None
                best_dist = float('inf')
                for slot in list(available_slots):
                    dist = abs(slot - node_pos)
                    if dist < best_dist:
                        best_dist = dist
                        best_slot = slot
                if best_slot is not None:
                    pos[neighbor] = best_slot
                    available_slots.remove(best_slot)
    
    # Place any remaining unplaced nodes
    for i in range(n):
        if pos[i] == -1:
            if available_slots:
                pos[i] = available_slots.pop()
    
    # Calculate wire usage
    total_wire = 0
    for u, v in edges:
        total_wire += abs(pos[u] - pos[v])
    
    if total_wire <= B:
        print("YES")
        result = [p + 1 for p in pos]
        print(' '.join(map(str, result)))
    else:
        print("NO")

if __name__ == "__main__":
    main()
