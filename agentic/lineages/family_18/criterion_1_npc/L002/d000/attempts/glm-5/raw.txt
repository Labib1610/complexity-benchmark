import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u-1, v-1))  # 0-indexed
    
    # Compute minimum wire length
    # For each edge (u, v), minimum contribution is 1
    min_wire = m
    
    if min_wire > B:
        print("NO")
        return
    
    # Check if we can achieve exactly min_wire
    # This requires all edges to be between adjacent units
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Try to find a Hamiltonian path in the graph
    # Using backtracking with pruning
    path = []
    used = [False] * n
    
    def backtrack():
        if len(path) == n:
            return True
        curr = path[-1] if path else -1
        # Get candidates
        if curr == -1:
            candidates = range(n)
        else:
            # Prioritize neighbors (required for min wire)
            neighbors = [v for v in adj[curr] if not used[v]]
            non_neighbors = [v for v in range(n) if not used[v] and v not in adj[curr]]
            candidates = neighbors + non_neighbors
        
        for nxt in candidates:
            if used[nxt]:
                continue
            used[nxt] = True
            path.append(nxt)
            if backtrack():
                return True
            path.pop()
            used[nxt] = False
        return False
    
    if backtrack():
        # Convert to 1-indexed slot assignment
        slots = [0] * n
        for slot, unit in enumerate(path):
            slots[unit] = slot + 1
        print("YES")
        print(" ".join(map(str, slots)))
        return
    
    # If no Hamiltonian path, we need more wire
    # Use a simple greedy approach
    # Start with unit 0, then greedily pick next unit minimizing additional wire
    path = [0]
    used = [False] * n
    used[0] = True
    total_wire = 0
    
    for _ in range(n - 1):
        curr = path[-1]
        best_next = -1
        best_cost = float('inf')
        
        for nxt in range(n):
            if used[nxt]:
                continue
            # Calculate additional wire cost
            cost = 0
            for u, v in edges:
                if (u == curr and v == nxt) or (v == curr and u == nxt):
                    cost += 1
                elif u == nxt and used[v]:
                    cost += abs(len(path) - path.index(v))
                elif v == nxt and used[u]:
                    cost += abs(len(path) - path.index(u))
            
            if cost < best_cost:
                best_cost = cost
                best_next = nxt
        
        if best_next == -1:
            break
        
        # Calculate actual wire added
        for u, v in edges:
            if u == best_next and used[v]:
                total_wire += len(path) - path.index(v)
            elif v == best_next and used[u]:
                total_wire += len(path) - path.index(u)
        
        path.append(best_next)
        used[best_next] = True
    
    if len(path) == n and total_wire <= B:
        slots = [0] * n
        for slot, unit in enumerate(path):
            slots[unit] = slot + 1
        print("YES")
        print(" ".join(map(str, slots)))
    else:
        # Try all permutations for small n
        from itertools import permutations
        
        for perm in permutations(range(n)):
            total = 0
            for u, v in edges:
                total += abs(perm[u] - perm[v])
            if total <= B:
                slots = [0] * n
                for slot, unit in enumerate(perm):
                    slots[unit] = slot + 1
                print("YES")
                print(" ".join(map(str, slots)))
                return
        
        print("NO")

if __name__ == "__main__":
    solve()
