import sys
from collections import defaultdict

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each clause (l1 or l2), we create implications
    # l1 or l2 is equivalent to (not l1 -> l2) and (not l2 -> l1)
    
    # We'll use 2-SAT
    # Variables: 1 to n (mode 1), -1 to -n (mode 0)
    # For each variable x, we have:
    # x = node 2*x
    # not x = node 2*x + 1
    # But it's easier to map: 
    # for variable x (1-based), 
    # mode 1 (x) -> node 2*(x-1)
    # mode 0 (-x) -> node 2*(x-1)+1
    
    adj = defaultdict(list)
    rev_adj = defaultdict(list)
    
    for _ in range(m):
        l1, l2 = map(int, sys.stdin.readline().split())
        # l1 or l2
        # not l1 -> l2
        # not l2 -> l1
        
        # Convert to 0-based indexing for nodes
        # For a variable x (1 to n):
        # Node for x (mode 1): 2*(x-1)
        # Node for not x (mode 0): 2*(x-1)+1
        # For a variable -x (meaning mode 0 for |x|):
        # Node for -x: 2*(|x|-1)+1
        # Node for not(-x) = x: 2*(|x|-1)
        
        # l1 or l2 means (not l1 -> l2) and (not l2 -> l1)
        
        # Convert l1 and l2 to nodes
        if l1 > 0:
            node_l1 = 2 * (l1 - 1)       # x in mode 1
            node_not_l1 = 2 * (l1 - 1) + 1  # x in mode 0
        else:
            x = -l1
            node_l1 = 2 * (x - 1) + 1    # x in mode 0
            node_not_l1 = 2 * (x - 1)    # x in mode 1
            
        if l2 > 0:
            node_l2 = 2 * (l2 - 1)       # x in mode 1
            node_not_l2 = 2 * (l2 - 1) + 1  # x in mode 0
        else:
            x = -l2
            node_l2 = 2 * (x - 1) + 1    # x in mode 0
            node_not_l2 = 2 * (x - 1)    # x in mode 1
            
        # Add implications
        adj[node_not_l1].append(node_l2)
        adj[node_not_l2].append(node_l1)
        rev_adj[node_l2].append(node_not_l1)
        rev_adj[node_l1].append(node_not_l2)
    
    # Now we do Kosaraju's algorithm for 2-SAT
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Second DFS on reverse graph
    visited = [False] * (2 * n)
    components = [-1] * (2 * n)
    comp_num = 0
    
    def dfs2(u):
        visited[u] = True
        components[u] = comp_num
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            comp_num += 1
    
    # Check if satisfiable
    for i in range(n):
        node_1 = 2 * i      # Machine i+1 in mode 1
        node_0 = 2 * i + 1  # Machine i+1 in mode 0
        if components[node_1] == components[node_0]:
            print("NO")
            return
    
    # Build solution
    result = [0] * n
    # Process components in reverse topological order
    # The component with higher number in first DFS is processed first in second DFS
    # So components with higher comp_num were finished later in first DFS
    # We want to assign values based on finishing time in reverse
    
    comp_order = list(range(comp_num))
    comp_order.sort(key=lambda x: x, reverse=True)
    
    assigned = [False] * (2 * n)
    
    # But it's easier: for each variable, if node for mode 1 has higher component number
    # than node for mode 0, then set to mode 1, else mode 0
    for i in range(n):
        node_1 = 2 * i
        node_0 = 2 * i + 1
        if components[node_1] > components[node_0]:
            result[i] = 1
        else:
            result[i] = 0
    
    print("YES")
    print(" ".join(map(str, result)))

if __name__ == "__main__":
    main()
