Here’s the Python 3 solution for the problem:  

```python
import sys

input = sys.stdin.readline

n, m = map(int, input().split())
edges = []
for _ in range(m):
    a, b = map(int, input().split())
    edges.append((a, b))

# Build graph for 2-SAT
# Each variable x_i: 0 = False, 1 = True
# For each interlock (c1, c2):
# strain = (s_u != b(c1)) + (s_v != b(c2))
# We need strain <= 1, so NOT (strain == 2)
# strain == 2 means both mismatch:
#   s_u != b(c1) AND s_v != b(c2)
# So we forbid: (s_u == 1-b(c1)) AND (s_v == 1-b(c2))
# In 2-SAT: (s_u != 1-b(c1)) OR (s_v != 1-b(c2))
# i.e. (s_u == b(c1)) OR (s_v == b(c2))

# Map variable index: machine i -> 2*i (False), 2*i+1 (True)
# For clause (x OR y): add implications (!x -> y) and (!y -> x)

g = [[] for _ in range(2 * n)]
gt = [[] for _ in range(2 * n)]

def add_imp(u, v):
    g[u].append(v)
    gt[v].append(u)

def add_clause(a, a_val, b, b_val):
    # (a == a_val) OR (b == b_val)
    # !(a == a_val) -> (b == b_val)
    # !(b == b_val) -> (a == a_val)
    na = 2 * a + (1 - a_val)
    nb = 2 * b + (1 - b_val)
    add_imp(na, 2 * b + b_val)
    add_imp(nb, 2 * a + a_val)

for c1, c2 in edges:
    u1 = abs(c1)
    u2 = abs(c2)
    b1 = 1 if c1 > 0 else 0
    b2 = 1 if c2 > 0 else 0
    # Forbid both mismatches: (s_u1 != b1) AND (s_u2 != b2)
    # So require: (s_u1 == b1) OR (s_u2 == b2)
    add_clause(u1 - 1, b1, u2 - 1, b2)

# Kosaraju SCC
visited = [False] * (2 * n)
order = []

def dfs1(v):
    visited[v] = True
    for to in g[v]:
        if not visited[to]:
            dfs1(to)
    order.append(v)

for i in range(2 * n):
    if not visited[i]:
        dfs1(i)

comp = [-1] * (2 * n)
def dfs2(v, cid):
    comp[v] = cid
    for to in gt[v]:
        if comp[to] == -1:
            dfs2(to, cid)

cid = 0
for v in reversed(order):
    if comp[v] == -1:
        dfs2(v, cid)
        cid += 1

# Check satisfiability
for i in range(n):
    if comp[2 * i] == comp[2 * i + 1]:
        print("NO")
        sys.exit()

print("YES")
res = []
for i in range(n):
    # Choose False if comp[2*i] < comp[2*i+1] (topological order)
    if comp[2 * i] < comp[2 * i + 1]:
        res.append(0)
    else:
        res.append(1)
print(" ".join(map(str, res)))
```
