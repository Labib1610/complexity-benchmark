```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    clauses = []
    index = 2
    for _ in range(m):
        c1 = int(data[index])
        c2 = int(data[index+1])
        index += 2
        clauses.append((c1, c2))
    
    # For each clause (c1, c2), the sum of strains must be <= 1
    # Strain for a linkage c is 1 if s[|c|] != b(c), 0 otherwise
    # b(c) = (c + |c|) / (2|c|) = 1 if c > 0, 0 if c < 0
    # So strain = 1 if s[|c|] != (1 if c>0 else 0) => s[|c|] != (c>0)
    # This is equivalent to s[|c|] != (c > 0) => s[|c|] == (c <= 0)
    # Let's define boolean variables: for machine i (1-based), variable i represents s_i = 1
    # We'll use 2-SAT
    
    # Each clause (c1, c2) means strain1 + strain2 <= 1
    # strain1 = 1 if s[|c1|] != b(c1) else 0
    # strain2 = 1 if s[|c2|] != b(c2) else 0
    # So we want NOT(strain1 = 1 AND strain2 = 1)
    # This is equivalent to (strain1 = 0 OR strain2 = 0)
    # strain1 = 0 means s[|c1|] == b(c1)
    # strain2 = 0 means s[|c2|] == b(c2)
    # So clause is (s[|c1|] == b(c1) OR s[|c2|] == b(c2))
    
    # In terms of boolean variables:
    # If c1 > 0, then b(c1) = 1, so s[|c1|] == 1 => variable |c1| is True
    # If c1 < 0, then b(c1) = 0, so s[|c1|] == 0 => variable |c1| is False => NOT(|c1|)
    # Similarly for c2
    
    # So the clause is:
    # ( (c1>0) ? var[|c1|] : NOT var[|c1|] ) OR ( (c2>0) ? var[|c2|] : NOT var[|c2|] )
    # This is a 2-CNF clause
    
    # Let's build the implication graph for 2-SAT
    # Variables: 1..n (s_i = 1) and -1..-n (s_i = 0)
    # For clause (a OR b), we add edges NOT a -> b and NOT b -> a
    
    # In our case, clause is (literal1 OR literal2)
    # where literal1 = (c1>0) ? |c1| : -|c1|
    #       literal2 = (c2>0) ? |c2| : -|c2|
    # So we add edges -literal1 -> literal2 and -literal2 -> literal1
    
    graph = defaultdict(list)
    reverse_graph = defaultdict(list)
    
    def add_clause(lit1, lit2):
        # Add edge NOT lit1 -> lit2
        graph[-lit1].append(lit2)
        reverse_graph[lit2].append(-lit1)
        # Add edge NOT lit2 -> lit1
        graph[-lit2].append(lit1)
        reverse_graph[lit1].append(-lit2)
    
    for c1, c2 in clauses:
        u1 = abs(c1)
        u2 = abs(c2)
        lit1 = u1 if c1 > 0 else -u1
        lit2 = u2 if c2 > 0 else -u2
        add_clause(lit1, lit2)
    
    # Now we do Kosaraju's algorithm to find strongly connected components
    visited = [False] * (2 * n + 1)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in graph[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    # We need to handle negative indices, so let's shift
    # Use index 0..2n where 0 is dummy, 1..n are positive vars, n+1..2n are negative vars
    # Actually, let's just use a dict for visited and order
    
    visited = {}
    order = []
    
    def dfs1(u):
        if u in visited:
            return
        visited[u] = True
        for v in graph[u]:
            dfs1(v)
        order.append(u)
    
    # First DFS on original graph
    nodes = set()
    for u in graph:
        nodes.add(u)
        for v in graph[u]:
            nodes.add(v)
    
    for u in nodes:
        dfs1(u)
    
    # Second DFS on reverse graph in reverse order
    visited = {}
    scc = {}
    scc_id = 0
    
    def dfs2(u):
        if u in visited:
            return
        visited[u] = True
        scc[u] = scc_id
        for v in reverse_graph[u]:
            dfs2(v)
    
    for u in reversed(order):
        if u not in visited:
            dfs2(u)
            scc_id += 1
    
    # Check if any variable and its negation are in the same SCC
    for i in range(1, n+1):
        if scc.get(i, -1) == scc.get(-i, -2):
            print("NO")
            return
    
    # Assign values: process SCCs in reverse topological order
    # SCC with higher id was finished later in first DFS, so should be processed first
    assignment = {}
    
    # Process nodes in order of decreasing SCC id
    scc_nodes = defaultdict(list)
    for node, sid in scc.items():
        scc_nodes[sid].append(node)
    
    for sid in reversed(range(scc_id)):
        for node in scc_nodes[sid]:
            var = abs(node)
            if var not in assignment:
                # If node is positive literal, set var to True
                # If node is negative literal, set var to False
                assignment[var] = 1 if node > 0 else 0
    
    # Output the result
    result = []
    for i in range(1, n+1):
        result.append(str(assignment.get(i, 0)))  # default to 0 if not assigned
    
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
