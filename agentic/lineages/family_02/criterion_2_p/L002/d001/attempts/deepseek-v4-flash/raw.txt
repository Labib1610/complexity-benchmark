 Here’s a possible solution using a 2-SAT approach, since each machine has two possible modes and each interlock forbids exactly one combination of two assignments.  

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def node(var, val):
        # var: 0-based machine index, val: 0 or 1
        return 2 * var + (1 if val else 0)

    def add_imp(a, va, b, vb):
        # a -> b
        g[node(a, va)].append(node(b, vb))
        rg[node(b, vb)].append(node(a, va))

    for _ in range(m):
        l1, l2 = map(int, input().split())
        a = abs(l1) - 1
        b = abs(l2) - 1
        va = 1 if l1 > 0 else 0
        vb = 1 if l2 > 0 else 0
        # forbidden: (a = 1-va) and (b = 1-vb)
        # so if a = 1-va then b must be vb
        # and if b = 1-vb then a must be va
        add_imp(a, 1 - va, b, vb)
        add_imp(b, 1 - vb, a, va)

    # Kosaraju for SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, c):
        comp[u] = c
        for v in rg[u]:
            if comp[v] == -1:
                dfs2(v, c)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    print("YES")
    res = []
    for i in range(n):
        # choose value: if comp[2*i] > comp[2*i+1] then True (1) else False (0)
        res.append('1' if comp[2*i] > comp[2*i+1] else '0')
    print(' '.join(res))

if __name__ == "__main__":
    solve()
```
