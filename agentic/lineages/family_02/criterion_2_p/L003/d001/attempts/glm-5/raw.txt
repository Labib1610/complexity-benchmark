import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    N = 2 * n + 2
    head = [-1] * N
    to = []
    nxt = []
    
    def add_edge(u, v):
        to.append(v)
        nxt.append(head[u])
        head[u] = len(to) - 1
    
    def node(x):
        if x > 0:
            return 2 * x
        else:
            return 2 * (-x) + 1
    
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        u = node(a)
        v = node(b)
        add_edge(u ^ 1, v)
        add_edge(v ^ 1, u)
    
    visited = [0] * N
    order = []
    
    for i in range(2, 2 * n + 2):
        if visited[i] == 0:
            stack = [(i, 0)]
            while stack:
                v, p = stack[-1]
                if visited[v] == 0:
                    visited[v] = 1
                e = p
                while e != -1:
                    w = to[e]
                    if visited[w] == 0:
                        stack[-1] = (v, head[e])
                        stack.append((w, head[w]))
                        break
                    e = nxt[e]
                else:
                    stack.pop()
                    order.append(v)
    
    comp = [-1] * N
    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            stack = [u]
            comp[u] = cid
            while stack:
                v = stack.pop()
                e = head[v]
                while e != -1:
                    w = to[e]
                    if comp[w] == -1:
                        comp[w] = cid
                        stack.append(w)
                    e = nxt[e]
            cid += 1
    
    assignment = [0] * (n + 1)
    for i in range(1, n + 1):
        if comp[2 * i] == comp[2 * i + 1]:
            sys.stdout.write("NO\n")
            return
        assignment[i] = 1 if comp[2 * i] > comp[2 * i + 1] else 0
    
    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(str(assignment[i]) for i in range(1, n + 1)) + '\n')

solve()
