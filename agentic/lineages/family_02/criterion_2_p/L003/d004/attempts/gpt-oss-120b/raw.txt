```python
import sys

sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = n * 2
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def var_index(x: int) -> int:
        """return node index for literal x (positive for xi, negative for ¬xi)"""
        v = abs(x) - 1
        if x > 0:
            return v * 2      # xi true
        else:
            return v * 2 + 1  # xi false (¬xi)

    def neg(idx: int) -> int:
        return idx ^ 1

    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        # literals that must be true to avoid warning
        lit1 = var_index(l1)
        lit2 = var_index(l2)
        # clause (lit1 OR lit2)
        # add implications: ¬lit1 -> lit2, ¬lit2 -> lit1
        g[neg(lit1)].append(lit2)
        g[neg(lit2)].append(lit1)
        rg[lit2].append(neg(lit1))
        rg[lit1].append(neg(lit2))

    visited = [False] * N
    order = []

    def dfs(v: int):
        stack = [(v, 0)]
        while stack:
            node, state = stack.pop()
            if state == 0:
                if visited[node]:
                    continue
                visited[node] = True
                stack.append((node, 1))
                for to in g[node]:
                    if not visited[to]:
                        stack.append((to, 0))
            else:
                order.append(node)

    for v in range(N):
        if not visited[v]:
            dfs(v)

    comp = [-1] * N
    cid = 0

    def rdfs(v: int, label: int):
        stack = [v]
        while stack:
            node = stack.pop()
            if comp[node] != -1:
                continue
            comp[node] = label
            for to in rg[node]:
                if comp[to] == -1:
                    stack.append(to)

    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, cid)
            cid += 1

    assignment = [0] * n
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return
        assignment[i] = 1 if comp[2*i] > comp[2*i+1] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
