import sys
import threading

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # For each variable x_i, we create two nodes: 2*i (x_i) and 2*i+1 (~x_i)
    # For convenience, we'll index from 0 to 2*n+1
    # But actually, let's think differently.
    # Variables: x_1, ..., x_n where x_i = 1 means primary env
    # For each clause (a OR b), we build implication graph
    # (~a => b) and (~b => a)
    
    # Let's map variables:
    # For microservice i (1-based), 
    # Literal +i means x_i = True (primary env)
    # Literal -i means x_i = False (secondary env)
    # So for node indices:
    # Node 2*(i-1) represents x_i = False (variable i is False)
    # Node 2*(i-1)+1 represents x_i = True (variable i is True)
    
    adj = [[] for _ in range(2 * n)]
    rev_adj = [[] for _ in range(2 * n)]
    
    def var_index(lit):
        # lit is a non-zero integer
        # returns the node index in implication graph
        v = abs(lit)
        idx = v - 1  # convert to 0-based
        if lit > 0:
            return 2 * idx + 1  # x_v = True
        else:
            return 2 * idx      # x_v = False
    
    def neg(lit):
        return -lit
    
    # Process clauses
    for i in range(2, 2 + 2*m, 2):
        l1 = int(data[i])
        l2 = int(data[i+1])
        
        # Clause is (l1 OR l2)
        # This is equivalent to (~l1 => l2) AND (~l2 => l1)
        # In terms of nodes:
        # neg(l1) => l2  and  neg(l2) => l1
        
        n1 = var_index(neg(l1))
        p2 = var_index(l2)
        n2 = var_index(neg(l2))
        p1 = var_index(l1)
        
        adj[n1].append(p2)
        adj[n2].append(p1)
        rev_adj[p2].append(n1)
        rev_adj[p1].append(n2)
    
    # Now run Kosaraju's algorithm for 2-SAT
    order = []
    visited = [False] * (2 * n)
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Reverse graph traversal
    visited = [False] * (2 * n)
    components = [-1] * (2 * n)
    comp_num = 0
    
    def dfs2(u):
        visited[u] = True
        components[u] = comp_num
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
    
    for i in reversed(order):
        if not visited[i]:
            dfs2(i)
            comp_num += 1
    
    # Check if satisfiable
    for i in range(n):
        false_node = 2 * i       # x_{i+1} = False
        true_node = 2 * i + 1    # x_{i+1} = True
        if components[false_node] == components[true_node]:
            print("NO")
            return
    
    # Satisfiable, construct assignment
    assignment = [0] * n
    comp_order = [-1] * comp_num
    for i in range(2 * n):
        c = components[i]
        if comp_order[c] == -1:
            comp_order[c] = i
    
    # Sort components in topological order
    sorted_comps = list(range(comp_num))
    sorted_comps.sort(key=lambda x: comp_order[x])
    
    comp_assignment = [-1] * comp_num
    for c in sorted_comps:
        # If not assigned yet, assign it to False (0)
        if comp_assignment[c] == -1:
            comp_assignment[c] = 0
            # Find the complement component
            # We need to make sure that if component c is assigned 0,
            # then the complement gets assigned 1
            pass  # Actually, we can just use the fact that if a comp is lower in order, it's False
    
    # Better approach: for each variable, check which of its literals is in a later SCC
    for i in range(n):
        false_comp = components[2 * i]
        true_comp = components[2 * i + 1]
        if comp_order[false_comp] < comp_order[true_comp]:
            assignment[i] = 1  # x_i = True (Primary)
        else:
            assignment[i] = 0  # x_i = False (Secondary)
    
    print("YES")
    print(' '.join(map(str, assignment)))

threading.Thread(target=main).start()
