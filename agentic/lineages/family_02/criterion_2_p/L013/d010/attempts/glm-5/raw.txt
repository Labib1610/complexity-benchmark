```python
import sys
from collections import deque

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    
    adj = [[] for _ in range(n)]
    
    idx = 2
    for _ in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx+1])
        idx += 2
        
        u = abs(l1) - 1
        v = abs(l2) - 1
        
        # We want to enforce t_{l1} * t_{l2} != 1
        # t_{l1} = 1 - s_u if l1 > 0 else s_u
        # t_{l2} = 1 - s_v if l2 > 0 else s_v
        # This simplifies to s_u XOR s_v = c, where c = 1 if l1*l2 > 0 else 0
        c = 1 if l1 * l2 > 0 else 0
        adj[u].append((v, c))
        adj[v].append((u, c))
        
    s = [-1] * n
    
    for i in range(n):
        if s[i] == -1:
            s[i] = 0
            q = deque([i])
            while q:
                u = q.popleft()
                for v, c in adj[u]:
                    if s[v] == -1:
                        s[v] = s[u] ^ c
                        q.append(v)
                    elif s[v] != (s[u] ^ c):
                        print("NO")
                        return
                        
    print("YES")
    print(" ".join(map(str, s)))

if __name__ == '__main__':
    solve()
```
