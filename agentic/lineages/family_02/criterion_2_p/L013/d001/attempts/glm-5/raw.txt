import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        u = 2 * (abs(l1) - 1) + (1 if l1 > 0 else 0)
        v = 2 * (abs(l2) - 1) + (1 if l2 > 0 else 0)
        
        adj[u ^ 1].append(v)
        adj[v ^ 1].append(u)
    
    index = 0
    stack = []
    indices = [0] * (2 * n)
    lowlink = [0] * (2 * n)
    on_stack = [False] * (2 * n)
    comp_id = [-1] * (2 * n)
    current_comp = 0
    
    def strongconnect(v):
        nonlocal index, current_comp
        indices[v] = index
        lowlink[v] = index
        index += 1
        stack.append(v)
        on_stack[v] = True
        
        for w in adj[v]:
            if indices[w] == 0:
                strongconnect(w)
                lowlink[v] = min(lowlink[v], lowlink[w])
            elif on_stack[w]:
                lowlink[v] = min(lowlink[v], indices[w])
        
        if lowlink[v] == indices[v]:
            while True:
                w = stack.pop()
                on_stack[w] = False
                comp_id[w] = current_comp
                if w == v:
                    break
            current_comp += 1
    
    for v in range(2 * n):
        if indices[v] == 0:
            strongconnect(v)
    
    for i in range(n):
        if comp_id[2 * i] == comp_id[2 * i + 1]:
            print("NO")
            return
    
    assignment = [0] * n
    for i in range(n):
        if comp_id[2 * i] < comp_id[2 * i + 1]:
            assignment[i] = 0
        else:
            assignment[i] = 1
    
    print("YES")
    print(' '.join(map(str, assignment)))

solve()
