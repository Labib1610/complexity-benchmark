We can model each switch as a boolean variable and each rule as a clause that must not be violated.  
A rule with literals \( l_1 \) and \( l_2 \) triggers a fault only if both literals are false.  
So we must avoid having both literals false simultaneously.  

This is equivalent to a **2-SAT** problem:  
For each rule \((a \lor b)\), we add implications \((\neg a \rightarrow b)\) and \((\neg b \rightarrow a)\).  

We then run SCC (Tarjan or Kosaraju) to check satisfiability.  
If satisfiable, we assign values based on SCC topological order.

```python
import sys
sys.setrecursionlimit(10**6)
input = sys.stdin.readline

n, m = map(int, input().split())
N = 2 * n  # each variable x has nodes: 2*i for false, 2*i+1 for true

def node(var):
    # var: positive -> true node, negative -> false node
    idx = abs(var) - 1
    if var > 0:
        return 2 * idx + 1
    else:
        return 2 * idx

def neg_node(var):
    idx = abs(var) - 1
    if var > 0:
        return 2 * idx
    else:
        return 2 * idx + 1

g = [[] for _ in range(N)]
gr = [[] for _ in range(N)]

for _ in range(m):
    a, b = map(int, input().split())
    # clause (a or b) -> implications: not a -> b, not b -> a
    na = neg_node(a)
    nb = neg_node(b)
    u = node(a)
    v = node(b)
    # not a -> b
    g[na].append(v)
    gr[v].append(na)
    # not b -> a
    g[nb].append(u)
    gr[u].append(nb)

# Kosaraju SCC
visited = [False] * N
order = []

def dfs1(v):
    visited[v] = True
    for to in g[v]:
        if not visited[to]:
            dfs1(to)
    order.append(v)

for i in range(N):
    if not visited[i]:
        dfs1(i)

comp = [-1] * N

def dfs2(v, cid):
    comp[v] = cid
    for to in gr[v]:
        if comp[to] == -1:
            dfs2(to, cid)

cid = 0
for v in reversed(order):
    if comp[v] == -1:
        dfs2(v, cid)
        cid += 1

# Check satisfiability
for i in range(n):
    if comp[2*i] == comp[2*i+1]:
        print("NO")
        sys.exit()

print("YES")
res = [0] * n
for i in range(n):
    # If true node has higher comp id (later in topological order), assign true
    res[i] = 1 if comp[2*i+1] > comp[2*i] else 0

print(" ".join(map(str, res)))
```
