```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    clauses = []
    index = 2
    for _ in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        index += 2
        clauses.append((l1, l2))
    
    # For each clause (l1, l2), we want P(l1) + P(l2) <= 0
    # P(l) = (-1)^(s_|l| + delta(l))
    # So P(l1) + P(l2) <= 0 means it's not the case that both P(l1) and P(l2) are 1
    # Which means at least one of P(l1) or P(l2) must be -1
    
    # Let's define boolean variables for the 2-SAT problem
    # For each distributor i, we have two boolean variables:
    # x_i represents s_i = 0
    # y_i represents s_i = 1
    # But since s_i is either 0 or 1, we can just use one variable per distributor
    # Let's say variable i represents s_i = 0 (True) or s_i = 1 (False)
    # Or better yet, let's use standard 2-SAT formulation:
    # variable i (1-based) represents s_i = 0
    # variable -i represents s_i = 1
    
    # For each l, P(l) = (-1)^(s_|l| + delta(l))
    # If P(l) = 1, then s_|l| + delta(l) is even
    # If P(l) = -1, then s_|l| + delta(l) is odd
    
    # So for a clause (l1, l2), we want P(l1) + P(l2) <= 0
    # This is equivalent to NOT(P(l1) = 1 AND P(l2) = 1)
    
    # Let's build the 2-SAT clauses
    sat_clauses = []
    
    for l1, l2 in clauses:
        # We want NOT(P(l1) = 1 AND P(l2) = 1)
        # Which is (P(l1) != 1) OR (P(l2) != 1)
        
        # When is P(l) = 1?
        # P(l) = (-1)^(s_|l| + delta(l)) = 1 when s_|l| + delta(l) is even
        # That is, s_|l| = delta(l) (mod 2)
        
        # So P(l) = 1 iff s_|l| = delta(l)
        # P(l) != 1 iff s_|l| != delta(l)
        
        # For l1:
        abs_l1 = abs(l1)
        delta_l1 = 1 if l1 < 0 else 0
        # P(l1) != 1 iff s_|l1| != delta_l1
        # If delta_l1 = 0, then s_|l1| != 0, so s_|l1| = 1, which is variable -|l1|
        # If delta_l1 = 1, then s_|l1| != 1, so s_|l1| = 0, which is variable |l1|
        var1 = -abs_l1 if delta_l1 == 0 else abs_l1
        
        # For l2:
        abs_l2 = abs(l2)
        delta_l2 = 1 if l2 < 0 else 0
        # P(l2) != 1 iff s_|l2| != delta_l2
        var2 = -abs_l2 if delta_l2 == 0 else abs_l2
        
        sat_clauses.append((var1, var2))
    
    # Now we solve 2-SAT
    # Build implication graph
    # Each variable x has nodes x and -x
    # For clause (a OR b), we add edges (-a -> b) and (-b -> a)
    
    graph = defaultdict(list)
    reverse_graph = defaultdict(list)
    
    for a, b in sat_clauses:
        # Add (-a -> b) and (-b -> a)
        graph[-a].append(b)
        graph[-b].append(a)
        reverse_graph[b].append(-a)
        reverse_graph[a].append(-b)
    
    # Find all variables
    variables = set()
    for a, b in sat_clauses:
        variables.add(abs(a))
        variables.add(abs(b))
    
    # Kosaraju's algorithm for SCC
    def dfs1(u, visited, stack):
        visited.add(u)
        for v in graph[u]:
            if v not in visited:
                dfs1(v, visited, stack)
        stack.append(u)
    
    def dfs2(u, visited, component):
        visited.add(u)
        component.append(u)
        for v in reverse_graph[u]:
            if v not in visited:
                dfs2(v, visited, component)
    
    # First DFS to get finishing times
    visited = set()
    stack = []
    for var in variables:
        if var not in visited:
            dfs1(var, visited, stack)
        if -var not in visited:
            dfs1(-var, visited, stack)
    
    # Second DFS on transpose graph
    visited = set()
    scc = {}
    scc_index = 0
    while stack:
        u = stack.pop()
        if u not in visited:
            component = []
            dfs2(u, visited, component)
            for node in component:
                scc[node] = scc_index
            scc_index += 1
    
    # Check if satisfiable
    for var in variables:
        if scc[var] == scc[-var]:
            print("NO")
            return
    
    # Assign values
    # Variables in later finishing SCCs are assigned True first
    # But we need to be careful about the meaning of variables
    
    # In our encoding:
    # variable i represents s_i = 0
    # variable -i represents s_i = 1
    
    # If scc[i] > scc[-i], then variable i is true, meaning s_i = 0
    # If scc[-i] > scc[i], then variable -i is true, meaning s_i = 1
    
    result = [0] * (n + 1)  # 1-indexed
    for var in range(1, n + 1):
        if scc[var] > scc[-var]:
            result[var] = 0
        else:
            result[var] = 1
    
    print("YES")
    print(" ".join(map(str, result[1:])))

if __name__ == "__main__":
    main()
```
