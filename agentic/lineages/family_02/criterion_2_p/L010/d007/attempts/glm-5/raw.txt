import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n)]
    
    # For each check, we need to determine when it's unstable
    # Unstable: both swaps happen -> permutation (1 3 2) -> no particle in original position
    # Stable: at most one swap happens -> at least one particle in original position
    # So we need: NOT(both swaps happen) = NOT(swap1 AND swap2) = NOT(swap1) OR NOT(swap2)
    # swap1 happens when spin[|l1|] != target1
    # swap2 happens when spin[|l2|] != target2
    # NOT(swap1): spin matches target1
    # NOT(swap2): spin matches target2
    # So we need: spin matches target1 OR spin matches target2
    
    # For each check, add constraint: (spin matches target1) OR (spin matches target2)
    # Convert to 2-SAT: (A OR B) = NOT(NOT A AND NOT B)
    # Let x_i = (spin[i] == 1), so NOT x_i = (spin[i] == 0)
    # target = 1 means we want x_i, target = 0 means we want NOT x_i
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        q1 = abs(l1) - 1
        q2 = abs(l2) - 1
        t1 = 1 if l1 > 0 else 0
        t2 = 1 if l2 > 0 else 0
        
        # We need: (spin[q1] == t1) OR (spin[q2] == t2)
        # Clause: (x_q1 if t1==1 else NOT x_q1) OR (x_q2 if t2==1 else NOT x_q2)
        
        # Variable indices: 2*i for x_i, 2*i+1 for NOT x_i
        def var_idx(qubit, target):
            # Returns index for "spin[qubit] == target"
            if target == 1:
                return 2 * qubit  # x_qubit
            else:
                return 2 * qubit + 1  # NOT x_qubit
        
        a = var_idx(q1, t1)
        b = var_idx(q2, t2)
        
        # Add clause (a OR b): implications NOT a -> b and NOT b -> a
        not_a = a ^ 1
        not_b = b ^ 1
        adj[not_a].append(b)
        adj[not_b].append(a)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack.pop()
            if idx == 0:
                if visited[node]:
                    continue
                visited[node] = True
                stack.append((node, 1))
                for v in adj[node]:
                    if not visited[v]:
                        stack.append((v, 0))
            else:
                order.append(node)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Reverse graph
    radj = [[] for _ in range(2 * n)]
    for u in range(2 * n):
        for v in adj[u]:
            radj[v].append(u)
    
    comp = [-1] * (2 * n)
    comp_id = 0
    
    def dfs2(u, cid):
        stack = [u]
        while stack:
            node = stack.pop()
            if comp[node] != -1:
                continue
            comp[node] = cid
            for v in radj[node]:
                if comp[v] == -1:
                    stack.append(v)
    
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, comp_id)
            comp_id += 1
    
    # Check satisfiability
    for i in range(n):
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return
    
    # Build assignment
    assignment = [0] * n
    for i in range(n):
        if comp[2 * i] > comp[2 * i + 1]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(" ".join(map(str, assignment)))

solve()
