import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n * 2)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        x1 = abs(l1) - 1
        s1 = 1 if l1 > 0 else -1
        x2 = abs(l2) - 1
        s2 = 1 if l2 > 0 else -1
        
        # The condition R >= B means:
        # v1*s1 + v2*s2 >= (s1+s2)/2
        # v1, v2 in {0,1}. This is a 2-CNF problem.
        # The implication graph edges are added based on satisfying the inequality.
        
        if s1 == 1 and s2 == 1:
            # (v1 + v2 >= 1) -> at least one is 1.
            # ~v1 -> v2, ~v2 -> v1
            adj[x1].append(x2 + n)
            adj[x2].append(x1 + n)
        elif s1 == 1 and s2 == -1:
            # (v1 - v2 >= 0) -> v1 >= v2.
            # v2 -> v1, ~v1 -> ~v2
            adj[x2 + n].append(x1 + n)
            adj[x1].append(x2)
        elif s1 == -1 and s2 == 1:
            # (-v1 + v2 >= 0) -> v2 >= v1.
            # v1 -> v2, ~v2 -> ~v1
            adj[x1 + n].append(x2 + n)
            adj[x2].append(x1)
        else: # s1 == -1 and s2 == -1
            # (-v1 - v2 >= -1) -> v1 + v2 <= 1.
            # at most one is 1. v1 -> ~v2, v2 -> ~v1
            adj[x1 + n].append(x2)
            adj[x2 + n].append(x1)
            
    # Kosaraju's algorithm to find SCCs
    N = n * 2
    vis = [False] * N
    ord = []
    
    for i in range(N):
        if not vis[i]:
            stack = [(i, 0)]
            while stack:
                u, idx = stack.pop()
                if idx == 0:
                    if vis[u]:
                        continue
                    vis[u] = True
                    stack.append((u, 1))
                    for v in adj[u]:
                        if not vis[v]:
                            stack.append((v, 0))
                else:
                    ord.append(u)
                    
    comp = [-1] * N
    cadj = [[] for _ in range(N)]
    for u in range(N):
        for v in adj[u]:
            cadj[v].append(u)
            
    cid = 0
    for u in reversed(ord):
        if comp[u] == -1:
            stack = [u]
            comp[u] = cid
            while stack:
                cur = stack.pop()
                for v in cadj[cur]:
                    if comp[v] == -1:
                        comp[v] = cid
                        stack.append(v)
            cid += 1
            
    assignment = [0] * n
    for i in range(n):
        if comp[i] == comp[i + n]:
            print("NO")
            return
        assignment[i] = 1 if comp[i] > comp[i + n] else 0
        
    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    solve()
