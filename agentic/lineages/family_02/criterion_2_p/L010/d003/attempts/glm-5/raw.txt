import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        u = abs(l1) - 1
        v = abs(l2) - 1
        
        # Determine the state that would cause a fault for each switch
        # If l_i > 0, target is ON (1), fault if switch is OFF (0)
        # If l_i < 0, target is OFF (0), fault if switch is ON (1)
        s1 = 1 if l1 > 0 else 0
        s2 = 1 if l2 > 0 else 0
        
        # Implication: if u is in fault state s1, then v must NOT be in fault state s2
        # (u, s1) -> (v, 1-s2)
        adj[2*u + s1].append(2*v + (1-s2))
        # Contrapositive: (v, s2) -> (u, 1-s1)
        adj[2*v + s2].append(2*u + (1-s1))
    
    N = 2 * n
    visited = [False] * N
    order = []
    
    # First pass: DFS to get finish order
    for i in range(N):
        if not visited[i]:
            stack = [(i, 0)]
            while stack:
                node, idx = stack[-1]
                if not visited[node]:
                    visited[node] = True
                
                found_next = False
                while idx < len(adj[node]):
                    neighbor = adj[node][idx]
                    idx += 1
                    if not visited[neighbor]:
                        stack[-1] = (node, idx)
                        stack.append((neighbor, 0))
                        found_next = True
                        break
                if not found_next:
                    stack.pop()
                    order.append(node)
    
    # Second pass: process in reverse order on transposed graph
    visited = [False] * N
    comp_id = [-1] * N
    current_comp = 0
    
    # Build transposed graph
    radj = [[] for _ in range(N)]
    for u in range(N):
        for v in adj[u]:
            radj[v].append(u)
    
    for u in reversed(order):
        if not visited[u]:
            stack = [u]
            visited[u] = True
            component_nodes = []
            while stack:
                node = stack.pop()
                component_nodes.append(node)
                for neighbor in radj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        stack.append(neighbor)
            
            for node in component_nodes:
                comp_id[node] = current_comp
            current_comp += 1
    
    # Check for contradictions and build assignment
    assignment = [0] * n
    for i in range(n):
        if comp_id[2*i] == comp_id[2*i + 1]:
            print("NO")
            return
        assignment[i] = 1 if comp_id[2*i] > comp_id[2*i + 1] else 0
    
    print("YES")
    print(*(assignment))

if __name__ == "__main__":
    solve()
