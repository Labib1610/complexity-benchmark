import sys
sys.setrecursionlimit(1000000)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # For each variable x_i, we have two nodes: 2*i (for x_i = 0) and 2*i+1 (for x_i = 1)
    # But in 2-SAT, it's easier to index from 0. So:
    # variable i (1-based) -> nodes 2*(i-1) and 2*(i-1)+1
    # literal +i -> node 2*(i-1)+1 (True)
    # literal -i -> node 2*(i-1) (False)
    
    # We'll build implication graph and solve 2-SAT
    adj = [[] for _ in range(2 * n)]
    adj_rev = [[] for _ in range(2 * n)]
    
    index = 2
    clauses = []
    for i in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        index += 2
        clauses.append((l1, l2))
        
    # For each clause (l1 OR l2), we add edges:
    # NOT l1 -> l2
    # NOT l2 -> l1
    
    for l1, l2 in clauses:
        # Convert literals to node indices
        # l1: if positive, True node; if negative, False node
        if l1 > 0:
            node_l1_F = 2 * (l1 - 1)      # x_l1 = False
            node_l1_T = 2 * (l1 - 1) + 1  # x_l1 = True
        else:
            var1 = -l1
            node_l1_F = 2 * (var1 - 1) + 1  # NOT False = True
            node_l1_T = 2 * (var1 - 1)      # NOT True = False
            
        if l2 > 0:
            node_l2_F = 2 * (l2 - 1)
            node_l2_T = 2 * (l2 - 1) + 1
        else:
            var2 = -l2
            node_l2_F = 2 * (var2 - 1) + 1
            node_l2_T = 2 * (var2 - 1)
            
        # Add implications: NOT l1 -> l2, NOT l2 -> l1
        adj[node_l1_F].append(node_l2_T)
        adj[node_l2_F].append(node_l1_T)
        adj_rev[node_l2_T].append(node_l1_F)
        adj_rev[node_l1_T].append(node_l2_F)
        
    # Now solve 2-SAT using Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
        
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
            
    # Second DFS on transpose graph
    visited = [False] * (2 * n)
    components = [-1] * (2 * n)
    comp_num = 0
    
    def dfs2(u):
        visited[u] = True
        components[u] = comp_num
        for v in adj_rev[u]:
            if not visited[v]:
                dfs2(v)
                
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            comp_num += 1
            
    # Check if satisfiable
    for i in range(n):
        node_F = 2 * i
        node_T = 2 * i + 1
        if components[node_F] == components[node_T]:
            print("NO")
            return
            
    # Satisfiable, construct solution
    # Assign True to the literal whose node has higher component number
    assignment = [0] * n
    for i in range(n):
        node_F = 2 * i
        node_T = 2 * i + 1
        if components[node_T] > components[node_F]:
            assignment[i] = 1
        else:
            assignment[i] = 0
            
    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    main()
