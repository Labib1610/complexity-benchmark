import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    # 2-SAT: variables 1..n, indices 0..n-1
    # For each variable x: true node = 2*x, false node = 2*x+1
    # Total nodes: 2*n
    adj = [[] for _ in range(2 * n)]
    radj = [[] for _ in range(2 * n)]

    def get_node(var):
        # var is signed integer, 1 <= |var| <= n
        idx = abs(var) - 1
        if var > 0:
            return 2 * idx      # true node
        else:
            return 2 * idx + 1  # false node

    def get_neg(node):
        return node ^ 1

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        # Clause: (u) OR (v)
        # Implications: not u -> v, not v -> u
        nu = get_node(u)
        nv = get_node(v)
        u_node = get_node(-u) # not u
        v_node = get_node(-v) # not v
        
        adj[u_node].append(nv)
        adj[v_node].append(nu)
        radj[nv].append(u_node)
        radj[nu].append(v_node)

    # Kosaraju's algorithm
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, i = stack[-1]
            if not visited[node]:
                visited[node] = True
            
            if i < len(adj[node]):
                neighbor = adj[node][i]
                stack[-1] = (node, i + 1)
                if not visited[neighbor]:
                    stack.append((neighbor, 0))
            else:
                stack.pop()
                order.append(node)

    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)

    comp_id = [-1] * (2 * n)
    current_comp = 0
    
    def dfs2(u, cid):
        stack = [u]
        while stack:
            node = stack.pop()
            if comp_id[node] != -1:
                continue
            comp_id[node] = cid
            for neighbor in radj[node]:
                if comp_id[neighbor] == -1:
                    stack.append(neighbor)

    for u in reversed(order):
        if comp_id[u] == -1:
            dfs2(u, current_comp)
            current_comp += 1

    assignment = [0] * n
    for i in range(n):
        true_node = 2 * i
        false_node = 2 * i + 1
        if comp_id[true_node] == comp_id[false_node]:
            print("NO")
            return
        if comp_id[true_node] > comp_id[false_node]:
            assignment[i] = 1
        else:
            assignment[i] = 0

    print("YES")
    print(*(assignment))

if __name__ == "__main__":
    solve()
