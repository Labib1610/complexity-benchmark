```python
import sys

sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = 2 * n
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def lit(x: int) -> int:
        """return node index for literal described by signed integer x"""
        v = abs(x) - 1
        if x > 0:          # trigger when xi = 0  -> literal ¬xi
            return 2 * v          # false node
        else:              # trigger when xi = 1  -> literal xi
            return 2 * v + 1      # true node

    def neg(u: int) -> int:
        return u ^ 1

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        A = lit(a)
        B = lit(b)
        # clause (¬A ∨ ¬B)  => implications A -> ¬B , B -> ¬A
        na = neg(A)
        nb = neg(B)
        g[A].append(nb)
        rg[nb].append(A)
        g[B].append(na)
        rg[na].append(B)

    # Kosaraju
    visited = [False] * N
    order = []

    sys.setrecursionlimit(1 << 25)

    def dfs(v: int):
        stack = [(v, 0)]
        while stack:
            node, idx = stack.pop()
            if idx == 0:
                if visited[node]:
                    continue
                visited[node] = True
                stack.append((node, 1))  # post processing marker
                for to in g[node]:
                    if not visited[to]:
                        stack.append((to, 0))
            else:
                order.append(node)

    for v in range(N):
        if not visited[v]:
            dfs(v)

    comp = [-1] * N
    cur = 0
    def rdfs(v: int, label: int):
        stack = [v]
        while stack:
            node = stack.pop()
            if comp[node] != -1:
                continue
            comp[node] = label
            for to in rg[node]:
                if comp[to] == -1:
                    stack.append(to)

    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, cur)
            cur += 1

    assignment = [0] * n
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return
        # component with larger id comes earlier in topological order
        assignment[i] = 1 if comp[2*i+1] > comp[2*i] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
