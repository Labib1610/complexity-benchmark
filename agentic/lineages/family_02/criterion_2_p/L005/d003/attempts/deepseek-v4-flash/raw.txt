We can model each qubit’s state as a boolean variable, and each route as an implication constraint.  

A route `l1 -> l2` means:  
- If `l1` fires, then `l2` must not fire.  
- Firing conditions:  
  - `+x` fires when qubit `x` is `0`.  
  - `-x` fires when qubit `x` is `1`.  

So the constraint becomes:  
If `l1` is true, then `l2` must be false.  
That is equivalent to: `(not l1) OR (not l2)`.  

This is a 2-SAT problem. We can solve it with implication graph and SCC.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(1 << 20)
input = sys.stdin.readline

n, m = map(int, input().split())
N = 2 * n  # each variable has two nodes: x and not x

# node mapping: 
# variable x (1-indexed) -> node for x: 2*(x-1)
# node for not x: 2*(x-1)+1

g = [[] for _ in range(N)]
gr = [[] for _ in range(N)]

def add_imp(u, v):
    g[u].append(v)
    gr[v].append(u)

def node(var):
    # var is positive for x, negative for not x
    if var > 0:
        return 2 * (var - 1)
    else:
        return 2 * (-var - 1) + 1

def neg_node(u):
    return u ^ 1

for _ in range(m):
    l1, l2 = map(int, input().split())
    # (l1 fires) => (l2 does not fire)
    # l1 => not l2
    # not l2 => not l1
    u = node(l1)
    v = node(l2)
    # u => neg(v)
    add_imp(u, neg_node(v))
    # v => neg(u)
    add_imp(v, neg_node(u))

# Kosaraju SCC
visited = [False] * N
order = []

def dfs1(u):
    visited[u] = True
    for w in g[u]:
        if not visited[w]:
            dfs1(w)
    order.append(u)

for i in range(N):
    if not visited[i]:
        dfs1(i)

comp = [-1] * N
def dfs2(u, cid):
    comp[u] = cid
    for w in gr[u]:
        if comp[w] == -1:
            dfs2(w, cid)

cid = 0
for u in reversed(order):
    if comp[u] == -1:
        dfs2(u, cid)
        cid += 1

# Check satisfiability
for i in range(n):
    if comp[2*i] == comp[2*i+1]:
        print("NO")
        sys.exit()

# Assign values
assignment = [False] * n
for i in range(n):
    # If not x comes after x in topological order, x is true
    assignment[i] = comp[2*i] > comp[2*i+1]

print("YES")
print(' '.join('1' if assignment[i] else '0' for i in range(n)))
```
