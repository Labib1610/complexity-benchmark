```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = 2 * n
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def node(var: int, val: int) -> int:
        # var 1-indexed, val 0 or 1
        return 2 * (var - 1) + val

    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))

        var1 = abs(l1)
        var2 = abs(l2)

        # literal a: true if condition holds
        a = node(var1, 0) if l1 > 0 else node(var1, 1)
        b = node(var2, 0) if l2 > 0 else node(var2, 1)

        na = a ^ 1  # negation
        nb = b ^ 1

        # a -> ¬b , b -> ¬a
        g[a].append(nb)
        rg[nb].append(a)
        g[b].append(na)
        rg[na].append(b)

    sys.setrecursionlimit(1 << 25)
    visited = [False] * N
    order = []

    def dfs(v: int):
        stack = [(v, 0)]
        while stack:
            v, state = stack.pop()
            if state == 0:
                if visited[v]:
                    continue
                visited[v] = True
                stack.append((v, 1))
                for to in g[v]:
                    if not visited[to]:
                        stack.append((to, 0))
            else:
                order.append(v)

    for v in range(N):
        if not visited[v]:
            dfs(v)

    comp = [-1] * N
    cid = 0

    def rdfs(v: int, label: int):
        stack = [v]
        while stack:
            v = stack.pop()
            if comp[v] != -1:
                continue
            comp[v] = label
            for to in rg[v]:
                if comp[to] == -1:
                    stack.append(to)

    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, cid)
            cid += 1

    assignment = [0] * n
    for i in range(1, n + 1):
        if comp[node(i, 0)] == comp[node(i, 1)]:
            print("NO")
            return
        # component with larger id comes earlier in topological order
        assignment[i - 1] = 1 if comp[node(i, 1)] > comp[node(i, 0)] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
