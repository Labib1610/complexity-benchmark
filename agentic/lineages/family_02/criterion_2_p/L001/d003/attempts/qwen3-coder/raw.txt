```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each assessment (a or b), we must satisfy at least one condition
    # This is a 2-SAT problem
    # Variable x_i represents hub i being ACTIVE (1) or DORMANT (0)
    # For each clause (l1 or l2), we add edges to the implication graph
    
    # Implication graph: for each variable, we have two nodes: x and ~x
    # Node ids: 0 to 2*n-1 where 2*i represents x_i (ACTIVE) and 2*i+1 represents ~x_i (DORMANT)
    
    adj = defaultdict(list)
    rev_adj = defaultdict(list)
    
    for _ in range(m):
        l1, l2 = map(int, sys.stdin.readline().split())
        
        # l1 and l2 are literals
        # Convert to node ids
        # Positive literal k means hub k is ACTIVE, corresponds to node 2*(k-1)
        # Negative literal -k means hub k is DORMANT, corresponds to node 2*(k-1)+1
        
        node1 = 2*(abs(l1)-1) + (0 if l1 > 0 else 1)
        node2 = 2*(abs(l2)-1) + (0 if l2 > 0 else 1)
        
        # Clause (l1 or l2) is equivalent to (~l1 -> l2) and (~l2 -> l1)
        # In terms of nodes: (node1^1 -> node2) and (node2^1 -> node1)
        # node^1 flips the last bit, so it's the negation of the node
        
        adj[node1^1].append(node2)
        adj[node2^1].append(node1)
        rev_adj[node2].append(node1^1)
        rev_adj[node1].append(node2^1)
    
    # Kosaraju's algorithm to find strongly connected components
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Second DFS on reverse graph
    visited = [False] * (2 * n)
    scc = [-1] * (2 * n)
    scc_id = 0
    
    def dfs2(u):
        visited[u] = True
        scc[u] = scc_id
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            scc_id += 1
    
    # Check if satisfiable
    for i in range(n):
        # Variable i corresponds to nodes 2*i and 2*i+1
        if scc[2*i] == scc[2*i+1]:
            print("NO")
            return
    
    # Assign values
    assignment = [-1] * n
    
    # Process SCCs in reverse topological order
    for i in range(2 * n):
        node = i
        var = node // 2
        is_neg = node % 2
        
        if assignment[var] == -1:
            # If node represents ACTIVE (is_neg == 0) and has higher SCC id,
            # then we set the variable to ACTIVE
            # If node represents DORMANT (is_neg == 1) and has higher SCC id,
            # then we set the variable to DORMANT
            assignment[var] = 1 - is_neg
    
    print("YES")
    print(" ".join(map(str, assignment)))

solve()
```
