import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    
    adj = [[] for _ in range(n)]
    
    idx = 2
    for _ in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx+1])
        idx += 2
        
        u = abs(l1) - 1
        v = abs(l2) - 1
        
        # We want to add an edge between u and v such that if the edge is satisfied, 
        # at least one of the parameters is aligned.
        # A parameter v is aligned if (v > 0 and s_{|v|} == 1) or (v < 0 and s_{|v|} == 0).
        # This is equivalent to s_{|v|} == (v > 0).
        # So we want s_u == (l1 > 0) OR s_v == (l2 > 0).
        # This is equivalent to NOT(s_u != (l1 > 0) AND s_v != (l2 > 0)).
        # Let x_u = (s_u != (l1 > 0)). Then we want NOT(x_u AND x_v), which is x_u -> NOT x_v.
        # This is a 2-SAT implication. We add edges x_u -> NOT x_v and x_v -> NOT x_u.
        # In the implication graph, x_u is represented by 2*u and NOT x_u by 2*u+1.
        
        u_node = 2 * u + (1 if l1 > 0 else 0)
        v_node = 2 * v + (1 if l2 > 0 else 0)
        
        adj[u_node ^ 1].append(v_node)
        adj[v_node ^ 1].append(u_node)
        
    # Kosaraju's algorithm to find SCCs
    sys.setrecursionlimit(3000000)
    order = []
    visited = [False] * (2 * n)
    
    for i in range(2 * n):
        if not visited[i]:
            stack = [(i, False)]
            while stack:
                node, processed = stack.pop()
                if processed:
                    order.append(node)
                    continue
                if visited[node]:
                    continue
                visited[node] = True
                stack.append((node, True))
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        stack.append((neighbor, False))
                        
    radj = [[] for _ in range(2 * n)]
    for u in range(2 * n):
        for v in adj[u]:
            radj[v].append(u)
            
    scc_id = [-1] * (2 * n)
    current_id = 0
    
    for i in range(2 * n - 1, -1, -1):
        node = order[i]
        if scc_id[node] == -1:
            stack = [node]
            scc_id[node] = current_id
            while stack:
                curr = stack.pop()
                for neighbor in radj[curr]:
                    if scc_id[neighbor] == -1:
                        scc_id[neighbor] = current_id
                        stack.append(neighbor)
            current_id += 1
            
    for i in range(n):
        if scc_id[2 * i] == scc_id[2 * i + 1]:
            print("NO")
            return
            
    assignment = [0] * n
    for i in range(n):
        # 2*i is x_i (s_i != (l1 > 0)), 2*i+1 is NOT x_i (s_i == (l1 > 0))
        # We want s_i = 1. If NOT x_i is true, s_i = 1. If x_i is true, s_i = 0.
        # So s_i = NOT x_i.
        # In 2-SAT, we pick the literal with the larger scc_id.
        # If 2*i+1 has larger scc_id, NOT x_i is true, so s_i = 1.
        # If 2*i has larger scc_id, x_i is true, so s_i = 0.
        assignment[i] = 1 if scc_id[2 * i + 1] > scc_id[2 * i] else 0
        
    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == '__main__':
    solve()
