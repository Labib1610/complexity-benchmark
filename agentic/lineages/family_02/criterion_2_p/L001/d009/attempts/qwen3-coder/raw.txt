import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    clauses = []
    index = 2
    for _ in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        index += 2
        clauses.append((l1, l2))
    
    # Create implication graph for 2-SAT
    # Variables: x1, x2, ..., xn
    # xi = 1 means si = 1, xi = 0 means si = 0
    # For each clause (l1 or l2), we add implications:
    # not l1 -> l2 and not l2 -> l1
    
    # Node numbering: 
    # For variable i (1-based): 
    # Node 2*i represents xi (variable i is True)
    # Node 2*i+1 represents not xi (variable i is False)
    
    graph = defaultdict(list)
    reverse_graph = defaultdict(list)
    
    def var_to_node(v):
        # v is +i or -i
        i = abs(v)
        if v > 0:
            return 2 * i  # xi
        else:
            return 2 * i + 1  # not xi
    
    def node_to_var(node):
        i = node // 2
        if node % 2 == 0:
            return i  # xi node
        else:
            return -i  # not xi node
    
    # Add implications for each clause
    for l1, l2 in clauses:
        # Clause (l1 or l2) is equivalent to (not l1 -> l2) and (not l2 -> l1)
        # not l1 -> l2
        not_l1 = var_to_node(-l1)
        l2_node = var_to_node(l2)
        graph[not_l1].append(l2_node)
        reverse_graph[l2_node].append(not_l1)
        
        # not l2 -> l1
        not_l2 = var_to_node(-l2)
        l1_node = var_to_node(l1)
        graph[not_l2].append(l1_node)
        reverse_graph[l1_node].append(not_l2)
    
    # Kosaraju's algorithm to find SCCs
    visited = [False] * (2 * (n + 1))
    order = []
    
    def dfs1(node):
        visited[node] = True
        for neighbor in graph[node]:
            if not visited[neighbor]:
                dfs1(neighbor)
        order.append(node)
    
    # Run DFS1 on all nodes
    for i in range(2, 2 * (n + 1)):
        if not visited[i]:
            dfs1(i)
    
    # Reverse the graph and run DFS in reverse order
    visited = [False] * (2 * (n + 1))
    scc_assignment = [-1] * (2 * (n + 1))
    scc_id = 0
    
    def dfs2(node):
        visited[node] = True
        scc_assignment[node] = scc_id
        for neighbor in reverse_graph[node]:
            if not visited[neighbor]:
                dfs2(neighbor)
    
    while order:
        node = order.pop()
        if not visited[node]:
            dfs2(node)
            scc_id += 1
    
    # Check if satisfiable
    for i in range(1, n + 1):
        xi = 2 * i       # node for xi
        not_xi = 2 * i + 1  # node for not xi
        if scc_assignment[xi] == scc_assignment[not_xi]:
            print("NO")
            return
    
    # Assign values based on SCC topological order
    # In Kosaraju, later SCCs should be assigned False first
    assignment = [0] * (n + 1)
    
    # Create mapping from SCC id to list of nodes
    scc_nodes = defaultdict(list)
    for node in range(2, 2 * (n + 1)):
        scc_nodes[scc_assignment[node]].append(node)
    
    # Process SCCs in reverse topological order (reverse of finishing times in dfs1)
    # We can process them in order of scc_id since Kosaraju gives topological order
    for scc in range(scc_id):
        for node in scc_nodes[scc]:
            var = node_to_var(node)
            if var == 0:
                continue
            idx = abs(var)
            if assignment[idx] == 0:  # not yet assigned
                if var > 0:
                    assignment[idx] = 1
                else:
                    assignment[idx] = 0
    
    print("YES")
    result = []
    for i in range(1, n + 1):
        result.append(str(assignment[i]))
    print(" ".join(result))

if __name__ == "__main__":
    main()
