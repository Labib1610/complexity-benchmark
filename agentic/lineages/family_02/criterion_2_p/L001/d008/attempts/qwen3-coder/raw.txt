import sys
from collections import defaultdict

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    assessments = []
    for _ in range(m):
        l1, l2 = map(int, sys.stdin.readline().split())
        assessments.append((l1, l2))
    return n, m, assessments

def encode_phase(v, s):
    """Returns how parameter v interacts with phase s"""
    if v > 0:
        return 1 if s == 0 else 0
    else:  # v < 0
        return 1 if s == 1 else 0

def compute_Z(l1, l2, phases):
    """Compute diagnostic code Z for an assessment"""
    Z = 0
    for v in [l1, l2]:
        hub_index = abs(v) - 1  # Convert to 0-based index
        phase = phases[hub_index]
        if encode_phase(v, phase) == 1:
            Z = 2 * Z + 1
        # else Z remains unchanged
    return Z

def is_stable(Z):
    """Check if diagnostic code is stable"""
    return Z < 3

def solve(n, m, assessments):
    if m == 0:
        return "YES\n" + " ".join(["0"] * n)
    
    # Try all 2^n combinations for small n, but since n can be up to 100000,
    # we need a more efficient approach.
    
    # Let's think about constraints:
    # For each assessment (l1, l2), we want Z < 3.
    # Z is computed as follows:
    # Start with Z=0
    # Process l1: if encode_phase(l1, s_|l1|) == 1, then Z = 2*Z + 1
    # Process l2: if encode_phase(l2, s_|l2|) == 1, then Z = 2*Z + 1
    
    # So Z can be 0, 1, 2, 3, ... depending on which steps increment it.
    # To keep Z < 3, we must avoid having too many increments.
    
    # Let's consider what values of encode_phase lead to Z >= 3:
    # If first step increments (Z becomes 1) and second increments (Z becomes 3) -> Z=3 (bad)
    # If first step doesn't increment (Z stays 0) and second increments (Z becomes 1) -> Z=1 (ok)
    # If first step increments (Z becomes 1) and second doesn't (Z stays 1) -> Z=1 (ok)
    # If neither increments -> Z=0 (ok)
    
    # So for each assessment, we want:
    # NOT (encode_phase(l1, s_|l1|) == 1 AND encode_phase(l2, s_|l2|) == 1)
    
    # This gives us a 2-SAT problem:
    # For each assessment (l1, l2), we want:
    # NOT (x_|l1| == (l1>0 ? 0 : 1) AND x_|l2| == (l2>0 ? 0 : 1))
    # Which is equivalent to:
    # (x_|l1| != (l1>0 ? 0 : 1)) OR (x_|l2| != (l2>0 ? 0 : 1))
    
    # In terms of boolean variables:
    # Let x_i be the boolean variable for s_i (0/1 phase)
    # For l1, if l1 > 0: x_|l1| should NOT be 0 to avoid increment
    #          if l1 < 0: x_|l1| should NOT be 1 to avoid increment
    # Same for l2
    
    # Actually, let's re-examine the logic:
    # If v > 0 and s_|v| = 0 => increment
    # If v > 0 and s_|v| = 1 => no increment
    # If v < 0 and s_|v| = 1 => increment  
    # If v < 0 and s_|v| = 0 => no increment
    
    # So for an assessment (l1, l2) to be stable (Z < 3):
    # We cannot have both l1 and l2 cause an increment.
    
    # For l1 to NOT cause increment:
    # if l1 > 0: s_|l1| must be 1
    # if l1 < 0: s_|l1| must be 0
    # 
    # For l2 to NOT cause increment:
    # if l2 > 0: s_|l2| must be 1
    # if l2 < 0: s_|l2| must be 0
    
    # But this is getting complex. Let's just simulate.
    
    # Key insight: Z can only be 0, 1, 2, 3, ...
    # After processing l1:
    # - If increment: Z = 1
    # - If no increment: Z = 0
    # Then processing l2:
    # - If Z was 0 and increment: Z = 1
    # - If Z was 0 and no increment: Z = 0
    # - If Z was 1 and increment: Z = 3
    # - If Z was 1 and no increment: Z = 1
    #
    # So Z >= 3 happens only when both l1 and l2 cause increments.
    
    # Therefore, for each assessment, we must ensure that NOT both l1 and l2 cause increments.
    # That is: (l1 does not cause increment) OR (l2 does not cause increment)
    
    # When does a parameter v cause an increment?
    # v > 0 and s_|v| = 0 => increment
    # v < 0 and s_|v| = 1 => increment
    # 
    # So v does NOT cause increment when:
    # v > 0 and s_|v| = 1
    # v < 0 and s_|v| = 0
    
    # This gives us 2-SAT clauses:
    # For each assessment (l1, l2), we want:
    # (l1 does NOT cause increment) OR (l2 does NOT cause increment)
    # 
    # Let's define boolean variables:
    # x_i corresponds to s_i = 1 (True means phase 1)
    # 
    # l1 does NOT cause increment:
    # if l1 > 0: x_|l1| must be True
    # if l1 < 0: x_|l1| must be False
    # 
    # l2 does NOT cause increment:
    # if l2 > 0: x_|l2| must be True  
    # if l2 < 0: x_|l2| must be False
    #
    # So clause is:
    # (l1 > 0 ? x_|l1| : NOT x_|l1|) OR (l2 > 0 ? x_|l2| : NOT x_|l2|)
    
    # Build 2-SAT instance
    # We'll use implication graph approach
    
    # Nodes: 0..n-1 for x_i, n..2n-1 for NOT x_i
    # But let's use a simpler representation
    
    from collections import defaultdict
    
    def two_sat(n, clauses):
        # clauses is list of [(a, nota, b, notb), ...] meaning (a OR b) where a = xa if not nota else NOT xa
        adj = defaultdict(list)
        rev_adj = defaultdict(list)
        
        # Create implication graph
        # Node i represents variable xi
        # Node i+n represents NOT xi
        
        def add_implication(a, nota, b, notb):
            # If (nota ? NOT a : a) => (notb ? NOT b : b)
            # This is equivalent to (NOT (nota ? NOT a : a)) OR (notb ? NOT b : b)
            # Which is ((nota ? a : NOT a)) OR (notb ? NOT b : b)
            
            # But we're building implications for SCC algorithm
            # a => b is equivalent to NOT a OR b
            
            # For clause (a OR b), we add edges:
            # NOT a => b
            # NOT b => a
            
            node_a = a if not nota else a + n
            node_na = a + n if not nota else a
            node_b = b if not notb else b + n  
            node_nb = b + n if not notb else b
            
            # NOT a => b means node_na => node_b
            adj[node_na].append(node_b)
            rev_adj[node_b].append(node_na)
            
            # NOT b => a means node_nb => node_a
            adj[node_nb].append(node_a)
            rev_adj[node_a].append(node_nb)
        
        for a, nota, b, notb in clauses:
            add_implication(a, nota, b, notb)
            
        # Kosaraju's algorithm for SCC
        visited = [False] * (2 * n)
        order = []
        component = [-1] * (2 * n)
        
        def dfs1(u):
            visited[u] = True
            for v in adj[u]:
                if not visited[v]:
                    dfs1(v)
            order.append(u)
            
        def dfs2(u, comp):
            component[u] = comp
            for v in rev_adj[u]:
                if component[v] == -1:
                    dfs2(v, comp)
                    
        # First DFS pass
        for i in range(2 * n):
            if not visited[i]:
                dfs1(i)
                
        # Second DFS pass in reverse order
        comp_id = 0
        for u in reversed(order):
            if component[u] == -1:
                dfs2(u, comp_id)
                comp_id += 1
                
        # Check if satisfiable
        for i in range(n):
            if component[i] == component[i + n]:
                return None  # Unsatisfiable
                
        # Extract solution
        assignment = [False] * n
        comp_to_assignment = {}
        
        # Process components in topological order
        comp_order = list(range(comp_id))
        comp_nodes = defaultdict(list)
        for i in range(2 * n):
            comp_nodes[component[i]].append(i)
            
        for c in sorted(comp_order, key=lambda x: len(comp_nodes[x]), reverse=True):
            val = True
            for node in comp_nodes[c]:
                var_idx = node % n
                is_neg = node >= n
                if var_idx in comp_to_assignment:
                    continue
                comp_to_assignment[var_idx] = val if not is_neg else not val
                
        for i in range(n):
            assignment[i] = comp_to_assignment.get(i, False)
            
        return assignment
    
    # Build clauses
    clauses = []
    for l1, l2 in assessments:
        # We want (l1 does NOT cause increment) OR (l2 does NOT cause increment)
        # l1 does NOT cause increment:
        #   if l1 > 0: s_|l1| = 1
        #   if l1 < 0: s_|l1| = 0
        # l2 does NOT cause increment:
        #   if l2 > 0: s_|l2| = 1
        #   if l2 < 0: s_|l2| = 0
        
        # Variables are 0-indexed
        var_l1 = abs(l1) - 1
        var_l2 = abs(l2) - 1
        
        # Does l1 cause increment?
        # l1 > 0 and s_|l1| = 0 => increment
        # l1 < 0 and s_|l1| = 1 => increment
        # So l1 does NOT cause increment when:
        # l1 > 0 and s_|l1| = 1  i.e. x_var_l1 = True
        # l1 < 0 and s_|l1| = 0  i.e. x_var_l1 = False
        
        # Clause: (x_var_l1 = (l1>0)) OR (x_var_l2 = (l2>0))
        # In 2-SAT: (x_var_l1 OR x_var_l2) if both should be True
        # But actually we want:
        # (l1 does NOT cause increment) OR (l2 does NOT cause increment)
        # 
        # l1 does NOT cause increment:
        #   if l1 > 0: x_var_l1 should be True
        #   if l1 < 0: x_var_l1 should be False
        #   
        # In clause form for 2-SAT solver:
        # (var_l1, l1<0, var_l2, l2<0)
        # This means: (x_var_l1 if not (l1<0) else NOT x_var_l1) OR (x_var_l2 if not (l2<0) else NOT x_var_l2)
        # Which is exactly what we want!
        
        clauses.append((var_l1, l1 < 0, var_l2, l2 < 0))
        
    try:
        result = two_sat(n, clauses)
        if result is None:
            return "NO"
        else:
            return "YES\n" + " ".join(str(int(b)) for b in result)
    except Exception:
        # Fallback for implementation issues
        pass
    
    # Simplified approach for small cases or debugging
    # Since 2-SAT might be overkill, let's try greedy for simple cases
    
    # Actually, let's implement a proper 2-SAT
    class TwoSAT:
        def __init__(self, n):
            self.n = n
            self.adj = defaultdict(list)
            self.rev_adj = defaultdict(list)
            
        def add_clause(self, a, nota, b, notb):
            # Add clause (a OR b) where a = xa if not nota else NOT xa
            # Convert to implications: NOT a => b and NOT b => a
            
            # Node IDs: 0..n-1 for x_i, n..2n-1 for NOT x_i
            node_a = a if not nota else a + self.n
            node_na = a + self.n if not nota else a
            node_b = b if not notb else b + self.n
            node_nb = b + self.n if not notb else b
            
            # NOT a => b
            self.adj[node_na].append(node_b)
            self.rev_adj[node_b].append(node_na)
            
            # NOT b => a
            self.adj[node_nb].append(node_a)
            self.rev_adj[node_a].append(node_nb)
            
        def solve(self):
            # Kosaraju's algorithm
            visited = [False] * (2 * self.n)
            order = []
            component = [-1] * (2 * self.n)
            
            def dfs1(u):
                visited[u] = True
                for v in self.adj[u]:
                    if not visited[v]:
                        dfs1(v)
                order.append(u)
                
            def dfs2(u, comp):
                component[u] = comp
                for v in self.rev_adj[u]:
                    if component[v] == -1:
                        dfs2(v, comp)
                        
            # First pass
            for i in range(2 * self.n):
                if not visited[i]:
                    dfs1(i)
                    
            # Second pass
            comp_id = 0
            for u in reversed(order):
                if component[u] == -1:
                    dfs2(u, comp_id)
                    comp_id += 1
                    
            # Check satisfiability
            for i in range(self.n):
                if component[i] == component[i + self.n]:
                    return None  # Unsatisfiable
                    
            # Build assignment
            assignment = [False] * self.n
            # Assign values based on component numbers - higher component number means later in topological sort
            comp_assign = {}
            for i in range(2 * self.n):
                var = i % self.n
                if var not in comp_assign:
                    comp_assign[var] = component[i] < component[i + self.n] if i < self.n else not (component[i - self.n] < component[i])
                    
            for i in range(self.n):
                assignment[i] = component[i] > component[i + self.n]
                
            return assignment
    
    # Use our 2-SAT solver
    sat = TwoSAT(n)
    for l1, l2 in assessments:
        var_l1 = abs(l1) - 1
        var_l2 = abs(l2) - 1
        # Add clause: (l1 does NOT cause increment) OR (l2 does NOT cause increment)
        # l1 does NOT cause increment: 
        #   l1 > 0 and s_|l1| = 1  i.e. x_var_l1 = True
        #   l1 < 0 and s_|l1| = 0  i.e. x_var_l1 = False
        # This is captured by (x_var_l1 if l1>0 else NOT x_var_l1)
        sat.add_clause(var_l1, l1 < 0, var_l2, l2 < 0)
        
    result = sat.solve()
    if result is None:
        return "NO"
    else:
        return "YES\n" + " ".join(str(int(b)) for b in result)

def main():
    n, m, assessments = read_input()
    print(solve(n, m, assessments))

if __name__ == "__main__":
    main()
