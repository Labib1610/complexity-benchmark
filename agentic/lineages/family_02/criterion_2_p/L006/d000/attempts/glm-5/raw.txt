import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    # 2-SAT: variables x_1..x_n, nodes 1..2n
    # node i: x_i = True, node i+n: x_i = False
    size = 2 * n + 1
    head = [-1] * size
    to = []
    nxt = []

    def add_edge(u, v):
        to.append(v)
        nxt.append(head[u])
        head[u] = len(to) - 1

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        # literal a: if a>0 -> x_a True, if a<0 -> x_|a| False
        # clause (a or b): add edges not(a)->b, not(b)->a
        if a > 0:
            na = a + n
            la = a
        else:
            na = -a
            la = -a + n
        if b > 0:
            nb = b + n
            lb = b
        else:
            nb = -b
            lb = -b + n
        add_edge(na, lb)
        add_edge(nb, la)

    # Tarjan's SCC iteratively
    index_counter = [0]
    stack = []
    on_stack = [False] * size
    indices = [-1] * size
    lowlink = [0] * size
    scc_id = [-1] * size
    scc_count = [0]

    # iterative DFS
    for start in range(1, size):
        if indices[start] != -1:
            continue
        call_stack = [(start, head[start], 0)]
        while call_stack:
            node, edge, state = call_stack[-1]
            if state == 0:
                indices[node] = index_counter[0]
                lowlink[node] = index_counter[0]
                index_counter[0] += 1
                stack.append(node)
                on_stack[node] = True
                call_stack[-1] = (node, edge, 1)
            if state == 1:
                if edge != -1:
                    w = to[edge]
                    if indices[w] == -1:
                        call_stack[-1] = (node, head[node], 2)
                        call_stack.append((w, head[w], 0))
                        continue
                    elif on_stack[w]:
                        if indices[w] < lowlink[node]:
                            lowlink[node] = indices[w]
                    call_stack[-1] = (node, nxt[edge], 1)
                    continue
                else:
                    # finished all edges
                    if lowlink[node] == indices[node]:
                        while True:
                            w = stack.pop()
                            on_stack[w] = False
                            scc_id[w] = scc_count[0]
                            if w == node:
                                break
                        scc_count[0] += 1
                    call_stack.pop()
                    if call_stack:
                        parent = call_stack[-1][0]
                        if lowlink[node] < lowlink[parent]:
                            lowlink[parent] = lowlink[node]
            else:
                # state == 2, returned from child
                if lowlink[node] < lowlink[parent]:
                    lowlink[parent] = lowlink[node]
                call_stack[-1] = (node, nxt[edge], 1)

    # check satisfiability
    for i in range(1, n + 1):
        if scc_id[i] == scc_id[i + n]:
            print("NO")
            return

    # assign values: x_i = True if scc_id[i] > scc_id[i+n] (topologically later)
    assignment = [0] * (n + 1)
    for i in range(1, n + 1):
        if scc_id[i] > scc_id[i + n]:
            assignment[i] = 1
        else:
            assignment[i] = 0

    print("YES")
    print(' '.join(str(assignment[i]) for i in range(1, n + 1)))

solve()
