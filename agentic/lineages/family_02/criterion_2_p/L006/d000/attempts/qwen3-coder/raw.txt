import sys
from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # Build implication graph for 2-SAT
    # Variables: 0 to n-1 for species 1 to n in humid (1)
    #            n to 2n-1 for species 1 to n in arid (0)
    
    # For literal l:
    # if l > 0: variable is (l-1) for humid
    # if l < 0: variable is (|l|-1)+n for arid
    
    def var_humid(i):
        return i - 1
    
    def var_arid(i):
        return (i - 1) + n
    
    # Adjacency list for implication graph
    adj = defaultdict(list)
    adj_rev = defaultdict(list)
    
    idx = 2
    for _ in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx+1])
        idx += 2
        
        # Clause (l1 OR l2) is equivalent to (~l1 -> l2) AND (~l2 -> l1)
        
        # Determine the nodes for l1 and l2 and their negations
        if l1 > 0:
            node_l1 = var_humid(l1)      # species l1 in humid
            node_not_l1 = var_arid(l1)   # species l1 in arid
        else:
            node_l1 = var_arid(-l1)      # species |l1| in arid
            node_not_l1 = var_humid(-l1) # species |l1| in humid
            
        if l2 > 0:
            node_l2 = var_humid(l2)      # species l2 in humid
            node_not_l2 = var_arid(l2)   # species l2 in arid
        else:
            node_l2 = var_arid(-l2)      # species |l2| in arid
            node_not_l2 = var_humid(-l2) # species |l2| in humid
            
        # Add implications: (~l1 -> l2) and (~l2 -> l1)
        adj[node_not_l1].append(node_l2)
        adj[node_not_l2].append(node_l1)
        adj_rev[node_l2].append(node_not_l1)
        adj_rev[node_l1].append(node_not_l2)
    
    # Kosaraju's algorithm to find SCCs
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Second DFS on transpose graph
    visited = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in adj_rev[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            scc_count += 1
    
    # Check if any variable and its negation are in the same SCC
    for i in range(n):
        if scc_id[i] == scc_id[i + n]:
            print("NO")
            return
    
    # Assign values: variables in later finishing SCCs are true
    assignment = [False] * (2 * n)
    scc_order = sorted(range(scc_count), key=lambda x: sum(1 for i in range(2*n) if scc_id[i] == x))
    
    for scc in reversed(scc_order):
        for i in range(2 * n):
            if scc_id[i] == scc and not assignment[i]:
                assignment[i] = True
                # Also assign the same value to all in the same SCC
                stack = [i]
                while stack:
                    u = stack.pop()
                    if assignment[u] is True:  # Not yet processed
                        assignment[u] = True
                        for v in adj_rev[u]:
                            if scc_id[v] == scc and assignment[v] is True:
                                stack.append(v)
    
    # Extract solution
    result = []
    for i in range(n):
        if assignment[i]:  # species (i+1) in humid
            result.append("1")
        else:  # species (i+1) in arid
            result.append("0")
    
    print("YES")
    print(" ".join(result))

solve()
