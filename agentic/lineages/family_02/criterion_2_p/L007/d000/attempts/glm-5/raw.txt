import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        # l1: if positive, zone l1 must be 1; if negative, zone -l1 must be 0
        # l2: if positive, zone l2 must be 1; if negative, zone -l2 must be 0
        # Clause: (l1 is satisfied) OR (l2 is satisfied)
        
        # Variable x: x is 1-indexed zone number
        # Node 2*(x-1) represents x=0, Node 2*(x-1)+1 represents x=1
        
        def get_nodes(val):
            if val > 0:
                zone = val
                # satisfied means zone = 1
                true_node = 2 * (zone - 1) + 1
                false_node = 2 * (zone - 1)
            else:
                zone = -val
                # satisfied means zone = 0
                true_node = 2 * (zone - 1)
                false_node = 2 * (zone - 1) + 1
            return true_node, false_node
        
        t1, f1 = get_nodes(l1)
        t2, f2 = get_nodes(l2)
        
        # Implication: not l1 -> l2  (f1 -> t2)
        adj[f1].append(t2)
        # Implication: not l2 -> l1  (f2 -> t1)
        adj[f2].append(t1)
    
    N = 2 * n
    visited = [False] * N
    order = []
    
    # DFS for topological sort (finishing order)
    stack = []
    for i in range(N):
        if not visited[i]:
            stack.append((i, 0))
            while stack:
                u, state = stack.pop()
                if state == 0:
                    if visited[u]:
                        continue
                    visited[u] = True
                    stack.append((u, 1))
                    for v in adj[u]:
                        if not visited[v]:
                            stack.append((v, 0))
                else:
                    order.append(u)
    
    # Reverse graph for SCC
    radj = [[] for _ in range(N)]
    for u in range(N):
        for v in adj[u]:
            radj[v].append(u)
    
    visited = [False] * N
    comp_id = [-1] * N
    
    cid = 0
    for u in reversed(order):
        if not visited[u]:
            # DFS to find SCC
            stack = [u]
            visited[u] = True
            component_nodes = []
            while stack:
                curr = stack.pop()
                component_nodes.append(curr)
                for v in radj[curr]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            for node in component_nodes:
                comp_id[node] = cid
            cid += 1
    
    # Check validity
    for i in range(n):
        if comp_id[2*i] == comp_id[2*i+1]:
            print("NO")
            return
    
    print("YES")
    result = []
    for i in range(n):
        # If (x=1) has higher topological order (larger comp_id) than (x=0), set x=1
        if comp_id[2*i+1] > comp_id[2*i]:
            result.append(1)
        else:
            result.append(0)
    
    print(' '.join(map(str, result)))

if __name__ == '__main__':
    solve()
