```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # For each zone, we'll have two nodes: 2*i for value 0, 2*i+1 for value 1
    # We'll use 2*i for zone i being 0, and 2*i+1 for zone i being 1
    adj = defaultdict(list)
    rev_adj = defaultdict(list)
    
    index = 2
    clauses = []
    for _ in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        index += 2
        clauses.append((l1, l2))
    
    # Build implication graph for 2-SAT
    # Each literal x is represented as:
    # if x > 0: node 2*(x-1)+1 (zone x is 1)
    # if x < 0: node 2*(-x-1) (zone -x is 0)
    # not x for x>0 is node 2*(x-1)
    # not x for x<0 is node 2*(-x-1)+1
    
    for l1, l2 in clauses:
        # clause (l1 or l2) is equivalent to (not l1 -> l2) and (not l2 -> l1)
        # Convert literals to nodes
        if l1 > 0:
            node_not_l1 = 2*(l1-1)      # zone l1 is 0
            node_l1 = 2*(l1-1)+1        # zone l1 is 1
        else:
            node_not_l1 = 2*(-l1-1)+1   # zone -l1 is 1
            node_l1 = 2*(-l1-1)         # zone -l1 is 0
            
        if l2 > 0:
            node_not_l2 = 2*(l2-1)      # zone l2 is 0
            node_l2 = 2*(l2-1)+1        # zone l2 is 1
        else:
            node_not_l2 = 2*(-l2-1)+1   # zone -l2 is 1
            node_l2 = 2*(-l2-1)         # zone -l2 is 0
            
        # Add implications
        adj[node_not_l1].append(node_l2)
        adj[node_not_l2].append(node_l1)
        rev_adj[node_l2].append(node_not_l1)
        rev_adj[node_l1].append(node_not_l2)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Second DFS on transpose graph
    visited = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            scc_count += 1
    
    # Check if satisfiable
    for i in range(n):
        node_0 = 2 * i
        node_1 = 2 * i + 1
        if scc_id[node_0] == scc_id[node_1]:
            print("NO")
            return
    
    # Build solution
    print("YES")
    result = [0] * n
    # For each variable, if scc_id of node_1 > scc_id of node_0, set to 1
    for i in range(n):
        node_0 = 2 * i
        node_1 = 2 * i + 1
        if scc_id[node_1] > scc_id[node_0]:
            result[i] = 1
            
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
```
