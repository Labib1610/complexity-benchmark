```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    constraints = []
    idx = 2
    for _ in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx+1])
        constraints.append((l1, l2))
        idx += 2
    
    # For each variable x_i (representing s_i), we'll create a 2-SAT instance
    # x_i = True means s_i = 1, x_i = False means s_i = 0
    
    # We'll use the implication graph approach for 2-SAT
    # Each variable x_i has two nodes: i*2 (for x_i false) and i*2+1 (for x_i true)
    
    # Helper functions
    def var(i, val):
        # i is 1-based index, val is 0 or 1
        # returns node index in implication graph
        if val:
            return i * 2 + 1
        else:
            return i * 2
    
    def not_var(node):
        if node % 2 == 0:
            return node + 1
        else:
            return node - 1
            
    # For each constraint (l1, l2), I(l1) + I(l2) > 0
    # I(v) = v*(2*s_|v| - 1) + |v|
    # So I(l1) = l1*(2*s_|l1| - 1) + |l1|
    #    I(l2) = l2*(2*s_|l2| - 1) + |l2|
    # Constraint: l1*(2*s_|l1| - 1) + |l1| + l2*(2*s_|l2| - 1) + |l2| > 0
    # Simplify: 2*l1*s_|l1| - l1 + |l1| + 2*l2*s_|l2| - l2 + |l2| > 0
    #          2*l1*s_|l1| + 2*l2*s_|l2| + (|l1| - l1) + (|l2| - l2) > 0
    # Let c1 = |l1| - l1, c2 = |l2| - l2
    # Then: 2*l1*s_|l1| + 2*l2*s_|l2| + c1 + c2 > 0
    
    # Since s_i is 0 or 1, this becomes checking cases
    # Let's rewrite the constraint more clearly:
    # If l1 > 0: I(l1) = l1*(2*s_l1 - 1) + l1 = 2*l1*s_l1 - l1 + l1 = 2*l1*s_l1
    # If l1 < 0: I(l1) = l1*(2*s_|l1| - 1) + |l1| = l1*(2*s_l1 - 1) + l1 = 2*l1*s_l1 - l1 + l1 = 2*l1*s_l1
    # Wait, let's recheck:
    # For l1 > 0: I(l1) = l1*(2*s_l1 - 1) + l1 = 2*l1*s_l1 - l1 + l1 = 2*l1*s_l1
    # For l1 < 0: I(l1) = l1*(2*s_|l1| - 1) + |l1| = l1*(2*s_l1 - 1) + (-l1) = 2*l1*s_l1 - l1 - l1 = 2*l1*s_l1 - 2*l1
    
    # Actually, let's just compute I(l1) + I(l2) directly based on s_|l1| and s_|l2|
    
    # For a constraint (l1, l2):
    # Let i1 = |l1|, i2 = |l2|
    # s1 = s_i1, s2 = s_i2
    # I(l1) = l1*(2*s1 - 1) + i1
    # I(l2) = l2*(2*s2 - 1) + i2
    # Sum = l1*(2*s1 - 1) + i1 + l2*(2*s2 - 1) + i2 
    #     = 2*l1*s1 - l1 + i1 + 2*l2*s2 - l2 + i2
    #     = 2*l1*s1 + 2*l2*s2 + (i1 - l1) + (i2 - l2)
    # Let c = (i1 - l1) + (i2 - l2)
    # Constraint: 2*l1*s1 + 2*l2*s2 + c > 0
    
    # This is equivalent to: l1*s1 + l2*s2 > -c/2
    # Since everything is integer, this is: l1*s1 + l2*s2 >= floor(-c/2) + 1
    
    # But since s1, s2 are binary, l1*s1 + l2*s2 can take at most 4 values.
    # It's easier to enumerate all 4 possibilities and see which ones satisfy the constraint.
    
    # Let's build implications:
    # For each constraint (l1, l2), with i1=|l1|, i2=|l2|
    # We want: I(l1) + I(l2) > 0
    # Which is: l1*(2*s1-1) + i1 + l2*(2*s2-1) + i2 > 0
    #           2*l1*s1 - l1 + i1 + 2*l2*s2 - l2 + i2 > 0
    #           2*l1*s1 + 2*l2*s2 + (i1 - l1) + (i2 - l2) > 0
    
    # Let's denote c = (i1 - l1) + (i2 - l2)
    # Then: 2*l1*s1 + 2*l2*s2 + c > 0
    #       l1*s1 + l2*s2 > -c/2
    
    # Since s1,s2 are in {0,1}, we can check all four combinations:
    # (s1=0,s2=0): 0 > -c/2 => c > 0
    # (s1=0,s2=1): l2 > -c/2 => 2*l2 + c > 0
    # (s1=1,s2=0): l1 > -c/2 => 2*l1 + c > 0
    # (s1=1,s2=1): l1 + l2 > -c/2 => 2*l1 + 2*l2 + c > 0
    
    # The constraint is satisfied if at least one of these four is true.
    # To encode in 2-SAT, we need to find when the constraint is NOT satisfied,
    # and add implications accordingly.
    
    # The constraint is NOT satisfied when ALL four conditions fail:
    # c <= 0 AND 2*l2 + c <= 0 AND 2*l1 + c <= 0 AND 2*l1 + 2*l2 + c <= 0
    
    # But actually, it's simpler to think in terms of forbidden assignments.
    # A constraint fails only when all satisfying assignments are forbidden.
    # Let's think differently: for each constraint, determine which (s1,s2) pairs violate it.
    
    # Actually, let's reformulate:
    # For constraint (l1,l2), with i1=abs(l1), i2=abs(l2):
    # Define f(s1,s2) = 2*l1*s1 + 2*l2*s2 + (i1-l1) + (i2-l2)
    # Constraint is satisfied if f(s1,s2) > 0
    
    # Forbidden pairs (s1,s2) are those where f(s1,s2) <= 0
    
    # Now we can build a 2-SAT formula:
    # For each forbidden pair (a,b), we add clause (!x1_a OR !x2_b)
    # where xi_v means "variable i is assigned value v"
    
    adj = defaultdict(list)
    rev_adj = defaultdict(list)
    
    # Process each constraint
    for l1, l2 in constraints:
        i1 = abs(l1)
        i2 = abs(l2)
        c = (i1 - l1) + (i2 - l2)
        
        forbidden = []
        # Check all four combinations
        # (s1=0, s2=0)
        if 2*l1*0 + 2*l2*0 + c <= 0:
            forbidden.append((0, 0))
        # (s1=0, s2=1)
        if 2*l1*0 + 2*l2*1 + c <= 0:
            forbidden.append((0, 1))
        # (s1=1, s2=0)
        if 2*l1*1 + 2*l2*0 + c <= 0:
            forbidden.append((1, 0))
        # (s1=1, s2=1)
        if 2*l1*1 + 2*l2*1 + c <= 0:
            forbidden.append((1, 1))
            
        # If all four are forbidden, then unsatisfiable
        if len(forbidden) == 4:
            print("NO")
            return
            
        # For each forbidden pair (a,b), we need (!x1_a OR !x2_b) to be false
        # Which means both x1_a and x2_b must be true in any satisfying assignment
        # But that's not how clauses work. Let me rethink.
        
        # In 2-SAT, if (a,b) is forbidden, then we cannot have x1=a AND x2=b
        # This is equivalent to (!x1_a OR !x2_b)
        # So we add this clause
        
        for a, b in forbidden:
            # Add clause (!x1_a OR !x2_b)
            # Which is (x1 != a OR x2 != b)
            node1 = var(i1, 1-a)  # if a=0, take x1_true; if a=1, take x1_false
            node2 = var(i2, 1-b)  # if b=0, take x2_true; if b=1, take x2_false
            adj[not_var(node1)].append(node2)
            adj[not_var(node2)].append(node1)
            rev_adj[node2].append(not_var(node1))
            rev_adj[node1].append(not_var(node2))
    
    # Strongly connected components to solve 2-SAT
    order = []
    visited = [False] * (2*n + 2)
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2, 2*n + 2):
        if not visited[i]:
            dfs1(i)
    
    # Build reverse graph and find SCCs
    visited = [False] * (2*n + 2)
    scc_id = [0] * (2*n + 2)
    cnt = 1
    
    def dfs2(u, cid):
        visited[u] = True
        scc_id[u] = cid
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v, cid)
    
    for u in reversed(order):
        if not visited[u]:
            dfs2(u, cnt)
            cnt += 1
    
    # Check satisfiability
    for i in range(1, n+1):
        if scc_id[var(i, 0)] == scc_id[var(i, 1)]:
            print("NO")
            return
    
    # Extract solution
    assignment = [0] * (n+1)
    # Process SCCs in reverse topological order
    scc_order = [(scc_id[var(i, b)], var(i, b)) for i in range(1, n+1) for b in [0,1]]
    scc_order.sort(reverse=True)
    
    assigned = [False] * (n+1)
    for _, node in scc_order:
        idx = node // 2
        val = node % 2
        if not assigned[idx]:
            assignment[idx] = val
            assigned[idx] = True
    
    print("YES")
    print(' '.join(str(assignment[i]) for i in range(1, n+1)))

if __name__ == "__main__":
    main()
```
