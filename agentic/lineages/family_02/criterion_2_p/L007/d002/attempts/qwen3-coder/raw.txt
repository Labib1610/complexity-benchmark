import sys
from collections import defaultdict

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # We'll build a graph for 2-SAT
    # Variables: for each zone i (1-based), we have boolean variable x_i
    # x_i = True means state 1, x_i = False means state 0
    # For each constraint (l1, l2):
    # If l1 > 0 and l2 > 0:
    #   Not both zones |l1| and |l2| can be in state 0.
    #   So not (not x_|l1| and not x_|l2|) => x_|l1| or x_|l2|
    # If l1 > 0 and l2 < 0:
    #   Zone |l1| is not 0 OR zone |l2| is not 1
    #   So x_|l1| or not x_|l2|
    # If l1 < 0 and l2 > 0:
    #   Zone |l1| is not 1 OR zone |l2| is not 0
    #   So not x_|l1| or x_|l2|
    # If l1 < 0 and l2 < 0:
    #   Not both zones |l1| and |l2| can be in state 1.
    #   So not x_|l1| or not x_|l2|
    
    # Let's map variable indices: for zone i (1-based), 
    # var[i] = 2*i - 2 (True) and not_var[i] = 2*i - 1 (False)
    # But actually, let's use standard 2-SAT formulation:
    # For variable i (0-based), node 2*i represents x_i, node 2*i+1 represents not x_i
    
    # But since zones are 1-based, let's define:
    # For zone i (1-based), variable index is i-1 (0-based)
    # Node 2*(i-1) represents x_i (zone i is state 1)
    # Node 2*(i-1)+1 represents not x_i (zone i is state 0)
    
    adj = defaultdict(list)
    rev_adj = defaultdict(list)
    
    idx = 2
    for i in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx+1])
        idx += 2
        
        # Convert to 0-based indexing for variables
        # Also, positive means we care about state 1, negative means state 0
        v1 = abs(l1)
        v2 = abs(l2)
        var1 = v1 - 1  # 0-based variable index
        var2 = v2 - 1
        
        # Interpretation:
        # If l1 > 0: zone v1 should NOT be in state 0 (so it should be in state 1)
        # If l1 < 0: zone v1 should NOT be in state 1 (so it should be in state 0)
        # Same for l2
        
        # The constraint is: NOT(strain w.r.t l1 AND strain w.r.t l2)
        # strain w.r.t l1 means:
        #   if l1 > 0: zone |l1| is in state 0
        #   if l1 < 0: zone |l1| is in state 1
        # Similarly for l2
        
        # So the forbidden case is:
        # ( (l1>0 and x_|l1| is False) or (l1<0 and x_|l1| is True) ) AND 
        # ( (l2>0 and x_|l2| is False) or (l2<0 and x_|l2| is True) )
        
        # This translates to forbidding the assignment:
        # (x_var1 = (l1<0)) AND (x_var2 = (l2<0))
        
        # So we add clause: NOT(x_var1 = (l1<0)) OR NOT(x_var2 = (l2<0))
        # Which is: (x_var1 != (l1<0)) OR (x_var2 != (l2<0))
        
        # In terms of implications:
        # If x_var1 == (l1<0) then x_var2 != (l2<0)
        # If x_var2 == (l2<0) then x_var1 != (l1<0)
        
        # Node IDs:
        # For variable i, node 2*i is x_i=True, node 2*i+1 is x_i=False
        
        node1_true = 2 * var1       # x_var1 = True
        node1_false = 2 * var1 + 1 # x_var1 = False
        node2_true = 2 * var2
        node2_false = 2 * var2 + 1
        
        val1 = (l1 < 0)  # if l1<0, then we want x_var1 = False to cause strain
        val2 = (l2 < 0)
        
        # Forbidden case: x_var1 == val1 AND x_var2 == val2
        # Add clauses: NOT(x_var1 == val1) OR NOT(x_var2 == val2)
        # Which gives two implications:
        # 1. If x_var1 == val1 then x_var2 != val2
        # 2. If x_var2 == val2 then x_var1 != val1
        
        # Case 1: x_var1 == val1 implies x_var2 != val2
        if val1:  # x_var1 = True -> x_var2 != val2
            # node1_true -> node2_false (if val2 is True) or node2_true (if val2 is False)
            if val2:
                adj[node1_true].append(node2_false)
                rev_adj[node2_false].append(node1_true)
            else:
                adj[node1_true].append(node2_true)
                rev_adj[node2_true].append(node1_true)
        else:  # x_var1 = False -> x_var2 != val2
            # node1_false -> node2_false (if val2 is True) or node2_true (if val2 is False)
            if val2:
                adj[node1_false].append(node2_false)
                rev_adj[node2_false].append(node1_false)
            else:
                adj[node1_false].append(node2_true)
                rev_adj[node2_true].append(node1_false)
                
        # Case 2: x_var2 == val2 implies x_var1 != val1
        if val2:  # x_var2 = True -> x_var1 != val1
            # node2_true -> node1_false (if val1 is True) or node1_true (if val1 is False)
            if val1:
                adj[node2_true].append(node1_false)
                rev_adj[node1_false].append(node2_true)
            else:
                adj[node2_true].append(node1_true)
                rev_adj[node1_true].append(node2_true)
        else:  # x_var2 = False -> x_var1 != val1
            # node2_false -> node1_false (if val1 is True) or node1_true (if val1 is False)
            if val1:
                adj[node2_false].append(node1_false)
                rev_adj[node1_false].append(node2_false)
            else:
                adj[node2_false].append(node1_true)
                rev_adj[node1_true].append(node2_false)
    
    # Now run Kosaraju's algorithm to check 2-SAT satisfiability
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for u in range(2 * n):
        if not visited[u]:
            dfs1(u)
    
    # Transpose graph
    visited = [False] * (2 * n)
    component = [-1] * (2 * n)
    comp_count = 0
    
    def dfs2(u, root):
        component[u] = root
        for v in rev_adj[u]:
            if component[v] == -1:
                dfs2(v, root)
    
    while order:
        u = order.pop()
        if component[u] == -1:
            dfs2(u, comp_count)
            comp_count += 1
            
    # Check if any variable and its negation are in the same SCC
    for i in range(n):
        node_true = 2 * i
        node_false = 2 * i + 1
        if component[node_true] == component[node_false]:
            print("NO")
            return
            
    # Satisfiable, reconstruct solution
    print("YES")
    assignment = [0] * n
    # Process components in reverse topological order
    comp_nodes = [[] for _ in range(comp_count)]
    for node in range(2 * n):
        comp_nodes[component[node]].append(node)
        
    comp_visited = [False] * comp_count
    comp_order = []
    
    def comp_dfs(c):
        comp_visited[c] = True
        for node in comp_nodes[c]:
            for nei in adj[node]:
                nei_c = component[nei]
                if nei_c != c and not comp_visited[nei_c]:
                    comp_dfs(nei_c)
        comp_order.append(c)
                    
    for c in range(comp_count):
        if not comp_visited[c]:
            comp_dfs(c)
            
    # Assign values
    assigned_comp = [False] * comp_count
    for c in reversed(comp_order):
        if assigned_comp[c]: continue
        # Mark complement components as false
        for node in comp_nodes[c]:
            var_idx = node // 2
            # If this component sets something to true/false,
            # the complement component sets it to the opposite
            comp_of_node = component[node ^ 1]  # XOR flips last bit
            if not assigned_comp[comp_of_node]:
                assigned_comp[comp_of_node] = True
                # This means node is False, so node^1 is True
                # But we're processing c which contains node
                # So if node is False, then node^1 (in another component) is True
                # We set the variable according to which node is in which component
                pass
                
    # Simpler way: process nodes in component order
    assigned = [False] * n
    for c in reversed(comp_order):
        if assigned_comp[c]: continue
        for node in comp_nodes[c]:
            var_idx = node // 2
            if not assigned[var_idx]:
                if node % 2 == 0:  # node represents x_var = True
                    assignment[var_idx] = 1
                else:  # node represents x_var = False
                    assignment[var_idx] = 0
                assigned[var_idx] = True
        assigned_comp[c] = True
        
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    main()
