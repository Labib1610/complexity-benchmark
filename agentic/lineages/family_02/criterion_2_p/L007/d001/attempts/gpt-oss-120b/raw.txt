```python
import sys

sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = n * 2
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def var_node(idx, val):
        # idx 1‑based, val 0/1
        return (idx - 1) * 2 + val

    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        i1, d1 = abs(l1), 1 if l1 > 0 else 0
        i2, d2 = abs(l2), 1 if l2 > 0 else 0
        a = var_node(i1, d1)          # literal a : xi == d1
        b = var_node(i2, d2)          # literal b : xj == d2
        na = a ^ 1                     # negation
        nb = b ^ 1
        # (a OR b) => (¬a -> b) and (¬b -> a)
        g[na].append(b)
        rg[b].append(na)
        g[nb].append(a)
        rg[a].append(nb)

    visited = [False] * N
    order = []

    def dfs(v):
        stack = [(v, 0)]
        while stack:
            v, state = stack.pop()
            if state == 0:
                if visited[v]:
                    continue
                visited[v] = True
                stack.append((v, 1))
                for to in g[v]:
                    if not visited[to]:
                        stack.append((to, 0))
            else:
                order.append(v)

    for v in range(N):
        if not visited[v]:
            dfs(v)

    comp = [-1] * N
    cid = 0

    def rdfs(v, label):
        stack = [v]
        while stack:
            x = stack.pop()
            if comp[x] != -1:
                continue
            comp[x] = label
            for to in rg[x]:
                if comp[to] == -1:
                    stack.append(to)

    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, cid)
            cid += 1

    assignment = [0] * n
    for i in range(1, n + 1):
        if comp[var_node(i, 0)] == comp[var_node(i, 1)]:
            print("NO")
            return
        # component with higher topological order gets false
        assignment[i - 1] = 0 if comp[var_node(i, 0)] > comp[var_node(i, 1)] else 1

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
