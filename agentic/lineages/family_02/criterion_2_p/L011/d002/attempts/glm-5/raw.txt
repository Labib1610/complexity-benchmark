```python
import sys
from collections import deque

# Constants
UP = 1
DOWN = 0

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return
    
    # Adjacency list for 2-SAT implication graph
    # Nodes: 0 to 2n-1
    # Variable x: UP node (2x-2), DOWN node (2x-1)
    # Implication u -> v means if u is true, v must be true
    
    adj = [[] for _ in range(2 * n)]
    
    def get_node(var):
        """Maps variable to node index. 
        Positive var (UP) -> 2*(var-1)
        Negative var (DOWN) -> 2*(-var-1) + 1
        """
        if var > 0:
            return 2 * (var - 1)
        else:
            return 2 * (-var - 1) + 1
    
    def neg_node(node):
        """Maps a node to its negation node."""
        return node ^ 1
    
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
        except StopIteration:
            break
        
        # Clause: (u OR v)
        # Implications: !u -> v and !v -> u
        nu = -u
        nv = -v
        
        adj[get_node(nu)].append(get_node(v))
        adj[get_node(nv)].append(get_node(u))
    
    # Kosaraju's Algorithm to find Strongly Connected Components
    
    # 1. First pass DFS to determine processing order
    visited = [False] * (2 * n)
    order = []
    
    for i in range(2 * n):
        if not visited[i]:
            stack = [(i, 0)]
            while stack:
                curr, idx = stack[-1]
                if not visited[curr]:
                    visited[curr] = True
                
                if idx < len(adj[curr]):
                    neighbor = adj[curr][idx]
                    stack[-1] = (curr, idx + 1)
                    if not visited[neighbor]:
                        stack.append((neighbor, 0))
                else:
                    stack.pop()
                    order.append(curr)
    
    # 2. Build transpose graph
    rev_adj = [[] for _ in range(2 * n)]
    for u in range(2 * n):
        for v in adj[u]:
            rev_adj[v].append(u)
    
    # 3. Second pass DFS to find SCCs in topological order
    visited = [False] * (2 * n)
    comp_id = [-1] * (2 * n)
    curr_comp_id = 0
    
    for u in reversed(order):
        if not visited[u]:
            # BFS for the component
            queue = deque([u])
            visited[u] = True
            component_nodes = []
            
            while queue:
                curr = queue.popleft()
                component_nodes.append(curr)
                
                for v in rev_adj[curr]:
                    if not visited[v]:
                        visited[v] = True
                        queue.append(v)
            
            for node in component_nodes:
                comp_id[node] = curr_comp_id
            curr_comp_id += 1
            
    # Check for unsatisfiability and assign values
    result = [0] * n
    
    for i in range(n):
        up_node = get_node(i + 1)
        down_node = neg_node(up_node)
        
        if comp_id[up_node] == comp_id[down_node]:
            print("NO")
            return
        
        # Assign TOPOLGICALLY LATER component as True
        # In typical SCC numbering, later component has smaller ID if using reverse postorder
        if comp_id[up_node] < comp_id[down_node]:
            result[i] = UP
        else:
            result[i] = DOWN
            
    print("YES")
    print(*(result))

if __name__ == '__main__':
    solve()
```
