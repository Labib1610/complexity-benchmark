import sys
sys.setrecursionlimit(1 << 25)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n, m = int(data[0]), int(data[1])
    
    # Each clause is a pair of literals
    # Literal x means variable x is true
    # Literal -x means variable x is false
    clauses = []
    index = 2
    for _ in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        index += 2
        clauses.append((l1, l2))
    
    # We model this as 2-SAT
    # Variables: x_1, ..., x_n corresponding to switches 1..n
    # x_i = True means switch i is UP
    # x_i = False means switch i is DOWN
    
    # Build implication graph
    # For each clause (a OR b), we add edges (!a -> b) and (!b -> a)
    adj = [[] for _ in range(2 * n + 2)]  # nodes -n to n, but use 1-based indexing with offset
    rev_adj = [[] for _ in range(2 * n + 2)]
    
    def var_to_node(x):
        # Maps variable x (1-based) to node index
        # x is from 1 to n => node indices 1 to n (positive)
        # -x is from -1 to -n => node indices n+1 to 2n (negative)
        if x > 0:
            return x
        else:
            return n - x  # -1 -> n+1, -2 -> n+2, etc
    
    def node_to_var(node):
        if 1 <= node <= n:
            return node
        else:
            return -(node - n)
    
    for l1, l2 in clauses:
        # Clause (l1 OR l2) => (!l1 -> l2) and (!l2 -> l1)
        # !l1 is represented as -l1
        not_l1 = -l1
        not_l2 = -l2
        
        # Add edge from not_l1 to l2
        u1 = var_to_node(not_l1)
        v1 = var_to_node(l2)
        adj[u1].append(v1)
        rev_adj[v1].append(u1)
        
        # Add edge from not_l2 to l1
        u2 = var_to_node(not_l2)
        v2 = var_to_node(l1)
        adj[u2].append(v2)
        rev_adj[v2].append(u2)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n + 2)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(1, 2*n + 1):
        if not visited[i]:
            dfs1(i)
    
    # Transpose graph and do DFS in reverse order
    visited = [False] * (2 * n + 2)
    components = [-1] * (2 * n + 2)
    comp_num = 0
    
    def dfs2(u):
        visited[u] = True
        components[u] = comp_num
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            comp_num += 1
            dfs2(u)
    
    # Check if any variable and its negation are in the same component
    assignment = [-1] * (n + 1)
    
    for i in range(1, n + 1):
        pos_node = var_to_node(i)
        neg_node = var_to_node(-i)
        if components[pos_node] == components[neg_node]:
            print("NO")
            return
        # pos_node corresponds to x_i = True
        # neg_node corresponds to x_i = False
        if components[pos_node] > components[neg_node]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    result = []
    for i in range(1, n + 1):
        result.append(str(assignment[i]))
    print(' '.join(result))

if __name__ == "__main__":
    main()
