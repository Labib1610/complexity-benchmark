We can model this as a **2-SAT** problem.  
Each feed line has two possible states: energized (true) or isolated (false).  
Each interlock is a clause of the form `(a OR b)`, where `a` and `b` are literals (positive or negative).  

We can solve 2-SAT using implication graph and SCC (Tarjan’s or Kosaraju’s).  

Here’s the implementation:  

```python
import sys
sys.setrecursionlimit(1 << 20)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # nodes: 2*i for false, 2*i+1 for true

    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)

    # literal to node: positive x -> 2*(x-1)+1, negative -x -> 2*(x-1)
    def node(lit):
        if lit > 0:
            return 2 * (lit - 1) + 1
        else:
            return 2 * (-lit - 1)

    def neg_node(v):
        return v ^ 1

    for _ in range(m):
        a, b = map(int, input().split())
        na = node(a)
        nb = node(b)
        # (a OR b) => (!a -> b) and (!b -> a)
        add_imp(neg_node(na), nb)
        add_imp(neg_node(nb), na)

    # Kosaraju SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    # Build assignment: if comp[true_node] > comp[false_node], set true
    res = []
    for i in range(n):
        if comp[2*i+1] > comp[2*i]:
            res.append(1)
        else:
            res.append(0)

    print("YES")
    print(" ".join(map(str, res)))

if __name__ == "__main__":
    solve()
```
