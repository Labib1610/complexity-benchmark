import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # Each tumbler i can be 0 or 1.
    # We create a 2-SAT problem.
    # Variable x_i is True if tumbler i is 1, False if tumbler i is 0.
    # For each pin, at least one prong must be triggered.
    # A prong x is triggered if:
    #   x > 0: tumbler x is 1 (x_x is True)
    #   x < 0: tumbler |x| is 0 (x_{|x|} is False)
    # So for each pin with prongs (a, b, c), the condition is (trigger(a) OR trigger(b) OR trigger(c)).
    # This is a 3-SAT clause. We convert it to three 2-SAT implications:
    # (NOT trigger(a) -> trigger(b)) AND (NOT trigger(a) -> trigger(c)) ... (and cyclic permutations)
    # Actually, (A OR B OR C) is equivalent to:
    # (!A -> B) AND (!A -> C) AND (!B -> A) AND (!B -> C) AND (!C -> A) AND (!C -> B)
    # We only need to ensure that if two are false, the third must be true.
    # So we add implications: !A -> B, !A -> C, !B -> A, !B -> C, !C -> A, !C -> B.
    # Or more simply, for each pair, add implications.
    
    # We map literals to nodes.
    # Literal: (tumbler_index, value) where value is 0 or 1.
    # Node index: 2 * (tumbler_index - 1) + value.
    # Total nodes: 2 * n.
    
    adj = [[] for _ in range(2 * n)]
    
    def get_node(tumbler_idx, val):
        # tumbler_idx is 1-based
        return 2 * (tumbler_idx - 1) + val
    
    def add_implication(u_tumbler, u_val, v_tumbler, v_val):
        u_node = get_node(u_tumbler, u_val)
        v_node = get_node(v_tumbler, v_val)
        adj[u_node].append(v_node)
    
    for _ in range(m):
        p1 = int(next(it))
        p2 = int(next(it))
        p3 = int(next(it))
        
        # Parse prongs to (tumbler_index, required_value)
        # prong x: if x > 0, requires tumbler x to be 1.
        # if x < 0, requires tumbler |x| to be 0.
        prongs = []
        for p in [p1, p2, p3]:
            if p > 0:
                prongs.append((p, 1))
            else:
                prongs.append((-p, 0))
        
        # Clause: (A OR B OR C)
        # Implications:
        # !A -> B, !A -> C
        # !B -> A, !B -> C
        # !C -> A, !C -> B
        A_t, A_v = prongs[0]
        B_t, B_v = prongs[1]
        C_t, C_v = prongs[2]
        
        # !A is (A_t, 1 - A_v)
        add_implication(A_t, 1 - A_v, B_t, B_v)
        add_implication(A_t, 1 - A_v, C_t, C_v)
        
        add_implication(B_t, 1 - B_v, A_t, A_v)
        add_implication(B_t, 1 - B_v, C_t, C_v)
        
        add_implication(C_t, 1 - C_v, A_t, A_v)
        add_implication(C_t, 1 - C_v, B_t, B_v)
    
    # Kosaraju's algorithm to find SCCs
    num_nodes = 2 * n
    visited = [False] * num_nodes
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(num_nodes):
        if not visited[i]:
            dfs1(i)
    
    # Reverse graph
    rev_adj = [[] for _ in range(num_nodes)]
    for u in range(num_nodes):
        for v in adj[u]:
            rev_adj[v].append(u)
    
    visited = [False] * num_nodes
    comp_id = [-1] * num_nodes
    current_comp = 0
    
    def dfs2(u, cid):
        visited[u] = True
        comp_id[u] = cid
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v, cid)
    
    for u in reversed(order):
        if not visited[u]:
            dfs2(u, current_comp)
            current_comp += 1
    
    # Check for contradiction and assign values
    assignment = [-1] * n
    possible = True
    
    # We iterate components in topological order (which is reverse of finishing times, 
    # but here comp_ids are assigned in that order, so higher id is "earlier" in topo sort).
    # To process in topological order, we iterate comp_ids from max to 0.
    # Actually, in standard Kosaraju, the first SCC found in 2nd pass is a sink.
    # So comp_id 0 is a sink. We want to process from sinks to sources.
    # So we iterate components in order 0, 1, 2...
    
    # We need to know which component contains which literals.
    # comp_id[u] stores the component of node u.
    
    # Check validity
    for i in range(n):
        node0 = get_node(i + 1, 0)
        node1 = get_node(i + 1, 1)
        if comp_id[node0] == comp_id[node1]:
            possible = False
            break
    
    if not possible:
        print("NO")
    else:
        # Assign values
        # Process components in topological order (0 to current_comp-1)
        # If a component is not assigned, set all its literals to True.
        # This forces the opposite literals (in the source component) to be False.
        # Wait, standard logic:
        # If comp_id[literal] == comp_id[!literal], impossible.
        # Otherwise, if comp_id[literal] < comp_id[!literal], then literal is False.
        # Because the component containing !literal is processed earlier (is a sink relative to literal's component),
        # so if we set !literal to True, then literal must be False.
        # Actually, the standard rule is: if comp_id[x] < comp_id[!x], then x is True.
        # Let's re-verify.
        # In Kosaraju, the first SCC found in 2nd pass is a sink in the component graph.
        # If comp_id[x] < comp_id[!x], then SCC(x) is a sink compared to SCC(!x).
        # If we set x to True, then !x must be False.
        # If SCC(x) is a sink, it has no outgoing edges to SCC(!x).
        # So setting x to True does not force !x to be True.
        # However, if we set !x to True, then since SCC(!x) has an edge to SCC(x) (because x -> !x is not an edge, but !x -> x is an edge? No.)
        # Let's stick to the rule: if comp_id[x] > comp_id[!x], then x is assigned False.
        # Because SCC(!x) is "later" in topological sort (closer to source), so it can force x.
        # Actually, let's just use the standard rule: x is True iff comp_id[x] > comp_id[!x].
        # Let's trace: if comp_id[x] > comp_id[!x], then SCC(x) appeared earlier in 2nd pass, so it is closer to source.
        # Edges go from source to sink.
        # If x is True, then !x is False.
        # If SCC(x) is source-like, and we set x=True, does it imply anything?
        # The implication !x -> x means if !x is True, x must be True.
        # If we set x=True, we satisfy this.
        # The implication x -> !x means if x is True, !x must be True. This would be a contradiction.
        # So if x -> !x exists, then comp_id[x] < comp_id[!x] (x is sink, !x is source).
        # In this case, we must have x=False.
        # So if comp_id[x] < comp_id[!x], x must be False.
        # If comp_id[x] > comp_id[!x], x can be True.
        # So assignment: x is True iff comp_id[x] > comp_id[!x].
        
        res = []
        for i in range(n):
            node0 = get_node(i + 1, 0)
            node1 = get_node(i + 1, 1)
            # node1 corresponds to x_i being True.
            if comp_id[node1] > comp_id[node0]:
                res.append(1)
            else:
                res.append(0)
        
        print("YES")
        print(" ".join(map(str, res)))

solve()
