import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # 2-SAT setup
    # Variable i: True means channel i is mode 1, False means mode 0
    # For each unit, at least one connection must match expected phase
    # Unit has connections c1, c2, c3
    # For connection ci: if ci > 0, expected mode 1; if ci < 0, expected mode 0
    # "Match expected" for ci means:
    #   - if ci > 0: channel |ci| is mode 1 (True)
    #   - if ci < 0: channel |ci| is mode 0 (False)
    # So literal for ci is: (|ci| is True) if ci > 0, else (|ci| is False)
    # We need at least one literal true -> clause (l1 or l2 or l3)
    # Convert to 2-SAT: (l1 or l2) and (l1 or l3) and (l2 or l3)
    
    # Node indexing: 0 to 2n-1
    # Node 2*(i-1) represents variable i is False
    # Node 2*(i-1)+1 represents variable i is True
    
    N = 2 * n
    adj = [[] for _ in range(N)]
    adj_rev = [[] for _ in range(N)]
    
    def get_node(channel, is_true):
        # channel is 1-indexed
        return 2 * (channel - 1) + (1 if is_true else 0)
    
    def add_implication(u, v):
        adj[u].append(v)
        adj_rev[v].append(u)
    
    def negate(node):
        return node ^ 1
    
    for _ in range(m):
        c1 = int(next(it))
        c2 = int(next(it))
        c3 = int(next(it))
        
        # Get literals (nodes that represent "match expected")
        l1 = get_node(abs(c1), c1 > 0)
        l2 = get_node(abs(c2), c2 > 0)
        l3 = get_node(abs(c3), c3 > 0)
        
        # Add clauses: (l1 or l2), (l1 or l3), (l2 or l3)
        # (a or b) is equivalent to (~a -> b) and (~b -> a)
        
        # (l1 or l2)
        add_implication(negate(l1), l2)
        add_implication(negate(l2), l1)
        
        # (l1 or l3)
        add_implication(negate(l1), l3)
        add_implication(negate(l3), l1)
        
        # (l2 or l3)
        add_implication(negate(l2), l3)
        add_implication(negate(l3), l2)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * N
    order = []
    
    def dfs1(v):
        stack = [(v, 0)]
        while stack:
            node, idx = stack[-1]
            if idx == 0:
                if visited[node]:
                    stack.pop()
                    continue
                visited[node] = True
            if idx < len(adj[node]):
                stack[-1] = (node, idx + 1)
                next_node = adj[node][idx]
                if not visited[next_node]:
                    stack.append((next_node, 0))
            else:
                order.append(node)
                stack.pop()
    
    for i in range(N):
        if not visited[i]:
            dfs1(i)
    
    visited = [False] * N
    comp = [-1] * N
    comp_id = 0
    
    def dfs2(v, cid):
        stack = [v]
        visited[v] = True
        while stack:
            node = stack.pop()
            comp[node] = cid
            for next_node in adj_rev[node]:
                if not visited[next_node]:
                    visited[next_node] = True
                    stack.append(next_node)
    
    for v in reversed(order):
        if not visited[v]:
            dfs2(v, comp_id)
            comp_id += 1
    
    # Check satisfiability
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return
    
    # Build assignment
    assignment = [0] * n
    for i in range(n):
        # If True node has higher component id, it's assigned True
        if comp[2*i+1] > comp[2*i]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(" ".join(map(str, assignment)))

solve()
