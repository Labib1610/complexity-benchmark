```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        c1 = int(next(it))
        c2 = int(next(it))
        c3 = int(next(it))
        clauses.append([c1, c2, c3])

    # assignment: None = unassigned, 0 = false, 1 = true
    assign = [None] * (n + 1)

    # helper to get literal value under current assignment
    def lit_val(lit):
        v = abs(lit)
        val = assign[v]
        if val is None:
            return None
        return val if lit > 0 else 1 - val

    # propagation of unit clauses
    def propagate():
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                sat = False
                unassigned_lits = []
                for lit in clause:
                    val = lit_val(lit)
                    if val is None:
                        unassigned_lits.append(lit)
                    elif val == 1:
                        sat = True
                        break
                if sat:
                    continue
                if not unassigned_lits:
                    # all false -> conflict
                    return False
                if len(unassigned_lits) == 1:
                    lit = unassigned_lits[0]
                    var = abs(lit)
                    needed = 1 if lit > 0 else 0
                    if assign[var] is None:
                        assign[var] = needed
                        changed = True
                    elif assign[var] != needed:
                        return False
        return True

    # pure literal elimination (optional but helpful)
    def pure_literal():
        while True:
            pos = [False] * (n + 1)
            neg = [False] * (n + 1)
            for clause in clauses:
                clause_sat = False
                for lit in clause:
                    v = abs(lit)
                    if assign[v] is not None:
                        if (assign[v] == 1 and lit > 0) or (assign[v] == 0 and lit < 0):
                            clause_sat = True
                            break
                if clause_sat:
                    continue
                for lit in clause:
                    v = abs(lit)
                    if assign[v] is None:
                        if lit > 0:
                            pos[v] = True
                        else:
                            neg[v] = True
            found = False
            for v in range(1, n + 1):
                if assign[v] is None:
                    if pos[v] and not neg[v]:
                        assign[v] = 1
                        found = True
                    elif neg[v] and not pos[v]:
                        assign[v] = 0
                        found = True
            if not found:
                break
            if not propagate():
                return False
        return True

    def dfs():
        if not propagate():
            return False
        if not pure_literal():
            return False
        # check if all clauses satisfied
        all_sat = True
        for clause in clauses:
            clause_sat = False
            for lit in clause:
                val = lit_val(lit)
                if val == 1:
                    clause_sat = True
                    break
            if not clause_sat:
                all_sat = False
                break
        if all_sat:
            return True
        # choose unassigned variable
        for v in range(1, n + 1):
            if assign[v] is None:
                var = v
                break
        # try 0 then 1
        saved = assign[:]
        assign[var] = 0
        if dfs():
            return True
        assign[:] = saved
        assign[var] = 1
        if dfs():
            return True
        assign[:] = saved
        return False

    if dfs():
        print("YES")
        res = [str(assign[i] if assign[i] is not None else 0) for i in range(1, n + 1)]
        print(" ".join(res))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
