from collections import defaultdict
import sys

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each diagnostic path, we need at least one of the three conditions to be true
    # This is a 3-CNF clause: (a OR b OR c)
    # We'll convert this to 2-CNF by introducing auxiliary variables
    
    # But since we have 3-CNF and n <= 90, we can use a SAT solver approach
    # However, let's try to solve it directly with backtracking since constraints are small
    
    # Let's represent clauses
    clauses = []
    for _ in range(m):
        line = list(map(int, input().split()))
        clause = []
        for x in line:
            # x is positive means variable x should be 1 (focusing)
            # x is negative means variable |x| should be 0 (defocusing)
            clause.append(x)
        clauses.append(clause)
    
    # Try all possible assignments
    # But 2^90 is too large, so let's think differently
    
    # This is exactly 3-SAT problem. Since n <= 90 and m <= 400, 
    # we need an efficient approach. Let's use a randomized or heuristic approach,
    # but for correctness, let's implement a proper 3-SAT solver with pruning.
    
    # Actually, let's try a simple backtracking with unit propagation
    
    def var_index(x):
        return abs(x) - 1
    
    def var_value(x, assignment):
        idx = var_index(x)
        if idx >= len(assignment):
            return None
        val = assignment[idx]
        if val is None:
            return None
        if x > 0:
            return val
        else:
            return 1 - val
    
    def is_satisfied(clause, assignment):
        for lit in clause:
            val = var_value(lit, assignment)
            if val == 1:
                return True
        return False
    
    def find_unit_clause(clauses, assignment):
        for clause in clauses:
            unknown_count = 0
            unknown_lit = 0
            satisfied = False
            for lit in clause:
                val = var_value(lit, assignment)
                if val == 1:
                    satisfied = True
                    break
                elif val is None:
                    unknown_count += 1
                    unknown_lit = lit
            if not satisfied and unknown_count == 1:
                return unknown_lit
        return None
    
    def find_pure_literal(clauses, assignment):
        lit_count = defaultdict(int)
        for clause in clauses:
            for lit in clause:
                idx = var_index(lit)
                if assignment[idx] is None:
                    lit_count[lit] += 1
        
        for lit, count in lit_count.items():
            neg_lit = -lit if lit > 0 else -lit
            if neg_lit not in lit_count or lit_count[neg_lit] == 0:
                return lit
        return None
    
    def solve_sat(clauses, assignment):
        # Unit propagation
        while True:
            unit_lit = find_unit_clause(clauses, assignment)
            if unit_lit is None:
                break
            idx = var_index(unit_lit)
            val = 1 if unit_lit > 0 else 0
            if assignment[idx] is not None:
                if assignment[idx] != val:
                    return False  # Conflict
                continue
            assignment[idx] = val
        
        # Check if all clauses are satisfied
        all_satisfied = True
        for clause in clauses:
            if not is_satisfied(clause, assignment):
                all_satisfied = False
                break
        if all_satisfied:
            return True
        
        # Pure literal elimination
        while True:
            pure_lit = find_pure_literal(clauses, assignment)
            if pure_lit is None:
                break
            idx = var_index(pure_lit)
            val = 1 if pure_lit > 0 else 0
            assignment[idx] = val
            # Check if all clauses are satisfied
            all_satisfied = True
            for clause in clauses:
                if not is_satisfied(clause, assignment):
                    all_satisfied = False
                    break
            if all_satisfied:
                return True
        
        # Find first unassigned variable
        for i in range(n):
            if assignment[i] is None:
                # Try setting it to 1
                assignment[i] = 1
                if solve_sat(clauses, assignment):
                    return True
                # Try setting it to 0
                assignment[i] = 0
                if solve_sat(clauses, assignment):
                    return True
                # Backtrack
                assignment[i] = None
                return False
        return False
    
    assignment = [None] * n
    if solve_sat(clauses, assignment):
        print("YES")
        result = []
        for i in range(n):
            if assignment[i] is None:
                result.append("0")
            else:
                result.append(str(assignment[i]))
        print(" ".join(result))
    else:
        print("NO")

solve()
