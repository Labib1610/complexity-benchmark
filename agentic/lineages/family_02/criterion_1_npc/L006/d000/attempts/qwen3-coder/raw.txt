from collections import defaultdict
import sys

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each constraint, we have a clause with 3 literals
    # This is a 3-SAT problem
    # We'll use the fact that 3-SAT can be solved in exponential time
    # But since n <= 90, we can't brute force
    # However, we can try to solve it as a 2-SAT if possible, or use other techniques
    # Actually, let's model this as a general SAT problem and use a heuristic approach
    
    # Let's try to solve this using a backtracking approach with pruning
    # But given the constraints, let's think of a better way
    
    # Each constraint is a disjunction of 3 literals
    # We can try to solve this using a SAT solver approach
    
    # Let's represent each variable as index 0 to n-1
    # And positive literal as 2*i, negative as 2*i+1 in a CNF encoding
    # But we'll implement a simple backtracking solution
    
    constraints = []
    for _ in range(m):
        line = list(map(int, input().split()))
        clause = []
        for lit in line:
            if lit > 0:
                clause.append((lit - 1, True))  # (variable_index, is_positive)
            else:
                clause.append((-lit - 1, False))  # (variable_index, is_positive)
        constraints.append(clause)
    
    # Try all possible assignments
    # With n up to 90, this is infeasible
    # But Python might be able to handle small cases
    
    # Let's try a more intelligent approach
    # We'll use backtracking with unit propagation
    
    def is_satisfied(assignment):
        # assignment[i] = 0 means false, 1 means true, -1 means unassigned
        for clause in constraints:
            satisfied = False
            for var_idx, is_pos in clause:
                val = assignment[var_idx]
                if val == -1:
                    continue
                if (val == 1 and is_pos) or (val == 0 and not is_pos):
                    satisfied = True
                    break
            if not satisfied:
                return False
        return True
    
    def backtrack(assignment, index):
        if index == n:
            if is_satisfied(assignment):
                return assignment[:]
            else:
                return None
        
        # Try setting to 0 (inactive)
        assignment[index] = 0
        if is_satisfied(assignment):  # Pruning
            result = backtrack(assignment, index + 1)
            if result is not None:
                return result
        
        # Try setting to 1 (active)
        assignment[index] = 1
        if is_satisfied(assignment):  # Pruning
            result = backtrack(assignment, index + 1)
            if result is not None:
                return result
        
        assignment[index] = -1
        return None
    
    # Start with all unassigned
    assignment = [-1] * n
    result = backtrack(assignment, 0)
    
    if result is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(str(x) for x in result))

solve()
