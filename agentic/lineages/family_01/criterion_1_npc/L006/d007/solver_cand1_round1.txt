import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The matrices M1, M2, M3 satisfy:
    # {M1, M1} = 2I, {M2, M2} = 2I, {M3, M3} = 2I
    # {M1, M2} = 0, {M1, M3} = 0, {M2, M3} = 0
    # The condition sum_{u,v} {Ms_u, Ms_v} = 0 is equivalent to:
    # sum_{u,v} delta_{s_u, s_v} * 2I = 0
    # This means for each state k in {1, 2, 3}, the number of edges (u, v)
    # such that s_u = s_v = k must be 0.
    # This is equivalent to saying that for each k, the set of nodes with state k
    # must be an independent set in the graph.
    # We need to partition the vertices into 3 independent sets.
    # This is the 3-coloring problem.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
