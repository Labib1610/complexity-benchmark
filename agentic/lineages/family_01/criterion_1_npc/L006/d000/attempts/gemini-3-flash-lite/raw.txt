import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    assignment = [0] * n
    
    # We want lexicographically smallest (s1, s2, ..., sn).
    # This means we try to assign 1 to s1, then 1 to s2, etc.
    # Backtracking with pruning.
    
    def backtrack(idx_sub):
        if idx_sub == n:
            return True
        
        # Try assigning vault 1, then 2, then 3
        for vault in [1, 2, 3]:
            # Check if any incompatible substance already in this vault
            conflict = False
            for prev in range(idx_sub):
                if (assignment[prev] == vault) and (adj[idx_sub + 1] & (1 << prev)):
                    conflict = True
                    break
            
            if not conflict:
                assignment[idx_sub] = vault
                if backtrack(idx_sub + 1):
                    return True
        
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
