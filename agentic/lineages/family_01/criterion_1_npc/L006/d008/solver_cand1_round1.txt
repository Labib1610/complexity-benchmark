```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition: s_u^3 + s_v^3 = s_u^2 * s_v + s_u * s_v^2 (mod 7)
    # s_u^3 - s_u^2 * s_v - s_u * s_v^2 + s_v^3 = 0 (mod 7)
    # s_u^2(s_u - s_v) - s_v^2(s_u - s_v) = 0 (mod 7)
    # (s_u^2 - s_v^2)(s_u - s_v) = 0 (mod 7)
    # (s_u - s_v)(s_u + s_v)(s_u - s_v) = 0 (mod 7)
    # (s_u - s_v)^2 * (s_u + s_v) = 0 (mod 7)
    # Since s_u, s_v in {1, 2, 3}, s_u - s_v is in {-2, -1, 0, 1, 2}.
    # (s_u - s_v)^2 is 0 only if s_u = s_v.
    # If s_u != s_v, then (s_u - s_v)^2 is 1 or 4.
    # So we need s_u + s_v = 0 (mod 7).
    # Since s_u, s_v in {1, 2, 3}, s_u + s_v is in {2, 3, 4, 5, 6}.
    # None of these are 0 mod 7.
    # Thus, the condition is satisfied if and only if s_u = s_v.
    # To avoid the collision, we must have s_u != s_v for all coupled pairs.
    # This is a graph coloring problem with 3 colors.

    assignment = [0] * n
    
    def can_color(u, c):
        for v in adj[u]:
            if assignment[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if can_color(u, c):
                assignment[u] = c
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
```
