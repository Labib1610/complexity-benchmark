```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Perfect squares: 1*1=1, 2*2=4, 3*3=9.
    # So, if u and v are connected, f[u]*f[v] cannot be 1, 4, or 9.
    # f[u]*f[v] = 1 iff f[u]=1 and f[v]=1.
    # f[u]*f[v] = 4 iff f[u]=2 and f[v]=2.
    # f[u]*f[v] = 9 iff f[u]=3 and f[v]=3.
    # This means for any edge (u, v), f[u] != f[v].
    # This is exactly the definition of a graph coloring problem with 3 colors.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
```
