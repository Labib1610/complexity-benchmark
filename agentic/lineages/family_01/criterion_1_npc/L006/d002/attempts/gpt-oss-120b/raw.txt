import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    colors = [-1] * n  # -1 uncolored, else 0,1,2

    # Precompute degree order for tie-breaking
    degree = [len(adj[i]) for i in range(n)]

    # DSATUR: select vertex with max number of distinct neighbor colors,
    # break ties by degree.
    def select_vertex():
        best = -1
        best_sat = -1
        best_deg = -1
        for v in range(n):
            if colors[v] != -1:
                continue
            neighbor_colors = set()
            for nb in adj[v]:
                c = colors[nb]
                if c != -1:
                    neighbor_colors.add(c)
            sat = len(neighbor_colors)
            deg = degree[v]
            if sat > best_sat or (sat == best_sat and deg > best_deg):
                best = v
                best_sat = sat
                best_deg = deg
        return best

    def backtrack(colored_cnt: int) -> bool:
        if colored_cnt == n:
            return True
        v = select_vertex()
        # compute forbidden colors
        forbidden = 0
        for nb in adj[v]:
            c = colors[nb]
            if c != -1:
                forbidden |= 1 << c
        for col in range(3):
            if not (forbidden & (1 << col)):
                colors[v] = col
                if backtrack(colored_cnt + 1):
                    return True
                colors[v] = -1
        return False

    possible = backtrack(0)
    out = []
    if possible:
        out.append("YES")
        out.append(' '.join(str(c+1) for c in colors))
    else:
        out.append("NO")
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
