import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition: |c_u - c_v| >= 1.
    # This means c_u != c_v.
    # Wait, the problem says: "Two assigned protocols are considered 
    # incompatible if and only if the absolute difference between their 
    # designated integers is at least 1."
    # If |c_u - c_v| >= 1, they are incompatible.
    # This means if there is a constraint between u and v, 
    # they MUST have the same protocol (c_u == c_v).
    # Let's re-read: "the protocols assigned to the two tanks in any 
    # proximity pair must be strictly incompatible with each other."
    # If they are incompatible when |c_u - c_v| >= 1, then they must 
    # satisfy |c_u - c_v| < 1.
    # Since c_i are integers {1, 2, 3}, |c_u - c_v| < 1 implies c_u = c_v.
    
    # So, for every edge (u, v), we must have c_u = c_v.
    # This means all tanks in the same connected component must have the same protocol.
    
    parent = list(range(n))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            
    for u in range(n):
        for v in adj[u]:
            union(u, v)
            
    # All nodes in the same component must have the same value.
    # Are there any constraints that prevent this?
    # The problem says: "Two assigned protocols are considered incompatible 
    # if and only if the absolute difference between their designated 
    # integers is at least 1."
    # This means if there is an edge, they MUST be incompatible.
    # So if there is an edge (u, v), we must have |c_u - c_v| >= 1.
    # Wait, my previous interpretation was wrong.
    # "The protocols assigned to the two tanks in any proximity pair must 
    # be strictly incompatible with each other."
    # Incompatible means |c_u - c_v| >= 1.
    # Since c_i are {1, 2, 3}, |c_u - c_v| >= 1 is true for any c_u != c_v.
    # If c_u == c_v, then |c_u - c_v| = 0, which is NOT >= 1.
    # So the condition is simply: c_u != c_v.
    # This is a standard graph coloring problem with 3 colors.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
