import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        # The condition: product of codes is not a perfect square.
        # Possible codes: 1, 2, 3.
        # Products:
        # 1*1=1 (sq), 1*2=2, 1*3=3
        # 2*2=4 (sq), 2*3=6
        # 3*3=9 (sq)
        # So, u and v cannot have the same code.
        # This means the graph must be 3-colorable.
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
