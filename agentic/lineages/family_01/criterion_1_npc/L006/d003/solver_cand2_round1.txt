```python
import sys

# The problem asks to color a graph with 3 colors such that no two adjacent
# vertices have the same color. This is the 3-coloring problem, which is NP-complete.
# However, n is at most 40. 3^40 is too large, but we can use backtracking with
# pruning or a simple recursive search. Given the constraints and the nature of
# the problem, a backtracking approach with a heuristic (like MRV) should work.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # For n=40, simple backtracking might be slow if the graph is dense.
    # However, the problem constraints and the nature of 3-coloring on small graphs
    # usually allow backtracking to pass if implemented efficiently.
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
