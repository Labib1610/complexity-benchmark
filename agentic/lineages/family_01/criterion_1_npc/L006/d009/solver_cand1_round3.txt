import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The condition: s_n^3 s_u^3 + 3 s_u^3 + s_n^3 s_v^3 + 3 s_v^3 \equiv s_u s_v (s_n^3 s_u + s_n^3 s_v + 3 s_u + 3 s_v) (mod 7)
    # Let f(s_u, s_v, s_n) = s_n^3 s_u^3 + 3 s_u^3 + s_n^3 s_v^3 + 3 s_v^3 - s_u s_v (s_n^3 s_u + s_n^3 s_v + 3 s_u + 3 s_v)
    # We need f(s_u, s_v, s_n) % 7 != 0
    
    def check(su, sv, sn):
        sn3 = pow(sn, 3, 7)
        su3 = pow(su, 3, 7)
        sv3 = pow(sv, 3, 7)
        lhs = (sn3 * su3 + 3 * su3 + sn3 * sv3 + 3 * sv3) % 7
        rhs = (su * sv * (sn3 * su + sn3 * sv + 3 * su + 3 * sv)) % 7
        return lhs != rhs

    # Try all possible values for s_n (1, 2, 3)
    for sn in [1, 2, 3]:
        # For a fixed sn, we need to assign s_i in {1, 2, 3} such that for all (u, v) in edges, check(s_u, s_v, sn) is true.
        # This is a 3-coloring problem variant. Since n <= 40, we can use backtracking.
        assignment = [0] * n
        assignment[n-1] = sn
        
        def backtrack(idx):
            if idx == n:
                return True
            if idx == n - 1:
                return backtrack(idx + 1)
            
            for val in [1, 2, 3]:
                possible = True
                for u, v in edges:
                    if u == idx and assignment[v] != 0:
                        if not check(val, assignment[v], sn):
                            possible = False; break
                    elif v == idx and assignment[u] != 0:
                        if not check(assignment[u], val, sn):
                            possible = False; break
                if possible:
                    assignment[idx] = val
                    if backtrack(idx + 1):
                        return True
                    assignment[idx] = 0
            return False

        if backtrack(0):
            print("YES")
            print(*(assignment))
            return

    print("NO")

solve()
