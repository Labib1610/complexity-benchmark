import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))

    # precompute forbidden for each sn, a,b in {1,2,3}
    forbidden = {}
    for sn in (1, 2, 3):
        f = [[False]*4 for _ in range(4)]  # 1-indexed
        for a in (1,2,3):
            for b in (1,2,3):
                left = (sn**3 * a**3 + 3*a**3 + sn**3 * b**3 + 3*b**3) % 7
                right = (a*b*(sn**3 * a + sn**3 * b + 3*a + 3*b)) % 7
                if left == right:
                    f[a][b] = True
                    f[b][a] = True
        forbidden[sn] = f

    order = sorted(range(n), key=lambda x: -len(adj[x]))

    for sn in (1,2,3):
        f = forbidden[sn]
        assign = [0]*n
        assign[n-1] = sn  # fix last node

        # check edges incident to n-1 with unassigned others later
        def dfs(pos):
            if pos == n:
                return True
            v = order[pos]
            if assign[v]:
                return dfs(pos+1)
            for col in (1,2,3):
                ok = True
                # check against already assigned neighbors
                for nb in adj[v]:
                    if assign[nb]:
                        if f[col][assign[nb]]:
                            ok = False
                            break
                if not ok:
                    continue
                assign[v] = col
                if dfs(pos+1):
                    return True
                assign[v] = 0
            return False

        # early check for edges where both ends are n-1 (none) or one is n-1
        early_ok = True
        for nb in adj[n-1]:
            if f[sn][assign[nb]]:  # assign[nb] is 0 now, ignore
                pass
        if dfs(0):
            print("YES")
            print(' '.join(map(str, assign)))
            return
    print("NO")

if __name__ == "__main__":
    solve()
