def solve():
    n, m = map(int, input().split())
    
    # Read edges
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u-1, v-1))  # Convert to 0-indexed
    
    # Precompute valid assignments
    # For each possible state of s_n (1,2,3), precompute which pairs (s_u, s_v) are valid
    valid_pairs = {}
    for s_n in [1, 2, 3]:
        valid_pairs[s_n] = set()
        for s_u in [1, 2, 3]:
            for s_v in [1, 2, 3]:
                left = (s_n**3 * s_u**3 + 3 * s_u**3 + s_n**3 * s_v**3 + 3 * s_v**3) % 7
                right = (s_u * s_v * (s_n**3 * s_u + s_n**3 * s_v + 3 * s_u + 3 * s_v)) % 7
                if left != right:
                    valid_pairs[s_n].add((s_u, s_v))
    
    # Try all 3 possible states for the last resonator
    for s_n in [1, 2, 3]:
        # Check if this s_n is valid by testing all edges
        valid = True
        for u, v in edges:
            if u == n-1 and v == n-1:
                continue
            # We need to check both directions since the constraint is symmetric
            if (s_n, s_n) not in valid_pairs[s_n]:
                valid = False
                break
            if u == n-1:  # u is the terminal
                found_valid = False
                for s_v in [1, 2, 3]:
                    if (s_n, s_v) in valid_pairs[s_n]:
                        found_valid = True
                        break
                if not found_valid:
                    valid = False
                    break
            elif v == n-1:  # v is the terminal
                found_valid = False
                for s_u in [1, 2, 3]:
                    if (s_u, s_n) in valid_pairs[s_n]:
                        found_valid = True
                        break
                if not found_valid:
                    valid = False
                    break
        if not valid:
            continue
            
        # Now do graph coloring with the constraint
        # We'll use backtracking
        assignment = [0] * n
        assignment[n-1] = s_n
        
        def is_valid_color(node, color):
            for u, v in edges:
                if u == node:
                    if assignment[v] != 0 and (color, assignment[v]) not in valid_pairs[s_n]:
                        return False
                elif v == node:
                    if assignment[u] != 0 and (assignment[u], color) not in valid_pairs[s_n]:
                        return False
            return True
        
        def backtrack(node):
            if node == n:
                return True
            if node == n-1:  # Already assigned
                return backtrack(node + 1)
                
            for color in [1, 2, 3]:
                if is_valid_color(node, color):
                    assignment[node] = color
                    if backtrack(node + 1):
                        return True
                    assignment[node] = 0
            return False
        
        if backtrack(0):
            print("YES")
            print(*assignment)
            return
    
    print("NO")

solve()
