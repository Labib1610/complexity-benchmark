```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_safe(u, hopper):
        for v in adj[u]:
            if assignment[v] == hopper:
                return False
        return True

    # Since we want to minimize sum(h_i * 4^(n-i)), we want to pick the 
    # smallest possible h_i for each i from 1 to n.
    # This is a greedy approach with backtracking.
    
    def backtrack(u):
        if u > n:
            return True
        
        for h in [1, 2, 3]:
            if is_safe(u, h):
                assignment[u] = h
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
