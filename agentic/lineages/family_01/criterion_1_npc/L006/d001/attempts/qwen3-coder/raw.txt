import sys
from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        # No conflicts, assign all to agent 1
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list for the graph
    adj = defaultdict(list)
    index = 2
    edges = []
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
        index += 2
    
    # Try to 3-color the graph with lexicographically smallest assignment
    # We can model this as a constraint satisfaction problem
    # Since n <= 40, but we have at most 3 colors, and we want lexicographically smallest,
    # we can use backtracking with pruning.
    
    # However, there's a more efficient way: since we want lexicographically smallest,
    # we can iterate through assignments in lexicographical order and check validity.
    # But even better: since we're looking for a 3-coloring, and want the lex smallest,
    # we can do backtracking with early pruning.
    
    color = [0] * n  # 0 means unassigned, 1,2,3 are colors
    
    def is_valid(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
        
        for c in range(1, 4):
            if is_valid(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

solve()
