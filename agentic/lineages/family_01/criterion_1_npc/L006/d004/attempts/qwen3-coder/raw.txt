```python
from itertools import product

def is_square(n):
    if n < 0:
        return False
    root = int(n**0.5)
    return root * root == n

def solve():
    n, m = map(int, input().split())
    
    # If no interactions, any assignment works
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # Try all 3^n combinations
    for assignment in product([1, 2, 3], repeat=n):
        valid = True
        for u, v in edges:
            product_uv = assignment[u-1] * assignment[v-1]
            if is_square(product_uv):
                valid = False
                break
        if valid:
            print("YES")
            print(" ".join(map(str, assignment)))
            return
    
    print("NO")

solve()
```
