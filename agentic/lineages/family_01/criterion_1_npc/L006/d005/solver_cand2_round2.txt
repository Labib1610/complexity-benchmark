import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Resonance failure condition: (c_u + c_v)^2 % (c_u * c_v) == 0
    # Possible pairs (c_u, c_v) where c_u, c_v in {1, 2, 3}:
    # (1, 1): (1+1)^2 % (1*1) = 4 % 1 = 0 (FAIL)
    # (1, 2): (1+2)^2 % (1*2) = 9 % 2 = 1 (OK)
    # (1, 3): (1+3)^2 % (1*3) = 16 % 3 = 1 (OK)
    # (2, 1): (2+1)^2 % (2*1) = 9 % 2 = 1 (OK)
    # (2, 2): (2+2)^2 % (2*2) = 16 % 4 = 0 (FAIL)
    # (2, 3): (2+3)^2 % (2*3) = 25 % 6 = 1 (OK)
    # (3, 1): (3+1)^2 % (3*1) = 16 % 3 = 1 (OK)
    # (3, 2): (3+2)^2 % (3*2) = 25 % 6 = 1 (OK)
    # (3, 3): (3+3)^2 % (3*3) = 36 % 9 = 0 (FAIL)
    #
    # Failures occur if c_u == c_v.
    # This is equivalent to a graph coloring problem with 3 colors where
    # adjacent nodes must have different colors.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
