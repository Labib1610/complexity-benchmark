import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The condition "no directed cycles" means we need a topological ordering.
    # If we assign values 1, 2, 3, the divisibility rules are:
    # 1 divides 2, 1 divides 3, 2 divides 2, 3 divides 3.
    # If we assign 1 to a set of nodes S1, 2 to S2, 3 to S3:
    # - Within S1, all pairs have bidirectional edges (cycle if |S1| > 1).
    # - Within S2, all pairs have bidirectional edges (cycle if |S2| > 1).
    # - Within S3, all pairs have bidirectional edges (cycle if |S3| > 1).
    # - Between S1 and S2: 1 divides 2, so edges go S1 -> S2.
    # - Between S1 and S3: 1 divides 3, so edges go S1 -> S3.
    # - Between S2 and S3: No divisibility, no edges.
    # Thus, we need to partition nodes into S1, S2, S3 such that:
    # |S1| <= 1, |S2| <= 1, |S3| <= 1 is not strictly required, 
    # but actually, if we assign 1, 2, 3, the only way to avoid cycles is
    # to have at most one node in each set if they are connected.
    # Actually, the condition is simpler: we need a topological sort.
    # If we assign values such that if u -> v is an edge, val(u) <= val(v)
    # and if val(u) == val(v), there is no edge.
    # Since we only have 3 values, we can try to partition the graph into 
    # 3 independent sets S1, S2, S3 such that all edges go S1->S2, S1->S3, S2->S3.
    
    # This is equivalent to finding a 3-coloring of the graph such that
    # if we color nodes with 1, 2, 3, all edges (u, v) satisfy color(u) < color(v).
    # This is exactly the definition of a graph being a comparability graph 
    # of a poset of height 3.
    
    # Given n <= 40, we can use backtracking to assign colors 1, 2, 3.
    colors = [0] * n
    
    def is_valid(u, c):
        for v1, v2 in edges:
            if v1 == u:
                if colors[v2] != 0 and not (c < colors[v2]):
                    return False
            elif v2 == u:
                if colors[v1] != 0 and not (colors[v1] < c):
                    return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
