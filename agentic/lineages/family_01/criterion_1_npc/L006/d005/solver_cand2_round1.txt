import sys

def solve():
    # The condition is: (c_u^2 + c_v^2) % (c_u * c_v) != 0
    # Let's check all pairs (c_u, c_v) where c_u, c_v in {1, 2, 3}:
    # (1, 1): 1^2 + 1^2 = 2. 2 % (1*1) = 0. (FAIL)
    # (1, 2): 1^2 + 2^2 = 5. 5 % (1*2) = 1 != 0. (OK)
    # (1, 3): 1^2 + 3^2 = 10. 10 % (1*3) = 1 != 0. (OK)
    # (2, 1): 1^2 + 2^2 = 5. 5 % (1*2) = 1 != 0. (OK)
    # (2, 2): 2^2 + 2^2 = 8. 8 % (2*2) = 0. (FAIL)
    # (2, 3): 2^2 + 3^2 = 13. 13 % (2*3) = 1 != 0. (OK)
    # (3, 1): 1^2 + 3^2 = 10. 10 % (1*3) = 1 != 0. (OK)
    # (3, 2): 2^2 + 3^2 = 13. 13 % (2*3) = 1 != 0. (OK)
    # (3, 3): 3^2 + 3^2 = 18. 18 % (3*3) = 0. (FAIL)
    
    # So, c_u and c_v cannot be equal.
    # This means the graph must be 3-colorable.
    # Since we have 3 colors {1, 2, 3}, any graph that is 3-colorable works.
    # However, the problem asks for a valid assignment for any given graph.
    # Wait, the condition is simply that no two connected nodes have the same color.
    # This is exactly the definition of graph coloring.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # Greedy coloring with 3 colors
    for i in range(1, n + 1):
        if colors[i] == 0:
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        # Try to assign a color different from u
                        used = {colors[u]}
                        for neighbor in adj[v]:
                            if colors[neighbor] != 0:
                                used.add(colors[neighbor])
                        
                        found = False
                        for c in [1, 2, 3]:
                            if c not in used:
                                colors[v] = c
                                found = True
                                break
                        if not found:
                            print("NO")
                            return
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
