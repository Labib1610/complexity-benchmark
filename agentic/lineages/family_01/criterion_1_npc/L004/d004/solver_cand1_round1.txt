```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    f = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        for val in [1, 2, 3]:
            possible = True
            for neighbor in adj[u]:
                if f[neighbor] == val:
                    possible = False
                    break
            
            if possible:
                f[u] = val
                if backtrack(u + 1):
                    return True
                f[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(f))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
