import sys
import math

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Resonance condition: gcd(b_u + 2*b_v, b_v + 2*b_u) > 1
    # Stable condition: gcd(b_u + 2*b_v, b_v + 2*b_u) == 1
    # Possible values for b_i are {1, 2, 3}
    # Pairs (b_u, b_v):
    # (1, 1): gcd(1+2, 1+2) = 3 > 1 (Invalid)
    # (1, 2): gcd(1+4, 2+2) = gcd(5, 4) = 1 (Valid)
    # (1, 3): gcd(1+6, 3+2) = gcd(7, 5) = 1 (Valid)
    # (2, 1): gcd(2+2, 1+4) = gcd(4, 5) = 1 (Valid)
    # (2, 2): gcd(2+4, 2+4) = 6 > 1 (Invalid)
    # (2, 3): gcd(2+6, 3+4) = gcd(8, 7) = 1 (Valid)
    # (3, 1): gcd(3+2, 1+6) = gcd(5, 7) = 1 (Valid)
    # (3, 2): gcd(3+4, 2+6) = gcd(7, 8) = 1 (Valid)
    # (3, 3): gcd(3+6, 3+6) = 9 > 1 (Invalid)
    
    # So, b_u != b_v is the only constraint.
    # This is equivalent to 3-coloring the graph.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
