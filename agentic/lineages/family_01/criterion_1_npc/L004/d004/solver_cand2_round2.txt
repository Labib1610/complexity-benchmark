import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Energy function f(x) = 17x^3 - 86x^2 + 73x where x = (b_u - b_v)^2
    # Possible values for b_u - b_v are {-2, -1, 0, 1, 2}
    # x = (b_u - b_v)^2 can be {0, 1, 4}
    # f(0) = 0
    # f(1) = 17 - 86 + 73 = 4
    # f(4) = 17(64) - 86(16) + 73(4) = 1088 - 1376 + 292 = 4
    # So, energy is 4 if b_u != b_v, and 0 if b_u == b_v.
    # Total energy = 4 * (number of edges where b_u != b_v)
    # We want total energy = 4m, which means for all edges (u, v), b_u != b_v.
    # This is equivalent to checking if the graph is 3-colorable.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
