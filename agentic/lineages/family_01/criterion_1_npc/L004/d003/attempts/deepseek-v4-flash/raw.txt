We need to assign each emitter a frequency from {1, 2, 3} such that every given pair has a difference of at least 1, i.e., they cannot be equal.

Since n ≤ 40, we can use backtracking with pruning and try to build the lexicographically smallest assignment.  
We’ll assign frequencies in order from emitter 1 to n, trying 1, then 2, then 3.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
    
    # We'll use backtracking
    freq = [0] * n
    
    def backtrack(pos):
        if pos == n:
            return True
        for f in (1, 2, 3):
            # Check constraints with already assigned neighbors
            ok = True
            for nb in adj[pos]:
                if nb < pos and freq[nb] == f:
                    ok = False
                    break
            if ok:
                freq[pos] = f
                if backtrack(pos + 1):
                    return True
                freq[pos] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, freq)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
