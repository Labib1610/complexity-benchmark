import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * n
    
    # We use backtracking with a heuristic to prune the search space.
    # Since n <= 40, we can use bitmasks to check constraints quickly.
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try buses 1, 2, 3 in order to ensure lexicographical smallest
        for bus in range(1, 4):
            # Check if any neighbor already assigned to this bus
            conflict = False
            for v in range(u):
                if (adj[u] & (1 << v)) and assignment[v] == bus:
                    conflict = True
                    break
            
            if not conflict:
                assignment[u] = bus
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Optimization: Sort nodes by degree (descending) to fail faster,
    # but the problem requires lexicographical order of the output array.
    # Given n=40, simple backtracking with pruning is usually sufficient
    # if the constraints are dense.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
