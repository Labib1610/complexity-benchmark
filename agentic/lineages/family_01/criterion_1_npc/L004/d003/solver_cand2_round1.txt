import sys

def solve():
    # The integral I = integral_0^pi sin(b_u * x) * sin(b_v * x) dx
    # Using the identity sin(A)sin(B) = 0.5 * (cos(A-B) - cos(A+B))
    # I = 0.5 * [integral_0^pi cos((b_u-b_v)x) dx - integral_0^pi cos((b_u+b_v)x) dx]
    # If b_u == b_v:
    # I = 0.5 * [integral_0^pi 1 dx - integral_0^pi cos(2*b_u*x) dx]
    # I = 0.5 * [pi - 0] = pi/2 > 0.
    # So b_u cannot equal b_v.
    # If b_u != b_v:
    # I = 0.5 * [sin((b_u-b_v)x)/(b_u-b_v) - sin((b_u+b_v)x)/(b_u+b_v)] evaluated from 0 to pi.
    # Since b_u, b_v in {1, 2, 3}, b_u-b_v and b_u+b_v are integers.
    # sin(k * pi) = 0 for any integer k.
    # So I = 0.5 * [0 - 0] = 0.
    # Thus, the condition "integral > 0" is equivalent to "b_u == b_v".
    # The problem reduces to: Can we color the graph with 3 colors such that no two 
    # adjacent nodes have the same color?
    # This is the 3-coloring problem, which is NP-complete, but n <= 40 is small.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
