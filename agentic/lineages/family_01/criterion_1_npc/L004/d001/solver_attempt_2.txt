```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Perfect square products:
    # 1*1 = 1 (sq)
    # 2*2 = 4 (sq)
    # 3*3 = 9 (sq)
    # 1*2 = 2 (no)
    # 1*3 = 3 (no)
    # 2*3 = 6 (no)
    # So, two modules cannot have the same multiplier.
    # This means the graph must be 3-colorable.
    
    assignment = [0] * n
    
    def is_safe(u, val):
        for v in adj[u]:
            if assignment[v] == val:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        for val in [1, 2, 3]:
            if is_safe(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
