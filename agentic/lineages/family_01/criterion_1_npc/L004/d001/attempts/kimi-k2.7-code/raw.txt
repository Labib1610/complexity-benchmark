import sys

def has_k4(adj):
    n = len(adj)
    for a in range(n - 3):
        for b in range(a + 1, n - 2):
            if not ((adj[a] >> b) & 1):
                continue
            for c in range(b + 1, n - 1):
                if not (((adj[a] >> c) & 1) and ((adj[b] >> c) & 1)):
                    continue
                if (adj[a] & adj[b] & adj[c]) >> (c + 1):
                    return True
    return False

def solve_component(comp, g):
    k = len(comp)
    if k == 1:
        return [1]
    loc = {v: i for i, v in enumerate(comp)}
    local_adj = [0] * k
    for i, v in enumerate(comp):
        mask = 0
        for u in g[v]:
            if u in loc:
                mask |= 1 << loc[u]
        local_adj[i] = mask
    if has_k4(local_adj):
        return None

    col = [0] * k
    cm = [0, 0, 0]
    col[0] = 1
    cm[0] = 1
    ans = None

    def dfs(start):
        nonlocal ans
        i = start
        while i < k and col[i]:
            i += 1
        if i == k:
            ans = col[:]
            return True
        a = local_adj[i]
        opts = []
        if a & cm[0] == 0:
            opts.append(0)
        if a & cm[1] == 0:
            opts.append(1)
        if a & cm[2] == 0:
            opts.append(2)
        for c in opts:
            col[i] = c + 1
            cm[c] |= 1 << i
            stack = []
            ok = True
            changed = True
            while changed and ok:
                changed = False
                for v in range(k):
                    if col[v]:
                        continue
                    av = local_adj[v]
                    mask = 0
                    if av & cm[0] == 0:
                        mask |= 1
                    if av & cm[1] == 0:
                        mask |= 2
                    if av & cm[2] == 0:
                        mask |= 4
                    if mask == 0:
                        ok = False
                        break
                    if mask == 1:
                        col[v] = 1
                        cm[0] |= 1 << v
                        stack.append(v)
                        changed = True
                    elif mask == 2:
                        col[v] = 2
                        cm[1] |= 1 << v
                        stack.append(v)
                        changed = True
                    elif mask == 4:
                        col[v] = 3
                        cm[2] |= 1 << v
                        stack.append(v)
                        changed = True
            if ok and dfs(i + 1):
                return True
            for v in stack:
                cc = col[v]
                col[v] = 0
                cm[cc - 1] ^= 1 << v
            col[i] = 0
            cm[c] ^= 1 << i
        return False

    if dfs(0):
        return ans
    return None

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    g = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            sys.stdout.write('NO\n')
            return
        if not ((adj[u] >> v) & 1):
            adj[u] |= 1 << v
            adj[v] |= 1 << u
            g[u].append(v)
            g[v].append(u)

    visited = [False] * n
    components = []
    for v in range(n):
        if not visited[v]:
            stack = [v]
            visited[v] = True
            comp = []
            while stack:
                x = stack.pop()
                comp.append(x)
                for y in g[x]:
                    if not visited[y]:
                        visited[y] = True
                        stack.append(y)
            comp.sort()
            components.append(comp)

    answer = [0] * n
    for comp in components:
        local = solve_component(comp, g)
        if local is None:
            sys.stdout.write('NO\n')
            return
        for idx, v in enumerate(comp):
            answer[v] = local[idx]

    sys.stdout.write('YES\n')
    sys.stdout.write(' '.join(map(str, answer)) + '\n')

if __name__ == '__main__':
    solve()
