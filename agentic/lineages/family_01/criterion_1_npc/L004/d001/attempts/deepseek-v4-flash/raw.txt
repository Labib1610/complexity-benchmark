We can model this as a graph coloring problem with 3 colors.  
Since \( n \le 40 \), we can use backtracking with pruning to find the lexicographically smallest valid assignment.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * n  # 0 = unassigned, 1,2,3 = bus

    # Precompute adjacency as bitmasks for speed
    adj_mask = [0] * n
    for i in range(n):
        mask = 0
        for j in adj[i]:
            mask |= (1 << j)
        adj_mask[i] = mask

    # Backtracking with pruning
    def backtrack(idx):
        if idx == n:
            return True
        # Try colors 1,2,3 in order for lexicographic smallest
        for c in (1, 2, 3):
            ok = True
            for v in adj[idx]:
                if color[v] == c:
                    ok = False
                    break
            if ok:
                color[idx] = c
                if backtrack(idx + 1):
                    return True
                color[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(str(c) for c in color))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
