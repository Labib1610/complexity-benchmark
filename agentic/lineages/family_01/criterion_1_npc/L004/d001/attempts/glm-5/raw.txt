import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    full = (1 << n) - 1
    # Precompute popcount
    popcnt = [0] * (1 << n)
    for i in range(1, 1 << n):
        popcnt[i] = popcnt[i >> 1] + (i & 1)
    
    # Find all maximal independent sets using Bron-Kerbosch with pivoting
    maximal_IS = []
    def bron_kerbosch(R, P, X):
        if P == 0 and X == 0:
            maximal_IS.append(R)
            return
        # Choose pivot u in P | X with max degree in P
        pivot = -1
        best = -1
        candidates = P | X
        while candidates:
            u = (candidates & -candidates).bit_length() - 1
            candidates &= candidates - 1
            cnt = popcnt[P & adj[u]]
            if cnt > best:
                best = cnt
                pivot = u
        # Iterate over P \ N(pivot)
        sub = P & ~adj[pivot]
        while sub:
            u = (sub & -sub).bit_length() - 1
            sub &= sub - 1
            Nu = adj[u]
            bron_kerbosch(R | (1 << u), P & ~Nu & ~(1 << u), X & ~Nu)
            P &= ~(1 << u)
            X |= (1 << u)
    
    bron_kerbosch(0, full, 0)
    
    # Sort maximal independent sets by popcount (ascending) for pruning
    maximal_IS.sort(key=lambda x: popcnt[x])
    
    # For each maximal IS, compute its complement (vertices not in it)
    complements = [(full ^ iset) for iset in maximal_IS]
    
    # For each complement, check if it's bipartite; if so, record the 2-coloring
    valid = []
    for idx, comp in enumerate(complements):
        color = [0] * n
        is_bip = True
        temp = comp
        while temp:
            start = (temp & -temp).bit_length() - 1
            stack = [start]
            color[start] = 1
            temp &= ~(1 << start)
            while stack and is_bip:
                u = stack.pop()
                for v in range(n):
                    if (comp >> v) & 1 and (adj[u] >> v) & 1:
                        if color[v] == 0:
                            color[v] = 3 - color[u]
                            stack.append(v)
                            temp &= ~(1 << v)
                        elif color[v] == color[u]:
                            is_bip = False
                            break
                if not is_bip:
                    break
            if not is_bip:
                break
        if is_bip:
            valid.append((maximal_IS[idx], comp, color))
    
    if not valid:
        print("NO")
        return
    
    # Find lexicographically smallest assignment
    best_assignment = None
    for iset, comp, color in valid:
        assignment = [0] * n
        for i in range(n):
            if (iset >> i) & 1:
                assignment[i] = 1
            else:
                assignment[i] = color[i] + 1  # color is 1 or 2, map to 2 or 3
        if best_assignment is None or assignment < best_assignment:
            best_assignment = assignment
    
    print("YES")
    print(' '.join(map(str, best_assignment)))

solve()
