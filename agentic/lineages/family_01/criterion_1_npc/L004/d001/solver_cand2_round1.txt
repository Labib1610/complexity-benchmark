import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem asks for an assignment b_i in {1, 2, 3} such that
    # if (u, v) is a conflict, b_u != b_v.
    # Specifically, if (u, v) is a conflict, one must be strictly earlier.
    # This means b_u < b_v or b_v < b_u.
    # This is equivalent to finding a 3-coloring of the graph.
    # Since n <= 40, we can use backtracking with pruning.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        # Heuristic: try colors 1, 2, 3
        for color in range(1, 4):
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # To optimize, sort nodes by degree descending
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to process in order of degree
    mapping = {node: i for i, node in enumerate(nodes)}
    new_adj = [[] for _ in range(n)]
    for u in range(1, n + 1):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    res = [0] * n
    def backtrack_optimized(idx):
        if idx == n:
            return True
        u = idx
        for color in range(1, 4):
            possible = True
            for v in new_adj[u]:
                if res[v] == color:
                    possible = False
                    break
            if possible:
                res[u] = color
                if backtrack_optimized(idx + 1):
                    return True
                res[u] = 0
        return False

    if backtrack_optimized(0):
        print("YES")
        final_assignment = [0] * (n + 1)
        for i in range(n):
            final_assignment[nodes[i]] = res[i]
        print(*(final_assignment[1:]))
    else:
        print("NO")

solve()
