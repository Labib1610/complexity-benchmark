import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    assignment = [0] * n
    
    # We need the lexicographically smallest assignment.
    # This means we try assigning 1, then 2, then 3 for each module in order 1..n.
    
    def backtrack(module_idx):
        if module_idx == n:
            return True
        
        # Try buses 1, 2, 3
        for bus in range(1, 4):
            # Check if any neighbor already assigned to this bus
            can_assign = True
            for prev in range(module_idx):
                if (adj[module_idx + 1] & (1 << prev)):
                    if assignment[prev] == bus:
                        can_assign = False
                        break
            
            if can_assign:
                assignment[module_idx] = bus
                if backtrack(module_idx + 1):
                    return True
                assignment[module_idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
