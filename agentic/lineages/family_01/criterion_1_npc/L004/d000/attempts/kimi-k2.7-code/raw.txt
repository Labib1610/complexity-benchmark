 ```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    g = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            continue
        g[u].append(v)
        g[v].append(u)

    color = [-1] * n          # -1 uncolored, 0/1/2 colors
    avail = [7] * n           # bitmask of possible colors
    bit_to_col = [0] * 8
    bit_to_col[1] = 0
    bit_to_col[2] = 1
    bit_to_col[4] = 2

    sys.setrecursionlimit(10000)

    def propagate(start, changes):
        stack = [start]
        while stack:
            x = stack.pop()
            cx = color[x]
            mask = 1 << cx
            for y in g[x]:
                if color[y] == cx:
                    return False
                if color[y] == -1:
                    old = avail[y]
                    new = old & ~mask
                    if new == 0:
                        return False
                    if new != old:
                        avail[y] = new
                        changes.append((y, old))
                        if new & (new - 1) == 0:
                            cy = bit_to_col[new]
                            color[y] = cy
                            changes.append((y, -1))
                            stack.append(y)
        return True

    def dfs(i):
        while i < n and color[i] != -1:
            i += 1
        if i == n:
            return True
        if avail[i] == 0:
            return False
        old_mask = avail[i]
        mask = old_mask
        while mask:
            lsb = mask & -mask
            c = bit_to_col[lsb]
            mask ^= lsb
            color[i] = c
            avail[i] = lsb
            changes = [(i, old_mask)]
            if propagate(i, changes):
                if dfs(i + 1):
                    return True
            for v, old in reversed(changes):
                if old == -1:
                    color[v] = -1
                else:
                    avail[v] = old
            color[i] = -1
            avail[i] = old_mask
        return False

    if dfs(0):
        out = ['YES', ' '.join(str(c + 1) for c in color)]
        sys.stdout.write('\n'.join(out))
    else:
        sys.stdout.write('NO')

if __name__ == '__main__':
    solve()
```
