```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # Degree-based heuristic for variable ordering is not needed because
    # we need the lexicographically smallest assignment (1 to n).
    # However, we need to prune the search space effectively.
    
    def is_valid(u, bus):
        for v in adj[u]:
            if assignment[v] == bus:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        # Try buses 1, 2, 3 in order to ensure lexicographical smallest
        for bus in range(1, 4):
            if is_valid(u, bus):
                assignment[u] = bus
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Optimization: Forward checking or MRV could be used, but for n=40,
    # simple backtracking with pruning is often sufficient if the graph
    # is not extremely dense. Given the constraints and the requirement
    # for lexicographical order, we must assign 1, 2, ..., n.
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
```
