import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    assignment = [0] * n
    
    # To find the lexicographically smallest assignment, we try bus 1, 2, 3 in order.
    # Given n <= 40, we use backtracking with pruning.
    # To handle the "adversarial" ordering, we can reorder nodes by degree
    # but the problem asks for the lexicographical smallest assignment for 1..n.
    # We must stick to the 1..n order for the output, but we can use a heuristic
    # for the search order if needed. However, simple backtracking with bitmask
    # checks is usually fast enough for n=40 if the graph isn't extremely dense.
    
    bus_masks = [0, 0, 0]
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try buses 1, 2, 3
        for b in range(3):
            mask = 1 << u
            if not (bus_masks[b] & adj[u + 1]):
                bus_masks[b] |= mask
                assignment[u] = b + 1
                if backtrack(u + 1):
                    return True
                bus_masks[b] &= ~mask
        return False

    # The problem states the numbering is adversarial.
    # Let's reorder nodes by degree to speed up the search,
    # but we must map back to the original 1..n indices.
    degrees = [(bin(adj[i]).count('1'), i) for i in range(1, n + 1)]
    degrees.sort(key=lambda x: x[0], reverse=True)
    order = [x[1] for x in degrees]
    
    res = [0] * (n + 1)
    
    def backtrack_optimized(idx_in_order):
        if idx_in_order == n:
            return True
        
        u = order[idx_in_order]
        for b in range(3):
            if not (bus_masks[b] & adj[u]):
                bus_masks[b] |= (1 << (u - 1))
                res[u] = b + 1
                if backtrack_optimized(idx_in_order + 1):
                    return True
                bus_masks[b] &= ~(1 << (u - 1))
        return False

    if backtrack_optimized(0):
        print("YES")
        print(*(res[1:]))
    else:
        print("NO")

solve()
