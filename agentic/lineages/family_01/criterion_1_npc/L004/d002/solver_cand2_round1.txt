```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Condition: (b_u^2 + b_v^2) % (b_u * b_v) != 0
    # Possible values: {1, 2, 3}
    # Pairs (b_u, b_v):
    # (1, 1): 1^2+1^2 = 2, 1*1 = 1. 2%1 == 0 (FAIL)
    # (1, 2): 1^2+2^2 = 5, 1*2 = 2. 5%2 != 0 (OK)
    # (1, 3): 1^2+3^2 = 10, 1*3 = 3. 10%3 != 0 (OK)
    # (2, 2): 2^2+2^2 = 8, 2*2 = 4. 8%4 == 0 (FAIL)
    # (2, 3): 2^2+3^2 = 13, 2*3 = 6. 13%6 != 0 (OK)
    # (3, 3): 3^2+3^2 = 18, 3*3 = 9. 18%9 == 0 (FAIL)
    
    # So:
    # 1 cannot be with 1
    # 2 cannot be with 2
    # 3 cannot be with 3
    # This is equivalent to a 3-coloring problem where adjacent nodes must have different colors.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
```
