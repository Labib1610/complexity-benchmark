import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To find the lexicographically smallest assignment, we try assigning
    # modules 1 to n in order, and for each, try buses 1, 2, 3 in order.
    # This is a standard backtracking approach.
    
    def is_valid(u, bus):
        for v in adj[u]:
            if assignment[v] == bus:
                return False
        return True
    
    # Heuristic: Degree-based ordering is usually good, but the problem
    # specifically asks for the lexicographically smallest assignment,
    # which forces us to assign modules 1, 2, ..., n in that order.
    
    def backtrack(u):
        if u > n:
            return True
        
        for bus in range(1, 4):
            if is_valid(u, bus):
                assignment[u] = bus
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
