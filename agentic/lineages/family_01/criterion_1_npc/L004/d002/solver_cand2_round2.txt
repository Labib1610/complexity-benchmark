import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    assignment = [0] * n
    
    # To find the lexicographically smallest assignment, we use backtracking.
    # To optimize, we order modules by degree or simply use the natural order 1..n.
    # Given n <= 40, we need a pruning strategy.
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try buses 1, 2, 3 in order for lexicographical smallest
        for bus in [1, 2, 3]:
            can_place = True
            for prev in range(u):
                if (adj[u + 1] & (1 << prev)):
                    if assignment[prev] == bus:
                        can_place = False
                        break
            
            if can_place:
                assignment[u] = bus
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # The problem asks for lexicographically smallest sequence b1, b2, ..., bn.
    # This means we want b1 to be as small as possible, then b2, etc.
    # The standard backtracking approach naturally explores this.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
