import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition: x = b_u (mod 3) and x = b_v (mod 5).
    # By Chinese Remainder Theorem, x is unique mod 15.
    # x = b_u + 3k. b_u + 3k = b_v (mod 5) => 3k = b_v - b_u (mod 5).
    # Since 3*2 = 6 = 1 (mod 5), k = 2(b_v - b_u) (mod 5).
    # x = b_u + 3 * (2 * (b_v - b_u) % 5).
    # We need x > 3.
    # Let's precompute the forbidden pairs (b_u, b_v).
    
    forbidden = []
    for bu in [1, 2, 3]:
        for bv in [1, 2, 3]:
            # Find smallest positive x such that x % 3 == bu % 3 and x % 5 == bv % 5
            # x % 3 = bu, x % 5 = bv
            for x in range(1, 16):
                if x % 3 == bu % 3 and x % 5 == bv % 5:
                    if x <= 3:
                        forbidden.append((bu, bv))
                    break
    
    # Backtracking to find assignment
    assignment = [0] * n
    
    def is_valid(u, val):
        for v in adj[u]:
            if assignment[v] != 0:
                if (val, assignment[v]) in forbidden:
                    return False
                if (assignment[v], val) in forbidden:
                    return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for val in [1, 2, 3]:
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
