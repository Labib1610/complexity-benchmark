import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [0] * n
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # dp[mask] = bitmask of colors assigned to vertices in mask
    # mask is a bitmask of vertices (0 to n-1)
    # dp[mask] stores a bitmask of colors for each vertex in mask
    # bit 2*v and 2*v+1 store the color (0, 1, or 2) for vertex v
    # We use a dictionary to store only reachable states
    dp = {0: 0}
    
    for v in range(n):
        next_dp = {}
        # Neighbors with smaller index
        prev_neighbors_mask = 0
        for u in range(v):
            if (adj[v] >> u) & 1:
                prev_neighbors_mask |= (1 << u)
        
        # Iterate over all states from previous step
        for mask, colors in dp.items():
            # Try to add vertex v with color c (0, 1, 2)
            for c in range(3):
                # Check if color c is compatible with all neighbors in mask
                ok = True
                # Iterate over neighbors u in mask
                nm = mask & prev_neighbors_mask
                while nm:
                    u = (nm & -nm).bit_length() - 1
                    nm ^= (1 << u)
                    # Extract color of u
                    u_color = (colors >> (2 * u)) & 3
                    if u_color == c:
                        ok = False
                        break
                
                if ok:
                    new_mask = mask | (1 << v)
                    new_colors = colors | (c << (2 * v))
                    
                    if new_mask not in next_dp:
                        next_dp[new_mask] = new_colors
                    else:
                        # Lexicographical check: smaller colors for smaller vertices
                        # Since we process vertices in order, we want to minimize the color of the current vertex v
                        # But we also need to ensure the entire sequence is lexicographically smallest
                        # Since we process v in increasing order, and try colors 0, 1, 2 in order,
                        # the first valid assignment we find for v is the best for this prefix.
                        # However, we need to compare the entire sequence.
                        # Actually, since we iterate colors 0, 1, 2, and we keep the smallest colors for previous vertices,
                        # we just need to check if the new assignment is better.
                        # But the colors for previous vertices are fixed in `colors`.
                        # So we just compare the color of v.
                        # But wait, `colors` for previous vertices might differ.
                        # We need to compare the full sequence.
                        # Let's compare `new_colors` with `next_dp[new_mask]`
                        # We compare from vertex 0 to v.
                        # Since `colors` is the same for both (from the same `mask`), we just need to compare `c`.
                        # But `next_dp[new_mask]` could come from a different `mask`? No, `new_mask` includes v.
                        # So `next_dp[new_mask]` comes from some `prev_mask` and some `prev_colors`.
                        # We need to compare `new_colors` with existing `next_dp[new_mask]`.
                        # We compare from vertex 0 to v.
                        # Since we process v in order, the first time we set a color for v, it's the smallest possible for that prefix.
                        # But different paths can lead to the same `new_mask` with different colors for previous vertices.
                        # We need to find the lexicographically smallest sequence.
                        # We compare the two color assignments.
                        existing_colors = next_dp[new_mask]
                        # Compare from vertex 0 to v
                        is_better = False
                        for k in range(v + 1):
                            if (new_colors >> (2 * k)) & 3 < (existing_colors >> (2 * k)) & 3:
                                is_better = True
                                break
                            elif (new_colors >> (2 * k)) & 3 > (existing_colors >> (2 * k)) & 3:
                                break
                        if is_better:
                            next_dp[new_mask] = new_colors
        dp = next_dp
        if not dp:
            print("NO")
            return

    full_mask = (1 << n) - 1
    if full_mask in dp:
        final_colors = dp[full_mask]
        result = []
        for i in range(n):
            result.append(str(((final_colors >> (2 * i)) & 3) + 1))
        print("YES")
        print(" ".join(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
