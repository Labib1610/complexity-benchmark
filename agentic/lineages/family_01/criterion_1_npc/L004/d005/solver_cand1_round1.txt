import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Compatibility: (b_u^2 + b_v^2) % 7 in {3, 5, 6}
    # b in {1, 2, 3} => b^2 in {1, 4, 2}
    # Pairs (b_u^2, b_v^2):
    # (1,1) -> 2 (No)
    # (1,4) -> 5 (Yes)
    # (1,2) -> 3 (Yes)
    # (4,4) -> 8%7=1 (No)
    # (4,2) -> 6 (Yes)
    # (2,2) -> 4 (No)
    
    # Mapping: 1->1, 2->4, 3->2
    # Compatible if {b_u^2, b_v^2} is {1, 4}, {1, 2}, or {4, 2}
    # This means b_u^2 != b_v^2, which is equivalent to b_u != b_v.
    # Wait, let's recheck:
    # b=1: b^2=1
    # b=2: b^2=4
    # b=3: b^2=2
    # Pairs (b_u, b_v):
    # (1,2): 1+4=5 (Yes)
    # (1,3): 1+2=3 (Yes)
    # (2,3): 4+2=6 (Yes)
    # (1,1): 1+1=2 (No)
    # (2,2): 4+4=8%7=1 (No)
    # (3,3): 2+2=4 (No)
    
    # So the condition is simply b_u != b_v.
    # This is a graph coloring problem with 3 colors.
    
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for color in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if assignment[v] == color:
                    possible = False
                    break
            if possible:
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
