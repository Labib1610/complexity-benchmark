import sys
from itertools import product

def compute_determinant_squared(mode_u, mode_v):
    # Signature vectors
    v = {1: (1, 0, 0), 2: (0, 1, 0), 3: (0, 0, 1)}
    
    # Construct the 3x3 matrix
    matrix = [
        list(v[mode_u]),
        list(v[mode_v]),
        [1, 1, 1]
    ]
    
    # Compute determinant using Sarrus rule for 3x3 matrices
    det = (
        matrix[0][0] * matrix[1][1] * matrix[2][2] +
        matrix[0][1] * matrix[1][2] * matrix[2][0] +
        matrix[0][2] * matrix[1][0] * matrix[2][1] -
        matrix[0][2] * matrix[1][1] * matrix[2][0] -
        matrix[0][0] * matrix[1][2] * matrix[2][1] -
        matrix[0][1] * matrix[1][0] * matrix[2][2]
    )
    
    return det * det

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Precompute tension values for all mode pairs
    tension = {}
    for mode_u, mode_v in product([1, 2, 3], repeat=2):
        tension[(mode_u, mode_v)] = compute_determinant_squared(mode_u, mode_v)
    
    # Try all possible assignments (3^n possibilities)
    # But since n can be up to 40, we need a more efficient approach
    
    # Let's think differently. For each edge, T_uv = (det)^2
    # The determinant of the matrix with rows v_u, v_v, (1,1,1) can be:
    # Let's compute all possible determinants:
    # Modes: 1=(1,0,0), 2=(0,1,0), 3=(0,0,1)
    
    # All combinations of two modes:
    det_values = {}
    for i in [1,2,3]:
        for j in [1,2,3]:
            # Matrix rows: v_i, v_j, (1,1,1)
            # v_i = (a,b,c)
            # v_j = (d,e,f)
            # (1,1,1)
            # Determinant = a(e*1 - f*1) - b(d*1 - f*1) + c(d*1 - e*1)
            #             = a(e-f) - b(d-f) + c(d-e)
            vi = [0]*3
            vj = [0]*3
            vi[i-1] = 1
            vj[j-1] = 1
            
            det = vi[0]*(vj[1] - vj[2]) - vi[1]*(vj[0] - vj[2]) + vi[2]*(vj[0] - vj[1])
            det_values[(i,j)] = det * det
    
    # We want sum of tensions = m
    # So we want sum over all edges of T_e = m
    # Each T_e is in {0, 1}
    # Actually let's check what values are possible:
    # det_values[(1,1)] = (1*(0-1) - 0*(1-1) + 0*(1-0))^2 = (-1)^2 = 1
    # det_values[(1,2)] = (1*(1-1) - 0*(0-1) + 0*(0-1))^2 = 0
    # det_values[(1,3)] = (1*(0-1) - 0*(0-1) + 0*(0-0))^2 = (-1)^2 = 1
    # det_values[(2,1)] = (0*(0-1) - 1*(1-1) + 0*(1-0))^2 = 0
    # det_values[(2,2)] = (0*(1-1) - 1*(0-1) + 0*(0-0))^2 = 1
    # det_values[(2,3)] = (0*(0-1) - 1*(0-1) + 0*(0-0))^2 = 1
    # det_values[(3,1)] = (0*(0-1) - 0*(1-1) + 1*(1-0))^2 = 1
    # det_values[(3,2)] = (0*(1-1) - 0*(0-1) + 1*(0-0))^2 = 0
    # det_values[(3,3)] = (0*(1-1) - 0*(0-1) + 1*(0-1))^2 = (-1)^2 = 1
    
    # So T_uv can be either 0 or 1.
    # T_uv = 0 when modes are (1,2), (2,1), (3,2)
    # T_uv = 1 otherwise
    
    # This means we're looking for an assignment where exactly m edges have T_e = 1.
    # This is equivalent to finding a 3-coloring where certain edges are "satisfied".
    # An edge is satisfied (T_e=1) unless it connects colors from the set {(1,2), (2,1), (3,2)}.
    # Actually, unsatisfied edges connect:
    # 1-2 or 2-1
    # 3-2 or 2-3? No, (2,3) gives det=-1, so T=1. 
    # Wait, let me recheck:
    # (2,3): rows (0,1,0), (0,0,1), (1,1,1)
    # det = 0*(0*1-1*1) - 1*(0*1-1*1) + 0*(0*1-0*1) = 0 - 1*(-1) + 0 = 1. So T=1.
    # (3,2): rows (0,0,1), (0,1,0), (1,1,1)
    # det = 0*(1*1-0*1) - 0*(0*1-1*1) + 1*(0*1-1*1) = 0 - 0 + 1*(-1) = -1. So T=1.
    
    # So only (1,2) and (2,1) give T=0.
    
    # We want exactly m edges with T=1, which means all edges must have T=1.
    # This means no edge can connect nodes of modes 1 and 2.
    
    # So in the graph, there must be no edge between a node of mode 1 and a node of mode 2.
    # This is a 3-coloring problem with constraint that no 1-2 edge exists.
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # We can try all 3-colorings but that's too slow for n=40.
    # However, the constraint is simpler: partition vertices into three sets V1, V2, V3
    # such that there is no edge between V1 and V2.
    
    # This means G[V1 U V2] is a bipartite graph complement.
    # In other words, we want to 3-color the vertices such that color 1 and color 2 are not adjacent.
    
    # One approach: try all possible ways to assign vertices to sets where V1-V2 has no edges.
    # But still exponential.
    
    # Better idea: since we just need one solution, we can use backtracking with pruning.
    
    # Or even better: note that we want ALL edges to contribute 1 to the sum.
    # So NO edge should connect mode 1 to mode 2.
    
    # Let's model this as a graph 3-coloring where color 1 and color 2 cannot be adjacent.
    # This is equivalent to: find a partition of vertices into V1, V2, V3 such that
    # there are no edges between V1 and V2.
    
    # A simple approach: try greedy assignment.
    # But let's think of it as: we can put vertices in three groups,
    # but group 1 and group 2 cannot have any edges between them.
    
    # This is equivalent to checking if the graph is tripartite with the constraint.
    
    # Let's do a search with backtracking.
    color = [-1] * n  # -1 uncolored, 0=mode1, 1=mode2, 2=mode3
    
    def is_safe(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                # Special case: if we're trying to color node with 0 (mode 1)
                # and neighbor has 1 (mode 2), or vice versa, it's invalid
                if (c == 0 and color[neighbor] == 1) or (c == 1 and color[neighbor] == 0):
                    return False
        return True
    
    def solve(node):
        if node == n:
            return True
        
        for c in [0, 1, 2]:
            if is_safe(node, c):
                color[node] = c
                if solve(node + 1):
                    return True
                color[node] = -1
        return False
    
    if solve(0):
        print("YES")
        result = [str(c + 1) for c in color]
        print(" ".join(result))
    else:
        print("NO")

if __name__ == "__main__":
    main()
