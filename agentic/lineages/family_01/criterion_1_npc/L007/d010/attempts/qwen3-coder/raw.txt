```python
import sys
from itertools import product

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u-1, v-1))  # Convert to 0-indexed
    return n, m, edges

def compute_trace_product(state_u, state_v):
    # Define the transformation matrices
    A = [
        None,  # 0 index not used
        [[0, 1, 0], [0, 0, 1], [1, 0, 0]],  # State 1
        [[0, 0, 1], [1, 0, 0], [0, 1, 0]],  # State 2
        [[1, 0, 0], [0, 1, 0], [0, 0, 1]]   # State 3
    ]
    
    # Compute A[state_v]^T (transpose)
    At = [[A[state_v][j][i] for j in range(3)] for i in range(3)]
    
    # Compute A[state_u] * A[state_v]^T
    product_matrix = [[0]*3 for _ in range(3)]
    for i in range(3):
        for j in range(3):
            for k in range(3):
                product_matrix[i][j] += A[state_u][i][k] * At[k][j]
    
    # Return trace (sum of diagonal elements)
    return product_matrix[0][0] + product_matrix[1][1] + product_matrix[2][2]

def precompute_scores():
    # Precompute all possible trace values for pairs of states
    scores = {}
    for s1 in range(1, 4):
        for s2 in range(1, 4):
            scores[(s1, s2)] = compute_trace_product(s1, s2)
    return scores

def is_valid_assignment(states, edges, scores):
    total_score = 0
    for u, v in edges:
        total_score += scores[(states[u], states[v])]
    return total_score == 0

def solve():
    n, m, edges = read_input()
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    scores = precompute_scores()
    
    # Try all possible assignments in lexicographical order
    for assignment in product(range(1, 4), repeat=n):
        if is_valid_assignment(assignment, edges, scores):
            print("YES")
            print(" ".join(map(str, assignment)))
            return
    
    print("NO")

if __name__ == "__main__":
    solve()
```
