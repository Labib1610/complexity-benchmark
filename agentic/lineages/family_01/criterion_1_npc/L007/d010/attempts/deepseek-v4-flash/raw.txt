We can solve this by noticing that the determinant squared for an edge is 1 if the two endpoints have different modes, and 0 if they have the same mode.  

So the problem reduces to: can we assign each node one of 3 colors such that the number of edges with different colors equals the total number of edges?  
That means **every edge must connect nodes of different colors**.  

Thus, we need to check if the graph is 3-colorable. Since \( n \le 40 \), we can use backtracking with pruning.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    edges = []
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n

    def backtrack(node):
        if node == n:
            return True
        # Try colors 0,1,2
        for c in range(3):
            ok = True
            for nb in adj[node]:
                if color[nb] == c:
                    ok = False
                    break
            if ok:
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = -1
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(str(c + 1) for c in color))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
