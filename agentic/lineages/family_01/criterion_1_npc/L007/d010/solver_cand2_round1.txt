import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Precompute stress for all pairs (c_u, c_v)
    # c_u, c_v in {1, 2, 3}
    stress_map = {}
    for cu in range(1, 4):
        for cv in range(1, 4):
            D = (cu - cv)**2
            F = 3 * (cu + cv)**2 - 24 * (cu + cv) + 49
            W = (D * F) % 17
            
            # Simulation: S = (S^2 + W) % 17, 1,000,000 cycles
            # Since 17 is small, the sequence S_k will eventually cycle.
            # Period is at most 17.
            S = 1
            history = {}
            path = []
            for i in range(1000000):
                if S in history:
                    start_idx = history[S]
                    cycle = path[start_idx:]
                    S = cycle[(1000000 - start_idx) % len(cycle)]
                    break
                history[S] = i
                path.append(S)
                S = (S * S + W) % 17
            else:
                # If loop finishes without cycle (not possible for 1M > 17)
                pass
            stress_map[(cu, cv)] = S

    # Backtracking to find lexicographically smallest assignment
    assignment = [0] * n
    
    def check():
        total_stress = 0
        for u, v in edges:
            total_stress += stress_map[(assignment[u], assignment[v])]
        return total_stress == 12 * m

    def backtrack(node_idx):
        if node_idx == n:
            return check()
        
        for c in [1, 2, 3]:
            assignment[node_idx] = c
            if backtrack(node_idx + 1):
                return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
