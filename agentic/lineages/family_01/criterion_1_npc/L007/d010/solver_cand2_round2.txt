import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The penalty function I(s_u, s_v) = 0 simplifies to:
    # (s_u - s_v)^2 * ((s_u - s_v)^2 - 5) = 0
    # This means (s_u - s_v)^2 = 0 or (s_u - s_v)^2 = 5.
    # Since s_u, s_v in {1, 2, 3}, (s_u - s_v)^2 can be 0, 1, or 4.
    # The only way to get 0 is (s_u - s_v)^2 = 0, i.e., s_u = s_v.
    # Wait, let's re-check:
    # If s_u=1, s_v=2: (1-2)^2 * (1-5) = 1 * (-4) = -4 != 0.
    # If s_u=1, s_v=3: (1-3)^2 * (4-5) = 4 * (-1) = -4 != 0.
    # If s_u=2, s_v=3: (2-3)^2 * (1-5) = 1 * (-4) = -4 != 0.
    # Actually, the condition I(s_u, s_v) = 0 is satisfied if and only if s_u = s_v.
    # Let's re-verify the example:
    # Link (1, 2) with states 1 and 2: I = 0.
    # Wait, the example says 1 and 2 gives 0. Let's re-calculate:
    # I(1, 2) = 1^4 + 2^4 - 4(1^3*2) - 4(1*2^3) + 6(1^2*2^2) - 5(1^2) - 5(2^2) + 10(1*2) + 4
    # = 1 + 16 - 8 - 32 + 24 - 5 - 20 + 20 + 4 = 0.
    # So s_u != s_v is allowed.
    # The condition I(s_u, s_v) = 0 is satisfied if |s_u - s_v| is 1 or 2.
    # Since s_u, s_v in {1, 2, 3}, |s_u - s_v| can be 0, 1, 2.
    # |s_u - s_v| = 0 is NOT allowed. |s_u - s_v| in {1, 2} is allowed.
    # This means s_u != s_v.
    
    # This is a graph coloring problem with 3 colors where adjacent nodes must have different colors.
    # This is equivalent to finding a 3-coloring of the graph.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
