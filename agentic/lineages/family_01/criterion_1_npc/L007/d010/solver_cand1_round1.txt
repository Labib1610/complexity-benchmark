import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2

    # D(u, v) = (5(s_u - s_v)^2 - (s_u - s_v)^4) / 4
    # If |s_u - s_v| = 1: D = (5(1) - 1) / 4 = 1
    # If |s_u - s_v| = 2: D = (5(4) - 16) / 4 = 1
    # If |s_u - s_v| = 0: D = 0
    # So D = 1 if s_u != s_v, and D = 0 if s_u == s_v.
    # We need sum of D = m, which means for every strut (u, v), s_u != s_v.
    # This is equivalent to checking if the graph is 3-colorable.
    
    # Since n <= 40, we can use backtracking with pruning.
    colors = [-1] * n
    
    def is_valid(u, c):
        for v1, v2 in adj:
            if v1 == u and colors[v2] != -1 and colors[v2] == c:
                return False
            if v2 == u and colors[v1] != -1 and colors[v1] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = -1
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
