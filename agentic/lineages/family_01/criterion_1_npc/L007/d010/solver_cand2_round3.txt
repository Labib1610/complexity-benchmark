import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The determinant of M_{u,v} is:
    # If s_u == s_v: det = 0, T = 0
    # If s_u != s_v:
    #   Let s_u = i, s_v = j. The rows are e_i, e_j, (1,1,1).
    #   The determinant is the signed volume of the parallelepiped.
    #   For distinct i, j in {1, 2, 3}, the determinant is +/- 1.
    #   Specifically, if (i, j) is (1, 2), det = 1. (1, 3), det = -1. (2, 3), det = 1.
    #   In all cases where s_u != s_v, T_{u,v} = 1.
    #   If s_u == s_v, T_{u,v} = 0.
    # We want sum(T_{u,v}) = m.
    # This means for every edge (u, v), we must have s_u != s_v.
    # This is equivalent to saying the graph must be 3-colorable.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    colors = [-1] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = -1
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
