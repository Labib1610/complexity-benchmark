import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Calculate C(s_u, s_v) for all pairs (s_u, s_v) in {1, 2, 3} x {1, 2, 3}
    def get_c(s1, s2):
        return (s1**5 + s2**5 - 3*s1**4*s2 - 3*s1*s2**4 + 2*s1**3*s2**2 + 2*s1**2*s2**3 
                - 5*s1**3 - 5*s2**3 + 5*s1**2*s2 + 5*s1*s2**2 + 4*s1 + 4*s2)

    # Precompute table
    table = [[0]*4 for _ in range(4)]
    for i in range(1, 4):
        for j in range(1, 4):
            table[i][j] = get_c(i, j)

    # Meet-in-the-middle: n <= 40 is too large for 3^n.
    # However, notice C(1, 2) = 0, C(1, 3) = 0, C(2, 3) = 0.
    # C(1, 1) = 1+1-3-3+2+2-5-5+5+5+4+4 = 2
    # C(2, 2) = 32+32-96-96+64+64-40-40+40+40+8+8 = -8
    # C(3, 3) = 243+243-729-729+486+486-135-135+225+225+12+12 = 18
    # The energy is 0 if we only use pairs (1,2), (1,3), (2,3).
    # If we only use states {1, 2, 3} such that no two connected nodes have the same state,
    # the energy is always 0. This is equivalent to 3-coloring the graph.
    
    # Since n is small (40), we can use backtracking with pruning to find a 3-coloring.
    assignment = [0] * n
    
    def is_valid(u, color):
        for v1, v2 in edges:
            if v1 == u and assignment[v2] == color: return False
            if v2 == u and assignment[v1] == color: return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
