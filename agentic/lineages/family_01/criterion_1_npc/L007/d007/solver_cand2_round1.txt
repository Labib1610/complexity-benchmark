import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # v_u dot v_v = 2*2 + (-1)*(-1) + (-1)*(-1) = 6 if c_u == c_v
    # v_u dot v_v = 2*(-1) + (-1)*2 + (-1)*(-1) = -3 if c_u != c_v
    # Total energy = sum_{edges} (v_u dot v_v)
    # We want total energy = -3m.
    # Since each edge contributes either 6 or -3, let x be the number of edges
    # where c_u == c_v, and y be the number of edges where c_u != c_v.
    # x + y = m
    # 6x - 3y = -3m
    # 6x - 3(m - x) = -3m
    # 6x - 3m + 3x = -3m
    # 9x = 0 => x = 0.
    # This means no two connected modules can have the same access level.
    # The problem is equivalent to finding a 3-coloring of the graph.
    # We need the lexicographically smallest 3-coloring.

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
