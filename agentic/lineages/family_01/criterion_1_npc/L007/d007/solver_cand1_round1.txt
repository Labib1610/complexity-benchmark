```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition trace(X^T M X) = -6m simplifies to:
    # sum_{(u,v) in E} (V_u dot V_v) = -3m
    # V_j values: v1=(1,1,-2), v2=(1,-2,1), v3=(-2,1,1)
    # Dot products:
    # v1.v1 = 1+1+4 = 6
    # v2.v2 = 1+4+1 = 6
    # v3.v3 = 4+1+1 = 6
    # v1.v2 = 1-2-2 = -3
    # v1.v3 = -2+1-2 = -3
    # v2.v3 = -2-2+1 = -3
    # Let c_i be the level of module i.
    # If c_u == c_v, dot product is 6.
    # If c_u != c_v, dot product is -3.
    # Let E be the set of edges.
    # sum_{(u,v) in E} (6 if c_u == c_v else -3) = -3m
    # Let k be the number of edges where c_u == c_v.
    # 6k - 3(m - k) = -3m
    # 6k - 3m + 3k = -3m
    # 9k = 0 => k = 0.
    # This means no two connected modules can have the same access level.
    # This is exactly the definition of a 3-coloring problem.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
```
