import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Phase projections for s = 1, 2, 3:
    # s=1: x=1, y=-1, z=-1
    # s=2: x=-1, y=1, z=-1
    # s=3: x=-1, y=-1, z=1
    # Dot products:
    # P1.P1 = 1+1+1 = 3
    # P2.P2 = 1+1+1 = 3
    # P3.P3 = 1+1+1 = 3
    # P1.P2 = -1-1+1 = -1
    # P1.P3 = -1+1-1 = -1
    # P2.P3 = 1-1-1 = -1
    
    # Let n1, n2, n3 be the number of modules in states 1, 2, 3.
    # Total energy W = sum_{edges} (P_u . P_v)
    # Let E12 be the number of edges between state 1 and 2, etc.
    # W = E12*(-1) + E13*(-1) + E23*(-1) + E11*(3) + E22*(3) + E33*(3)
    # We want W = -m.
    # Note that E12 + E13 + E23 + E11 + E22 + E33 = m.
    # So -m = - (E12 + E13 + E23 + E11 + E22 + E33)
    # This implies:
    # -E12 - E13 - E23 + 3(E11 + E22 + E33) = -E12 - E13 - E23 - E11 - E22 - E33
    # 4(E11 + E22 + E33) = 0
    # Since E_ii >= 0, this requires E11 = E22 = E33 = 0.
    # This means no two modules connected by an edge can have the same state.
    # This is equivalent to saying the graph must be 3-colorable.
    
    # Since n <= 40, we can use backtracking with pruning to find a 3-coloring.
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
