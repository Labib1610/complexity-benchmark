```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [set() for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
    
    color = [0] * n
    
    def feasible(i, c):
        for j in adj[i]:
            if color[j] == c:
                return False
        return True
    
    def backtrack(pos):
        if pos == n:
            return True
        for c in (1, 2, 3):
            if feasible(pos, c):
                color[pos] = c
                if backtrack(pos + 1):
                    return True
                color[pos] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(' '.join(map(str, color)))
    else:
        print("NO")

solve()
```
