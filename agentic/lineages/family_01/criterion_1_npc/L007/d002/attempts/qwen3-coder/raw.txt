def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n, m = int(data[0]), int(data[1])
    
    # If no dependencies, any assignment works
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Precompute valid pairs: for shifts a,b in {1,2,3}, check if (a^2+b^2) % (a+b) > 0
    valid_pairs = set()
    shifts = [1, 2, 3]
    for a in shifts:
        for b in shifts:
            if (a*a + b*b) % (a + b) > 0:
                valid_pairs.add((a, b))
                valid_pairs.add((b, a))  # Since the graph is undirected
    
    # Try all 3^n assignments (but since n<=40, we need to be smarter)
    # However, with n<=40, even 3^40 is too large. But looking at constraints,
    # let's try a backtracking approach which might prune effectively.
    
    # Actually, given n<=40 is too large for brute force, but m<=n(n-1)/2 and small examples,
    # maybe intended solution is backtracking with pruning.
    
    # Let's implement a backtracking solution
    assignment = [-1] * n  # -1 means unassigned
    
    def is_valid(u, shift_u, v, shift_v):
        return (shift_u, shift_v) in valid_pairs
    
    # We'll use a greedy backtracking with constraint checking
    def backtrack(node):
        if node == n:
            return True
        
        # Try assigning shifts 1, 2, 3 to current node
        for shift in [1, 2, 3]:
            # Check if this assignment violates any edge constraints
            valid = True
            neighbors = []
            # Collect all edges involving 'node'
            for u, v in edges:
                if u == node:
                    neighbors.append((u, v))
                elif v == node:
                    neighbors.append((u, v))
            
            # Check if assignment is valid with already assigned neighbors
            for u, v in neighbors:
                other_node = v if u == node else u
                if assignment[other_node] != -1:  # Already assigned
                    other_shift = assignment[other_node]
                    if u == node:
                        if not is_valid(u, shift, v, other_shift):
                            valid = False
                            break
                    else:
                        if not is_valid(u, other_shift, v, shift):
                            valid = False
                            break
            
            if valid:
                assignment[node] = shift
                if backtrack(node + 1):
                    return True
                assignment[node] = -1  # Backtrack
                
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, assignment)))
    else:
        print("NO")

if __name__ == "__main__":
    main()
