def solve():
    import sys
    from collections import defaultdict
    
    n, m = map(int, sys.stdin.readline().split())
    
    # Create adjacency list for the graph
    graph = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1  # Convert to 0-based indexing
        v -= 1
        graph[u].append(v)
        graph[v].append(u)
        edges.append((u, v))
    
    # Try all possible colorings using backtracking
    # Colors are 1, 2, 3
    coloring = [0] * n  # 0 means uncolored
    
    def is_valid_color(node, color):
        # Check if assigning 'color' to 'node' violates any constraints
        for neighbor in graph[node]:
            if coloring[neighbor] == color:
                return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
            
        for color in [1, 2, 3]:
            if is_valid_color(node, color):
                coloring[node] = color
                if backtrack(node + 1):
                    return True
                coloring[node] = 0  # Backtrack
        
        return False
    
    # Optimization: sort nodes by degree descending to detect infeasibility early
    node_degrees = [(i, len(graph[i])) for i in range(n)]
    node_degrees.sort(key=lambda x: -x[1])
    order = [node for node, _ in node_degrees]
    
    # Re-index the problem according to this order
    old_to_new = {old_idx: new_idx for new_idx, old_idx in enumerate(order)}
    new_to_old = {new_idx: old_idx for old_idx, new_idx in old_to_new.items()}
    
    # Create a new graph with reordered nodes
    new_graph = defaultdict(list)
    for u, v in edges:
        new_u = old_to_new[u]
        new_v = old_to_new[v]
        new_graph[new_u].append(new_v)
        new_graph[new_v].append(new_u)
    
    # Now we solve on the reordered graph
    new_coloring = [0] * n
    
    def is_valid_in_new_graph(node, color):
        for neighbor in new_graph[node]:
            if new_coloring[neighbor] == color:
                return False
        return True
    
    def backtrack_new(node):
        if node == n:
            return True
            
        for color in [1, 2, 3]:
            if is_valid_in_new_graph(node, color):
                new_coloring[node] = color
                if backtrack_new(node + 1):
                    return True
                new_coloring[node] = 0
                
        return False
    
    if backtrack_new(0):
        # Reorder the solution back to original indexing
        final_coloring = [0] * n
        for new_idx in range(n):
            old_idx = new_to_old[new_idx]
            final_coloring[old_idx] = new_coloring[new_idx]
        
        print("YES")
        print(' '.join(map(str, final_coloring)))
    else:
        print("NO")

solve()
