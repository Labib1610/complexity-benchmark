import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To find the lexicographically smallest assignment, we use backtracking.
    # Since n is small (3^3 + 13 = 40), backtracking with pruning is efficient.
    # To avoid worst-case scenarios, we use a simple heuristic: 
    # process nodes in order 1..n and try tiers 1, 2, 3.
    
    def is_valid(u, tier):
        for v in adj[u]:
            if assignment[v] == tier:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for tier in range(1, 4):
            if is_valid(u, tier):
                assignment[u] = tier
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
