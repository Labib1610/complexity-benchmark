import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Condition: (s_u^2 + s_v^2) % (s_u + s_v) > 0
    # Possible shifts: 1, 2, 3
    # Pairs (s_u, s_v):
    # (1, 1): (1+1) % 2 = 0 (Invalid)
    # (1, 2): (1+4) % 3 = 2 > 0 (Valid)
    # (1, 3): (1+9) % 4 = 2 > 0 (Valid)
    # (2, 1): (4+1) % 3 = 2 > 0 (Valid)
    # (2, 2): (4+4) % 4 = 0 (Invalid)
    # (2, 3): (4+9) % 5 = 3 > 0 (Valid)
    # (3, 1): (9+1) % 4 = 2 > 0 (Valid)
    # (3, 2): (9+4) % 5 = 3 > 0 (Valid)
    # (3, 3): (9+9) % 6 = 0 (Invalid)
    
    # Summary: s_u != s_v is required.
    # This is equivalent to finding a 3-coloring of the graph.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
