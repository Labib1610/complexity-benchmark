import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Delta E = (c_u - c_v)^4 - 5(c_u - c_v)^2 + 4
    # Let d = |c_u - c_v|. Possible values for d are 0, 1, 2.
    # If d = 0: 0^4 - 5(0)^2 + 4 = 4
    # If d = 1: 1^4 - 5(1)^2 + 4 = 1 - 5 + 4 = 0
    # If d = 2: 2^4 - 5(2)^2 + 4 = 16 - 20 + 4 = 0
    # Total energy is 0 iff for all edges (u, v), |c_u - c_v| != 0.
    # This means c_u != c_v for all edges (u, v).
    # This is equivalent to finding a 3-coloring of the graph.
    
    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
