import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Constraint: s_u + 2*s_v != 0 (mod 3)
    # s_i in {1, 2, 3}. Modulo 3, these are {1, 2, 0}.
    # Let x_i = s_i mod 3. x_i in {0, 1, 2}.
    # Constraint: x_u + 2*x_v != 0 (mod 3)
    # If x_v = 0, then x_u + 0 != 0 => x_u != 0.
    # If x_v = 1, then x_u + 2 != 0 => x_u != 1.
    # If x_v = 2, then x_u + 4 != 0 => x_u + 1 != 0 => x_u != 2.
    
    assignment = [0] * n
    
    def is_valid(u, val):
        # val is 0, 1, or 2
        for v in adj[u]:
            if assignment[v] != -1:
                # Check constraint for (u, v)
                if (val + 2 * assignment[v]) % 3 == 0:
                    return False
                # Check constraint for (v, u)
                if (assignment[v] + 2 * val) % 3 == 0:
                    return False
        return True

    assignment = [-1] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        for val in range(3):
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = -1
        return False

    if backtrack(0):
        print("YES")
        res = []
        for x in assignment:
            # Map 0, 1, 2 back to 3, 1, 2
            if x == 0: res.append(3)
            elif x == 1: res.append(1)
            else: res.append(2)
        print(*(res))
    else:
        print("NO")

solve()
