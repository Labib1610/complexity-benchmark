import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2

    # S(u, v) = c_u^2 + c_v^2 + c_u * c_v
    # We need R = product(S(u, v)) not divisible by 3.
    # This means for every protocol (u, v), S(u, v) % 3 != 0.
    # Possible values for c_i are {1, 2, 3}.
    # Modulo 3: 1 -> 1, 2 -> 2, 3 -> 0.
    # S(u, v) % 3:
    # If c_u = 3 (0 mod 3): S = 0 + c_v^2 + 0 = c_v^2.
    #   If c_v = 1, S = 1. If c_v = 2, S = 4 = 1. If c_v = 3, S = 0.
    #   So if c_u = 3, c_v cannot be 3.
    # If c_u = 1: S = 1 + c_v^2 + c_v.
    #   If c_v = 1, S = 3 = 0. (Invalid)
    #   If c_v = 2, S = 1 + 4 + 2 = 7 = 1. (Valid)
    #   If c_v = 3, S = 1 + 0 + 0 = 1. (Valid)
    # If c_u = 2: S = 4 + c_v^2 + 2*c_v = 1 + c_v^2 + 2*c_v.
    #   If c_v = 1, S = 1 + 1 + 2 = 4 = 1. (Valid)
    #   If c_v = 2, S = 1 + 4 + 4 = 9 = 0. (Invalid)
    #   If c_v = 3, S = 1 + 0 + 0 = 1. (Valid)
    
    # Summary of constraints:
    # 1. If c_u = 1, c_v cannot be 1.
    # 2. If c_u = 2, c_v cannot be 2.
    # 3. If c_u = 3, c_v cannot be 3.
    # This is equivalent to saying c_u != c_v for all protocols.
    # This is exactly the 3-coloring problem.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v1, v2 in adj:
            if v1 == u:
                if assignment[v2] == color: return False
            elif v2 == u:
                if assignment[v1] == color: return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
