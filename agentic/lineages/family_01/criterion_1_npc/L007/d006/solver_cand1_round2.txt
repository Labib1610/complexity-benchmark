import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Delta E = (c_u - c_v)^2 * ((c_u - c_v)^2 - 5) + 4
    # Let d = |c_u - c_v|. d can be 0, 1, or 2.
    # If d=0: 0*(0-5) + 4 = 4
    # If d=1: 1*(1-5) + 4 = 0
    # If d=2: 4*(4-5) + 4 = 0
    # So Delta E = 0 if and only if |c_u - c_v| != 0.
    # This means c_u != c_v for all edges.
    # The problem is equivalent to finding a 3-coloring of the graph.
    # We need the lexicographically smallest assignment.

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    assignment = [0] * n

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
