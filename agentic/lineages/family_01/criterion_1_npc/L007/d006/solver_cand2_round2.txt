import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # I(u, v) = (R1 - 2*R2)^2 - 5*R1 + 10*R2 + 4
    # R1 = cu^2 + cv^2, R2 = cu*cv
    # R1 - 2*R2 = cu^2 + cv^2 - 2*cu*cv = (cu - cv)^2
    # I(u, v) = (cu - cv)^4 - 5(cu^2 + cv^2) + 10(cu*cv) + 4
    # Let's precompute I(cu, cv) for cu, cv in {1, 2, 3}
    def get_interference(cu, cv):
        r1 = cu**2 + cv**2
        r2 = cu * cv
        return (r1 - 2*r2)**2 - 5*r1 + 10*r2 + 4

    # Interference table
    # I(1,1)=0, I(2,2)=0, I(3,3)=0
    # I(1,2) = (1-2)^4 - 5(1+4) + 10(2) + 4 = 1 - 25 + 20 + 4 = 0
    # I(1,3) = (1-3)^4 - 5(1+9) + 10(3) + 4 = 16 - 50 + 30 + 4 = 0
    # I(2,3) = (2-3)^4 - 5(4+9) + 10(6) + 4 = 1 - 65 + 60 + 4 = 0
    # Wait, all I(cu, cv) are 0 for any cu, cv in {1, 2, 3}!
    # Let's recheck:
    # I(1,2): (1-4)^2 - 5(5) + 10(2) + 4 = 9 - 25 + 20 + 4 = 8? No.
    # Formula: (R1 - 2*R2)^2 - 5*R1 + 10*R2 + 4
    # R1 - 2*R2 = (cu - cv)^2
    # I = (cu - cv)^4 - 5(cu^2 + cv^2) + 10(cu*cv) + 4
    # I(1,1) = 0 - 5(2) + 10(1) + 4 = 4
    # I(2,2) = 0 - 5(8) + 10(4) + 4 = 4
    # I(3,3) = 0 - 5(18) + 10(9) + 4 = 4
    # I(1,2) = 1 - 5(5) + 10(2) + 4 = 1 - 25 + 20 + 4 = 0
    # I(1,3) = 16 - 5(10) + 10(3) + 4 = 16 - 50 + 30 + 4 = 0
    # I(2,3) = 1 - 5(13) + 10(6) + 4 = 1 - 65 + 60 + 4 = 0
    
    # So I(u, v) == 0 iff cu != cv.
    # The problem is equivalent to: Can we color the graph with 3 colors such that
    # every edge connects nodes of different colors?
    # This is exactly the 3-coloring problem.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            valid = True
            for v in adj[u]:
                if assignment[v] == color:
                    valid = False
                    break
            if valid:
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
