```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n  # 0 = unassigned

    def backtrack(idx: int) -> bool:
        if idx == n:
            return True
        for col in (1, 2, 3):          # try colors in increasing order
            ok = True
            for nb in adj[idx]:
                if nb < idx and colors[nb] == col:
                    ok = False
                    break
            if not ok:
                continue
            colors[idx] = col
            if backtrack(idx + 1):
                return True
            colors[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(' '.join(map(str, colors)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
