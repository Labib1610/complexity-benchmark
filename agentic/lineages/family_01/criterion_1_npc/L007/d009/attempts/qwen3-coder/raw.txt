import math
from itertools import product

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Define the signature vectors
    w = {
        1: (-1, math.sqrt(3)),
        2: (-1, -math.sqrt(3)),
        3: (2, 0)
    }
    
    # Precompute the stress for each pair of channels
    stress = {}
    for c1 in [1, 2, 3]:
        for c2 in [1, 2, 3]:
            wx = w[c1][0] + w[c2][0]
            wy = w[c1][1] + w[c2][1]
            stress[(c1, c2)] = wx*wx + wy*wy
            stress[(c2, c1)] = stress[(c1, c2)]
    
    target_total = 4 * m
    
    # Try all 3^n combinations in lexicographical order
    for assignment in product([1, 2, 3], repeat=n):
        total_stress = 0
        for u, v in edges:
            total_stress += stress[(assignment[u], assignment[v])]
        if abs(total_stress - target_total) < 1e-9:
            print("YES")
            print(" ".join(map(str, assignment)))
            return
    
    print("NO")

solve()
