import sys
from itertools import product

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Define phase vectors
    p = {
        1: (1.0, 0.0),
        2: (-0.5, 3**0.5 / 2),
        3: (-0.5, -(3**0.5) / 2)
    }
    
    # Precompute D values for all pairs of bands
    D = {}
    for i in range(1, 4):
        for j in range(1, 4):
            dot = p[i][0] * p[j][0] + p[i][1] * p[j][1]
            D[(i,j)] = (2/3) * (1 - dot)
    
    # Try all 3^n assignments
    for assignment in product([1,2,3], repeat=n):
        total = 0.0
        for u, v in edges:
            total += D[(assignment[u], assignment[v])]
        if abs(total - m) < 1e-9:
            print("YES")
            print(" ".join(map(str, assignment)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
