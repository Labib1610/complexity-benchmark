import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition D_{u,v} = 1 means p_{s_u} . p_{s_v} = -0.5.
    # The dot products are:
    # p1.p1 = 1, p2.p2 = 1, p3.p3 = 1
    # p1.p2 = -0.5, p1.p3 = -0.5, p2.p3 = -0.5
    # So D_{u,v} = 1 if s_u != s_v, and D_{u,v} = 0 if s_u == s_v.
    # We need sum(D_{u,v}) = m, which means for all struts (u, v), s_u != s_v.
    # This is equivalent to checking if the graph is 3-colorable.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
