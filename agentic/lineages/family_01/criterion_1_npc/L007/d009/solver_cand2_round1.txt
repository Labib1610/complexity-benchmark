import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The dot products are:
    # V1.V1 = 2, V2.V2 = 2, V3.V3 = 2
    # V1.V2 = 1, V1.V3 = 1, V2.V3 = 1
    # We want sum(V_su . V_sv) = m.
    # If we assign all gyroscopes to the same band, V_si . V_sj = 2.
    # Then W = 2 * m. This is only m if m=0.
    # If we use two different bands, V_si . V_sj = 1.
    # If we use all three bands, V_si . V_sj = 1.
    # So if we can partition the graph into two sets (or use 3 bands such that
    # all edges connect different bands), the dot product is always 1.
    # This is equivalent to checking if the graph is 3-colorable.
    # Actually, the condition is simpler: if we assign each node a color 
    # from {1, 2, 3}, the dot product is 1 if color(u) != color(v), 
    # and 2 if color(u) == color(v).
    # We want sum_{edges} (1 if c(u)!=c(v) else 2) = m.
    # Let k be the number of edges with c(u) == c(v).
    # Then m + k = m => k = 0.
    # So we need a 3-coloring of the graph.
    
    colors = [0] * (n + 1)
    
    def is_3_colorable(u, c):
        colors[u] = c
        for v in adj[u]:
            if colors[v] == c:
                return False
            if colors[v] == 0:
                # Try colors 1, 2, 3
                for next_c in [1, 2, 3]:
                    if next_c != c:
                        # Check if this color is valid for neighbors
                        possible = True
                        for neighbor in adj[v]:
                            if colors[neighbor] == next_c:
                                possible = False
                                break
                        if possible:
                            if is_3_colorable(v, next_c):
                                return True
                return False
        return True

    # Standard backtracking for 3-coloring
    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            valid = True
            for v in adj[u]:
                if colors[v] == c:
                    valid = False
                    break
            if valid:
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
