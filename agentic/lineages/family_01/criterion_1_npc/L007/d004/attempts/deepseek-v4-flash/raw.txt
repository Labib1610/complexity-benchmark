import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[False] * n for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u][v] = adj[v][u] = True

    # Try all 3^n assignments? n <= 40, too large.
    # But we can use backtracking with pruning.
    # Since 3^40 is huge, we need a smarter approach.
    # However, n <= 40, m <= 780. We can use a simple backtracking
    # with constraint propagation, which should be fast enough for n=40
    # because the graph is small and 3-coloring is easy for small n.

    color = [0] * n  # 0 = uncolored, 1,2,3 = colors

    # Precompute adjacency list
    graph = [[] for _ in range(n)]
    for i in range(n):
        for j in range(n):
            if adj[i][j]:
                graph[i].append(j)

    # Backtracking with forward checking
    def backtrack(idx):
        if idx == n:
            return True
        # Try colors 1,2,3
        for c in (1, 2, 3):
            ok = True
            for nb in graph[idx]:
                if color[nb] == c:
                    ok = False
                    break
            if ok:
                color[idx] = c
                if backtrack(idx + 1):
                    return True
                color[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(str(c) for c in color))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
