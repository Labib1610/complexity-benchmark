 We need to assign each node a value from {1, 2, 3} such that no two connected nodes share the same value, and we want the lexicographically smallest assignment.

This is essentially a graph coloring problem with 3 colors. Since n ≤ 40, we can use backtracking with pruning, but we must be careful to avoid exponential blow-up in worst cases.  

A good approach is to use backtracking with forward checking and choose the next variable with the smallest domain (MRV heuristic). This will handle the adversarial cases efficiently.

Here’s the implementation:

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    colors = [0] * n  # 0 = uncolored, 1,2,3 = colors
    
    # Precompute possible colors for each node
    possible = [set([1,2,3]) for _ in range(n)]
    
    # MRV backtracking
    def backtrack(index):
        if index == n:
            return True
        # Choose uncolored node with smallest domain
        best = -1
        best_len = 4
        for i in range(n):
            if colors[i] == 0:
                l = len(possible[i])
                if l < best_len:
                    best_len = l
                    best = i
                    if l == 1:
                        break
        if best == -1:
            return True
        
        # Try colors in ascending order for lexicographic minimality
        for c in sorted(possible[best]):
            # Check constraints
            ok = True
            for nb in adj[best]:
                if colors[nb] == c:
                    ok = False
                    break
            if not ok:
                continue
            
            # Save state
            saved_possible = [set(p) for p in possible]
            colors[best] = c
            
            # Remove c from neighbors' possible sets
            conflict = False
            for nb in adj[best]:
                if colors[nb] == 0:
                    if c in possible[nb]:
                        possible[nb].remove(c)
                        if not possible[nb]:
                            conflict = True
                            break
            if not conflict:
                if backtrack(index + 1):
                    return True
            
            # Restore state
            colors[best] = 0
            possible[:] = saved_possible
        
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(str(colors[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
