import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    # Heuristic: Sort nodes by degree descending to fail fast
    # But we need lexicographical order, so we must assign 1..n in order.
    # To optimize, we use MRV (Minimum Remaining Values) for variable selection
    # but we must respect the lexicographical requirement.
    # Actually, for n=40, simple backtracking with forward checking is sufficient
    # if we use bitmasks for available colors.
    
    def get_valid_colors(u, current_assignment):
        used = 0
        for v in adj[u]:
            if current_assignment[v] != 0:
                used |= (1 << (current_assignment[v] - 1))
        return [c for c in range(1, 4) if not (used & (1 << (c - 1)))]

    def backtrack(u):
        if u == n:
            return True
        
        # To ensure lexicographical smallest, we try colors 1, 2, 3 in order
        # for each module 0, 1, ..., n-1.
        for color in range(1, 4):
            possible = True
            for v in adj[u]:
                if assignment[v] == color:
                    possible = False
                    break
            
            if possible:
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # The problem asks for lexicographically smallest sequence.
    # The standard backtracking trying 1, 2, 3 for each index 0..n-1
    # naturally finds the lexicographically smallest solution.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
