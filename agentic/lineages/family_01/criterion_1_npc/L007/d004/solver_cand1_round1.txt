import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Condition: s_u^2 + s_v^2 + s_u*s_v != 0 (mod 3)
    # s in {1, 2, 3} => s mod 3 in {1, 2, 0}
    # Let x = s_u mod 3, y = s_v mod 3
    # If x=0 or y=0:
    #   If x=0, y=1: 0+1+0 = 1 != 0 (ok)
    #   If x=0, y=2: 0+4+0 = 4 = 1 != 0 (ok)
    #   If x=0, y=0: 0+0+0 = 0 (bad)
    # If x=1, y=1: 1+1+1 = 3 = 0 (bad)
    # If x=1, y=2: 1+4+2 = 7 = 1 != 0 (ok)
    # If x=2, y=2: 4+4+4 = 12 = 0 (bad)
    
    # Valid pairs (x, y) mod 3:
    # (0, 1), (0, 2), (1, 0), (1, 2), (2, 0), (2, 1)
    # Note: 3 mod 3 is 0. So s_i can be 3 (0 mod 3), 1, or 2.
    
    states = [0] * n
    
    def is_valid(u, val):
        for v in adj[u]:
            if states[v] != 0:
                s_u = val % 3
                s_v = states[v] % 3
                if (s_u**2 + s_v**2 + s_u*s_v) % 3 == 0:
                    return False
        return True

    def backtrack(u):
        if u == n:
            return True
        # Try 1, 2, 3 (mod 3: 1, 2, 0)
        for val in [1, 2, 3]:
            if is_valid(u, val):
                states[u] = val
                if backtrack(u + 1):
                    return True
                states[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(states))
    else:
        print("NO")

solve()
