import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    # To find the lexicographically smallest assignment, we use backtracking.
    # To handle the "adversarial" constraint, we use a simple heuristic:
    # order variables by degree (MRV - Minimum Remaining Values is implicit in 
    # the fixed order, but we can use degree-based ordering for pruning).
    # However, for n=40, simple backtracking with forward checking is usually sufficient.
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # The problem asks for the lexicographically smallest sequence c1, c2, ..., cn.
    # This means we must try to assign the smallest possible tier to module 1,
    # then module 2, and so on. The standard backtracking does exactly this.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
