import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    def check(s):
        for u, v in edges:
            su, sv = s[u], s[v]
            if su == sv:
                if not (su * sv > 10):
                    return False
            elif su < sv:
                if not (sv - su < 3):
                    return False
            else: # su > sv
                if not (su + sv > 2):
                    return False
        return True

    # The constraints are:
    # 1. su == sv => su*sv > 10. Since s in {1, 2, 3}, 
    #    1*1=1, 2*2=4, 3*3=9. None are > 10.
    #    Therefore, su == sv is NEVER allowed.
    # 2. su < sv => sv - su < 3.
    #    If su=1, sv can be 2 (2-1=1 < 3) or 3 (3-1=2 < 3).
    #    If su=2, sv can be 3 (3-2=1 < 3).
    # 3. su > sv => su + sv > 2.
    #    If su=2, sv=1 (2+1=3 > 2).
    #    If su=3, sv=1 (3+1=4 > 2) or 2 (3+2=5 > 2).
    
    # This means for any edge (u, v), su != sv.
    # If su < sv, then sv - su < 3.
    # If su > sv, then su + sv > 2.
    # Since su, sv in {1, 2, 3} and su != sv:
    # If su=1, sv must be 2 or 3. (1 < 2, 1 < 3)
    # If su=2, sv can be 1 (2 > 1, 2+1=3 > 2) or 3 (2 < 3, 3-2=1 < 3).
    # If su=3, sv can be 1 (3 > 1, 3+1=4 > 2) or 2 (3 > 2, 3+2=5 > 2).
    
    # This is equivalent to a 3-coloring problem where edges are directed.
    # Actually, it's simpler: su != sv for all edges.
    # Let's use backtracking to find a valid 3-coloring.
    
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            assignment[u] = color
            valid = True
            for i in range(u + 1):
                for v_idx, (a, b) in enumerate(edges):
                    if (a == i and b == u) or (a == u and b == i):
                        su, sv = assignment[a], assignment[b]
                        if su == 0 or sv == 0: continue
                        if su == sv: valid = False; break
                        if su < sv and not (sv - su < 3): valid = False; break
                        if su > sv and not (su + sv > 2): valid = False; break
                if not valid: break
            if valid and backtrack(u + 1):
                return True
        assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
