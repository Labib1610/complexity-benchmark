import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    assignment = [-1] * n
    
    # Degree-based heuristic for variable ordering (MRV)
    # To get lexicographically smallest, we must try terminals 0 to n-1 in order.
    # However, we can use forward checking and MRV to prune the search space.
    
    def get_valid_colors(u, current_assignment):
        used = set()
        for v in adj[u]:
            if current_assignment[v] != -1:
                used.add(current_assignment[v])
        return [c for c in [1, 2, 3] if c not in used]

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3 in order for lexicographical smallest
        for color in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if assignment[v] == color:
                    possible = False
                    break
            
            if possible:
                assignment[u] = color
                
                # Forward checking: check if any unassigned neighbor has 0 options
                valid_forward = True
                for v in range(u + 1, n):
                    if assignment[v] == -1:
                        if not get_valid_colors(v, assignment):
                            valid_forward = False
                            break
                
                if valid_forward:
                    if backtrack(u + 1):
                        return True
                
                assignment[u] = -1
        return False

    # For n=40, simple backtracking with forward checking is usually sufficient
    # if the graph is sparse or highly constrained.
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
