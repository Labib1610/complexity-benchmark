import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The formula C(s_u, s_v) = -2 if (s_u, s_v) is (1, 2), (1, 3), (2, 1), (3, 1), (2, 3), (3, 2)
    # C(1, 1) = 3+3-12-12+2+2+18-27-27+30+22+22-14 = 10
    # C(2, 2) = 48+48-192-192+16+16+288-108-108+120+44+44-14 = 10
    # C(3, 3) = 243+243-972-972+54+54+1458-243-243+270+66+66-14 = 10
    # The condition C(s_u, s_v) = -2 is satisfied if and only if {s_u, s_v} = {1, 2} or {1, 3} or {2, 3}.
    # This means s_u != s_v and s_u, s_v in {1, 2, 3}.
    # This is equivalent to saying the graph must be 3-colorable.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
