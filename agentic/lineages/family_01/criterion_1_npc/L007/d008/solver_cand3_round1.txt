import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # D(s_u, s_v) = (s_u^2 + s_v^2) % (s_u * s_v)
    # Possible values:
    # (1, 1): (1+1)%1 = 0
    # (1, 2): (1+4)%2 = 1
    # (1, 3): (1+9)%3 = 1
    # (2, 1): (4+1)%2 = 1
    # (2, 2): (4+4)%4 = 0
    # (2, 3): (4+9)%6 = 1
    # (3, 1): (9+1)%3 = 1
    # (3, 2): (9+4)%6 = 1
    # (3, 3): (9+9)%9 = 0
    
    # Drift is 1 if s_u != s_v, and 0 if s_u == s_v.
    # We want sum of drifts = m.
    # Since there are m edges, and each edge contributes 0 or 1,
    # we need every edge to contribute 1.
    # This means for every edge (u, v), s_u must not equal s_v.
    # This is equivalent to saying the graph must be 3-colorable.
    # Since n <= 40, we can use backtracking to find a 3-coloring.

    colors = [-1] * n

    def is_valid(u, c):
        for v1, v2 in edges:
            if v1 == u and colors[v2] == c: return False
            if v2 == u and colors[v1] == c: return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = -1
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
