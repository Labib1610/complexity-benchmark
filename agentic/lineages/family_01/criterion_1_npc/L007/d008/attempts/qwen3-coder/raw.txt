from itertools import product

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u-1, v-1))  # Convert to 0-indexed
    
    # Precompute all possible drift values for pairs of frequencies
    drift = {}
    for s1 in [1, 2, 3]:
        for s2 in [1, 2, 3]:
            d = (s1*s1 + s2*s2) % (s1 * s2)
            drift[(s1, s2)] = d
            drift[(s2, s1)] = d  # Symmetric
    
    # Try all 3^n possible assignments
    for assignment in product([1, 2, 3], repeat=n):
        total_drift = 0
        valid = True
        
        for u, v in edges:
            freq_u = assignment[u]
            freq_v = assignment[v]
            total_drift += drift[(freq_u, freq_v)]
            
            # Early termination if we already exceed m
            if total_drift > m:
                valid = False
                break
                
        if valid and total_drift == m:
            print("YES")
            print(" ".join(map(str, assignment)))
            return
    
    print("NO")

solve()
