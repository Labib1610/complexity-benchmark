import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        edges.append((u, v))
        idx += 2

    # The formula for C_{u,v} simplifies significantly.
    # Let f(s) = 3s^5 - 15s^3 + 12s.
    # Let g(s) = -9s^4 + 6s^3 + 15s^2.
    # C_{u,v} = f(s_u) + f(s_v) + g(s_u)s_v + g(s_v)s_u + u^2 + v^2
    # We want sum(C_{u,v}) = sum(u^2 + v^2)
    # This is equivalent to sum(f(s_u) + f(s_v) + g(s_u)s_v + g(s_v)s_u) = 0
    
    def f(s):
        return 3 * (s**5) - 15 * (s**3) + 12 * s
    
    def g(s):
        return -9 * (s**4) + 6 * (s**3) + 15 * (s**2)

    # Precompute values for s in {1, 2, 3}
    # f(1) = 3 - 15 + 12 = 0
    # f(2) = 3*32 - 15*8 + 12*2 = 96 - 120 + 24 = 0
    # f(3) = 3*243 - 15*27 + 12*3 = 729 - 405 + 36 = 360
    # g(1) = -9 + 6 + 15 = 12
    # g(2) = -9*16 + 6*8 + 15*4 = -144 + 48 + 60 = -36
    # g(3) = -9*81 + 6*27 + 15*9 = -729 + 162 + 135 = -432
    
    vals_f = {1: 0, 2: 0, 3: 360}
    vals_g = {1: 12, 2: -36, 3: -432}
    
    # We need sum_{(u,v) in E} (f(s_u) + f(s_v) + g(s_u)s_v + g(s_v)s_u) = 0
    # Let deg(i) be the degree of node i.
    # sum_{i=1}^n (deg(i) * f(s_i)) + sum_{(u,v) in E} (g(s_u)s_v + g(s_v)s_u) = 0
    
    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    deg = [len(adj[i]) for i in range(n + 1)]
    
    # Since n is small (40), we can use backtracking with pruning.
    # To optimize, order nodes by degree or connectivity.
    assignment = [0] * (n + 1)
    
    def backtrack(idx):
        if idx > n:
            total = 0
            for u, v in edges:
                total += vals_f[assignment[u]] + vals_f[assignment[v]] + \
                         vals_g[assignment[u]] * assignment[v] + \
                         vals_g[assignment[v]] * assignment[u]
            return total == 0
        
        for s in [1, 2, 3]:
            assignment[idx] = s
            if backtrack(idx + 1):
                return True
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
