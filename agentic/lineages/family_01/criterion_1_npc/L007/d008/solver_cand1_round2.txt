import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The condition W = B is equivalent to W - B = 0.
    # C(s_u, s_v) - 2*s_u*s_v = 0
    # Let f(x, y) = C(x, y) - 2*x*y
    # f(1, 1) = 1+1-3-3+2+2-5-5+5+5+2+4+4 - 2 = 0
    # f(1, 2) = 1+32-3(16)-3(8)+2(4)+2(16)-5-40+5(4)+5(2)+2(2)+4+8 - 4 = 0
    # f(1, 3) = 1+243-3(81)-3(27)+2(9)+2(243)-5-135+5(9)+5(3)+2(3)+4+12 - 6 = 0
    # f(2, 2) = 32+32-3(16*2)-3(2*16)+2(8*4)+2(4*8)-5(8)-5(8)+5(4*2)+5(2*4)+2(4)+4(2)+4(2) - 8 = 0
    # f(2, 3) = 32+243-3(16*3)-3(2*27)+2(8*9)+2(4*27)-5(8)-5(27)+5(4*3)+5(2*9)+2(6)+4(2)+4(3) - 12 = 0
    # f(3, 3) = 243+243-3(81*3)-3(3*81)+2(27*9)+2(9*27)-5(27)-5(27)+5(9*3)+5(3*9)+2(9)+4(3)+4(3) - 18 = 0
    
    # Calculating f(x, y) values:
    # f(1,1) = 0
    # f(1,2) = 0
    # f(1,3) = 0
    # f(2,2) = 0
    # f(2,3) = 0
    # f(3,3) = 0
    # It turns out f(x, y) is always 0 for any x, y in {1, 2, 3}.
    # Let's double check:
    # C(x, y) = x^5 + y^5 - 3x^4y - 3xy^4 + 2x^3y^2 + 2x^2y^3 - 5x^3 - 5y^3 + 5x^2y + 5xy^2 + 2xy + 4x + 4y
    # f(x, y) = C(x, y) - 2xy
    # For x=1, y=1: 1+1-3-3+2+2-5-5+5+5+2+4+4 - 2 = 0. Correct.
    # For x=1, y=2: 1+32-3(16)-3(8)+2(4)+2(16)-5-40+5(4)+5(2)+2(2)+4+8 - 4 = 33-48-24+8+32-45+20+10+4+12-4 = 0. Correct.
    # For x=2, y=2: 32+32-3(32)-3(32)+2(32)+2(32)-40-40+40+40+8+8+8 - 8 = 64-192+128-80+80+24-8 = 16. Wait.
    # Re-calculating f(2,2): 32+32-3(64)-3(64)+2(32)+2(32)-5(8)-5(8)+5(8)+5(8)+2(4)+4(2)+4(2) - 8
    # = 64 - 192 - 192 + 64 + 64 - 40 - 40 + 40 + 40 + 8 + 8 + 8 - 8 = -128.
    # Actually, the problem implies there might be NO solution.
    
    # Since n <= 40, we can use backtracking with pruning or meet-in-the-middle.
    # Given the constraints and the nature of the problem, simple backtracking is sufficient.
    
    assignment = [0] * n
    
    def check():
        total_w = 0
        total_b = 0
        for u, v in edges:
            su, sv = assignment[u], assignment[v]
            total_b += 2 * su * sv
            total_w += (su**5 + sv**5 - 3*su**4*sv - 3*su*sv**4 + 2*su**3*sv**2 + 2*su**2*sv**3 - 5*su**3 - 5*sv**3 + 5*su**2*sv + 5*su*sv**2 + 2*su*sv + 4*su + 4*sv)
        return total_w == total_b

    def backtrack(idx):
        if idx == n:
            return check()
        for s in [1, 2, 3]:
            assignment[idx] = s
            if backtrack(idx + 1):
                return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
