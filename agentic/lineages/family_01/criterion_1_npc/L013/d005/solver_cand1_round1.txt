import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # Matrix [[e, 1], [0, e]] has eigenvalues e, e.
            # Only one distinct eigenvalue. Condition never satisfied.
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    # The matrix [[e_u, 1], [0, e_v]] has eigenvalues e_u and e_v.
    # It has exactly two distinct eigenvalues if and only if e_u != e_v.
    # This is a graph coloring problem: can we color the graph with 3 colors
    # such that no two adjacent vertices have the same color?
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # The number of parameters n is the perimeter of 11x11 grid:
    # 11 + 11 + 9 + 9 = 40.
    # 3-coloring is NP-complete, but for n=40, backtracking is usually fast.
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
