def solve():
    import sys
    from itertools import product

    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    incompatibilities = [set() for _ in range(n + 1)]
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        incompatibilities[u].add(v)
        incompatibilities[v].add(u)

    # Try all possible assignments of blocks
    for assignment in product(range(1, 4), repeat=n):
        valid = True
        for i in range(1, n + 1):
            for j in incompatibilities[i]:
                if assignment[i - 1] == assignment[j - 1]:
                    valid = False
                    break
            if not valid:
                break
        
        if valid:
            print("YES")
            print(" ".join(map(str, assignment)))
            return

    print("NO")
