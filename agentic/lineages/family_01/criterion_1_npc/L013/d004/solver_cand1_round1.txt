```python
import sys

# The problem asks to color a graph with 3 colors such that no two adjacent
# vertices have the same color. This is the 3-coloring problem.
# Given n <= 40, we can use a backtracking algorithm with heuristics.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [-1] * n
    
    # Heuristic: Degree-based ordering (MRV - Minimum Remaining Values)
    # Sort vertices by degree descending to fail fast
    order = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    
    def backtrack(idx_in_order):
        if idx_in_order == n:
            return True
        
        u = order[idx_in_order]
        
        # Find forbidden colors
        forbidden = 0
        for v in adj[u]:
            if colors[v] != -1:
                forbidden |= (1 << (colors[v] - 1))
        
        # Try colors 1, 2, 3
        for c in range(1, 4):
            if not (forbidden & (1 << (c - 1))):
                colors[u] = c
                if backtrack(idx_in_order + 1):
                    return True
                colors[u] = -1
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(2000)
    solve()
```
