import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Constraint: t_u * t_v is not a perfect square.
    # Possible products:
    # 1*1=1 (sq), 1*2=2, 1*3=3
    # 2*1=2, 2*2=4 (sq), 2*3=6
    # 3*1=3, 3*2=6, 3*3=9 (sq)
    # So, if t_u == t_v, it's a perfect square.
    # Thus, t_u != t_v is a necessary condition.
    # Additionally, 2*2=4 (sq) and 3*3=9 (sq) are forbidden.
    # Wait, the condition is t_u * t_v is NOT a perfect square.
    # 1*1=1 (sq), 2*2=4 (sq), 3*3=9 (sq).
    # So t_u != t_v is required.
    # Are there any other pairs?
    # 1*2=2 (no), 1*3=3 (no), 2*3=6 (no).
    # So the only forbidden pairs are (1,1), (2,2), (3,3).
    # This is exactly the graph coloring problem with 3 colors.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
