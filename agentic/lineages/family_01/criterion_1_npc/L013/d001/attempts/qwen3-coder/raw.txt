import sys
from itertools import product

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Try all 3^n assignments, but in a smarter way
    # We'll use backtracking to find the lexicographically smallest valid one
    
    assignment = [0] * n
    
    def is_valid(emp, shift):
        for u, v in edges:
            if u == emp and assignment[v] == shift:
                return False
            if v == emp and assignment[u] == shift:
                return False
        return True
    
    def backtrack(emp):
        if emp == n:
            return True
            
        for shift in [1, 2, 3]:
            if is_valid(emp, shift):
                assignment[emp] = shift
                if backtrack(emp + 1):
                    return True
                assignment[emp] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, assignment)))
    else:
        print("NO")

solve()
