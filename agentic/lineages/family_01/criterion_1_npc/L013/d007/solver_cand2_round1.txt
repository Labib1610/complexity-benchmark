import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2
        
    # Constraint: gcd(2*eu + ev, eu + 2*ev) % 3 != 0
    # Let A = 2*eu + ev, B = eu + 2*ev
    # gcd(A, B) = gcd(A, B - A) = gcd(2*eu + ev, -eu + ev)
    # gcd(2*eu + ev, ev - eu)
    # If eu = ev, gcd(3*eu, 0) = 3*eu. Since eu in {1, 2, 3}, 3*eu is always a multiple of 3.
    # So eu != ev is a necessary condition.
    # If eu != ev:
    # Case 1: {eu, ev} = {1, 2}. A = 2(1)+2 = 4, B = 1+2(2) = 5. gcd(4, 5) = 1. (Valid)
    # Case 2: {eu, ev} = {1, 3}. A = 2(1)+3 = 5, B = 1+2(3) = 7. gcd(5, 7) = 1. (Valid)
    # Case 3: {eu, ev} = {2, 3}. A = 2(2)+3 = 7, B = 2+2(3) = 8. gcd(7, 8) = 1. (Valid)
    # So the condition is simply eu != ev.
    # This is equivalent to the graph coloring problem with 3 colors.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for i, j in adj:
            if i == u and colors[j] == c:
                return False
            if j == u and colors[i] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
