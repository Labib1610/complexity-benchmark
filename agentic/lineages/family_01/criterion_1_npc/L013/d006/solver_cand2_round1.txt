import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    conditions = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        conditions.append((u, v))
        idx += 2

    # P(x) = x^3 - 6x^2 + 11x - 6
    # P(1) = 1 - 6 + 11 - 6 = 0
    # P(2) = 8 - 24 + 22 - 6 = 0
    # P(3) = 27 - 54 + 33 - 6 = 0
    # Since P(1)=P(2)=P(3)=0, the condition R(u, v) % 3 != 0 simplifies to:
    # (e_u^e_v + e_v^e_u + e_u * e_v) % 3 != 0

    # Precompute valid pairs (e_u, e_v)
    valid_pairs = []
    for eu in [1, 2, 3]:
        for ev in [1, 2, 3]:
            val = (pow(eu, ev, 3) + pow(ev, eu, 3) + (eu * ev)) % 3
            if val != 0:
                valid_pairs.append((eu, ev))

    assignment = [0] * (n + 1)

    def backtrack(u):
        if u > n:
            return True
        
        for e in [1, 2, 3]:
            assignment[u] = e
            possible = True
            for v1, v2 in conditions:
                if v1 == u and assignment[v2] != 0:
                    if (assignment[u], assignment[v2]) not in valid_pairs:
                        possible = False
                        break
                elif v2 == u and assignment[v1] != 0:
                    if (assignment[v1], assignment[u]) not in valid_pairs:
                        possible = False
                        break
            
            if possible:
                if backtrack(u + 1):
                    return True
            assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
