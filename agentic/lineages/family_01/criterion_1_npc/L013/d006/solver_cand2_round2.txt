import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        edges.append((u, v))
        idx += 2

    def F(x):
        return x**3 - 6*x**2 + 11*x - 6

    # Precompute R(u, v) mod 3 for all pairs (e_u, e_v)
    # R(u, v) = e_u^e_v + e_v^e_u + e_u*e_v + u*F(e_u) + v*F(e_v)
    # F(1) = 0, F(2) = 0, F(3) = 0.
    # So R(u, v) mod 3 = (e_u^e_v + e_v^e_u + e_u*e_v) mod 3
    
    allowed = {}
    for u in range(1, n + 1):
        for v in range(1, n + 1):
            if u == v: continue
            valid_pairs = []
            for eu in [1, 2, 3]:
                for ev in [1, 2, 3]:
                    val = (pow(eu, ev, 3) + pow(ev, eu, 3) + (eu * ev)) % 3
                    if val != 0:
                        valid_pairs.append((eu, ev))
            allowed[(u, v)] = valid_pairs

    assignment = [0] * (n + 1)

    def backtrack(u):
        if u > n:
            return True
        
        for state in [1, 2, 3]:
            possible = True
            for v in range(1, u):
                if (v, u) in allowed:
                    if (assignment[v], state) not in allowed[(v, u)]:
                        possible = False
                        break
            
            if possible:
                assignment[u] = state
                if backtrack(u + 1):
                    return True
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
