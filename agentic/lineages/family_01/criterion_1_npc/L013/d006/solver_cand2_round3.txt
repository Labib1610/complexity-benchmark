import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj.append((u, v))
        idx += 2
        
    # The condition: u*e_u + v*e_v != u*e_v + v*e_u
    # u*e_u - v*e_u - u*e_v + v*e_v != 0
    # (u - v) * e_u - (u - v) * e_v != 0
    # (u - v) * (e_u - e_v) != 0
    # Since u != v, u - v != 0.
    # Thus, e_u - e_v != 0, which means e_u != e_v.
    # This is simply a graph coloring problem with 3 colors.
    
    graph = [[] for _ in range(n + 1)]
    for u, v in adj:
        graph[u].append(v)
        graph[v].append(u)
        
    colors = [0] * (n + 1)
    
    def is_valid(node, color):
        for neighbor in graph[node]:
            if colors[neighbor] == color:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(node, color):
                colors[node] = color
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
