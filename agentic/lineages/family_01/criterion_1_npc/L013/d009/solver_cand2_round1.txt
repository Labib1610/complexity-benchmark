```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # W(e_u, e_v) = (e_u - e_v)^4 - 5(e_u - e_v)^2 + 7
    # Possible values for diff = |e_u - e_v|:
    # diff = 0: W = 0 - 0 + 7 = 7
    # diff = 1: W = 1 - 5 + 7 = 3
    # diff = 2: W = 16 - 20 + 7 = 3
    # We want sum(W) = 3m.
    # Since W is 3 if |e_u - e_v| is 1 or 2, and 7 if |e_u - e_v| is 0,
    # we need |e_u - e_v| != 0 for all edges.
    # This means e_u != e_v for all edges.
    # This is equivalent to 3-coloring the graph.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
