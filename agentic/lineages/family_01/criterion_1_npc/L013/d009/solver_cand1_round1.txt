import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    conditions = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        conditions.append((u, v))
        idx += 2
        
    # The condition is (u*e_v + v*e_u) % (u + v) != 0
    # Since n is small (up to 40), we can use backtracking with pruning.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, v, eu, ev):
        return (u * ev + v * eu) % (u + v) != 0
    
    def backtrack(curr):
        if curr > n:
            return True
        
        for state in [1, 2, 3]:
            assignment[curr] = state
            
            # Check all conditions involving curr and already assigned nodes
            possible = True
            for u, v in conditions:
                if u == curr and v < curr:
                    if not is_valid(u, v, assignment[u], assignment[v]):
                        possible = False
                        break
                elif v == curr and u < curr:
                    if not is_valid(u, v, assignment[u], assignment[v]):
                        possible = False
                        break
            
            if possible:
                if backtrack(curr + 1):
                    return True
        
        assignment[curr] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
