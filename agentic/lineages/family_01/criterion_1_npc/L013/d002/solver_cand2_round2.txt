import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # Sort nodes by degree descending to optimize backtracking (heuristic)
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    def backtrack(idx_in_nodes):
        if idx_in_nodes == n:
            return True
        
        u = nodes[idx_in_nodes]
        
        # Find forbidden periods
        forbidden = set()
        for v in adj[u]:
            if assignment[v] != 0:
                forbidden.add(assignment[v])
        
        for p in [1, 2, 3]:
            if p not in forbidden:
                assignment[u] = p
                if backtrack(idx_in_nodes + 1):
                    return True
                assignment[u] = 0
        
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
