import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning period 1, 2, or 3
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Heuristic: sort nodes by degree descending to prune search space
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to process in order of degree
    mapping = {node: i for i, node in enumerate(nodes)}
    rev_mapping = {i: node for i, node in enumerate(nodes)}
    new_adj = [[] for _ in range(n)]
    for u in range(1, n + 1):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    final_assignment = [0] * n
    
    def backtrack_optimized(idx):
        if idx == n:
            return True
        
        u = idx
        for color in [1, 2, 3]:
            possible = True
            for v in new_adj[u]:
                if final_assignment[v] == color:
                    possible = False
                    break
            if possible:
                final_assignment[u] = color
                if backtrack_optimized(idx + 1):
                    return True
                final_assignment[u] = 0
        return False

    if backtrack_optimized(0):
        print("YES")
        res = [0] * n
        for i in range(n):
            res[rev_mapping[i] - 1] = final_assignment[i]
        print(*(res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
