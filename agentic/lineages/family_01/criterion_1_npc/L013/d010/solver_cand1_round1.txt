import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition C = 0 means for every edge (u, v):
    # C_{uv} = e_u^2 + e_v^2 + e_u*e_v - 6(e_u + e_v) + 11 = 0
    # Let f(x, y) = x^2 + y^2 + xy - 6x - 6y + 11
    # Possible values for e_i are {1, 2, 3}:
    # f(1, 1) = 1 + 1 + 1 - 6 - 6 + 11 = 2
    # f(1, 2) = 1 + 4 + 2 - 6 - 12 + 11 = 0
    # f(1, 3) = 1 + 9 + 3 - 6 - 18 + 11 = 0
    # f(2, 2) = 4 + 4 + 4 - 12 - 12 + 11 = -1
    # f(2, 3) = 4 + 9 + 6 - 12 - 18 + 11 = 0
    # f(3, 3) = 9 + 9 + 9 - 18 - 18 + 11 = 2
    
    # Valid pairs (e_u, e_v) are (1, 2), (1, 3), (2, 1), (2, 3), (3, 1), (3, 2)
    # This is equivalent to saying e_u != e_v AND (e_u, e_v) != (2, 2) is not quite right.
    # Actually, the valid pairs are:
    # (1, 2), (2, 1), (1, 3), (3, 1), (2, 3), (3, 2)
    # This is exactly the condition that e_u != e_v AND {e_u, e_v} != {2, 2} is wrong.
    # Wait, the valid pairs are all pairs {x, y} from {1, 2, 3} such that x != y,
    # EXCEPT {2, 2} is invalid, {1, 1} is invalid, {3, 3} is invalid.
    # Actually, the valid pairs are exactly those where e_u != e_v.
    # Let's recheck:
    # f(1, 2) = 0 (Valid)
    # f(1, 3) = 0 (Valid)
    # f(2, 3) = 0 (Valid)
    # So the condition is simply e_u != e_v.
    
    # Wait, let's re-verify f(2, 3):
    # 4 + 9 + 6 - 12 - 18 + 11 = 19 - 30 + 11 = 0. Correct.
    # So the condition is simply e_u != e_v for all edges (u, v).
    # This is a 3-coloring problem.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
