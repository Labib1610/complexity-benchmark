import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    conditions = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        conditions.append((u, v))
        idx += 2

    # The condition for two distinct real roots for t^2 - C1*t + C2 = 0
    # is the discriminant D = C1^2 - 4*C2 > 0.
    # C1 = (u+v)(eu+ev)
    # C2 = (u^2+v^2)eu*ev + uv(eu^2+ev^2)
    # D = (u+v)^2(eu+ev)^2 - 4((u^2+v^2)eu*ev + uv(eu^2+ev^2))
    # D = (u+v)^2(eu^2 + 2eu*ev + ev^2) - 4(u^2*eu*ev + v^2*eu*ev + uv*eu^2 + uv*ev^2)
    # D = (u^2+2uv+v^2)(eu^2 + 2eu*ev + ev^2) - 4(u^2*eu*ev + v^2*eu*ev + uv*eu^2 + uv*ev^2)
    # After expansion and simplification:
    # D = u^2*eu^2 + 2*u^2*eu*ev + u^2*ev^2 + 2*uv*eu^2 + 4*uv*eu*ev + 2*uv*ev^2 + v^2*eu^2 + 2*v^2*eu*ev + v^2*ev^2
    #     - 4*u^2*eu*ev - 4*v^2*eu*ev - 4*uv*eu^2 - 4*uv*ev^2
    # D = u^2*eu^2 - 2*u^2*eu*ev + u^2*ev^2 + v^2*eu^2 - 2*v^2*eu*ev + v^2*ev^2 - 2*uv*eu^2 + 4*uv*eu*ev - 2*uv*ev^2
    # D = u^2(eu-ev)^2 + v^2(eu-ev)^2 - 2uv(eu^2 - 2eu*ev + ev^2)
    # D = (u^2 + v^2 - 2uv)(eu-ev)^2
    # D = (u-v)^2 * (eu-ev)^2
    # For D > 0, we need (u-v)^2 * (eu-ev)^2 > 0.
    # Since u != v (emitters are distinct), (u-v)^2 is always > 0.
    # Thus, we need (eu-ev)^2 > 0, which means eu != ev.
    
    # The problem reduces to: Can we color the graph with 3 colors such that
    # no two adjacent vertices have the same color?
    # This is the 3-coloring problem. Since n <= 40, we can use backtracking.
    
    adj = [[] for _ in range(n + 1)]
    for u, v in conditions:
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
