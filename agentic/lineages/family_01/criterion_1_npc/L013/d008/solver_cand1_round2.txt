import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        edges.append((u, v))
        idx += 2

    # The condition: t^2 - C1*t + C2 = 0 has two distinct real solutions.
    # This is equivalent to the discriminant D = C1^2 - 4*C2 > 0.
    # C1 = (u+v)(eu+ev)
    # C2 = uv(eu+ev)^2 + (u-v)^2 * eu * ev
    # D = (u+v)^2(eu+ev)^2 - 4(uv(eu+ev)^2 + (u-v)^2 * eu * ev)
    # D = (eu+ev)^2 * ((u+v)^2 - 4uv) - 4(u-v)^2 * eu * ev
    # D = (eu+ev)^2 * (u-v)^2 - 4(u-v)^2 * eu * ev
    # D = (u-v)^2 * ((eu+ev)^2 - 4*eu*ev)
    # D = (u-v)^2 * (eu^2 + 2*eu*ev + ev^2 - 4*eu*ev)
    # D = (u-v)^2 * (eu^2 - 2*eu*ev + ev^2)
    # D = (u-v)^2 * (eu - ev)^2
    # Since u != v, (u-v)^2 > 0.
    # Thus, D > 0 is equivalent to (eu - ev)^2 > 0, which means eu != ev.
    # The problem reduces to: assign eu in {1, 2, 3} such that for all (u, v) in edges, eu != ev.
    # This is exactly the 3-coloring problem.

    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    assignment = [0] * (n + 1)

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
