import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data: return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0]*n
    for _ in range(m):
        u = int(next(it))-1
        v = int(next(it))-1
        if u == v:
            print("NO")
            return
        adj[u] |= 1<<v
        adj[v] |= 1<<u

    try:
        bc = int.bit_count
    except AttributeError:
        def bc(x):
            return bin(x).count('1')

    def propagate(forb, mask):
        # build queue of singletons
        queue = []
        mm = mask
        while mm:
            lsb = mm & -mm
            i = lsb.bit_length() - 1
            fi = forb[i]
            if fi == 0:
                return None
            if fi & (fi - 1) == 0:
                queue.append(i)
            mm ^= lsb
        while queue:
            v = queue.pop()
            bit = 1 << v
            if not (mask & bit):
                continue
            fv = forb[v]
            if fv & (fv - 1):
                continue
            c = fv.bit_length() - 1
            mask &= ~bit
            nb = adj[v] & mask
            while nb:
                lsb = nb & -nb
                j = lsb.bit_length() - 1
                if (forb[j] >> c) & 1:
                    forb[j] &= ~(1 << c)
                    if forb[j] == 0:
                        return None
                    if forb[j] & (forb[j] - 1) == 0:
                        queue.append(j)
                nb ^= lsb
        return mask

    def reduce_mask(mask, forb):
        changed = True
        while changed:
            changed = False
            mm = mask
            while mm:
                lsb = mm & -mm
                i = lsb.bit_length() - 1
                if bc(forb[i]) > bc(adj[i] & mask):
                    mask ^= lsb
                    changed = True
                    break
                mm ^= lsb
        return mask

    sys.setrecursionlimit(10000)

    def can_color(forb, mask):
        # forb is owned by this call
        mask = propagate(forb, mask)
        if mask is None:
            return False
        mask = reduce_mask(mask, forb)
        if mask == 0:
            return True
        # choose vertex with smallest domain
        v = -1
        best = 4
        mm = mask
        while mm:
            lsb = mm & -mm
            i = lsb.bit_length() - 1
            cnt = bc(forb[i])
            if cnt < best:
                best = cnt
                v = i
                if cnt == 1:
                    break
            mm ^= lsb
        av = forb[v]
        c = 0
        # local references
        adj_local = adj
        while av:
            if av & 1:
                nforb = forb.copy()
                nmask = mask & ~(1 << v)
                nforb[v] = 1 << c
                nb = adj_local[v] & nmask
                ok = True
                while nb:
                    lsb = nb & -nb
                    j = lsb.bit_length() - 1
                    if (nforb[j] >> c) & 1:
                        nforb[j] &= ~(1 << c)
                        if nforb[j] == 0:
                            ok = False
                            break
                    nb ^= lsb
                if ok and can_color(nforb, nmask):
                    return True
            av >>= 1
            c += 1
        return False

    full_mask = (1 << n) - 1
    init_forb = [7] * n
    if not can_color(init_forb, full_mask):
        print("NO")
        return

    forb = [7] * n
    mask = full_mask
    ans = [0] * n
    for i in range(n):
        bit_i = 1 << i
        for c in range(3):
            if not ((forb[i] >> c) & 1):
                continue
            nforb = forb.copy()
            nmask = mask & ~bit_i
            nforb[i] = 1 << c
            nb = adj[i] & nmask
            ok = True
            while nb:
                lsb = nb & -nb
                j = lsb.bit_length() - 1
                if (nforb[j] >> c) & 1:
                    nforb[j] &= ~(1 << c)
                    if nforb[j] == 0:
                        ok = False
                        break
                nb ^= lsb
            if ok and can_color(nforb, nmask):
                ans[i] = c
                forb = nforb
                mask = nmask
                break
        else:
            print("NO")
            return

    out = sys.stdout
    out.write("YES\n")
    out.write(' '.join(str(c + 1) for c in ans))
    out.write('\n')
