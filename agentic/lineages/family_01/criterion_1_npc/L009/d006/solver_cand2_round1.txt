import sys

def solve():
    # The problem states:
    # c_i < 12 (number of seasons in a year is 12)
    # Constraint: (c_u * c_u + c_v * c_v) % (c_u + c_v) != 0
    # Note: c_u^2 + c_v^2 = (c_u + c_v)^2 - 2 * c_u * c_v
    # So (c_u^2 + c_v^2) % (c_u + c_v) != 0 is equivalent to
    # (-2 * c_u * c_v) % (c_u + c_v) != 0
    # Let S = c_u + c_v. Then -2 * c_u * c_v = -2 * c_u * (S - c_u) = -2 * c_u * S + 2 * c_u^2
    # So the condition is (2 * c_u^2) % (c_u + c_v) != 0
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We need to color the graph with colors {1, 2, ..., 11}
    # such that for every edge (u, v), (2 * c_u^2) % (c_u + c_v) != 0
    # This is a constraint satisfaction problem. Since n is small (40),
    # and we want the lexicographically smallest sequence, we can use backtracking.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] != 0:
                cv = colors[v]
                if (2 * c * c) % (c + cv) == 0:
                    return False
                if (2 * cv * cv) % (c + cv) == 0:
                    return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in range(1, 12):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
