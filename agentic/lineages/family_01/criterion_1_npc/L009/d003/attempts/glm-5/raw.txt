import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    FULL = (1 << (n + 1)) - 1
    memo = {}
    
    def can_finish(remaining, forbidden):
        if remaining == 0:
            return True
        key = (remaining, forbidden)
        if key in memo:
            return memo[key]
        res = False
        for v in range(1, n + 1):
            if remaining & (1 << v):
                if adj[v] & forbidden:
                    continue
                if can_finish(remaining ^ (1 << v), forbidden | (1 << v)):
                    res = True
                    break
        memo[key] = res
        return res
    
    color = [0] * (n + 1)
    remaining = FULL
    forbidden = 0
    
    for pos in range(1, n + 1):
        chosen = None
        for c in (1, 2, 3):
            if adj[pos] & (1 << c):
                continue
            new_remaining = remaining ^ (1 << pos)
            new_forbidden = forbidden | (1 << pos)
            if can_finish(new_remaining, new_forbidden):
                chosen = c
                break
        if chosen is None:
            print("NO")
            return
        color[pos] = chosen
        remaining ^= (1 << pos)
        forbidden |= (1 << pos)
    
    print("YES")
    print(" ".join(str(color[i]) for i in range(1, n + 1)))

solve()
