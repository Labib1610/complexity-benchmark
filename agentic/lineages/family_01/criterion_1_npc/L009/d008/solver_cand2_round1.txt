```python
import sys

def solve():
    # A perfectly alternating schedule of length c_u + c_v with c_u Morning and c_v Evening
    # shifts exists if and only if c_u == c_v.
    # Proof: In a perfectly alternating cycle, the number of Morning shifts must equal
    # the number of Evening shifts.
    # Thus, the condition "impossible to construct a perfectly alternating schedule"
    # is equivalent to c_u != c_v.
    
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m = map(int, line1)
    except ValueError:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)

    # We need to assign c_i in {1, 2, 3} such that for all (u, v) in constraints, c_u != c_v.
    # This is equivalent to 3-coloring the graph.
    # We want the lexicographically smallest sequence.
    
    res = [0] * n
    
    def backtrack(idx):
        if idx == n:
            return True
        
        for val in [1, 2, 3]:
            possible = True
            for neighbor in adj[idx]:
                if neighbor < idx and res[neighbor] == val:
                    possible = False
                    break
            
            if possible:
                res[idx] = val
                if backtrack(idx + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
```
