import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    constraints = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        constraints.append((u, v))
        idx += 2

    # S can range from n*1 to n*3, i.e., n to 3n.
    # For each possible sum S, try to find the lexicographically smallest sequence.
    best_seq = None

    for s in range(n, 3 * n + 1):
        # Backtracking to find the lexicographically smallest sequence for a fixed S
        current_seq = [0] * n
        
        def is_valid(idx, val):
            current_seq[idx] = val
            # Check constraints involving idx and already filled indices
            for u, v in constraints:
                if u == idx and current_seq[v] != 0:
                    if abs(val - current_seq[v]) % s == 0:
                        return False
                elif v == idx and current_seq[u] != 0:
                    if abs(val - current_seq[u]) % s == 0:
                        return False
            
            # Pruning: check if remaining sum is possible
            remaining_slots = n - 1 - idx
            current_sum = sum(current_seq[:idx+1])
            if current_sum + remaining_slots > s or current_sum + 3 * remaining_slots < s:
                return False
            return True

        def backtrack(idx, current_sum):
            if idx == n:
                return current_sum == s
            
            for val in [1, 2, 3]:
                if is_valid(idx, val):
                    if backtrack(idx + 1, current_sum + val):
                        return True
                    current_seq[idx] = 0
            current_seq[idx] = 0
            return False

        if backtrack(0, 0):
            if best_seq is None or current_seq < best_seq:
                best_seq = list(current_seq)

    if best_seq:
        print("YES")
        print(*(best_seq))
    else:
        print("NO")

solve()
