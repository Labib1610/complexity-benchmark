import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The polynomial P(x) = x^3 - (c_u + c_v + 4)x^2 + (c_u c_v + 4c_u + 4c_v)x - 4c_u c_v
    # Factors as (x - c_u)(x - c_v)(x - 4) = 0
    # The roots are c_u, c_v, and 4.
    # For the roots to be distinct, c_u != c_v, c_u != 4, and c_v != 4.
    # Since c_i in {1, 2, 3}, c_i != 4 is always true.
    # Thus, the condition is simply c_u != c_v for all links (u, v).
    # This is equivalent to finding a 3-coloring of the graph such that
    # adjacent nodes have different colors, minimizing the sequence lexicographically.

    colors = [0] * (n + 1)

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
