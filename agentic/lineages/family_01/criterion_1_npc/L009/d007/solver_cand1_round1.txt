import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The polynomial P(x) = x^3 - (c_u + c_v)x^2 + (c_u c_v)x = x(x - c_u)(x - c_v)
    # The roots are 0, c_u, c_v.
    # For these to be three distinct real roots, we must have:
    # 0 != c_u, 0 != c_v, and c_u != c_v.
    # Since c_i in {1, 2, 3}, the condition c_u != c_v is the only one that matters.
    # This means for every constraint (u, v), c_u must not equal c_v.
    # This is equivalent to finding a 3-coloring of the graph.
    
    res = [0] * n
    
    def is_valid(u, val):
        for v in adj[u]:
            if res[v] == val:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for val in [1, 2, 3]:
            if is_valid(u, val):
                res[u] = val
                if backtrack(u + 1):
                    return True
                res[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
