import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    pairs = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        pairs.append((u, v))
        idx += 2

    # The condition for coplanarity of (0,0,0), (1, c_u, c_u^2), (1, c_v, c_v^2), (1, 4, 16)
    # is that the determinant of the matrix formed by these vectors is 0.
    # Matrix:
    # | 1  c_u  c_u^2 |
    # | 1  c_v  c_v^2 |
    # | 1   4    16   |
    # Determinant = 1*(c_v*16 - 4*c_v^2) - c_u*(1*16 - 1*c_v^2) + c_u^2*(1*4 - 1*c_v)
    # = 16*c_v - 4*c_v^2 - 16*c_u + c_u*c_v^2 + 4*c_u^2 - c_u^2*c_v
    # = -4*c_v^2 + 16*c_v - 16*c_u + c_u*c_v^2 + 4*c_u^2 - c_u^2*c_v
    # = (c_u - 4)(c_v - 4)(c_u - c_v) = 0
    # This is 0 if c_u == 4 or c_v == 4 or c_u == c_v.
    # So, for stability, we need c_u != 4, c_v != 4, and c_u != c_v for all pairs (u, v).

    c = [0] * n
    
    def is_stable(idx, val):
        c[idx] = val
        for u, v in pairs:
            if u <= idx and v <= idx:
                if c[u] == 4 or c[v] == 4 or c[u] == c[v]:
                    return False
        return True

    def backtrack(idx):
        if idx == n:
            return True
        
        for val in range(1, 11):
            if val == 4:
                continue
            
            # Check constraints for current index
            possible = True
            for u, v in pairs:
                if u == idx and v < idx:
                    if c[v] == 4 or val == 4 or val == c[v]:
                        possible = False; break
                elif v == idx and u < idx:
                    if c[u] == 4 or val == 4 or val == c[u]:
                        possible = False; break
            
            if possible:
                c[idx] = val
                if backtrack(idx + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(c))
    else:
        print("NO")

solve()
