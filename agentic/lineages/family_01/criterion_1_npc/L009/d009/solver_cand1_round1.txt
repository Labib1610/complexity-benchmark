import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # F(u, v) = c_u^2 + c_u*c_v + c_v^2
    # Modulo 3:
    # c_i in {1, 2, 3} -> c_i mod 3 in {1, 2, 0}
    # If c_i = 3, c_i mod 3 = 0.
    # F(u, v) % 3 == 0:
    # If c_u=3, c_v=3: 0+0+0 = 0 (divisible by 3)
    # If c_u=3, c_v=1: 0+0+1 = 1
    # If c_u=3, c_v=2: 0+0+4 = 1
    # If c_u=1, c_v=1: 1+1+1 = 3 = 0 (divisible by 3)
    # If c_u=1, c_v=2: 1+2+4 = 7 = 1
    # If c_u=2, c_v=2: 4+4+4 = 12 = 0 (divisible by 3)
    
    # Forbidden pairs (c_u, c_v):
    # (3, 3), (1, 1), (2, 2)
    # This means for any edge (u, v), c_u != c_v.
    # This is equivalent to 3-coloring the graph.
    
    res = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if res[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                res[u] = color
                if backtrack(u + 1):
                    return True
                res[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
