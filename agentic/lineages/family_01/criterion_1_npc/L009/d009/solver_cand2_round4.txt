import sys

def solve():
    # The energy function is E(x, y) = 1/2 * (c_u x + c_v y)^2 + 1/2 * (c_v x + c_u y)^2 - x - y
    # This is a quadratic form E(x, y) = 1/2 * [x y] * M * [x y]^T - [1 1] * [x y]^T
    # where M = [[c_u^2 + c_v^2, 2*c_u*c_v], [2*c_u*c_v, c_u^2 + c_v^2]]
    # A unique minimum exists if and only if M is positive definite.
    # M is positive definite if det(M) > 0 and trace(M) > 0.
    # trace(M) = 2*(c_u^2 + c_v^2) > 0 (always true for c_i in {1, 2, 3})
    # det(M) = (c_u^2 + c_v^2)^2 - (2*c_u*c_v)^2
    #        = (c_u^2 + c_v^2 - 2*c_u*c_v) * (c_u^2 + c_v^2 + 2*c_u*c_v)
    #        = (c_u - c_v)^2 * (c_u + c_v)^2
    # For det(M) > 0, we need (c_u - c_v)^2 > 0, which means c_u != c_v.
    # Thus, the condition is simply c_u != c_v for all constraints (u, v).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_valid(u, val):
        for v in adj[u]:
            if assignment[v] == val:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for val in [1, 2, 3]:
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
