def solve():
    n, m = map(int, input().split())
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Precompute valid pairs: F(u,v) = c_u^2 + c_u*c_v + c_v^2 should not be divisible by 3
    valid_pairs = [[False] * 4 for _ in range(4)]
    for c_u in range(1, 4):
        for c_v in range(1, 4):
            F = c_u * c_u + c_u * c_v + c_v * c_v
            if F % 3 != 0:
                valid_pairs[c_u][c_v] = True
    
    # Try all 3^n assignments, but prune early using backtracking
    # We want the lexicographically smallest one
    assignment = [0] * (n + 1)
    
    def is_valid(vertex, color):
        for neighbor in adj[vertex]:
            if assignment[neighbor] != 0:
                if not valid_pairs[color][assignment[neighbor]]:
                    return False
        return True
    
    def backtrack(index):
        if index == n + 1:
            return True
        
        for color in range(1, 4):
            if is_valid(index, color):
                assignment[index] = color
                if backtrack(index + 1):
                    return True
                assignment[index] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(' '.join(map(str, assignment[1:])))
    else:
        print("NO")

solve()
