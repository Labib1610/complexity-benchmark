import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # Precompute valid pairs (c_u, c_v) such that F(u,v) is NOT divisible by 3
    # F(u,v) = c_u^2 + c_u*c_v + c_v^2
    # Valid pairs: (1,1), (2,2), (3,3), (1,2), (2,1)
    # Invalid pairs: (1,3), (3,1), (2,3), (3,2)
    valid_pairs = [[False] * 4 for _ in range(4)]
    for i in range(1, 4):
        for j in range(1, 4):
            F = i * i + i * j + j * j
            if F % 3 != 0:
                valid_pairs[i][j] = True
    
    assignment = [0] * n
    
    def is_valid(node, val):
        for neighbor in adj[node]:
            if assignment[neighbor] != 0:
                if not valid_pairs[val][assignment[neighbor]]:
                    return False
        return True
    
    def backtrack(idx):
        if idx == n:
            return True
        
        for val in [1, 2, 3]:
            if is_valid(idx, val):
                assignment[idx] = val
                if backtrack(idx + 1):
                    return True
                assignment[idx] = 0
        
        return False
    
    if backtrack(0):
        print("YES")
        print(' '.join(map(str, assignment)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
