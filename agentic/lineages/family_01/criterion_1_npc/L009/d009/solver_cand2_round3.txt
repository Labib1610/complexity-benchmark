import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # A number has an odd number of divisors if and only if it is a perfect square.
    # Possible levels are {1, 2, 3}.
    # Products:
    # 1*1=1 (sq), 1*2=2, 1*3=3
    # 2*1=2, 2*2=4 (sq), 2*3=6
    # 3*1=3, 3*2=6, 3*3=9 (sq)
    # Invalid pairs (product is square): (1,1), (2,2), (3,3)
    # So, for any linked pair (u, v), L_u != L_v.
    # This is equivalent to finding a 3-coloring of the graph.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for color in range(1, 4):
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
