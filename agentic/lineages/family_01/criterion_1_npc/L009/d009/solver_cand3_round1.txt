import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The equation f = 6 / (f^2 - 6f + 11)
    # f(f^2 - 6f + 11) = 6
    # f^3 - 6f^2 + 11f - 6 = 0
    # (f-1)(f-2)(f-3) = 0
    # So f_i must be in {1, 2, 3}.
    # The condition f_u/f_v + f_v/f_u > 2 is equivalent to (f_u - f_v)^2 / (f_u * f_v) > 0
    # This is true if and only if f_u != f_v.
    # So the problem is equivalent to finding the lexicographically smallest 3-coloring
    # of the graph such that colors are {1, 2, 3}.

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
