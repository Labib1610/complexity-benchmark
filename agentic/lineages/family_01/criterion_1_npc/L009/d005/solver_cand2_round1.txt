import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Constraint: sum of first c_i multiples <= 18
    # c_i * (1 + 2 + ... + c_i) = c_i * (c_i * (c_i + 1) / 2) <= 18
    # c_i^2 * (c_i + 1) <= 36
    # If c_i = 1: 1 * 2 = 2 <= 36 (True)
    # If c_i = 2: 4 * 3 = 12 <= 36 (True)
    # If c_i = 3: 9 * 4 = 36 <= 36 (True)
    # If c_i = 4: 16 * 5 = 80 > 36 (False)
    # So c_i can be 1, 2, or 3.
    
    # Determinant of M = [[cu, cv, u], [cv, cu, v], [1, 1, 0]]
    # det = cu(0 - v) - cv(0 - v) + u(cv - cu)
    # det = -cu*v + cv*v + u*cv - u*cu
    # det = cv(u + v) - cu(u + v) = (cv - cu)(u + v)
    # det != 0 implies cv != cu
    
    c = [0] * (n + 1)
    
    def is_valid(node, val):
        for neighbor in adj[node]:
            if c[neighbor] != 0 and c[neighbor] == val:
                return False
        return True

    def backtrack(node):
        if node > n:
            return True
        
        for val in [1, 2, 3]:
            if is_valid(node, val):
                c[node] = val
                if backtrack(node + 1):
                    return True
                c[node] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(c[1:]))
    else:
        print("NO")

solve()
