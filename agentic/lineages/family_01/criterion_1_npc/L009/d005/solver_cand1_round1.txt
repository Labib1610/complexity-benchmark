```python
import sys
import math

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The uniform wear property requires that for two gears with teeth T1 and T2,
    # every tooth on T1 must touch every tooth on T2. This happens if and only if
    # gcd(T1, T2) = 1.
    # Teeth counts: 5, 6, 7.
    # Pairs (T1, T2) with gcd(T1, T2) = 1:
    # (5, 6), (5, 7), (6, 7)
    # Note: (5, 5), (6, 6), (7, 7) have gcd > 1, so they are invalid.
    # This means the graph must be 3-colorable, where colors are {5, 6, 7}.
    
    teeth = [5, 6, 7]
    assignment = [0] * (n + 1)

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] != 0:
                if math.gcd(color, assignment[v]) != 1:
                    return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for color in teeth:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
```
