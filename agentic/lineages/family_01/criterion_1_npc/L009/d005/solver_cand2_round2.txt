```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition (c_u % c_v) + (c_v % c_u) > 0 is equivalent to:
    # NOT (c_u % c_v == 0 AND c_v % c_u == 0)
    # This is equivalent to: NOT (c_u == c_v)
    # So the constraint is simply c_u != c_v.
    # This is a graph coloring problem with 3 colors.
    
    res = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if res[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                res[u] = color
                if backtrack(u + 1):
                    return True
                res[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
```
