def can_assign(n, m, constraints):
    from collections import defaultdict
    
    graph = defaultdict(list)
    
    for u, v in constraints:
        graph[u].append(v)
        graph[v].append(u)
    
    colors = [0] * (n + 1)

    def dfs(node, color):
        colors[node] = color
        for neighbor in graph[node]:
            if colors[neighbor] == color:
                return False
            if colors[neighbor] == 0 and not dfs(neighbor, 3 - color):
                return False
        return True

    for i in range(1, n + 1):
        if colors[i] == 0:
            if not dfs(i, 1):
                return "NO", []
    
    assignment = []
    for i in range(1, n + 1):
        if colors[i] == 1:
            assignment.append(1)
        elif colors[i] == 2:
            assignment.append(2)
        else:
            assignment.append(3)

    return "YES", assignment

import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
constraints = [tuple(map(int, line.split())) for line in data[1:m + 1]]

result, assignment = can_assign(n, m, constraints)
if result == "NO":
    print("NO")
else:
    print("YES")
    print(' '.join(map(str, assignment)))
