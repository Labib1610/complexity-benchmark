import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # E(u, v) = (c_u - c_v)^4 - 5(c_u - c_v)^2 + 2
    # Possible values for diff = c_u - c_v (where c_i in {1, 2, 3}):
    # diff = 0: 0^4 - 5(0)^2 + 2 = 2
    # diff = 1: 1^4 - 5(1)^2 + 2 = 1 - 5 + 2 = -2
    # diff = -1: (-1)^4 - 5(-1)^2 + 2 = 1 - 5 + 2 = -2
    # diff = 2: 2^4 - 5(2)^2 + 2 = 16 - 20 + 2 = -2
    # diff = -2: (-2)^4 - 5(-2)^2 + 2 = 16 - 20 + 2 = -2
    # The only way to get -2 is if |c_u - c_v| != 0, i.e., c_u != c_v.
    # We need sum E(u, v) = -2m. Since each E(u, v) is at least -2,
    # this is only possible if E(u, v) = -2 for all edges.
    # This means c_u != c_v for all edges (u, v).
    # This is equivalent to finding a 3-coloring of the graph.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
