import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2

    # The quadratic equation is a_u * x^2 + (a_u + a_v) * x + a_v = 0
    # This factors as (a_u * x + a_v) * (x + 1) = 0
    # The roots are x = -a_v / a_u and x = -1.
    # For the roots to be distinct, -a_v / a_u != -1, which means a_v != a_u.
    # So for every pair (u, v), we must have a_u != a_v.
    # This is equivalent to saying the graph must be 3-colorable.
    
    graph = [[] for _ in range(n)]
    for u, v in adj:
        graph[u].append(v)
        graph[v].append(u)
        
    ans = [0] * n
    
    def is_valid(u, color):
        for v in graph[u]:
            if ans[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                ans[u] = color
                if backtrack(u + 1):
                    return True
                ans[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(ans))
    else:
        print("NO")

solve()
