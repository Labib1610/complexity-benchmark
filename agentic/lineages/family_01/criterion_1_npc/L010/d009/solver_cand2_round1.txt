import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # P1 = 998244353, P2 = 1000000007, P3 = 1000000009
    # All are primes.
    # Phi(X) = Phi(P_a * P_b)
    # If a == b: X = P_a^2, Phi(X) = P_a^2 - P_a = P_a(P_a - 1)
    # Inequality: P_a(P_a - 1) <= (P_a - 1)^2
    # P_a <= P_a - 1 => False.
    # So, a != b is required for any pair (u, v) in proximity.
    # If a != b: X = P_a * P_b, Phi(X) = (P_a - 1)(P_b - 1)
    # Inequality: (P_a - 1)(P_b - 1) <= (P_a - 1)(P_b - 1) => True.
    # Thus, the condition is equivalent to: for all (u, v) in proximity, f_u != f_v.
    # This is exactly the 3-coloring problem.

    assignment = [0] * n

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
