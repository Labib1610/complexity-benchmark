import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Condition: f_u^3 + f_v^3 > (f_u + f_v) * f_u * f_v
    # Let's check pairs (f_u, f_v) from {1, 2, 3}:
    # (1,1): 1+1 > 2*1*1 => 2 > 2 (False)
    # (1,2): 1+8 > 3*1*2 => 9 > 6 (True)
    # (1,3): 1+27 > 4*1*3 => 28 > 12 (True)
    # (2,1): True
    # (2,2): 8+8 > 4*2*2 => 16 > 16 (False)
    # (2,3): 8+27 > 5*2*3 => 35 > 30 (True)
    # (3,1): True
    # (3,2): True
    # (3,3): 27+27 > 6*3*3 => 54 > 54 (False)
    # Stable if {f_u, f_v} is {1, 2}, {1, 3}, or {2, 3}.
    # Essentially, f_u != f_v.
    # This is equivalent to 3-coloring the graph.

    res = [0] * n

    def is_valid(u, color):
        for v in adj[u]:
            if res[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                res[u] = color
                if backtrack(u + 1):
                    return True
                res[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
