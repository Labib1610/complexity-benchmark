```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # precompute allowed pairs (unordered)
    allowed = [[False]*4 for _ in range(4)]  # 1..3
    for a in (1,2,3):
        for b in (1,2,3):
            if a**3 + b**3 > (a+b)*a*b:
                allowed[a][b] = allowed[b][a] = True

    colors = [0]*n
    sys.setrecursionlimit(10000)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        for c in (1,2,3):
            ok = True
            for nb in adj[idx]:
                if colors[nb] and not allowed[c][colors[nb]]:
                    ok = False
                    break
            if not ok:
                continue
            colors[idx] = c
            if dfs(idx+1):
                return True
            colors[idx] = 0
        return False

    if dfs(0):
        print("YES")
        print(' '.join(map(str, colors)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
