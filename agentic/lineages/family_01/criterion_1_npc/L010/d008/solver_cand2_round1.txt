import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        edges.append(list(map(int, sys.stdin.readline().split())))

    # W = [[9, -7], [13, -10]]
    # M1 = W^1 = [[9, -7], [13, -10]]
    # M2 = W^2 = [[9*9-7*13, 9*-7-7*-10], [13*9-10*13, 13*-7-10*-10]]
    #          = [[81-91, -63+70], [117-130, -91+100]] = [[-10, 7], [-13, 9]]
    # M3 = W^3 = [[-10*9+7*13, -10*-7+7*-10], [-13*9+9*13, -13*-7+9*-10]]
    #          = [[-90+91, 70-70], [-117+117, 91-90]] = [[1, 0], [0, 1]]
    
    M = {
        1: ((9, -7), (13, -10)),
        2: ((-10, 7), (-13, 9)),
        3: ((1, 0), (0, 1))
    }

    def get_det(u_f, v_f):
        m1 = M[u_f]
        m2 = M[v_f]
        # C = M1 + M2
        c00 = m1[0][0] + m2[0][0]
        c01 = m1[0][1] + m2[0][1]
        c10 = m1[1][0] + m2[1][0]
        c11 = m1[1][1] + m2[1][1]
        return c00 * c11 - c01 * c10

    # Precompute det(f_u, f_v) for all pairs of f in {1, 2, 3}
    det_table = {}
    for f1 in range(1, 4):
        for f2 in range(1, 4):
            det_table[(f1, f2)] = get_det(f1, f2)

    f = [0] * (n + 1)

    def backtrack(idx):
        if idx > n:
            total_det = 0
            for u, v in edges:
                total_det += det_table[(f[u], f[v])]
            return total_det == m

        for val in range(1, 4):
            f[idx] = val
            # Pruning: check edges only involving nodes <= idx
            # This is a simple check, but for n=40, we need more efficiency.
            # However, the problem constraints and structure might allow this.
            if backtrack(idx + 1):
                return True
        return False

    # Since n=40 is too large for simple backtracking, check if there's a pattern.
    # The problem asks for lexicographically smallest.
    # Let's try a greedy approach with backtracking.
    
    f = [0] * (n + 1)
    def solve_recursive(idx):
        if idx > n:
            total_det = 0
            for u, v in edges:
                total_det += det_table[(f[u], f[v])]
            return total_det == m
        
        for val in range(1, 4):
            f[idx] = val
            # Partial check: sum of det for edges where both u, v <= idx
            possible = True
            current_sum = 0
            count = 0
            for u, v in edges:
                if u <= idx and v <= idx:
                    current_sum += det_table[(f[u], f[v])]
                    count += 1
            
            # This is a heuristic; if the remaining edges can't satisfy the sum, prune.
            # But we don't know the future values.
            if solve_recursive(idx + 1):
                return True
        return False

    if solve_recursive(1):
        print("YES")
        print(*(f[1:]))
    else:
        print("NO")

solve()
