import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    # For each pair of values (a, b), check if E(x)=0 has exactly 3 distinct real roots
    # E(x) = x(x-a)(x-b), roots are 0, a, b. Need exactly 3 distinct: 0, a, b all distinct.
    # So a != 0, b != 0, a != b. Since a,b in {1,2,3}, always true. So all pairs are valid.
    # Thus the answer is always YES, and lexicographically smallest is all 1s.
    
    print("YES")
    print(" ".join(["1"] * n))

solve()
