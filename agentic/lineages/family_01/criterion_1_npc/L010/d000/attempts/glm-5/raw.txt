import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    # Precompute independent sets
    indep_sets = []
    for mask in range(1 << n):
        valid = True
        bits = mask
        while bits:
            lb = bits & -bits
            u = (lb.bit_length() - 1)
            if adj[u] & mask:
                valid = False
                break
            bits ^= lb
        if valid:
            indep_sets.append(mask)
    
    # Find valid triplets
    valid_triplets = []
    for s1 in indep_sets:
        rem1 = ((1 << n) - 1) ^ s1
        for s2 in indep_sets:
            if s2 & s1:
                continue
            rem2 = rem1 ^ s2
            if rem2 in indep_sets:
                valid_triplets.append((s1, s2, rem2))
    
    if not valid_triplets:
        print("NO")
        return
    
    # Find lexicographically smallest assignment
    best_seq = None
    
    for s1, s2, s3 in valid_triplets:
        seq = [0] * n
        for i in range(n):
            bit = 1 << i
            if s1 & bit:
                seq[i] = 1
            elif s2 & bit:
                seq[i] = 2
            else:
                seq[i] = 3
        
        if best_seq is None or seq < best_seq:
            best_seq = seq
    
    print("YES")
    print(' '.join(map(str, best_seq)))

if __name__ == "__main__":
    solve()
