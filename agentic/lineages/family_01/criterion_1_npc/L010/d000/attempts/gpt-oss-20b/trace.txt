def dfs(idx):
    if idx>n: solution found
    for c in 1..3:
        if (adj_mask[idx] & color_mask[c])==0:
            # check avail for neighbors
            changes=[]
            ok=True
            for j in neighbors[idx]:
                if color[j]==0:
                    prev=avail_mask[j]
                    new=prev & ~(1<<(c-1))
                    if new==0:
                        ok=False
                        break
                    avail_mask[j]=new
                    changes.append((j,prev))
            if not ok:
                # revert
                for j,prev in changes:
                    avail_mask[j]=prev
                continue
            # assign
            color[idx]=c
            color_mask[c] |= 1<<(idx-1)
            if dfs(idx+1): return True
            # revert assignment
            color[idx]=0
            color_mask[c] ^= 1<<(idx-1)
            for j,prev in changes:
                avail_mask[j]=prev
    return False
