import sys

def solve():
    # The function W_{u,v}(x) = x^3/3 - ((f_u + f_v)/2)x^2 + (f_u * f_v)x
    # Its derivative is W'_{u,v}(x) = x^2 - (f_u + f_v)x + f_u * f_v
    # This is a quadratic (x - f_u)(x - f_v).
    # A local maximum exists if and only if the derivative changes sign from + to -.
    # The roots of the derivative are f_u and f_v.
    # If f_u != f_v, the roots are distinct, and the function has a local max and a local min.
    # If f_u == f_v, the derivative is (x - f_u)^2, which is >= 0 everywhere.
    # In this case, the function is monotonically increasing and has no local maximum.
    # Thus, a resonant loop exists if and only if f_u == f_v.
    # The condition "no resonant loops" means for all proximity pairs (u, v), f_u != f_v.
    # This is exactly the graph coloring problem with 3 colors.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
