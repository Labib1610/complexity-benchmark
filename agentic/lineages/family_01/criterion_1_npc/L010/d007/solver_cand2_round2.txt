```python
import sys

def solve():
    # W(t) = 2t^3 - 3(f_u + f_v)t^2 + 6f_u f_v t
    # W'(t) = 6t^2 - 6(f_u + f_v)t + 6f_u f_v = 6(t^2 - (f_u + f_v)t + f_u f_v)
    # W'(t) = 6(t - f_u)(t - f_v)
    # W''(t) = 12t - 6(f_u + f_v)
    # Stationary point of inflection: W'(t) = 0 and W''(t) = 0
    # W'(t) = 0 at t = f_u and t = f_v
    # W''(t) = 0 at t = (f_u + f_v) / 2
    # For a stationary point of inflection, we need f_u = f_v = (f_u + f_v) / 2
    # This simplifies to f_u = f_v.
    # Thus, a resonant loop exists if and only if f_u = f_v for any proximity pair (u, v).
    # This is equivalent to finding a 3-coloring of the graph such that no two adjacent
    # vertices have the same color.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
