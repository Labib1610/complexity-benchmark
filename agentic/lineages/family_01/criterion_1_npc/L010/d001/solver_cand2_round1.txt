import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    constraints = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        constraints.append((u, v))
        idx += 2

    # x_u + 2*x_v != 0 (mod 3)
    # Let y_i = x_i mod 3. Since x_i in {1, 2, 3}, y_i in {0, 1, 2}.
    # If y_i = 0, it corresponds to x_i = 3.
    # If y_i = 1, it corresponds to x_i = 1.
    # If y_i = 2, it corresponds to x_i = 2.
    # Condition: y_u + 2*y_v != 0 (mod 3)
    # y_u - y_v != 0 (mod 3) => y_u != y_v (mod 3)
    
    # This is equivalent to a graph coloring problem where each constraint
    # (u, v) means y_u != y_v.
    # Since we have 3 colors {0, 1, 2}, this is 3-coloring.
    # n <= 40 is small enough for backtracking with pruning.
    
    adj = [[] for _ in range(n)]
    for u, v in constraints:
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [-1] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 0, 1, 2
        for color in range(3):
            possible = True
            for v in adj[u]:
                if assignment[v] == color:
                    possible = False
                    break
            if possible:
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = -1
        return False

    if backtrack(0):
        print("YES")
        res = []
        for val in assignment:
            if val == 0:
                res.append(3)
            else:
                res.append(val)
        print(*(res))
    else:
        print("NO")

solve()
