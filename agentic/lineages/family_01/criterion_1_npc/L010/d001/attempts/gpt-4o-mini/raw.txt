def main():
    import sys
    from collections import defaultdict

    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    edges = [tuple(map(int, line.split())) for line in data[1:m+1]]

    # Create adjacency list for interference rules
    graph = defaultdict(set)
    for u, v in edges:
        graph[u].add(v)
        graph[v].add(u)

    # Array to hold the tier assignments, initialized as 0 (unassigned)
    tiers = [0] * (n + 1)

    def can_assign(task, assigned_tier):
        for neighbor in graph[task]:
            if tiers[neighbor] == assigned_tier:
                return False
        return True

    def assign_tiers(task):
        if task > n:
            return True  # All tasks assigned

        for tier in range(1, 4):
            if can_assign(task, tier):
                tiers[task] = tier
                if assign_tiers(task + 1):
                    return True
                tiers[task] = 0  # Reset on backtrack
        return False

    if assign_tiers(1):
        print("YES")
        print(' '.join(map(str, tiers[1:])))
    else:
        print("NO")

if __name__ == "__main__":
    main()
