 We need to check if the graph is 3-colorable. Since \( n \leq 40 \), we can use backtracking with pruning.  

Here’s the Python solution:  

```python
n, m = map(int, input().split())
edges = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    edges[u].append(v)
    edges[v].append(u)

colors = [0] * n

def is_safe(node, color):
    for neighbor in edges[node]:
        if colors[neighbor] == color:
            return False
    return True

def solve(node):
    if node == n:
        return True
    for color in range(1, 4):
        if is_safe(node, color):
            colors[node] = color
            if solve(node + 1):
                return True
            colors[node] = 0
    return False

if solve(0):
    print("YES")
    print(" ".join(map(str, colors)))
else:
    print("NO")
```
