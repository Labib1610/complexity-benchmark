import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Condition: sum is even => resonant loop, UNLESS (1, 3) or (3, 1).
    # Multipliers: 1, 2, 3.
    # Pairs (f_u, f_v):
    # (1, 1): sum 2 (even) -> Loop
    # (1, 2): sum 3 (odd) -> OK
    # (1, 3): sum 4 (even) -> Exception (OK)
    # (2, 1): sum 3 (odd) -> OK
    # (2, 2): sum 4 (even) -> Loop
    # (2, 3): sum 5 (odd) -> OK
    # (3, 1): sum 4 (even) -> Exception (OK)
    # (3, 2): sum 5 (odd) -> OK
    # (3, 3): sum 6 (even) -> Loop
    #
    # Forbidden pairs: (1, 1), (2, 2), (3, 3).
    # This is exactly the condition for graph coloring with 3 colors!
    # If f_u == f_v, it's a loop.
    # Wait, let's re-read: "if the sum is even, loop, EXCEPT (1, 3)".
    # So (1, 1) is bad, (2, 2) is bad, (3, 3) is bad.
    # (1, 2) is good, (1, 3) is good, (2, 3) is good.
    # Yes, this is exactly 3-coloring.
    
    res = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if res[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                res[u] = color
                if backtrack(u + 1):
                    return True
                res[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
