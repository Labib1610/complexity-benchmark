import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # A resonant loop occurs if (f_u + f_v) is even, EXCEPT if {f_u, f_v} = {1, 3}.
    # Possible pairs (f_u, f_v):
    # (1, 1) -> sum 2 (even) -> Loop
    # (1, 2) -> sum 3 (odd)  -> OK
    # (1, 3) -> sum 4 (even) -> Exception: OK
    # (2, 2) -> sum 4 (even) -> Loop
    # (2, 3) -> sum 5 (odd)  -> OK
    # (3, 3) -> sum 6 (even) -> Loop
    
    # Forbidden pairs: (1, 1), (2, 2), (3, 3)
    # Also (1, 3) is allowed.
    # So, f_u == f_v is forbidden.
    # This is equivalent to a graph coloring problem with 3 colors where
    # adjacent nodes must have different colors.
    
    assignment = [0] * n

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
