import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # C_i = (1, f_i, f_i^2)
    # C_u x C_v = (f_u*f_v^2 - f_v*f_u^2, f_u^2 - f_v^2, f_v - f_u)
    # = (f_u*f_v*(f_v - f_u), (f_u - f_v)*(f_u + f_v), -(f_u - f_v))
    # This is (0,0,0) if and only if f_u = f_v.
    # Proof:
    # If f_u = f_v, then C_u = C_v, so C_u x C_v = 0.
    # If f_u != f_v, then the third component -(f_u - f_v) is non-zero.
    # Thus, the condition "no resonant loop" is equivalent to "no two adjacent relays have the same multiplier".
    # This is exactly the graph coloring problem with 3 colors.

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
