import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition: integral from f_u to f_v of e^t dt = 0
    # [e^t] from f_u to f_v = e^{f_v} - e^{f_u} = 0
    # This implies e^{f_v} = e^{f_u}, which means f_v = f_u.
    # So, for any pair (u, v) in proximity, we must have f_u != f_v.
    # This is exactly the graph coloring problem with 3 colors.
    # We want to minimize the sequence lexicographically.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
