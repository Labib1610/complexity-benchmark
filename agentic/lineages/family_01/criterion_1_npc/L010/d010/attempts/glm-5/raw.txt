import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [set() for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
    
    # For each pair of multipliers (a,b) in {1,2,3}^2, check if they are compatible.
    # Compatible means: for all x in [a-1/3, a+1/3] and y in [b-1/3, b+1/3], |x-y| > 1/4.
    # We use exact rational arithmetic.
    compatible = [[False]*3 for _ in range(3)]
    for a in range(3):
        for b in range(3):
            lo_a = (3*a - 1, 3)  # a - 1/3
            hi_a = (3*a + 1, 3)  # a + 1/3
            lo_b = (3*b - 1, 3)
            hi_b = (3*b + 1, 3)
            # We need min |x-y| > 1/4 over the rectangle.
            # The minimum of |x-y| over intervals [lo_a,hi_a] and [lo_b,hi_b] is:
            # 0 if they overlap, otherwise the gap between them.
            # Intervals overlap iff lo_a <= hi_b and lo_b <= hi_a.
            # Compare lo_a <= hi_b: (3a-1)/3 <= (3b+1)/3  => 3a-1 <= 3b+1 => 3a <= 3b+2 => a <= b + 2/3
            # Since a,b integers, a <= b.
            # Compare lo_b <= hi_a: similarly b <= a.
            # So overlap iff a == b.
            if a == b:
                compatible[a][b] = False
            else:
                # gap = lo_b - hi_a if b > a, else lo_a - hi_b
                # gap > 1/4 ?
                # If b > a: gap = (3b-1)/3 - (3a+1)/3 = (3(b-a)-2)/3 = (3(b-a)-2)/3
                # Need (3(b-a)-2)/3 > 1/4 => 4(3(b-a)-2) > 3 => 12(b-a) - 8 > 3 => 12(b-a) > 11 => b-a >= 1 (true for b-a>=1)
                # If b < a: gap = (3a-1)/3 - (3b+1)/3 = (3(a-b)-2)/3, same thing.
                # So for a != b, gap = (3*|a-b| - 2)/3. For |a-b|=1: gap = 1/3 > 1/4. For |a-b|=2: gap = 4/3 > 1/4.
                compatible[a][b] = True
    
    # Now we need to assign each vertex a value in {0,1,2} (representing multiplier 1,2,3)
    # such that for every edge (u,v), compatible[assign[u]][assign[v]] is True.
    # This means adjacent vertices must have different multipliers (since a==b is the only incompatible case).
    # Wait, let's double check: compatible[a][b] is False only when a==b.
    # So the constraint is simply that adjacent vertices have different multipliers.
    # This is graph 3-coloring.
    
    # But wait, the problem says the system is stable iff for ALL x_i in bounds, |x_u - x_v| > 1/4 for all edges.
    # We showed this holds iff f_u != f_v. So it's exactly 3-coloring.
    
    # For lexicographically smallest, we try to assign vertex 0 the smallest value (0), then vertex 1, etc.
    # Using backtracking with constraint propagation.
    
    color = [-1] * n
    
    def backtrack(v):
        if v == n:
            return True
        for c in range(3):
            ok = True
            for u in adj[v]:
                if color[u] == c:
                    ok = False
                    break
            if ok:
                color[v] = c
                if backtrack(v + 1):
                    return True
                color[v] = -1
        return False
    
    if backtrack(0):
        out = ['YES']
        out.append(' '.join(str(color[i] + 1) for i in range(n)))
        sys.stdout.write('\n'.join(out) + '\n')
    else:
        sys.stdout.write('NO\n')

solve()
