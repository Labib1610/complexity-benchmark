import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Multipliers f_i must be positive integers such that f_i^3 < 30.
    # Possible values: 1, 2, 3 (since 3^3 = 27 < 30, 4^3 = 64 > 30).
    possible_values = [1, 2, 3]
    
    # The condition is sum((f_u - f_v)^4) = 5 * sum((f_u - f_v)^2) - 4 * m
    # Let x = f_u - f_v.
    # If x = 0, x^4 = 0, 5*x^2 = 0. Contribution: 0 = 0 - 0.
    # If x = 1, x^4 = 1, 5*x^2 = 5. Contribution: 1 = 5 - 4.
    # If x = -1, x^4 = 1, 5*x^2 = 5. Contribution: 1 = 5 - 4.
    # If x = 2, x^4 = 16, 5*x^2 = 20. Contribution: 16 = 20 - 4.
    # If x = -2, x^4 = 16, 5*x^2 = 20. Contribution: 16 = 20 - 4.
    # For any integer x, x^4 = 5*x^2 - 4 is true if and only if x^4 - 5*x^2 + 4 = 0.
    # (x^2 - 1)(x^2 - 4) = 0.
    # This holds if x^2 = 1 or x^2 = 4, i.e., x in {1, -1, 2, -2}.
    # The condition holds if for every edge (u, v), |f_u - f_v| is in {1, 2}.
    # This means |f_u - f_v| cannot be 0.
    # So f_u != f_v for all edges (u, v).
    # This is equivalent to saying the graph is 3-colorable using {1, 2, 3}.

    current_f = [0] * n

    def is_valid(idx, val):
        for u, v in edges:
            if u == idx:
                if current_f[v] != 0 and abs(val - current_f[v]) == 0:
                    return False
            elif v == idx:
                if current_f[u] != 0 and abs(val - current_f[u]) == 0:
                    return False
        return True

    def backtrack(idx):
        if idx == n:
            return True
        for val in possible_values:
            if is_valid(idx, val):
                current_f[idx] = val
                if backtrack(idx + 1):
                    return True
                current_f[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(current_f))
    else:
        print("NO")

solve()
