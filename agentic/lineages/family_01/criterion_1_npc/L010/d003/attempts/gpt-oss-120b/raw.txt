```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    # masks: bit 0 -> color 1, bit1 -> color2, bit2 -> color3
    ALL = 0b111
    allowed = [ALL] * n
    color = [0] * n  # 0 = unassigned, else 1..3

    sys.setrecursionlimit(10000)

    # order is fixed 0..n-1 for lexicographic minimality
    order = list(range(n))

    # Preprocess: if any vertex has degree >= 3 and graph is complete? not needed

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = order[idx]
        # try colors 1..3
        for c in range(1, 4):
            if not (allowed[v] & (1 << (c - 1))):
                continue
            # assign
            color[v] = c
            changed = []  # list of (u, old_mask)
            fail = False
            for u in adj[v]:
                if color[u] == 0:
                    old = allowed[u]
                    new = old & ~(1 << (c - 1))
                    if new != old:
                        allowed[u] = new
                        changed.append((u, old))
                        if new == 0:
                            fail = True
                            break
            if not fail and dfs(idx + 1):
                return True
            # revert
            for u, old in reversed(changed):
                allowed[u] = old
            color[v] = 0
        return False

    possible = dfs(0)
    out = sys.stdout
    if not possible:
        out.write("NO")
    else:
        out.write("YES\n")
        out.write(" ".join(str(c) for c in color))

if __name__ == "__main__":
    solve()
```
