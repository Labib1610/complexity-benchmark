import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Condition: (a_u^2 + a_w^2 + a_u * a_w) % 3 != 0
    # Possible values for a_i in {1, 2, 3}:
    # If a_i = 3, a_i % 3 = 0.
    # If a_i = 1, a_i % 3 = 1.
    # If a_i = 2, a_i % 3 = 2.
    # Let x = a_u % 3, y = a_w % 3.
    # Condition: (x^2 + y^2 + xy) % 3 != 0
    # If x=0 or y=0:
    #   If x=0, y=1: (0+1+0) = 1 != 0 (OK)
    #   If x=0, y=2: (0+4+0) = 4 = 1 != 0 (OK)
    #   If x=0, y=0: (0+0+0) = 0 (FAIL)
    # If x=1, y=1: (1+1+1) = 3 = 0 (FAIL)
    # If x=1, y=2: (1+4+2) = 7 = 1 != 0 (OK)
    # If x=2, y=2: (4+4+4) = 12 = 0 (FAIL)
    
    # Mapping: 3 -> 0, 1 -> 1, 2 -> 2
    # Valid pairs (x, y): (0, 1), (0, 2), (1, 2)
    # Invalid pairs: (0, 0), (1, 1), (2, 2)
    # This is equivalent to saying x != y.
    # So we need to color the graph with 3 colors such that no two adjacent nodes have the same color.
    # This is exactly the 3-coloring problem.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
