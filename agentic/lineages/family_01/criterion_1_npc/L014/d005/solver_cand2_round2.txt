import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Constraint: gcd(a, b) + lcm(a, b) > 2 * min(a, b)
    # Possible pairs (a, b) from {1, 2, 3}:
    # (1, 1): gcd=1, lcm=1, min=1. 1+1 > 2*1 => 2 > 2 (False)
    # (1, 2): gcd=1, lcm=2, min=1. 1+2 > 2*1 => 3 > 2 (True)
    # (1, 3): gcd=1, lcm=3, min=1. 1+3 > 2*1 => 4 > 2 (True)
    # (2, 2): gcd=2, lcm=2, min=2. 2+2 > 2*2 => 4 > 4 (False)
    # (2, 3): gcd=1, lcm=6, min=2. 1+6 > 2*2 => 7 > 4 (True)
    # (3, 3): gcd=3, lcm=3, min=3. 3+3 > 2*3 => 6 > 6 (False)
    
    # Valid pairs: (1, 2), (1, 3), (2, 3)
    # Invalid pairs: (1, 1), (2, 2), (3, 3)
    # This means no two connected components can have the same voltage.
    # This is equivalent to finding a 3-coloring of the graph.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
