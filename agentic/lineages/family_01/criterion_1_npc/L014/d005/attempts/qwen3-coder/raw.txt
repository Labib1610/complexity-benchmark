import sys
from collections import defaultdict

def is_perfect_square(n):
    if n < 0:
        return False
    root = int(n**0.5)
    return root * root == n

def is_bad_pair(a, b):
    return is_perfect_square(a * b)

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # If no edges, any assignment works
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Try all possible assignments (1, 2, 3 GHz)
    # We'll use backtracking
    assignment = [0] * n  # 0 means unassigned
    
    def is_valid(node, speed):
        # Check if assigning 'speed' to 'node' conflicts with neighbors
        for neighbor in adj[node]:
            if assignment[neighbor] != 0:  # if neighbor is assigned
                if is_bad_pair(speed, assignment[neighbor]):
                    return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
            
        for speed in [1, 2, 3]:
            if is_valid(node, speed):
                assignment[node] = speed
                if backtrack(node + 1):
                    return True
                assignment[node] = 0  # backtrack
                
        return False
    
    # Optimization: if the graph is not bipartite, it might still be solvable
    # But let's try a more systematic approach for small n
    
    # For small graphs, we can try all 3^n combinations
    # But for n=40, that's too much. Let's stick with backtracking with optimizations
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, assignment)))
    else:
        print("NO")

# 3*1=3           3*2=6           3*3=9 (square)

# So forbidden pairs are: (1,1), (2,2), (3,3)
# This is simply a graph coloring problem where adjacent nodes cannot have the same color!

# Wait, that's not right. Let me recheck:
# Perfect squares up to 9: 1, 4, 9

# 2*1=2    2*2=4 ✓  2*3=6
# 3*1=3    3*2=6   3*3=9 ✓

# So nodes connected by an edge cannot be assigned the same speed.
# This is a graph 3-coloring problem!

def solve_optimized():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    # If no edges, any assignment works
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Graph 3-coloring with colors {1, 2, 3}
    color = [0] * n  # 0 means uncolored
    
    def is_safe(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def graph_coloring(node):
        if node == n:
            return True
            
        for c in [1, 2, 3]:
            if is_safe(node, c):
                color[node] = c
                if graph_coloring(node + 1):
                    return True
                color[node] = 0
                
        return False
    
    if graph_coloring(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

solve_optimized()
