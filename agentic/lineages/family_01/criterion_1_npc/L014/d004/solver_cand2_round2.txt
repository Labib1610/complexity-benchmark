import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # A number has an even number of divisors if and only if it is not a perfect square.
    # Firm IDs are {1, 2, 3}.
    # Products:
    # 1*1 = 1 (square, odd divisors) -> INVALID
    # 1*2 = 2 (not square, even) -> VALID
    # 1*3 = 3 (not square, even) -> VALID
    # 2*1 = 2 (not square, even) -> VALID
    # 2*2 = 4 (square, odd) -> INVALID
    # 2*3 = 6 (not square, even) -> VALID
    # 3*1 = 3 (not square, even) -> VALID
    # 3*2 = 6 (not square, even) -> VALID
    # 3*3 = 9 (square, odd) -> INVALID
    
    # Constraint: If u and v are connected, f[u]*f[v] must not be a square.
    # This means f[u] != f[v] if f[u] == f[v] is a square.
    # Since 1*1=1, 2*2=4, 3*3=9 are all squares, the constraint is simply:
    # If u and v are connected, f[u] != f[v].
    # This is exactly the 3-coloring problem.

    assignment = [0] * (n + 1)

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
