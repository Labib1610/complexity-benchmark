import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Two vectors (1, k1) and (1, k2) are linearly independent 
    # if and only if k1 != k2.
    # The problem reduces to finding a 3-coloring of the graph.
    # We need to minimize the lexicographical sequence (f1, f2, ..., fn).
    
    assignment = [0] * (n + 1)
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning 1, 2, 3 in order to minimize the sum
        for color in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if assignment[v] == color:
                    possible = False
                    break
            
            if possible:
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
