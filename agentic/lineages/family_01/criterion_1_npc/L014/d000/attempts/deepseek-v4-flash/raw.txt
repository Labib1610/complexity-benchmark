We can model this as a graph coloring problem with 3 colors.  
Since \( n \leq 40 \), we can use backtracking with pruning to find the lexicographically smallest valid assignment.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n

    def is_safe(node, color):
        for nei in adj[node]:
            if colors[nei] == color:
                return False
        return True

    def backtrack(node):
        if node == n:
            return True
        for c in range(1, 4):
            if is_safe(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(map(str, colors)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
