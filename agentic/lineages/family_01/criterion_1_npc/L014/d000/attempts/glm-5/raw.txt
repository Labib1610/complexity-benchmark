import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # We need to find the lexicographically smallest valid assignment.
    # Since n <= 40, we can use a recursive backtracking approach with pruning.
    # We assign domains one by one from 0 to n-1.
    # For each domain, we try tags 1, 2, 3 in increasing order.
    # We check if the tag is valid against already assigned neighbors.
    # If valid, we proceed. If we reach a dead end, we backtrack.
    
    assignment = [0] * n
    
    def backtrack(index):
        if index == n:
            return True
        
        for tag in range(1, 4):
            valid = True
            for neighbor in adj[index]:
                if neighbor < index and assignment[neighbor] == tag:
                    valid = False
                    break
            
            if valid:
                assignment[index] = tag
                if backtrack(index + 1):
                    return True
                # Backtrack: assignment[index] will be overwritten or remain 0
        
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
