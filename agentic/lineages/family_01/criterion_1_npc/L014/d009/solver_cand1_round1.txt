```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        w = int(input_data[idx+1])
        adj.append((u, w))
        idx += 2

    def check(u_idx, w_idx, x, y):
        # V_u = x^2 * (w * x + u * y) + y^2 * (x^2 + y)
        # V_w = y^2 * (u * y + w * x) + x^2 * (y^2 + x)
        # u and w are 1-indexed node labels
        vu = (x**2) * (w_idx * x + u_idx * y) + (y**2) * (x**2 + y)
        vw = (y**2) * (u_idx * y + w_idx * x) + (x**2) * (y**2 + x)
        return vu != vw

    assignment = [0] * (n + 1)

    def backtrack(node):
        if node > n:
            return True
        
        for speed in [1, 2, 3]:
            assignment[node] = speed
            possible = True
            for u, w in adj:
                if u <= node and w <= node:
                    if not check(u, w, assignment[u], assignment[w]):
                        possible = False
                        break
            if possible:
                if backtrack(node + 1):
                    return True
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
```
