We can model this as a constraint satisfaction problem.  
Since \( n \leq 40 \), we can use a backtracking search with pruning, trying assignments in lexicographic order.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    edges = []
    deg = [0] * (n + 1)
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        deg[u] += 1
        deg[v] += 1

    # Precompute edge list per vertex
    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    # Collaborative load table
    load = {}
    for a in range(1, 4):
        for b in range(a, 4):
            if a == 1 and b == 1:
                load[(a, b)] = 10
            elif a == 2 and b == 2:
                load[(a, b)] = 16
            elif a == 3 and b == 3:
                load[(a, b)] = 26
            elif a == 1 and b == 2:
                load[(a, b)] = 5
            elif a == 1 and b == 3:
                load[(a, b)] = 10
            elif a == 2 and b == 3:
                load[(a, b)] = 13

    # Precompute baseline risk contribution for each vertex and agency
    base_risk = [[0] * 4 for _ in range(n + 1)]
    for i in range(1, n + 1):
        for f in range(1, 4):
            base_risk[i][f] = (f * f) * deg[i]

    # Precompute collaborative load contribution for each edge and agency pair
    edge_load = []
    for u, v in edges:
        edge_load.append([[0] * 4 for _ in range(4)])
        for a in range(1, 4):
            for b in range(1, 4):
                if a <= b:
                    edge_load[-1][a][b] = load[(a, b)]
                else:
                    edge_load[-1][a][b] = load[(b, a)]

    assign = [0] * (n + 1)
    used = [False] * (n + 1)

    # We'll do DFS with pruning
    # We'll compute current total risk and load incrementally
    total_risk = 0
    total_load = 0

    # Precompute edge indices per vertex for quick updates
    edge_idx = [[] for _ in range(n + 1)]
    for idx, (u, v) in enumerate(edges):
        edge_idx[u].append((idx, v))
        edge_idx[v].append((idx, u))

    found = False

    def dfs(pos):
        nonlocal total_risk, total_load, found
        if found:
            return
        if pos == n + 1:
            if total_risk == total_load:
                found = True
            return

        # Try agencies in lexicographic order
        for f in range(1, 4):
            # Add risk
            total_risk += base_risk[pos][f]
            # Add loads for edges to already assigned neighbors
            added_load = 0
            for idx, nb in edge_idx[pos]:
                if assign[nb] != 0:
                    a = f
                    b = assign[nb]
                    if a > b:
                        a, b = b, a
                    added_load += edge_load[idx][a][b]
            total_load += added_load

            assign[pos] = f

            # Prune: if total_load > total_risk, no need to continue
            if total_load <= total_risk:
                dfs(pos + 1)
                if found:
                    return

            # Backtrack
            assign[pos] = 0
            total_load -= added_load
            total_risk -= base_risk[pos][f]

    dfs(1)

    if not found:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, assign[1:])))

if __name__ == "__main__":
    solve()
```
