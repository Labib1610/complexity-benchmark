```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        w = int(input_data[idx+1])
        adj.append((u, w))
        idx += 2

    def check(u_idx, w_idx, x, y):
        # u_idx, w_idx are 1-based
        # V_u = x * (u_idx + y)^3 + y * (w_idx + x)^3
        # V_w = y * (u_idx + x)^3 + x * (w_idx + y)^3
        vu = x * (u_idx + y)**3 + y * (w_idx + x)**3
        vw = y * (u_idx + x)**3 + x * (w_idx + y)**3
        return vu != vw

    assignment = [0] * (n + 1)

    def backtrack(node):
        if node > n:
            return True
        
        for speed in [1, 2, 3]:
            assignment[node] = speed
            possible = True
            for u, w in adj:
                if u == node and assignment[w] != 0:
                    if not check(u, w, assignment[u], assignment[w]):
                        possible = False
                        break
                elif w == node and assignment[u] != 0:
                    if not check(u, w, assignment[u], assignment[w]):
                        possible = False
                        break
            
            if possible:
                if backtrack(node + 1):
                    return True
            assignment[node] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
```
