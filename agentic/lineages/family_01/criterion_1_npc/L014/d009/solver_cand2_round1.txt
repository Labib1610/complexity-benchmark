import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    def get_load(fu, fv):
        return fu * fv * (fu + fv) + 8 * (fu // fv) * (fv // fu)

    assignment = [0] * n

    def check(f):
        total_system_risk = 0
        for i in range(n):
            sum_neighbors = sum(f[neighbor] for neighbor in adj[i])
            total_system_risk += (f[i] ** 2) * sum_neighbors
        
        total_collaborative_load = 0
        for u, v in edges:
            total_collaborative_load += get_load(f[u], f[v])
            
        return total_system_risk == total_collaborative_load

    def backtrack(idx):
        if idx == n:
            if check(assignment):
                return True
            return False
        
        for val in [1, 2, 3]:
            assignment[idx] = val
            if backtrack(idx + 1):
                return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
