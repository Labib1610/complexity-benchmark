import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # L(u, v) = (f_u - f_v)^4 - 5(f_u - f_v)^2 + 10
    # If f_u == f_v: L = 0 - 0 + 10 = 10
    # If |f_u - f_v| == 1: L = 1 - 5 + 10 = 6
    # If |f_u - f_v| == 2: L = 16 - 20 + 10 = 6
    # Total load <= 6m
    # Let x be number of edges with f_u == f_v, y be number of edges with f_u != f_v
    # x + y = m
    # Total load = 10x + 6y = 10x + 6(m - x) = 4x + 6m
    # Condition: 4x + 6m <= 6m => 4x <= 0 => x = 0
    # This means no two connected departments can have the same agency.
    # The problem is equivalent to finding a 3-coloring of the graph.
    # We need the lexicographically smallest assignment.

    assignment = [0] * n

    def is_valid(u, color):
        for v1, v2 in edges:
            if v1 == u:
                if assignment[v2] == color:
                    return False
            elif v2 == u:
                if assignment[v1] == color:
                    return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
