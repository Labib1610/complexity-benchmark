import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        edges.append((u, v))
        idx += 2

    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    C = [0] * (n + 1)
    for u, v in edges:
        C[u] += (u + v)
        C[v] += (u + v)

    K = sum(u + v for u, v in edges)
    target = 4 * K

    def get_risk(f_i, i):
        return C[i] * (5 * f_i**2 - f_i**4)

    def get_load(u, v, f_u, f_v):
        return (u + v) * f_u * f_v * (4 * f_u**2 - 6 * f_u * f_v + 4 * f_v**2 - 10)

    f = [0] * (n + 1)

    def backtrack(idx):
        if idx > n:
            total_risk = sum(get_risk(f[i], i) for i in range(1, n + 1))
            total_load = sum(get_load(u, v, f[u], f[v]) for u, v in edges)
            return (total_risk + total_load) == target

        for agency in [1, 2, 3]:
            f[idx] = agency
            if backtrack(idx + 1):
                return True
        return False

    if backtrack(1):
        print("YES")
        print(*(f[1:]))
    else:
        print("NO")

solve()
