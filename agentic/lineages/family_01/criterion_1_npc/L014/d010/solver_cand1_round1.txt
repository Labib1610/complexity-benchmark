import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        w = int(input_data[idx+1])
        edges.append((u, w))
        idx += 2

    C = [1, 2, 6, 7]

    def get_v(u, w, speed):
        c = C[(u + w) % 4]
        i = (u * w) % 17
        for _ in range(speed):
            i = (i * i + c) % 17
        return i

    # Precompute validity matrix
    # valid[u][w][speed_u][speed_w]
    # Since n is small (40), we can use backtracking
    assignment = [0] * (n + 1)

    def is_safe(node, speed):
        for u, w in edges:
            if u == node:
                other = w
                if assignment[other] != 0:
                    if get_v(u, other, speed) == get_v(u, other, assignment[other]):
                        return False
            elif w == node:
                other = u
                if assignment[other] != 0:
                    if get_v(other, w, assignment[other]) == get_v(other, w, speed):
                        return False
        return True

    def backtrack(node):
        if node > n:
            return True
        for speed in [1, 2, 3]:
            if is_safe(node, speed):
                assignment[node] = speed
                if backtrack(node + 1):
                    return True
                assignment[node] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
