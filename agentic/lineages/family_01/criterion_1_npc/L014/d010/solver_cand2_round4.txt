import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        w = int(input_data[idx+1])
        adj.append((u, w))
        idx += 2

    # V = 4x^2 + 3xy + 4y^2 + 11(u+w)(x+y)
    # V % 11 = (4x^2 + 3xy + 4y^2) % 11
    # We need (4x^2 + 3xy + 4y^2) % 11 != 0
    # Possible values for x, y in {1, 2, 3}:
    # (1,1): 4+3+4 = 11 = 0 (mod 11) -> Fail
    # (1,2): 4+6+16 = 26 = 4 (mod 11) -> OK
    # (1,3): 4+9+36 = 49 = 5 (mod 11) -> OK
    # (2,1): 4+6+16 = 26 = 4 (mod 11) -> OK
    # (2,2): 16+12+16 = 44 = 0 (mod 11) -> Fail
    # (2,3): 16+18+36 = 70 = 4 (mod 11) -> OK
    # (3,1): 4+9+36 = 49 = 5 (mod 11) -> OK
    # (3,2): 16+18+36 = 70 = 4 (mod 11) -> OK
    # (3,3): 36+27+36 = 99 = 0 (mod 11) -> Fail
    
    # So, x != y is the condition for validity.
    # This is equivalent to 3-coloring the graph.
    
    assignment = [0] * (n + 1)
    
    def is_valid(node, color):
        for u, w in adj:
            if u == node:
                if assignment[w] == color:
                    return False
            elif w == node:
                if assignment[u] == color:
                    return False
        return True

    def backtrack(node):
        if node > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(node, color):
                assignment[node] = color
                if backtrack(node + 1):
                    return True
                assignment[node] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
