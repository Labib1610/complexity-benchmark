import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Resonance condition: 4x + y = 0 mod 5 or 4y + x = 0 mod 5
    # 4x + y = 0 mod 5  => -x + y = 0 mod 5 => x = y mod 5
    # Since x, y in {1, 2, 3}, x = y mod 5 implies x = y.
    # 4y + x = 0 mod 5  => -y + x = 0 mod 5 => x = y mod 5
    # So, resonance occurs if x = y.
    # Wait, let's recheck:
    # If x=1, y=1: 4(1)+1 = 5 (multiple of 5) -> Fail
    # If x=1, y=2: 4(1)+2 = 6, 4(2)+1 = 9 -> OK
    # If x=1, y=3: 4(1)+3 = 7, 4(3)+1 = 13 -> OK
    # If x=2, y=2: 4(2)+2 = 10 (multiple of 5) -> Fail
    # If x=2, y=3: 4(2)+3 = 11, 4(3)+2 = 14 -> OK
    # If x=3, y=3: 4(3)+3 = 15 (multiple of 5) -> Fail
    # Conclusion: x and y must be different.
    # This is exactly the graph coloring problem with 3 colors.

    colors = [0] * (n + 1)

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
