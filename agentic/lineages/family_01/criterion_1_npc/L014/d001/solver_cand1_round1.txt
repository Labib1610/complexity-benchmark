```python
import sys

# The problem asks for a 3-coloring of a graph with n vertices and m edges.
# We need to find the lexicographically smallest coloring.
# Since n can be up to 40 (implied by 400kDa/10kDa), a backtracking approach
# with pruning is efficient enough for 3-coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    # Increase recursion depth for deep backtracking trees
    sys.setrecursionlimit(2000)
    solve()
```
