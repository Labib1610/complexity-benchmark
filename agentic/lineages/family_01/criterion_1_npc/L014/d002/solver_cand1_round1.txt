```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    # Sort nodes by degree descending to optimize backtracking
    nodes = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    
    def backtrack(idx_in_nodes):
        if idx_in_nodes == n:
            return True
        
        u = nodes[idx_in_nodes]
        
        # Find forbidden colors
        forbidden = set()
        for v in adj[u]:
            if colors[v] != 0:
                forbidden.add(colors[v])
        
        for c in [1, 2, 3]:
            if c not in forbidden:
                colors[u] = c
                if backtrack(idx_in_nodes + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
