```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in range(1, 4):
            possible = True
            for v in adj[u]:
                if colors[v] == c:
                    possible = False
                    break
            
            if possible:
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # Optimization: Sort nodes by degree descending to fail faster
    # However, for n=40, simple backtracking with pruning is usually fine
    # if the graph is dense.
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
