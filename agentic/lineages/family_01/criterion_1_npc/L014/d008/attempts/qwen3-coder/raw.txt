from itertools import product

def compute_collaborative_load(fu, fv):
    diff = fu - fv
    return fu**2 + fv**2 + 2*(diff**4) - 10*(diff**2) + 8

def is_compliant(n, m, edges, assignment):
    # Calculate total system risk
    # For each department, baseline risk = f_i^2 * degree_i
    degree = [0] * (n + 1)
    for u, v in edges:
        degree[u] += 1
        degree[v] += 1
    
    total_system_risk = sum(assignment[i]**2 * degree[i] for i in range(1, n + 1))
    
    # Calculate total collaborative load
    total_collaborative_load = 0
    for u, v in edges:
        fu, fv = assignment[u], assignment[v]
        total_collaborative_load += compute_collaborative_load(fu, fv)
    
    return total_system_risk == total_collaborative_load

def solve():
    n, m = map(int, input().split())
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
    
    # Try all possible assignments in lexicographical order
    for assignment_tuple in product([1, 2, 3], repeat=n):
        assignment = [0] + list(assignment_tuple)  # 1-indexed
        if is_compliant(n, m, edges, assignment):
            print("YES")
            print(' '.join(map(str, assignment[1:])))
            return
    
    print("NO")

solve()
