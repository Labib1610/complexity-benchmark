```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        # Ensure u < v for the conflict rule
        if u > v:
            u, v = v, u
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    def backtrack(task_idx):
        if task_idx > n:
            return True
        
        # Find which threads are forbidden by previous tasks
        forbidden = set()
        for neighbor in adj[task_idx]:
            forbidden.add(assignment[neighbor])
            
        for thread in [1, 2, 3]:
            if thread not in forbidden:
                assignment[task_idx] = thread
                if backtrack(task_idx + 1):
                    return True
                assignment[task_idx] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
