import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # W(a, b) where a, b in {0, 1, 2}
    # W = -18(a^2 + b^2) - 3ab(1 + a + b) + 54(a+b) - 3a^2 b^2 - 24
    def get_w(a, b):
        return -18 * (a**2 + b**2) - 3 * a * b * (1 + a + b) + 54 * (a + b) - 3 * (a**2) * (b**2) - 24

    # Precompute valid pairs (a, b) such that W(a, b) = 12
    valid_pairs = []
    for a in range(3):
        for b in range(3):
            if get_w(a, b) == 12:
                valid_pairs.append((a + 1, b + 1))

    assignment = [0] * (n + 1)

    def backtrack(u):
        if u > n:
            return True
        
        for agency in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if assignment[v] != 0:
                    # Check if (agency, assignment[v]) is a valid pair
                    if (agency, assignment[v]) not in valid_pairs:
                        possible = False
                        break
            
            if possible:
                assignment[u] = agency
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
