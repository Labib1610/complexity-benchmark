import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        w = int(input_data[idx+1])
        edges.append((u, w))
        idx += 2

    def check(freqs):
        for u, w in edges:
            x = freqs[u-1]
            y = freqs[w-1]
            # V_u = x * (u * (x^2 + y^2) - 2 * w * y^2)
            # V_w = y * (w * (x^2 + y^2) - 2 * u * x^2)
            # V_u + V_w = x*u*(x^2+y^2) - 2*x*w*y^2 + y*w*(x^2+y^2) - 2*y*u*x^2
            #           = (x*u + y*w)*(x^2+y^2) - 2*x*w*y^2 - 2*y*u*x^2
            vu = x * (u * (x*x + y*y) - 2 * w * y*y)
            vw = y * (w * (x*x + y*y) - 2 * u * x*x)
            if vu + vw == 0:
                return False
        return True

    # Since n is small (up to 40), we can try a randomized approach or backtracking.
    # Given the constraints and the nature of the problem, a simple randomized search
    # is often effective for constraint satisfaction problems of this size.
    import random
    
    for _ in range(20000):
        freqs = [random.randint(1, 3) for _ in range(n)]
        
        # Local search / hill climbing
        for _ in range(100):
            if check(freqs):
                print("YES")
                print(*(freqs))
                return
            
            # Find a bad edge
            bad_edges = []
            for u, w in edges:
                x, y = freqs[u-1], freqs[w-1]
                if x * (u * (x*x + y*y) - 2 * w * y*y) + y * (w * (x*x + y*y) - 2 * u * x*x) == 0:
                    bad_edges.append((u, w))
            
            if not bad_edges:
                break
                
            # Flip a random node in a bad edge
            u, w = random.choice(bad_edges)
            node = random.choice([u, w])
            freqs[node-1] = random.randint(1, 3)
            
    print("NO")

solve()
