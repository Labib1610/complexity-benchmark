import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        w = int(input_data[idx+1])
        edges.append((u, w))
        idx += 2

    def check(assignment):
        for u, w in edges:
            x = assignment[u-1]
            y = assignment[w-1]
            vu = x * ((u + w) * x - 2 * u * y)
            vw = y * ((u + w) * y - 2 * w * x)
            if vu + vw == 0:
                return False
        return True

    # Since n is small (up to 40), we can try a randomized approach or backtracking.
    # Given the constraints and the nature of the problem, a simple randomized search
    # or a greedy approach with backtracking is likely to find a solution if one exists.
    
    import random
    
    # Try random assignments
    for _ in range(20000):
        assignment = [random.randint(1, 3) for _ in range(n)]
        if check(assignment):
            print("YES")
            print(*(assignment))
            return

    # If random fails, try a simple backtracking
    assignment = [0] * n
    def backtrack(node):
        if node == n:
            return True
        for speed in [1, 2, 3]:
            assignment[node] = speed
            valid = True
            for u, w in edges:
                if u-1 <= node and w-1 <= node:
                    x = assignment[u-1]
                    y = assignment[w-1]
                    vu = x * ((u + w) * x - 2 * u * y)
                    vw = y * ((u + w) * y - 2 * w * x)
                    if vu + vw == 0:
                        valid = False
                        break
            if valid and backtrack(node + 1):
                return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
