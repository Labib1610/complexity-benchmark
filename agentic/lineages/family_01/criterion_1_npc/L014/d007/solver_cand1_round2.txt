```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # score(diff) = 4*diff^4 - 20*diff^2 + 12
    # diff = 0: 4(0) - 20(0) + 12 = 12
    # diff = 1: 4(1) - 20(1) + 12 = -4
    # diff = 2: 4(16) - 20(4) + 12 = 64 - 80 + 12 = -4
    # Total sum = -4m means every link must have diff != 0
    # i.e., c_u != c_v. This is exactly the 3-coloring problem.

    assignment = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if assignment[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                assignment[u] = c
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
```
