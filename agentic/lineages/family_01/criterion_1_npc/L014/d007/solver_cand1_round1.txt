import sys

def solve():
    # The characteristic polynomial of M = [[x, u], [w, y]] is:
    # P(t) = det(tI - M) = det([[t-x, -u], [-w, t-y]])
    # P(t) = (t-x)(t-y) - uw = t^2 - (x+y)t + xy - uw
    # Resonance condition: P((x+y)/2) = -uw
    # ((x+y)/2)^2 - (x+y)((x+y)/2) + xy - uw = -uw
    # (x+y)^2 / 4 - (x+y)^2 / 2 + xy = 0
    # -(x+y)^2 / 4 + xy = 0
    # -(x^2 + 2xy + y^2) + 4xy = 0
    # -x^2 + 2xy - y^2 = 0
    # -(x-y)^2 = 0
    # x = y
    # So, the condition for resonance is x == y.
    # To avoid resonance, we must have x != y for all connected nodes.
    # This is equivalent to finding a 3-coloring of the graph.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
