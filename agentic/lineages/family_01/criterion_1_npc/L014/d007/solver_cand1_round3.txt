import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Interaction value: 2(f_u - f_v)^4 - 10(f_u - f_v)^2 + 4
    # Let d = |f_u - f_v|. d can be 0, 1, 2.
    # If d=0: 2(0)^4 - 10(0)^2 + 4 = 4
    # If d=1: 2(1)^4 - 10(1)^2 + 4 = 2 - 10 + 4 = -4
    # If d=2: 2(16) - 10(4) + 4 = 32 - 40 + 4 = -4
    # We want sum = -4m.
    # Since each edge contributes either 4 or -4, and we want sum = -4m,
    # every edge must contribute -4.
    # This means |f_u - f_v| must be 1 or 2.
    # |f_u - f_v| != 0 means f_u != f_v.
    # This is equivalent to saying the graph is 3-colorable.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    assignment = [0] * n

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
