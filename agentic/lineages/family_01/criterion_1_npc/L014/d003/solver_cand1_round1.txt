import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition: sum(u, v) != 2*u and sum(u, v) != 2*v
    # This simplifies to: u + v != 2*u => v != u
    # AND u + v != 2*v => u != v
    # So the condition is simply: u != v for all connected components.
    # This is equivalent to 2-coloring the graph.
    # However, we have 3 colors (1, 2, 3).
    # If the graph is bipartite, we can use colors 1 and 2.
    # If it's not bipartite, we might need color 3.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # Heuristic: sort nodes by degree descending to fail fast
    nodes = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes for the backtracking
    mapping = {node: i for i, node in enumerate(nodes)}
    new_adj = [[] for _ in range(n)]
    for u in range(n):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    res_colors = [0] * n
    
    def backtrack_optimized(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            possible = True
            for v in new_adj[u]:
                if res_colors[v] == c:
                    possible = False
                    break
            if possible:
                res_colors[u] = c
                if backtrack_optimized(u + 1):
                    return True
                res_colors[u] = 0
        return False

    if backtrack_optimized(0):
        print("YES")
        final_ans = [0] * n
        for i in range(n):
            final_ans[nodes[i]] = res_colors[i]
        print(*(final_ans))
    else:
        print("NO")

solve()
