import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # We need to find an assignment f such that
    # sum_{(u,v) in E} W_{f_u} . W_{f_v} = -4m
    # Since W_i . W_j = 12 if i=j, and -4 if i!=j,
    # Let k be the number of edges with endpoints having same agency.
    # Total sum = 12k - 4(m-k) = 16k - 4m.
    # We need 16k - 4m = -4m => 16k = 0 => k = 0.
    # So we need a valid 3-coloring of the graph.
    # We want the lexicographically smallest assignment.
    
    # Greedy assignment: for each vertex u from 0 to n-1,
    # assign the smallest color in {1, 2, 3} that is not used by any processed neighbor.
    
    result = [0] * n
    possible = True
    
    for u in range(n):
        # Check colors used by neighbors with index < u
        used_colors = set()
        for v in adj[u]:
            if v < u:
                used_colors.add(result[v])
        
        # Try colors 1, 2, 3 in order
        assigned = False
        for color in range(1, 4):
            if color not in used_colors:
                result[u] = color
                assigned = True
                break
        
        if not assigned:
            possible = False
            break
            
    if possible:
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
