import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # W1 = (2, 2, 2), W2 = (-2, 2, -2), W3 = (2, -2, -2)
    # Dot products:
    # W1.W1 = 4+4+4 = 12
    # W2.W2 = 4+4+4 = 12
    # W3.W3 = 4+4+4 = 12
    # W1.W2 = -4+4-4 = -4
    # W1.W3 = 4-4-4 = -4
    # W2.W3 = -4-4+4 = -4
    # The condition is sum(W_fu . W_fv) = -4m.
    # Since W_i . W_i = 12 and W_i . W_j = -4 for i != j,
    # let x_i be 1 if f_u == f_v, 0 otherwise.
    # Sum = sum_{edges} (12 * x_i - 4 * (1 - x_i)) = sum (16 * x_i - 4) = 16 * sum(x_i) - 4m.
    # We need 16 * sum(x_i) - 4m = -4m, which means sum(x_i) = 0.
    # This implies x_i must be 0 for all edges, meaning f_u != f_v for all edges.
    # The problem is equivalent to finding a 3-coloring of the graph.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
