import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Clock speeds:
    # 1: swap(1, 2) -> permutation (2, 1, 3)
    # 2: swap(2, 3) -> permutation (1, 3, 2)
    # 3: swap(1, 3) -> permutation (3, 2, 1)
    # Composition of two swaps:
    # 1 then 1: (2, 1, 3) * (2, 1, 3) = (1, 2, 3) -> Identity
    # 2 then 2: (1, 3, 2) * (1, 3, 2) = (1, 2, 3) -> Identity
    # 3 then 3: (3, 2, 1) * (3, 2, 1) = (1, 2, 3) -> Identity
    # 1 then 2: (2, 1, 3) * (1, 3, 2) = (2, 3, 1) -> Not Identity
    # 2 then 1: (1, 3, 2) * (2, 1, 3) = (3, 1, 2) -> Not Identity
    # 1 then 3: (2, 1, 3) * (3, 2, 1) = (3, 2, 1) * (2, 1, 3) = (3, 1, 2) -> Not Identity
    # 3 then 1: (3, 2, 1) * (2, 1, 3) = (2, 3, 1) -> Not Identity
    # 2 then 3: (1, 3, 2) * (3, 2, 1) = (3, 1, 2) -> Not Identity
    # 3 then 2: (3, 2, 1) * (1, 3, 2) = (2, 3, 1) -> Not Identity
    
    # Condition: f(u) != f(v)
    # This is equivalent to 2-coloring the graph.
    # If the graph is bipartite, we can assign 1 to one set and 2 to the other.
    
    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    res = []
    for c in color:
        if c == 0:
            res.append(1)
        else:
            res.append(2)
    print(*(res))

solve()
