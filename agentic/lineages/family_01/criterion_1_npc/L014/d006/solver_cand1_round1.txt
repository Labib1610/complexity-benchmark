import sys

def solve():
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m = map(int, line1)
    except ValueError:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)

    # Condition: (x^3 + y^3 + x*y) % 10 == 1
    # Possible values for x, y in {1, 2, 3}:
    # (1,1): 1+1+1 = 3 (No)
    # (1,2): 1+8+2 = 11 (Yes)
    # (1,3): 1+27+3 = 31 (Yes)
    # (2,1): 8+1+2 = 11 (Yes)
    # (2,2): 8+8+4 = 20 (No)
    # (2,3): 8+27+6 = 41 (Yes)
    # (3,1): 27+1+3 = 31 (Yes)
    # (3,2): 27+8+6 = 41 (Yes)
    # (3,3): 27+27+9 = 63 (No)
    
    # Valid pairs (x, y):
    # (1,2), (1,3), (2,1), (2,3), (3,1), (3,2)
    # Invalid pairs: (1,1), (2,2), (3,3)
    # This means x != y for all connected nodes.
    # This is equivalent to 3-coloring the graph.

    assignment = [0] * n

    def is_valid(node, color):
        for neighbor in adj[node]:
            if assignment[neighbor] == color:
                return False
        return True

    def backtrack(node):
        if node == n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(node, color):
                assignment[node] = color
                if backtrack(node + 1):
                    return True
                assignment[node] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
