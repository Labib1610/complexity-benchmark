import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        edges.append((u, v))
        idx += 2

    def get_det(su, sv, u, v):
        # Matrix:
        # [[su, sv, su^2 + u + v + 10],
        #  [1,  1,  sv],
        #  [sv, su, su*sv]]
        # Det = su(1*su*sv - sv*su) - sv(1*su*sv - sv*sv) + (su^2+u+v+10)(1*su - 1*sv)
        # Det = su(0) - sv(su*sv - sv^2) + (su^2+u+v+10)(su-sv)
        # Det = -sv*sv*(su-sv) + (su^2+u+v+10)(su-sv)
        # Det = (su-sv) * (su^2 + u + v + 10 - sv^2)
        return (su - sv) * (su**2 + u + v + 10 - sv**2)

    # Precompute valid pairs (su, sv) for each edge (u, v)
    # Since n is small (40), we can use backtracking with pruning
    # The signature V_n = 4*V_{n-1} + s_n is equivalent to interpreting
    # the sequence s_1, s_2, ..., s_n as digits in base 4.
    # To minimize V_n, we want s_1 to be as small as possible, then s_2, etc.
    
    assignment = [0] * (n + 1)
    
    def is_valid(curr_node, val):
        for u, v in edges:
            if u == curr_node and assignment[v] != 0:
                if get_det(val, assignment[v], u, v) == 0:
                    return False
            elif v == curr_node and assignment[u] != 0:
                if get_det(assignment[u], val, u, v) == 0:
                    return False
        return True

    def backtrack(curr_node):
        if curr_node > n:
            return True
        
        for val in [1, 2, 3]:
            if is_valid(curr_node, val):
                assignment[curr_node] = val
                if backtrack(curr_node + 1):
                    return True
                assignment[curr_node] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
