import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    def get_det(su, sv):
        # Matrix:
        # su+1, su^2-1, 3
        # sv+1, sv^2-1, 3
        # su+sv+2, su^2+sv^2-2, 7
        # Det = (su+1)((sv^2-1)*7 - 3(su^2+sv^2-2)) - (su^2-1)((sv+1)*7 - 3(su+sv+2)) + 3((sv+1)(su^2+sv^2-2) - (sv^2-1)(su+sv+2))
        
        def det3x3(a, b, c, d, e, f, g, h, i):
            return a*(e*i - f*h) - b*(d*i - f*g) + c*(d*h - e*g)
        
        return det3x3(su+1, su**2-1, 3, sv+1, sv**2-1, 3, su+sv+2, su**2+sv**2-2, 7)

    # Precompute validity for all pairs (su, sv)
    # A link is stable if det(su, sv) > 0 OR det(sv, su) > 0
    valid_pairs = [[False for _ in range(4)] for _ in range(4)]
    for su in range(1, 4):
        for sv in range(1, 4):
            if get_det(su, sv) > 0 or get_det(sv, su) > 0:
                valid_pairs[su][sv] = True

    # Backtracking to find the lexicographically smallest sequence
    # V_i = 4 * V_{i-1} + s_i. To minimize V_n, we want s_1 smallest, then s_2, etc.
    assignment = [0] * n
    
    def is_consistent(idx, val):
        for u, v in edges:
            if u == idx:
                if assignment[v] != 0 and not valid_pairs[val][assignment[v]]:
                    return False
            elif v == idx:
                if assignment[u] != 0 and not valid_pairs[val][assignment[u]]:
                    return False
        return True

    def backtrack(idx):
        if idx == n:
            return True
        for val in range(1, 4):
            if is_consistent(idx, val):
                assignment[idx] = val
                if backtrack(idx + 1):
                    return True
                assignment[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
