import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Resonance signature R = 2^s_u + 2^s_v
    # s_u, s_v in {1, 2, 3}
    # Possible R:
    # 1+1: 2^1+2^1 = 4 = 2^2 (Collapse)
    # 1+2: 2^1+2^2 = 6 (Stable)
    # 1+3: 2^1+2^3 = 10 (Stable)
    # 2+2: 2^2+2^2 = 8 = 2^3 (Collapse)
    # 2+3: 2^2+2^3 = 12 (Stable)
    # 3+3: 2^3+2^3 = 16 = 2^4 (Collapse)
    
    # Forbidden pairs: (1,1), (2,2), (3,3)
    # Allowed pairs: (1,2), (1,3), (2,3)
    # This means no two connected nodes can have the same multiplier.
    # This is equivalent to a 3-coloring problem.
    
    assignment = [0] * n
    
    def is_valid(u, val):
        for v in adj[u]:
            if assignment[v] == val:
                return False
        return True

    # We need to minimize sum_{i=1}^n s_i * 4^{n-i}
    # This is equivalent to lexicographically smallest sequence (s_1, s_2, ..., s_n)
    
    def backtrack(u):
        if u == n:
            return True
        for val in [1, 2, 3]:
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
