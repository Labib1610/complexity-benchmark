import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Geometric mean sqrt(s_u * s_v) is irrational iff s_u * s_v is not a perfect square.
    # Possible values for s_u, s_v are {1, 2, 3}.
    # Products:
    # 1*1=1 (sq), 1*2=2, 1*3=3
    # 2*1=2, 2*2=4 (sq), 2*3=6
    # 3*1=3, 3*2=6, 3*3=9 (sq)
    # So, s_u * s_v must not be 1, 4, or 9.
    # This means:
    # If s_u = 1, s_v cannot be 1.
    # If s_u = 2, s_v cannot be 2.
    # If s_u = 3, s_v cannot be 3.
    # Essentially, s_u != s_v.
    # This is equivalent to saying the graph must be 3-colorable.

    best_s = None
    min_r = float('inf')

    # Since n <= 40, we can't do 3^40. But wait, the constraint is 3-coloring.
    # 3-coloring is NP-hard, but n=40 is small enough for backtracking with pruning.
    
    assignment = [0] * n

    def is_valid(u, val):
        for v in adj[u]:
            if assignment[v] == val:
                return False
        return True

    def backtrack(u):
        nonlocal best_s, min_r
        if u == n:
            # Calculate R = sum(s_i / 4^(i+1))
            # To minimize R, we want smaller values at smaller indices.
            # Since we iterate 1, 2, 3, the first valid solution found 
            # by lexicographical order of (s_1, s_2, ..., s_n) 
            # will minimize the sum.
            return list(assignment)
        
        for val in [1, 2, 3]:
            if is_valid(u, val):
                assignment[u] = val
                res = backtrack(u + 1)
                if res:
                    return res
                assignment[u] = 0
        return None

    result = backtrack(0)
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
