import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    neighbors = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            print("NO")
            return
        neighbors[u].append(v)
        neighbors[v].append(u)

    color = [-1] * n
    avail = [7] * n

    def assign(v, c, colored, changed):
        color[v] = c
        colored.append(v)
        bit = 1 << c
        for u in neighbors[v]:
            if color[u] == -1:
                old = avail[u]
                if old & bit:
                    new = old ^ bit
                    avail[u] = new
                    changed.append((u, old))
                    if new == 0:
                        return False
        return True

    def undo(colored, changed):
        for u, old in changed:
            avail[u] = old
        for v in colored:
            color[v] = -1

    sys.setrecursionlimit(10000)

    def dfs(pos):
        while pos < n and color[pos] != -1:
            pos += 1
        if pos == n:
            return True
        mask = avail[pos]
        for c in (0, 1, 2):
            if mask & (1 << c):
                colored = []
                changed = []
                if assign(pos, c, colored, changed):
                    ok = True
                    while True:
                        sv = -1
                        for u in range(n):
                            if color[u] == -1:
                                am = avail[u]
                                if am and (am & (am - 1)) == 0:
                                    sv = u
                                    break
                        if sv == -1:
                            break
                        sc = avail[sv].bit_length() - 1
                        if not assign(sv, sc, colored, changed):
                            ok = False
                            break
                    if ok and dfs(pos + 1):
                        return True
                undo(colored, changed)
        return False

    if dfs(0):
        print("YES")
        print(' '.join(str(color[i] + 1) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
