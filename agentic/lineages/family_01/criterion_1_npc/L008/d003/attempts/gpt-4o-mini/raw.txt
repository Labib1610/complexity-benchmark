def can_assign_pressure(n, m, conflicts):
    from itertools import product

    # Create a conflict adjacency matrix
    conflict_matrix = [[False] * n for _ in range(n)]
    for u, v in conflicts:
        conflict_matrix[u - 1][v - 1] = True
        conflict_matrix[v - 1][u - 1] = True

    # Try all combinations of assignments
    for pressures in product(range(1, 4), repeat=n):
        valid = True
        for u in range(n):
            for v in range(n):
                if u != v and conflict_matrix[u][v]:
                    if pressures[u] == pressures[v]:
                        valid = False
                        break
            if not valid:
                break
        
        if valid:
            return pressures

    return None

import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
conflicts = [tuple(map(int, line.split())) for line in data[1:m+1]]

assignment = can_assign_pressure(n, m, conflicts)

if assignment is None:
    print("NO")
else:
    print("YES")
    print(" ".join(map(str, assignment)))
