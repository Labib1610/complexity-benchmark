```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning phase 1, 2, or 3
        for phase in range(1, 4):
            can_assign = True
            for neighbor in adj[u]:
                if assignment[neighbor] == phase:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[u] = phase
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        
        return False

    # Optimization: Sort nodes by degree descending (heuristic)
    # But since n is small (40), simple backtracking is sufficient.
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
