import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2

    # The recurrence X_k = (u+v)X_{k-1} - uvX_{k-2} has characteristic equation
    # r^2 - (u+v)r + uv = 0, which factors to (r-u)(r-v) = 0.
    # Thus X_k = A*u^k + B*v^k.
    # For X_0 = s_u + s_v, X_1 = s_u*u + s_v*v:
    # A + B = s_u + s_v
    # A*u + B*v = s_u*u + s_v*v
    # Solving this: A = s_u, B = s_v.
    # So X_k = s_u * u^k + s_v * v^k.
    # Similarly, Y_k = s_v * u^k + s_u * v^k.
    # The ratio X_k / Y_k = (s_u * u^k + s_v * v^k) / (s_v * u^k + s_u * v^k).
    # Assume u > v. As k -> inf, ratio -> s_u / s_v.
    # If u == v, ratio = (s_u + s_v) / (s_u + s_v) = 1.
    # But u, v are node indices (1 to n), so u != v is guaranteed.
    # The limit is 1 iff s_u / s_v == 1, i.e., s_u == s_v.
    # So the condition is: for every link (u, v), s_u != s_v.
    # This is equivalent to saying the graph must be 3-colorable.
    
    # Since n <= 40, we can use backtracking to find the lexicographically smallest
    # configuration (minimizing sum s_i * 2^(2n-2i)).
    # This is equivalent to minimizing s_1, then s_2, ..., then s_n.
    
    colors = [0] * n
    
    def is_valid(node, c):
        for u, v in adj:
            if u == node and colors[v] == c: return False
            if v == node and colors[u] == c: return False
        return True

    def backtrack(node):
        if node == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
