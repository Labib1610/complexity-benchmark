import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        edges.append((u, v))
        idx += 2

    # Condition: M_{u->v} != M_{v->u}
    # s_u * v + s_v * u != s_v * v + s_u * u
    # s_u * v - s_u * u != s_v * v - s_v * u
    # s_u * (v - u) != s_v * (v - u)
    # Since u != v, (v - u) != 0, so s_u != s_v.
    # The condition simplifies to s_u != s_v for all linked pairs (u, v).
    # This is equivalent to finding a 3-coloring of the graph.
    # We need to minimize V_n = sum_{i=1}^n s_i * 4^{n-i}.
    # This is equivalent to lexicographically smallest (s_1, s_2, ..., s_n).

    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    s = [0] * (n + 1)

    def backtrack(u):
        if u > n:
            return True
        
        # Try multipliers 1, 2, 3 in order to minimize lexicographically
        for val in [1, 2, 3]:
            possible = True
            for neighbor in adj[u]:
                if s[neighbor] == val:
                    possible = False
                    break
            
            if possible:
                s[u] = val
                if backtrack(u + 1):
                    return True
                s[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(s[1:]))
    else:
        print("NO")

solve()
