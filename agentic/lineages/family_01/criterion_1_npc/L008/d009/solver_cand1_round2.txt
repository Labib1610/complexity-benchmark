import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # P(s_u, s_v) = (s_u^2 + s_u*s_v + s_v^2) % 3
    # s in {1, 2, 3} -> s % 3 in {1, 2, 0}
    # Let x = s % 3. x in {0, 1, 2}
    # P = (x_u^2 + x_u*x_v + x_v^2) % 3
    # If x=0, P = x_v^2 % 3. If x_v=1, P=1. If x_v=2, P=4%3=1.
    # If x=1, P = (1 + x_v + x_v^2) % 3. If x_v=0, P=1. If x_v=1, P=3%3=0. If x_v=2, P=(1+2+4)%3=1.
    # If x=2, P = (4 + 2*x_v + x_v^2) % 3 = (1 + 2*x_v + x_v^2) % 3. If x_v=0, P=1. If x_v=1, P=(1+2+1)%3=1. If x_v=2, P=(1+4+4)%3=0.
    
    # Mapping: 1->1, 2->2, 3->0
    # P(x_u, x_v):
    # x_u=0: x_v=0->0, x_v=1->1, x_v=2->1
    # x_u=1: x_v=0->1, x_v=1->0, x_v=2->1
    # x_u=2: x_v=0->1, x_v=1->1, x_v=2->0
    # This is P = 1 if x_u != x_v else 0.
    # G = sum(P) = sum(1 if x_u != x_v else 0) = m
    # This means exactly m links must have x_u != x_v.
    # But there are m links total. So all links must have x_u != x_v.
    # This is equivalent to saying the graph is 3-colorable with colors {0, 1, 2}.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [-1] * n
    
    def get_s(x):
        return 3 if x == 0 else x

    # We need to find the lexicographically smallest sequence (s1, s2, ..., sn)
    # which corresponds to the smallest V_n.
    # V_n = s_n + 4*s_{n-1} + 16*s_{n-2} + ...
    # To minimize V_n, we want s1 to be as small as possible, then s2, etc.
    # s values are {1, 2, 3}. x values are {1, 2, 0}.
    # Order of preference for x: 1 (s=1), 2 (s=2), 0 (s=3).
    
    best_x = [None] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try x in order 1, 2, 0
        for x in [1, 2, 0]:
            possible = True
            for v in adj[u]:
                if colors[v] != -1 and colors[v] == x:
                    possible = False
                    break
            if possible:
                colors[u] = x
                if backtrack(u + 1):
                    return True
                colors[u] = -1
        return False

    if backtrack(0):
        print("YES")
        print(*(get_s(c) for c in colors))
    else:
        print("NO")

solve()
