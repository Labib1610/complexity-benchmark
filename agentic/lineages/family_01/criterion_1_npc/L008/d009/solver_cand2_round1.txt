import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # I(u, v) = (s_u - s_v)^4 - 5(s_u - s_v)^2 + 4
    # Let d = s_u - s_v. Possible values for s_i are {1, 2, 3}.
    # d can be {-2, -1, 0, 1, 2}.
    # If d = 0: 0 - 0 + 4 = 4
    # If d = 1 or -1: 1 - 5 + 4 = 0
    # If d = 2 or -2: 16 - 20 + 4 = 0
    # So I(u, v) = 4 if s_u == s_v, else 0.
    # Total interference = 4 * (number of edges where s_u == s_v).
    # For total interference to be 0, we need s_u != s_v for all edges.
    # This is equivalent to checking if the graph is 3-colorable.
    
    # Since n <= 40, we can use backtracking with pruning to find the lexicographically
    # smallest sequence (which minimizes the footprint F_n).
    # F_n = 4^n * s_1 + 4^{n-1} * s_2 + ... + 4^1 * s_n.
    # To minimize F_n, we want s_1 to be as small as possible, then s_2, etc.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        for s in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if assignment[v] == s:
                    possible = False
                    break
            if possible:
                assignment[u] = s
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
