import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # Precompute valid pairs (su, sv)
    # Matrix C:
    # Row 1: 1, su+2, su^2 + 4sv
    # Row 2: 1, sv, sv^2
    # Row 3: 1, sv+1, sv^2 + 2sv
    # det = (sv - su - 2)(sv - su)(sv - su + 2)
    valid_pairs = set()
    vals = [1, 2, 3]
    for su in vals:
        for sv in vals:
            d1 = sv - su - 2
            d2 = sv - su
            d3 = sv - su + 2
            if d1 != 0 and d2 != 0 and d3 != 0:
                valid_pairs.add((su, sv))
    
    assignment = [0] * n
    
    def dfs(u):
        for v in adj[u]:
            if assignment[v] == 0:
                for sv in vals:
                    if (assignment[u], sv) in valid_pairs and (sv, assignment[u]) in valid_pairs:
                        assignment[v] = sv
                        if not dfs(v):
                            assignment[v] = 0
                        else:
                            break
                if assignment[v] == 0:
                    return False
        return True
    
    # Try assignments starting with smallest signature
    # Since signature is V_n = sum_{i=1 to n} s_i * 4^{n-i},
    # minimizing signature is equivalent to lexicographically minimizing (s_1, s_2, ..., s_n)
    # We iterate through all possible assignments in lexicographical order
    
    # Generate all possible assignments in lexicographical order
    # Total 3^n possibilities, n <= 40, so too many to iterate directly
    # However, the graph structure constraints reduce possibilities
    # We use backtracking with pruning
    
    # Since we need lexicographically smallest, we try values 1, 2, 3 for each position
    # and backtrack if constraints are violated
    
    # We'll use iterative approach with stack to avoid recursion depth issues
    
    # But first, let's check if graph is bipartite (2-colorable) with valid_pairs
    # Actually, valid_pairs allow all pairs except (1,1), (2,2), (3,3), (1,3), (3,1)
    # So valid pairs are: (1,2), (2,1), (2,3), (3,2)
    # This is essentially a 2-coloring problem where adjacent nodes must have different values
    # and values must be consecutive (1-2 or 2-3)
    
    # So the problem reduces to: can we assign values such that for every edge (u,v),
    # |s_u - s_v| = 1?
    
    # This is a graph coloring problem with constraints
    # We can use BFS/DFS to check components
    
    # For each component, we can try starting with 1 or 2 or 3
    # But since we need lexicographically smallest, we prefer smaller values for earlier nodes
    
    # Let's implement a backtracking search with pruning
    
    # We'll process nodes in order 0 to n-1
    # For each node, try values 1, 2, 3 in order, check constraints with already assigned neighbors
    
    best_assignment = None
    
    def backtrack(pos):
        nonlocal best_assignment
        if best_assignment is not None:
            return True
        if pos == n:
            best_assignment = assignment[:]
            return True
        for val in vals:
            # Check constraints with already assigned neighbors
            valid = True
            for v in adj[pos]:
                if assignment[v] != 0:
                    if (val, assignment[v]) not in valid_pairs or (assignment[v], val) not in valid_pairs:
                        valid = False
                        break
            if valid:
                assignment[pos] = val
                if backtrack(pos + 1):
                    return True
                assignment[pos] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(' '.join(map(str, best_assignment)))
    else:
        print("NO")

solve()
