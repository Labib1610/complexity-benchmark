import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # R(s_u, s_v) = 7*s_u*s_v - 2*s_u^2 - 2*s_v^2
    # s in {1, 2, 3}
    # R(1,1) = 7-2-2 = 3 (Not power of 2)
    # R(1,2) = 14-2-8 = 4 (2^2)
    # R(1,3) = 21-2-18 = 1 (2^0)
    # R(2,1) = 4
    # R(2,2) = 28-8-8 = 12 (Not power of 2)
    # R(2,3) = 42-8-18 = 16 (2^4)
    # R(3,1) = 1
    # R(3,2) = 16
    # R(3,3) = 63-18-18 = 27 (Not power of 2)

    # Valid pairs (s_u, s_v):
    # (1,2), (2,1), (1,3), (3,1), (2,3), (3,2)
    # Note: s_u, s_v cannot be (1,1), (2,2), (3,3)
    
    # This is a 3-coloring problem where adjacent nodes must have different colors,
    # and specifically, the pair (s_u, s_v) must be in {(1,2), (1,3), (2,3)}
    # This is equivalent to saying s_u != s_v.
    # So we need to check if the graph is 3-colorable.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    colors = [-1] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = -1
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
