import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # P(x) = x^3 - 1.5(su + sv)x^2 + 3*su*sv*x
    # P'(x) = 3x^2 - 3(su + sv)x + 3*su*sv
    # P''(x) = 6x - 3(su + sv)
    # Stationary inflection point: P''(x) = 0 => x = (su + sv) / 2
    # P'(x) = 0 at x = (su + sv) / 2
    # 3((su + sv)/2)^2 - 3(su + sv)((su + sv)/2) + 3*su*sv = 0
    # 3/4 * (su + sv)^2 - 3/2 * (su + sv)^2 + 3*su*sv = 0
    # -3/4 * (su + sv)^2 + 3*su*sv = 0
    # -3/4 * (su^2 + 2*su*sv + sv^2) + 3*su*sv = 0
    # -3/4 * su^2 - 3/2 * su*sv - 3/4 * sv^2 + 3*su*sv = 0
    # -3/4 * su^2 + 3/2 * su*sv - 3/4 * sv^2 = 0
    # -3/4 * (su^2 - 2*su*sv + sv^2) = 0
    # -3/4 * (su - sv)^2 = 0
    # This is 0 if and only if su == sv.
    # So, a stable configuration exists if and only if for every linked pair (u, v), su != sv.
    # This is equivalent to checking if the graph is 3-colorable.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
