import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Precompute pulse condition
    # M(s1, s2) = (17*s1^3 + 27*s2^2) % 64
    # M(s2, s1) = (17*s2^3 + 27*s1^2) % 64
    # Condition: M(s1, s2) != M(s2, s1)
    
    memo_valid = {}
    def is_valid(s1, s2):
        if (s1, s2) in memo_valid: return memo_valid[(s1, s2)]
        m12 = (17 * (s1**3) + 27 * (s2**2)) % 64
        m21 = (17 * (s2**3) + 27 * (s1**2)) % 64
        res = (m12 != m21)
        memo_valid[(s1, s2)] = res
        return res

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    assignment = [0] * n

    def backtrack(u):
        if u == n:
            return True
        
        # Try 1, 2, 3 to minimize signature (lexicographical order)
        for s in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if v < u:
                    if not is_valid(assignment[v], s):
                        possible = False
                        break
            if possible:
                assignment[u] = s
                if backtrack(u + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
