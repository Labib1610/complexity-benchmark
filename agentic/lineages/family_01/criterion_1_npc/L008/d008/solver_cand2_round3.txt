import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Stability factor f(su, sv) = floor(((su - sv)^2 + 3) / 4)
    # f(1,1)=0, f(2,2)=0, f(3,3)=0
    # f(1,2)=floor((1+3)/4)=1, f(2,1)=1
    # f(2,3)=floor((1+3)/4)=1, f(3,2)=1
    # f(1,3)=floor((4+3)/4)=1, f(3,1)=1
    # The condition G = m means for every link (u, v), f(su, sv) must be 1.
    # This means su != sv.
    # So the problem is equivalent to: Can we color the graph with 3 colors
    # such that no two adjacent nodes have the same color?
    # This is the 3-coloring problem.
    # We need to find the lexicographically smallest sequence (s1, s2, ..., sn)
    # where each si is in {1, 2, 3}.

    assignment = [0] * (n + 1)

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
