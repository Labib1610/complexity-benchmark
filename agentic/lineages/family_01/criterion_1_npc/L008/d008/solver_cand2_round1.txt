import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    links = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        links.append((u, v))
        idx += 2

    # s_i! < 10 => s_i can be 1, 2, 3 (since 3! = 6 < 10, 4! = 24 >= 10)
    # F_{u,v} = (s_u^2 + s_v^2) % (s_u * s_v)
    # If s_u = s_v = s: F = (2s^2) % s^2 = 0.
    # If s_u = 1, s_v = 2: F = (1+4) % 2 = 1.
    # If s_u = 1, s_v = 3: F = (1+9) % 3 = 1.
    # If s_u = 2, s_v = 3: F = (4+9) % 6 = 1.
    # Stability condition: sum(F_{u,v}) = m.
    # Since F_{u,v} is 0 if s_u == s_v and 1 if s_u != s_v,
    # the condition sum(F_{u,v}) = m means that for every link (u, v),
    # we must have s_u != s_v.
    # This is equivalent to saying the graph must be bipartite.

    adj = [[] for _ in range(n)]
    for u, v in links:
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    # To minimize V_n = 4*V_{n-1} + s_n, we want smaller s_i for smaller i.
    # Possible values for s_i are {1, 2, 3}.
    # If color[i] == 0, s_i can be from {1, 2, 3}.
    # If color[i] == 1, s_i can be from {1, 2, 3}.
    # But we need s_u != s_v for all links.
    # If we assign all color 0 nodes to value A and all color 1 nodes to value B,
    # we need A != B. To minimize V_n, we should pick A, B from {1, 2, 3}
    # such that A != B and the sequence is lexicographically smallest.
    
    best_s = None
    min_v = float('inf')
    
    # Try all pairs (A, B) from {1, 2, 3} where A != B
    for A in [1, 2, 3]:
        for B in [1, 2, 3]:
            if A == B: continue
            
            current_s = [0] * n
            for i in range(n):
                current_s[i] = A if color[i] == 0 else B
            
            v_val = 0
            for x in current_s:
                v_val = 4 * v_val + x
            
            if v_val < min_v:
                min_v = v_val
                best_s = current_s
                
    print("YES")
    print(*(best_s))

solve()
