```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n  # 0 = uncolored, else 1..3

    sys.setrecursionlimit(10000)

    def dfs(v: int) -> bool:
        if v == n:
            return True
        # compute forbidden colors from already colored neighbours
        forbidden = 0  # bitmask 1<<c
        for nb in adj[v]:
            col = colors[nb]
            if col:
                forbidden |= 1 << col
        for c in (1, 2, 3):
            if not (forbidden & (1 << c)):
                colors[v] = c
                # simple forward check: any future vertex already impossible?
                ok = True
                for u in range(v + 1, n):
                    forb = 0
                    for nb in adj[u]:
                        col = colors[nb]
                        if col:
                            forb |= 1 << col
                    if forb == 0b1110:   # bits 1,2,3 set (mask 0b1110 = 14)
                        ok = False
                        break
                if ok and dfs(v + 1):
                    return True
                colors[v] = 0
        return False

    if dfs(0):
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(map(str, colors)) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == "__main__":
    solve()
```
