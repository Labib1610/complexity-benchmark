import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To find the lexicographically smallest solution, we must assign colors
    # to elements 1, 2, ..., n in order.
    # The problem suggests dynamic prioritization for efficiency, but the 
    # lexicographical requirement forces a specific order of assignment.
    # We use backtracking with pruning.
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        # Try colors 1, 2, 3 in order to ensure lexicographical smallest
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # The problem states n <= 40 (since 40*39/2 = 780).
    # Backtracking with simple pruning is sufficient for 3-coloring on small graphs.
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
