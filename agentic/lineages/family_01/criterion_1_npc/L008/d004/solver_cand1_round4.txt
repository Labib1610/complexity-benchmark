import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    colors = [0] * n
    
    # To minimize S = sum(s_i * 4^(n-i)), we want s_1 to be 1, then s_2 to be 1, etc.
    # This is equivalent to finding the lexicographically smallest sequence (s_1, s_2, ..., s_n).
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3 in order to minimize the signature
        neighbor_colors = 0
        for v in range(n):
            if (adj[u] >> v) & 1:
                if colors[v] != 0:
                    neighbor_colors |= (1 << (colors[v] - 1))
        
        for c in range(1, 4):
            if not ((neighbor_colors >> (c - 1)) & 1):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
