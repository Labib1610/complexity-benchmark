import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To handle the adversarial nature and ensure lexicographical smallest,
    # we use backtracking with a simple heuristic: degree-based ordering
    # or just standard backtracking. Given n=40, we need to be efficient.
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    # Pre-calculate degrees to order nodes by constraint intensity
    # However, the problem asks for lexicographical smallest, so we must
    # assign colors to 1, 2, ..., n in that order.
    
    def backtrack(u):
        if u > n:
            return True
        
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # The problem states that simple lexicographical order might be slow.
    # We can optimize by checking connectivity or using MRV (Minimum Remaining Values).
    # But since we need the lexicographically smallest result, we must try 
    # colors 1, 2, 3 for node 1, then 1, 2, 3 for node 2, etc.
    
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
