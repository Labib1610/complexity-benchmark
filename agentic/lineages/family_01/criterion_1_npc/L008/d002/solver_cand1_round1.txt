import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
            
    assignment = [0] * n
    
    # We need the lexicographically smallest sequence.
    # This means we try assigning 1, then 2, then 3 for each part in order 0 to n-1.
    
    def backtrack(u):
        if u == n:
            return True
        
        # Find forbidden colors for current part u
        forbidden = 0
        for v in adj[u]:
            if assignment[v] != 0:
                forbidden |= (1 << (assignment[v] - 1))
        
        # Try colors 1, 2, 3
        for color in range(1, 4):
            if not (forbidden & (1 << (color - 1))):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # To optimize, we can use a simple heuristic: 
    # sort nodes by degree or just use the natural order.
    # Given n=40, simple backtracking with pruning is usually sufficient 
    # for 3-coloring if the graph isn't specifically designed to be hard.
    # If it is hard, we use the fact that we need the lexicographically smallest.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
