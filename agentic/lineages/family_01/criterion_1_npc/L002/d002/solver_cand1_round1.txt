```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Condition: (c_u + 2 * c_v) % 3 != 0
    # Let c_u, c_v be in {1, 2, 3}. Map to {0, 1, 2} by x -> x-1.
    # (c_u-1 + 1 + 2*(c_v-1 + 1)) % 3 != 0
    # (c_u' + 1 + 2*c_v' + 2) % 3 != 0
    # (c_u' + 2*c_v') % 3 != 0
    # Let x_u = c_u' in {0, 1, 2}.
    # Condition: x_u + 2*x_v != 0 (mod 3)
    # x_u != -2*x_v (mod 3) => x_u != x_v (mod 3)
    # Wait, let's recheck:
    # If x_u = 0, x_v = 0: 0 + 0 = 0 (Error)
    # If x_u = 1, x_v = 1: 1 + 2 = 3 = 0 (Error)
    # If x_u = 2, x_v = 2: 2 + 4 = 6 = 0 (Error)
    # So x_u != x_v is the condition.
    # This is simply a graph coloring problem with 3 colors.
    
    assignment = [-1] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in range(3):
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = -1
        return False

    if backtrack(0):
        print("YES")
        print(*(c + 1 for c in assignment))
    else:
        print("NO")

solve()
```
