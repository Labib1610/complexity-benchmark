```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    # Sort nodes by degree descending to improve pruning (heuristic)
    nodes = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    
    def backtrack(idx_in_nodes):
        if idx_in_nodes == n:
            return True
        
        u = nodes[idx_in_nodes]
        
        # Try frequencies 1, 2, 3
        used_neighbors = set()
        for v in adj[u]:
            if assignment[v] != 0:
                used_neighbors.add(assignment[v])
        
        for freq in [1, 2, 3]:
            if freq not in used_neighbors:
                assignment[u] = freq
                if backtrack(idx_in_nodes + 1):
                    return True
                assignment[u] = 0
        
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
