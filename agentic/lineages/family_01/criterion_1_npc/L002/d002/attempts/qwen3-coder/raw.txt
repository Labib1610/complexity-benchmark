def solve():
    import sys
    from collections import defaultdict
    
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(set)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].add(v)
        adj[v].add(u)
    
    # Color the graph with 3 colors using backtracking
    colors = [0] * (n + 1)  # colors[i] represents color of node i (1, 2, or 3)
    
    def is_safe(node, color):
        # Check if assigning 'color' to 'node' conflicts with neighbors
        for neighbor in adj[node]:
            if colors[neighbor] == color:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
            
        for color in [1, 2, 3]:
            if is_safe(node, color):
                colors[node] = color
                if backtrack(node + 1):
                    return True
                colors[node] = 0  # backtrack
                
        return False
    
    # Since n <= 40, but we need to check if it's 3-colorable,
    # and the graph might be disconnected, we still try backtracking
    # But given the constraints, a simple backtracking with proper pruning should work
    
    if backtrack(1):
        print("YES")
        print(*colors[1:])
    else:
        print("NO")

solve()
