import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem is equivalent to 3-coloring a graph.
    # Since n <= 40, we can use backtracking with pruning.
    # To optimize, we use the Welsh-Powell heuristic or simple degree-based ordering.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    # Sort nodes by degree descending to fail fast
    nodes = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    
    def backtrack(idx):
        if idx == n:
            return True
        
        u = nodes[idx]
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(idx + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
