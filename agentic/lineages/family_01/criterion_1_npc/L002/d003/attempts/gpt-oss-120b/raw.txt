```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n          # bitmask of neighbours
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # order vertices by descending degree for better pruning
    order = sorted(range(n), key=lambda x: adj[x].bit_count(), reverse=True)
    pos_in_order = [0] * n
    for i, v in enumerate(order):
        pos_in_order[v] = i

    colors = [-1] * n  # -1 uncolored, else 0,1,2

    # precompute neighbour list for quick iteration
    neigh = [ [j for j in range(n) if (adj[i] >> j) & 1] for i in range(n) ]

    sys.setrecursionlimit(10000)

    def dfs(k: int) -> bool:
        if k == n:
            return True
        v = order[k]
        # try each color
        for c in range(3):
            ok = True
            for nb in neigh[v]:
                if colors[nb] == c:
                    ok = False
                    break
            if not ok:
                continue
            colors[v] = c
            if dfs(k + 1):
                return True
            colors[v] = -1
        return False

    possible = dfs(0)
    out = []
    if possible:
        out.append("YES")
        # convert to 1..3
        out.append(' '.join(str(c + 1) for c in colors))
    else:
        out.append("NO")
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
```
