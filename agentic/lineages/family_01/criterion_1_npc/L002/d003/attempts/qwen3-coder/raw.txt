```python
import sys
from collections import defaultdict

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u-1, v-1))  # Convert to 0-indexed
    return n, m, edges

def is_stable(c1, c2):
    # Determinant = c1*c1 - c2*c2
    # For stability, determinant != 0, i.e., c1^2 != c2^2, i.e., |c1| != |c2|
    # Since codes are in {1,2,3}, this means c1 != c2
    return c1 != c2

def solve(n, m, edges):
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Try all possible assignments in lexicographical order using backtracking
    assignment = [0] * n  # 0 means unassigned
    
    def backtrack(node):
        if node == n:
            return True
            
        # Try colors 1, 2, 3 in order
        for color in [1, 2, 3]:
            # Check if this color is valid for this node
            valid = True
            for neighbor in adj[node]:
                if assignment[neighbor] != 0:  # If neighbor is already assigned
                    if not is_stable(color, assignment[neighbor]):
                        valid = False
                        break
            if valid:
                assignment[node] = color
                if backtrack(node + 1):
                    return True
                assignment[node] = 0  # Backtrack
                
        return False
    
    if backtrack(0):
        return "YES", assignment
    else:
        return "NO", []

def main():
    n, m, edges = read_input()
    result, assignment = solve(n, m, edges)
    print(result)
    if result == "YES":
        print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    main()
```
