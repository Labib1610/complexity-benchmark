import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Resonance condition: R = (tu^2 + tv^2) % (tu * tv) == 0
    # Possible tiers: {1, 2, 3}
    # Pairs (tu, tv):
    # (1, 1): R = (1+1) % 1 = 0 (Resonance)
    # (1, 2): R = (1+4) % 2 = 1 (Stable)
    # (1, 3): R = (1+9) % 3 = 1 (Stable)
    # (2, 2): R = (4+4) % 4 = 0 (Resonance)
    # (2, 3): R = (4+9) % 6 = 1 (Stable)
    # (3, 3): R = (9+9) % 9 = 0 (Resonance)
    # Forbidden pairs: (1,1), (2,2), (3,3)
    # This means the graph must be 3-colorable.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
