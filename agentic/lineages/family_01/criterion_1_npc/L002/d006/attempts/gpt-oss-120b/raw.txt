```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)          # direction irrelevant for coloring constraint

    order = sorted(range(n), key=lambda x: -len(adj[x]))
    color = [-1] * n
    sys.setrecursionlimit(10000)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = order[idx]
        # try colors 0,1,2
        for c in range(3):
            ok = True
            for nb in adj[v]:
                if color[nb] == c:
                    ok = False
                    break
            if ok:
                color[v] = c
                if dfs(idx + 1):
                    return True
                color[v] = -1
        return False

    if dfs(0):
        print("YES")
        # output in original order, convert to 1..3
        print(' '.join(str(color[i] + 1) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
