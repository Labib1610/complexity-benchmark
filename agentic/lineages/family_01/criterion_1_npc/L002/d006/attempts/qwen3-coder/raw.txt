```python
import sys
from collections import defaultdict

def solve():
    n, m = map(int, input().split())
    
    # Build adjacency list
    graph = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        graph[u].append(v)
        graph[v].append(u)  # For finding connected components
    
    # Check if a valid 3-coloring exists using backtracking
    colors = [0] * (n + 1)  # 0 means unassigned, 1,2,3 are colors
    
    def is_safe(node, color):
        # Check if assigning 'color' to 'node' conflicts with neighbors
        for edge in edges:
            u, v = edge
            # If this edge involves our node
            if u == node:
                if colors[v] == color:
                    return False
            elif v == node:
                if colors[u] == color:
                    return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
            
        if colors[node] != 0:
            return backtrack(node + 1)
            
        for color in [1, 2, 3]:
            if is_safe(node, color):
                colors[node] = color
                if backtrack(node + 1):
                    return True
                colors[node] = 0
                
        return False
    
    # Special case: no edges
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Try to find a valid coloring
    if backtrack(1):
        print("YES")
        print(" ".join(map(str, colors[1:])))
    else:
        print("NO")

solve()
```
