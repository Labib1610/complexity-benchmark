```python
import sys

def solve():
    # The number of trailing zeros in n! is floor(n/5) + floor(n/25) + ...
    # For 9 zeros:
    # n=40: 8 + 1 = 9. So K=40.
    # The condition "number of ways to express C[i] as sum of two positive integers < 2"
    # means C[i] can be 1 (0 ways: 1 = x+y, no sol) or 2 (1 way: 2 = 1+1).
    # So C[i] must be in {1, 2}.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # C[i] in {1, 2}. |C[u] - C[v]| must be power of 2.
    # If C[u] == C[v], |C[u] - C[v]| = 0, not a power of 2.
    # So C[u] != C[v] for all edges (u, v).
    # This means the graph must be bipartite.
    
    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, we can assign 1 to color 0 and 2 to color 1.
    # |1 - 2| = 1 = 2^0, which is a power of 2.
    res = [0] * n
    for i in range(n):
        if color[i] == -1:
            res[i] = 1
        else:
            res[i] = color[i] + 1
            
    print("YES")
    print(*(res))

solve()
```
