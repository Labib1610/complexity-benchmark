import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Condition: gcd(C[u]!, C[v]!) < max(C[u], C[v])
    # C[i] in {1, 2, 3}
    # 1! = 1, 2! = 2, 3! = 6
    # Pairs (C[u], C[v]):
    # (1, 1): gcd(1, 1) = 1, max = 1. 1 < 1 is False.
    # (1, 2): gcd(1, 2) = 1, max = 2. 1 < 2 is True.
    # (1, 3): gcd(1, 6) = 1, max = 3. 1 < 3 is True.
    # (2, 2): gcd(2, 2) = 2, max = 2. 2 < 2 is False.
    # (2, 3): gcd(2, 6) = 2, max = 3. 2 < 3 is True.
    # (3, 3): gcd(6, 6) = 6, max = 3. 6 < 3 is False.
    
    # Valid pairs (u, v) must have different values, OR one of them must be 3 and the other 2,
    # OR one of them must be 3 and the other 1, OR one of them must be 2 and the other 1.
    # Actually, the condition is:
    # (1, 1) -> False
    # (2, 2) -> False
    # (3, 3) -> False
    # (1, 2) -> True
    # (1, 3) -> True
    # (2, 3) -> True
    # So the condition is simply C[u] != C[v].
    # This is a graph coloring problem with 3 colors.
    
    color = [-1] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if color[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                color[u] = c
                if backtrack(u + 1):
                    return True
                color[u] = -1
        return False
    
    if backtrack(0):
        print("YES")
        print(*(color))
    else:
        print("NO")

solve()
