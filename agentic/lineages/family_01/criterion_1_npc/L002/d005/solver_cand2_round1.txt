import sys
import math

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
        
    # 120 / C[i] = k, where 35 < k < 125
    # k must be a divisor of 120.
    # Divisors of 120: 1, 2, 3, 4, 5, 6, 8, 10, 12, 15, 20, 24, 30, 40, 60, 120
    # k in (35, 125) => k can be 40, 60, 120
    # C[i] = 120 / k => C[i] can be 3, 2, 1
    
    possible_values = [1, 2, 3]
    
    def check(seq):
        for u, v in edges:
            val_u = seq[u] + 5
            val_v = seq[v] + 5
            lcm_val = (val_u * val_v) // math.gcd(val_u, val_v)
            if not (lcm_val > seq[u] + seq[v] + 5):
                return False
        return True

    # Since n is small (up to 40), we can try to assign values.
    # With only 3 possible values, we can use backtracking or a greedy approach.
    # Given the constraints and the nature of the condition, let's try backtracking.
    
    assignment = [0] * n
    
    def backtrack(idx):
        if idx == n:
            if check(assignment):
                return True
            return False
        
        for val in possible_values:
            assignment[idx] = val
            if backtrack(idx + 1):
                return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
