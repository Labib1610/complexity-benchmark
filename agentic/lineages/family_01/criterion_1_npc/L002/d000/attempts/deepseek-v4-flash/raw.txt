We can model this as a graph coloring problem with 3 colors, where adjacent nodes must have different colors. Since \( n \leq 40 \), we can use backtracking with pruning to find the lexicographically smallest valid assignment.

We’ll assign colors from 1 to 3 in order, trying the smallest possible value for each node first, and backtrack if a conflict occurs.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * n

    def is_safe(node, c):
        for nei in adj[node]:
            if color[nei] == c:
                return False
        return True

    def backtrack(idx):
        if idx == n:
            return True
        for c in range(1, 4):
            if is_safe(idx, c):
                color[idx] = c
                if backtrack(idx + 1):
                    return True
                color[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
