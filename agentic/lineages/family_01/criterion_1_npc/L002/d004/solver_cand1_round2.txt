```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The quadratic equation z^2 - (C[u]+C[v])z + C[u]C[v] = 0
    # factors to (z - C[u])(z - C[v]) = 0.
    # The roots are C[u] and C[v].
    # For the roots to be distinct, C[u] must not equal C[v].
    # This is a graph coloring problem: can we color the graph with 3 colors
    # such that every edge (u, v) has C[u] != C[v]?
    
    # Since n <= 40, we can use backtracking with pruning.
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
