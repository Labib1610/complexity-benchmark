import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)

    # The metric M = sum_{k=1}^n (3 * C_k * 4^{n-k-1}) + C_n
    # where C_k = sum_{i=1}^k b_i.
    # M = 3 * (b_1 * 4^{n-2} + (b_1+b_2) * 4^{n-3} + ... + (b_1+...+b_{n-1}) * 4^0) + (b_1+...+b_n)
    # M = b_1 * (3 * sum_{j=0}^{n-2} 4^j + 1) + b_2 * (3 * sum_{j=0}^{n-3} 4^j + 1) + ... + b_n * 1
    # 3 * sum_{j=0}^{k} 4^j = 3 * (4^{k+1}-1)/(4-1) = 4^{k+1}-1
    # So coefficient of b_i is (4^{n-i}-1) + 1 = 4^{n-i} for i < n, and 1 for i = n.
    # M = sum_{i=1}^n b_i * 4^{n-i}
    # To minimize M, we want smaller b_i for smaller i (lexicographical priority).
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3 in order to minimize M
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # Since n <= 40, standard backtracking might be slow.
    # However, the graph is 3-colorable. 
    # For n=40, we use a simple heuristic or just standard backtracking.
    # Given the constraints and the nature of the problem, 
    # backtracking with pruning is usually sufficient for 3-coloring.
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
