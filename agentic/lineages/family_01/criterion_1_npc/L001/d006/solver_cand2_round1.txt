import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Weights calculation
    # W_n = 1, W_{n-1} = 4
    # W_i = 5*W_{i+1} - 4*W_{i+2}
    # Characteristic eq: r^2 - 5r + 4 = 0 => (r-4)(r-1) = 0
    # W_i = A * 4^i + B * 1^i
    # W_n = A * 4^n + B = 1
    # W_{n-1} = A * 4^{n-1} + B = 4
    # Subtracting: A * 4^{n-1} * (4 - 1) = 1 - 4 = -3
    # A * 4^{n-1} * 3 = -3 => A = -1 / 4^{n-1}
    # B = 1 - A * 4^n = 1 - (-1/4^{n-1}) * 4^n = 1 + 4 = 5
    # W_i = 5 - 4^{i-n+1}
    weights = [0] * n
    for i in range(1, n + 1):
        weights[i-1] = 5 - (4**(i - n + 1))

    best_score = -float('inf')
    best_profile = None

    # Since n <= 40, we use backtracking with pruning
    # To optimize, use connected components
    visited = [0] * n
    
    def get_best(comp_nodes, current_assignment):
        nonlocal best_score, best_profile
        
        def backtrack(idx, current_profile):
            if idx == len(comp_nodes):
                score = sum((4 - current_profile[i]) * weights[i] for i in range(n))
                return score, current_profile
            
            u = comp_nodes[idx]
            best_s = -float('inf')
            best_p = None
            
            for f in [1, 2, 3]:
                valid = True
                for neighbor in adj[u]:
                    if neighbor < u and current_profile[neighbor] == f:
                        valid = False
                        break
                if valid:
                    current_profile[u] = f
                    s, p = backtrack(idx + 1, current_profile)
                    if s > best_s:
                        best_s, best_p = s, list(p)
            return best_s, best_p

        return backtrack(0, current_assignment)

    # For n=40, simple backtracking might be slow, but the constraints 
    # on 3-coloring are tight.
    assignment = [0] * n
    
    def solve_recursive(u, current_assignment):
        if u == n:
            score = sum((4 - current_assignment[i]) * weights[i] for i in range(n))
            return score, list(current_assignment)
        
        best_s = -float('inf')
        best_p = None
        
        for f in [1, 2, 3]:
            valid = True
            for neighbor in adj[u]:
                if neighbor < u and current_assignment[neighbor] == f:
                    valid = False
                    break
            if valid:
                current_assignment[u] = f
                s, p = solve_recursive(u + 1, current_assignment)
                if s > best_s:
                    best_s, best_p = s, p
        return best_s, best_p

    score, profile = solve_recursive(0, assignment)
    
    if profile is None:
        print("NO")
    else:
        print("YES")
        print(*(profile))

solve()
