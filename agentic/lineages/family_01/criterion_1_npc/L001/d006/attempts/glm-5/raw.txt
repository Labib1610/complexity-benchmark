import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # 3-coloring using backtracking with constraint propagation
    color = [0] * n
    
    # Order vertices by degree (descending) for better pruning
    order = list(range(n))
    order.sort(key=lambda x: -len(adj[x]))
    pos_in_order = [0] * n
    for i, v in enumerate(order):
        pos_in_order[v] = i
    
    def backtrack(idx):
        if idx == n:
            return True
        v = order[idx]
        used = 0
        for nb in adj[v]:
            if color[nb] != 0:
                used |= (1 << color[nb])
        for c in [1, 2, 3]:
            if not (used & (1 << c)):
                color[v] = c
                if backtrack(idx + 1):
                    return True
                color[v] = 0
        return False
    
    if not backtrack(0):
        print("NO")
        return
    
    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
