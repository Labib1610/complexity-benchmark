def solve():
    import sys
    from itertools import product
    
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list for the graph
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    # Try all possible assignments (3^n possibilities)
    # But we can do better with backtracking for efficiency and lex order
    assignment = [0] * (n + 1)
    
    def is_valid(material, chamber):
        for neighbor in adj[material]:
            if assignment[neighbor] == chamber:
                return False
        return True
    
    def backtrack(material):
        if material > n:
            return True
            
        for chamber in [1, 2, 3]:
            if is_valid(material, chamber):
                assignment[material] = chamber
                if backtrack(material + 1):
                    return True
                assignment[material] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(' '.join(map(str, assignment[1:])))
    else:
        print("NO")

solve()
