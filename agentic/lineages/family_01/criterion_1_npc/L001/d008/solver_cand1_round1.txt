import sys
import math

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Precompute binomial coefficients
    C = [[0] * (n + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        C[i][0] = 1
        for j in range(1, i + 1):
            C[i][j] = C[i-1][j-1] + C[i-1][j]

    # M = sum_{i=1}^n sum_{j=1}^i b_j * C(n-j, i-j) * 3^(i-j)
    # Swap sums: M = sum_{j=1}^n b_j * sum_{i=j}^n C(n-j, i-j) * 3^(i-j)
    # Let k = i-j: sum_{k=0}^{n-j} C(n-j, k) * 3^k
    # By binomial theorem, sum_{k=0}^N C(N, k) * x^k = (1+x)^N
    # Here N = n-j, x = 3, so sum = (1+3)^(n-j) = 4^(n-j)
    # M = sum_{j=1}^n b_j * 4^(n-j)
    # To minimize M, we want smaller b_j for larger weights 4^(n-j)
    # Weights are 4^(n-1), 4^(n-2), ..., 4^0.
    # This is a greedy problem: minimize b_1, then b_2, ..., then b_n.

    best_config = None
    
    # Backtracking with greedy choice (try 1, 2, 3 in order)
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
