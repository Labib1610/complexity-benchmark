import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * n
    
    # To find the lexicographically smallest, we try to assign 1, 2, 3 in order
    # for each shard from 1 to n.
    # Since n is small (up to 40), we use backtracking with pruning.
    
    def get_possible_colors(u, current_assignment):
        forbidden = 0
        for v in range(n):
            if (adj[u] & (1 << v)) and current_assignment[v] != 0:
                forbidden |= (1 << (current_assignment[v] - 1))
        return [c for c in [1, 2, 3] if not (forbidden & (1 << (c - 1)))]

    # Check if the remaining graph is 3-colorable.
    # For n=40, we can use a simple backtracking with MRV (Minimum Remaining Values)
    # heuristic to speed up the search.
    
    def backtrack(assigned_count, current_assignment):
        if assigned_count == n:
            return True
        
        # MRV heuristic: pick the unassigned shard with the fewest available colors
        best_u = -1
        min_colors = 4
        best_possible = []
        
        for u in range(n):
            if current_assignment[u] == 0:
                possible = get_possible_colors(u, current_assignment)
                if len(possible) < min_colors:
                    min_colors = len(possible)
                    best_u = u
                    best_possible = possible
                if min_colors == 0:
                    return False
        
        # To ensure lexicographical order, we must assign shards in order 1..n.
        # However, the problem asks for the lexicographically smallest sequence.
        # This means we MUST try to assign shard 0, then 1, then 2...
        # The MRV heuristic is for speed, but we must respect the order.
        
        # Let's stick to the order 0..n-1 to guarantee lexicographical smallest.
        u = 0
        while u < n and current_assignment[u] != 0:
            u += 1
            
        possible = get_possible_colors(u, current_assignment)
        for color in possible:
            current_assignment[u] = color
            if backtrack(assigned_count + 1, current_assignment):
                return True
            current_assignment[u] = 0
        return False

    if backtrack(0, assignment):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
