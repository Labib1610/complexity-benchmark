```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # Sort adjacency lists for deterministic processing
    for i in range(n):
        adj[i].sort()
    
    freq = [0] * n
    
    def dfs(idx):
        if idx == n:
            return True
        
        for f in [1, 2, 3]:
            valid = True
            for neighbor in adj[idx]:
                if neighbor < idx and freq[neighbor] == f:
                    valid = False
                    break
            if valid:
                freq[idx] = f
                if dfs(idx + 1):
                    return True
                freq[idx] = 0
        return False
    
    if dfs(0):
        print("YES")
        print(' '.join(map(str, freq)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
