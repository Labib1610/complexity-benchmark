import sys

def solve():
    # The condition:
    # Let S be the sum of all frequencies.
    # For a pair (u, v), the condition is:
    # S - f_u - f_v != S - 2 * f_u
    # This simplifies to:
    # -f_v != -f_u
    # f_v != f_u
    # So the condition is simply that for every intersecting pair (u, v),
    # f_u must not equal f_v.
    # This is a standard graph coloring problem with 3 colors.
    # We need the lexicographically smallest coloring.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # Since we need the lexicographically smallest, we greedily try 1, 2, 3
    # for each node in order 1 to n.
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
