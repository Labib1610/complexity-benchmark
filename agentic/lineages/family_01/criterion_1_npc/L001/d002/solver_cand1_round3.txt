import sys

# The problem asks for the lexicographically smallest 3-coloring of a graph.
# Since n <= 40 (8 racks * 5 materials = 40), we can use backtracking with pruning.
# Lexicographical order means we try assigning chamber 1, then 2, then 3 for each material.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_safe(u, chamber):
        for v in adj[u]:
            if assignment[v] == chamber:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for chamber in [1, 2, 3]:
            if is_safe(u, chamber):
                assignment[u] = chamber
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Optimization: The graph might be disconnected. 
    # However, for n=40, simple backtracking with lexicographical order is usually 
    # efficient enough if the graph is sparse or has constraints.
    # Given the constraints and the nature of 3-coloring (NP-complete), 
    # we rely on the small n.
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
