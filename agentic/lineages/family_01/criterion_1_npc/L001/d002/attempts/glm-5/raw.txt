import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # Greedy assignment for lexicographically smallest 3-coloring
    colors = [0] * n
    for u in range(n):
        used = set()
        for v in adj[u]:
            if v < u:
                used.add(colors[v])
        for c in range(1, 4):
            if c not in used:
                colors[u] = c
                break
        if colors[u] == 0:
            print("NO")
            return
    
    # Verify validity
    for u in range(n):
        for v in adj[u]:
            if colors[u] == colors[v]:
                print("NO")
                return
    
    print("YES")
    print(' '.join(map(str, colors)))

if __name__ == "__main__":
    solve()
