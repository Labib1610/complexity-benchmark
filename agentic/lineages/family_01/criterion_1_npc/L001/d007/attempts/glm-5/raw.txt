import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # Find connected components
    visited = [False] * n
    components = []
    
    for i in range(n):
        if not visited[i]:
            comp = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            comp.sort()
            components.append(comp)
    
    # Check if each component is 3-colorable
    color = [-1] * n
    
    def is_bipartite(nodes):
        c = {}
        for start in nodes:
            if start not in c:
                c[start] = 0
                stack = [start]
                while stack:
                    u = stack.pop()
                    for v in adj[u]:
                        if v not in c:
                            c[v] = 1 - c[u]
                            stack.append(v)
                        elif c[v] == c[u]:
                            return None
        return c
    
    for comp in components:
        if len(comp) >= 3:
            # Check if component contains a triangle
            node_set = set(comp)
            has_triangle = False
            for u in comp:
                for v in adj[u]:
                    if v > u:
                        for w in adj[v]:
                            if w > v and w in node_set:
                                if u in adj[w] or any(x == u for x in adj[w]):
                                    has_triangle = True
                                    break
                        if has_triangle:
                            break
                    if has_triangle:
                        break
                if has_triangle:
                    break
            
            if has_triangle:
                # Check if 3-colorable using backtracking
                comp_idx = {node: i for i, node in enumerate(comp)}
                comp_adj = [[] for _ in range(len(comp))]
                for i, u in enumerate(comp):
                    for v in adj[u]:
                        if v in node_set:
                            comp_adj[i].append(comp_idx[v])
                
                comp_color = [-1] * len(comp)
                possible = [False]
                
                def backtrack(pos):
                    if pos == len(comp):
                        possible[0] = True
                        return True
                    u = comp[pos]
                    used = 0
                    for v_idx in comp_adj[u]:
                        if comp_color[v_idx] != -1:
                            used |= (1 << comp_color[v_idx])
                    for c in range(3):
                        if not (used & (1 << c)):
                            comp_color[u] = c
                            if backtrack(pos + 1):
                                return True
                            comp_color[u] = -1
                    return False
                
                if not backtrack(0):
                    print("NO")
                    return
                
                for i, node in enumerate(comp):
                    color[node] = comp_color[i]
        else:
            # Bipartite component
            c = is_bipartite(comp)
            if c is None:
                print("NO")
                return
            for node, col in c.items():
                color[node] = col
    
    # Assign colors to minimize metric
    # For each component, assign bands to minimize contribution
    result = [0] * n
    
    for comp in components:
        if len(comp) == 1:
            result[comp[0]] = 1
        elif len(comp) == 2:
            # Bipartite: assign 1 to smaller index, 2 to larger
            comp_sorted = sorted(comp)
            result[comp_sorted[0]] = 1
            result[comp_sorted[1]] = 2
        else:
            # 3-colorable component: assign 1,2,3 to minimize metric
            # Sort by original index
            comp_sorted = sorted(comp)
            # color[] contains 0,1,2 for this component
            # We need to map 0,1,2 to bands 1,2,3
            # To minimize metric, smallest index should get band 1, etc.
            # Group nodes by their color
            color_groups = {0: [], 1: [], 2: []}
            for node in comp:
                color_groups[color[node]].append(node)
            
            # Sort each group
            for c in color_groups:
                color_groups[c].sort()
            
            # We want to assign band 1 to the color group containing the smallest index
            # band 2 to the color group containing the second smallest index among group mins
            # band 3 to the remaining group
            
            group_mins = []
            for c in [0, 1, 2]:
                if color_groups[c]:
                    group_mins.append((color_groups[c][0], c))
            
            group_mins.sort()
            
            band_assignment = {}
            for band, (_, c) in enumerate(group_mins, 1):
                band_assignment[c] = band
            
            for node in comp:
                result[node] = band_assignment[color[node]]
    
    print("YES")
    print(' '.join(map(str, result)))

solve()
