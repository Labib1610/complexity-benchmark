```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    assigned = [0] * n          # 0 = unassigned, else 1..3
    poss = [0b111] * n          # possible colors mask

    # initial forward checking not needed

    def dfs(v: int) -> bool:
        if v == n:
            return True
        # try colors 1..3 in order
        for c in (1, 2, 3):
            if not (poss[v] & (1 << (c - 1))):
                continue
            # assign c to v, propagate
            assigned[v] = c
            changed = []  # list of (neighbor, previous mask)
            ok = True
            for nb in adj[v]:
                if assigned[nb]:
                    if assigned[nb] == c:
                        ok = False
                        break
                else:
                    if poss[nb] & (1 << (c - 1)):
                        prev = poss[nb]
                        new = prev & ~(1 << (c - 1))
                        if new == 0:
                            ok = False
                            break
                        poss[nb] = new
                        changed.append((nb, prev))
            if ok and dfs(v + 1):
                return True
            # revert
            for nb, prev in reversed(changed):
                poss[nb] = prev
            assigned[v] = 0
        return False

    if dfs(0):
        print("YES")
        print(' '.join(map(str, assigned)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
