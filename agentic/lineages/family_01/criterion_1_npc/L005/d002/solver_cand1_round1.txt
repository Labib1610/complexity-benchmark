import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Sort nodes by degree descending to optimize backtracking (heuristic)
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    assignment = [0] * (n + 1)
    
    def backtrack(idx):
        if idx == n:
            return True
        
        u = nodes[idx]
        
        # Find forbidden zones for current node
        forbidden = set()
        for neighbor in adj[u]:
            if assignment[neighbor] != 0:
                forbidden.add(assignment[neighbor])
        
        # Try assigning to zones 1, 2, 3
        for zone in [1, 2, 3]:
            if zone not in forbidden:
                assignment[u] = zone
                if backtrack(idx + 1):
                    return True
                assignment[u] = 0
        
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment[i] for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
