import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition |{c_u, c_v}| = 2 means c_u != c_v.
    # This is equivalent to finding a 3-coloring of the graph.
    # Since n <= 40, we can use backtracking with pruning.
    
    colors = [0] * (n + 1)
    
    def is_valid(node, color):
        for neighbor in adj[node]:
            if colors[neighbor] == color:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
        
        # Heuristic: try colors in order
        for c in [1, 2, 3]:
            if is_valid(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False

    # To speed up, sort nodes by degree (most constrained first)
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to process in order of degree
    mapping = {node: i for i, node in enumerate(nodes)}
    new_adj = [[] for _ in range(n)]
    for u in range(1, n + 1):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    res_colors = [0] * n
    
    def backtrack_optimized(idx):
        if idx == n:
            return True
        
        u = idx
        # Try colors
        for c in [1, 2, 3]:
            possible = True
            for v in new_adj[u]:
                if res_colors[v] == c:
                    possible = False
                    break
            if possible:
                res_colors[u] = c
                if backtrack_optimized(idx + 1):
                    return True
                res_colors[u] = 0
        return False

    if backtrack_optimized(0):
        print("YES")
        final_colors = [0] * (n + 1)
        for i in range(n):
            final_colors[nodes[i]] = res_colors[i]
        print(*(final_colors[1:]))
    else:
        print("NO")

solve()
