```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # I(u, v) = p_u^2 + p_u*p_v + p_v^2
    # Modulo 3:
    # p in {1, 2, 3} -> p mod 3 in {1, 2, 0}
    # If p_u or p_v is 3 (mod 3 is 0):
    # I(u, v) = p_u^2 + p_u*p_v + p_v^2
    # If p_u=3, p_v=1: I = 0 + 0 + 1 = 1 (OK)
    # If p_u=3, p_v=2: I = 0 + 0 + 4 = 1 (OK)
    # If p_u=3, p_v=3: I = 0 + 0 + 0 = 0 (BAD)
    # If p_u=1, p_v=1: I = 1 + 1 + 1 = 3 = 0 (BAD)
    # If p_u=1, p_v=2: I = 1 + 2 + 4 = 7 = 1 (OK)
    # If p_u=2, p_v=2: I = 4 + 4 + 4 = 12 = 0 (BAD)
    
    # Valid pairs (p_u, p_v) mod 3:
    # (0, 1), (0, 2), (1, 0), (1, 2), (2, 0), (2, 1)
    # Mapping 3 -> 0, 1 -> 1, 2 -> 2
    # Valid: (0, 1), (0, 2), (1, 0), (1, 2), (2, 0), (2, 1)
    # Invalid: (0, 0), (1, 1), (2, 2)
    
    # We want to minimize V_n = p_1 * 4^(n-1) + p_2 * 4^(n-2) + ... + p_n
    # This is equivalent to lexicographical minimization of (p_1, p_2, ..., p_n)
    # since 4^k > sum_{i=0}^{k-1} 3 * 4^i = 4^k - 1.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, val):
        for v in adj[u]:
            if assignment[v] != 0:
                p_u = val % 3
                p_v = assignment[v] % 3
                if p_u == p_v:
                    return False
        return True

    def backtrack(u):
        if u > n:
            return True
        # Try 1, 2, 3 to minimize lexicographically
        for val in [1, 2, 3]:
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
```
