import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # Precompute valid pairs (p_u, p_v) such that p_u^2 + p_u*p_v + p_v^2 is not divisible by 3
    # Values are 1, 2, 3. Indices 0, 1, 2.
    valid_pairs = [[False] * 3 for _ in range(3)]
    for pu in range(3):
        for pv in range(3):
            val = (pu + 1)**2 + (pu + 1) * (pv + 1) + (pv + 1)**2
            if val % 3 != 0:
                valid_pairs[pu][pv] = True
    
    # DP state: dp[mask] = (min_Vn, parent_mask, last_node, last_val)
    # mask is a bitmask of visited nodes. We track the last node and its value.
    # To reconstruct, we need to know the full assignment.
    # Since n <= 40, we cannot store all assignments.
    # We use a greedy approach: for each node, try values 1, 2, 3 in order.
    # The first valid assignment found is the lexicographically smallest,
    # which corresponds to the minimum V_n.
    
    assignment = [0] * n
    
    for u in range(n):
        found_val = False
        for val in range(3):
            # Check compatibility with all previously assigned neighbors
            ok = True
            for v in adj[u]:
                if v < u and assignment[v] > 0: # Neighbor already assigned
                    if not valid_pairs[assignment[v] - 1][val]:
                        ok = False
                        break
            if ok:
                assignment[u] = val + 1
                found_val = True
                break
        if not found_val:
            print("NO")
            return
            
    print("YES")
    print(*(assignment))

if __name__ == '__main__':
    solve()
