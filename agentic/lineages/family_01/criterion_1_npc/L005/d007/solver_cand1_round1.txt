import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition: f(x) = x^3 - (p_u + p_v + 4)x^2 + (p_u p_v + 4p_u + 4p_v)x - 4p_u p_v
    # f(x) = (x - p_u)(x - p_v)(x - 4)
    # The roots are p_u, p_v, and 4.
    # For the roots to be distinct, p_u != p_v, p_u != 4, and p_v != 4.
    # Since p_i in {1, 2, 3}, p_i != 4 is always true.
    # Thus, the condition is simply p_u != p_v for all connected u, v.
    # This means the graph must be 3-colorable.
    
    # We want to minimize E = sum_{i=1}^n p_i * 4^{n-i}.
    # This is equivalent to lexicographically minimizing (p_1, p_2, ..., p_n).
    # Since we want the smallest p_1, then smallest p_2, etc.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
