```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def backtrack(unit):
        if unit > n:
            return True
        
        # Try frequencies 1, 2, 3 in order to ensure lexicographical smallest
        for freq in [1, 2, 3]:
            can_assign = True
            for neighbor in adj[unit]:
                if assignment[neighbor] == freq:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[unit] = freq
                if backtrack(unit + 1):
                    return True
                assignment[unit] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
