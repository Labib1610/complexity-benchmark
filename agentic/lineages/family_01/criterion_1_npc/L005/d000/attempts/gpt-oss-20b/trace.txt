allowed_mask = [7]*n

assigned = [0]*n

def dfs(idx):
    if idx==n: return True
    u=idx
    mask=allowed_mask[u]
    for c in (1,2,3):
        bit=1<<(c-1)
        if mask & bit:
            assigned[u]=c
            changes=[]
            conflict=False
            for v in adj[u]:
                if assigned[v]==0:
                    if allowed_mask[v] & bit:
                        prev=allowed_mask[v]
                        allowed_mask[v] ^= bit
                        changes.append((v, prev))
                        if allowed_mask[v]==0:
                            conflict=True
                            break
            if not conflict:
                if dfs(idx+1):
                    return True
            # revert
            for v,prev in changes:
                allowed_mask[v]=prev
            assigned[u]=0
    return False
