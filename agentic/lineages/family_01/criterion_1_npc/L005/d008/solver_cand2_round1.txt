import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # State vectors:
    # 1: (1, -1, 0)
    # 2: (0, 1, -1)
    # 3: (-1, 0, 1)
    # Dot products:
    # 1.1 = 1+1+0 = 2
    # 2.2 = 0+1+1 = 2
    # 3.3 = 1+0+1 = 2
    # 1.2 = 0-1+0 = -1
    # 2.3 = 0+0-1 = -1
    # 3.1 = -1+0+0 = -1
    # Constraint: dot product must be -1.
    # This means if u is 1, v must be 2.
    # If u is 2, v must be 3.
    # If u is 3, v must be 1.
    # This is equivalent to: state(v) = (state(u) % 3) + 1
    
    assignment = [0] * n
    
    for i in range(n):
        if assignment[i] == 0:
            # Try assigning state 1 to the start of a connected component
            stack = [(i, 1)]
            assignment[i] = 1
            while stack:
                u, state_u = stack.pop()
                for v in adj[u]:
                    # Required state for v is (state_u % 3) + 1
                    required_v = (state_u % 3) + 1
                    if assignment[v] == 0:
                        assignment[v] = required_v
                        stack.append((v, required_v))
                    elif assignment[v] != required_v:
                        print("NO")
                        return
    
    print("YES")
    print(*(assignment))

solve()
