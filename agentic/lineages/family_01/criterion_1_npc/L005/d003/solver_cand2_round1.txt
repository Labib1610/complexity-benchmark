import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        # Heuristic: try colors 1, 2, 3
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # To optimize, sort nodes by degree descending
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to process in order of degree
    mapping = {node: i + 1 for i, node in enumerate(nodes)}
    rev_mapping = {i + 1: node for i, node in enumerate(nodes)}
    new_adj = [[] for _ in range(n + 1)]
    for u in range(1, n + 1):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    adj = new_adj
    
    if backtrack(1):
        print("YES")
        res = [0] * n
        for i in range(1, n + 1):
            res[rev_mapping[i] - 1] = colors[i]
        print(*(res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
