import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition: (c_u + 2 * c_v) % 3 != 0
    # Since c_i in {1, 2, 3}, let x_i = c_i % 3.
    # If c_i = 3, x_i = 0.
    # The condition is (x_u + 2 * x_v) % 3 != 0.
    # If x_u = 0, then 2 * x_v % 3 != 0 => x_v != 0.
    # If x_u = 1, then (1 + 2 * x_v) % 3 != 0 => 2 * x_v != 2 => x_v != 1.
    # If x_u = 2, then (2 + 2 * x_v) % 3 != 0 => 2 * x_v != 1 => x_v != 2.
    
    # This is equivalent to saying x_u != x_v (if x_u, x_v != 0)
    # and if one is 0, the other cannot be 0.
    # Actually, let's map {1, 2, 3} to {0, 1, 2}.
    # Condition: (c_u + 2 * c_v) % 3 != 0.
    # Let c_i be in {0, 1, 2}.
    # (c_u - c_v) % 3 != 0 is not quite it.
    # Let's try backtracking with pruning.
    
    assignment = [-1] * (n + 1)
    
    def is_valid(u, val):
        for v in adj[u]:
            if assignment[v] != -1:
                if (val + 2 * assignment[v]) % 3 == 0:
                    return False
                if (assignment[v] + 2 * val) % 3 == 0:
                    return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for val in [0, 1, 2]:
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = -1
        return False

    if backtrack(1):
        print("YES")
        res = []
        for i in range(1, n + 1):
            # Map 0, 1, 2 back to 1, 2, 3
            res.append(str(assignment[i] + 1))
        print(" ".join(res))
    else:
        print("NO")

solve()
