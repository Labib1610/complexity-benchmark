import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Shard 1: [1, 1, 1] -> [0, 2, 1]
    # Shard 2: [1, 1, 1] -> [1, 0, 2]
    # Shard 3: [1, 1, 1] -> [2, 1, 0]
    
    # Let S1 = [0, 2, 1], S2 = [1, 0, 2], S3 = [2, 1, 0]
    # Collision if sum(S_u * S_v) > 3 (element-wise multiplication)
    # S1*S1 = [0, 4, 1], sum = 5 > 3 (Collision)
    # S1*S2 = [0, 0, 2], sum = 2 <= 3 (OK)
    # S1*S3 = [0, 2, 0], sum = 2 <= 3 (OK)
    # S2*S1 = [0, 0, 2], sum = 2 <= 3 (OK)
    # S2*S2 = [1, 0, 4], sum = 5 > 3 (Collision)
    # S2*S3 = [2, 0, 0], sum = 2 <= 3 (OK)
    # S3*S1 = [0, 2, 0], sum = 2 <= 3 (OK)
    # S3*S2 = [2, 0, 0], sum = 2 <= 3 (OK)
    # S3*S3 = [4, 1, 0], sum = 5 > 3 (Collision)
    
    # Conclusion: u and v cannot have the same shard ID.
    # This is equivalent to 3-coloring the graph.
    
    colors = [0] * n
    
    def is_valid(node, c):
        for neighbor in adj[node]:
            if colors[neighbor] == c:
                return False
        return True

    def backtrack(node):
        if node == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
