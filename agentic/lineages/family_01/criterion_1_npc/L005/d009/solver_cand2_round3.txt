import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Vectors: V1=(1,-1,0), V2=(0,1,-1), V3=(-1,0,1)
    # Dot products:
    # V1.V1 = 2, V2.V2 = 2, V3.V3 = 2
    # V1.V2 = -1, V2.V3 = -1, V3.V1 = -1
    # Let c_i be the shard ID (1, 2, or 3).
    # We want sum_{(u,v) in E} (W_u . W_v) = -m
    # Let x_i be the vector assigned to node i.
    # If x_i = x_j, dot product is 2.
    # If x_i != x_j, dot product is -1.
    # Let k be the number of edges where nodes have the same shard.
    # Let m-k be the number of edges where nodes have different shards.
    # Total sum = 2k - 1(m-k) = 3k - m.
    # We want 3k - m = -m => 3k = 0 => k = 0.
    # This means no two nodes connected by an edge can have the same shard.
    # This is equivalent to finding a 3-coloring of the graph.

    colors = [-1] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = -1
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
