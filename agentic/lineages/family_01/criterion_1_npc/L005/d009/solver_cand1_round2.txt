import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition L == R simplifies to:
    # (c_u * u - c_v * v)^2 + (c_u * v - c_v * u)^2 = (c_u^2 + c_v^2)(u^2 + v^2) - 2uv(c_u^2 + c_v^2)
    # Expanding L: c_u^2 u^2 - 2 c_u c_v u v + c_v^2 v^2 + c_u^2 v^2 - 2 c_u c_v u v + c_v^2 u^2
    # = c_u^2(u^2 + v^2) + c_v^2(u^2 + v^2) - 4 c_u c_v u v
    # = (c_u^2 + c_v^2)(u^2 + v^2) - 4 c_u c_v u v
    # Setting L = R:
    # (c_u^2 + c_v^2)(u^2 + v^2) - 4 c_u c_v u v = (c_u^2 + c_v^2)(u^2 + v^2) - 2uv(c_u^2 + c_v^2)
    # -4 c_u c_v u v = -2uv(c_u^2 + c_v^2)
    # 2 c_u c_v = c_u^2 + c_v^2
    # 0 = c_u^2 - 2 c_u c_v + c_v^2
    # 0 = (c_u - c_v)^2
    # This implies c_u = c_v.
    # So a collision occurs if and only if c_u == c_v.
    # We need to assign shard IDs {1, 2, 3} such that for all edges (u, v), c_u != c_v.
    # This is equivalent to 3-coloring the graph.

    colors = [0] * (n + 1)

    def is_valid(node, c):
        for neighbor in adj[node]:
            if colors[neighbor] == c:
                return False
        return True

    def backtrack(node):
        if node > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
