import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Metrics:
    # 1: {Temp, Sal, Pres}
    # 2: {Temp, Sal, Turb}
    # 3: {Temp, Pres, Turb}
    # Pair (p1, p2) covers all 4 if their union is {Temp, Sal, Pres, Turb}
    # 1 & 2: {Temp, Sal, Pres, Turb} (Yes)
    # 1 & 3: {Temp, Sal, Pres, Turb} (Yes)
    # 2 & 3: {Temp, Sal, Pres, Turb} (Yes)
    # 1 & 1: {Temp, Sal, Pres} (No)
    # 2 & 2: {Temp, Sal, Turb} (No)
    # 3 & 3: {Temp, Pres, Turb} (No)
    # So, adjacent buoys must have different firmware packages.
    # This is equivalent to 3-coloring the graph.

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
