import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Sector mapping: 0: Sector 1, 1: Sector 2, 2: Sector 3
    sectors = [i % 3 for i in range(n)]
    
    # Shard mapping: 0: Shard 1, 1: Shard 2, 2: Shard 3
    # Sector 1 (0): Shard 0->X, 1->Y, 2->Z
    # Sector 2 (1): Shard 0->Y, 1->Z, 2->X
    # Sector 3 (2): Shard 0->Z, 1->X, 2->Y
    
    # Signature mapping: 0: X, 1: Y, 2: Z
    def get_sig(sector, shard):
        return (sector + shard) % 3

    # Collision rules:
    # 1. Same sector: sig1 == sig2
    # 2. Sector 1 (0) and 2 (1): (0,1) -> X,Y (0,1); (1,2) -> Y,Z (1,2); (2,0) -> Z,X (2,0)
    #    In general: sig1 == (sig2 + 2) % 3
    # 3. Sector 2 (1) and 3 (2): (0,1) -> X,Y (0,1); (1,2) -> Y,Z (1,2); (2,0) -> Z,X (2,0)
    #    In general: sig1 == (sig2 + 2) % 3
    # 4. Sector 3 (2) and 1 (0): (0,1) -> X,Y (0,1); (1,2) -> Y,Z (1,2); (2,0) -> Z,X (2,0)
    #    In general: sig1 == (sig2 + 2) % 3
    
    # Wait, let's re-verify:
    # Sector 1 (0): S1->X(0), S2->Y(1), S3->Z(2)
    # Sector 2 (1): S1->Y(1), S2->Z(2), S3->X(0)
    # Sector 3 (2): S1->Z(2), S2->X(0), S3->Y(1)
    
    # Collision if:
    # Same sector: shard1 == shard2
    # Diff sector:
    # (0,1): S1=0,S2=1 (X,Y); S1=1,S2=2 (Y,Z); S1=2,S2=0 (Z,X) -> shard2 == (shard1 + 1) % 3
    # (1,2): S2=0,S3=1 (X,Y); S2=1,S3=2 (Y,Z); S2=2,S3=0 (Z,X) -> shard3 == (shard2 + 1) % 3
    # (2,0): S3=0,S1=1 (X,Y); S3=1,S1=2 (Y,Z); S3=2,S1=0 (Z,X) -> shard1 == (shard3 + 1) % 3

    assignment = [0] * n
    
    def backtrack(node_idx):
        if node_idx == n:
            return True
        
        for shard in range(3):
            assignment[node_idx] = shard
            possible = True
            for u, v in edges:
                if u > node_idx or v > node_idx:
                    continue
                s1, s2 = sectors[u], sectors[v]
                sh1, sh2 = assignment[u], assignment[v]
                
                if s1 == s2:
                    if sh1 == sh2:
                        possible = False; break
                else:
                    # Check if (u,v) or (v,u) matches the collision rule
                    # Rule: shard_next == (shard_prev + 1) % 3
                    # Sector order: 0->1, 1->2, 2->0
                    if (s1 + 1) % 3 == s2:
                        if sh2 == (sh1 + 1) % 3:
                            possible = False; break
                    elif (s2 + 1) % 3 == s1:
                        if sh1 == (sh2 + 1) % 3:
                            possible = False; break
            if possible:
                if backtrack(node_idx + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(a + 1 for a in assignment))
    else:
        print("NO")

solve()
