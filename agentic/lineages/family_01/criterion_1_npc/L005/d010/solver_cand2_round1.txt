import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Package 1: Write A, Read B
    # Package 2: Write B, Read C
    # Package 3: Write C, Read A
    # Unidirectional link between u and v:
    # (u writes to X, v reads from X) OR (v writes to X, u reads from X)
    # Let W(p) be write block, R(p) be read block.
    # Condition: (W(u) == R(v)) XOR (W(v) == R(u))
    # If u=1 (W=A, R=B), v=2 (W=B, R=C): W(u)=A, R(v)=C; W(v)=B, R(u)=B.
    # Condition: (A==C) XOR (B==B) -> False XOR True -> True. Valid.
    # If u=1 (W=A, R=B), v=1 (W=A, R=B): W(u)=A, R(v)=B; W(v)=A, R(u)=B.
    # Condition: (A==B) XOR (A==B) -> False XOR False -> False. Invalid.
    # If u=1 (W=A, R=B), v=3 (W=C, R=A): W(u)=A, R(v)=A; W(v)=C, R(u)=B.
    # Condition: (A==A) XOR (C==B) -> True XOR False -> True. Valid.
    # If u=2 (W=B, R=C), v=3 (W=C, R=A): W(u)=B, R(v)=A; W(v)=C, R(u)=C.
    # Condition: (B==A) XOR (C==C) -> False XOR True -> True. Valid.
    
    # Summary:
    # 1 and 2: Valid
    # 1 and 3: Valid
    # 2 and 3: Valid
    # 1 and 1: Invalid
    # 2 and 2: Invalid
    # 3 and 3: Invalid
    # So, for any edge (u, v), c[u] != c[v].
    # This is exactly the 3-coloring problem.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
