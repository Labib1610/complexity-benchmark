from itertools import product

def is_perfect_square(n):
    if n < 0:
        return False
    root = int(n**0.5)
    return root * root == n

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1  # Convert to 0-indexed
        v = int(data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
    
    # Frequencies for packages 1, 2, 3
    freq = [5409, 5616, 5200]
    
    # Precompute which package pairs have sums that are perfect squares
    valid_pairs = [[False]*3 for _ in range(3)]
    for i in range(3):
        for j in range(3):
            if is_perfect_square(freq[i] + freq[j]):
                valid_pairs[i][j] = True
    
    # Try all possible assignments (3^n possibilities)
    for assignment in product(range(3), repeat=n):
        valid = True
        for u, v in edges:
            if not valid_pairs[assignment[u]][assignment[v]]:
                valid = False
                break
        if valid:
            print("YES")
            result = [str(x + 1) for x in assignment]  # Convert back to 1-indexed packages
            print(" ".join(result))
            return
    
    print("NO")

solve()
