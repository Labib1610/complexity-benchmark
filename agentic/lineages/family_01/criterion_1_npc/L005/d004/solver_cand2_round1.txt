```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition: c_u * c_v is not a perfect square.
    # Possible states: {1, 2, 3}
    # Products:
    # 1*1 = 1 (square)
    # 2*2 = 4 (square)
    # 3*3 = 9 (square)
    # 1*2 = 2 (not square)
    # 1*3 = 3 (not square)
    # 2*3 = 6 (not square)
    #
    # So, if (u, v) are connected, they cannot have the same state.
    # This means the graph must be 3-colorable.
    # Since n <= 40, we can use backtracking to find a 3-coloring.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
