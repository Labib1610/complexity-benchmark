```python
import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        adj[u][v] = True
        adj[v][u] = True
        idx += 2
        
    # Precompute valid pairs
    valid = [[False] * 4 for _ in range(4)]
    for i in range(1, 4):
        for j in range(1, 4):
            if (i * j) not in (1, 4, 9):
                valid[i][j] = True
                
    best = None
    
    def dfs(u, curr):
        nonlocal best
        if best is not None:
            return
            
        if u == n + 1:
            best = curr[:]
            return
            
        for c in range(1, 4):
            ok = True
            for v in range(1, u):
                if adj[u][v] and not valid[c][curr[v-1]]:
                    ok = False
                    break
            if ok:
                curr.append(c)
                dfs(u + 1, curr)
                curr.pop()
                if best is not None:
                    return

    dfs(1, [])
    
    if best is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, best)))

if __name__ == '__main__':
    solve()
```
