```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Footprint: c_u^2 + c_u*c_v + c_v^2
    # Modulo 3:
    # 1^2 = 1, 2^2 = 4 = 1, 3^2 = 9 = 0
    # Pairs (c_u, c_v) mod 3:
    # (1, 1): 1 + 1 + 1 = 3 = 0 (Bad)
    # (1, 2): 1 + 2 + 1 = 4 = 1 (Good)
    # (1, 3): 1 + 3 + 0 = 4 = 1 (Good)
    # (2, 1): 4 + 2 + 1 = 7 = 1 (Good)
    # (2, 2): 4 + 4 + 4 = 12 = 0 (Bad)
    # (2, 3): 4 + 6 + 0 = 10 = 1 (Good)
    # (3, 1): 0 + 3 + 1 = 4 = 1 (Good)
    # (3, 2): 0 + 6 + 4 = 10 = 1 (Good)
    # (3, 3): 0 + 0 + 0 = 0 (Bad)
    
    # So, c_u and c_v cannot be (1,1), (2,2), or (3,3).
    # This is equivalent to saying c_u != c_v.
    # This is exactly the 3-coloring problem.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
```
