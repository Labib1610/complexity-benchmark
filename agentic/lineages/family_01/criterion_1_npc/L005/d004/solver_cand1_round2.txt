import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # P(u, v) = c_u^2 + c_u*c_v + c_v^2
    # Modulo 3:
    # If c_i = 3, c_i % 3 = 0.
    # If c_i = 1, c_i % 3 = 1.
    # If c_i = 2, c_i % 3 = 2.
    # P(u, v) % 3:
    # 1, 1 -> 1+1+1 = 3 = 0 (Bad)
    # 2, 2 -> 4+4+4 = 12 = 0 (Bad)
    # 3, 3 -> 0+0+0 = 0 (Bad)
    # 1, 2 -> 1+2+4 = 7 = 1 (Good)
    # 1, 3 -> 1+0+0 = 1 (Good)
    # 2, 3 -> 4+0+0 = 4 = 1 (Good)
    
    # So, c_u and c_v cannot be the same.
    # This is equivalent to 3-coloring the graph.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
