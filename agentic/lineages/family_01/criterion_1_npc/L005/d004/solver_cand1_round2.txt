import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Constraint: product of tuning parameters (1, 2, 3) must not be a perfect square.
    # 1*1 = 1 (sq), 2*2 = 4 (sq), 3*3 = 9 (sq)
    # 1*2 = 2, 1*3 = 3, 2*3 = 6
    # So, if u and v are connected:
    # - If p[u] == 1, p[v] cannot be 1.
    # - If p[u] == 2, p[v] cannot be 2.
    # - If p[u] == 3, p[v] cannot be 3.
    # This is equivalent to saying p[u] != p[v].
    # This is a graph coloring problem with 3 colors.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
