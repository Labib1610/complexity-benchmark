import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition Z = X + Y means that if c_u != c_v, then the count of 
    # installations with frequency c_u plus the count of installations with 
    # frequency c_v must equal the count of installations with frequency c_u 
    # or c_v. This is always true if c_u != c_v.
    # If c_u == c_v, then X = Y = Z, so X + Y = 2X, and Z = X.
    # Thus, 2X = X implies X = 0, which is impossible since at least one 
    # installation has frequency c_u.
    # Therefore, the condition is equivalent to: for every interacting pair (u, v),
    # c_u must not be equal to c_v.
    # This is exactly the definition of a graph coloring problem with 3 colors.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
