```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition: product of multipliers c_u * c_v must not be a perfect square.
    # Multipliers are {1, 2, 3}.
    # Products:
    # 1*1 = 1 (square) -> c_u != c_v if c_u=1, c_v=1
    # 2*2 = 4 (square) -> c_u != c_v if c_u=2, c_v=2
    # 3*3 = 9 (square) -> c_u != c_v if c_u=3, c_v=3
    # 1*2 = 2 (not square)
    # 1*3 = 3 (not square)
    # 2*3 = 6 (not square)
    # So, for any edge (u, v), c_u must not equal c_v.
    # This is equivalent to 3-coloring the graph.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
```
