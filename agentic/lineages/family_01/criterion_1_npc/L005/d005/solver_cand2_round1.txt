import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition: LCM(c_u, c_v) / GCD(c_u, c_v) > 1
    # This is equivalent to saying c_u != c_v.
    # Proof:
    # If c_u = c_v = k, then LCM(k, k) / GCD(k, k) = k / k = 1.
    # If c_u != c_v:
    # (1, 2) -> LCM(1, 2)/GCD(1, 2) = 2/1 = 2 > 1
    # (1, 3) -> LCM(1, 3)/GCD(1, 3) = 3/1 = 3 > 1
    # (2, 3) -> LCM(2, 3)/GCD(2, 3) = 6/1 = 6 > 1
    # So the condition is simply that no two adjacent nodes have the same color.
    # This is a 3-coloring problem. Since n <= 40, we can use backtracking.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
