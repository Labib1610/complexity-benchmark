```python
import sys
from itertools import product

def is_perfect_square(n):
    if n < 0:
        return False
    root = int(n**0.5)
    return root * root == n

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # If no constraints, any assignment works
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Try all possible assignments
    for assignment in product([1, 2, 3], repeat=n):
        valid = True
        for u, v in edges:
            product_uv = assignment[u-1] * assignment[v-1]
            if is_perfect_square(product_uv):
                valid = False
                break
        
        if valid:
            print("YES")
            print(" ".join(map(str, assignment)))
            return
    
    print("NO")

solve()
```
