from itertools import product

def is_valid_pair(pu, pv):
    sum_freq = pu + pv
    sum_squares = pu * pu + pv * pv
    return sum_squares % sum_freq != 0

def is_valid_solution(n, edges, assignment):
    for u, v in edges:
        if not is_valid_pair(assignment[u-1], assignment[v-1]):
            return False
    return True

def solve():
    n, m = map(int, input().split())
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
    
    # Try all possible assignments in lexicographical order
    for assignment in product([1, 2, 3], repeat=n):
        if is_valid_solution(n, edges, assignment):
            print("YES")
            print(" ".join(map(str, assignment)))
            return
    
    print("NO")

solve()
