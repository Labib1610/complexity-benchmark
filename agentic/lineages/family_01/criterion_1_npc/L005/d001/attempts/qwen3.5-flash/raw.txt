    # bit 1 corresponds to color 1, bit 2 to color 2, bit 3 to color 3
    # forbidden[i] == 7 (binary 111) means node i cannot be colored
    forbidden = [0] * (n + 1)

    def dfs(u):
        # If we have assigned colors to all nodes, we found a solution
        if u > n:

        if forbidden[u] == 7:
            return False
