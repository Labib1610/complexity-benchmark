def solve():
    import sys
    from collections import defaultdict

    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Try to 3-color the graph with colors 1, 2, 3
    # This is equivalent to checking if the graph is bipartite in a generalized way
    # We'll use backtracking with constraint propagation
    
    color = [0] * (n + 1)  # color[i] represents the value of dial i
    
    def is_safe(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
            
        for c in [1, 2, 3]:
            if is_safe(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = 0
        return False
    
    # Since we want lexicographically smallest, we start with node 1 and go in order
    if backtrack(1):
        print("YES")
        print(" ".join(map(str, color[1:])))
    else:
        print("NO")

solve()
