```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def backtrack(dial):
        if dial > n:
            return True
        
        # Try values 1, 2, 3 in order to ensure lexicographical smallest
        for val in range(1, 4):
            can_place = True
            for neighbor in adj[dial]:
                if assignment[neighbor] == val:
                    can_place = False
                    break
            
            if can_place:
                assignment[dial] = val
                if backtrack(dial + 1):
                    return True
                assignment[dial] = 0
        
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
