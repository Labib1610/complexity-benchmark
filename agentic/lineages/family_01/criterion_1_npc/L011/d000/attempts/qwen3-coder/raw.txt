def solve():
    import sys
    from itertools import product
    
    n, m = map(int, sys.stdin.readline().split())
    
    # Create adjacency list representation of conflicts
    conflicts = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        conflicts[u].append(v)
        conflicts[v].append(u)
    
    # Try all possible assignments (3^n possibilities)
    for assignment in product([1, 2, 3], repeat=n):
        valid = True
        # Check if this assignment violates any conflict
        for i in range(1, n + 1):
            for neighbor in conflicts[i]:
                if assignment[i-1] == assignment[neighbor-1]:
                    valid = False
                    break
            if not valid:
                break
        
        if valid:
            print("YES")
            print(' '.join(map(str, assignment)))
            return
    
    print("NO")

solve()
