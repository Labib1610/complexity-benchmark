import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    colors = [0] * n
    best_val = float('inf')
    best_config = None

    def get_matrix_val(c_list):
        # M = B_n * ... * B_1
        # B_i = [[7, c_i], [0, 1]]
        # Let M_k = B_k * ... * B_1
        # M_k = [[7^k, sum_{i=1}^k c_i * 7^{k-i}], [0, 1]]
        # The top-right element is sum_{i=1}^n c_i * 7^{n-i}
        val = 0
        for i in range(n):
            val += c_list[i] * (7**(n - 1 - i))
        return val

    def backtrack(u):
        nonlocal best_val, best_config
        if u == n:
            val = get_matrix_val(colors)
            if val < best_val:
                best_val = val
                best_config = list(colors)
            return

        # Try colors 1, 2, 3
        used = {colors[v] for v in adj[u] if colors[v] != 0}
        for c in [1, 2, 3]:
            if c not in used:
                colors[u] = c
                backtrack(u + 1)
                colors[u] = 0

    backtrack(0)

    if best_config is None:
        print("NO")
    else:
        print("YES")
        print(*(best_config))

solve()
