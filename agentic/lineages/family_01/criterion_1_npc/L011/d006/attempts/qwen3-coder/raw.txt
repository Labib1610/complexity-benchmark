from collections import defaultdict
import sys

def solve():
    n, m = map(int, input().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Try all possible colorings (1, 2, 3) for each node
    # We want the lexicographically smallest coloring
    best_coloring = None
    best_penalty = None
    
    def calculate_penalty(coloring):
        W = 0
        for c in coloring:
            W = W * 4 + c
        return W
    
    def is_valid_coloring(coloring):
        for u in range(1, n + 1):
            for v in adj[u]:
                if coloring[u-1] == coloring[v-1]:
                    return False
        return True
    
    # Generate all possible colorings in lexicographic order
    def generate_colorings():
        coloring = [0] * n
        def backtrack(pos):
            nonlocal best_coloring, best_penalty
            if pos == n:
                if is_valid_coloring(coloring):
                    penalty = calculate_penalty(coloring)
                    if best_coloring is None or penalty < best_penalty:
                        best_coloring = coloring[:]
                        best_penalty = penalty
                return
            
            # Try colors 1, 2, 3 in order
            for color in range(1, 4):
                coloring[pos] = color
                # Early pruning: check if current partial coloring is valid
                valid = True
                for v in adj[pos + 1]:
                    if v <= pos + 1 and coloring[pos] == coloring[v - 1]:
                        valid = False
                        break
                if valid:
                    backtrack(pos + 1)
        
        backtrack(0)
    
    generate_colorings()
    
    if best_coloring is None:
        print("NO")
    else:
        print("YES")
        print(*best_coloring)

solve()
