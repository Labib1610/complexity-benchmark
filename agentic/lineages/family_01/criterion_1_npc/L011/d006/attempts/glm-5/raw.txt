import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n
    
    for i in range(n):
        if color[i] != 0:
            continue
        
        queue = [i]
        color[i] = 1
        head = 0
        component_nodes = []
        
        while head < len(queue):
            u = queue[head]
            head += 1
            component_nodes.append(u)
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - color[u]
                    queue.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
        
        # Check if component is bipartite
        is_bipartite = True
        for u in component_nodes:
            for v in adj[u]:
                if color[u] == color[v]:
                    is_bipartite = False
                    break
            if not is_bipartite:
                break
        
        if is_bipartite:
            # Greedy assignment for minimum penalty
            for u in component_nodes:
                forbidden = set()
                for v in adj[u]:
                    if color[v] != 0:
                        forbidden.add(color[v])
                for c in [1, 2, 3]:
                    if c not in forbidden:
                        color[u] = c
                        break
        else:
            # Odd cycle component: use 3 colors
            # Process nodes in order, assigning smallest valid color
            for u in component_nodes:
                forbidden = set()
                for v in adj[u]:
                    if color[v] != 0:
                        forbidden.add(color[v])
                for c in [1, 2, 3]:
                    if c not in forbidden:
                        color[u] = c
                        break
    
    print("YES")
    print(' '.join(map(str, color)))

if __name__ == '__main__':
    solve()
