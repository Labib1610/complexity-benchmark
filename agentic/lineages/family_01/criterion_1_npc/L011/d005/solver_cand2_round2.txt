import sys

def solve():
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m = map(int, line1)
    except ValueError:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)

    # Stable condition: c_u * c_v is not a perfect square.
    # Possible states: {1, 2, 3}
    # Products:
    # 1*1 = 1 (square)
    # 1*2 = 2
    # 1*3 = 3
    # 2*2 = 4 (square)
    # 2*3 = 6
    # 3*3 = 9 (square)
    # So, c_u != c_v is required.
    # This means the graph must be 3-colorable.
    # We need the lexicographically smallest sequence (c_1, c_2, ..., c_n).

    assignment = [0] * n

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
