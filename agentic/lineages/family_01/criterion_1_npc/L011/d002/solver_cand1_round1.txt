import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To find the lexicographically smallest assignment, we use backtracking.
    # To avoid the performance trap mentioned, we use a heuristic:
    # Always pick the node with the most constraints (highest degree) first.
    # However, the problem asks for the lexicographically smallest sequence c1, c2, ..., cn.
    # This means we must determine c1, then c2, etc.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # The problem states that sequential order (1 to n) is a trap.
    # To get the lexicographically smallest, we must assign c1, c2, ..., cn.
    # We can use MRV (Minimum Remaining Values) or degree heuristic to prune,
    # but we must ensure the final output is the lexicographically smallest.
    # Actually, the lexicographical requirement is on the sequence (c1, c2, ..., cn).
    # This implies we should try to assign c1=1, then c2=1, etc.
    
    # To optimize, we pre-calculate the order of nodes to visit to satisfy the 
    # constraints quickly, but we must map back to the original indices.
    
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
