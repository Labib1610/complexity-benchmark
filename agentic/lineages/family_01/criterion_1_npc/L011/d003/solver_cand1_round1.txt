import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    # Heuristic: order nodes by degree (descending) to prune search space
    # However, we need the lexicographically smallest sequence for the original indices.
    # We use backtracking with MRV (Minimum Remaining Values) heuristic.
    
    def get_valid_bands(u, current_assignment):
        used = set()
        for v in adj[u]:
            if current_assignment[v] != 0:
                used.add(current_assignment[v])
        return [b for b in [1, 2, 3] if b not in used]

    def backtrack(u):
        if u == n:
            return True
        
        # To ensure lexicographical minimality, we must fill indices 0 to n-1 in order.
        # But we can use MRV to pick the next node to assign to prune the tree.
        # Wait, the problem asks for the lexicographically smallest sequence of (c1, c2, ..., cn).
        # This means we MUST try c1=1, then c1=2, then c1=3.
        # For a fixed c1, we try c2=1, then c2=2, then c2=3, and so on.
        
        valid_bands = get_valid_bands(u, assignment)
        for b in valid_bands:
            assignment[u] = b
            if backtrack(u + 1):
                return True
            assignment[u] = 0
        return False

    # The standard backtracking approach is sufficient for n=40 if we prune well.
    # Given the lexicographical requirement, we must assign in order 0, 1, ..., n-1.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
