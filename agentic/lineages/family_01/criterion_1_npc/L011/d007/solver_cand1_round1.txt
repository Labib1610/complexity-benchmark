import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The penalty function V_i = c_i + (V_{i+1}^2)/20
    # Since c_i is in {1, 2, 3}, V_i is strictly increasing with respect to c_i
    # and V_{i+1}. To minimize V_1, we should greedily minimize c_i starting from i=1.
    # However, the choice of c_i affects the constraints for future particles.
    # Since n is small (up to 40), we can use backtracking with pruning.
    
    best_c = None
    min_v = float('inf')
    
    # Precompute V values for a given sequence c
    def calculate_penalty(c):
        v = c[n-1]
        for i in range(n-2, -1, -1):
            v = c[i] + (v * v) / 20.0
        return v

    current_c = [0] * n
    
    def backtrack(idx):
        nonlocal best_c, min_v
        if idx == n:
            val = calculate_penalty(current_c)
            if val < min_v:
                min_v = val
                best_c = list(current_c)
            return

        # Try assigning 1, 2, 3
        for val in [1, 2, 3]:
            possible = True
            for neighbor in adj[idx + 1]:
                if neighbor <= idx: # neighbor already assigned
                    if current_c[neighbor - 1] == val:
                        possible = False
                        break
            if possible:
                current_c[idx] = val
                backtrack(idx + 1)
                current_c[idx] = 0

    backtrack(0)
    
    if best_c is None:
        print("NO")
    else:
        print("YES")
        print(*(best_c))

solve()
