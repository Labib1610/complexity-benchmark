def solve():
    import sys
    from collections import defaultdict
    
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list for the graph
    adj = defaultdict(list)
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1  # Convert to 0-indexed
        v = int(data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    # Try all 3^10 possibilities for the first 10 nodes
    # For remaining nodes, we'll do a greedy assignment
    assignment = [-1] * n
    
    def is_valid(player, team):
        for neighbor in adj[player]:
            if assignment[neighbor] == team:
                return False
        return True
    
    def backtrack(pos):
        if pos == n:
            return True
            
        # Try teams 1, 2, 3 in order for tie-breaking
        for team in [1, 2, 3]:
            if is_valid(pos, team):
                assignment[pos] = team
                if backtrack(pos + 1):
                    return True
                assignment[pos] = -1
        return False
    
    # Since n <= 40, but we have to be smart
    # Actually, let's just do backtracking with early pruning
    # The key is to try in order 1,2,3 for each player to satisfy tie-breaking
    
    if backtrack(0):
        print("YES")
        print(' '.join(map(str, assignment)))
    else:
        print("NO")

solve()
