```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n  # bitmask of neighbors
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    colors = [0] * n
    sys.setrecursionlimit(10000)
    found = False

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        # try colors 1..3
        neighbor_mask = adj[idx]
        for c in (1, 2, 3):
            ok = True
            # check already colored neighbors
            mask = neighbor_mask
            while mask:
                v = (mask & -mask).bit_length() - 1
                if colors[v] == c:
                    ok = False
                    break
                mask &= mask - 1
            if not ok:
                continue
            colors[idx] = c
            if dfs(idx + 1):
                return True
            colors[idx] = 0
        return False

    if dfs(0):
        print("YES")
        print(' '.join(map(str, colors)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
