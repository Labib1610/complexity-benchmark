```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def backtrack(player):
        if player > n:
            return True
        
        # Try assigning team 1, 2, 3 in order to satisfy the lexicographical requirement
        for team in range(1, 4):
            can_assign = True
            for neighbor in adj[player]:
                if assignment[neighbor] == team:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[player] = team
                if backtrack(player + 1):
                    return True
                assignment[player] = 0
        
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
