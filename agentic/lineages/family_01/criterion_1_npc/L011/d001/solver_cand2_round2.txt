import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * n
    
    # We need the lexicographically smallest assignment.
    # This means we try assigning 1, then 2, then 3 for each sensor in order 0 to n-1.
    # Since n <= 40, we use backtracking with pruning.
    
    def backtrack(u):
        if u == n:
            return True
        
        # Check which bands are forbidden by already assigned neighbors
        forbidden = 0
        for v in range(u):
            if (adj[u] & (1 << v)):
                forbidden |= (1 << (assignment[v] - 1))
        
        for band in range(1, 4):
            if not (forbidden & (1 << (band - 1))):
                assignment[u] = band
                if backtrack(u + 1):
                    return True
        return False

    # The problem asks for the lexicographically smallest sequence.
    # The standard backtracking approach with trying 1, 2, 3 in order
    # naturally finds the lexicographically smallest solution first.
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
