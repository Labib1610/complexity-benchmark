from collections import defaultdict

    edges = []
    for _ in range(m):

        v -= 1
        adj[u].append(v)

    color = [-1] * n  # -1 means uncolored

                return False
        return True

        if node == n:
            return True

                color[node] = -1  # Backtrack
        
        return False
    
    # For better performance, start with node of highest degree
    degrees = [(len(adj[i]), i) for i in range(n)]
