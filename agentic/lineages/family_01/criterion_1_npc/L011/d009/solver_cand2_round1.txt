import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Structural equations: x+y+z=2, x^2+y^2+z^2=2
    # Possible non-negative integer solutions:
    # (1, 1, 0), (1, 0, 1), (0, 1, 1)
    # Tier codes:
    # c=1 if z=0 => (1, 1, 0)
    # c=2 if y=0 => (1, 0, 1)
    # c=3 if x=0 => (0, 1, 1)
    
    vectors = {
        1: (1, 1, 0),
        2: (1, 0, 1),
        3: (0, 1, 1)
    }

    def dot(v1, v2):
        return v1[0]*v2[0] + v1[1]*v2[1] + v1[2]*v2[2]

    # We need to assign c_i in {1, 2, 3} for each node i
    # such that for all edges (u, v), dot(V_u, V_v) == 1
    
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        for c in [1, 2, 3]:
            v_u = vectors[c]
            possible = True
            for neighbor in adj[u]:
                if assignment[neighbor] != 0:
                    if dot(v_u, vectors[assignment[neighbor]]) != 1:
                        possible = False
                        break
            
            if possible:
                assignment[u] = c
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
