import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    constraints = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        constraints.append((u, v))
        idx += 2

    # A polynomial of degree at most n-1 is uniquely determined by n points.
    # Any sequence c_1, ..., c_n can be represented by a unique polynomial
    # of degree at most n-1 via Lagrange interpolation.
    # Thus, the only constraint is the perfect square condition.
    # c_i * c_j is a perfect square if:
    # 1*1=1 (sq), 1*2=2, 1*3=3, 2*2=4 (sq), 2*3=6, 3*3=9 (sq).
    # So c_i * c_j is a square iff c_i == c_j.
    # The condition "P(u) * P(v) is not a perfect square" simplifies to c_u != c_v.
    
    # This is equivalent to finding a coloring of a graph with 3 colors {1, 2, 3}
    # such that for every constraint (u, v), color[u] != color[v].
    # This is exactly the 3-coloring problem.
    
    adj = [[] for _ in range(n)]
    for u, v in constraints:
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * n
    
    def is_valid(node, c):
        for neighbor in adj[node]:
            if colors[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
