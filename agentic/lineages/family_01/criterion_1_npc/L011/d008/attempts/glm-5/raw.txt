import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u][v] = True
        adj[v][u] = True
    
    color = [0] * (n + 1)
    
    def get_available_colors(u):
        used = set()
        for v in range(1, n + 1):
            if adj[u][v] and color[v] != 0:
                used.add(color[v])
        for c in [1, 2, 3]:
            if c not in used:
                return c
        return None
    
    def check_bipartite(start_node, c1, c2):
        # Check if component containing start_node can be colored with c1 and c2
        # Returns True if possible, False otherwise
        # Uses BFS to check bipartiteness
        q = [start_node]
        side = {start_node: 0}
        head = 0
        while head < len(q):
            u = q[head]
            head += 1
            for v in range(1, n + 1):
                if adj[u][v]:
                    if v not in side:
                        side[v] = 1 - side[u]
                        q.append(v)
                    elif side[v] == side[u]:
                        return False
        return True
    
    # Try to color vertices 1 to n
    for u in range(1, n + 1):
        if color[u] != 0:
            continue
        
        # Find connected component of u
        q = [u]
        visited_comp = {u}
        head = 0
        while head < len(q):
            curr = q[head]
            head += 1
            for v in range(1, n + 1):
                if adj[curr][v] and v not in visited_comp:
                    visited_comp.add(v)
                    q.append(v)
        
        # Try to color this component with colors 1 and 2 first (to minimize lexicographic order)
        # Check if component is bipartite
        is_bipartite = check_bipartite(u, 1, 2)
        
        if is_bipartite:
            # Color with 1 and 2
            # BFS to assign colors
            color[u] = 1
            q = [u]
            head = 0
            while head < len(q):
                curr = q[head]
                head += 1
                for v in range(1, n + 1):
                    if adj[curr][v] and color[v] == 0:
                        # Find available colors for v
                        used = set()
                        for w in range(1, n + 1):
                            if adj[v][w] and color[w] != 0:
                                used.add(color[w])
                        # Assign smallest available from {1, 2}
                        for c in [1, 2]:
                            if c not in used:
                                color[v] = c
                                break
                        q.append(v)
        else:
            # Component is not bipartite, need all 3 colors
            # Use greedy approach
            for v in sorted(visited_comp):
                if color[v] == 0:
                    c = get_available_colors(v)
                    if c is None:
                        print("NO")
                        return
                    color[v] = c
    
    # Verify solution
    for i in range(1, n + 1):
        for j in range(i + 1, n + 1):
            if adj[i][j] and color[i] == color[j]:
                print("NO")
                return
    
    print("YES")
    print(' '.join(map(str, color[1:])))

solve()
