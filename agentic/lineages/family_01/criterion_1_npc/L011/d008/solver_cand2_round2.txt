import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # We need to find the lexicographically smallest assignment (c1, c2, ..., cn)
    # where each ci is in {1, 2, 3}.
    # Since n <= 40, we can use backtracking with pruning.
    
    assignment = [0] * (n + 1)

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def can_complete(u):
        if u > n:
            return True
        for color in range(1, 4):
            if is_valid(u, color):
                assignment[u] = color
                if can_complete(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Check if any solution exists
    if not can_complete(1):
        print("NO")
        return

    # Find the specific roster by greedily picking the smallest color for each player
    final_assignment = [0] * (n + 1)
    
    def find_roster(u):
        if u > n:
            return True
        for color in range(1, 4):
            if is_valid(u, color):
                assignment[u] = color
                # Check if this choice allows a valid completion for the remaining players
                if can_complete(u + 1):
                    final_assignment[u] = color
                    if find_roster(u + 1):
                        return True
                assignment[u] = 0
        return False

    # Reset assignment for the greedy search
    assignment = [0] * (n + 1)
    find_roster(1)
    
    print("YES")
    print(*(final_assignment[1:]))

solve()
