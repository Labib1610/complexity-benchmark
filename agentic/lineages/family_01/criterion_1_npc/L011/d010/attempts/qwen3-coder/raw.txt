def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Profiles: 1->(1,2,3), 2->(2,3,1), 3->(3,1,2)
    profiles = {
        1: (1, 2, 3),
        2: (2, 3, 1),
        3: (3, 1, 2)
    }
    
    # Build adjacency list for constraints
    adj = [[] for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # For each pair of profiles, precompute if they conflict
    # conflict[p1][p2] = True if profiles p1 and p2 conflict at any step
    conflict = [[False]*4 for _ in range(4)]
    for p1 in range(1, 4):
        for p2 in range(1, 4):
            for k in range(3):
                if profiles[p1][k] == profiles[p2][k]:
                    conflict[p1][p2] = True
                    break
    
    # Try to assign profiles greedily in lexicographical order
    assignment = [0] * (n + 1)
    
    def is_valid(task, profile):
        for neighbor in adj[task]:
            if assignment[neighbor] != 0:
                if conflict[profile][assignment[neighbor]]:
                    return False
        return True
    
    for task in range(1, n + 1):
        assigned = False
        for profile in range(1, 4):
            if is_valid(task, profile):
                assignment[task] = profile
                assigned = True
                break
        if not assigned:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, assignment[1:])))

solve()
