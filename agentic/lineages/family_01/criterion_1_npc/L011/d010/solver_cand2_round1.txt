import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Condition: (2^cu + 2^cv) % 7 not in {1, 2, 4}
    # Possible values for 2^c % 7:
    # c=1: 2^1 = 2
    # c=2: 2^2 = 4
    # c=3: 2^3 = 8 = 1
    # Let x = 2^cu % 7, y = 2^cv % 7.
    # x, y in {1, 2, 4}.
    # Forbidden sums (x+y)%7:
    # 1+1=2 (Forbidden)
    # 1+2=3 (OK)
    # 1+4=5 (OK)
    # 2+1=3 (OK)
    # 2+2=4 (Forbidden)
    # 2+4=6 (OK)
    # 4+1=5 (OK)
    # 4+2=6 (OK)
    # 4+4=8=1 (Forbidden)
    
    # Mapping:
    # c=1 -> val=2
    # c=2 -> val=4
    # c=3 -> val=1
    # Forbidden pairs (c_u, c_v):
    # (1, 1) -> 2+2=4 (Forbidden)
    # (2, 2) -> 4+4=1 (Forbidden)
    # (3, 3) -> 1+1=2 (Forbidden)
    
    # So c_u != c_v for all constraints (u, v).
    # This is equivalent to graph coloring with 3 colors.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
