import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Profiles:
    # 1: (1, 2, 3)
    # 2: (2, 3, 1)
    # 3: (3, 1, 2)
    # Constraint: P_u[k] != P_v[k] for k in {0, 1, 2}
    # Profile 1 vs 2: (1,2,3) vs (2,3,1) -> 1!=2, 2!=3, 3!=1 (Valid)
    # Profile 1 vs 3: (1,2,3) vs (3,1,2) -> 1!=3, 2!=1, 3!=2 (Valid)
    # Profile 2 vs 3: (2,3,1) vs (3,1,2) -> 2!=3, 3!=1, 1!=2 (Valid)
    # Profile 1 vs 1: (1,2,3) vs (1,2,3) -> 1==1 (Invalid)
    # So, P_u != P_v is the only constraint.
    
    assignment = [0] * (n + 1)
    
    def is_valid(task, profile):
        for neighbor in adj[task]:
            if assignment[neighbor] == profile:
                return False
        return True
    
    def backtrack(task):
        if task > n:
            return True
        
        for profile in range(1, 4):
            if is_valid(task, profile):
                assignment[task] = profile
                if backtrack(task + 1):
                    return True
                assignment[task] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
