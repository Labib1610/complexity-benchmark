import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition cos(2pi/3 * (cu - cv)) = -0.5
    # implies (cu - cv) mod 3 is 1 or 2.
    # This means cu != cv.
    # This is equivalent to 3-coloring the graph.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    best_config = None

    def backtrack(u):
        nonlocal best_config
        if u > n:
            current_config = tuple(colors[1:])
            if best_config is None or current_config < best_config:
                best_config = current_config
            return True
        
        found = False
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    found = True
                colors[u] = 0
        return found

    # Since we want to minimize F_n = 4*F_{n-1} + c_n,
    # this is equivalent to lexicographical minimization of (c_1, c_2, ..., c_n).
    # The backtracking naturally explores in lexicographical order if we try 1, 2, 3.
    
    if backtrack(1):
        print("YES")
        print(*(best_config))
    else:
        print("NO")

solve()
