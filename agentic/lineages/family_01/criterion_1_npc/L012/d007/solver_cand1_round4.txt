```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition det([[c_u, c_v], [1, 1]]) != 0 simplifies to c_u - c_v != 0,
    # which means c_u != c_v.
    # This is a graph coloring problem with 3 colors.
    # We want to minimize sum_{i=1}^n c_i * 2^(2n - 2i).
    # Since 2^(2n - 2i) is strictly decreasing with i, we want to make
    # c_1 as small as possible, then c_2, and so on.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
```
