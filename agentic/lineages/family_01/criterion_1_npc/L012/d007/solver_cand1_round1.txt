import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition: for every edge (u, v), {c_u, c_v} must have exactly 2 distinct values.
    # This means c_u != c_v.
    # This is equivalent to 3-coloring the graph.
    # We want to minimize sum c_i * 2^(2n - 2i).
    # This is equivalent to lexicographically minimizing (c_1, c_2, ..., c_n).
    
    colors = [0] * (n + 1)
    
    # Heuristic: Degree of uncolored nodes
    def get_next_node():
        best_node = -1
        max_deg = -1
        for i in range(1, n + 1):
            if colors[i] == 0:
                deg = 0
                for neighbor in adj[i]:
                    if colors[neighbor] == 0:
                        deg += 1
                if deg > max_deg:
                    max_deg = deg
                    best_node = i
        return best_node

    def backtrack(node):
        if node == -1:
            return True
        
        # Try colors 1, 2, 3 in order to minimize lexicographically
        forbidden = set()
        for neighbor in adj[node]:
            if colors[neighbor] != 0:
                forbidden.add(colors[neighbor])
        
        for c in [1, 2, 3]:
            if c not in forbidden:
                colors[node] = c
                next_node = get_next_node()
                if backtrack(next_node):
                    return True
                colors[node] = 0
        return False

    # To minimize lexicographically, we must pick c_1 first, then c_2, etc.
    # But the problem says "adversarial dependency structures" and "dynamically prioritize".
    # Actually, the lexicographical requirement is on the sequence (c_1, ..., c_n).
    # So we MUST pick node 1, then node 2, etc.
    
    def backtrack_lex(node):
        if node > n:
            return True
        
        forbidden = set()
        for neighbor in adj[node]:
            if neighbor < node and colors[neighbor] != 0:
                forbidden.add(colors[neighbor])
        
        for c in [1, 2, 3]:
            if c not in forbidden:
                colors[node] = c
                if backtrack_lex(node + 1):
                    return True
                colors[node] = 0
        return False

    if backtrack_lex(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
