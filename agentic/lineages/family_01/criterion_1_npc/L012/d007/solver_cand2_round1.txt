import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)

    matrices = {}
    for c in [1, 2, 3]:
        matrices[c] = [[5 + c, -3 - c], [-3 + c, 5 - c]]

    def multiply(A, B):
        C = [[0, 0], [0, 0]]
        for i in range(2):
            for j in range(2):
                for k in range(2):
                    C[i][j] += A[i][k] * B[k][j]
        return C

    best_intensity = float('inf')
    best_config = None

    # Since n <= 40, but the graph is 3-colorable, we use backtracking with pruning
    # The constraints are small enough for a well-pruned search
    colors = [0] * n

    def get_intensity(current_colors):
        res = [[1, 0], [0, 1]]
        for i in range(n):
            res = multiply(res, matrices[current_colors[i]])
        # V_out = F * [1, 0]^T = [res[0][0], res[1][0]]
        return res[0][0] + res[1][0]

    def backtrack(idx):
        nonlocal best_intensity, best_config
        if idx == n:
            intensity = get_intensity(colors)
            if intensity < best_intensity:
                best_intensity = intensity
                best_config = list(colors)
            return

        for c in [1, 2, 3]:
            valid = True
            for neighbor in adj[idx]:
                if neighbor < idx and colors[neighbor] == c:
                    valid = False
                    break
            if valid:
                colors[idx] = c
                backtrack(idx + 1)
                colors[idx] = 0

    backtrack(0)

    if best_config is None:
        print("NO")
    else:
        print("YES")
        print(*(best_config))

solve()
