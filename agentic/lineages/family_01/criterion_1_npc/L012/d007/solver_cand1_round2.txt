import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)

    # The audit hash is sum_{i=1}^n c_i * 2^(2n - 2i).
    # Since 2^(2n - 2i) is strictly decreasing with i, we want to make
    # c_1 as small as possible, then c_2, and so on.
    # This is a greedy approach with backtracking.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    # To minimize the hash, we try colors 1, 2, 3 in order for each index.
    # Because the graph might be disconnected, we process components.
    # However, the problem asks to minimize the hash based on index order.
    # This means we must decide c_1, then c_2, ..., c_n.
    
    def backtrack(idx):
        if idx == n:
            return True
        
        # Try colors 1, 2, 3
        for color in [1, 2, 3]:
            if is_valid(idx, color):
                assignment[idx] = color
                if backtrack(idx + 1):
                    return True
                assignment[idx] = 0
        return False

    # The constraints n <= 40 are small enough for backtracking if we prune well.
    # The greedy choice (1, 2, 3) is optimal for the hash.
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
