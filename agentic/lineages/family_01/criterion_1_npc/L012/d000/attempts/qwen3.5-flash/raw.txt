                            if rem[v] == 0:
                                valid = False
                                break
                
                if valid:
                    assigned[u] = c
                    if backtrack(u + 1):
                        return True
                    assigned[u] = 0  # Backtrack
                
                # Restore available colors for neighbors
                for v in modified:
                    rem[v] |= (1 << (c - 1))
        
        return False

    if backtrack(1):
        print("YES")
        print(*(assigned[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
