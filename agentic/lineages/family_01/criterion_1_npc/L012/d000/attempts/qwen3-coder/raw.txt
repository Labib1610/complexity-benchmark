def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list for the graph
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    # Try all possible assignments in lexicographical order
    # We use backtracking with early termination
    assignment = [0] * (n + 1)  # assignment[i] = tier of microservice i
    
    def is_valid(node, tier):
        for neighbor in adj[node]:
            if assignment[neighbor] == tier:
                return False
        return True
    
    def backtrack(pos):
        if pos > n:
            return True
        
        for tier in range(1, 4):  # Try tiers 1, 2, 3
            if is_valid(pos, tier):
                assignment[pos] = tier
                if backtrack(pos + 1):
                    return True
                assignment[pos] = 0  # Backtrack
                
        return False
    
    if backtrack(1):
        print("YES")
        print(' '.join(map(str, assignment[1:])))
    else:
        print("NO")

solve()
