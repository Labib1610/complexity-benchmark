import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u][v] = True
        adj[v][u] = True
    
    tiers = [0] * (n + 1)
    
    def is_valid(node, t):
        for i in range(1, node):
            if adj[node][i] and tiers[i] == t:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
        
        for t in range(1, 4):
            if is_valid(node, t):
                tiers[node] = t
                if backtrack(node + 1):
                    return True
                tiers[node] = 0
        
        return False
    
    if backtrack(1):
        print("YES")
        print(' '.join(map(str, tiers[1:])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
