import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    color = [0] * (n + 1)
    best = None
    
    def dfs(i):
        nonlocal best
        if best is not None:
            return
        if i > n:
            best = color[1:n+1][:]
            return
        used = 0
        for j in range(1, i):
            if (adj[i] >> j) & 1:
                used |= (1 << color[j])
        for c in range(1, 4):
            if not (used >> c) & 1:
                color[i] = c
                dfs(i + 1)
                if best is not None:
                    return
        color[i] = 0
    
    dfs(1)
    if best is None:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, best)))

solve()
