```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    colors = [0] * n  # 0 = uncolored, else 1..3

    def dfs(v: int) -> bool:
        if v == n:
            return True
        for c in (1, 2, 3):
            ok = True
            for nb in adj[v]:
                if nb < v and colors[nb] == c:  # only already colored neighbours
                    ok = False
                    break
            if not ok:
                continue
            colors[v] = c
            if dfs(v + 1):
                return True
            colors[v] = 0
        return False

    if dfs(0):
        out = "YES\n" + " ".join(map(str, colors))
    else:
        out = "NO"
    sys.stdout.write(out)

if __name__ == "__main__":
    solve()
```
