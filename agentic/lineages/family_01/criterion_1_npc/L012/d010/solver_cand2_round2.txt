```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning 1, 2, 3 in order for lexicographical smallest
        for val in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if v < u:
                    if assignment[v] == val:
                        possible = False
                        break
            
            if possible:
                assignment[u] = val
                if backtrack(u + 1):
                    return True
        
        assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
