```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # We want to minimize sum_{i=1}^n a_i * 4^{n-i}.
    # This is equivalent to lexicographically minimizing the sequence (a_1, a_2, ..., a_n).
    # Since we want to minimize the total sum, we greedily pick the smallest possible 
    # value for a_1, then a_2, and so on.
    
    def is_valid(u, val):
        for v in adj[u]:
            if assignment[v] == val:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        for val in [1, 2, 3]:
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
```
