We need to assign each node a tier (1, 2, or 3) such that no edge connects two nodes with the same tier.  
We also need the lexicographically smallest assignment (by node index) among all valid ones.

Since \( n \le 40 \), we can use a backtracking search with pruning.  
We’ll try assigning tiers from 1 to 3 for each node in order, checking constraints with already assigned neighbors.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    assignment = [0] * n
    
    def backtrack(idx):
        if idx == n:
            return True
        for tier in (1, 2, 3):
            ok = True
            for nb in adj[idx]:
                if assignment[nb] == tier:
                    ok = False
                    break
            if ok:
                assignment[idx] = tier
                if backtrack(idx + 1):
                    return True
                assignment[idx] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, assignment)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
