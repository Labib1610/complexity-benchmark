adj = [set() for _ in range(n)].

def dfs(idx):
    if idx==n:
        return True
    for c in [1,2,3]:
        ok = True
        for nb in adj[idx]:
            if color[nb]==c:
                ok=False
                break
        if not ok: continue
        color[idx]=c
        if dfs(idx+1):
            return True
        color[idx]=0
    return False
