import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)
    
    # We use backtracking with pruning.
    # The state is the current assignment array.
    # We try to assign tiers 1, 2, 3 to each node in order 0 to n-1.
    # Since we want the lexicographically smallest assignment, we try smaller tiers first.
    
    assignment = [0] * n
    
    def backtrack(index):
        if index == n:
            return True
        
        u = index
        # Determine available tiers for u
        # Tiers are 1, 2, 3
        # A tier t is available if for all neighbors v < u that are assigned, assignment[v] != t
        # and for all neighbors v > u, we can potentially assign a tier != t.
        # However, checking future constraints is complex. We just check past constraints here
        # and rely on recursion to handle the rest. If a future node has no options, backtrack returns False.
        
        # Check constraints with already assigned neighbors
        possible_tiers = []
        for t in range(1, 4):
            ok = True
            for v in adj[u]:
                if v < index and assignment[v] == t:
                    ok = False
                    break
            if ok:
                possible_tiers.append(t)
        
        # Try tiers in increasing order to satisfy tie-breaking (lexicographically smallest)
        for t in possible_tiers:
            assignment[u] = t
            if backtrack(index + 1):
                return True
            assignment[u] = 0 # Backtrack
        
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
