import sys

def solve():
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m = map(int, line1)
    except ValueError:
        return

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)

    assignment = [0] * (n + 1)

    def is_valid(pillar, color):
        for neighbor in adj[pillar]:
            if assignment[neighbor] == color:
                return False
        return True

    def backtrack(pillar):
        if pillar > n:
            return True
        
        for color in range(1, 4):
            if is_valid(pillar, color):
                assignment[pillar] = color
                if backtrack(pillar + 1):
                    return True
                assignment[pillar] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
