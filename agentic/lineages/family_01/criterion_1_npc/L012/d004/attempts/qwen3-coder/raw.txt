def solve():
    import sys
    from collections import defaultdict
    
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Colors: 1, 2, 3
    # We want lexicographically smallest coloring
    color = [0] * (n + 1)  # color[i] represents color of node i
    
    def is_valid(node, c):
        """Check if we can assign color c to node"""
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        """Try to color nodes from 'node' to 'n' in order"""
        if node > n:
            return True
            
        # Try colors 1, 2, 3 in order to ensure lexicographically smallest
        for c in range(1, 4):
            if is_valid(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = 0  # backtrack
                
        return False
    
    # Since we need lexicographically smallest, we must use a different approach
    # We'll model this as a constraint satisfaction problem
    # For each node, try colors 1,2,3 in order, but we need to be more careful
    
    def solve_lexicographically():
        # For each connected component, we need to 3-color it
        # We'll do this with backtracking, trying colors in order 1,2,3 for each node
        visited = [False] * (n + 1)
        
        for start_node in range(1, n + 1):
            if visited[start_node]:
                continue
                
            # BFS to get all nodes in this component
            component = []
            queue = [start_node]
            visited[start_node] = True
            
            while queue:
                node = queue.pop(0)
                component.append(node)
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        queue.append(neighbor)
            
            # Sort component nodes to ensure we process them in order
            component.sort()
            
            # Now we need to color this component with backtracking
            # But we want the lexicographically smallest assignment
            # This means we try to assign the smallest possible color to the earliest node
            
            def bt_component(nodes, idx):
                if idx == len(nodes):
                    return True
                    
                current_node = nodes[idx]
                for c in range(1, 4):
                    if is_valid(current_node, c):
                        color[current_node] = c
                        if bt_component(nodes, idx + 1):
                            return True
                        color[current_node] = 0
                return False
            
            if not bt_component(component, 0):
                return False
                
        return True
    
    # Actually, simpler approach: just do standard backtracking but process nodes in order 1 to n
    # and for each node, try colors 1,2,3 in that order. This naturally gives lex smallest.
    
    success = backtrack(1)
    
    if not success:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, color[1:])))

solve()
