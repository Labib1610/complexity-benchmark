import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # We use backtracking with pruning.
    # The goal is to find the lexicographically smallest valid assignment.
    # Since the cost function is sum(c_i * 4^(n-i)), minimizing the sum
    # is equivalent to finding the lexicographically smallest sequence (c_1, c_2, ..., c_n).
    # We assign values to nodes 0 to n-1 sequentially.
    
    assignment = [0] * n
    
    # To speed up constraint checking, we maintain the set of available colors for each node.
    # available[i] is a bitmask of allowed colors.
    # Bit 0 -> color 1, Bit 1 -> color 2, Bit 2 -> color 3.
    # Initially all colors {1, 2, 3} are available.
    available = [0b111] * n
    
    # Stack for backtracking: stores (node_index, color_assigned, list_of_affected_neighbors)
    stack = []
    
    def backtrack(k):
        # If we have assigned all n nodes, we are done.
        if k == n:
            return True
        
        # Pruning: if the current node has no available colors, backtrack.
        if available[k] == 0:
            return False
            
        # Try colors 1, 2, 3 in increasing order.
        for color in range(1, 4):
            bit = 1 << (color - 1)
            if available[k] & bit:
                # Tentatively assign color to node k
                assignment[k] = color
                
                # Track neighbors affected by this assignment to restore state later
                affected = []
                
                # Update available colors for unassigned neighbors
                for neighbor in adj[k]:
                    if neighbor > k: # We only care about unassigned nodes (indices > k)
                        if available[neighbor] & bit:
                            available[neighbor] ^= bit
                            affected.append(neighbor)
                
                # Push state to stack (conceptually, though here we just recurse)
                # Recurse to next node
                if backtrack(k + 1):
                    return True
                
                # Backtrack: restore state
                for neighbor in affected:
                    available[neighbor] |= bit
                    
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
