import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Constraint: product of tiers must not be a perfect square.
    # Possible tiers: {1, 2, 3}
    # Products:
    # 1*1 = 1 (square)
    # 1*2 = 2
    # 1*3 = 3
    # 2*1 = 2
    # 2*2 = 4 (square)
    # 2*3 = 6
    # 3*1 = 3
    # 3*2 = 6
    # 3*3 = 9 (square)
    # Invalid pairs: (1,1), (2,2), (3,3)
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, tier):
        for v in adj[u]:
            if assignment[v] != 0:
                if assignment[v] == tier:
                    return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for tier in [1, 2, 3]:
            if is_valid(u, tier):
                assignment[u] = tier
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
