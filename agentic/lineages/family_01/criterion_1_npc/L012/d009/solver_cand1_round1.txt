```python
import sys

# The condition |c_u - c_v| is a root of x^2 - 3x + 2 = 0.
# Roots are x = 1 and x = 2.
# So |c_u - c_v| must be 1 or 2.
# Since c_i in {1, 2, 3}, the only forbidden case is |c_u - c_v| = 0,
# which means c_u != c_v.
# This is a graph coloring problem: assign colors {1, 2, 3} to vertices
# such that no two adjacent vertices have the same color.
# This is equivalent to finding the lexicographically smallest 3-coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To find the lexicographically smallest coloring, we use backtracking.
    # We try to assign 1, then 2, then 3 to each node in order 1 to n.
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # Optimization: The graph might be disconnected.
    # However, the lexicographical requirement forces us to decide 
    # colors in order 1..n.
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(2000)
    solve()
```
