import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # A number has an odd number of divisors if and only if it is a perfect square.
    # p_u * p_v is a perfect square if:
    # 1. p_u = p_v (1*1=1, 2*2=4, 3*3=9)
    # 2. p_u, p_v are 1 and 4 (not possible here as states are 1, 2, 3)
    # So, p_u * p_v is a perfect square iff p_u == p_v.
    # To avoid shattering, we must have p_u != p_v for all connected u, v.
    # This is equivalent to finding a 3-coloring of the graph.
    
    # We want to minimize D_n = 4*D_{n-1} + p_n.
    # This is equivalent to minimizing the integer represented by (p_1, p_2, ..., p_n)
    # in base 4, i.e., p_1 * 4^{n-1} + p_2 * 4^{n-2} + ... + p_n * 4^0.
    # Since we want to minimize this, we should greedily pick the smallest possible
    # state for p_1, then p_2, and so on.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
