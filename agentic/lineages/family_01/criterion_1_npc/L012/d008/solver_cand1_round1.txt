import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Condition: det([[c_u, c_v], [1, 1]]) != 0 => c_u - c_v != 0 => c_u != c_v
    # This is a graph coloring problem with 3 colors.
    
    colors = [0] * n
    best_hash = float('inf')
    best_config = None

    def get_matrix(c):
        # W_i = [[7-2c, 4c-6], [3-c, 2c-2]]
        return ((7 - 2 * c, 4 * c - 6), (3 - c, 2 * c - 2))

    def multiply(A, B):
        a, b = A[0][0], A[0][1]
        c, d = A[1][0], A[1][1]
        e, f = B[0][0], B[0][1]
        g, h = B[1][0], B[1][1]
        return ((a * e + b * g, a * f + b * h), (c * e + d * g, c * f + d * h))

    def backtrack(u):
        nonlocal best_hash, best_config
        if u == n:
            # Compute G = W_n * ... * W_1
            res = ((1, 0), (0, 1))
            for i in range(n):
                res = multiply(get_matrix(colors[i]), res)
            val = res[0][1]
            if val < best_hash:
                best_hash = val
                best_config = list(colors)
            return

        for c in [1, 2, 3]:
            valid = True
            for v in adj[u]:
                if v < u and colors[v] == c:
                    valid = False
                    break
            if valid:
                colors[u] = c
                backtrack(u + 1)

    backtrack(0)

    if best_config is None:
        print("NO")
    else:
        print("YES")
        print(*(best_config))

solve()
