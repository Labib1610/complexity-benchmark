import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)

    assignment = [0] * (n + 1)

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    # The priority rules:
    # 1. Maximize usage of band 1 for smallest indices.
    # 2. Then maximize usage of band 2 for smallest indices.
    # This is equivalent to trying colors in the order 1, 2, 3 for each antenna.
    # If we try colors 1, 2, 3 in order, the first valid assignment found
    # by backtracking will be the lexicographically smallest sequence of colors.
    # Wait, the priority is:
    # - Maximize presence of 1 at index 1, then 2, then 3...
    # - Then maximize presence of 2 at index 1, then 2, then 3...
    # This means we should try colors in the order 1, 2, 3.
    # If we assign color 1 to index i, it is better than assigning 2 or 3.
    # If we assign color 2 to index i, it is better than assigning 3.
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try colors 1, 2, 3 in order
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
