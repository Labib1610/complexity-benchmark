```python
import sys

def solve():
    # The available density classes are positive integers whose squares are < 10.
    # 1^2 = 1 < 10, 2^2 = 4 < 10, 3^2 = 9 < 10.
    # So classes are {1, 2, 3}.
    classes = [1, 2, 3]
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want to minimize sum_{i=1}^n c_i * 2^{2(n-i)}.
    # This is equivalent to lexicographically minimizing the sequence (c_1, c_2, ..., c_n).
    # Since we want the smallest c_1, then smallest c_2, etc., we use backtracking.
    
    assignment = [0] * (n + 1)
    
    # To optimize, we can use a simple greedy approach with backtracking.
    # Given the constraints and the objective, we try 1, 2, 3 in order.
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning classes 1, 2, 3 in order to minimize the sequence
        for c in classes:
            possible = True
            for v in adj[u]:
                if v < u and assignment[v] == c:
                    possible = False
                    break
            
            if possible:
                assignment[u] = c
                if backtrack(u + 1):
                    return True
        
        assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
```
