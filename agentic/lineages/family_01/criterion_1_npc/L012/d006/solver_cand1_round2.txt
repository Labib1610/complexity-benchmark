import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition "exactly two distinct values" for every edge (u, v)
    # means c_u != c_v AND {c_u, c_v} != {1, 2, 3} is not possible?
    # Wait, the condition is: |{c_u, c_v}| == 2.
    # This means c_u != c_v.
    # Also, it means we cannot have {c_u, c_v} = {1, 2, 3} (impossible for 2 elements)
    # and we cannot have c_u == c_v.
    # So the condition is simply c_u != c_v.
    # Wait, let's re-read: "the set of risk tiers assigned to the two transactions 
    # must contain exactly two distinct values."
    # Yes, that is exactly c_u != c_v.
    # If c_u != c_v, then the set {c_u, c_v} has size 2.
    # So the problem is just 3-coloring a graph.
    # We want to minimize sum c_i * 2^(2n - 2i).
    # This is equivalent to lexicographically smallest sequence (c_1, c_2, ..., c_n).

    colors = [0] * (n + 1)

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
