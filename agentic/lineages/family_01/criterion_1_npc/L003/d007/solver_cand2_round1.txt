import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    # V_k = (k, k^2, k^3), W = (1, 0, 0)
    # Linear independence of V_x, V_y, W is equivalent to det([V_x, V_y, W]) != 0
    # det = | x  y  1 |
    #       | x^2 y^2 0 |
    #       | x^3 y^3 0 |
    # = 1 * (x^2 * y^3 - y^2 * x^3) = x^2 * y^2 * (y - x)
    # Since x, y in {1, 2, 3}, x^2 * y^2 is always > 0.
    # So det != 0 iff y - x != 0, i.e., x != y.
    # The constraint is simply that for any pair (u, v), mode[u] != mode[v].
    # This is a graph coloring problem with 3 colors.
    
    modes = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if modes[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            # Check energy constraint
            energy = 0
            for p in modes:
                energy += (p**4 - p**2)
            return energy <= 4000
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                modes[u] = color
                if backtrack(u + 1):
                    return True
                modes[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(modes))
    else:
        print("NO")

solve()
