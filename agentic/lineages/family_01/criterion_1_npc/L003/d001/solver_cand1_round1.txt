```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition: product of codes (1, 2, 3) is not a perfect square.
    # 1*1 = 1 (sq), 2*2 = 4 (sq), 3*3 = 9 (sq)
    # 1*2 = 2 (not sq), 1*3 = 3 (not sq), 2*3 = 6 (not sq)
    # So, if u and v are connected, they cannot have the same code.
    # This is equivalent to 3-coloring the graph.
    
    colors = [0] * n
    
    def is_safe(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for c in [1, 2, 3]:
            if is_safe(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
