```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Dot products:
    # v1·v1 = 6, v2·v2 = 6, v3·v3 = 6
    # v1·v2 = -2-2+1 = -3
    # v1·v3 = -2+1-2 = -3
    # v2·v3 = 1-2-2 = -3
    # Let x_i be the mode of antenna i.
    # If x_i == x_j, dot product is 6.
    # If x_i != x_j, dot product is -3.
    # Total interference = sum_{edges (i,j)} dot(v_i, v_j)
    # Let k be the number of edges where x_i == x_j.
    # Total interference = k * 6 + (m - k) * (-3) = 6k - 3m + 3k = 9k - 3m.
    # We want 9k - 3m = -3m, which implies 9k = 0, so k = 0.
    # This means no two connected antennas can have the same mode.
    # This is exactly the definition of a 3-coloring problem.

    colors = [-1] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = -1
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
