```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Condition: (a^2 + b^2) % 3 == (2ab) % 3
    # This is equivalent to (a^2 - 2ab + b^2) % 3 == 0
    # Which is (a - b)^2 % 3 == 0
    # Since 3 is prime, this means (a - b) % 3 == 0
    # So a % 3 == b % 3.
    # The modes are {1, 2, 3}.
    # 1 % 3 = 1
    # 2 % 3 = 2
    # 3 % 3 = 0
    # So the condition is: a % 3 != b % 3.
    # This is equivalent to saying that for every edge (u, v),
    # the modes assigned to u and v must be different modulo 3.
    # This is exactly the definition of a 3-coloring problem.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning modes 1, 2, 3
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
