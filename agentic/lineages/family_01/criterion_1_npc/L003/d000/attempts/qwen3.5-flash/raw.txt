sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))

        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * (n + 1)

    forbidden = [0] * (n + 1)
    
    # To optimize, we can store changes to forbidden array to rollback
    # But since we pass state, maybe just a stack of changes?
    # Or just pass the current state? No, mutable array is better.
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try shifts 1, 2, 3
        for c in range(1, 4):
            # Check if c is valid for u
            if (forbidden[u] >> (c - 1)) & 1:
                continue
            
            # Assign c to u
            colors[u] = c
            
            # Record changes to forbidden to rollback later
            changes = []
            valid_assignment = True
            
            # Update neighbors
            for v in adj[u]:
                if v > u:
                    old_val = forbidden[v]
                    forbidden[v] |= (1 << (c - 1))
                    changes.append((v, old_val))
                    # Check forward consistency
                    if forbidden[v] == 7: # 111 binary
                        valid_assignment = False
                        break
            
            if valid_assignment:
                if backtrack(u + 1):
                    return True
            
            # Backtrack
            for v, old_val in changes:
                forbidden[v] = old_val
            colors[u] = 0
            
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

    colors = [0] * (n + 1)

    # If bit 0 is set, shift 1 is forbidden.
    # If bit 1 is set, shift 2 is forbidden.
    # If bit 2 is set, shift 3 is forbidden.
    forbidden = [0] * (n + 1)
    
    def backtrack(u):
        # Base case: all employees assigned
        if u > n:
            return True

        for c in range(1, 4):

            if (forbidden[u] >> (c - 1)) & 1:

            colors[u] = c

            # and check if any neighbor becomes invalid (forbidden == 7)
            # We store changes to rollback later
            changes = []
            valid_branch = True
            
            for v in adj[u]:
                if v > u:
                    old_val = forbidden[v]
                    forbidden[v] |= (1 << (c - 1))
                    changes.append((v, old_val))

                    if forbidden[v] == 7:
                        valid_branch = False
                        break
            
            if valid_branch:
                if backtrack(u + 1):
                    return True
            
            # Backtrack: restore forbidden sets
            for v, old_val in changes:
                forbidden[v] = old_val
            colors[u] = 0
            
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")
