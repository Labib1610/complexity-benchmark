import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Resonance energy E(u, v) = u^v + v^u + (u * v)
    # We need E(u, v) % 3 != 0
    # Possible codes: 1, 2, 3
    # E(1, 1) = 1^1 + 1^1 + 1*1 = 3 (divisible by 3) -> Invalid
    # E(1, 2) = 1^2 + 2^1 + 1*2 = 1 + 2 + 2 = 5 (not div by 3) -> Valid
    # E(1, 3) = 1^3 + 3^1 + 1*3 = 1 + 0 + 0 = 1 (not div by 3) -> Valid
    # E(2, 2) = 2^2 + 2^2 + 2*2 = 4 + 4 + 4 = 12 (divisible by 3) -> Invalid
    # E(2, 3) = 2^3 + 3^2 + 2*3 = 8 + 0 + 0 = 8 (not div by 3) -> Valid
    # E(3, 3) = 3^3 + 3^3 + 3*3 = 0 + 0 + 0 = 0 (divisible by 3) -> Invalid
    
    # Valid pairs: (1, 2), (1, 3), (2, 3)
    # Invalid pairs: (1, 1), (2, 2), (3, 3)
    # This means no two connected nodes can have the same code.
    # This is exactly the 3-coloring problem.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
