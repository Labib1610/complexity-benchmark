import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Condition: x^2 + y^2 > 3 * min(x, y)^2
    # Let x <= y. Condition: x^2 + y^2 > 3 * x^2 => y^2 > 2 * x^2
    # Possible pairs (x, y) where x, y in {1, 2, 3}:
    # (1, 1): 1^2 + 1^2 = 2, 3 * 1^2 = 3. 2 > 3 False.
    # (1, 2): 1^2 + 2^2 = 5, 3 * 1^2 = 3. 5 > 3 True.
    # (1, 3): 1^2 + 3^2 = 10, 3 * 1^2 = 3. 10 > 3 True.
    # (2, 2): 2^2 + 2^2 = 8, 3 * 2^2 = 12. 8 > 12 False.
    # (2, 3): 2^2 + 3^2 = 13, 3 * 2^2 = 12. 13 > 12 True.
    # (3, 3): 3^2 + 3^2 = 18, 3 * 3^2 = 27. 18 > 27 False.
    
    # Forbidden pairs: (1,1), (2,2), (3,3)
    # Allowed pairs: (1,2), (1,3), (2,3)
    # This is exactly the definition of a 3-coloring problem.
    
    assignment = [0] * n
    
    def is_safe(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_safe(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
