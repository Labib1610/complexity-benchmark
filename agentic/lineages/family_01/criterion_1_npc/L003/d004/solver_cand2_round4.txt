import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Condition: floor(pu/pv) != floor(pv/pu)
    # Possible values: {1, 2, 3}
    # Pairs (pu, pv):
    # (1, 1): floor(1/1) = 1, floor(1/1) = 1. 1 != 1 is False. (Invalid)
    # (2, 2): floor(2/2) = 1, floor(2/2) = 1. 1 != 1 is False. (Invalid)
    # (3, 3): floor(3/3) = 1, floor(3/3) = 1. 1 != 1 is False. (Invalid)
    # (1, 2): floor(1/2) = 0, floor(2/1) = 2. 0 != 2 is True. (Valid)
    # (1, 3): floor(1/3) = 0, floor(3/1) = 3. 0 != 3 is True. (Valid)
    # (2, 3): floor(2/3) = 0, floor(3/2) = 1. 0 != 1 is True. (Valid)
    
    # This means no two emitters connected by an edge can have the same value.
    # This is equivalent to 3-coloring the graph.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
