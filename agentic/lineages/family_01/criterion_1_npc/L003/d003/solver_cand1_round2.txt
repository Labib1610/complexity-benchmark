import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    # We want to minimize sum s_i * 4^(n-i).
    # This is equivalent to lexicographical minimization of (s_1, s_2, ..., s_n).
    # Since n <= 40, we can use backtracking with pruning.
    
    assignment = [0] * n
    
    def backtrack(u, current_mask):
        if u == n:
            return True
        
        # Try bands 1, 2, 3 in order to minimize the value
        for band in range(1, 4):
            # Check if any neighbor already assigned has this band
            conflict = False
            for v in range(u):
                if (adj[u+1] & (1 << v)) and assignment[v] == band:
                    conflict = True
                    break
            
            if not conflict:
                assignment[u] = band
                if backtrack(u + 1, current_mask):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0, 0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
