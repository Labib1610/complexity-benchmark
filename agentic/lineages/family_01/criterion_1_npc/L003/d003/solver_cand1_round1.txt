import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # A product of two numbers has an odd number of divisors if and only if
    # the product is a perfect square.
    # Possible products:
    # 1*1=1 (sq), 1*2=2, 1*3=3
    # 2*1=2, 2*2=4 (sq), 2*3=6
    # 3*1=3, 3*2=6, 3*3=9 (sq)
    # Safe pairs (product not a square): (1,2), (1,3), (2,3), (2,1), (3,1), (3,2)
    # Forbidden pairs (product is a square): (1,1), (2,2), (3,3)
    # This means for every edge (u, v), code[u] != code[v].
    # This is exactly the definition of a 3-coloring problem.

    colors = [0] * n

    def is_safe(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            if is_safe(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
