import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try buffers 1, 2, 3 in order to minimize the sum
        # The sum is sum(b_i * 4^(n-i)). Since 4^(n-i) is positive,
        # we want the smallest b_i for the smallest i.
        # This is naturally handled by trying 1, then 2, then 3.
        
        neighbor_colors = 0
        for v in range(u):
            if (adj[u] >> v) & 1:
                neighbor_colors |= (1 << (assignment[v] - 1))
        
        for color in range(1, 4):
            if not ((neighbor_colors >> (color - 1)) & 1):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
        return False

    # The problem asks to minimize sum_{i=1}^n b_i * 4^{n-i}.
    # This is equivalent to lexicographical minimization of the sequence (b_1, b_2, ..., b_n).
    # Since we want the smallest b_1, then smallest b_2, etc.,
    # a simple greedy backtracking with 1, 2, 3 order works.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
