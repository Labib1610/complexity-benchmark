import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The system:
    # x * pu + y * pv = 1
    # x * pv + y * pu = 1
    # Determinant = pu^2 - pv^2 = (pu - pv)(pu + pv)
    # Unique solution exists if det != 0, i.e., pu != pv and pu != -pv
    # Since codes are {1, 2, 3}, pu != -pv is always true (as codes are positive).
    # Thus, the condition is simply pu != pv.
    # This is a graph coloring problem with 3 colors.
    
    colors = [0] * n
    
    def is_safe(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for c in [1, 2, 3]:
            if is_safe(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # Since n <= 40, we use backtracking with heuristic (degree ordering)
    # Sort nodes by degree descending to fail fast
    nodes = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes for backtracking
    mapping = [0] * n
    for i, node in enumerate(nodes):
        mapping[node] = i
    
    new_adj = [[] for _ in range(n)]
    for u in range(n):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    final_colors = [0] * n
    
    def backtrack_optimized(u):
        if u == n:
            return True
        
        for c in [1, 2, 3]:
            safe = True
            for v in new_adj[u]:
                if final_colors[v] == c:
                    safe = False
                    break
            if safe:
                final_colors[u] = c
                if backtrack_optimized(u + 1):
                    return True
                final_colors[u] = 0
        return False

    if backtrack_optimized(0):
        print("YES")
        res = [0] * n
        for i in range(n):
            res[nodes[i]] = final_colors[i]
        print(*(res))
    else:
        print("NO")

solve()
