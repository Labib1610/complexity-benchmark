```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Phase codes: 1, 2, 3
    # Products:
    # 1*1 = 1 (square)
    # 1*2 = 2
    # 1*3 = 3
    # 2*1 = 2
    # 2*2 = 4 (square)
    # 2*3 = 6
    # 3*1 = 3
    # 3*2 = 6
    # 3*3 = 9 (square)
    
    # Constraints:
    # If u, v share a region:
    # p[u]*p[v] != 1, 4, 9
    # This means:
    # If p[u] == 1, p[v] != 1
    # If p[u] == 2, p[v] != 2
    # If p[u] == 3, p[v] != 3
    # This is equivalent to a 3-coloring problem on the graph.
    
    colors = [0] * (n + 1)
    
    def is_safe(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            if is_safe(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
```
