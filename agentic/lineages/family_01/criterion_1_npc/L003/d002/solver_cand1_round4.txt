import sys

def solve():
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m = map(int, line1)
    except ValueError:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)

    # The condition: product of codes (1, 2, 3) must not be a perfect square.
    # 1*1 = 1 (sq), 2*2 = 4 (sq), 3*3 = 9 (sq).
    # So, no two connected emitters can have the same code.
    # This means the graph must be 3-colorable.
    # Since n <= 40, we can use backtracking with pruning.

    colors = [0] * n

    def is_safe(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            if is_safe(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # Heuristic: sort nodes by degree descending to fail faster
    nodes = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    
    # Map original indices to sorted order
    mapping = [0] * n
    for i, node in enumerate(nodes):
        mapping[node] = i
    
    new_adj = [[] for _ in range(n)]
    for u in range(n):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    # Re-run backtracking with sorted nodes
    adj = new_adj
    if backtrack(0):
        print("YES")
        res = [0] * n
        for i in range(n):
            res[nodes[i]] = colors[i]
        print(*(res))
    else:
        print("NO")

solve()
