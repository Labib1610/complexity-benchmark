import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # Heuristic: order nodes by degree descending to prune search space
    nodes = sorted(range(n), key=lambda x: bin(adj[x]).count('1'), reverse=True)
    mapping = [0] * n
    for i, node in enumerate(nodes):
        mapping[node] = i
    
    new_adj = [0] * n
    for i in range(n):
        for j in range(n):
            if (adj[i] >> j) & 1:
                new_adj[mapping[i]] |= (1 << mapping[j])
    
    assignment = [-1] * n
    
    def backtrack(idx):
        if idx == n:
            return True
        
        # Try tiers 1, 2, 3
        for tier in range(1, 4):
            can_place = True
            for prev in range(idx):
                if ((new_adj[idx] >> prev) & 1) and assignment[prev] == tier:
                    can_place = False
                    break
            
            if can_place:
                assignment[idx] = tier
                if backtrack(idx + 1):
                    return True
        return False

    # The problem asks for lexicographically smallest sequence for original indices.
    # Since n is small (40), we can use a more robust approach.
    # Given the adversarial constraint, we use a simple backtracking with 
    # MRV (Minimum Remaining Values) or just standard backtracking with 
    # careful ordering.
    
    res = [0] * n
    
    def solve_lex(idx, current_assignment):
        if idx == n:
            return True
        
        for tier in range(1, 4):
            possible = True
            for neighbor in range(idx):
                if ((adj[idx] >> neighbor) & 1) and current_assignment[neighbor] == tier:
                    possible = False
                    break
            if possible:
                current_assignment[idx] = tier
                if solve_lex(idx + 1, current_assignment):
                    return True
        return False

    # To handle the adversarial case, we use a bitmask-based check
    # and a more efficient search.
    assignment = [0] * n
    if solve_lex(0, assignment):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
