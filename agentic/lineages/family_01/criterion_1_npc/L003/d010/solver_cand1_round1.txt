import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The determinant of M_{u,v} is:
    # det = 1*(v*16 - 4*v^2) - p_u*(1*16 - 4*1) + p_u^2*(1*v - 1*1)
    # det = 16v - 4v^2 - 12p_u + p_u^2(v - 1)
    # Let's check for all pairs (p_u, p_v) in {1, 2, 3} x {1, 2, 3}
    # p_u, p_v \in {1, 2, 3}
    # det(p_u, p_v) = 16*p_v - 4*p_v^2 - 12*p_u + p_u^2*(p_v - 1)
    
    def get_det(pu, pv):
        return 16 * pv - 4 * (pv**2) - 12 * pu + (pu**2) * (pv - 1)

    # Precompute valid pairs
    valid_pairs = []
    for pu in [1, 2, 3]:
        for pv in [1, 2, 3]:
            if get_det(pu, pv) != 0:
                valid_pairs.append((pu, pv))

    assignment = [0] * n

    def backtrack(u):
        if u == n:
            return True
        
        for mode in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if assignment[v] != 0:
                    if get_det(mode, assignment[v]) == 0 or get_det(assignment[v], mode) == 0:
                        possible = False
                        break
            if possible:
                assignment[u] = mode
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
