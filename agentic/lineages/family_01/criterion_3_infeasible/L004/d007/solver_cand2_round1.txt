```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # For a 3x3 matrix Z with Z_ii = 1 and Z_ij = Z_ji:
    # Z = [[1, a, b], [a, 1, c], [b, c, 1]]
    # where a, b, c are 0 if connected, 1 if not.
    # Trace(Z^5) = 3 + 5(a^2 + b^2 + c^2) + 10abc
    # We want Trace(Z^5) = 3, so 5(a^2 + b^2 + c^2) + 10abc = 0
    # Since a, b, c are 0 or 1, this implies a=0, b=0, c=0.
    # This means all three pairs must have interference links.
    # The condition is equivalent to finding a triangle (clique of size 3).
    
    # To satisfy the lexicographical requirement:
    # Iterate u1 from 1 to n-2, u2 from u1+1 to n-1, u3 from u2+1 to n
    # Check if (u1, u2), (u1, u3), (u2, u3) are all in adj.
    
    for u1 in range(1, n - 1):
        adj_u1 = adj[u1]
        for u2 in range(u1 + 1, n):
            if u2 in adj_u1:
                adj_u2 = adj[u2]
                # We need u3 such that u3 in adj_u1 and u3 in adj_u2
                # Intersection of adj_u1 and adj_u2
                common = adj_u1.intersection(adj_u2)
                # Filter for u3 > u2
                valid_u3 = [x for x in common if x > u2]
                if valid_u3:
                    print(f"{u1} {u2} {min(valid_u3)}")
                    return

solve()
```
