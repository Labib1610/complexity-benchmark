import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triplet
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists c > b
    # such that (a, c) and (b, c) are also edges.
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b from a+1 to n, then check common neighbors c > b.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Find common neighbors c > b
            # We want the smallest c, so we intersect sorted lists
            neighbors_b = adj[b]
            
            # Optimization: iterate through the smaller neighbor list
            if len(neighbors_a) < len(neighbors_b):
                for c in neighbors_a:
                    if c > b and c in neighbors_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in neighbors_b:
                    if c > b and c in neighbors_a:
                        # Since we need the smallest c for a fixed (a, b),
                        # we should collect and sort or just find min.
                        # Actually, iterating through sorted neighbors_a is better.
                        pass
                # Correct approach for lexicographical smallest:
                for c in neighbors_a:
                    if c > b and c in neighbors_b:
                        print(f"{a} {b} {c}")
                        return

solve()
