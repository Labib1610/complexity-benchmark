import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].append(v)
    
    # Sort adjacency lists to ensure we find the lexicographically smallest sequence
    for i in range(1, n + 1):
        adj[i].sort()
        
    # To find the lexicographically smallest (a, b, c):
    # Iterate a from 1 to n
    # Iterate b in adj[a] (where b > a)
    # Iterate c in adj[a] (where c > b)
    # Check if c is in adj[b]
    
    # Convert adj[b] to a set for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Since neighbors_a is sorted, we look for c in neighbors_a[i+1:]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj_sets[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
