```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        a, c = min(u, v), max(u, v)
        edges.append((a, c))
    
    # Build adjacency list for outgoing edges
    adj = defaultdict(list)
    edge_set = set()
    for u, v in edges:
        adj[u].append(v)
        edge_set.add((u, v))
    
    # Sort adjacency lists for deterministic processing
    for u in adj:
        adj[u].sort()
    
    # Iterate through all possible triplets (a, b, c) with a < b < c
    # To find the lexicographically smallest, we iterate in order
    for a in range(1, n+1):
        if a not in adj:
            continue
        for b in adj[a]:
            if b not in adj:
                continue
            for c in adj[b]:
                # Check if edge (a, c) exists
                if (a, c) in edge_set:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
