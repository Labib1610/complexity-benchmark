import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically first trio (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, check if there exists c > b such that c is in adj[a] and adj[b]
    
    # Sort neighbors to help with lexicographical order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Only consider neighbors b > a
        valid_b = [b for b in neighbors_a if b > a]
        for i in range(len(valid_b)):
            b = valid_b[i]
            # Find common neighbors c > b
            # We can intersect the sorted lists of neighbors of a and b
            # or just iterate through neighbors of b that are > b
            neighbors_b = set(adj[b])
            for c in valid_b[i+1:]:
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
