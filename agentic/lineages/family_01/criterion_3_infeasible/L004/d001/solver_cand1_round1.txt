import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].append(v)
    
    # Sort adjacency lists to ensure we find the lexicographically smallest
    for i in range(1, n + 1):
        adj[i].sort()
        
    # We need a < b < c such that a->b, a->c, and b->c.
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # For a fixed 'a', we need to find b and c such that
        # b is in adj[a], c is in adj[a], and c is in adj[b].
        # To get lexicographically smallest, we iterate b and c in order.
        targets_of_a = adj[a]
        
        # Use a set for O(1) lookup of connections from b
        # But since we need to check if c is in adj[b], 
        # let's pre-process adj into sets for fast lookup.
        # Actually, given n=2000, we can use a boolean matrix or sets.
        # Let's use a list of sets.
        pass

    # Re-implement with sets for efficiency
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        targets_a = adj[a]
        for i in range(len(targets_a)):
            b = targets_a[i]
            # We need c > b such that c is in targets_a AND c is in adj[b]
            # To find the smallest c, iterate through targets_a[i+1:]
            for j in range(i + 1, len(targets_a)):
                c = targets_a[j]
                if c in adj_sets[b]:
