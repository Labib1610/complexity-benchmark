import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need to find a triangle (x, y, z) such that x < y < z.
    # The problem asks for the lexicographically smallest triplet (x, y, z).
    # We can iterate through all possible x, then y, then z.
    # To optimize, iterate x from 1 to n-2.
    # For a fixed x, iterate through all neighbors y of x.
    # If y > x, check if there exists a z such that z > y and z is a neighbor of both x and y.
    
    best_triplet = None
    
    for x in range(1, n + 1):
        neighbors_x = sorted(list(adj[x]))
        for i in range(len(neighbors_x)):
            y = neighbors_x[i]
            if y <= x:
                continue
            
            # We have x and y. Now look for z > y such that z is in adj[x] and adj[y].
            # To find the smallest z, we can intersect the sorted neighbor lists.
            # Since we want the lexicographically smallest (x, y, z), 
            # we should check y in increasing order.
            
            # Optimization: intersect adj[x] and adj[y]
            # Only consider z > y
            common = sorted(list(adj[x].intersection(adj[y])))
            for z in common:
                if z > y:
                    triplet = (x, y, z)
                    if best_triplet is None or triplet < best_triplet:
                        best_triplet = triplet
                    # Since we want the smallest x, then y, then z, 
                    # we can break early if we find a valid z for this (x, y)
                    # but we must ensure we check all y for this x.
                    break 
        
        if best_triplet is not None:
            # Since we iterate x from 1 upwards, the first x that yields a result
            # is the smallest possible x.
            break
            
    if best_triplet:
        print(f"{best_triplet[0]} {best_triplet[1]} {best_triplet[2]}")

solve()
