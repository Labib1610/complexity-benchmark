import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (i, j, k) with i < j < k
    # Lexicographical order: (i, j, k) < (i', j', k') if i < i' or (i == i' and j < j') or (i == i' and j == j' and k < k')
    
    best_triangle = None
    
    # Iterate through each node i as the smallest node in the triangle
    for i in range(1, n + 1):
        # Optimization: if we already found a triangle with a smaller i, we can stop
        if best_triangle and i > best_triangle[0]:
            break
            
        neighbors = sorted(list(adj[i]))
        len_n = len(neighbors)
        
        for idx_j in range(len_n):
            j = neighbors[idx_j]
            if j <= i:
                continue
            
            # Optimization: if we have a best_triangle, and i == best_triangle[0] and j > best_triangle[1], skip
            if best_triangle and i == best_triangle[0] and j > best_triangle[1]:
                continue
                
            # Check for common neighbors k > j
            # Intersection of adj[i] and adj[j]
            common = adj[i].intersection(adj[j])
            for k in common:
                if k > j:
                    if best_triangle is None or (i < best_triangle[0]) or \
                       (i == best_triangle[0] and j < best_triangle[1]) or \
                       (i == best_triangle[0] and j == best_triangle[1] and k < best_triangle[2]):
                        best_triangle = (i, j, k)
                        
    if best_triangle:
        print(f"{best_triangle[0]} {best_triangle[1]} {best_triangle[2]}")

if __name__ == '__main__':
    solve()
