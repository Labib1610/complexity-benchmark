import sys

def solve():
    # The problem asks for a set of 3 modules that cannot be distributed into 2 environments.
    # This is equivalent to finding a 3-clique (a triangle) in the conflict graph.
    # A triangle (u, v, w) means u-v, v-w, and w-u are all conflicts.
    # If we have a triangle, we cannot put 3 modules into 2 environments (Pigeonhole Principle).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all possible 'a', then 'b', then check if 'c' exists.
    # Since we need the lexicographically smallest, we iterate a from 1 to n,
    # b from a+1 to n, and check if there exists c > b such that c is in adj[a] and adj[b].
    
    # Sort adjacency lists to make finding c easier
    sorted_adj = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        for b in sorted_adj[a]:
            if b <= a:
                continue
            # Find common neighbors of a and b that are > b
            # Intersection of sorted_adj[a] and sorted_adj[b]
            # Since we need the smallest c, we can just iterate
            set_a = adj[a]
            set_b = adj[b]
            
            # Optimization: iterate over the smaller set
            if len(set_a) < len(set_b):
                for c in sorted(list(set_a)):
                    if c > b and c in set_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in sorted(list(set_b)):
                    if c > b and c in set_a:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
