import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (i, j, k) with i < j < k
    # To minimize (i, j, k) lexicographically:
    # Iterate i from 1 to n-2
    #   Iterate j from i+1 to n-1
    #     If j is connected to i:
    #       Check if there exists k > j such that k is connected to both i and j
    
    # Sort neighbors to make intersection faster or just use set intersection
    # Since we need the lexicographically smallest (i, j, k), we iterate i, j, k
    
    for i in range(1, n + 1):
        neighbors_i = sorted(list(adj[i]))
        for j in neighbors_i:
            if j <= i:
                continue
            
            # Find common neighbors k > j
            # We want the smallest k > j
            # Intersection of adj[i] and adj[j]
            common = adj[i].intersection(adj[j])
            
            best_k = float('inf')
            found = False
            for k in common:
                if k > j:
                    if k < best_k:
                        best_k = k
                        found = True
            
            if found:
                print(f"{i} {j} {best_k}")
                return

if __name__ == '__main__':
    solve()
