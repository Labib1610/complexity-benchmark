import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triplet (i, j, k) with i < j < k
    # such that i, j, k form a triangle.
    # Iterate i from 1 to n
    for i in range(1, n + 1):
        # Get neighbors of i that are > i
        neighbors_i = [v for v in adj[i] if v > i]
        neighbors_i.sort()
        
        # Iterate j in neighbors_i
        for idx_j in range(len(neighbors_i)):
            j = neighbors_i[idx_j]
            
            # Check for k in neighbors_i such that k > j and k is connected to j
            # To optimize, intersect adj[i] and adj[j]
            # Since we need the smallest triplet, we check k in sorted order
            # The intersection of adj[i] and adj[j] gives all k
            
            # Optimization: only check k > j
            # We can use set intersection
            common = adj[i].intersection(adj[j])
            valid_k = [k for k in common if k > j]
            
            if valid_k:
                k = min(valid_k)
                print(f"{i} {j} {k}")
                return

if __name__ == '__main__':
    solve()
