import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To satisfy the tie-breaking rule:
    # 1. Lowest first element
    # 2. Lowest second element
    # 3. Lowest third element
    # We iterate i from 1 to n, then j in sorted(adj[i]), then k in sorted(adj[i] & adj[j])
    
    best_triplet = None
    
    for i in range(1, n + 1):
        neighbors_i = sorted(list(adj[i]))
        for j in neighbors_i:
            if j <= i:
                continue
            
            # Check if there exists k such that k > j and k is in both adj[i] and adj[j]
            # To optimize, find intersection of adj[i] and adj[j]
            common = adj[i].intersection(adj[j])
            for k in common:
                if k > j:
                    triplet = (i, j, k)
                    if best_triplet is None or triplet < best_triplet:
                        best_triplet = triplet
        
        # If we found a triplet starting with i, and we have finished checking all j > i,
        # we can potentially stop if we are sure no better triplet exists.
        # Since we iterate i from 1 upwards, the first i that yields a triplet is the best i.
        if best_triplet is not None and best_triplet[0] == i:
            # We need to ensure we checked all j for this i to find the best j and k
            # The current loop structure does this.
            pass
        elif best_triplet is not None and best_triplet[0] < i:
            break
            
    print(f"{best_triplet[0]} {best_triplet[1]} {best_triplet[2]}")

if __name__ == '__main__':
    solve()
