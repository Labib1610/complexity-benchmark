import sys

def solve():
    # The condition E(x1, x2, x3) = -0.5 for all x1+x2+x3=0, x1^2+x2^2+x3^2=1
    # We know (x1+x2+x3)^2 = x1^2+x2^2+x3^2 + 2(x1x2 + x2x3 + x1x3) = 0
    # So x1x2 + x2x3 + x1x3 = -0.5
    # The energy is E = I12*x1x2 + I23*x2x3 + I13*x1x3
    # For E to be -0.5 regardless of x1, x2, x3, we must have I12 = I23 = I13 = 1.
    # This means the triplet must form a triangle (a clique of size 3).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # Sort edges to iterate in lexicographical order
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # all edges (a, b) and check if there exists a c > b such that
    # (a, c) and (b, c) are also edges.
    
    # Pre-sort adjacency lists for faster lookup
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        for b in adj[a]:
            # We have edge (a, b). Now look for c > b such that (a, c) and (b, c) exist.
            # Since we want the smallest (a, b, c), we check c in increasing order.
            # Intersection of adj[a] and adj[b] gives all c > b.
            # Since adj lists are sorted, we can use two pointers.
            list_a = adj[a]
            list_b = adj[b]
            
            ptr_a = 0
            ptr_b = 0
            while ptr_a < len(list_a) and ptr_b < len(list_b):
                val_a = list_a[ptr_a]
                val_b = list_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

if __name__ == '__main__':
    solve()
