import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v: u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # The quadratic form E(x1, x2, x3) = 0 has a non-zero solution if and only if
    # the determinant of the symmetric matrix associated with the quadratic form is 0.
    # The matrix M is:
    # [ 4-2I12, 2(2-S), 3(2-S) ]
    # [ 2(2-S), 16-8I13, 6(2-S) ]
    # [ 3(2-S), 6(2-S), 36-18I23 ]
    # Let k = 2-S. Matrix is:
    # [ 2(2-I12), 2k, 3k ]
    # [ 2k, 8(2-I13), 6k ]
    # [ 3k, 6k, 18(2-I23) ]
    # Determinant = 2(2-I12) * [8(2-I13)*18(2-I23) - 36k^2]
    #             - 2k * [2k*18(2-I23) - 18k^2]
    #             + 3k * [12k^2 - 3k*8(2-I13)]
    # After simplification, det = 0 if and only if S = 2.
    # So we need to find a triplet (u1, u2, u3) with u1 < u2 < u3 such that
    # I12 + I23 + I13 = 2.
    
    for i in range(1, n + 1):
        for j in sorted(list(adj[i])):
            # i < j. Look for k > j such that exactly one of (i,k) or (j,k) is an edge.
            # S = Iij + Iik + Ijk = 1 + Iik + Ijk.
            # We need S = 2, so Iik + Ijk = 1.
            # This means k must be connected to exactly one of i or j.
            
            # Check all k > j
            # To optimize, iterate over neighbors of i or j
            neighbors_i = adj[i]
            neighbors_j = adj[j]
            
            # Candidates for k are neighbors of i or neighbors of j (excluding i, j)
            candidates = (neighbors_i | neighbors_j)
            for k in sorted(list(candidates)):
                if k <= j:
                    continue
                
                s = 1 + (1 if k in neighbors_i else 0) + (1 if k in neighbors_j else 0)
                if s == 2:
                    print(f"{i} {j} {k}")
                    return

solve()
