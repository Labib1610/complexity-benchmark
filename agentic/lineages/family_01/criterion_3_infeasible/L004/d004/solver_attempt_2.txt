import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triplet
    # We want to minimize x*10^8 + y*10^4 + z
    # This is equivalent to finding the smallest x, then smallest y, then smallest z
    edges.sort()
    
    best_triplet = None
    
    # Iterate through all edges (u, v) as the first two nodes of the triplet
    # Since we want the smallest x, y, z, we iterate u from 1 to n
    for u in range(1, n + 1):
        # Optimization: if we already found a triplet with a smaller x, we can stop
        if best_triplet and u >= best_triplet[0]:
            break
            
        neighbors_u = sorted(list(adj[u]))
        for i in range(len(neighbors_u)):
            v = neighbors_u[i]
            
            # If we have a best_triplet, check if (u, v) can possibly lead to a smaller one
            if best_triplet:
                if u > best_triplet[0] or (u == best_triplet[0] and v >= best_triplet[1]):
                    continue
            
            # Find common neighbors of u and v
            # To minimize z, we look at neighbors of v that are also neighbors of u
            # Since neighbors_u is sorted, we can intersect efficiently
            # But since n is 2000, we can just iterate through neighbors of v
            neighbors_v = adj[v]
            for w in neighbors_v:
                if w > v and w in adj[u]:
                    # Found a triplet (u, v, w)
                    if best_triplet is None or (u < best_triplet[0]) or \
                       (u == best_triplet[0] and v < best_triplet[1]) or \
                       (u == best_triplet[0] and v == best_triplet[1] and w < best_triplet[2]):
                        best_triplet = (u, v, w)
    
    if best_triplet:
        print(f"{best_triplet[0]} {best_triplet[1]} {best_triplet[2]}")

solve()
