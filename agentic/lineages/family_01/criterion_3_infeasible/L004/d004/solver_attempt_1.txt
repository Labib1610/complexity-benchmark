```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # A collection of three modules that cannot be distributed across two environments
    # is a triangle (a clique of size 3) in the graph where edges represent 
    # "cannot be in the same environment".
    # We need to find all triangles (u, v, w) with u < v < w,
    # and pick the lexicographically smallest one.
    
    best = None
    
    # Iterate through all edges (u, v) with u < v
    # For each edge, check common neighbors w
    for u in range(1, n + 1):
        # Optimization: if we already found a triangle starting with a smaller u,
        # we don't need to check larger u.
        if best and u > best[0]:
            break
            
        sorted_neighbors = sorted(list(adj[u]))
        for i in range(len(sorted_neighbors)):
            v = sorted_neighbors[i]
            if v <= u:
                continue
            
            # Check if there exists w such that (u, w) and (v, w) are edges
            # To minimize lexicographical order, we want smallest u, then v, then w.
            # We only need to check w > v.
            for w in sorted_neighbors[i+1:]:
                if w in adj[v]:
                    current = (u, v, w)
                    if best is None or current < best:
                        best = current
                    # Since we want the smallest w for a fixed (u, v),
                    # and we iterate w in increasing order, we could break here
                    # but we need to be careful about the lexicographical requirement.
                    # Actually, for a fixed u, we want the smallest v, then smallest w.
                    # The current loop structure finds triangles in lexicographical order.
                    break 
            if best and best[0] == u and best[1] < v:
                break
                
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
