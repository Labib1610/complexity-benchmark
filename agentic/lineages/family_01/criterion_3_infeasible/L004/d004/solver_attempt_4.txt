import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # We iterate through all possible 'a' from 1 to n-2.
    # For a fixed 'a', we look for 'b' > 'a' such that 'b' is connected to 'a'.
    # Then we check if there exists 'c' > 'b' such that 'c' is connected to both 'a' and 'b'.
    
    # Sort edges to help with the search
    edges.sort()
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need b > a, which is guaranteed by the way we store edges
            # Now look for c > b such that c is in adj[a] and adj[b]
            neighbors_b = adj[b]
            
            # Intersection of two sorted lists
            ptr_a = i + 1
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    # Found a triangle (a, b, val_a)
                    print(f"{a} {b} {val_a}")
                    return
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

if __name__ == '__main__':
    solve()
