import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (A, B, C) such that A < B < C
    # and (A, B), (B, C), (A, C) are all in the registry.
    # To minimize (A, B, C) lexicographically:
    # Iterate A from 1 to n-2
    #   Iterate B from A+1 to n-1
    #     If (A, B) is an edge:
    #       Find common neighbors C > B
    
    for a in range(1, n + 1):
        # Sort neighbors to help with lexicographical order
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need a c such that c > b and c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can look for c in neighbors_a[i+1:]
            # and check if c is in adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
