```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u][v] = True
        adj[v][u] = True
    
    # Iterate through all triples in lexicographical order
    # The first valid triple found is the answer
    for i in range(1, n - 1):
        for j in range(i + 1, n):
            for k in range(j + 1, n + 1):
                # Check if the triple (i, j, k) forms a triangle in the graph
                # A triangle means every pair is connected by an edge
                if adj[i][j] and adj[j][k] and adj[i][k]:
                    print(i, j, k)
                    return

if __name__ == "__main__":
    solve()
```
