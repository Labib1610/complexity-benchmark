```python
import sys

def solve():
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr += 1
    m = int(input[ptr])
    ptr += 1

    adj = [[False] * (n + 1) for _ in range(n + 1)]
    deg = [0] * (n + 1)
    
    for _ in range(m):
        u = int(input[ptr])
        ptr += 1
        v = int(input[ptr])
        ptr += 1
        adj[u][v] = True
        adj[v][u] = True
        deg[u] += 1
        deg[v] += 1

    # We need to find a triangle (u, v, w) such that all three edges exist.
    # A triangle is a "bad" set because if all three pairs are co-dependent,
    # no 2-coloring (Long/Short) is possible (it's an odd cycle of length 3).
    
    # To optimize, we iterate over edges (u, v) and check for common neighbors w.
    # We order u < v to avoid duplicate checks.
    # To speed up intersection, we iterate over the neighborhood of the node with smaller degree.
    
    best = None
    
    for u in range(1, n + 1):
        # We can iterate neighbors v > u
        # To avoid iterating all neighbors and checking v > u condition inside,
        # we can just iterate v from u+1 to n and check adjacency.
        # However, iterating neighbors list is faster if graph is sparse.
        # But we need v > u.
        
        # Let's iterate v from u+1 to n
        for v in range(u + 1, n + 1):
            if not adj[u][v]:
                continue
            
            # Check for common neighbor w
            # Iterate neighbors of the node with smaller degree
            if deg[u] < deg[v]:
                source, target = u, v
            else:
                source, target = v, u
                
            for w in range(1, n + 1):
                if adj[source][w] and adj[target][w]:
                    # Found triangle u, v, w
                    # We need distinct nodes. Since u != v and adj[u][v] is true, u != v.
                    # w is connected to both, so w != u and w != v.
                    # We need to output sorted(u, v, w).
                    # We iterate u from 1 to n, v > u.
                    # w can be anything.
                    tri = sorted([u, v, w])
                    if best is None or tri < best:
                        best = tri
                        # Optimization: since we want lexicographically smallest,
                        # and we iterate u in increasing order, the first u we find
                        # is the smallest possible first element.
                        # We can break and return immediately.
                        print(f"{best[0]} {best[1]} {best[2]}")
                        return
    
    # The problem guarantees a solution exists, so this part is theoretically unreachable.
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
