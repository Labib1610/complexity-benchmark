 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [0] * (n + 1)
    edge = [bytearray(n + 1) for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v or edge[u][v]:
            continue
        edge[u][v] = 1
        edge[v][u] = 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for a in range(1, n):
        ea = edge[a]
        adja = adj[a]
        for b in range(a + 1, n + 1):
            if ea[b]:
                common = adja & adj[b] & (-1 << (b + 1))
                if common:
                    c = (common & -common).bit_length() - 1
                    sys.stdout.write(f"{a} {b} {c}\n")
                    return

if __name__ == "__main__":
    solve()
```
