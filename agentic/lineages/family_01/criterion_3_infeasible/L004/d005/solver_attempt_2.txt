import sys

def solve():
    # The problem asks for a set of three assets such that no matter how you assign
    # Long/Short, at least two are co-dependent. This is equivalent to finding
    # a triangle (a clique of size 3) in the graph of co-dependent assets.
    # If three assets form a triangle, any assignment of 2 positions (Long/Short)
    # to 3 assets must result in at least two assets having the same position
    # by the Pigeonhole Principle. If those two are connected, the limit is breached.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all possible 'a' from 1 to n.
    # Then iterate through all neighbors 'b' of 'a' such that b > a.
    # Then check if there exists a 'c' such that c > b and c is a neighbor of both a and b.
    
    # Sort neighbors to ensure we find the lexicographically smallest triangle
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c > b such that c is in adj[a] and adj[b]
            # Since adj lists are sorted, we can intersect them efficiently
            # or just iterate through neighbors of b that are > b
            neighbors_b = adj[b]
            
            # Use a set for O(1) lookup if needed, but since we want the smallest c,
            # we can just iterate through the sorted neighbors of b.
            ptr_a = 0
            ptr_b = 0
            len_a = len(neighbors_a)
            len_b = len(neighbors_b)
            
            while ptr_a < len_a and ptr_b < len_b:
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

if __name__ == '__main__':
    solve()
