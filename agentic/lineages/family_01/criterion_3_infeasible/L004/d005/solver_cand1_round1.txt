import sys

def solve():
    # The determinant of a 3x3 matrix S with 0s on the diagonal:
    # S = [[0, a, b], [a, 0, c], [b, c, 0]]
    # det(S) = 0*(0*0 - c*c) - a*(a*0 - c*b) + b*(a*c - 0*b)
    # det(S) = a*c*b + b*a*c = 2*a*b*c
    # We need det(S) = 2, so 2*a*b*c = 2 => a*b*c = 1.
    # Since a, b, c are 0 or 1, this means a=1, b=1, c=1.
    # This means the three transceivers must form a triangle (a clique of size 3).

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v: u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need to find a triangle (i, j, k) with i < j < k.
    # The problem asks for the lexicographically smallest (i, j, k).
    # Iterate i from 1 to n-2
    for i in range(1, n + 1):
        # Iterate j from i+1 to n-1
        # To optimize, only check j that are connected to i
        neighbors_i = sorted(list(adj[i]))
        for j in neighbors_i:
            if j <= i: continue
            
            # Check for k > j such that k is connected to both i and j
            # Intersection of adj[i] and adj[j]
            common = sorted(list(adj[i].intersection(adj[j])))
            for k in common:
                if k > j:
                    print(f"{i} {j} {k}")
                    return

if __name__ == '__main__':
    solve()
