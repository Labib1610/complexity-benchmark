import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # A fully locked subset of 3 pillars is a triangle (clique of size 3).
    # We need the lexicographically smallest (i, j, k) such that i < j < k
    # and (i, j), (j, k), (i, k) are all edges.
    
    # To find the lexicographically smallest, we iterate i from 1 to n-2,
    # j from i+1 to n-1, and k from j+1 to n.
    # Optimization: iterate over edges (i, j) and check if there exists k > j
    # such that k is in adj[i] and adj[j].
    
    best = None
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    # To find the lexicographically smallest (i, j, k), we can iterate i, j, k
    # but that's O(N^3). With N=2000, we need something faster.
    # Iterate i from 1 to n, then j in adj[i] where j > i, then k in adj[i] & adj[j] where k > j.
    
    for i in range(1, n + 1):
        # Optimization: if we already found a solution (i, j, k),
        # any other solution starting with i' > i will be lexicographically larger.
        # So we can stop once we finish checking all triangles starting with i.
        found_for_i = []
        
        neighbors_i = sorted([x for x in adj[i] if x > i])
        for j in neighbors_i:
            # Find common neighbors k > j
            common = adj[i].intersection(adj[j])
            for k in common:
                if k > j:
                    found_for_i.append((i, j, k))
        
        if found_for_i:
            found_for_i.sort()
            print(f"{found_for_i[0][0]} {found_for_i[0][1]} {found_for_i[0][2]}")
            return

solve()
