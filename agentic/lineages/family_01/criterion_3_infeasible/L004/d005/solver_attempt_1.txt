import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triplet (i, j, k) with i < j < k
    # such that (i, j), (i, k), and (j, k) are all in the registry.
    # We iterate i from 1 to n, j from i+1 to n, and k from j+1 to n.
    
    # Optimization: Sort adjacency lists to speed up intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for i in range(1, n + 1):
        neighbors_i = adj[i]
        for idx_j in range(len(neighbors_i)):
            j = neighbors_i[idx_j]
            if j <= i:
                continue
            
            # We need k such that k > j and k is in adj[i] and adj[j]
            # Since adj lists are sorted, we can find intersection efficiently
            neighbors_j = adj[j]
            
            # Use two pointers or set intersection to find common neighbors > j
            ptr_i = 0
            ptr_j = 0
            len_i = len(neighbors_i)
            len_j = len(neighbors_j)
            
            while ptr_i < len_i and ptr_j < len_j:
                val_i = neighbors_i[ptr_i]
                val_j = neighbors_j[ptr_j]
                
                if val_i == val_j:
                    if val_i > j:
                        print(f"{i} {j} {val_i}")
                        return
                    ptr_i += 1
                    ptr_j += 1
                elif val_i < val_j:
                    ptr_i += 1
                else:
                    ptr_j += 1

if __name__ == '__main__':
    solve()
