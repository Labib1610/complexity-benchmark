import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # The matrix W for {u1, u2, u3} is:
    # [[-1, a12, a13],
    #  [a12, -1, a23],
    #  [a13, a23, -1]]
    # where aij = 1 if connected, 0 otherwise.
    # det(W) = -1(1 - a23^2) - a12(-a12 - a13*a23) + a13(a12*a23 + a13)
    # det(W) = -1 + a23^2 + a12^2 + a12*a13*a23 + a12*a13*a23 + a13^2
    # det(W) = a12^2 + a13^2 + a23^2 + 2*a12*a13*a23 - 1
    # We want det(W) = 4, so a12^2 + a13^2 + a23^2 + 2*a12*a13*a23 = 5
    # Since aij is 0 or 1, the only way to get 5 is if a12=1, a13=1, a23=1.
    # 1^2 + 1^2 + 1^2 + 2(1*1*1) = 3 + 2 = 5.
    # So we are looking for a triangle (a clique of size 3).
    
    # To find the lexicographically smallest (a, b, c), iterate a, then b, then c.
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Sort neighbors to ensure we check b and c in order
        sorted_neighbors_a = sorted(list(neighbors_a))
        for i in range(len(sorted_neighbors_a)):
            b = sorted_neighbors_a[i]
            if b <= a:
                continue
            
            # We need c > b such that c is connected to both a and b
            neighbors_b = adj[b]
            # Intersection of neighbors_a and neighbors_b
            # We only care about c > b
            common = neighbors_a.intersection(neighbors_b)
            for c in sorted(list(common)):
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
