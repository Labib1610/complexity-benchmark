```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c
    # We iterate through all edges (u, v) where u < v.
    # For each edge, we look for a common neighbor w.
    # To ensure lexicographical order, we check potential triplets.
    
    best = None
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    for u, v in edges:
        # We want to find w such that (u, v, w) form a triangle.
        # The triplet will be sorted as (min(u,v,w), mid(u,v,w), max(u,v,w))
        # We iterate through all neighbors of u and check if they are in v's neighbors
        
        # To optimize, iterate over the smaller adjacency list
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
            
        for w in adj[u]:
            if w in adj[v]:
                # Found a triangle (u, v, w)
                triplet = sorted([u, v, w])
                if best is None or triplet < best:
                    best = triplet
                    
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
