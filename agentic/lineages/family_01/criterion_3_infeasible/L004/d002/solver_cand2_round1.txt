import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # We iterate through all possible a from 1 to n-2.
    # For a fixed a, we look for b > a such that (a, b) is an edge.
    # Then we look for c > b such that (a, c) and (b, c) are edges.
    
    # Sort edges to help find smallest a, b, c
    edges.sort()
    
    # Pre-sort adjacency lists for faster lookup and ordered iteration
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through all possible a
    for a in range(1, n + 1):
        # Neighbors of a that are greater than a
        neighbors_a = adj[a]
        
        # Iterate through all b in neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Since adj[a] and adj[b] are sorted, we can use two pointers or set intersection
            # Given n <= 2000, set intersection is efficient enough
            
            # Find intersection of adj[a] and adj[b]
            # Only consider elements > b
            set_a = set(neighbors_a[i+1:])
            set_b = set(adj[b])
            
            common = sorted(list(set_a.intersection(set_b)))
            
            if common:
                print(f"{a} {b} {common[0]}")
                return

if __name__ == '__main__':
    solve()
