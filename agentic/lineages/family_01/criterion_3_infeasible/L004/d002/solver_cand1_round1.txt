import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triplet (i, j, k) with i < j < k
    # We iterate i from 1 to n-2
    # Then j from i+1 to n-1
    # Then check if there exists k > j such that k is in adj[i] and adj[j]
    
    # Sort adjacency lists to potentially speed up intersection
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for i in range(1, n + 1):
        neighbors_i = sorted_adj[i]
        for idx_j, j in enumerate(neighbors_i):
            if j <= i:
                continue
            
            # We need to find k > j such that k is in adj[i] and adj[j]
            # Since neighbors_i is sorted, we can intersect efficiently
            neighbors_j = adj[j]
            
            # Check neighbors of i that are greater than j
            for k in neighbors_i[idx_j + 1:]:
                if k in neighbors_j:
                    # Found the lexicographically smallest triplet
                    res = sorted([i, j, k])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
