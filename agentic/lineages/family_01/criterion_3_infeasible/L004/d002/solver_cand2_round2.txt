```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # adj[u] will store a list of v such that u < v
    # We want to find a < b < c such that (a,b), (a,c), (b,c) are edges.
    # This means b is in adj[a] and c is in adj[a] AND c is in adj[b].
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].append(v)
    
    # Sort adjacency lists to ensure we find the lexicographically smallest triad
    for i in range(1, n + 1):
        adj[i].sort()
        
    # To find the lexicographically smallest (a, b, c):
    # Iterate a from 1 to n-2
    # Iterate b in adj[a]
    # Iterate c in adj[a] such that c > b
    # Check if c is in adj[b]
    
    # To make the check "c in adj[b]" efficient, use a set or boolean array
    # Since n <= 2000, a 2D boolean array or list of sets is fine.
    is_edge = [set() for _ in range(n + 1)]
    for u in range(1, n + 1):
        for v in adj[u]:
            is_edge[u].add(v)
            
    for a in range(1, n + 1):
        for b in adj[a]:
            for c in adj[a]:
                if c > b:
                    if c in is_edge[b]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
```
