```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # We need to find a triangle (u, v, w) such that (u,v), (u,w), (v,w) are edges.
    # To find the lexicographically smallest (u, v, w), we iterate u from 1 to n,
    # then v from u+1 to n, then w from v+1 to n.
    
    # Sort edges to help with finding triangles efficiently
    # For each edge (u, v), check if there exists w such that (u, w) and (v, w) exist.
    
    # To find the lexicographically smallest, we can iterate through all possible
    # triangles. Since n is up to 2000, O(n^3) is too slow, but we can iterate
    # over edges and check common neighbors.
    
    best_triangle = None
    
    # Sort vertices by degree or just iterate
    # For each u, iterate over neighbors v > u
    for u in range(1, n + 1):
        neighbors_u = sorted(list(adj[u]))
        for i in range(len(neighbors_u)):
            v = neighbors_u[i]
            if v <= u: continue
            
            # Check common neighbors w > v
            # Intersection of adj[u] and adj[v]
            common = adj[u].intersection(adj[v])
            for w in common:
                if w > v:
                    triangle = (u, v, w)
                    if best_triangle is None or triangle < best_triangle:
                        best_triangle = triangle
        
        # Optimization: if we found a triangle starting with u, 
        # we don't necessarily need to check higher u, but we need the lexicographically smallest.
        # Since we iterate u from 1, the first u that yields a triangle is the smallest u.
        if best_triangle is not None and best_triangle[0] < u:
            break
            
    if best_triangle:
        print(f"{best_triangle[0]} {best_triangle[1]} {best_triangle[2]}")

solve()
```
