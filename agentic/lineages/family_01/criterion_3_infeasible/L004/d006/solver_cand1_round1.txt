```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # Let U = {a, b, c}.
    # k(S) is the number of edges with both endpoints in S.
    # Let e_ab = 1 if (a,b) is an edge, else 0.
    # k({a,b}) = e_ab, k({a,c}) = e_ac, k({b,c}) = e_bc
    # k({a,b,c}) = e_ab + e_ac + e_bc
    # k({a}) = k({b}) = k({c}) = k(empty) = 0
    # Capacity = 2^0 + 2^0 + 2^0 + 2^0 + 2^e_ab + 2^e_ac + 2^e_bc + 2^(e_ab+e_ac+e_bc)
    # Capacity = 4 + 2^e_ab + 2^e_ac + 2^e_bc + 2^(e_ab+e_ac+e_bc) = 18
    # Let x = e_ab, y = e_ac, z = e_bc
    # 4 + 2^x + 2^y + 2^z + 2^(x+y+z) = 18
    # 2^x + 2^y + 2^z + 2^(x+y+z) = 14
    # Possible values for x, y, z are 0 or 1.
    # If x=1, y=1, z=0: 2+2+1 + 2^2 = 5 + 4 = 9 != 14
    # If x=1, y=1, z=1: 2+2+2 + 2^3 = 6 + 8 = 14. Correct.
    # So the condition is equivalent to the triplet forming a triangle (3 edges).

    # We need to find a triangle (a, b, c) with a < b < c.
    # Lexicographical order: smallest a, then smallest b, then smallest c.
    
    adj_list = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        neighbors_a = adj_list[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a: continue
            
            # Check for c > b such that (a,c) and (b,c) are edges
            # Intersection of neighbors of a and b
            set_a = adj[a]
            set_b = adj[b]
            
            # To optimize, iterate over the smaller neighbor list
            if len(set_a) < len(set_b):
                for c in set_a:
                    if c > b and c in set_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in set_b:
                    if c > b and c in set_a:
                        print(f"{a} {b} {c}")
                        return

solve()
```
