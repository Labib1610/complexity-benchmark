```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (i, j, k) with i < j < k
    # To minimize (i, j, k) lexicographically, we iterate i from 1 to n,
    # then j from i+1 to n, then check if k exists in adj[i] and adj[j].
    
    for i in range(1, n + 1):
        # Sort neighbors to ensure we find the smallest k efficiently
        neighbors_i = sorted(list(adj[i]))
        for j_idx in range(len(neighbors_i)):
            j = neighbors_i[j_idx]
            if j <= i:
                continue
            
            # Find common neighbors of i and j
            common = sorted(list(adj[i].intersection(adj[j])))
            for k in common:
                if k > j:
                    print(f"{i} {j} {k}")
                    return

if __name__ == '__main__':
    solve()
```
