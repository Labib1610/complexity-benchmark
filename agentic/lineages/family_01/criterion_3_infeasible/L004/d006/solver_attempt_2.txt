import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A set of three compounds {a, b, c} cannot be stored in two vaults
    # if and only if they form a triangle (a-b, b-c, c-a).
    # If they form a triangle, any distribution into two vaults will
    # force at least two of them into the same vault, which is a reactive pair.
    
    # We need to find all triangles and pick the lexicographically smallest one.
    # A triangle is (i, j, k) with i < j < k.
    # To minimize (i, j, k) lexicographically:
    # Iterate i from 1 to n-2
    #   Iterate j from i+1 to n-1
    #     If i and j are connected:
    #       Find common neighbors k > j
    
    for i in range(1, n + 1):
        adj_i = adj[i]
        # Sort neighbors to ensure we check in order
        neighbors_i = sorted([x for x in adj_i if x > i])
        for idx_j, j in enumerate(neighbors_i):
            # Check if there exists k > j such that k is in adj[i] and adj[j]
            # Intersection of neighbors_i[idx_j+1:] and adj[j]
            adj_j = adj[j]
            for k in neighbors_i[idx_j + 1:]:
                if k in adj_j:
                    print(f"{i} {j} {k}")
                    return

if __name__ == '__main__':
    solve()
