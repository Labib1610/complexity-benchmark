import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A subset of 3 nodes {a, b, c} has a resonance loop if:
    # Let x, y, z be tuning values for a, b, c.
    # Sum of neighbors in subset = 2 * own value.
    # Let A be the adjacency matrix of the subset.
    # A * [x, y, z]^T = 2 * [x, y, z]^T
    # This means 2 is an eigenvalue of the adjacency matrix of the subgraph induced by {a, b, c}.
    
    # Possible subgraphs of 3 nodes:
    # 0 edges: A = [[0,0,0],[0,0,0],[0,0,0]]. Eigenvalues: 0, 0, 0.
    # 1 edge: A = [[0,1,0],[1,0,0],[0,0,0]]. Eigenvalues: 1, -1, 0.
    # 2 edges (path of 2): A = [[0,1,0],[1,0,1],[0,1,0]]. Eigenvalues: sqrt(2), 0, -sqrt(2).
    # 3 edges (triangle): A = [[0,1,1],[1,0,1],[1,1,0]]. Eigenvalues: 2, -1, -1.
    
    # The only case where 2 is an eigenvalue is the triangle (3 edges).
    # So we need to find a triangle {a, b, c} such that 2 is an eigenvalue.
    # For a triangle, the adjacency matrix is:
    # [[0, 1, 1],
    #  [1, 0, 1],
    #  [1, 1, 0]]
    # Eigenvalues are 2, -1, -1.
    # The condition is satisfied if and only if the subset forms a triangle.
    
    # We need the lexicographically smallest triangle.
    # Iterate through all possible triangles (a, b, c) with a < b < c.
    
    # To optimize, iterate a from 1 to n, b from a+1 to n, c from b+1 to n.
    # Check if b in adj[a], c in adj[a], c in adj[b].
    
    for a in range(1, n + 1):
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Check common neighbors of a and b that are > b
            for c in neighbors_a[i+1:]:
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
