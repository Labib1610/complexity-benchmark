import sys
from itertools import combinations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    edge_set = set()
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        edge_set.add((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Build list of neighbors for each node
    # We'll iterate through all combinations of 3 nodes
    
    # To optimize, we can go through all combinations of 3 nodes
    # But since n <= 2000, combinations could be too much
    # Instead, we can iterate through each node, then pairs of its neighbors
    
    found = False
    result = None
    
    # Try method: for each node, look at pairs of its neighbors
    candidates = []
    for x in range(1, n+1):
        if len(adj[x]) >= 2:
            neighbors = list(adj[x])
            for i in range(len(neighbors)):
                for j in range(i+1, len(neighbors)):
                    u = neighbors[i]
                    v = neighbors[j]
                    # Now we have a triangle: x, u, v
                    triplet = tuple(sorted([x, u, v]))
                    # Check if we already considered this triplet
                    # To avoid duplicates, ensure x is the smallest
                    if x < u and x < v:
                        candidates.append(triplet)
    
    # Since the number of triangles can still be large, we sort and check in order
    candidates.sort()
    
    for triplet in candidates:
        x, y, z = triplet
        count_edges = 0
        edges_in_subset = []
        edge_pairs = [(x,y), (x,z), (y,z)]
        for u,v in edge_pairs:
            if u > v:
                u,v = v,u
            if (u,v) in edge_set:
                count_edges += 1
                edges_in_subset.append((u,v))
        
        # Compute interference capacity
        # All 8 subsets of {x,y,z}
        total = 0
        # Generate all subsets implicitly
        # S = {}
        k_empty = 0
        total += 1 << k_empty
        # S = {x}
        k_x = 0
        total += 1 << k_x
        # S = {y}
        k_y = 0
        total += 1 << k_y
        # S = {z}
        k_z = 0
        total += 1 << k_z
        # S = {x,y}
        k_xy = 1 if (min(x,y), max(x,y)) in edge_set else 0
        total += 1 << k_xy
        # S = {x,z}
        k_xz = 1 if (min(x,z), max(x,z)) in edge_set else 0
        total += 1 << k_xz
        # S = {y,z}
        k_yz = 1 if (min(y,z), max(y,z)) in edge_set else 0
        total += 1 << k_yz
        # S = {x,y,z}
        k_xyz = count_edges
        total += 1 << k_xyz
        
        if total == 18:
            print(x, y, z)
            return

    # Fallback: check all combinations if above doesn't work
    # This should not happen given constraints but just in case
    for comb in combinations(range(1,n+1), 3):
        x, y, z = sorted(comb)
        count_edges = 0
        edge_pairs = [(x,y), (x,z), (y,z)]
        for u,v in edge_pairs:
            if u > v:
                u,v = v,u
            if (u,v) in edge_set:
                count_edges += 1
        
        # Compute interference capacity
        total = 0
        total += 1 << 0  # {}
        total += 1 << 0  # {x}
        total += 1 << 0  # {y}
        total += 1 << 0  # {z}
        k_xy = 1 if (min(x,y), max(x,y)) in edge_set else 0
        total += 1 << k_xy
        k_xz = 1 if (min(x,z), max(x,z)) in edge_set else 0
        total += 1 << k_xz
        k_yz = 1 if (min(y,z), max(y,z)) in edge_set else 0
        total += 1 << k_yz
        k_xyz = count_edges
        total += 1 << k_xyz
        
        if total == 18:
            print(x, y, z)
            return

if __name__ == "__main__":
    main()
