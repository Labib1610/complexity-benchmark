import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # For a triplet {u1, u2, u3}, let e12 = 1 if (u1, u2) is an edge, else 0.
    # d1 = e12 + e13, d2 = e12 + e23, d3 = e13 + e23.
    # W = [[2+d1, -2*e12, -2*e13],
    #      [-2*e12, 2+d2, -2*e23],
    #      [-2*e13, -2*e23, 2+d3]]
    # det(W) = 0 is the condition.
    
    for i in range(1, n + 1):
        for j in range(i + 1, n + 1):
            e12 = 1 if j in adj[i] else 0
            for k in range(j + 1, n + 1):
                e13 = 1 if k in adj[i] else 0
                e23 = 1 if k in adj[j] else 0
                
                d1 = e12 + e13
                d2 = e12 + e23
                d3 = e13 + e23
                
                # Matrix W:
                # w11 = 2+d1, w12 = -2*e12, w13 = -2*e13
                # w21 = -2*e12, w22 = 2+d2, w23 = -2*e23
                # w31 = -2*e13, w32 = -2*e23, w33 = 2+d3
                
                w11, w12, w13 = 2 + d1, -2 * e12, -2 * e13
                w21, w22, w23 = -2 * e12, 2 + d2, -2 * e23
                w31, w32, w33 = -2 * e13, -2 * e23, 2 + d3
                
                det = (w11 * (w22 * w33 - w23 * w32) - 
                       w12 * (w21 * w33 - w23 * w31) + 
                       w13 * (w21 * w32 - w22 * w31))
                
                if det == 0:
                    print(f"{i} {j} {k}")
                    return

solve()
