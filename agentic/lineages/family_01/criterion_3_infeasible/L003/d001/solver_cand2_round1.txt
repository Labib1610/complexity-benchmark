```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # Sort edges to help find lexicographically smallest (a, b, c)
    # We want smallest a, then smallest b, then smallest c.
    # Iterate through all edges (u, v) as potential (a, b)
    # Then check if there exists a w such that (u, w) and (v, w) exist.
    
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # possible a, then b, then c.
    # Since n <= 2000, we can iterate through all pairs (a, b) that have an edge.
    
    for a in range(1, n + 1):
        # Get all neighbors of a
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a < b < c. Since we iterate a from 1 to n,
            # and b is a neighbor of a, we just need to ensure b > a.
            if b < a:
                continue
            
            # Now find c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # To be efficient, use the smaller adjacency set
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
                other = adj[b]
            else:
                candidates = adj[b]
                other = adj[a]
            
            for c in sorted(list(candidates)):
                if c > b and c in other:
                    print(f"{a} {b} {c}")
                    return

solve()
```
