import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # Sort edges to help find the minimum signature triad
    # Signature is a*10^8 + b*10^4 + c
    # To minimize this, we want the smallest 'a', then smallest 'b', then smallest 'c'
    edges.sort()
    
    min_a, min_b, min_c = float('inf'), float('inf'), float('inf')
    
    # Iterate through all edges (a, b) and check for a common neighbor c > b
    # To ensure we find the minimum, we iterate in order
    for a in range(1, n + 1):
        # Optimization: if a is already larger than min_a, we can't find a smaller signature
        if a > min_a:
            break
            
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if a > b: continue
            
            # If (a, b) is already worse than (min_a, min_b), skip
            if a == min_a and b > min_b:
                continue
            
            # Find common neighbors c > b
            # We only need the smallest c to minimize the signature for this (a, b)
            for c in neighbors_a[i+1:]:
                if c in adj[b]:
                    # Found a triad (a, b, c)
                    if a < min_a or (a == min_a and b < min_b) or (a == min_a and b == min_b and c < min_c):
                        min_a, min_b, min_c = a, b, c
                    # Since we want the smallest c for this (a, b), we can break here
                    break
                    
    print(f"{min_a} {min_b} {min_c}")

if __name__ == '__main__':
    solve()
