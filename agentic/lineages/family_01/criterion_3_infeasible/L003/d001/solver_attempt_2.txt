import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically first (a, b, c) with a < b < c:
    # Iterate through all possible 'a' from 1 to n
    # Then iterate through all neighbors 'b' of 'a' such that b > a
    # Then check if there exists a 'c' such that c > b and c is in both adj[a] and adj[b]
    
    # Sort neighbors to ensure we find the lexicographically smallest (a, b, c)
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c > b such that c is in adj[a] and adj[b]
            # Since adj[a] and adj[b] are sorted, we can find common elements efficiently
            # But since we want the smallest c, we just iterate through the smaller list
            
            # Optimization: check intersection of adj[a] and adj[b]
            # Only need to check elements > b
            set_b = set(adj[b])
            for c in neighbors_a:
                if c > b and c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
