```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list using sets for O(1) lookup
    adj = defaultdict(set)
    edges = []
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort adjacency lists for consistent iteration
    for key in adj:
        adj[key] = sorted(list(adj[key]))
    
    # Iterate through all edges to find a triangle
    found = False
    result = None
    
    # We'll try all edges as potential first edge of triangle
    # For each edge (u,v), we look for a third node w that connects to both
    for u, v in edges:
        if found:
            break
        if u > v:
            u, v = v, u  # Ensure u < v for consistency
        
        # Get neighbors of both u and v
        neighbors_u = adj[u]
        neighbors_v = adj[v]
        
        # Use two pointers to find common neighbors efficiently
        i, j = 0, 0
        while i < len(neighbors_u) and j < len(neighbors_v):
            if neighbors_u[i] == neighbors_v[j]:
                w = neighbors_u[i]
                # Check that all three are distinct and form a triangle
                if w != u and w != v:
                    # Return the lexicographically smallest triplet
                    triplet = sorted([u, v, w])
                    if not result or triplet < result:
                        result = triplet
                        found = True
                        break
                i += 1
                j += 1
            elif neighbors_u[i] < neighbors_v[j]:
                i += 1
            else:
                j += 1
    
    # Output the result
    print(result[0], result[1], result[2])

if __name__ == "__main__":
    main()
```
