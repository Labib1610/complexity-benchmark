 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    neigh = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        neigh[u].append(v)
        neigh[v].append(u)
    for lst in neigh:
        lst.sort()
    out = sys.stdout.write
    for a in range(n):
        mask = adj[a] & ~((1 << a) - 1)
        for b in neigh[a]:
            if b <= a:
                continue
            cand = adj[b] & mask & ~((1 << b) - 1)
            if cand:
                c = (cand & -cand).bit_length() - 1
                out(f"{a + 1} {b + 1} {c + 1}")
                return

if __name__ == "__main__":
    solve()
```
