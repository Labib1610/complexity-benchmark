import sys

def solve():
    # The condition "no matter which of the three gears is disengaged, 
    # the remaining two must be compatible" means that for a set {a, b, c},
    # (a, b), (b, c), and (a, c) must all be compatible.
    # This is equivalent to finding a triangle (clique of size 3) in the graph.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest {a, b, c}, we iterate through
    # possible values of a, then b, then c.
    # Since n <= 2000, we can iterate through all edges (a, b) and check
    # if there exists a c such that (a, c) and (b, c) are edges.
    
    edges.sort()
    
    # To optimize, for each edge (u, v), we check the intersection of adj[u] and adj[v].
    # We want the smallest a, then b, then c.
    
    best = None
    
    for a in range(1, n + 1):
        # Optimization: if we already found a solution with a smaller 'a', we can stop.
        # But we need to check all edges.
        # Let's iterate through all edges (u, v) and see if they form a triangle.
        pass
    
    # Actually, iterating through all edges (u, v) and finding common neighbors is efficient.
    for u in range(1, n + 1):
        sorted_neighbors = sorted(list(adj[u]))
        for i in range(len(sorted_neighbors)):
            v = sorted_neighbors[i]
            # We have edge (u, v). Now find common neighbor w > v.
            # To ensure lexicographical order, we check all common neighbors.
            common = adj[u].intersection(adj[v])
            for w in common:
                if w > v:
                    # Found a triangle (u, v, w)
                    res = sorted([u, v, w])
                    if best is None or res < best:
                        best = res
        # If we found a best for this 'a', we can potentially prune, 
        # but given the constraints, this approach is O(m * sqrt(m)) or similar.
        if best is not None and best[0] < u:
            break
            
    print(f"{best[0]} {best[1]} {best[2]}")

solve()
