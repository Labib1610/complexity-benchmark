import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Sort adjacency lists to potentially speed up intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Neighbors of a that are > a
        neighbors_a = [x for x in adj[a] if x > a]
        for b in neighbors_a:
            # We need a c such that c > b and c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we want the smallest c, we can iterate through sorted neighbors
            neighbors_b = adj[b]
            # Use two pointers or set intersection
            # Since we want the smallest c > b, we can just iterate
            for c in neighbors_b:
                if c > b and c in adj_sets[a]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
