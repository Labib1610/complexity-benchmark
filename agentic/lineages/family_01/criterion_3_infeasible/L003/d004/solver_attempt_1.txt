import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # A Hedge Ring is a triangle (a, b, c) where a < b < c
    # and edges (a, b), (a, c), (b, c) exist.
    # To find the lexicographically smallest, we iterate a from 1 to n,
    # then b from a+1 to n, then check if c exists.
    
    # Sort adjacency lists to help with finding c
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # To optimize, iterate over edges (u, v) and check for a common neighbor w
    # Since we need the lexicographically smallest (a, b, c),
    # we can iterate a, then b, then check if any c > b exists in both adj[a] and adj[b].
    
    # Convert adj to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    best = None
    
    # Iterate through all possible a
    for a in range(1, n + 1):
        # Iterate through all possible b > a
        # To be efficient, only consider b that are connected to a
        for b in adj[a]:
            if b <= a:
                continue
            
            # Find common neighbors c > b
            # Intersection of adj[a] and adj[b]
            # Since adj lists are sorted, we can find common elements efficiently
            # But sets are already O(1) lookup
            
            # Optimization: iterate over the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                for c in adj[a]:
                    if c > b and c in adj_sets[b]:
                        if best is None or (a, b, c) < best:
                            best = (a, b, c)
                            # Since we want the smallest a, then b, then c,
                            # we can return immediately if we find one for this a, b
                            print(f"{a} {b} {c}")
                            return
            else:
                for c in adj[b]:
                    if c > b and c in adj_sets[a]:
                        if best is None or (a, b, c) < best:
                            best = (a, b, c)
                            print(f"{a} {b} {c}")
                            return

solve()
