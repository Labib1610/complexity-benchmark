import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    degree = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        degree[u] += 1
        degree[v] += 1
        idx += 2
        
    # Filter nodes that have at least 2 grievances
    valid_nodes = [i for i in range(1, n + 1) if degree[i] >= 2]
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n, then b from a+1 to n, then c from b+1 to n
    for a in range(1, n + 1):
        if degree[a] < 2:
            continue
        
        # Neighbors of a that are > a
        neighbors_a = [v for v in adj[a] if v > a]
        neighbors_a.sort()
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if degree[b] < 2:
                continue
            
            # Check for c such that c > b, c in adj[a], and c in adj[b]
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            
            candidates = []
            for c in common:
                if c > b and degree[c] >= 2:
                    candidates.append(c)
            
            if candidates:
                print(f"{a} {b} {min(candidates)}")
                return

if __name__ == '__main__':
    solve()
