import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # A trilateral impasse is a triangle (a, b, c) where a < b < c.
    # Condition 4: Exactly 3 grievance records involve only {a, b, c}.
    # This is always true for any triangle (a, b, c) because the only 
    # possible edges are (a, b), (a, c), and (b, c).
    # Condition 5: c - a >= 2.
    
    # We need the lexicographically smallest (a, b, c).
    # Iterate through all possible a, b, c in increasing order.
    
    # To optimize, iterate through all edges (u, v) as potential (a, b)
    # and check for a common neighbor w.
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    best = None
    
    # Iterate through all possible a
    for a in range(1, n - 1):
        # Iterate through all possible b > a
        for b in range(a + 1, n):
            if b in adj[a]:
                # Find common neighbors c > b
                # Intersection of adj[a] and adj[b]
                common = adj[a].intersection(adj[b])
                for c in common:
                    if c > b:
                        # Check condition 5: c - a >= 2
                        if c - a >= 2:
                            if best is None or (a, b, c) < best:
                                best = (a, b, c)
        if best is not None:
            # Since we iterate a and b in increasing order, 
            # the first valid one found for a specific a might not be the global min,
            # but we can break early if we ensure we check all b for this a.
            # Actually, just keep track of the best.
            pass
            
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
