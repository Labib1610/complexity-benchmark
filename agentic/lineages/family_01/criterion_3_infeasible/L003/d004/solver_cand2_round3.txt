import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest (a, b, c)
    # We want to minimize a, then b, then c.
    # Iterate through all possible 'a' from 1 to n
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted(list(adj[a]))
        
        # For each pair (b, c) in neighbors_a, check if (b, c) is an edge
        # To minimize lexicographical order, we iterate b and c in increasing order
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need to check if there exists c > b such that c is in adj[a] and c is in adj[b]
            # To keep it efficient, intersect adj[a] and adj[b]
            # Only consider c > b
            
            # Optimization: iterate through the smaller adjacency list
            if len(adj[b]) < len(adj[a]):
                candidates = [c for c in adj[b] if c > b and c in adj[a]]
            else:
                candidates = [c for c in adj[a] if c > b and c in adj[b]]
            
            if candidates:
                c = min(candidates)
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
