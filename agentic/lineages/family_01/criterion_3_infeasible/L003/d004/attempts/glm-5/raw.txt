import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Find lexicographically smallest triangle
    for a in range(1, n - 1):
        adj_a = adj[a]
        # Get neighbors of a that are > a
        neighbors_a = [v for v in adj_a if v > a]
        if len(neighbors_a) < 2:
            continue
        neighbors_a.sort()
        # Check all pairs of neighbors
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            adj_b = adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj_b:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
