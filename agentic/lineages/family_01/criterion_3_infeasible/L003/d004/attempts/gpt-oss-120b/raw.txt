```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    masks = [0] * (n + 1)          # bitmask of neighbours for each vertex (1‑based)
    adj = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        masks[u] |= 1 << v
        masks[v] |= 1 << u
        adj[u].append(v)
        adj[v].append(u)

    # sort adjacency lists to guarantee increasing order of b
    for lst in adj:
        lst.sort()

    for a in range(1, n - 1):
        mask_a = masks[a]
        for b in adj[a]:
            if b <= a:
                continue
            # common neighbours of a and b
            common = mask_a & masks[b]
            # we need c > b
            common >>= b
            if common == 0:
                continue
            # lowest set bit gives smallest c > b
            low_bit = common & -common
            c_offset = low_bit.bit_length() - 1   # position relative to shifted bits
            c = b + c_offset
            # output lexicographically smallest triangle
            print(a, b, c)
            return

if __name__ == "__main__":
    solve()
```
