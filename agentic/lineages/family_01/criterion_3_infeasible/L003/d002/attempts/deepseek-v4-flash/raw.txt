We can solve this by checking each vertex in increasing order and looking for a triangle that includes it.  
To keep it efficient, we’ll use adjacency sets for O(1) edge checks.

```python
import sys

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)

    for a in range(1, n + 1):
        for b in adj[a]:
            if b > a:
                for c in adj[a]:
                    if c > b and c in adj[b]:
                        print(a, b, c)
                        return

if __name__ == "__main__":
    main()
```
