import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_mask = [0] * (n + 1)
    adj_list = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        adj_mask[u] |= 1 << v
        adj_mask[v] |= 1 << u
        adj_list[u].append(v)
        adj_list[v].append(u)
    for i in range(1, n + 1):
        adj_list[i].sort()
    for a in range(1, n + 1):
        mask_a = adj_mask[a]
        for b in adj_list[a]:
            if b <= a:
                continue
            common = mask_a & adj_mask[b]
            if common == 0:
                continue
            shifted = common >> (b + 1)
            if shifted:
                low = shifted & -shifted
                c = (b + 1) + (low.bit_length() - 1)
                sys.stdout.write(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
