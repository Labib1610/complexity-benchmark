import sys
from collections import defaultdict

def main():
    import sys

    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list using sets for fast lookup
    adj = defaultdict(set)
    edges = []
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort nodes by ID to ensure lexicographical order
    nodes = sorted(adj.keys())
    
    # For each edge (u,v), look for a common neighbor w
    # such that (u,w) and (v,w) are also edges
    found = False
    result = None
    
    # To maintain lexicographical order, we iterate through edges in order
    # and for each edge (u,v), check neighbors of u that are > v
    for u in nodes:
        for v in adj[u]:
            if v <= u:
                continue
            # Now look for a node w such that w is connected to both u and v
            # and w > v to maintain u < v < w
            for w in adj[u]:
                if w > v and w in adj[v]:
                    # Found a triangle: u-v-w
                    print(u, v, w)
                    return

if __name__ == "__main__":
    main()
