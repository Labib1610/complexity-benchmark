import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate through possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we iterate b in adj[a] in increasing order.
    # Then we check if there exists c in adj[a] such that c > b and c in adj[b].
    
    # Sort adjacency lists to ensure we check in lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Since neighbors_a is sorted, we look for c in neighbors_a[i+1:]
            # and check if c is in adj[b].
            # To make this efficient, convert adj[b] to a set for O(1) lookup.
            set_b = set(adj[b])
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

solve()
