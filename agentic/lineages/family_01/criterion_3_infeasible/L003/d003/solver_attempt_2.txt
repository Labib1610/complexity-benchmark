```python
import sys

def solve():
    # The condition "no matter which of the three commodities is selected as the 
    # 'base asset', the remaining two commodities in the portfolio must be a 
    # pair that has a verified exchange parity" means that for a set {a, b, c},
    # the pairs (a, b), (b, c), and (a, c) must all exist in the input.
    # This is equivalent to finding a triangle (a clique of size 3) in the graph.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest {a, b, c} with a < b < c:
    # Iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, we can iterate through all edges (a, b) and check
    # for a common neighbor c.
    
    edges.sort()
    
    # To optimize, iterate through all pairs (a, b) that are connected.
    # For each pair, find common neighbors.
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and adj[b]
            # To ensure lexicographical order, we check c in increasing order
            # Intersection of adj[a] and adj[b]
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
