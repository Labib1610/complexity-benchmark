We can solve this by checking each possible triple in sorted order and stopping at the first valid one.  
Since we need the lexicographically smallest set, we can iterate `a` from 1 to n, then `b` from `a+1` to n, then `c` from `b+1` to n, and check if all three pairs exist.

```python
import sys

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b not in adj[a]:
                continue
            for c in range(b + 1, n + 1):
                if c in adj[a] and c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
