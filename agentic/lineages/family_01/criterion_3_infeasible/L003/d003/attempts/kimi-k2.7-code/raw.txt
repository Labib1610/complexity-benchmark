 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_mask = [0] * n
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj_mask[u] |= 1 << v
        adj_mask[v] |= 1 << u
        adj[u].append(v)
        adj[v].append(u)
    for lst in adj:
        lst.sort()
    for a in range(n):
        mask_a = adj_mask[a]
        for b in adj[a]:
            if b <= a:
                continue
            common = mask_a & adj_mask[b]
            shifted = common >> (b + 1)
            if shifted:
                low = shifted & -shifted
                c = low.bit_length() - 1 + b + 1
                sys.stdout.write(f"{a + 1} {b + 1} {c + 1}\n")
                return

if __name__ == "__main__":
    solve()
```
