import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to iterate in lexicographical order
    edges.sort()
    
    # To find the lexicographically first (a, b, c) with a < b < c:
    # Iterate through all possible 'a' from 1 to n
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # For a fixed 'a', we need to find b, c such that b, c are neighbors of a
        # and b is a neighbor of c.
        # To ensure lexicographical order, we iterate b in increasing order
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need a c such that c is a neighbor of a and c is a neighbor of b
            # c must be > b to satisfy a < b < c
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            
            # Filter for c > b
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
