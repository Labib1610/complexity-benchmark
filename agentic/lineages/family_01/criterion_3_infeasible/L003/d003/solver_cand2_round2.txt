```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we iterate b in adj[a] such that b > a.
    # Then we check if there exists c in adj[a] intersect adj[b] such that c > b.
    
    # Sort adjacency lists to ensure we check in lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # To optimize, we can iterate through all edges (u, v) with u < v
    # and check if there is a common neighbor w > v.
    # But we need the lexicographically smallest (a, b, c).
    # Let's iterate a from 1 to n.
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # b must be a neighbor of a and b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # c must be a neighbor of a and b, and c > b
            # We can intersect the sorted lists adj[a] and adj[b]
            # Since they are sorted, we can use two pointers or set intersection
            # But since we want the smallest c, we just iterate through adj[b]
            # and check if it's in adj[a] (using a set for O(1) lookup)
            
            set_a = set(neighbors_a)
            for c in adj[b]:
                if c > b and c in set_a:
                    print(f"{a} {b} {c}")
                    return

solve()
```
