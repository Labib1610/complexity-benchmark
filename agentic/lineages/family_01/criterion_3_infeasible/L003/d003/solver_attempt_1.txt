import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, b, c in order.
    # Since n <= 2000, we can optimize by iterating over edges (a, b)
    # and checking if there exists a c > b such that (a, c) and (b, c) exist.
    
    edges.sort()
    
    best = None
    
    # Iterate through all possible a
    for a in range(1, n + 1):
        # Iterate through all possible b > a
        # We only care about b if (a, b) is an edge
        neighbors_a = sorted(list(adj[a]))
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Now find c > b such that (a, c) and (b, c) are edges
            # c must be in both adj[a] and adj[b]
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
                    # Since we want the lexicographically smallest,
                    # and we are iterating a, then b, then c,
                    # the first one we find is the smallest.
                    print(f"{a} {b} {c}")
                    return

solve()
