```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency set for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
    
    # A redundant storage cell is a triangle (a, b, c)
    # If any one is removed, the other two must be connected.
    # This means all three pairs (a,b), (b,c), (a,c) must exist.
    # This is exactly the definition of a triangle in the graph.
    
    # Sort edges to iterate in a way that helps find the lexicographically smallest
    edges.sort()
    
    best = None
    
    # Iterate over all edges (u, v) and check for a common neighbor w
    for u, v in edges:
        # We want to find w such that (u, w) and (v, w) exist
        # To ensure lexicographical order, we check all w
        # Optimization: iterate over the smaller adjacency list
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
            
        for w in adj[u]:
            if w in adj[v]:
                # Found a triangle (u, v, w)
                # Sort the vertices to get (a, b, c) with a < b < c
                res = sorted([u, v, w])
                if best is None or res < best:
                    best = res
                    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
