```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj_bits = [0] * (n + 1)
    neighbors = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        adj_bits[u] |= 1 << v
        adj_bits[v] |= 1 << u
        neighbors[u].append(v)
        neighbors[v].append(u)

    # mask of vertices strictly greater than i
    full = ((1 << n) - 1) << 1
    mask_gt = [0] * (n + 2)
    for i in range(n + 1):
        mask_gt[i] = full & ~((1 << (i + 1)) - 1)

    for i in range(1, n + 1):
        neighbors[i].sort()

    for a in range(1, n + 1):
        for b in neighbors[a]:
            if b <= a:
                continue
            common = adj_bits[a] & adj_bits[b] & mask_gt[b]
            if common:
                lsb = common & -common
                c = lsb.bit_length() - 1
                sys.stdout.write(f"{a} {b} {c}\n")
                return

if __name__ == "__main__":
    solve()
```
