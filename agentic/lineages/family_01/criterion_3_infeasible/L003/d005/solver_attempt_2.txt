import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) with a < b < c.
    # To find the lexicographically smallest, we iterate a from 1 to n,
    # then b from a+1 to n, then c from b+1 to n.
    # To optimize, we only check b if b is in adj[a],
    # and then check if there exists c such that c is in adj[a] and adj[b].
    
    # Sort neighbors to ensure we check in increasing order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Find common neighbors of a and b
            # Since neighbors_a is sorted, we can intersect efficiently
            # or just iterate through neighbors of b
            neighbors_b = adj[b]
            
            # We want the smallest c > b such that c is in neighbors_a and neighbors_b
            # To be lexicographically smallest, we check all c > b
            # and pick the smallest one.
            for c in neighbors_a:
                if c > b and c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
