```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Convert to sorted lists for consistent iteration
    for u in adj:
        adj[u] = sorted(adj[u])
    
    # We want the lexicographically smallest triple (a,b,c) where a<b<c
    # and each pair among {a,b,c} has both connected to the third.
    # That means:
    # - a is connected to b and c
    # - b is connected to a and c
    # - c is connected to a and b
    # Which is equivalent to saying the subgraph induced by {a,b,c} is a triangle.
    
    # Let's iterate through all edges (u,v) with u < v
    # For each such edge, we look for a node w such that:
    # w is connected to both u and v, and u < v < w
    
    # To make it efficient, let's build adjacency sets
    adj_set = {u: set(adj[u]) for u in adj}
    
    # Try all pairs (u,v) with u < v
    candidates = []
    for u in sorted(adj_set.keys()):
        for v in adj_set[u]:
            if v > u:
                # Now find w such that w > v and w connected to both u and v
                common_neighbors = adj_set[u].intersection(adj_set[v])
                for w in common_neighbors:
                    if w > v:
                        candidates.append((u, v, w))
    
    # Find the lexicographically smallest
    candidates.sort()
    result = candidates[0]
    print(result[0], result[1], result[2])

if __name__ == "__main__":
    main()
```
