```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
    
    # A cell {a, b, c} is valid if:
    # (a, b) in E, (b, c) in E, (a, c) in E
    # This is simply a triangle in the graph.
    # We need the lexicographically smallest {a, b, c} with a < b < c.
    
    # Iterate through all possible a, b, c
    # Since n <= 2000, O(n^3) is too slow.
    # However, we only need to check triangles.
    # We can iterate over all edges (u, v) and check for common neighbors.
    
    # To find the lexicographically smallest, we iterate a from 1 to n-2,
    # b from a+1 to n-1, and c from b+1 to n.
    # To optimize, iterate a, then iterate over neighbors b of a,
    # then check if there exists c such that c is a neighbor of both a and b.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Find common neighbors c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            
            if valid_c:
                c = min(valid_c)
                # We found a triangle {a, b, c}.
                # Since we iterate a then b, we need to ensure we find the smallest.
                # The current loop structure finds the smallest a, then smallest b.
                # For a fixed a and b, any c works, but we want the smallest c.
                print(f"{a} {b} {c}")
                return

solve()
```
