```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c,
    # we iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    # We iterate through all edges (u, v) and check for a common neighbor w.
    
    best = None
    
    # Sort neighbors to help find smallest triplet
    for i in range(1, n + 1):
        neighbors = sorted(list(adj[i]))
        len_n = len(neighbors)
        for j_idx in range(len_n):
            u = neighbors[j_idx]
            if u <= i:
                continue
            
            # Check for common neighbors w such that (i, u, w) is a triangle
            # To ensure lexicographical order, we check w < i < u, i < w < u, or i < u < w
            # Actually, just find all triangles and keep the min.
            # Optimization: iterate over the smaller degree node's neighbors
            if len(adj[i]) < len(adj[u]):
                for w in adj[i]:
                    if w in adj[u]:
                        res = sorted([i, u, w])
                        if best is None or res < best:
                            best = res
            else:
                for w in adj[u]:
                    if w in adj[i]:
                        res = sorted([i, u, w])
                        if best is None or res < best:
                            best = res
                            
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
