import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c,
    # we iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we look for 'b' > 'a' such that 'b' is connected to 'a'.
    # Then we look for 'c' > 'b' such that 'c' is connected to both 'a' and 'b'.
    
    # Sort edges to help iterate in order
    edges.sort()
    
    # Pre-sort adjacency lists to ensure we check 'b' and 'c' in increasing order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Since both adj[a] and adj[b] are sorted, we can find intersection efficiently
            # or just iterate through neighbors of b that are > b
            neighbors_b = adj[b]
            
            # Use a set for O(1) lookup if needed, but since we want the smallest c,
            # we can just iterate through the smaller adjacency list
            if len(neighbors_a) < len(neighbors_b):
                for c in neighbors_a:
                    if c > b and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in neighbors_b:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

solve()
