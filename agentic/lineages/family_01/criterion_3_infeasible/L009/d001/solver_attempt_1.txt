```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, we can iterate through all edges (a, b) and check
    # for a common neighbor c.
    
    best = None
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    for a, b in edges:
        # We have a < b. We need c such that a-c and b-c exist.
        # c can be < a, between a and b, or > b.
        # The condition is a < b < c OR a < c < b OR c < a < b.
        # We want the smallest (a, b, c) tuple.
        
        common = adj[a].intersection(adj[b])
        for c in common:
            # Form the triplet and sort it
            triplet = sorted([a, b, c])
            if best is None or triplet < best:
                best = triplet
                
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
