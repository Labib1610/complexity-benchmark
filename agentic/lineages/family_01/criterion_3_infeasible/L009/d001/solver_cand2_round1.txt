import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need to find a trio (a, b, c) such that a-b, b-c, a-c are all edges.
    # To find the lexicographically first, we iterate through all edges (u, v)
    # and check for a common neighbor w.
    # To ensure lexicographical order, we check candidates in order.
    
    # Sort edges to help find the lexicographically first trio
    # A trio (a, b, c) with a < b < c is formed by edge (a, b) and common neighbor c.
    # We want the smallest a, then smallest b, then smallest c.
    
    found_trio = None
    
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # Iterate through all possible 'b' > a
        # To be efficient, iterate through neighbors of 'a'
        neighbors_a = sorted(list(adj[a]))
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Now find 'c' such that c > b and c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we want the smallest c, we can iterate through sorted neighbors
            neighbors_b = adj[b]
            
            # Find common neighbors
            common = sorted(list(adj[a].intersection(neighbors_b)))
            for c in common:
                if c > b:
                    # This is a valid trio (a, b, c)
                    # Since we iterate a, then b, then c in increasing order,
                    # the first one we find is the lexicographically smallest.
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
