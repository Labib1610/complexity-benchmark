import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
    
    # A trio {a, b, c} satisfies the condition if:
    # 1. At least one edge exists.
    # 2. For any node in the trio, if it has an edge to one of the other two,
    #    it must have an edge to both.
    # This means the induced subgraph must be either:
    # - A triangle (all 3 edges present)
    # - A single edge (one node isolated, two connected)
    
    # We need the lexicographically smallest (a, b, c).
    # Iterate through all possible a < b < c.
    # Since n <= 2000, O(n^3) is too slow.
    # However, the condition is equivalent to:
    # The number of edges in the induced subgraph is 1 or 3.
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            # Check if a and b are connected
            connected_ab = b in adj[a]
            
            # We need to find c > b such that {a, b, c} is a certificate.
            # Case 1: {a, b, c} has 1 edge.
            # If connected_ab is True, then c must be connected to neither a nor b.
            # If connected_ab is False, then c must be connected to exactly one of a or b.
            
            # Case 2: {a, b, c} has 3 edges (a triangle).
            # This requires a-b, b-c, and a-c to be edges.
            
            # Optimization: iterate c
            for c in range(b + 1, n + 1):
                edges_count = 0
                if b in adj[a]: edges_count += 1
                if c in adj[a]: edges_count += 1
                if c in adj[b]: edges_count += 1
                
                if edges_count == 1 or edges_count == 3:
                    # Verify symmetry condition:
                    # For each node, degree in induced subgraph must be 0 or 2.
                    deg_a = (1 if b in adj[a] else 0) + (1 if c in adj[a] else 0)
                    deg_b = (1 if a in adj[b] else 0) + (1 if c in adj[b] else 0)
                    deg_c = (1 if a in adj[c] else 0) + (1 if b in adj[c] else 0)
                    
                    if (deg_a == 0 or deg_a == 2) and \
                       (deg_b == 0 or deg_b == 2) and \
                       (deg_c == 0 or deg_c == 2):
                        print(f"{a} {b} {c}")
                        return

solve()
