import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list to store the graph
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A valid panel of 3 interpreters involving exactly 3 languages
    # means we are looking for a triangle (a, b, c) in the graph
    # where edges (a, b), (b, c), and (a, c) exist.
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through a from 1 to n, b from a+1 to n, and c from b+1 to n.
    # To optimize, we only check if c is in adj[a] and adj[b].
    
    # Sort neighbors to potentially speed up intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that c > b and c is in both adj[a] and adj[b]
            # Since adj lists are sorted, we can find common elements efficiently
            neighbors_b = adj[b]
            
            # Use two pointers to find common elements in sorted lists
            ptr_a = i + 1
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

if __name__ == '__main__':
    solve()
