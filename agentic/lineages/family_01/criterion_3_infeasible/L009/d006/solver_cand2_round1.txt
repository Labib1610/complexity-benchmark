import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A triplet {a, b, c} is a core instability triplet if:
    # (a, b) is volatile, (b, c) is volatile, (a, c) is volatile.
    # Wait, the condition is: "if any one of the three materials is set aside, 
    # the remaining two are always documented as a mutually volatile pair."
    # This means:
    # If we remove 'a', {b, c} must be volatile.
    # If we remove 'b', {a, c} must be volatile.
    # If we remove 'c', {a, b} must be volatile.
    # This is exactly the definition of a triangle (a clique of size 3).
    
    # We need the lexicographically smallest triplet (a, b, c) with a < b < c.
    # Iterate through all possible a, b, c.
    # Since n <= 2000, O(n^3) is too slow.
    # However, we only need to check if a, b, c form a triangle.
    # We can iterate over all edges (u, v) and check for a common neighbor w.
    
    triplets = []
    
    # To find the lexicographically smallest, we iterate a from 1 to n.
    for a in range(1, n + 1):
        # To keep it efficient, iterate over neighbors of a
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now find common neighbors c such that c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    # Found a triplet (a, b, c)
                    # Since we want the lexicographically smallest, 
                    # we can collect them and sort, or just keep track of the min.
                    # Given the constraints and the nature of the problem,
                    # we can just return the first one found if we iterate correctly.
                    print(f"{a} {b} {c}")
                    return

solve()
