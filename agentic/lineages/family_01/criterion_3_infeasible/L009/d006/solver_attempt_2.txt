import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c
    # We iterate through possible 'a' from 1 to n-2
    for a in range(1, n - 1):
        # For a fixed 'a', we want the smallest 'b' > a
        # 'b' must be connected to 'a'
        neighbors_a = sorted(list(adj[a]))
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Now we need a 'c' > b such that (a, c) and (b, c) are edges
            # This means c must be in adj[a] AND adj[b]
            # To minimize (a, b, c), we want the smallest c > b
            # Intersection of adj[a] and adj[b]
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
