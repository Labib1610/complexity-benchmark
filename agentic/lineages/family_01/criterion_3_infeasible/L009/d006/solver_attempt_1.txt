```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Represent the graph as an adjacency list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A valid panel of 3 interpreters covering exactly 3 languages
    # means the 3 interpreters must form a triangle (a-b, b-c, c-a).
    # We need to find the lexicographically smallest (a, b, c) with a < b < c.
    
    # Iterate through all possible a, b, c
    # To optimize, iterate a from 1 to n-2
    for a in range(1, n + 1):
        # Sort neighbors to help find b and c in order
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that c is connected to both a and b
            # c must be > b to satisfy a < b < c
            # c must be in adj[a] and adj[b]
            
            # Intersection of adj[a] and adj[b]
            # Since we need c > b, we only look at neighbors of a > b
            for j in range(i + 1, len_a):
                c = neighbors_a[j]
                if c in adj[b]:
                    # Found the lexicographically smallest triangle
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
