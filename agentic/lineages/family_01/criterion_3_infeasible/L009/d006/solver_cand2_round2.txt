import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triplet
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # possible a, then b, then c.
    # A triplet (a, b, c) with a < b < c is a core instability triplet if
    # (a, b), (a, c), and (b, c) are all in the set of reactive pairs.
    
    # Optimization: Iterate through all edges (a, b) and check for a common neighbor c
    # such that c > b.
    
    # To ensure lexicographical order, we can iterate a from 1 to n,
    # then b from a+1 to n, then check if (a, b) is an edge.
    # If it is, find all c > b such that (a, c) and (b, c) are edges.
    
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # To be efficient, iterate through the smaller set
            if len(adj[b]) < len(neighbors_a) - i - 1:
                for c in adj[b]:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for j in range(i + 1, len(neighbors_a)):
                    c = neighbors_a[j]
                    if c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

solve()
