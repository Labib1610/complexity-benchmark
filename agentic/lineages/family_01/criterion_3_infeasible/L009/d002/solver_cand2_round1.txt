```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # A minimal certificate is a triangle (a, b, c) such that
    # (a, b), (b, c), (a, c) are all edges.
    # We need the lexicographically smallest (a, b, c) with a < b < c.
    
    edges.sort()
    
    # Iterate through all edges (u, v) as the "volatile pair"
    # and find a common neighbor w.
    # To ensure lexicographical order, we check candidates in order.
    
    best = None
    
    for u in range(1, n + 1):
        # Sort neighbors to check potential 'c' in order
        neighbors = sorted(list(adj[u]))
        for i in range(len(neighbors)):
            v = neighbors[i]
            if v <= u:
                continue
            
            # Now we have edge (u, v). Look for w such that (u, w) and (v, w) exist.
            # w must be a neighbor of both u and v.
            # To keep (a, b, c) sorted, we check w < u, u < w < v, and v < w.
            
            common = sorted(list(adj[u].intersection(adj[v])))
            for w in common:
                # Form the triplet and sort it
                triplet = sorted([u, v, w])
                if best is None or triplet < best:
                    best = triplet
                    
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
