```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n, m = int(data[0]), int(data[1])
    edges = []
    adj = [set() for _ in range(n+1)]
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # For each edge (u,v), we want to find a common neighbor w
    # such that (u,w) and (v,w) are also edges.
    # This forms a triangle u-v-w where every pair is connected.
    
    # We'll iterate through all edges and try to complete them into triangles
    # We want the lexicographically smallest triangle
    
    best = None
    
    # To make it efficient, for each node, sort its neighbors
    for i in range(1, n+1):
        adj[i] = sorted(adj[i])
    
    # For each edge, find a common neighbor
    for u, v in edges:
        if u > v:
            continue  # to avoid duplicate checking
        # Find common neighbors of u and v
        nu = set(adj[u])
        nv = set(adj[v])
        common = nu & nv
        for w in sorted(common):
            if w <= u or w <= v:
                continue  # to ensure u < v < w
            candidate = (u, v, w)
            if best is None or candidate < best:
                best = candidate
    
    if best is not None:
        print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
```
