import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # A certificate {a, b, c} means:
    # (a, b) is volatile, (a, c) is volatile, (b, c) is volatile.
    # Wait, the condition is:
    # If we pick one for primary, the other two must be volatile.
    # If we pick 'a', {b, c} must be volatile.
    # If we pick 'b', {a, c} must be volatile.
    # If we pick 'c', {a, b} must be volatile.
    # This means {a, b, c} must form a triangle (a clique of size 3).
    
    # We need the lexicographically smallest {a, b, c} such that a < b < c.
    # Iterate through all edges (u, v) and check for a common neighbor w.
    # Since we need the lexicographically smallest, we iterate a from 1 to n.
    
    edges.sort()
    
    for a in range(1, n + 1):
        neighbors = sorted(list(adj[a]))
        len_n = len(neighbors)
        for i in range(len_n):
            b = neighbors[i]
            if b <= a:
                continue
            for j in range(i + 1, len_n):
                c = neighbors[j]
                # Check if (b, c) is an edge
                if c in adj[b]:
                    # Found a triangle {a, b, c}
                    # Since we iterate a, then b, then c in increasing order,
                    # the first one we find is the lexicographically smallest.
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
