```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest trio (a, b, c) with a < b < c,
    # we iterate through all edges (a, b) and check if there exists a c > b
    # such that (a, c) and (b, c) are also edges.
    
    # Sort edges to help find the smallest trio
    edges.sort()
    
    # To optimize, we can iterate through all possible 'a'
    # For a fixed 'a', we look for 'b' in adj[a]
    # Then we look for 'c' in the intersection of adj[a] and adj[b]
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we only look at indices > i
            neighbors_b = adj[b]
            
            # Use two pointers to find common elements in sorted lists
            ptr_a = i + 1
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    print(f"{a} {b} {val_a}")
                    return
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
```
