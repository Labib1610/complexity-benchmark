import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Convert to sorted lists for consistent iteration
    adj_list = [sorted(s) for s in adj]
    
    for a in range(1, n + 1):
        neighbors_a = adj_list[a]
        len_a = len(neighbors_a)
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            # Check intersection of neighbors of a and b (that are > b)
            neighbors_b = adj_list[b]
            # Iterate through neighbors of a that are > b
            for j in range(i + 1, len_a):
                c = neighbors_a[j]
                if c > b and c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == '__main__':
    solve()
