```python
import sys

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    adj = [0] * (n + 1)
    p = 2
    for _ in range(m):
        u = data[p]
        v = data[p + 1]
        p += 2
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    out = sys.stdout.write
    for a in range(1, n - 1):
        bits = adj[a] >> a
        while bits:
            low = bits & -bits
            b = a + low.bit_length()
            common = adj[a] & adj[b]
            x = common >> b
            if x:
                lowc = x & -x
                c = b + lowc.bit_length()
                out(f"{a} {b} {c}\n")
                return
            bits ^= low

if __name__ == "__main__":
    solve()
```
