```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to process in order
    edges.sort()
    
    # For each edge (a,b), look for a vertex c such that (a,c) and (b,c) exist
    for a, b in edges:
        # Common neighbors of a and b
        common = adj[a] & adj[b]
        if common:
            # Find the smallest c in common neighbors
            c = min(common)
            # Ensure a < b < c
            if a < b < c:
                print(a, b, c)
                return
            else:
                # Need to sort them properly
                triple = sorted([a, b, c])
                print(triple[0], triple[1], triple[2])
                return

if __name__ == "__main__":
    main()
```
