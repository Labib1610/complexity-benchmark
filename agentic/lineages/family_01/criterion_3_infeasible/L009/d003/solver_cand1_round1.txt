import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
    
    # A certificate of 3 materials {a, b, c} is impossible to split into 2 units
    # if and only if the subgraph induced by {a, b, c} contains an odd cycle.
    # Since it's 3 nodes, the only odd cycle is a triangle (K3).
    # So we are looking for a triangle (a, b, c) in the graph.
    # We need the lexicographically smallest triangle.
    
    # Sort edges to help find smallest a, b, c
    edges.sort()
    
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    
    # Convert adj to list for faster iteration if needed, but set is fine for lookup
    # Sort nodes by degree to optimize triangle finding
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]))
    
    # Iterate through all edges (u, v) and check for a common neighbor w
    # To ensure lexicographical order, we check all possible triplets.
    # Actually, iterating through all edges and checking neighbors is efficient.
    
    best = None
    
    for a in range(1, n + 1):
        neighbors = sorted(list(adj[a]))
        for i in range(len(neighbors)):
            b = neighbors[i]
            if b <= a: continue
            # Check for common neighbor c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    candidate = (a, b, c)
                    if best is None or candidate < best:
                        best = candidate
        if best and best[0] < a:
            break
            
    print(f"{best[0]} {best[1]} {best[2]}")

solve()
