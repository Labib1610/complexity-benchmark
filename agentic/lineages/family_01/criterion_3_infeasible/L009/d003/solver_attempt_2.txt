```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # Sort edges to help find the lexicographically smallest trio
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # sorted edges (a, b) and check if there exists a c > b such that
    # (a, c) and (b, c) are edges.
    # Since we want the smallest a, then b, then c, we iterate through
    # edges (a, b) in order.
    
    for a, b in edges:
        # We need a c > b such that c is in adj[a] and c is in adj[b]
        # To find the smallest c, we can intersect the two sets
        common = adj[a].intersection(adj[b])
        if common:
            c = min(common)
            print(f"{a} {b} {c}")
            return

if __name__ == '__main__':
    solve()
```
