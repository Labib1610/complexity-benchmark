import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # A certificate is a set of 3 materials {a, b, c} such that 
    # the edges (a, b), (b, c), and (a, c) all exist in the registry.
    # This is equivalent to finding a triangle in the graph.
    
    # To find the lexicographically smallest {a, b, c}, we iterate
    # through all possible a, then b, then c.
    # Since n <= 2000, we can iterate over all edges (u, v) and check
    # for a common neighbor w.
    
    found_triangles = []
    
    # To ensure we find the lexicographically smallest, we iterate
    # a from 1 to n, b from a+1 to n, c from b+1 to n.
    # A triangle {a, b, c} exists if (a, b), (b, c), and (a, c) are edges.
    
    # Optimization: iterate over edges (u, v) and check common neighbors.
    # For each edge (u, v) with u < v, check all w such that (u, w) and (v, w) exist.
    # The triangle is {min(u, v, w), mid(u, v, w), max(u, v, w)}.
    
    for u in range(1, n + 1):
        # Sort neighbors to help with lexicographical order
        neighbors = sorted(list(adj[u]))
        for i in range(len(neighbors)):
            v = neighbors[i]
            if v <= u:
                continue
            # Now we have edge (u, v) with u < v.
            # Find common neighbors w such that w > v.
            # Actually, to get all triangles, we just need to check all w.
            # To avoid duplicates, only consider w > v.
            for w in neighbors[i+1:]:
                if w in adj[v]:
                    # Found triangle {u, v, w}
                    res = sorted([u, v, w])
                    found_triangles.append(tuple(res))
    
    if found_triangles:
        found_triangles.sort()
        print(f"{found_triangles[0][0]} {found_triangles[0][1]} {found_triangles[0][2]}")

solve()
