import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triplet
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists c > b
    # such that (a, c) and (b, c) are also edges.
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b from a+1 to n, then check for c.
    
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that b < c, c in adj[a], and c in adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # To be efficient, iterate through the smaller set
            if len(neighbors_a) - i - 1 < len(adj[b]):
                for j in range(i + 1, len(neighbors_a)):
                    c = neighbors_a[j]
                    if c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                # Iterate through sorted neighbors of b
                neighbors_b = sorted(list(adj[b]))
                for c in neighbors_b:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
