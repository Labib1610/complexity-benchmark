import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges lexicographically by (u, v)
    edges.sort()
    
    # For each edge (u, v), find the smallest common neighbor w > v.
    # Since we want lexicographically smallest (a, b, c) with a < b < c,
    # and edges are sorted by (u, v), the first valid triangle found is the answer.
    for u, v in edges:
        # Iterate over neighbors of u that are > v (to ensure u < v < w)
        # To find the smallest w, we check neighbors of the smaller degree vertex.
        if len(adj[u]) <= len(adj[v]):
            smaller, larger = adj[u], adj[v]
        else:
            smaller, larger = adj[v], adj[u]
        
        for w in smaller:
            if w > v and w in larger:
                print(u, v, w)
                return
    
    # Should not reach here per problem guarantee
    print(-1, -1, -1)

solve()
