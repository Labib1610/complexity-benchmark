import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    deg = [0] * (n + 1)
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u][v] = True
        adj[v][u] = True
        deg[u] += 1
        deg[v] += 1
    
    # Check for triangle
    for i in range(1, n + 1):
        for j in range(i + 1, n + 1):
            if adj[i][j]:
                for k in range(j + 1, n + 1):
                    if adj[i][k] and adj[j][k]:
                        print(f"{i} {j} {k}")
                        return
    
    # No triangle, look for independent set of size 3
    # Since complement has a triangle, there are 3 vertices with no edges between them
    # We can find one by checking vertices with degree <= n - 3
    # Start with vertex 1 (lexicographically smallest)
    u = 1
    # Collect non-neighbors of u
    non_neighbors_u = []
    for v in range(1, n + 1):
        if v != u and not adj[u][v]:
            non_neighbors_u.append(v)
    
    # Find two non-neighbors of u that are not connected
    for i in range(len(non_neighbors_u)):
        v = non_neighbors_u[i]
        for j in range(i + 1, len(non_neighbors_u)):
            w = non_neighbors_u[j]
            if not adj[v][w]:
                res = sorted([u, v, w])
                print(f"{res[0]} {res[1]} {res[2]}")
                return

if __name__ == '__main__':
    solve()
