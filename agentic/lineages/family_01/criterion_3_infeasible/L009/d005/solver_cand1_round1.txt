import sys

def solve():
    # The problem asks for a trio (a, b, c) such that it's impossible to assign
    # w_a, w_b, w_c >= 0 with w_a + w_b + w_c = 1 such that for every volatile
    # pair (u, v) in the trio, w_u + w_v <= 1/2.
    #
    # Let the trio be {a, b, c}.
    # If all three pairs are volatile:
    # w_a + w_b <= 1/2
    # w_a + w_c <= 1/2
    # w_b + w_c <= 1/2
    # Summing these: 2(w_a + w_b + w_c) <= 3/2 => 2(1) <= 1.5 => 2 <= 1.5 (False).
    # So a triangle (clique of size 3) is always a certificate.
    #
    # If only two pairs are volatile (e.g., (a,b) and (a,c)):
    # w_a + w_b <= 1/2
    # w_a + w_c <= 1/2
    # Summing: 2w_a + w_b + w_c <= 1. Since w_a + w_b + w_c = 1, w_b + w_c = 1 - w_a.
    # 2w_a + 1 - w_a <= 1 => w_a <= 0. Since w_a >= 0, w_a must be 0.
    # Then w_b <= 1/2 and w_c <= 1/2. Since w_b + w_c = 1, this is possible (w_b=0.5, w_c=0.5).
    # So two pairs are not enough.
    #
    # Thus, a certificate is a triangle (clique of size 3).
    # We need the lexicographically smallest triangle.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v: u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), iterate a, then b, then c.
    # a ranges from 1 to n-2
    for a in range(1, n + 1):
        # b ranges from a+1 to n-1
        # b must be connected to a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        for b in neighbors_a:
            # c must be > b and connected to both a and b
            # Check intersection of adj[a] and adj[b]
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
