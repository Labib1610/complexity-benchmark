import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # Sort edges to help find lexicographically smallest trio
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all possible 'a' from 1 to n
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # For a fixed 'a', we need to find b, c such that b, c are neighbors of a
        # and (b, c) is an edge.
        # To ensure lexicographical order, we iterate b in increasing order
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is a neighbor of a AND c is a neighbor of b
            # Intersection of adj[a] and adj[b]
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
