import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # The condition "no matter which single substance is excluded, 
    # the remaining two react" means that for a triplet {a, b, c},
    # (a, b), (b, c), and (a, c) must all be in the set of reactive pairs.
    # This is equivalent to finding a triangle (clique of size 3) in the graph.
    
    # To find the lexicographically smallest triangle {a, b, c} with a < b < c:
    # Iterate through all possible 'a' from 1 to n-2.
    # For each 'a', iterate through all neighbors 'b' of 'a' such that b > a.
    # Then check if there exists a 'c' such that c > b and c is a neighbor of both a and b.
    
    # Sort neighbors to ensure we find the lexicographically smallest triplet
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and adj[b]
            # Since adj lists are sorted, we can intersect them efficiently
            # or just iterate through the smaller list.
            # Given the constraints and the requirement for lexicographical order,
            # we check neighbors of b that are also neighbors of a.
            
            # Optimization: iterate through neighbors of b that are > b
            for c in adj[b]:
                if c <= b:
                    continue
                if c in adj[a]:
                    print(f"{a} {b} {c}")
                    return

solve()
