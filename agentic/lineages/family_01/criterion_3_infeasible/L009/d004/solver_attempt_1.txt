import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest trio (a, b, c) with a < b < c
    # such that (a, b), (b, c), and (a, c) are all edges.
    # We iterate through all possible 'a' from 1 to n-2.
    # For a fixed 'a', we iterate through all neighbors 'b' of 'a' where b > a.
    # Then we check if there exists a 'c' such that c > b and c is a neighbor of both a and b.
    
    # Sort neighbors to ensure we find the lexicographically smallest trio
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # Since neighbors_a is sorted, we can intersect efficiently
            # or just iterate through neighbors_a after index i
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
