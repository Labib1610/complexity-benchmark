import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list to store edges
    # Since we need to check if (u, v), (v, w), and (u, w) exist,
    # an adjacency matrix or a set of edges is efficient.
    # Given n <= 2000, an adjacency matrix of booleans is fine.
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # A dense triplet is formed by three pairs (a, b), (b, c), (a, c).
    # This means the three items {a, b, c} form a triangle in the graph.
    # We need to find the lexicographically smallest (a, b, c) such that
    # all three edges (a, b), (b, c), and (a, c) exist.
    
    # Iterate through all possible triplets (a, b, c) with 1 <= a < b < c <= n
    # To optimize, iterate over edges (a, b) and check for a common neighbor c.
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we can iterate
    # through all possible a, then b, then c.
    # Since n is 2000, O(n^3) is too slow, but O(m * n) or O(n * degree^2) is fine.
    
    for a in range(1, n + 1):
        # Get all neighbors of a
        neighbors_a = sorted(list(adj[a]))
        len_na = len(neighbors_a)
        if len_na < 2:
            continue
            
        for i in range(len_na):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # neighbors_a is sorted, so we look for c in neighbors_a[i+1:]
            # and check if c is in adj[b]
            for j in range(i + 1, len_na):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
