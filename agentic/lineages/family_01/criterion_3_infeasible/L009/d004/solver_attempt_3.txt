import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest trio (a, b, c) with a < b < c,
    # we iterate through all edges (a, b) and check for a common neighbor c.
    # Since we want the smallest (a, b, c), we sort the edges first.
    edges.sort()
    
    # To optimize, we can iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we look for 'b' and 'c' such that (a,b), (a,c), and (b,c) exist.
    # This is equivalent to finding a triangle in the graph.
    
    # Sort neighbors for each node to help find 'c' efficiently
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need to find c > b such that c is in adj[a] and c is in adj[b]
            # Since neighbors are sorted, we can use two pointers or set intersection
            # Given the constraints and the requirement for the lexicographically smallest,
            # we check neighbors of b that are also neighbors of a.
            
            # Optimization: only check c > b
            neighbors_b = adj[b]
            
            # Use set for O(1) lookup if needed, but here we can just iterate
            # Since we want the smallest (a, b, c), we iterate b and c in order.
            ptr_b = 0
            for c in neighbors_b:
                if c <= b:
                    continue
                # Check if c is also a neighbor of a
                # Since adj[a] is sorted, use binary search or just check
                # Actually, since we iterate through neighbors_a, we can just check
                # if c is in adj[a].
                if c in adj[a]:
                    print(f"{a} {b} {c}")
                    return

solve()
