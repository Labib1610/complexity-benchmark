```python
import sys

def solve():
    # The problem asks for three materials {a, b, c} such that the maximum 
    # size of a safe set (independent set) is 1.
    # In a graph of 3 vertices, the maximum independent set size is 1 
    # if and only if the graph contains at least 3 edges (i.e., it is a triangle).
    # Wait, let's re-check:
    # If the graph is a triangle (edges (a,b), (b,c), (a,c)), 
    # the max independent set is 1.
    # If the graph has 2 edges (e.g., (a,b), (b,c)), 
    # the max independent set is 2 (e.g., {a, c}).
    # If the graph has 1 edge, max independent set is 2.
    # If the graph has 0 edges, max independent set is 3.
    # So the condition "max independent set is 1" is equivalent to 
    # the three materials forming a triangle (a clique of size 3).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # We need the lexicographically smallest {a, b, c} such that 
    # (a,b), (b,c), (a,c) are all edges.
    # Iterate through all edges (u, v) and check for a common neighbor w.
    # To ensure lexicographical order, we can iterate through all possible 
    # triplets or optimize.
    
    # Optimization: iterate through all edges (u, v) and find common neighbors.
    # Since n <= 2000, O(m * n / 64) or O(m * sqrt(m)) is fine.
    
    best = None
    
    # Sort edges to help find the smallest triplet
    edges.sort()
    
    for u, v in edges:
        # We want to find w such that {u, v, w} is a triangle.
        # Let the sorted triplet be (a, b, c).
        # w can be smaller than u, between u and v, or larger than v.
        
        # Common neighbors of u and v
        common = adj[u].intersection(adj[v])
        for w in common:
            # Form the triplet and sort it
            triplet = sorted([u, v, w])
            if best is None or triplet < best:
                best = triplet
                
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
