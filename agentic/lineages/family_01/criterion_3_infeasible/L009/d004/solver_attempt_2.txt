import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Optimization: Sort adjacency lists to make intersection faster
    sorted_adj = [sorted(list(s)) for s in adj]
    
    for a in range(1, n - 1):
        adj_a = sorted_adj[a]
        # Only consider neighbors b > a
        neighbors_a = [x for x in adj_a if x > a]
        
        for b in neighbors_a:
            # We need a c such that c > b and c is in adj[a] and adj[b]
            # Since we want the smallest (a, b, c), we check neighbors of b
            # that are also neighbors of a and are > b.
            adj_b = adj[b]
            for c in sorted_adj[b]:
                if c > b:
                    if c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

solve()
