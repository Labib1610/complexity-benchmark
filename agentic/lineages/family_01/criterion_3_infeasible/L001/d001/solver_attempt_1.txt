```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # A runaway condition is a triangle (a, b, c) where all pairs are reactive.
    # We need to find the lexicographically smallest (a, b, c) with a < b < c.
    # Iterate through all edges (u, v) and find a common neighbor w.
    
    best = None
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    for u, v in edges:
        # u < v is guaranteed by the input processing
        # We look for w such that (u, w) and (v, w) exist.
        # To minimize (a, b, c), we check neighbors of u and v.
        
        # Common neighbors
        common = adj[u].intersection(adj[v])
        for w in common:
            # Form the triplet (a, b, c)
            triplet = sorted([u, v, w])
            if best is None or triplet < best:
                best = triplet
                
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
