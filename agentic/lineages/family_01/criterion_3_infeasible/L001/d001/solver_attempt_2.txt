import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # We iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we look for 'b' such that 'a' and 'b' are connected.
    # Then we check if there exists 'c' such that 'b' and 'c' are connected AND 'c' and 'a' are connected.
    # To ensure lexicographical order, we iterate a, then b, then c.
    
    # Sort adjacency lists to help with lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Optimization: iterate through all edges (u, v) as potential (a, b)
    # and check for a common neighbor c > v.
    # But we need the lexicographically smallest (a, b, c).
    
    # Let's iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # For a fixed a, we want the smallest b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Now we need the smallest c > b such that c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can intersect efficiently
            # or just iterate through neighbors_b
            neighbors_b = adj[b]
            
            # Intersection of neighbors_a and neighbors_b
            # Both are sorted.
            ptr_a = 0
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    c = val_a
                    if c > b:
                        print(f"{a} {b} {c}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
