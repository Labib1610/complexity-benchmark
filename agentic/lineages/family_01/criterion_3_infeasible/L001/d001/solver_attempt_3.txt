import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Sort neighbors to potentially speed up intersection
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Only consider neighbors b > a
        neighbors_a = [x for x in adj_list[a] if x > a]
        for b in neighbors_a:
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # Since adj_list is sorted, we can find common elements efficiently
            # or just iterate through the smaller neighbor list
            
            # Optimization: check intersection of adj[a] and adj[b]
            # Since we need c > b, we only care about elements > b
            
            # Using set intersection for speed
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
