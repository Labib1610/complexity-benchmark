import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    suffix = [0] * (n + 1)
    for i in range(1, n + 1):
        suffix[i] = ~((1 << i) - 1)

    for a in range(1, n + 1):
        mask_b = adj[a] >> a
        while mask_b:
            lsb = mask_b & -mask_b
            b = lsb.bit_length() + a
            common = adj[a] & adj[b]
            common_gt = common & suffix[b]
            if common_gt:
                c = (common_gt & -common_gt).bit_length()
                sys.stdout.write(f"{a} {b} {c}")
                return
            mask_b ^= lsb

solve()
