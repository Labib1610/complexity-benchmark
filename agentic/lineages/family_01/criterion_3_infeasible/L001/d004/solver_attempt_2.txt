import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we iterate b in adj[a] such that b > a.
    # Then we check if there exists c in adj[a] intersect adj[b] such that c > b.
    
    # Sort adjacency lists to ensure we check in lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Find common neighbors c > b
            # Since adj[a] and adj[b] are sorted, we can use two pointers
            # or just iterate through the smaller list.
            # Given n=2000, O(n^2) or O(m * sqrt(m)) is acceptable.
            
            set_b = set(adj[b])
            for c in neighbors_a[i+1:]:
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

solve()
