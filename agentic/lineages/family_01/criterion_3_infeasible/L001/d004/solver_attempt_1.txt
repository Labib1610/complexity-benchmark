import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        if len_a < 2:
            continue
            
        # Iterate b from neighbors of a
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c such that c is a neighbor of a AND c is a neighbor of b
            # c must be > b
            # Intersection of adj[a] and adj[b]
            # Since we want smallest b, we check neighbors of a in order
            
            # Optimization: intersect sets
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
