import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to process in order (u, v) with u < v
    edges.sort()
    
    # For each edge (u, v), check common neighbors in sorted order
    for u, v in edges:
        # Iterate through neighbors of u in sorted order
        neighbors_u = sorted(adj[u])
        for w in neighbors_u:
            if w != v and w in adj[v]:
                # Found a triangle (u, v, w)
                result = sorted([u, v, w])
                print(f"{result[0]} {result[1]} {result[2]}")
                return
    
    # Should not reach here as problem guarantees at least one valid subsystem
    pass

if __name__ == "__main__":
    solve()
