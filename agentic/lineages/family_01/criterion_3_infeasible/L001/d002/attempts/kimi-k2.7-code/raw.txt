```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        N[u] |= 1 << v
        N[v] |= 1 << u
    bit = [1 << i for i in range(n)]
    suffix = [~((1 << (i + 1)) - 1) for i in range(n)]
    for a in range(n):
        Na = N[a]
        for b in range(a + 1, n):
            if Na & bit[b]:
                common = Na & N[b] & suffix[b]
                if common:
                    c = (common & -common).bit_length() - 1
                    print(f"{a + 1} {b + 1} {c + 1}")
                    return

solve()
```
