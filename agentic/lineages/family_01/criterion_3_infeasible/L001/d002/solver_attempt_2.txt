```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n-2, b from a+1 to n-1, c from b+1 to n.
    # Since we need to check if a, b, c form a triangle,
    # we check if b in adj[a], c in adj[a], and c in adj[b].
    
    for a in range(1, n - 1):
        # Optimization: only consider b if a is connected to b
        for b in sorted(list(adj[a])):
            if b <= a:
                continue
            
            # Intersection of neighbors of a and b gives potential c's
            common = adj[a].intersection(adj[b])
            
            # Filter common neighbors that are greater than b
            valid_c = [c for c in common if c > b]
            
            if valid_c:
                c = min(valid_c)
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
```
