import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through possible a, then b, then c.
    # A triangle (a, b, c) exists if a < b < c and
    # (a, b) in E, (b, c) in E, (a, c) in E.
    
    # Sort edges to help iterate efficiently
    # We want to minimize a, then b, then c.
    # Iterate a from 1 to n-2
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        # For each pair (b, c) in neighbors_a, check if (b, c) is an edge
        # To minimize b and c, we iterate through sorted neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need to check if there exists c > b such that
            # c is in neighbors_a AND c is in adj[b]
            # To get lexicographically smallest, we check c in increasing order
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
