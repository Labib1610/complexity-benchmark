import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for deterministic iteration
    adj_lists = [sorted(s) for s in adj]
    
    for a in range(1, n + 1):
        neighbors_a = adj_lists[a]
        # Try to find b and c in neighbors of a such that b < c and edge (b, c) exists
        # Iterate b from smallest to largest
        for b in neighbors_a:
            if b <= a:
                continue
            # For c, iterate neighbors of a that are > b
            for c in neighbors_a:
                if c <= b:
                    continue
                # Check if edge (b, c) exists
                if c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
