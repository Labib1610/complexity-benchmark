import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, b from a+1 to n, and c from b+1 to n.
    # Since we need to check if a, b, c form a triangle,
    # we check if b in adj[a] and c in adj[a] and c in adj[b].
    
    # Optimization: iterate a, then iterate b in adj[a], then check intersection.
    # To ensure lexicographical order, we sort the adjacency lists.
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of sorted_adj[a] and sorted_adj[b]
            # Since we need the smallest c, we can just iterate through
            # the intersection.
            
            # To make it efficient, use the smaller adjacency list to check
            if len(adj[a]) < len(adj[b]):
                for c in sorted_adj[a]:
                    if c > b and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in sorted_adj[b]:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

solve()
