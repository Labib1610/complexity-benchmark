import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we can iterate through sorted edges
    # or iterate through nodes.
    
    # Sorting nodes by degree can help, but for n=2000, O(n^3) is too slow.
    # However, we only need to check triangles.
    # A triangle (a, b, c) exists if a < b < c and b in adj[a] and c in adj[a] and c in adj[b].
    
    # Sort adjacency lists to help with lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Iterate b in neighbors_a such that b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c such that c > b and c is in adj[a] and c is in adj[b]
            # Intersection of sorted lists neighbors_a and neighbors_b
            # Since we want the smallest c, we just need the first common element > b
            neighbors_b = adj[b]
            
            # Use two pointers or set intersection to find common neighbors
            ptr_a = i + 1
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
