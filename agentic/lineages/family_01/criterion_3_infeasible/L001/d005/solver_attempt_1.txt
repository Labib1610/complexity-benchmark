import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To minimize x * 10^8 + y * 10^4 + z, we iterate x from 1 to n,
    # then y from x+1 to n, then check if there exists z > y such that
    # z is in adj[x] and adj[y].
    
    # Sort edges to help iterate efficiently
    edges.sort()
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # We need to find the lexicographically smallest (x, y, z)
    # Iterate through all pairs (x, y) that have an edge
    for x in range(1, n + 1):
        for y in adj[x]:
            if y <= x:
                continue
            
            # Now find the smallest z > y such that z is in adj[x] and adj[y]
            # Since adj lists are sorted, we can use intersection
            # But we only need the first one
            ptr_x = 0
            ptr_y = 0
            list_x = adj[x]
            list_y = adj[y]
            
            while ptr_x < len(list_x) and ptr_y < len(list_y):
                val_x = list_x[ptr_x]
                val_y = list_y[ptr_y]
                
                if val_x == val_y:
                    if val_x > y:
                        print(f"{x} {y} {val_x}")
                        return
                    ptr_x += 1
                    ptr_y += 1
                elif val_x < val_y:
                    ptr_x += 1
                else:
                    ptr_y += 1

solve()
