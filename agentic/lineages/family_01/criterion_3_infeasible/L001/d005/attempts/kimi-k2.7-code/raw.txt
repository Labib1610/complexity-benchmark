import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for a in range(1, n + 1):
        # iterate b > a in increasing order
        remaining = adj[a] & ~((1 << (a + 1)) - 1)
        while remaining:
            b_bit = remaining & -remaining
            b = b_bit.bit_length() - 1
            # intersection of neighbors of a and b, only c > b
            common = adj[a] & adj[b]
            candidates = common & ~((1 << (b + 1)) - 1)
            if candidates:
                c_bit = candidates & -candidates
                c = c_bit.bit_length() - 1
                sys.stdout.write(f"{a} {b} {c}")
                return
            remaining ^= b_bit

if __name__ == "__main__":
    solve()
