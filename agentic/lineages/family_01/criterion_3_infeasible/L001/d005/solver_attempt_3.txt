import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) in the graph where edges represent inert pairs.
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n,
    # then b from a+1 to n, then c from b+1 to n.
    # Optimization: Only check b if b is connected to a.
    # Then check if c is connected to both a and b.
    
    # Sort adjacency lists to iterate in order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        if len(neighbors_a) < 2:
            continue
            
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # To make this efficient, use set intersection or sorted list intersection
            # Since we want the smallest c, we can iterate through neighbors of b
            # that are also neighbors of a.
            
            neighbors_b = adj[b]
            for c in neighbors_a[i+1:]:
                if c in neighbors_b:
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
