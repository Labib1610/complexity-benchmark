import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges lexicographically by (u, v)
    edges.sort()
    
    # For each edge (u, v), find the smallest w > v such that w is adjacent to both u and v
    for u, v in edges:
        # Find common neighbors > v
        # Iterate over the smaller adjacency set
        if len(adj[u]) <= len(adj[v]):
            for w in adj[u]:
                if w > v and w in adj[v]:
                    print(u, v, w)
                    return
        else:
            for w in adj[v]:
                if w > v and w in adj[u]:
                    print(u, v, w)
                    return
    
    # Fallback (should not reach here given problem guarantees)
    # Brute force search for any triangle
    for a in range(1, n + 1):
        for b in adj[a]:
            if b <= a:
                continue
            for c in adj[a]:
                if c > b and c in adj[b]:
                    print(a, b, c)
                    return

solve()
