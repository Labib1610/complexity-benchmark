```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    all_bits = (1 << n) - 1
    # mask of vertices with index > b (1-indexed)
    mask_gt = [0] * (n + 1)
    for b in range(1, n + 1):
        # keep bits b .. n-1 (0-indexed), i.e., vertices b+1 .. n
        mask_gt[b] = all_bits ^ ((1 << b) - 1)

    for a in range(1, n + 1):
        bits_b = adj[a] & mask_gt[a]
        while bits_b:
            low = bits_b & -bits_b
            b = low.bit_length()  # since bit_length of 1<< (b-1) is b
            common = adj[a] & adj[b]
            masked = common & mask_gt[b]
            if masked:
                c = (masked & -masked).bit_length()
                sys.stdout.write(f"{a} {b} {c}")
                return
            bits_b ^= low

if __name__ == "__main__":
    solve()
```
