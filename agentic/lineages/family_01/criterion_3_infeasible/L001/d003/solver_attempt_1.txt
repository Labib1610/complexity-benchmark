```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triangle (i, j, k) with i < j < k:
    # Iterate i from 1 to n-2
    # Iterate j from i+1 to n-1
    # Check if j is connected to i
    # If so, iterate k from j+1 to n
    # Check if k is connected to both i and j
    
    # Optimization: Only iterate over neighbors of i for j, and neighbors of i and j for k
    for i in range(1, n + 1):
        neighbors_i = sorted(list(adj[i]))
        for idx_j in range(len(neighbors_i)):
            j = neighbors_i[idx_j]
            if j <= i:
                continue
            
            # We need k > j such that k is in adj[i] and adj[j]
            # Intersection of sorted lists is efficient
            common = sorted(list(adj[i].intersection(adj[j])))
            for k in common:
                if k > j:
                    print(f"{i} {j} {k}")
                    return

if __name__ == '__main__':
    solve()
```
