import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # Sort edges to help find lexicographically smallest triangle
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (u, v) as potential (a, b)
    # Then check if there exists a w such that (u, w) and (v, w) exist.
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b from a+1 to n, then check if they share a neighbor c > b.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        adj_a = adj[a]
        for i in range(len(adj_a)):
            b = adj_a[i]
            # We need to find c > b such that c is in adj[a] and adj[b]
            # Since adj lists are sorted, we can use two pointers or set intersection
            adj_b = adj[b]
            
            # Optimization: use set for faster lookup if needed, 
            # but here we can just iterate through the smaller adjacency list
            if len(adj_a) < len(adj_b):
                for c in adj_a:
                    if c > b and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in adj_b:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

solve()
