```python
import sys

def solve():
    # The condition "if any one of the three systems were to temporarily suspend 
    # its trade operations, the remaining two systems in the alliance must still 
    # have an active, direct trade route between them" means that for any 
    # subset of 3 nodes {a, b, c}, the edges (a,b), (b,c), and (a,c) must all exist.
    # This is simply the definition of a triangle (a clique of size 3).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, we can iterate through all edges (a, b) and check 
    # for a common neighbor c.
    
    best = None
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    # To optimize, iterate through all nodes as 'a'
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that c > b and c is in both adj[a] and adj[b]
            # Intersection of neighbors
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
        
        # If we found a triangle starting with 'a', we don't need to check 
        # larger 'a' values if we want the lexicographically smallest.
        if best is not None:
            print(f"{best[0]} {best[1]} {best[2]}")
            return

solve()
```
