```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that a < b < c
    # b is in adj[a], c is in adj[a], c is in adj[b]
    # This is equivalent to finding a triangle (a, b, c)
    # Lexicographical order: iterate a from 1 to n, b from a+1 to n, c from b+1 to n
    
    # To optimize, iterate a, then iterate b in adj[a] where b > a
    # Then check if there exists c in adj[a] AND adj[b] where c > b
    
    for a in range(1, n + 1):
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for b in neighbors_a:
            # Find common neighbors of a and b that are > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
```
