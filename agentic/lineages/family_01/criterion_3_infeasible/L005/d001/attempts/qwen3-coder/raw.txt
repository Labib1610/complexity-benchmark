import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Convert to adjacency list for faster lookup
    adj_list = {i: adj[i] for i in range(1, n+1)}
    
    # Create a set of edges for O(1) lookup
    edge_set = set(edges)
    # Add reverse edges as well since the graph is undirected
    for u, v in edges:
        edge_set.add((v, u))
    
    # We need to find a triangle (3 nodes all connected to each other)
    # That means we need to find three nodes a, b, c such that:
    # (a,b), (a,c), (b,c) all exist
    
    # To ensure lexicographical order, we'll iterate through nodes in order
    found = False
    result = None
    
    # Try all pairs (a,b) where a < b
    for a in range(1, n+1):
        if found:
            break
        for b in range(a+1, n+1):
            if found:
                break
            # Check if there's an edge between a and b
            if b not in adj_list[a]:
                continue
            # Now look for a c > b such that both (a,c) and (b,c) exist
            for c in range(b+1, n+1):
                if c in adj_list[a] and c in adj_list[b]:
                    print(a, b, c)
                    found = True
                    break

if __name__ == "__main__":
    main()
