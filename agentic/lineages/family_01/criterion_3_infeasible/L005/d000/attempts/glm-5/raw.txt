import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort adjacency lists to ensure we find the lexicographically smallest triangle
    # when iterating through neighbors.
    # However, we iterate through edges. For an edge (u, v), we iterate through common neighbors w.
    # To find the smallest (u, v, w), we should iterate w in sorted order.
    # We can convert sets to sorted lists for iteration.
    # But sorting all adjacency lists takes O(sum(deg) log deg) which is O(m log n).
    # Given constraints, this is acceptable.
    
    # Optimization: We can iterate through edges and find common neighbors.
    # To find the lexicographically smallest triangle, we can iterate u from 1 to n,
    # then v from u+1 to n, then w from v+1 to n.
    # But iterating all pairs is O(n^2).
    # Better approach: iterate through edges (u, v) where u < v.
    # For each edge, iterate through common neighbors w.
    # We want the smallest (u, v, w) where u < v < w.
    # If we iterate edges in arbitrary order, we might find a triangle (u, v, w) but there might be a smaller one.
    # However, the lexicographically smallest triangle (a, b, c) will be found when we process edge (a, b).
    # At that point, we need to find the smallest w > b such that w is a common neighbor.
    # So for each edge (u, v) with u < v, we iterate common neighbors w in ascending order.
    # The first valid w we find for this edge gives us a candidate triangle (u, v, w).
    # We keep track of the minimum such triangle found so far.
    
    # To iterate common neighbors efficiently, we iterate through the smaller degree node's neighbors.
    # And we check if the neighbor is in the larger degree node's set.
    # To ensure we find the smallest w, we should iterate neighbors in sorted order.
    
    # Let's pre-sort neighbors for each node.
    sorted_adj = [sorted(s) for s in adj]
    
    best_a, best_b, best_c = n + 1, n + 1, n + 1
    
    for u, v in edges:
        if u > v:
            u, v = v, u
        
        # We are looking for w such that u < v < w and w is connected to both.
        # This corresponds to the triangle (u, v, w).
        # Any triangle (a, b, c) will be found when processing edge (a, b).
        
        # Iterate through neighbors of the node with smaller degree
        if len(adj[u]) < len(adj[v]):
            small_node, large_node = u, v
        else:
            small_node, large_node = v, u
            
        # Iterate neighbors of small_node in sorted order to find smallest w
        # We need w > v to maintain order u < v < w
        for w in sorted_adj[small_node]:
            if w > v and w in adj[large_node]:
                # Found a candidate triangle (u, v, w)
                if (u < best_a) or \
                   (u == best_a and v < best_b) or \
                   (u == best_a and v == best_b and w < best_c):
                    best_a, best_b, best_c = u, v, w
                    # Since we are iterating w in sorted order, this is the best w for this edge.
                    # We can break to save time, but we must continue checking other edges
                    # because another edge might yield a smaller (u, v) pair.
                    break
    
    print(best_a, best_b, best_c)

if __name__ == '__main__':
    solve()
