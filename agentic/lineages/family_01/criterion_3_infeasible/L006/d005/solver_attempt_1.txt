```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that 1 <= a < b < c <= n
    # and (a,b), (a,c), (b,c) are all in the compatibility set.
    # The problem asks for the lexicographically smallest (a, b, c).
    
    for a in range(1, n + 1):
        # Get all neighbors of a that are greater than a
        neighbors_a = [v for v in adj[a] if v > a]
        neighbors_a.sort()
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that b < c <= n, c is in adj[a], and c is in adj[b]
            # So c must be in the intersection of adj[a] and adj[b]
            # and c > b
            
            # Optimization: find intersection of adj[a] and adj[b]
            # Since n is small (up to 2000), this is efficient.
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
