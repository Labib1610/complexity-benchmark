import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically first (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and adj[b]
    
    for a in range(1, n - 1):
        # Get neighbors of a that are greater than a
        neighbors_a = [x for x in adj[a] if x > a]
        neighbors_a.sort()
        
        for b in neighbors_a:
            # We need a c > b such that c is in adj[a] and adj[b]
            # Intersection of adj[a] and adj[b]
            # Since we want the smallest c, we can iterate through sorted neighbors
            # or just check intersection
            
            # Optimization: check intersection of smaller set
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
                other = adj[b]
            else:
                candidates = adj[b]
                other = adj[a]
            
            for c in sorted(candidates):
                if c > b and c in other:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
