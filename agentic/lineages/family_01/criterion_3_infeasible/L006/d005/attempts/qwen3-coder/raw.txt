import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # For each node, sort its neighbors
    for u in adj:
        adj[u] = sorted(adj[u])
    
    # We want the lexicographically smallest triangle
    # So we iterate through all edges (u,v) with u < v
    # and for each such edge, we look for a third node w
    # such that w > v and w is connected to both u and v
    # We want the smallest such w for the smallest (u,v)
    
    # Sort edges to process in lexicographical order of u, then v
    edges.sort()
    
    for u, v in edges:
        if u >= v:
            continue
        # Find common neighbors of u and v that are > v
        # Since adj[u] and adj[v] are sorted, we can use two pointers or just iterate
        # We want the smallest w in adj[u] ∩ adj[v] such that w > v
        neighbors_u = adj[u]
        neighbors_v = adj[v]
        
        i = 0
        j = 0
        while i < len(neighbors_u) and j < len(neighbors_v):
            w_u = neighbors_u[i]
            w_v = neighbors_v[j]
            if w_u == w_v:
                w = w_u
                if w > v:
                    print(u, v, w)
                    return
                elif w <= v:
                    i += 1
                    j += 1
            elif w_u < w_v:
                i += 1
            else:
                j += 1
    
    # As per problem statement, at least one valid profile exists
    # So we should always find one before reaching here

if __name__ == "__main__":
    main()
