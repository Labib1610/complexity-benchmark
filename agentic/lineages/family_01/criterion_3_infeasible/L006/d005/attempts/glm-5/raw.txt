```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    
    for a in range(1, n - 1):
        neighbors = sorted(adj[a])
        size = len(neighbors)
        for i in range(size):
            b = neighbors[i]
            if b <= a:
                continue
            for j in range(i + 1, size):
                c = neighbors[j]
                if c <= a:
                    continue
                if c in adj[b]:
                    print(a, b, c)
                    return
    
    # Fallback (should not be reached due to problem guarantees)
    for a in range(1, n - 1):
        for b in range(a + 1, n):
            if b not in adj[a]:
                continue
            for c in range(b + 1, n + 1):
                if c in adj[a] and c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
```
