```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that:
    # a is connected to b
    # a is connected to c
    # b is connected to c
    # This is equivalent to finding a triangle (a, b, c) in the graph.
    # We need to output a, b, c such that a < b < c and (a, b, c) is lexicographically smallest.
    
    # Iterate through all possible a, b, c
    # Since n <= 2000, O(n^3) is too slow, but we can optimize.
    # A triangle is a set of 3 vertices where each pair is connected.
    
    for a in range(1, n + 1):
        # Get neighbors of a
        neighbors_a = sorted(list(adj[a]))
        len_na = len(neighbors_a)
        if len_na < 2:
            continue
            
        for i in range(len_na):
            b = neighbors_a[i]
            for j in range(i + 1, len_na):
                c = neighbors_a[j]
                
                # Check if b and c are connected
                if c in adj[b]:
                    # Found a triangle (a, b, c)
                    # Sort them to satisfy a < b < c
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

if __name__ == '__main__':
    solve()
```
