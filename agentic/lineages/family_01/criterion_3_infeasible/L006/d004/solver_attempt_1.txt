```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically first (a, b, c) with a < b < c,
    # we iterate through all possible a, then b, then c.
    # Since we need a triangle, we check if b is in adj[a] and c is in adj[a] and c is in adj[b].
    
    # Sort edges to help iterate efficiently
    edges.sort()
    
    # We can iterate through all pairs (a, b) that are connected,
    # then check if there exists a c > b such that (a, c) and (b, c) are connected.
    
    # To optimize, iterate a from 1 to n-2
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # This is the intersection of neighbors_a[i+1:] and adj[b]
            
            # Optimization: iterate through the smaller set
            if len(neighbors_a) - i - 1 < len(adj[b]):
                for j in range(i + 1, len(neighbors_a)):
                    c = neighbors_a[j]
                    if c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                # Iterate through neighbors of b that are > b
                for c in sorted(adj[b]):
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

solve()
```
