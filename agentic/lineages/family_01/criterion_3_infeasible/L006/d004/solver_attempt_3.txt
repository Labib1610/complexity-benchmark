import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically first (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is connected to a
    # If so, iterate c from b+1 to n
    # Check if c is connected to both a and b
    
    # Optimization: sort adjacency lists to iterate efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # We need c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we need the smallest c, we can iterate through sorted neighbors
            # But we need to be careful about complexity.
            # Given n=2000, O(n^3) is too slow, but O(m * n) or O(n * degree^2) is fine.
            
            # Find common neighbors of a and b that are > b
            # Using sets for O(1) lookup
            set_a = adj[a]
            set_b = adj[b]
            
            # Iterate through the smaller neighbor list to find common ones
            if len(set_a) < len(set_b):
                candidates = [x for x in sorted_adj[a] if x > b and x in set_b]
            else:
                candidates = [x for x in sorted_adj[b] if x > b and x in set_a]
            
            if candidates:
                print(f"{a} {b} {min(candidates)}")
                return

solve()
