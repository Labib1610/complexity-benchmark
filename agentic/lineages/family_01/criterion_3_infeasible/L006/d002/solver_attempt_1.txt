import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # Iterate through all edges (u, v) where u < v.
    # For each edge, check if there exists a node w such that (u, w) and (v, w) exist.
    # To ensure lexicographical order, we iterate u from 1 to n, then v from u+1 to n.
    
    edges.sort()
    
    # Pre-sort adjacency lists to help find w efficiently
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    best_triad = None
    
    for u in range(1, n + 1):
        for v in adj[u]:
            # We have edge (u, v) with u < v.
            # Look for w such that w > v and w is in both adj[u] and adj[v].
            # Or w < u, or u < w < v.
            # Actually, just find any w that forms a triangle and keep the min.
            
            # Intersection of sorted lists
            i = 0
            j = 0
            list_u = adj[u]
            list_v = adj[v]
            while i < len(list_u) and j < len(list_v):
                w = list_u[i]
                if w == list_v[j]:
                    # Found a triangle (u, v, w)
                    nodes = sorted([u, v, w])
                    if best_triad is None or nodes < best_triad:
                        best_triad = nodes
                    i += 1
                    j += 1
                elif w < list_v[j]:
                    i += 1
                else:
                    j += 1
                    
    if best_triad:
        print(f"{best_triad[0]} {best_triad[1]} {best_triad[2]}")

solve()
