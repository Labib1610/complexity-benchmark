import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges lexicographically by (u, v)
    edges.sort()
    
    # For each edge (u, v), find the smallest w > v such that w is adjacent to both u and v.
    # Since we iterate edges in sorted order, the first found triangle is lexicographically smallest.
    for u, v in edges:
        # Iterate over the smaller adjacency set for efficiency
        if len(adj[u]) <= len(adj[v]):
            smaller = adj[u]
            larger = adj[v]
        else:
            smaller = adj[v]
            larger = adj[u]
        
        # We need w > v to ensure u < v < w
        # Find the smallest w in the intersection that is > v
        best_w = n + 1
        for w in smaller:
            if w > v and w in larger:
                if w < best_w:
                    best_w = w
        if best_w <= n:
            print(f"{u} {v} {best_w}")
            return
    
    # Should not reach here because problem guarantees at least one valid set exists
    print("-1 -1 -1")

solve()
