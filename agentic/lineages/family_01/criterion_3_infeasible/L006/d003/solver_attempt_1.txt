```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need to find a triangle (a, b, c) such that a < b < c
    # To find the lexicographically smallest, we iterate through possible a, b, c
    # Since n <= 2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    # Sorting edges by u then v helps.
    edges.sort()
    
    best = None
    
    # Iterate through all edges (u, v) as the first two nodes of the triangle
    for i in range(m):
        u, v = edges[i]
        # We want to find w such that (u, v, w) is a triangle.
        # w must be connected to both u and v.
        # To ensure lexicographical order, we check potential w's.
        
        # Intersection of adj[u] and adj[v]
        common = adj[u].intersection(adj[v])
        for w in common:
            # Form the trio and sort it
            trio = sorted([u, v, w])
            if best is None or trio < best:
                best = trio
                
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
