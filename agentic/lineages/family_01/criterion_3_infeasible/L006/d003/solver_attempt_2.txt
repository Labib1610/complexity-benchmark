```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically first (a, b, c)
    # We want smallest a, then smallest b, then smallest c.
    # Iterate through all possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    
    edges.sort()
    
    # To optimize, iterate through all edges (a, b) where a < b
    # Then check if there exists a c > b such that c is in adj[a] and adj[b]
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we have a < b. We need c > b such that c in adj[a] and c in adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we want the smallest c, we can iterate through sorted neighbors
            
            # Optimization: only check c > b
            # We can use set intersection or just iterate
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

solve()
```
