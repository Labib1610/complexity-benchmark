import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically first (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2,
    # b from a+1 to n-1, and c from b+1 to n.
    # To optimize, we only check b in adj[a] and c in adj[a] & adj[b].
    
    # Sort adjacency lists to ensure we check in lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Neighbors of a that are greater than a
        neighbors_a = [x for x in adj[a] if x > a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and adj[b]
            # To keep it lexicographical, we check neighbors of b that are > b
            for c in adj[b]:
                if c > b:
                    if c in adj_sets[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
