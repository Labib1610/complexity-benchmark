```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) with a < b < c.
    # To ensure lexicographical order, we iterate a from 1 to n-2,
    # b from a+1 to n-1, and c from b+1 to n.
    # Since n <= 2000, O(n^3) is too slow, but we can optimize by
    # checking if c exists in adj[a] and adj[b].
    
    for a in range(1, n + 1):
        # Sort neighbors to help with lexicographical order
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that b < c and c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # To keep it efficient, use set intersection
            common = adj[a].intersection(adj[b])
            # Filter for c > b
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
```
