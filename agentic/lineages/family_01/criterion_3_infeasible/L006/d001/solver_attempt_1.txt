import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        idx += 2
        
    # We need to find a, b, c such that a < b < c
    # and (a,b), (a,c), (b,c) are all in the set of forbidden pairs.
    # To find the lexicographically smallest (a, b, c),
    # we iterate through a from 1 to n, b from a+1 to n, c from b+1 to n.
    
    # Optimization: iterate through a, then iterate through neighbors of a (b),
    # then check if there exists a c such that c is a neighbor of both a and b.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of adj[a] and adj[b]
            # Since we need the lexicographically smallest, we check candidates for c
            # in increasing order.
            
            # Only consider b > a
            if b <= a:
                continue
                
            # Find common neighbors c > b
            # To be efficient, use set intersection
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

solve()
