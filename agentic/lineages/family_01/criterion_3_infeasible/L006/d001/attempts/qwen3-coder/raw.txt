```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list using sets for O(1) lookup
    adj = defaultdict(set)
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        adj[u].add(v)
        adj[v].add(u)
        index += 2
    
    # Convert to list for ordered iteration
    nodes = sorted(adj.keys())
    
    # For each node, check pairs among its neighbors
    for a in nodes:
        neighbors = sorted(list(adj[a]))
        for i in range(len(neighbors)):
            b = neighbors[i]
            if b <= a:
                continue
            for j in range(i+1, len(neighbors)):
                c = neighbors[j]
                if c <= b:
                    continue
                # Check if b and c are connected
                if c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
