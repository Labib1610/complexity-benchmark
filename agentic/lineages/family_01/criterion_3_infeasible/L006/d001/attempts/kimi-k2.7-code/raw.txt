import sys

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = data[idx]
        v = data[idx + 1]
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for a in range(1, n - 1):
        nb = adj[a] & ~((1 << (a + 1)) - 1)
        while nb:
            low = nb & -nb
            b = low.bit_length() - 1
            common = adj[a] & adj[b] & ~((1 << (b + 1)) - 1)
            if common:
                c = (common & -common).bit_length() - 1
                sys.stdout.write(f"{a} {b} {c}")
                return
            nb ^= low

if __name__ == "__main__":
    solve()
