import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    
    # Iterate in lexicographic order
    for a in range(1, n - 1):
        adj_a = adj[a]
        # Candidates b > a that are adjacent to a
        candidates = [b for b in adj_a if b > a]
        # Sort candidates to ensure lexicographic order
        candidates.sort()
        L = len(candidates)
        for i in range(L):
            b = candidates[i]
            adj_b = adj[b]
            for j in range(i + 1, L):
                c = candidates[j]
                if c in adj_b:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
