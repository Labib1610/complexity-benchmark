 ```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    out_adj = [0] * (n + 1)
    edges = []

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        if u > v:
            u, v = v, u
        out_adj[u] |= 1 << v
        edges.append((u, v))

    edges.sort()

    for u, v in edges:
        common = out_adj[u] & out_adj[v]
        if common:
            low = common & -common
            w = low.bit_length() - 1
            sys.stdout.write(f"{u} {v} {w}")
            return

if __name__ == "__main__":
    solve()
```
