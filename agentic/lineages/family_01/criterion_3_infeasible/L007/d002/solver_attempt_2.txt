import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    adj_set = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].append(v)
        adj_set[u].add(v)
        
    # Sort adjacency lists to ensure we find the lexicographically smallest triad
    for i in range(1, n + 1):
        adj[i].sort()
        
    # We need to find a, b, c such that a < b < c
    # and (a,b), (a,c), (b,c) are all edges.
    # Iterate through a, then b, then check if there exists c > b
    # such that c is in adj[a] and c is in adj[b].
    
    for a in range(1, n + 1):
        for b in adj[a]:
            # Now we look for c > b such that c is in adj[a] and adj[b]
            # Since adj[a] and adj[b] are sorted, we can intersect them
            # efficiently or just iterate.
            # To optimize, we only care about c > b.
            
            # Intersection of adj[a] and adj[b]
            # Since we need the smallest a, then b, then c,
            # we can just iterate through the smaller list.
            
            # Optimization: use the set for O(1) lookup
            # Iterate through the smaller adjacency list to find c
            if len(adj[a]) < len(adj[b]):
                for c in adj[a]:
                    if c > b and c in adj_set[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in adj[b]:
                    if c > b and c in adj_set[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
