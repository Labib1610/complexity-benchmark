import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for consistent ordering
    adj_lists = [sorted(s) for s in adj]
    
    # Try to find triad with smallest a, then b, then c
    for a in range(1, n - 1):
        # Get neighbors of a that are > a
        neighbors_a = [v for v in adj_lists[a] if v > a]
        if len(neighbors_a) < 2:
            continue
        
        for b in neighbors_a:
            # Get neighbors of b that are > b
            neighbors_b = [v for v in adj_lists[b] if v > b]
            if not neighbors_b:
                continue
            
            # Find common neighbor c > b
            # neighbors_a already filtered for > a, now filter for > b
            candidates = [v for v in neighbors_a if v > b]
            
            # Check intersection with neighbors_b
            set_b = set(neighbors_b)
            for c in candidates:
                if c in set_b:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
