import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to ensure we check triplets in lexicographical order
    edges.sort()
    
    # A critical triplet is a clique of size 3.
    # We iterate through all edges (u, v) and check for a common neighbor w.
    # To satisfy the lexicographical requirement, we iterate u from 1 to n,
    # then v from u+1 to n, then check common neighbors w > v.
    
    for u in range(1, n + 1):
        adj_u = adj[u]
        # Sort neighbors to ensure w is checked in increasing order
        sorted_adj_u = sorted(list(adj_u))
        for i in range(len(sorted_adj_u)):
            v = sorted_adj_u[i]
            if v <= u:
                continue
            
            # Find common neighbors w such that u < v < w
            adj_v = adj[v]
            # Intersection of adj_u and adj_v
            # We only care about w > v
            for w in sorted_adj_u[i+1:]:
                if w in adj_v:
                    print(f"{u} {v} {w}")
                    return

if __name__ == '__main__':
    solve()
