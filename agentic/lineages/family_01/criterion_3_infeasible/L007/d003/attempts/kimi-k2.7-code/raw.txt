 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    masks = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        masks[u] |= 1 << (v - 1)
        masks[v] |= 1 << (u - 1)
        adj[u].append(v)
        adj[v].append(u)

    for i in range(1, n + 1):
        adj[i].sort()

    for a in range(1, n + 1):
        mask_a = masks[a]
        for b in adj[a]:
            if b <= a:
                continue
            common = mask_a & masks[b]
            shifted = common >> b
            if shifted:
                lsb = shifted & -shifted
                c = b + lsb.bit_length()
                sys.stdout.write(f"{a} {b} {c}")
                return

solve()
```
