import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # Sort edges to help find the lexicographically smallest triplet
    # We want to minimize a, then b, then c.
    # Iterate through all possible a, then b, then check if c exists.
    
    # To optimize, iterate through all edges (a, b) with a < b
    # and check if there exists a c > b such that (a, c) and (b, c) exist.
    
    # Sort edges by (a, b)
    edges.sort()
    
    best_triplet = None
    
    for a in range(1, n + 1):
        # Optimization: if we already found a triplet with a smaller 'a', we can stop
        if best_triplet and a > best_triplet[0]:
            break
            
        neighbors_a = adj[a]
        # Sort neighbors to check b in increasing order
        sorted_neighbors_a = sorted(list(neighbors_a))
        
        for b in sorted_neighbors_a:
            if b <= a:
                continue
            
            if best_triplet and a == best_triplet[0] and b > best_triplet[1]:
                continue
                
            # Find common neighbors c > b
            neighbors_b = adj[b]
            
            # Intersection of neighbors_a and neighbors_b
            # We only care about c > b
            common = neighbors_a.intersection(neighbors_b)
            
            for c in common:
                if c > b:
                    if best_triplet is None or (a, b, c) < best_triplet:
                        best_triplet = (a, b, c)
    
    if best_triplet:
        print(f"{best_triplet[0]} {best_triplet[1]} {best_triplet[2]}")

solve()
