import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # We iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we look at all neighbors 'b' of 'a' where b > a.
    # Then we check if there exists a 'c' such that c > b and c is in both adj[a] and adj[b].
    
    # Sort neighbors to ensure we check smaller identifiers first
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # To optimize, we iterate through all edges (u, v) with u < v
    # and check for a common neighbor w > v.
    # To satisfy the lexicographical requirement, we collect all valid triads
    # and pick the best one.
    
    best_triad = None
    
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # For each neighbor b of a, where b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c > b such that c is in adj[a] and adj[b]
            # Since adj lists are sorted, we can use set intersection or two pointers
            # Given the constraints, set intersection is efficient enough
            set_a = set(neighbors_a[i+1:])
            set_b = set(adj[b])
            common = set_a.intersection(set_b)
            
            if common:
                c = min(common)
                triad = (a, b, c)
                if best_triad is None or triad < best_triad:
                    best_triad = triad
        
        # Optimization: if we found a triad starting with 'a', 
        # we don't necessarily need to check larger 'a's if we want the smallest 'a'.
        if best_triad is not None and best_triad[0] < a:
            break
            
    if best_triad:
        print(f"{best_triad[0]} {best_triad[1]} {best_triad[2]}")

solve()
