import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find the lexicographically smallest (a, b, c) with a < b < c
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # Iterate b from neighbors of a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c such that c > b and c is in adj[a] and adj[b]
            # Intersection of adj[a] and adj[b]
            # To optimize, iterate through the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                common = adj[a].intersection(adj[b])
            else:
                common = adj[b].intersection(adj[a])
            
            valid_c = []
            for c in common:
                if c > b:
                    valid_c.append(c)
            
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
