import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        idx += 2
        
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Sort adjacency lists to potentially speed up intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Neighbors of a that are > a
        neighbors_a = [x for x in adj[a] if x > a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # To minimize b, we iterate b in increasing order
            # To minimize c, we check the smallest c in the intersection
            
            # Optimization: iterate through the smaller adjacency list
            if len(neighbors_a) - i - 1 < len(adj[b]):
                for j in range(i + 1, len(neighbors_a)):
                    c = neighbors_a[j]
                    if c in adj_sets[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in adj[b]:
                    if c > b and c in adj_sets[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
