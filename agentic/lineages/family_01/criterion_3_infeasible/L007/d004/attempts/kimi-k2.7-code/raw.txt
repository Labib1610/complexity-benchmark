import sys

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)

    adj = [0] * (n + 1)
    for _ in range(m):
        u = next(it)
        v = next(it)
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # high[i] has bits > i set to 1, lower bits cleared
    high = [~((1 << (i + 1)) - 1) for i in range(n + 1)]

    out = sys.stdout.write
    for a in range(1, n + 1):
        Sa = adj[a] & high[a]          # neighbors of a that are greater than a
        while Sa:
            lowb = Sa & -Sa
            b = lowb.bit_length() - 1
            cand = (adj[b] & Sa) & high[b]   # common neighbors c with a < b < c
            if cand:
                c = (cand & -cand).bit_length() - 1
                out(f"{a} {b} {c}\n")
                return
            Sa ^= lowb

if __name__ == "__main__":
    solve()
