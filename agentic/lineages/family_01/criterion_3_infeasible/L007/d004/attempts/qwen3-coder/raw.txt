import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for consistent ordering
    for key in adj:
        adj[key] = sorted(adj[key])
    
    # We'll iterate through all edges and try to find a triangle
    # To ensure we get the lexicographically smallest one,
    # we'll process potential triangles in order.
    
    # First, sort edges to help with ordering
    edges.sort()
    
    best = None
    
    # For each edge (u,v), look for a common neighbor w such that u < v < w
    for u, v in edges:
        if u >= v:
            continue
        # Find common neighbors of u and v that are greater than v
        neighbors_u = adj[u]
        neighbors_v = adj[v]
        
        i = j = 0
        while i < len(neighbors_u) and j < len(neighbors_v):
            w_u = neighbors_u[i]
            w_v = neighbors_v[j]
            if w_u == w_v:
                w = w_u
                if w > v:
                    candidate = (u, v, w)
                    if best is None or candidate < best:
                        best = candidate
                i += 1
                j += 1
            elif w_u < w_v:
                i += 1
            else:
                j += 1
        
        # Early exit if we found something and we're past possible better ones
        if best is not None and (u > best[0] or (u == best[0] and v > best[1])):
            break
    
    if best is not None:
        print(best[0], best[1], best[2])
    else:
        # Fallback brute-force for correctness (should not be needed given constraints)
        nodes = sorted(adj.keys())
        found = False
        for i in range(len(nodes)):
            u = nodes[i]
            for j in range(i+1, len(nodes)):
                v = nodes[j]
                if v not in adj[u]:
                    continue
                for k in range(j+1, len(nodes)):
                    w = nodes[k]
                    if w in adj[u] and w in adj[v]:
                        print(u, v, w)
                        found = True
                        break
                if found:
                    break
            if found:
                break

if __name__ == "__main__":
    main()
