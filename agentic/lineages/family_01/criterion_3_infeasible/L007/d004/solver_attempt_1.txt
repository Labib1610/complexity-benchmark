import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        idx += 2
        
    # We need to find a, b, c such that a < b < c
    # and (a, b) is an edge, (a, c) is an edge, (b, c) is an edge.
    # This is equivalent to finding a triangle in the graph.
    # To satisfy the lexicographical requirement:
    # Iterate a from 1 to n-2
    #   Iterate b from a+1 to n-1
    #     If (a, b) is an edge:
    #       Find common neighbors c > b
    
    for a in range(1, n + 1):
        # Sort neighbors to help with lexicographical order
        neighbors_a = sorted(list(adj[a]))
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Find common neighbors c of a and b such that c > b
            # Intersection of adj[a] and adj[b]
            # Since we need the smallest c, we can iterate through sorted neighbors
            neighbors_b = adj[b]
            
            # Optimization: iterate through the smaller set
            if len(neighbors_a) < len(neighbors_b):
                for c in neighbors_a:
                    if c > b and c in neighbors_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in neighbors_b:
                    if c > b and c in neighbors_a:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
