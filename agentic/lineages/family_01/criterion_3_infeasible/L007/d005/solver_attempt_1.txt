import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that a < b < c and
    # b in adj[a], c in adj[a], c in adj[b]
    # To satisfy the smallest a, then b, then c requirement,
    # we iterate a from 1 to n, b from a+1 to n, and c from b+1 to n.
    
    # Optimization: iterate a, then iterate b in adj[a] where b > a,
    # then check if there exists c in adj[a] such that c in adj[b] and c > b.
    
    for a in range(1, n + 1):
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Find common neighbors of a and b that are > b
            # Intersection of adj[a] and adj[b]
            common = sorted([x for x in adj[a] if x in adj[b] and x > b])
            if common:
                print(f"{a} {b} {common[0]}")
                return

if __name__ == '__main__':
    solve()
