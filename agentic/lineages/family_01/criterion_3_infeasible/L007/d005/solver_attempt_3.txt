```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find the lexicographically smallest (a, b, c) with a < b < c
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Get neighbors of a that are > a
        neighbors_a = [v for v in adj[a] if v > a]
        neighbors_a.sort()
        
        # Iterate b from neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # To find the smallest c, we can intersect the sorted neighbors
            # or just iterate through neighbors of b that are > b
            # Since we want the smallest c, we check neighbors of b
            for c in sorted(adj[b]):
                if c > b and c in adj[a]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
