```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [set() for _ in range(n + 1)]
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # For each node, sort its neighbors for consistent ordering
    for i in range(1, n + 1):
        adj[i] = sorted(adj[i])
    
    # Map each edge to a unique id for quick lookup
    edge_to_id = {}
    for idx, (u, v) in enumerate(edges):
        if u > v:
            u, v = v, u
        edge_to_id[(u, v)] = True
    
    # We want the lexicographically smallest triangle
    # So we iterate through all edges and for each edge (a,b), 
    # we look for a node c such that (a,c) and (b,c) are also edges
    
    found = False
    result = None
    
    # Sort edges to process in order that helps get lexicographically smallest first
    edges_sorted = []
    for u, v in edges:
        if u > v:
            u, v = v, u
        edges_sorted.append((u, v))
    
    # Sort edges by first vertex, then second
    edges_sorted.sort()
    
    for a, b in edges_sorted:
        if found:
            break
        # Get common neighbors of a and b
        # Iterate through neighbors of a, check if they are also neighbors of b
        # and ensure we form a triangle a < b < c
        if a >= b:
            continue
        for c in adj[a]:
            if c <= b:
                continue
            if c in adj[b]:
                # Check if the third edge (b, c) exists
                # Since adj[b] contains c, and we built adj from the edge list,
                # it's guaranteed to exist. Similarly for (a, c).
                print(a, b, c)
                return

if __name__ == "__main__":
    main()
```
