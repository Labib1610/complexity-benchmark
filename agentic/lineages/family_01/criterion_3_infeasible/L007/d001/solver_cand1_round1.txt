import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to ensure we check in lexicographical order
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # all possible a, then b, then c.
    # A tri-resonance is a triangle (a, b, c).
    # We can iterate through all edges (u, v) and check if there exists a w
    # such that (u, w) and (v, w) are edges.
    
    best = None
    
    # To optimize, iterate through all possible 'a' from 1 to n
    for a in range(1, n + 1):
        # Get neighbors of a
        neighbors_a = sorted(list(adj[a]))
        if len(neighbors_a) < 2:
            continue
            
        # For each pair of neighbors b, c of a, check if (b, c) is an edge
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                
                # Check if (b, c) is an edge
                if c in adj[b]:
                    # Found a triangle (a, b, c)
                    # Since we iterate a, then b, then c in sorted order,
                    # the first one we find is the lexicographically smallest.
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
