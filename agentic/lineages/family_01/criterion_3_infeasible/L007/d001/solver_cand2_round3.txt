```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest (a, b, c)
    # We want smallest a, then smallest b, then smallest c.
    # Iterate through all edges (u, v) as potential (a, b)
    # Then check for a common neighbor w > v
    
    edges.sort()
    
    # To optimize, we can iterate through all edges (u, v)
    # and check common neighbors.
    # Since n <= 2000, O(m * n) is acceptable (2000 * 2000 = 4e6)
    
    best = None
    
    for u in range(1, n + 1):
        # Optimization: if we already found a solution with a smaller 'a',
        # we don't need to check larger 'a'.
        if best and u > best[0]:
            break
            
        neighbors_u = sorted(list(adj[u]))
        for i in range(len(neighbors_u)):
            v = neighbors_u[i]
            if v <= u:
                continue
            
            # Now we have edge (u, v). Look for w > v such that w is in adj[u] and adj[v]
            # Intersection of adj[u] and adj[v]
            common = sorted(list(adj[u].intersection(adj[v])))
            for w in common:
                if w > v:
                    candidate = (u, v, w)
                    if best is None or candidate < best:
                        best = candidate
                    # Since we want the smallest a, then b, then c,
                    # and we iterate u, v, w in order, the first one we find
                    # might not be the absolute smallest if we don't check all.
                    # But with the current loops, we can just keep track of the min.
    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
