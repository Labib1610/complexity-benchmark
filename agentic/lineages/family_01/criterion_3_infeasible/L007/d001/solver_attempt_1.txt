import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through all possible a, then b, then c.
    # Since we need to find the smallest a, then b, then c,
    # we can iterate a from 1 to n, b from a+1 to n, and check if
    # there exists c > b such that (a,b), (b,c), (c,a) are edges.
    
    # Optimization: Sort adjacency lists to speed up intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # To find the smallest (a, b, c), we iterate a, then b, then c.
    # A trio (a, b, c) exists if b is in adj[a] and there exists c in adj[a] 
    # such that c is also in adj[b].
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Since neighbors_a is sorted, we can iterate through neighbors_a
            # starting from i+1.
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    # Found the smallest a, then b, then c
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
