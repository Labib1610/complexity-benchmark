import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triad
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # sorted edges (a, b) and check if there exists a c > b such that
    # (a, c) and (b, c) are edges.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # We want the smallest a, then smallest b, then smallest c.
    # Iterating through edges (a, b) in sorted order covers this.
    for a, b in edges:
        # Find common neighbors of a and b that are > b
        # Since adj lists are sorted, we can use two pointers or set intersection
        set_a = adj[a]
        set_b = adj[b]
        
        # Optimization: iterate over the smaller adjacency list
        if len(set_a) > len(set_b):
            small, large = set_b, set_a
        else:
            small, large = set_a, set_b
            
        # We need c > b
        # Since small and large are sorted, we can find intersection efficiently
        i = 0
        j = 0
        while i < len(small) and j < len(large):
            if small[i] == large[j]:
                if small[i] > b:
                    print(f"{a} {b} {small[i]}")
                    return
                i += 1
                j += 1
            elif small[i] < large[j]:
                i += 1
            else:
                j += 1

if __name__ == '__main__':
    solve()
