import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To minimize a * 10^8 + b * 10^4 + c, we want the smallest a, then smallest b, then smallest c.
    # We iterate through all edges (u, v) with u < v.
    # For each edge, we look for a common neighbor w.
    # To ensure we find the lexicographically smallest (a, b, c), we sort the edges.
    
    edges.sort()
    
    best_abc = (float('inf'), float('inf'), float('inf'))
    
    # Optimization: iterate through edges and check common neighbors
    # To keep it efficient, iterate over the smaller degree vertex's neighbors
    for u, v in edges:
        # Ensure u < v
        # Check neighbors of u and v
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        
        # Now u is the one with potentially fewer neighbors
        for w in adj[u]:
            if w in adj[v]:
                # Found a triangle (u, v, w)
                # Sort them to get a < b < c
                triangle = sorted([u, v, w])
                if triangle < list(best_abc):
                    best_abc = tuple(triangle)
                    
    print(f"{best_abc[0]} {best_abc[1]} {best_abc[2]}")

if __name__ == '__main__':
    solve()
