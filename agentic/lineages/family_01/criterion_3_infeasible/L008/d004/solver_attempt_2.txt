```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To minimize a * 10^8 + b * 10^4 + c, we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # c is in adj[a] and adj[b].
    
    best_triple = None
    min_score = float('inf')
    
    # Sort neighbors to potentially speed up intersection
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Optimization: if a * 10^8 is already >= min_score, we can stop
        if a * 100000000 > min_score:
            break
            
        neighbors_a = adj_list[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            score_ab = a * 100000000 + b * 10000
            if score_ab > min_score:
                continue
                
            # Find common neighbors c > b
            # Using set intersection for efficiency
            set_a = adj[a]
            set_b = adj[b]
            
            # Iterate through the smaller set to find common elements
            if len(set_a) < len(set_b):
                candidates = set_a
            else:
                candidates = set_b
                
            for c in candidates:
                if c > b:
                    if c in set_a and c in set_b:
                        score = score_ab + c
                        if score < min_score:
                            min_score = score
                            best_triple = (a, b, c)
                            
    if best_triple:
        print(f"{best_triple[0]} {best_triple[1]} {best_triple[2]}")

if __name__ == '__main__':
    solve()
```
