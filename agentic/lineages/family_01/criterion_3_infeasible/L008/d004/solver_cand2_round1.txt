```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need a subset of 3 workers {a, b, c} such that each appears in exactly 2 files.
    # This means the edges (a,b), (b,c), (c,a) must all exist in the input.
    # If these 3 edges exist, each node is connected to the other two, 
    # so each node appears in exactly 2 files within the subset.
    
    # To minimize a, then b, then c:
    for a in range(1, n + 1):
        # Sort neighbors to help with lexicographical order
        neighbors = sorted(list(adj[a]))
        len_n = len(neighbors)
        for i in range(len_n):
            b = neighbors[i]
            if b <= a:
                continue
            for j in range(i + 1, len_n):
                c = neighbors[j]
                # Check if b and c are connected
                if c in adj[b]:
                    # Found the triangle {a, b, c}
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

if __name__ == '__main__':
    solve()
```
