```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To minimize a * 10^8 + b * 10^4 + c, we iterate a from 1 to n,
    # then b from a+1 to n, then check if c exists in adj[a] and adj[b].
    # Since we need the lexicographically smallest (a, b, c),
    # we iterate in increasing order.
    
    best_a, best_b, best_c = float('inf'), float('inf'), float('inf')
    
    # Optimization: sort adjacency lists to iterate c efficiently
    sorted_adj = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        if a >= best_a:
            break
        for b in sorted_adj[a]:
            if b <= a:
                continue
            if b >= best_b and a == best_a:
                break
            
            # Find common neighbors c > b
            # Using set intersection for speed
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    if a < best_a or (a == best_a and b < best_b) or (a == best_a and b == best_b and c < best_c):
                        best_a, best_b, best_c = a, b, c
    
    print(f"{best_a} {best_b} {best_c}")

if __name__ == '__main__':
    solve()
```
