import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # We are looking for three reagents (a, b, c) such that no two are safe to mix.
    # A pair is safe if it is NOT in the precipitate list.
    # So we need a, b, c such that (a, b), (b, c), and (a, c) are ALL in the precipitate list.
    # This is equivalent to finding a triangle in the graph where edges are precipitate reactions.
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To minimize a * 10^8 + b * 10^4 + c, we iterate a from 1 to n, b from a+1 to n, c from b+1 to n.
    # Since we need to find the lexicographically smallest (a, b, c), we can stop early.
    
    best = None
    
    # Sort adjacency lists to potentially speed up intersection
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        sorted_neighbors_a = adj_list[a]
        for b in sorted_neighbors_a:
            if b <= a:
                continue
            
            # We need c such that c is in adj[a] and c is in adj[b]
            # To minimize a*10^8 + b*10^4 + c, we want the smallest c > b
            neighbors_b = adj[b]
            
            # Intersection of neighbors_a and neighbors_b
            # Since we want the smallest c > b, we iterate through sorted neighbors
            for c in sorted_neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
