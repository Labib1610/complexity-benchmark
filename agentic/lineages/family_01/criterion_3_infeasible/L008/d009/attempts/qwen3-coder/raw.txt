```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # For each node, store its degree
    degree = {i: len(adj[i]) for i in range(1, n+1)}
    
    # Create a mapping from node to its neighbors
    neighbors = {i: adj[i] for i in range(1, n+1)}
    
    # Build a set of edges for O(1) lookup
    edge_set = set()
    for u, v in edges:
        if u > v:
            u, v = v, u
        edge_set.add((u, v))
    
    # Try all combinations of 3 nodes, in sorted order
    found = False
    result = None
    
    for a in range(1, n+1):
        if found:
            break
        for b in range(a+1, n+1):
            if found:
                break
            for c in range(b+1, n+1):
                # Check if this triplet satisfies T = E + 6
                
                # T is degree[a] + degree[b] + degree[c]
                T = degree[a] + degree[b] + degree[c]
                
                # Count E: number of edges with exactly one endpoint in {a,b,c}
                count_E = 0
                
                # Edges from a to outside {a,b,c}
                for neigh in neighbors[a]:
                    if neigh != b and neigh != c:
                        count_E += 1
                        
                # Edges from b to outside {a,b,c}
                for neigh in neighbors[b]:
                    if neigh != a and neigh != c:
                        count_E += 1
                        
                # Edges from c to outside {a,b,c}
                for neigh in neighbors[c]:
                    if neigh != a and neigh != b:
                        count_E += 1
                
                if T == count_E + 6:
                    result = (a, b, c)
                    found = True
                    break
    
    if result:
        print(result[0], result[1], result[2])

if __name__ == "__main__":
    main()
```
