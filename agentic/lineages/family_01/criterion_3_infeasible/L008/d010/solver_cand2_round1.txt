import sys

def solve():
    # The problem asks for a group of 3 workers {a, b, c} such that there exist 
    # 3 distinct incident files (edges) involving only these 3 workers.
    # A group of 3 workers {a, b, c} has exactly 3 possible edges: (a, b), (b, c), (a, c).
    # If all 3 edges exist in the input, then the condition is satisfied:
    # - File 1: (a, b) involves a and b.
    # - File 2: (b, c) involves b and c.
    # - File 3: (a, c) involves a and c.
    # Each worker is assigned to two files, but the condition says:
    # "assign each of the three workers in your core group to a different one of these chosen files"
    # This means we need a matching of size 3 in a graph with 3 vertices, which is impossible.
    # Wait, re-reading: "assign each of the three workers... to a different one of these chosen files"
    # This implies a bijection between the 3 workers and the 3 files.
    # Let the files be F1, F2, F3. Worker A -> F1, B -> F2, C -> F3.
    # This is only possible if each worker is in exactly one file.
    # But each file has 2 workers. Total workers in 3 files = 6 (with multiplicity).
    # If each worker is in exactly one file, total workers = 3.
    # This is only possible if the 3 files are disjoint, which is impossible for 3 workers.
    # Re-reading again: "A worker is always assigned to an incident file that lists them."
    # This means we need to pick 3 files such that we can map {a, b, c} to {f1, f2, f3} 
    # where worker i is in file f_i.
    # This is equivalent to finding a system of distinct representatives.
    # For a set of 3 vertices {a, b, c}, the edges are subsets of {a, b, c}.
    # The only way to have 3 files using only these 3 vertices is to have all 3 edges (a,b), (b,c), (a,c).
    # If we have edges (a,b), (b,c), (a,c), we can assign:
    # a -> (a, c), b -> (a, b), c -> (b, c). This works!
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # Iterate through all triplets (a, b, c) with a < b < c
    # To optimize, iterate over a, then b in adj[a], then c in adj[a] intersect adj[b]
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a: continue
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if (b, c) is an edge
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
