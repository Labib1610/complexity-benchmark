import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    degree = [0] * (n + 1)
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
        edges.append((min(u, v), max(u, v)))
    
    # G = deg(a) + deg(b) + deg(c)
    # K = m - (number of edges involving at least one of {a, b, c})
    # Condition: G + K = m + 3
    # G + m - (edges involving a, b, or c) = m + 3
    # G - (edges involving a, b, or c) = 3
    # Let E(a,b,c) be the set of edges involving a, b, or c.
    # |E(a,b,c)| = deg(a) + deg(b) + deg(c) - (edges between a,b, c)
    # G - (|E(a,b,c)|) = G - (G - edges_between_abc) = edges_between_abc
    # So the condition is: edges_between_abc = 3
    # This means a, b, and c must form a triangle in the graph.
    
    # We need to find a triangle (a, b, c) with a < b < c, minimizing a, then b, then c.
    
    # To find triangles efficiently:
    # Sort vertices by degree to optimize triangle finding
    adj_set = [set(adj[i]) for i in range(n + 1)]
    
    # Iterate through all possible a, b, c
    # Since we need the lexicographically smallest, we iterate a from 1 to n
    for a in range(1, n + 1):
        # b must be a neighbor of a
        neighbors_a = sorted(adj[a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # c must be a neighbor of both a and b
            # To minimize b and c, we check neighbors of a that are > a
            # and check if they are connected to b
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj_set[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
