import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    degree = [0] * (n + 1)
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
        edges.append((min(u, v), max(u, v)))
    
    # S = deg(a) + deg(b) + deg(c)
    # U = m - (number of edges involving at least one of a, b, c)
    # Condition: S + U = m + 3
    # S + m - (edges involving a, b, or c) = m + 3
    # S - (edges involving a, b, or c) = 3
    # Let E(a,b,c) be the set of edges involving a, b, or c.
    # |E(a,b,c)| = deg(a) + deg(b) + deg(c) - (edges between a,b; b,c; a,c)
    # So, (deg(a) + deg(b) + deg(c)) - (deg(a) + deg(b) + deg(c) - count_edges_between_abc) = 3
    # count_edges_between_abc = 3
    # This means the three workers must form a triangle (a,b), (b,c), (a,c)
    
    is_edge = [[False] * (n + 1) for _ in range(n + 1)]
    for u, v in edges:
        is_edge[u][v] = is_edge[v][u] = True
        
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if is_edge[a][b]:
                for c in range(b + 1, n + 1):
                    if is_edge[a][c] and is_edge[b][c]:
                        # Found a triangle (a, b, c)
                        # Check condition: S + U = m + 3
                        # S = deg(a) + deg(b) + deg(c)
                        # U = m - (deg(a) + deg(b) + deg(c) - 3)
                        # S + U = deg(a) + deg(b) + deg(c) + m - deg(a) - deg(b) - deg(c) + 3 = m + 3
                        # The condition is always satisfied for any triangle.
                        # Since we iterate a, b, c in increasing order, the first one found is the answer.
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
