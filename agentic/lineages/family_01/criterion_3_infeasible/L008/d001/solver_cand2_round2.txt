import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we iterate b in adj[a] (sorted) such that b > a.
    # Then we check if there exists c in adj[a] intersect adj[b] such that c > b.
    
    # Sort adjacency lists to ensure we find the lexicographically smallest triplet
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Pre-sort edges to help find smallest a, b
    edges.sort()
    
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # Iterate through all 'b' connected to 'a'
        for b in adj[a]:
            if b <= a:
                continue
            
            # We need a 'c' such that c is in adj[a] and adj[b] and c > b
            # Since adj[a] and adj[b] are sorted, we can use two pointers or set intersection
            # Given n=2000, set intersection or simple iteration is fine.
            # To be efficient, use the smaller adjacency list to check existence.
            
            set_b = set(adj[b])
            for c in adj[a]:
                if c > b and c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
