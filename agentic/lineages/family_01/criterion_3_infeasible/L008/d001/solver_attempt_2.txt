```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest (a, b, c)
    edges.sort()
    
    # We need to find a, b, c such that (a,b), (b,c), (a,c) are all edges.
    # Iterate through all edges (u, v) as the first edge (a, b)
    # Then check for a common neighbor w such that (a, w) and (b, w) exist.
    
    best = None
    
    for u, v in edges:
        # u < v is guaranteed by the input processing
        # We want to find w such that w is connected to both u and v
        # To minimize (a, b, c), we should check potential w values
        
        # Intersection of neighbors of u and v
        common = adj[u].intersection(adj[v])
        
        for w in common:
            # Form the triplet (a, b, c)
            triplet = sorted([u, v, w])
            if best is None or triplet < best:
                best = triplet
                
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
