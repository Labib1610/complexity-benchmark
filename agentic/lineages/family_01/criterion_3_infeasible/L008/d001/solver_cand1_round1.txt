import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest trio (a, b, c) with a < b < c
    # We iterate through possible 'a' from 1 to n
    for a in range(1, n + 1):
        # Get all neighbors of 'a' that are greater than 'a'
        neighbors_a = sorted(list(adj[a]))
        
        # We need to find b, c in neighbors_a such that (b, c) is an edge
        # To ensure lexicographical order, we iterate b then c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Check if there exists c in neighbors_a such that c > b and (b, c) is an edge
            # We can intersect neighbors_a with neighbors_b
            # Optimization: only check neighbors of b that are also in neighbors_a
            neighbors_b = adj[b]
            
            # Find common neighbors c > b
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
