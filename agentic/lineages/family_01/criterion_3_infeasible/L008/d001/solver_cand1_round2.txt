import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest trio (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is connected to a
    # If so, check if there exists c > b such that c is connected to both a and b
    
    # Sort neighbors to potentially speed up intersection
    adj_list = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        sorted_neighbors_a = sorted([x for x in neighbors_a if x > a])
        
        for b in sorted_neighbors_a:
            neighbors_b = adj[b]
            # We need c > b such that c in neighbors_a and c in neighbors_b
            # To keep it lexicographically smallest, we want the smallest such c
            # Intersection of neighbors_a and neighbors_b
            common = sorted([x for x in neighbors_a if x > b and x in neighbors_b])
            if common:
                print(f"{a} {b} {common[0]}")
                return

if __name__ == '__main__':
    solve()
