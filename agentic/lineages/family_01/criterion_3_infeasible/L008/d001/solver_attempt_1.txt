import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To minimize a * 10^8 + b * 10^4 + c, we want the smallest a, then smallest b, then smallest c.
    # We iterate through all edges (u, v) with u < v.
    # For each edge, we look for a common neighbor w such that (u, w) and (v, w) exist.
    # To ensure we find the lexicographically smallest (a, b, c), we sort the edges.
    
    edges.sort()
    
    best_triplet = None
    
    for u, v in edges:
        # Potential triplet is (u, v, w) where w > v or w < u or u < w < v
        # We need to check all w in adj[u] that are also in adj[v]
        common = adj[u].intersection(adj[v])
        for w in common:
            # Form the triplet and sort it
            triplet = sorted([u, v, w])
            a, b, c = triplet
            
            if best_triplet is None:
                best_triplet = (a, b, c)
            else:
                # Compare lexicographically
                if (a < best_triplet[0]) or \
                   (a == best_triplet[0] and b < best_triplet[1]) or \
                   (a == best_triplet[0] and b == best_triplet[1] and c < best_triplet[2]):
                    best_triplet = (a, b, c)
                    
    print(f"{best_triplet[0]} {best_triplet[1]} {best_triplet[2]}")

if __name__ == '__main__':
    solve()
