import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a trio (a, b, c) with a < b < c such that
    # b is in adj[a], c is in adj[a], and c is in adj[b].
    # To minimize a, then b, then c:
    
    best_trio = None
    
    # Iterate through possible a
    for a in range(1, n + 1):
        neighbors = sorted(list(adj[a]))
        if len(neighbors) < 2:
            continue
            
        # Iterate through pairs of neighbors (b, c)
        for i in range(len(neighbors)):
            b = neighbors[i]
            if best_trio and a > best_trio[0]:
                break
            
            # Optimization: if a is fixed, we want smallest b, then smallest c
            # Since neighbors are sorted, we just need to check if b and c are connected
            for j in range(i + 1, len(neighbors)):
                c = neighbors[j]
                
                # Check if b and c are connected
                if c in adj[b]:
                    # Found a trio (a, b, c)
                    # Since we iterate a from 1 to n, and b, c from sorted neighbors,
                    # the first one we find is the lexicographically smallest.
                    # Wait, the order is a < b < c.
                    # Let's ensure a < b < c.
                    trio = sorted([a, b, c])
                    if best_trio is None or trio < best_trio:
                        best_trio = trio
                    
                    # Since we want to minimize a, then b, then c:
                    # If we found a trio for this 'a', we don't need to check larger 'a's.
                    # But we need to be careful about the order.
                    # Actually, if we find any trio for a given 'a', 
                    # we just need to find the smallest (b, c) for that 'a'.
                    # The current loop structure finds the smallest b, then c for a fixed a.
                    print(f"{trio[0]} {trio[1]} {trio[2]}")
                    return

solve()
