import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # Sort edges to help find the lexicographically smallest triplet
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # sorted edges (a, b) and check if there exists a c such that
    # (a, c) and (b, c) are in the adjacency set.
    # Since we want the smallest a, then b, then c, we iterate through
    # all pairs (a, b) and check for common neighbors.
    
    # Optimization: iterate through all pairs (a, b) and check for common neighbors
    # To ensure lexicographical order, we can iterate a from 1 to n,
    # then b from a+1 to n, and check if b is in adj[a].
    # If so, find the smallest c > b such that c is in adj[a] and adj[b].
    
    for a in range(1, n + 1):
        # Sort neighbors to find smallest c efficiently
        neighbors_a = sorted(list(adj[a]))
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Find common neighbors c > b
            # We can intersect the sorted lists of neighbors
            neighbors_b = adj[b]
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
