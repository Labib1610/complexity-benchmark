import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through all edges (u, v) with u < v.
    # For each edge, we look for a common neighbor w such that w > v.
    # To ensure we find the smallest (a, b, c), we sort the edges
    # and for each edge, check neighbors.
    
    edges.sort()
    
    best = (float('inf'), float('inf'), float('inf'))
    
    # Optimization: iterate through edges (u, v) and check common neighbors
    # To minimize a*10^8 + b*10^4 + c, we want smallest a, then smallest b, then smallest c.
    for u in range(1, n + 1):
        # If we already found a triplet starting with a smaller 'a', we can stop
        if u >= best[0]:
            break
            
        neighbors_u = sorted(list(adj[u]))
        for i in range(len(neighbors_u)):
            v = neighbors_u[i]
            if v <= u:
                continue
            
            # Now we have edge (u, v) with u < v.
            # We need a w such that w > v and w is connected to both u and v.
            # To minimize the triplet, we check neighbors of u that are also in adj[v]
            # and are greater than v.
            
            # Intersection of adj[u] and adj[v]
            common = adj[u].intersection(adj[v])
            for w in common:
                if w > v:
                    # Found a triplet (u, v, w)
                    triplet = (u, v, w)
                    if triplet < best:
                        best = triplet
    
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
