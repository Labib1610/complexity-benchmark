```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To minimize a * 10^8 + b * 10^4 + c, we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # c is in adj[a] and c is in adj[b].
    
    # Sort adjacency lists to make intersection faster and ordered
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    best_trio = None
    min_score = float('inf')
    
    for a in range(1, n + 1):
        # Optimization: if a * 10^8 is already >= min_score, we can stop
        if a * 100000000 > min_score:
            break
            
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            score_ab = a * 100000000 + b * 10000
            if score_ab > min_score:
                continue
            
            # Find common neighbors c > b
            # Since neighbors_a and neighbors_b are sorted, use two pointers
            neighbors_b = sorted_adj[b]
            
            ptr_a = i + 1
            ptr_b = 0
            
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        c = val_a
                        score = score_ab + c
                        if score < min_score:
                            min_score = score
                            best_trio = (a, b, c)
                        # Since we want the smallest c for this (a, b), 
                        # and we are iterating b in increasing order, 
                        # we found the best c for this (a, b).
                        break
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1
                    
    if best_trio:
        print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

if __name__ == '__main__':
    solve()
```
