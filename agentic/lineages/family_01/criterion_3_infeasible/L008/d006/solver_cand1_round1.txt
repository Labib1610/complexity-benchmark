import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A deadlock is a set of 3 incident files involving exactly 3 workers.
    # This means there exist 3 workers {a, b, c} such that all three pairs
    # (a, b), (b, c), and (a, c) are in the incident list.
    # This is equivalent to finding a triangle in the graph.
    
    # To minimize a, then b, then c, we iterate through possible a, b, c.
    # Since n <= 2000, we can iterate through all edges (u, v) and check
    # for a common neighbor w.
    
    best = None
    
    # Sort edges to help with lexicographical order
    edges = []
    for u in range(1, n + 1):
        for v in adj[u]:
            if u < v:
                edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors = sorted(list(adj[a]))
        len_n = len(neighbors)
        for i in range(len_n):
            b = neighbors[i]
            if b <= a: continue
            
            # Check for common neighbor c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    # Found a triangle (a, b, c)
                    # Since we iterate a, then b, then c in increasing order,
                    # the first one we find is the lexicographically smallest.
                    print(f"{a} {b} {c}")
                    return

solve()
