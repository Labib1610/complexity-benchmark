import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To minimize a * 10^8 + b * 10^4 + c, we want the smallest a, then smallest b, then smallest c.
    # We iterate through all edges (u, v) with u < v.
    # For each edge, we look for a common neighbor w.
    # To ensure we find the lexicographically smallest (a, b, c), we sort the edges.
    
    edges.sort()
    
    best = (float('inf'), float('inf'), float('inf'))
    
    # Optimization: iterate through edges and check common neighbors
    # Since n <= 2000, O(m * n / word_size) or O(m * sqrt(m)) is acceptable.
    # Using set intersection is efficient.
    
    for u, v in edges:
        # We need a third node w such that (u, w) and (v, w) exist.
        # To keep a < b < c, we check w > v, or if u < w < v, or w < u.
        # Actually, for any triangle {a, b, c} with a < b < c, 
        # it will be found when processing edge (a, b), (a, c), or (b, c).
        # We just need to check all common neighbors w.
        
        # To minimize a*10^8 + b*10^4 + c, we check all w in intersection
        common = adj[u].intersection(adj[v])
        for w in common:
            # Form the sorted triplet
            triplet = sorted([u, v, w])
            if triplet < list(best):
                best = tuple(triplet)
                
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
