import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to process in order
    edges.sort()
    
    # For each edge (u,v), look for a neighbor of u and a neighbor of v that are connected
    best = None
    
    for i in range(len(edges)):
        u, v = edges[i]
        # We want to find a node x such that:
        #   x is adjacent to u but not v
        #   there exists a node w adjacent to v and x
        # This creates the triangle u-v-x or u-x-v
        
        # Get neighbors of u and v
        neigh_u = adj[u]
        neigh_v = adj[v]
        
        # Check each neighbor of u (other than v) to see if it connects to a neighbor of v
        for x in neigh_u:
            if x == v:
                continue
            # Now check if there is a node w such that edge (x,w) exists and w is in neigh_v
            # That would make the triplet u,x,v with edges (u,x), (u,v), (x,v) 
            # Wait, let's reframe:
            # We have edge (u,v). We need another edge sharing one node with (u,v).
            # Say we pick edge (u, x) where x != v. Then we need edge (v, x) to close the triangle.
            # So we just check if both (u,x) and (v,x) exist.
            
            if x in neigh_v:
                # Found triangle u, v, x
                a, b, c = sorted([u, v, x])
                if best is None or (a < best[0]) or (a == best[0] and b < best[1]) or (a == best[0] and b == best[1] and c < best[2]):
                    best = (a, b, c)
                    
    if best:
        print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
