```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # A deadlock is a triangle (a, b, c) such that edges (a,b), (b,c), (a,c) exist.
    # The problem states: "count how many of them have both of their listed workers 
    # belonging to this specific trio, the count must be exactly 3."
    # This is exactly the definition of a triangle in the graph.
    
    triangles = []
    
    # To find triangles efficiently:
    # Iterate over each edge (u, v) and find common neighbors.
    # Since we need the lexicographically smallest (a, b, c), we sort them.
    
    # Optimization: iterate over nodes with smaller degree to find triangles
    # But n=2000 is small enough for O(m * sqrt(m)) or similar.
    
    for u in range(1, n + 1):
        neighbors = sorted(list(adj[u]))
        for i in range(len(neighbors)):
            v = neighbors[i]
            if v <= u:
                continue
            # Find common neighbors of u and v
            # Intersection of adj[u] and adj[v]
            common = adj[u].intersection(adj[v])
            for w in common:
                if w > v:
                    triangles.append((u, v, w))
                elif w < u:
                    triangles.append((w, u, v))
                else: # u < w < v
                    triangles.append((u, w, v))
                    
    triangles.sort()
    
    # The problem guarantees at least one exists.
    # The first one in sorted order is the answer.
    a, b, c = triangles[0]
    print(f"{a} {b} {c}")

if __name__ == '__main__':
    solve()
```
