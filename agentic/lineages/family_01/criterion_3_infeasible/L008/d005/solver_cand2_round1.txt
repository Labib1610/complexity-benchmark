```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # We need to find a triangle (a, b, c) in the graph.
    # A deadlock is a set of 3 incident files involving exactly 3 workers.
    # This is equivalent to finding a cycle of length 3 (a triangle).
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # Iterate b from neighbors of a, b > a
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Check for common neighbors c such that c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            
            valid_c = [c for c in common if c > b]
            if valid_c:
                c = min(valid_c)
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
```
