import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To minimize a * 10^8 + b * 10^4 + c, we want the smallest a, then smallest b, then smallest c.
    # We iterate through all edges (u, v) with u < v.
    # For each edge, we look for a common neighbor w.
    # The trio will be (min(u, v, w), mid(u, v, w), max(u, v, w)).
    
    best_trio = (float('inf'), float('inf'), float('inf'))
    
    # Sort edges to process in a way that helps find the smallest trio
    edges.sort()
    
    for u, v in edges:
        # We need a w such that (u, w) and (v, w) exist.
        # To keep the trio sorted, w can be < u, between u and v, or > v.
        
        # Optimization: only check common neighbors
        # If u is small, we are more likely to find a small trio.
        
        # Intersection of adj[u] and adj[v]
        # To be efficient, iterate over the smaller adjacency list
        if len(adj[u]) < len(adj[v]):
            common = adj[u].intersection(adj[v])
        else:
            common = adj[v].intersection(adj[u])
            
        for w in common:
            trio = sorted([u, v, w])
            if trio[0] < best_trio[0] or \
               (trio[0] == best_trio[0] and trio[1] < best_trio[1]) or \
               (trio[0] == best_trio[0] and trio[1] == best_trio[1] and trio[2] < best_trio[2]):
                best_trio = tuple(trio)
                
    print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

if __name__ == '__main__':
    solve()
