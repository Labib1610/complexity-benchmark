import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To minimize a * 10^8 + b * 10^4 + c, we want the smallest a, then smallest b, then smallest c.
    # We iterate through all edges (u, v) with u < v.
    # For each edge, we look for a common neighbor w.
    # The triangle is (min(u, v, w), mid(u, v, w), max(u, v, w)).
    
    best = (float('inf'), float('inf'), float('inf'))
    
    # Sort edges to help find the lexicographically smallest triangle
    edges.sort()
    
    # Optimization: iterate through edges and check common neighbors
    # To keep it efficient, iterate over the smaller adjacency list
    for u, v in edges:
        # We want to find w such that (u, w) and (v, w) exist.
        # To minimize the score, we check w in increasing order.
        # Since we need to find the global minimum, we can iterate through all w.
        
        # To speed up, intersect the smaller adjacency list
        if len(adj[u]) < len(adj[v]):
            common = adj[u].intersection(adj[v])
        else:
            common = adj[v].intersection(adj[u])
            
        for w in common:
            # Triangle is u, v, w. Sort them.
            t = sorted([u, v, w])
            if (t[0], t[1], t[2]) < best:
                best = (t[0], t[1], t[2])
                
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
