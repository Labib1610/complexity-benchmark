import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # A deadlock is a trio {a, b, c} such that the edges between them
    # form a cycle of length 3 (a triangle).
    # In a triangle {a, b, c}, each node is connected to the other two,
    # so each node appears exactly twice in the set of edges {(a,b), (b,c), (a,c)}.
    # Thus, any triangle is a deadlock.
    
    # We need to find the lexicographically smallest (a, b, c) with a < b < c.
    # Iterate through all possible a, then b, then c.
    # Since n <= 2000, O(n^3) is too slow, but we can iterate over edges.
    # For each edge (u, v), check all w such that (u, w) and (v, w) exist.
    
    results = []
    
    # To find the smallest (a, b, c), we iterate a from 1 to n-2
    for a in range(1, n + 1):
        # Find all neighbors of a
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Find common neighbors of a and b
            # These are candidates for c
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    # Found a triangle (a, b, c)
                    # Since we iterate a, then b, then c in increasing order,
                    # the first one we find is the lexicographically smallest.
                    print(f"{a} {b} {c}")
                    return

solve()
