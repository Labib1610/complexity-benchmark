import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Convert edges to a set for O(1) lookup
    edge_set = set(edges)
    for i in range(len(edges)):
        u, v = edges[i]
        # We want triangles, so we look for a third node w such that 
        # (u,w) and (v,w) are also edges
        common_neighbors = adj[u] & adj[v]
        for w in common_neighbors:
            # Ensure we have a triangle u, v, w
            nodes = sorted([u, v, w])
            a, b, c = nodes[0], nodes[1], nodes[2]
            # Check if all three edges exist (to form a triangle)
            edges_in_triangle = [
                tuple(sorted((a, b))),
                tuple(sorted((a, c))),
                tuple(sorted((b, c)))
            ]
            if all(edge in edge_set or (edge[1], edge[0]) in edge_set for edge in [(a,b), (a,c), (b,c)]):
                print(a, b, c)
                return

if __name__ == "__main__":
    main()
