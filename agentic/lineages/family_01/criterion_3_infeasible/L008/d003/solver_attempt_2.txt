import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To minimize a * 10^8 + b * 10^4 + c, we want the smallest a, then smallest b, then smallest c.
    # A trio (a, b, c) with a < b < c is a hazard if b in adj[a], c in adj[a], and c in adj[b].
    
    best_trio = None
    
    # Sort edges to iterate in a way that helps find the lexicographically smallest trio
    # Actually, iterating through all edges (u, v) and checking common neighbors is efficient.
    # For each edge (u, v) with u < v, find common neighbors w.
    # The trio is (min(u, v, w), mid(u, v, w), max(u, v, w)).
    
    # Optimization: iterate through u from 1 to n
    for u in range(1, n + 1):
        neighbors = sorted(list(adj[u]))
        len_n = len(neighbors)
        for i in range(len_n):
            v = neighbors[i]
            if v <= u:
                continue
            
            # Check common neighbors w > v
            # To minimize score, we want smallest u, then smallest v, then smallest w.
            # Since we iterate u and v in increasing order, we just need to find the smallest w.
            
            # Intersection of adj[u] and adj[v]
            # Only need to check w > v
            for j in range(i + 1, len_n):
                w = neighbors[j]
                if w in adj[v]:
                    # Found a trio (u, v, w)
                    trio = sorted([u, v, w])
                    if best_trio is None or trio < best_trio:
                        best_trio = trio
                    # Since we want the absolute minimum, and we are iterating u and v,
                    # we can't break early easily unless we are sure.
                    # Given the constraints and the lexicographical requirement,
                    # checking all is feasible if we optimize the intersection.
    
    print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

solve()
