import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # Iterate a from 1 to n
    # Iterate b from a+1 to n
    # If b is connected to a, check if there exists c > b such that c is connected to both a and b
    
    # Sort neighbors to potentially speed up intersection
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Only consider b > a
        for b in sorted(list(neighbors_a)):
            if b <= a:
                continue
            
            # We need c > b such that c is in neighbors_a AND c is in neighbors_b
            # Intersection of neighbors_a and neighbors_b
            neighbors_b = adj[b]
            
            # To find the smallest c > b, iterate through sorted neighbors of b
            for c in sorted_adj[b]:
                if c > b and c in neighbors_a:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
