import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To minimize a * 10^8 + b * 10^4 + c, we iterate through a, then b, then c.
    # Since we need to find the lexicographically smallest (a, b, c),
    # we can iterate through all edges (u, v) and check for a common neighbor w.
    
    best_a, best_b, best_c = float('inf'), float('inf'), float('inf')
    
    # Sort edges to process in order of potential a, b
    edges.sort()
    
    for u, v in edges:
        # u < v is guaranteed by the input processing
        # We look for w such that (u, w) and (v, w) exist.
        # To minimize a*10^8 + b*10^4 + c, we check all possible w.
        
        # Common neighbors of u and v
        common = adj[u].intersection(adj[v])
        for w in common:
            # Form the triplet (a, b, c) sorted
            triplet = sorted([u, v, w])
            a, b, c = triplet
            
            if a < best_a or (a == best_a and b < best_b) or (a == best_a and b == best_b and c < best_c):
                best_a, best_b, best_c = a, b, c
                
    print(f"{best_a} {best_b} {best_c}")

if __name__ == '__main__':
    solve()
