import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # We need to find a, b, c such that a < b < c and
    # (a, b), (a, c), (b, c) are all edges.
    # This is a triangle in the graph.
    # To minimize a * 10^8 + b * 10^4 + c, we iterate a from 1 to n,
    # then b from a+1 to n, then check if c exists in intersection of adj[a] and adj[b].
    
    best_a, best_b, best_c = float('inf'), float('inf'), float('inf')
    
    # Sort edges to iterate efficiently
    edges.sort()
    
    # To optimize, iterate over edges (a, b) and check for common neighbors c > b
    for a, b in edges:
        # Ensure a < b
        if a > b:
            a, b = b, a
        
        # Find common neighbors c > b
        # Intersection of adj[a] and adj[b]
        common = adj[a].intersection(adj[b])
        for c in common:
            if c > b:
                # Found a triangle (a, b, c) with a < b < c
                if a < best_a or (a == best_a and b < best_b) or (a == best_a and b == best_b and c < best_c):
                    best_a, best_b, best_c = a, b, c
            elif c < a:
                # Triangle (c, a, b)
                if c < best_a or (c == best_a and a < best_b) or (c == best_a and a == best_b and b < best_c):
                    best_a, best_b, best_c = c, a, b
            else:
                # Triangle (a, c, b)
                if a < best_a or (a == best_a and c < best_b) or (a == best_a and c == best_b and b < best_c):
                    best_a, best_b, best_c = a, c, b
                    
    print(f"{best_a} {best_b} {best_c}")

if __name__ == '__main__':
    solve()
