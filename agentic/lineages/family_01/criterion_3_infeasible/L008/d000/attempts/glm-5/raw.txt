import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]); v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    
    best = None
    
    for a in range(1, n + 1):
        if best is not None and a >= best[0]:
            break
        adj_a = adj[a]
        # candidates b must be > a and adjacent to a
        # we want smallest b, then smallest c
        # iterate b in sorted order
        for b in sorted(adj_a):
            if b <= a:
                continue
            if best is not None and b >= best[1]:
                # since a < best[0], any b >= best[1] would give (a,b,*) > (best[0],best[1],best[2])? 
                # Actually lexicographic: (a,b,c) vs (best[0],best[1],best[2])
                # a < best[0] already, so (a,b,c) < best regardless of b,c. So we can't break here.
                # But we want smallest b. So we should still check this b.
                pass
            # find c > b, c in adj[a] and adj[b]
            # iterate over smaller set
            if len(adj_a) <= len(adj[b]):
                for c in adj_a:
                    if c > b and c in adj[b]:
                        if best is None or (a, b, c) < best:
                            best = (a, b, c)
                        break  # smallest c for this b
            else:
                for c in adj[b]:
                    if c > b and c in adj_a:
                        if best is None or (a, b, c) < best:
                            best = (a, b, c)
                        break  # smallest c for this b
        # Since we iterate a in increasing order, the first a that yields any triangle
        # gives the lexicographically smallest (because a is smallest possible).
        # But we need smallest b then c for that a. We iterate b in sorted order and take first c.
        # So once we find best with best[0] == a, we can stop? 
        # Actually best[0] == a means we found a triangle with smallest a. 
        # But could there be a smaller a later? No, a increases. So we can break.
        if best is not None and best[0] == a:
            # we have found the lexicographically smallest triangle
            # because a is minimal, and for this a we took minimal b and minimal c
            break
    
    if best is not None:
        sys.stdout.write(f"{best[0]} {best[1]} {best[2]}\n")

solve()
