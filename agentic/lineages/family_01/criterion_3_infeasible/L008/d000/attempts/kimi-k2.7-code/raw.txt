 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    full = (1 << (n + 1)) - 1
    high_mask = [0] * (n + 1)
    for x in range(n + 1):
        high_mask[x] = full ^ ((1 << (x + 1)) - 1)

    for a in range(1, n + 1):
        cand = adj[a] & high_mask[a]
        while cand:
            lsb = cand & -cand
            b = lsb.bit_length() - 1
            common = adj[b] & cand & high_mask[b]
            if common:
                c = (common & -common).bit_length() - 1
                sys.stdout.write(f"{a} {b} {c}\n")
                return
            cand ^= lsb

if __name__ == "__main__":
    solve()
```
