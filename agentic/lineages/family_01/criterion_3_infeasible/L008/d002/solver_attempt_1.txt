import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    best_score = float('inf')
    best_triple = None
    
    # Iterate through all edges (u, v) and check for a common neighbor w
    # To minimize a*10^8 + b*10^4 + c, we iterate in a way that respects the order
    # or simply check all valid triples and keep the minimum.
    
    for u, v in edges:
        # u < v is guaranteed by the input processing
        # We look for w such that (u, w) and (v, w) exist
        # To ensure a < b < c, we check three cases for w:
        # 1. w < u < v: triple is (w, u, v)
        # 2. u < w < v: triple is (u, w, v)
        # 3. u < v < w: triple is (u, v, w)
        
        # Optimization: intersect the adjacency sets
        common = adj[u].intersection(adj[v])
        for w in common:
            # Sort the triple to get a < b < c
            triple = sorted([u, v, w])
            a, b, c = triple
            score = a * 100000000 + b * 10000 + c
            if score < best_score:
                best_score = score
                best_triple = (a, b, c)
                
    if best_triple:
        print(f"{best_triple[0]} {best_triple[1]} {best_triple[2]}")

if __name__ == '__main__':
    solve()
