import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to help find the lexicographically smallest (a, b, c)
    # We want to minimize a*10^8 + b*10^4 + c
    # This is equivalent to finding the smallest (a, b, c) lexicographically
    
    best = (float('inf'), float('inf'), float('inf'))
    
    # Iterate through all edges (u, v) as potential (a, b)
    # For each edge, find common neighbors w
    # To optimize, iterate over the vertex with smaller degree
    
    # Pre-sort edges by (u, v)
    edges.sort()
    
    for u, v in edges:
        # We need to find w such that (u, w) and (v, w) exist
        # To ensure a < b < c, we check w > v
        # But wait, the triplet could be (w, u, v) or (u, w, v) or (u, v, w)
        # The problem asks for a < b < c.
        
        # Let's iterate over all edges (u, v) with u < v
        # Find common neighbors w
        # For each w, form the triplet (min(u,v,w), mid(u,v,w), max(u,v,w))
        
        # Optimization: only check w > v to avoid duplicates and ensure order
        # Actually, just check all common neighbors and sort the triplet
        
        u_set = adj[u]
        v_set = adj[v]
        
        # Intersect sets
        if len(u_set) < len(v_set):
            common = [w for w in u_set if w in v_set]
        else:
            common = [w for w in v_set if w in u_set]
            
        for w in common:
            triplet = sorted([u, v, w])
            if triplet < list(best):
                best = tuple(triplet)
                
    print(f"{best[0]} {best[1]} {best[2]}")

solve()
