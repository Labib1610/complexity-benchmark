import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that a < b < c
    # and (a,b), (a,c), (b,c) are all in the incident list.
    # To minimize a, then b, then c:
    
    # Sort adjacency lists to help with iteration
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # We only care about neighbors b > a
        valid_b = [b for b in neighbors_a if b > a]
        
        for i in range(len(valid_b)):
            b = valid_b[i]
            # Check for c > b such that c is in both adj[a] and adj[b]
            # Since adj lists are sorted, we can intersect efficiently
            # or just iterate through neighbors of b that are > b
            for c in adj[b]:
                if c > b:
                    if c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
