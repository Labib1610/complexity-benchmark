import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A group of 3 workers {a, b, c} is contradictory if:
    # Let E_sub be the set of edges within {a, b, c}.
    # deg_a = count of edges in E_sub connected to a
    # deg_b = count of edges in E_sub connected to b
    # deg_c = count of edges in E_sub connected to c
    # All must be even and > 0.
    # Since the sum of degrees in a graph is 2 * |E|, and we have 3 nodes,
    # the only way for all degrees to be even and > 0 is if all degrees are 2.
    # This means the subgraph induced by {a, b, c} must be a triangle (a-b, b-c, c-a).
    # Wait, if it's a triangle, degrees are 2, 2, 2. 2 is even and > 0.
    # So the condition is simply that {a, b, c} must form a triangle.
    
    # Iterate through all triplets (a, b, c) with a < b < c
    # To optimize, iterate over a, then b in adj[a], then c in adj[a] intersect adj[b]
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Find common neighbors c such that c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in sorted(list(common)):
                if c > b:
                    # Found a triangle {a, b, c}
                    # Check if degrees are even and > 0
                    # In a triangle, deg_a=2, deg_b=2, deg_c=2. All are even and > 0.
                    print(f"{a} {b} {c}")
                    return

solve()
