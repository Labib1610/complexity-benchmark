import sys

def solve():
    # A trio {u, v, w} is critically flawed if every assignment of {u, v, w} 
    # to Set A or Set B results in an edge within A or an edge within B.
    # This is equivalent to saying that the subgraph induced by {u, v, w} 
    # is not bipartite. A graph is bipartite if and only if it contains no odd cycles.
    # A trio of 3 vertices forms an odd cycle if and only if it is a triangle (K3).
    # If the trio is a triangle (u, v, w), then:
    # - Assignment (A, A, A) has edges (u,v), (v,w), (u,w) in A.
    # - Assignment (B, B, B) has edges (u,v), (v,w), (u,w) in B.
    # - Any assignment with 2 in A and 1 in B has an edge in A.
    # - Any assignment with 1 in A and 2 in B has an edge in B.
    # Thus, a trio is critically flawed if and only if it forms a triangle.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v: u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, we can optimize triangle finding.
    # Sort adjacency lists to help with lexicographical order.
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a: continue
            
            # Check for common neighbors c > b
            # Intersection of adj[a] and adj[b]
            # Since adj lists are sorted, we can use two pointers or set intersection
            set_a = adj[a]
            set_b = adj[b]
            
            # We need c > b such that c is in both adj[a] and adj[b]
            # Since we want the smallest c, we just need the first one > b
            for c in set_b:
                if c > b and c in set_a:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
