```python
import sys

def solve():
    # A subset of three elements {a, b, c} is locked if its defect score is > 0
    # for every binary mapping M: {a, b, c} -> {0, 1}.
    # There are 2^3 = 8 possible mappings.
    # Let the links be E. The defect score for M is the number of edges {u, v}
    # such that u, v in {a, b, c} and M(u) == M(v).
    # For a subset to be locked, for every M, there must be at least one edge
    # {u, v} such that M(u) == M(v).
    # This is equivalent to saying that the complement graph (the non-edges)
    # does not contain an independent set of size 2 that covers all vertices
    # in a way that allows M(u) != M(v) for all edges.
    # Actually, a simpler condition: {a, b, c} is locked if and only if
    # the edges form a triangle (K3) OR the edges form a path of length 2
    # plus the third edge (which is K3) OR...
    # Let's analyze the condition:
    # For every M, there exists an edge {u, v} with M(u) == M(v).
    # This is equivalent to: there is NO mapping M such that for all edges {u, v}, M(u) != M(v).
    # M(u) != M(v) for all edges means the graph is bipartite.
    # So, {a, b, c} is locked if and only if the subgraph induced by {a, b, c}
    # is NOT bipartite.
    # A graph is bipartite if and only if it contains no odd cycles.
    # The only odd cycle in a 3-element subset is a triangle (K3).
    # Wait, let's re-check:
    # If the induced subgraph is K3 (all 3 edges present), it's not bipartite.
    # If the induced subgraph has 2 edges (a path of length 2), it is bipartite.
    # If the induced subgraph has 1 edge, it is bipartite.
    # If the induced subgraph has 0 edges, it is bipartite.
    # So {a, b, c} is locked if and only if it contains all 3 edges (a triangle).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need the lexicographically smallest {a, b, c} with a < b < c
    # such that {a, b}, {b, c}, and {a, c} are all edges.
    for a in range(1, n + 1):
        # Optimization: only check b > a and c > b
        # To be a triangle, b must be a neighbor of a, and c must be a neighbor of both a and b.
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # c must be a neighbor of a and b, and c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in sorted(common):
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
