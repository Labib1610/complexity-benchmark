import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Check for triangle (3-clique)
    for u, v in edges:
        common = adj[u] & adj[v]
        for w in common:
            if w > v:
                print(u, v, w)
                return
    
    # Check for independent set of size 3
    # Find non-neighbor pairs
    non_edge_pairs = []
    for i in range(1, n + 1):
        for j in range(i + 1, n + 1):
            if j not in adj[i]:
                non_edge_pairs.append((i, j))
    
    # Group by first vertex
    groups = [[] for _ in range(n + 1)]
    for u, v in non_edge_pairs:
        groups[u].append(v)
    
    # For each group, find pair with no edge between them
    for u in range(1, n + 1):
        lst = groups[u]
        sz = len(lst)
        for i in range(sz):
            vi = lst[i]
            for j in range(i + 1, sz):
                vj = lst[j]
                if vj not in adj[vi]:
                    print(u, vi, vj)
                    return

if __name__ == "__main__":
    solve()
