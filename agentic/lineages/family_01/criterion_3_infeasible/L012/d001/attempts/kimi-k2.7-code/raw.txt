import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    bits = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        bits[u] |= 1 << (v - 1)
        bits[v] |= 1 << (u - 1)
        adj[u].append(v)
        adj[v].append(u)

    for i in range(1, n + 1):
        adj[i].sort()

    for a in range(1, n + 1):
        ba = bits[a]
        for b in adj[a]:
            if b <= a:
                continue
            common = ba & bits[b]
            if common:
                # keep only common neighbors with identifier greater than b
                common &= -(1 << b)
                if common:
                    c = (common & -common).bit_length()
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
