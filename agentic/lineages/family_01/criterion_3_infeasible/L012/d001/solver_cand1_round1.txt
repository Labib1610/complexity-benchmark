```python
import sys

def solve():
    # The problem asks for a subset of 3 sensors that cannot be assigned to 2 phases.
    # This is equivalent to finding a triangle (a clique of size 3) in the graph.
    # If three sensors form a triangle, each pair is connected, so no two can share a phase.
    # With only 2 phases available, 3 sensors forming a triangle cannot be assigned.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append(tuple(sorted((u, v))))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # We iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, we can optimize by iterating over edges (u, v)
    # and checking for a common neighbor w.
    
    best = None
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    # To optimize, iterate through all nodes as 'a'
    for a in range(1, n + 1):
        neighbors = sorted(list(adj[a]))
        len_n = len(neighbors)
        for i in range(len_n):
            b = neighbors[i]
            if b <= a:
                continue
            
            # Check if there exists c > b such that c is connected to both a and b
            # We want the smallest c, so we check neighbors of b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    # Found a triangle (a, b, c)
                    # Since we want the lexicographically smallest, we keep track
                    res = tuple(sorted((a, b, c)))
                    if best is None or res < best:
                        best = res
        
        # Optimization: if we found a triangle starting with 'a', 
        # we don't necessarily need to check higher 'a's if we are sure 
        # we found the smallest. However, the current loop structure 
        # is efficient enough for N=2000.
        if best is not None and a > best[0]:
            break
            
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
