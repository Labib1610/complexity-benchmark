import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to process in lexicographic order
    edges.sort()
    
    # For each edge (u, v), check for common neighbors w where w > v
    # This ensures u < v < w, giving us sorted output
    for u, v in edges:
        # Find common neighbors w where w > v
        # Iterate through the smaller adjacency set for efficiency
        if len(adj[u]) > len(adj[v]):
            neighbors_u = adj[v]
            neighbors_v = adj[u]
        else:
            neighbors_u = adj[u]
            neighbors_v = adj[v]
        
        for w in neighbors_u:
            if w > v and w in neighbors_v:
                print(u, v, w)
                return
    
    # Fallback: brute force check all triples (should not reach here per problem guarantee)
    for a in range(1, n - 1):
        for b in range(a + 1, n):
            if b not in adj[a]:
                continue
            for c in range(b + 1, n + 1):
                if c in adj[a] and c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
