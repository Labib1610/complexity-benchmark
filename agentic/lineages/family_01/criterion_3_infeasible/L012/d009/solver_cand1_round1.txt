```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A subset T = {a, b, c} is locked if for every assignment W:
    # W(a)W(b) + W(b)W(c) + W(a)W(c) > -L_T
    # where L_T is the number of links in T.
    # Possible values for L_T are 0, 1, 2, 3.
    # If L_T = 0, 1, 2:
    #   L_T = 0: score = 0 > 0 (False)
    #   L_T = 1: score = W(u)W(v) can be -1. -1 > -1 (False)
    #   L_T = 2: score = W(u)W(v) + W(v)W(w) can be -2. -2 > -2 (False)
    #   L_T = 3: score = W(a)W(b) + W(b)W(c) + W(a)W(c).
    #     Possible values for W(a)W(b) + W(b)W(c) + W(a)W(c):
    #     If W(a)=W(b)=W(c)=1: 1+1+1 = 3
    #     If W(a)=W(b)=1, W(c)=-1: 1-1-1 = -1
    #     The minimum value is -1.
    #     Condition: -1 > -3 (True).
    # So a subset is locked if and only if it forms a triangle (L_T = 3).
    
    # We need the lexicographically smallest triangle (a, b, c) with a < b < c.
    # Iterate a from 1 to n-2
    for a in range(1, n + 1):
        adj_a = adj[a]
        # Iterate b from a+1 to n-1
        for b in range(a + 1, n + 1):
            if b in adj_a:
                # Check for c > b such that c is connected to both a and b
                # Intersection of adj[a] and adj[b]
                common = adj_a.intersection(adj[b])
                # Filter c > b
                valid_c = [c for c in common if c > b]
                if valid_c:
                    print(f"{a} {b} {min(valid_c)}")
                    return

if __name__ == '__main__':
    solve()
```
