import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)          # bitset of neighbors (1‑based indices, bit 0 for vertex 1)
    neigh_gt = [[] for _ in range(n + 1)]   # neighbors > vertex, sorted

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        # store undirected
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)
        if u < v:
            neigh_gt[u].append(v)
        else:
            neigh_gt[v].append(u)

    # ensure neighbor lists are sorted (they already are by input order, but sort to be safe)
    for lst in neigh_gt:
        lst.sort()

    for a in range(1, n - 1):
        bits_a = adj[a]
        for b in neigh_gt[a]:          # b > a and edge a‑b exists
            # common neighbors of a and b
            common = bits_a & adj[b]
            # we need c > b
            # mask out bits <= b
            mask = ~((1 << b) - 1)      # bits with index >= b (0‑based) i.e., vertex >= b+1
            common &= mask
            if common:
                # lowest set bit gives smallest c
                low = common & -common
                c = low.bit_length()    # gives position (1‑based) of that bit
                # c is vertex number (since bit 0 -> vertex 1)
                print(a, b, c)
                return

if __name__ == "__main__":
    solve()
