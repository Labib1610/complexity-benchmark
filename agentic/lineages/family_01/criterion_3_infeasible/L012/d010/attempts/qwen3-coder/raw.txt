import sys
from itertools import combinations

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    edges = set()
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.add((u, v) if u < v else (v, u))
        adj[u].add(v)
        adj[v].add(u)
    return n, m, edges, adj

def is_connected(a, b, adj):
    return b in adj[a]

def is_unresolvable(a, b, c, adj):
    # Check connectivity
    connections = 0
    if is_connected(a, b, adj):
        connections += 1
    if is_connected(a, c, adj):
        connections += 1
    if is_connected(b, c, adj):
        connections += 1
    
    # If less than 2 connections, it's unresolvable
    if connections < 2:
        return True
    
    # If fully connected (triangle), check if the only solution is trivial
    if connections == 3:
        # For a triangle, the system of equations is:
        # P_b + P_c = 0
        # P_a + P_c = 0
        # P_a + P_b = 0
        # This has only the trivial solution P_a = P_b = P_c = 0
        return True
    
    # If exactly 2 connections, it's resolvable (non-trivial solution exists)
    return False

def solve():
    n, m, edges, adj = read_input()
    
    # Try all combinations of 3 modules in lexicographical order
    for a, b, c in combinations(range(1, n + 1), 3):
        if is_unresolvable(a, b, c, adj):
            print(a, b, c)
            return

if __name__ == "__main__":
    solve()
