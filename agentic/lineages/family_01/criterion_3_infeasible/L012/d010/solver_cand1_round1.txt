import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A triplet {a, b, c} is unresolvable if the system of equations:
    # E(a,b)Pb + E(a,c)Pc = 0
    # E(b,a)Pa + E(b,c)Pc = 0
    # E(c,a)Pa + E(c,b)Pb = 0
    # has only the trivial solution Pa=Pb=Pc=0.
    # This happens if the determinant of the coefficient matrix is non-zero.
    # Matrix:
    # [ 0, Eab, Eac ]
    # [ Eba, 0, Ebc ]
    # [ Eca, Ecb, 0 ]
    # Determinant = 0*0*0 + Eab*Ebc*Eca + Eac*Eba*Ecb - (Eac*0*Eca + Eab*Eba*0 + 0*Ebc*Ecb)
    # Determinant = Eab*Ebc*Eca + Eac*Eba*Ecb
    # Since Euv = Evu, Det = 2 * Eab * Ebc * Eca.
    # The triplet is unresolvable if Det != 0, which means Eab*Ebc*Eca != 0.
    # This means Eab=1, Ebc=1, Eca=1.
    # So the triplet is unresolvable if and only if it forms a triangle (a clique of size 3).
    
    # We need the lexicographically smallest triplet {a, b, c} with a < b < c
    # such that a, b, c form a triangle.
    
    # To find the lexicographically smallest, iterate a from 1 to n-2,
    # b from a+1 to n-1, c from b+1 to n.
    # Check if a, b, c form a triangle.
    
    # Optimization: iterate a, then b, then check if c exists in adj[a] and adj[b].
    for a in range(1, n + 1):
        adj_a = adj[a]
        sorted_adj_a = sorted(list(adj_a))
        for i in range(len(sorted_adj_a)):
            b = sorted_adj_a[i]
            if b <= a:
                continue
            # We need c > b such that c is in adj[a] and adj[b]
            adj_b = adj[b]
            # Intersection of adj_a and adj_b
            common = adj_a.intersection(adj_b)
            for c in sorted(list(common)):
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
