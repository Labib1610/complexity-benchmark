```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest trio (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Optimization: Sort adjacency lists to check for common neighbors efficiently
    # or just use set intersection. Given n=2000, O(n^3) is too slow, but
    # O(n * m) or O(n * (n + m)) is acceptable.
    
    # Pre-sort adjacency lists to help with lexicographical order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider neighbors b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of sorted_adj[a] and sorted_adj[b]
            # Since we want the smallest c > b, we can iterate through
            # the intersection.
            
            # Using set intersection for speed
            common = adj[a].intersection(adj[b])
            # Filter for c > b
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
```
