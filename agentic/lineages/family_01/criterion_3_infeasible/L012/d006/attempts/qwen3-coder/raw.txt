```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list using sets for O(1) lookup
    adj = defaultdict(set)
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        adj[u].add(v)
        adj[v].add(u)
        index += 2

    # Convert to list of neighbors for each node and sort them
    neighbors = {}
    for node in adj:
        neighbors[node] = sorted(list(adj[node]))

    # Get all nodes sorted
    nodes = sorted(neighbors.keys())
    
    # For each node a (smallest first)
    for a in nodes:
        # For each neighbor b of a, where b > a
        neigh_a = neighbors[a]
        for b in neigh_a:
            if b <= a:
                continue
            # For each neighbor c of a, where c > b
            for c in neigh_a:
                if c <= b:
                    continue
                # Check if b and c are also connected
                if c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
