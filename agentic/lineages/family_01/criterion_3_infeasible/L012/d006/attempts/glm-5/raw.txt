import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges for lexicographically smallest triangle search
    edges.sort()
    
    for u, v in edges:
        # Find common neighbors w where w > v (to maintain a < b < c)
        # We need w such that u-w and v-w are edges
        # Iterate through smaller adjacency list
        if len(adj[u]) < len(adj[v]):
            smaller, larger = adj[u], adj[v]
        else:
            smaller, larger = adj[v], adj[u]
        
        for w in smaller:
            if w > v and w in larger:
                print(u, v, w)
                return
    
    # Should not reach here as solution is guaranteed
    print(-1)

if __name__ == "__main__":
    solve()
