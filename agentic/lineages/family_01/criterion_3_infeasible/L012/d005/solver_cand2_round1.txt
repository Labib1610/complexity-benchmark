import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that a < b < c, b in P(a), c in P(a), c in P(b)
    # This is equivalent to finding a triangle (a, b, c)
    # To get the lexicographically smallest (a, b, c), we iterate a from 1 to n,
    # then b from a+1 to n, then c from b+1 to n.
    
    # Sort adjacency lists to ensure we check in order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Filter neighbors to only those > a
        neighbors_a_gt = [x for x in neighbors_a if x > a]
        
        for i in range(len(neighbors_a_gt)):
            b = neighbors_a_gt[i]
            # We need c such that c > b, c in P(a), and c in P(b)
            # Intersection of neighbors_a and neighbors_b
            # Since we want smallest c, we iterate through sorted neighbors of b
            neighbors_b = sorted_adj[b]
            
            # Optimization: use set intersection for faster lookup
            # But we need the smallest c > b
            set_a = adj[a]
            for c in neighbors_b:
                if c > b and c in set_a:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
