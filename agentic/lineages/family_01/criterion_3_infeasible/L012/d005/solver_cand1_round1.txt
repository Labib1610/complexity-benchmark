import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
    
    # A Fault-Tolerant Triplet {a, b, c} must satisfy:
    # 1. At least one internal interlock.
    # 2. If any one internal interlock is removed, every sensor in the triplet
    #    still has degree >= 1 within the triplet.
    # This condition is equivalent to saying that every sensor in the triplet
    # must have degree >= 2 within the triplet.
    # A set of 3 sensors {a, b, c} has degree >= 2 for all if and only if
    # it forms a triangle (a-b, b-c, c-a).
    # Proof:
    # If it's a triangle, each node has degree 2. Removing any edge leaves
    # the two nodes of that edge with degree 1, and the third node with degree 2.
    # If it's not a triangle, the max degree is 2 (e.g., a-b, b-c).
    # Removing b-c leaves c with degree 0.
    
    # So we need to find the lexicographically smallest triangle {a, b, c}.
    
    adj_list = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        neighbors_a = adj_list[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Check for c such that {a, b, c} is a triangle
            # c must be a neighbor of both a and b
            # To ensure lexicographical order, we check c > b
            # But we also need to check c < b (if c < b, then {c, a, b} is the set)
            # Actually, we just need to find all triangles and pick the smallest.
            
            # Optimization: intersect neighbors of a and b
            set_b = adj[b]
            for c in neighbors_a:
                if c > b and c in set_b:
                    print(f"{a} {b} {c}")
                    return

solve()
