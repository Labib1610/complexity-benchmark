```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list using sets for O(1) lookup
    adj = defaultdict(set)
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        adj[u].add(v)
        adj[v].add(u)
        index += 2

    # Convert adjacency list to sorted lists for ordered iteration
    adj_list = {}
    for node in adj:
        adj_list[node] = sorted(list(adj[node]))

    # Iterate through all possible triplets (a,b,c) in lexicographical order
    for a in range(1, n+1):
        if a not in adj_list:
            continue
        neighbors_a = adj_list[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            if b not in adj_list:
                continue
            neighbors_b = adj_list[b]
            # Check for common neighbors of a and b that are > b
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
