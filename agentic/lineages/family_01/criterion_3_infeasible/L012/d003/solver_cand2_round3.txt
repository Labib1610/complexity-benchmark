import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) such that a < b < c.
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n,
    # then b from a+1 to n, then c from b+1 to n.
    # A triangle exists if c is in adj[a] and c is in adj[b].
    
    # Optimization: iterate a, then iterate b in adj[a] (where b > a),
    # then check if there exists c in (adj[a] intersection adj[b]) such that c > b.
    
    # To ensure lexicographical order, we sort the adjacency lists.
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Find intersection of neighbors_a and neighbors_b
            # Since both are sorted, we can find common elements efficiently
            neighbors_b = sorted_adj[b]
            
            # We need the smallest c > b such that c is in both
            ptr_a = 0
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

if __name__ == '__main__':
    solve()
