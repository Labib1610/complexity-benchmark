import sys

def solve():
    # The problem asks for a subset of 3 sensors that cannot be 2-colored.
    # A graph is 2-colorable if and only if it is bipartite (contains no odd cycles).
    # A subset of 3 sensors is not 2-colorable if and only if it contains an odd cycle.
    # The only odd cycle of length 3 is a triangle (a clique of size 3).
    # Thus, we need to find a triangle (u, v, w) in the graph.
    # We need the lexicographically smallest (a, b, c) where a < b < c.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c):
    # Iterate through all possible a from 1 to n-2.
    # For a fixed a, iterate through all b > a such that (a, b) is an edge.
    # For fixed a and b, check if there exists c > b such that (a, c) and (b, c) are edges.
    # To minimize (a, b, c), we want the smallest a, then smallest b, then smallest c.
    
    # Sort adjacency lists to help find smallest c
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    best = None
    
    # Iterate a
    for a in range(1, n - 1):
        # Iterate b > a
        for b in adj[a]:
            if b <= a: continue
            
            # We need c > b such that c is in adj[a] and adj[b]
            # Since adj[a] is sorted, we can iterate through it
            for c in adj[a]:
                if c <= b: continue
                if c in adj_sets[b]:
                    # Found a triangle (a, b, c)
                    # Since we iterate a, b, c in increasing order, 
                    # the first one we find is the lexicographically smallest.
                    print(f"{a} {b} {c}")
                    return

solve()
