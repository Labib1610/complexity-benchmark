```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (x, y, z) such that x, y, z are connected to each other.
    # The problem defines a critical flaw as x, y, z such that y in L(x), z in L(x), and z in L(y).
    # This is exactly the definition of a triangle (a clique of size 3).
    # We need the lexicographically smallest (a, b, c) with a < b < c.
    
    # To find the lexicographically smallest triangle:
    # Iterate a from 1 to n-2
    #   Iterate b from a+1 to n-1
    #     If b is in adj[a]:
    #       Find common neighbors c of a and b
    #       If any c > b exists, the smallest such c gives the smallest triangle starting with (a, b)
    
    # Sort neighbors to make finding smallest c efficient
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Find common neighbors of a and b
            # Since neighbors are sorted, we can use two pointers or set intersection
            set_a = adj[a]
            for c in sorted_adj[b]:
                if c > b and c in set_a:
                    # Found the smallest triangle (a, b, c)
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
```
