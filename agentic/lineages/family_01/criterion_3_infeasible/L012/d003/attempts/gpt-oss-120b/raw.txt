import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    # adjacency matrix for quick edge check
    adj_mat = [bytearray(n) for _ in range(n)]
    # bitmask of neighbors for each vertex (0‑based)
    adj_bits = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj_mat[u][v] = adj_mat[v][u] = 1
        adj_bits[u] |= 1 << v
        adj_bits[v] |= 1 << u

    full_mask = (1 << n) - 1

    for i in range(n):
        bi = adj_bits[i]
        for j in range(i + 1, n):
            if adj_mat[i][j]:          # edge exists, cannot be in independent set
                continue
            # vertices not adjacent to i or j
            common = full_mask ^ (bi | adj_bits[j])
            # keep only vertices with index > j
            common &= ~((1 << (j + 1)) - 1)
            if common:
                # lowest set bit gives smallest k
                low = common & -common
                k = low.bit_length() - 1
                print(i + 1, j + 1, k + 1)
                return

if __name__ == "__main__":
    solve()
