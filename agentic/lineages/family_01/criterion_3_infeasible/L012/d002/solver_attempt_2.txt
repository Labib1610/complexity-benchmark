import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # We iterate through all possible a, then b, then check if c exists.
    # Since we need the smallest, we sort the edges or iterate carefully.
    # Given n=2000, O(n*m) or O(n^2) is acceptable.
    
    # Sort edges to help find smallest a, b
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # all possible a from 1 to n-2.
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # For a fixed a, we want the smallest b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b < a:
                continue
            
            # Now we need the smallest c > b such that c is connected to both a and b
            # c must be in adj[a] and adj[b]
            # To optimize, intersect the two sets
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
