import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find the lexicographically smallest triplet (a, b, c)
    # such that a < b < c and (a,b), (b,c), (c,a) are all edges.
    # Iterate through possible a, then b, then c.
    
    # To optimize, sort the adjacency lists
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Only consider b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need c > b such that c is in adj[a] and adj[b]
            # Intersection of sorted_adj[a] and sorted_adj[b]
            # Since we need c > b, we can look at elements in sorted_adj[b]
            # that are also in sorted_adj[a] and are > b.
            
            # Optimization: use set intersection for speed
            common = adj[a].intersection(adj[b])
            
            # Find the smallest c > b
            best_c = float('inf')
            found = False
            for c in common:
                if c > b:
                    if c < best_c:
                        best_c = c
                        found = True
            
            if found:
                print(f"{a} {b} {best_c}")
                return

if __name__ == '__main__':
    solve()
