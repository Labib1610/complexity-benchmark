```python
import sys

def solve():
    # The condition "x_u + x_v = 0" for all edges in a subset of 3 nodes
    # means that if we have edges (u, v), (v, w), (w, u), then:
    # x_u + x_v = 0
    # x_v + x_w = 0
    # x_w + x_u = 0
    # This implies 2(x_u + x_v + x_w) = 0, so x_u + x_v + x_w = 0.
    # Substituting x_v = -x_u and x_w = -x_v = x_u, we get x_u + (-x_u) + x_u = 0,
    # which means x_u = 0. Thus, a triangle (3-cycle) forces all x_i = 0.
    # The problem asks for a subset of 3 components where the only solution is 0.
    # This happens if and only if the subgraph induced by the 3 nodes contains a cycle of odd length.
    # In a 3-node set, the only odd cycle is a triangle.
    # So we are looking for a triangle (a, b, c).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append(tuple(sorted((u, v))))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    
    # Sort adjacency lists to help with lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            # Check for common neighbor c > b
            # Intersection of sorted lists
            set_a = set(adj[a])
            set_b = set(adj[b])
            common = sorted(list(set_a.intersection(set_b)))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
