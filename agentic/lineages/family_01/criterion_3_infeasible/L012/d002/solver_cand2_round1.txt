```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
    
    # We need to find a triangle (a, b, c) such that (a,b), (b,c), (a,c) are edges.
    # The problem asks for the lexicographically smallest (a, b, c).
    # Since n <= 2000, we can iterate through all edges (u, v) and check for common neighbors.
    
    best = None
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    for u, v in edges:
        # Find common neighbors of u and v
        # To optimize, iterate over the smaller adjacency list
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        
        for w in adj[u]:
            if w in adj[v]:
                # Found a triangle (u, v, w)
                # Sort the triplet
                triplet = sorted([u, v, w])
                if best is None or triplet < best:
                    best = triplet
                    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
