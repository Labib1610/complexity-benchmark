```python
import sys

def solve():
    # The problem describes a "concurrency deadlock" as a set of three processes
    # where each pair is incompatible. This is equivalent to finding a triangle
    # (a clique of size 3) in the graph of incompatibilities.
    # We need to find the triangle (a, b, c) with a < b < c that minimizes
    # a * 10^8 + b * 10^4 + c.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To minimize a * 10^8 + b * 10^4 + c, we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # c is in adj[a] and adj[b].
    
    best_a, best_b, best_c = float('inf'), float('inf'), float('inf')
    
    # Sort adjacency lists to make intersection faster and ordered
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Optimization: if a is already worse than best_a, we can't improve
        if a > best_a:
            break
            
        for b in adj_list[a]:
            if b <= a:
                continue
            
            # Check if (a, b, c) is a triangle
            # We need c > b and c in adj[a] and c in adj[b]
            # Since adj_list is sorted, we can intersect
            set_a = adj[a]
            set_b = adj[b]
            
            # Find common neighbors c > b
            for c in set_a:
                if c > b and c in set_b:
                    # Found a triangle (a, b, c)
                    # Since we iterate a and b in increasing order,
                    # we check if this is the lexicographically smallest
                    if a < best_a or (a == best_a and b < best_b) or (a == best_a and b == best_b and c < best_c):
                        best_a, best_b, best_c = a, b, c
                        # Since we want the minimum trace, and we iterate a, b, c
                        # in increasing order, the first one we find might not be the
                        # absolute minimum if we don't finish the loops.
                        # However, the trace formula a*10^8 + b*10^4 + c is 
                        # lexicographical.
    
    print(f"{best_a} {best_b} {best_c}")

solve()
```
