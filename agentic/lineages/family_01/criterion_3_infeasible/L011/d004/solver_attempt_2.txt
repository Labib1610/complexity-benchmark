```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To minimize a*x^2 + b*x + c with x = n^2,
    # we prioritize smaller a, then smaller b, then smaller c.
    # This is equivalent to lexicographical comparison of (a, b, c).
    
    best_triplet = None
    
    # Iterate through all edges (u, v) and check for a common neighbor w
    # To ensure a < b < c, we check neighbors w such that w > v or w < u or u < w < v
    for u, v in edges:
        # Common neighbors w
        # We need to find w such that (u, w) and (v, w) are edges
        # To avoid duplicates and ensure ordering, we check all w
        # Intersection of adj[u] and adj[v]
        common = adj[u].intersection(adj[v])
        for w in common:
            # Sort the three to get a < b < c
            triplet = sorted([u, v, w])
            if best_triplet is None or triplet < best_triplet:
                best_triplet = triplet
                
    print(f"{best_triplet[0]} {best_triplet[1]} {best_triplet[2]}")

if __name__ == '__main__':
    solve()
```
