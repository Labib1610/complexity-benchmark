import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the minimum hardware trace a*10^8 + b*10^4 + c,
    # we need to iterate through possible triplets (a, b, c) with a < b < c.
    # Since we want the minimum, we iterate a from 1 to n, b from a+1 to n,
    # and check if c exists such that (a,b), (b,c), and (a,c) are edges.
    
    # Optimization: iterate over edges (a, b) and check common neighbors
    # Sort edges to help find the smallest trace
    edges.sort()
    
    min_trace = float('inf')
    best_triplet = None
    
    # For each edge (u, v), find common neighbors w
    for u, v in edges:
        # Ensure u < v
        a, b = (u, v) if u < v else (v, u)
        
        # Find common neighbors of a and b
        # To minimize trace, we want to check potential c values
        # Common neighbors are nodes w such that w in adj[a] and w in adj[b]
        common = adj[a].intersection(adj[b])
        
        for w in common:
            # Form triplet (a, b, w) sorted
            triplet = sorted([a, b, w])
            a_val, b_val, c_val = triplet
            
            trace = a_val * 100000000 + b_val * 10000 + c_val
            if trace < min_trace:
                min_trace = trace
                best_triplet = (a_val, b_val, c_val)
                
    if best_triplet:
        print(f"{best_triplet[0]} {best_triplet[1]} {best_triplet[2]}")

if __name__ == '__main__':
    solve()
