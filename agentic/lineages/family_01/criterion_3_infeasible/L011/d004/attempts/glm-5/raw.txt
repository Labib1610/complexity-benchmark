import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    deg = [0] * (n + 1)
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u][v] = True
        adj[v][u] = True
        deg[u] += 1
        deg[v] += 1
    
    # We need to find a triangle (K3) in the complement graph.
    # A triangle in complement means all 3 pairs are non-edges in original graph.
    # For 3 nodes a, b, c, if all pairs are non-edges, then in any 2-coloring,
    # at least one pair is monochromatic, causing a collision.
    # Since the problem guarantees a solution exists, we search for such a triangle.
    
    # Optimization: iterate a from 1 to n.
    # For each a, iterate b from a+1 to n.
    # If (a,b) is a non-edge, iterate c from b+1 to n.
    # If (a,c) and (b,c) are also non-edges, we found the lexicographically smallest triplet.
    
    # To speed up the inner check for c, we can iterate over nodes c > b
    # and check if they are non-neighbors of both a and b.
    # Since we stop at the first found triplet, efficiency depends on how early it is found.
    
    for a in range(1, n + 1):
        # We look for b > a such that a and b are not connected
        for b in range(a + 1, n + 1):
            if adj[a][b]:
                continue
            # a and b are not connected, now find c > b
            # such that a-c and b-c are also not connected
            for c in range(b + 1, n + 1):
                if not adj[a][c] and not adj[b][c]:
                    print(a, b, c)
                    return

if __name__ == '__main__':
    solve()
