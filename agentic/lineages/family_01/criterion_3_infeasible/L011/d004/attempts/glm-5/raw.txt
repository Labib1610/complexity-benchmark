import sys

def solve() -> None:
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    min_a = n + 1
    min_b = n + 1
    min_c = n + 1
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Pruning: if current a is already larger than the best a found, we can stop
        if a > min_a:
            break
        
        # Sort neighbors to check in order, enabling early termination
        sorted_neighbors = sorted(neighbors_a)
        
        for b in sorted_neighbors:
            if b <= a:
                continue
            
            # Pruning: if a == min_a and b > min_b, no need to check further for this a
            if a == min_a and b > min_b:
                break
            
            for c in sorted_neighbors:
                if c <= b:
                    continue
                
                # Pruning: if a == min_a and b == min_b and c > min_c
                if a == min_a and b == min_b and c > min_c:
                    break
                
                # Check if b and c are connected
                if c in adj[b]:
                    min_a, min_b, min_c = a, b, c
                    # Once we find a valid triplet starting with (a, b), 
                    # any subsequent c will be larger, so we can break
                    break
    
    print(f"{min_a} {min_b} {min_c}")

if __name__ == "__main__":
    solve()
