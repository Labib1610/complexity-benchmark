```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A tertiary aggregation knot is a triangle (clique of size 3)
    # We need to find the triplet (a, b, c) with a < b < c
    # that minimizes a * 10^8 + b * 10^4 + c.
    # This is equivalent to finding the lexicographically smallest (a, b, c).
    
    best_triplet = None
    
    # Iterate through all possible a
    for a in range(1, n + 1):
        # Optimization: if we already found a triplet starting with a smaller a,
        # we don't need to check larger a's.
        if best_triplet and a > best_triplet[0]:
            break
            
        neighbors_a = sorted(list(adj[a]))
        
        # Iterate through all possible b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            if best_triplet and a == best_triplet[0] and b > best_triplet[1]:
                break
            
            # Check for c > b such that c is connected to both a and b
            # c must be in adj[a] and adj[b]
            # To minimize c, we look at the intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    if best_triplet is None or (a, b, c) < best_triplet:
                        best_triplet = (a, b, c)
    
    if best_triplet:
        print(f"{best_triplet[0]} {best_triplet[1]} {best_triplet[2]}")

if __name__ == '__main__':
    solve()
```
