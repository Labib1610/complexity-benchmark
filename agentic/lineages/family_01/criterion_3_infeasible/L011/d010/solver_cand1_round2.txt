import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    def get_val(i, j):
        if i == j or j not in adj[i]:
            return 0
        return i + j

    # We need to find the lexicographically smallest triple (a, b, c)
    # such that the matrix M is invertible (det != 0).
    # M = [[0, s1+s2, s1+s3], [s2+s1, 0, s2+s3], [s3+s1, s3+s2, 0]]
    # det(M) = 0 + (s1+s2)(s2+s3)(s3+s1) + (s1+s3)(s1+s2)(s3+s2) - 0
    # det(M) = 2 * (s1+s2) * (s2+s3) * (s3+s1)
    # Since s_i >= 1, s_i + s_j > 0.
    # The matrix is linearly independent if and only if det(M) != 0.
    # det(M) != 0 iff (s1+s2) != 0, (s2+s3) != 0, and (s3+s1) != 0.
    # This means all three pairs must have a shared tool.
    # So the condition is simply that the three steps form a triangle in the graph.
    
    # Sort adjacency lists to ensure we find the lexicographically smallest triple
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        adj_a = adj[a]
        for i in range(len(adj_a)):
            b = adj_a[i]
            if b <= a:
                continue
            # We need c > b such that c is connected to a and b
            # Intersection of sorted lists
            adj_b = adj[b]
            ptr_a = i + 1
            ptr_b = 0
            while ptr_a < len(adj_a) and ptr_b < len(adj_b):
                val_a = adj_a[ptr_a]
                val_b = adj_b[ptr_b]
                if val_a == val_b:
                    c = val_a
                    if c > b:
                        print(f"{a} {b} {c}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
