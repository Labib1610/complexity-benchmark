import sys
from itertools import combinations

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    edges = set()
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        if u > v:
            u, v = v, u
        edges.add((u, v))
    return n, edges

def are_linearly_independent(s1, s2, s3, edges):
    # Create vectors v1, v2, v3
    # v1 = (0, s1+s2, s1+s3) if (s1,s2) and (s1,s3) in edges, else 0
    # v2 = (s2+s1, 0, s2+s3) if (s2,s1) and (s2,s3) in edges, else 0
    # v3 = (s3+s1, s3+s2, 0) if (s3,s1) and (s3,s2) in edges, else 0
    
    # Since we're checking if vectors are linearly independent,
    # we can simplify by just checking the determinant of the matrix formed by these vectors
    
    x12 = s1 + s2 if (min(s1,s2), max(s1,s2)) in edges else 0
    x13 = s1 + s3 if (min(s1,s3), max(s1,s3)) in edges else 0
    x23 = s2 + s3 if (min(s2,s3), max(s2,s3)) in edges else 0
    
    # The matrix is:
    # [0,    x12, x13]
    # [x12,  0,   x23]
    # [x13,  x23, 0  ]
    
    # Determinant = 0*(0*0 - x23*x23) - x12*(x12*0 - x13*x23) + x13*(x12*x23 - 0*x13)
    #             = -x12*(-x13*x23) + x13*(x12*x23)
    #             = x12*x13*x23 + x13*x12*x23
    #             = 2*x12*x13*x23
    
    det = 2 * x12 * x13 * x23
    
    return det != 0

def solve():
    n, edges = read_input()
    
    # Try all combinations of 3 steps in lexicographical order
    for s1, s2, s3 in combinations(range(1, n+1), 3):
        if are_linearly_independent(s1, s2, s3, edges):
            print(s1, s2, s3)
            return

if __name__ == "__main__":
    solve()
