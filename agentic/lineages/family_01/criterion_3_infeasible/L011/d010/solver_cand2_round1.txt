import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2

    # Let T = {i, j, k}.
    # Exposure of v in T to A is |{u in A : u ~ v}|.
    # A is balanced if for all v in {i, j, k}, exposure is even.
    # Let x_i, x_j, x_k be indicators for A.
    # For v=i: x_j * (1 if j~i else 0) + x_k * (1 if k~i else 0) = 0 (mod 2)
    # For v=j: x_i * (1 if i~j else 0) + x_k * (1 if k~j else 0) = 0 (mod 2)
    # For v=k: x_i * (1 if i~k else 0) + x_j * (1 if j~k else 0) = 0 (mod 2)
    
    # Let e_ij = 1 if i~j else 0.
    # System:
    # x_j * e_ij + x_k * e_ik = 0 (mod 2)
    # x_i * e_ij + x_k * e_jk = 0 (mod 2)
    # x_i * e_ik + x_j * e_jk = 0 (mod 2)
    
    # A = {i, j, k} is always balanced because:
    # i: e_ij + e_ik = 0 (mod 2)
    # j: e_ij + e_jk = 0 (mod 2)
    # k: e_ik + e_jk = 0 (mod 2)
    # This implies e_ij = e_ik = e_jk (mod 2).
    # So {i, j, k} is balanced iff e_ij = e_ik = e_jk.
    # This means either 0 or 2 edges exist among {i, j, k}.
    
    # We need the ONLY non-empty subset to be {i, j, k}.
    # This means for any proper subset A, it is NOT balanced.
    # Subsets: {i}, {j}, {k}, {i, j}, {i, k}, {j, k}.
    
    for i in range(1, n + 1):
        for j in range(i + 1, n + 1):
            e_ij = 1 if j in adj[i] else 0
            for k in range(j + 1, n + 1):
                e_ik = 1 if k in adj[i] else 0
                e_jk = 1 if k in adj[j] else 0
                
                # Check if {i, j, k} is balanced
                if (e_ij + e_ik) % 2 == 0 and (e_ij + e_jk) % 2 == 0 and (e_ik + e_jk) % 2 == 0:
                    # Check if any proper subset is balanced
                    # {i}: e_ij=0, e_ik=0
                    # {j}: e_ij=0, e_jk=0
                    # {k}: e_ik=0, e_jk=0
                    # {i, j}: i: e_ij=0, j: e_ij=0, k: e_ik+e_jk=0
                    # {i, k}: i: e_ik=0, j: e_ij+e_jk=0, k: e_ik=0
                    # {j, k}: i: e_ij+e_ik=0, j: e_jk=0, k: e_jk=0
                    
                    if not (e_ij == 0 and e_ik == 0): continue
                    if not (e_ij == 0 and e_jk == 0): continue
                    if not (e_ik == 0 and e_jk == 0): continue
                    if not (e_ij == 0 and (e_ik + e_jk) % 2 == 0): continue
                    if not (e_ik == 0 and (e_ij + e_jk) % 2 == 0): continue
                    if not (e_jk == 0 and (e_ij + e_ik) % 2 == 0): continue
                    
                    print(f"{i} {j} {k}")
                    return

solve()
