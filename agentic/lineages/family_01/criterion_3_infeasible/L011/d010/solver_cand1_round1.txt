import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2

    def d(u, v):
        if v in adj[u]:
            return u + v
        return 0

    # The volume of the tetrahedron with vertices (0,0,0), P1, P2, P3 is 
    # 1/6 * |det([[0, d12, d13], [d21, 0, d23], [d31, d32, 0]])|
    # The determinant is:
    # 0*(0 - d23*d32) - d12*(d21*0 - d23*d31) + d13*(d21*d32 - 0*d31)
    # = d12*d23*d31 + d13*d21*d32
    # Since d(u, v) = d(v, u), this is 2 * d(s1, s2) * d(s2, s3) * d(s3, s1)
    # The volume is strictly positive if and only if d(s1, s2) * d(s2, s3) * d(s3, s1) > 0
    # This means all three pairs must have a shared tool.
    
    for s1 in range(1, n + 1):
        for s2 in range(s1 + 1, n + 1):
            if s2 not in adj[s1]:
                continue
            for s3 in range(s2 + 1, n + 1):
                if s3 in adj[s1] and s3 in adj[s2]:
                    print(f"{s1} {s2} {s3}")
                    return

if __name__ == '__main__':
    solve()
