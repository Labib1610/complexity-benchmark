import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, b from a+1 to n, and c from b+1 to n.
    # To optimize, we only check if b is in adj[a] and c is in adj[a] and adj[b].
    
    # Sort adjacency lists to potentially speed up or simplify
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Only consider b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a[i+1:] and neighbors_b
            neighbors_b = adj[b]
            
            # Iterate through neighbors of a that are > b
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c > b and c in neighbors_b:
                    # Found the lexicographically smallest
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
