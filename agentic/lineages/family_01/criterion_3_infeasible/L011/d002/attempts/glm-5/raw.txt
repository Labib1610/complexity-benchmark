```python
import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    for a in range(1, n - 1):
        adj_a = adj[a]
        for b in range(a + 1, n):
            if b in adj_a:
                adj_b = adj[b]
                for c in range(b + 1, n + 1):
                    if c in adj_a and c in adj_b:
                        print(a, b, c)
                        return

solve()
```
