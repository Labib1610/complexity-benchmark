We can solve this by checking each possible triple, but that would be too slow for large n.  
Instead, we can iterate over edges and check for a common neighbor, keeping track of the smallest fingerprint.

```python
import sys

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    best = None
    best_fp = float('inf')

    for u, v in edges:
        # Find common neighbors of u and v
        common = adj[u] & adj[v]
        for w in common:
            a, b, c = sorted((u, v, w))
            fp = a * 10**8 + b * 10**4 + c
            if fp < best_fp:
                best_fp = fp
                best = (a, b, c)

    print(*best)

if __name__ == "__main__":
    main()
```
