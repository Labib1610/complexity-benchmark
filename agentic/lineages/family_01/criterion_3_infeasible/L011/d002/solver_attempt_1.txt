import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To minimize a * 10^8 + b * 10^4 + c, we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # c is in adj[a] and c is in adj[b].
    
    # Sort adjacency lists to potentially speed up intersection
    sorted_adj = [sorted(list(s)) for s in adj]
    
    best_trio = None
    
    for a in range(1, n + 1):
        # Optimization: if a is already larger than the first element of 
        # the best found trio, we can stop.
        if best_trio and a > best_trio[0]:
            break
            
        neighbors_a = adj[a]
        sorted_neighbors_a = sorted_adj[a]
        
        for b in sorted_neighbors_a:
            if b <= a:
                continue
            if best_trio and a == best_trio[0] and b > best_trio[1]:
                continue
            
            # Find common neighbors c > b
            neighbors_b = adj[b]
            # Intersection of neighbors_a and neighbors_b
            # We only care about c > b
            for c in sorted_neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    # Found a trio (a, b, c)
                    if best_trio is None or (a < best_trio[0]) or \
                       (a == best_trio[0] and b < best_trio[1]) or \
                       (a == best_trio[0] and b == best_trio[1] and c < best_trio[2]):
                        best_trio = (a, b, c)
                    # Since we iterate a and b in increasing order, 
                    # the first c we find for a fixed (a, b) is the smallest c.
                    break
                    
    if best_trio:
        print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

if __name__ == '__main__':
    solve()
