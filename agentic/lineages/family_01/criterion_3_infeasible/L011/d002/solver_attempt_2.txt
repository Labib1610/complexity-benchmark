```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To minimize a * 10^8 + b * 10^4 + c, we want the smallest a, then smallest b, then smallest c.
    # We iterate through all edges (u, v) with u < v.
    # For each edge, we look for a common neighbor w.
    # The triplet will be (min(u, v, w), mid(u, v, w), max(u, v, w)).
    
    best_a, best_b, best_c = float('inf'), float('inf'), float('inf')
    
    # Sort edges to process in a way that helps find the minimum fingerprint
    edges.sort()
    
    for u, v in edges:
        # Optimization: if u is already larger than best_a, we can't improve
        if u > best_a:
            break
            
        # Find common neighbors
        # To keep it efficient, iterate over the smaller adjacency list
        if len(adj[u]) < len(adj[v]):
            common = adj[u].intersection(adj[v])
        else:
            common = adj[v].intersection(adj[u])
            
        for w in common:
            # Form the sorted triplet
            nodes = sorted([u, v, w])
            a, b, c = nodes[0], nodes[1], nodes[2]
            
            if a < best_a:
                best_a, best_b, best_c = a, b, c
            elif a == best_a:
                if b < best_b:
                    best_b, best_c = b, c
                elif b == best_b:
                    if c < best_c:
                        best_c = c
                        
    print(f"{best_a} {best_b} {best_c}")

if __name__ == '__main__':
    solve()
```
