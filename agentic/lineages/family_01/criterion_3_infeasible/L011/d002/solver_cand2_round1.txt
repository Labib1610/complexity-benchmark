import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) with a < b < c
    # To find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, b from a+1 to n, and c from b+1 to n.
    # Optimization: only check b in adj[a] and c in (adj[a] intersect adj[b])
    
    # Sort adjacency lists to help with lexicographical order
    sorted_adj = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Only consider b > a
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since neighbors are sorted, we can find common elements efficiently
            set_b = adj[b]
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
