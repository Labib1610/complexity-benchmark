```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A group of 3 steps {a, b, c} is an unresolvable bottleneck if 
    # exactly 4 out of 8 assignments are destructive.
    # Let x, y, z be the phase shifts (0 or 1).
    # Destructive if (x*y*E_ab + x*z*E_ac + y*z*E_bc) is odd, where E_ij is 1 if edge exists.
    # Let k be the number of edges in the triangle.
    # If k=0: 0 is never odd. 0 destructive.
    # If k=1: (x*y) is odd only if x=1, y=1. 2 assignments (1,1,0), (1,1,1). 2 destructive.
    # If k=2: (x*y + x*z) = x(y+z) is odd if x=1 and (y+z) is odd. 
    #         (1,1,0), (1,0,1). 2 destructive.
    # If k=3: (x*y + x*z + y*z) is odd if 2 or 3 variables are 1.
    #         (1,1,0), (1,0,1), (0,1,1), (1,1,1). 4 destructive.
    # Thus, a group is an unresolvable bottleneck iff it forms a triangle (k=3).
    
    # We need the lexicographically smallest triangle.
    # Iterate through all nodes i, then j > i, then k > j.
    # Check if i, j, k form a triangle.
    
    # To optimize, iterate i, then j in adj[i] where j > i, then k in adj[i] intersect adj[j] where k > j.
    
    adj_list = [sorted(list(s)) for s in adj]
    
    for i in range(1, n + 1):
        neighbors_i = adj_list[i]
        for idx_j in range(len(neighbors_i)):
            j = neighbors_i[idx_j]
            if j <= i:
                continue
            
            # Find common neighbors k > j
            # Since adj_list is sorted, we can intersect efficiently
            neighbors_j = adj[j]
            for k in neighbors_i[idx_j + 1:]:
                if k in neighbors_j:
                    print(f"{i} {j} {k}")
                    return

solve()
```
