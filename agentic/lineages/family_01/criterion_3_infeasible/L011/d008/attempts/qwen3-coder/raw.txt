import sys
from itertools import combinations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        index += 2
        edges.append((u, v))
    
    # Build adjacency list
    adj = [set() for _ in range(n)]
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # Function to count destructive assignments for a triangle
    def count_destructive(a, b, c):
        # Get the edges present
        e_ab = b in adj[a]
        e_ac = c in adj[a]
        e_bc = c in adj[b]
        
        count = 0
        # Iterate through all 8 phase assignments
        for mask in range(8):
            phases = [(mask >> i) & 1 for i in range(3)]  # 0 or 1 for a,b,c
            interfering = 0
            if e_ab and phases[0] == 1 and phases[1] == 1:
                interfering += 1
            if e_ac and phases[0] == 1 and phases[2] == 1:
                interfering += 1
            if e_bc and phases[1] == 1 and phases[2] == 1:
                interfering += 1
            if interfering % 2 == 1:
                count += 1
        return count
    
    # Iterate through all combinations of 3 nodes, in sorted order
    for a in range(n):
        for b in range(a+1, n):
            if b not in adj[a]:
                continue
            for c in range(b+1, n):
                if c not in adj[a] or c not in adj[b]:
                    continue
                # Found a triangle, check if it's unresolvable
                if count_destructive(a, b, c) == 4:
                    print(a+1, b+1, c+1)
                    return

solve()
