import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # Sort edges to help find the lexicographically smallest trio
    # We want to find (i, j, k) with i < j < k such that
    # (i, j), (i, k), and (j, k) are all in the conflict set.
    
    # Iterate through all possible i
    for i in range(1, n + 1):
        neighbors = sorted(list(adj[i]))
        # For each pair of neighbors j, k of i, check if (j, k) is an edge
        len_n = len(neighbors)
        for idx1 in range(len_n):
            j = neighbors[idx1]
            for idx2 in range(idx1 + 1, len_n):
                k = neighbors[idx2]
                # Check if j and k are connected
                if k in adj[j]:
                    # Found a trio (i, j, k)
                    # Since we iterate i from 1 to n, and neighbors are sorted,
                    # the first one we find is the lexicographically smallest.
                    res = sorted([i, j, k])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

if __name__ == '__main__':
    solve()
