```python
import sys
from itertools import combinations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list
    adj = [set() for _ in range(n + 1)]
    edges = set()
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        index += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.add((u, v))
    
    # Find all triangles (3-cliques) by checking all combinations of 3 nodes
    min_trace = float('inf')
    result = None
    
    # For each combination of 3 nodes, check if they form a triangle
    for a, b, c in combinations(range(1, n + 1), 3):
        # Check if all three edges exist
        if b in adj[a] and c in adj[a] and c in adj[b]:
            # This is a triangle
            trace = a * 100000000 + b * 10000 + c
            if trace < min_trace:
                min_trace = trace
                result = (a, b, c)
    
    print(result[0], result[1], result[2])

if __name__ == "__main__":
    main()
```
