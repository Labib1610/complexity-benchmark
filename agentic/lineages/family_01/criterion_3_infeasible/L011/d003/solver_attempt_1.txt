import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To minimize a * 10^8 + b * 10^4 + c, we want the smallest a, then smallest b, then smallest c.
    # We iterate through all edges (u, v) with u < v.
    # For each edge, we look for a common neighbor w.
    # The trio will be (min(u, v, w), mid(u, v, w), max(u, v, w)).
    
    min_fingerprint = float('inf')
    best_trio = None
    
    # Sort edges to process smaller IDs first
    edges.sort()
    
    for u, v in edges:
        # Find common neighbors
        # To optimize, iterate over the smaller adjacency list
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        
        # Now u is the one with smaller degree
        for w in adj[u]:
            if w in adj[v]:
                # Found a triangle (u, v, w)
                trio = sorted([u, v, w])
                a, b, c = trio
                fingerprint = a * 100000000 + b * 10000 + c
                if fingerprint < min_fingerprint:
                    min_fingerprint = fingerprint
                    best_trio = (a, b, c)
                    
    print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

if __name__ == '__main__':
    solve()
