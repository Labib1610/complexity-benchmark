import sys

def solve():
    # The problem asks for a group of three steps (a, b, c) such that
    # they form a triangle in the conflict graph.
    # A triangle (a, b, c) means (a,b), (b,c), and (a,c) are all edges.
    # If these three steps are in a graph, any assignment to two lines
    # will result in a conflict because at least two of the three must
    # share a line (Pigeonhole Principle).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check for a common neighbor c.
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # c is in adj[a] and c is in adj[b].
    
    # Sort adjacency lists to help find c efficiently
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    # Iterate through all pairs (a, b)
    # Since we need the smallest (a, b, c), we iterate a, then b, then c.
    for a in range(1, n + 1):
        for b in adj[a]:
            if b <= a:
                continue
            # Now find smallest c > b such that c is in adj[a] and adj[b]
            # Intersection of sorted lists
            common = []
            i, j = 0, 0
            list_a = adj[a]
            list_b = adj[b]
            while i < len(list_a) and j < len(list_b):
                if list_a[i] == list_b[j]:
                    if list_a[i] > b:

                        return
                    i += 1
                    j += 1
                elif list_a[i] < list_b[j]:
                    i += 1

solve()
