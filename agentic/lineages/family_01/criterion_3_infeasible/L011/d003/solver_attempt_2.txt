```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To minimize a * 10^8 + b * 10^4 + c, we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # c is in adj[a] and c is in adj[b].
    # Since we want the minimum trace, we stop as soon as we find the first
    # valid (a, b, c) triplet.
    
    # Optimization: sort adjacency lists to check c efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider b > a
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # Find common neighbors c > b
                # We can intersect the sorted lists or just iterate
                # Since we want the smallest c, we iterate through sorted_adj[a]
                # and check if it's in adj[b]
                for c in sorted_adj[a]:
                    if c > b:
                        if c in adj[b]:
                            print(f"{a} {b} {c}")
                            return

if __name__ == '__main__':
    solve()
```
