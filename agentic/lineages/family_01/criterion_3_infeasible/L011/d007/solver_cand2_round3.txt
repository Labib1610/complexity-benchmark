import sys

def solve():
    # The condition "no matter how each of the three nodes is assigned either a High or Low state, 
    # there will always be two entangled nodes within the group that receive the same state"
    # is equivalent to saying that the three nodes must form a triangle (a clique of size 3).
    # If we have nodes {a, b, c} and they form a triangle, then any assignment of 2 states 
    # to 3 nodes must result in at least two nodes having the same state (Pigeonhole Principle).
    # If those two nodes are connected, the condition is satisfied.
    # In a triangle, all pairs are connected, so any assignment of 2 states to 3 nodes 
    # will always have at least two nodes with the same state, and those two nodes 
    # will be connected.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
        
    # Sort edges to iterate in lexicographical order
    edges.sort()
    
    # To find the lexicographically smallest triplet (a, b, c),
    # we iterate through all possible triplets (a, b, c) with a < b < c.
    # However, n is up to 2000, so O(n^3) is too slow.
    # We can iterate over all edges (a, b) and check for a common neighbor c.
    
    # To ensure we find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, then b from a+1 to n, then c from b+1 to n.
    # A triplet (a, b, c) is a triangle if b in adj[a], c in adj[a], and c in adj[b].
    
    for a in range(1, n + 1):
        adj_a = sorted(list(adj[a]))
        for i in range(len(adj_a)):
            b = adj_a[i]
            if b <= a:
                continue
            # Now we need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

solve()
