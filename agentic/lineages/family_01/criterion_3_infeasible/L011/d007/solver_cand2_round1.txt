import sys

def solve():
    # Read N and M
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    N = int(input_data[0])
    M = int(input_data[1])
    
    adj = [set() for _ in range(N + 1)]
    idx = 2
    for _ in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A trio (a, b, c) is frustrated if for all 8 spin assignments, 
    # the sum of (Su + Sv)^2 > 0.
    # (Su + Sv)^2 is 0 if Su == -Sv, and 4 if Su == Sv.
    # Let E(a,b,c) be the set of edges within {a, b, c}.
    # The sum is 4 * (number of edges (u, v) such that Su == Sv).
    # The sum is 0 if and only if there exists an assignment such that 
    # for all edges (u, v) in the trio, Su != Sv.
    # This is equivalent to saying the subgraph induced by {a, b, c} is bipartite.
    # A triangle (a, b, c) is bipartite if and only if it is not a 3-cycle.
    # Wait, the condition is: for every assignment, sum > 0.
    # This means there is NO assignment where sum == 0.
    # Sum == 0 iff for all edges (u, v) in the trio, Su != Sv.
    # This is possible iff the induced subgraph is bipartite.
    # A trio is frustrated iff the induced subgraph is NOT bipartite.
    # A subgraph of 3 nodes is bipartite iff it contains no odd cycles.
    # The only odd cycle in a 3-node graph is a 3-cycle (triangle).
    # So, a trio is frustrated iff it forms a triangle (a, b, c) where 
    # all three edges (a,b), (b,c), (a,c) exist.
    
    # Wait, let's re-check:
    # If the trio has 0 edges, sum is always 0. Not frustrated.
    # If 1 edge (a,b), sum is 4 if Sa=Sb, 0 if Sa!=Sb. Can be 0. Not frustrated.
    # If 2 edges (a,b), (b,c), sum is 4*( (Sa+Sb)^2/4 + (Sb+Sc)^2/4 ).
    # If Sa=1, Sb=-1, Sc=1, sum = 4*(0+0) = 0. Can be 0. Not frustrated.
    # If 3 edges (a,b), (b,c), (a,c), sum is 4*( (Sa+Sb)^2/4 + (Sb+Sc)^2/4 + (Sa+Sc)^2/4 ).
    # This is 0 iff Sa=-Sb, Sb=-Sc, Sa=-Sc.
    # Sa=-Sb and Sb=-Sc implies Sa=Sc. But Sa=-Sc implies Sa=-Sa, impossible.
    # So a triangle is always frustrated.
    
    # The problem asks for the sequentially earliest trio.
    # We iterate a from 1 to N, b from a+1 to N, c from b+1 to N.
    # Check if (a,b), (b,c), (a,c) are all in edges.
    
    # Optimization: iterate a, then b in adj[a], then c in adj[a] & adj[b].
    # To ensure sequentially earliest, we need to be careful.
    
    for a in range(1, N + 1):
        adj_a = adj[a]
        sorted_adj_a = sorted([x for x in adj_a if x > a])
        for i in range(len(sorted_adj_a)):
            b = sorted_adj_a[i]
            # We need c > b such that c is in adj[a] and adj[b]
            common = sorted(list(adj_a.intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
