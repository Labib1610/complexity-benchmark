```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A group of 3 steps {a, b, c} has 3 edges (a,b), (b,c), (a,c)
    # Let k be the number of edges among the 3 steps.
    # If k=0: tension is always 0. Relaxed states = 0.
    # If k=1: tension = |x_i - x_j|. Relaxed if |x_i - x_j| = 2.
    #         Pairs (x_i, x_j) are (1,3) or (3,1). 3 choices for the 3rd node.
    #         Total relaxed = 2 * 3 = 6.
    # If k=2: tension = |x_i - x_j| + |x_j - x_k|. Relaxed if sum = 2.
    #         Possible (x_i, x_j, x_k): (1,2,3), (3,2,1), (2,1,1), (2,3,3), (1,1,2), (3,3,2).
    #         Total relaxed = 6.
    # If k=3: tension = |x_i - x_j| + |x_j - x_k| + |x_k - x_i|. Relaxed if sum = 2.
    #         Possible (x_i, x_j, x_k) permutations:
    #         (1,2,3) -> |1-2|+|2-3|+|3-1| = 1+1+2 = 4
    #         (1,1,2) -> |1-1|+|1-2|+|2-1| = 0+1+1 = 2 (Relaxed)
    #         (1,2,2) -> |1-2|+|2-2|+|2-1| = 1+0+1 = 2 (Relaxed)
    #         For a triangle, there are 3! = 6 permutations of (1,2,3) -> tension 4.
    #         There are 3 choices for the node with value 2, and 2 choices for the others (1,1) -> 3 * 2 = 6.
    #         There are 3 choices for the node with value 1, and 2 choices for the others (2,2) -> 3 * 2 = 6.
    #         Total relaxed = 6 + 6 = 12.
    # Thus, the condition "exactly 12 relaxed states" is equivalent to the 3 steps forming a triangle.

    for a in range(1, n + 1):
        adj_a = adj[a]
        sorted_adj_a = sorted(list(adj_a))
        len_a = len(sorted_adj_a)
        for i in range(len_a):
            b = sorted_adj_a[i]
            if b <= a: continue
            adj_b = adj[b]
            # Intersection of adj[a] and adj[b] gives nodes c that form a triangle
            for c in sorted_adj_a[i+1:]:
                if c in adj_b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
