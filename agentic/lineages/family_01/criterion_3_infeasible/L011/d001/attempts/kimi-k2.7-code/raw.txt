import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        if u < v:
            edges.append((u, v))
        else:
            edges.append((v, u))
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    edges.sort()
    # suffix[v] has bits for vertices > v set
    suffix = [0] * (n + 1)
    all_bits = (1 << (n + 1)) - 1
    for v in range(1, n + 1):
        suffix[v] = all_bits ^ ((1 << (v + 1)) - 1)

    best_sig = 10**18
    best_a = n + 1
    best_b = n + 1
    best_triplet = (0, 0, 0)

    for u, v in edges:
        if u > best_a:
            break
        if u == best_a and v > best_b:
            break
        common = adj[u] & adj[v]
        if common:
            cand = common & suffix[v]
            if cand:
                low = cand & -cand
                c = low.bit_length() - 1
                sig = u * 100000000 + v * 10000 + c
                if sig < best_sig:
                    best_sig = sig
                    best_a = u
                    best_b = v
                    best_triplet = (u, v, c)

    sys.stdout.write(f"{best_triplet[0]} {best_triplet[1]} {best_triplet[2]}")

solve()
