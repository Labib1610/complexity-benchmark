We can solve this by checking all triples, but with n up to 2000, that’s too slow.  
Instead, we can use adjacency sets and iterate over edges, checking for a common neighbor.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    best = None
    best_key = None

    for u, v in edges:
        common = adj[u] & adj[v]
        for w in common:
            a, b, c = sorted((u, v, w))
            key = a * 10**8 + b * 10**4 + c
            if best_key is None or key < best_key:
                best_key = key
                best = (a, b, c)

    print(*best)

if __name__ == "__main__":
    solve()
```
