import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c
    # We iterate through a from 1 to n-2
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # We need b and c such that b, c are in neighbors_a and b is in adj[c]
        # To ensure a < b < c, we only consider neighbors > a
        valid_neighbors = [x for x in neighbors_a if x > a]
        
        # Sort valid neighbors to ensure we find the smallest b, then c
        valid_neighbors.sort()
        
        for i in range(len(valid_neighbors)):
            b = valid_neighbors[i]
            # Check if there exists c > b such that c is in neighbors_a and c is in adj[b]
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            
            # Filter common neighbors that are > b
            candidates_c = [x for x in common if x > b]
            if candidates_c:
                c = min(candidates_c)
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
