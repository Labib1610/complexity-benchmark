import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n
    # Iterate b in sorted(adj[a]) where b > a
    # Check if there exists c in adj[a] intersect adj[b] where c > b
    
    # Sort adjacency lists to ensure we find the smallest lexicographical trio
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Optimization: convert to set for O(1) lookup
        set_a = adj[a]
        
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need to find the smallest c > b such that c is in adj[a] and adj[b]
            # Since we want the smallest (a, b, c), we iterate b in increasing order
            # and check the intersection.
            set_b = adj[b]
            
            # Intersection of neighbors of a and b
            # To keep it efficient, iterate through the smaller adjacency list
            if len(neighbors_a) < len(sorted_adj[b]):
                for c in neighbors_a:
                    if c > b and c in set_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in sorted_adj[b]:
                    if c > b and c in set_a:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
