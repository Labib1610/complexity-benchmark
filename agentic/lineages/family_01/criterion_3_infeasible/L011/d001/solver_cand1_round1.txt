```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) with a < b < c.
    # To find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, b from a+1 to n, and c from b+1 to n.
    # However, checking all triples is O(n^3), which is too slow.
    # Instead, iterate over a, then iterate over neighbors b of a,
    # then check if any neighbor c of b is also a neighbor of a.
    
    # Sort neighbors to ensure we find the lexicographically smallest triple
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need to find c such that b < c and c is in adj[a] and adj[b]
            # Since adj[a] and adj[b] are sorted, we can use intersection
            # or just iterate through the smaller list.
            # To keep it efficient, use a set for O(1) lookup.
            
            # Optimization: only check c > b
            # We want the smallest c, so we iterate through sorted neighbors of b
            for c in adj[b]:
                if c > b:
                    if c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

solve()
```
