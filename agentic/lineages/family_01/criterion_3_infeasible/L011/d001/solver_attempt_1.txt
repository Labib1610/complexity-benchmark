```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To minimize a * 10^8 + b * 10^4 + c, we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # c is in adj[a] and c is in adj[b].
    # To optimize, we convert adj sets to sorted lists or use bitsets if n was smaller.
    # Given n=2000, O(n^3) is too slow, but O(n * m) or O(n^2 + m * sqrt(m)) is fine.
    
    # Sort adjacency lists to make intersection faster
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    best_a, best_b, best_c = float('inf'), float('inf'), float('inf')
    
    for a in range(1, n + 1):
        if a >= best_a:
            break
        for b in adj_list[a]:
            if b <= a:
                continue
            # We need c > b such that c in adj[a] and c in adj[b]
            # Since adj_list is sorted, we can intersect
            # But we only need the smallest c > b
            # Optimization: iterate through the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                for c in adj_list[a]:
                    if c > b and c in adj[b]:
                        if a < best_a or (a == best_a and b < best_b) or (a == best_a and b == best_b and c < best_c):
                            best_a, best_b, best_c = a, b, c
                        break
            else:
                for c in adj_list[b]:
                    if c > b and c in adj[a]:
                        if a < best_a or (a == best_a and b < best_b) or (a == best_a and b == best_b and c < best_c):
                            best_a, best_b, best_c = a, b, c
                        break
                        
    print(f"{best_a} {best_b} {best_c}")

solve()
```
