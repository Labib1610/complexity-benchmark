import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # The problem asks for a group of three steps {a, b, c} such that
    # no matter which one is removed, the remaining two have a collision.
    # This means all three pairs (a, b), (b, c), and (a, c) must exist
    # in the input list of collisions.
    # This is equivalent to finding a triangle (clique of size 3) in the graph.
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, iterate c from b+1 to n
    # Check if c is in adj[a] and c is in adj[b]
    
    for a in range(1, n + 1):
        # Sort neighbors to ensure we check in order
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we want the smallest c, we can iterate through sorted neighbors
            neighbors_b = adj[b]
            for c in neighbors_a[i+1:]:
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
