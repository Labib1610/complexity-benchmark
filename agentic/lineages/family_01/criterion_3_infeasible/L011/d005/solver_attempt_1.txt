import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) with a < b < c
    # To minimize a * 10^8 + b * 10^4 + c, we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # c is in adj[a] and c is in adj[b].
    
    # To optimize, for a fixed a and b, we need to check if there is any c > b
    # such that c is in adj[a] and c is in adj[b].
    # This is equivalent to checking if the intersection of adj[a] and adj[b]
    # contains any element > b.
    
    best_a, best_b, best_c = float('inf'), float('inf'), float('inf')
    
    # Sort adjacency lists to make intersection faster and ordered
    adj_list = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        # Optimization: if a is already worse than best_a, skip
        if a > best_a:
            break
            
        for b in adj[a]:
            if b <= a:
                continue
            
            # We need to find c > b such that c is in adj[a] and adj[b]
            # Intersection of adj[a] and adj[b]
            # Since we want the smallest c, we can iterate through the smaller set
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
                other = adj[b]
            else:
                candidates = adj[b]
                other = adj[a]
            
            for c in candidates:
                if c > b and c in other:
                    # Found a triangle (a, b, c)
                    # Ensure a < b < c
                    nodes = sorted([a, b, c])
                    curr_a, curr_b, curr_c = nodes
                    if (curr_a < best_a) or (curr_a == best_a and curr_b < best_b) or (curr_a == best_a and curr_b == best_b and curr_c < best_c):
                        best_a, best_b, best_c = curr_a, curr_b, curr_c
    
    print(f"{best_a} {best_b} {best_c}")

if __name__ == '__main__':
    solve()
