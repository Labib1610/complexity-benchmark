import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    d = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        d[u] += 1
        d[v] += 1
        idx += 2
        
    # The condition: number of edges with exactly one endpoint in {a, b, c}
    # equals (D_a + D_b + D_c - 6).
    # Let E(S) be the number of edges with at least one endpoint in S.
    # Let E_in(S) be the number of edges with both endpoints in S.
    # The number of edges with exactly one endpoint in S is:
    # (D_a + D_b + D_c) - 2 * (edges within S).
    # The condition given is: (D_a + D_b + D_c) - 2 * E_in(S) = D_a + D_b + D_c - 6
    # This simplifies to: 2 * E_in(S) = 6, or E_in(S) = 3.
    # E_in(S) = 3 means the subgraph induced by {a, b, c} has 3 edges.
    # This means {a, b, c} must form a triangle (a clique of size 3).
    
    # We need to find the lexicographically smallest triangle.
    # Iterate through all possible a, b, c such that 1 <= a < b < c <= n.
    # To optimize, iterate over a, then b in adj[a], then c in adj[a] intersect adj[b].
    
    triangles = []
    for a in range(1, n + 1):
        adj_a = adj[a]
        sorted_adj_a = sorted(list(adj_a))
        for i in range(len(sorted_adj_a)):
            b = sorted_adj_a[i]
            if b <= a:
                continue
            # Find common neighbors c > b
            common = adj_a.intersection(adj[b])
            for c in common:
                if c > b:
                    triangles.append((a, b, c))
    
    if triangles:
        triangles.sort()
        a, b, c = triangles[0]
        print(f"{a} {b} {c}")

solve()
