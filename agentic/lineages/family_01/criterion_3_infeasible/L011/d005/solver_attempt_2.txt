import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the minimum sequence trace p1*10^8 + p2*10^4 + p3,
    # we need to iterate through p1, then p2, then p3 in increasing order.
    # Sorting edges by (u, v) helps.
    edges.sort()
    
    best_p1, best_p2, best_p3 = float('inf'), float('inf'), float('inf')
    
    # Iterate through each edge (u, v) as the first two nodes of the triangle
    for u, v in edges:
        # We need a node w such that (u, w) and (v, w) exist.
        # To minimize p1*10^8 + p2*10^4 + p3, we fix p1=u, p2=v,
        # and look for the smallest w > v.
        # Also consider cases where w < u or u < w < v.
        
        # Check common neighbors
        # Optimization: intersect smaller adjacency list with larger
        if len(adj[u]) < len(adj[v]):
            common = adj[u].intersection(adj[v])
        else:
            common = adj[v].intersection(adj[u])
            
        for w in common:
            # Form the triplet and sort it
            triplet = sorted([u, v, w])
            p1, p2, p3 = triplet
            
            if p1 < best_p1:
                best_p1, best_p2, best_p3 = p1, p2, p3
            elif p1 == best_p1:
                if p2 < best_p2:
                    best_p1, best_p2, best_p3 = p1, p2, p3
                elif p2 == best_p2:
                    if p3 < best_p3:
                        best_p1, best_p2, best_p3 = p1, p2, p3
                        
    print(f"{best_p1} {best_p2} {best_p3}")

if __name__ == '__main__':
    solve()
