import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [set() for _ in range(n + 1)]
    edges = set()
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.add((u, v))
    
    # Case 1: Check for a triangle (3-clique)
    # A triangle (u,v,w) has vectors (0,1,1), (1,0,1), (1,1,0) which are linearly independent.
    # We iterate u < v < w to find the lexicographically smallest triangle.
    for u in range(1, n - 1):
        # To check if u and v are connected, we iterate v in adj[u]
        # We need to sort neighbors to ensure we check in increasing order for correctness,
        # though the problem asks for smallest u, then v, then w.
        # Since we iterate u from small to large, and v from adj[u] (which are > u),
        # we just need to ensure we check v in increasing order.
        
        neighbors_u = sorted(adj[u])
        for v in neighbors_u:
            if v <= u: 
                continue
            # Check for common neighbor w such that w > v
            # Iterate through neighbors of u again to find potential w
            for w in neighbors_u:
                if w <= v:
                    continue
                if w in adj[v]:
                    print(u, v, w)
                    return

    # Case 2: Check for an independent set of size 3 (no edges between them)
    # Vectors are (0,0,0), (0,0,0), (0,0,0) which are linearly dependent (rank 0).
    # This case does NOT form a bottleneck.
    
    # Case 3: Check for a path of length 2 (u-v-w) where u and w are not connected
    # Vectors: u=(0,1,0), v=(1,0,1), w=(0,1,0). 
    # These are linearly dependent (u and w are identical).
    # This case does NOT form a bottleneck.
    
    # Case 4: Check for a 'V' shape (u-v, u-w) where v and w are not connected, 
    # and neither u-v nor u-w is part of a triangle.
    # Vectors: u=(0,1,1), v=(1,0,0), w=(1,0,0).
    # These are linearly dependent (v and w are identical).
    # This case does NOT form a bottleneck.
    
    # Case 5: Check for a single edge (u-v) and an isolated node w
    # Vectors: u=(0,1,0), v=(1,0,0), w=(0,0,0).
    # Rank is 2. Linearly dependent.
    # This case does NOT form a bottleneck.

    # Case 6: The only remaining structure for 3 nodes is a single edge (u-v) and a node w
    # connected to exactly one of them (say v), but not the other (u), and no triangle.
    # Structure: u-v-w (path) is covered in Case 3.
    # Structure: u-v, v-w, u not connected to w. This is a path.
    # Structure: u-v, u-w, v not connected to w. This is Case 4.
    
    # Wait, let's re-evaluate linear independence conditions.
    # Let the adjacency matrix subgraph for {s1, s2, s3} be A.
    # Vectors v1, v2, v3 are the rows of A (with 0 diagonal).
    # We want det(A) != 0.
    # det(A) = 2 * (number of triangles) - (sum of degrees in the subgraph).
    # Let k be number of edges in the subgraph induced by {s1, s2, s3}.
    # Let t be number of triangles (0 or 1).
    # det(A) = 2*t - k.
    # For det != 0, we need 2*t != k.
    # Possible k values: 0, 1, 2, 3.
    # k=0 (independent set): det = 0. Dependent.
    # k=1 (single edge): det = -1 != 0. INDEPENDENT. BOTTLENECK.
    # k=2 (path or V-shape): det = -2 != 0. INDEPENDENT. BOTTLENECK.
    # k=3 (triangle): det = 2 - 3 = -1 != 0. INDEPENDENT. BOTTLENECK.
    
    # So we are looking for any set of 3 nodes that is NOT an independent set.
    # Basically, any set of 3 nodes with at least one edge.
    # We already checked triangles (k=3).
    # Now we need to check for k=1 or k=2.
    
    # We iterate u < v < w.
    # We want the smallest u.
    # If we find a pair (u, v) connected, we can pick any w.
    # To get smallest u, then smallest v, then smallest w:
    # We iterate u from 1 to n.
    # For each u, iterate v from u+1 to n.
    # If (u, v) is an edge, we pick the smallest w > v such that {u, v, w} is not a triangle.
    # Since we already checked all triangles in Case 1, any w found here will form a bottleneck.
    # The bottleneck will have k=1 (if w has no edges to u or v) or k=2 (if w connects to exactly one of u, v).
    
    # To optimize, we don't iterate all w. We just pick the smallest valid w.
    # Valid w: w > v.
    # We need to check if {u, v, w} is a triangle. If not, it's a bottleneck.
    # Since we check u, v in increasing order, the first valid w we find gives the answer.
    
    for u in range(1, n - 1):
        for v in range(u + 1, n):
            if (u, v) in edges:
                # We have an edge u-v.
                # We need a third node w > v.
                # We want smallest w.
                # w is valid if {u, v, w} is not a triangle.
                # {u, v, w} is a triangle iff w is connected to both u and v.
                # So we need smallest w > v such that NOT (w in adj[u] AND w in adj[v]).
                
                # Let's find the smallest w > v.
                # Ideally w = v + 1.
                # We check if w is connected to both u and v.
                
                # Optimization: 
                # We can iterate w from v+1 to n.
                # Since we want the smallest, we stop at the first valid one.
                
                # However, iterating w from v+1 to n is O(N) per edge, total O(NM) which is too slow.
                # We need a faster way to find the smallest w > v that is NOT a common neighbor of u and v.
                
                # Let common = adj[u] & adj[v]. These are common neighbors.
                # Any w > v that is NOT in common is a valid bottleneck.
                
                # We can iterate through sorted common neighbors to skip them.
                # Or better, iterate w from v+1 and check presence in common set.
                # To make this efficient, we only check a few candidates for w.
                # Since we want the smallest w, we check v+1, v+2, ...
                # If v+1 is not a common neighbor, we are done.
                # Is it possible that v+1, v+2, ... are all common neighbors?
                # Common neighbors of u and v are part of a clique.
                # Max clique size is limited by graph density.
                # In sparse graph, common neighbors are few.
                # In dense graph, adj[u] is large, but checking membership is O(1).
                # The number of checks is size of common neighbor set + 1.
                # Sum of sizes of common neighbor sets can be large O(M sqrt M) or O(M^1.5).
                # With N=2000, M=2e5, this might be acceptable.
                
                common_neighbors = adj[u] & adj[v]
                
                w = v + 1
                while w <= n:
                    if w not in common_neighbors:
                        print(u, v, w)
                        return
                    w += 1

if __name__ == '__main__':
    solve()
