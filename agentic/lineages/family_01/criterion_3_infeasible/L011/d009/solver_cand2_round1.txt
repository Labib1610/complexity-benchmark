import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # For a group of 3 steps {i, j, k}, let x_i, x_j, x_k be polarities.
    # Let e_ij = 1 if (i, j) share a tool, 0 otherwise.
    # T = e_ij * x_i * x_j + e_jk * x_j * x_k + e_ki * x_k * x_i
    # We want sum_{x in {-1, 1}^3} T^3 = 48.
    # T^3 = (e_ij x_i x_j + e_jk x_j x_k + e_ki x_k x_i)^3
    # Expanding this, the sum over all 8 assignments is non-zero only for terms
    # where each variable x_i, x_j, x_k appears an even number of times.
    # The only terms that survive are 6 * e_ij * e_jk * e_ki.
    # Sum = 8 * (6 * e_ij * e_jk * e_ki) = 48 * e_ij * e_jk * e_ki.
    # Thus, the condition is e_ij * e_jk * e_ki = 1, which means {i, j, k}
    # must form a triangle in the graph.
    
    # Sort nodes by degree to optimize triangle finding
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]))
    
    # To find the lexicographically smallest triangle:
    # Iterate through all possible i < j < k
    for i in range(1, n + 1):
        adj_i = adj[i]
        sorted_adj_i = sorted([x for x in adj_i if x > i])
        for idx_j, j in enumerate(sorted_adj_i):
            adj_j = adj[j]
            # Intersection of adj[i] and adj[j] gives k
            # We need k > j
            for k in sorted_adj_i[idx_j + 1:]:
                if k in adj_j:
                    print(f"{i} {j} {k}")
                    return

solve()
