```python
import sys

def solve():
    # A group of 3 steps {a, b, c} is an unresolvable bottleneck if every 
    # assignment of these 3 steps to 2 lines results in a collision.
    # With 3 steps, there are 2^3 = 8 assignments.
    # A collision occurs if a pair (u, v) is assigned to the same line.
    # For a group of 3 to be an unresolvable bottleneck, every assignment 
    # must have at least one pair (u, v) in the group assigned to the same line.
    # This happens if and only if the group forms a triangle (a,b), (b,c), (a,c)
    # in the collision graph.
    # Proof: If {a, b, c} is a triangle, any assignment of 3 items into 2 bins
    # must put at least two items in the same bin (Pigeonhole Principle).
    # Since all pairs are connected, those two items will collide.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all edges (u, v) and check common neighbors.
    
    # Sort edges to ensure we check in lexicographical order
    edges.sort()
    
    # To optimize, iterate through all nodes as 'a'
    for a in range(1, n + 1):
        neighbors = sorted(list(adj[a]))
        len_n = len(neighbors)
        for i in range(len_n):
            b = neighbors[i]
            if b <= a:
                continue
            # Check common neighbors c > b
            for j in range(i + 1, len_n):
                c = neighbors[j]
                if c in adj[b]:
                    # Found triangle (a, b, c)
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

solve()
```
