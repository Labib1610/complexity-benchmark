```python
import sys

def solve():
    # A subset of 3 steps {a, b, c} is an unresolvable bottleneck if all 8 
    # assignments to Line 1 or Line 2 result in a collision.
    # A collision occurs if two steps in the same line have an edge.
    # For a set {a, b, c}, the possible assignments are:
    # (L1, L1, L1), (L1, L1, L2), (L1, L2, L1), (L1, L2, L2),
    # (L2, L1, L1), (L2, L1, L2), (L2, L2, L1), (L2, L2, L2)
    #
    # If {a, b, c} forms a triangle (K3), then (L1, L1, L1) and (L2, L2, L2)
    # are invalid. To make all 8 invalid, we need more edges.
    # Specifically, for any partition of {a, b, c} into two sets (Line 1, Line 2),
    # at least one set must contain an edge.
    # This happens if and only if the graph induced by {a, b, c} contains a K3.
    # Wait, let's re-check:
    # If {a, b, c} is a K3, then (L1, L1, L1) is invalid (all edges exist).
    # (L1, L1, L2) is invalid because (a, b) is an edge.
    # (L1, L2, L1) is invalid because (a, c) is an edge.
    # (L2, L1, L1) is invalid because (b, c) is an edge.
    # (L1, L2, L2) is invalid because (b, c) is an edge.
    # (L2, L1, L2) is invalid because (a, c) is an edge.
    # (L2, L2, L1) is invalid because (a, b) is an edge.
    # (L2, L2, L2) is invalid because all edges exist.
    # So any K3 is an unresolvable bottleneck.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append(tuple(sorted((u, v))))
        idx += 2
        
    # We need the lexicographically smallest {a, b, c} that is a K3.
    # Iterate through all edges (u, v) and check for a common neighbor w.
    # To ensure lexicographical order, we can iterate through all possible
    # triplets (a, b, c) with a < b < c and check if they form a K3.
    # Since n <= 2000, O(n^3) is too slow, but O(m * n) is 2*10^5 * 2000 = 4*10^8.
    # This might pass within 2 seconds in Python if optimized.
    
    # Sort edges to help find smallest triplets
    edges.sort()
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # Check for common neighbor c > b
                # Intersection of adj[a] and adj[b]
                common = adj[a].intersection(adj[b])
                # Filter for c > b
                valid_c = [c for c in common if c > b]
                if valid_c:
                    print(f"{a} {b} {min(valid_c)}")
                    return

solve()
```
