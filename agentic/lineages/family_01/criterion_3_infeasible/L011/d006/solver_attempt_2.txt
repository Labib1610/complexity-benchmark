import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the minimum sequence trace (a, b, c) with a < b < c,
    # we iterate through a from 1 to n-2, b from a+1 to n-1,
    # and check if there exists c > b such that c is in adj[a] and adj[b].
    
    # Sort adjacency lists to potentially speed up intersection
    adj_list = [sorted(list(s)) for s in adj]
    
    min_trace = float('inf')
    result = None
    
    # Optimization: iterate through a, then b, then check intersection
    # Since we need the minimum trace, we want the smallest a, then smallest b, then smallest c.
    for a in range(1, n + 1):
        # Optimization: if a*10^8 is already > min_trace, we can stop
        if a * 100000000 > min_trace:
            break
            
        neighbors_a = adj[a]
        for b in range(a + 1, n + 1):
            if b not in neighbors_a:
                continue
            
            # Check if there exists c > b such that c is in adj[a] and adj[b]
            # We want the smallest such c
            neighbors_b = adj[b]
            
            # Intersection of neighbors_a and neighbors_b
            # We only care about c > b
            common = neighbors_a.intersection(neighbors_b)
            for c in common:
                if c > b:
                    trace = a * 100000000 + b * 10000 + c
                    if trace < min_trace:
                        min_trace = trace
                        result = (a, b, c)
    
    if result:
        print(f"{result[0]} {result[1]} {result[2]}")

if __name__ == '__main__':
    solve()
