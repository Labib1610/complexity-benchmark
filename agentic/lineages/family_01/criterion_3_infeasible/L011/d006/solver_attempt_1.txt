import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A structural deadlock is a triangle (clique of size 3).
    # We need to find all triangles (u, v, w) with u < v < w.
    # The sequence trace is x*10^8 + y*10^4 + z where x < y < z.
    # This is equivalent to minimizing x, then y, then z.
    
    best_triple = None
    
    # To find triangles efficiently:
    # For each vertex u, iterate over neighbors v > u.
    # Then check if there is a common neighbor w > v.
    
    # Sort neighbors to optimize intersection
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for u in range(1, n + 1):
        neighbors_u = [v for v in adj_list[u] if v > u]
        for i in range(len(neighbors_u)):
            v = neighbors_u[i]
            # Intersection of adj[u] and adj[v] for w > v
            # Since adj[u] and adj[v] are sorted, we can use two pointers
            # or set intersection.
            
            # Optimization: only check if a triangle exists
            # We want the smallest x, then y, then z.
            # Since we iterate u from 1 to n, and v from u+1 to n,
            # the first triangle we find with the smallest u, then v,
            # will be the one with the smallest x, then y.
            # We just need to find the smallest w for that (u, v).
            
            # Using set for O(1) lookup
            for w in neighbors_u[i+1:]:
                if w in adj[v]:
                    # Found a triangle (u, v, w)
                    # Since we iterate u, v, w in increasing order,
                    # the first one we find is the lexicographically smallest.
                    print(f"{u} {v} {w}")
                    return

solve()
