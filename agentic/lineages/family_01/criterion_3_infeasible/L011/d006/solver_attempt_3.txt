import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To minimize a * 10^8 + b * 10^4 + c, we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # (a, c) and (b, c) are edges.
    
    # Sort edges to help with finding c efficiently
    # For a fixed (a, b), we need to find the smallest c > b such that
    # c is in adj[a] and c is in adj[b].
    
    # Pre-sort adjacency lists to make intersection faster
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # To find the minimum trace, we iterate a, then b, then find the smallest c
    # Since we want the smallest a, then smallest b, then smallest c:
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # Find smallest c > b such that c in adj[a] and c in adj[b]
                # Since adj lists are sorted, we can use intersection
                set_a = adj[a]
                set_b = adj[b]
                
                # We only care about c > b
                # Use two pointers or set intersection
                ptr_a = 0
                ptr_b = 0
                len_a = len(set_a)
                len_b = len(set_b)
                
                while ptr_a < len_a and ptr_b < len_b:
                    val_a = set_a[ptr_a]
                    val_b = set_b[ptr_b]
                    
                    if val_a == val_b:
                        if val_a > b:
                            print(f"{a} {b} {val_a}")
                            return
                        ptr_a += 1
                        ptr_b += 1
                    elif val_a < val_b:
                        ptr_a += 1
                    else:
                        ptr_b += 1

solve()
