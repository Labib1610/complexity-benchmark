```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    p = 2
    for _ in range(m):
        u = int(data[p])
        v = int(data[p + 1])
        p += 2
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    out = sys.stdout
    for u in range(1, n + 1):
        bits = adj[u] >> u
        while bits:
            lsb = bits & -bits
            v = u + lsb.bit_length()
            high = (adj[u] & adj[v]) >> v
            if high:
                lsb2 = high & -high
                w = v + lsb2.bit_length()
                out.write(f"{u} {v} {w}\n")
                return
            bits ^= lsb

if __name__ == "__main__":
    solve()
```
