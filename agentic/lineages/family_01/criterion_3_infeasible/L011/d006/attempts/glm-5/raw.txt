import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    
    adj = [0] * (n + 1)
    for i in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    best = None
    
    for a in range(1, n + 1):
        adj_a = adj[a]
        if adj_a == 0:
            continue
        b = a + 1
        while b <= n:
            if adj_a & (1 << b):
                common = adj_a & adj[b]
                while common:
                    c_bit = common & (-common)
                    c = c_bit.bit_length() - 1
                    if c > b:
                        if best is None or a < best[0] or (a == best[0] and b < best[1]):
                            best = (a, b, c)
                    common ^= c_bit
            b += 1
    
    print(f"{best[0]} {best[1]} {best[2]}")

solve()
