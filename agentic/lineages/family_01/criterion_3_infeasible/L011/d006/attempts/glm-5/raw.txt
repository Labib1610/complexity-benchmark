import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Find lexicographically smallest unresolvable bottleneck
    for a in range(1, n - 1):
        for b in range(a + 1, n):
            if b not in adj[a]:
                continue
            for c in range(b + 1, n + 1):
                if c not in adj[a] or c not in adj[b]:
                    continue
                # Check if {a, b, c} is an unresolvable bottleneck
                # Need all 8 assignments to have collision within {a,b,c}
                # Edges within: (a,b), (a,c), (b,c) - all present
                # Check edges to outside: for each of a,b,c, check if there's
                # a vertex outside that is NOT connected to it
                has_non_neighbor_a = False
                has_non_neighbor_b = False
                has_non_neighbor_c = False
                
                # Check for a
                if len(adj[a]) < n - 1:
                    has_non_neighbor_a = True
                # Check for b
                if len(adj[b]) < n - 1:
                    has_non_neighbor_b = True
                # Check for c
                if len(adj[c]) < n - 1:
                    has_non_neighbor_c = True
                
                # If any vertex has no non-neighbor outside, that vertex
                # can be isolated (assigned to one line while all others
                # go to the other line), avoiding collision within {a,b,c}
                if not has_non_neighbor_a or not has_non_neighbor_b or not has_non_neighbor_c:
                    continue
                
                # Find actual non-neighbors outside
                non_neighbor_a = []
                non_neighbor_b = []
                non_neighbor_c = []
                
                for v in range(1, n + 1):
                    if v != a and v != b and v != c:
                        if v not in adj[a]:
                            non_neighbor_a.append(v)
                        if v not in adj[b]:
                            non_neighbor_b.append(v)
                        if v not in adj[c]:
                            non_neighbor_c.append(v)
                
                # Check if there's a valid assignment
                # Try to find v_a, v_b, v_c such that:
                # v_a not connected to a, v_b not connected to b, v_c not connected to c
                # and v_a, v_b, v_c can be assigned to create valid config
                
                # Assignment: a,b,c on line 0, v_a,v_b,v_c on line 1
                # Need: a-v_a, b-v_b, c-v_c all non-edges (satisfied by choice)
                # Need: v_a, v_b, v_c all on same line (line 1)
                # Need: no edge between any pair of v_a, v_b, v_c
                
                # Try all combinations
                found_valid = False
                for v_a in non_neighbor_a:
                    for v_b in non_neighbor_b:
                        if v_b in adj[v_a]:
                            continue
                        for v_c in non_neighbor_c:
                            if v_c in adj[v_a] or v_c in adj[v_b]:
                                continue
                            found_valid = True
                            break
                        if found_valid:
                            break
                    if found_valid:
                        break
                
                if not found_valid:
                    print(a, b, c)
                    return
    
    # Fallback: shouldn't reach here per problem statement
    print(-1, -1, -1)

if __name__ == "__main__":
    solve()
