```python
import sys

def solve():
    # A set of three chemicals cannot be divided into two planes if and only if
    # they form a triangle (a clique of size 3).
    # If they form a triangle, any assignment of 3 items into 2 sets
    # (by Pigeonhole Principle) forces at least two items to be in the same set.
    # Since all pairs in a triangle are reactive, those two will react.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, we can optimize the search.
    
    # Sort edges to help find smallest a, b, c
    edges.sort()
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through all edges (a, b) and check for a common neighbor c
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b in adj[a], then check if any c in adj[a] is also in adj[b].
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and adj[b]
            # Since adj lists are sorted, we can use two pointers or set intersection
            set_b = set(adj[b])
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
