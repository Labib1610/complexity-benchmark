import sys

def solve():
    # A critical triplet of 3 substances that cannot be partitioned into 2 safe groups
    # means that the subgraph induced by these 3 substances must contain an odd cycle.
    # The only odd cycle of length 3 is a triangle (a clique of size 3).
    # If 3 substances form a triangle, any partition into 2 groups will force
    # at least two substances to be in the same group, and since they are all
    # connected, that group will contain an incompatible pair.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triplet (a, b, c),
    # we iterate through all possible a, b, c.
    # Since n <= 2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    # A triangle (a, b, c) exists if a-b, b-c, and a-c are edges.
    
    # Sort edges to help find triangles efficiently
    edges.sort()
    
    # For each edge (u, v), check if there exists a w such that
    # (u, w) and (v, w) are edges.
    # To ensure lexicographical order, we check all possible triplets.
    
    # Optimization: iterate through all u, then all neighbors v of u,
    # then check if any neighbor w of v is also a neighbor of u.
    
    found_triplets = []
    
    # To find the lexicographically smallest, we can iterate a from 1 to n-2
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Sort neighbors to maintain order
        sorted_neighbors_a = sorted(list(neighbors_a))
        for i in range(len(sorted_neighbors_a)):
            b = sorted_neighbors_a[i]
            if b <= a: continue
            
            # Check common neighbors of a and b
            neighbors_b = adj[b]
            # Intersection of neighbors_a and neighbors_b
            common = neighbors_a.intersection(neighbors_b)
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
