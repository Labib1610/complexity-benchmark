 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    bits = [0] * (n + 1)
    adj = [bytearray(n + 1) for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        bits[u] |= 1 << (v - 1)
        bits[v] |= 1 << (u - 1)
        adj[u][v] = 1
        adj[v][u] = 1

    out = sys.stdout.write
    for a in range(1, n):
        row_a = adj[a]
        ba = bits[a]
        for b in range(a + 1, n + 1):
            if row_a[b]:
                shifted = (ba & bits[b]) >> b
                if shifted:
                    c = b + (shifted & -shifted).bit_length()
                    out(f"{a} {b} {c}\n")
                    return

if __name__ == "__main__":
    solve()
```
