import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then check if c exists.
    # Since we need the smallest, we sort the edges or iterate carefully.
    
    # Sort edges to help find smallest a, b
    edges.sort()
    
    # To optimize, iterate through all edges (a, b) and check for common neighbors c
    # where c > b.
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that c is in adj[a] and adj[b] and c > b
            # To ensure lexicographical order, we check neighbors of b that are > b
            # and also in neighbors_a.
            
            # Optimization: intersect sets
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

solve()
