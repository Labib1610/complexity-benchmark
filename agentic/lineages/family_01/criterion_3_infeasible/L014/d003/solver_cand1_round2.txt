import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # We iterate through all possible a, then b, then check if c exists.
    # Since we need the lexicographically smallest, we iterate a from 1 to n,
    # b from a+1 to n, and check if there exists c > b such that
    # (a, c) and (b, c) are edges.
    
    # Sort edges to help with iteration or just use the adjacency sets.
    # Optimization: iterate through all edges (u, v) as potential (a, b)
    # and check for common neighbors c > b.
    
    # To ensure lexicographical order, we can iterate a from 1 to n,
    # then iterate through all neighbors b of a where b > a.
    # Then find the smallest c > b that is in both adj[a] and adj[b].
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We have a and b. Now find smallest c > b such that c in adj[a] and c in adj[b]
            # To make this efficient, intersect the two sets
            common = adj[a].intersection(adj[b])
            
            # Filter for c > b
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

solve()
