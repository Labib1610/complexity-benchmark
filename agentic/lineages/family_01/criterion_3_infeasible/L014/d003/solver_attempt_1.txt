```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find the lexicographically smallest triplet (a, b, c)
    # such that a < b < c and (a,b), (b,c), (a,c) are all edges.
    # Iterate a from 1 to n-2
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # Iterate b from neighbors of a, such that b > a
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Now check for c in neighbors of a AND neighbors of b
            # such that c > b
            # To be efficient, intersect the two sets
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
