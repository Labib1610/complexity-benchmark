import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, b from a+1 to n, c from b+1 to n.
    # Optimization: a, b, c must form a triangle.
    # For a fixed a, b must be a neighbor of a.
    # Then c must be a neighbor of both a and b.
    
    # Sort neighbors to ensure we check in lexicographical order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Find common neighbors of a and b
            # Since we need the smallest c, we look at neighbors of a
            # that are also neighbors of b.
            # To keep it efficient, intersect the two sorted lists.
            neighbors_b = adj[b]
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
