import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is connected to a
    # Iterate c from b+1 to n
    # Check if c is connected to a and b
    
    # Optimization: Only consider nodes with degree >= 2
    # Sort neighbors to potentially speed up
    for a in range(1, n + 1):
        if len(adj[a]) < 2:
            continue
        
        neighbors_a = sorted(list(adj[a]))
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of adj[a] and adj[b]
            # Since we want smallest c, we can iterate through sorted neighbors of b
            # that are also > b
            
            # To make this efficient, use set intersection
            common = adj[a].intersection(adj[b])
            valid_c = [x for x in common if x > b]
            
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
