import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2,
    # b from a+1 to n-1, and check if c exists in adj[a] and adj[b].
    
    # Sort adjacency lists to make searching faster
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # To optimize, iterate through sorted edges or nodes
    # Given n=2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    # We can iterate over all edges (u, v) and check for a common neighbor w.
    
    # Pre-sort edges to ensure we find the lexicographically smallest (a, b, c)
    # Actually, iterating a, b, c directly is better if we use bitsets or sets.
    # With n=2000, a simple loop over edges and checking neighbors is efficient.
    
    # Convert adj to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    # Sort edges by (u, v)
    edges.sort()
    
    for i in range(len(edges)):
        u, v = edges[i]
        # We want to find w such that (u, w) and (v, w) exist.
        # To ensure a < b < c, we check w > v, or w < u, or u < w < v.
        # The smallest (a, b, c) will be found by checking all common neighbors.
        
        # Common neighbors of u and v
        common = adj_sets[u].intersection(adj_sets[v])
        for w in common:
            res = sorted([u, v, w])

            # But we need the lexicographically smallest.
            # Let's collect all and find min.
            pass
            
    # Actually, the simplest way to get lexicographically smallest:
    for a in range(1, n + 1):
        for b in adj[a]:
            if b <= a: continue
            # Find common neighbors c > b
            common = adj_sets[a].intersection(adj_sets[b])
