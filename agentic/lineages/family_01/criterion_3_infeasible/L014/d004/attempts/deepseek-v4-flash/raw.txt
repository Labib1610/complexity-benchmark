We can solve this by iterating over the first pillar `a` from smallest to largest, then checking pairs among its neighbors to find the smallest valid triangle.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    # Try each possible smallest vertex a
    for a in range(1, n + 1):
        # Check neighbors of a in increasing order
        neighbors = sorted(adj[a])
        for b in neighbors:
            if b <= a:
                continue
            # Find smallest c > b that is connected to both a and b
            for c in neighbors:
                if c <= b:
                    continue
                if c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
```
