import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list where each node stores a sorted list of neighbors
    # or a set for O(1) lookup. Given n=2000, a bitset or boolean matrix is efficient.
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that a < b < c and
    # b in P(a), c in P(a), c in P(b).
    # This is equivalent to finding a triangle (a, b, c) in the graph.
    # To satisfy the canonical order (min a, then min b, then min c),
    # we iterate a from 1 to n, b from a+1 to n, and c from b+1 to n.
    
    # Optimization: iterate a, then iterate b in adj[a] where b > a,
    # then check if there exists c in adj[a] such that c in adj[b] and c > b.
    
    # Sort neighbors to ensure we can iterate in order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Filter neighbors > a
        neighbors_a_gt = [x for x in neighbors_a if x > a]
        
        for b in neighbors_a_gt:
            # We need c such that c > b and c is in adj[a] and c is in adj[b]
            # This is the intersection of neighbors_a and neighbors_b
            neighbors_b = adj[b]
            
            # To find the smallest c > b efficiently:
            for c in neighbors_a_gt:
                if c <= b:
                    continue
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
