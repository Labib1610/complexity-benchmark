```python
import sys

def solve():
    # The problem asks for a "deadlock" of three materials (a, b, c)
    # such that they form a cycle of length 3 in the reactivity graph.
    # A cycle of length 3 means (a, b), (b, c), and (c, a) are all reactive pairs.
    # This is equivalent to finding a triangle in the graph.
    # We need the lexicographically smallest sorted triple (a, b, c).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and adj[b]
    
    # Convert sets to sorted lists for faster iteration if needed, 
    # but checking membership in a set is O(1).
    
    for a in range(1, n + 1):
        # Sort neighbors to help find smallest c
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for b in neighbors_a:
            # We need c > b such that c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we want the smallest c, we can iterate through sorted neighbors
            neighbors_b = adj[b]
            
            # Optimization: iterate through the smaller neighbor list
            if len(neighbors_a) < len(neighbors_b):
                for c in neighbors_a:
                    if c > b and c in neighbors_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                # We need to find the smallest c > b in neighbors_b that is also in neighbors_a
                # Actually, just iterating through sorted neighbors_a is fine
                for c in neighbors_a:
                    if c > b and c in neighbors_b:
                        print(f"{a} {b} {c}")
                        return

solve()
```
