import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # Sort edges to help find lexicographically smallest triangle
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # all possible pairs (a, b) and check if there exists a c > b such that
    # (a, c) and (b, c) are edges.
    # Since we want the smallest (a, b, c), we iterate a from 1 to n,
    # then b from a+1 to n, then check for c.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # To optimize, iterate through all edges (a, b) and check common neighbors
    # To ensure lexicographical order, we can iterate a, then b, then check c.
    for a in range(1, n + 1):
        adj_a = adj[a]
        for b in adj_a:
            if b <= a:
                continue
            
            # Find common neighbors c > b
            # Since adj_a and adj_b are sorted, we can use two pointers
            adj_b = adj[b]
            ptr_a = 0
            ptr_b = 0
            while ptr_a < len(adj_a) and ptr_b < len(adj_b):
                val_a = adj_a[ptr_a]
                val_b = adj_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
