import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triplet
    # A triangle (a, b, c) with a < b < c exists if (a, b), (a, c), and (b, c) are edges.
    # We want to minimize a, then b, then c.
    
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # Get all neighbors of a that are greater than a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        # For each pair of neighbors (b, c) of a, check if (b, c) is an edge
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                if c in adj[b]:
                    # Found the smallest triplet because we iterate a, b, c in order
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
