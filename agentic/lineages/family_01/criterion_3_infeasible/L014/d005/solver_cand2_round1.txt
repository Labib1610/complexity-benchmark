import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use a bitset (represented as a list of integers or sets) to store adjacency
    # Since n <= 2000, a list of sets is efficient enough for lookups
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A configuration {a, b, c} is a deadlock if:
    # 1. At least one pair is prohibited (in adj).
    # 2. For any two materials, they have the same prohibition status with the third.
    # Let P(x, y) be 1 if x, y are prohibited, 0 otherwise.
    # Condition 2:
    # P(a, b) == P(a, c)
    # P(b, a) == P(b, c)
    # P(c, a) == P(c, b)
    # This simplifies to:
    # P(a, b) = P(a, c) = P(b, c)
    # Condition 1 says at least one is 1.
    # Thus, P(a, b) = P(a, c) = P(b, c) = 1 (a triangle)
    # OR P(a, b) = P(a, c) = P(b, c) = 0 (an independent set of size 3)
    
    # We need the lexicographically smallest {a, b, c} with a < b < c.
    # Iterate a from 1 to n-2, b from a+1 to n-1, c from b+1 to n.
    for a in range(1, n - 1):
        for b in range(a + 1, n):
            # Check if a and b are connected
            ab_connected = b in adj[a]
            
            # We need c such that P(a, c) == P(b, c) == ab_connected
            # If ab_connected is True, we need c such that c in adj[a] and c in adj[b]
            # If ab_connected is False, we need c such that c not in adj[a] and c not in adj[b]
            
            if ab_connected:
                # Find smallest c > b such that c in adj[a] and c in adj[b]
                # Intersection of adj[a] and adj[b]
                common = adj[a].intersection(adj[b])
                for c in sorted(common):
                    if c > b:
                        print(f"{a} {b} {c}")
                        return
            else:
                # Find smallest c > b such that c not in adj[a] and c not in adj[b]
                for c in range(b + 1, n + 1):
                    if c not in adj[a] and c not in adj[b]:
                        print(f"{a} {b} {c}")
                        return

solve()
