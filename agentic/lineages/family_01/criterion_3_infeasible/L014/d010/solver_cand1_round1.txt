import sys

def solve():
    # A sequence (x, y, z) is compliant if (x, y) is NOT prohibited OR (y, z) is NOT prohibited.
    # A sequence is NOT compliant if (x, y) IS prohibited AND (y, z) IS prohibited.
    # A deadlock configuration {a, b, c} is one where NO ordering is compliant.
    # This means for all 6 permutations (x, y, z), the sequence is NOT compliant.
    # A sequence (x, y, z) is NOT compliant iff (x, y) is prohibited AND (y, z) is prohibited.
    # For {a, b, c} to be a deadlock, every permutation must fail.
    # This means for every permutation (x, y, z), (x, y) is prohibited AND (y, z) is prohibited.
    # This implies that for any choice of middle element y, the other two elements x and z
    # must both be prohibited with y.
    # So, {a, b, c} is a deadlock iff every element in the set is prohibited with the other two.
    # This is equivalent to saying {a, b, c} forms a triangle (a clique of size 3) in the 
    # graph of prohibited pairs.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find the lexicographically smallest {a, b, c} such that
    # a, b, c form a triangle in the prohibited graph.
    # Iterate a from 1 to n-2
    for a in range(1, n + 1):
        # Iterate b from a+1 to n-1
        # b must be connected to a
        if not adj[a]: continue
        
        # Sort neighbors to ensure lexicographical order
        neighbors_a = sorted([x for x in adj[a] if x > a])
        
        for b in neighbors_a:
            # c must be connected to both a and b
            # c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            for c in sorted(common):
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
