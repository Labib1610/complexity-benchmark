import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for lexicographic iteration
    sorted_adj = [sorted(s) for s in adj]
    
    # Iterate over all triples a < b < c
    for a in range(1, n - 1):
        for b in sorted_adj[a]:
            if b <= a:
                continue
            # Check if c exists such that a < b < c and (a,c) and (b,c) are edges
            # Iterate over neighbors of b that are > b
            for c in sorted_adj[b]:
                if c <= b:
                    continue
                if c in adj[a]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
