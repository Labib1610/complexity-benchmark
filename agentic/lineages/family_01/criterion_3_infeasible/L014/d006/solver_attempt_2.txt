```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we iterate b in adj[a] such that b > a.
    # Then we check if there exists c in adj[a] intersect adj[b] such that c > b.
    
    # Sort adjacency lists to help find smallest c
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        sorted_neighbors_a = adj[a]
        for b in sorted_neighbors_a:
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and adj[b]
            # Since sorted_neighbors_a is sorted, we can iterate through it
            for c in sorted_neighbors_a:
                if c <= b:
                    continue
                if c in adj_sets[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
```
