import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    full_mask = (1 << n) - 1
    for a in range(n - 2):
        neigh_a = adj[a]
        # consider b > a
        b_mask = neigh_a >> (a + 1)  # bits for vertices > a
        while b_mask:
            # get lowest set bit in b_mask
            low = b_mask & -b_mask
            b = (low.bit_length() - 1) + (a + 1)
            # common neighbors of a and b
            common = adj[a] & adj[b]
            # keep only c > b
            mask = full_mask ^ ((1 << (b + 1)) - 1)
            common &= mask
            if common:
                lowc = common & -common
                c = lowc.bit_length() - 1
                # output 1‑based indices
                print(a + 1, b + 1, c + 1)
                return
            # clear this b bit and continue
            b_mask ^= low

if __name__ == "__main__":
    solve()
