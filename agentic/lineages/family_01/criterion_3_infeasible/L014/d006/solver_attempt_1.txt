import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to help find lexicographically smallest (a, b, c)
    # We want smallest a, then smallest b, then smallest c.
    # Iterate through all possible a, then b, then check if c exists.
    
    # To optimize, iterate through all edges (u, v) as potential (a, b)
    # where a < b. Then check for a common neighbor c > b.
    
    # Sort edges by (u, v)
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we can iterate
    # through all possible 'a' from 1 to n-2.
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        # For each pair of neighbors b, c of a, check if (b, c) is an edge
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
