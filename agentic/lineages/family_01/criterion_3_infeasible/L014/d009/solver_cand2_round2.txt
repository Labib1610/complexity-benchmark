```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # We need to find a, b, c such that (a,b), (b,c), (a,c) are all prohibited.
    # This is equivalent to finding a triangle in the graph of prohibited pairs.
    # Adjacency list using sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is connected to a
    # If so, iterate c from b+1 to n
    # Check if c is connected to both a and b
    
    # Optimization: sort adjacency lists to iterate efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we need the smallest c, we can iterate through the smaller list
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
            
            # Find common neighbors c > b
            for c in sorted(list(candidates)):
                if c > b:
                    if c in adj[a] and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
```
