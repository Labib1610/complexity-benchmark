```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n  # bitset per vertex, 0-indexed
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for a in range(n - 2):
        a_adj = adj[a]
        for b in range(a + 1, n - 1):
            common = a_adj & adj[b]
            # keep only vertices with index > b
            mask = ~((1 << (b + 1)) - 1)
            common &= mask
            if common:
                lowbit = common & -common
                c = lowbit.bit_length() - 1
                sys.stdout.write(f"{a+1} {b+1} {c+1}")
                return

if __name__ == "__main__":
    solve()
```
