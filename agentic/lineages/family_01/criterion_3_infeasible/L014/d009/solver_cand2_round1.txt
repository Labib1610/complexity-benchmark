import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # We need to find a, b, c such that (a,b), (b,c), (a,c) are all prohibited.
    # This is equivalent to finding a triangle in the complement graph of the 
    # prohibited pairs graph.
    # Let G be the graph where edges are prohibited pairs.
    # We want a, b, c such that (a,b) in E, (b,c) in E, (a,c) in E.
    # This is simply finding a triangle in the graph of prohibited pairs.
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n,
    # then b from a+1 to n, then c from b+1 to n.
    # Optimization: only check if b is in adj[a] and c is in adj[a] and c is in adj[b].
    
    # Sort adjacency lists to potentially speed up
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    # Iterate to find the lexicographically first triangle
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Since neighbors_a is sorted, we can intersect or just check
            # To be efficient, iterate through the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                for c in neighbors_a:
                    if c > b and c in adj_sets[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in adj[b]:
                    if c > b and c in adj_sets[a]:
                        print(f"{a} {b} {c}")
                        return

solve()
