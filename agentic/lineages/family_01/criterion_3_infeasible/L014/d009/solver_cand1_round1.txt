import sys

def solve():
    # A deadlock configuration of three materials (a, b, c) means they cannot be
    # distributed across two vessels without a prohibited pair being together.
    # This is equivalent to saying that the subgraph induced by {a, b, c}
    # contains at least one edge (a prohibited pair) and is not bipartite.
    # Since we have 3 nodes, the only non-bipartite graph is a triangle (K3).
    # Thus, a deadlock configuration of 3 materials is simply a triangle in the graph.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all possible a from 1 to n-2
    for a in range(1, n - 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        # For each pair of neighbors (b, c) of a, check if (b, c) is an edge
        # To minimize lexicographically, we iterate b then c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
