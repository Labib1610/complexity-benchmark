import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # We need to find a triangle (a, b, c) in the graph of PROHIBITED pairs.
    # A deadlock is a set of 3 materials where NO pair is safe.
    # This means all 3 pairs (a,b), (b,c), (a,c) must be prohibited.
    # So we are looking for a triangle in the graph of prohibited pairs.
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Optimization: sort adjacency lists to make intersection faster
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Candidates for b must be neighbors of a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # Candidates for c must be neighbors of both a and b, and c > b
            # We can intersect sorted_adj[a] and sorted_adj[b]
            # Since we need the smallest c, we just need the first one > b
            
            # Using set intersection for efficiency
            # But we need the smallest c > b
            # Since sorted_adj is sorted, we can use two pointers or set intersection
            
            # To keep it fast, use set intersection
            common = adj[a].intersection(adj[b])
            
            # Find smallest c in common such that c > b
            best_c = float('inf')
            found = False
            for c in common:
                if c > b:
                    if c < best_c:
                        best_c = c
                        found = True
            
            if found:
                print(f"{a} {b} {best_c}")
                return

if __name__ == '__main__':
    solve()
