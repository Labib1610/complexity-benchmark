import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A configuration {a, b, c} is a deadlock if:
    # {a, b, c} subset (E_a intersect E_b) union (E_b intersect E_c) union (E_c intersect E_a)
    # This is equivalent to saying:
    # a in (E_a intersect E_b) or a in (E_b intersect E_c) or a in (E_c intersect E_a)
    # AND b in (E_a intersect E_b) or b in (E_b intersect E_c) or b in (E_c intersect E_a)
    # AND c in (E_a intersect E_b) or c in (E_b intersect E_c) or c in (E_c intersect E_a)
    
    # Note: a in (E_a intersect E_b) is impossible because a is not in E_a.
    # So the condition simplifies to:
    # a in (E_b intersect E_c)
    # b in (E_a intersect E_c)
    # c in (E_a intersect E_b)
    # This is exactly the definition of a triangle in the graph.
    # Let's re-read: S subset (E_a intersect E_b) U (E_b intersect E_c) U (E_c intersect E_a)
    # If {a, b, c} is a triangle, then:
    # c in E_a and c in E_b => c in (E_a intersect E_b)
    # a in E_b and a in E_c => a in (E_b intersect E_c)
    # b in E_a and b in E_c => b in (E_a intersect E_c)
    # So any triangle is a deadlock.
    
    # Are there non-triangle deadlocks?
    # Example 1: {1, 2, 3} is a triangle.
    # Example 2: {1, 2, 4} is a triangle.
    # The condition S subset (E_a intersect E_b) U (E_b intersect E_c) U (E_c intersect E_a)
    # is satisfied if {a, b, c} is a triangle.
    # If it's not a triangle, say (a,b) is not an edge, then a not in E_b and b not in E_a.
    # Then a cannot be in (E_a intersect E_b) or (E_c intersect E_a).
    # So a must be in (E_b intersect E_c), meaning a in E_b and a in E_c.
    # Similarly, b must be in (E_a intersect E_c), meaning b in E_a and b in E_c.
    # This implies a, b are both connected to c.
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            for c in range(b + 1, n + 1):
                s = {a, b, c}
                e_a = adj[a]
                e_b = adj[b]
                e_c = adj[c]
                
                union_set = (e_a & e_b) | (e_b & e_c) | (e_c & e_a)
                if s.issubset(union_set):
                    print(f"{a} {b} {c}")
                    return

solve()
