```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to process in order
    edges.sort()
    
    # For each edge (a,b), look for a common neighbor c
    for a, b in edges:
        if a >= b:
            continue
        # Check for common neighbors
        for c in adj[a]:
            if c > b and c in adj[b]:
                print(a, b, c)
                return

if __name__ == "__main__":
    main()
```
