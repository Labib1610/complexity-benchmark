import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through all edges (a, b) and check if there exists a c
    # such that (a, c) and (b, c) are edges.
    # To ensure lexicographical order, we sort the edges first.
    edges.sort()
    
    for a, b in edges:
        # We need to find c > b such that c is in adj[a] and adj[b]
        # To keep it lexicographically smallest, we check candidates for c
        # in increasing order.
        # Intersection of adj[a] and adj[b]
        common = adj[a].intersection(adj[b])
        if common:
            c = min(common)
            # The triplet is (a, b, c). Since a < b and we need a < b < c,
            # we must ensure c > b.
            # If c < b, the triplet is (a, c, b).
            # If c > b, the triplet is (a, b, c).
            # Actually, for any triangle {u, v, w} with u < v < w,
            # the edges are (u, v), (u, w), (v, w).
            # The loop over edges (a, b) will eventually hit (u, v)
            # and find w.
            
            # Let's collect all triangles and find the minimum.
            # Since we want the smallest a, then b, then c:
            pass

    # Optimized approach:
    # Iterate a from 1 to n
    #   Iterate b in adj[a] where b > a
    #     Iterate c in adj[a] where c > b
    #       If c in adj[b], we found (a, b, c)
    # This is O(n * m) or O(n^3) worst case, but with bitsets or sets it's fast.
    
    for a in range(1, n + 1):
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Check if there is a c > b such that c is in adj[a] and adj[b]
            # To be lexicographically smallest, we want the smallest c.
            # We can intersect adj[a] and adj[b] and find min > b.
            common = adj[a].intersection(adj[b])
            candidates = [x for x in common if x > b]
            if candidates:
                print(f"{a} {b} {min(candidates)}")
                return

solve()
