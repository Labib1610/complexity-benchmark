```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate through a from 1 to n, b from a+1 to n, c from b+1 to n.
    # Optimization: a, b, c must form a triangle.
    # For a fixed a, b, we only need to check if c is in adj[a] and adj[b].
    
    # Sort neighbors to help find c efficiently
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through possible a
    for a in range(1, n + 1):
        # Iterate through possible b
        for b in adj[a]:
            if b <= a:
                continue
            
            # Find common neighbors of a and b
            # Since adj lists are sorted, we can find intersection efficiently
            set_a = set(adj[a])
            common = [c for c in adj[b] if c in set_a]
            
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
