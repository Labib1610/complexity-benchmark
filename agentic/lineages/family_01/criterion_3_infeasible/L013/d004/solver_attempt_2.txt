import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2,
    # b from a+1 to n-1, and check if c exists.
    
    # Sort edges to help with efficient lookup
    # Actually, iterating through sorted edges or nodes is better.
    # Given the constraints and the requirement for lexicographical order:
    
    # Sort adjacency lists to iterate in order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # We want smallest a, then b, then c.
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        # Iterate b in adj[a] such that b > a
        for b in adj[a]:
            if b <= a:
                continue
            
            # Now we need a c such that c > b, c in adj[a], and c in adj[b]
            # Since adj lists are sorted, we can intersect them efficiently
            # or just iterate through the smaller list.
            
            # To optimize, check if c exists in both adj[a] and adj[b]
            # Since we want the smallest c, we iterate through the sorted adj[a]
            # starting from the first element > b.
            
            # Using sets for O(1) lookup
            adj_b_set = set(adj[b])
            for c in adj[a]:
                if c > b and c in adj_b_set:
                    print(f"{a} {b} {c}")
                    return

solve()
