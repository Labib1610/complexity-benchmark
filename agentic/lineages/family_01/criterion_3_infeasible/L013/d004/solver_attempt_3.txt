```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest trio (a, b, c) with a < b < c,
    # we iterate through all possible a, then b, then c.
    # Since we need a triangle, we check if b is in adj[a] and c is in adj[a] and adj[b].
    
    # Sort neighbors to help with lexicographical order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c such that c > b and c is connected to both a and b.
            # Intersection of neighbors_a and neighbors_b
            # Since neighbors are sorted, we can find common elements efficiently.
            # But given n=2000, a simple intersection is fine.
            
            # Optimization: only check c > b
            set_b = adj[b]
            for c in neighbors_a[i+1:]:
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
