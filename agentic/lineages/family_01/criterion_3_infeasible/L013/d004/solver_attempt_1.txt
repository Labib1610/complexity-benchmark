import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A group of three species requires at least 3 enclosures if they form a triangle
    # (i.e., every pair in the group is forbidden from sharing an enclosure).
    # We need to find the lexicographically smallest triangle (a, b, c) with a < b < c.
    
    # Iterate through all possible a, b, c in lexicographical order
    for a in range(1, n + 1):
        # Get neighbors of a
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        if len_a < 2:
            continue
            
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that c > b and c is a neighbor of both a and b
            # To ensure lexicographical order, we check neighbors of b that are also neighbors of a
            neighbors_b = adj[b]
            
            # Find common neighbors c > b
            common = []
            for c in neighbors_a:
                if c > b and c in neighbors_b:
                    common.append(c)
            
            if common:
                print(f"{a} {b} {min(common)}")
                return

if __name__ == '__main__':
    solve()
