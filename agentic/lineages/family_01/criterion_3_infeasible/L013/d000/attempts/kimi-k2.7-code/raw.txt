import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    all_mask = (1 << n) - 1
    gt_mask = [0] * (n + 2)
    for b in range(1, n + 1):
        gt_mask[b] = all_mask ^ ((1 << b) - 1)

    for a in range(1, n + 1):
        bits_b = adj[a] & gt_mask[a]
        while bits_b:
            lsb = bits_b & -bits_b
            b = lsb.bit_length()
            bits_b ^= lsb
            cand = adj[a] & adj[b] & gt_mask[b]
            if cand:
                c = (cand & -cand).bit_length()
                sys.stdout.write(f"{a} {b} {c}\n")
                return

if __name__ == "__main__":
    solve()
