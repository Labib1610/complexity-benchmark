import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we iterate b from a+1 to n-1.
    # If a and b are connected, we check if there exists c > b
    # such that c is connected to both a and b.
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    # Pre-sort adjacency lists for faster intersection
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider b > a
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Find common neighbors of a and b that are > b
            # Since neighbors_a and neighbors_b are sorted, we can intersect efficiently
            neighbors_b = sorted_adj[b]
            
            ptr_a = 0
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
