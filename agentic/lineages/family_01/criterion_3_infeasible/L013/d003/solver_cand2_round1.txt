```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through all possible a, then b, then c.
    # An unstable core is a triangle (a, b, c).
    # We can iterate through all edges (u, v) and check if there exists a w
    # such that (u, w) and (v, w) are edges.
    
    best = None
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    # To minimize (a, b, c), we iterate a from 1 to n
    for a in range(1, n + 1):
        # Get neighbors of a
        neighbors_a = sorted(list(adj[a]))
        if len(neighbors_a) < 2:
            continue
            
        # Try all pairs of neighbors (b, c) of a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b < a: continue
            
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                
                # Check if b and c are connected
                if c in adj[b]:
                    # Found a triangle (a, b, c)
                    # Ensure a < b < c
                    res = sorted([a, b, c])
                    if best is None or res < best:
                        best = res
        
        # Optimization: if we found a triangle starting with 'a', 
        # and we have checked all possible b, c for this 'a', 
        # we don't necessarily stop, but since we want the smallest 'a',
        # once we find any triangle for a specific 'a', we can't 
        # immediately stop because there might be a smaller 'b' or 'c' 
        # for the same 'a'. However, the current approach is O(N * degree^2),
        # which is acceptable for N=2000.
        if best is not None:
            # Since we iterate 'a' in increasing order, the first 'a' 
            # that yields a triangle is the smallest possible 'a'.
            # We just need to ensure we find the smallest 'b' and 'c' for this 'a'.
            # We can break after checking all triangles for this 'a'.
            break
            
    print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
