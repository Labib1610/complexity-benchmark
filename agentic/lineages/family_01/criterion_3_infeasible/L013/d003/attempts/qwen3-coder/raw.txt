```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # We need to find a triangle (three nodes all connected to each other)
    # Among all triangles, we want the lexicographically smallest one
    
    # To efficiently find triangles, we can iterate over all edges
    # For each edge (u,v), we look for a node w such that (u,w) and (v,w) are also edges
    # To ensure lexicographical order, we process nodes in order
    
    # First, let's store neighbors in sorted order for consistent processing
    neighbors = [sorted(list(adj[i])) for i in range(n+1)]
    
    best_triangle = None
    
    # Iterate through all edges
    for u, v in edges:
        if u > v:
            continue  # Only consider each edge once, with u < v
        
        # Now check for common neighbors of u and v
        nu = neighbors[u]
        nv = neighbors[v]
        
        i = j = 0
        common = []
        while i < len(nu) and j < len(nv):
            if nu[i] == nv[j]:
                if nu[i] != u and nu[i] != v:  # Avoid including u or v themselves
                    common.append(nu[i])
                i += 1
                j += 1
            elif nu[i] < nv[j]:
                i += 1
            else:
                j += 1
                
        # For each common neighbor w, we have a triangle (u,v,w)
        for w in common:
            if w < u or w < v:
                continue  # Ensure u <= v <= w to maintain ordering
            
            triangle = tuple(sorted([u, v, w]))
            if best_triangle is None or triangle < best_triangle:
                best_triangle = triangle
    
    if best_triangle:
        print(best_triangle[0], best_triangle[1], best_triangle[2])

if __name__ == "__main__":
    main()
```
