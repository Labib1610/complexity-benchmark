import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To minimize S1 * (n+1)^2 + S2 * (n+1) + S3, we iterate S1 from 1 to n,
    # then S2 from S1+1 to n, then check if there exists S3 > S2 such that
    # S3 is in adj[S1] and adj[S2].
    
    best_trio = None
    min_score = float('inf')
    
    # Pre-sort adjacency lists to make intersection faster
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for s1 in range(1, n + 1):
        # Optimization: if s1 * (n+1)^2 is already worse than min_score, break
        if s1 * (n + 1)**2 >= min_score:
            break
            
        neighbors_s1 = adj[s1]
        for s2 in range(s1 + 1, n + 1):
            if s2 not in neighbors_s1:
                continue
            
            score_prefix = s1 * (n + 1)**2 + s2 * (n + 1)
            if score_prefix >= min_score:
                continue
                
            # Find common neighbors s3 > s2
            neighbors_s2 = adj[s2]
            common = neighbors_s1.intersection(neighbors_s2)
            
            for s3 in common:
                if s3 > s2:
                    score = score_prefix + s3
                    if score < min_score:
                        min_score = score
                        best_trio = (s1, s2, s3)
                        
    if best_trio:
        print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

if __name__ == '__main__':
    solve()
