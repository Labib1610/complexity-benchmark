import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need the lexicographically smallest trio (a, b, c) with a < b < c
    # such that (a,b), (b,c), (c,a) are all edges.
    # Iterate through all possible a, then b, then c.
    # To optimize, iterate a from 1 to n-2.
    # For a fixed a, iterate b in adj[a] such that b > a.
    # Then check if there exists c in adj[a] such that c > b and c is in adj[b].
    
    # Sort adjacency lists to ensure we check in lexicographical order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Since neighbors_a is sorted, we can look for c in neighbors_a[i+1:]
            # and check if c is in adj[b].
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    # Found the lexicographically smallest trio
                    res = sorted([a, b, c])
                    print(f"{res[0]} {res[1]} {res[2]}")
                    return

if __name__ == '__main__':
    solve()
