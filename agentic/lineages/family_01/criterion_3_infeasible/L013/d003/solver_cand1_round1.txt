```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (i, j, k) with i < j < k
    # We iterate i from 1 to n-2
    # Then j from i+1 to n-1
    # Then check if k exists in adj[i] and adj[j]
    
    for i in range(1, n + 1):
        # Sort neighbors to check potential j's in order
        neighbors_i = sorted(list(adj[i]))
        for idx_j in range(len(neighbors_i)):
            j = neighbors_i[idx_j]
            if j <= i:
                continue
            
            # We need a k > j such that k is in adj[i] and adj[j]
            # To minimize k, we look for the smallest k > j
            # Intersection of adj[i] and adj[j]
            common = adj[i].intersection(adj[j])
            
            best_k = float('inf')
            found = False
            for k in common:
                if k > j:
                    if k < best_k:
                        best_k = k
                        found = True
            
            if found:
                print(f"{i} {j} {best_k}")
                return

solve()
```
