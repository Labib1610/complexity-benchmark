import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) with a < b < c.
    # To find the lexicographically smallest, we iterate a from 1 to n-2,
    # b from a+1 to n-1, and c from b+1 to n.
    # To optimize, we only check if c exists in adj[a] and adj[b].
    
    # Sort adjacency lists to potentially speed up intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Convert back to sets for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Neighbors of a that are > a
        neighbors_a = [x for x in adj[a] if x > a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that b < c and c is in adj[a] and adj[b]
            # Iterate through neighbors of b that are > b
            for c in adj[b]:
                if c > b:
                    if c in adj_sets[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
