import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is connected to a
    # If so, check if there exists c > b such that c is connected to both a and b
    
    # Sort neighbors to ensure we find the smallest c efficiently
    adj_sorted = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = adj_sorted[a]
        # Only consider neighbors b > a
        valid_b = [b for b in neighbors_a if b > a]
        for b in valid_b:
            # We need a c > b such that c is in adj[a] and adj[b]
            # To be lexicographically smallest, we want the smallest such c
            # Since neighbors_a is sorted, we can iterate through it
            for c in neighbors_a:
                if c > b:
                    if c in adj[b]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
