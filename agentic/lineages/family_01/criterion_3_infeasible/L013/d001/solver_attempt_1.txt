import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c
    # We iterate through all possible a, then b, then c.
    # Since we need the smallest, we iterate a from 1 to n, b from a+1 to n,
    # and check if c exists in both adj[a] and adj[b].
    
    # Optimization: Sort adjacency lists to check c efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider neighbors of a as potential b's
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need a c such that c > b, c is in adj[a] and c is in adj[b]
            # Intersection of sorted_adj[a] and sorted_adj[b]
            # Since we want the smallest c, we can iterate through the intersection
            
            # Using set intersection for speed
            common = adj[a].intersection(adj[b])
            for c in sorted(list(common)):
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
