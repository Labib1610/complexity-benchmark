```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj_bits = [0] * (n + 1)          # adjacency as bitset
    neigh = [[] for _ in range(n + 1)]  # neighbor lists for iteration

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj_bits[u] |= 1 << v
        adj_bits[v] |= 1 << u
        neigh[u].append(v)
        neigh[v].append(u)

    for a in range(1, n + 1):
        # ensure neighbor list is sorted for lexicographic order
        neigh[a].sort()
        for b in neigh[a]:
            if b <= a:
                continue
            # common neighbors of a and b
            common = adj_bits[a] & adj_bits[b]
            # keep only vertices > b
            mask = (1 << (b + 1)) - 1   # bits 0..b set
            cand = common & ~mask
            if cand:
                low = cand & -cand      # lowest set bit
                c = low.bit_length() - 1
                # output in ascending order (a<b<c already)
                sys.stdout.write(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
```
