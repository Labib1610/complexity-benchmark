import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest trio (i, j, k) with i < j < k:
    # We iterate through all possible i, then j, then check if k exists.
    # To optimize, we sort the adjacency lists.
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # We want to minimize i, then j, then k.
    # Iterate i from 1 to n
    for i in range(1, n + 1):
        # Neighbors of i that are > i
        neighbors_i = [v for v in adj[i] if v > i]
        if len(neighbors_i) < 2:
            continue
        
        # Sort neighbors to ensure we check j in increasing order
        neighbors_i.sort()
        
        for idx_j in range(len(neighbors_i)):
            j = neighbors_i[idx_j]
            # We need a k such that k > j and k is in both adj[i] and adj[j]
            # To minimize k, we look at the intersection of adj[i] and adj[j]
            # Since adj lists are sorted, we can find intersection efficiently
            
            # Optimization: only check k > j
            # Use set for O(1) lookup
            set_j = set(adj[j])
            for k in neighbors_i[idx_j + 1:]:
                if k in set_j:
                    print(f"{i} {j} {k}")
                    return

if __name__ == '__main__':
    solve()
