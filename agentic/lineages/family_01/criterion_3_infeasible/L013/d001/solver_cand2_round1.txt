import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and adj[b]
    
    # Sort adjacency lists to help find smallest c efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider neighbors b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # We need a c > b such that c is in adj[a] and adj[b]
            # Since sorted_adj[a] is sorted, we can intersect efficiently
            # or just iterate through neighbors of b that are > b
            # To be lexicographically smallest, we want the smallest c
            
            # Optimization: use set intersection for speed
            # But we need the smallest c > b
            # Let's use the smaller degree node to iterate
            if len(adj[a]) < len(adj[b]):
                for c in sorted_adj[a]:
                    if c > b and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in sorted_adj[b]:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

solve()
