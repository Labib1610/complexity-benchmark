import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest trio (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, check if there exists c > b such that c is in adj[a] and adj[b]
    
    for a in range(1, n - 1):
        # Get neighbors of a that are > a
        neighbors_a = [v for v in adj[a] if v > a]
        neighbors_a.sort()
        
        for b in neighbors_a:
            # We need a c > b such that c is in adj[a] and adj[b]
            # To find the smallest c, we can intersect the sorted neighbors
            # of a and b that are > b
            
            # Optimization: use set intersection for faster lookup
            # But since we need the smallest c, we can iterate through
            # the smaller neighbor set
            
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
                
            found_c = -1
            for c in candidates:
                if c > b:
                    if c in adj[a] and c in adj[b]:
                        if found_c == -1 or c < found_c:
                            found_c = c
            
            if found_c != -1:
                print(f"{a} {b} {found_c}")
                return

solve()
