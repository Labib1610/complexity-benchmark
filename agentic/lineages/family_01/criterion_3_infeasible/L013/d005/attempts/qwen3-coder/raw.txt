import sys
from collections import defaultdict

def find_triangle(n, edges):
    # Build adjacency list
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # Create a list of neighbors for each node, sorted for consistent processing
    neighbors = {}
    for node in range(1, n + 1):
        neighbors[node] = sorted(list(adj[node]))
    
    # For each node, check all pairs of its neighbors
    # If a pair of neighbors are also connected, we found a triangle
    best_triangle = None
    
    for node in range(1, n + 1):
        node_neighbors = neighbors[node]
        neighbor_set = set(node_neighbors)
        
        # Check all pairs of neighbors of 'node'
        for i in range(len(node_neighbors)):
            u = node_neighbors[i]
            for j in range(i + 1, len(node_neighbors)):
                v = node_neighbors[j]
                # If u and v are connected, then (node, u, v) form a triangle
                if v in adj[u]:
                    triangle = tuple(sorted([node, u, v]))
                    if best_triangle is None or triangle < best_triangle:
                        best_triangle = triangle
    
    return best_triangle

# Read input
n, m = map(int, sys.stdin.readline().split())
edges = []
for _ in range(m):
    u, v = map(int, sys.stdin.readline().split())
    edges.append((u, v))

# Find and print the lexicographically smallest triangle
result = find_triangle(n, edges)
print(result[0], result[1], result[2])
