import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (i, j, k) with i < j < k.
    # To minimize (i, j, k) lexicographically, we iterate i from 1 to n,
    # then j from i+1 to n, then k from j+1 to n.
    # Since we need to find the first one, we can optimize:
    # For a fixed i, iterate through all neighbors j of i such that j > i.
    # Then check if there exists a k such that k > j and k is a neighbor of both i and j.
    
    # Sort neighbors to ensure we check in lexicographical order
    for i in range(1, n + 1):
        neighbors_i = sorted([x for x in adj[i] if x > i])
        for idx_j, j in enumerate(neighbors_i):
            # We need a k > j such that k is in adj[i] and adj[j]
            # Intersection of adj[i] and adj[j]
            common = adj[i].intersection(adj[j])
            valid_k = [k for k in common if k > j]
            if valid_k:
                print(f"{i} {j} {min(valid_k)}")
                return

if __name__ == '__main__':
    solve()
