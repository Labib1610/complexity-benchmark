import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a triangle (a, b, c) with a < b < c,
    # we can iterate through all edges (u, v) and check for a common neighbor w.
    
    best = None
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    for u in range(1, n + 1):
        # Get neighbors of u that are > u
        neighbors = sorted(list(adj[u]))
        len_n = len(neighbors)
        for i in range(len_n):
            v = neighbors[i]
            # Now we need a w such that w > v and w is in adj[u] and adj[v]
            # To ensure lexicographical order, we check w in increasing order
            for j in range(i + 1, len_n):
                w = neighbors[j]
                if w in adj[v]:
                    # Found a triangle (u, v, w)
                    # Since u < v < w, this is a candidate
                    if best is None or (u, v, w) < best:
                        best = (u, v, w)
                        # Since we iterate u from 1, v from neighbors, w from neighbors,
                        # the first one we find might not be the absolute smallest if we don't
                        # check all. But we can optimize: if we find a triangle with current u,
                        # we don't need to check larger u.
                        print(f"{u} {v} {w}")
                        return

solve()
