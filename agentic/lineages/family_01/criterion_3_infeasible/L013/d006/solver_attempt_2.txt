```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through a from 1 to n-2, b from a+1 to n-1,
    # and check if there exists c > b such that c is connected to both a and b.
    
    # Sort neighbors to make finding c easier
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Neighbors of a that are > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # Since we want the lexicographically smallest, we check c in order
            # Intersection of neighbors of a and b that are > b
            # Using sets for O(1) lookup
            set_a = adj[a]
            set_b = adj[b]
            
            # Iterate through potential c's
            # To be efficient, iterate through the smaller neighbor set
            if len(set_a) < len(set_b):
                candidates = sorted([x for x in set_a if x > b])
            else:
                candidates = sorted([x for x in set_b if x > b])
                
            for c in candidates:
                if c in set_a and c in set_b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
