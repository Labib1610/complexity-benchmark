 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    high = [~((1 << i) - 1) for i in range(n + 1)]

    for a in range(1, n + 1):
        adj_a = adj[a]
        cand = adj_a & high[a]
        while cand:
            lsb = cand & -cand
            b = lsb.bit_length()
            common = adj_a & adj[b] & high[b]
            if common:
                c = (common & -common).bit_length()
                sys.stdout.write(f"{a} {b} {c}\n")
                return
            cand ^= lsb

if __name__ == "__main__":
    solve()
```
