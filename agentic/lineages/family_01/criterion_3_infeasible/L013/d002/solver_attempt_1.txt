```python
import sys

def solve():
    # A deadlock of 3 drones (a, b, c) means that every way to assign them to 
    # 2 altitudes results in a conflict. This is equivalent to saying that 
    # the subgraph induced by {a, b, c} contains all 3 possible edges (a,b), 
    # (b,c), and (a,c). In other words, a deadlock is simply a triangle 
    # (a clique of size 3) in the graph of overlapping trajectories.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then check if c exists.
    # Since n <= 2000, O(n^3) is too slow, but O(n * m) or O(n^2) is fine.
    
    # Sort adjacency lists to help with lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Iterate b in adj[a] such that b > a
        for b in adj[a]:
            if b <= a:
                continue
            
            # We need a c such that c > b and c is in both adj[a] and adj[b]
            # Since adj lists are sorted, we can find intersection efficiently
            # or just iterate through the smaller list.
            
            # Optimization: check intersection of adj[a] and adj[b]
            # Only need to check c > b
            set_a = set(adj[a])
            for c in adj[b]:
                if c > b and c in set_a:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
