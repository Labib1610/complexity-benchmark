import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Sort neighbors to potentially speed up intersection
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Only consider neighbors of a that are > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # We need a c > b such that c is in adj[a] and c is in adj[b]
            # Since we want the lexicographically smallest, we check c in order
            # Intersection of neighbors_a and neighbors_b
            # To optimize, use set intersection
            set_a = adj[a]
            set_b = adj[b]
            
            # Find common neighbors c > b
            common = sorted([x for x in set_a if x > b and x in set_b])
            if common:
                print(f"{a} {b} {common[0]}")
                return

if __name__ == '__main__':
    solve()
