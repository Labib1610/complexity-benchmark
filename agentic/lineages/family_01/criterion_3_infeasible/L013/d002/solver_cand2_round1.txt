import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c,
    # we iterate through all possible a, then b, then c.
    # Since n <= 2000 (40*50), an O(n^3) approach is too slow,
    # but we can optimize by checking if c exists in adj[a] and adj[b].
    
    # Sort neighbors to help with lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        # Neighbors of a that are > a
        neighbors_a = [x for x in adj[a] if x > a]
        if len(neighbors_a) < 2:
            continue
        
        # Sort neighbors to ensure we check b in increasing order
        neighbors_a.sort()
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that c > b and c is in both adj[a] and adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # Since adj[b] is sorted, we can use set intersection or binary search
            set_b = set(adj[b])
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
