```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Build adjacency list
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # A deadlock is a triangle (a, b, c) where (a,b), (b,c), (a,c) are edges.
    # We need the lexicographically smallest (a, b, c) with a < b < c.
    # Iterate through all edges (u, v) and check for a common neighbor w.
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b from a+1 to n, then check if there exists c > b such that
    # (a, c) and (b, c) are edges.
    
    # Sort edges to help find triangles efficiently
    # For each edge (u, v) with u < v, check all w > v such that
    # (u, w) and (v, w) are edges.
    
    best = None
    
    # To find the lexicographically smallest (a, b, c), we iterate a, b, c.
    # Since n <= 2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    # We can iterate over all edges (u, v) and find common neighbors.
    
    for a in range(1, n + 1):
        # Optimization: if we already found a triangle starting with a smaller 'a',
        # we don't need to check larger 'a'.
        if best and a > best[0]:
            break
            
        # Neighbors of a
        neighbors_a = adj[a]
        for b in sorted(list(neighbors_a)):
            if b <= a:
                continue
            if best and (a > best[0] or (a == best[0] and b >= best[1])):
                continue
            
            # Find common neighbors c > b
            common = neighbors_a.intersection(adj[b])
            for c in common:
                if c > b:
                    if best is None or (a < best[0]) or (a == best[0] and b < best[1]) or (a == best[0] and b == best[1] and c < best[2]):
                        best = (a, b, c)
    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
