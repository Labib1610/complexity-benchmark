```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # We iterate through all possible 'a' from 1 to n-2.
    # For each 'a', we iterate through all neighbors 'b' of 'a' such that b > a.
    # Then we check if there exists a 'c' such that c > b and c is a neighbor of both a and b.
    # To ensure lexicographical order, we sort the neighbors of each node.
    
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need a c > b such that c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can iterate through it
            # or use set intersection.
            # To be efficient, intersect the two neighbor sets.
            
            # Optimization: only check c > b
            # Using set intersection for speed
            common = adj[a].intersection(adj[b])
            for c in sorted(list(common)):
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
