import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # We need a trio {a, b, c} such that (a,b), (b,c), (a,c) are all in edges.
    # This is equivalent to finding a triangle in the graph.
    # To find the lexicographically smallest, we iterate a from 1 to n-2,
    # b from a+1 to n-1, and c from b+1 to n.
    
    # Sort edges for each node to optimize lookup
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # To find the lexicographically smallest triangle (a, b, c):
    # Iterate a, then b, then check if there exists c > b such that
    # c is in adj[a] and c is in adj[b].
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in neighbors_a and c is in adj[b]
            # Since neighbors_a is sorted, we only look at elements after i
            neighbors_b = adj[b]
            
            # Use set intersection for efficiency
            # But we need the smallest c, so we iterate through neighbors_a[i+1:]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
