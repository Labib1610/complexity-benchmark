import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest trio (i, j, k) with i < j < k:
    # Iterate i from 1 to n-2
    # Iterate j from i+1 to n-1
    # Check if j is connected to i
    # If so, check if there exists k > j such that k is connected to both i and j
    
    # Sort adjacency lists to make finding k easier
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for i in range(1, n + 1):
        # Only consider neighbors j > i
        neighbors_i = [x for x in sorted_adj[i] if x > i]
        for j in neighbors_i:
            # We need a k > j such that k is in adj[i] and adj[j]
            # Intersection of sorted_adj[i] and sorted_adj[j]
            # Since we need k > j, we can just iterate through sorted_adj[i]
            # and check if it's > j and in adj[j]
            
            # Optimization: use set for O(1) lookup
            adj_j = adj[j]
            for k in sorted_adj[i]:
                if k > j and k in adj_j:
                    print(f"{i} {j} {k}")
                    return

if __name__ == '__main__':
    solve()
