```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list representation of conflicts
    conflicts = defaultdict(set)
    edges = []
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        conflicts[u].add(v)
        conflicts[v].add(u)
        edges.append((u, v))
    
    # Sort edges to process in order
    edges.sort()
    
    # For each edge (u,v), look for a node w such that (u,w) and (v,w) are also edges
    # This forms a triangle (u,v,w)
    for u, v in edges:
        if u >= v:
            continue
        # Get neighbors of both u and v
        neighbors_u = conflicts[u]
        neighbors_v = conflicts[v]
        # Find common neighbors
        common = neighbors_u.intersection(neighbors_v)
        for w in common:
            if u < v < w:
                print(u, v, w)
                return

if __name__ == "__main__":
    main()
```
