 ```python
import sys

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n = data[0]
    m = data[1]
    adj = [0] * (n + 1)
    neigh = [[] for _ in range(n + 1)]
    p = 2
    for _ in range(m):
        u = data[p]
        v = data[p + 1]
        p += 2
        if u == v:
            continue
        if (adj[u] >> v) & 1:
            continue
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        neigh[u].append(v)
        neigh[v].append(u)
    for i in range(1, n + 1):
        neigh[i].sort()
    all_mask = (1 << (n + 1)) - 1
    mask_gt = [0] * (n + 1)
    for b in range(1, n + 1):
        mask_gt[b] = all_mask ^ ((1 << (b + 1)) - 1)
    for a in range(1, n - 1):
        for b in neigh[a]:
            if b <= a or b == n:
                continue
            cand = adj[a] & adj[b] & mask_gt[b]
            if cand:
                c = (cand & -cand).bit_length() - 1
                sys.stdout.write(f"{a} {b} {c}")
                return

solve()
```
