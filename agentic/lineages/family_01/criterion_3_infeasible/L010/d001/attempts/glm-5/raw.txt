import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # Use a set for O(1) conflict lookup
    conflicts = set()
    # Also build adjacency list for iteration
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        conflicts.add((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # Sort adjacency lists for lexicographic order
    for i in range(1, n + 1):
        adj[i].sort()
    
    # For each pair (a, b) with a < b that conflict, check for c
    for a in range(1, n + 1):
        for b in adj[a]:
            if b <= a:
                continue
            # a < b and (a, b) is a conflict
            # Find smallest c > b such that (a, c) and (b, c) are conflicts
            # Use two pointers on sorted lists adj[a] and adj[b]
            i, j = 0, 0
            len_a = len(adj[a])
            len_b = len(adj[b])
            while i < len_a and j < len_b:
                ca = adj[a][i]
                cb = adj[b][j]
                if ca <= b:
                    i += 1
                    continue
                if cb <= b:
                    j += 1
                    continue
                if ca == cb:
                    c = ca
                    # Check if (a, c) and (b, c) are conflicts (they should be by construction)
                    # But verify c > b
                    if c > b:
                        print(a, b, c)
                        return
                    i += 1
                    j += 1
                elif ca < cb:
                    i += 1
                else:
                    j += 1
    
    # Should not reach here as deadlock is guaranteed
    pass

if __name__ == "__main__":
    solve()
