```python
import sys

def solve():
    # A deadlock of 3 reports (u1, v1), (u2, v2), (u3, v3) means
    # every worker involved appears in an even number of reports.
    # Since there are 3 reports, each worker must appear either 0 or 2 times.
    # Let the set of workers be {a, b, c}.
    # The only way to have 3 edges such that each vertex has even degree
    # is to form a triangle (a,b), (b,c), (c,a).
    # The problem asks for the lexicographically smallest {a, b, c}
    # such that (a,b), (b,c), (a,c) are all in the log.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # We need to find a triangle (a, b, c) with a < b < c.
    # To find the lexicographically smallest (a, b, c), we iterate
    # through all possible a, then b, then c.
    # Since n <= 2000, we can iterate over all edges (u, v) and check
    # for a common neighbor w.
    
    best = None
    
    # Sort edges to help find smallest a, b, c
    edges.sort()
    
    for i in range(m):
        u, v = edges[i]
        # Triangle is (u, v, w) where w is a common neighbor
        # We need to check all w such that (u, w) and (v, w) exist.
        # To ensure lexicographical order, we check all possible w.
        
        # Intersection of adj[u] and adj[v]
        common = adj[u].intersection(adj[v])
        for w in common:
            # The triangle vertices are u, v, w.
            # Sort them to get a < b < c
            t = sorted([u, v, w])
            if best is None or t < best:
                best = t
                
    print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
