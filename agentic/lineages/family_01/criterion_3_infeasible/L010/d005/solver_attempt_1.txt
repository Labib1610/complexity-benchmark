```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # A signature meal is a triangle (a, b, c) in the graph.
    # We need the lexicographically smallest (a, b, c) with a < b < c.
    # Iterate through all edges (u, v) with u < v, and check for common neighbors w.
    
    edges.sort()
    
    best = None
    
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n.
    # Then iterate b from a+1 to n.
    # Then check if there exists c > b such that (a, c) and (b, c) are edges.
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Sort neighbors to ensure we find the smallest b and c
        sorted_neighbors_a = sorted(list(neighbors_a))
        
        for b in sorted_neighbors_a:
            if b <= a:
                continue
            
            # We need a c such that c > b and c is in both adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            neighbors_b = adj[b]
            
            # Optimization: iterate over the smaller neighbor set
            if len(neighbors_a) < len(neighbors_b):
                common = [c for c in neighbors_a if c > b and c in neighbors_b]
            else:
                common = [c for c in neighbors_b if c > b and c in neighbors_a]
            
            if common:
                c = min(common)
                if best is None or (a, b, c) < best:
                    best = (a, b, c)
        
        # If we found a triangle starting with 'a', and we've checked all b > a,
        # we can potentially stop if we are sure no smaller 'a' exists.
        # Since we iterate 'a' from 1, the first 'a' that yields a triangle
        # will be the smallest 'a'.
        if best is not None:
            break
            
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
