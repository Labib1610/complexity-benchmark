import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # A ternary alloy {a, b, c} is sound if (a,b), (b,c), and (a,c) are all in the catalogue.
    # This is equivalent to finding a triangle in the graph.
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to n,
    # b from a+1 to n, and c from b+1 to n.
    
    # Sort edges to help with iteration or just use the adjacency sets.
    # Since n <= 2000, O(n^3) is too slow, but we can iterate over edges (a, b)
    # and check for a common neighbor c.
    
    edges.sort()
    
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that c > b and c is in adj[a] and adj[b]
            # To ensure lexicographical order, we check c in increasing order
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

solve()
