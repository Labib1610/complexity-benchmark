```python
import sys

def solve():
    # A deadlock is a subset of 3 distinct reports (edges) such that every worker
    # involved appears an even number of times.
    # Let the three edges be (u1, v1), (u2, v2), (u3, v3).
    # The total number of occurrences of any worker must be even.
    # With 3 edges, there are 6 endpoints.
    # If we have 3 distinct workers {a, b, c}, the only way to have 3 edges
    # such that each appears an even number of times is to have the edges
    # (a, b), (b, c), and (a, c).
    # This forms a triangle (a cycle of length 3).
    # In a triangle (a, b), (b, c), (a, c):
    # a appears in (a, b) and (a, c) -> 2 times (even)
    # b appears in (a, b) and (b, c) -> 2 times (even)
    # c appears in (b, c) and (a, c) -> 2 times (even)
    # This satisfies the condition.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # We need the lexicographically smallest (a, b, c) where a < b < c.
    # A triangle (a, b, c) exists if (a, b), (b, c), and (a, c) are in the log.
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # possible a, then b, then c.
    
    # Sort edges to help find triangles efficiently
    edges.sort()
    
    # For each edge (u, v), check if there exists a w such that (u, w) and (v, w) exist.
    # To get the smallest (a, b, c), we can iterate through all possible a, b, c.
    # Since n <= 2000, O(n^2) or O(m * sqrt(m)) is acceptable.
    
    best = None
    
    # Iterate over all edges (u, v) as the base
    for u in range(1, n + 1):
        sorted_neighbors = sorted(list(adj[u]))
        for i in range(len(sorted_neighbors)):
            v = sorted_neighbors[i]
            # Now we look for w such that (u, w) and (v, w) exist.
            # w must be > v to ensure a < b < c order (u < v < w)
            # Actually, the triangle is {u, v, w}. Let a < b < c.
            # We need to check all triangles and pick the smallest (a, b, c).
            
            # Optimization: iterate over the smaller degree vertex
            if len(adj[u]) > len(adj[v]):
                u_small, v_small = v, u
            else:
                u_small, v_small = u, v
                
            for w in adj[u_small]:
                if w in adj[v_small]:
                    # Found triangle {u, v, w}
                    triangle = sorted([u, v, w])
                    if best is None or triangle < best:
                        best = triangle
                        
    print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
