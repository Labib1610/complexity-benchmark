```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # A trio (i, j, k) with i < j < k is perfectly harmonious if 
    # it forms a triangle (i,j), (j,k), (i,k) are all edges.
    # The problem states the number of synergies involving only dishes 
    # from the trio is exactly 3. This means the trio must form a triangle.
    
    # We need to find all triangles and pick the lexicographically smallest one.
    # Since n <= 2000, we can iterate over edges (u, v) and check common neighbors.
    
    best_trio = None
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    for u, v in edges:
        # Find common neighbors w such that u < v < w or u < w < v or w < u < v
        # To ensure we find all triangles, we can iterate over all w
        # But to be efficient, iterate over the smaller adjacency list
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
            
        for w in adj[u]:
            if w in adj[v]:
                # Found a triangle (u, v, w)
                trio = sorted([u, v, w])
                if best_trio is None or trio < best_trio:
                    best_trio = trio
                    
    print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

if __name__ == '__main__':
    solve()
```
