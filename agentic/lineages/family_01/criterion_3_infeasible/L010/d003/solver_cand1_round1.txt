import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # The problem defines a deadlock as 3 columns (edges) such that 
    # the sum of these 3 columns has exactly three 2s and 0s elsewhere.
    # Each column (edge) has exactly two 1s (the two workers involved).
    # Let the three edges be (u1, v1), (u2, v2), (u3, v3).
    # The sum of these columns is a vector where each row index is a worker.
    # The value at row i is the number of edges in the set that contain worker i.
    # We need exactly three 2s, meaning three workers are each involved in 
    # exactly two of the three edges, and all other workers are involved in 0.
    # This implies the three edges must form a triangle (a cycle of length 3).
    # If the edges are (a, b), (b, c), (c, a), then:
    # Worker a is in (a, b) and (c, a) -> count 2
    # Worker b is in (a, b) and (b, c) -> count 2
    # Worker c is in (b, c) and (c, a) -> count 2
    # All other workers are in 0 of these edges.
    # Thus, the problem is simply to find a triangle in the graph.
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # We need the lexicographically smallest triplet (a, b, c)
    # Iterate through all possible triplets (a, b, c) with a < b < c
    # and check if they form a triangle.
    # Since n <= 2000, O(n^3) is too slow, but we can iterate over edges.
    
    triangles = []
    # To find the lexicographically smallest, we can iterate a from 1 to n
    for a in range(1, n + 1):
        # For a fixed a, find all b > a such that (a, b) is an edge
        neighbors_a = sorted([b for b in adj[a] if b > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Check for c > b such that (a, c) and (b, c) are edges
            # We want the smallest a, then b, then c.
            # So we can iterate a, then b, then check common neighbors c > b.
            for c in neighbors_a[i+1:]:
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

solve()
