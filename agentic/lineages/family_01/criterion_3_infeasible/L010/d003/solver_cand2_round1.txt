```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
        idx += 2
    
    # A deadlock is three edges (u1, v1), (u2, v2), (u3, v3)
    # such that the set of vertices {u1, v1, u2, v2, u3, v3}
    # contains three vertices each appearing twice.
    # This is equivalent to saying the three edges form a triangle
    # in the line graph, or more simply, the three edges form a 
    # structure where the vertices are {a, b, c} and the edges are
    # (a, b), (b, c), (a, c).
    # Wait, the condition is: 3 workers appear twice, others 0.
    # If we pick edges (a, b), (b, c), (a, c), then:
    # a appears in (a, b) and (a, c) -> count 2
    # b appears in (a, b) and (b, c) -> count 2
    # c appears in (b, c) and (a, c) -> count 2
    # This is exactly a triangle!
    
    # Let's check if any triangle exists.
    # The problem says "exactly three workers will have a tally of 2".
    # A triangle (a, b), (b, c), (a, c) satisfies this.
    
    triangles = []
    for a in range(1, n + 1):
        adj_a = set(adj[a])
        for b in adj[a]:
            if b > a:
                for c in adj[b]:
                    if c > b and c in adj_a:
                        triangles.append(tuple(sorted((a, b, c))))
    
    triangles.sort()
    if triangles:
        print(f"{triangles[0][0]} {triangles[0][1]} {triangles[0][2]}")

solve()
```
