import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u][v] = True
        adj[v][u] = True
    
    # We need to find a triangle (a, b, c) such that
    # edges (a,b), (b,c), (c,a) exist.
    # We want the lexicographically smallest sorted triple.
    
    # Iterate 'a' from 1 to n
    for a in range(1, n + 1):
        # Iterate 'b' from a+1 to n
        for b in range(a + 1, n + 1):
            if adj[a][b]:
                # If edge (a, b) exists, check for any c > b
                # such that (a, c) and (b, c) also exist.
                for c in range(b + 1, n + 1):
                    if adj[a][c] and adj[b][c]:
                        print(a, b, c)
                        return

if __name__ == '__main__':
    solve()
