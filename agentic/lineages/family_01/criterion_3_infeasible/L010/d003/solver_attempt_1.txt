```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest (a, b, c)
    # We want to minimize a, then b, then c.
    # Iterate through all edges (u, v) as potential (a, b)
    # Since we need a < b < c, we iterate a from 1 to n
    # Then iterate b in adj[a] such that b > a
    # Then check if there exists c in adj[a] intersect adj[b] such that c > b
    
    edges.sort()
    
    # To optimize, iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Find common neighbors c > b
            # Intersection of adj[a] and adj[b]
            # Since adj sets are small or we can use bitsets/intersection
            # For n=2000, simple intersection is fine
            common = adj[a].intersection(adj[b])
            
            # We need the smallest c > b
            best_c = float('inf')
            found = False
            for c in common:
                if c > b:
                    if c < best_c:
                        best_c = c
                        found = True
            
            if found:
                print(f"{a} {b} {best_c}")
                return

solve()
```
