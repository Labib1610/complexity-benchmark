import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Get neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        # Iterate b from neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # We need a c such that c > b and c is in adj[a] and adj[b]
            # Intersection of adj[a] and adj[b]
            # Since we want the smallest c, we can iterate through sorted neighbors of b
            # that are > b
            neighbors_b = adj[b]
            
            # Optimization: iterate through the smaller set
            if len(neighbors_a) < len(neighbors_b):
                for c in neighbors_a:
                    if c > b and c in neighbors_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                # Sort neighbors of b to find smallest c
                sorted_neighbors_b = sorted([v for v in neighbors_b if v > b])
                for c in sorted_neighbors_b:
                    if c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
