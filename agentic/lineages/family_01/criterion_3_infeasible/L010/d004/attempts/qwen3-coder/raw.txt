```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to process in order
    edges.sort()
    
    # For each edge (u,v), look for a node w such that (u,w) and (v,w) exist
    for u, v in edges:
        # Get neighbors of u and v
        neighbors_u = adj[u]
        neighbors_v = adj[v]
        # Find common neighbors
        common = neighbors_u & neighbors_v
        if common:
            # Get the smallest common neighbor
            w = min(common)
            # Output in sorted order
            triple = sorted([u, v, w])
            print(triple[0], triple[1], triple[2])
            return

if __name__ == "__main__":
    main()
```
