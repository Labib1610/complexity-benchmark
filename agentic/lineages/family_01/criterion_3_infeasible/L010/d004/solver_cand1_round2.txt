```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # A deadlock group is a triangle (3-clique).
    # We need to find the lexicographically smallest (a, b, c) such that
    # a < b < c and (a,b), (b,c), (a,c) are all in the log.
    
    # To find the lexicographically smallest, we iterate through possible a, b, c.
    # Since n <= 2000, we can iterate through all edges (u, v) and check
    # for a common neighbor w.
    
    found_triangles = []
    
    # Sort edges to help find lexicographically smallest
    edges.sort()
    
    # To optimize, iterate through each edge (u, v) and check common neighbors
    for i in range(m):
        u, v = edges[i]
        # We want to find w such that (u, v, w) form a triangle.
        # The sorted order of the triangle will be (a, b, c).
        # We check all w. To keep it efficient, iterate over the smaller adjacency list.
        if len(adj[u]) < len(adj[v]):
            common = adj[u].intersection(adj[v])
        else:
            common = adj[v].intersection(adj[u])
            
        for w in common:
            # Triangle is {u, v, w}. Sort them to get (a, b, c)
            t = sorted([u, v, w])
            found_triangles.append(tuple(t))
            
    # The problem guarantees at least one exists.
    # We need the lexicographically smallest.
    print(*(min(found_triangles)))

if __name__ == '__main__':
    solve()
```
