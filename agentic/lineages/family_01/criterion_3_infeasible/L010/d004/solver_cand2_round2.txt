import sys

def solve():
    # The problem asks for three distinct pairs (u1, v1), (u2, v2), (u3, v3)
    # such that their symmetric difference is empty.
    # Let the sets be {u1, v1}, {u2, v2}, {u3, v3}.
    # The symmetric difference is empty if and only if each element appears
    # an even number of times across the three sets.
    # Since there are 3 sets of 2 elements (6 total slots), and each element
    # must appear an even number of times, each of the 3 distinct workers
    # involved must appear exactly twice.
    # This means the three pairs must form a triangle: (a, b), (b, c), (c, a).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].append(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through all possible triangles (a, b, c).
    # A triangle exists if (a, b), (b, c), and (a, c) are all in the input.
    
    # Sort adjacency lists to help with finding triangles efficiently
    for i in range(1, n + 1):
        adj[i].sort()
        
    # Use a set for O(1) lookup
    edge_set = set()
    for u, v in edges:
        edge_set.add((u, v))
        
    found_triangles = []
    
    # Iterate through all pairs (u, v) and check for a common neighbor w
    # To ensure lexicographical order, we iterate a, b, c
    for a in range(1, n + 1):
        for i in range(len(adj[a])):
            b = adj[a][i]
            # Now look for c such that (b, c) and (a, c) exist
            # c must be > b
            for c in adj[b]:
                if c > b:
                    if (a, c) in edge_set:
                        found_triangles.append(tuple(sorted((a, b, c))))
    
    found_triangles.sort()
    res = found_triangles[0]
    print(f"{res[0]} {res[1]} {res[2]}")

solve()
