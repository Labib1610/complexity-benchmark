import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # A deadlock of 3 reports (u1, v1), (u2, v2), (u3, v3)
    # results in all workers being 'cleared' if each worker appears
    # an even number of times across the three reports.
    # Since there are 3 reports (6 endpoints), and each worker must appear
    # an even number of times, the only way to have 3 reports is if
    # the 3 reports form a triangle (u, v), (v, w), (w, u).
    # In this case, u, v, and w each appear twice.
    
    deadlocks = []
    
    # Iterate through all possible triangles (u, v, w)
    # To find the lexicographically smallest (a, b, c),
    # we iterate u from 1 to n, v from u+1 to n, w from v+1 to n.
    for u in range(1, n + 1):
        for v in sorted(list(adj[u])):
            if v <= u: continue
            # Check for w such that (u, w) and (v, w) exist
            # To optimize, intersect adj[u] and adj[v]
            common = adj[u].intersection(adj[v])
            for w in sorted(list(common)):
                if w <= v: continue
                # Found a triangle (u, v, w)
                # The workers involved are u, v, w.
                # The problem asks for the workers whose status was flipped.
                # In a triangle (u, v), (v, w), (w, u), each worker is flipped twice.
                deadlocks.append(tuple(sorted((u, v, w))))
    
    if deadlocks:
        deadlocks.sort()
        print(f"{deadlocks[0][0]} {deadlocks[0][1]} {deadlocks[0][2]}")

solve()
