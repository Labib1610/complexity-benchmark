```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we iterate b from a+1 to n-1.
    # If (a, b) is an edge, we check if there exists c > b such that
    # (a, c) and (b, c) are edges.
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through all possible a
    for a in range(1, n + 1):
        # Iterate through all b such that (a, b) is an edge and b > a
        for b in adj[a]:
            if b <= a:
                continue
            
            # Now we need to find the smallest c > b such that
            # c is in adj[a] and c is in adj[b]
            # Since adj lists are sorted, we can use two pointers or set intersection
            # Given the constraints, set intersection or simple iteration is fine
            
            # Optimization: iterate through the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                for c in adj[a]:
                    if c > b and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in adj[b]:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

solve()
```
