import sys

def solve():
    # Read W and R
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    W = int(input_data[0])
    R = int(input_data[1])
    
    # A deadlock of 3 reports (u1, v1), (u2, v2), (u3, v3)
    # means the sum of vectors is 0 mod 2.
    # Each worker ID appears in the sum an even number of times.
    # Since there are 3 reports, each with 2 workers, there are 6 total slots.
    # For the sum to be 0, each of the 3 workers involved must appear exactly twice.
    # This forms a triangle: (a, b), (b, c), (c, a).
    
    adj = [set() for _ in range(W + 1)]
    idx = 2
    for _ in range(R):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a, b, c such that (a,b), (b,c), (c,a) are all in the log.
    # To find the lexicographically smallest (a, b, c), we iterate a from 1 to W,
    # b from a+1 to W, and c from b+1 to W.
    
    for a in range(1, W + 1):
        for b in sorted(list(adj[a])):
            if b <= a:
                continue
            # Check if there exists c such that (b, c) and (c, a) are in the log
            # Intersection of adj[a] and adj[b] gives all possible c's
            common = adj[a].intersection(adj[b])
            if common:
                c = min(common)
                # The triplet is (a, b, c) sorted
                res = sorted([a, b, c])
                print(f"{res[0]} {res[1]} {res[2]}")
                return

solve()
