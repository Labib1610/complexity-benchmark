import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    bits = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        bits[u] |= 1 << (v - 1)
        bits[v] |= 1 << (u - 1)

    for a in range(1, n + 1):
        ba = bits[a]
        nb = ba & ~((1 << a) - 1)
        while nb:
            lsb = nb & -nb
            b = lsb.bit_length()
            inter = ba & bits[b] & ~((1 << b) - 1)
            if inter:
                c = (inter & -inter).bit_length()
                sys.stdout.write(f"{a} {b} {c}")
                return
            nb ^= lsb

if __name__ == "__main__":
    solve()
