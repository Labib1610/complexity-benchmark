import sys

def solve() -> None:
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Iterate to find lexicographically smallest triple
    for a in range(1, n - 1):
        # adj[a] contains neighbors of a
        # We need to find b, c in adj[a] such that b < c and b and c are adjacent
        neighbors = adj[a]
        # To find smallest b, we iterate b in increasing order
        # b must be > a to satisfy a < b < c
        # We iterate b from a+1 up to n
        for b in range(a + 1, n + 1):
            if b in neighbors:
                # Now we need c in neighbors such that c > b and c in adj[b]
                # To find smallest c, iterate c from b+1 up to n
                for c in range(b + 1, n + 1):
                    if c in neighbors and c in adj[b]:
                        print(a, b, c)
                        return

if __name__ == '__main__':
    solve()
