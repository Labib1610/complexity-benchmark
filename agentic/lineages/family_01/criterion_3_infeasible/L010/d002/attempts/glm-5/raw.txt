```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    deg = [0] * (n + 1)
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u][v] = True
        adj[v][u] = True
        deg[u] += 1
        deg[v] += 1
    
    # Iterate through all pairs of nodes (u, v) in lexicographical order
    for u in range(1, n + 1):
        for v in range(u + 1, n + 1):
            if not adj[u][v]:
                continue
            # We have an edge u-v. We need a third node w such that
            # u-w and v-w are also edges.
            # To find the lexicographically smallest triangle,
            # we iterate w in increasing order.
            for w in range(v + 1, n + 1):
                if adj[u][w] and adj[v][w]:
                    print(u, v, w)
                    return

if __name__ == '__main__':
    solve()
```
