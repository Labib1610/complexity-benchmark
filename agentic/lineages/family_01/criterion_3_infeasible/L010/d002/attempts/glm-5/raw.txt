import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to process in order (helps find lexicographically smallest)
    edges.sort()
    
    # For each edge (u, v), check all common neighbors w
    # We want to find the smallest (u, v, w) such that exactly 3 edges exist among {u, v, w}
    # Since edges are sorted, the first valid triple found will be lexicographically smallest
    
    for u, v in edges:
        # Find common neighbors w where w > v (to ensure u < v < w)
        # We iterate through the smaller adjacency set for efficiency
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        
        for w in adj[u]:
            if w > v and w in adj[v]:
                # Found a triangle (u, v, w)
                # Count edges among these three vertices
                # We know (u,v), (u,w), (v,w) exist = 3 edges
                # Check if there are exactly 3 edges (no extra edges)
                # Since we found a triangle, there are at least 3 edges
                # We need to verify there are exactly 3
                
                # Count edges: (u,v), (u,w), (v,w) = 3
                # Check if any other pair has an edge (should not)
                # But since we only have these three pairs in a triangle, count = 3
                
                # Actually, we need to count edges among {u, v, w}
                # The pairs are: (u,v), (u,w), (v,w)
                # We already know all three exist (triangle)
                # So count = 3, which is exactly what we need
                
                # Output the result
                print(u, v, w)
                return
    
    # If no triangle found (shouldn't happen per problem statement)
    # But let's handle it anyway
    print(-1, -1, -1)

solve()
