import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # A triad {a, b, c} is a deadlock if every 2-coloring of the triad
    # results in at least one monochromatic edge.
    # This happens if and only if the subgraph induced by {a, b, c}
    # contains at least 3 edges (i.e., it is a triangle).
    # If it's a triangle, any 2-coloring of 3 vertices will have at least
    # one edge monochromatic (Pigeonhole Principle).
    # We need the lexicographically smallest triangle.
    
    # Sort edges to help find the smallest triangle
    edges.sort()
    
    # Iterate through all possible a, b, c
    # To find the lexicographically smallest, we iterate a from 1 to n,
    # b from a+1 to n, and c from b+1 to n.
    # Check if {a, b}, {b, c}, and {a, c} are all in the adjacency set.
    
    for a in range(1, n + 1):
        # Optimization: only consider b such that a is connected to b
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now check for c > b such that c is connected to both a and b
            # Intersection of neighbors of a and neighbors of b
            common = adj[a].intersection(adj[b])
            for c in sorted(list(common)):
                if c > b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
