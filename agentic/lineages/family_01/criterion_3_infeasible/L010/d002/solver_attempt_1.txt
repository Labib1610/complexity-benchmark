import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
    
    # A meal {a, b, c} is perfectly balanced if it contains exactly 3 pairings.
    # This means the subgraph induced by {a, b, c} must have exactly 3 edges.
    # This is equivalent to saying {a, b, c} forms a triangle (a clique of size 3).
    # We need to find the lexicographically smallest triangle.
    
    # Sort edges to help find triangles efficiently
    edges.sort()
    
    # To find the lexicographically smallest triangle (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then c.
    # Since n <= 2000, we can iterate through all edges (a, b) and check for common neighbors c.
    
    # Pre-sort adjacency lists for consistent ordering
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # To find the lexicographically smallest, we iterate a from 1 to n-2,
    # b from a+1 to n-1, and c from b+1 to n.
    # A triangle exists if c is in adj[a] and c is in adj[b].
    
    for a in range(1, n + 1):
        adj_a = adj[a]
        for i in range(len(adj_a)):
            b = adj_a[i]
            if b <= a:
                continue
            
            # Now we need a c > b such that c is in adj[a] and adj[b]
            # Since adj lists are sorted, we can use intersection
            adj_b = adj[b]
            
            # Find common neighbors c > b
            ptr_a = i + 1
            ptr_b = 0
            while ptr_a < len(adj_a) and ptr_b < len(adj_b):
                val_a = adj_a[ptr_a]
                val_b = adj_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

solve()
