```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Store conflicts as an adjacency list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # A deadlock is a set of 3 edges (a,b), (a,c), (b,c)
    # This is equivalent to finding a triangle in the graph.
    # We need to find the lexicographically smallest (a, b, c)
    # where a < b < c.
    
    found_triangles = []
    
    # Iterate through all edges (u, v) and check for a common neighbor w
    # To ensure lexicographical order, we iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors = sorted(list(adj[a]))
        len_n = len(neighbors)
        for i in range(len_n):
            b = neighbors[i]
            # Check common neighbors of a and b
            # We only care about c > b
            # Common neighbors are in adj[a] and adj[b]
            # Since we want a < b < c, we look for c in adj[a] and adj[b]
            # where c > b
            common = adj[a].intersection(adj[b])
            for c in common:
                if c > b:
                    found_triangles.append((a, b, c))
        
        if found_triangles:
            break
            
    found_triangles.sort()
    a, b, c = found_triangles[0]
    print(f"{a} {b} {c}")

if __name__ == '__main__':
    solve()
```
