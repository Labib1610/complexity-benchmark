```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
    
    # A perfectly balanced meal of 3 recipes (a, b, c) has exactly 3 pairings.
    # This means the subgraph induced by {a, b, c} must have 3 edges.
    # This is equivalent to saying {a, b, c} forms a triangle (a clique of size 3).
    
    # Sort edges to help with lexicographical order
    edges.sort()
    
    # Iterate through all edges (u, v) and check for a common neighbor w
    # To ensure lexicographical order, we iterate through possible a, b, c
    # A triangle (a, b, c) with a < b < c exists if (a, b), (b, c), and (a, c) are edges.
    
    # To optimize, iterate over edges (u, v) and find common neighbors w
    # Since we need the lexicographically smallest (a, b, c), we can iterate
    # through all possible a, then b, then c.
    
    for a in range(1, n + 1):
        # Get neighbors of a
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need a c such that c > b and c is connected to both a and b
            # Intersection of adj[a] and adj[b]
            common = sorted(list(adj[a].intersection(adj[b])))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
