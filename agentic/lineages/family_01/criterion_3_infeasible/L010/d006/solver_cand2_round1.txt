```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [{} for _ in range(n + 1)]
    edges = []
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u][v] = i
        adj[v][u] = i
        idx += 2
    
    # A deadlock is R1 ^ R2 ^ R3 = empty set.
    # Let R1 = {u, v}, R2 = {v, w}, R3 = {u, w}.
    # Then R1 ^ R2 = {u, w}, and (R1 ^ R2) ^ R3 = empty set.
    # This is exactly a triangle in the graph.
    
    best_res = None
    
    # Iterate over all edges (u, v) and check for a common neighbor w
    for i in range(m):
        u, v = edges[i]
        # To find the lexicographically smallest, we need to check all triangles
        # and keep the best one.
        # Optimization: iterate over the smaller adjacency list
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        
        for w in adj[u]:
            if w in adj[v]:
                # Found a triangle (u, v, w)
                res = sorted([u, v, w])
                if best_res is None or res < best_res:
                    best_res = res
                    
    print(*(best_res))

solve()
```
