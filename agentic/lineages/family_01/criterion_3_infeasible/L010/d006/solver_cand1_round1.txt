import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # A deadlock is 3 edges (u1, v1), (u2, v2), (u3, v3)
    # Let the set of vertices be {a, b, c}.
    # The only way to satisfy the condition is if the edges are (a, b), (b, c), (a, c).
    # This is a triangle in the graph.
    # The condition: "if you take any two of these three reports, the workers who are 
    # named in exactly one of those two reports must be precisely the two workers 
    # named in the third report."
    # Let edges be e1={a,b}, e2={b,c}, e3={a,c}.
    # e1, e2: symmetric difference is {a, c} = e3. Correct.
    # e1, e3: symmetric difference is {b, c} = e2. Correct.
    # e2, e3: symmetric difference is {a, b} = e1. Correct.
    # So a deadlock is simply a triangle (a, b, c).
    
    best_res = None
    
    # To find the lexicographically smallest sorted sequence, we iterate through
    # all triangles and keep the minimum.
    # Since n <= 2000, we can iterate over edges (u, v) and check common neighbors.
    
    for u in range(1, n + 1):
        adj_u = sorted(list(adj[u]))
        for i in range(len(adj_u)):
            v = adj_u[i]
            if v <= u: continue
            
            # Find common neighbors w > v
            # To optimize, intersect adj[u] and adj[v]
            common = adj[u].intersection(adj[v])
            for w in common:
                if w > v:
                    res = tuple(sorted([u, v, w]))
                    if best_res is None or res < best_res:
                        best_res = res
                        
    print(*(best_res))

solve()
