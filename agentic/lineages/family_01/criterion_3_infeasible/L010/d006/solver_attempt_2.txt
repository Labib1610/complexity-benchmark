import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # A group of three servers {a, b, c} is valid if every pair is connected.
    # This means {a, b, c} must form a triangle (a clique of size 3).
    # The condition "survive the loss of any one member" means if we remove
    # one, the remaining two must be connected.
    # If {a, b, c} is a triangle, then removing any one leaves the other two
    # connected by the edge that was part of the triangle.
    # Thus, any triangle is a valid group.
    
    # We need the lexicographically smallest triangle (a, b, c) with a < b < c.
    
    # Sort neighbors to help find lexicographically smallest
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = adj_list[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Check for c > b such that (a, c) and (b, c) exist
            # To be lexicographically smallest, we want smallest a, then b, then c.
            # Since we iterate a and b in increasing order, we just need the
            # smallest c > b that is connected to both a and b.
            
            # Intersection of neighbors of a and neighbors of b
            # Since neighbors are sorted, we can find common elements efficiently
            set_a = adj[a]
            set_b = adj[b]
            
            # We only care about c > b
            for c in neighbors_a[i+1:]:
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
