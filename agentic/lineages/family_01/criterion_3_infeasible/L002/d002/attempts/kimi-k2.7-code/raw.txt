import sys

def solve():
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    full = (1 << n) - 1
    mask_gt = [0] * n
    for i in range(n):
        mask_gt[i] = full ^ ((1 << (i + 1)) - 1)
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    out = sys.stdout.write
    for a in range(n):
        Na = adj[a] & mask_gt[a]
        if not Na:
            continue
        temp = Na
        while temp:
            lsb = temp & -temp
            b = lsb.bit_length() - 1
            temp ^= lsb
            cand = Na & adj[b] & mask_gt[b]
            if cand:
                c = (cand & -cand).bit_length() - 1
                out(f"{a + 1} {b + 1} {c + 1}\n")
                return

if __name__ == "__main__":
    solve()
