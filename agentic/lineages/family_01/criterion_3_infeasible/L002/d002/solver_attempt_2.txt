```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically first (a, b, c)
    # We want smallest a, then smallest b, then smallest c.
    # Iterate through all edges (u, v) as potential (a, b)
    # Then check if there exists a node w such that w > v and w is connected to both u and v
    
    edges.sort()
    
    for a, b in edges:
        # We need a third node c such that a-c and b-c exist.
        # Since we want lexicographically first, we check possible c values.
        # c must be connected to both a and b.
        # To ensure a < b < c, we look for c > b.
        
        # Intersection of neighbors of a and b
        common = adj[a].intersection(adj[b])
        
        # Filter for c > b
        candidates = [c for c in common if c > b]
        
        if candidates:
            print(f"{a} {b} {min(candidates)}")
            return

if __name__ == '__main__':
    solve()
```
