import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically first trio (a, b, c) with a < b < c,
    # we iterate through all edges (a, b) and check for a common neighbor c.
    # To ensure lexicographical order, we sort the edges and check neighbors.
    
    edges.sort()
    
    best_trio = None
    
    # Optimization: iterate through sorted edges (a, b)
    # For each edge (a, b), find common neighbors c.
    # Since we want the lexicographically smallest (a, b, c),
    # we can iterate through all possible 'a' from 1 to n.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a < b < c.
            # Check neighbors of b that are also neighbors of a.
            # To keep it efficient, intersect the smaller adjacency list.
            if len(adj[a]) < len(adj[b]):
                common = [c for c in adj[a] if c in adj[b]]
            else:
                common = [c for c in adj[b] if c in adj[a]]
            
            for c in common:
                if c > b:
                    trio = tuple(sorted((a, b, c)))
                    if best_trio is None or trio < best_trio:
                        best_trio = trio
        
        # If we found a trio starting with 'a', and we have checked all 
        # potential 'b's for this 'a', we can potentially stop if we 
        # are sure no smaller 'a' exists.
        if best_trio and best_trio[0] < a:
            break
            
    if best_trio:
        print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

solve()
