import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Iterate to find the lexicographically smallest triangle
    for a in range(1, n + 1):
        # We need b > a and c > b.
        # To find the lexicographically smallest (a, b, c), for a fixed a,
        # we iterate b in increasing order from a+1 to n.
        for b in range(a + 1, n + 1):
            if b not in adj[a]:
                continue
            # If edge (a, b) exists, check for c > b such that (a, c) and (b, c) exist.
            # We iterate c in increasing order from b+1 to n.
            for c in range(b + 1, n + 1):
                if c in adj[a] and c in adj[b]:
                    print(a, b, c)
                    return

if __name__ == '__main__':
    solve()
