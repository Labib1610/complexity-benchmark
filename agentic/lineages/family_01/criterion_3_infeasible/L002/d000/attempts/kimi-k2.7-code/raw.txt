```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    full = (1 << n) - 1
    for a in range(1, n - 1):
        cand = adj[a] & (full ^ ((1 << a) - 1))  # neighbors > a
        if cand.bit_count() < 2:
            continue
        rem = cand
        while rem:
            lsb = rem & -rem
            b = lsb.bit_length()                    # vertex number from bit
            cands = adj[b] & cand & (full ^ ((1 << b) - 1))  # neighbors > b
            if cands:
                c = (cands & -cands).bit_length()
                sys.stdout.write(f"{a} {b} {c}")
                return
            rem ^= lsb

if __name__ == "__main__":
    solve()
```
