import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically first triangle
    edges.sort()
    
    # To find the lexicographically first (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists a c > b
    # such that (a, c) and (b, c) are edges.
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b from a+1 to n, then check for common neighbors c > b.
    
    # Pre-sort adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We have edge (a, b). Now look for c > b such that (a, c) and (b, c)
            # Since neighbors_a and neighbors_b are sorted, we can intersect them.
            neighbors_b = adj[b]
            
            # Find common neighbors c > b
            ptr_a = 0
            ptr_b = 0
            while ptr_a < len(neighbors_a) and ptr_b < len(neighbors_b):
                val_a = neighbors_a[ptr_a]
                val_b = neighbors_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

if __name__ == '__main__':
    solve()
