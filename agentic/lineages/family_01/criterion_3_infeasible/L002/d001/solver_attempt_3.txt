import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # We need to find three workers {a, b, c} such that 
    # all three pairs (a, b), (b, c), (a, c) are in the conflict list.
    # This is equivalent to finding a triangle in the conflict graph.
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest {a, b, c},
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we check if (a, b), (b, c), and (a, c) exist.
    
    # Sort adjacency lists to help with ordered search
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through all possible a
    for a in range(1, n + 1):
        # Iterate through all neighbors b of a where b > a
        for b in adj[a]:
            if b <= a:
                continue
            
            # Now we need a c such that c > b and c is a neighbor of both a and b
            # We can intersect the sorted lists adj[a] and adj[b]
            # Since adj[a] and adj[b] are sorted, we can find common elements efficiently
            
            # Optimization: only check c > b
            # We can use the fact that adj[a] and adj[b] are sorted
            ptr_a = 0
            ptr_b = 0
            list_a = adj[a]
            list_b = adj[b]
            
            while ptr_a < len(list_a) and ptr_b < len(list_b):
                val_a = list_a[ptr_a]
                val_b = list_b[ptr_b]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    ptr_a += 1
                    ptr_b += 1
                elif val_a < val_b:
                    ptr_a += 1
                else:
                    ptr_b += 1

if __name__ == '__main__':
    solve()
