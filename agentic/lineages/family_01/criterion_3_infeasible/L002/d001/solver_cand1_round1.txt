import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        idx += 2
        
    # We need to find the lexicographically smallest (a, b, c)
    # such that a < b < c and (a,b), (a,c), (b,c) are in the conflict set.
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted(list(adj[a]))
        
        # Iterate through pairs of neighbors (b, c)
        # Since neighbors_a is sorted, b and c will be picked in increasing order
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We only care about b > a, which is guaranteed by neighbors_a
            
            # Check if there exists c > b such that c is a neighbor of a AND c is a neighbor of b
            # To find the smallest c, we look at neighbors of b that are > b
            # and check if they are also in neighbors_a
            
            # Optimization: intersect neighbors of a and neighbors of b
            # Only consider neighbors > b
            neighbors_b = adj[b]
            
            # Find common neighbors c > b
            # We want the smallest c, so iterate through sorted neighbors_a[i+1:]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
