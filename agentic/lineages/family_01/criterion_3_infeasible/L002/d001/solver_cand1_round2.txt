```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, b from a+1 to n, and c from b+1 to n.
    # To optimize, we only check b in adj[a] and c in (adj[a] & adj[b]).
    
    # Sort adjacency lists to help with lexicographical order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Only consider neighbors b > a
        valid_b = [b for b in neighbors_a if b > a]
        
        for i in range(len(valid_b)):
            b = valid_b[i]
            # Find common neighbors c > b
            # Using set intersection for efficiency
            common = adj[a].intersection(adj[b])
            
            # Filter and sort common neighbors to find the smallest c
            valid_c = sorted([c for c in common if c > b])
            
            if valid_c:
                print(f"{a} {b} {valid_c[0]}")
                return

if __name__ == '__main__':
    solve()
```
