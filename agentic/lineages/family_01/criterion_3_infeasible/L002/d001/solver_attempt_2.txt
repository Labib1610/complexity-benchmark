import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    best_trio = (float('inf'), float('inf'), float('inf'))
    
    # To find the lexicographically smallest trio (X, Y, Z) with X < Y < Z,
    # we iterate through X from 1 to n, then Y from X+1 to n,
    # then check if there exists Z > Y such that Z is in adj[X] and adj[Y].
    # Since we want the minimum X*10^8 + Y*10^4 + Z, we can stop early.
    
    for x in range(1, n + 1):
        if x >= best_trio[0]:
            break
        
        neighbors_x = sorted(list(adj[x]))
        for y in neighbors_x:
            if y <= x:
                continue
            
            # Current candidate prefix (x, y)
            # We need to find the smallest z > y such that z is in adj[x] and adj[y]
            # To optimize, intersect the two sets
            common = adj[x].intersection(adj[y])
            
            for z in common:
                if z > y:
                    # Found a trio (x, y, z)
                    # Check if it's better than current best
                    if (x, y, z) < best_trio:
                        best_trio = (x, y, z)
            
            # If we found a trio with this x, y, we don't need to check larger y for this x
            # because any trio (x, y', z') with y' > y would have a larger value.
            # However, we must ensure we checked all z for this (x, y).
            # The current logic finds the smallest z for a fixed (x, y).
            # If we found any z, then (x, y, min_z) is the best for this (x, y).
            # Since we want the global minimum, we can break if we found a trio for this x.
        
        if best_trio[0] != float('inf'):
            break
            
    print(f"{best_trio[0]} {best_trio[1]} {best_trio[2]}")

if __name__ == '__main__':
    solve()
