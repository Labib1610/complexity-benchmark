 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = map(int, data)
    n = next(it)
    m = next(it)

    adj = [0] * (n + 1)
    up = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = next(it)
        v = next(it)
        if u > v:
            u, v = v, u
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)
        up[u].append(v)

    for i in range(1, n + 1):
        up[i].sort()

    full = (1 << n) - 1
    suffix = [0] * (n + 1)
    for b in range(1, n + 1):
        suffix[b] = full ^ ((1 << b) - 1)

    out = sys.stdout
    for a in range(1, n + 1):
        abits = adj[a]
        for b in up[a]:
            inter = abits & adj[b]
            if inter:
                cand = inter & suffix[b]
                if cand:
                    c = (cand & -cand).bit_length()
                    out.write(f"{a} {b} {c}\n")
                    return

if __name__ == "__main__":
    solve()
```
