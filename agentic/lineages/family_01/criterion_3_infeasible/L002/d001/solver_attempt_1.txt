```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find three workers {a, b, c} such that there are 3 edges
    # among them. This means the subgraph induced by {a, b, c} must be a triangle (K3).
    # We need the lexicographically smallest triplet (a, b, c).
    
    # Iterate through all possible triplets (a, b, c) with 1 <= a < b < c <= n
    # To optimize, iterate a, then b, then check if c exists.
    # A triangle exists if a-b, b-c, and a-c are all edges.
    
    for a in range(1, n + 1):
        # Get neighbors of a
        neighbors_a = sorted(list(adj[a]))
        len_a = len(neighbors_a)
        if len_a < 2:
            continue
            
        for i in range(len_a):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that b < c and c is connected to both a and b
            # c must be in adj[a] and adj[b]
            # To ensure lexicographical order, we check c in increasing order
            
            # Intersection of adj[a] and adj[b]
            # Since we need the smallest triplet, we iterate through sorted neighbors
            # of a that are > b
            for j in range(i + 1, len_a):
                c = neighbors_a[j]
                if c in adj[b]:
                    # Found the smallest triplet
                    print(f"{a} {b} {c}")
                    return

solve()
```
