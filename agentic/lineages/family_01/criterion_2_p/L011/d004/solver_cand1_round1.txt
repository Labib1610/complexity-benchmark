```python
import sys

# The problem asks for x_i >= 1 such that:
# 1. If (u, v) is an edge, x_u != x_v.
# 2. sum(d_i * x_i) = 3m.
#
# Since x_i >= 1, let x_i = 1 + y_i, where y_i >= 0.
# sum(d_i * (1 + y_i)) = 3m
# sum(d_i) + sum(d_i * y_i) = 3m
# Since sum(d_i) = 2m, we have:
# 2m + sum(d_i * y_i) = 3m
# sum(d_i * y_i) = m
#
# We need to assign y_i >= 0 such that if (u, v) is an edge, x_u != x_v.
# To get the lexicographically smallest x_i, we want y_i to be 0 as much as possible.
# If we set y_i = 0 for all i, then x_i = 1 for all i. This violates the edge constraint
# if m > 0.
# If the graph is bipartite, we can color it with 1 and 2.
# Let the two sets be A and B. If we set x_i = 1 for i in A and x_i = 2 for i in B:
# sum(d_i * x_i) = sum_{i in A} d_i * 1 + sum_{i in B} d_i * 2
# Let S_A = sum_{i in A} d_i and S_B = sum_{i in B} d_i.
# S_A + S_B = 2m.
# We need S_A + 2*S_B = 3m.
# Subtracting: S_B = m.
# Since S_A + S_B = 2m, this implies S_A = m.
# So we need a bipartite coloring such that the sum of degrees in each partition is m.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    if m == 0:
        print("YES")
        print(*( [1] * n ))
        return

    color = [-1] * (n + 1)
    degrees = [len(adj[i]) for i in range(n + 1)]
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    sum_deg0 = sum(degrees[i] for i in range(1, n + 1) if color[i] == 0)
    sum_deg1 = sum(degrees[i] for i in range(1, n + 1) if color[i] == 1)
    
    # We need sum(d_i * x_i) = 3m.
    # If we use colors 1 and 2, we need sum of degrees of nodes with color 2 to be m.
    # Currently, sum_deg0 + sum_deg1 = 2m.
    # If sum_deg1 == m, we are good.
    # If sum_deg0 == m, we can swap colors.
    
    if sum_deg1 == m:
        res = [1 if color[i] == 0 else 2 for i in range(1, n + 1)]
    elif sum_deg0 == m:
        res = [1 if color[i] == 1 else 2 for i in range(1, n + 1)]
    else:
        print("NO")
        return

    print("YES")
    print(*(res))

solve()
```
