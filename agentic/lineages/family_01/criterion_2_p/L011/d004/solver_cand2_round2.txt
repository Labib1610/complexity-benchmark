```python
import sys

# The problem constraints imply that for any edge (u, v), the conditions on D(x_u) and D(x_v)
# must be satisfied.
# D(x_u) U D(x_v) contains exactly two elements whose product is 2.
# Since 1 is always a divisor, the only way to have two elements with product 2 is {1, 2}.
# Thus, {1, 2} must be a subset of D(x_u) U D(x_v).
# This means 2 must divide at least one of x_u or x_v.
# The symmetric difference D(x_u) Δ D(x_v) contains exactly one element.
# If x_u = 1, D(x_u) = {1}. If x_v = 2, D(x_v) = {1, 2}.
# D(1) U D(2) = {1, 2}, product 1*2 = 2. Correct.
# D(1) Δ D(2) = {2}, size 1. Correct.
# If x_u = 1, x_v = 1, D(1) U D(1) = {1}, no product 2.
# If x_u = 2, x_v = 2, D(2) U D(2) = {1, 2}, product 2. D(2) Δ D(2) = empty, size 0.
# Thus, the graph must be bipartite, with one set assigned 1 and the other 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    # Lexicographically smallest: 1 is smaller than 2.
    # We want x_i = 1 if possible.
    # The bipartite coloring gives two sets. We can swap colors if needed.
    # To make the sequence lexicographically smallest, we want 1s at the smallest indices.
    # Let count1 be the number of nodes with color 1, count2 with color 2.
    # If we assign color 1 -> 1 and color 2 -> 2, we get a sequence.
    # If we assign color 1 -> 2 and color 2 -> 1, we get another.
    # We compare them index by index.
    
    res1 = [0] * n
    res2 = [0] * n
    for i in range(1, n + 1):
        if color[i] == 1:
            res1[i-1] = 1
            res2[i-1] = 2
        else:
            res1[i-1] = 2
            res2[i-1] = 1
            
    if res1 <= res2:
        print(*(res1))
    else:
        print(*(res2))

solve()
```
