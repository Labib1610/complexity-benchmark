```python
import sys

# The condition f_u(f_v(1)) = 2 means:
# If f_v(1) = 1, then f_u(1) = 2.
# If f_v(1) = 2, then f_u(2) = 2.
# Since f_u is either identity (f(1)=1, f(2)=2) or inversion (f(1)=2, f(2)=1):
# If f_u is identity, f_u(1)=1, f_u(2)=2.
# If f_u is inversion, f_u(1)=2, f_u(2)=1.
# Let x_i = f_i(1).
# If f_u is identity, x_u = 1. If f_u is inversion, x_u = 2.
# The condition f_u(f_v(1)) = 2 is equivalent to:
# If x_v = 1, then x_u = 2.
# If x_v = 2, then f_u(2) = 2, which means x_u = 1.
# In both cases, x_u != x_v.
# This is a standard 2-coloring problem on a graph.
# We want the lexicographically smallest sequence (x_1, ..., x_n).
# To minimize the sequence, we want x_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # x_i can be 1 or 2. x_u != x_v.
    # This is bipartite coloring.
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To make the sequence lexicographically smallest,
            # we prefer color 1 for the smallest node in the component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
