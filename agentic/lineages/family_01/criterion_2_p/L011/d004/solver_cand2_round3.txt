import sys

# The determinant of the matrix [[2*x_u, 3], [3, x_v]] is (2*x_u * x_v) - 9.
# The requirement is: 2*x_u * x_v - 9 = 3*(x_u + x_v) - 14
# 2*x_u * x_v - 3*x_u - 3*x_v + 5 = 0
# Multiply by 2: 4*x_u * x_v - 6*x_u - 6*x_v + 10 = 0
# (2*x_u - 3) * (2*x_v - 3) - 9 + 10 = 0
# (2*x_u - 3) * (2*x_v - 3) = -1
# Since x_u, x_v >= 1, 2*x_u - 3 can be -1 (if x_u=1) or >= 1 (if x_u >= 2).
# If 2*x_u - 3 = -1, then 2*x_v - 3 = 1 => x_v = 2.
# If 2*x_u - 3 = 1, then 2*x_v - 3 = -1 => x_v = 1.
# Thus, for every edge (u, v), one node must be 1 and the other must be 2.
# This means the graph must be bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    # We want to minimize sum(x_i * 3^(n-i)).
    # For each connected component, we have two choices:
    # Choice 0: nodes with color 0 get value A, nodes with color 1 get value B.
    # Choice 1: nodes with color 0 get value B, nodes with color 1 get value A.
    # Where {A, B} = {1, 2}.
    # For each component, calculate the cost for both choices and pick the smaller one.
    
    res = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            cost0 = 0
            cost1 = 0
            for node in component:
                # Choice 0: color 0 -> 1, color 1 -> 2
                val0 = 1 if color[node] == 0 else 2
                # Choice 1: color 0 -> 2, color 1 -> 1
                val1 = 2 if color[node] == 0 else 1
                
                # Cost is x_i * 3^(n-i)
                # Since 3^(n-i) is large, we compare lexicographically or by value
                # Actually, we can just compare the contribution of each component.
                # The weight 3^(n-i) is strictly decreasing with i.
                # For a component, find the smallest index node 'min_idx'.
                # The choice is determined by the value at 'min_idx'.
                min_idx = min(component)
                if color[min_idx] == 0:
                    # Choice 0: x[min_idx] = 1
                    # Choice 1: x[min_idx] = 2
                    # Since 1 < 2, Choice 0 is better.
                    for node in component:
                        res[node] = 1 if color[node] == 0 else 2
                else:
                    # color[min_idx] == 1
                    # Choice 0: x[min_idx] = 2
                    # Choice 1: x[min_idx] = 1
                    # Since 1 < 2, Choice 1 is better.
                    for node in component:
                        res[node] = 2 if color[node] == 0 else 1

    print("YES")
    print(*(res[1:]))

solve()
