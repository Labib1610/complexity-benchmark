import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The condition "one is twice the other" means if one is 1, the other must be 2.
    # This is a bipartite coloring problem where 1 and 2 are the two colors.
    # We want the lexicographically smallest sequence, so we prefer 1 over 2.
    
    colors = [0] * (n + 1) # 0: unvisited, 1: duration 1, 2: duration 2
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 to the current node
            stack = [(i, 1)]
            colors[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                target_neighbor_color = 3 - c
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = target_neighbor_color
                        stack.append((v, target_neighbor_color))
                    elif colors[v] != target_neighbor_color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
