```python
import sys

# The condition is x_u^{x_v} + x_v^{x_u} = 3.
# x_i must be a positive integer divisor of 2, so x_i can be 1 or 2.
# Possible pairs (x_u, x_v):
# If x_u = 1, x_v = 1: 1^1 + 1^1 = 2 != 3
# If x_u = 1, x_v = 2: 1^2 + 2^1 = 1 + 2 = 3 (Valid)
# If x_u = 2, x_v = 1: 2^1 + 1^2 = 2 + 1 = 3 (Valid)
# If x_u = 2, x_v = 2: 2^2 + 2^2 = 4 + 4 = 8 != 3
# Thus, for every link (u, v), one must be 1 and the other must be 2.
# This means the graph must be bipartite.
# To get the lexicographically smallest sequence, we want x_i = 1 whenever possible.
# We can use BFS/DFS to color the graph with 1s and 2s.
# For each connected component, we have two choices:
# 1. Set nodes in partition A to 1 and partition B to 2.
# 2. Set nodes in partition A to 2 and partition B to 1.
# To make the sequence lexicographically smallest, for each component,
# we check the smallest index node 's' in that component.
# If we set x_s = 1, we must set all nodes in its partition to 1 and the other to 2.
# If we set x_s = 2, we must set all nodes in its partition to 2 and the other to 1.
# Since 1 < 2, we always prefer x_s = 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # In this component, nodes with color 1 and color 2.
            # We want the smallest index in the component to be 1.
            # Let 'start_node' be the smallest index in this component.
            start_node = min(component)
            # If color[start_node] is 1, we assign 1 to color 1 and 2 to color 2.
            # If color[start_node] is 2, we assign 2 to color 1 and 1 to color 2.
            # Actually, we just need to ensure the smallest index gets 1.
            if color[start_node] == 1:
                for node in component:
                    res[node] = color[node]
            else:
                for node in component:
                    res[node] = 3 - color[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
```
