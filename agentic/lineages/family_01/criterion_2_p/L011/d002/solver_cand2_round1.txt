import sys

# The constraint: (x_u + x_v) / 2 = 1.5
# This implies x_u + x_v = 3.
# Since x_i is either 1 or 2, this means x_u != x_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (x_1, x_2, ..., x_n).
# To make it lexicographically smallest, we prefer 1 over 2.
# For each connected component, we have two possible colorings (swap 1s and 2s).
# We choose the coloring that results in a smaller value for the smallest index in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_idx = min(node for node, c in component)
            # Find the color assigned to min_idx
            color_of_min = 0
            for node, c in component:
                if node == min_idx:
                    color_of_min = c
                    break
            
            # We want the color of min_idx to be 1 if possible.
            # If color_of_min is 1, we keep it. If it's 2, we flip all colors.
            flip = (color_of_min == 2)
            for node, c in component:
                if flip:
                    ans[node] = 3 - c
                else:
                    ans[node] = c
                    
    print("YES")
    print(*(ans[1:]))

solve()
