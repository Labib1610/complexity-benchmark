```python
import sys

# The condition is: frequency_u^2 + frequency_v^2 = 5.
# Since frequencies are 1 or 2, 1^2 + 2^2 = 1 + 4 = 5.
# This means if u and v are connected, they must have different frequencies.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If a node is not colored, we try assigning it 1.
    # This is a greedy approach for bipartite coloring.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from this component
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
