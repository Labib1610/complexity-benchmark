import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unvisited, 1 = slot 1, 2 = slot 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest configuration, we iterate 1 to n.
    # If a component is not visited, we greedily assign the first node to slot 1.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            stack = [(i, 1)]
            color[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                # Neighbors must have the opposite color
                target_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = target_c
                        stack.append((v, target_c))
                    elif color[v] != target_c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
