```python
import sys

# The constraints are:
# 1. Link Interference: For edge (u, v), x_u * x_v = 2.
#    Since x_i is 1 or 2, this means one must be 1 and the other 2.
#    This implies the graph must be bipartite.
# 2. Local Load Balancing: For station u with degree d, sum of neighbors = d * (3 - x_u).
#    If x_u = 1, sum of neighbors = 2 * d. Since each neighbor is 2, this is always true.
#    If x_u = 2, sum of neighbors = 1 * d. Since each neighbor is 1, this is always true.
#    So condition 2 is satisfied if the graph is bipartite and condition 1 holds.
# 3. Extended Stability: For any walk of length 3 (u-v-w-z), x_u + x_v + x_w + x_z = 6.
#    If the graph is bipartite with partitions A and B:
#    If u is in A, v in B, w in A, z in B: x_u + x_v + x_w + x_z = x_A + x_B + x_A + x_B = 2*x_A + 2*x_B.
#    Since x_A and x_B are 1 and 2 (or vice versa), 2(1) + 2(2) = 6.
#    This is always satisfied if the graph is bipartite.
#
# Conclusion: The problem reduces to checking if the graph is bipartite.
# If bipartite, we have two possible assignments for each connected component:
# (A=1, B=2) or (A=2, B=1).
# To get the lexicographically smallest assignment, for each component,
# we pick the assignment that makes the smallest index in the component 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_node = min(node for node, c in component)
            # Find the color of the smallest index
            for node, c in component:
                if node == min_node:
                    min_color = c
                    break
            
            # We want assignment[min_node] = 1.
            # If min_color is 1, then nodes with color 1 get 1, color 2 get 2.
            # If min_color is 2, then nodes with color 2 get 1, color 1 get 2.
            for node, c in component:
                if min_color == 1:
                    assignment[node] = c
                else:
                    assignment[node] = 3 - c
                    
    print("YES")
    print(*(assignment[1:]))

solve()
```
