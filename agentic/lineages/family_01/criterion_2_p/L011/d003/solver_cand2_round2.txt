```python
import sys

# The equation 3*x_u*x_v = 2*(x_u + x_v) can be rewritten as:
# 9*x_u*x_v - 6*x_u - 6*x_v = 0
# (3*x_u - 2)*(3*x_v - 2) = 4
# Since x_u, x_v are positive integers, 3*x_u - 2 and 3*x_v - 2 must be divisors of 4.
# Possible values for 3*x - 2:
# 1) 3*x - 2 = 1 => 3*x = 3 => x = 1
# 2) 3*x - 2 = 2 => 3*x = 4 (no integer solution)
# 3) 3*x - 2 = 4 => 3*x = 6 => x = 2
# 4) 3*x - 2 = -1 => 3*x = 1 (no)
# 5) 3*x - 2 = -2 => 3*x = 0 (no)
# 6) 3*x - 2 = -4 => 3*x = -2 (no)
# So, for any connected u, v, the pair (x_u, x_v) must be (1, 2) or (2, 1).
# This means the graph must be bipartite.
# If the graph is bipartite, we have two sets of nodes (color 0 and color 1).
# We can assign color 0 nodes to 1 and color 1 nodes to 2, or vice versa.
# To get the lexicographically smallest sequence, we compare the two assignments.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    # Two possible assignments:
    # Option A: color 0 -> 1, color 1 -> 2
    # Option B: color 0 -> 2, color 1 -> 1
    # We want the lexicographically smallest.
    
    res1 = [0] * n
    res2 = [0] * n
    for i in range(1, n + 1):
        if color[i] == 0:
            res1[i-1] = 1
            res2[i-1] = 2
        else:
            res1[i-1] = 2
            res2[i-1] = 1
            
    print("YES")
    if res1 <= res2:
        print(*(res1))
    else:
        print(*(res2))

solve()
```
