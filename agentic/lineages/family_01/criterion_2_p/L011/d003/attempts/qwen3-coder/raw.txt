```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print("1 " * n)
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Analyze the constraint: 3x_u + 3x_v - 2x_u*x_v = 5
    # Rearranging: 3x_u + 3x_v - 2x_u*x_v = 5
    # This can be rewritten as: x_u(3 - 2x_v) = 5 - 3x_v
    # So: x_u = (5 - 3x_v) / (3 - 2x_v)
    
    # For this to be a positive integer, we need to find valid pairs (x_u, x_v)
    # Let's check small values of x_v to see what x_u values we get:
    valid_pairs = {}
    
    # Check for small positive integer values
    for x_v in range(1, 100):
        numerator = 5 - 3 * x_v
        denominator = 3 - 2 * x_v
        
        if denominator == 0:
            continue
            
        if numerator % denominator == 0:
            x_u = numerator // denominator
            if x_u > 0:
                if x_v not in valid_pairs:
                    valid_pairs[x_v] = []
                valid_pairs[x_v].append(x_u)
    
    # From analysis, the valid pairs are limited. Let's find them systematically.
    # We solve: x_u = (5 - 3*x_v) / (3 - 2*x_v)
    # For this to be a positive integer, we need (5-3*x_v) to be divisible by (3-2*x_v)
    
    valid_combinations = set()
    for x_v in range(1, 1000):
        denom = 3 - 2 * x_v
        if denom == 0:
            continue
        num = 5 - 3 * x_v
        if num % denom == 0:
            x_u = num // denom
            if x_u > 0:
                valid_combinations.add((x_u, x_v))
                valid_combinations.add((x_v, x_u))
    
    # The constraint essentially means nodes connected by an edge can only have specific value pairs
    # This forms a bipartite graph constraint - nodes must be 2-colored with specific values
    
    # Check if the graph is bipartite and find components
    color = [-1] * n
    components = []
    visited = [False] * n
    
    for i in range(n):
        if not visited[i] and adj[i]:
            # BFS to find component and check bipartiteness
            component = []
            queue = deque([i])
            visited[i] = True
            color[i] = 0
            component.append(i)
            
            is_bipartite = True
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        color[neighbor] = 1 - color[node]
                        component.append(neighbor)
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        is_bipartite = False
            
            if not is_bipartite:
                print("NO")
                return
                
            components.append(component)
        elif not visited[i] and not adj[i]:
            # Isolated node
            visited[i] = True
    
    # For each component, we need to assign values such that all edges satisfy the constraint
    # In a bipartite graph, we can assign one value to one set and another value to the other set
    # But the values must be a valid pair from our constraint
    
    # Valid pairs we found: (1,2) and (2,1) are the only small positive integer solutions
    # Let's verify: 3*1 + 3*2 - 2*1*2 = 3 + 6 - 4 = 5 ✓
    
    x = [0] * n  # 1-indexed in output, 0-indexed here
    
    # Assign values to isolated nodes (those with no edges)
    for i in range(n):
        if not adj[i]:  # No edges
            x[i] = 1  # Can be any positive value, choose 1 for lex smallest
    
    # Process each component
    for component in components:
        # Try assigning value 1 to color 0 and value 2 to color 1
        assignment_valid = True
        
        # Check if this assignment satisfies all edges
        for u, v in edges:
            if u in component and v in component:
                # Both nodes are in this component
                cu = color[u]
                cv = color[v]
                # cu should have value 1, cv should have value 2, or vice versa
                # Check if (1,2) or (2,1) satisfies constraint
                val_u = 1 if cu == 0 else 2
                val_v = 2 if cu == 0 else 1
                if 3*val_u + 3*val_v - 2*val_u*val_v != 5:
                    assignment_valid = False
                    break
        
        if assignment_valid:
            # Assign values
            for node in component:
                x[node] = 1 if color[node] == 0 else 2
        else:
            # Try the reverse assignment
            assignment_valid = True
            for u, v in edges:
                if u in component and v in component:
                    cu = color[u]
                    cv = color[v]
                    val_u = 2 if cu == 0 else 1
                    val_v = 1 if cu == 0 else 2
                    if 3*val_u + 3*val_v - 2*val_u*val_v != 5:
                        assignment_valid = False
                        break
            
            if assignment_valid:
                for node in component:
                    x[node] = 2 if color[node] == 0 else 1
            else:
                print("NO")
                return
    
    # Verify the solution
    for u, v in edges:
        if 3*x[u] + 3*x[v] - 2*x[u]*x[v] != 5:
            print("NO")
            return
    
    # Make sure all values are assigned
    for i in range(n):
        if x[i] == 0:
            x[i] = 1  # Default for isolated nodes
    
    print("YES")
    print(' '.join(map(str, x)))

solve()
```
