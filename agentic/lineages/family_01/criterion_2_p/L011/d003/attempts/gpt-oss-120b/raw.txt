```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [ -1 ] * (n + 1)          # parity 0/1 inside component
    answer = [0] * (n + 1)            # final x_i
    visited = [False] * (n + 1)

    for start in range(1, n + 1):
        if visited[start]:
            continue
        if not adj[start]:            # isolated vertex
            visited[start] = True
            answer[start] = 1
            continue

        # BFS to get component and check bipartite
        q = deque([start])
        visited[start] = True
        color[start] = 0
        comp_nodes = [start]
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    color[v] = color[u] ^ 1
                    q.append(v)
                    comp_nodes.append(v)
                else:
                    if color[v] == color[u]:
                        ok = False
        if not ok:
            print("NO")
            return

        # decide orientation: smallest index node gets value 1
        min_node = min(comp_nodes)
        if color[min_node] == 0:
            # parity 0 -> 1, parity 1 -> 2
            for v in comp_nodes:
                answer[v] = 1 if color[v] == 0 else 2
        else:
            # parity 1 -> 1, parity 0 -> 2
            for v in comp_nodes:
                answer[v] = 1 if color[v] == 1 else 2

    # all nodes assigned
    print("YES")
    print(' '.join(str(answer[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
