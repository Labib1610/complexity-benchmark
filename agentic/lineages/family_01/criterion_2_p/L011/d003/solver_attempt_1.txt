import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The constraint "sum of modes is 3" means if lever u is 1, v must be 2,
    # and if u is 2, v must be 1. This is a bipartite coloring problem.
    # We want to prioritize mode 1 for the smallest index.
    
    modes = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if modes[i] == 0:
            # Start a BFS/DFS for this connected component
            # To prioritize mode 1 for the smallest index, we set the 
            # first unvisited node in the component to 1.
            stack = [(i, 1)]
            modes[i] = 1
            
            while stack:
                u, mode = stack.pop()
                
                for v in adj[u]:
                    target_mode = 3 - mode
                    if modes[v] == 0:
                        modes[v] = target_mode
                        stack.append((v, target_mode))
                    elif modes[v] != target_mode:
                        print("NO")
                        return
                        
    print("YES")
    print(*(modes[1:]))

if __name__ == '__main__':
    solve()
