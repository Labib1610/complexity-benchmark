```python
import sys

# The problem asks for an assignment of 1 or 2 to each node such that for every edge (u, v),
# cycle(u) * cycle(v) = 2.
# This means if cycle(u) = 1, then cycle(v) = 2, and if cycle(u) = 2, then cycle(v) = 1.
# This is equivalent to a 2-coloring problem where the two colors are 1 and 2.
# A graph is 2-colorable if and only if it is bipartite.
# We need the lexicographically smallest assignment, meaning we prefer 1 over 2.
# For each connected component, we can pick the color of the first node (the one with the smallest index).
# To make the sequence lexicographically smallest, for each component, we should assign 1 to the 
# smallest index node in that component if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we assign 1 to the smallest index i
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
