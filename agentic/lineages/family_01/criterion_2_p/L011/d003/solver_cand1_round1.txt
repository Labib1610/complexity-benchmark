import sys

# The equation is 3x_u + 3x_v - 2x_u x_v = 5
# Rearranging: 2x_u x_v - 3x_u - 3x_v + 5 = 0
# Multiply by 2: 4x_u x_v - 6x_u - 6x_v + 10 = 0
# (2x_u - 3)(2x_v - 3) - 9 + 10 = 0
# (2x_u - 3)(2x_v - 3) = -1
# Let A = 2x_u - 3, B = 2x_v - 3. Then AB = -1.
# Since x_u, x_v >= 1 are integers, 2x_u - 3 can be -1 (if x_u=1) or 1 (if x_u=2).
# If 2x_u - 3 = -1, then 2x_v - 3 = 1 => x_v = 2.
# If 2x_u - 3 = 1, then 2x_v - 3 = -1 => x_v = 1.
# So for every edge (u, v), one node must be 1 and the other must be 2.
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # colors: 0 = unvisited, 1 = state 1, 2 = state 2
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # To get lexicographically smallest, we want x_i = 1 whenever possible.
            # For each connected component, we have two choices:
            # Choice A: start node i with color 1
            # Choice B: start node i with color 2
            # We compare the two assignments and pick the lexicographically smaller one.
            
            # BFS/DFS to find component
            component = []
            stack = [i]
            colors[i] = 1
            component.append(i)
            
            possible = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        component.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
            
            # Check if swapping colors (1->2, 2->1) gives a smaller result
            # We only need to check the first node of the component
            # If we set colors[i] = 2, we must flip all colors in this component
            comp_colors_1 = [colors[node] for node in component]
            
            # Try setting i to 1 (already done)
            # Try setting i to 2
            comp_colors_2 = [3 - c for c in comp_colors_1]
            
            # Compare lexicographically
            # Since we want the smallest sequence, we compare the values at the smallest index
            # The smallest index in this component is min(component)
            # We need to map node IDs to their assigned values
            
            def get_assignment(c_list):
                res = [0] * (n + 1)
                for idx, node in enumerate(component):
                    res[node] = c_list[idx]
                return res

            # We want the smallest value at the smallest node index in the component
            min_node = min(component)
            if comp_colors_1[component.index(min_node)] <= comp_colors_2[component.index(min_node)]:
                final_colors = comp_colors_1
            else:
                final_colors = comp_colors_2
            
            for idx, node in enumerate(component):
                colors[node] = final_colors[idx]

    print("YES")

solve()
