```python
import sys

# The thermal footprint of a link between u and v is (s_u | s_v) * (s_u & s_v).
# Given s_i in {1, 2}:
# If s_u = 1, s_v = 1: (1|1) * (1&1) = 1 * 1 = 1
# If s_u = 2, s_v = 2: (2|2) * (2&2) = 2 * 2 = 4
# If s_u = 1, s_v = 2: (1|2) * (1&2) = 3 * 0 = 0
#
# The total footprint is the sum of these values. For the sum to be 0,
# every link must have a footprint of 0.
# This means for every link (u, v), one unit must be 1 and the other must be 2.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we want s_1 = 1.
# For each connected component, we can either assign 1 to the set containing
# the smallest index node or 2. To minimize lexicographically, we greedily
# assign 1 to the smallest index node in each component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

solve()
```
