import sys

# The condition for each unit i is:
# d_i * s_i + sum_{j in neighbors(i)} s_j = 3 * d_i
# Let x_i = s_i - 1. Since s_i in {1, 2}, x_i in {0, 1}.
# d_i * (x_i + 1) + sum_{j in neighbors(i)} (x_j + 1) = 3 * d_i
# d_i * x_i + d_i + sum_{j in neighbors(i)} x_j + d_i = 3 * d_i
# d_i * x_i + sum_{j in neighbors(i)} x_j = d_i
#
# This is a system of linear equations over the integers.
# For a connected component, if d_i = 0, the equation is 0 = 0 (always true).
# If d_i > 0, this looks like a variation of the 2-coloring problem.
# Specifically, if the graph is bipartite, we can solve this.
# If the graph is not bipartite, we check for consistency.
# Actually, the equation d_i * x_i + sum_{j in neighbors(i)} x_j = d_i
# is equivalent to: if we set x_i = 1 - x_j for all neighbors, it works.
# This implies the graph must be bipartite.
# If bipartite, let the two sets be A and B.
# For i in A: d_i * x_i + sum_{j in neighbors(i)} x_j = d_i
# If we set all x_i = 1 for i in A and x_j = 0 for j in B:
# For i in A: d_i * 1 + 0 = d_i (True)
# For j in B: d_j * 0 + sum_{i in neighbors(j)} 1 = d_j (True)
# So x_i = 1 for A, x_j = 0 for B is a solution.
# Also x_i = 0 for A, x_j = 1 for B is a solution.
# We want lexicographically smallest s_i, so smallest x_i.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        idx += 2
        if u == v:
            # Self-loop: d_i increases by 2.
            # d_i * x_i + sum_{j in neighbors} x_j = d_i
            # If u has a self-loop, x_u must satisfy:
            # (d_i_orig + 2) * x_u + (sum_{j} x_j + 2 * x_u) = d_i_orig + 2
            # This forces x_u = 0.
            adj[u].append(u)
            adj[u].append(u)
        else:
            adj[u].append(v)
            adj[v].append(u)

    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if v == u: continue
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if self-loops are consistent with color
            for u, c in component:
                for v in adj[u]:
                    if v == u and c != 0:
                        print("NO")
                        return
            
            # Two possible assignments: 
            # 1: color 0 -> x=0, color 1 -> x=1
            # 2: color 0 -> x=1, color 1 -> x=0
            # We want lexicographically smallest s_i, so smallest x_i.
            # Compare the two options for the first node in the component.
            # Actually, we just need to pick the one that makes the sequence smallest.
            
            # Option 1: color 0 -> x=0, color 1 -> x=1
            # Option 2: color 0 -> x=1, color 1 -> x=0
            # Pick the one that gives smaller x for the smallest index in component.
            first_node = min(u for u, c in component)
            # If color[first_node] == 0, Option 1 gives x=0, Option 2 gives x=1.
            # So Option 1 is better.
            if color[first_node] == 0:
                for u, c in component: res[u] = c
            else:
                for u, c in component: res[u] = 1 - c
                
    print("YES")
    print(*(res[i] + 1 for i in range(1, n + 1)))

solve()
