```python
import sys

# The stability value for a pair (p_u, p_v) is 6(p_u + p_v) - (p_u + p_v)^2.
# Possible values for p_u, p_v are {1, 2}.
# Sum (p_u + p_v) can be 2, 3, or 4.
# If p_u + p_v = 2 (1, 1): 6(2) - 2^2 = 12 - 4 = 8.
# If p_u + p_v = 3 (1, 2 or 2, 1): 6(3) - 3^2 = 18 - 9 = 9.
# If p_u + p_v = 4 (2, 2): 6(4) - 4^2 = 24 - 16 = 8.
# We need the sum of stability values to be 9m.
# Let k be the number of edges where p_u + p_v = 3.
# Let m-k be the number of edges where p_u + p_v is 2 or 4.
# Total sum = k * 9 + (m - k) * 8 = 8m + k.
# We need 8m + k = 9m, which implies k = m.
# This means for every edge (u, v), p_u + p_v must be 3.
# Since p_u, p_v in {1, 2}, p_u + p_v = 3 implies p_u != p_v.
# This is exactly the condition for a graph to be bipartite.
# We need to find the lexicographically smallest assignment (1 or 2).
# For each connected component, we can pick the color of the first node.
# To make it lexicographically smallest, for each component, we try to assign 1 to the
# smallest index node in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
        else:
            print("NO")
            return
        idx += 2

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color component
            # To be lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
