```python
import sys

# The Laplace transform of f(t) = p_u^4 cos(t) + p_v^4 sin(t) at s=1 is:
# L{cos(t)}(s) = s / (s^2 + 1)
# L{sin(t)}(s) = 1 / (s^2 + 1)
# At s=1: L{cos(t)}(1) = 1/2, L{sin(t)}(1) = 1/2
# So, L{f(t)}(1) = p_u^4 * (1/2) + p_v^4 * (1/2) = 8.5
# p_u^4 + p_v^4 = 17
# Since p_u, p_v are positive integers, the only solution to x^4 + y^4 = 17
# is {x, y} = {1, 2} because 1^4 + 2^4 = 1 + 16 = 17.
# This means for every edge (u, v), one must be 1 and the other must be 2.
# This is a bipartite coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    # To satisfy the tie-breaking rule (lexicographically smallest sequence),
    # we want p_i = 1 whenever possible.
    # We iterate 1 to n. If not colored, color it 1.
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 2 if c == 1 else 1
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_c
                        stack.append((v, next_c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
