```python
import sys

# The coupling matrix C = [[x_u, 1], [1, x_v]]
# The characteristic equation is det(C - lambda*I) = 0
# (x_u - lambda)(x_v - lambda) - 1 = 0
# lambda^2 - (x_u + x_v)lambda + (x_u * x_v - 1) = 0
# The eigenvalues are irrational if and only if the discriminant D > 0 is not a perfect square.
# D = (x_u + x_v)^2 - 4(x_u * x_v - 1)
# D = x_u^2 + 2*x_u*x_v + x_v^2 - 4*x_u*x_v + 4
# D = x_u^2 - 2*x_u*x_v + x_v^2 + 4
# D = (x_u - x_v)^2 + 4
# Since x_i must be a positive integer divisor of 2, x_i can be 1 or 2.
# Possible pairs (x_u, x_v):
# 1) (1, 1): D = (1-1)^2 + 4 = 4. sqrt(4) = 2 (rational). Not balanced.
# 2) (2, 2): D = (2-2)^2 + 4 = 4. sqrt(4) = 2 (rational). Not balanced.
# 3) (1, 2) or (2, 1): D = (1-2)^2 + 4 = 1 + 4 = 5. sqrt(5) is irrational. Balanced.
# Thus, for every edge (u, v), x_u must be different from x_v.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we greedily assign 1 to the first node
# in each connected component, then propagate the bipartite coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color component
            # To get lexicographically smallest, we want x_i = 1 if possible.
            # Since we iterate 1 to n, we assign 1 to the first uncolored node.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
