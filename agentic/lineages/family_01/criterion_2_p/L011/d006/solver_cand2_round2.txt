import sys

# The characteristic matrix M(x_u, x_v) = [[x_u, 1], [1, x_v]]
# The characteristic polynomial is det(M - lambda*I) = (x_u - lambda)(x_v - lambda) - 1
# = lambda^2 - (x_u + x_v)lambda + (x_u*x_v - 1)
# The eigenvalues are irrational if the discriminant D = (x_u + x_v)^2 - 4(x_u*x_v - 1) is not a perfect square.
# D = x_u^2 + 2*x_u*x_v + x_v^2 - 4*x_u*x_v + 4
# D = x_u^2 - 2*x_u*x_v + x_v^2 + 4 = (x_u - x_v)^2 + 4
# Since x_u, x_v in {1, 2}:
# If x_u = x_v, D = 0^2 + 4 = 4 = 2^2 (perfect square)
# If x_u != x_v, D = (1-2)^2 + 4 = 1 + 4 = 5 (not a perfect square)
# Thus, the condition "irrational eigenvalues" is equivalent to x_u != x_v.
# This is a 2-coloring problem on a graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # We want the lexicographically smallest sequence.
    # This means we prefer x_i = 1 whenever possible.
    # We use BFS/DFS to color each connected component.
    # For each component, we try setting the first node to 1.
    # If that fails, we set it to 2.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we greedily pick 1 for the smallest index
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard 2-coloring
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
