import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unassigned, 1 = 1 microsecond, 2 = 2 microseconds
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is unassigned, we try assigning 1. If that leads to a conflict,
    # we must assign 2.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this connected component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Use a stack to perform DFS to find all nodes in the component
            # and check for bipartiteness.
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring the component, we check if this assignment is 
            # lexicographically smallest. The current assignment for this 
            # component is fixed by the choice of color[i] = 1.
            # Since we process nodes 1..n in order, if we encounter an 
            # unassigned node, setting it to 1 is always the best choice.
            
    print("YES")
    print(*(color[1:]))

solve()
