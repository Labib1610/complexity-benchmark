```python
import sys

# The transformation matrices are:
# T1 = [[1, 3], [0, 1]]
# T2 = [[1, 0], [3, 1]]
# Initial vector V = [1, 1]^T
# For a channel (u, v):
# If u=1, v=1: T1(T1(1, 1)) = T1(4, 1) = (4+3, 1) = (7, 1). Sum = 8 != 17.
# If u=1, v=2: T2(T1(1, 1)) = T2(4, 1) = (4, 12+1) = (4, 13). Sum = 17. (Valid)
# If u=2, v=1: T1(T2(1, 1)) = T1(1, 4) = (1+12, 4) = (13, 4). Sum = 17. (Valid)
# If u=2, v=2: T2(T2(1, 1)) = T2(1, 4) = (1, 3+4) = (1, 7). Sum = 8 != 17.
# Thus, for every edge (u, v), x_u != x_v.
# This is a bipartite graph coloring problem.
# We need to color the graph with 2 colors (1 and 2) such that no two adjacent nodes have the same color.
# To get the lexicographically smallest sequence, we greedily assign 1 to the smallest available node index.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for each connected component
            # To get lexicographically smallest, we prefer color 1 for the smallest node
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
