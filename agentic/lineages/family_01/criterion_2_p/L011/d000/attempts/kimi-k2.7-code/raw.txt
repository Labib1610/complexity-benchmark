import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    bad = False
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            bad = True
        else:
            adj[u].append(v)
            adj[v].append(u)
    if bad:
        sys.stdout.write("NO")
        return

    color = [-1] * n
    for i in range(n):
        if color[i] != -1:
            continue
        dq = deque([i])
        color[i] = 0
        comp = [i]
        mn = i
        while dq:
            u = dq.popleft()
            cu = color[u]
            for w in adj[u]:
                if color[w] == -1:
                    color[w] = cu ^ 1
                    dq.append(w)
                    comp.append(w)
                    if w < mn:
                        mn = w
                elif color[w] == cu:
                    sys.stdout.write("NO")
                    return
        if color[mn] == 1:
            for v in comp:
                color[v] ^= 1

    out = ['YES', ' '.join(str(color[i] + 1) for i in range(n))]
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
