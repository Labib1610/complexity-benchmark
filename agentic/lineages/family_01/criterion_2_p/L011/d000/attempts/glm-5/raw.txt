import sys
from sys import stdin

def solve():
    input_data = stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    head = list(range(n + 1))
    nxt = [0] * (2 * m)
    to = [0] * (2 * m)
    
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        idx = 2 * i
        nxt[idx] = head[u]
        to[idx] = v
        head[u] = idx
        
        idx2 = 2 * i + 1
        nxt[idx2] = head[v]
        to[idx2] = u
        head[v] = idx2
    
    color = [0] * (n + 1)
    
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        color[start] = 1
        stack = [start]
        while stack:
            u = stack.pop()
            cu = color[u]
            e = head[u]
            while e != 0:
                v = to[e]
                cv = 3 - cu
                if color[v] == 0:
                    color[v] = cv
                    stack.append(v)
                elif color[v] != cv:
                    print("NO")
                    return
                e = nxt[e]
    
    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

solve()
