import sys

# The phase deviation formula:
# D(u, v) = 599999(x_u + x_v) - 299999(x_u * x_v) - 1199999
# Let x_i be 1 or 2.
# If x_u=1, x_v=1: D = 599999(2) - 299999(1) - 1199999 = 1199998 - 299999 - 1199999 = -299999
# If x_u=1, x_v=2: D = 599999(3) - 299999(2) - 1199999 = 1799997 - 599998 - 1199999 = 0
# If x_u=2, x_v=2: D = 599999(4) - 299999(4) - 1199999 = 2399996 - 1199996 - 1199999 = 1
# Let n11 be the number of edges with (1, 1), n12 be (1, 2), n22 be (2, 2).
# Total deviation = n11 * (-299999) + n12 * (0) + n22 * (1) = 0
# n22 = 299999 * n11
# Since n11 + n12 + n22 = m, and n11, n12, n22 >= 0:
# If n11 > 0, then n22 >= 299999. But m <= 200000.
# Thus, n11 must be 0.
# If n11 = 0, then n22 = 0.
# This means there are no edges between two 1s and no edges between two 2s.
# This is equivalent to saying the graph must be bipartite, and we must color it
# such that all edges are between 1 and 2.
# If the graph is bipartite, we can color each connected component with 2 colors.
# Let the two colors be A and B.
# For each component, we have two choices: (A=1, B=2) or (A=2, B=1).
# To get the lexicographically smallest sequence, for each component,
# we pick the assignment that makes the first node (smallest index) 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            first_node = min(node for node, c in component)
            # We want res[first_node] to be 1 if possible.
            # If color[first_node] is 1, we map 1->1, 2->2.
            # If color[first_node] is 2, we map 1->2, 2->1.
            c_first = color[first_node]
            for node, c in component:
                if c_first == 1:
                    res[node] = c
                else:
                    res[node] = 3 - c
                    
    print("YES")
    print(*(res[1:]))

solve()
