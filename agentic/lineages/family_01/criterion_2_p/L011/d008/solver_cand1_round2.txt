```python
import sys

# The condition for a channel (u, v) to have phase deviation 0 is:
# (x_u^2 + x_v^2 - 5)^2 = 0
# x_u^2 + x_v^2 = 5
# Since x_i \in {1, 2}, x_i^2 \in {1, 4}.
# The only way to get 1 + 4 = 5 is if one node is 1 and the other is 2.
# This means the graph must be bipartite.
# We need to find the lexicographically smallest assignment (x_1, ..., x_n).
# This means we want x_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # Bipartite coloring: 1 or 2.
    # To get lexicographically smallest, we try to assign 1 to node i if possible.
    # We process components. For each component, we have two choices of coloring.
    # We pick the one that makes the sequence lexicographically smaller.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS/DFS to find component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a component. We can either keep the current colors
            # or flip them (1->2, 2->1).
            # We want the lexicographically smallest result.
            # Compare the two options for the first node in the component (smallest index).
            # Actually, we want the smallest sequence overall.
            # For each component, we check which assignment (color[v] or 3-color[v])
            # is better by looking at the smallest index node in the component.
            
            first_node = min(component)
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
```
