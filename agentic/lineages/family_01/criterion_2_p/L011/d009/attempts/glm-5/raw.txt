import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(input_data[idx]); idx += 1
        v = int(input_data[idx]); idx += 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0]*(n+1)
    parent = [0]*(n+1)
    comp = [0]*(n+1)
    comp_id = 0
    comp_roots = []
    
    for start in range(1, n+1):
        if color[start] == 0:
            comp_id += 1
            comp_roots.append(start)
            color[start] = 1
            parent[start] = 0
            stack = [start]
            while stack:
                u = stack.pop()
                cu = color[u]
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - cu
                        parent[v] = u
                        comp[v] = comp_id
                        stack.append(v)
                    elif color[v] == cu:
                        print("NO")
                        return
            comp[start] = comp_id
    
    comp_has_cycle = [False]*(comp_id+1)
    for u in range(1, n+1):
        if not comp_has_cycle[comp[u]]:
            for v in adj[u]:
                if color[v] == color[u]:
                    comp_has_cycle[comp[u]] = True
                    break
    
    comp_flipped = [False]*(comp_id+1)
    for c in range(1, comp_id+1):
        if not comp_has_cycle[c]:
            comp_flipped[c] = True
    
    final_color = [0]*(n+1)
    for u in range(1, n+1):
        c = comp[u]
        if comp_flipped[c]:
            final_color[u] = 3 - color[u]
        else:
            final_color[u] = color[u]
    
    print("YES")
    print(' '.join(str(final_color[i]) for i in range(1, n+1)))

solve()
