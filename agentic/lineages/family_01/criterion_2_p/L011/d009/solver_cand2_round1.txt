import sys

# The condition is: for every node i, sum_{j in neighbors(i)} (x_i + x_j) = 3 * d_i
# This simplifies to: d_i * x_i + sum_{j in neighbors(i)} x_j = 3 * d_i
# Let y_i = x_i - 1. Since x_i in {1, 2}, y_i in {0, 1}.
# The equation becomes: d_i * (y_i + 1) + sum_{j in neighbors(i)} (y_j + 1) = 3 * d_i
# d_i * y_i + d_i + sum_{j in neighbors(i)} y_j + d_i = 3 * d_i
# d_i * y_i + sum_{j in neighbors(i)} y_j = d_i
# This is a system of linear equations over GF(2) if we consider the variables y_i.
# Actually, it's a system of linear equations over integers, but since y_i is 0 or 1,
# we can solve it component-wise on each connected component.
# For a connected component, if d_i > 0 for all i, the equation is:
# y_i + sum_{j in neighbors(i)} (y_j / d_i) = 1.
# This looks like a bipartite graph property. If the graph is bipartite, we can
# partition nodes into sets A and B. For u in A, x_u = 1 => x_v = 2 for all neighbors v.
# If x_u = 1, then sum(x_v) = 3 * d_u => sum(x_v) = 3 * d_u.
# If all neighbors v have x_v = 2, then sum(x_v) = 2 * d_u.
# This only works if 2 * d_u = 3 * d_u, which implies d_u = 0.
# Wait, the condition is x_u + x_v = 3 for all edges (u, v).
# If x_u + x_v = 3, then if x_u = 1, x_v = 2. If x_u = 2, x_v = 1.
# This is exactly the condition for a bipartite graph.
# If the graph is bipartite, we can color it with 1 and 2.
# For each connected component, we have two choices (1, 2, 1, 2... or 2, 1, 2, 1...).
# We pick the one that is lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Try both assignments for this component
            # Assignment 1: nodes with color 1 get x=1, color 2 get x=2
            # Assignment 2: nodes with color 1 get x=2, color 2 get x=1
            # Check if valid: for all edges (u, v), x_u + x_v = 3
            # This is always true if color[u] != color[v]
            
            # Lexicographically smallest: compare the two assignments
            # Assignment 1: x_u = color[u]
            # Assignment 2: x_u = 3 - color[u]
            
            # Find the smallest node index in this component
            comp_nodes = [u for u, c in component]
            min_node = min(comp_nodes)
            
            # If color[min_node] == 1, assignment 1 gives x[min_node] = 1
            # If color[min_node] == 2, assignment 2 gives x[min_node] = 1
            if color[min_node] == 1:
                for u, c in component:
                    res[u] = c
            else:
                for u, c in component:
                    res[u] = 3 - c
                    
    print("YES")
    print(*(res[1:]))

solve()
