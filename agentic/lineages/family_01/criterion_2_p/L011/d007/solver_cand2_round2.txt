```python
import sys

# The problem requires x_i in {1, 2} for all i.
# For each edge (u, v):
# 1. At least one of {x_u, x_v} must be 1.
# 2. At least one of {x_u, x_v} must be 2.
# This means x_u != x_v.
# If x_u != x_v, then one is 1 and the other is 2.
# Condition 1: {1, 2} contains 1 (True).
# Condition 2: {1, 2} contains 2 (True).
# So the condition is equivalent to: the graph must be bipartite,
# and for every edge (u, v), x_u != x_v.
# This means the graph must be 2-colorable.
# If the graph is bipartite, each connected component has exactly 2 colorings.
# To get the lexicographically smallest sequence, for each component,
# we pick the coloring that makes the smallest node index in that component 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # 1. color[i] = 1 (original)
            # 2. color[i] = 2 (flipped)
            # We want the lexicographically smallest result.
            # Compare the two options by looking at the first node in the component.
            # Actually, we just need to ensure the smallest node in the component
            # gets the smallest possible value (1).
            
            # Find the smallest node index in this component
            min_node = min(component)
            
            # If we set color[min_node] = 1, is it valid?
            # Yes, because the graph is bipartite.
            # We just need to shift the colors so that color[min_node] becomes 1.
            
            target_color_of_min = 1
            shift = color[min_node] - target_color_of_min
            
            for node in component:
                res[node] = color[node] - shift
                if res[node] == 0:
                    res[node] = 2
                elif res[node] == -1:
                    res[node] = 1
                    
    print("YES")
    print(*(res[1:]))

solve()
```
