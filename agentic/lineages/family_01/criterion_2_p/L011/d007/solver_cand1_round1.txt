```python
import sys

# The transformation for state 1 is V' = V + k * D(V) = (X + k(X+2Y), Y + k(3X+4Y))
# The transformation for state 2 is V'' = 2V - k * D(V) = (2X - k(X+2Y), 2Y - k(3X+4Y))
# Let V = (1, 1).
# State 1: V1 = (1 + k(1+2), 1 + k(3+4)) = (1+3k, 1+7k)
# State 2: V2 = (2 - k(1+2), 2 - k(3+4)) = (2-3k, 2-7k)
# For a channel (u, v) with k = u+v, applying states x_u and x_v:
# Let f(x) be the transformation. f(1)(V) = (1+3k)X + 2kY, (3k)X + (1+4k)Y
# f(2)(V) = (2-3k)X - 2kY, (-3k)X + (2-4k)Y
# The sum of components of f(x)(V) is:
# S(1, V) = (1+3k)X + 2kY + 3kX + (1+4k)Y = (1+6k)X + (1+6k)Y = (1+6k)(X+Y)
# S(2, V) = (2-3k)X - 2kY - 3kX + (2-4k)Y = (2-6k)X + (2-6k)Y = (2-6k)(X+Y)
# Applying two states x_u, x_v:
# Total sum = S(x_v, f(x_u)(1, 1)) = S(x_v, (1+6k(x_u-1), 1+6k(x_u-1)))
# If x_u=1: S(1, (1, 1)) = 1+6k. Then S(x_v, (1+6k, 1+6k)) = (1+6k) * S(x_v, (1, 1))
# If x_u=1, x_v=1: (1+6k)^2 = 1 + 12k + 36k^2
# If x_u=1, x_v=2: (1+6k)(2-6k) = 2 - 6k + 12k - 36k^2 = 2 + 6k - 36k^2
# If x_u=2, x_v=1: (2-6k)(1+6k) = 2 + 6k - 36k^2
# If x_u=2, x_v=2: (2-6k)^2 = 4 - 24k + 36k^2
# Requirement: sum = 4 + 10k - 54k^2
# This implies x_u != x_v.
# Thus, the graph must be bipartite, and for each edge, x_u != x_v.
# To get lexicographically smallest, assign 1 to the first node of each component,
# then propagate 1s and 2s.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    print("YES")
    print(*(color[1:]))

solve()
```
