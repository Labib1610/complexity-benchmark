```python
import sys

# The problem states:
# 1 <= x_i <= 2
# For each link (u, v), x_u + x_v = max_allowed + min_allowed
# max_allowed = 2, min_allowed = 1
# So, x_u + x_v = 3
# This means for every edge (u, v), x_u != x_v.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest assignment.
# Lexicographically smallest means we want x_1 to be 1 if possible, then x_2 to be 1 if possible, etc.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # color 1 represents x_i = 1, color 2 represents x_i = 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, try assigning 1 to the first node of each component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
