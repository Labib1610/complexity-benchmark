import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest plan, we try to assign 1 to each node
    # whenever possible. This is a 2-coloring problem on a graph.
    # Each connected component can be colored in two ways (swapping 1s and 2s).
    # We want the smallest sequence, so for each component, we check which 
    # assignment results in a smaller sequence.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # Let the current coloring be C. The other valid coloring is 3-C.
            # We compare the two colorings lexicographically.
            # Since we want the smallest overall sequence, we look at the 
            # smallest index 'k' in the component and pick the coloring 
            # that assigns 1 to 'k'.
            
            first_node = min(component)
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
