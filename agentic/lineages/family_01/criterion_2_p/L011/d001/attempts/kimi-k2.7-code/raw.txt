 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    for i in range(n):
        if color[i] != -1:
            continue
        color[i] = 0
        stack = [i]
        while stack:
            u = stack.pop()
            cu = color[u] ^ 1
            for w in adj[u]:
                cw = color[w]
                if cw == -1:
                    color[w] = cu
                    stack.append(w)
                elif cw != cu:
                    sys.stdout.write("NO\n")
                    return

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(str(c + 1) for c in color))

if __name__ == "__main__":
    solve()
```
