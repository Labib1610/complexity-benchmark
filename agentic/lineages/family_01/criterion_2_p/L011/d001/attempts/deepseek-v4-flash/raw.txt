We can model this as a graph coloring problem with two colors, where each edge means the two nodes must have different colors.  
This is equivalent to checking if the graph is bipartite and, if so, finding the lexicographically smallest 2-coloring.

We can solve it with BFS/DFS, assigning colors greedily in index order to get the lexicographically smallest assignment.

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n

def bfs(start):
    q = deque([start])
    color[start] = 1
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = 3 - color[u]
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

for i in range(n):
    if color[i] == -1:
        if not bfs(i):
            print("NO")
            sys.exit()

print("YES")
print(" ".join(str(color[i]) for i in range(n)))
```
