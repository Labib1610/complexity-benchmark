import sys

# The problem asks for a 2-coloring of a graph such that each edge connects nodes of different colors.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest assignment, we iterate through nodes 1 to n.
# If a node is not yet colored, we try assigning it color 1. If that leads to a contradiction,
# we try color 2. If both fail, it's impossible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we process nodes 1 to n.
    # For each connected component, we try to assign the smallest possible color (1)
    # to the first node encountered.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            stack = [(i, 1)]

            q = [i]

            head = 0
            while head < len(q):
