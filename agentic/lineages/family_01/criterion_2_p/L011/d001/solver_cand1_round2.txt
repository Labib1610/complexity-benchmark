# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The condition "sum must be 3" means if u is 1, v must be 2, and vice versa.
    # This is a standard bipartite coloring problem.
    # To get the lexicographically smallest sequence, we want 1 at index 1 if possible.
    # We iterate 1 to n, and if not visited, try assigning 1.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To be lexicographically smallest, we prefer 1 over 2.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process nodes in a way that respects lexicographical order
            # However, standard DFS/BFS is fine as long as we are consistent.
            # To ensure lexicographical smallest, we always try to assign 1 to the
            # smallest uncolored node in the component.
            
            q = [i]
            color[i] = 1
            while q:
                u = q.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
