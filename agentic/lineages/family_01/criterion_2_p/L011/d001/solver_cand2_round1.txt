```python
import sys

# The condition for parallel resistance of R_u and R_v to be 2/3 is:
# (R_u * R_v) / (R_u + R_v) = 2/3
# Given R_u, R_v in {1, 2}:
# If R_u = 1, R_v = 1: 1/2 = 0.5 != 2/3
# If R_u = 1, R_v = 2: 2/3 = 0.666... (Valid)
# If R_u = 2, R_v = 1: 2/3 = 0.666... (Valid)
# If R_u = 2, R_v = 2: 4/4 = 1 != 2/3
# Thus, for every link (u, v), R_u must be different from R_v.
# This is a bipartite coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    # To get lexicographically smallest, we want R_i = 1 whenever possible.
    # We iterate 1 to n. If node i is uncolored, assign 1 and BFS/DFS.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
