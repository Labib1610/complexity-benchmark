import sys

# The problem states that for any two welders u and v connected by a bus,
# the XOR of their phase settings (1 or 2) must be 3.
# 1 XOR 2 = 3.
# This means if u is 1, v must be 2, and if u is 2, v must be 1.
# This is a standard bipartite coloring problem where each connected component
# must be 2-colorable.
# To get the lexicographically smallest sequence, we want s_i = 1 whenever possible.
# For each connected component, we have two choices:
# 1. Assign the first node (smallest index) to 1.
# 2. Assign the first node (smallest index) to 2.
# We compare the resulting sequences for each component and pick the one
# that is lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # Option 1: color[i] = 1
            # Option 2: color[i] = 2 (which means flipping all colors in component)
            
            # Generate Option 1
            opt1 = {}
            for node in component:
                opt1[node] = color[node]
            
            # Generate Option 2
            opt2 = {}
            for node in component:
                opt2[node] = 3 - color[node]
            
            # Compare lexicographically
            # We want the smallest value at the smallest index in the component
            # Since we process components in order of their smallest node,
            # we just need to pick the one that gives a smaller value at the
            # first node of the component.
            if opt1[i] < opt2[i]:
                for node in component:
                    ans[node] = opt1[node]
            else:
                for node in component:
                    ans[node] = opt2[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
