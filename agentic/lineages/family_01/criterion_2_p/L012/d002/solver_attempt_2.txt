import sys

# The stability equation is s_u^{s_v} + s_v^{s_u} = 3.
# Possible values for s_i are {1, 2}.
# If s_u = 1, s_v = 1: 1^1 + 1^1 = 2 != 3
# If s_u = 1, s_v = 2: 1^2 + 2^1 = 1 + 2 = 3 (Valid)
# If s_u = 2, s_v = 1: 2^1 + 1^2 = 2 + 1 = 3 (Valid)
# If s_u = 2, s_v = 2: 2^2 + 2^2 = 4 + 4 = 8 != 3
# Thus, the condition s_u^{s_v} + s_v^{s_u} = 3 is equivalent to s_u != s_v.
# This is a standard 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest sequence, we iterate from 1 to n.
# If a node is not colored, we try assigning 1 (the smaller value).
# If that leads to a contradiction, we try 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning 1 to the current component
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            possible = True
            idx_comp = 0
            component.append(i)
            
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # If 1 failed, try 2 for the root of this component
                for node in component:
                    color[node] = 0
                
                color[i] = 2
                component = [i]
                idx_comp = 0
                possible = True
                while idx_comp < len(component):
                    u = component[idx_comp]
                    idx_comp += 1
                    for v in adj[u]:
                        if color[v] == 0:
                            color[v] = 3 - color[u]
                            component.append(v)
                        elif color[v] == color[u]:
                            print("NO")
                            return
    
    print("YES")
    print(*(color[1:]))

solve()
