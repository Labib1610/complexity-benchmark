```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    graph = defaultdict(list)
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        graph[u].append(v)
        graph[v].append(u)
        index += 2
    
    # We need s_u & s_v = 0
    # Modes are 1 and 2
    # 1 & 1 = 1 != 0
    # 1 & 2 = 0
    # 2 & 2 = 2 != 0
    # So, adjacent nodes must have different values: one 1 and one 2
    # This is a bipartite graph check
    
    color = [-1] * n  # -1: unvisited, 0: color A, 1: color B
    
    for start in range(n):
        if color[start] == -1:
            queue = deque()
            queue.append(start)
            color[start] = 0  # Assign first color
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in graph[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - current_color
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        print("NO")
                        return
    
    # It's bipartite. Now we need to assign 1 and 2 to the two color classes
    # For lex smallest, we prefer s_i = 1 if possible
    # Color 0 -> 1
    # Color 1 -> 2
    
    result = []
    for i in range(n):
        if color[i] == 0:
            result.append("1")
        else:
            result.append("2")
    
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
