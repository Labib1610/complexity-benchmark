```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Since XOR of phase settings must be 3, and phases are 1 or 2:
    # 1 XOR 2 = 3
    # So connected nodes must have different phases.
    # This is a bipartite graph check.
    
    color = [0] * (n + 1)  # 0: unvisited, 1: color A, 2: color B
    
    # Check each component
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        # BFS to check bipartiteness and assign colors
        queue = deque()
        queue.append(start)
        color[start] = 1  # Assign color 1 to start node
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - color[u]  # Assign opposite color
                    queue.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
    
    # If bipartite, construct lexicographically smallest configuration
    # We want s_i \in {1, 2}, and connected nodes have different values
    # Color 1 -> 1, Color 2 -> 2 to get lex smallest
    result = []
    for i in range(1, n + 1):
        if color[i] == 1:
            result.append("1")
        else:
            result.append("2")
    
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
