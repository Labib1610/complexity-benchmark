import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # State 1 or 2. Difference must be 1.
    # This is equivalent to 2-coloring a graph where 1 maps to 0 and 2 maps to 1.
    # If color[u] == color[v], then |s_u - s_v| = 0, which is invalid.
    # We need |s_u - s_v| = 1, so color[u] != color[v].
    
    color = [-1] * (n + 1)
    result = [0] * (n + 1)
    
    # To get lexicographically smallest sequence, we want s_i = 1 whenever possible.
    # s_i = 1 corresponds to color 0.
    # We iterate 1 to n. If not colored, try color 0 (s=1).
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            component = []
            
            # Standard DFS to color component
            q = [i]
            color[i] = 0
            component.append(i)
            
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring, check if we can flip the whole component to make it lexicographically smaller
            # We want the first node in the component (by index) to be 1 if possible.
            # The component nodes are already colored.
            # Let's find the smallest index in this component.
            min_node = min(component)
            if color[min_node] == 1:
                # Flip all colors in this component
                for node in component:
                    color[node] = 1 - color[node]
                    
    print("YES")
    ans = []
    for i in range(1, n + 1):
        ans.append(str(color[i] + 1))
    print(" ".join(ans))

solve()
