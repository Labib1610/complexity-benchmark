```python
import sys

# The condition: s_u + s_v must have exactly 2 set bits in binary.
# s_i is in {1, 2}.
# Possible sums:
# 1 + 1 = 2 (binary 10, 1 set bit) -> Invalid
# 1 + 2 = 3 (binary 11, 2 set bits) -> Valid
# 2 + 1 = 3 (binary 11, 2 set bits) -> Valid
# 2 + 2 = 4 (binary 100, 1 set bit) -> Invalid
# Thus, for every critical pair (u, v), s_u must be different from s_v.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we have two sets of vertices (color 1 and color 2).
# We can assign state 1 to one set and state 2 to the other.
# There are two possible assignments:
# Assignment A: Color 1 -> 1, Color 2 -> 2
# Assignment B: Color 1 -> 2, Color 2 -> 1
# We want to minimize sum_{i=1}^n s_i * 2^{n-i}.
# This is equivalent to minimizing the integer formed by the sequence (s_1, s_2, ..., s_n).
# We compare the two assignments lexicographically.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # Two possible assignments:
    # 1. color 1 -> 1, color 2 -> 2
    # 2. color 1 -> 2, color 2 -> 1
    
    def get_assignment(c1_val):
        res = []
        for i in range(1, n + 1):
            if color[i] == 1:
                res.append(c1_val)
            else:
                res.append(3 - c1_val)
        return res

    a1 = get_assignment(1)
    a2 = get_assignment(2)
    
    # Compare lexicographically
    if a1 <= a2:
        print("YES")
        print(*(a1))
    else:
        print("YES")
        print(*(a2))

solve()
```
