```python
import sys

# The problem states that for any two zones connected by a valve, 
# the ratio of their pressures must be 2.
# Since pressures can only be 1 or 2, this means one zone must be 1 
# and the other must be 2.
# This is equivalent to saying the graph must be bipartite.
# We want the lexicographically smallest sequence (p1, p2, ..., pn).
# To make it lexicographically smallest, we prefer 1 over 2 for smaller indices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If a node is not colored, we color it 1 (the smallest possible value).
    # Then we perform BFS/DFS to color its connected component.
    for i in range(1, n + 1):
        if colors[i] == 0:
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
