import sys

# The condition for the matrix M_{u,v} = [[h_u, 2], [1, h_v]] to be singular is:
# det(M_{u,v}) = h_u * h_v - 2 * 1 = 0
# h_u * h_v = 2
# Since h_i must be 1 or 2, the only solutions are:
# (h_u, h_v) = (1, 2) or (2, 1)
# This means for every edge (u, v), h_u != h_v.
# This is equivalent to checking if the graph is bipartite.
# To minimize sum_{i=1}^n h_i * 3^{n-i}, we want h_1 to be as small as possible (1),
# then h_2 as small as possible, and so on.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    # To minimize the sum, we greedily set h_i = 1 if possible.
    # For each connected component, we have two choices for coloring (1, 2) or (2, 1).
    # We pick the one that makes the first node in the component 1 if possible.
    
    res = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # We want to choose the coloring such that the smallest index in the component
            # gets the value 1.
            # Actually, the problem asks to minimize sum h_i * 3^{n-i}.
            # This is equivalent to lexicographical comparison of (h_1, h_2, ..., h_n).
            # We iterate i from 1 to n. If i is not colored, we force h_i = 1.
            
    # Re-run logic to ensure lexicographical smallest
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Force color[i] = 1 to minimize h_i
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
