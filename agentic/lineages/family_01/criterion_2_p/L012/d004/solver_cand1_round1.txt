import sys

# The condition: (s_u % s_v) + (s_v % s_u) == 1
# If s_u = 1, s_v = 1: (1%1) + (1%1) = 0 + 0 = 0 != 1
# If s_u = 1, s_v = 2: (1%2) + (2%1) = 1 + 0 = 1 (Valid)
# If s_u = 2, s_v = 1: (2%1) + (1%2) = 0 + 1 = 1 (Valid)
# If s_u = 2, s_v = 2: (2%2) + (2%2) = 0 + 0 = 0 != 1
# Thus, the condition is equivalent to s_u != s_v.
# This is a standard 2-coloring problem (bipartite graph check).
# We want to minimize sum(s_i * 3^(n-i)).
# Since 3^(n-i) is larger for smaller i, we want s_1 to be 1 if possible.
# If s_1 is fixed to 1, the rest of the component is determined.
# If a component is not connected to 1, we can choose the color of the 
# smallest index in that component to be 1 to minimize the sum.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To minimize sum, we want the smallest index in this component to be 1
            # Actually, for each component, we have two choices:
            # Choice A: smallest_node = 1, others follow
            # Choice B: smallest_node = 2, others follow
            # We compare the two configurations lexicographically based on indices.
            
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if we should flip colors
            # We want the smallest index in the component to have color 1
            # Find smallest index in component
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
