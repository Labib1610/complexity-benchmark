import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition 1/s_u + 1/s_v = 3/2
    # If s_u = 1, then 1 + 1/s_v = 3/2 => 1/s_v = 1/2 => s_v = 2
    # If s_u = 2, then 1/2 + 1/s_v = 3/2 => 1/s_v = 1 => s_v = 1
    # If s_u >= 3, then 1/s_u <= 1/3. 1/s_v = 3/2 - 1/s_u >= 3/2 - 1/3 = 7/6.
    # This implies s_v < 1, which is impossible for positive integers.
    # Thus, for any connected component, the values must be 1 and 2, alternating.
    # This is a bipartite graph coloring problem.
    
    colors = [0] * (n + 1) # 0: unvisited, 1: color 1, 2: color 2
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # BFS to color component
            component = []
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
            
            # To get lexicographically smallest, we want s_i to be 1 whenever possible.
            # For each component, we have two choices:
            # Choice A: color 1 -> 1, color 2 -> 2
            # Choice B: color 1 -> 2, color 2 -> 1
            # We pick the one that makes the first node in the component 1 if possible.
            
            # Find the smallest index in this component
            comp_nodes = [node for node, c in component]
            first_node = min(comp_nodes)
            
            # If we set s[first_node] = 1, then its color must map to 1.
            # If colors[first_node] == 1, then color 1 -> 1, color 2 -> 2.
            # If colors[first_node] == 2, then color 2 -> 1, color 1 -> 2.
            
            target_color_for_1 = colors[first_node]
            for node, c in component:
                if c == target_color_for_1:
                    ans[node] = 1
                else:
                    ans[node] = 2
                    
    print("YES")
    print(*(ans[1:]))

solve()
