```python
import sys

# The problem asks for an assignment s_i in {1, 2} such that for every edge (u, v),
# {s_u, s_v} = {1, 2}.
# This means s_u != s_v for all edges (u, v).
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# For a connected component, there are two possible colorings (swap 1s and 2s).
# To get the lexicographically smallest sequence, for each connected component,
# we should try to assign 1 to the smallest index node in that component.
# If that leads to a valid 2-coloring, we check if it's better than assigning 2.
# Actually, for each component, we have two choices:
# Choice A: node with min index gets 1.
# Choice B: node with min index gets 2.
# We want the smallest sequence, so we greedily pick the assignment that puts 1s
# at the earliest possible indices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # 1. Keep as is (color[v])
            # 2. Flip (3 - color[v])
            # We want the lexicographically smallest result.
            # Find the smallest index in this component.
            min_node = min(component)
            
            # Try coloring such that min_node has color 1
            # If color[min_node] is 1, use current. Else flip.
            if color[min_node] == 1:
                for node in component:
                    ans[node] = color[node]
            else:
                for node in component:
                    ans[node] = 3 - color[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
