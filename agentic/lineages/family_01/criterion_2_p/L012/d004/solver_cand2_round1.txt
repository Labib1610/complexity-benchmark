```python
import sys

# The eigenvalue condition for matrix [[s_u, 1], [1, s_v]] is:
# det(M - lambda*I) = 0
# (s_u - lambda)(s_v - lambda) - 1 = 0
# lambda^2 - (s_u + s_v)lambda + (s_u*s_v - 1) = 0
# Given lambda = (3 + sqrt(5)) / 2
# lambda^2 = (9 + 5 + 6*sqrt(5)) / 4 = (14 + 6*sqrt(5)) / 4 = (7 + 3*sqrt(5)) / 2
# Substitute lambda^2:
# (7 + 3*sqrt(5))/2 - (s_u + s_v)(3 + sqrt(5))/2 + (s_u*s_v - 1) = 0
# Multiply by 2:
# 7 + 3*sqrt(5) - (s_u + s_v)(3 + sqrt(5)) + 2*s_u*s_v - 2 = 0
# 5 + 3*sqrt(5) - 3(s_u + s_v) - (s_u + s_v)sqrt(5) + 2*s_u*s_v = 0
# Group rational and irrational parts:
# (5 - 3(s_u + s_v) + 2*s_u*s_v) + (3 - (s_u + s_v))sqrt(5) = 0
# For this to hold, both parts must be zero:
# 1) 3 - (s_u + s_v) = 0  => s_u + s_v = 3
# 2) 5 - 3(3) + 2*s_u*s_v = 0 => 5 - 9 + 2*s_u*s_v = 0 => 2*s_u*s_v = 4 => s_u*s_v = 2
# Since s_u, s_v in {1, 2}, s_u + s_v = 3 and s_u*s_v = 2 implies {s_u, s_v} = {1, 2}.
# This is a bipartite coloring problem where connected nodes must have different states.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, try assigning 1 first
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, color[v]))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
