import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unassigned, 1 = Mode 1, 2 = Mode 2
    color = [0] * (n + 1)
    
    # To minimize sum_{i=1}^n (s_i - 1) * 2^{n-i}, we want s_i = 1 as much as possible
    # for smaller i. This is a greedy approach on connected components.
    # For each component, we have two choices of coloring (swap 1 and 2).
    # We pick the one that results in a smaller value for the first node in the component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # We can either keep it as is, or flip all 1s to 2s and 2s to 1s.
            # Let's compare the two options.
            # Option 1: current color[i] is the assignment.
            # Option 2: flip all colors in component.
            
            # We want to minimize sum_{j in component} (s_j - 1) * 2^{n-j}
            # Let val1 = sum_{j in component} (color[j]-1) * 2^{n-j}
            # Let val2 = sum_{j in component} ( (3-color[j])-1 ) * 2^{n-j}
            
            val1 = 0
            val2 = 0
            for node in component:
                val1 += (color[node] - 1) * (1 << (n - node))
                val2 += (2 - color[node]) * (1 << (n - node))
            
            if val2 < val1:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
