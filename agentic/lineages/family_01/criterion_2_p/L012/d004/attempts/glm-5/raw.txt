import sys
from collections import deque

def solve():
    input = sys.stdin.read
    data = input().split()
    
    idx = 0

    m = int(data[idx]); idx += 1

        v = int(data[idx]); idx += 1
        adj[u].append(v)
        adj[v].append(u)
    
    # -1: unvisited, 1 or 2: assigned value
    color = [-1] * (n + 1)
    
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        
        # BFS from start
        color[start] = 1
        queue = deque([start])
        
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 3 - color[u]  # 1 -> 2, 2 -> 1
                    queue.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
    
    print("YES")
    print(' '.join(map(str, color[1:])))
