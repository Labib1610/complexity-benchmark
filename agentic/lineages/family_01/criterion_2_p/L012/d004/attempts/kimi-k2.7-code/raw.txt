```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    bad = False
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            bad = True
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)
    q = deque()

    for i in range(1, n + 1):
        if color[i] != -1:
            continue
        color[i] = 0
        q.append(i)
        while q:
            x = q.popleft()
            cx = color[x]
            for y in adj[x]:
                if color[y] == -1:
                    color[y] = cx ^ 1
                    q.append(y)
                elif color[y] == cx:
                    bad = True

    if bad:
        sys.stdout.write("NO\n")
        return

    out = ['YES']
    out.append(' '.join(str(color[i] + 1) for i in range(1, n + 1)))
    sys.stdout.write('\n'.join(out))

if __name__ == '__main__':
    solve()
```
