```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        # No constraints, choose the lexicographically smallest configuration
        res = [1] * n
        print("YES")
        print(' '.join(map(str, res)))
        return
    
    edges = []
    graph = [[] for _ in range(n)]
    for i in range(m):
        u = int(data[2 + 2*i]) - 1
        v = int(data[2 + 2*i + 1]) - 1
        edges.append((u, v))
        graph[u].append(v)
        graph[v].append(u)
    
    # For the constraint between u and v:
    # (s_u % s_v) + (s_v % s_u) = 1
    # Let's analyze what values of s_u, s_v satisfy this:
    # s_u, s_v ∈ {1, 2}
    # Possible pairs:
    # (1,1): (1%1)+(1%1) = 0+0 = 0 ❌
    # (1,2): (1%2)+(2%1) = 1+0 = 1 ✅
    # (2,1): (2%1)+(1%2) = 0+1 = 1 ✅
    # (2,2): (2%2)+(2%2) = 0+0 = 0 ❌
    # So, adjacent nodes must be assigned different values.
    # This is a bipartite graph check.
    
    color = [-1] * n  # -1: unvisited, 0: color 0, 1: color 1
    
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 0
        comp_nodes = []
        while queue:
            u = queue.popleft()
            comp_nodes.append(u)
            for v in graph[u]:
                if color[v] == -1:
                    color[v] = 1 - color[u]
                    queue.append(v)
                elif color[v] == color[u]:
                    return False, []
        return True, comp_nodes
    
    components = []
    for i in range(n):
        if color[i] == -1:
            is_bipartite, comp = bfs(i)
            if not is_bipartite:
                print("NO")
                return
            components.append(comp)
    
    # Now we need to assign values to minimize sum s_i * 3^(n-1-i)
    # This means earlier indices should prefer smaller values.
    # In each component, we can choose which color gets 1 and which gets 2.
    # We want to minimize the weighted sum.
    result = [0] * n
    
    for comp in components:
        # Two options:
        # Option 0: color 0 -> 1, color 1 -> 2
        # Option 1: color 0 -> 2, color 1 -> 1
        sum0 = 0
        sum1 = 0
        nodes_color0 = []
        nodes_color1 = []
        for u in comp:
            if color[u] == 0:
                nodes_color0.append(u)
                sum0 += 1 * (3 ** (n - 1 - u))
                sum1 += 2 * (3 ** (n - 1 - u))
            else:
                nodes_color1.append(u)
                sum0 += 2 * (3 ** (n - 1 - u))
                sum1 += 1 * (3 ** (n - 1 - u))
        
        if sum0 <= sum1:
            # Assign color 0 -> 1, color 1 -> 2
            for u in nodes_color0:
                result[u] = 1
            for u in nodes_color1:
                result[u] = 2
        else:
            # Assign color 0 -> 2, color 1 -> 1
            for u in nodes_color0:
                result[u] = 2
            for u in nodes_color1:
                result[u] = 1
                
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
```
