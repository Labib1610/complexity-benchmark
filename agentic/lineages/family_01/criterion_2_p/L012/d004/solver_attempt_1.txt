import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To minimize C_n = sum_{i=1}^n (s_i - 1) * 2^{n-i},
    # we want s_i = 1 whenever possible for smaller i.
    # Since the graph is bipartite, each connected component has two possible colorings.
    # For each component, we pick the coloring that makes the first node (smallest index) 1.
    # If that's not possible, we pick the other.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            first_node = min(component)
            
            # We want assignment[first_node] = 1 if possible.
            # If color[first_node] is 1, we keep the current coloring.
            # If color[first_node] is 2, we flip all colors in this component.
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                assignment[node] = color[node]
                
    print("YES")
    print(*(assignment[1:]))

solve()
