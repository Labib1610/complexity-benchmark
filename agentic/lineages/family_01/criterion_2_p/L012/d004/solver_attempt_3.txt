import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The penalty is sum_{i=1}^n (b_i - 1) * 2^{n-i}.
    # To minimize this, we want b_i = 1 for as many small i as possible.
    # This is a bipartite coloring problem. For each connected component,
    # there are at most two ways to color it.
    # Let a component have nodes with colors {c1, c2}.
    # If we assign color c1 to blend 1, then all nodes with color c2 get blend 2.
    # If we assign color c1 to blend 2, then all nodes with color c2 get blend 1.
    # We choose the assignment that results in a smaller penalty.
    
    color = [-1] * (n + 1)
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            component.append(i)
            
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        component.append(v)
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible assignments for this component:
            # Option 0: color 0 -> blend 1, color 1 -> blend 2
            # Option 1: color 0 -> blend 2, color 1 -> blend 1
            
            # Calculate penalty for Option 0
            penalty0 = 0
            for node in component:
                if color[node] == 1:
                    penalty0 += (1 << (n - node))
            
            # Calculate penalty for Option 1
            penalty1 = 0
            for node in component:
                if color[node] == 0:
                    penalty1 += (1 << (n - node))
            
            if penalty0 <= penalty1:
                for node in component:
                    assignment[node] = 1 if color[node] == 0 else 2
            else:
                for node in component:
                    assignment[node] = 2 if color[node] == 0 else 1
                    
    print("YES")
    print(*(assignment[1:]))

solve()
