import sys

# The condition "product of bands is 2" means if one is 1, the other must be 2.
# This is a bipartite graph coloring problem where each edge (u, v) implies
# color(u) != color(v).
# We want the lexicographically smallest assignment (s1, s2, ..., sn).
# This means we prefer 1 over 2 for each sensor in increasing order of index.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # colors: 0 = unassigned, 1 = band 1, 2 = band 2
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 first
            stack = [i]
            colors[i] = 1
            component = []
            
            # Use a stack for DFS to avoid recursion depth issues
            # We need to process components. Since we want lexicographically
            # smallest, we greedily assign 1 to the smallest uncolored node.
            q = [i]
            colors[i] = 1
            while q:
                u = q.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        q.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
