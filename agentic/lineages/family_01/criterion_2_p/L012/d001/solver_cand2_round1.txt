import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # We want to minimize F = sum_{i=1}^n s_i * 2^{n-i}
    # Since s_i is 1 or 2, this is equivalent to minimizing s_1, then s_2, etc.
    # We greedily try to set s_i = 1 if possible.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Component not yet colored.
            # To minimize F, we want s_i = 1.
            # If we set s_i = 1, we must check if it's valid.
            # This is a standard 2-coloring problem.
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # We have two choices:
            # 1. Keep current colors (s_i = color[i])
            # 2. Flip colors (s_i = 3 - color[i])
            # We want to pick the one that makes the sequence lexicographically smaller.
            # Since we process i from 1 to n, we just need to check the first node
            # in the component (which is the smallest index i).
            # If color[i] is 2, we should flip to make s_i = 1.
            if color[i] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
