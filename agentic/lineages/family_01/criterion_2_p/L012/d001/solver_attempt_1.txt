import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # state: 0 = unvisited, 1 = state 1, 2 = state 2
    state = [0] * (n + 1)
    
    # To get lexicographically smallest, we prefer state 1 for smaller indices.
    # We iterate 1 to n. If unvisited, try setting to 1.
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS for this component
            stack = [(i, 1)]
            state[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to keep track of the component
            comp_stack = [i]
            while comp_stack:
                u = comp_stack.pop()
                component.append(u)
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = 3 - state[u]
                        comp_stack.append(v)
                    elif state[v] == state[u]:
                        print("NO")
                        return
            
    print("YES")
    print(*(state[1:]))

solve()
