```python
import sys

# The condition s_u * s_v = 2 means:
# If s_u = 1, then s_v = 2.
# If s_u = 2, then s_v = 1.
# This is equivalent to saying s_u != s_v.
# This is a bipartite graph coloring problem.
# We need to color the graph with 2 colors (1 and 2).
# To get the lexicographically smallest sequence, we want s_1 to be 1.
# If the graph is bipartite, each connected component has two possible colorings.
# For each component, we choose the coloring that makes the smallest index in that
# component have the value 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_node = min(component)
            # We want color[min_node] to be 1.
            # If color[min_node] is 1, we keep the current coloring.
            # If color[min_node] is 2, we flip the colors (1->2, 2->1).
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
```
