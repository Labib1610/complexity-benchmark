import sys

# The problem states that for every edge (u, v), s_u * s_v = 2.
# Since s_i is either 1 or 2, the only way s_u * s_v = 2 is if one is 1 and the other is 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# For each connected component, there are two possible colorings (swapping 1s and 2s).
# To get the lexicographically smallest sequence, for each component, we want to
# assign 1 to the smallest index in that component if possible.
# Specifically, if a component has nodes {v_1, v_2, ..., v_k} with v_1 being the smallest,
# we should try to set s_{v_1} = 1. If that leads to a valid coloring, we take it.
# Since the graph is bipartite, once we fix the color of one node in a component,
# all other nodes in that component are fixed.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            idx_in_comp = 0
            while idx_in_comp < len(component):
                u = component[idx_in_comp]
                idx_in_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # We have a valid coloring for this component.
            # Check if we should flip it to make it lexicographically smaller.
            # The current coloring has color[i] = 1.
            # If we flip, color[i] becomes 2.
            # We want the smallest index in the component to have 1.
            # Let min_node be the smallest index in this component.
            # If color[min_node] == 2, we must flip.
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
