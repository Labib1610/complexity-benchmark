import sys

# The problem asks to minimize sum_{i=1}^n s_i * 2^{n-i} where s_i in {1, 2}.
# This is equivalent to minimizing the number represented by the sequence s_1, s_2, ..., s_n
# in base 2, where 1 corresponds to bit 0 and 2 corresponds to bit 1.
# Since we want to minimize the sum, we want s_1 to be 1 if possible.
# If s_1 is fixed, we want s_2 to be 1 if possible, and so on.
# This is a standard 2-coloring problem (bipartite graph check).
# We can solve this using BFS/DFS on each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To minimize the sum, we greedily assign 1 to the smallest index available.
    # For each connected component, we have two choices:
    # 1. Start with color 1 for the smallest index node in the component.
    # 2. Start with color 2 for the smallest index node in the component.
    # We compare the resulting sequences lexicographically.
    
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # Let the current coloring be c1, c2, ..., ck.
            # The other valid coloring is 3-c1, 3-c2, ..., 3-ck.
            # We want to pick the one that is lexicographically smaller.
            # Since we want to minimize sum s_i * 2^{n-i}, we compare
            # the sequences at the first index where they differ.
            
            # Find the smallest index in this component
            first_node = min(component)
            
            # If color[first_node] is 1, the sequence starts with 1.
            # If color[first_node] is 2, the sequence starts with 2.
            # We prefer 1.
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
