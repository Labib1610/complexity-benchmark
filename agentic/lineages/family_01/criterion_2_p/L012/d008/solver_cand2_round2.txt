```python
import sys

# The determinant of the stress tensor S = [[h_u + h_v, h_u - h_v], [h_u - h_v, h_u + h_v]]
# det(S) = (h_u + h_v)^2 - (h_u - h_v)^2
# det(S) = (h_u^2 + 2*h_u*h_v + h_v^2) - (h_u^2 - 2*h_u*h_v + h_v^2)
# det(S) = 4 * h_u * h_v
# We require det(S) = 8, so 4 * h_u * h_v = 8, which means h_u * h_v = 2.
# Since h_i must be 1 or 2, the only way h_u * h_v = 2 is if one is 1 and the other is 2.
# This is a bipartite coloring problem.
# We need to color the graph with 2 colors (1 and 2) such that every edge connects different colors.
# To minimize the base-3 number h_1 h_2 ... h_n, we want h_i to be 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the connected component
            # To minimize the number, we prefer color 1 for the smallest index
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
