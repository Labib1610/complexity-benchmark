```python
import sys

# The calibration rule:
# 1. C = h_u + 4
# 2. If h_v == 1: C = 2 * C - 1
#    If h_v == 2: C = C + 3
# 3. (C ^ h_u) == 9
#
# Let's test the 4 combinations:
# h_u=1, h_v=1: C = (1+4)*2 - 1 = 9. 9 ^ 1 = 8 != 9. (Invalid)
# h_u=1, h_v=2: C = (1+4) + 3 = 8. 8 ^ 1 = 9. (Valid)
# h_u=2, h_v=1: C = (2+4)*2 - 1 = 11. 11 ^ 2 = 9. (Valid)
# h_u=2, h_v=2: C = (2+4) + 3 = 9. 9 ^ 2 = 11 != 9. (Invalid)
#
# So:
# If h_u=1, then h_v must be 2.
# If h_u=2, then h_v must be 1.
# This means h_u != h_v must hold for all struts.
# This is a bipartite graph problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unassigned, 1 = height 1, 2 = height 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To minimize the sequence as a base-3 number, 
            # we prefer height 1 over height 2.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, color[v]))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
