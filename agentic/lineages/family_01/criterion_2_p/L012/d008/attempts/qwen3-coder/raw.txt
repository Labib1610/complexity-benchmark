```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2

    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return

    # For each edge (u,v), we derive a constraint between h[u] and h[v]
    # The calibration rule:
    # C = h_u + 4
    # if h_v == 1: C = C * 2 - 1
    # if h_v == 2: C = C + 3
    # final check: C ^ h_u == 9

    # Let's compute for all 4 combinations of (h_u, h_v) whether the constraint holds
    # (h_u, h_v) in {1,2} x {1,2}
    valid = {}
    for hu in [1, 2]:
        for hv in [1, 2]:
            C = hu + 4
            if hv == 1:
                C = C * 2 - 1
            else:  # hv == 2
                C = C + 3
            if C ^ hu == 9:
                valid[(hu, hv)] = True
            else:
                valid[(hu, hv)] = False

    # Build implication graph for 2-SAT
    # Variables: for each node i, we have two variables:
    # i*2 (represents h[i] == 1) and i*2+1 (represents h[i] == 2)
    # We'll build a graph where an edge from a to b means "if a then b"
    
    adj = defaultdict(list)
    rev_adj = defaultdict(list)

    for u, v in edges:
        # For each edge, we add constraints based on valid assignments
        # If (h_u, h_v) is not valid, we add implications
        for hu in [1, 2]:
            for hv in [1, 2]:
                if not valid[(hu, hv)]:
                    # If h_u == hu, then h_v != hv
                    # Which means if h_u == hu, then h_v == (3 - hv)
                    # In 2-SAT terms:
                    # not (u_var_hu) OR (v_var_(3-hv))
                    # Which is (u_var_hu) => (v_var_(3-hv))
                    u_var = u * 2 if hu == 1 else u * 2 + 1
                    v_var = v * 2 if (3 - hv) == 1 else v * 2 + 1
                    adj[u_var ^ 1].append(v_var)
                    adj[v_var ^ 1].append(u_var)
                    rev_adj[v_var].append(u_var ^ 1)
                    rev_adj[u_var].append(v_var ^ 1)

    # Now we do 2-SAT using Kosaraju's algorithm
    # First, topological sort on original graph
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(v):
        visited[v] = True
        for u in adj[v]:
            if not visited[u]:
                dfs1(u)
        order.append(v)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Second, DFS on transpose graph in reverse order
    visited = [False] * (2 * n)
    components = [-1] * (2 * n)
    comp_num = 0
    
    def dfs2(v):
        visited[v] = True
        components[v] = comp_num
        for u in rev_adj[v]:
            if not visited[u]:
                dfs2(u)
    
    while order:
        v = order.pop()
        if not visited[v]:
            dfs2(v)
            comp_num += 1
    
    # Check if satisfiable
    for i in range(n):
        if components[i * 2] == components[i * 2 + 1]:
            print("NO")
            return
    
    # Assign values: if component of (i*2) > component of (i*2+1), then h[i] = 1
    # We want the lexicographically smallest assignment (in base 3, so prefer 1 over 2)
    assignment = [0] * n
    for i in range(n):
        if components[i * 2] > components[i * 2 + 1]:
            assignment[i] = 1
        else:
            assignment[i] = 2
    
    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    main()
```
