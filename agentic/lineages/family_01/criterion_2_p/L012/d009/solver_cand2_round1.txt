import sys

# The determinant of the matrix:
# | h_u  u    0 |
# | 0    h_v  v |
# | 1    0    uv|
# is h_u * (h_v * uv - 0) - u * (0 * uv - v * 1) + 0 = h_u * h_v * uv + u * v
# The condition is: h_u * h_v * uv + u * v = 3 * u * v
# Dividing by u * v (since u, v >= 1):
# h_u * h_v + 1 = 3
# h_u * h_v = 2
# Since h_i is in {1, 2}, the only way h_u * h_v = 2 is if one is 1 and the other is 2.
# This means the graph must be bipartite, with the two sets being {h=1} and {h=2}.
# To minimize the base-3 number h_1 h_2 ... h_n, we want h_i = 1 whenever possible.
# This is a standard 2-coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    # To minimize the base-3 number, we prefer h_i = 1.
    # For each connected component, we try setting the first node to 1.
    # If that fails, we try setting it to 2.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try setting color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            component = [i]
            possible = True
            
            # Standard BFS/DFS to 2-color
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
