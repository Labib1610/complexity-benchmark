import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    g = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        g[u].append(v)
        g[v].append(u)

    color = [-1] * n
    ans = [0] * n
    from collections import deque

    for start in range(n):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        comp = [start]
        ok = True
        while q:
            u = q.popleft()
            for v in g[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                    comp.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
        # decide orientation: make smallest index node have value 1 (i.e., parity 0)
        min_node = min(comp)
        if color[min_node] == 1:
            # flip whole component
            for v in comp:
                color[v] ^= 1
        # set answer values
        for v in comp:
            ans[v] = 1 if color[v] == 0 else 2

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
