```python
import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # We'll build a graph where each constraint creates edges
    # The constraints are:
    # 1. (x_u OR x_v) => at least one of x_u, x_v is True
    # 2. (NOT x_u OR NOT x_v) => at least one of x_u, x_v is False
    
    # This means:
    # - If both x_u and x_v are False, constraint 1 fails
    # - If both x_u and x_v are True, constraint 2 fails
    
    # So for each constraint (u,v):
    # - Not both False
    # - Not both True
    
    # This is equivalent to saying x_u != x_v for each constraint.
    
    # So we're looking for a 2-coloring of a graph where each constraint
    # creates an edge between u and v.
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1; v -= 1  # Convert to 0-based indexing
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 means unvisited, 1 means True, 2 means False
    color = [0] * n
    
    # Check if bipartite (2-colorable) using BFS/DFS
    def dfs(node, c):
        color[node] = c
        for nei in adj[node]:
            if color[nei] == 0:
                if not dfs(nei, 3 - c):  # 3-1=2, 3-2=1
                    return False
            elif color[nei] == c:
                return False
        return True
    
    # Process each connected component
    for i in range(n):
        if color[i] == 0:
            if not dfs(i, 1):
                print("NO")
                return
    
    print("YES")
    # For lexicographically smallest, we want earlier nodes to be 1 if possible.
    # Our DFS already ensures this by starting with color 1.
    result = []
    for c in color:
        result.append(str(c))
    print(" ".join(result))

threading.Thread(target=main).start()
```
