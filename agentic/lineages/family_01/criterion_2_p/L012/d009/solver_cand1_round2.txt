```python
import sys

# The constraints (x_u OR x_v) AND (NOT x_u OR NOT x_v)
# are equivalent to x_u XOR x_v = True.
# This means x_u != x_v.
# This is a 2-coloring problem on a graph where edges represent x_u != x_v.
# We want the lexicographically smallest assignment (1 for True, 2 for False).
# 1 is smaller than 2.
# For each connected component, we can pick the value of the first node
# (the one with the smallest index) to be 1, then propagate.
# If we find a contradiction, output NO.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer 1 over 2.
            # For the first node in the component, assign 1.
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, val = stack.pop()
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - val
                        stack.append((v, 3 - val))
                    elif assignment[v] == val:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
