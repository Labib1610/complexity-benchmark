```python
import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem is equivalent to 2-coloring a graph.
    # If the graph is bipartite, a solution exists.
    # To get the lexicographically smallest assignment, we want h_1 = 1.
    # If h_1 = 1, then all nodes in its connected component are fixed.
    # For components not containing 1, we want the smallest value at the 
    # smallest index to be 1.
    
    color = [0] * (n + 1)
    
    def dfs(u, c):
        color[u] = c
        for v in adj[u]:
            if color[v] == 0:
                if not dfs(v, 3 - c):
                    return False
            elif color[v] == c:
                return False
        return True

    # We want h_1 = 1. If 1 is part of a component, color it.
    # Then iterate 1..n, if not colored, color it with 1 to keep it lexicographically small.
    
    # First, process the component containing 1
    if not dfs(1, 1):
        print("NO")
        return
    
    # Then process other components
    for i in range(1, n + 1):
        if color[i] == 0:
            if not dfs(i, 1):
                print("NO")
                return
                
    print("YES")
    print(*(color[1:]))

solve()
```
