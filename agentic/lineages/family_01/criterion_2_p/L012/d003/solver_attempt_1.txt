```python
import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# color(u) != color(v). This is equivalent to checking if the graph is bipartite.
# The penalty function is sum_{i=1}^n (s_i - 1) * 2^{n-i}.
# Since s_i is 1 or 2, (s_i - 1) is 0 or 1.
# The penalty is sum_{i=1}^n (s_i - 1) * 2^{n-i}.
# To minimize this, we want s_i = 1 whenever possible, especially for small i.
# In each connected component, if we fix the color of one node, all other
# nodes in that component are determined.
# For each component, we have two choices:
# Choice A: Assign colors based on setting the first node (by index) to 1.
# Choice B: Assign colors based on setting the first node (by index) to 2.
# We compare the penalty contribution of both choices and pick the smaller one.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            idx_stack = 0
            while idx_stack < len(component):
                u, c = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible colorings for this component
            # Option 1: color[u] as is
            # Option 2: flip all colors (1->2, 2->1)
            
            penalty1 = 0
            penalty2 = 0
            for node, c in component:
                if c == 2:
                    penalty1 += (1 << (n - node))
                else:
                    penalty2 += (1 << (n - node))
            
            if penalty1 <= penalty2:
                for node, c in component:
                    ans[node] = c
            else:
                for node, c in component:
                    ans[node] = 3 - c
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
