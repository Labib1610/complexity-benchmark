import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The condition p_u + p_v = 3 means p_u != p_v.
    # This is a bipartite coloring problem.
    # We want the lexicographically smallest sequence.
    # For each connected component, we have two choices:
    # 1. Start with 1 (then neighbors are 2, etc.)
    # 2. Start with 2 (then neighbors are 1, etc.)
    # We want to pick the choice that makes the first node in the component 1 if possible.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a component. We want the smallest lexicographical result.
            # Find the smallest index in this component.
            start_node = min(component)
            
            # If we set color[start_node] = 1, is it valid?
            # Yes, because it's bipartite.
            # We need to ensure the smallest index in the component gets 1.
            if color[start_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
