```python
import sys

# The condition is that for every edge (u, v), s[u] * s[v] = 2.
# Since s[i] is in {1, 2}, the only way s[u] * s[v] = 2 is if one is 1 and the other is 2.
# This means the graph must be bipartite.
# We want the lexicographically smallest sequence s_1, s_2, ..., s_n.
# This means we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # 0: unvisited, 1: color 1, 2: color 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, we try to set color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
