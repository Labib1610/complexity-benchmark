```python
import sys

# The problem asks for a configuration of 1s and 2s such that for every edge (u, v),
# s[u] * s[v] = 2.
# Since s[i] can only be 1 or 2, the only way s[u] * s[v] = 2 is if one is 1 and the other is 2.
# This is exactly the definition of a bipartite graph.
# We need to find the lexicographically smallest sequence s_1, s_2, ..., s_n.
# To make the sequence lexicographically smallest, we want s_1 = 1.
# If a component contains node 1, s_1 must be 1.
# For other components, we want the smallest node in that component to be 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If node i is not colored, we start a BFS/DFS.
    # To make it lexicographically smallest, we want the smallest node in the component to be 1.
    # Actually, for each connected component, we have two choices:
    # (color 1, color 2) or (color 2, color 1).
    # We want the smallest index in the component to have the smallest possible value (1).
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if the smallest node in this component is assigned 1.
            # If not, flip all colors in this component.
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
```
