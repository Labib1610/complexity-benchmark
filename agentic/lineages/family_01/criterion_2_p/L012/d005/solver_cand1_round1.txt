import sys

# The condition is: sum_{j in neighbors(i)} s_j = k_i * (3 - s_i)
# Let x_i = s_i - 1. Then s_i = x_i + 1, where x_i in {0, 1}.
# The condition becomes: sum_{j in neighbors(i)} (x_j + 1) = k_i * (3 - (x_i + 1))
# sum_{j in neighbors(i)} x_j + k_i = k_i * (2 - x_i)
# sum_{j in neighbors(i)} x_j + k_i = 2*k_i - k_i * x_i
# sum_{j in neighbors(i)} x_j + k_i * x_i = k_i
# This is equivalent to: sum_{j in neighbors(i)} x_j = k_i * (1 - x_i)
# If x_i = 0, then sum_{j in neighbors(i)} x_j = k_i.
# If x_i = 1, then sum_{j in neighbors(i)} x_j = 0.
# This means if x_i = 1, all neighbors must have x_j = 0.
# If x_i = 0, all neighbors must have x_j = 1.
# This is exactly the condition for a bipartite graph.
# If the graph is bipartite, we can color it with 0 and 1.
# Let the two sets be A and B.
# Option 1: All nodes in A have x=0, all nodes in B have x=1.
# Option 2: All nodes in A have x=1, all nodes in B have x=0.
# We want to minimize sum s_i * 3^{n-i} = sum (x_i + 1) * 3^{n-i}.
# This is equivalent to minimizing sum x_i * 3^{n-i}.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            components.append(comp)

    # For each component, we have two choices:
    # Choice 0: nodes with color 0 get x=0, nodes with color 1 get x=1
    # Choice 1: nodes with color 0 get x=1, nodes with color 1 get x=0
    # We want to minimize sum x_i * 3^{n-i}
    
    res_x = [0] * (n + 1)
    for comp in components:
        # Calculate contribution of Choice 0 vs Choice 1
        # Choice 0: sum_{u in comp, color[u]==1} 3^{n-u}
        # Choice 1: sum_{u in comp, color[u]==0} 3^{n-u}
        val0 = 0
        val1 = 0
        for u, c in comp:
            if c == 1:
                val0 += pow(3, n - u)
            else:
                val1 += pow(3, n - u)
        
        if val0 < val1:
            for u, c in comp:
                res_x[u] = c
        else:
            for u, c in comp:
                res_x[u] = 1 - c
                
    print("YES")
    print(*(res_x[i] + 1 for i in range(1, n + 1)))

solve()
