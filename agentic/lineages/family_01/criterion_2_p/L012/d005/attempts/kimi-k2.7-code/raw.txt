import sys

def solve() -> None:
    it = map(int, sys.stdin.buffer.read().split())
    try:
        n = next(it)
        m = next(it)
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        adj[u].append(v)
        adj[v].append(u)

    # 2 = unvisited, 0/1 = chosen color
    color = bytearray([2]) * n

    for i in range(n):
        if color[i] != 2:
            continue
        stack = [i]
        color[i] = 0
        while stack:
            v = stack.pop()
            cv = color[v]
            for to in adj[v]:
                ct = color[to]
                if ct == 2:
                    color[to] = cv ^ 1
                    stack.append(to)
                elif ct == cv:
                    sys.stdout.write("NO\n")
                    return

    sys.stdout.write(
        "YES\n" + " ".join(["1" if c == 0 else "2" for c in color]) + "\n"
    )

if __name__ == "__main__":
    solve()
