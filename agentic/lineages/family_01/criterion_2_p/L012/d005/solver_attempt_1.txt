import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # We want to minimize P(3) = sum_{i=1}^n s_i * 3^{n-i}
    # This is equivalent to minimizing the value lexicographically from s_1 to s_n.
    # Since s_i \in {1, 2}, we prefer s_i = 1 whenever possible.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Component traversal
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard BFS/DFS to find bipartite coloring
            # To minimize cost, we need to check if we can set s_i = 1
            # for the smallest index i in the component.
            # Actually, for each connected component, there are only 2 possible
            # assignments (swap 1s and 2s). We check both and pick the one
            # that is lexicographically smaller.
            
            q = [i]
            visited = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        visited.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible assignments for this component:
            # 1. color[v] as is
            # 2. 3 - color[v]
            
            # Find the first node in this component
            first_node = min(visited)
            
            # Option 1: color[first_node] = 1 (if it's 1) or 2 (if it's 2)
            # We want the assignment where the first node in the component
            # that appears earliest in the sequence 1..n is 1.
            
            # Actually, just compare the two assignments lexicographically.
            # Since we process components independently, we just need to ensure
            # that for each component, the assignment is chosen to be lexicographically
            # smallest.
            
            # Let's store the two possible assignments
            assignment1 = {}
            assignment2 = {}
            for node in visited:
                assignment1[node] = color[node]
                assignment2[node] = 3 - color[node]
            
            # Compare lexicographically
            # Find the smallest index in this component
            comp_nodes = sorted(visited)
            first = comp_nodes[0]
            
            if assignment1[first] < assignment2[first]:
                for node in visited:
                    color[node] = assignment1[node]
            else:
                for node in visited:
                    color[node] = assignment2[node]
                    
    print("YES")

solve()
