```python
import sys

# The energy formula for a bus (u, v) is E(u, v) = 3(s_u + s_v) - 2(s_u * s_v).
# Possible values for (s_u, s_v):
# (1, 1) -> 3(2) - 2(1) = 4
# (1, 2) -> 3(3) - 2(2) = 5
# (2, 1) -> 3(3) - 2(2) = 5
# (2, 2) -> 3(4) - 2(4) = 4
# We want the sum of energies to be 5m.
# Since each bus contributes either 4 or 5, and we want the sum to be 5m,
# every bus must contribute exactly 5.
# This means for every edge (u, v), s_u must be different from s_v.
# This is equivalent to checking if the graph is bipartite.
# To minimize sum(s_i * 3^(n-i)), we want s_i to be 1 whenever possible.
# For each connected component, we have two choices for the 2-coloring.
# Let the two sets be A and B. We can assign 1 to A and 2 to B, or vice versa.
# We want to pick the assignment that makes the sequence lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # For this component, we have two colorings:
            # 1: nodes with color 1 get s=1, nodes with color 2 get s=2
            # 2: nodes with color 1 get s=2, nodes with color 2 get s=1
            # We want to pick the one that is lexicographically smaller.
            # Find the first node in this component.
            first_node = min(node for node, c in component)
            
            # If color[first_node] == 1:
            # Option 1: s[first_node] = 1
            # Option 2: s[first_node] = 2
            # Clearly Option 1 is better.
            if color[first_node] == 1:
                for node, c in component:
                    res[node] = c
            else:
                for node, c in component:
                    res[node] = 3 - c
                    
    print("YES")
    print(*(res[1:]))

solve()
```
