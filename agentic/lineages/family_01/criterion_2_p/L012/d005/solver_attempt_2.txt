import sys

# The cost function C_n = 2 * C_{n-1} + s_i - 1
# C_n = (s_1 - 1) * 2^{n-1} + (s_2 - 1) * 2^{n-2} + ... + (s_n - 1) * 2^0
# Since s_i \in {1, 2}, s_i - 1 \in {0, 1}.
# To minimize C_n, we want s_i - 1 to be 0 whenever possible, starting from the most significant bit (i=1).
# This means we prefer s_i = 1 over s_i = 2 for smaller i.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # s_i + s_j = 3 means s_i != s_j. This is a bipartite coloring problem.
    # We want to minimize the cost, so for each connected component,
    # we try to assign s_i = 1 to the smallest index in that component if possible.
    
    color = [-1] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            # Start BFS/DFS for component
            component = []
            stack = [(i, 0)] # 0 or 1
            color[i] = 0
            component.append(i)
            
            # To minimize cost, we want the smallest index in the component to be 1 (s_i-1 = 0)
            # Let's find all nodes in component first
            q = [i]
            visited = {i}
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if v not in visited:
                        visited.add(v)
                        q.append(v)
            
            # Now color the component. Try setting color[min_idx] = 0 (s=1)
            min_idx = min(q)
            color[min_idx] = 0
            stack = [min_idx]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    res = []
    for i in range(1, n + 1):
        # color 0 -> s=1, color 1 -> s=2
        res.append(str(color[i] + 1))
    print(" ".join(res))

solve()
