import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The hash H_n = sum_{i=1}^n (t_i - 1) * 2^{n-i}
    # To minimize H_n, we want t_i = 1 whenever possible, starting from i=1.
    # This is a 2-coloring problem. We want to assign color 0 (representing 1)
    # or 1 (representing 2) to each node.
    # To minimize the hash, for each connected component, we try to assign
    # the smallest possible value (1) to the smallest index in that component.
    
    color = [-1] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            # Start BFS/DFS for this component
            # We want to assign color 0 to node i to minimize the hash
            component = []
            stack = [(i, 0)]
            color[i] = 0
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            idx_comp = 0
            q = [i]
            color[i] = 0
            
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    # If we reach here, a valid coloring exists.
    # The color array contains 0 or 1. We map 0 -> 1 and 1 -> 2.
    result = []
    for i in range(1, n + 1):
        result.append(str(color[i] + 1))
        
    print("YES")
    print(" ".join(result))

solve()
