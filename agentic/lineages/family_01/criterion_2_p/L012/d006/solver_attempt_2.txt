```python
import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To minimize the hash H_n = 2*H_{n-1} + p_n - 1,
    # we want p_1 to be 1, then p_2 to be 1 if possible, etc.
    # This is equivalent to lexicographically smallest sequence (p_1, ..., p_n)
    # where p_i in {1, 2}.
    
    color = {} # 1 or 2
    
    for i in range(1, n + 1):
        if i not in color:
            # Try assigning 1 first to minimize the hash
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                for v in adj[u]:
                    if v not in color:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if this component could have been smaller by starting with 2
            # Actually, for each connected component, we have two choices:
            # (c1, c2, ...) or (3-c1, 3-c2, ...).
            # We want the smallest p_i for the smallest i in the component.
            # If the smallest index i in the component has color[i] == 2,
            # we should flip all colors in this component.
            
            first_node = min(node for node, c in component)
            if color[first_node] == 2:
                for node, c in component:
                    color[node] = 3 - c
                    
    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

solve()
```
