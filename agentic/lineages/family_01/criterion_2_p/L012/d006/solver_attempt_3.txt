```python
import sys

# The problem asks to minimize D_n = 2 * D_{n-1} + c_n - 1.
# Expanding this:
# D_n = (c_1 - 1) * 2^{n-1} + (c_2 - 1) * 2^{n-2} + ... + (c_n - 1) * 2^0
# Since c_i is 1 or 2, (c_i - 1) is 0 or 1.
# To minimize D_n, we want (c_1 - 1) to be 0, then (c_2 - 1) to be 0, and so on.
# This means we prefer c_i = 1 whenever possible.
# This is a 2-coloring problem (bipartite graph check).
# For each connected component, we have two possible colorings (swap 1s and 2s).
# We want to choose the coloring that makes the first node in the component 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    ans = [0] * (n + 1)

    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            idx_stack = 0
            while idx_stack < len(component):
                u, c = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if we can make the first node in this component 1
            # The first node is i. If color[i] is 1, we are good.
            # If color[i] is 2, we must flip all colors in this component.
            if color[i] == 2:
                for node, c in component:
                    ans[node] = 3 - c
            else:
                for node, c in component:
                    ans[node] = c
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
