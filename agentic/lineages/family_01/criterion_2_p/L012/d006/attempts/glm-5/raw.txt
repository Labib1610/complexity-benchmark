import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    visited = [False] * (n + 1)
    
    for start in range(1, n + 1):
        if visited[start]:
            continue
        stack = [(start, 0)]
        visited[start] = True
        while stack:
            node, p = stack.pop()
            parent[node] = p
            order.append(node)
            for nb in adj[node]:
                if not visited[nb]:
                    visited[nb] = True
                    stack.append((nb, node))
    
    for node in order:
        c = 1
        for nb in adj[node]:
            if nb == parent[node]:
                continue
            if color[nb] == c:
                c = 2
        color[node] = c
    
    for u in range(1, n + 1):
        for v in adj[u]:
            if color[u] == color[v]:
                print("NO")
                return
    
    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

solve()
