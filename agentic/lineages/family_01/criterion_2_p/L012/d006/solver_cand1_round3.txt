import sys

# The condition: distance between (u, h_u) and (v, h_v) is sqrt((u-v)^2 + 1).
# Distance squared = (u-v)^2 + (h_u - h_v)^2.
# We need (u-v)^2 + (h_u - h_v)^2 = (u-v)^2 + 1.
# This implies (h_u - h_v)^2 = 1, so |h_u - h_v| = 1.
# Since h_i is 1 or 2, this means h_u != h_v.
# This is a bipartite graph coloring problem.
# We need to color the graph with 2 colors (1 and 2).
# To minimize sum(h_i * 3^(n-i)), we want h_i to be 1 whenever possible.
# For each connected component, we have two choices of coloring (swap 1 and 2).
# Let a component have nodes with indices S1 and S2.
# If we assign color 1 to S1 and 2 to S2, the contribution is sum_{i in S1} 1*3^(n-i) + sum_{j in S2} 2*3^(n-j).
# If we assign color 2 to S1 and 1 to S2, the contribution is sum_{i in S1} 2*3^(n-i) + sum_{j in S2} 1*3^(n-j).
# We choose the one that gives the smaller sum.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Compare two colorings
            # Coloring A: nodes with color 1 get 1, nodes with color 2 get 2
            # Coloring B: nodes with color 1 get 2, nodes with color 2 get 1
            # We want to minimize sum(h_i * 3^(n-i))
            # Let S1 be nodes with color 1, S2 be nodes with color 2
            # Sum A = sum_{u in S1} 1*3^(n-u) + sum_{v in S2} 2*3^(n-v)
            # Sum B = sum_{u in S1} 2*3^(n-u) + sum_{v in S2} 1*3^(n-v)
            # Sum A - Sum B = sum_{u in S1} (-3^(n-u)) + sum_{v in S2} (3^(n-v))
            # If Sum A < Sum B, choose A.
            
            diff = 0
            for u, c in component:
                if c == 1:
                    diff -= (3**(n - u))
                else:
                    diff += (3**(n - u))
            
            if diff <= 0:
                for u, c in component:
                    ans[u] = c
            else:
                for u, c in component:
                    ans[u] = 3 - c
                    
    print("YES")
    print(*(ans[1:]))

solve()
