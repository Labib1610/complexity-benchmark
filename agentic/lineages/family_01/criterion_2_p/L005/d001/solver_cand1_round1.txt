import sys

# The problem asks to assign s_i in {1, 2} such that for every edge (u, v),
# s_u * s_v = 2.
# Let x_i = log2(s_i). Since s_i in {1, 2}, x_i in {0, 1}.
# The condition s_u * s_v = 2 becomes log2(s_u) + log2(s_v) = log2(2) = 1.
# So x_u + x_v = 1 (mod 2).
# This is a standard bipartite graph coloring problem.
# If the graph is bipartite, we can color it with 0 and 1.
# x_i = 0 corresponds to s_i = 2^0 = 1.
# x_i = 1 corresponds to s_i = 2^1 = 2.
# To get the lexicographically smallest assignment, we want s_i = 1 as much as possible.
# For each connected component, we have two choices for the coloring (x_i or 1-x_i).
# We choose the one that results in a smaller sequence of s_i values.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible assignments for this component:
            # Option 0: node u gets s = 2^color[u]
            # Option 1: node u gets s = 2^(1-color[u])
            # We want the lexicographically smallest sequence.
            # Compare the first node in the component (smallest index).
            # Actually, we need to compare the whole sequence.
            # Since we want lexicographical smallest, we compare the values
            # at the smallest index in the component.
            
            comp_nodes = sorted([node for node, c in component])
            first_node = comp_nodes[0]
            
            # If we use color[first_node] as the exponent, s = 2^color[first_node]
            # If we use 1-color[first_node] as the exponent, s = 2^(1-color[first_node])
            # We pick the one that gives s=1 if possible.
            
            # Let's check which assignment makes the first node 1.
            # If color[first_node] == 0, then s = 2^0 = 1.
            # If color[first_node] == 1, then s = 2^1 = 2.
            # If we flip, s becomes 2^(1-color[first_node]).
            
            # We want to minimize the sequence.
            # For each node u in component, s_u = 2^(color[u] ^ flip)
            # We want to choose flip in {0, 1} to minimize the sequence.
            
            def get_seq(f):
                return [2**(color[u] ^ f) for u in comp_nodes]
            
            seq0 = get_seq(0)
            seq1 = get_seq(1)
            
            if seq0 <= seq1:
                best_flip = 0
            else:
                best_flip = 1
                
            for u, c in component:
                ans[u] = 2**(c ^ best_flip)
                
    print("YES")
    print(*(ans[1:]))

solve()
