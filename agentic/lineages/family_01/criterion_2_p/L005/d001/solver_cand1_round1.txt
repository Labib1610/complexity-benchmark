import sys

# The problem states that for any two substations u and v with a proximity report,
# power(u) + power(v) = 3.
# Since power levels are {1, 2}, this means power(u) != power(v).
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest configuration, we want to assign 1
# to the smallest possible indices whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest configuration, we iterate 1 to n.
    # If a node is not colored, we start a BFS/DFS.
    # In each connected component, we have two choices:
    # 1. Assign 1 to the first node encountered, 2 to its neighbors, etc.
    # 2. Assign 2 to the first node encountered, 1 to its neighbors, etc.
    # We pick the one that results in a lexicographically smaller sequence.
    
    ans = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard BFS/DFS to find component and check bipartiteness
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a component. We have two colorings:
            # C1: color[v]
            # C2: 3 - color[v]
            # We want to choose the one that makes the sequence lexicographically smallest.
            # We compare the values at the smallest index in the component.
            
            # Let's check which coloring is better for the first node in the component
            # (which is the smallest index in the component).
            # Actually, we need to compare the whole sequence.
            # Since we want the lexicographically smallest, we want the smallest
            # possible value at the smallest index.
            
            # Let 'first' be the smallest index in this component.
            # If we set color[first] = 1, then all nodes with color[v] == color[first] get 1,
            # and all nodes with color[v] != color[first] get 2.
            # If we set color[first] = 2, then all nodes with color[v] == color[first] get 2,
            # and all nodes with color[v] != color[first] get 1.
            
            # We compare the two options:
            # Option A: nodes with color[v] == color[first] get 1, others get 2.
            # Option B: nodes with color[v] == color[first] get 2, others get 1.
            
            # We want the one that puts a 1 at the smallest index possible.
            # The smallest index in the component is 'first'.
            # In Option A, ans[first] = 1. In Option B, ans[first] = 2.
            # So Option A is always better.
            
            for node in component:
                if color[node] == color[i]:
                    ans[node] = 1
                else:
                    ans[node] = 2
                    
    print("YES")
    print(*(ans[1:]))

solve()
