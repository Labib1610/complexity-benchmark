import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a node is not colored, we try assigning 1 first.
    # Since it's a bipartite check, we use DFS.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # Using a stack for iterative DFS to avoid recursion limits
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Collect component nodes
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # The component is bipartite. We have two choices for the root:
            # color[root] = 1 or color[root] = 2.
            # To make the sequence lexicographically smallest, we want the 
            # smallest values at the smallest indices.
            # We check both assignments for the component and pick the one 
            # that results in a lexicographically smaller sequence.
            
            # Actually, for each component, we can flip all colors.
            # We want the smallest value at the smallest index in the component.
            # Let's find the smallest index 'min_idx' in this component.
            # We want color[min_idx] to be 1.
            
            min_idx = min(component)
            if color[min_idx] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
