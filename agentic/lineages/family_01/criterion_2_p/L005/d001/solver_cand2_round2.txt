import sys

# The condition is that for every edge (u, v), state[u] * state[v] = 2.
# Since states are in {1, 2}, the only way to get a product of 2 is (1, 2) or (2, 1).
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with two colors (say 0 and 1).
# Let color[u] be the color of node u.
# If we assign state[u] = 1 if color[u] == 0 and state[u] = 2 if color[u] == 1,
# then for any edge (u, v), color[u] != color[v], so one is 1 and the other is 2.
# Product is 1 * 2 = 2.
# To get the lexicographically smallest configuration, we want state[1] to be 1 if possible.
# If state[1] = 1, then for any node u in the same connected component as 1,
# its state is determined by its distance from 1.
# If node u is in a different component, we can choose state[u] = 1 to be lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we want state[i] = 1 if possible.
            # This corresponds to color[i] = 0.
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    res = []
    for i in range(1, n + 1):
        # color 0 -> state 1, color 1 -> state 2
        res.append(str(color[i] + 1))
        
    print("YES")
    print(" ".join(res))

solve()
