import sys

# The problem states that for any pair (u, v) in a proximity record,
# the sum of their assigned frequency bands must be 3.
# Since the bands are {1, 2}, this means if u is 1, v must be 2, and vice versa.
# This is equivalent to saying the graph must be bipartite, and for each
# connected component, we must assign colors such that adjacent nodes have
# different colors.
# To get the lexicographically smallest assignment, for each connected component,
# we try assigning 1 to the smallest index node in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want the smallest index
            # in the component to be 1.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a list to keep track of nodes in this component
            # to verify bipartiteness
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
    print("YES")
    print(*(color[1:]))

solve()
