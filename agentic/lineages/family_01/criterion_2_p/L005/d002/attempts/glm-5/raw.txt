```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        # BFS to assign colors; we want lexicographically smallest,
        # so we try to assign 1 to the smallest-indexed node in each component.
        # Since we iterate start from 1 to n, the first uncolored node
        # gets color 1.
        color[start] = 1
        q = deque([start])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - color[u]
                    q.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
    
    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

solve()
```
