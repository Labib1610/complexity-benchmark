import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # s_u + s_v = 3 means s_u != s_v.
    # This is a bipartite coloring problem.
    # We want the lexicographically smallest assignment.
    # For each connected component, we have two choices:
    # 1. Start with s_root = 1
    # 2. Start with s_root = 2
    # We want to pick the choice that makes the sequence lexicographically smallest.
    
    color = [0] * (n + 1) # 0: unvisited, 1: s=1, 2: s=2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            possible = True
            idx_comp = 0
            while idx_comp < len(component):
                u, c = component[idx_comp]
                idx_comp += 1
                
                for v in adj[u]:
                    if color[v] == 0:
                        new_c = 3 - c
                        color[v] = new_c
                        component.append((v, new_c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check which assignment is better for this component
            # We want the smallest value at the smallest index in the component.
            # Let's find the smallest index in this component.
            min_node = min(node for node, c in component)
            
            # If we set color[min_node] = 1, is it better than color[min_node] = 2?
            # Yes, 1 is always better.
            # So we force the smallest index in the component to be 1.
            # If the current coloring has color[min_node] == 2, flip all colors.
            if color[min_node] == 2:
                for node, c in component:
                    color[node] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
