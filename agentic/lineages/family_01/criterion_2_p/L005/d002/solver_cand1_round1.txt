```python
import sys

# The condition is x^2 + 3xy + y^2 = 11
# Possible modes are {1, 2}.
# If x=1, y=1: 1 + 3 + 1 = 5 != 11
# If x=1, y=2: 1 + 6 + 4 = 11 (Valid)
# If x=2, y=1: 4 + 6 + 1 = 11 (Valid)
# If x=2, y=2: 4 + 12 + 4 = 20 != 11
# Thus, for every adjacency pair (u, v), mode(u) must be different from mode(v).
# This means the graph must be bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS to check bipartiteness and assign lexicographically smallest
            # To get lexicographically smallest, we want node 1 to be 1,
            # then node 2 to be 1 if possible, etc.
            # Actually, for each connected component, we have two choices:
            # 1. Start node with 1, neighbors with 2
            # 2. Start node with 2, neighbors with 1
            # We want the resulting array to be lexicographically smallest.
            # This means for each component, we pick the coloring that makes
            # the first node in the component (by index) have mode 1.
            
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # After coloring, check if the smallest index in this component is 1
            # Actually, the rule is: for each component, we can swap 1s and 2s.
            # To make the whole array lexicographically smallest, we look at the
            # smallest index 'k' in the component. We want color[k] to be 1.
            min_node = min(node for node, c in component)
            if color[min_node] == 2:
                for node, c in component:
                    color[node] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
```
