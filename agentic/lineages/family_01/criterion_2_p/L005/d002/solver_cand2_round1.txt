```python
import sys

# The condition is that for any connected u, v, state[u] ^ state[v] = 3.
# Since states are 1 or 2:
# 1 ^ 2 = 3
# 1 ^ 1 = 0
# 2 ^ 2 = 0
# This means if u and v are connected, they must have different states.
# This is a standard bipartite graph coloring problem.
# We want the lexicographically smallest configuration.
# To get the lexicographically smallest, we prefer state 1 over state 2.
# For each connected component, we can pick the state of the first node (smallest index)
# to be 1, then propagate. If we find a conflict, output NO.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    state = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want state[i] = 1
            stack = [(i, 1)]
            state[i] = 1
            while stack:
                u, s = stack.pop()
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = 3 - s
                        stack.append((v, state[v]))
                    elif state[v] == s:
                        print("NO")
                        return
                        
    print("YES")
    print(*(state[1:]))

solve()
```
