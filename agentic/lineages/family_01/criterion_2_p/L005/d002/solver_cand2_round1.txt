import sys

# The problem states: for any pair (u, v) in proximity, 
# their signal overlap (u_band & v_band) must be 0.
# Available bands are 1 (binary 01) and 2 (binary 10).
# 1 & 1 = 1 (not 0)
# 2 & 2 = 2 (not 0)
# 1 & 2 = 0 (valid)
# 2 & 1 = 0 (valid)
# This means if u and v are connected, they must have different bands.
# This is a standard 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest assignment, we want 1s as much as possible.
# For each connected component, we can either:
# 1. Assign 1 to the first node (and propagate).
# 2. Assign 2 to the first node (and propagate).
# We compare the resulting assignments lexicographically.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    result = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Try assigning 1 to the first node of the component
            # vs assigning 2 to the first node.
            # Since we want lexicographically smallest, we compare the 
            # resulting colorings for this component.
            
            # Option 1: color[i] = 1
            # Option 2: color[i] = 2 (which flips all colors in component)
            
            # To be lexicographically smallest, we want the smallest index 
            # in the component to have 1 if possible.
            # Actually, for each component, we just need to pick the 
            # coloring that results in a smaller value at the smallest index.
            
            # Let's just pick the coloring where the smallest node index 
            # in the component gets 1.
            
            # Find smallest node in component
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
