```python
import sys

# The condition is p_u^{p_v} + p_v^{p_u} = 3.
# Possible states are {1, 2}.
# If p_u = 1, p_v = 1: 1^1 + 1^1 = 2 != 3.
# If p_u = 1, p_v = 2: 1^2 + 2^1 = 1 + 2 = 3. (Valid)
# If p_u = 2, p_v = 1: 2^1 + 1^2 = 2 + 1 = 3. (Valid)
# If p_u = 2, p_v = 2: 2^2 + 2^2 = 4 + 4 = 8 != 3.
# Thus, for every edge (u, v), p_u must be different from p_v.
# This is exactly the definition of a bipartite graph.
# We need to check if the graph is bipartite and find the lexicographically
# smallest assignment of {1, 2}.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, we prefer color 1 for the
            # smallest node in each connected component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
