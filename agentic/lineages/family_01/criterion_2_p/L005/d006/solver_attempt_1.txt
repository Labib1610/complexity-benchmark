```python
import sys

# The problem asks for an assignment of 1 or 2 to each node such that for every edge (u, v),
# the product of the multipliers is 2.
# If multiplier(u) * multiplier(v) = 2, then one must be 1 and the other must be 2.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we want to assign 1 to the smallest
# possible index whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If a node is not colored, we start a BFS/DFS.
    # To make the sequence lexicographically smallest, we prefer assigning 1 to the
    # smallest index node in each connected component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if we can swap colors in this component to make it lexicographically smaller
            # The current assignment is color[i] = 1.
            # If we swap, color[i] becomes 2.
            # We want the smallest value at the smallest index.
            # Let's check the first node in the component (the one with the smallest index).
            # If color[first_node] is 2, we should swap to make it 1.
            
            first_node = min(node for node, c in component)
            if color[first_node] == 2:
                for node, c in component:
                    color[node] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
```
