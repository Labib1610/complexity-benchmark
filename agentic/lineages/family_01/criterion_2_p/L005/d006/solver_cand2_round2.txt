```python
import sys

# The matrix S = [[0, x, y], [x, 0, 0], [y, 0, 0]]
# The characteristic equation is det(S - lambda*I) = 0
# det([[ -L, x, y], [x, -L, 0], [y, 0, -L]]) = 0
# -L(L^2) - x(-xL) + y(xy) = 0
# -L^3 + x^2*L + y^2*L = 0
# -L(L^2 - (x^2 + y^2)) = 0
# Eigenvalues are 0, sqrt(x^2 + y^2), -sqrt(x^2 + y^2)
# The largest eigenvalue is sqrt(x^2 + y^2).
# We need sqrt(x^2 + y^2) = sqrt(5), so x^2 + y^2 = 5.
# Since x, y in {1, 2}, the only way to get 1^2 + 2^2 = 5 is {x, y} = {1, 2}.
# This means for every critical pair (u, v), the modes c_u and c_v must be different.
# If c_u = 1, then c_v = 2. If c_u = 2, then c_v = 1.
# This is equivalent to 2-coloring the graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, we prefer 1 over 2.
            # Start BFS/DFS with color 1.
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
