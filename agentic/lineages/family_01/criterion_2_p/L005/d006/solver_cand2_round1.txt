import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition: Volume1 + Volume2 = 6*pi
    # Volume = pi * r^2 * h
    # pi * x^2 * y + pi * y^2 * x = 6 * pi
    # x*y*(x + y) = 6
    # Possible settings are {1, 2}.
    # If x=1, y=1: 1*1*(1+1) = 2 != 6
    # If x=1, y=2: 1*2*(1+2) = 6 (Valid)
    # If x=2, y=1: 2*1*(2+1) = 6 (Valid)
    # If x=2, y=2: 2*2*(2+2) = 16 != 6
    # Thus, for every pair (u, v), x and y must be different (one 1, one 2).
    # This is a bipartite coloring problem.
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
