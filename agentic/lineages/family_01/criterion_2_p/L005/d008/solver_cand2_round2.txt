import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Stress formula: (x+3)^y + (y+3)^x
    # If x=1, y=1: (1+3)^1 + (1+3)^1 = 4 + 4 = 8
    # If x=1, y=2: (1+3)^2 + (2+3)^1 = 16 + 5 = 21
    # If x=2, y=1: (2+3)^1 + (1+3)^2 = 5 + 16 = 21
    # If x=2, y=2: (2+3)^2 + (2+3)^2 = 25 + 25 = 50
    # We need product to be 21^m.
    # This means every connection must have stress 21, which happens if and only if
    # one machine is 1 and the other is 2.
    # This is a bipartite graph coloring problem.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # We want lexicographically smallest sequence.
    # The graph might have multiple connected components.
    # For each component, we have two choices of coloring (swap 1s and 2s).
    # To make it lexicographically smallest, for each component, we check
    # which coloring gives a smaller value at the first node of the component.
    
    final_colors = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Try both colorings
            c1 = color[i]
            # Option 1: color[i] = c1
            # Option 2: color[i] = 3 - c1
            
            # We want the smallest value at the smallest index in the component.
            # Actually, we want the smallest sequence overall.
            # For each component, we can flip all colors.
            # Let's pick the flip that makes the smallest index in the component 1.
            
            min_node = min(component)
            if color[min_node] == 1:
                for node in component:
                    final_colors[node] = color[node]
            else:
                for node in component:
                    final_colors[node] = 3 - color[node]
                    
    print("YES")
    print(*(final_colors[1:]))

solve()
