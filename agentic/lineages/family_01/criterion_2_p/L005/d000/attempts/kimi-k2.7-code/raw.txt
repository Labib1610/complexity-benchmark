import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    ans = ['0'] * n

    for i in range(n):
        if color[i] != -1:
            continue
        q = deque([i])
        color[i] = 0
        comp = [i]
        mn = i
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                    comp.append(v)
                    if v < mn:
                        mn = v
                elif color[v] == color[u]:
                    sys.stdout.write("NO")
                    return

        # Put the smallest-index vertex in this component in shift 1.
        if color[mn] == 0:
            for u in comp:
                ans[u] = '1' if color[u] == 0 else '2'
        else:
            for u in comp:
                ans[u] = '2' if color[u] == 0 else '1'

    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(ans))

if __name__ == "__main__":
    solve()
