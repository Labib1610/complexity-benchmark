import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
        
    # Power modes are 1 or 2.
    # Cubes: 1^3 = 1, 2^3 = 8.
    # Token for (u, v) = (mode(u)^3 + mode(v)^3) % 9.
    # Let x_i = 1 if mode(i) = 1, x_i = 0 if mode(i) = 2.
    # If mode(i) = 1, cube = 1. If mode(i) = 2, cube = 8.
    # Token = (cube(u) + cube(v)) % 9.
    # If both 1: (1+1)%9 = 2.
    # If both 2: (8+8)%9 = 16%9 = 7.
    # If one 1, one 2: (1+8)%9 = 0.
    
    # Let c_i be the cube of mode i. c_i \in {1, 8}.
    # Sum_{edges} (c_u + c_v) % 9 = 0.
    # Let d_i be the degree of node i.
    # Sum_{i=1}^n d_i * c_i \equiv 0 (mod 9).
    # Let k be the number of machines set to mode 1.
    # Then (n-k) machines are set to mode 2.
    # Sum_{i=1}^n d_i * c_i = Sum_{i in Set1} d_i * 1 + Sum_{i in Set2} d_i * 8
    # = Sum_{i in Set1} d_i + 8 * (Sum_{all} d_i - Sum_{i in Set1} d_i)
    # = Sum_{i in Set1} d_i + 8 * (2m) - 8 * Sum_{i in Set1} d_i
    # = 16m - 7 * Sum_{i in Set1} d_i \equiv 0 (mod 9).
    # 16m \equiv 7m (mod 9).
    # 7m - 7 * Sum_{i in Set1} d_i \equiv 0 (mod 9).
    # 7 * (m - Sum_{i in Set1} d_i) \equiv 0 (mod 9).
    # Since gcd(7, 9) = 1, this is equivalent to:
    # Sum_{i in Set1} d_i \equiv m (mod 9).
    
    degrees = [0] * n
    for u, v in edges:
        degrees[u] += 1
        degrees[v] += 1
        
    # We want lexicographically smallest, so we want as many 1s as possible at the start.
    # This is a variation of the subset sum problem (or partition problem).
    # Since we need Sum_{i in Set1} d_i \equiv m (mod 9), we can use DP.
    # dp[i][rem] = can we reach remainder 'rem' using a subset of first 'i' nodes.
    
    target = m % 9
    dp = {0: []} # remainder: list of indices in Set1
    
    # To get lexicographically smallest, we want c_i=1 if possible.
    # c_i=1 means i is in Set1.
    # We iterate i from 0 to n-1.
    # If we pick i for Set1, new_rem = (old_rem + degrees[i]) % 9.
    
    # Since n is 2*10^5, we can't store the full set.
    # But we only need to know if it's possible.
    # Let's use a bitmask or boolean array for reachability.
    
    reachable = [None] * (n + 1)
    reachable[0] = {0}
    
    for i in range(n):
        d = degrees[i] % 9
        reachable[i+1] = reachable[i].copy()
        for r in reachable[i]:
            reachable[i+1].add((r + d) % 9)
            
    if target not in reachable[n]:
        print("NO")
        return
    
    # Backtrack to find the lexicographically smallest
    res = [0] * n
    curr_rem = target
    for i in range(n - 1, -1, -1):
        d = degrees[i] % 9
        # Can we set res[i] = 1? (i.e., i in Set1)
        # We need (curr_rem - d) % 9 to be reachable using first i elements
        prev_rem = (curr_rem - d) % 9
        if prev_rem in reachable[i]:
            res[i] = 1
            curr_rem = prev_rem
        else:
            res[i] = 2
            
    print("YES")
    print(*(res))

solve()
