import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Let x_i be the mode of machine i (1 or 2).
    # Let k_i be the degree of machine i.
    # If x_i = 1, sum_{j in N(i)} x_j = 2 * k_i
    # If x_i = 2, sum_{j in N(i)} x_j = k_i
    # Let y_i = x_i - 1 (so y_i is 0 or 1).
    # If x_i = 1 (y_i = 0), sum (y_j + 1) = 2 * k_i => sum y_j + k_i = 2 * k_i => sum y_j = k_i
    # If x_i = 2 (y_i = 1), sum (y_j + 1) = k_i => sum y_j + k_i = k_i => sum y_j = 0
    
    # This is equivalent to:
    # If y_i = 0, sum_{j in N(i)} y_j = k_i
    # If y_i = 1, sum_{j in N(i)} y_j = 0
    
    # This means if y_i = 1, all neighbors must have y_j = 0.
    # If y_i = 0, all neighbors must have y_j = 1.
    # This is a bipartite coloring problem.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # Now we have two possible assignments for each connected component:
    # Option A: y_i = color[i]
    # Option B: y_i = 1 - color[i]
    # We need to check if these satisfy the conditions.
    # Condition: If y_i = 1, sum y_j = 0. If y_i = 0, sum y_j = k_i.
    
    def check(y):
        for i in range(1, n + 1):
            s = sum(y[v] for v in adj[i])
            if y[i] == 1:
                if s != 0: return False
            else:
                if s != len(adj[i]): return False
        return True

    # Since the graph is bipartite, let the two sets be U and V.
    # If we set y_i = 0 for all i in U and y_i = 1 for all i in V:
    # For i in U: y_i=0, sum y_j = sum_{j in V} 1 = k_i. (Satisfied)
    # For i in V: y_i=1, sum y_j = sum_{j in U} 0 = 0. (Satisfied)
    # So y_i = color[i] is always a valid solution if the graph is bipartite.
    # To get lexicographically smallest x_i, we want y_i = 0 as much as possible.
    
    # For each component, we can choose y_i = color[i] or y_i = 1 - color[i].
    # We want to minimize the sequence x_1, x_2, ..., x_n.
    # x_i = y_i + 1. So we want y_i = 0.
    
    y = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            comp = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Try setting y_i = color[i] for all i in comp
            # vs y_i = 1 - color[i] for all i in comp
            # Check which one gives smaller x_i at the first index
            first = min(comp)
            y1 = [0] * (n + 1)
            y2 = [0] * (n + 1)
            for node in comp:
                y1[node] = color[node]
                y2[node] = 1 - color[node]
            
            if y1[first] <= y2[first]:
                for node in comp: y[node] = y1[node]
            else:
                for node in comp: y[node] = y2[node]
                
    if not check(y):
        print("NO")
    else:
        print("YES")
        print(*(y[i] + 1 for i in range(1, n + 1)))

solve()
