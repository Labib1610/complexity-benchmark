import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    s = [0] * (n + 1)
    for i in range(1, n + 1):
        if s[i] == 0:
            s[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                k = len(adj[u])
                target = 2 * k if s[u] == 1 else k
                cur = sum(s[v] for v in adj[u] if s[v] != 0)
                rem = target - cur
                cnt = sum(1 for v in adj[u] if s[v] == 0)
                if rem < cnt or rem > 2 * cnt:
                    print("NO")
                    return
                for v in adj[u]:
                    if s[v] == 0:
                        if rem == 2 * cnt:
                            s[v] = 2
                            rem -= 2
                            cnt -= 1
                            stack.append(v)
                        elif rem == cnt:
                            s[v] = 1
                            rem -= 1
                            cnt -= 1
                            stack.append(v)
                        else:
                            s[v] = 1
                            rem -= 1
                            cnt -= 1
                            stack.append(v)
    
    for u in range(1, n + 1):
        k = len(adj[u])
        target = 2 * k if s[u] == 1 else k
        cur = sum(s[v] for v in adj[u])
        if cur != target:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, s[1:])))

solve()
