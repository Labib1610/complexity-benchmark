```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Color nodes with BFS to check bipartiteness
    color = [0] * (n + 1)  # 0: unvisited, 1: color 1, 2: color 2
    visited = [False] * (n + 1)
    
    # Check each connected component
    for start_node in range(1, n + 1):
        if visited[start_node]:
            continue
            
        queue = deque([start_node])
        color[start_node] = 1
        visited[start_node] = True
        
        while queue:
            node = queue.popleft()
            
            for neighbor in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = 3 - color[node]  # Flip color (1->2, 2->1)
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:
                    print("NO")
                    return
    
    # Verify the coloring satisfies all edges
    for u, v in edges:
        if color[u] + color[v] != 3:
            print("NO")
            return
    
    # Construct result with lexicographically smallest assignment
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:  # Isolated node, assign 1
            result.append("1")
        else:
            result.append(str(color[i]))
    
    print("YES")
    print(" ".join(result))

solve()
```
