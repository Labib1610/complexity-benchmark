```python
import sys

# The problem states that for every contact point (u, v), the product of their 
# resonance settings must be 2. Since settings are in {1, 2}, the product is 2 
# if and only if one block is 1 and the other is 2.
# This is equivalent to a 2-coloring problem on a graph.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest configuration, we want s_1 to be 1.
# If s_1 = 1, then all neighbors of 1 must be 2, their neighbors must be 1, etc.
# We perform a BFS/DFS on each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To get lexicographically smallest, we want s_i to be 1 whenever possible.
    # For each component, we can try setting the first node to 1.
    # Since we want the lexicographically smallest sequence, we iterate 1 to n.
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we prefer 1 over 2.
            # If we haven't colored this component, set the first node to 1.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                target_c = 3 - c
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = target_c
                        stack.append((v, target_c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
