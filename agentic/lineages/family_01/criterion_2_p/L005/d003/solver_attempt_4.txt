```python
import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    roles = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a node is not visited, we start a BFS/DFS.
    # To ensure lexicographical smallest, we prefer assigning 1 whenever possible.
    
    for i in range(1, n + 1):
        if roles[i] == 0:
            # Start BFS/DFS
            stack = [i]
            roles[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if roles[v] == 0:
                        roles[v] = 3 - roles[u]
                        stack.append(v)
                    elif roles[v] == roles[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(roles[1:]))

if __name__ == '__main__':
    solve()
```
