import sys

# The condition is x^3 + y^3 + xy(u + v) = 9 + 2(u + v)
# Let S = u + v.
# If x=1, y=1: 1 + 1 + 1*S = 2 + S. This matches 9 + 2S only if S = 7.
# If x=2, y=2: 8 + 8 + 4*S = 16 + 4S. This matches 9 + 2S only if 2S = -7 (impossible).
# If x=1, y=2: 1 + 8 + 2*S = 9 + 2S. This matches 9 + 2S for any S.
# If x=2, y=1: 8 + 1 + 2*S = 9 + 2S. This matches 9 + 2S for any S.
#
# So, for every pair (u, v), we must have x != y.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 1 and 2.
# To get the lexicographically smallest configuration, we want 1s at the 
# smallest indices possible.
# In each connected component, there are two ways to color:
# 1. Color A with 1, Color B with 2
# 2. Color A with 2, Color B with 1
# We choose the one that makes the configuration lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            is_bipartite = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two colorings: 
            # Option 1: color[v] as is
            # Option 2: flip colors (1->2, 2->1)
            # We want the lexicographically smallest result.
            # Find the smallest index in the component.
            comp_nodes = sorted(component)
            
            # Try coloring the smallest node with 1
            # If color[smallest] is 1, keep as is. If 2, flip.
            first = comp_nodes[0]
            if color[first] == 1:
                # Option 1: color[v]
                # Option 2: 3 - color[v]
                # Compare lexicographically
                opt1 = [color[node] for node in comp_nodes]
                opt2 = [3 - color[node] for node in comp_nodes]
            else:
                opt1 = [3 - color[node] for node in comp_nodes]
                opt2 = [color[node] for node in comp_nodes]
            
            if opt1 <= opt2:
                for node in comp_nodes:
                    ans[node] = color[node] if color[first] == 1 else 3 - color[node]
            else:
                for node in comp_nodes:
                    ans[node] = 3 - color[node] if color[first] == 1 else color[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
