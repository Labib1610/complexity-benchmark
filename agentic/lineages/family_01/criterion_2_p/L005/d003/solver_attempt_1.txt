```python
import sys

# The problem asks for a 2-coloring of a graph such that every edge connects
# vertices of different colors (1 and 2). This is equivalent to checking if
# the graph is bipartite. If it is, we need the lexicographically smallest
# configuration, which means we prefer setting 1 over 2 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color array: 0 = unvisited, 1 = color 1, 2 = color 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest configuration, we iterate from 1 to n.
    # If a component is not visited, we start BFS/DFS.
    # For the lexicographically smallest, we want node 1 to be 1, then node 2 to be 1 if possible, etc.
    # Actually, for a bipartite graph, once the color of one node in a connected
    # component is fixed, all others are determined.
    # To make the configuration lexicographically smallest, for each component,
    # we try assigning 1 to the smallest index node in that component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
