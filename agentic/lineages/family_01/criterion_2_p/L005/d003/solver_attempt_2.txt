import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition "sum to 3" means if u is 1, v must be 2, and vice versa.
    # This is a standard 2-coloring problem (bipartite check).
    # To prioritize lower states for lower-numbered blocks, we greedily
    # assign state 1 to the smallest unvisited node in each connected component.
    
    state = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS from i, assigning state 1
            stack = [(i, 1)]
            state[i] = 1
            while stack:
                u, s = stack.pop()
                target_s = 3 - s
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = target_s
                        stack.append((v, target_s))
                    elif state[v] != target_s:
                        print("NO")
                        return
                        
    print("YES")
    print(*(state[1:]))

if __name__ == '__main__':
    solve()
