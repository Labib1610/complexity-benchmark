```python
import sys

# The system of equations:
# p*x + q*y = u
# p*y + q*x = -v
# This is a linear system for p and q:
# [x y] [p] = [ u]
# [y x] [q] = [-v]
# Determinant D = x^2 - y^2.
# If x = y, then x^2 - y^2 = 0.
# If x = y = 1, then p + q = u and p + q = -v, so u = -v.
# If x = y = 2, then 2p + 2q = u and 2p + 2q = -v, so u = -v.
# Since u, v are machine indices (1 to n), u, v > 0, so u = -v is impossible.
# Thus, x must not equal y.
# Since modes are {1, 2}, one must be 1 and the other 2.
# If x=1, y=2:
# p + 2q = u
# 2p + q = -v
# Multiply first by 2: 2p + 4q = 2u
# Subtract second: 3q = 2u + v => q = (2u + v) / 3
# p = u - 2q = u - 2(2u + v)/3 = (3u - 4u - 2v)/3 = (-u - 2v)/3
# If x=2, y=1:
# 2p + q = u
# p + 2q = -v
# 4p + 2q = 2u
# 3p = 2u + v => p = (2u + v) / 3
# q = -v - 2p = -v - 2(2u + v)/3 = (-3v - 4u - 2v)/3 = (-4u - 5v)/3
# In both cases, for p and q to exist, the equations must be consistent.
# Actually, the problem implies that for any pair (u, v), the modes x and y
# must be different. If x != y, then x=1, y=2 or x=2, y=1.
# In either case, the determinant x^2 - y^2 = 1 - 4 = -3 or 4 - 1 = 3.
# Since the determinant is non-zero, p and q always exist for any u, v.
# The constraint is simply that for every pair (u, v), mode(u) != mode(v).
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
