```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    adj = defaultdict(list)
    edges = []
    
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
        index += 2
    
    # For each edge (u,v), we need s[u] * s[v] = 2
    # So the pair (s[u], s[v]) must be (1,2) or (2,1)
    # This defines a bipartite graph where nodes must be colored with two colors
    # such that neighbors have different "types"
    
    # We can model this as a 2-SAT problem or as a graph coloring problem
    # But there's a simpler way: since the product must be 2, 
    # for every edge, one endpoint must be 1 and the other must be 2.
    # So the graph must be bipartite.
    
    color = [-1] * n
    visited = [False] * n
    
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 0
        while queue:
            node = queue.popleft()
            if visited[node]:
                continue
            visited[node] = True
            for neighbor in adj[node]:
                if color[neighbor] == -1:
                    color[neighbor] = 1 - color[node]
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:
                    return False
        return True
    
    for i in range(n):
        if not visited[i] and adj[i]:
            if not bfs(i):
                print("NO")
                return
    
    # Now we need to assign values 1 and 2 based on the bipartition
    # Let's say color 0 -> value 1, color 1 -> value 2
    # But we also need to check that all edges satisfy the constraint
    result = [0] * n
    
    # Handle isolated nodes: set them to 1 (lexicographically smallest)
    for i in range(n):
        if color[i] == -1:
            result[i] = 1
        else:
            result[i] = 1 if color[i] == 0 else 2
    
    # Verify all edges satisfy the constraint
    for u, v in edges:
        if result[u] * result[v] != 2:
            print("NO")
            return
    
    print("YES")
    print(" ".join(map(str, result)))

if __name__ == "__main__":
    main()
```
