```python
import sys

# The problem asks to assign 1 or 2 to each participant such that for every contract (u, v),
# status(u) * status(v) = 2.
# Since statuses are in {1, 2}, the product is 2 if and only if one is 1 and the other is 2.
# This is equivalent to a 2-coloring problem on a graph where edges represent constraints.
# If status(u) = 1, then status(v) must be 2. If status(u) = 2, then status(v) must be 1.
# This is a bipartite graph check.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # status: 0 = unassigned, 1 = status 1, 2 = status 2
    status = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a component is not colored, we try assigning 1 first.
    for i in range(1, n + 1):
        if status[i] == 0:
            # Start BFS/DFS
            stack = [(i, 1)]
            status[i] = 1
            while stack:
                u, s = stack.pop()
                target_s = 3 - s
                for v in adj[u]:
                    if status[v] == 0:
                        status[v] = target_s
                        stack.append((v, target_s))
                    elif status[v] != target_s:
                        print("NO")
                        return
    
    print("YES")
    print(*(status[1:]))

if __name__ == '__main__':
    solve()
```
