import sys

# The condition is x^3 * y + x * y^3 = 10, where x, y in {1, 2}.
# If x=1, y=1: 1*1 + 1*1 = 2 != 10
# If x=1, y=2: 1*2 + 1*8 = 10 (Valid)
# If x=2, y=1: 8*1 + 2*1 = 10 (Valid)
# If x=2, y=2: 8*2 + 2*8 = 32 != 10
# Thus, for every critical pair (u, v), we must have c[u] != c[v].
# This is a bipartite graph coloring problem.
# We need the lexicographically smallest assignment:
# For each connected component, we try assigning 1 to the smallest index node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for component
            # To get lexicographically smallest, we want the smallest index 
            # in the component to be 1.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if this assignment is the lexicographically smallest
            # We have two choices for the component: color[u] or 3-color[u]
            # We want the smallest index in the component to have color 1.
            # The current assignment already sets the first visited node to 1.
            # Since we iterate 1 to n, the first node encountered in a component
            # is the smallest index.

    print("YES")
    print(*(color[1:]))

solve()
