```python
import sys

# The condition: for machine i, let S_i be its state (1 or 2).
# Let N(i) be the list of states of its partners.
# The condition is: S_i + (sum of states in N(i)) / count(i) = 3
# Let d_i be the degree of machine i (number of pairs involving i).
# S_i + (sum of states in N(i)) / d_i = 3
# d_i * S_i + sum of states in N(i) = 3 * d_i
# sum of states in N(i) = d_i * (3 - S_i)
# If S_i = 1, sum of states in N(i) = 2 * d_i
# If S_i = 2, sum of states in N(i) = 1 * d_i
# Let x_i = S_i - 1, so x_i is 0 or 1.
# If S_i = 1 (x_i = 0), sum of (x_j + 1) = 2 * d_i => sum(x_j) + d_i = 2 * d_i => sum(x_j) = d_i
# If S_i = 2 (x_i = 1), sum of (x_j + 1) = 1 * d_i => sum(x_j) + d_i = d_i => sum(x_j) = 0
# This means if x_i = 1, all neighbors must have x_j = 0.
# If x_i = 0, the number of neighbors with x_j = 1 must be exactly d_i.
# Wait, if x_i = 0, sum(x_j) = d_i. Since x_j is 0 or 1, this means all neighbors must have x_j = 1.
# So: if x_i = 0, all neighbors must be 1. If x_i = 1, all neighbors must be 0.
# This is exactly the condition for a bipartite graph!
# We need to check if the graph is bipartite. If so, the two partitions define the states.
# Lexicographically smallest: we want as many 1s as possible at the start.
# S_i = 1 corresponds to x_i = 0. S_i = 2 corresponds to x_i = 1.
# We want x_i = 0 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        idx += 2
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
        else:
            # If a machine is paired with itself, S_i + S_i = 3 => 2*S_i = 3, impossible.
            print("NO")
            return

    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            # Try to set color[i] = 0 (S_i = 1) to be lexicographically smallest
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # Check the condition: sum(x_j) = d_i if x_i=0, else 0
    # Actually, the bipartite condition is sufficient.
    # If x_i=0, neighbors are 1. sum(x_j) = d_i * 1 = d_i. Correct.
    # If x_i=1, neighbors are 0. sum(x_j) = d_i * 0 = 0. Correct.
    
    res = [color[i] + 1 for i in range(1, n + 1)]
    print("YES")
    print(*(res))

solve()
```
