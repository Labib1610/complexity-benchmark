import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition: sum of (c_u | c_v) for all m pairs = 3m.
    # Since c_i is 1 or 2, (c_u | c_v) can be:
    # (1|1)=1, (1|2)=3, (2|1)=3, (2|2)=2.
    # To get sum = 3m, every pair must result in 3.
    # (c_u | c_v) = 3 means one is 1 and the other is 2.
    # This is a bipartite coloring problem.
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, we have a valid coloring.
    # We need the lexicographically smallest.
    # The components can be colored in two ways (1,2) or (2,1).
    # To make it lexicographically smallest, for each component,
    # we check if setting the smallest node to 1 is better.
    
    final_colors = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Try both colorings for this component
            # Option 1: color[i] stays as is
            # Option 2: flip all colors in component
            
            # Find smallest node in component
            min_node = min(component)
            
            # If we set min_node to 1, what are the colors?
            # Current color of min_node is color[min_node]
            # If color[min_node] == 1, keep as is.
            # If color[min_node] == 2, flip.
            
            if color[min_node] == 1:
                for node in component:
                    final_colors[node] = color[node]
            else:
                for node in component:
                    final_colors[node] = 3 - color[node]
                    
    print("YES")
    print(*(final_colors[1:]))

solve()
