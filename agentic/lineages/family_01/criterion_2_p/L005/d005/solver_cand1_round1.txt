```python
import sys

# The matrix M = [[s_u, sqrt(2)], [sqrt(2), s_v]]
# The characteristic equation is det(M - lambda*I) = 0
# (s_u - lambda)(s_v - lambda) - 2 = 0
# lambda^2 - (s_u + s_v)lambda + (s_u*s_v - 2) = 0
# The eigenvalues are rational if and only if the discriminant D is a perfect square.
# D = (s_u + s_v)^2 - 4(s_u*s_v - 2)
# D = s_u^2 + 2*s_u*s_v + s_v^2 - 4*s_u*s_v + 8
# D = s_u^2 - 2*s_u*s_v + s_v^2 + 8
# D = (s_u - s_v)^2 + 8
#
# Possible values for s_u, s_v are {1, 2}:
# If s_u = 1, s_v = 1: D = (1-1)^2 + 8 = 8 (not a square)
# If s_u = 2, s_v = 2: D = (2-2)^2 + 8 = 8 (not a square)
# If s_u = 1, s_v = 2: D = (1-2)^2 + 8 = 1 + 8 = 9 = 3^2 (a square)
# If s_u = 2, s_v = 1: D = (2-1)^2 + 8 = 1 + 8 = 9 = 3^2 (a square)
#
# Thus, for every flagged pair (u, v), we must have s_u != s_v.
# This is equivalent to checking if the graph is bipartite.
# To minimize the base-3 number (s_1, s_2, ..., s_n), we want s_i to be 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To minimize the base-3 number, we greedily assign 1 to the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
