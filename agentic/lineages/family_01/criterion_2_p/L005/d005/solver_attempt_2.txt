import sys

# The problem asks to assign 1 or 2 to each node such that for every edge (u, v),
# pressure(u) + pressure(v) = 3.
# This means pressure(u) != pressure(v).
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest sequence of 1s and 2s.
# To get the lexicographically smallest sequence, we want to assign 1 to node 1,
# then 1 to node 2 if possible, etc.
# For each connected component, we have two choices:
# 1. Assign the first node in the component to 1.
# 2. Assign the first node in the component to 2.
# We pick the choice that results in a smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Component found. Perform BFS/DFS to check bipartiteness
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            is_bipartite = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a bipartite component.
            # We can either keep the current coloring or flip it (1->2, 2->1).
            # We want the lexicographically smallest result.
            # Find the smallest index in this component.
            first_node = min(component)
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
