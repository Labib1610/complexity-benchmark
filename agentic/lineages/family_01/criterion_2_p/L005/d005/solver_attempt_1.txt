```python
import sys

# The problem asks to assign 1 or 2 to each node such that for every edge (u, v),
# mode(u) + mode(v) = 3.
# This means mode(u) != mode(v).
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest sequence of 1s and 2s.
# For each connected component, we can assign 1 to the smallest index node,
# then propagate the colors.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To get lexicographically smallest sequence, we iterate 1 to n.
    # If a node is uncolored, we start BFS/DFS.
    # To make it lexicographically smallest, we prefer assigning 1 to the 
    # smallest index node in each component.
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS for this component
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
