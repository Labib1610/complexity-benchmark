```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    ans = [0] * n

    for i in range(n):
        if color[i] != -1:
            continue
        comp = []
        dq = deque([i])
        color[i] = 0
        while dq:
            v = dq.popleft()
            comp.append(v)
            cv = color[v]
            for w in adj[v]:
                if color[w] == -1:
                    color[w] = cv ^ 1
                    dq.append(w)
                elif color[w] == cv:
                    sys.stdout.write("NO\n")
                    return
        min_v = min(comp)
        if color[min_v] == 0:
            for v in comp:
                ans[v] = color[v] + 1
        else:
            for v in comp:
                ans[v] = 2 - color[v]

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, ans)))

if __name__ == "__main__":
    solve()
```
