```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Color array: -1 = unvisited, 0 = color 1, 1 = color 2
    color = [-1] * n
    result = [0] * n
    
    # Check if bipartite for each connected component
    for start in range(n):
        if color[start] == -1:
            queue = deque([start])
            color[start] = 0
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # If bipartite, construct the answer
    # We want lexicographically smallest solution
    # Assign 1 to color 0 nodes, 2 to color 1 nodes
    for i in range(n):
        if color[i] == 0:
            result[i] = 1
        else:
            result[i] = 2
    
    # Verify the solution
    for u, v in edges:
        if result[u] + result[v] != 3:
            print("NO")
            return
    
    print("YES")
    print(" ".join(map(str, result)))

solve()
```
