```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    side = [-1] * (n + 1)          # 0 / 1 for bipartition
    ans = [0] * (n + 1)            # final values 1 or 2
    for start in range(1, n + 1):
        if side[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        side[start] = 0
        comp = [start]
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if side[v] == -1:
                    side[v] = side[u] ^ 1
                    q.append(v)
                    comp.append(v)
                elif side[v] == side[u]:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            print("NO")
            return
        # choose orientation so that minimal index gets value 1
        min_node = min(comp)
        s_min = side[min_node]
        for v in comp:
            ans[v] = 1 if side[v] == s_min else 2

    print("YES")
    print(' '.join(str(ans[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
