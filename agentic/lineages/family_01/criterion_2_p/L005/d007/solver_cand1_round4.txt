import sys

# The resonance factor for a pair (x, y) is x^3 + y^3.
# Possible values for x, y are {1, 2}.
# Possible resonance factors:
# (1, 1) -> 1^3 + 1^3 = 2
# (1, 2) -> 1^3 + 2^3 = 9
# (2, 1) -> 2^3 + 1^3 = 9
# (2, 2) -> 2^3 + 2^3 = 16
# We need the product of m factors to be 9^m.
# This means every pair must have a resonance factor of 9.
# Thus, for every pair (u, v), we must have (c_u, c_v) = (1, 2) or (2, 1).
# This is equivalent to saying the graph must be bipartite, and for every edge (u, v),
# c_u != c_v.
# Since we want the lexicographically smallest calibration, we want as many 1s as possible.
# In each connected component, we have two choices for coloring (1, 2, 1, 2... or 2, 1, 2, 1...).
# To make the sequence lexicographically smallest, for each component, we check which
# coloring gives a smaller value at the first node (the node with the smallest index).
# If the first node in the component is i, we prefer c_i = 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            # to decide the coloring
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring, check if we can make the first node 1
            if color[i] == 2:
                for node in q:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
