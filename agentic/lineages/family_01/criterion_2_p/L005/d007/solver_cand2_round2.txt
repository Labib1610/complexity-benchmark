```python
import sys

# The determinant of S = [[x^2, x+y], [x+y, y^2]] is x^2 * y^2 - (x+y)^2.
# We are given x, y in {1, 2}.
# If x=1, y=1: det = 1*1 - (1+1)^2 = 1 - 4 = -3
# If x=1, y=2: det = 1*4 - (1+2)^2 = 4 - 9 = -5
# If x=2, y=1: det = 4*1 - (2+1)^2 = 4 - 9 = -5
# If x=2, y=2: det = 4*4 - (2+2)^2 = 16 - 16 = 0
# The condition det = -5 is satisfied if and only if x != y.
# This means the graph must be bipartite.
# We need to find the lexicographically smallest assignment (1 or 2).
# For each connected component, we can assign the first node to 1 or 2.
# To be lexicographically smallest, we prefer 1 for the smallest index in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for component
            # To get lexicographically smallest, we assign 1 to the smallest node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
