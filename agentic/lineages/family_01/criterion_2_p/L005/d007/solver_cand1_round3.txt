import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(1000000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # 2-SAT: 1 -> 0, 2 -> 1
    # Node i mode 1: 2*(i-1), mode 2: 2*(i-1)+1
    # Constraint: u != v => (u1, v2), (u2, v1)
    adj_2sat = [[] for _ in range(2 * n)]
    for i in range(1, n + 1):
        for neighbor in adj[i]:
            u, v = i - 1, neighbor - 1
            adj_2sat[2 * u].append(2 * v + 1)
            adj_2sat[2 * u + 1].append(2 * v)
            adj_2sat[2 * v].append(2 * u + 1)
            adj_2sat[2 * v + 1].append(2 * u)
            
    dfn = [-1] * (2 * n)
    low = [-1] * (2 * n)
    scc = [-1] * (2 * n)
    stk = []
    in_stk = [False] * (2 * n)
    timer = 0
    scc_cnt = 0
    
    def tarjan(u):
        nonlocal timer, scc_cnt
        dfn[u] = low[u] = timer
        timer += 1
        stk.append(u)
        in_stk[u] = True
        for v in adj_2sat[u]:
            if dfn[v] == -1:
                tarjan(v)
                low[u] = min(low[u], low[v])
            elif in_stk[v]:
                low[u] = min(low[u], dfn[v])
        if low[u] == dfn[u]:
            while True:
                node = stk.pop()
                in_stk[node] = False
                scc[node] = scc_cnt
                if node == u: break
            scc_cnt += 1
            
    for i in range(2 * n):
        if dfn[i] == -1:
            tarjan(i)
            
    for i in range(n):
        if scc[2 * i] == scc[2 * i + 1]:
            print("NO")
            return
            
    # To check if a partial assignment is valid:
    # We need to check if there exists a valid 2-SAT assignment
    # given fixed values for 1..i.
    # This is equivalent to checking if the remaining constraints are satisfiable.
    # Since we need the lexicographically smallest sequence (1 before 2),
    # we greedily pick 1 if possible.
    
    res = [0] * n
    fixed = [None] * n
    
    # Precompute reachability or use the property that 2-SAT is satisfiable
    # if no variable is forced to both values.
    # For the greedy approach, we can use the SCCs.
    # A variable x is forced to true if scc[x_false] -> scc[x_true]
    
    def is_possible(current_fixed):
        # Check if current_fixed is consistent with 2-SAT
        # This is a standard 2-SAT check with forced assignments
        temp_adj = [list(x) for x in adj_2sat]
        for i, val in enumerate(current_fixed):
            if val is not None:
                # If val is 1 (mode 1), force 2*i to be true, 2*i+1 to be false
                # Add edge 2*i+1 -> 2*i
                if val == 1:
                    temp_adj[2 * i + 1].append(2 * i)
                else:
                    temp_adj[2 * i].append(2 * i + 1)
        
        # Re-run SCC on temp_adj
        dfn_t = [-1] * (2 * n)
        low_t = [-1] * (2 * n)
        scc_t = [-1] * (2 * n)
        stk_t = []
        in_stk_t = [False] * (2 * n)
        timer_t = 0
        scc_cnt_t = 0
        
        for i in range(2 * n):
            if dfn_t[i] == -1:
                # Iterative Tarjan to avoid recursion limit
                stack = [(i, 0)]
                while stack:
                    u, pi = stack[-1]
                    if pi == 0:
                        dfn_t[u] = low_t[u] = timer_t
                        timer_t += 1
                        stk_t.append(u)
                        in_stk_t[u] = True
                    found = False
                    for j in range(pi, len(temp_adj[u])):
                        v = temp_adj[u][j]
                        if dfn_t[v] == -1:
                            stack[-1] = (u, j + 1)
                            stack.append((v, 0))
                            found = True
                            break
                        elif in_stk_t[v]:
                            low_t[u] = min(low_t[u], dfn_t[v])
                    if not found:
                        if low_t[u] == dfn_t[u]:
                            while True:
                                node = stk_t.pop()
                                in_stk_t[node] = False
                                scc_t[node] = scc_cnt_t
                                if node == u: break
                            scc_cnt_t += 1
                        stack.pop()
                        if stack:
                            low_t[stack[-1][0]] = min(low_t[stack[-1][0]], low_t[u])
        
        for i in range(n):
            if scc_t[2 * i] == scc_t[2 * i + 1]:
                return False
        return True

    print("YES")
    for i in range(n):
        fixed[i] = 1
        if is_possible(fixed):
            res[i] = 1
        else:
            fixed[i] = 2
            res[i] = 2
            
    print(*(res))

solve()
