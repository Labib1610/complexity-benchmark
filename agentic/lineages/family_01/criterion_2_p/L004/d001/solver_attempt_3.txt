import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a node is not colored, we try to color it 1.
    # This is a 2-coloring problem (bipartite check).
    # Since we want the lexicographically smallest, we greedily assign 1
    # whenever possible.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [i]
            color[i] = 1
            component = []
            
            # Collect component nodes
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if this component's coloring is optimal.
            # Actually, for a bipartite component, there are only two ways to color it:
            # 1. The current way (color[i]=1)
            # 2. The flipped way (color[i]=2)
            # We compare them lexicographically.
            
            # Store current coloring
            comp_colors = {node: color[node] for node in component}
            
            # Try flipping
            flipped_colors = {node: 3 - comp_colors[node] for node in component}
            
            # Find the first node in this component (by index)
            first_node = min(component)
            
            # If the current coloring has a 2 at the first node, 
            # but the flipped one has a 1, we must flip.
            if comp_colors[first_node] == 2:
                for node in component:
                    color[node] = flipped_colors[node]
            else:
                # Current is already better or equal at the first node.
                # Since it's a bipartite component, the coloring is fixed 
                # once the first node's color is chosen.
                pass

    print("YES")
    print(*(color[1:]))

solve()
