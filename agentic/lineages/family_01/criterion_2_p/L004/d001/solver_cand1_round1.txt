```python
import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# the sum of their tuning parameters (1 or 2) is odd.
# 1 + 2 = 3 (odd), 1 + 1 = 2 (even), 2 + 2 = 4 (even).
# This means u and v must have different colors (1 and 2).
# This is a standard bipartite graph check.
# To get the lexicographically smallest sequence, we iterate from 1 to n.
# If a node is unvisited, we assign it 1 (the smallest possible value)
# and perform BFS/DFS to color its connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS from i, assign 1 to i to keep it lexicographically smallest
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
