 ```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    ans = [0] * n

    for i in range(n):
        if color[i] != -1:
            continue

        dq = deque([i])
        color[i] = 0
        comp = []

        while dq:
            v = dq.popleft()
            comp.append(v)
            for w in adj[v]:
                if color[w] == -1:
                    color[w] = color[v] ^ 1
                    dq.append(w)
                elif color[w] == color[v]:
                    print("NO")
                    return

        min_v = min(comp)
        target_color = color[min_v]
        for v in comp:
            ans[v] = 1 if color[v] == target_color else 2

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
```
