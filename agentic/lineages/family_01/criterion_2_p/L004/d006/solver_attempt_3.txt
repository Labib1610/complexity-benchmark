```python
import sys

# The problem asks for a configuration of power states {1, 2} such that for every
# interference edge (u, v), the power states are different (since 1 < 2 is the 
# only way to satisfy "one state is strictly greater than the other").
# This is equivalent to checking if the graph is bipartite.
# To minimize the configuration lexicographically, we iterate through nodes 1 to n.
# For each connected component, we try assigning 1 to the first node encountered.
# If that leads to a valid 2-coloring, we keep it. If not, we try 2.
# Since we want to minimize node 1, then node 2, etc., we process nodes in order.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning 1 to the first node of the component to minimize lexicographically
            stack = [(i, 1)]
            color[i] = 1
            component = [i]
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, color[v]))
                        component.append(v)
                    elif color[v] == c:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # If 1 didn't work, try 2 for the root of this component
                for node in component:
                    color[node] = 0
                
                stack = [(i, 2)]
                color[i] = 2
                possible = True
                while stack:
                    u, c = stack.pop()
                    for v in adj[u]:
                        if color[v] == 0:
                            color[v] = 3 - c
                            stack.append((v, color[v]))
                        elif color[v] == c:
                            possible = False
                            break
                    if not possible:
                        break
            
            if not possible:
                print("NO")
                return

    print("YES")
    print(*(color[1:]))

solve()
```
