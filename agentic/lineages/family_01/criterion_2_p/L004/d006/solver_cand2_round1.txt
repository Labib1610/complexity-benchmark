```python
import sys

# The problem states:
# Level 1: {Gateway}
# Level 2: {Gateway, Processor}
# Symmetric difference of modules for (u, v) must be {Processor}.
# Let S(u) be the set of modules for u.
# S(u) Δ S(v) = {Processor}
# If S(u) = {G}, S(v) = {G, P}, then S(u) Δ S(v) = {P}. (Valid)
# If S(u) = {G, P}, S(v) = {G}, then S(u) Δ S(v) = {P}. (Valid)
# If S(u) = {G}, S(v) = {G}, then S(u) Δ S(v) = {}. (Invalid)
# If S(u) = {G, P}, S(v) = {G, P}, then S(u) Δ S(v) = {}. (Invalid)
# Thus, for every edge (u, v), S(u) must be different from S(v).
# This is equivalent to a 2-coloring problem on a graph.
# If the graph is bipartite, we can assign 1s and 2s.
# To get the lexicographically smallest sequence, we want z_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we prefer color 1 for the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
