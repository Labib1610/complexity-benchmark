import sys

# The problem asks for an assignment of 1 or 2 to each hydrophone such that
# for every edge (u, v), z_u * z_v = 2.
# This means if z_u = 1, then z_v = 2, and if z_u = 2, then z_v = 1.
# This is equivalent to a 2-coloring problem where the two colors are 1 and 2.
# If the graph is bipartite, we can color each connected component in two ways.
# To minimize z_1, z_2, ..., z_n lexicographically, for each component,
# we should pick the coloring that assigns the smaller value to the smallest
# index hydrophone in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            idx_in_comp = 0
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_node = min(component)
            
            # If we set color[min_node] = 1, then for any node v,
            # if color[v] == color[min_node], z_v = 1, else z_v = 2.
            # This ensures z_{min_node} = 1, which is the minimum possible.
            # Since we process components in order of their smallest index,
            # this greedy approach works.
            
            target_color_for_1 = color[min_node]
            for node in component:
                if color[node] == target_color_for_1:
                    ans[node] = 1
                else:
                    ans[node] = 2
                    
    print("YES")
    print(*(ans[1:]))

solve()
