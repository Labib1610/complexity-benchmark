import sys

# The problem states: for every edge (u, v), factor[u] * factor[v] = 2.
# Since factors are in {1, 2}, this means one must be 1 and the other must be 2.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1 and 2.
# For each connected component, there are two possible colorings (swap 1s and 2s).
# We want to minimize the sequence (z_1, z_2, ..., z_n) lexicographically.
# This means for each component, we choose the coloring that makes the smallest
# index in that component have a 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # BFS to find component and check bipartiteness
            component = []
            stack = [(i, 1)]
            colors[i] = 1
            component.append(i)
            
            is_bipartite = True
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        q.append(v)
                        component.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_node = min(component)
            # We want colors[min_node] to be 1.
            # If it's currently 2, flip all colors in this component.
            if colors[min_node] == 2:
                for node in component:
                    colors[node] = 3 - colors[node]
                    
    print("YES")
    print(*(colors[1:]))

solve()
