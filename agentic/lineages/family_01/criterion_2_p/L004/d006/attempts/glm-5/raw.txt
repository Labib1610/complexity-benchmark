import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    possible = True
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break
    
    if not possible:
        sys.stdout.write("NO\n")
        return
    
    # We have a valid bipartition. Now we need to choose the lexicographically smallest assignment.
    # The constraint is: for every edge (u, v), z[u] * z[v] = 2.
    # So one endpoint must be 1 and the other must be 2.
    # This means the graph must be bipartite, and the two colors correspond to the two values {1, 2}.
    # To minimize the sequence lexicographically, we want as many early vertices as possible to be 1.
    # This is equivalent to finding a 2-coloring that maximizes the number of 1's in a greedy manner from left to right.
    # We can do this by processing components and trying to assign 1 to the smallest vertex in each component.
    # But we must ensure consistency. A simple way is to use a greedy approach with a priority queue (or just BFS/DFS from the smallest unassigned vertex).
    # Actually, since the graph is bipartite, for each connected component, there are exactly two valid colorings (swapping 1 and 2).
    # We want to choose the one that is lexicographically smaller.
    # For a single component, the lexicographically smaller coloring is the one where the smallest vertex gets 1.
    # But wait, is that always true? Let's think.
    # If we have a component, and we fix the smallest vertex to 1, then all other vertices are determined.
    # If we fix it to 2, then all others are swapped.
    # The coloring with the smallest vertex as 1 will be lexicographically smaller because at the first position where they differ, it has 1 and the other has 2.
    # So for each component, we just need to assign 1 to the smallest vertex and propagate.
    # This gives the lexicographically smallest overall assignment.
    
    z = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            # Start BFS/DFS from i, assign z[i] = 1
            z[i] = 1
            visited[i] = True
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if not visited[v]:
                        z[v] = 3 - z[u]
                        visited[v] = True
                        stack.append(v)
    
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(str(z[i]) for i in range(1, n + 1)) + "\n")

solve()
