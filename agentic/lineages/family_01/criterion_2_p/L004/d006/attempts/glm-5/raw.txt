```python
import sys
from sys import stdin

def solve():
    input = stdin.readline
    n, m = map(int, input().split())
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    valid = True
    
    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            stack = [start]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        valid = False
                        break
                if not valid:
                    break
        if not valid:
            break
    
    if not valid:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, color[1:])))

solve()
```
