```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build graph
    graph = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        graph[u].append(v)
        graph[v].append(u)
    
    # For each edge, the resonance contribution must be either:
    # - If z_u=1, z_v=1: 1/1 + 1/1 = 2
    # - If z_u=1, z_v=2: 1/2 + 2/1 = 2.5
    # - If z_u=2, z_v=1: 2/1 + 1/2 = 2.5
    # - If z_u=2, z_v=2: 2/2 + 2/2 = 2
    
    # To get total resonance = 2.5 * m, every edge must contribute 2.5
    # This means for each edge (u,v), we need (z_u, z_v) = (1,2) or (2,1)
    
    # This is a 2-coloring problem where:
    # - Each node must be colored 1 or 2
    # - Each edge must connect a node of value 1 to a node of value 2
    
    # This is equivalent to checking if the graph is bipartite
    
    color = [-1] * n  # -1: uncolored, 0: color 1, 1: color 2
    
    def bfs(start):
        queue = deque([start])
        color[start] = 0
        
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if color[neighbor] == -1:
                    color[neighbor] = 1 - color[node]
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:
                    return False
        return True
    
    # Check if graph is bipartite
    for i in range(n):
        if color[i] == -1:
            if not bfs(i):
                print("NO")
                return
    
    # If bipartite, construct solution
    # We want lexicographically smallest assignment
    # So we prefer z_i = 1 over z_i = 2
    # In our coloring, color 0 -> z=1, color 1 -> z=2
    
    print("YES")
    result = []
    for i in range(n):
        if color[i] == -1:  # Isolated vertex
            result.append("1")
        elif color[i] == 0:
            result.append("1")
        else:
            result.append("2")
    
    print(" ".join(result))

solve()
```
