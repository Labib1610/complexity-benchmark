import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build graph
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # For each edge (u,v), the interference profile I(x) = (x - s_u)(x - s_v)
    # I(x) = x^2 - (s_u + s_v)x + s_u * s_v
    # The minimum occurs at x = (s_u + s_v)/2
    # We want this minimum to be at x = 1.5
    # So (s_u + s_v)/2 = 1.5 => s_u + s_v = 3
    # Since s_u, s_v ∈ {1,2}, the only way s_u + s_v = 3 is if one is 1 and the other is 2.
    
    # So for every edge (u,v), we must have s_u ≠ s_v
    # This is equivalent to 2-coloring the graph (bipartite check)
    
    color = [-1] * (n + 1)
    
    # Check if the graph is bipartite
    for start in range(1, n + 1):
        if color[start] == -1:
            queue = deque()
            queue.append(start)
            color[start] = 0
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    # If bipartite, assign colors as tuning parameters
    # Color 0 -> 1, Color 1 -> 2
    # To get lexicographically smallest, we prefer s_i = 1 if possible
    # That means we want color[i] = 0 for as many early indices as possible
    # Since we colored the components independently, and we started BFS from the smallest node,
    # and we always colored with 0 first, this should give us the lex smallest
    
    # Actually, let's re-color to ensure lex smallest
    # We can model this as: for each component, try both colorings and pick the lex smaller one
    # But that's complicated. Easier way:
    # Just use the coloring we have, but ensure that in each component, 
    # the node with the smallest label gets color 0
    
    # Let's do a proper approach:
    # For each connected component, we can choose which color class gets 1 and which gets 2
    # To get lex smallest, we want earlier nodes to be 1 if possible
    
    visited = [False] * (n + 1)
    final_color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            # BFS to get the component
            component = []
            queue = deque()
            queue.append(i)
            visited[i] = True
            component.append(i)
            
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        queue.append(v)
                        component.append(v)
            
            # In this component, decide which color class gets 1
            # Try both and see which gives lex smaller result
            comp_set = set(component)
            
            # Get the induced subgraph edges
            c_edges = [(u, v) for u, v in edges if u in comp_set and v in comp_set]
            
            # Try assigning color 0 to the first node in component
            temp_color = [-1] * (n + 1)
            temp_color[component[0]] = 0
            q = deque([component[0]])
            ok = True
            
            while q and ok:
                u = q.popleft()
                for v in adj[u]:
                    if v in comp_set:
                        if temp_color[v] == -1:
                            temp_color[v] = 1 - temp_color[u]
                            q.append(v)
                        elif temp_color[v] == temp_color[u]:
                            ok = False
                            break
            
            if not ok:
                print("NO")
                return
                
            # Now we have a valid 2-coloring in temp_color for this component
            # Map to {1,2}
            for u in component:
                final_color[u] = temp_color[u] + 1
    
    print("YES")
    print(" ".join(str(final_color[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    main()
