```python
import sys

# The condition: bitwise AND of amplification factors (1 or 2) must be 0.
# 1 is binary 01, 2 is binary 10.
# 1 & 1 = 1 (not 0), 2 & 2 = 2 (not 0), 1 & 2 = 0.
# So, for every edge (u, v), z_u and z_v must be different (one is 1, one is 2).
# This is equivalent to checking if the graph is bipartite.
# The tuning score is sum(z_i * 2^(n-i)).
# To minimize this, we want z_i to be 1 whenever possible, starting from i=1.
# For each connected component, we have two possible colorings (swapping 1s and 2s).
# Let a component have nodes with indices S1 and S2.
# If we assign 1 to S1 and 2 to S2, the contribution is sum_{i in S1} 2^(n-i) + sum_{j in S2} 2 * 2^(n-j).
# If we assign 2 to S1 and 1 to S2, the contribution is sum_{i in S1} 2 * 2^(n-i) + sum_{j in S2} 2^(n-j).
# We compare these two sums for each component and pick the smaller one.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Compare two colorings for this component
            # Coloring A: nodes with color 1 get 1, nodes with color 2 get 2
            # Coloring B: nodes with color 1 get 2, nodes with color 2 get 1
            # We want to minimize sum(z_i * 2^(n-i))
            # Let S1 be nodes with color 1, S2 be nodes with color 2
            # Score A: sum_{u in S1} 1*2^(n-u) + sum_{v in S2} 2*2^(n-v)
            # Score B: sum_{u in S1} 2*2^(n-u) + sum_{v in S2} 1*2^(n-v)
            # Score A < Score B iff sum_{u in S1} 2^(n-u) < sum_{v in S2} 2^(n-v)
            
            s1_val = 0
            s2_val = 0
            for u, c in component:
                if c == 1:
                    s1_val += (1 << (n - u))
                else:
                    s2_val += (1 << (n - u))
            
            if s1_val < s2_val:
                for u, c in component:
                    ans[u] = c
            else:
                for u, c in component:
                    ans[u] = 3 - c
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
