```python
import sys

# The condition A_i + s_i = 3 means:
# (1/d_i) * sum(s_j for j in neighbors of i) + s_i = 3
# sum(s_j for j in neighbors of i) = d_i * (3 - s_i)
# If s_i = 1, sum(s_j) = 2 * d_i
# If s_i = 2, sum(s_j) = 1 * d_i
# Let x_i = s_i - 1, so x_i in {0, 1}.
# s_i = x_i + 1.
# sum(x_j + 1) = d_i * (3 - (x_i + 1)) = d_i * (2 - x_i)
# sum(x_j) + d_i = 2*d_i - d_i * x_i
# sum(x_j) + d_i * x_i = d_i
# This is a system of linear equations over GF(2) if we consider the parity.
# Actually, the equation is sum(s_j) = d_i * (3 - s_i).
# If s_i = 1, sum(s_j) = 2 * d_i. Since s_j in {1, 2}, sum(s_j) = 2 * d_i implies all s_j = 2.
# If s_i = 2, sum(s_j) = 1 * d_i. Since s_j in {1, 2}, sum(s_j) = 1 * d_i implies all s_j = 1.
# This means for every edge (u, v), s_u != s_v.
# This is exactly the condition for a graph to be bipartite.
# If the graph is bipartite, we can color it with 1 and 2.
# Lexicographically smallest means we prefer 1 over 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try to set color[i] = 1 (lexicographically smallest)
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, color[v]))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
