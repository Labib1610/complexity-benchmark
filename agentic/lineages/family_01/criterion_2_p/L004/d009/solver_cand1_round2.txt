```python
import sys

# The condition is that the product of (s_u | s_v) + 2 for all m hazards must be 5^m.
# s_i is in {1, 2}.
# Possible values for (s_u | s_v):
# If s_u=1, s_v=1: (1|1) = 1, multiplier = 1+2 = 3
# If s_u=1, s_v=2: (1|2) = 3, multiplier = 3+2 = 5
# If s_u=2, s_v=1: (2|1) = 3, multiplier = 3+2 = 5
# If s_u=2, s_v=2: (2|2) = 2, multiplier = 2+2 = 4
#
# We need the product to be 5^m.
# This means every hazard (u, v) must have a multiplier of 5.
# A multiplier of 5 occurs if and only if (s_u | s_v) = 3.
# Since s_i in {1, 2}, (s_u | s_v) = 3 means one of them is 1 and the other is 2.
# This is exactly the condition for a bipartite graph.
# If the graph is bipartite, we can color it with 1 and 2 such that every edge
# connects a 1 and a 2.
# To get the lexicographically smallest sequence, we perform BFS/DFS on each
# connected component, assigning 1 to the first node and propagating.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    ans = [0] * (n + 1)
    for i in range(1, n + 1):
        if ans[i] == 0:
            # Start BFS/DFS for component
            # To get lexicographically smallest, we want ans[i] = 1
            stack = [(i, 1)]
            ans[i] = 1
            while stack:
                u, color = stack.pop()
                next_color = 3 - color
                for v in adj[u]:
                    if ans[v] == 0:
                        ans[v] = next_color
                        stack.append((v, next_color))
                    elif ans[v] == color:
                        print("NO")
                        return
    
    print("YES")
    print(*(ans[1:]))

solve()
```
