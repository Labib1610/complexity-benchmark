import sys

# The problem states that for every critical link (u, v), the product of 
# amplification factors must be 2. Since factors are in {1, 2}, the only 
# way for the product to be 2 is if one relay is 1 and the other is 2.
# This is equivalent to a 2-coloring problem on a graph.
# If the graph is bipartite, a solution exists.
# To get the lexicographically smallest assignment, we want 1s as early as possible.
# For each connected component, we have two possible colorings (swapping 1s and 2s).
# We choose the coloring that results in a smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # 1. color[v] as is
            # 2. 3 - color[v]
            # We want the one that makes the sequence lexicographically smallest.
            # Since we want 1s at the smallest indices, we check the first node in the component.
            # If color[i] == 1, the sequence starts with 1. If color[i] == 2, it starts with 2.
            # We prefer the coloring where the first node in the component (by index) is 1.
            
            first_node = min(component)
            if color[first_node] == 1:
                for node in component:
                    ans[node] = color[node]
            else:
                for node in component:
                    ans[node] = 3 - color[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
