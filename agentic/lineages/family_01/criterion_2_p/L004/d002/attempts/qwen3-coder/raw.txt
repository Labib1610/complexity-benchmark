```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # For each edge, the product of amplifications must be 2
    # This means one node gets 1 and the other gets 2
    # This is a bipartite graph problem
    
    # Color the graph with 2 colors (0 and 1)
    # Nodes with color 0 will get amplification 1
    # Nodes with color 1 will get amplification 2
    
    color = [-1] * n
    visited = [False] * n
    
    for i in range(n):
        if not visited[i]:
            # BFS to color component
            queue = deque([i])
            color[i] = 0
            visited[i] = True
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        color[neighbor] = 1 - current_color
                        visited[neighbor] = True
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        print("NO")
                        return
    
    # Verify the solution
    for u, v in edges:
        if color[u] == color[v]:
            print("NO")
            return
    
    # Construct the answer
    result = []
    for i in range(n):
        if color[i] == 0:
            result.append("1")
        else:
            result.append("2")
    
    print("YES")
    print(" ".join(result))

solve()
```
