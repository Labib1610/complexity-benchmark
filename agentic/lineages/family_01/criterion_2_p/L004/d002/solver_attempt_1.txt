```python
import sys

# The problem states:
# 1. Load tier z must be a positive integer such that z^2 <= 4.
#    Possible values for z are 1 and 2.
# 2. For any pair (u, v), (z_u + z_v) / 2 = 1.5, which means z_u + z_v = 3.
#    Since z_u, z_v are in {1, 2}, this implies one must be 1 and the other 2.
# This is a bipartite coloring problem.
# If z_u = 1, then z_v must be 2. If z_u = 2, then z_v must be 1.
# We can model this as a graph where each connected component must be 2-colorable.
# We want the lexicographically smallest assignment, so for each component,
# we try setting the first node to 1 and see if it's valid. If it is, we compare
# the resulting assignment with setting it to 2. Actually, for each component,
# there are only two possible assignments (swapping 1s and 2s). We pick the one
# that makes the sequence lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = {}
    components = []
    
    for i in range(1, n + 1):
        if i not in color:
            comp = []
            stack = [(i, 1)]
            color[i] = 1
            comp.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if v not in color:
                        color[v] = 3 - c
                        comp.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            components.append(comp)
            
    res = [0] * (n + 1)
    for comp in components:
        # Two possible colorings for this component:
        # 1. color[u] as is
        # 2. 3 - color[u]
        # We want the lexicographically smallest result.
        # Since we want the smallest sequence, we look at the smallest index in the component.
        # Let the smallest index be 'start_node'.
        # We want res[start_node] to be 1 if possible.
        
        start_node = min(comp)
        # If we set color[start_node] to 1, then for all v in comp,
        # if color[v] == color[start_node], res[v] = 1, else res[v] = 2.
        # If we set color[start_node] to 2, then for all v in comp,
        # if color[v] == color[start_node], res[v] = 2, else res[v] = 1.
        
        # Option A: color[start_node] maps to 1
        # Option B: color[start_node] maps to 2
        
        # To make it lexicographically smallest, we compare the two options.
        # Actually, we just need to ensure the smallest index in the component gets 1.
        # If we set color[start_node] to 1, then all nodes v with color[v] == color[start_node] get 1.
        # This is always better than setting them to 2.
        
        c_start = color[start_node]
        for node in comp:
            if color[node] == c_start:
                res[node] = 1
            else:
                res[node] = 2
                
    print("YES")
    print(*(res[1:]))

solve()
```
