```python
import sys

# The problem asks for a 2-coloring of a graph where each edge (u, v)
# requires the product of states s_u * s_v = 2.
# States are {1, 2}.
# If s_u = 1, then s_v must be 2.
# If s_u = 2, then s_v must be 1.
# This is equivalent to a standard bipartite coloring problem where
# s_u != s_v.
# We want the lexicographically smallest sequence (s_1, s_2, ..., s_n).
# This means we prefer 1 over 2 for each node in increasing order of index.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # state: 0 = unassigned, 1 = state 1, 2 = state 2
    state = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we greedily pick 1 for the
            # first node of each component.
            stack = [i]
            state[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = 3 - state[u]
                        stack.append(v)
                    elif state[v] == state[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(state[1:]))

solve()
```
