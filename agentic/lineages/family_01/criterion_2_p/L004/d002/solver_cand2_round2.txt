```python
import sys

# The condition: |s_i^2 - s_j^2| = 3
# Possible states are {1, 2}.
# Squares are {1, 4}.
# |1 - 4| = 3.
# This means if there is an edge between u and v, s_u^2 must be different from s_v^2.
# Since s_i^2 is 1 if s_i=1 and 4 if s_i=2, this is equivalent to s_u != s_v.
# This is a standard 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest sequence, we iterate from 1 to n.
# If a node is not visited, we start a BFS/DFS. To make it lexicographically smallest,
# we prefer assigning 1 over 2 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS. To be lexicographically smallest, 
            # we assign 1 to the first node of each connected component.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
