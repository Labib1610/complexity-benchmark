```python
import sys

# The condition is |s_i^3 - s_j^3| = 7 for each hazard (i, j).
# Possible values for s_i are {1, 2}.
# s_i^3 can be 1^3 = 1 or 2^3 = 8.
# |1 - 1| = 0, |8 - 8| = 0, |1 - 8| = 7, |8 - 1| = 7.
# Thus, for every hazard (i, j), we must have s_i != s_j.
# This is a standard 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest sequence, we iterate from 1 to n.
# If a node is unvisited, we assign it 1 (the smaller value) and perform BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the connected component
            # To ensure lexicographical smallest, we prefer color 1
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
