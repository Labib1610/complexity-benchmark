```python
import sys

# The transformation matrix is:
# M = [[s_u + s_v, 1], [8, 6 - s_u - s_v]]
# The area is preserved if the determinant of M is 1.
# det(M) = (s_u + s_v)(6 - s_u - s_v) - 8 = 1
# Let S = s_u + s_v.
# S(6 - S) - 8 = 1
# 6S - S^2 - 8 = 1
# S^2 - 6S + 9 = 0
# (S - 3)^2 = 0
# S = 3
# Since s_u, s_v \in {1, 2}, s_u + s_v = 3 implies one is 1 and the other is 2.
# This means for every hazard (u, v), s_u != s_v.
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, we prefer 1 over 2.
            # Start BFS/DFS with color 1 for the first node of each component.
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
