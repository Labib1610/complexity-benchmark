```python
import sys

# Let V1 = (1, 2, 2) and V2 = (2, 1, 2).
# V1 x V1 = (0, 0, 0), norm squared = 0.
# V2 x V2 = (0, 0, 0), norm squared = 0.
# V1 x V2 = (2*2 - 2*1, 2*2 - 1*2, 1*1 - 2*2) = (2, 2, -3).
# Norm squared of (2, 2, -3) = 4 + 4 + 9 = 17.
# V2 x V1 = -(V1 x V2), norm squared = 17.
#
# The interference for a hazard (u, v) is 17 if s_u != s_v, and 0 if s_u == s_v.
# We need the sum of interferences to be 17 * m.
# This means for every hazard (u, v), we must have s_u != s_v.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest assignment.
# For each connected component, we can assign the first node to 1,
# then propagate the constraints. If we have a choice, we pick 1 to be
# lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    s = [0] * (n + 1)
    for i in range(1, n + 1):
        if s[i] == 0:
            # Start BFS/DFS for component
            stack = [(i, 1)]
            s[i] = 1
            while stack:
                u, color = stack.pop()
                for v in adj[u]:
                    if s[v] == 0:
                        s[v] = 3 - color
                        stack.append((v, s[v]))
                    elif s[v] == color:
                        print("NO")
                        return
    
    print("YES")
    print(*(s[1:]))

solve()
```
