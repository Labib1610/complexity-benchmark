```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build graph
    adj = defaultdict(list)
    edges = []
    degree = [0] * (n + 1)
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
    
    # For each connected component, we need to check if it's bipartite
    # and assign states accordingly.
    
    # Color array: 0 = unvisited, 1 and 2 are the two parts of bipartition
    color = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    # Check if the graph is bipartite and assign colors
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 1
        visited[start] = True
        
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    color[v] = 3 - color[u]  # Flip color: 1->2, 2->1
                    queue.append(v)
                elif color[v] == color[u]:
                    return False  # Not bipartite
        return True
    
    # Check all components
    for i in range(1, n + 1):
        if not visited[i] and adj[i]:
            if not bfs(i):
                print("NO")
                return
    
    # If bipartite, assign s values based on color
    # We want the lexicographically smallest sequence
    # So we prefer s=1 for lower indices when possible
    # In a bipartite graph, we can assign all nodes in one part to have C_i = d_i (s=1)
    # and all in the other part to have C_i = -d_i (s=2) or vice versa
    
    # To get lexicographically smallest, we try to assign s=1 to node 1 if possible
    # That means: if color[1] == 1, we assign part 1 -> s=1, part 2 -> s=2
    #            if color[1] == 2, we assign part 1 -> s=2, part 2 -> s=1
    # But if node 1 is isolated, color[1] = 0, so we assign s=1
    
    s = [0] * (n + 1)
    
    # Determine mapping
    # We want s sequence to be lexicographically smallest
    # So we look at the first node that has an edge and decide based on that
    
    # Find first node with degree > 0
    first_node = -1
    for i in range(1, n + 1):
        if degree[i] > 0:
            first_node = i
            break
    
    if first_node == -1:
        # All nodes are isolated
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # For bipartite graphs, we can always make a valid assignment
    # Map color 1 to s=1 and color 2 to s=2, unless that makes the sequence larger
    # To get lex smallest, if color[first_node] == 1, keep as is
    # if color[first_node] == 2, swap the mapping
    
    map_1_to_s1 = True
    if color[first_node] == 2:
        map_1_to_s1 = False
    
    for i in range(1, n + 1):
        if degree[i] == 0:
            s[i] = 1
        else:
            if map_1_to_s1:
                if color[i] == 1:
                    s[i] = 1
                else:
                    s[i] = 2
            else:
                if color[i] == 1:
                    s[i] = 2
                else:
                    s[i] = 1
    
    print("YES")
    print(" ".join(map(str, s[1:])))

if __name__ == "__main__":
    main()
```
