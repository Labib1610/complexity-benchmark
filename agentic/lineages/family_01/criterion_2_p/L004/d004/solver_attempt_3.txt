import sys

# The constraint is: for every contact (u, v), speed(u) * speed(v) = 2.
# Since speeds are in {1, 2}, the only way to get a product of 2 is (1, 2) or (2, 1).
# This means the graph must be bipartite. If it's bipartite, we can color it with 1 and 2.
# To get the lexicographically smallest sequence, we want 1s as early as possible.
# For each connected component, we have two choices:
# 1. Assign color 1 to set A and color 2 to set B.
# 2. Assign color 2 to set A and color 1 to set B.
# We choose the one that makes the first node in the component 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            component = []
            stack = [(i, 1)]
            colors[i] = 1
            component.append((i, 1))
            
            idx_stack = 0
            while idx_stack < len(component):
                u, c = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        component.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_node = min(node for node, c in component)
            # Find the color assigned to the smallest index
            color_of_min = 0
            for node, c in component:
                if node == min_node:
                    color_of_min = c
                    break
            
            # If color_of_min is 2, we can flip all colors in this component
            # to make the smallest index 1.
            if color_of_min == 2:
                for node, c in component:
                    colors[node] = 3 - c
                    
    print("YES")
    print(*(colors[1:]))

solve()
