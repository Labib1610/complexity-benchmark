```python
import sys

# The problem asks for an assignment of 1 or 2 to each relay such that
# for every edge (u, v), (gain[u] & gain[v]) == 0.
# Since gain[i] is 1 or 2:
# 1 is binary 01, 2 is binary 10.
# (1 & 1) = 1 != 0
# (2 & 2) = 2 != 0
# (1 & 2) = 0
# (2 & 1) = 0
# This means for every edge (u, v), gain[u] must be different from gain[v].
# This is exactly the definition of a bipartite graph.
# We need to check if the graph is bipartite. If it is, we assign 1 or 2.
# To get the lexicographically smallest assignment, we want 1s as much as possible.
# For each connected component, we have two choices for the 2-coloring.
# Let the two sets of the bipartition be A and B.
# We can assign 1 to A and 2 to B, or 2 to A and 1 to B.
# To make the assignment lexicographically smallest, for each component,
# we look at the smallest index 'i' in the component.
# If we assign 1 to the set containing 'i', the assignment starts with 1.
# If we assign 2 to the set containing 'i', the assignment starts with 2.
# Clearly, we prefer 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_node = min(component)
            # If color[min_node] is 1, we can assign 1 to color 1 and 2 to color 2
            # to make assignment[min_node] = 1.
            # If color[min_node] is 2, we assign 2 to color 1 and 1 to color 2
            # to make assignment[min_node] = 1.
            
            if color[min_node] == 1:
                for node in component:
                    assignment[node] = color[node]
            else:
                for node in component:
                    assignment[node] = 3 - color[node]
                    
    print("YES")
    print(*(assignment[1:]))

solve()
```
