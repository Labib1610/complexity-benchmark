```python
import sys

# The condition for E(x) = 2x^3 - 3(z_u + z_v)x^2 + 6 z_u z_v x to have exactly one local max and one local min
# is that its derivative E'(x) = 6x^2 - 6(z_u + z_v)x + 6 z_u z_v = 0 must have two distinct real roots.
# E'(x) = 6(x^2 - (z_u + z_v)x + z_u z_v) = 6(x - z_u)(x - z_v) = 0.
# The roots are x = z_u and x = z_v.
# For these to be distinct, we must have z_u != z_v.
# Thus, the problem is equivalent to checking if the graph is bipartite.
# If bipartite, we assign 1 and 2 to the two partitions.
# To get the lexicographically smallest assignment, we greedily assign 1 to the first node
# of each connected component and propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
