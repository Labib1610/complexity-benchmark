```python
import sys

# The problem states that for any link (u, v), the product of coefficients must be 2.
# Since coefficients are in {1, 2}, the only way for the product to be 2 is if
# one relay is 1 and the other is 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we iterate from 1 to n.
# If a node is not yet colored, we assign it 1 (the smaller value).
# Then we propagate the constraints using BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # colors: 0 = unassigned, 1 = coefficient 1, 2 = coefficient 2
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                
                target_c = 3 - c # If c is 1, target is 2; if c is 2, target is 1
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = target_c
                        stack.append((v, target_c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
```
