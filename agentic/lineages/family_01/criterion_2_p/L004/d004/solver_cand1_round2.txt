```python
import sys

# The number of distinct arrangements of s_u alpha pulses and s_v beta pulses
# is given by the binomial coefficient (s_u + s_v) choose s_u.
# Possible values for s_u, s_v are {1, 2}.
# (1+1) choose 1 = 2! / (1!1!) = 2
# (1+2) choose 1 = 3! / (1!2!) = 3
# (2+1) choose 2 = 3! / (2!1!) = 3
# (2+2) choose 2 = 4! / (2!2!) = 6
# The condition is that the number of arrangements must be 3.
# This happens if and only if (s_u, s_v) is (1, 2) or (2, 1).
# This is equivalent to s_u != s_v.
# Thus, the problem is to check if the graph is bipartite.
# If bipartite, we want the lexicographically smallest assignment (s_1, ..., s_n).
# To get the lexicographically smallest, we greedily try s_i = 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            stack = [i]
            assignment[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
