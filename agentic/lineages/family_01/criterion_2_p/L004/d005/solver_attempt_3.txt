```python
import sys

# The problem states that for any two processors linked by a bus, 
# the integer division of their maximum multiplier by their minimum 
# multiplier must be 2.
# Since the multipliers are {1, 2}, the only way for max/min = 2 is 
# if one processor is 1 and the other is 2.
# This means the graph must be bipartite.
# We need to find the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for processor 1, 
# then for processor 2, and so on.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is not colored, we try assigning 1.
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [i]
            assignment[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
