```python
import sys

# The problem states that for every edge (u, v), one relay must be 1 and the other 2.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest assignment, we want relay 1 to be 1 if possible.
# If relay 1 is 1, its neighbors must be 2, their neighbors must be 1, and so on.
# We use BFS/DFS to color each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If not assigned, assign 1 and propagate.
    for i in range(1, n + 1):
        if assignment[i] == 0:
            assignment[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
