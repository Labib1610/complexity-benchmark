```python
import sys

# The problem states: amplification factors are 1 or 2.
# The non-interference rule: for any two relays u, v connected by a link,
# (factor[u] & factor[v]) == 0.
# Since factors are 1 (binary 01) and 2 (binary 10),
# (1 & 1) = 1 != 0
# (2 & 2) = 2 != 0
# (1 & 2) = 0
# (2 & 1) = 0
# This means the graph must be bipartite, where one set of the partition
# is assigned 1 and the other is assigned 2.
# We want to minimize the sequence (z1, z2, ..., zn) lexicographically.
# This means we prefer 1 over 2 for each node in order.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unvisited, 1 = factor 1, 2 = factor 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To minimize lexicographically, we try to assign 1 to the current node
            # if it's the start of a new connected component.
            stack = [(i, 1)]
            color[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                for v in adj[u]:
                    if color[v] == 0:
                        # If u is 1, v must be 2. If u is 2, v must be 1.
                        new_c = 3 - c
                        color[v] = new_c
                        stack.append((v, new_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
