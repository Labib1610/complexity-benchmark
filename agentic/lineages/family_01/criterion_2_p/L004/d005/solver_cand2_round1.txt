```python
import sys

# The problem asks us to assign a multiplier of 1 or 2 to each endpoint of each edge (u, v).
# Let x_{uv} = 1 if u gets 1 and v gets 2, and x_{uv} = 0 if u gets 2 and v gets 1.
# For each satellite i, let E_i be the set of edges connected to i.
# For each edge e in E_i, let c_{i,e} be the multiplier assigned to i (1 or 2).
# P_i = product_{e in E_i} c_{i,e}.
# Resonance condition: P_i = 1 or P_i = 2^{k_i}.
# Since c_{i,e} is either 1 or 2, P_i = 2^{count of 2s assigned to i}.
# The condition P_i = 1 means count of 2s is 0.
# The condition P_i = 2^{k_i} means count of 2s is k_i (i.e., all multipliers are 2).
# This is equivalent to saying: for each edge (u, v), one endpoint gets 1 and the other gets 2.
# This is exactly the definition of a bipartite graph.
# If the graph is bipartite, we can 2-color it with colors 1 and 2.
# If a node is colored 1, it receives 1 from all incident edges. P_i = 1^k = 1.
# If a node is colored 2, it receives 2 from all incident edges. P_i = 2^k.
# Lexicographically smallest: we want s_i = 1 whenever possible.
# For each connected component, we have two choices of coloring.
# We pick the one that makes the sequence s_1, ..., s_n lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = map(int, input_data[:2])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = map(int, input_data[idx:idx+2])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Try both colorings for this component
            # Option 1: color[i] = 1
            # Option 2: color[i] = 2
            # We want the lexicographically smallest result.
            # Since we process nodes 1 to n, we check the first node in the component.
            
            comp_nodes = sorted([node for node, c in component])
            first_node = comp_nodes[0]
            
            # If we set color[first_node] = 1, what are the values?
            # If color[first_node] was 1, then s_j = color[j].
            # If color[first_node] was 2, then s_j = 3 - color[j].
            
            # Let's calculate the two possible assignments for this component
            assign1 = {}
            assign2 = {}
            for node, c in component:
                assign1[node] = c
                assign2[node] = 3 - c
            
            # Compare assign1 and assign2 lexicographically
            # We only care about the values at the nodes in this component
            is_assign1_better = False
            for node in comp_nodes:
                if assign1[node] < assign2[node]:
                    is_assign1_better = True
                    break
                elif assign1[node] > assign2[node]:
                    is_assign1_better = False
                    break
            
            best = assign1 if is_assign1_better else assign2
            for node in comp_nodes:
                res[node] = best[node]

    print("YES")
    print(*(res[1:]))

solve()
```
