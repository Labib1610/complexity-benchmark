import sys

# The resonance equation: d_i * s_i + L_i = 3 * d_i
# L_i = sum_{j in neighbors(i)} s_j
# d_i * s_i + sum_{j in neighbors(i)} s_j = 3 * d_i
# Let x_i = s_i - 1. Then s_i = x_i + 1, where x_i in {0, 1}.
# d_i * (x_i + 1) + sum_{j in neighbors(i)} (x_j + 1) = 3 * d_i
# d_i * x_i + d_i + sum_{j in neighbors(i)} x_j + d_i = 3 * d_i
# d_i * x_i + sum_{j in neighbors(i)} x_j = d_i
# This is a system of linear equations over GF(2) if we consider the constraints.
# Actually, this is a system of linear equations over integers, but since s_i in {1, 2},
# x_i in {0, 1}.
# d_i * x_i + sum_{j in neighbors(i)} x_j = d_i
# If d_i is even, d_i * x_i is even, so sum_{j in neighbors(i)} x_j must be even.
# If d_i is odd, d_i * x_i has the same parity as x_i, so x_i + sum_{j in neighbors(i)} x_j is odd.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    # x_i * d_i + sum_{j in N(i)} x_j = d_i
    # This is a system of linear equations over GF(2).
    # We want the lexicographically smallest s, which means smallest x.
    # We can use Gaussian elimination or solve connected components.
    
    ans = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i] and len(adj[i]) > 0:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Solve for this component
            # Try x_i = 0, then x_i = 1
            def check(comp, start_val):
                res = {node: -1 for node in comp}
                res[comp[0]] = start_val
                queue = [comp[0]]
                head = 0
                while head < len(queue):
                    u = queue[head]
                    head += 1
                    # d_u * x_u + sum_{v in N(u)} x_v = d_u (mod 2)
                    # sum_{v in N(u)} x_v = d_u - d_u * x_u = d_u * (1 - x_u) (mod 2)
                    rhs = (len(adj[u]) * (1 - res[u])) % 2
                    unknowns = []
                    known_sum = 0
                    for v in adj[u]:
                        if res[v] == -1:
                            unknowns.append(v)
                        else:
                            known_sum += res[v]
                    
                    if not unknowns:
                        if (known_sum % 2) != rhs: return None
                    elif len(unknowns) == 1:
                        res[unknowns[0]] = (rhs - known_sum) % 2
                        queue.append(unknowns[0])
                    else:
                        # This is a cycle or complex component, simple propagation fails
                        # But for this specific problem, we can use backtracking or 2-SAT
                        return "complex"
                return res

            # Since the system might be complex, use a simple backtracking for the component
            def solve_comp(idx_in_comp, current_res):
                if idx_in_comp == len(component):
                    for u in component:
                        d_u = len(adj[u])
                        if (d_u * current_res[u] + sum(current_res[v] for v in adj[u])) % 2 != (d_u % 2):
                            return None
                    return current_res
                
                u = component[idx_in_comp]
                for val in [0, 1]:
                    current_res[u] = val
                    res = solve_comp(idx_in_comp + 1, current_res)
                    if res: return res
                return None

            final_res = solve_comp(0, {node: -1 for node in component})
            if not final_res:
                print("NO")
                return
            for node in component:
                ans[node] = final_res[node] + 1
        elif not visited[i]:
            ans[i] = 1
            
    print("YES")
    print(*(ans[1:]))

solve()
