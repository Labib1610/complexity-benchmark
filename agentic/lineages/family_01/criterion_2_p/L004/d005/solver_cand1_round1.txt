import sys

# The resonance equation: L_i + d_i * s_i = 3 * d_i
# L_i = sum_{j in neighbors(i)} s_j
# So, sum_{j in neighbors(i)} s_j + d_i * s_i = 3 * d_i
# Let x_i = s_i - 1. Then s_i = x_i + 1, where x_i in {0, 1}.
# sum_{j in neighbors(i)} (x_j + 1) + d_i * (x_i + 1) = 3 * d_i
# sum_{j in neighbors(i)} x_j + d_i + d_i * x_i + d_i = 3 * d_i
# sum_{j in neighbors(i)} x_j + d_i * x_i = d_i
# This is a system of linear equations over GF(2) if we consider the parity.
# Actually, it's simpler: sum_{j in neighbors(i)} x_j + d_i * x_i = d_i
# If d_i is even, sum_{j in neighbors(i)} x_j = 0 (mod 2)
# If d_i is odd, sum_{j in neighbors(i)} x_j + x_i = 1 (mod 2)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    # x_i in {0, 1}
    # For each connected component, we can solve the system.
    # Since we want lexicographically smallest s_i, we want smallest x_i.
    # x_i = 0 is preferred.
    
    x = [-1] * (n + 1)
    for i in range(1, n + 1):
        if len(adj[i]) == 0:
            x[i] = 0
            continue
        if x[i] == -1:
            # BFS/DFS to solve component
            component = []
            stack = [i]
            x[i] = 0
            component.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if x[v] == -1:
                        # Equation for u: sum_{j in adj[u]} x_j + (d_u % 2) * x_u = d_u % 2
                        # sum_{j in adj[u], j != v} x_j + x_v + (d_u % 2) * x_u = d_u % 2
                        # x_v = (d_u % 2) - (d_u % 2) * x_u - sum_{j in adj[u], j != v} x_j
                        # All calculations mod 2.
                        val = (len(adj[u]) % 2) - (len(adj[u]) % 2) * x[u]
                        for neighbor in adj[u]:
                            if neighbor != v and x[neighbor] != -1:
                                val -= x[neighbor]
                        x[v] = val % 2
                        stack.append(v)
                        component.append(v)
            
            # Check consistency
            for u in component:
                s = 0
                for v in adj[u]:
                    s += x[v]
                if (s + (len(adj[u]) % 2) * x[u]) % 2 != (len(adj[u]) % 2):
                    print("NO")
                    return

    print("YES")
    print(*(x[i] + 1 for i in range(1, n + 1)))

solve()
