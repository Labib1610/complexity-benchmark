import sys

# The problem asks us to assign each satellite i a state s_i in {1, 2}.
# If s_i = 1, satellite i is a source for all its hazards.
# If s_i = 2, satellite i is a sink for all its hazards.
# For a hazard between u and v, one must be a source and one must be a sink.
# This means s_u != s_v.
# This is equivalent to 2-coloring the graph of hazards.
# If the graph is bipartite, we can color it with 1s and 2s.
# If a component is bipartite, we have two choices for coloring (swapping 1s and 2s).
# We want the lexicographically smallest sequence s_1, s_2, ..., s_n.
# For each connected component, we want to assign the smallest possible values.
# Isolated nodes (no hazards) must be 1 (as per problem statement).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            if not adj[i]:
                color[i] = 1
                continue
            
            # Component found, perform BFS/DFS to 2-color
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            possible = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if swapping colors gives a lexicographically smaller sequence
            # We only care about the first node in the component (the smallest index)
            # If color[i] is 1, we keep it. If it's 2, we check if swapping is better.
            # Actually, for each component, we want the smallest value at the smallest index.
            # Let min_node be the smallest index in this component.
            # If we set color[min_node] = 1, we are good.
            
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
