```python
import sys
from collections import defaultdict, deque

def solve():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        # No constraints, all relays can be set to 1
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color assignment: 0 means unset, 1 and 2 are the two colors (factors)
    color = [0] * (n + 1)
    
    # Check if bipartite using BFS/DFS for each component
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
            
        # Start BFS coloring
        queue = deque()
        queue.append((start, 1))  # Assign color 1 to start node
        color[start] = 1
        
        while queue:
            node, c = queue.popleft()
            next_color = 3 - c  # Toggle between 1 and 2
            
            for neighbor in adj[node]:
                if color[neighbor] == 0:
                    color[neighbor] = next_color
                    queue.append((neighbor, next_color))
                elif color[neighbor] == c:
                    # Conflict found: not bipartite
                    print("NO")
                    return
    
    # If we reach here, graph is bipartite and a valid configuration exists
    print("YES")
    result = []
    for i in range(1, n + 1):
        result.append(str(color[i]))
    print(" ".join(result))

solve()
```
