import sys

# The problem states that for every edge (u, v), one relay must be 1 and the other 2.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we want 1s at the smallest indices possible.
# For each connected component, we have two choices:
# 1. Assign 1 to the smallest index in the component.
# 2. Assign 2 to the smallest index in the component.
# We should choose the assignment that results in a smaller value at the smallest index.
# If the smallest index in the component is 'i', we want z[i] = 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            # to check for bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring, check if we need to flip to make the smallest index 1
            # Find the smallest index in this component
            min_node = min(q)
            if color[min_node] == 2:
                for node in q:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
