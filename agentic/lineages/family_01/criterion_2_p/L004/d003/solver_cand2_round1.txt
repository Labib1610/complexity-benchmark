```python
import sys

# The condition is (s_u * s_v) ^ (s_u + s_v) == 1
# Possible values for s_u, s_v are {1, 2}.
# Case 1: s_u = 1, s_v = 1
# (1 * 1) ^ (1 + 1) = 1 ^ 2 = 3 (Not 1)
# Case 2: s_u = 1, s_v = 2
# (1 * 2) ^ (1 + 2) = 2 ^ 3 = 1 (Valid)
# Case 3: s_u = 2, s_v = 1
# (2 * 1) ^ (2 + 1) = 2 ^ 3 = 1 (Valid)
# Case 4: s_u = 2, s_v = 2
# (2 * 2) ^ (2 + 2) = 4 ^ 4 = 0 (Not 1)
#
# Conclusion: For every edge (u, v), s_u must be different from s_v.
# This is a standard bipartite graph coloring problem.
# We want the lexicographically smallest sequence (s_1, s_2, ..., s_n).
# To minimize lexicographically, we prefer 1 over 2.
# For each connected component, we can pick the color of the first node
# (the one with the smallest index) to be 1, then propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS from i, assign color 1 to i
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
