```python
import sys

# The matrices are:
# M1 = [[1, 2, 0], [0, 1, 2], [2, 0, 1]]
# M2 = [[1, 0, 2], [2, 1, 0], [0, 2, 1]]
#
# Check determinant of (Ai + Aj):
# If Ai = M1, Aj = M1: M1+M1 = [[2, 4, 0], [0, 2, 4], [4, 0, 2]]
# det(M1+M1) = 2(4-0) - 4(0-16) + 0 = 8 + 64 = 72 != 0
# If Ai = M2, Aj = M2: M2+M2 = [[2, 0, 4], [4, 2, 0], [0, 4, 2]]
# det(M2+M2) = 2(4-0) - 0 + 4(16-0) = 8 + 64 = 72 != 0
# If Ai = M1, Aj = M2: M1+M2 = [[2, 2, 2], [2, 2, 2], [2, 2, 2]]
# det(M1+M2) = 0
#
# Thus, for det(Ai + Aj) = 0, we must have Ai != Aj.
# This is a bipartite coloring problem.
# We want to minimize E = sum(si * 2^(n-i)).
# To minimize E, we want s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # color 1: s_i = 1, color 2: s_i = 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To minimize E, we greedily assign s_i = 1 to the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
