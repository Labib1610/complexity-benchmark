import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    temp = [-1] * (n + 1)          # 0/1 temporary colors
    ans = [0] * (n + 1)            # final 1/2 values

    for start in range(1, n + 1):
        if temp[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        temp[start] = 0
        comp = [start]
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if temp[v] == -1:
                    temp[v] = temp[u] ^ 1
                    q.append(v)
                    comp.append(v)
                elif temp[v] == temp[u]:
                    print("NO")
                    return
        # lexicographically smallest: make start (the smallest index in component) = 1
        if temp[start] == 0:
            # map 0->1, 1->2
            for v in comp:
                ans[v] = 1 if temp[v] == 0 else 2
        else:
            # map 1->1, 0->2
            for v in comp:
                ans[v] = 1 if temp[v] == 1 else 2

    print("YES")
    print(' '.join(str(ans[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
