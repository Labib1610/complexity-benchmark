```python
import sys

# The characteristic equation of M = [[s_u, 2], [1, s_v]] is:
# det(M - lambda*I) = (s_u - lambda)(s_v - lambda) - 2 = 0
# lambda^2 - (s_u + s_v)lambda + (s_u*s_v - 2) = 0
# The eigenvalues are rational if and only if the discriminant D is a perfect square.
# D = (s_u + s_v)^2 - 4(s_u*s_v - 2)
# D = s_u^2 + 2*s_u*s_v + s_v^2 - 4*s_u*s_v + 8
# D = s_u^2 - 2*s_u*s_v + s_v^2 + 8
# D = (s_u - s_v)^2 + 8
# Possible values for s_u, s_v in {1, 2}:
# If s_u = s_v: D = 0^2 + 8 = 8 (not a perfect square)
# If s_u != s_v: D = (1-2)^2 + 8 = 1 + 8 = 9 = 3^2 (perfect square)
# Thus, the condition is s_u != s_v for all connected u, v.
# This is equivalent to checking if the graph is bipartite.
# To minimize I = sum(s_i / 3^i), we want s_i to be 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # To minimize sum(s_i / 3^i), we prefer s_i = 1.
    # For each connected component, we have two choices of coloring (1,2) or (2,1).
    # We pick the one that makes the first node in the component 1 if possible.
    
    res = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check which coloring is better for this component
            # We want the smallest index in the component to have s_i = 1
            first_node = min(component)
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                res[node] = color[node]
                
    print("YES")
    print(*(res[1:]))

solve()
```
