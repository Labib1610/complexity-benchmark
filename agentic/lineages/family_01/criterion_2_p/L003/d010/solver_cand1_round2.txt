```python
import sys

# The stability index term for an edge (u, v) is (2*s_u - 3) * (2*s_v - 3).
# Let x_i = 2*s_i - 3. Since s_i in {1, 2}, x_i in {-1, 1}.
# The condition is sum_{j=1}^m x_{u_j} * x_{v_j} = -m.
# Since each term x_{u_j} * x_{v_j} is either -1 or 1, the sum equals -m
# if and only if every term x_{u_j} * x_{v_j} = -1.
# This means for every edge (u, v), x_u != x_v.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with two colors (1 and 2).
# To minimize E = sum s_i * 2^{n-i}, we want s_i = 1 as much as possible for small i.
# This is equivalent to finding the lexicographically smallest sequence (s_1, ..., s_n).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # To minimize lexicographically, we prefer s_i = 1 (color 1)
            # for the smallest index in each connected component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

solve()
```
