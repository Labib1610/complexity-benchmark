import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Let x_i = s_i - 1. Then x_i \in {0, 1}.
    # s_i = x_i + 1.
    # The equation: s^T A s + s^T D s - 6 * 1^T D s = -9m
    # s^T (A + D) s - 6 * 1^T D s = -9m
    # (x+1)^T (A+D) (x+1) - 6 * 1^T D (x+1) = -9m
    # x^T (A+D) x + 2 * 1^T (A+D) x + 1^T (A+D) 1 - 6 * 1^T D x - 6 * 1^T D 1 = -9m
    # Note: 1^T A 1 = 2m, 1^T D 1 = 2m. So 1^T (A+D) 1 = 4m.
    # 1^T D 1 = 2m.
    # x^T (A+D) x + 2 * 1^T A x + 2 * 1^T D x + 4m - 6 * 1^T D x - 12m = -9m
    # x^T (A+D) x + 2 * 1^T A x - 4 * 1^T D x = -m
    # x^T A x + x^T D x + 2 * 1^T A x - 4 * 1^T D x = -m
    # x^T A x + \sum d_i x_i^2 + 2 \sum_{u,v \in E} (x_u + x_v) - 4 \sum d_i x_i = -m
    # Since x_i^2 = x_i:
    # \sum_{u,v \in E} 2 x_u x_v + \sum d_i x_i + 2 \sum_{u,v \in E} (x_u + x_v) - 4 \sum d_i x_i = -m
    # \sum_{u,v \in E} (2 x_u x_v + 2 x_u + 2 x_v) - 3 \sum d_i x_i = -m
    # \sum_{u,v \in E} (2 x_u x_v + 2 x_u + 2 x_v - 3 x_u - 3 x_v) = -m
    # \sum_{u,v \in E} (2 x_u x_v - x_u - x_v) = -m
    
    # Let x_i = 0 or 1.
    # If x_u=0, x_v=0: term is 0
    # If x_u=0, x_v=1: term is -1
    # If x_u=1, x_v=0: term is -1
    # If x_u=1, x_v=1: term is 2-1-1 = 0
    # So we need the number of edges with exactly one endpoint as 1 to be m.
    # This is exactly the definition of a cut in the graph.
    # We need a cut of size m.
    # A graph has a cut of size m if and only if it is bipartite.
    # If bipartite, the two partitions (or any subset of connected components) form a cut.
    # To minimize E = \sum s_i 2^{n-i} = \sum (x_i+1) 2^{n-i}, we want x_i=0 as much as possible for small i.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # For each component, we can choose color 0 or 1 to be x_i=0.
    # To minimize \sum (x_i+1) 2^{n-i}, we want x_i=0 for small i.
    # For each component, find the smallest index i. If color[i] == 0, 
    # we prefer setting color 0 to x=0.
    print("YES")
    res = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            comp = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Find smallest index in component
            first = min(comp)
            # If color[first] == 0, we set color 0 -> x=0, color 1 -> x=1
            # If color[first] == 1, we set color 1 -> x=0, color 0 -> x=1
            target_c = color[first]
            for node in comp:
                res[node] = 1 if color[node] != target_c else 0
                
    print(*(res[i] + 1 for i in range(1, n + 1)))

solve()
