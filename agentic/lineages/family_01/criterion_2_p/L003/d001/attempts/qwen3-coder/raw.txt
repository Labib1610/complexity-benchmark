```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Color nodes with BFS to check bipartiteness
    color = {}
    
    for start_node in range(1, n + 1):
        if start_node not in color:
            queue = deque([start_node])
            color[start_node] = 0
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if neighbor not in color:
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # If bipartite, construct solution
    # Assign 1 to color 0, 2 to color 1 for lexicographically smallest
    result = [0] * (n + 1)
    
    # Check which coloring gives lexicographically smaller sequence
    # We want to minimize the first position where they differ
    for i in range(1, n + 1):
        if color[i] == 0:
            result[i] = 1
        else:
            result[i] = 2
    
    # Verify this satisfies all constraints
    for u, v in edges:
        if result[u] * result[v] != 2:
            # Try the other coloring
            for i in range(1, n + 1):
                if color[i] == 0:
                    result[i] = 2
                else:
                    result[i] = 1
            break
    
    # Final verification
    valid = True
    for u, v in edges:
        if result[u] * result[v] != 2:
            valid = False
            break
    
    if not valid:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, result[1:])))

solve()
```
