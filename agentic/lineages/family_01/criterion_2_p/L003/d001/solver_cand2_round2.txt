import sys

# The problem asks for a 2-coloring of a graph where edges represent constraints.
# If two nodes u, v have an edge, their states s_u, s_v must satisfy s_u * s_v = 2.
# Since s_i \in {1, 2}, this means:
# If s_u = 1, then s_v = 2.
# If s_u = 2, then s_v = 1.
# This is exactly the definition of a bipartite graph.
# If the graph is bipartite, we can color each connected component with two colors.
# To get the lexicographically smallest sequence, for each component, we try
# assigning the smallest possible value (1) to the smallest index node in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we want the smallest index
            # in the component to be 1.
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard bipartite check
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring, check if we can flip the colors to make it smaller
            # The current coloring assigns color[i] = 1.
            # If we flip, color[i] becomes 2.
            # We want the smallest index in the component to have color 1.
            # Since we started BFS from the smallest uncolored node i,
            # color[i] is already 1, which is the smallest possible value.
            
    print("YES")
    print(*(color[1:]))

solve()
