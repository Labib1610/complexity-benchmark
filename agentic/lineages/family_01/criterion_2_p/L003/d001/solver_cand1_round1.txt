```python
import sys

# The problem states that for each critical link (u, v), the product of their 
# calibration factors must be 2. Since factors are 1 or 2, the only way 
# their product is 2 is if one is 1 and the other is 2.
# This is a bipartite coloring problem. We need to color the graph with 
# colors {1, 2} such that no two adjacent nodes have the same color.
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If a node is not visited, we assign it 1 (the smaller value) and perform BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS from i, assign 1 to i to keep it lexicographically smallest
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
```
