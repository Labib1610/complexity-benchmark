```python
import sys

# The pulse in a box of size (w, h) starting at (0,0) with velocity (1,1)
# hits a corner when the time t satisfies:
# t = k1 * w = k2 * h for some integers k1, k2.
# The pulse hits the top-right corner (w, h) if and only if
# the first time it hits a corner is at (w, h).
# This happens if w/gcd(w, h) and h/gcd(w, h) are both odd.
# Let w = s_u, h = s_v.
# If s_u = 1, s_v = 1: gcd(1, 1) = 1. w/1 = 1 (odd), h/1 = 1 (odd). Hits (1, 1). FAIL.
# If s_u = 1, s_v = 2: gcd(1, 2) = 1. w/1 = 1 (odd), h/1 = 2 (even). Hits (0, 2). OK.
# If s_u = 2, s_v = 1: gcd(2, 1) = 1. w/1 = 2 (even), h/1 = 1 (odd). Hits (2, 0). OK.
# If s_u = 2, s_v = 2: gcd(2, 2) = 2. w/2 = 1 (odd), h/2 = 1 (odd). Hits (2, 2). FAIL.
# Thus, the constraint is: s_u != s_v.
# This is a 2-coloring problem on a graph.
# We want the lexicographically smallest assignment (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS to color component, greedily assign 1 to the first node
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

solve()
```
