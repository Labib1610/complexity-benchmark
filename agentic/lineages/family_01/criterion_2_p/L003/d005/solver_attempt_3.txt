import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition "sum of treatments is 3" means if u is 1, v must be 2, and vice versa.
    # This is a standard 2-coloring problem (bipartite graph check).
    # To get the lexicographically smallest sequence, we prefer 1 over 2.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i. To get lexicographically smallest,
            # we want to assign 1 to the smallest node in the component if possible.
            # Actually, for each connected component, we have two choices:
            # 1. Assign 1 to the starting node i.
            # 2. Assign 2 to the starting node i.
            # We want the resulting sequence to be lexicographically smallest.
            # This means for each component, we check both colorings and pick the one
            # that is lexicographically smaller.
            
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # Let's see if swapping 1s and 2s gives a better lexicographical result.
            # The current coloring is 'color'. The swapped one is '3 - color'.
            # We compare the two options for the nodes in this component.
            
            # Actually, the lexicographical requirement is global.
            # For each component, we can choose to keep the current coloring or swap it.
            # We should choose the one that makes the sequence smaller at the first index
            # where they differ. Since we process components in order 1..n,
            # we just need to ensure the first node of the component gets 1 if possible.
            
            if color[i] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
