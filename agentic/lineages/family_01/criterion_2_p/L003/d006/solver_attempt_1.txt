```python
import sys

# The problem states that for any pair (u, v) in an interference zone,
# the bitwise XOR of their configurations must be 3.
# Configurations are 1 (binary 01) and 2 (binary 10).
# 1 XOR 2 = 3.
# This means if u and v are connected, they must have different configurations.
# This is a standard 2-coloring problem (bipartite graph check).
# We want the lexicographically smallest sequence of 1s and 2s.
# For each connected component, we can assign the first node (smallest index)
# to 1, which forces the rest of the component. If that leads to a conflict,
# the graph is not bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer 1 over 2.
            # Since we process nodes 1 to n, we assign color 1 to the 
            # first uncolored node in each component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
