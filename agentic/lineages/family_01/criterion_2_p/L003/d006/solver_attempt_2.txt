```python
import sys

# The constraint is that for any two satellites u and v sharing an overlap,
# mode(u) * mode(v) = 2.
# Since modes are {1, 2}, the only way to get a product of 2 is:
# (1 * 2) = 2 or (2 * 1) = 2.
# This means if u and v share an overlap, they must have different modes.
# This is a standard 2-coloring problem on a graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # modes: 0 = unassigned, 1 = mode 1, 2 = mode 2
    modes = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a component is not colored, we start coloring it.
    # To make it lexicographically smallest, we prefer mode 1 whenever possible.
    
    for i in range(1, n + 1):
        if modes[i] == 0:
            # BFS to color the component
            stack = [i]
            modes[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if modes[v] == 0:
                        modes[v] = 3 - modes[u]
                        stack.append(v)
                    elif modes[v] == modes[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(modes[1:]))

solve()
```
