```python
import sys

# The matrices are:
# M1 = [[0, 1], [0, 0]]
# M2 = [[0, 0], [1, 0]]
# M1 * M1 = [[0, 0], [0, 0]] = Z
# M2 * M2 = [[0, 0], [0, 0]] = Z
# M1 * M2 = [[1, 0], [0, 0]] != Z
# M2 * M1 = [[0, 0], [0, 1]] != Z
#
# The condition M_u * M_v != Z and M_v * M_u != Z means:
# If s_u = 1 and s_v = 1, product is Z.
# If s_u = 2 and s_v = 2, product is Z.
# So, s_u must not equal s_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence s_1, ..., s_n.
# We can use BFS/DFS to color each connected component.
# For each component, we try assigning 1 to the first node and see if it works.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer color 1 for the smallest node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
