```python
import sys

# The problem states:
# Instance i has s_i in {1, 2}.
# Payload size is s_i + 1.
# If s_i = 1, size = 2.
# If s_i = 2, size = 3.
# Combined payload size for (u, v) is (s_u + 1) * (s_v + 1).
# Stability constraint: (s_u + 1) * (s_v + 1) must NOT be a perfect square.
# Possible sizes:
# If s_u=1, s_v=1: 2*2 = 4 (Square) -> Forbidden
# If s_u=1, s_v=2: 2*3 = 6 (Not square) -> Allowed
# If s_u=2, s_v=1: 3*2 = 6 (Not square) -> Allowed
# If s_u=2, s_v=2: 3*3 = 9 (Square) -> Forbidden
#
# This means for every edge (u, v), s_u must not equal s_v.
# This is exactly the definition of a bipartite graph.
# We need to 2-color the graph with colors {1, 2}.
# To get the lexicographically smallest sequence, we greedily assign 1 to 
# the smallest index in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
