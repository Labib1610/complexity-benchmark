```python
import sys

# The problem states that for every edge (u, v), one relay must be double the other.
# Since power levels must be positive integers and no relay can exceed 2,
# the only possible power levels are 1 and 2.
# If power(u) = 1, then power(v) must be 2.
# If power(u) = 2, then power(v) must be 1.
# This is equivalent to a 2-coloring problem on a graph.
# A graph is 2-colorable if and only if it is bipartite.
# We need the lexicographically smallest assignment, which means we want
# as many 1s as possible at the beginning.
# For each connected component, we can assign 1 to the first node encountered
# and propagate the coloring. To get the lexicographically smallest,
# we should try to assign 1 to the smallest index in each component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
