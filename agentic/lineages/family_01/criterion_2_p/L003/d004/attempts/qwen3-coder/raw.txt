```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    adj = defaultdict(list)
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # The allowed pairs are (1,2) and (2,1)
    # This is equivalent to a bipartite graph coloring problem
    # where each edge must have different colors on its endpoints
    
    color = [-1] * n  # -1 means unvisited, 0 and 1 are the two colors
    
    for start in range(n):
        if color[start] == -1:
            queue = deque()
            queue.append(start)
            color[start] = 0  # Assign color 0 to the starting node
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - current_color
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        print("NO")
                        return
    
    # Check if the coloring satisfies all edges
    for u, v in edges:
        if color[u] == color[v]:
            print("NO")
            return
    
    # Convert colors to configurations (1 and 2)
    # We want the lexicographically smallest sequence
    # That means we prefer node 0 to be 1 if possible
    # Since we started coloring node 0 with 0, we need to map 0->1 and 1->2
    result = []
    for c in color:
        if c == 0:
            result.append(1)
        else:
            result.append(2)
    
    print("YES")
    print(" ".join(map(str, result)))

if __name__ == "__main__":
    main()
```
