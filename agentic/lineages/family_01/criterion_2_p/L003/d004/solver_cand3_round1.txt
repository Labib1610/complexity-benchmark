```python
import sys

# The problem states:
# s_i = 1 => v_i = (1, 1)
# s_i = 2 => v_i = (1, -1)
# Two vectors are linearly dependent if they are the same or negatives of each other.
# (1, 1) and (1, 1) are dependent.
# (1, -1) and (1, -1) are dependent.
# (1, 1) and (1, -1) are independent.
# Thus, for any overlapping pair (u, v), s_u must not equal s_v.
# This is a 2-coloring problem on a graph.
# To get the lexicographically smallest sequence, we greedily assign 1 to the 
# first node of each connected component, then propagate the constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we want s_i = 1 if possible.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
