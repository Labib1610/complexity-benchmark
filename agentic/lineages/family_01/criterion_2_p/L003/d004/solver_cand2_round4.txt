```python
import sys

# The problem states:
# s_i is in {1, 2}.
# Cycle lengths are s_i + 1, so they are in {2, 3}.
# Constraint: gcd(s_u + 1, s_v + 1) == 1.
# Possible pairs (s_u + 1, s_v + 1):
# (2, 2) -> gcd = 2 (Invalid)
# (2, 3) -> gcd = 1 (Valid)
# (3, 2) -> gcd = 1 (Valid)
# (3, 3) -> gcd = 3 (Invalid)
#
# This means if u and v are connected, s_u + 1 != s_v + 1,
# which implies s_u != s_v.
# This is exactly the definition of a bipartite graph coloring problem.
# We want the lexicographically smallest sequence s_1, ..., s_n.
# To make it lexicographically smallest, we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try to assign 1 first.
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
