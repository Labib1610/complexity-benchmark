```python
import sys

# The condition is x << y == 4.
# Possible values for x, y are {1, 2}.
# 1 << 1 = 2 (False)
# 1 << 2 = 4 (True)
# 2 << 1 = 4 (True)
# 2 << 2 = 8 (False)
# Thus, for any link (u, v), the calibration factors must be different.
# This is a bipartite graph coloring problem.
# We need to assign 1 or 2 such that adjacent nodes have different values.
# To get the lexicographically smallest assignment, we greedily assign 1
# to the smallest available node in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer 1 over 2.
            # For each component, we can either set the first node to 1 or 2.
            # Setting the first node to 1 is always better for lexicographical order.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_c
                        stack.append((v, next_c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
