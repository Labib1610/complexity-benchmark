```python
import sys

# The constraint: |s_u - s_v| = min(s_u, s_v)
# Possible values for s_i are {1, 2}.
# If s_u = 1, s_v = 1: |1 - 1| = 0, min(1, 1) = 1. 0 != 1 (Invalid)
# If s_u = 1, s_v = 2: |1 - 2| = 1, min(1, 2) = 1. 1 == 1 (Valid)
# If s_u = 2, s_v = 1: |2 - 1| = 1, min(2, 1) = 1. 1 == 1 (Valid)
# If s_u = 2, s_v = 2: |2 - 2| = 0, min(2, 2) = 2. 0 != 2 (Invalid)
# Thus, for every edge (u, v), s_u must be different from s_v.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we greedily assign 1 
# to the smallest available node in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign 1 to i to be lexicographically smallest
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
