```python
import sys

# The matrices are:
# M1 = [[42, -30], [56, -40]]
# M2 = [[-60, 45], [-84, 63]]
# C_{u,v} = u * M_{s_u} + v * M_{s_v}
# det(C_{u,v}) = (u*a1 + v*a2)(u*d1 + v*d2) - (u*b1 + v*b2)(u*c1 + v*c2)
# Let M1 = [[a1, b1], [c1, d1]], M2 = [[a2, b2], [c2, d2]]
# det(u*M_i + v*M_j) = u^2*det(Mi) + v^2*det(Mj) + u*v*(a1*d2 + a2*d1 - b1*c2 - b2*c1)
# det(M1) = 42*(-40) - (-30)*56 = -1680 + 1680 = 0
# det(M2) = (-60)*63 - 45*(-84) = -3780 + 3780 = 0
# det(u*M_i + v*M_j) = u*v * (a1*d2 + a2*d1 - b1*c2 - b2*c1)
# Let K = a1*d2 + a2*d1 - b1*c2 - b2*c1
# K = 42*63 + (-60)*(-40) - (-30)*(-84) - 45*56
# K = 2646 + 2400 - 2520 - 2520 = 6
# So det(C_{u,v}) = u * v * 6 * (1 if s_u != s_v else something else)
# If s_u = s_v = 1: det = u*v * det(M1+M1) = u*v * 4 * det(M1) = 0
# If s_u = s_v = 2: det = u*v * det(M2+M2) = u*v * 4 * det(M2) = 0
# If s_u = 1, s_v = 2: det = u*v * det(M1+M2) = u*v * 6
# If s_u = 2, s_v = 1: det = u*v * det(M2+M1) = u*v * 6
# Thus, det(C_{u,v}) != 0 if and only if s_u != s_v.
# This is a 2-coloring problem on the graph of constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
