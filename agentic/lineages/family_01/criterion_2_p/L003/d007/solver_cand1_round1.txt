```python
import sys

# The matrices M1 and M2 are given.
# Let's check if M1*M1 = 0, M1*M2 = 0, M2*M1 = 0, M2*M2 = 0.
# M1 = [[-8, -5, 4, 5], [-10, -6, 5, 6], [-16, -10, 8, 10], [-10, -6, 5, 6]]
# M2 = [[6, 1, -2, -2], [10, -4, -2, 0], [10, 13, -6, -10], [13, -12, -1, 4]]

def mat_mul(A, B):
    C = [[0]*4 for _ in range(4)]
    for i in range(4):
        for k in range(4):
            if A[i][k] == 0: continue
            for j in range(4):
                C[i][j] += A[i][k] * B[k][j]
    return C

M1 = [[-8, -5, 4, 5], [-10, -6, 5, 6], [-16, -10, 8, 10], [-10, -6, 5, 6]]
M2 = [[6, 1, -2, -2], [10, -4, -2, 0], [10, 13, -6, -10], [13, -12, -1, 4]]

def is_zero(M):
    return all(all(x == 0 for x in row) for row in M)

# Check constraints:
# M1*M1 = 0? Yes.
# M1*M2 = 0? No.
# M2*M1 = 0? No.
# M2*M2 = 0? Yes.

# So:
# If u, v have same mode (1,1 or 2,2), product is 0.
# If u, v have different modes (1,2 or 2,1), product is not 0.
# This is a 2-coloring problem: adjacent nodes must have different colors.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
