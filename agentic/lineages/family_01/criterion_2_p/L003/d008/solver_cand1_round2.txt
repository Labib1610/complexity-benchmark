import sys

# The problem asks for a lexicographically smallest assignment of states {1, 2}
# to n items such that for m pairs (u, v), M_u * M_v != 0 and M_v * M_u != 0.
# Let's analyze the matrices.
# For any c_i > 0, M_i(s=1) and M_i(s=2) are non-zero.
# Let A = M_i(s=1) and B = M_i(s=2).
# Checking the products:
# A*A = 0, B*B = 0, A*B = 0, B*A = 0.
# Wait, let's re-verify:
# For any c, M_i(1)^2 = 0, M_i(2)^2 = 0, M_i(1)M_i(2) = 0, M_i(2)M_i(1) = 0.
# This means for any pair (u, v), if s_u = s_v, then M_u * M_v = c_u * c_v * (M_1 * M_1) = 0.
# Thus, the condition M_u * M_v != 0 implies s_u != s_v.
# This is a 2-coloring problem on a graph.
# The graph has n nodes and m edges. We need to check if it's bipartite.
# If bipartite, the lexicographically smallest assignment is found by
# greedily assigning 1 to the first node of each connected component,
# then propagating the constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a new component. To be lexicographically smallest,
            # we want s_i = 1 if possible.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            possible = True
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                res[u] = c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
    print("YES")
    print(*(res[1:]))

solve()
