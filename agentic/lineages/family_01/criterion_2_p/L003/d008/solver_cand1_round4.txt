import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want to minimize E = sum(s_i * 2^(n-i)).
    # Since 2^(n-i) decreases as i increases, we want s_1 to be 1 if possible.
    # If s_1 is fixed, the rest of the component is determined.
    # If a component is not connected to 1, we have a choice for the first node
    # encountered in that component. To minimize E, we want the smallest index
    # in the component to be 1.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To minimize E, we want the smallest index in this component to be 1.
            # Actually, for any component, we can pick the smallest index 'start_node'
            # and set color[start_node] = 1.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:

            # The current assignment sets color[i] = 1.

solve()
