```python
import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# the matrices M_u and M_v satisfy M_u * M_v != 0 and M_v * M_u != 0.
# Since M_i = c_i * A^(f_i), M_u * M_v = c_u * c_v * A^(f_u) * A^(f_v).
# The condition M_u * M_v != 0 is equivalent to A^(f_u) * A^(f_v) != 0.
# Let's check the four combinations:
# A1*A1:
# [-8 -5 4 5]   [-8 -5 4 5]   [0 0 0 0]
# [-10 -6 5 6] * [-10 -6 5 6] = [0 0 0 0]
# [-16 -10 8 10] [-16 -10 8 10] [0 0 0 0]
# [-10 -6 5 6]   [-10 -6 5 6]   [0 0 0 0]
# So A1*A1 = 0.
# A2*A2:
# [6 1 -2 -2]   [6 1 -2 -2]   [1 1 0 0]
# [10 -4 -2 0] * [10 -4 -2 0] = [20 26 -10 -10]
# [10 13 -6 -10] [10 13 -6 -10] [1 1 0 0]
# [13 -12 -1 4]  [13 -12 -1 4]  [1 1 0 0]
# A2*A2 != 0.
# A1*A2:
# [-8 -5 4 5]   [6 1 -2 -2]   [0 0 0 0]
# [-10 -6 5 6] * [10 -4 -2 0] = [0 0 0 0]
# [-16 -10 8 10] [10 13 -6 -10] [0 0 0 0]
# [-10 -6 5 6]   [13 -12 -1 4]  [0 0 0 0]
# So A1*A2 = 0.
# A2*A1:
# [6 1 -2 -2]   [-8 -5 4 5]   [0 0 0 0]
# [10 -4 -2 0] * [-10 -6 5 6] = [0 0 0 0]
# [10 13 -6 -10] [-16 -10 8 10] [0 0 0 0]
# [13 -12 -1 4]  [-10 -6 5 6]  [0 0 0 0]
# So A2*A1 = 0.
# Conclusion: The only valid combination is (f_u, f_v) = (2, 2).
# Wait, if (2, 2) is the only valid combination, then for any edge (u, v),
# both u and v must be 2. This implies that if there is any edge,
# all connected nodes must be 2. If a connected component has an edge,
# all nodes in that component must be 2. If a node is isolated, it can be 1.
# To get the lexicographically smallest, we set isolated nodes to 1 and
# nodes in components with edges to 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    ans = [0] * (n + 1)
    for i in range(1, n + 1):
        if not adj[i]:
            ans[i] = 1
        else:
            ans[i] = 2
            
    print("YES")
    print(*(ans[1:]))

solve()
```
