```python
import sys

# The problem states that for every interference zone (u, v), the product of 
# their assigned bands must be 2. Since the bands are {1, 2}, the only way 
# for the product to be 2 is if one relay is assigned 1 and the other is 2.
# This is a classic 2-coloring problem on a graph.
# We need to check if the graph is bipartite. If it is, we want the 
# lexicographically smallest assignment. To achieve this, we iterate 
# through nodes 1 to n, and if a node is not yet colored, we assign it 
# the smallest possible value (1) and perform BFS/DFS to color its component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we assign 1 to the first node
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
