import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The constraint is: s_u = 2 * s_v or s_v = 2 * s_u.
    # Since s_i in {1, 2}, this means:
    # If s_u = 1, then s_v must be 2.
    # If s_u = 2, then s_v must be 1.
    # This is equivalent to s_u != s_v.
    # So the graph must be bipartite.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # To get lexicographically smallest, try assigning 1 first
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, val = stack.pop()
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - val
                        stack.append((v, 3 - val))
                    elif assignment[v] == val:
                        print("NO")
                        return
    
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    solve()
