 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    ans = [0] * n
    ok = True

    for start in range(n):
        if color[start] != -1:
            continue
        stack = [start]
        color[start] = 0
        min_v = start
        verts = []
        while stack:
            v = stack.pop()
            verts.append(v)
            if v < min_v:
                min_v = v
            cv = color[v]
            for nb in adj[v]:
                if color[nb] == -1:
                    color[nb] = cv ^ 1
                    stack.append(nb)
                elif color[nb] == cv:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            break
        cmin = color[min_v]
        for v in verts:
            ans[v] = 1 if color[v] == cmin else 2

    if not ok:
        sys.stdout.write("NO")
    else:
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(map(str, ans)))

if __name__ == "__main__":
    solve()
```
