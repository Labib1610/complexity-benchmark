import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] = 0 (unvisited), 1, or 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If an instance is not colored, we start a BFS/DFS.
    # To make it lexicographically smallest, we prefer assigning 1 whenever possible.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS component
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        # Assign the smallest possible color (1, then 2)
                        # If u is 1, v must be 2. If u is 2, v must be 1.
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
