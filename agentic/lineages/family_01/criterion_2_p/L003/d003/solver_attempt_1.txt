```python
import sys

# The problem states that for any two branches u and v involved in a dispute,
# the product of their tiers must be 2.
# Since tiers are in {1, 2}, the only way for the product to be 2 is if one
# branch is 1 and the other is 2.
# This is equivalent to saying the graph of disputes must be bipartite.
# We need to find the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for branch 1, then 2, etc.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate through
    # branches 1 to n. If a branch is not colored, we assign it 1 (the smaller tier)
    # and perform BFS/DFS to color its connected component.
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
