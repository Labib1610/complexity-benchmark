```python
import sys

# The problem asks for a lexicographically smallest assignment of s_i in {1, 2}
# such that for every pair (u, v) sharing a backup array, the concatenated
# sequence of s_u + s_v pulses is not a square (i.e., not of the form XX).
# The length of the merged sequence is s_u + s_v.
# Possible lengths are 1+1=2, 1+2=3, 2+1=3, 2+2=4.
# A sequence of length 2 is a square if it is "11" (1+1).
# A sequence of length 3 is never a square (cannot be split into two equal halves).
# A sequence of length 4 is a square if it is "1111" or "2222".
# Since s_i is 1 or 2, the merged sequence is s_u 'sync's followed by s_v 'sync's.
# If s_u=1, s_v=1, sequence is "11", which is "1"+"1" (square).
# If s_u=1, s_v=2, sequence is "111", not a square.
# If s_u=2, s_v=1, sequence is "221", not a square.
# If s_u=2, s_v=2, sequence is "2222", which is "22"+"22" (square).
# Thus, the constraint is: if u and v share an array, s_u + s_v cannot be 2 (1+1)
# and s_u + s_v cannot be 4 (2+2).
# This simplifies to: s_u != s_v.
# This is a 2-coloring problem on the graph of backup arrays.
# We want the lexicographically smallest sequence s_1, s_2, ..., s_n.
# We greedily assign s_i = 1 if possible, else s_i = 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS/DFS to color the connected component
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
