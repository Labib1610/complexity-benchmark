```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        # No constraints, assign all to environment 1 (lexicographically smallest)
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build graph
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 = unvisited, 1 = env1, 2 = env2
    color = [0] * (n + 1)
    
    # Check bipartiteness using BFS for each component
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
            
        queue = deque()
        queue.append((start, 1))  # Assign color 1 to start node
        color[start] = 1
        
        while queue:
            node, c = queue.popleft()
            next_c = 3 - c  # Flip color: 1->2, 2->1
            
            for neighbor in adj[node]:
                if color[neighbor] == 0:
                    color[neighbor] = next_c
                    queue.append((neighbor, next_c))
                elif color[neighbor] == c:
                    # Conflict: adjacent nodes have same color
                    print("NO")
                    return
    
    # If we reach here, a valid assignment exists
    # Since we started with color 1 for each component's first node,
    # this gives us the lexicographically smallest assignment
    print("YES")
    result = []
    for i in range(1, n + 1):
        result.append(str(color[i]))
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
