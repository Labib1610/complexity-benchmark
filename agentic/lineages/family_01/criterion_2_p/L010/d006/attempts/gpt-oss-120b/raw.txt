import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n          # 0 or 1
    result = [0] * n
    for start in range(n):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        comp_nodes = [start]
        min_node = start
        while q:
            v = q.popleft()
            if v < min_node:
                min_node = v
            for to in adj[v]:
                if color[to] == -1:
                    color[to] = color[v] ^ 1
                    q.append(to)
                    comp_nodes.append(to)
                elif color[to] == color[v]:
                    print("NO")
                    return
        # decide orientation: min_node should be color 0 (value 1)
        if color[min_node] == 1:
            for v in comp_nodes:
                color[v] ^= 1
        for v in comp_nodes:
            result[v] = 1 if color[v] == 0 else 2

    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    solve()
