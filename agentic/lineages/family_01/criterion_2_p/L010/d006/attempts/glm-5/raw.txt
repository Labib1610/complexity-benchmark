```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    visited = [False] * (n + 1)

    for start in range(1, n + 1):
        if visited[start]:
            continue
        visited[start] = True
        color[start] = 1
        stack = [start]
        while stack:
            u = stack.pop()
            order.append(u)
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    parent[v] = u
                    color[v] = 3 - color[u]
                    stack.append(v)

    for u in range(1, n + 1):
        cu = color[u]
        for v in adj[u]:
            if color[v] == cu:
                print("NO")
                return

    for u in reversed(order):
        cnt1 = 0
        cnt2 = 0
        for v in adj[u]:
            if v == parent[u]:
                continue
            if color[v] == 1:
                cnt1 += 1
            else:
                cnt2 += 1
        if cnt1 > cnt2:
            color[u] = 2
        elif cnt2 > cnt1:
            color[u] = 1

    print("YES")
    print(' '.join(map(str, color[1:])))

solve()
```
