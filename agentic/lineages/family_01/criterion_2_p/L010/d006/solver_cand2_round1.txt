import sys

# The condition is: s_v = 5 % (s_u + 2)
# If s_u = 1: s_v = 5 % (1 + 2) = 5 % 3 = 2
# If s_u = 2: s_v = 5 % (2 + 2) = 5 % 4 = 1
# This means s_v must be different from s_u.
# Specifically, s_v = 3 - s_u.
# This is a bipartite graph problem. Each connected component must be 2-colorable.
# To get the lexicographically smallest sequence, for each component,
# we try setting the first node (smallest index) to 1, then propagate.
# If that fails, we try setting it to 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if ans[i] == 0:
            # Try setting ans[i] = 1
            component = []
            stack = [(i, 1)]
            ans[i] = 1
            component.append(i)
            
            possible = True
            ptr = 0
            while ptr < len(component):
                u = component[ptr]
                ptr += 1
                for v in adj[u]:
                    if ans[v] == 0:
                        ans[v] = 3 - ans[u]
                        component.append(v)
                    elif ans[v] == ans[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Reset and try setting ans[i] = 2
                for node in component:
                    ans[node] = 0
                
                stack = [(i, 2)]
                ans[i] = 2
                component = [i]
                possible = True
                ptr = 0
                while ptr < len(component):
                    u = component[ptr]
                    ptr += 1
                    for v in adj[u]:
                        if ans[v] == 0:
                            ans[v] = 3 - ans[u]
                            component.append(v)
                        elif ans[v] == ans[u]:
                            possible = False
                            break
                    if not possible:
                        break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(ans[1:]))

solve()
