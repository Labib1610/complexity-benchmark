import sys

# The system of equations:
# s_u * x + 2 * y = 0
# x + s_v * y = 0
# This system has non-trivial solutions if and only if the determinant is 0:
# det = (s_u * s_v) - (2 * 1) = 0
# s_u * s_v = 2
# Since s_i \in {1, 2}, the only way s_u * s_v = 2 is if one is 1 and the other is 2.
# This means for every edge (u, v), s_u != s_v.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest assignment, we want s_i = 1 whenever possible.
# For each connected component, we have two choices for the 2-coloring.
# We pick the one that results in a smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            is_bipartite = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Try setting the first node of the component to 1
            # If color[i] == 1, then nodes with color 1 get 1, color 2 get 2
            # If we set node i to 2, then nodes with color 1 get 2, color 2 get 1
            # We want the lexicographically smallest result.
            # Compare the two possible assignments for this component.
            
            # Assignment 1: color 1 -> 1, color 2 -> 2
            # Assignment 2: color 1 -> 2, color 2 -> 1
            
            # Find the smallest index in this component
            first_node = min(component)
            
            # If color[first_node] == 1, Assignment 1 gives 1 at first_node
            # If color[first_node] == 2, Assignment 2 gives 1 at first_node
            
            if color[first_node] == 1:
                for node in component:
                    res[node] = color[node]
            else:
                for node in component:
                    res[node] = 3 - color[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
