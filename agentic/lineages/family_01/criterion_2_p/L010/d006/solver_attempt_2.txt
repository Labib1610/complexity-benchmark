import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem is a 2-coloring problem (bipartite check).
    # To get the lexicographically smallest sequence, we want to assign 1
    # to the smallest possible index whenever possible.
    # For each connected component, we have two choices:
    # 1. Assign the first node in the component to 1.
    # 2. Assign the first node in the component to 2.
    # We pick the choice that results in a smaller value at the first index
    # of the component.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Use a list to store nodes in order of discovery
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring, check if we should flip the colors
            # We want the smallest value at the smallest index in the component.
            # The smallest index in this component is 'i'.
            # Currently color[i] is 1. If we flip, color[i] becomes 2.
            # Since 1 < 2, we keep the current coloring.
            
    print("YES")
    print(*(color[1:]))

solve()
