```python
import sys

# The problem asks for a 2-coloring of a graph where each edge (u, v)
# implies that u and v must have different colors (100g vs 200g).
# This is equivalent to checking if the graph is bipartite.
# The sequential commitment protocol implies we should try to assign 1
# to the current node if it doesn't lead to a contradiction.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unassigned, 1 = 100g, 2 = 200g
    color = [0] * (n + 1)
    
    # To check if an assignment is possible given current constraints:
    # We use a standard bipartite check (BFS/DFS).
    # The sequential protocol: for each node i from 1 to n,
    # if color[i] is already set, continue.
    # If not, try setting color[i] = 1. Check if this is valid.
    # If valid, keep it. If not, try color[i] = 2. If that's valid, keep it.
    # If neither, return NO.
    
    def is_consistent(start_node, start_color, current_colors):
        stack = [(start_node, start_color)]
        temp_colors = current_colors[:]
        temp_colors[start_node] = start_color
        
        history = []
        
        while stack:
            u, c = stack.pop()
            history.append(u)
            for v in adj[u]:
                if temp_colors[v] == 0:
                    temp_colors[v] = 3 - c
                    stack.append((v, 3 - c))
                elif temp_colors[v] == c:
                    return False, None
        return True, temp_colors

    for i in range(1, n + 1):
        if color[i] == 0:
            # Try setting color[i] = 1
            possible, new_colors = is_consistent(i, 1, color)
            if possible:
                color = new_colors
            else:
                # Try setting color[i] = 2
                possible, new_colors = is_consistent(i, 2, color)
                if possible:
                    color = new_colors
                else:
                    print("NO")
                    return

    print("YES")
    print(*(color[1:]))

solve()
```
