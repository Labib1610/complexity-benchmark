import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n          # 0 or 1
    comp_id = [-1] * n
    comps = []                # list of vertices per component
    from collections import deque
    comp_cnt = 0
    for i in range(n):
        if color[i] != -1:
            continue
        q = deque([i])
        color[i] = 0
        comp_vertices = []
        comp_id[i] = comp_cnt
        while q:
            v = q.popleft()
            comp_vertices.append(v)
            for to in adj[v]:
                if color[to] == -1:
                    color[to] = color[v] ^ 1
                    comp_id[to] = comp_cnt
                    q.append(to)
                elif color[to] == color[v]:
                    print("NO")
                    return
        comps.append(comp_vertices)
        comp_cnt += 1

    # decide flips to get lexicographically minimal (base-3) sequence
    flip = [0] * comp_cnt   # 0 = keep, 1 = invert colors
    decided = [False] * comp_cnt

    for i in range(n):
        cid = comp_id[i]
        if decided[cid]:
            continue
        # we want s_i = 1, i.e., final color 0
        # current color is color[i]; if we flip, final = color[i]^1
        if color[i] == 0:
            flip[cid] = 0
        else:
            flip[cid] = 1
        decided[cid] = True

    res = []
    for i in range(n):
        final = color[i] ^ flip[comp_id[i]]
        res.append('1' if final == 0 else '2')
    print("YES")
    print(' '.join(res))

if __name__ == "__main__":
    solve()
