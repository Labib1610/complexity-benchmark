```python
import sys

# The condition s_u^3 + s_v^3 = 9 with s_i in {1, 2}
# 1^3 + 1^3 = 2 != 9
# 1^3 + 2^3 = 1 + 8 = 9
# 2^3 + 1^3 = 8 + 1 = 9
# 2^3 + 2^3 = 16 != 9
# Thus, for every overlap (u, v), s_u must be different from s_v.
# This is a bipartite graph coloring problem.
# We want to minimize E = sum_{i=1}^n s_i * 3^{n-i}.
# To minimize E, we want s_1 to be as small as possible (1), then s_2, etc.
# This is equivalent to finding a 2-coloring such that the sequence (s_1, ..., s_n)
# is lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To minimize lexicographically, we iterate i from 1 to n.
    # If i is not colored, we try to color it 1.
    # Then we propagate the constraints using BFS/DFS.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS component
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Within this component, we have two possible colorings:
            # 1. The one we just found (color[u] = c)
            # 2. The flipped one (color[u] = 3 - c)
            # We want the one that makes the sequence lexicographically smallest.
            # Since we process i from 1 to n, the first node in the component
            # we encounter is the one with the smallest index.
            # Let this node be 'start_node'. We want color[start_node] = 1.
            # If our current coloring has color[start_node] = 2, we flip it.
            
            # Find the node with the smallest index in this component
            start_node = min(component, key=lambda x: x[0])[0]
            if color[start_node] == 2:
                for node, c in component:
                    color[node] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
```
