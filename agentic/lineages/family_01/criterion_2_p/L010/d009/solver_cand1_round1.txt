```python
import sys

# The area of a triangle with sides a, b, and c is given by Heron's formula.
# Here, sides are s_u, s_v, and sqrt(5).
# Area = 1/4 * sqrt(4 * s_u^2 * s_v^2 - (s_u^2 + s_v^2 - 5)^2) = 1
# 4 * s_u^2 * s_v^2 - (s_u^2 + s_v^2 - 5)^2 = 16
# Let x = s_u^2, y = s_v^2.
# 4xy - (x + y - 5)^2 = 16
# 4xy - (x^2 + y^2 + 25 + 2xy - 10x - 10y) = 16
# -x^2 - y^2 + 2xy + 10x + 10y - 25 = 16
# -(x - y)^2 + 10(x + y) = 41
# Since s_u, s_v are positive integers, x, y are perfect squares.
# Let s_u = a, s_v = b.
# -(a^2 - b^2)^2 + 10(a^2 + b^2) = 41
# If a = 1, -(1 - b^2)^2 + 10(1 + b^2) = 41
# -(1 - 2b^2 + b^4) + 10 + 10b^2 = 41
# -b^4 + 12b^2 - 1 + 10 = 41
# -b^4 + 12b^2 - 32 = 0
# b^4 - 12b^2 + 32 = 0
# (b^2 - 4)(b^2 - 8) = 0
# b^2 = 4 => b = 2.
# If a = 2, b = 1.
# So if s_u = 1, s_v = 2 (or vice versa).
# This means the graph must be bipartite.
# We need to color the graph with 1 and 2 such that adjacent nodes have different colors.
# To get the lexicographically smallest sequence, we greedily assign 1 to the smallest
# available node in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
