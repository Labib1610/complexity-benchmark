import sys

def solve():
    # The problem states s_i must be a positive integer divisor of 2.
    # The divisors of 2 are {1, 2}.
    # Resonance condition: s_u^5 + s_v^5 - 3 * s_u^2 * s_v^2 = 21
    # Let's test the possible pairs (s_u, s_v) where s_u, s_v in {1, 2}:
    # (1, 1): 1^5 + 1^5 - 3(1^2 * 1^2) = 1 + 1 - 3 = -1 != 21
    # (1, 2): 1^5 + 2^5 - 3(1^2 * 2^2) = 1 + 32 - 3(4) = 33 - 12 = 21. (Valid)
    # (2, 1): 2^5 + 1^5 - 3(2^2 * 1^2) = 32 + 1 - 12 = 21. (Valid)
    # (2, 2): 2^5 + 2^5 - 3(2^2 * 2^2) = 32 + 32 - 3(16) = 64 - 48 = 16 != 21
    
    # So, for every overlap (u, v), we must have (s_u, s_v) = (1, 2) or (2, 1).
    # This means s_u != s_v, and specifically, the graph must be bipartite.
    # If the graph is bipartite, we can color it with 1 and 2.
    # To get the lexicographically smallest sequence, we want s_i = 1 whenever possible.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To make it lexicographically smallest, we prefer color 1 for the smallest index
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 2 if c == 1 else 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
