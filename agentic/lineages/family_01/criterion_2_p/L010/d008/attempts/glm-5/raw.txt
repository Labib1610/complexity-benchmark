```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)

    # We want lexicographically smallest sequence.
    # Since 1 < 2, we prefer assigning 1 whenever possible.
    # We use BFS starting from unassigned nodes, trying to assign 1 first.
    
    color = [0] * (n + 1)  # 0: unassigned, 1 or 2: assigned value
    q = deque()
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS from this node
            color[i] = 1
            q.append(i)
            
            while q:
                u = q.popleft()
                for v in adj[u]:
                    # Neighbor must satisfy u^3 + v^3 = 9
                    # If u is 1, v must be 2. If u is 2, v must be 1.
                    required_val = 3 - color[u]
                    
                    if color[v] == 0:
                        color[v] = required_val
                        q.append(v)
                    elif color[v] != required_val:
                        # Conflict found
                        sys.stdout.write("NO\n")
                        return

    # Validation pass (safety check)
    # Though BFS logic ensures constraints, we verify for correctness.
    # This also handles cases where logic might be complex, though here it's straightforward.
    for u in range(1, n + 1):
        for v in adj[u]:
            if color[u]**3 + color[v]**3 != 9:
                sys.stdout.write("NO\n")
                return

    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(map(str, color[1:])) + '\n')

if __name__ == '__main__':
    solve()
```
