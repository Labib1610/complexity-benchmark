import sys

# The resonance condition: s_u^2 + 3*s_u*s_v + s_v^2 = 11
# Possible values for s_u, s_v are {1, 2}.
# If s_u=1, s_v=1: 1 + 3 + 1 = 5 != 11
# If s_u=1, s_v=2: 1 + 6 + 4 = 11 (Valid)
# If s_u=2, s_v=1: 4 + 6 + 1 = 11 (Valid)
# If s_u=2, s_v=2: 4 + 12 + 4 = 20 != 11
# Thus, for every overlap (u, v), s_u must be different from s_v.
# This is a bipartite graph coloring problem.

# Energy footprint E = sum_{i=1}^n s_i * W_{n-i}
# W_0 = 1
# W_k = 3 + sum_{j=1}^{k-1} (2 * W_j) for k >= 1
# W_1 = 3
# W_2 = 3 + 2*W_1 = 3 + 6 = 9
# W_3 = 3 + 2*(W_1 + W_2) = 3 + 2*(3 + 9) = 27
# W_k = 3 * 3^{k-1} = 3^k for k >= 1.
# So W_0 = 1, W_k = 3^k for k >= 1.
# E = s_n * W_0 + s_{n-1} * W_1 + ... + s_1 * W_{n-1}
# E = s_n * 1 + s_{n-1} * 3^1 + s_{n-2} * 3^2 + ... + s_1 * 3^{n-1}
# To minimize E, we want smaller s_i for larger powers of 3.
# The powers are 3^{n-1}, 3^{n-2}, ..., 3^0.
# So we want s_1 to be 1 if possible, then s_2, etc.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # To minimize E, we prefer color 1 for smaller indices.
            # For each connected component, we have two choices:
            # 1. Start with color 1
            # 2. Start with color 2
            # We compare the contribution to E for both choices.
            comp = []
            stack = [i]
            color[i] = 1
            comp.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        comp.append(v)
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Try both assignments for this component
            # Assignment A: current colors
            # Assignment B: flip colors (1->2, 2->1)
            # Contribution to E: sum_{v in comp} color[v] * 3^{n-v}
            val_a = 0
            val_b = 0
            for v in comp:
                power = 3**(n - v)
                if color[v] == 1:
                    val_a += 1 * power
                    val_b += 2 * power
                else:
                    val_a += 2 * power
                    val_b += 1 * power
            
            if val_b < val_a:
                for v in comp:
                    color[v] = 3 - color[v]
                    
    print("YES")
    print(*(color[1:]))

solve()
