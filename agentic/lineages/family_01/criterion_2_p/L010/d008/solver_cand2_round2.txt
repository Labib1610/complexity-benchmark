```python
import sys

# The condition s_u^2 + 3*s_u*s_v + s_v^2 = 11
# Possible values for s_u, s_v are {1, 2}:
# If s_u=1, s_v=1: 1 + 3 + 1 = 5 != 11
# If s_u=1, s_v=2: 1 + 6 + 4 = 11 (Valid)
# If s_u=2, s_v=1: 4 + 6 + 1 = 11 (Valid)
# If s_u=2, s_v=2: 4 + 12 + 4 = 20 != 11
# Thus, s_u != s_v. This is a bipartite graph coloring problem.
# We want to maximize the number of 1s at lower indices.
# This means for each connected component, we try to assign 1 to the smallest index.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for component
            # To prioritize 1 at lower indices, we greedily assign 1 to the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
