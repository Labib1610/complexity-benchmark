We can model the sensors as nodes and interfering pairs as edges. This is a classic bipartite graph checking problem. We just need to check whether the graph is bipartite; if yes, we assign channel 1/2 based on color, but choose the lexicographically smallest assignment.

Here’s one way to do it:

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * n
    possible = True
    
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break
    
    if not possible:
        print("NO")
    else:
        # Ensure lexicographically smallest: 
        # Re-check with BFS from smallest possible index and assign greedily
        color = [-1] * n
        for i in range(n):
            if color[i] == -1:
                color[i] = 1  # try 1 first
                stack = [i]
                while stack:
                    u = stack.pop()
                    for v in adj[u]:
                        if color[v] == -1:
                            # Try to make v as small as possible: if color[u] == 1 -> v = 2, else v could be 1 if no conflict
                            cand = 1 if color[u] == 2 else 2
                            if cand == 1:
                                # check all neighbors: if any neighbor already 1, must pick 2
                                conflict = False
                                for w in adj[v]:
                                    if color[w] == 1:
                                        conflict = True
                                        break
                                if conflict:
                                    cand = 2
                            color[v] = cand
                            stack.append(v)
                        elif color[v] == color[u]:
                            possible = False
                            break
                    if not possible:
                        break
            if not possible:
                break
        
        if not possible:
            print("NO")
        else:
            print("YES")
            print(" ".join(str(c) for c in color))

if __name__ == "__main__":
    solve()
```
