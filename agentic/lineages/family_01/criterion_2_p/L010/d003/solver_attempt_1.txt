# Increase recursion depth for deep DFS trees

    if not input_data:
        return

    m = int(input_data[1])

    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # This is a standard 2-coloring problem (bipartite graph check).
    # We want the lexicographically smallest sequence:
    # For each connected component, we try assigning 1 to the smallest node.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):

            # 2. Assign 2 to the smallest node in the component.
            # We pick the one that results in a smaller sequence.
            
            # Collect component nodes
            q = [i]
            comp_nodes = [i]

            # Actually, the lexicographical requirement is global.
            # For each component, we must choose the coloring that makes the 
            # sequence lexicographically smallest.
            # Since we process nodes 1 to n, if we encounter an unassigned node,
            # we must decide its value. To be lexicographically smallest,
            # we should always try to assign 1 to the smallest unassigned node.
