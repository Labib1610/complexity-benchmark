```python
import sys

# The problem states that for any two arms u and v sharing a workspace,
# the two-digit code formed by their modes (s_u, s_v) must be a multiple of 3.
# Possible modes are {1, 2}.
# Possible codes:
# 11: 11 % 3 = 2 (No)
# 12: 12 % 3 = 0 (Yes)
# 21: 21 % 3 = 0 (Yes)
# 22: 22 % 3 = 1 (No)
# Thus, if u and v share a workspace, s_u must be different from s_v.
# This is a standard 2-coloring problem (bipartite graph check).
# We want the lexicographically smallest configuration.
# This means for each connected component, we want to assign 1 to the 
# smallest index node if possible, or generally ensure the sequence is minimal.
# Since we want the lexicographically smallest sequence, we iterate from 1 to n.
# If a node is unvisited, we start a BFS/DFS. To make it lexicographically smallest,
# we should try to assign 1 to the smallest index in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we assign 1 to the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
