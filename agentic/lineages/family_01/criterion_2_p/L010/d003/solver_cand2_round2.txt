import sys

# The condition s_i! = s_i implies s_i = 1 or s_i = 2.
# The determinant of the matrix M = [[a, b, 1], [b, a, 1], [1, 1, 1]] is:
# det(M) = a(a - 1) - b(b - 1) + 1(b - a)
# det(M) = a^2 - a - b^2 + b + b - a
# det(M) = a^2 - 2a - b^2 + 2b
# det(M) = (a^2 - 2a + 1) - (b^2 - 2b + 1)
# det(M) = (a - 1)^2 - (b - 1)^2
# For det(M) != 0, we need (a - 1)^2 != (b - 1)^2.
# Since a, b in {1, 2}, (a-1) and (b-1) are in {0, 1}.
# (a-1)^2 is 0 if a=1, and 1 if a=2.
# Thus, (a-1)^2 != (b-1)^2 is equivalent to a != b.
# The problem reduces to: assign 1 or 2 to each node such that for every edge (u, v), s_u != s_v.
# This is exactly the 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest sequence, we greedily assign 1 to the smallest 
# available node in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color component
            # To be lexicographically smallest, we prefer 1 over 2.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
