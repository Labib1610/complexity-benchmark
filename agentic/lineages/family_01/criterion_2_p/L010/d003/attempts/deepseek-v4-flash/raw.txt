We can model this as a graph bipartiteness problem.  
If the graph is bipartite, we can assign facilities; otherwise, it’s impossible.

To get the lexicographically smallest assignment, we can use BFS/DFS coloring, always trying to assign `1` first when possible.

```python
import sys
from collections import deque

input = sys.stdin.readline

def solve():
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * n
    possible = True
    
    for i in range(n):
        if color[i] == -1:
            color[i] = 1
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break
    
    if not possible:
        print("NO")
        return
    
    print("YES")
    print(' '.join(str(c) for c in color))

if __name__ == "__main__":
    solve()
```
