```python
import sys

# The condition: s_u * x^2 + s_v * x + k = 0 has a rational root for some odd k.
# A quadratic equation ax^2 + bx + c = 0 has rational roots if and only if the 
# discriminant D = b^2 - 4ac is a perfect square.
# Here, D = s_v^2 - 4 * s_u * k.
# We need s_v^2 - 4 * s_u * k = y^2 for some odd k.
# This implies y^2 = s_v^2 - 4 * s_u * k.
# Since k is odd, let k = 2m + 1.
# y^2 = s_v^2 - 4 * s_u * (2m + 1) = s_v^2 - 8 * s_u * m - 4 * s_u.
#
# Case 1: s_u = 1, s_v = 1. D = 1 - 4k. For D to be a square y^2, y^2 = 1 - 4k.
# If k = -2, D = 9 = 3^2. k is even (not allowed).
# If k = -6, D = 25 = 5^2. k is even.
# If k = -3, D = 13 (not square).
# If k = -1, D = 5 (not square).
# If k = 0, D = 1 = 1^2. k is even.
# If k = -2, D = 9.
# Actually, for any odd k, 1 - 4k is odd. If 1 - 4k = y^2, then y must be odd.
# Let y = 2j + 1. (2j+1)^2 = 4j^2 + 4j + 1 = 1 - 4k => 4j^2 + 4j = -4k => j^2 + j = -k.
# Since j^2 + j = j(j+1) is always even, -k must be even, so k must be even.
# But k must be odd. Thus, s_u=1, s_v=1 is impossible.
#
# Case 2: s_u = 1, s_v = 2. D = 4 - 4k = 4(1-k). For D to be a square, 1-k must be a square.
# Let 1-k = m^2. k = 1 - m^2. For k to be odd, m^2 must be even, so m must be even.
# Let m = 2. k = 1 - 4 = -3 (odd). This works. So (1, 2) is allowed.
#
# Case 3: s_u = 2, s_v = 1. D = 1 - 8k. For D to be a square, 1-8k = y^2.
# y^2 = 1 mod 8. This is always true for any odd y.
# Let y = 3, 9 = 1 - 8k => 8 = -8k => k = -1 (odd). This works. So (2, 1) is allowed.
#
# Case 4: s_u = 2, s_v = 2. D = 4 - 8k = 4(1-2k). For D to be a square, 1-2k = m^2.
# Since 1-2k is odd, m must be odd. Let m = 1. 1-2k = 1 => k = 0 (even).
# Let m = 3. 1-2k = 9 => -2k = 8 => k = -4 (even).
# 1-2k is always odd, so m^2 is odd, so m is odd. 1-2k = m^2 => 2k = 1-m^2.
# Since m is odd, m^2 = 1 mod 8 or 1 mod 4. 1-m^2 is even.
# Let m=1, k=0. Let m=3, k=-4. Let m=5, k=-12. All k are even.
# Thus, (2, 2) is impossible.
#
# Summary: (1, 1) NO, (1, 2) YES, (2, 1) YES, (2, 2) NO.
# This is equivalent to: s_u != s_v.
# This is a 2-coloring problem on a graph. If the graph is bipartite, YES.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    print("YES")
    print(*(color[1:]))

solve()
```
