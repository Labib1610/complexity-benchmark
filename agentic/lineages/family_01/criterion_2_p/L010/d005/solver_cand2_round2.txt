import sys

# The condition is s_u^2 + s_v^2 = 5.
# Since s_i \in {1, 2}, s_i^2 \in {1, 4}.
# The only way to get 1 + 4 = 5 is if one arm is 1 and the other is 2.
# This means the graph must be bipartite.
# We need to color the graph with 1 and 2 such that every edge connects a 1 and a 2.
# To get the lexicographically smallest sequence, we want s_i = 1 whenever possible.
# For each connected component, we have two choices for coloring (if bipartite).
# Let the two sets of the bipartition be A and B.
# We can either assign 1 to A and 2 to B, or 2 to A and 1 to B.
# To make the sequence lexicographically smallest, for each component,
# we look at the smallest index i in that component.
# If we assign s_i = 1, we check if that forces a valid coloring.
# Since it's a bipartite graph, once we fix one node, all others are fixed.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Component found. Try setting color[i] = 1
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            idx_stack = 0
            while idx_stack < len(component):
                u = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # We have two choices:
            # Choice 1: nodes with color 1 get value 1, nodes with color 2 get value 2.
            # Choice 2: nodes with color 1 get value 2, nodes with color 2 get value 1.
            # We want the lexicographically smallest result.
            # Find the smallest index 'first' in this component.
            # If we pick Choice 1, res[first] = 1. If Choice 2, res[first] = 2.
            # Clearly Choice 1 is better.
            for node in component:
                res[node] = color[node]
                
    print("YES")
    print(*(res[1:]))

solve()
