```python
import sys

# The problem states that for each critical pair (u, v), the combined mass must be 300g.
# Since weights are 100g (Type 1) and 200g (Type 2), the only way to get 300g is 100+200.
# This means for every pair (u, v), slot u and slot v must have different types.
# This is a 2-coloring problem (bipartite graph check).
# If the graph is bipartite, we have connected components.
# For each component, there are two possible colorings (swapping 1s and 2s).
# To get the lexicographically smallest sequence, for each component, we check
# which of the two colorings results in a smaller sequence.
# Specifically, we want the first node in the component (by index) to be 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings:
            # Option 1: color[i] = 1
            # Option 2: color[i] = 2
            # We want the lexicographically smallest result.
            # Find the smallest index in this component.
            comp_nodes = sorted(component)
            first = comp_nodes[0]
            
            # If we set color[first] = 1, then for all v in component:
            # if color[v] == color[first], weight is 1, else 2.
            # If we set color[first] = 2, then for all v in component:
            # if color[v] == color[first], weight is 2, else 1.
            
            # Compare the two options at the first index where they differ.
            # Actually, we just need to ensure the smallest index in the component
            # gets a 1 if possible.
            
            # Let's pick the coloring such that color[first] becomes 1.
            # This is always possible because we can flip all colors in a component.
            # Wait, the lexicographical requirement is global.
            # For each component, we choose the coloring that makes the sequence
            # lexicographically smallest. Since we process components independently,
            # we just need to make the smallest index in the component 1.
            
            base_color = color[first]
            for node in component:
                if color[node] == base_color:
                    res[node] = 1
                else:
                    res[node] = 2
                    
    print("YES")
    print(*(res[1:]))

solve()
```
