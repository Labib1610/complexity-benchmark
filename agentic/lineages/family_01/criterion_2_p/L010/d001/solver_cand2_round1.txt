import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Constraint: mode(u) * mode(v) = 2
    # Possible modes are {1, 2}.
    # If mode(u) = 1, then mode(v) must be 2.
    # If mode(u) = 2, then mode(v) must be 1.
    # This is a bipartite coloring problem where colors are 1 and 2.
    
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # To get lexicographically smallest, try assigning 1 first
            stack = [(i, 1)]
            colors[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                for v in adj[u]:
                    target_c = 3 - c
                    if colors[v] == 0:
                        colors[v] = target_c
                        stack.append((v, target_c))
                    elif colors[v] != target_c:
                        print("NO")
                        return
            
            # Check if this component could have started with 2 instead
            # But wait, the lexicographical requirement applies to the whole sequence.
            # We must decide the color of the smallest node in each component.
            # If we start with 1, the smallest node gets 1. If we start with 2, it gets 2.
            # Since 1 < 2, we always prefer 1 for the smallest node of each component.
            
    print("YES")
    print(*(colors[1:]))

solve()
