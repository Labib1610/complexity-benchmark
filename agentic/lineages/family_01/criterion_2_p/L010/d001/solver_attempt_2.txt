import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(10**6)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want the lexicographically smallest assignment (s1, s2, ..., sn)
    # where si in {1, 2}. This means we prefer 1 over 2.
    # This is a 2-SAT problem where we want to set variables to 1 if possible.
    # Since we want the lexicographically smallest sequence, we iterate 1 to n.
    # If a node is not assigned, try assigning 1. If that leads to a contradiction,
    # assign 2. If that also leads to a contradiction, then NO.
    
    assignment = [0] * (n + 1)
    
    def get_components(start_node, current_assignment):
        # Standard 2-coloring (bipartite check)
        stack = [(start_node, 1)]
        comp = []
        current_assignment[start_node] = 1
        comp.append((start_node, 1))
        
        while stack:
            u, color = stack.pop()
            for v in adj[u]:
                if current_assignment[v] == 0:
                    current_assignment[v] = 3 - color
                    comp.append((v, 3 - color))
                    stack.append((v, 3 - color))
                elif current_assignment[v] == color:
                    return None
        return comp

    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Try assigning i to 1
            # We need to check if this choice forces any other nodes
            # Since we want lexicographically smallest, we greedily pick 1
            # for the first unassigned node in each connected component.
            comp = get_components(i, assignment)
            if comp is None:
                print("NO")
                return
            
            # Check if this component's assignment is valid
            # The current assignment is already set by get_components
            # We need to ensure that for the first node i, we picked 1.
            # If the component forced i to be 2, we flip the whole component.
            if assignment[i] == 2:
                for node, color in comp:
                    assignment[node] = 3 - color
                    
    print("YES")
    print(*(assignment[1:]))

solve()
