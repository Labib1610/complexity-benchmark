```python
import sys

# The condition: for every pair (u, v) with overlapping ranges, 
# multiplier(u) * multiplier(v) = 2.
# Possible multipliers are {1, 2}.
# If multiplier(u) = 1, then multiplier(v) must be 2.
# If multiplier(u) = 2, then multiplier(v) must be 1.
# This is a bipartite coloring problem.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest assignment, we want 1s as much as possible
# for smaller indices.
# For each connected component, we have two choices of coloring (swap 1s and 2s).
# We choose the one that makes the smallest index in the component 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if we should flip colors to make the smallest index 1
            # The smallest index in this component is min(component)
            # We want color[min(component)] to be 1.
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
```
