import sys

# The problem states that for every connected pair (u, v), the sum of their 
# voltage states must be 3. Since states are {1, 2}, this means if u is 1, 
# v must be 2, and if u is 2, v must be 1. This is equivalent to saying 
# that the graph must be bipartite, and for every connected component, 
# once one node's state is fixed, all others are determined.
# To get the lexicographically smallest sequence, we iterate from 1 to n.
# If a node is not yet assigned, we try assigning it 1. If that leads to 
# a contradiction, we try 2. If both fail, then no solution exists.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Try assigning 1 to the current component
            component = []
            stack = [(i, 1)]
            assignment[i] = 1
            component.append(i)
            
            possible = True
            ptr = 0
            while ptr < len(component):
                u = component[ptr]
                ptr += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Reset and try assigning 2
                for node in component:
                    assignment[node] = 0
                
                assignment[i] = 2
                component = [i]
                ptr = 0
                possible = True
                while ptr < len(component):
                    u = component[ptr]
                    ptr += 1
                    for v in adj[u]:
                        if assignment[v] == 0:
                            assignment[v] = 3 - assignment[u]
                            component.append(v)
                        elif assignment[v] == assignment[u]:
                            possible = False
                            break
                    if not possible:
                        break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(assignment[1:]))

solve()
