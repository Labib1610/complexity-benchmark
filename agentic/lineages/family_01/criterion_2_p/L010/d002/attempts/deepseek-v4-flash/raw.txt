```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    possible = True

    # BFS to check bipartiteness and assign initial colors
    from collections import deque
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        q.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break

    if not possible:
        print("NO")
        return

    # Now we have a valid 2-coloring, but we need lexicographically smallest.
    # We can flip colors in each connected component if it makes the sequence smaller.
    visited = [False] * n
    for i in range(n):
        if not visited[i]:
            # collect component
            comp = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            # Try to see if flipping this component gives lexicographically smaller
            # Compare current vs flipped at first differing index
            flip = False
            for u in sorted(comp):
                if color[u] == 0:
                    # current has 0, flipped would have 1 -> current is smaller
                    break
                else:
                    # current has 1, flipped would have 0 -> flipped is smaller
                    flip = True
                    break
            if flip:
                for u in comp:
                    color[u] ^= 1

    print("YES")
    print(' '.join(str(c + 1) for c in color))

if __name__ == "__main__":
    solve()
```
