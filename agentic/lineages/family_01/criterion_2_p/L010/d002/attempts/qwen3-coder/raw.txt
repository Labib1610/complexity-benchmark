```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    adj = defaultdict(list)
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    color = [-1] * n  # -1: unvisited, 0: phase 1, 1: phase 2
    
    # Check bipartiteness and find lexicographically smallest coloring
    for start in range(n):
        if color[start] == -1:
            # BFS to color component
            queue = deque()
            queue.append((start, 0))  # Assign phase 1 (0) to start
            color[start] = 0
            
            while queue:
                u, c = queue.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        queue.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If we reach here, it's bipartite. Output the coloring.
    print("YES")
    result = []
    for i in range(n):
        if color[i] == -1:
            result.append("1")  # Default to phase 1 if isolated node
        else:
            result.append(str(color[i] + 1))  # Convert 0,1 to 1,2
    
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
