import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # polarity: 0 = unassigned, 1 = polarity 1, 2 = polarity 2
    polarity = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a compound is unassigned, we try assigning it 1 first.
    # If that leads to a contradiction, we try 2.
    
    for i in range(1, n + 1):
        if polarity[i] == 0:
            # Start a BFS/DFS to color the connected component
            # We need to check if this component is bipartite.
            # Since we want the lexicographically smallest, we prioritize 1.
            component = []
            stack = [i]
            polarity[i] = 1
            component.append(i)
            
            # Standard bipartite check
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if polarity[v] == 0:
                        polarity[v] = 3 - polarity[u]
                        component.append(v)
                    elif polarity[v] == polarity[u]:
                        print("NO")
                        return
            
    print("YES")
    print(*(polarity[1:]))

solve()
