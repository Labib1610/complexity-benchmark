import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    g = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        g[u].append(v)
        g[v].append(u)

    color = [-1] * n          # 0/1 bipartite color
    comp_id = [-1] * n
    comps = []                # list of (nodes list, colors list)

    for i in range(n):
        if color[i] != -1:
            continue
        # BFS this component
        q = deque([i])
        color[i] = 0
        nodes = []
        cols = []
        ok = True
        while q:
            v = q.popleft()
            nodes.append(v)
            cols.append(color[v])
            for to in g[v]:
                if color[to] == -1:
                    color[to] = color[v] ^ 1
                    q.append(to)
                elif color[to] == color[v]:
                    print("NO")
                    return
        comps.append((nodes, cols))

    # decide orientation for each component to get lexicographically smallest sequence
    ans = [0] * n
    for nodes, cols in comps:
        # sort nodes to compare sequences in index order
        order = sorted(range(len(nodes)), key=lambda k: nodes[k])
        seq1 = []  # orientation: col 0 -> 1, col 1 -> 2
        seq2 = []  # swapped
        for idx in order:
            c = cols[idx]
            seq1.append(1 if c == 0 else 2)
            seq2.append(2 if c == 0 else 1)
        # choose smaller lexicographically
        if seq1 <= seq2:
            # use seq1
            for idx in range(len(nodes)):
                ans[nodes[idx]] = 1 if cols[idx] == 0 else 2
        else:
            for idx in range(len(nodes)):
                ans[nodes[idx]] = 2 if cols[idx] == 0 else 1

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
