```python
import sys

# The condition for each arm u is:
# sum(s_v for v in neighbors(u)) + prod(s_v for v in neighbors(u)) = k_u * (3 - s_u) + (3 - s_u)^k_u
# Let x_u = 3 - s_u. Since s_u in {1, 2}, x_u in {2, 1}.
# The condition becomes:
# sum(s_v) + prod(s_v) = k_u * x_u + x_u^k_u
# If s_v = 1, x_v = 2. If s_v = 2, x_v = 1.
# Note that s_v = 3 - x_v.
# sum(3 - x_v) + prod(3 - x_v) = k_u * x_u + x_u^k_u
# 3*k_u - sum(x_v) + prod(3 - x_v) = k_u * x_u + x_u^k_u
# This looks like a constraint on the parity or specific values.
# Testing small k_u:
# If k_u = 1: s_v = 1*(3-s_u) + (3-s_u)^1 = 3-s_u + 3-s_u = 6 - 2*s_u
# If s_u=1, s_v=4 (impossible). If s_u=2, s_v=2.
# Actually, the condition simplifies to s_u != s_v for all edges (u, v).
# This is equivalent to checking if the graph is bipartite.
# If bipartite, the two colorings are the candidates.
# Lexicographically smallest means we prefer s_i=1 over s_i=2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # We have connected components. For each component, we can swap 1s and 2s.
    # To get lexicographically smallest, for each component, pick the coloring
    # that has 1 at the smallest index node in that component.
    
    final_colors = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Check which coloring is better
            # Option 1: color[i] stays as is
            # Option 2: flip all colors in component
            # We want the one that results in a smaller sequence.
            # Compare color[i] and 3-color[i] at the first index.
            # Actually, just pick the one where the smallest index in component has color 1.
            min_node = min(component)
            if color[min_node] == 1:
                for node in component:
                    final_colors[node] = color[node]
            else:
                for node in component:
                    final_colors[node] = 3 - color[node]
                    
    print("YES")
    print(*(final_colors[1:]))

solve()
```
