```python
import sys

# The problem asks for a 2-coloring of a graph such that every edge connects
# vertices of different colors (1 and 2). This is equivalent to checking if
# the graph is bipartite. If it is, we want the lexicographically smallest
# sequence of 1s and 2s.
# To get the lexicographically smallest sequence, we want s_1 to be 1 if possible.
# If the graph has multiple connected components, for each component, we want
# the smallest possible value for the smallest index in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is uncolored, we start a BFS/DFS.
    # To make the sequence lexicographically smallest, we prefer color 1
    # for the smallest index in each component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for this component
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
