import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want the lexicographically smallest assignment (1s before 2s).
    # This is a 2-coloring problem. For each connected component,
    # we have two possible colorings (swap 1s and 2s).
    # To get the lexicographically smallest, for each component,
    # we pick the coloring that results in a smaller value at the 
    # smallest index in that component.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            idx_comp = 0
            while idx_comp < len(component):
                u, c = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_node = min(node for node, c in component)
            # Check the color of the min_node in the current coloring
            # If color[min_node] is 2, we must flip the colors of this component
            # to make the min_node 1 (lexicographically smaller).
            if color[min_node] == 2:
                for node, c in component:
                    color[node] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
