import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want the lexicographically smallest sequence (s1, s2, ..., sn)
    # where si is 1 or 2.
    # This means we want to assign 1 to as many early modules as possible.
    # This is a 2-coloring problem. For each connected component, we have 
    # two possible colorings (swap 1s and 2s).
    # To make the sequence lexicographically smallest, for each component,
    # we find the smallest index 'i' in that component. We must decide
    # whether to assign 1 or 2 to 'i'. Since we want the smallest sequence,
    # we always try to assign 1 to the smallest index in the component.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the component
            component = []
            stack = [i]
            assignment[i] = 1
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            # to verify consistency and then potentially flip
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # The current assignment is valid. 
            # Check if it's the lexicographically smallest for this component.
            # The smallest index in this component is 'i'.
            # We assigned assignment[i] = 1. This is already the smallest possible.
            # So we don't need to flip.
            
    print("YES")
    print(*(assignment[1:]))

solve()
