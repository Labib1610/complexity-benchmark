import sys

# The condition for sensor i is:
# Product_{j in neighbors(i)} b_j = 2^(d_i * (2 - b_i))
# Let x_i = b_i - 1, so x_i in {0, 1}.
# b_i = x_i + 1.
# The condition is: Product_{j in neighbors(i)} (x_j + 1) = 2^(d_i * (1 - x_i))
# Taking log2: Sum_{j in neighbors(i)} log2(x_j + 1) = d_i * (1 - x_i)
# Since x_j in {0, 1}, log2(x_j + 1) is x_j.
# So: Sum_{j in neighbors(i)} x_j = d_i * (1 - x_i)
# Sum_{j in neighbors(i)} x_j = d_i - d_i * x_i
# Sum_{j in neighbors(i)} x_j + d_i * x_i = d_i
# This is a system of linear equations over GF(2).
# For each connected component, we can solve this system.
# Since we want to minimize sum b_i * 3^(n-i), we want to minimize x_i * 3^(n-i).
# This means we want x_i = 0 whenever possible, prioritizing smaller i.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # System: x_i * d_i + sum_{j in adj[i]} x_j = d_i (mod 2)
    # Let A be the adjacency matrix + diagonal matrix D.
    # A x = d (mod 2)
    
    x = [-1] * n
    visited = [False] * n
    
    for i in range(n):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Solve for this component
            # Map component nodes to 0..k-1
            comp_map = {node: idx for idx, node in enumerate(component)}
            k = len(component)
            matrix = [[0] * (k + 1) for _ in range(k)]
            for u in component:
                r = comp_map[u]
                matrix[r][r] = len(adj[u]) % 2
                for v in adj[u]:
                    matrix[r][comp_map[v]] = 1
                matrix[r][k] = len(adj[u]) % 2
            
            # Gaussian elimination
            pivot_row = 0
            pivot_col = [ -1 ] * k
            for c in range(k):
                sel = -1
                for r in range(pivot_row, k):
                    if matrix[r][c]:
                        sel = r
                        break
                if sel != -1:
                    matrix[pivot_row], matrix[sel] = matrix[sel], matrix[pivot_row]
                    pivot_col[c] = pivot_row
                    for r in range(k):
                        if r != pivot_row and matrix[r][c]:
                            for j in range(c, k + 1):
                                matrix[r][j] ^= matrix[pivot_row][j]
                    pivot_row += 1
            
            # Check consistency
            for r in range(pivot_row, k):
                if matrix[r][k]:
                    print("NO")
                    return
            
            # Back substitution: set free variables to 0 to minimize x_i
            res = [0] * k
            for c in range(k):
                if pivot_col[c] != -1:
                    res[c] = matrix[pivot_col[c]][k]
            
            for idx, node in enumerate(component):
                x[node] = res[idx]

    print("YES")
    print(*(val + 1 for val in x))

solve()
