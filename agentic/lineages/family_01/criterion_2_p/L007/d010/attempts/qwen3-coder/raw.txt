```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [[] for _ in range(n)]
    
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        index += 2
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # Precompute traces
    # S1 = [[0,1,0],[0,0,1],[1,0,0]]
    # S2 = [[0,0,1],[1,0,0],[0,1,0]]
    # S1*S1 = [[0,0,1],[1,0,0],[0,1,0]] = S2  -> Tr = 0+0+0 = 0
    # S2*S2 = [[0,1,0],[0,0,1],[1,0,0]] = S1  -> Tr = 0+0+0 = 0
    # S1*S2 = [[1,0,0],[0,1,0],[0,0,1]]       -> Tr = 3
    # S2*S1 = [[1,0,0],[0,1,0],[0,0,1]]       -> Tr = 3
    
    # So Tr(Si*Sj) = 3 iff i != j
    # For u=v: Tr(Si*Si) = 0 always, so condition Tr(M_u*M_u)=3 is never satisfied.
    # Therefore, no self-loops allowed.
    
    # Check for self-loops
    for u, v in edges:
        if u == v:
            print("NO")
            return
    
    # For each edge (u,v), we need Tr(M_u * M_v) = 3 => M_u != M_v => b_u != b_v
    # So this is a 2-coloring (bipartite) problem
    
    color = [-1] * n  # -1: unvisited, 0: color 1, 1: color 2
    
    # Check each connected component
    for start in range(n):
        if color[start] == -1:
            queue = deque()
            queue.append(start)
            color[start] = 0
            while queue:
                u = queue.popleft()
                c = color[u]
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        queue.append(v)
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, assign colors. We want to minimize the sequence.
    # color 0 -> basis 1, color 1 -> basis 2
    # To minimize, we should try to assign 1 (basis 1) to earlier nodes if possible.
    # But since we did BFS from each component's start node and assigned 0 first,
    # we need to decide for each component: whether to flip colors or not.
    # To minimize the sequence, for each component, if the first node (in order 0 to n-1)
    # in that component is colored 1, we should flip all colors in that component.
    
    # Find components
    visited = [False] * n
    components = []
    for i in range(n):
        if not visited[i]:
            comp = []
            queue = deque()
            queue.append(i)
            visited[i] = True
            comp.append(i)
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        queue.append(v)
                        comp.append(v)
            components.append(comp)
    
    # For each component, determine if we should flip
    final_color = color[:]
    for comp in components:
        # Get the first node in the component (smallest index)
        first_node = min(comp)
        # If it's colored 1, we should flip to make it 0 (which maps to basis 1)
        if color[first_node] == 1:
            for node in comp:
                final_color[node] = 1 - final_color[node]
    
    # Now assign basis: 0 -> 1, 1 -> 2
    basis = [1 if final_color[i] == 0 else 2 for i in range(n)]
    
    print("YES")
    print(' '.join(map(str, basis)))

if __name__ == "__main__":
    main()
```
