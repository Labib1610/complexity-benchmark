 ```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            dq = deque([i])
            while dq:
                u = dq.popleft()
                cu = color[u]
                for v in adj[u]:
                    cv = color[v]
                    if cv == -1:
                        color[v] = cu ^ 1
                        dq.append(v)
                    elif cv == cu:
                        sys.stdout.write("NO\n")
                        return

    out = ['YES', ' '.join('1' if c == 0 else '2' for c in color)]
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
```
