import sys
from collections import defaultdict, deque

def solve():
    import sys
    import threading
    
    def main():
        import sys
        sys.setrecursionlimit(1 << 25)
        
        n, m = map(int, sys.stdin.readline().split())
        
        if m == 0:
            print("YES")
            print(" ".join(["1"] * n))
            return
            
        # Build adjacency list
        adj = defaultdict(list)
        edges = []
        for _ in range(m):
            u, v = map(int, sys.stdin.readline().split())
            u -= 1
            v -= 1
            adj[u].append(v)
            adj[v].append(u)
            edges.append((u, v))
            
        # For each edge, we need to ensure the two endpoints don't simultaneously be in Ping or Ack
        # Let's think about what states each flag produces over time
        # Flag 1: Sync(0) -> Ping(1) -> Ack(2) -> Sync(0) ...
        # Flag 2: Sync(0) -> Ack(2) -> Ping(1) -> Sync(0) ...
        # So at time t:
        # Flag 1: t%3 == 1 -> Ping, t%3 == 2 -> Ack
        # Flag 2: t%3 == 1 -> Ack, t%3 == 2 -> Ping
        
        # For conflict-freeness:
        # At no time t >= 1, both nodes can be Ping or both Ack
        # This means for every time step t, it's not the case that:
        #   (f_u == 1 and t%3==1) and (f_v == 1 and t%3==1)  [both Ping]
        #   (f_u == 1 and t%3==2) and (f_v == 1 and t%3==2)  [both Ack]
        #   (f_u == 2 and t%3==1) and (f_v == 2 and t%3==1)  [both Ack]
        #   (f_u == 2 and t%3==2) and (f_v == 2 and t%3==2)  [both Ping]
        
        # This simplifies to checking that for each edge (u,v):
        # It's never true that f_u == f_v and (t%3 == 1 or t%3 == 2)
        # But since this must hold for ALL t >= 1, let's look at t=1,2,3,4,...
        # t=1: f_u=1 -> Ping, f_u=2 -> Ack
        # t=2: f_u=1 -> Ack,  f_u=2 -> Ping  
        # t=3: f_u=1 -> Ping, f_u=2 -> Ack
        # t=4: f_u=1 -> Ack,  f_u=2 -> Ping
        # Pattern repeats every 3 steps.
        
        # So conflicts happen when:
        # At times where both should be Ping: when f_u=f_v=1 and t%3=1, or f_u=f_v=2 and t%3=2
        # At times where both should be Ack: when f_u=f_v=1 and t%3=2, or f_u=f_v=2 and t%3=1
        
        # Wait, let me recheck the cycles:
        # Flag 1: Sync(0) -> Ping(1) -> Ack(2) -> Sync(0) ...
        # Time t: 0       -> 1        -> 2       -> 3 ...
        # State:  Sync      Ping       Ack        Sync ...
        # So f=1: t%3=1 -> Ping, t%3=2 -> Ack
        
        # Flag 2: Sync(0) -> Ack(2) -> Ping(1) -> Sync(0) ... 
        # Time t: 0       -> 1        -> 2       -> 3 ...
        # State:  Sync      Ack        Ping      Sync ...
        # So f=2: t%3=1 -> Ack, t%3=2 -> Ping
        
        # Now for an edge (u,v), they conflict if at some time t>=1:
        # Both are Ping: (f_u=1,t%3=1 and f_v=1,t%3=1) OR (f_u=2,t%3=2 and f_v=2,t%3=2) 
        # Both are Ack:  (f_u=1,t%3=2 and f_v=1,t%3=2) OR (f_u=2,t%3=1 and f_v=2,t%3=1)
        
        # Since this must hold for all t, and the pattern repeats every 3 steps,
        # we just need to check t%3 = 1 and t%3 = 2.
        
        # For t%3=1: f_u=1->Ping, f_v=1->Ping => conflict if f_u=f_v=1
        #             f_u=2->Ack,  f_v=2->Ack  => conflict if f_u=f_v=2
        # For t%3=2: f_u=1->Ack,  f_v=1->Ack  => conflict if f_u=f_v=1  
        #             f_u=2->Ping, f_v=2->Ping => conflict if f_u=f_v=2
         
        # So in both cases, if f_u == f_v, there's a conflict!
        # That means for every edge, f_u != f_v.
        
        # This is a 2-coloring problem! Each node must have a different color from its neighbors.
        
        # But wait, that's not quite right either. Let me reconsider...
        
        # Actually, let's think more carefully.
        # We want to avoid both being Ping at same time, and both being Ack at same time.
        
        # Let's enumerate all combinations for an edge (u,v):
        # Time t | f_u=1 state | f_v=1 state | f_u=2 state | f_v=2 state
        # t=1    | Ping         | Ping         | Ack          | Ack         -> conflict if f_u=f_v 
        # t=2    | Ack          | Ack          | Ping         | Ping        -> conflict if f_u=f_v
        # t=3    | Ping         | Ping         | Ack          | Ack         -> conflict if f_u=f_v
        # ...
        
        # So indeed, if f_u == f_v, then at times t%3=1 both are Ping/Ack (depending on flag),
        # and at times t%3=2 both are Ack/Ping. So they conflict.
        
        # Therefore, for every edge, we must have f_u != f_v.
        # This is exactly a bipartite graph check!
        
        # Check if the graph is bipartite
        color = [-1] * n  # -1 unvisited, 0 and 1 are colors
        
        def bfs(start):
            queue = deque()
            queue.append(start)
            color[start] = 0
            
            while queue:
                u = queue.popleft()
                c = color[u]
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        queue.append(v)
                    elif color[v] == c:
                        return False
            return True
                        
        for i in range(n):
            if color[i] == -1:
                if not bfs(i):
                    print("NO")
                    return
                    
        print("YES")
        # Convert colors to flags (1 and 2), preferring lexicographically smallest
        # That means we want node 0 to be 1 if possible
        # Since colors are 0 and 1, we map 0->1, 1->2
        result = []
        for i in range(n):
            result.append(str(color[i] + 1))
        print(" ".join(result))
        
    threading.Thread(target=main).start()

solve()
