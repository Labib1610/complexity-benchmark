import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Local resonance values:
    # c_u=1, c_v=1: 1^3 + 1^3 - (1*1)^2 = 1 + 1 - 1 = 1
    # c_u=1, c_v=2: 1^3 + 2^3 - (1*2)^2 = 1 + 8 - 4 = 5
    # c_u=2, c_v=1: 2^3 + 1^3 - (2*1)^2 = 8 + 1 - 4 = 5
    # c_u=2, c_v=2: 2^3 + 2^3 - (2*2)^2 = 8 + 8 - 16 = 0
    
    # Let x_i = 1 if c_i = 1, x_i = 0 if c_i = 2.
    # If c_u=1, c_v=1: val=1
    # If c_u=1, c_v=2: val=5
    # If c_u=2, c_v=1: val=5
    # If c_u=2, c_v=2: val=0
    # We want sum of values = 5m.
    # Let n1 be number of edges with (1,1), n5 be number of edges with (1,2) or (2,1), n0 be number of edges with (2,2).
    # n1 + n5 + n0 = m
    # 1*n1 + 5*n5 + 0*n0 = 5m
    # n1 + 5n5 = 5(n1 + n5 + n0) => n1 + 5n5 = 5n1 + 5n5 + 5n0 => 4n1 + 5n0 = 0
    # Since n1, n0 >= 0, this implies n1 = 0 and n0 = 0.
    # Thus, all edges must be (1,2) or (2,1).
    # This means the graph must be bipartite.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, we have two sets of nodes (color 0 and color 1).
    # We can assign c=1 to color 0 and c=2 to color 1, or vice versa.
    # To get lexicographically smallest, we check both assignments.
    
    def get_assignment(c0_val):
        res = [0] * (n + 1)
        for i in range(1, n + 1):
            res[i] = c0_val if color[i] == 0 else (3 - c0_val)
        return res[1:]

    a1 = get_assignment(1)
    a2 = get_assignment(2)
    
    if a1 < a2:
        print("YES")
        print(*(a1))
    else:
        print("YES")
        print(*(a2))

solve()
