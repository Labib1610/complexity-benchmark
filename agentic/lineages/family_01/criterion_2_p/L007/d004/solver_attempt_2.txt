import sys

# The problem states that for any two connected dampeners with amplitudes a and b,
# |a^2 - b^2| = 3.
# Since a, b are in {1, 2}, a^2, b^2 are in {1, 4}.
# |1 - 4| = 3.
# This means for every edge (u, v), one dampener must have amplitude 1 and the other 2.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we want to assign 1 to as many
# early indices as possible.
# For each connected component, we have two choices:
# 1. Assign 1 to the first node (in index order) of the component.
# 2. Assign 2 to the first node (in index order) of the component.
# We compare the resulting sequences for each component and pick the one that
# is lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # Option 1: color[i] = 1
            # Option 2: color[i] = 2 (which means flipping all colors)
            
            # To be lexicographically smallest, we want the smallest value at the 
            # smallest index in the component.
            # Let's check the value at the smallest index 'i' in this component.
            # If we set color[i] = 1, the sequence is fixed.
            # If we set color[i] = 2, the sequence is fixed.
            
            # Actually, for each component, we just need to decide whether to 
            # keep the current coloring or flip it.
            # We want the smallest value at the smallest index 'i'.
            # If color[i] is 1, we keep it. If color[i] is 2, we flip it to 1.
            
            if color[i] == 2:
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
