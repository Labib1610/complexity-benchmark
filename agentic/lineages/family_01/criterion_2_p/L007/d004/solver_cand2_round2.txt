```python
import sys

# The problem states that a link between u and v is stable if the combined
# effect of shifts f_u and f_v restores the sequence (x1, x2, x3).
# A 1-step shift is (x3, x1, x2).
# A 2-step shift is (x2, x3, x1).
# Let S1 be the 1-step shift and S2 be the 2-step shift.
# S1(S1(x)) = S2(x)
# S1(S2(x)) = S2(S1(x)) = (x1, x2, x3) (Identity)
# S2(S2(x)) = S1(x)
#
# For the combined effect to be the identity:
# If f_u = 1 and f_v = 1: S1(S1(x)) = S2(x) != Identity
# If f_u = 1 and f_v = 2: S1(S2(x)) = Identity (Valid)
# If f_u = 2 and f_v = 1: S2(S1(x)) = Identity (Valid)
# If f_u = 2 and f_v = 2: S2(S2(x)) = S1(x) != Identity
#
# Thus, for every link (u, v), we must have f_u != f_v.
# This is equivalent to checking if the graph is bipartite.
# We need the lexicographically smallest assignment, so we prefer 1 over 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
