import sys

# The resonance formula R(cu, cv) = 3(cu + cv) - 2*cu*cv
# If cu=1, cv=1: R = 3(2) - 2(1) = 4
# If cu=1, cv=2: R = 3(3) - 2(2) = 5
# If cu=2, cv=1: R = 3(3) - 2(2) = 5
# If cu=2, cv=2: R = 3(4) - 2(4) = 4
# We want sum(R) = 5m.
# Let x be the number of edges with (1, 2) or (2, 1) and y be the number of edges with (1, 1) or (2, 2).
# x + y = m
# 5x + 4y = 5m
# 5x + 4(m - x) = 5m => x = m.
# This means every edge must connect a 1 and a 2.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we want 1s at the beginning.
# For each connected component, we have two choices of coloring (swap 1s and 2s).
# We choose the coloring that puts a 1 at the smallest index in the component.

    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
