```python
import sys

# The condition is integral_{-1}^{1} S_u(x) S_v(x) dx < 0.
# If b_i = 1, S_i(x) = x + 1.
# If b_i = 2, S_i(x) = x - 1.
# Let's calculate the integrals:
# Case 1: b_u = 1, b_v = 1: integral (x+1)(x+1) dx = integral (x^2 + 2x + 1) dx = [x^3/3 + x^2 + x] from -1 to 1 = (1/3 + 1 + 1) - (-1/3 + 1 - 1) = 7/3 - (-1/3) = 8/3 > 0.
# Case 2: b_u = 2, b_v = 2: integral (x-1)(x-1) dx = integral (x^2 - 2x + 1) dx = [x^3/3 - x^2 + x] from -1 to 1 = (1/3 - 1 + 1) - (-1/3 - 1 - 1) = 1/3 - (-7/3) = 8/3 > 0.
# Case 3: b_u = 1, b_v = 2: integral (x+1)(x-1) dx = integral (x^2 - 1) dx = [x^3/3 - x] from -1 to 1 = (1/3 - 1) - (-1/3 + 1) = -2/3 - 2/3 = -4/3 < 0.
# Case 4: b_u = 2, b_v = 1: integral (x-1)(x+1) dx = -4/3 < 0.
#
# Thus, the condition is satisfied if and only if b_u != b_v.
# This is a bipartite graph coloring problem.
# We need to color the graph with 2 colors (1 and 2) such that no two adjacent nodes have the same color.
# To get the lexicographically smallest assignment, we iterate through nodes 1 to n.
# If a node is not colored, we assign it color 1 (the smaller value) and perform BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
