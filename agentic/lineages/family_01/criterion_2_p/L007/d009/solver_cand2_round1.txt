import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Stability condition:
    # P1(x) = x^2 - 1
    # P2(x) = -x^2 + x + 1
    # S(u,v) = P_u(x) + P_v(x)
    # If both 1: 2x^2 - 2 = 0 => x^2 = 1 => x = +/- 1 (2 solutions)
    # If both 2: -2x^2 + 2x + 2 = 0 => x^2 - x - 1 = 0 => D = 1 - 4(-1) = 5 (2 solutions)
    # If one 1, one 2: (x^2 - 1) + (-x^2 + x + 1) = x = 0 (1 solution)
    # Thus, for each link (u, v), u and v must have different bases.
    # This is a bipartite graph coloring problem.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # To get lexicographically smallest, for each component,
            # we check if setting the smallest index node to 1 is valid.
            # Actually, the bipartite coloring is fixed once the first node is set.
            # We just need to ensure the first node of the component gets the smallest possible color.
            # Since we iterate 1 to n, the first node encountered in a component is the smallest.
            # We already set it to 1, which is the smallest possible basis.
            
    print("YES")
    print(*(color[1:]))

solve()
