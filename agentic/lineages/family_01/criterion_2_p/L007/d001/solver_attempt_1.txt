import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color array: 0 = unvisited, 1 = type 1, 2 = type 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest configuration,
    # we iterate from 1 to n. If an employee is not colored,
    # we start a BFS/DFS. To ensure lexicographical smallest,
    # we prefer assigning 1 whenever possible.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                
                # Sort neighbors to process in order, though for 
                # bipartiteness, the order doesn't change feasibility.
                # To get lexicographically smallest, we need to be careful.
                # Actually, for each connected component, we have two choices:
                # start with 1 or start with 2. We pick the one that makes
                # the sequence lexicographically smallest.
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
