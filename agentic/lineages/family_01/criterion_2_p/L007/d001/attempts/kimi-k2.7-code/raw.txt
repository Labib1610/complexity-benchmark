 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    ok = True

    for i in range(1, n + 1):
        if color[i] != 0:
            continue
        color[i] = 1
        stack = [i]
        while stack:
            u = stack.pop()
            want = 3 - color[u]
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = want
                    stack.append(v)
                elif color[v] != want:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            break

    if not ok:
        sys.stdout.write("NO")
    else:
        out = ["YES", ' '.join(str(color[i]) for i in range(1, n + 1))]
        sys.stdout.write('\n'.join(out) + '\n')

if __name__ == "__main__":
    solve()
```
