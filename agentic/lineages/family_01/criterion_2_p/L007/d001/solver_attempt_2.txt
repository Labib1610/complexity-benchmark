import sys

# The problem states that for every linkage (u, v), s[u] + s[v] = 3.
# Since s[i] is either 1 or 2, this means s[u] != s[v].
# This is equivalent to checking if the graph is bipartite.
# If it is bipartite, we want the lexicographically smallest sequence.
# For each connected component, we have two choices:
# 1. Assign the first node in the component (smallest index) to 1.
# 2. Assign the first node in the component (smallest index) to 2.
# To get the lexicographically smallest sequence, for each component,
# we check which assignment makes the smallest index in that component 1.
# Since we want the overall sequence to be lexicographically smallest,
# we should always assign 1 to the smallest index node in each component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
