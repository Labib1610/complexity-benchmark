```python
import sys

# The local resonance value for a dampener connecting pillars with grades c_u and c_v:
# If c_u = 1, c_v = 1: (1%1) + (1%1) = 0 + 0 = 0
# If c_u = 1, c_v = 2: (1%2) + (2%1) = 1 + 0 = 1
# If c_u = 2, c_v = 1: (2%1) + (1%2) = 0 + 1 = 1
# If c_u = 2, c_v = 2: (2%2) + (2%2) = 0 + 0 = 0
#
# The sum of resonance values must be exactly m.
# Let k be the number of dampeners connecting a 1 and a 2.
# The sum is 0 * (m - k) + 1 * k = k.
# So we need k = m.
# This means every dampener must connect a 1 and a 2.
# This is equivalent to saying the graph must be bipartite.
# We want the lexicographically smallest sequence (c_1, ..., c_n).
# This means we prefer 1 over 2 for each pillar.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

solve()
```
