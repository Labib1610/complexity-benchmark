```python
import sys

# The condition E_u(t) + E_v(t) != 2 for all t >= 1:
# E_i(t) = (t * f_i) % 3
# Let f_i, f_j in {1, 2}.
# t=1: (f_u % 3) + (f_v % 3) != 2
# t=2: (2*f_u % 3) + (2*f_v % 3) != 2
#
# Case 1: f_u = 1, f_v = 1
# t=1: 1+1 = 2 (Conflict!)
# Case 2: f_u = 2, f_v = 2
# t=1: 2+2 = 4 % 3 = 1 (OK)
# t=2: 4+4 = 8 % 3 = 2 (Conflict!)
# Case 3: f_u = 1, f_v = 2
# t=1: 1+2 = 3 % 3 = 0 (OK)
# t=2: 2+4 = 6 % 3 = 0 (OK)
#
# So, f_u = 1 and f_v = 1 is forbidden.
# f_u = 2 and f_v = 2 is forbidden.
# Thus, f_u must be different from f_v.
# This is a standard 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
