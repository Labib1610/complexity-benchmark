import sys

# The problem states:
# For every link (u, v):
# 1. Not (f_u == 1 and f_v == 1)
# 2. Not (f_u == 2 and f_v == 2)
# This implies f_u != f_v.
# If f_u != f_v, then for every edge (u, v), u and v must have different flags.
# This is exactly the definition of a bipartite graph.
# If the graph is bipartite, we can color it with 2 colors (1 and 2).
# To get the lexicographically smallest assignment:
# For each connected component, we have two choices:
# Choice A: Assign the smallest possible flag to the first node encountered in the component.
# Choice B: Swap all 1s and 2s in that component.
# We want to minimize f_1, then f_2, ..., f_n.
# This is equivalent to: for each component, pick the coloring that makes the 
# smallest index node in that component have flag 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # To make it lexicographically smallest:
            # Find the smallest index node in this component.
            # We want that node to have flag 1.
            min_node = min(component)
            if color[min_node] == 2:
                # Flip colors for this component
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")

solve()
