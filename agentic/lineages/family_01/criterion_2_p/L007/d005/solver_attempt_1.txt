```python
import sys

# The problem states that for each linkage (u, v), the sum of the squares of 
# their amplitudes must be 5.
# Let a_u and a_v be the amplitudes (1 or 2).
# a_u^2 + a_v^2 = 5.
# Since a_i is 1 or 2, a_i^2 is 1 or 4.
# The only way to get 1 + 4 = 5 is if one amplitude is 1 and the other is 2.
# This means the graph must be bipartite, and for every edge (u, v), 
# a_u != a_v.
# We want the lexicographically smallest sequence (a_1, a_2, ..., a_n).
# This means we prefer 1 over 2 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1) # 0: unvisited, 1: amplitude 1, 2: amplitude 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try setting color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
