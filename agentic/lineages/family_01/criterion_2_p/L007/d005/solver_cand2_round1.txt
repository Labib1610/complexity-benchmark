```python
import sys

# The equation is: V_u + d_u * c_u = 3 * d_u
# V_u = sum_{v in neighbors(u)} c_v
# So, sum_{v in neighbors(u)} c_v + d_u * c_u = 3 * d_u
# Let x_u = c_u - 1. Since c_u in {1, 2}, x_u in {0, 1}.
# c_u = x_u + 1
# sum_{v in neighbors(u)} (x_v + 1) + d_u * (x_u + 1) = 3 * d_u
# sum_{v in neighbors(u)} x_v + d_u + d_u * x_u + d_u = 3 * d_u
# sum_{v in neighbors(u)} x_v + d_u * x_u = d_u
# This is a system of linear equations over the integers.
# Since x_u are 0 or 1, this is a constraint satisfaction problem.
# For each connected component, if it's bipartite, we can solve it.
# If not bipartite, it might have no solution.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # x_u + sum_{v in N(u)} x_v = d_u
    # This is equivalent to: x_u + sum_{v in N(u)} x_v = degree(u)
    # Let's check components.
    visited = [False] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Try setting x_i = 0 then x_i = 1 to find lexicographically smallest
            found = False
            for start_val in [0, 1]:
                temp_x = {node: -1 for node in component}
                temp_x[i] = start_val
                q = [i]
                possible = True
                head = 0
                while head < len(q):
                    u = q[head]
                    head += 1
                    # Equation: x_u + sum_{v in N(u)} x_v = d_u
                    # sum_{v in N(u)} x_v = d_u - x_u
                    # If all but one neighbor are known, we can determine the last one.
                    unknowns = [v for v in adj[u] if temp_x[v] == -1]
                    known_sum = sum(temp_x[v] for v in adj[u] if temp_x[v] != -1)
                    
                    if len(unknowns) == 0:
                        if temp_x[u] + known_sum != len(adj[u]):
                            possible = False; break
                    elif len(unknowns) == 1:
                        v = unknowns[0]
                        val = len(adj[u]) - temp_x[u] - known_sum
                        if val not in [0, 1]:
                            possible = False; break
                        temp_x[v] = val
                        q.append(v)
                
                if possible:
                    # Check all nodes in component
                    for u in component:
                        if temp_x[u] == -1:
                            # If still unknown, try 0 then 1
                            # This happens in cycles.
                            pass 
                    if all(temp_x[u] != -1 for u in component):
                        if all(temp_x[u] + sum(temp_x[v] for v in adj[u]) == len(adj[u]) for u in component):
                            for u in component: ans[u] = temp_x[u] + 1
                            found = True
                            break
            if not found:
                print("NO")
                return

    print("YES")
    print(*(ans[1:]))

solve()
```
