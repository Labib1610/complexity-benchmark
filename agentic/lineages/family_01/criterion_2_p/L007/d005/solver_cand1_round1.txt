import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    deg = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1
        deg[v] += 1
        idx += 2
        
    # The expression is 3 * sum(deg[i] * c[i]) - sum_{u,v in edges} (c[u] - c[v])^2
    # Let c[i] = 1 + x[i], where x[i] in {0, 1}.
    # 3 * sum(deg[i] * (1 + x[i])) - sum_{u,v} (x[u] - x[v])^2 = 5m
    # 3 * sum(deg[i]) + 3 * sum(deg[i] * x[i]) - sum_{u,v} (x[u] - x[v])^2 = 5m
    # Since sum(deg[i]) = 2m:
    # 6m + 3 * sum(deg[i] * x[i]) - sum_{u,v} (x[u] - x[v])^2 = 5m
    # sum_{u,v} (x[u] - x[v])^2 - 3 * sum(deg[i] * x[i]) = m
    
    # Let x[i] be 0 or 1.
    # If x[u] != x[v], (x[u] - x[v])^2 = 1.
    # Let E_diff be the set of edges where x[u] != x[v].
    # |E_diff| - 3 * sum(deg[i] * x[i]) = m
    
    # This is equivalent to:
    # sum_{i=1}^n x[i] * (3 * deg[i] - (deg[i] - count_neighbors_with_different_x)) = m
    # Actually, simpler:
    # sum_{u,v} (x[u] - x[v])^2 = sum_{u,v} (x[u] + x[v] - 2*x[u]*x[v])
    # This is a standard min-cut / max-flow or 2-SAT problem.
    # Let x[i] = 0 for all i. Expression = 0. We need it to be m.
    # Each x[i] = 1 changes the expression by:
    # 3 * deg[i] - 2 * (number of neighbors j where x[j] == 0)
    
    # Actually, the expression is:
    # sum_{i=1}^n (3 * deg[i] * c[i]) - sum_{i=1}^n deg[i] * c[i]^2 + 2 * sum_{u<v, edge} c[u]*c[v]
    # This simplifies to:
    # sum_{i=1}^n deg[i] * (3*c[i] - c[i]^2) + 2 * sum_{u<v, edge} c[u]*c[v] = 5m
    # If c[i] = 1: 3*1 - 1^2 = 2. If c[i] = 2: 3*2 - 2^2 = 2.
    # So 2 * sum(deg[i]) + 2 * sum_{u<v, edge} c[u]*c[v] = 5m
    # 4m + 2 * sum_{u<v, edge} c[u]*c[v] = 5m
    # 2 * sum_{u<v, edge} c[u]*c[v] = m
    # sum_{u<v, edge} c[u]*c[v] = m/2
    
    if m % 2 != 0:
        print("NO")
        return
    
    target = m // 2
    # We need sum_{u<v, edge} c[u]*c[v] = target, where c[i] in {1, 2}
    # Let c[i] = 1 + x[i], x[i] in {0, 1}
    # sum_{u<v, edge} (1 + x[u])(1 + x[v]) = target
    # sum_{u<v, edge} (1 + x[u] + x[v] + x[u]*x[v]) = target
    # m + sum_{u<v, edge} (x[u] + x[v]) + sum_{u<v, edge} x[u]*x[v] = target
    # sum_{i=1}^n deg[i]*x[i] + sum_{u<v, edge} x[u]*x[v] = target - m
    
    # If we set all x[i] = 0, sum = 0. If target - m < 0, impossible.
    # Since target = m/2, target - m = -m/2.
    # This implies we need sum_{u<v, edge} c[u]*c[v] = m/2.
    # Since c[i] >= 1, c[u]*c[v] >= 1.
    # sum_{u<v, edge} c[u]*c[v] >= m.
    # So m/2 >= m, which means m=0.
    # If m=0, any c works. Lexicographically smallest: all 1s.
    
    if m == 0:
        print("YES")
        print(*( [1] * n ))
        return
    
    print("NO")

solve()
