import sys

# The condition is: there is no t >= 1 such that D_u(t) = 998244352 and D_v(t) = 998244352.
# D_i(t) = (t * f_i) % 998244353.
# Let P = 998244353 (a prime).
# D_i(t) = P - 1 iff t * f_i = -1 (mod P).
# If f_i = 1, t = -1 = P - 1 (mod P).
# If f_i = 2, t = -1 * inv(2) = (P - 1) / 2 = 499122176 (mod P).
#
# If f_u = 1 and f_v = 1:
#   Both are P-1 at t = P-1. Conflict exists.
# If f_u = 2 and f_v = 2:
#   Both are P-1 at t = (P-1)/2. Conflict exists.
# If f_u = 1 and f_v = 2:
#   u is P-1 at t = P-1, 2(P-1), ...
#   v is P-1 at t = (P-1)/2, (P-1)/2 + (P-1), ...
#   These sets of t are disjoint. No conflict.
#
# Thus, the condition is equivalent to: for every edge (u, v), f_u != f_v.
# This is exactly the definition of a bipartite graph.
# We need to check if the graph is bipartite and find the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for each node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a component is not colored, we try assigning 1 to the first node.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

solve()
