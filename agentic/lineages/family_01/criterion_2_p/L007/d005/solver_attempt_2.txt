import sys

# The condition is that for every edge (u, v), key[u] ^ key[v] == 3.
# The possible keys are 1 (binary 01) and 2 (binary 10).
# 1 ^ 2 = 3.
# This means for every edge (u, v), key[u] must be different from key[v].
# If key[u] = 1, then key[v] must be 2.
# If key[u] = 2, then key[v] must be 1.
# This is equivalent to saying the graph must be bipartite.
# To get the lexicographically smallest sequence, we want key[1] to be 1 if possible.
# If key[1] = 1, then all nodes in its connected component are determined.
# If the graph is bipartite, we can color each component with 1s and 2s.
# To make the sequence lexicographically smallest, for each component,
# we check if setting the smallest node index in that component to 1
# results in a valid assignment.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we want the smallest index
            # in the component to be 1.
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if this component's assignment is optimal.
            # We want the smallest index in this component to have color 1.
            min_node = min(component)
            if color[min_node] == 2:
                # Flip colors for this component
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
