```python
import sys

# The condition "sum of phase state identifiers must be 3" means:
# If unit u is 1, unit v must be 2.
# If unit u is 2, unit v must be 1.
# This is equivalent to saying u and v must have different states.
# This is a standard bipartite graph coloring problem.
# We want the lexicographically smallest assignment (1, 2, 1, 2...).
# For each connected component, we can pick the state of the smallest index node.
# To make the sequence lexicographically smallest, for each component,
# we should try to assign 1 to the smallest index node in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign color 1 to i
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
