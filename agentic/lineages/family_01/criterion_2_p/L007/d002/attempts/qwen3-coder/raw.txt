```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build graph
    graph = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        graph[u].append(v)
        graph[v].append(u)
        edges.append((u, v))
    
    # For each edge, the product of the two nodes must be 2
    # This means one node must be 1 and the other must be 2
    # This is equivalent to a 2-coloring problem where:
    # - If node u has value 1, then all its neighbors must have value 2
    # - If node u has value 2, then all its neighbors must have value 1
    
    # Color array: 0 = unvisited, 1 = color 1, 2 = color 2
    color = [0] * n
    visited = [False] * n
    
    # Check if graph is bipartite
    for i in range(n):
        if not visited[i]:
            queue = deque([i])
            color[i] = 1
            visited[i] = True
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                next_color = 3 - current_color  # 1->2, 2->1
                
                for neighbor in graph[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        color[neighbor] = next_color
                        queue.append(neighbor)
                    elif color[neighbor] != next_color:
                        print("NO")
                        return
    
    # Verify the solution
    for u, v in edges:
        if color[u] * color[v] != 2:
            print("NO")
            return
    
    # Convert colors to values (1 and 2)
    result = []
    for i in range(n):
        if color[i] == 0:  # Isolated vertex, assign 1
            result.append("1")
        else:
            result.append(str(color[i]))
    
    print("YES")
    print(" ".join(result))

solve()
```
