```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # The stability condition f_u^{f_v} + f_v^{f_u} = 3
    # Let's evaluate all possibilities:
    # If f_u=1, f_v=1: 1^1 + 1^1 = 2 ≠ 3
    # If f_u=1, f_v=2: 1^2 + 2^1 = 1 + 2 = 3 ✓
    # If f_u=2, f_v=1: 2^1 + 1^2 = 2 + 1 = 3 ✓
    # If f_u=2, f_v=2: 2^2 + 2^2 = 4 + 4 = 8 ≠ 3
    #
    # So, two nodes connected by a critical link must have different values.
    # This is equivalent to checking if the graph is bipartite.
    
    color = [0] * (n + 1)  # 0: unvisited, 1: color A, 2: color B
    visited = [False] * (n + 1)
    
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 1
        visited[start] = True
        
        while queue:
            u = queue.popleft()
            current_color = color[u]
            next_color = 3 - current_color  # flips 1↔2
            
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    color[v] = next_color
                    queue.append(v)
                elif color[v] != next_color:
                    return False
        return True
    
    # Check all components
    for i in range(1, n + 1):
        if not visited[i] and adj[i]:  # Only start BFS if node has neighbors
            if not bfs(i):
                print("NO")
                return
    
    # If we reach here, a valid assignment exists.
    # Assign 1 to unvisited nodes (isolated nodes)
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
    
    print("YES")
    print(" ".join(str(color[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    main()
```
