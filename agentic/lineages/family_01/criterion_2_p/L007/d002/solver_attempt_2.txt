import sys

# The problem asks for a configuration of multipliers (1 or 2) such that for each
# correlator (u, v), the product of multipliers is 2.
# Let x_i be the multiplier for oscillator i.
# x_i * x_j = 2.
# Since x_i can only be 1 or 2, the only way for the product to be 2 is:
# (x_i = 1 and x_j = 2) or (x_i = 2 and x_j = 1).
# This is a bipartite coloring problem.
# If we assign 1 to one set of the bipartition and 2 to the other, we satisfy
# the condition. If the graph is not bipartite, no solution exists.
# To get the lexicographically smallest sequence, we want x_i = 1 whenever possible.
# In each connected component, we have two choices for coloring (1, 2) or (2, 1).
# We choose the one that results in a smaller value for the first node in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            idx_stack = 0
            while idx_stack < len(component):
                u = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if we should flip colors to make the first node 1
            # The first node in the component is 'i'.
            # If color[i] is 2, we flip all colors in this component.
            if color[i] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
