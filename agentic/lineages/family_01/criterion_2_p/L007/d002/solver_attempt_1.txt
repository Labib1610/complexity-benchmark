```python
import sys

# The condition "average of voltage levels is 1.5" means (v1 + v2) / 2 = 1.5,
# which implies v1 + v2 = 3. Since v1, v2 are in {1, 2}, this means
# one must be 1 and the other must be 2.
# This is equivalent to saying the graph must be bipartite, and for each
# connected component, once one node is assigned, all others are fixed.
# To get the lexicographically smallest sequence, for each component,
# we try assigning 1 to the smallest index node and see if it's valid.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want the smallest index
            # in the component to be 1 if possible.
            # Actually, for each component, we have two choices:
            # Choice A: node i is 1, neighbors are 2, etc.
            # Choice B: node i is 2, neighbors are 1, etc.
            # We compare the resulting sequences for the component.
            
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if assigning 1 to the smallest index in component is better
            # The smallest index in this component is i.
            # If color[i] == 1, sequence is fixed. If color[i] == 2, we can flip.
            if color[i] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
```
