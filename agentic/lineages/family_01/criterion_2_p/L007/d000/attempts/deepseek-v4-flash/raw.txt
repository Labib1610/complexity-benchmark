 We can solve this by checking if the graph is bipartite.  
If it is, we assign shift 1 to one color and shift 2 to the other.  
To get the lexicographically smallest assignment, we try to assign 1 whenever possible during BFS/DFS.

Here’s the code:

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
graph = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    graph[u-1].append(v-1)
    graph[v-1].append(u-1)

color = [-1] * n  # -1 = uncolored, 0 = shift 1, 1 = shift 2

def bfs(start):
    q = deque([start])
    color[start] = 0
    while q:
        u = q.popleft()
        for v in graph[u]:
            if color[v] == -1:
                color[v] = 1 - color[u]
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

possible = True
for i in range(n):
    if color[i] == -1:
        if not bfs(i):
            possible = False
            break

if not possible:
    print("NO")
else:
    print("YES")
    print(" ".join(str(c + 1) for c in color))
```
