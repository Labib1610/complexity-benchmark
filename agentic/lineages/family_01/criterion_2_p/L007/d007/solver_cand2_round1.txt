```python
import sys

# The condition for a link (u, v) to be successful:
# M_1 = [[1, 1], [1, 1]], M_2 = [[1, -1], [-1, 1]]
# If f_u = 1, f_v = 1: M_1 + M_1 = [[2, 2], [2, 2]], det = 0
# If f_u = 2, f_v = 2: M_2 + M_2 = [[2, -2], [-2, 2]], det = 0
# If f_u = 1, f_v = 2: M_1 + M_2 = [[2, 0], [0, 2]], det = 4 != 0
# If f_u = 2, f_v = 1: M_2 + M_1 = [[2, 0], [0, 2]], det = 4 != 0
# Thus, f_u must be different from f_v. This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is not colored, we start a BFS/DFS.
    # To minimize f_i, we prefer color 1 over color 2.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
