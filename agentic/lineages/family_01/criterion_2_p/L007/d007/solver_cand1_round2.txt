import sys

# The problem states:
# State 1: R
# State 2: I
# Handshake condition: u and v must result in R o I.
# If u=1 (R) and v=1 (R), result is R o R = Id.
# If u=2 (I) and v=2 (I), result is I o I = Id.
# If u=1 (R) and v=2 (I), result is R o I.
# If u=2 (I) and v=1 (R), result is I o R = R o I.
# Thus, for every link (u, v), f_u must be different from f_v.
# This is a bipartite graph coloring problem.
# We need the lexicographically smallest assignment.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is not colored, we try assigning 1, then 2.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            stack = [i]
            color[i] = 1
            component = []
            
            # Use a stack to traverse the component
            # To ensure lexicographical order, we must be careful.
            # Actually, for a bipartite graph, once the first node of a 
            # component is fixed, the rest are determined.
            # To minimize lexicographically, we want color[1] to be 1,
            # then color[2] to be 1 if possible, etc.
            
            # Standard bipartite check:
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
