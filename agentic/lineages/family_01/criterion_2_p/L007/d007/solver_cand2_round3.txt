```python
import sys

# The condition is that for an entanglement link (u, v), O_u and O_v must anti-commute.
# Pauli-X and Pauli-Z anti-commute (XZ + ZX = 0).
# Pauli-X and Pauli-X commute (XX + XX = 2I != 0).
# Pauli-Z and Pauli-Z commute (ZZ + ZZ = 2I != 0).
# Thus, for every link (u, v), sensor u and sensor v must have DIFFERENT bases.
# This is a bipartite graph coloring problem.
# We need to color the graph with 2 colors (1 and 2) such that no two adjacent
# nodes have the same color.
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If a node is not colored, we start a BFS/DFS from it.
# To make it lexicographically smallest, when we start a component, we assign
# the smallest possible color (1) to the smallest index node in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To ensure lexicographical smallest, we assign color 1 to the current node i
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
