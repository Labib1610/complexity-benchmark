We can model this as a graph problem.  
The condition that two observables anti-commute means that if we assign basis 1 (X) or basis 2 (Z), then two connected sensors must have different bases.  
That is, the graph must be bipartite.  

We can solve it with BFS/DFS coloring, and for lexicographically smallest assignment, we try to assign 1 whenever possible, starting from sensor 1.

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n

for start in range(n):
    if color[start] != -1:
        continue
    color[start] = 0
    q = deque([start])
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = color[u] ^ 1
                q.append(v)
            elif color[v] == color[u]:
                print("NO")
                sys.exit()

# Now we have a valid bipartition, but we need lexicographically smallest.
# We can flip colors in each connected component independently.
# For each component, we check which assignment (0 or 1 for the first node) gives smaller lexicographic order.

visited = [False] * n
assignment = [0] * n

for start in range(n):
    if visited[start]:
        continue
    # BFS to collect component
    comp = []
    q = deque([start])
    visited[start] = True
    while q:
        u = q.popleft()
        comp.append(u)
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                q.append(v)
    # Try both colorings for this component
    # coloring A: keep color as is (0 for start)
    # coloring B: flip (1 for start)
    # We'll compare lexicographically
    # Build candidate arrays
    candA = [0] * n
    candB = [0] * n
    for u in comp:
        candA[u] = color[u] + 1  # 1 or 2
        candB[u] = (color[u] ^ 1) + 1
    # Choose smaller lexicographically
    # Compare only within this component's positions
    for u in sorted(comp):
        if candA[u] < candB[u]:
            for v in comp:
                assignment[v] = candA[v]
            break
        elif candB[u] < candA[u]:
            for v in comp:
                assignment[v] = candB[v]
            break
    else:
        # equal, pick A
        for v in comp:
            assignment[v] = candA[v]

print("YES")
print(" ".join(map(str, assignment)))
```
