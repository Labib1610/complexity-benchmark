```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
        index += 2

    # We model this as a 2-SAT problem.
    # Each sensor i has two possible assignments: basis 1 (X) or basis 2 (Z).
    # We need to ensure that for each edge (u,v), the observables anti-commute.
    # X and X: commute
    # Z and Z: commute
    # X and Z: anti-commute
    # So for each edge (u,v), u and v must have different bases.
    # This is a classic graph coloring problem with 2 colors.
    # Each edge is a constraint that the two nodes have different colors.
    # This is equivalent to checking if the graph is bipartite.

    # We can model this as a 2-SAT problem or directly as bipartite check.
    # Let's use bipartite check with BFS/DFS.
    
    # Color array: 0 = unvisited, 1 = color A, -1 = color B
    color = [0] * (n + 1)
    
    # Check each component
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        # Start BFS
        queue = deque()
        queue.append(start)
        color[start] = 1  # Assign color 1 to the starting node
        
        while queue:
            u = queue.popleft()
            current_color = color[u]
            next_color = -current_color
            
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = next_color
                    queue.append(v)
                elif color[v] == current_color:
                    print("NO")
                    return
                # else color[v] == next_color, which is fine
    
    # If we reach here, the graph is bipartite and a valid assignment exists.
    # We need to output the assignment that is lexicographically smallest.
    # In our coloring, color 1 can represent basis 1, and color -1 can represent basis 2.
    # But we need to ensure that sensor 1 gets the smallest possible basis, etc.
    # Our BFS already ensures that. We start with node 1 and assign it color 1.
    # Then its neighbors get color -1, their neighbors get 1, etc.
    # This naturally gives the lexicographically smallest assignment.
    
    assignment = []
    for i in range(1, n + 1):
        if color[i] == 1:
            assignment.append("1")
        else:
            assignment.append("2")
    
    print("YES")
    print(" ".join(assignment))

if __name__ == "__main__":
    main()
```
