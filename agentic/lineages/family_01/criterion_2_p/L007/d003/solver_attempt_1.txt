import sys

# The problem states: for each correlator (u, v), one oscillator must be 
# double the other. Since phase modes are 1 or 2, this means one must be 1 
# and the other must be 2.
# This is equivalent to a 2-coloring problem on a graph where each edge 
# (u, v) implies that u and v must have different colors (1 and 2).
# If the graph is bipartite, a valid configuration exists.
# To get the lexicographically smallest sequence, we want to assign 1 to 
# as many early indices as possible.
# We iterate through each connected component. For each component, we try 
# assigning 1 to the smallest index in that component and propagate the 
# constraints. If we find a contradiction, the graph is not bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we prefer color 1 for the 
            # smallest node in the component.
            stack = [(i, 1)]
            colors[i] = 1
            component = []
            
            # Collect component nodes
            q = [i]
            visited = {i}
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if v not in visited:
                        visited.add(v)
                        q.append(v)
            
            # Try coloring starting with 1 for the smallest node
            # Since it's a bipartite graph, once the first node is fixed, 
            # all others are determined.
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
