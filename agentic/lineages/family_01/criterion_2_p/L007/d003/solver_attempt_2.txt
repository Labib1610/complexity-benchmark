```python
import sys

# The problem states: for two vehicles sharing a bay, the max of their tiers
# must be twice the min. Since tiers are 1 or 2, the only possible pair is (1, 2).
# This means for every edge (u, v), u and v must have different tiers.
# This is a standard 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest sequence, we greedily assign 1 to the
# smallest available node in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is uncolored, we start a BFS/DFS.
    # To make the sequence lexicographically smallest, we prefer 1 over 2.
    # For each component, we try assigning 1 to the smallest node.
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS for this component
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
