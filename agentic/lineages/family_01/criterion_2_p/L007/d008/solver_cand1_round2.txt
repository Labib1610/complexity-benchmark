import sys

# The condition: det(Cu + Cv) != 0
# C1 = [[1, 1], [1, 1]], C2 = [[1, -1], [-1, 1]]
# C1 + C1 = [[2, 2], [2, 2]], det = 0
# C2 + C2 = [[2, -2], [-2, 2]], det = 0
# C1 + C2 = [[2, 0], [0, 2]], det = 4 != 0
# C2 + C1 = [[2, 0], [0, 2]], det = 4 != 0
# Thus, for every link (u, v), we must have b_u != b_v.
# This is a bipartite graph coloring problem.
# We need to check if the graph is bipartite.
# If it is, we want the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2.
# For each connected component, we have two choices:
# 1. Assign 1 to the first node encountered (in index order) and 2 to its neighbors.
# 2. Assign 2 to the first node encountered and 1 to its neighbors.
# We pick the one that makes the assignment lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible assignments for this component:
            # Option A: color 1 -> basis 1, color 2 -> basis 2
            # Option B: color 1 -> basis 2, color 2 -> basis 1
            # We want the one that is lexicographically smaller.
            # Find the smallest index in the component.
            start_node = min(component)
            
            # If color[start_node] == 1, Option A gives basis 1 to start_node.
            # If color[start_node] == 2, Option B gives basis 1 to start_node.
            if color[start_node] == 1:
                for node in component:
                    assignment[node] = color[node]
            else:
                for node in component:
                    assignment[node] = 3 - color[node]
                    
    print("YES")
    print(*(assignment[1:]))

solve()
