```python
import sys

# The problem states that for each experiment (u, v), the mean of the oxidation states
# must be 1.5. Since the states are 1 or 2, (s_u + s_v) / 2 = 1.5 implies s_u + s_v = 3.
# This means s_u must be different from s_v (one is 1, the other is 2).
# This is a bipartite graph coloring problem.
# We need to check if the graph is bipartite.
# To get the lexicographically smallest sequence, we iterate from 1 to n.
# If a node is not visited, we assign it 1 (the smallest possible value) and perform BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS from i, assign 1 to i to keep it lexicographically smallest
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
