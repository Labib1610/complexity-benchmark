import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # v1 = (1, 2, 3), v2 = (3, 1, 2)
    # v1 x v1 = (0, 0, 0), mag^2 = 0
    # v2 x v2 = (0, 0, 0), mag^2 = 0
    # v1 x v2 = (2*2 - 3*1, 3*3 - 1*2, 1*1 - 2*3) = (1, 7, -5)
    # mag^2 = 1^2 + 7^2 + (-5)^2 = 1 + 49 + 25 = 75
    # v2 x v1 = -(v1 x v2), mag^2 = 75
    
    # The condition is sum of stress dissipation factors = 75 * m.
    # Since each dampener contributes 75 if the pillars have different grades
    # and 0 if they have the same grade, the condition is equivalent to:
    # (number of dampeners connecting pillars with different grades) == m.
    # This means every dampener must connect pillars with different grades.
    # This is equivalent to checking if the graph is bipartite.
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # To get lexicographically smallest, we prefer 1 over 2.
    # For each connected component, we try assigning 1 to the first node.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
