```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # For each edge (u,v), we have a constraint:
    # f[u] != f[v] (because u's primary must match v's backup, which is the opposite of v's primary)
    # This means we need a bipartite coloring of the graph
    
    color = [-1] * n  # -1: uncolored, 0: protocol 1, 1: protocol 2
    
    # Check if the graph is bipartite
    for i in range(n):
        if color[i] == -1:
            queue = deque([i])
            color[i] = 0
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # The graph is bipartite. Now we need to find the lexicographically smallest valid assignment.
    # In a bipartite graph, we have two partitions. We need to decide which partition gets protocol 1.
    # We want the lexicographically smallest assignment, so we try assigning partition containing node 0 to protocol 1.
    
    # But we need to be more careful. We need to consider the constraints from the first node that has neighbors.
    # Actually, let's think differently. For each connected component, we have two possible colorings.
    # We need to choose the coloring that gives the lexicographically smallest result.
    
    result = [0] * n
    visited = [False] * n
    
    for i in range(n):
        if not visited[i] and adj[i]:  # Only process nodes with edges
            # Find the connected component
            component = []
            queue = deque([i])
            visited[i] = True
            component.append(i)
            
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        component.append(neighbor)
                        queue.append(neighbor)
            
            # For this component, we have two possible colorings
            # Try both and pick the lexicographically smaller one
            comp_min = min(component)
            
            # First coloring: color[comp_min] = 0 (protocol 1)
            coloring1 = color[:]
            
            # Second coloring: flip all colors in this component
            coloring2 = color[:]
            for node in component:
                coloring2[node] = 1 - coloring2[node]
            
            # Determine which coloring gives protocol 1 to comp_min
            if coloring1[comp_min] == 1:  # comp_min gets protocol 2 in coloring1
                coloring1, coloring2 = coloring2, coloring1  # Swap so that coloring1 gives protocol 1 to comp_min
            
            # Now coloring1 gives protocol 1 to the minimum node in component
            # This should be the lexicographically smaller assignment
            for node in component:
                color[node] = coloring1[node]
    
    # For isolated nodes (no edges), we can assign protocol 1
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
    
    # Convert to protocol numbers (1 or 2)
    for i in range(n):
        result[i] = color[i] + 1
    
    # Verify the solution
    for u, v in edges:
        if result[u] == result[v]:
            print("NO")
            return
    
    print("YES")
    print(" ".join(map(str, result)))

solve()
```
