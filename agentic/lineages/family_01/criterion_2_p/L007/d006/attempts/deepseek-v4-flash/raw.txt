We need to assign each reagent an oxidation state of 1 or 2 such that for every edge (u, v), the average is 1.5, i.e.:

\[
\frac{s_u + s_v}{2} = 1.5 \implies s_u + s_v = 3
\]

So if one is 1, the other must be 2, and vice versa. This is exactly a bipartite graph condition: the graph must be bipartite, with the two parts being oxidation state 1 and 2.

We also need the lexicographically smallest sequence. That means we should try to assign 1 to as many early-indexed nodes as possible, but respecting the bipartite constraint.

We can solve with BFS/DFS coloring, but to get lexicographically smallest, we process nodes in order and assign 1 if possible, otherwise 2, propagating constraints.

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n  # -1 uncolored, 0 -> state 1, 1 -> state 2

def bfs(start):
    q = deque([start])
    color[start] = 0
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = 1 - color[u]
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

# Try to assign lexicographically smallest
for i in range(n):
    if color[i] == -1:
        # Try assigning 0 (state 1) first
        # Temporarily set and check
        temp_color = color[:]
        temp_color[i] = 0
        # BFS from i with this assignment
        q = deque([i])
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if temp_color[v] == -1:
                    temp_color[v] = 1 - temp_color[u]
                    q.append(v)
                elif temp_color[v] == temp_color[u]:
                    ok = False
                    break
            if not ok:
                break
        if ok:
            color = temp_color
        else:
            # Must assign 1 (state 2)
            color[i] = 1
            if not bfs(i):
                print("NO")
                sys.exit()

# Final check
for u in range(n):
    for v in adj[u]:
        if color[u] == color[v]:
            print("NO")
            sys.exit()

print("YES")
print(' '.join(str(c + 1) for c in color))
```
