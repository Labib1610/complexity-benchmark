import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition: product of keys is 2.
    # Keys are 1 or 2.
    # Product is 2 if and only if one key is 1 and the other is 2.
    # This means the graph must be bipartite.
    # If a component is bipartite, we have two ways to color it (1,2) or (2,1).
    # To get the lexicographically smallest sequence, for each component,
    # we try to assign 1 to the smallest index in the component.
    
    color = [0] * (n + 1)
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            is_bipartite = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        is_bipartite = False
                        break
                if not is_bipartite:
                    break
            
            if not is_bipartite:
                print("NO")
                return
            
            # Two possible colorings for this component:
            # 1. color[v] == 1 -> key 1, color[v] == 2 -> key 2
            # 2. color[v] == 1 -> key 2, color[v] == 2 -> key 1
            
            # Find the smallest index in this component
            min_node = min(component)
            
            # Option 1: color[min_node] is 1, assign key 1 to color 1
            # Option 2: color[min_node] is 1, assign key 2 to color 1
            
            # We want the lexicographically smallest sequence.
            # Compare the two options at the smallest index.
            # Option 1: key[min_node] = 1
            # Option 2: key[min_node] = 2
            # Clearly Option 1 is better.
            
            for node in component:
                if color[node] == 1:
                    assignment[node] = 1
                else:
                    assignment[node] = 2
                    
    print("YES")
    print(*(assignment[1:]))

solve()
