```python
import sys

# The problem asks us to assign a mode (1 or 2) to each server such that
# for every edge (u, v), one server is mode 1 and the other is mode 2.
# If u is mode 1, it must be the origin for all its edges.
# If u is mode 2, it must be the destination for all its edges.
# This implies that if u is mode 1 and v is mode 2, the edge must be (u, v).
# If u is mode 2 and v is mode 1, the edge must be (v, u).
# If both are mode 1, they cannot be connected.
# If both are mode 2, they cannot be connected.
# Thus, the graph must be bipartite.
# The modes 1 and 2 correspond to the two partitions of the bipartite graph.
# To get the lexicographically smallest sequence, we want as many 1s as possible
# at the beginning. For each connected component, there are two ways to assign
# modes (swap 1s and 2s). We choose the one that makes the sequence lexicographically
# smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible assignments for this component:
            # Option A: color 1 -> mode 1, color 2 -> mode 2
            # Option B: color 1 -> mode 2, color 2 -> mode 1
            # We want the lexicographically smallest sequence.
            # Find the first node in the component and compare.
            
            comp_nodes = sorted(component)
            first_node = comp_nodes[0]
            
            # If color[first_node] == 1:
            # Option A: first_node is 1
            # Option B: first_node is 2
            # We prefer Option A.
            if color[first_node] == 1:
                for node in component:
                    ans[node] = color[node]
            else:
                for node in component:
                    ans[node] = 3 - color[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
