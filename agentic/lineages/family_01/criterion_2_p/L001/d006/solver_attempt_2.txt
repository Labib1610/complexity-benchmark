import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Bipartite check using 2-coloring
    # To get the lexicographically smallest sequence, we want machine 1 to be 1.
    # If machine 1 is part of a component, we assign it 1.
    # For each component, we try to assign the smallest possible values.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To ensure lexicographical smallest, we prefer color 1 for the smallest node
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Collect component nodes
            q = [i]
            visited_idx = 0
            while visited_idx < len(q):
                u = q[visited_idx]
                visited_idx += 1
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = -1 # Temporary
                        q.append(v)
            
            # Now color the component starting from the smallest node
            # Actually, the requirement is the smallest n-digit integer.
            # This means we want machine 1 to be 1, machine 2 to be 1 (if possible), etc.
            # We process nodes 1 to n. If a node is uncolored, color it 1.
            
    # Re-run coloring to ensure lexicographical smallest
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
