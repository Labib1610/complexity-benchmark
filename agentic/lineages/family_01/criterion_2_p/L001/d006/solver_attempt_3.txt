```python
import sys

# The problem asks for a configuration of 1s and 2s such that for every edge (u, v),
# power(u) + power(v) = 3.
# This means power(u) != power(v).
# This is equivalent to 2-coloring the graph with colors {1, 2}.
# If the graph is bipartite, we can color it.
# To get the lexicographically smallest sequence, for each connected component,
# we have two choices:
# 1. Assign color 1 to the node with the smallest index in the component.
# 2. Assign color 2 to the node with the smallest index in the component.
# We want the smallest value at the smallest index, so we should assign 1 to the
# smallest index node in each component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make the sequence lexicographically smallest,
            # we want the smallest index in this component to be 1.
            # Since we iterate 1 to n, i is the smallest index in this component.
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
