import sys

# The problem states that for each strut (u, v), the sum of positions must be 3.
# Since positions are {1, 2}, this means x_u + x_v = 3, so x_u != x_v.
# This is a standard 2-coloring problem on a graph.
# We want to minimize sum(x_i * 3^(n-i)).
# Since 3^(n-i) is much larger than any sum of subsequent terms,
# we want to make x_1 as small as possible (1), then x_2 as small as possible, etc.
# This is equivalent to a greedy approach on the connected components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning color 1 to the first node of the component
            # to minimize the instability (since 1 < 2).
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # We have a component. We need to decide if we assign 1 or 2 to the first node.
            # Actually, the greedy choice is to assign the smallest possible value to the 
            # smallest index in the component.
            # Let's re-evaluate: we want to minimize sum x_i * 3^(n-i).
            # This is lexicographical minimization of (x_1, x_2, ..., x_n).
            # For each component, we have two choices: (c_i) or (3-c_i).
            # We pick the one that makes the first node in the component 1.
            
            # Find the smallest index in this component
            comp_nodes = [node for node, c in component]
            first_node = min(comp_nodes)
            
            # If color[first_node] is 2, flip all colors in this component
            if color[first_node] == 2:
                for node in comp_nodes:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
