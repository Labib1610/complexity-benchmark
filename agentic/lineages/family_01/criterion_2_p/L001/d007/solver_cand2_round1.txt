import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # f(1)=10, f(2)=15
    # lcm(10, 10) = 10, d(10) = d(2*5) = 4
    # lcm(10, 15) = 30, d(30) = d(2*3*5) = 8
    # lcm(15, 15) = 15, d(15) = d(3*5) = 4
    # Total score = sum_{edges} d(lcm) = 8m
    # Let x_i = 1 if r_i=1, x_i = 0 if r_i=2
    # If r_u=r_v, score is 4. If r_u != r_v, score is 8.
    # We want sum_{edges} (8 if r_u != r_v else 4) = 8m
    # Let k be the number of edges with r_u != r_v.
    # 8k + 4(m-k) = 8m => 4k + 4m = 8m => 4k = 4m => k = m.
    # This means every edge must have r_u != r_v.
    # This is equivalent to checking if the graph is bipartite.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, we have two sets of nodes (color 0 and color 1).
    # We can assign r=1 to color 0 and r=2 to color 1, or vice versa.
    # Lexicographically smallest:
    # Option 1: color 0 -> 1, color 1 -> 2
    # Option 2: color 0 -> 2, color 1 -> 1
    # Compare the resulting arrays.
    
    res1 = [0] * n
    res2 = [0] * n
    for i in range(n):
        if color[i] == 0:
            res1[i] = 1
            res2[i] = 2
        else:
            res1[i] = 2
            res2[i] = 1
            
    ans = res1 if res1 < res2 else res2
    print("YES")
    print(*(ans))

solve()
