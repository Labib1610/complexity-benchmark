import sys

# The condition T_i = (3 + D_{i,j}) / 2 implies:
# If D_{i,j} = 1, then T_i = 2.
# If D_{i,j} = -1, then T_i = 1.
# Since D_{j,i} = -D_{i,j}, the condition for server j is:
# T_j = (3 + D_{j,i}) / 2 = (3 - D_{i,j}) / 2.
# If D_{i,j} = 1, then T_j = 1.
# If D_{i,j} = -1, then T_j = 2.
# Thus, for any link (i, j), we must have T_i != T_j.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 1 and 2.
# To get the lexicographically smallest sequence, we want T_1 = 1.
# For each connected component, we can either assign T_i = color[i] or T_i = 3 - color[i].
# To minimize lexicographically, for each component, we check the smallest index i in it.
# If we set T_i = 1, then all other nodes in that component are fixed.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            start_node = min(component)
            # We want ans[start_node] to be 1 if possible.
            # If color[start_node] is 1, we keep the current coloring.
            # If color[start_node] is 2, we flip the coloring.
            if color[start_node] == 1:
                for node in component:
                    ans[node] = color[node]
            else:
                for node in component:
                    ans[node] = 3 - color[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
