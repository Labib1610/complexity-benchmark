```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # For each synchronization pair (u,v), we need M_u * M_v = swap_matrix
    # Let's represent mode 1 as 0 (identity) and mode 2 as 1 (swap)
    # Then M_u * M_v = swap iff (mode_u == 1 and mode_v == 2) or (mode_u == 2 and mode_v == 1)
    # That means mode_u XOR mode_v = 1
    
    # So for each edge (u,v), we have the constraint: mode_u XOR mode_v = 1
    # This is a 2-SAT problem with constraints of the form x XOR y = 1
    
    # We can model this as a graph where each variable has two nodes: x and ~x
    # For constraint x XOR y = 1, we add clauses:
    # (x OR y) AND (~x OR ~y)
    # Which translates to edges: (~x -> y) and (~y -> x) and (x -> ~y) and (y -> ~x)
    
    # But since we know that x XOR y = 1, it's simpler:
    # x = NOT y  =>  x = 1 - y
    # So we can use a graph-based approach similar to bipartite checking
    
    # Build a graph where each connected component must be bipartite
    # with the additional constraint that neighbors have different values
    
    # In our case, for each edge (u,v): mode_u XOR mode_v = 1
    # This means mode_u != mode_v (they must be different)
    # So we're essentially trying to 2-color the graph such that adjacent nodes have different colors
    # But we also want the lexicographically smallest assignment
    
    # Let's build the adjacency list and then do BFS/DFS for each component
    
    color = [-1] * (n + 1)  # -1: unvisited, 0: mode 1, 1: mode 2
    
    # Check if the graph is bipartite
    for i in range(1, n + 1):
        if color[i] == -1:
            # Start BFS from node i, assign color 0 (mode 1) to get lex smallest
            queue = deque()
            queue.append(i)
            color[i] = 0
            
            while queue:
                u = queue.popleft()
                current_color = color[u]
                
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - current_color
                        queue.append(v)
                    elif color[v] == current_color:
                        # Conflict: adjacent nodes have same color
                        print("NO")
                        return
    
    # If we reach here, a valid coloring exists
    # Convert colors back to modes: 0 -> 1, 1 -> 2
    modes = [color[i] + 1 for i in range(1, n + 1)]
    print("YES")
    print(' '.join(map(str, modes)))

if __name__ == "__main__":
    main()
```
