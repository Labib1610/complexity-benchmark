```python
import sys

# The condition r_u * r_v = 2 means:
# If r_u = 1, then r_v must be 2.
# If r_u = 2, then r_v must be 1.
# This is a bipartite coloring problem where each edge forces the two nodes
# to have different values (1 and 2).
# If a node has a self-loop (u, u), then r_u * r_u = 2, which is impossible
# since r_u is 1 or 2 (1*1=1, 2*2=4).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # colors: 0 = unassigned, 1 = band 1, 2 = band 2
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # To get lexicographically smallest, we prefer 1 for the first node
            # in each connected component.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, colors[v]))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
