```python
import sys

# The condition is product_{j=1}^m (s_{u_j} + s_{v_j}) = 3^m.
# Since s_i is either 1 or 2, s_u + s_v can be 2 (1+1), 3 (1+2), or 4 (2+2).
# For the product to be 3^m, every factor (s_u + s_v) must be 3.
# This means for every link (u, v), s_u + s_v = 3, which implies one is 1 and the other is 2.
# This is equivalent to saying the graph must be bipartite, and for every edge (u, v),
# s_u != s_v.
# If the graph is bipartite, we can color it with 0 and 1.
# Let color[u] be the color of node u.
# If we set s_u = 1 if color[u] == 0 and s_u = 2 if color[u] == 1,
# then for every edge (u, v), s_u + s_v = 1 + 2 = 3.
# If we swap the colors (s_u = 2 if color[u] == 0 and s_u = 1 if color[u] == 1),
# we also get s_u + s_v = 3.
# For each connected component, we have two choices of assignment.
# To get the lexicographically smallest sequence, for each component, we pick the
# assignment that makes the first node (smallest index) in that component 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            component.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        component.append(v)
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            start_node = min(component)
            # If color[start_node] is 0, we want s[start_node] = 1.
            # If color[start_node] is 1, we want s[start_node] = 2.
            # So if color[start_node] == 0, map 0->1, 1->2.
            # If color[start_node] == 1, map 0->2, 1->1.
            
            if color[start_node] == 0:
                for node in component:
                    res[node] = color[node] + 1
            else:
                for node in component:
                    res[node] = 2 - color[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
```
