import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The condition s[u] + s[v] = 3 means s[u] != s[v].
    # This is a standard 2-coloring problem (bipartite graph).
    # To get the lexicographically smallest sequence, we want s[i] = 1 whenever possible.
    # We iterate 1 to n. If not colored, start BFS/DFS.
    # To make it lexicographically smallest, we prefer 1 over 2.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS component
            # To be lexicographically smallest, we assign 1 to the first node
            stack = [(i, 1)]
            color[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                # Neighbors must have color 3 - c
                target_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = target_c
                        stack.append((v, target_c))
                    elif color[v] != target_c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
