```python
import sys

# The problem requires that for every pair (u, v), s[u] + s[v] = 3.
# This means s[u] must be different from s[v].
# This is equivalent to 2-coloring a graph.
# To get the lexicographically smallest sequence, we want s[i] = 1 whenever possible.
# We iterate from 1 to n. If a node is not colored, we start a BFS/DFS.
# To make it lexicographically smallest, we prefer assigning 1 to the smallest
# available node in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign 1 to i to be lexicographically smallest
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
