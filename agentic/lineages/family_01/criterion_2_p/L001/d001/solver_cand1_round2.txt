```python
import sys

# The problem asks for a 2-coloring of a graph where edges represent 
# constraints that the two nodes must have different values (1 and 2).
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we iterate from 1 to n.
# If a node is not yet colored, we assign it 1 (the smaller value) 
# and perform a BFS/DFS to color its connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # Assign 1 to the first node to keep it lexicographically smallest
            stack = [(i, 1)]
            colors[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                for v in adj[u]:
                    if colors[v] == 0:
                        # Must be different from u
                        new_c = 3 - c
                        colors[v] = new_c
                        stack.append((v, new_c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
