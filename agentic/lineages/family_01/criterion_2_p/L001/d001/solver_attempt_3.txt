import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # To get the lexicographically smallest assignment, we want p_i = 1 whenever possible.
    # This is a 2-coloring problem on a graph.
    # For each connected component, we can pick the color of the smallest node.
    # To make the sequence lexicographically smallest, for each component,
    # we try setting the smallest node to 1. If that fails, we try 2.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i
            # We want to assign color 1 to i to be lexicographically smallest
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring the component, check if it's the lexicographically smallest
            # Actually, the greedy choice of color 1 for the smallest node in the component
            # is sufficient because the choice of color for one node in a connected
            # component determines the color of all other nodes in that component.
            
    print("YES")
    print(*(color[1:]))

solve()
