import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # p_u + p_v = 3 means p_u != p_v.
    # This is a bipartite coloring problem.
    # We want the lexicographically smallest sequence (p_1, ..., p_n).
    # To make it lexicographically smallest, we prefer p_i = 1 whenever possible.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this connected component
            # To make it lexicographically smallest, we try to set color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # After coloring the component, check if we could have made it smaller.
            # Actually, for each component, there are only two valid colorings:
            # 1. color[u] = c, color[v] = 3-c
            # 2. color[u] = 3-c, color[v] = c
            # We choose the one that makes the first node (smallest index) in the component 1.
            # Since we iterate i from 1 to n, the first node we encounter in a component
            # is always the smallest index in that component. Setting it to 1 is always
            # lexicographically better than setting it to 2.
            
    print("YES")
    print(*(color[1:]))

solve()
