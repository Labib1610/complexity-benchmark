```python
import sys

# The problem requires that for every bond (u, v), spin(u) + spin(v) = 3.
# Since spin states are {1, 2}, this means spin(u) != spin(v).
# This is equivalent to 2-coloring a graph where each edge (u, v)
# forces u and v to have different colors.
# To get the lexicographically smallest sequence, we want to assign 1
# to as many early monomers as possible.
# We use BFS/DFS to color each connected component. For each component,
# we try assigning 1 to the smallest uncolored node and propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we prefer color 1 for the
            # smallest index in the component.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
