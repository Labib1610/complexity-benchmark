We can model the problem as a bipartite check: each artifact is a node, discordant pairs are edges that must connect nodes in different halls.

The key is:
1. Build the graph from the edges.
2. Run BFS/DFS to color each connected component in two colors (1 and 2).
3. If any connected component is not bipartite, output `"NO"`.
4. If all components are bipartite, we must also choose the lexicographically smallest assignment.  
   For each unvisited node in increasing order, try assigning it hall `1` if it doesn’t cause a conflict; BFS color the whole component starting with `1` if possibleinches; otherwise start with `2`.

Here’s the implementation:

```python
from collections import deque
import sys

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n
possible = True

for i in range(n):
    if color[i] == -1:
        color[i] = 0
        q = deque([i])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                elif color[v] == color[u]:
                    possible = False
                    break
            if not possible:
                break
    if not possible:
        break

if not possible:
    print("NO")
else:
    # Now lexicographically smallest: for each comp, try starting with hall 1
    final = [-1] * n
    for i in range(n):
        if final[i] != -1:
            continue
        # try color 0 (hall 1) for i
        c1 = [-1] * n
        c1[i] = 0
        q = deque([i])
        ok1 = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if c1[v] == -1:
                    c1[v] = c1[u] ^ 1
                    q.append(v)
                elif c1[v] == c1[u]:
                    ok1 = False
                    break
            if not ok1:
                break
        if ok1:
            for j in range(n):
                if c1[j] != -1:
                    final[j] = c1[j]
        else:
            c2 = [-1] * n
            c2[i] = 1
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if c2[v] == -1:
                        c2[v] = c2[u] ^ 1
                        q.append(v)
            for j in range(n):
                if c2[j] != -1:
                    final[j] = c2[j]

    print("YES")
    print(' '.join(str(x + 1) for x in final))
```
