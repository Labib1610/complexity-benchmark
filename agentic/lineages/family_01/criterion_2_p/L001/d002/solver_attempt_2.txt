```python
import sys

# The problem asks for a configuration of 1s and 2s such that for every edge (u, v),
# p[u] * p[v] = 2.
# Since p[i] is in {1, 2}, p[u] * p[v] = 2 implies that one must be 1 and the other 2.
# This is equivalent to a 2-coloring problem on a graph where 1 and 2 are the colors.
# We want the lexicographically smallest sequence, so we prefer 1 over 2.
# For each connected component, we can try assigning 1 to the first node and see if it works.
# If it works, we get a valid coloring. If not, the graph is not bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 2 if c == 1 else 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
