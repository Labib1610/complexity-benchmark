import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If an artifact is not yet colored, we try assigning it 1.
    # This is a standard 2-coloring problem (bipartite check).
    # Since we want the lexicographically smallest, we prioritize 1 over 2.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assigning 1 to the first node
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                # To maintain lexicographical order, we should process neighbors
                # in a way that respects the smallest assignment.
                # However, for a connected component, once the first node is fixed,
                # all others are determined.
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
