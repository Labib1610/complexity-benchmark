import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unassigned, 1 = Hall 1, 2 = Hall 2
    color = [0] * (n + 1)
    
    # To handle the preference, we use a greedy approach with 2-SAT logic.
    # For each connected component, we can pick the color of the first node.
    # The preference rule: try to set color[i] = 1. If it leads to a contradiction,
    # set color[i] = 2.
    
    def get_component(start_node):
        comp = []
        stack = [start_node]
        color[start_node] = 1
        comp.append(start_node)
        while stack:
            u = stack.pop()
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - color[u]
                    comp.append(v)
                    stack.append(v)
                elif color[v] == color[u]:
                    return None
        return comp

    # We need to process nodes 1 to n.
    # If a node is already colored, skip.
    # If not, try setting it to 1. If that fails, try 2.
    
    assignment = [0] * (n + 1)
    
    # We need to check consistency for the whole graph
    # The preference rule implies we fix the color of the first node in each component
    # to 1 if possible.
    
    # First, check if the graph is bipartite
    comp_id = [0] * (n + 1)
    components = []
    curr_id = 0
    for i in range(1, n + 1):
        if comp_id[i] == 0:
            curr_id += 1
            comp = []
            stack = [i]
            comp_id[i] = curr_id
            comp.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if comp_id[v] == 0:
                        comp_id[v] = curr_id
                        comp.append(v)
                        stack.append(v)
            components.append(comp)
            
    # For each component, check bipartiteness and store the two partitions
    partitions = []
    for comp in components:
        c1 = {comp[0]}
        c2 = set()
        stack = [(comp[0], 1)]
        visited = {comp[0]: 1}
        while stack:
            u, col = stack.pop()
            for v in adj[u]:
                if v not in visited:
                    visited[v] = 3 - col
                    if visited[v] == 1: c1.add(v)
                    else: c2.add(v)
                    stack.append((v, visited[v]))
                elif visited[v] == col:
                    print("NO")
                    return
        partitions.append((c1, c2))
        
    # Now apply the preference rule
    final_assignment = [0] * (n + 1)
    comp_assigned = [False] * (curr_id + 1)
    
    for i in range(1, n + 1):
        cid = comp_id[i]
        if not comp_assigned[cid]:
            c1, c2 = partitions[cid-1]
            # Try to put i in Hall 1
            if i in c1:
                for node in c1: final_assignment[node] = 1
                for node in c2: final_assignment[node] = 2
            else:
                for node in c2: final_assignment[node] = 1
                for node in c1: final_assignment[node] = 2
            comp_assigned[cid] = True
            
    print("YES")
    print(*(final_assignment[1:]))

solve()
