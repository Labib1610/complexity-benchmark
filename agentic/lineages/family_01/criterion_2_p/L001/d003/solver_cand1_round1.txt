```python
import sys

# The constraint is that for every bond (u, v), s_u * s_v = 2.
# Since s_i is in {1, 2}, the only way s_u * s_v = 2 is if one is 1 and the other is 2.
# This means the graph must be bipartite.
# We want the lexicographically smallest sequence (s_1, s_2, ..., s_n).
# This means we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unvisited, 1 = state 1, 2 = state 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, try setting color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                for v in adj[u]:
                    if color[v] == 0:
                        # If u is 1, v must be 2. If u is 2, v must be 1.
                        # So color[v] = 3 - color[u]
                        color[v] = 3 - color[u]
                        stack.append((v, color[v]))
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
