```python
import sys

# The problem requires that for every pair (u, v), state[u] + state[v] = 3.
# Since states are {1, 2}, this means state[u] != state[v].
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest sequence of states.
# Lexicographically smallest means we prefer 1 over 2 for the smallest indices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    state = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is unvisited, we assign it 1 (the smallest possible state).
    for i in range(1, n + 1):
        if state[i] == 0:
            stack = [i]
            state[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = 3 - state[u]
                        stack.append(v)
                    elif state[v] == state[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(state[1:]))

solve()
```
