import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)

    # 0: unassigned, 1: level 1, 2: level 2
    # To get lexicographically smallest, we prefer assigning 1.
    # If we assign 1 to a node, its neighbors must be 2.
    # If we assign 2 to a node, its neighbors must be 1.
    # We process nodes 1 to n. If a node is unassigned, we try to assign 1.
    # This greedy choice (1) is smaller than (2).
    # However, if the node is already assigned, we keep it.
    # If assigning 1 leads to a contradiction, we report NO.
    # Wait, is it possible that assigning 1 leads to contradiction but assigning 2 works?
    # Yes. But the problem asks for the lexicographically smallest valid sequence.
    # If we assign 1 to node $i$, and it propagates constraints, we fix those.
    # If this creates a conflict, it means there is NO valid sequence where node $i$ is 1.
    # Since we want lexicographically smallest, we want $r_i=1$ if possible.
    # If $r_i=1$ is impossible, then in ANY valid sequence $r_i$ must be 2.
    # If $r_i=2$ is also impossible, then NO solution exists.
    # So the strategy is:
    # Iterate $i$ from 1 to $n$.
    # If $i$ is unassigned, tentatively assign 1. Run BFS/DFS to check consistency and propagate.
    # If consistent, great, we locked in the smallest choice for $i$.
    # If inconsistent, it means $r_i$ cannot be 1. So $r_i$ MUST be 2.
    # We then assign 2 and propagate. If this is also inconsistent, output NO.
    
    # Optimization: We don't need to re-run BFS from scratch for neighbors if they are already assigned.
    # We just need to propagate from the current node.
    
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if res[i] == 0:
            # Try assigning 1
            # We use a queue for BFS propagation
            # We need to check if assigning 1 to i is consistent with existing assignments
            # and propagate.
            
            # Since we process in order, and we only assign values when we visit a node index,
            # any previously assigned node $j < i$ is fixed.
            # If $i$ is unassigned, it means it's not connected to any $j < i$.
            # So assigning 1 to $i$ is locally consistent with all $j < i$.
            # We just need to propagate to $k > i$.
            
            # Let's try to assign 1.
            # We perform a BFS to assign values.
            # If we hit a contradiction, we backtrack?
            # Backtracking is costly. 
            # Let's analyze the contradiction.
            # If assigning 1 to $i$ causes a contradiction, it implies there is a path
            # from $i$ to some node $k$ (possibly $i$ itself via a cycle) that forces $i$ to be 2.
            # This happens if there is an odd-length cycle involving $i$.
            # Or if $i$ is connected to a node $j < i$ which forces $i$ to be 2.
            # But we established $i$ is not connected to any $j < i$ (since res[i] is 0).
            # So the only way assigning 1 fails is if the connected component containing $i$
            # is not bipartite.
            # If the component is bipartite, both (1, 2) and (2, 1) assignments are valid.
            # Since we want lexicographically smallest, we pick the one starting with 1.
            # If the component is NOT bipartite, NO solution exists.
            
            # So the logic simplifies:
            # If res[i] == 0:
            #   Start BFS from i with res[i] = 1.
            #   If bipartite check fails, return NO.
            #   Else, the assignments made are optimal.
            
            q = deque([i])
            res[i] = 1
            is_bipartite = True
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if res[v] == 0:
                        res[v] = 3 - res[u]
                        q.append(v)
                    elif res[v] == res[u]:
                        is_bipartite = False
                        break
                if not is_bipartite:
                    break
            
            if not is_bipartite:
                print("NO")
                return

    print("YES")
    print(*(res[1:]))

if __name__ == '__main__':
    solve()
