import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        
        # BFS to check bipartiteness and assign colors
        # We want lexicographically smallest, so we try to assign 1 to the start node
        # If the component is bipartite, this determines the assignment uniquely
        # (up to swapping all colors in the component)
        # We choose the assignment where the start node gets color 1
        
        q = deque()
        q.append(start)
        color[start] = 1
        is_bipartite = True
        
        while q and is_bipartite:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - color[u]
                    q.append(v)
                elif color[v] == color[u]:
                    is_bipartite = False
                    break
        
        if not is_bipartite:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color[1:])))

if __name__ == "__main__":
    solve()
