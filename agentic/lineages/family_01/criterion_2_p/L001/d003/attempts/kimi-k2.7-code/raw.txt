 ```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    g = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        g[u].append(v)
        g[v].append(u)

    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] != -1:
            continue
        dq = deque([i])
        color[i] = 0
        while dq:
            x = dq.popleft()
            cx = color[x]
            for y in g[x]:
                if color[y] == -1:
                    color[y] = cx ^ 1
                    dq.append(y)
                elif color[y] == cx:
                    sys.stdout.write('NO')
                    return

    out = ['YES']
    out.append(' '.join(str(color[i] + 1) for i in range(1, n + 1)))
    sys.stdout.write('\n'.join(out))

if __name__ == '__main__':
    solve()
```
