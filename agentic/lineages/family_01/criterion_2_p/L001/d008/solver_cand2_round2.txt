```python
import sys

# The condition Tr(Mu * Mv) = -2:
# A = [[1, 0], [0, -1]], B = [[-1, 0], [0, 1]]
# A*A = [[1, 0], [0, 1]], Tr(A*A) = 2
# B*B = [[1, 0], [0, 1]], Tr(B*B) = 2
# A*B = [[-1, 0], [0, -1]], Tr(A*B) = -2
# B*A = [[-1, 0], [0, -1]], Tr(B*A) = -2
# So, Mu and Mv must be different (one A, one B).
# This is a 2-coloring problem on a graph.
# A = 1, B = 2. If u and v are connected, they must have different values.
# Lexicographically smallest: try to assign 1 to node i if not already colored.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we prefer 1 over 2
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
