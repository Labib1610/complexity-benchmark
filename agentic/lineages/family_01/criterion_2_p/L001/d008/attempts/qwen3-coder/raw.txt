```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list and degree count
    adj = defaultdict(list)
    degree = [0] * (n + 1)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
        index += 2
    
    # If no constraints, assign all mode 1 (lexicographically smallest)
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Color graph with 2 colors (Bipartite check)
    color = [-1] * (n + 1)  # -1: unvisited, 0: color 0, 1: color 1
    queue = deque()
    
    # Process each connected component
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # Start BFS from 'start'
        color[start] = 0
        queue.append(start)
        while queue:
            node = queue.popleft()
            current_color = color[node]
            for neighbor in adj[node]:
                if color[neighbor] == -1:
                    color[neighbor] = 1 - current_color
                    queue.append(neighbor)
                elif color[neighbor] == current_color:
                    print("NO")
                    return
    
    # Construct result: assign mode based on color
    # We want lexicographically smallest, so prefer mode 1 (color 0)
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:
            result.append('1')
        else:
            result.append('2')
    
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
