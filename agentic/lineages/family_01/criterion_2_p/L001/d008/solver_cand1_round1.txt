import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    d = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        d[u] += 1
        d[v] += 1
        
    # E = sum(6 * D_i * r_i^2) - sum(14 * r_u * r_v) = 2m
    # Let r_i = 1 + x_i, where x_i in {0, 1}
    # r_i^2 = 1 + 2x_i + x_i^2 = 1 + 3x_i (since x_i^2 = x_i)
    # E = sum(6 * D_i * (1 + 3x_i)) - sum(14 * (1 + x_u)(1 + x_v))
    # E = sum(6*D_i + 18*D_i*x_i) - sum(14 * (1 + x_u + x_v + x_u*x_v))
    # Since sum(D_i) = 2m, sum(6*D_i) = 12m
    # E = 12m + sum(18*D_i*x_i) - 14m - sum(14*x_u) - sum(14*x_v) - sum(14*x_u*x_v)
    # E = -2m + sum(18*D_i*x_i) - 28*sum(D_i*x_i) - sum(14*x_u*x_v)
    # Wait, sum(14*x_u) + sum(14*x_v) = sum(14*D_i*x_i)
    # E = -2m + sum(18*D_i*x_i) - sum(14*D_i*x_i) - sum(14*x_u*x_v)
    # E = -2m + sum(4*D_i*x_i) - sum(14*x_u*x_v)
    # We want E = 2m, so:
    # 4m = sum(4*D_i*x_i) - sum(14*x_u*x_v)
    # 2m = sum(2*D_i*x_i) - sum(7*x_u*x_v)
    
    # This is equivalent to: sum_{i} (2*D_i*x_i) - sum_{(u,v)} (7*x_u*x_v) = 2m
    # Let's test r_i = 1 (x_i = 0) for all i: E = 0 - 0 = 0 != 2m (unless m=0)
    # Actually, the formula simplifies to:
    # sum_{i} (6*D_i*r_i^2) - sum_{(u,v)} (14*r_u*r_v) = 2m
    # If r_i = 1 for all i: sum(6*D_i) - sum(14) = 12m - 14m = -2m.
    # If r_i = 2 for all i: sum(6*D_i*4) - sum(14*4) = 48m - 56m = -8m.
    
    # Let's re-evaluate:
    # If r_i = 1, energy is -2m. If r_i = 2, energy is -8m.
    # This suggests the formula might be different or I misread.
    # Re-reading: E = sum(6 * D_i * r_i^2) - sum(14 * r_u * r_v)
    # Let r_i = 1: 6 * sum(D_i) - 14 * m = 6 * 2m - 14m = -2m.
    # Let r_i = 2: 6 * sum(D_i * 4) - 14 * 4 * m = 48m - 56m = -8m.
    # The only way to get 2m is to mix 1s and 2s.
    
    # Let x_i = r_i - 1 (x_i in {0, 1})
    # E = sum(6*D_i*(1+x_i)^2) - sum(14*(1+x_u)*(1+x_v))
    # E = sum(6*D_i*(1+2x_i+x_i^2)) - sum(14*(1+x_u+x_v+x_u*x_v))
    # E = 12m + sum(12*D_i*x_i) + sum(6*D_i*x_i) - 14m - 14*sum(x_u+x_v) - 14*sum(x_u*x_v)
    # E = -2m + 18*sum(D_i*x_i) - 14*sum(D_i*x_i) - 14*sum(x_u*x_v)
    # E = -2m + 4*sum(D_i*x_i) - 14*sum(x_u*x_v) = 2m
    # 4*sum(D_i*x_i) - 14*sum(x_u*x_v) = 4m
    # 2*sum(D_i*x_i) - 7*sum(x_u*x_v) = 2m
    
    # This is a classic problem: 2*sum(D_i*x_i) - 7*sum(x_u*x_v) = 2m
    # For each edge (u, v), if x_u=1, x_v=1, it contributes -7.
    # If x_u=1, x_v=0, it contributes 0.
    # If x_u=0, x_v=0, it contributes 0.
    # This is equivalent to: sum_{i} (2*D_i*x_i) - 7 * (number of edges with both endpoints 1) = 2m
    
    # Actually, the condition is: sum_{i} (2*D_i*x_i) - 7 * (number of edges with both endpoints 1) = 2m
    # This is satisfied if x_i = 0 for all i (0 = 2m, only if m=0).
    # Wait, the example 1: 4 nodes, 4 edges (1-2, 2-3, 3-4, 4-1). m=4.
    # r = [1, 2, 1, 2], x = [0, 1, 0, 1].
    # D = [2, 2, 2, 2].
    # 2*sum(D_i*x_i) = 2*(2*0 + 2*1 + 2*0 + 2*1) = 8.
    # Edges: (1,2), (2,3), (3,4), (4,1).
    # x_u*x_v: 0*1=0, 1*0=0, 0*1=0, 1*0=0. Sum = 0.
    # 8 - 7*0 = 8. 2m = 8. Matches!
    
    # So we need to pick x_i in {0, 1} such that 2*sum(D_i*x_i) - 7*sum(x_u*x_v) = 2m.
    # This is equivalent to: sum_{i} (2*D_i*x_i) - 7 * (edges with x_u=1, x_v=1) = 2m.
    # Since 2*D_i = sum_{j: u_j=i} 2, this is sum_{edges (u,v)} (2*x_u + 2*x_v) - 7*sum(x_u*x_v) = 2m.
    # sum_{edges (u,v)} (2*x_u + 2*x_v - 7*x_u*x_v) = 2m.
    # For each edge, the contribution is:
    # x_u=0, x_v=0: 0
    # x_u=1, x_v=0: 2
    # x_u=0, x_v=1: 2
    # x_u=1, x_v=1: 2+2-7 = -3
    
    # We want sum_{edges} f(x_u, x_v) = 2m.
    # If we set x_i = 0 for all i, sum = 0.
    # If we set x_i = 1 for all i, sum = -3m.
    # This looks like we need to find x_i to satisfy the sum.
    # Actually, the only way to get 2m is to have some edges contribute 2 and some -3.
    # This is only possible if the graph is bipartite or has specific structure.
    # But wait, the example 1 works with x=[0, 1, 0, 1].
    # This is just a 2-coloring!
    
    # If the graph is bipartite, we can color it with 0 and 1.
    # Let the two sets be A and B. If we set x_i=1 for i in A, x_i=0 for i in B:
    # Edges are only between A and B, so x_u*x_v is always 0.
    # The sum becomes sum_{edges} (2*x_u + 2*x_v) = 2 * sum_{i} D_i * x_i = 2 * sum_{i in A} D_i.
    # We need 2 * sum_{i in A} D_i = 2m, so sum_{i in A} D_i = m.
    # This is always true for any bipartite graph!
    
    # So: 1. Check if bipartite. 2. If yes, find the two sets A and B.
    # 3. We need to pick x_i such that sum_{i in A} D_i = m.
    # But for any bipartite graph, sum_{i in A} D_i = m is always true!
    # So any bipartite graph works.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    res = [str(color[i] + 1) for i in range(1, n + 1)]
    print(" ".join(res))

solve()
