```python
import sys

# The condition f_u(f_v(x)) = -x means:
# If f_u(x) = s_u * x and f_v(x) = s_v * x, then s_u * s_v * x = -x.
# This implies s_u * s_v = -1.
# Since s_i is either 1 (Mode 1) or -1 (Mode 2),
# this means s_u != s_v.
# This is a classic 2-coloring problem on a graph where edges represent
# constraints that the two nodes must have different colors.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # colors: 0 = unassigned, 1 = Mode 1, 2 = Mode 2
    # Mode 1 corresponds to s=1, Mode 2 corresponds to s=-1
    # Constraint: s_u * s_v = -1 => colors must be different
    # Lexicographically smallest: try to assign 1 whenever possible.
    
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we greedily pick 1 for the first node
            stack = [(i, 1)]
            colors[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                for v in adj[u]:
                    target_c = 3 - c
                    if colors[v] == 0:
                        colors[v] = target_c
                        stack.append((v, target_c))
                    elif colors[v] != target_c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
