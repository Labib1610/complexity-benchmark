import sys

# The orthogonality condition:
# Integral from -1 to 1 of phi_u(t) * phi_v(t) dt = 0
# If phi(t) = t (1) and phi(t) = t^2 (2):
# Integral of t * t = Integral of t^2 = [t^3/3] from -1 to 1 = 2/3 != 0
# Integral of t^2 * t^2 = Integral of t^4 = [t^5/5] from -1 to 1 = 2/5 != 0
# Integral of t * t^2 = Integral of t^3 = [t^4/4] from -1 to 1 = 0
# So, two channels are orthogonal if and only if one is t and the other is t^2.
# This means for every constraint (u, v), u and v must have different values.
# This is a 2-coloring problem on a graph.
# We want the lexicographically smallest sequence (r_1, ..., r_n) where r_i in {1, 2}.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # If a channel is constrained with itself, it must be orthogonal to itself.
            # Integral of t*t = 2/3, t^2*t^2 = 2/5. Neither is 0.
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If not colored, try color 1, then color 2.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [i]
            color[i] = 1
            component = [i]
            
            # We need to be careful: the lexicographical requirement applies to the 
            # whole sequence, not just components.
            # Actually, for each component, we have two choices: 
            # (1, 2, 1, ...) or (2, 1, 2, ...).
            # We pick the one that makes the first node in the component 1 if possible.
            
            q = [i]
            color[i] = 1
            comp_nodes = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        comp_nodes.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if we could have started with 1. 
            # Since we process 1..n, the first uncolored node i is the smallest index.
            # We already set color[i] = 1, which is the smallest possible value.
            
    print("YES")
    print(*(color[1:]))

solve()
