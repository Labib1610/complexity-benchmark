```python
import sys

# The problem asks for an assignment of modes (1: Transmitter, 2: Receiver)
# such that for every edge (u, v), if u is 1, it must be directed u->v.
# If u is 2, it must be directed v->u.
# This implies that if u is 1 and v is 2, the edge must be directed u->v.
# If u is 2 and v is 1, the edge must be directed v->u.
# If both are 1, they cannot be connected (a link cannot be directed away from both).
# If both are 2, they cannot be connected (a link cannot be directed towards both).
# Thus, the graph must be bipartite, and the modes must correspond to the two partitions.
# To get the lexicographically smallest sequence, we try to assign 1 to server 1.
# This forces the modes of all connected components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    modes = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if modes[i] == 0:
            # Try assigning 1 to the first node of the component
            modes[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if modes[v] == 0:
                        modes[v] = 3 - modes[u]
                        stack.append(v)
                    elif modes[v] == modes[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(modes[1:]))

solve()
```
