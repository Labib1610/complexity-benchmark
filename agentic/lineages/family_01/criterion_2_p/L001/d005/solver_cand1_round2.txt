import sys

# The condition sum((r_u - r_v)^2) = m for r_i in {1, 2}
# (r_u - r_v)^2 is 1 if r_u != r_v, and 0 if r_u == r_v.
# So the condition is: the number of edges (u, v) such that r_u != r_v must be exactly m.
# This means every edge must connect monomers with different resonance levels.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 2 colors (1 and 2).
# For each connected component, there are two ways to color it (swap 1 and 2).
# To get the lexicographically smallest sequence, for each component, we pick the
# coloring that assigns 1 to the smallest index in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            # to decide which coloring is lexicographically smaller
            comp_nodes = [i]
            idx_in_comp = 0
            while idx_in_comp < len(comp_nodes):
                u = comp_nodes[idx_in_comp]
                idx_in_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        comp_nodes.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # The two possible colorings are:
            # 1. color[v]
            # 2. 3 - color[v]
            # We want the one that makes the sequence lexicographically smallest.
            # Find the smallest index in this component.
            min_node = min(comp_nodes)
            if color[min_node] == 2:
                for node in comp_nodes:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
