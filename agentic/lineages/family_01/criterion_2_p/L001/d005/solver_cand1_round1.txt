import sys

# The condition M * s = 0 means for each bond (u, v), s_u + s_v = 0.
# Since s_i = r_i - 1.5, where r_i in {1, 2}, s_i is in {-0.5, 0.5}.
# s_u + s_v = 0 implies s_u = -s_v.
# If u = v, s_u + s_u = 0 => 2 * s_u = 0 => s_u = 0, which is impossible.
# So if there is a self-loop, no solution exists.
# Otherwise, for each connected component, if we pick s_u, all other s_v are determined.
# This is a bipartite graph coloring problem.
# s_i = 0.5 corresponds to r_i = 2, s_i = -0.5 corresponds to r_i = 1.
# The condition s_u = -s_v means r_u - 1.5 = -(r_v - 1.5) = -r_v + 1.5
# r_u + r_v = 3.
# If u and v are connected, r_u != r_v.
# This means each connected component must be bipartite.
# If bipartite, we have two choices for each component (swap 1s and 2s).
# To get lexicographically smallest, for each component, we want the first node
# (smallest index) to be 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # To make lexicographically smallest, find the smallest index in component
            # and ensure its r value is 1.
            # color 1 -> r=1, color 2 -> r=2 (or vice versa)
            # If we set color[min_node] = 1, then r[min_node] = 1.
            min_node = min(component)
            if color[min_node] == 1:
                for node in component:
                    res[node] = color[node]
            else:
                for node in component:
                    res[node] = 3 - color[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
