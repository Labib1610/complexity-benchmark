import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The condition "sum of binding sites = 3" means if node u has x, node v must have 3-x.
    # This is a 2-coloring problem. If u has 1, v must have 2. If u has 2, v must have 1.
    # We want the lexicographically smallest sequence, so we prefer 1 over 2.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this connected component
            # To get lexicographically smallest, try assigning 1 to the first node
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, val = stack.pop()
                
                target = 3 - val
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = target
                        stack.append((v, target))
                    elif assignment[v] != target:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
