import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Color the graph with 2 colors (Bipartite check)
    color = [0] * (n + 1)  # 0: uncolored, 1: color 1, 2: color 2
    visited = [False] * (n + 1)
    
    for start in range(1, n + 1):
        if not visited[start]:
            queue = deque([start])
            color[start] = 1
            visited[start] = True
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        color[neighbor] = 3 - color[node]  # Flip color (1->2, 2->1)
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # If bipartite, assign modes based on colors
    # We want lexicographically smallest, so we prefer color 1 to be mode 1
    print("YES")
    print(" ".join(str(color[i]) for i in range(1, n + 1)))

solve()
