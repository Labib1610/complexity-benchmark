import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n+1)
    possible = True
    
    for i in range(1, n+1):
        if color[i] == 0:
            queue = deque([i])
            color[i] = 1
            comp = [i]
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                        comp.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
            if not possible:
                break
            # Decide which side gets 1 to minimize the integer
            # Count how many in comp are currently colored 1
            cnt1 = sum(1 for x in comp if color[x] == 1)
            if cnt1 > len(comp) - cnt1:
                for x in comp:
                    color[x] = 3 - color[x]
    
    if not possible:
        sys.stdout.write("NO\n")
    else:
        out = ' '.join(str(color[i]) for i in range(1, n+1))
        sys.stdout.write("YES\n")
        sys.stdout.write(out + "\n")

solve()
