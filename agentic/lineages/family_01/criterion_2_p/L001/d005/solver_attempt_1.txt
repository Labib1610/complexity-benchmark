```python
import sys

# The problem states that for each strut (u, v), the sum of positions must be 3.
# Since positions are 1 or 2, this means if machine u is at 1, v must be at 2,
# and if u is at 2, v must be at 1.
# This is equivalent to saying u and v must have different positions.
# This is a standard 2-coloring problem on a graph.
# We want the lexicographically smallest sequence of positions (1s and 2s).
# To minimize the sequence, we prefer 1 over 2 for each machine.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component.
            # To get the lexicographically smallest sequence, we prefer 1.
            # If a node is uncolored, assign it 1.
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
