import sys

# The condition: "exactly one path from (0,0) to (u,v) that never visits (1,1)"
# Paths from (0,0) to (u,v) are sequences of E and N steps. Total paths: (u+v)! / (u!v!)
# A path visits (1,1) if it goes (0,0) -> (1,0) -> (1,1) or (0,0) -> (0,1) -> (1,1).
# If u=0 or v=0, the path is unique and never visits (1,1) unless u=v=0 (not possible as values are positive).
# If u=1, v=1: Paths are EN, NE. Both visit (1,1). Count = 0.
# If u=1, v>1: Paths are E N...N (1 path) and N...N E (v paths).
#   Paths avoiding (1,1): Only the path that goes (0,0)->(0,1)->(0,2)...->(0,v)->(1,v).
#   This is 1 path. So u=1, v>1 is valid.
# If u>1, v=1: Symmetric to u=1, v>1. Valid.
# If u>1, v>1: Paths avoiding (1,1) are those that don't pass through (1,1).
#   Total paths = (u+v)! / (u!v!)
#   Paths through (1,1) = (Paths (0,0)->(1,1)) * (Paths (1,1)->(u,v))
#   Paths (0,0)->(1,1) = 2.
#   Paths (1,1)->(u,v) = (u-1+v-1)! / ((u-1)!(v-1)!) = (u+v-2)! / ((u-1)!(v-1)!)
#   Condition: Total - 2 * (u+v-2)! / ((u-1)!(v-1)!) = 1
#   For u=2, v=2: Total 6, 2 * (2!/1!1!) = 4. 6-4 = 2 != 1.
#   For u=2, v=3: Total 10, 2 * (3!/1!2!) = 6. 10-6 = 4 != 1.
#   Actually, the only solutions are (1, v) for v > 1, (u, 1) for u > 1.
#   This means for every edge (u, v), one must be 1 and the other must be > 1.
#   This is a bipartite graph condition.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # To get lexicographically smallest, we want 1s where possible.
            # In each component, we have two sets: color 0 and color 1.
            # We can assign 1 to color 0 and 2 to color 1, or vice versa.
            # Compare the two options lexicographically.
            comp_nodes = sorted([u for u, c in component])
            
            # Option 1: color 0 -> 1, color 1 -> 2
            # Option 2: color 0 -> 2, color 1 -> 1
            opt1 = {}
            opt2 = {}
            for u, c in component:
                opt1[u] = 1 if c == 0 else 2
                opt2[u] = 2 if c == 0 else 1
            
            # Compare
            is_opt1_better = False
            for node in comp_nodes:
                if opt1[node] < opt2[node]:
                    is_opt1_better = True
                    break
                elif opt1[node] > opt2[node]:
                    is_opt1_better = False
                    break
            
            best = opt1 if is_opt1_better else opt2
            for node in comp_nodes:
                res[node] = best[node]
                
    print("YES")
    print(*(res[1:]))

solve()
