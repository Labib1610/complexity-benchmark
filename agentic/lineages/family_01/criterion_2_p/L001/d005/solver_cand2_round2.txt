import sys

# The condition r_u^3 + r_v^3 = 9 with r_i in {1, 2}
# 1^3 = 1, 2^3 = 8.
# 1 + 8 = 9.
# So for every bond (u, v), one must be 1 and the other must be 2.
# This is equivalent to a 2-coloring problem on a graph.
# A graph is 2-colorable if and only if it is bipartite.
# We need the lexicographically smallest sequence (1, 2, ...).
# This means for each connected component, we try to set the first node to 1.
# If that fails, we set it to 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try setting color[i] = 1
            component = []
            stack = [(i, 1)]
            color[i] = 1
            possible = True
            
            # BFS/DFS to color component
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Try setting color[i] = 2
                for node in q:
                    color[node] = 0
                
                q = [i]
                color[i] = 2
                head = 0
                possible = True
                while head < len(q):
                    u = q[head]
                    head += 1
                    for v in adj[u]:
                        if color[v] == 0:
                            color[v] = 3 - color[u]
                            q.append(v)
                        elif color[v] == color[u]:
                            possible = False
                            break
                    if not possible:
                        break
            
            if not possible:
                print("NO")
                return

    print("YES")
    print(*(color[1:]))

solve()
