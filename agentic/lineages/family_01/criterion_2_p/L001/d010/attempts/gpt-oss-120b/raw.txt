import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    g = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        g[u].append(v)
        g[v].append(u)

    color = [-1] * (n + 1)          # 0 / 1
    flip = [0] * (n + 1)            # flip per component (store at root)
    comp_nodes = []                # list of nodes per component (for later assignment)

    from collections import deque
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        nodes = [start]
        ok = True
        while q:
            u = q.popleft()
            for v in g[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                    nodes.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
        # decide orientation: smallest index gets mode 1
        min_node = min(nodes)
        # we need after possible flip, its final color be 0
        # final_color = color[node] ^ flip
        # so flip = color[min_node]
        f = color[min_node]
        for v in nodes:
            flip[v] = f
        comp_nodes.append(nodes)

    # build answer
    ans = [0] * (n + 1)
    for i in range(1, n + 1):
        final_col = color[i] ^ flip[i]
        ans[i] = 1 if final_col == 0 else 2

    print("YES")
    print(' '.join(str(ans[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
