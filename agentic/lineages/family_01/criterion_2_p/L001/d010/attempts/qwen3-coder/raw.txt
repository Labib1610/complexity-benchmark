import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    V1 = [3, 1, 4, 2]
    V2 = [2, 4, 1, 3]
    
    # Precompute dot products
    dot11 = sum(V1[i] * V1[i] for i in range(4))  # 3*3 + 1*1 + 4*4 + 2*2 = 9+1+16+4 = 30
    dot12 = sum(V1[i] * V2[i] for i in range(4))  # 3*2 + 1*4 + 4*1 + 2*3 = 6+4+4+6 = 20
    dot21 = dot12  # 20
    dot22 = sum(V2[i] * V2[i] for i in range(4))  # 2*2 + 4*4 + 1*1 + 3*3 = 4+16+1+9 = 30
    
    # The constraint is that synchronized pairs must have dot product 20
    # So:
    # Mode 1 with Mode 1: 30 (invalid)
    # Mode 1 with Mode 2: 20 (valid)
    # Mode 2 with Mode 1: 20 (valid)
    # Mode 2 with Mode 2: 30 (invalid)
    # So two filters in a sync pair must have different modes.
    
    n, m = map(int, sys.stdin.readline().split())
    
    # Build graph
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color the graph with 2 colors (1 and 2)
    # -1: unvisited, 1: color 1, 2: color 2
    color = [-1] * (n + 1)
    visited = [False] * (n + 1)
    
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 1
        visited[start] = True
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    color[v] = 3 - color[u]  # Flip color: 1->2, 2->1
                    queue.append(v)
                else:
                    # Check for consistency
                    if color[v] == color[u]:
                        return False
        return True
    
    # Process each connected component
    for i in range(1, n + 1):
        if not visited[i]:
            if not bfs(i):
                print("NO")
                return
    
    # If we reach here, a valid coloring exists
    # Output the lexicographically smallest assignment
    # Since we start with color 1 for each component's first node,
    # and we only flip when needed, this should already be lex smallest
    # But to ensure lex smallest, we should start each component with color 1
    
    # Actually, let's re-implement to ensure lex smallest:
    # For each connected component, we can choose to start with 1 or 2.
    # To get lex smallest, we try starting with 1. If that leads to a valid coloring, great.
    # If not, we would need to try starting with 2, but that would make the first node 2 instead of 1,
    # which is lexicographically larger. So starting with 1 for each component is correct for lex smallest.
    
    # But wait, our BFS already ensures that. The only choice is the initial color of each component.
    # To get lex smallest, we should always start a component with color 1.
    # Our current implementation does that.
    
    # So the current coloring should be correct.
    print("YES")
    result = []
    for i in range(1, n + 1):
        if color[i] == -1:
            result.append("1")  # Isolated node, can be either, choose 1 for lex smallest
        else:
            result.append(str(color[i]))
    print(" ".join(result))

if __name__ == "__main__":
    main()
