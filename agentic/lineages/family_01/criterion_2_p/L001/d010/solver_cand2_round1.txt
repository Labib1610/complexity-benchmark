```python
import sys

# Dot product calculation:
# V1 = [3, 1, 4, 2]
# V2 = [2, 4, 1, 3]
# V1 dot V1 = 3*3 + 1*1 + 4*4 + 2*2 = 9 + 1 + 16 + 4 = 30
# V2 dot V2 = 2*2 + 4*4 + 1*1 + 3*3 = 4 + 16 + 1 + 9 = 30
# V1 dot V2 = 3*2 + 1*4 + 4*1 + 2*3 = 6 + 4 + 4 + 6 = 20
# V2 dot V1 = 20
#
# Condition: dot product must be 20.
# If u and v are connected, they must have different modes (one 1, one 2).
# This is a 2-coloring problem on a graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we prefer mode 1 (color 1)
            # for the smallest index in the component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
