import sys

# The problem states:
# 1. Number of divisors of x equals x.
#    Divisors(x) = x.
#    For x=1: divisors are {1}, count=1. Correct.
#    For x=2: divisors are {1, 2}, count=2. Correct.
#    For x=3: divisors are {1, 3}, count=2 != 3.
#    For x > 2, the number of divisors is strictly less than x.
#    So, possible values are {1, 2}.
# 2. For any baseline (u, v), u^3 + v^3 = 9.
#    If u=1, v^3 = 8 => v=2.
#    If u=2, v^3 = 1 => v=1.
#    So, every baseline must connect a dish with value 1 to a dish with value 2.
#    This means the graph must be bipartite.
# 3. We need the lexicographically smallest sequence.
#    For each connected component, we have two choices:
#    - Assign 1 to one partition and 2 to the other.
#    - Assign 2 to the first partition and 1 to the second.
#    To get the lexicographically smallest sequence, for each component,
#    we look at the smallest index 'i' in that component.
#    We prefer assigning 1 to 'i' if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1) # 0: unvisited, 1: value 1, 2: value 2
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            is_bipartite = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find smallest index in component
            min_node = min(component)
            # If we set color[min_node] = 1, then all nodes with color 1 get value 1, color 2 get value 2
            # If we set color[min_node] = 2, then all nodes with color 1 get value 2, color 2 get value 1
            # We want the smallest value at min_node.
            # If color[min_node] is 1, we can assign 1 to min_node.
            # If color[min_node] is 2, we can assign 2 to min_node.
            # To make it lexicographically smallest, we want 1 at min_node.
            # If color[min_node] == 1, we keep the mapping (1->1, 2->2).
            # If color[min_node] == 2, we swap the mapping (1->2, 2->1).
            
            if color[min_node] == 1:
                for node in component:
                    ans[node] = color[node]
            else:
                for node in component:
                    ans[node] = 3 - color[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
