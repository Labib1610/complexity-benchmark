```python
import sys

# The condition u*r_u + v*r_v != u*r_v + v*r_u
# Rearranging: u*r_u - u*r_v - v*r_u + v*r_v != 0
# u(r_u - r_v) - v(r_u - r_v) != 0
# (u - v)(r_u - r_v) != 0
# Since u != v (assuming no self-loops), this simplifies to:
# r_u - r_v != 0 => r_u != r_v
# This is a standard 2-coloring problem (bipartite graph check).
# We want the lexicographically smallest sequence (r_1, r_2, ..., r_n).
# To get the smallest sequence, we greedily try r_i = 1 for each component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if res[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer 1 over 2
            stack = [i]
            res[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if res[v] == 0:
                        res[v] = 3 - res[u]
                        stack.append(v)
                    elif res[v] == res[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(res[1:]))

if __name__ == '__main__':
    solve()
```
