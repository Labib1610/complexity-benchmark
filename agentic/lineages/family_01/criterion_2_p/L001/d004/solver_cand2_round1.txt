```python
import sys

# The polynomial f^3 - 3f^2 + 2f = 0 factors as f(f-1)(f-2) = 0.
# Since f must be strictly positive, f can be 1 or 2.
# The system of equations:
# f_u * x + y = 0
# 2 * x + f_v * y = 0
# This is a linear system with matrix [[f_u, 1], [2, f_v]].
# It has a non-zero solution (x, y) if and only if the determinant is 0:
# f_u * f_v - 2 = 0 => f_u * f_v = 2.
# Since f_u, f_v in {1, 2}, the only way f_u * f_v = 2 is:
# (f_u=1, f_v=2) or (f_u=2, f_v=1).
# This means the graph must be bipartite, and for every edge (u, v),
# f_u must be different from f_v.
# Since we want the lexicographically smallest sequence, we want f_i = 1
# whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # 0: unassigned, 1: frequency 1, 2: frequency 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 2 if c == 1 else 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
