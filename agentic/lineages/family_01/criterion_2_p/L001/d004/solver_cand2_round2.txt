```python
import sys

# The problem asks for a 2-coloring of a graph such that every edge connects
# vertices of different colors (1 and 2). This is equivalent to checking if
# the graph is bipartite. To find the lexicographically smallest assignment,
# we iterate from 1 to n. If a node is unvisited, we assign it 1 (the smallest
# possible value) and perform a BFS/DFS to color its connected component.
# Since we want the lexicographically smallest sequence, we must ensure that
# for each connected component, we pick the coloring that results in the
# smallest values. However, in a bipartite graph, there are only two possible
# colorings for a connected component (swap 1s and 2s). We choose the one
# that makes the first node encountered in the component 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we assign 1 to the first node
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
