```python
import sys

# The equation x^3 - 3x^2 + 2x = 0 factors to x(x-1)(x-2) = 0.
# The positive real roots are 1 and 2.
# The entanglement rule: sum of squares = 5.
# If qubit u has state x_u and qubit v has state x_v, then x_u^2 + x_v^2 = 5.
# Possible states are {1, 2}.
# 1^2 + 1^2 = 2 != 5
# 2^2 + 2^2 = 8 != 5
# 1^2 + 2^2 = 5
# Thus, for every link (u, v), one qubit must be 1 and the other must be 2.
# This is a bipartite graph coloring problem.
# We need to assign 1 or 2 to each node such that connected nodes have different values.
# To get the lexicographically smallest sequence, we want to assign 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # 0: unassigned, 1: state 1, 2: state 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we prefer 1 for the smallest index
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
