import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # We need to assign 1 or 2 to each robot.
    # Constraint: |x_u - x_v| = 1.
    # This is equivalent to 2-coloring a graph where 1 and 2 are the colors.
    # To get the lexicographically smallest sequence, we want x_i = 1 whenever possible.
    # For each connected component, we can pick the color of the smallest index node.
    # If we set the smallest index node to 1, the rest of the component is fixed.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we want the smallest index
            # in this component to be 1.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process nodes
            comp_nodes = []
            q = [i]
            color[i] = 1
            comp_nodes.append(i)
            
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        comp_nodes.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring, check if this component's smallest index is 1.
            # If color[min_index] is 2, we flip all colors in this component.
            min_idx = min(comp_nodes)
            if color[min_idx] == 2:
                for node in comp_nodes:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
