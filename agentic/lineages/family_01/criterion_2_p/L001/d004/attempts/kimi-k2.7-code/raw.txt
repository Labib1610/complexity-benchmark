 ```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        if u != v:
            adj[v].append(u)

    color = [-1] * (n + 1)
    q = deque()

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        q.append(start)
        color[start] = 0
        comp = []
        while q:
            u = q.popleft()
            comp.append(u)
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    q.append(v)
                elif color[v] == cu:
                    sys.stdout.write("NO\n")
                    return
        minv = min(comp)
        if color[minv] == 1:
            for v in comp:
                color[v] ^= 1

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(str(color[i] + 1) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
