import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    parent = [0] * (n + 1)
    depth = [0] * (n + 1)
    order = []
    failed = False
    
    for s in range(1, n + 1):
        if color[s] != 0:
            continue
        color[s] = 1
        parent[s] = 0
        depth[s] = 0
        stack = [s]
        while stack:
            u = stack.pop()
            order.append(u)
            cu = color[u]
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - cu
                    parent[v] = u
                    depth[v] = depth[u] + 1
                    stack.append(v)
                elif color[v] == cu:
                    nonlocal failed
                    failed = True
                    break
            if failed:
                break
        if failed:
            break
    
    if failed:
        sys.stdout.write("NO\n")
        return
    
    flip = [False] * (n + 1)
    for u in order:
        if flip[u]:
            color[u] = 3 - color[u]
        for v in adj[u]:
            if parent[v] == u:
                if color[u] == 1:
                    flip[v] = flip[v] ^ True
    
    out = ['YES']
    out.append(' '.join(str(color[i]) for i in range(1, n + 1)))
    sys.stdout.write('\n'.join(out) + '\n')

solve()
