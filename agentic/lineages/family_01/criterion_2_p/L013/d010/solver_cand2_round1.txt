```python
import sys

# The problem states W_i = cos(pi * p_i).
# If p_i = 1, W_i = cos(pi) = -1.
# If p_i = 2, W_i = cos(2*pi) = 1.
# The condition W_u + W_v = 0 means one must be -1 and the other 1.
# This is equivalent to saying p_u != p_v.
# This is a standard 2-coloring problem on a graph.
# We want the lexicographically smallest sequence p_1, p_2, ..., p_n.
# To make the sequence lexicographically smallest, we prefer p_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unassigned, 1 = p_i=1, 2 = p_i=2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 first
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
