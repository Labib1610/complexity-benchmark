import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    d = [0] * (n + 1)
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        d[u] += 1
        d[v] += 1
        edges.append((u, v))
        idx += 2
        
    # Equation: 3 * sum(d_i * p_i) - 2 * sum(p_u * p_v) = 5 * m
    # Let x_i = p_i - 1, so x_i in {0, 1}. p_i = x_i + 1
    # 3 * sum(d_i * (x_i + 1)) - 2 * sum((x_u + 1) * (x_v + 1)) = 5 * m
    # 3 * sum(d_i * x_i) + 3 * sum(d_i) - 2 * sum(x_u * x_v + x_u + x_v + 1) = 5 * m
    # Note: sum(d_i) = 2 * m
    # 3 * sum(d_i * x_i) + 6 * m - 2 * sum(x_u * x_v) - 2 * sum(x_u + x_v) - 2 * m = 5 * m
    # 3 * sum(d_i * x_i) - 2 * sum(x_u * x_v) - 2 * sum(d_i * x_i) = m
    # sum(d_i * x_i) - 2 * sum(x_u * x_v) = m
    
    # We want lexicographically smallest p_i, which means smallest x_i.
    # Try all x_i = 0 (all p_i = 1): sum(d_i * 0) - 2 * sum(0) = 0. 
    # If m == 0, this works.
    
    # Since we want lexicographically smallest, we want as many 0s as possible at the start.
    # This is equivalent to finding if there exists a subset of vertices S (where x_i=1)
    # such that sum_{i in S} d_i - 2 * (number of edges with both endpoints in S) = m.
    
    # This is a classic problem: the expression sum_{i in S} d_i - 2 * |E(S)|
    # is exactly the number of edges with exactly one endpoint in S (the cut size).
    # So we need to find a cut (S, V\S) such that the number of edges crossing the cut is m.
    # The number of edges crossing the cut is always <= m.
    # It equals m if and only if there are no edges within S and no edges within V\S.
    # This means the graph must be bipartite, and the cut must be the bipartition.
    
    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, we have two sets: S0 = {i | color[i] == 0}, S1 = {i | color[i] == 1}
    # Let x_i = color[i]. Then sum(d_i * x_i) - 2 * sum(x_u * x_v) = m
    # If we pick S = S1, then x_i = 1 if i in S1, 0 otherwise.
    # The cut size is exactly m.
    # We need to check if this assignment works.
    
    def check(assignment):
        w = sum(d[i] * assignment[i-1] for i in range(1, n + 1))
        u_val = sum(assignment[u-1] * assignment[v-1] for u, v in edges)
        return 3 * w - 2 * u_val == 5 * m

    # Try S = S1 (x_i = color[i])
    res1 = [color[i] + 1 for i in range(1, n + 1)]
    # Try S = S0 (x_i = 1 - color[i])
    res2 = [2 - color[i] for i in range(1, n + 1)]
    
    valid1 = check(res1)
    valid2 = check(res2)
    
    if not valid1 and not valid2:
        print("NO")
    else:
        if valid1 and valid2:
            print("YES")
            print(*(res1 if res1 < res2 else res2))
        elif valid1:
            print("YES")
            print(*res1)
        else:
            print("YES")
            print(*res2)

solve()
