```python
import sys

# Rule 1: f1(x) = -x
# Rule 2: f2(x) = 1/x
# Composition:
# f1(f1(x)) = -(-x) = x
# f1(f2(x)) = -1/x
# f2(f1(x)) = 1/(-x) = -1/x
# f2(f2(x)) = 1/(1/x) = x
#
# Stability condition: f_u(f_v(x)) = -1/x
# This means one node must be Rule 1 and the other must be Rule 2.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence (s1, s2, ..., sn).
# This means we prefer 1 over 2 for each node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unassigned, 1 = Rule 1, 2 = Rule 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 to the smallest node
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
