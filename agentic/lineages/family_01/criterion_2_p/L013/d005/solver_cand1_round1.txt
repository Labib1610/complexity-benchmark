```python
import sys

# The condition Tr(Au * Av) = 0:
# M1 = [[1, 0], [0, -1]], M2 = [[0, 1], [1, 0]]
# M1*M1 = [[1, 0], [0, 1]], Tr = 2
# M1*M2 = [[0, 1], [-1, 0]], Tr = 0
# M2*M1 = [[0, -1], [1, 0]], Tr = 0
# M2*M2 = [[1, 0], [0, 1]], Tr = 2
# Thus, Tr(Au * Av) = 0 if and only if Au != Av.
# This is a 2-coloring problem: adjacent nodes must have different states.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # colors: 0 = unassigned, 1 = state 1, 2 = state 2
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # To get lexicographically smallest, try state 1 first
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
