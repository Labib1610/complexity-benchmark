```python
import sys

# The phase matrix P = [[x, 2], [2, y]] has eigenvalues determined by the characteristic equation:
# det(P - lambda*I) = (x - lambda)(y - lambda) - 4 = 0
# lambda^2 - (x + y)lambda + (xy - 4) = 0
# The eigenvalues are irrational if and only if the discriminant D = (x + y)^2 - 4(xy - 4) is not a perfect square.
# D = x^2 + 2xy + y^2 - 4xy + 16 = x^2 - 2xy + y^2 + 16 = (x - y)^2 + 16.
# Possible values for x, y in {1, 2}:
# If x=1, y=1: D = (1-1)^2 + 16 = 16 = 4^2 (Perfect square, rational eigenvalues)
# If x=2, y=2: D = (2-2)^2 + 16 = 16 = 4^2 (Perfect square, rational eigenvalues)
# If x=1, y=2: D = (1-2)^2 + 16 = 1 + 16 = 17 (Not a perfect square, irrational eigenvalues)
# If x=2, y=1: D = (2-1)^2 + 16 = 1 + 16 = 17 (Not a perfect square, irrational eigenvalues)
# Thus, a tether is balanced if and only if x != y.
# This is a bipartite coloring problem. We need to color nodes with 1 or 2 such that
# for every edge (u, v), color[u] != color[v].
# This is possible if and only if the graph is bipartite.
# To minimize lexicographically, we greedily assign 1 to the first node of each component,
# then propagate the bipartite coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
        else:
            print("NO")
            return
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color component
            # To be lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
