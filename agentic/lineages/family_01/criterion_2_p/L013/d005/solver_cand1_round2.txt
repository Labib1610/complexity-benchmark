```python
import sys

# The problem asks for s_i in {1, 2} such that s_u * s_v = 2 for all constraints (u, v).
# Since s_i can only be 1 or 2, s_u * s_v = 2 implies that one must be 1 and the other 2.
# This is a bipartite coloring problem.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we want s_1 to be 1 if possible.
# If s_1 = 1, then all neighbors of 1 must be 2, their neighbors must be 1, and so on.
# We perform a BFS/DFS on each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we prefer color 1 for the smallest index
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, color[v]))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
