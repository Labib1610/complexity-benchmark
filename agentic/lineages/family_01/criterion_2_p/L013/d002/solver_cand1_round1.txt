```python
import sys

# The problem states:
# Alpha state (from 1): 1 -> Low-Halt, 2 -> Stable
# Beta state (from 2): 1 -> Stable, 2 -> High-Halt
# Stable state is reached if:
# (Field1=1 and Field2=2) OR (Field1=2 and Field2=1)
# This means Field1 != Field2.
# This is a 2-coloring problem on a graph.
# We need to check if the graph is bipartite.
# If bipartite, we want the lexicographically smallest configuration.
# For each connected component, we can assign the first node to 1,
# then propagate the constraints. If we have a choice, we pick 1 for the
# smallest index to keep it lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            ans[i] = 1
            
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        ans[v] = color[v]
                        stack.append((v, color[v]))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(ans[1:]))

solve()
```
