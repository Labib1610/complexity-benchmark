import sys

# The problem states that for each cross-beam (u, v), the sum of dampeners must be 3.
# Since dampeners are 1 or 2, this means one must be 1 and the other must be 2.
# This is equivalent to saying the graph is bipartite, and for each edge (u, v),
# the dampener at u must be different from the dampener at v (if we map 1 -> 0, 2 -> 1).
# Specifically, if we assign 1 to one set of the bipartition and 2 to the other,
# the sum will always be 1+2=3.
# To get the lexicographically smallest sequence, we want s_1 to be 1 if possible.
# If s_1 = 1, then all nodes in its connected component are fixed.
# If a component is not yet assigned, we try assigning 1 to the smallest index in it.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To make it lexicographically smallest, we want the first node
            # in the component to be 1.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a stack for DFS
            q = [i]
            color[i] = 1
            comp = [i]
            head = 0
            while head < len(comp):
                u = comp[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        comp.append(v)
                    elif color[v] == color[u]:

    print("YES")
    res = []
    for i in range(1, n + 1):
        res.append(str(color[i]))
    print(" ".join(res))

solve()
