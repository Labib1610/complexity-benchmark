```python
import sys

# The problem states that for each comparator (u, v), one field must be 
# exactly twice the other. Since the only allowed positions are 1 and 2,
# this means if one is 1, the other must be 2.
# This is equivalent to saying s_u != s_v.
# This is a bipartite graph coloring problem.
# We need to check if the graph is bipartite. If it is, we want the 
# lexicographically smallest configuration.
# For each connected component, we can choose the color of the first node 
# (the one with the smallest index) to be 1, which forces the colors of 
# all other nodes in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we always set the 
            # smallest index in the component to 1.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_c
                        stack.append((v, next_c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
