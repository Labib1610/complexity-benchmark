```python
import sys

# The problem states that for each monitor (u, v), the bitwise AND of their settings must be 0.
# Settings are 1 (binary 01) or 2 (binary 10).
# 1 & 1 = 1 (non-zero)
# 2 & 2 = 2 (non-zero)
# 1 & 2 = 0
# 2 & 1 = 0
# This means if there is an edge between u and v, they must have different settings.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest sequence of 1s and 2s.
# To minimize the sequence lexicographically, we want s_1 = 1, then s_2 = 1 if possible, etc.
# Actually, for each connected component, we have two choices:
# Choice A: Assign color 1 to set U and color 2 to set V.
# Choice B: Assign color 2 to set U and color 1 to set V.
# We want to pick the choice that makes the sequence lexicographically smallest.
# For each component, find the smallest index i in that component.
# If we set s_i = 1, the rest of the component is determined.
# If we set s_i = 2, the rest of the component is determined.
# We compare the two resulting sequences for that component and pick the better one.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible assignments for this component:
            # 1. color[x] == 1 -> s[x] = 1, color[x] == 2 -> s[x] = 2
            # 2. color[x] == 1 -> s[x] = 2, color[x] == 2 -> s[x] = 1
            
            # Find the smallest index in this component
            comp_nodes = sorted(component)
            first = comp_nodes[0]
            
            # Option 1: s[first] = 1
            # If color[first] == 1, then map 1->1, 2->2
            # If color[first] == 2, then map 2->1, 1->2
            
            # Option 2: s[first] = 2
            # If color[first] == 1, then map 1->2, 2->1
            # If color[first] == 2, then map 2->2, 1->1
            
            # We want to compare the sequences generated by these two options.
            # Actually, we just need to pick the one that gives a smaller value at the first index.
            # Since we want the lexicographically smallest, we always prefer s[first] = 1.
            # So we set the mapping such that s[first] = 1.
            
            mapping = {}
            if color[first] == 1:
                mapping = {1: 1, 2: 2}
            else:
                mapping = {1: 2, 2: 1}
                
            for node in component:
                res[node] = mapping[color[node]]
                
    print("YES")
    print(*(res[1:]))

solve()
```
