```python
import sys

# The problem states that for each constraint (u, v), the product of their 
# frequency multipliers must be 2. Since multipliers are in {1, 2}, 
# the only way for the product to be 2 is if one is 1 and the other is 2.
# This is a bipartite coloring problem.
# We want the lexicographically smallest sequence, meaning we prefer 1 over 2
# for smaller indices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unvisited, 1 = multiplier 1, 2 = multiplier 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, we try assigning 1 to the 
            # first node of each connected component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 2 if c == 1 else 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
