import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Protocol 1: 1->3, 2->5, 3->4, 4->2, 5->1
    # Permutation P1: [5, 4, 1, 3, 2] (0-indexed: 0->2, 1->4, 2->3, 3->1, 4->0)
    p1 = [2, 4, 3, 1, 0]
    # Protocol 2: 1->5, 2->4, 3->1, 4->3, 5->2
    # Permutation P2: [3, 5, 4, 2, 1] (0-indexed: 0->4, 1->3, 2->0, 3->2, 4->1)
    p2 = [4, 3, 0, 2, 1]

    def compose(p_a, p_b):
        return [p_b[p_a[i]] for i in range(5)]

    identity = [0, 1, 2, 3, 4]
    
    # Check if P_i composed with P_j is identity
    # Transparent if P_i(P_j(x)) = x for all x
    # Since these are permutations, P_i(P_j(x)) = x iff P_j = P_i^-1
    # P1^-1: 0->4, 1->3, 2->0, 3->2, 4->1 (which is P2)
    # P2^-1: 0->2, 1->4, 2->3, 3->1, 4->0 (which is P1)
    # So transparent if (P1, P2) or (P2, P1)
    
    # This is a 2-coloring problem:
    # If u and v are connected, they must have different protocols.
    # If the graph is bipartite, we can color it.
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # To get lexicographically smallest, for each component,
            # we have two choices: (start with 1) or (start with 2).
            # We pick the one that makes the smallest node index have 1.
            comp_nodes = sorted([u for u, c in component])
            first_node = comp_nodes[0]
            
            # If we set color[first_node] = 1, check the sequence
            # If we set color[first_node] = 2, check the sequence
            # Actually, just set color[first_node] = 1 and propagate.
            # Since it's bipartite, the coloring is fixed once one node is set.
            # We need to ensure the smallest node in the component gets 1 if possible.
            # Wait, the coloring is fixed relative to the first node.
            # If we set color[first_node] = 1, all others are determined.
            # If we set color[first_node] = 2, all others are flipped.
            # We compare the two resulting sequences for this component.
            
            c1 = {}
            c2 = {}
            for u, c in component:
                c1[u] = c
                c2[u] = 3 - c
            
            seq1 = [c1[u] for u in comp_nodes]
            seq2 = [c2[u] for u in comp_nodes]
            
            if seq1 < seq2:
                for u in comp_nodes: color[u] = c1[u]
            else:
                for u in comp_nodes: color[u] = c2[u]

    print("YES")
    print(*(color[1:]))

solve()
