import sys
from collections import deque

def solve():
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr]); ptr += 1
    m = int(input[ptr]); ptr += 1
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input[ptr]); ptr += 1
        v = int(input[ptr]); ptr += 1
        adj[u].append(v)
        adj[v].append(u)
    
    # We want lexicographically smallest assignment.
    # Operators are 1 and 2. Constraint: adjacent nodes must have different operators.
    # To minimize lexicographically, we prefer assigning 1 to all nodes if possible.
    # This is exactly bipartite graph coloring with colors 1 and 2, minimizing lexicographic order.
    # Strategy: BFS from each unvisited node, assigning 1 to the start node.
    
    color = [0] * (n + 1)
    possible = True
    
    for i in range(1, n + 1):
        if color[i] == 0:
            q = deque([i])
            color[i] = 1
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break
    
    if not possible:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, color[1:])))

if __name__ == "__main__":
    solve()
