import sys
from collections import defaultdict

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        # No dependencies, any assignment works. Maximize confidence: set all to 1
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    adj = [[] for _ in range(n)]
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # For each edge (u,v), the stability index is p_u * p_v / (p_u + p_v)^2
    # We want sum over all edges = 2m/9
    # So average per edge is 2/9
    
    # Let's see what values p_u * p_v / (p_u + p_v)^2 can take:
    # If p_u=1, p_v=1: 1*1/(1+1)^2 = 1/4
    # If p_u=1, p_v=2: 1*2/(1+2)^2 = 2/9
    # If p_u=2, p_v=1: 2*1/(2+1)^2 = 2/9
    # If p_u=2, p_v=2: 2*2/(2+2)^2 = 4/16 = 1/4
    
    # So we need the average to be 2/9. That means every edge should contribute 2/9.
    # That means for every edge, one endpoint should be 1 and the other 2.
    # I.e., the graph must be bipartite and we assign 1 to one side, 2 to the other.
    
    # Check if the graph is bipartite
    color = [-1] * n  # -1 unvisited, 0 or 1 are colors
    
    def is_bipartite(start):
        stack = [start]
        color[start] = 0
        while stack:
            u = stack.pop()
            c = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 1 - c
                    stack.append(v)
                elif color[v] == c:
                    return False
        return True
    
    # Check all components
    for i in range(n):
        if color[i] == -1:
            if not is_bipartite(i):
                print("NO")
                return
    
    # Graph is bipartite. Now we need to choose the coloring that maximizes confidence.
    # Confidence = sum (2 - p_i) / i!
    # So for node i, if we put p_i = 1, we get (2-1)/i! = 1/i!
    #              if we put p_i = 2, we get (2-2)/i! = 0
    # So we want to put p_i = 1 for nodes where we can, especially those with small i.
    
    # In a bipartite graph with parts A and B, we can either:
    # - assign 1 to A and 2 to B, or
    # - assign 1 to B and 2 to A
    # We choose the one that gives higher confidence.
    
    # Compute confidence for both options
    fact = [1] * (n + 1)
    for i in range(1, n + 1):
        fact[i] = fact[i - 1] * i
    
    conf_A_is_1 = 0  # A gets 1, B gets 2
    conf_B_is_1 = 0  # B gets 1, A gets 2
    
    for i in range(n):
        if color[i] == 0:
            conf_A_is_1 += 1 / fact[i + 1]  # node i (0-indexed) gets 1
        else:
            conf_B_is_1 += 1 / fact[i + 1]  # node i gets 1
    
    # Decide which assignment to use
    assign_1_to_color_0 = conf_A_is_1 >= conf_B_is_1
    
    result = [0] * n
    for i in range(n):
        if (color[i] == 0 and assign_1_to_color_0) or (color[i] == 1 and not assign_1_to_color_0):
            result[i] = 1
        else:
            result[i] = 2
    
    print("YES")
    print(" ".join(map(str, result)))

if __name__ == "__main__":
    main()
