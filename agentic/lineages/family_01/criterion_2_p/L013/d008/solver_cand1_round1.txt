```python
import sys

# The condition for self-adjointness of T_u o T_v:
# T1 = D + I, T2 = D - I
# T1* = -D + I, T2* = -D - I
# (T1 o T1)* = T1* o T1* = (-D + I)(-D + I) = D^2 - 2D + I
# (T1 o T1) = (D + I)(D + I) = D^2 + 2D + I
# For T1 o T1 to be self-adjoint, D^2 - 2D + I = D^2 + 2D + I => 4D = 0, impossible.
# (T2 o T2)* = T2* o T2* = (-D - I)(-D - I) = D^2 + 2D + I
# (T2 o T2) = (D - I)(D - I) = D^2 - 2D + I
# For T2 o T2 to be self-adjoint, D^2 + 2D + I = D^2 - 2D + I => 4D = 0, impossible.
# (T1 o T2)* = T2* o T1* = (-D - I)(-D + I) = D^2 - D + D - I = D^2 - I
# (T1 o T2) = (D + I)(D - I) = D^2 - I
# This is always self-adjoint.
# Thus, a link (u, v) is stable if and only if the operators assigned to u and v are different.
# This is a 2-coloring problem on a graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, we prefer 1 over 2.
            # For each connected component, try assigning 1 to the first node.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
