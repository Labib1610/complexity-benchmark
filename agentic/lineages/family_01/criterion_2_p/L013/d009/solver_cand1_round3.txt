```python
import sys

# The problem asks for a 2-coloring of a graph where nodes are 1..n.
# A conflict cell (u, v) means row u and column v must have different phases.
# Since row i and column i share the same phase p_i, the condition is:
# p_u != p_v.
# This is a standard bipartite graph coloring problem.
# We want the lexicographically smallest sequence p_1, p_2, ..., p_n.
# This means we want p_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate i from 1 to n.
    # If node i is not colored, we try to color it with 1.
    # This is a greedy approach on connected components.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            stack = [i]
            color[i] = 1
            component = []
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
