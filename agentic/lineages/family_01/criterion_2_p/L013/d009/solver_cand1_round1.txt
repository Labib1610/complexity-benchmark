```python
import sys

# The stability function is f_k(x) = 6x - x^2 * (p_u/p_v + p_v/p_u + 2).
# Let A = (p_u/p_v + p_v/p_u + 2).
# f_k(x) = -A * x^2 + 6x.
# This is a downward parabola with vertex at x = -6 / (2 * -A) = 3/A.
# The maximum value is f_k(3/A) = -A * (9/A^2) + 6 * (3/A) = -9/A + 18/A = 9/A.
# The stability index is |9/A|. Since A > 0, it is 9/A.
# A = (p_u^2 + p_v^2) / (p_u * p_v) + 2 = (p_u^2 + p_v^2 + 2*p_u*p_v) / (p_u*p_v) = (p_u + p_v)^2 / (p_u*p_v).
# Stability index = 9 / ((p_u + p_v)^2 / (p_u*p_v)) = 9 * p_u * p_v / (p_u + p_v)^2.
# If p_u = 1, p_v = 1: index = 9 * 1 / 4 = 2.25
# If p_u = 1, p_v = 2: index = 9 * 2 / 9 = 2
# If p_u = 2, p_v = 2: index = 9 * 4 / 16 = 2.25
# We need sum of indices = 2m.
# Let m1 be the number of edges with (1, 2) or (2, 1), m2 be edges with (1, 1), m3 with (2, 2).
# m1 + m2 + m3 = m.
# 2*m1 + 2.25*m2 + 2.25*m3 = 2m
# 2*m1 + 2.25*(m - m1) = 2m
# 2*m1 + 2.25*m - 2.25*m1 = 2m
# -0.25*m1 = -0.25*m
# m1 = m.
# This means every edge must connect a 1 and a 2.
# This is exactly the condition for a graph to be bipartite.
# If bipartite, we want to maximize sum (2-p_i)/i!.
# (2-p_i) is 1 if p_i=1, and 0 if p_i=2.
# So we want to maximize the number of 1s at smaller indices.
# In a bipartite graph, there are two colorings (swapping 1s and 2s).
# For each connected component, we can choose which partition is 1 and which is 2.
# To maximize the score, for each component, pick the partition that has the smaller index as 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            comp = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                comp.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find smallest index in component
            min_idx = min(u for u, c in comp)
            # If color of min_idx is 1, we keep it, else flip
            c_min = next(c for u, c in comp if u == min_idx)
            if c_min == 2:
                for u, c in comp:
                    color[u] = 3 - c
    
    print("YES")
    print(*(color[1:]))

solve()
```
