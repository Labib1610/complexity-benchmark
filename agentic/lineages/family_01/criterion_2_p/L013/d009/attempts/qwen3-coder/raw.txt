```python
import sys
from collections import defaultdict

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    # We'll build a graph where each node is a variable (1..n)
    # Each variable can take value 1 or 2
    # For each conflict (u,v), we have p[u] != p[v]
    # This is a 2-SAT kind of problem but simpler since each variable has only 2 values
    
    # For each pair (u,v) with u != v, we add constraint p[u] != p[v]
    # This means if p[u] = 1 then p[v] = 2 and vice versa
    
    adj = defaultdict(set)
    rev_adj = defaultdict(set)
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        if u == v:
            # p[u] != p[u] is impossible
            print("NO")
            return
        adj[u].add(v)
        adj[v].add(u)
        
    # Now we need to 2-color the graph (since p[i] can be 1 or 2)
    # If there's an odd cycle, it's not bipartite => NO solution
    # Otherwise, we get a valid coloring
    
    color = [-1] * n  # -1: unvisited, 0: color 1, 1: color 2
    
    def dfs(node, c):
        color[node] = c
        for nei in adj[node]:
            if color[nei] == -1:
                if not dfs(nei, 1 - c):
                    return False
            elif color[nei] == c:
                return False
        return True
    
    for i in range(n):
        if color[i] == -1:
            if not dfs(i, 0):
                print("NO")
                return
                
    # We want the lexicographically smallest assignment
    # That means we prefer p[i] = 1 over p[i] = 2
    # In our coloring, color 0 corresponds to p=1, color 1 to p=2
    # Since we started DFS with color 0, and always chose 0 when possible,
    # the result should already be lex smallest
    
    print("YES")
    print(' '.join(str(color[i] + 1) for i in range(n)))

if __name__ == "__main__":
    main()
```
