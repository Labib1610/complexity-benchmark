```python
import sys

# The condition: area of a right-angled triangle with legs a and b is (a * b) / 2.
# We need (a * b) / 2 = 1, so a * b = 2.
# Since a, b are in {1, 2}, the only way a * b = 2 is if one is 1 and the other is 2.
# This means for every tether (u, v), node u and node v must have different resonance tiers.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we prefer 1 over 2.
    # We iterate from 1 to n. If a node is uncolored, we assign it 1.
    # Then we perform BFS/DFS to color its connected component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
