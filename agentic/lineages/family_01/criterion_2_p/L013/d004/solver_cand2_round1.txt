```python
import sys

# The stability polynomial is P(x) = x^3 + ax^2 + bx + 1.
# By the Rational Root Theorem, any rational root must be a divisor of the constant term 1.
# Possible rational roots are 1 or -1.
# P(1) = 1 + a + b + 1 = a + b + 2.
# P(-1) = -1 + a - b + 1 = a - b.
# P(x) has a rational root if P(1) = 0 or P(-1) = 0.
# P(1) = 0 => a + b = -2. Since a, b in {1, 2}, a+b >= 2, so this is impossible.
# P(-1) = 0 => a = b.
# Thus, a tether is balanced if and only if a != b.
# This is a 2-coloring problem on a graph where edges represent the constraint a != b.
# We want the lexicographically smallest assignment (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, try assigning 1 first
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if this component's assignment is optimal.
            # Since we want the smallest sequence, we want the first node in the 
            # component to be 1 if possible. Our BFS/DFS approach naturally 
            # assigns 1 to the first uncolored node encountered.
            
    print("YES")
    print(*(color[1:]))

solve()
```
