import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    parent = [0] * (n + 1)
    is_back_edge = [False] * (n + 1)

    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        stack = [(start, 0, 0)]
        color[start] = 1
        parent[start] = 0
        while stack:
            u, p, idx = stack.pop()
            if idx == 0:
                parent[u] = p
            if idx < len(adj[u]):
                stack.append((u, p, idx + 1))
                v = adj[u][idx]
                if v == p:
                    continue
                if color[v] == 0:
                    color[v] = 1
                    stack.append((v, u, 0))
                else:
                    is_back_edge[v] = True
            else:
                color[u] = 2

    comp = [0] * (n + 1)
    comp_id = 0
    for start in range(1, n + 1):
        if comp[start] != 0:
            continue
        comp_id += 1
        stack = [start]
        comp[start] = comp_id
        while stack:
            u = stack.pop()
            for v in adj[u]:
                if comp[v] == 0 and not is_back_edge[v]:
                    comp[v] = comp_id
                    stack.append(v)

    comp_adj = [set() for _ in range(comp_id + 1)]
    for u in range(1, n + 1):
        cu = comp[u]
        for v in adj[u]:
            cv = comp[v]
            if cu != cv:
                comp_adj[cu].add(cv)
                comp_adj[cv].add(cu)

    comp_color = [0] * (comp_id + 1)
    comp_parent = [0] * (comp_id + 1)
    comp_depth = [0] * (comp_id + 1)
    comp_order = []

    for start in range(1, comp_id + 1):
        if comp_color[start] != 0:
            continue
        comp_color[start] = 1
        comp_parent[start] = 0
        comp_depth[start] = 0
        stack = [(start, 0)]
        while stack:
            u, idx = stack.pop()
            if idx == 0:
                comp_order.append(u)
            if idx < len(comp_adj[u]):
                stack.append((u, idx + 1))
                v = next(iter(comp_adj[u][idx:]))
                if v == comp_parent[u]:
                    continue
                if comp_color[v] == 0:
                    comp_color[v] = 1
                    comp_parent[v] = u
                    comp_depth[v] = comp_depth[u] + 1
                    stack.append((v, 0))
                else:
                    l = comp_depth[u] + comp_depth[v]
                    if l % 2 == 0:
                        print("NO")
                        return
            else:
                comp_color[u] = 2

    comp_val = [0] * (comp_id + 1)
    for u in comp_order:
        if comp_val[u] == 0:
            comp_val[u] = 1
        for v in comp_adj[u]:
            if comp_depth[v] > comp_depth[u]:
                if comp_val[v] == 0:
                    comp_val[v] = 3 - comp_val[u]

    ans = [0] * (n + 1)
    for u in range(1, n + 1):
        ans[u] = comp_val[comp[u]]

    print("YES")
    print(' '.join(map(str, ans[1:])))

solve()
