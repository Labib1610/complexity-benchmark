```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build graph
    graph = defaultdict(list)
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        graph[u].append(v)
        graph[v].append(u)
    
    # Color array: 0 means unvisited, 1 and 2 are colors
    color = [0] * n
    visited = [False] * n
    
    # Check if bipartite for each connected component
    for start in range(n):
        if visited[start]:
            continue
            
        if not graph[start]:  # Isolated vertex
            continue
            
        # BFS to check bipartiteness and assign colors
        queue = deque([start])
        color[start] = 1
        visited[start] = True
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            
            for neighbor in graph[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = 3 - current_color  # Flip color (1->2, 2->1)
                    queue.append(neighbor)
                elif color[neighbor] == current_color:
                    print("NO")
                    return
    
    # If we reach here, a valid assignment exists
    # Assign color 1 to unvisited isolated nodes to get lexicographically smallest
    for i in range(n):
        if color[i] == 0:
            color[i] = 1
    
    print("YES")
    print(" ".join(str(c) for c in color))

solve()
```
