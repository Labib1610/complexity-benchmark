```python
import sys

# The problem states: A link between u and v is active if and only if 
# overriding the state of u with the state of v results in a modified 
# global state sequence.
# This means s_u != s_v.
# Since every link must be active, for every edge (u, v), we must have s_u != s_v.
# This is equivalent to saying the graph must be bipartite.
# We need to find the lexicographically smallest assignment (s_1, ..., s_n)
# where s_i in {1, 2}.
# Lexicographically smallest means we prefer 1 over 2 for s_1, then s_2, etc.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a node is not colored, we assign it 1 (the smallest possible value).
    # Then we perform BFS/DFS to color the connected component.
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
