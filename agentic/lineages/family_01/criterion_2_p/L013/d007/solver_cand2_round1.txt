```python
import sys

# Protocol 1: +1 mod 4
# Protocol 2: -1 mod 4
# Initial state: 0 ('A')
# After node u (p_u) and node v (p_v):
# If p_u=1, p_v=1: 0 + 1 + 1 = 2 ('C') -> Unstable
# If p_u=2, p_v=2: 0 - 1 - 1 = -2 = 2 ('C') -> Unstable
# If p_u=1, p_v=2: 0 + 1 - 1 = 0 ('A') -> Stable
# If p_u=2, p_v=1: 0 - 1 + 1 = 0 ('A') -> Stable
#
# Stability condition: p_u != p_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence s_1, ..., s_n.
# We can use BFS/DFS to color each connected component.
# For each component, we have two choices:
# 1. Start with node i (the smallest index in the component) as 1.
# 2. Start with node i as 2.
# To minimize lexicographically, we greedily pick 1 for the smallest node in each component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for this component
            # To make it lexicographically smallest, we want color[i] = 1
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
