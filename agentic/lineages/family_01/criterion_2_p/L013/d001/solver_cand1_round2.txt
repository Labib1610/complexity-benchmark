```python
import sys

# The problem asks for an assignment of 1 or 2 to each person such that for every edge (u, v),
# s[u] + s[v] is odd. This means s[u] != s[v].
# This is equivalent to 2-coloring a graph.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, for each connected component,
# we want to assign 1 to the smallest index node in that component if possible.
# Since we want the sequence (s_1, s_2, ..., s_n) to be lexicographically smallest,
# we iterate i from 1 to n. If i is not yet colored, we color it 1 and propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign color 1 to i
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
