import sys

# The condition s_u * s_v = 2 means:
# If s_u = 1, then s_v = 2.
# If s_u = 2, then s_v = 1.
# This is equivalent to s_u != s_v.
# The problem is to check if the graph is bipartite.
# If it is, we want the lexicographically smallest assignment.
# For each connected component, we have two possible colorings (1, 2) or (2, 1).
# To make the sequence lexicographically smallest, for each component,
# we look at the smallest index 'i' in that component.
# We should assign s_i = 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            # to determine the coloring that makes the smallest index 1
            idx_in_comp = 0
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # The two possible colorings are:
            # 1. color[v]
            # 2. 3 - color[v]
            # We want the one where the smallest index in the component is 1.
            # Let 'min_node' be the smallest index in this component.
            # If color[min_node] == 1, we keep it.
            # If color[min_node] == 2, we flip all colors in this component.
            
            min_node = min(q)
            if color[min_node] == 2:
                for node in q:
                    color[node] = 3 - color[node]
                    
    print("YES")

solve()
