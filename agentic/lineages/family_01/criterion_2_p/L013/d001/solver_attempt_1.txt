```python
import sys

# The problem states that for each balancer (u, v), the sum of values must be 3.
# Since values are 1 or 2, this means if field u is 1, field v must be 2, and vice versa.
# This is equivalent to saying the graph must be bipartite, and for each connected
# component, once one node's value is fixed, all others are determined.
# Specifically, if we color the graph with 0 and 1, nodes with color 0 get value 1
# and nodes with color 1 get value 2 (or vice versa).
# To get the lexicographically smallest sequence, for each component, we try
# setting the first node (smallest index) to 1. If that works, we check if it's
# better than setting it to 2. Actually, for each component, we have two choices:
# Choice A: color 0 -> 1, color 1 -> 2
# Choice B: color 0 -> 2, color 1 -> 1
# We want the smallest value at the smallest index in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        component.append(v)
                        stack.append((v, color[v]))
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible assignments for this component:
            # 1: color 0 -> 1, color 1 -> 2
            # 2: color 0 -> 2, color 1 -> 1
            # We want the lexicographically smallest result.
            # Find the smallest index in this component.
            first_node = min(component)
            
            # Try assignment 1
            res1 = {}
            for node in component:
                res1[node] = 1 if color[node] == 0 else 2
            
            # Try assignment 2
            res2 = {}
            for node in component:
                res2[node] = 2 if color[node] == 0 else 1
                
            if res1[first_node] < res2[first_node]:
                for node in component:
                    ans[node] = res1[node]
            else:
                for node in component:
                    ans[node] = res2[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
