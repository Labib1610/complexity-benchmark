We can model this as a **bipartite graph** problem: each asset is a node, and each mandate is an edge requiring the two endpoints to be in different portfolios. This is exactly a **2-coloring** problem.

If the graph is bipartite, we can color it with two colors (1 and 2).  
To get the **lexicographically smallest** sequence, we should try to assign asset 1 to portfolio 1, then propagate colors via BFS/DFS, and if a conflict occurs, we try the opposite assignment for that component.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n

def bfs(start):
    q = deque([start])
    color[start] = 1
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = 3 - color[u]
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

for i in range(n):
    if color[i] == -1:
        # try starting with 1
        color[i] = 1
        q = deque([i])
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 3 - color[u]
                    q.append(v)
                elif color[v] == color[u]:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            # try starting with 2
            color[i] = 2
            q = deque([i])
            ok = True
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        ok = False
                        break
                if not ok:
                    break
            if not ok:
                print("NO")
                sys.exit()

print("YES")
print(" ".join(str(c) for c in color))
```
