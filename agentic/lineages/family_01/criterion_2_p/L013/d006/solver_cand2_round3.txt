import sys

# The condition G = 3^m means that for every dependency (u, v),
# (p_u + p_v) must be 3.
# Since p_i is in {1, 2}, p_u + p_v = 3 implies that p_u != p_v.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we want p_1 = 1.
# For each connected component, we can either set the first node to 1 or 2.
# If we set the first node to 1, the rest of the component is fixed.
# If we set it to 2, the rest is also fixed (the inverse).
# To be lexicographically smallest, for each component, we check if
# setting the smallest index node to 1 results in a smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            idx_stack = 0
            while idx_stack < len(component):
                u, c = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # We have a component. We can either keep colors as is,
            # or flip them (1->2, 2->1).
            # We want the lexicographically smallest result.
            # Find the smallest index in this component.
            min_node = min(node for node, c in component)
            
            # If color[min_node] is 2, we should flip to make it 1.
            if color[min_node] == 2:
                for node, c in component:
                    color[node] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
