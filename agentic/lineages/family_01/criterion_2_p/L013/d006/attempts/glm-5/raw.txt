import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    
    adj = [[] for _ in range(n)]
    for i in range(m):
        u = int(data[2 + 2 * i]) - 1
        v = int(data[2 + 2 * i + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        
    color = [-1] * n
    is_bipartite = True
    
    for i in range(n):
        if color[i] == -1:
            queue = [i]
            color[i] = 0
            while queue:
                u = queue.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        is_bipartite = False
                        break
                if not is_bipartite:
                    break
        if not is_bipartite:
            break
            
    if not is_bipartite:
        print("NO")
        return
    
    # To get the lexicographically smallest sequence, we want to assign 1 
    # to as many nodes as possible. This means we should assign 1 to the 
    # color class that has more nodes.
    count0 = sum(1 for x in color if x == 0)
    count1 = n - count0
    
    if count0 >= count1:
        # Assign tier 1 to color 0
        for i in range(n):
            if color[i] == 0:
                color[i] = 1
            else:
                color[i] = 2
    else:
        # Assign tier 1 to color 1
        for i in range(n):
            if color[i] == 1:
                color[i] = 1
            else:
                color[i] = 2
                
    print("YES")
    print(" ".join(map(str, color)))

if __name__ == '__main__':
    solve()
