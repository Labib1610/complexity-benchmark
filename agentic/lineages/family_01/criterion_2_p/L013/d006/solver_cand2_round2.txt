```python
import sys

# The condition is 7^{p_u} + 7^{p_v} \equiv 6 \pmod{10}.
# p_i \in {1, 2}.
# 7^1 = 7, 7^2 = 49 \equiv 9 \pmod{10}.
# Possible sums:
# p_u=1, p_v=1: 7+7 = 14 \equiv 4
# p_u=1, p_v=2: 7+9 = 16 \equiv 6
# p_u=2, p_v=1: 9+7 = 16 \equiv 6
# p_u=2, p_v=2: 9+9 = 18 \equiv 8
# The condition 7^{p_u} + 7^{p_v} \equiv 6 \pmod{10} is satisfied if and only if
# one of {p_u, p_v} is 1 and the other is 2.
# This is a bipartite coloring problem: p_u != p_v.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unassigned, 1 = tier 1, 2 = tier 2
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we prefer 1 over 2.
    # We iterate 1 to n. If uncolored, start BFS/DFS.
    # To make it lexicographically smallest, we want p_i = 1 whenever possible.
    # This is a standard bipartite check.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
