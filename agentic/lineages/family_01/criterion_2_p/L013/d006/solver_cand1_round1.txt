import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    d = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        d[u] += 1
        d[v] += 1
        
    # The equation V^2 - 4mS = m^2 can be rewritten.
    # Let x_i = c_i - 1. Then x_i \in {0, 1}.
    # c_i = x_i + 1.
    # V = sum d_i(x_i + 1) = sum d_i x_i + sum d_i = sum d_i x_i + 2m.
    # S = sum (x_u + 1)(x_v + 1) = sum (x_u x_v + x_u + x_v + 1)
    #   = sum x_u x_v + sum (d_u x_u) + m.
    # V^2 - 4mS = (sum d_i x_i + 2m)^2 - 4m(sum x_u x_v + sum d_i x_i + m)
    # = (sum d_i x_i)^2 + 4m(sum d_i x_i) + 4m^2 - 4m sum x_u x_v - 4m sum d_i x_i - 4m^2
    # = (sum d_i x_i)^2 - 4m sum x_u x_v = m^2.
    
    # This looks like (sum d_i x_i)^2 = m^2 + 4m sum x_u x_v.
    # If we set x_i = 0 for all i, V = 2m, S = m.
    # V^2 - 4mS = 4m^2 - 4m^2 = 0. We need m^2.
    # If we set x_i = 1 for all i, V = 4m, S = m + m = 2m.
    # V^2 - 4mS = 16m^2 - 8m^2 = 8m^2.
    
    # Consider the case where the graph is bipartite.
    # Let the two sets be A and B.
    # If we set c_i = 1 for i in A and c_i = 2 for i in B:
    # V = sum_{i in A} d_i + 2 * sum_{i in B} d_i
    # S = sum_{u in A, v in B} (1 * 2) = m.
    # V^2 - 4m^2 = m^2 => V^2 = 5m^2. Not generally true.
    
    # Actually, the equation V^2 - 4mS = m^2 is equivalent to (V - 2m)^2 = m^2 + 4m(S - m).
    # If we set c_i = 1 for all i, V = 2m, S = m. 0 = m^2 (False unless m=0).
    # If we set c_i = 2 for all i, V = 4m, S = m. 16m^2 - 4m^2 = 12m^2 (False).
    
    # Let's test the bipartite coloring: c_i = 1 if i in A, c_i = 2 if i in B.
    # V = sum_{i in A} d_i + 2 * sum_{i in B} d_i.
    # Let D_A = sum_{i in A} d_i, D_B = sum_{i in B} d_i. Note D_A = D_B = m.
    # V = m + 2m = 3m.
    # S = sum_{u in A, v in B} (1 * 2) = m.
    # V^2 - 4mS = (3m)^2 - 4m(m) = 9m^2 - 4m^2 = 5m^2.
    
    # What if we pick a subset of vertices to be 2?
    # Let x_i = 1 if c_i = 2, else 0.
    # V = sum d_i(x_i + 1) = sum d_i x_i + 2m.
    # S = sum_{uv} (x_u+1)(x_v+1) = sum x_u x_v + sum d_i x_i + m.
    # (sum d_i x_i)^2 - 4m sum x_u x_v = m^2.
    # If we pick a set of vertices X such that sum_{i in X} d_i = m and no two vertices in X are connected,
    # then sum x_u x_v = 0.
    # Then (m)^2 - 4m(0) = m^2. This works!
    
    # So we need to find a subset X such that sum_{i in X} d_i = m and X is an independent set.
    # Lexicographically smallest c means we want x_i = 0 as much as possible (i.e., c_i = 1).
    # This is equivalent to finding the lexicographically largest independent set X such that sum_{i in X} d_i = m.
    
    # Since we want lexicographically smallest c, we want x_1, x_2, ... to be 0 if possible.
    # This is a variation of the subset sum problem / knapsack.
    
    dp = {0: []}
    for i in range(1, n + 1):
        new_dp = dp.copy()
        for s, path in dp.items():
            if s + d[i] <= m:
                # Check if i is independent of all in path
                if all(i not in adj[p] for p in path):
                    new_s = s + d[i]
                    new_path = path + [i]
                    if new_s not in new_dp or len(new_path) < len(new_dp[new_s]):
                        new_dp[new_s] = new_path
        dp = new_dp
        
    if m in dp:
        print("YES")
        res = [1] * n
        for i in dp[m]:
            res[i-1] = 2
        print(*(res))
    else:
        print("NO")

solve()
