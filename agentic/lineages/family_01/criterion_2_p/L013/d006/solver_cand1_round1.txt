import sys

# Let f1(V) = M*V + B1, f2(V) = M*V + B2
# M = [[0, 1], [1, 0]], B1 = [[1], [-1]], B2 = [[-1], [1]]
# f1(f1(V)) = M(M*V + B1) + B1 = M^2*V + M*B1 + B1 = I*V + [[-1], [1]] + [[1], [-1]] = V + [[0], [0]] = V
# So f1(f1(V)) = V for all V. Link (u, v) is unstable if f_u(f_v(V)) = V.
# f1(f1(V)) = V (always unstable)
# f2(f2(V)) = M(M*V + B2) + B2 = V + [[1], [-1]] + [[-1], [1]] = V (always unstable)
# f1(f2(V)) = M(M*V + B2) + B1 = V + [[-1], [1]] + [[1], [-1]] = V (always unstable)
# f2(f1(V)) = M(M*V + B1) + B2 = V + [[1], [-1]] + [[-1], [1]] = V (always unstable)
# Wait, the problem says "composition ... has no fixed points".
# Let's re-check:
# f1(V) = (y+1, x-1)
# f2(V) = (y-1, x+1)
# f1(f1(V)) = f1(y+1, x-1) = (x-1+1, y+1-1) = (x, y) = V. Fixed points exist.
# f2(f2(V)) = f2(y-1, x+1) = (x+1-1, y-1+1) = (x, y) = V. Fixed points exist.
# f1(f2(V)) = f1(y-1, x+1) = (x+1+1, y-1-1) = (x+2, y-2). Fixed points exist if x+2=x and y-2=y, which is impossible.
# f2(f1(V)) = f2(y+1, x-1) = (x-1-1, y+1+1) = (x-2, y+2). Fixed points exist if x-2=x and y+2=y, which is impossible.
# So a link (u, v) is stable if and only if the transformations are different (one is 1, one is 2).
# This is a 2-coloring problem (bipartite graph).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
