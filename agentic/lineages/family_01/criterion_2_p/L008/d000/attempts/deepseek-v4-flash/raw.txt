We can model this as a bipartite graph problem.  
We need to check if the graph is bipartite and, if so, assign shifts 1 and 2 with the lexicographically smallest assignment.

```python
import sys
sys.setrecursionlimit(1 << 20)
input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n
possible = True

for i in range(n):
    if color[i] == -1:
        color[i] = 0
        stack = [i]
        while stack:
            u = stack.pop()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    stack.append(v)
                elif color[v] == color[u]:
                    possible = False
                    break
            if not possible:
                break
    if not possible:
        break

if not possible:
    print("NO")
else:
    print("YES")
    # Lexicographically smallest: assign 1 to color 0, 2 to color 1
    ans = [1 if c == 0 else 2 for c in color]
    print(" ".join(map(str, ans)))
```
