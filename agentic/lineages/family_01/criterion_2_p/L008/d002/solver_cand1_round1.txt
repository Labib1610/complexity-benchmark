```python
import sys

# The problem states that for each workstation (u, v), the sum of their 
# schedule rotations must be 3. Since rotations are 1 or 2, this means
# rotation(u) + rotation(v) = 3.
# This implies rotation(u) != rotation(v).
# This is a classic 2-coloring problem on a graph where workers are nodes
# and workstations are edges. If the graph is bipartite, a solution exists.
# We need the lexicographically smallest configuration, so we prefer 1 over 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest, we iterate from 1 to n.
    # If a node is not colored, we start a BFS/DFS and assign 1 to the 
    # first node of each connected component.
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
