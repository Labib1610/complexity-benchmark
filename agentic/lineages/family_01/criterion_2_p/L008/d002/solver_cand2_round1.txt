```python
import sys

# The condition is gcd(s_u + 1, s_v + 1) = 1.
# s_i can be 1 or 2.
# If s_i = 1, s_i + 1 = 2.
# If s_i = 2, s_i + 1 = 3.
# The condition gcd(s_u + 1, s_v + 1) = 1 means:
# - If s_u = 1 and s_v = 1, gcd(2, 2) = 2 != 1. (Invalid)
# - If s_u = 1 and s_v = 2, gcd(2, 3) = 1. (Valid)
# - If s_u = 2 and s_v = 1, gcd(3, 2) = 1. (Valid)
# - If s_u = 2 and s_v = 2, gcd(3, 3) = 3 != 1. (Invalid)
# Thus, for every edge (u, v), s_u must be different from s_v.
# This is exactly the definition of a bipartite graph.
# We need to check if the graph is bipartite.
# If it is, we want the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2.
# For each connected component, we can assign 1 to the first node,
# then propagate the bipartite coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
