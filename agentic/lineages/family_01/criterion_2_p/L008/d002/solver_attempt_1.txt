import sys

# The problem asks to assign 1 or 2 to each node such that for every edge (u, v),
# dosage[u] + dosage[v] = 3.
# This means if dosage[u] = 1, then dosage[v] must be 2, and vice versa.
# This is equivalent to checking if the graph is bipartite.
# If the graph is bipartite, we can color each connected component with 2 colors (1 and 2).
# To get the lexicographically smallest assignment, for each component, we try
# assigning 1 to the smallest node index in that component and propagate.
# If that leads to a conflict, the graph is not bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want the smallest node
            # in the component to be 1.
            stack = [(i, 1)]
            assignment[i] = 1
            component = []
            
            # Use a stack for DFS
            idx_stack = 0
            q = [i]
            assignment[i] = 1
            
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
