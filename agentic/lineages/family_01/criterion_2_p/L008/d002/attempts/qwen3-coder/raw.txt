import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Try to 2-color the graph (check if bipartite)
    color = [0] * (n + 1)  # 0: uncolored, 1: color 1, 2: color 2
    visited = [False] * (n + 1)
    
    for start in range(1, n + 1):
        if not visited[start]:
            queue = deque([start])
            color[start] = 1
            visited[start] = True
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        color[neighbor] = 3 - color[node]  # Flip color (1->2, 2->1)
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # If we get here, the graph is bipartite
    # We need the lexicographically smallest assignment
    # Color 1 -> dosage 1, Color 2 -> dosage 2
    
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:  # Isolated vertex, assign 1
            result.append("1")
        else:
            result.append(str(color[i]))
    
    print("YES")
    print(" ".join(result))

solve()
