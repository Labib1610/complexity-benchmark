import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition "sum to 3" means if worker u is 1, v must be 2, and vice versa.
    # This is a standard 2-coloring problem (bipartite graph check).
    # To get the lexicographically smallest assignment, we prefer 1 over 2.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS from this component
            # To get lexicographically smallest, we try assigning 1 to the smallest node
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, color = stack.pop()
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - color
                        stack.append((v, assignment[v]))
                    elif assignment[v] == color:
                        print("NO")
                        return
    
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    solve()
