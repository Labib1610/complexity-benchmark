import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(10**6)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We need to find the lexicographically smallest 2-coloring.
    # This means for each connected component, we try to assign 1 to the 
    # smallest node, then propagate.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from node i, assigning color 1
            stack = [(i, 1)]
            color[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                # To ensure lexicographical smallest, we should process neighbors
                # in a way that respects the order. However, for a fixed component,
                # once the first node is colored, the rest are forced.
                # The lexicographical requirement applies to the whole sequence.
                # We iterate 1 to n, if uncolored, color it 1.
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
