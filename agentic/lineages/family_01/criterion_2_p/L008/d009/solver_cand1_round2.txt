```python
import sys

# The problem asks us to assign each city a type 1 or 2.
# Type 1: No road directed towards it (all roads connected to it must be directed away).
# Type 2: No road directed away from it (all roads connected to it must be directed towards).
# If a road connects city u and city v:
# - If u is type 1 and v is type 1: Impossible (road must be directed away from both).
# - If u is type 2 and v is type 2: Impossible (road must be directed towards both).
# - If u is type 1 and v is type 2: Road must be directed from u to v.
# - If u is type 2 and v is type 1: Road must be directed from v to u.
# - If u == v: If u is type 1, it must be directed away from itself (impossible).
#              If u is type 2, it must be directed towards itself (impossible).
#              So, if there is a self-loop, it's impossible.
#
# This is a 2-coloring problem on a graph. If the graph is bipartite, we can color it.
# To get the lexicographically smallest sequence, we want as many 1s as possible at the start.
# For each connected component, we have two choices for coloring (if bipartite).
# We pick the one that results in a smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two colorings: 
            # Option 1: color[i] = 1, Option 2: color[i] = 2
            # We want the lexicographically smallest sequence.
            # Compare the two options for this component.
            
            # Option A: color[i] remains as is
            opt1 = {}
            for node in component:
                opt1[node] = color[node]
            
            # Option B: flip colors
            opt2 = {}
            for node in component:
                opt2[node] = 3 - color[node]
            
            # Find the first node in this component to decide which option is better
            first_node = min(component)
            if opt1[first_node] < opt2[first_node]:
                for node in component:
                    res[node] = opt1[node]
            else:
                for node in component:
                    res[node] = opt2[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
```
