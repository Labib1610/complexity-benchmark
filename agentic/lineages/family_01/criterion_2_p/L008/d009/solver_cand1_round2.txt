import sys

# The resonance footprint is the product of (s_u + s_v) for all edges (u, v).
# s_i is in {1, 2}.
# The possible values for (s_u + s_v) are:
# 1+1 = 2
# 1+2 = 3
# 2+2 = 4
# The product is 3^m if and only if every term (s_u + s_v) is 3.
# This means for every edge (u, v), s_u + s_v = 3.
# Since s_i in {1, 2}, this implies s_u != s_v.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1 and 2.
# To minimize the configuration lexicographically, we want s_1 = 1.
# For each connected component, if node 1 is in it, we set its color to 1.
# If node 1 is not in a component, we set the first node of that component to 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To minimize lexicographically, we want s_1=1, s_2=1, etc.
    # But the bipartite coloring is fixed once the color of one node in a component is set.
    # For each component, we have two choices: (1, 2, 1, 2...) or (2, 1, 2, 1...).
    # We want to pick the choice that makes the sequence lexicographically smallest.
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # BFS to find component and color
            component = []
            stack = [i]
            colors[i] = 1
            component.append(i)
            
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        component.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
            
            # Now we have a component. We can either keep colors as is, or flip them.
            # We want the smallest value at the smallest index in the component.
            # Let 'first' be the smallest index in this component.
            # If colors[first] == 2, we flip all colors in this component.
            first = min(component)
            if colors[first] == 2:
                for node in component:
                    colors[node] = 3 - colors[node]
                    
    print("YES")
    print(*(colors[1:]))

solve()
