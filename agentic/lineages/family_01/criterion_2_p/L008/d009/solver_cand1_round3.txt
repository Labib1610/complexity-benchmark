import sys

# The focal value of a lens i is f_i = 3 * c_i + 1.
# If c_i = 1, f_i = 3(1) + 1 = 4.
# If c_i = 2, f_i = 3(2) + 1 = 7.
# The condition is lcm(f_u, f_v) = 28 for every tube (u, v).
# lcm(4, 4) = 4 != 28
# lcm(4, 7) = 28
# lcm(7, 7) = 7 != 28
# Thus, for every edge (u, v), one lens must have c=1 (f=4) and the other must have c=2 (f=7).
# This means the graph must be bipartite.
# We need to find a 2-coloring of the graph.
# To get the lexicographically smallest sequence, we want c_i = 1 whenever possible.
# For each connected component, we have two possible colorings (swap 1s and 2s).
# We choose the one that makes the sequence lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check which coloring is lexicographically smaller
            # Option 1: color[i] = 1
            # Option 2: color[i] = 2 (flip all colors in component)
            # We want the smallest values at the smallest indices.
            # Find the smallest index in the component.
            start_node = min(component)
            if color[start_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]

    print("YES")
    print(*(color[1:]))

solve()
