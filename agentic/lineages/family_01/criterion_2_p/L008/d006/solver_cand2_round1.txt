import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    if m == 0:
        print("YES")
        print(*( [1] * n ))
        return

    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Let c_i be the axis of lens i (1 or 2).
    # Potential = sum_{i=1}^n 3 * d_i * c_i
    # Loss = sum_{(u,v) in E} 2 * c_u * c_v
    # Score = sum_{i=1}^n 3 * d_i * c_i - sum_{(u,v) in E} 2 * c_u * c_v
    # Let c_i = 1 + x_i, where x_i in {0, 1}.
    # Score = sum 3 * d_i * (1 + x_i) - sum 2 * (1 + x_u)(1 + x_v)
    # Score = sum 3 * d_i + sum 3 * d_i * x_i - sum 2 * (1 + x_u + x_v + x_u * x_v)
    # Since sum d_i = 2m, sum 3 * d_i = 6m.
    # Score = 6m + sum 3 * d_i * x_i - 2m - 2 * sum (x_u + x_v) - 2 * sum (x_u * x_v)
    # Note sum (x_u + x_v) = sum d_i * x_i.
    # Score = 4m + sum (3 * d_i * x_i) - 2 * sum (d_i * x_i) - 2 * sum (x_u * x_v)
    # Score = 4m + sum (d_i * x_i) - 2 * sum (x_u * x_v)
    # We want Score = 5m, so sum (d_i * x_i) - 2 * sum (x_u * x_v) = m.
    
    # Try setting x_i = 1 for all i.
    # sum d_i - 2 * m = 2m - 2m = 0 != m.
    # Try setting x_i = 0 for all i.
    # 0 - 0 = 0 != m.
    
    # Let's test if c_i = 1 for all i works:
    # Score = sum 3 * d_i * 1 - sum 2 * 1 * 1 = 3 * (2m) - 2m = 4m. (Not 5m)
    # Let's test if c_i = 2 for all i works:
    # Score = sum 3 * d_i * 2 - sum 2 * 2 * 2 = 6 * (2m) - 8m = 4m. (Not 5m)
    
    # The equation is sum_{i=1}^n d_i * x_i - 2 * sum_{(u,v) in E} x_u * x_v = m.
    # If we set x_i = 1 for a subset S of vertices, and x_i = 0 otherwise:
    # sum_{v in S} d_v - 2 * (number of edges with both endpoints in S) = m.
    # This is exactly the formula for the number of edges with exactly one endpoint in S (the cut size).
    # We need a subset S such that the cut size is m.
    # The total number of edges is m. The cut size is at most m.
    # The cut size is m if and only if there are no edges with both endpoints in S and no edges with both endpoints in V \ S.
    # This means the graph must be bipartite, and S must be one of the partitions.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, we have two sets S0 = {i | color[i] == 0} and S1 = {i | color[i] == 1}.
    # If we pick S = S0, cut size is m. If we pick S = S1, cut size is m.
    # We want lexicographically smallest sequence.
    # c_i = 1 + x_i. x_i = 1 if i in S, else 0.
    # So c_i = 2 if i in S, else 1.
    # To make it lexicographically smallest, we want as many 1s as possible at the start.
    # This means we want x_i = 0 as much as possible.
    
    # Option 1: S = S0. Then c_i = 2 if color[i] == 0 else 1.
    # Option 2: S = S1. Then c_i = 2 if color[i] == 1 else 1.
    
    res1 = [2 if color[i] == 0 else 1 for i in range(1, n + 1)]
    res2 = [2 if color[i] == 1 else 1 for i in range(1, n + 1)]
    
    ans = min(res1, res2)
    print("YES")
    print(*(ans))

solve()
