```python
import sys

# The problem asks us to assign 1 or 2 to each node such that for every edge (u, v),
# |opacity[u] - opacity[v]| = 1.
# Since the only possible values are 1 and 2, |1-2| = 1 and |2-1| = 1.
# This means for every edge (u, v), opacity[u] must be different from opacity[v].
# This is exactly the definition of a bipartite graph.
# We need to check if the graph is bipartite. If it is, we assign 1 or 2 to each
# connected component. To get the lexicographically smallest sequence, for each
# component, we assign the smaller value (1) to the node with the smallest index
# in that component, and then propagate the coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
