import sys

# The condition is that for every restriction (u, v), s_u^2 + s_v^2 = 5.
# Since s_i is either 1 or 2, s_i^2 is either 1 or 4.
# 1^2 + 2^2 = 1 + 4 = 5.
# This means for every edge (u, v), one must be 1 and the other must be 2.
# This is equivalent to saying the graph must be bipartite.
# We need to color the graph with 1 and 2 such that adjacent nodes have different colors.
# To minimize sum(s_i * 3^(n-i)), we want s_i to be 1 whenever possible,
# starting from the most significant bit (i=1).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS to color the connected component
            # To minimize the sum, we want color[i] = 1 if possible.
            # For each component, we have two choices:
            # Choice A: color[i] = 1, then neighbors are 2, etc.
            # Choice B: color[i] = 2, then neighbors are 1, etc.
            # We compare the lexicographical order of the resulting assignments.
            
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if flipping the colors in this component gives a smaller value
            # We want the first node in the component (by index) to be 1 if possible.
            # Actually, we want the smallest index in the component to have 1.
            # Let min_idx be the smallest index in this component.
            # If color[min_idx] == 2, we flip all colors in this component.
            min_idx = min(component)
            if color[min_idx] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
