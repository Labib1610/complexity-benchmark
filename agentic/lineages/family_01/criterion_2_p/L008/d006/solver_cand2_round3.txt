```python
import sys

# The condition is x^3 + y^3 - (x*y)^2 is prime.
# Possible modes are 1 and 2.
# Case 1: x=1, y=1 => 1^3 + 1^3 - (1*1)^2 = 1 + 1 - 1 = 1 (Not prime)
# Case 2: x=1, y=2 => 1^3 + 2^3 - (1*2)^2 = 1 + 8 - 4 = 5 (Prime)
# Case 3: x=2, y=1 => 2^3 + 1^3 - (2*1)^2 = 8 + 1 - 4 = 5 (Prime)
# Case 4: x=2, y=2 => 2^3 + 2^3 - (2*2)^2 = 8 + 8 - 16 = 0 (Not prime)
#
# Conclusion: For any link (u, v), one must be 1 and the other must be 2.
# This means the graph must be bipartite.
# To get the lexicographically smallest sequence, we want 1s as early as possible.
# For each connected component, we can assign 1 to the first node encountered
# and then propagate the bipartite coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we prefer color 1 for the smallest index
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
