```python
import sys

# The condition s_u^{s_v} + s_v^{s_u} = 3
# If s_u = 1, s_v = 1: 1^1 + 1^1 = 2 != 3
# If s_u = 1, s_v = 2: 1^2 + 2^1 = 1 + 2 = 3 (Valid)
# If s_u = 2, s_v = 1: 2^1 + 1^2 = 2 + 1 = 3 (Valid)
# If s_u = 2, s_v = 2: 2^2 + 2^2 = 4 + 4 = 8 != 3
# Thus, the condition is equivalent to s_u != s_v.
# This is a standard bipartite graph coloring problem.
# We want to minimize sum(s_i * 3^{n-i}).
# Since 3^{n-i} decreases as i increases, we want to assign s_i = 1
# whenever possible for smaller i.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To minimize the weight, we prefer color 1 for the smallest index
            # in each connected component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
