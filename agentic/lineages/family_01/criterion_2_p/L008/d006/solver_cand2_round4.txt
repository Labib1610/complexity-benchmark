```python
import sys

# The interior angle of a regular n-sided polygon is (n-2)*180/n.
# We want (n-2)*180/n = 60, which implies 180n - 360 = 60n, so 120n = 360, n = 3.
# The number of sides is s_u + s_v.
# So s_u + s_v = 3.
# Since s_i is in {1, 2}, this means for every edge (u, v), s_u + s_v = 3.
# This implies s_u != s_v.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we have two sets of vertices (color 0 and color 1).
# We can assign s_i = 1 or s_i = 2 for each set.
# To minimize sum(s_i * 3^(n-i)), we want s_i to be 1 whenever possible.
# For each connected component, we have two choices:
# Choice A: color 0 -> 1, color 1 -> 2
# Choice B: color 0 -> 2, color 1 -> 1
# We pick the choice that makes the sequence lexicographically smaller.
# Since 3^(n-i) is the weight, we compare the choices based on the first index i
# where the assignments differ.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two choices for this component:
            # Choice 0: color 0 -> 1, color 1 -> 2
            # Choice 1: color 0 -> 2, color 1 -> 1
            # Find the first node in this component
            first_node = min(u for u, c in component)
            
            # Compare lexicographically:
            # We want the smallest value at the smallest index.
            # For the first node in the component, if color is 0:
            # Choice 0 gives 1, Choice 1 gives 2.
            # If color is 1:
            # Choice 0 gives 2, Choice 1 gives 1.
            
            c_first = next(c for u, c in component if u == first_node)
            if c_first == 0:
                # Choice 0 is better
                for u, c in component:
                    assignment[u] = 1 if c == 0 else 2
            else:
                # Choice 1 is better
                for u, c in component:
                    assignment[u] = 2 if c == 0 else 1
                    
    print("YES")
    print(*(assignment[1:]))

solve()
```
