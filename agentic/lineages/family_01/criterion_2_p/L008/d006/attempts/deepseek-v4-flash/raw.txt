# x_i in {0,1}
x = [-1] * n

def propagate():
    q = deque()
    for i in range(n):
        if x[i] != -1:
            q.append(i)
    while q:
        u = q.popleft()
        du = degree[u]
        known_sum = 0

    # else try 1
