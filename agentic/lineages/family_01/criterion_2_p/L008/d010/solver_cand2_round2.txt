import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Tension = sum((c_u - c_v)^2).
    # Since c_i in {1, 2}, (c_u - c_v)^2 is 1 if c_u != c_v, and 0 if c_u == c_v.
    # Total tension = number of edges between lenses with different modes.
    # This is exactly the number of edges in a bipartite cut.
    # We need to partition the graph into two sets (1 and 2) such that the number of edges
    # between the sets is exactly m.
    # This is only possible if the graph is bipartite and we can find a cut of size m.
    # Actually, for any bipartite graph, if we color it with 2 colors, the number of edges
    # between different colors is exactly the number of edges in the graph (m).
    # So if the graph is bipartite, any 2-coloring works.
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, the number of edges between different colors is exactly m.
    # We need the lexicographically smallest sequence.
    # The coloring is fixed for each connected component.
    # For each component, we can either keep the coloring or swap 1s and 2s.
    # To make it lexicographically smallest, for each component, we want the first node
    # (smallest index) to be 1 if possible.
    
    final_colors = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Check color of the smallest index in component
            root = min(component)
            if color[root] == 2:
                for node in component:
                    final_colors[node] = 3 - color[node]
            else:
                for node in component:
                    final_colors[node] = color[node]
                    
    print("YES")
    print(*(final_colors[1:]))

solve()
