import sys

# The condition s_i * sum(s_{v_{i,k}}) = 2 * d_i
# If s_i = 1: sum(s_{v_{i,k}}) = 2 * d_i
# Since s_{v_{i,k}} is either 1 or 2, the sum is at most 2 * d_i.
# The only way to get 2 * d_i is if all s_{v_{i,k}} = 2.
# If s_i = 2: 2 * sum(s_{v_{i,k}}) = 2 * d_i => sum(s_{v_{i,k}}) = d_i
# Since s_{v_{i,k}} is either 1 or 2, the sum is at least d_i.
# The only way to get d_i is if all s_{v_{i,k}} = 1.
#
# This means if worker i has s_i = 1, all neighbors must have s = 2.
# If worker i has s_i = 2, all neighbors must have s = 1.
# This is exactly the definition of a bipartite graph.
# If the graph is bipartite, we can color it with 1s and 2s.
# If a component has no edges (d_i = 0), the condition 0 = 0 is satisfied for any s_i.
# To get the lexicographically smallest sequence, we assign 1 to all isolated nodes.
# For bipartite components, we have two choices (swap 1s and 2s).
# We pick the one that makes the first node in the component 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            is_bipartite = True
            
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        is_bipartite = False
            
            if not is_bipartite:
                print("NO")
                return
            
            # Check if we should flip colors to make the first node 1
            # The first node in the component is the one with the smallest index
            first_node = min(component)
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
