import sys

# The condition s_u^2 + s_v^2 = 5 with s_i in {1, 2}
# 1^2 + 1^2 = 2
# 1^2 + 2^2 = 5
# 2^2 + 2^2 = 8
# Thus, for every link (u, v), one must be 1 and the other must be 2.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest sequence.
# To get the lexicographically smallest sequence, we want s_i = 1 whenever possible.
# For each connected component, we have two choices for the 2-coloring (A, B).
# Let color(i) be 0 or 1. We can assign s_i = 1 if color(i) == c and s_i = 2 if color(i) != c,
# where c is 0 or 1.
# To make the sequence lexicographically smallest, for each component,
# we look at the smallest index i in that component. We prefer s_i = 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            component.append(i)
            
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # For this component, we have two colorings:
            # Option 1: color 0 -> 1, color 1 -> 2
            # Option 2: color 0 -> 2, color 1 -> 1
            # We want the smallest index i in component to have s_i = 1.
            # If color[i] == 0, Option 1 gives s_i = 1.
            # If color[i] == 1, Option 2 gives s_i = 1.
            
            target_color_for_1 = color[i]
            for node in component:
                if color[node] == target_color_for_1:
                    res[node] = 1
                else:
                    res[node] = 2
                    
    print("YES")
    print(*(res[1:]))

solve()
