```python
import sys

# The equation is 3W - 2C = 5m
# W = sum(d_i * s_i)
# C = sum(s_u * s_v) for each edge (u, v)
# Let x_i = s_i - 1, so x_i is in {0, 1}.
# s_i = x_i + 1
# W = sum(d_i * (x_i + 1)) = sum(d_i * x_i) + sum(d_i)
# Since sum(d_i) = 2m, W = sum(d_i * x_i) + 2m
# C = sum((x_u + 1) * (x_v + 1)) = sum(x_u * x_v + x_u + x_v + 1)
# C = sum(x_u * x_v) + sum(d_i * x_i) + m
# Substitute into 3W - 2C = 5m:
# 3(sum(d_i * x_i) + 2m) - 2(sum(x_u * x_v) + sum(d_i * x_i) + m) = 5m
# 3*sum(d_i * x_i) + 6m - 2*sum(x_u * x_v) - 2*sum(d_i * x_i) - 2m = 5m
# sum(d_i * x_i) - 2*sum(x_u * x_v) + 4m = 5m
# sum(d_i * x_i) - 2*sum(x_u * x_v) = m
# Let S1 be the set of vertices where x_i = 1 (s_i = 2), S0 be where x_i = 0 (s_i = 1).
# sum_{i in S1} d_i - 2 * (number of edges with both endpoints in S1) = m
# Let E(S1, S1) be edges with both endpoints in S1, E(S1, S0) be edges between S1 and S0.
# sum_{i in S1} d_i = 2 * E(S1, S1) + E(S1, S0)
# So, (2 * E(S1, S1) + E(S1, S0)) - 2 * E(S1, S1) = m
# E(S1, S0) = m
# This means the number of edges between S1 and S0 must be exactly m.
# This is satisfied if we partition the graph into two sets S1 and S0 such that
# all edges are between S1 and S0. This is exactly the definition of a bipartite graph.
# If the graph is bipartite, we can color it with 2 colors.
# Let the two partitions be A and B.
# If we set S1 = A, S0 = B, then E(S1, S0) = m.
# If we set S1 = B, S0 = A, then E(S1, S0) = m.
# We want the lexicographically smallest sequence s_1, ..., s_n.
# s_i = 1 if i in S0, s_i = 2 if i in S1.
# To make s lexicographically smallest, we want s_1=1, then s_2=1, etc.
# This means we want as many 1s as possible at the beginning.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # color[i] is 0 or 1. s_i = color[i] + 1.
    # We have two valid assignments:
    # 1. s_i = color[i] + 1
    # 2. s_i = (1 - color[i]) + 1
    # We want the lexicographically smallest.
    
    res1 = [color[i] + 1 for i in range(1, n + 1)]
    res2 = [(1 - color[i]) + 1 for i in range(1, n + 1)]
    
    print("YES")
    if res1 < res2:
        print(*(res1))
    else:
        print(*(res2))

solve()
```
