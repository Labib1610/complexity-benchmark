```python
import sys

# The problem asks for polarization modes c_i in {1, 2} such that there exists
# an orientation of edges where the resonance weight at lens i is d_i * (5*c_i - 2).
# Let x_i be the number of primary endpoints at lens i.
# The resonance weight at lens i is 8 * x_i + 3 * (d_i - x_i) = 5 * x_i + 3 * d_i.
# We require 5 * x_i + 3 * d_i = d_i * (5 * c_i - 2).
# 5 * x_i = d_i * (5 * c_i - 2 - 3) = d_i * (5 * c_i - 5) = 5 * d_i * (c_i - 1).
# So x_i = d_i * (c_i - 1).
# If c_i = 1, x_i = 0. If c_i = 2, x_i = d_i.
# This means if c_i = 1, all edges connected to i must be secondary.
# If c_i = 2, all edges connected to i must be primary.
# For an edge (u, v), one end must be primary and one secondary.
# If c_u = 1, u must be secondary. If c_u = 2, u must be primary.
# If c_v = 1, v must be secondary. If c_v = 2, v must be primary.
# Edge (u, v) requires:
# - If c_u = 1 and c_v = 1: Impossible (both secondary).
# - If c_u = 2 and c_v = 2: Impossible (both primary).
# - If c_u = 1 and c_v = 2: u is secondary, v is primary (Valid).
# - If c_u = 2 and c_v = 1: u is primary, v is secondary (Valid).
# Thus, for every edge (u, v), we must have c_u != c_v.
# This is possible if and only if the graph is bipartite.
# We need the lexicographically smallest sequence c_1, ..., c_n.
# This means we want c_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
