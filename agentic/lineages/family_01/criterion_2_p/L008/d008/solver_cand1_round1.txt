```python
import sys

# The resonance condition for lens i:
# Let a_i be the number of anode endpoints at lens i, c_i be the cathode endpoints.
# a_i + c_i = d_i (total endpoints)
# 3 * a_i + 8 * c_i = d_i * (5 * p_i - 2), where p_i is polarization (1 or 2).
# Substitute a_i = d_i - c_i:
# 3 * (d_i - c_i) + 8 * c_i = d_i * (5 * p_i - 2)
# 3 * d_i + 5 * c_i = 5 * d_i * p_i - 2 * d_i
# 5 * c_i = 5 * d_i * p_i - 5 * d_i
# c_i = d_i * (p_i - 1)
# If p_i = 1, c_i = 0. If p_i = 2, c_i = d_i.
# So for each tube (u, v), we must choose one endpoint to be the cathode.
# If p_u = 1, u cannot be a cathode. If p_u = 2, u must be a cathode.
# This means if p_u = 1 and p_v = 1, no cathode exists for the tube (impossible).
# If p_u = 2 and p_v = 2, both must be cathodes (impossible, only one cathode per tube).
# Thus, for every edge (u, v), we must have p_u != p_v.
# This is possible if and only if the graph is bipartite.
# If bipartite, we have two components (or more). For each connected component,
# we can assign p_i = 1 or 2 for one node, which forces the rest.
# To get lexicographically smallest, for each component, try p_start = 1.
# If that works, great. If not, try p_start = 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            # Try assigning color 0 (p=1) to node i
            component = []
            stack = [(i, 0)]
            color[i] = 0
            possible = True
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        possible = False
            
            if not possible:
                print("NO")
                return
            
            for node, c in component:
                res[node] = c + 1
                
    print("YES")
    print(*(res[1:]))

solve()
```
