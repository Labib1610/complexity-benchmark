```python
import sys

# The problem asks to assign dosage levels 1 or 2 to each species such that
# for every edge (u, v), dosage[u] * dosage[v] = 2.
# Since dosage levels are in {1, 2}, the only way for the product to be 2
# is if one is 1 and the other is 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 2 colors (1 and 2).
# The requirement "compare them starting from species 1 upwards, and output
# the assignment that grants a lower dosage to the first species where they differ"
# implies that for each connected component, we should try to assign 1 to the
# smallest index node in that component if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To satisfy the lexicographical requirement, we want the smallest
            # index in the component to have dosage 1.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
    print("YES")
    print(*(color[1:]))

solve()
```
