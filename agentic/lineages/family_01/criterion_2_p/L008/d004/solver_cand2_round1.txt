```python
import sys

# The condition for a link (u, v) to be stable:
# (s_u * (u + v^2) + s_v * (v + u^2)) % (u + v) != (s_v * (u + v^2) + s_u * (v + u^2)) % (u + v)
# Let A = u + v^2 and B = v + u^2.
# (s_u * A + s_v * B) % (u + v) != (s_v * A + s_u * B) % (u + v)
# (s_u - s_v) * A + (s_v - s_u) * B % (u + v) != 0
# (s_u - s_v) * (A - B) % (u + v) != 0
# A - B = (u + v^2) - (v + u^2) = (u - v) - (u^2 - v^2) = (u - v) - (u - v)(u + v)
# A - B = (u - v)(1 - (u + v))
# So, (s_u - s_v) * (u - v) * (1 - (u + v)) % (u + v) != 0
# Since (1 - (u + v)) % (u + v) = 1, this simplifies to:
# (s_u - s_v) * (u - v) % (u + v) != 0
# Let s_u, s_v be in {1, 2}. Then (s_u - s_v) is in {-1, 0, 1}.
# If s_u = s_v, then (s_u - s_v) = 0, and 0 % (u + v) = 0.
# So we must have s_u != s_v.
# If s_u != s_v, then |s_u - s_v| = 1.
# The condition becomes: (u - v) % (u + v) != 0.
# Since u, v >= 1, u + v > |u - v|.
# The only way (u - v) % (u + v) == 0 is if u - v = 0, which is impossible for distinct workers.
# Thus, the condition is simply s_u != s_v for all edges (u, v).
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
