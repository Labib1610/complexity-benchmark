```python
import sys

# The problem asks for transparency modes (1 or 2) such that for each constraint (u, v),
# mode(u) * mode(v) = 2.
# Let x_i be the mode of layer i. x_i in {1, 2}.
# x_u * x_v = 2.
# If x_u = 1, then x_v = 2.
# If x_u = 2, then x_v = 1.
# This is equivalent to a bipartite coloring problem where 1 and 2 are the two colors.
# If the graph is bipartite, we can color it.
# To get the lexicographically smallest sequence, we want x_1 to be 1 if possible.
# If x_1 = 1, then all nodes in its connected component are determined.
# If a component is not connected to 1, we want the smallest index in that component
# to be 1 to keep the sequence lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we want the smallest index
            # in the component to be 1.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if this component's smallest index can be 1
            # Actually, the coloring is fixed once we pick the color of one node.
            # If we pick color[min_node] = 1, the whole component is determined.
            # Since we want the lexicographically smallest, we check if color[i] == 1.
            # If color[i] == 2, we can flip all colors in this component.
            if color[i] == 2:
                for node, c in component:
                    color[node] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
```
