```python
import sys

# The problem states that for every edge (u, v), dosage[u] + dosage[v] = 3.
# Since dosages must be positive integers less than 3, the only possible values are 1 and 2.
# If dosage[u] = 1, then dosage[v] must be 2.
# If dosage[u] = 2, then dosage[v] must be 1.
# This means the graph must be bipartite.
# We need to find the lexicographically smallest assignment.
# For each connected component, we can pick the dosage of the first node (the one with the smallest index)
# to be 1, then propagate the constraints. If we find a contradiction, output NO.
# If we have a choice (e.g., a component is not yet colored), we should pick 1 for the smallest
# uncolored node to keep the assignment lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    dosage = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if dosage[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we assign 1 to the smallest node
            stack = [(i, 1)]
            dosage[i] = 1
            while stack:
                u, d = stack.pop()
                target = 3 - d
                for v in adj[u]:
                    if dosage[v] == 0:
                        dosage[v] = target
                        stack.append((v, target))
                    elif dosage[v] != target:
                        print("NO")
                        return
                        
    print("YES")
    print(*(dosage[1:]))

solve()
```
