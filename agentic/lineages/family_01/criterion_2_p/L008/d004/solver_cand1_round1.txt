import sys

# The condition for each worker i is:
# sum_{j in neighbors(i)} s_j + d_i * s_i = 3 * d_i
# Let s_i = 1 + x_i, where x_i in {0, 1}.
# sum_{j in neighbors(i)} (1 + x_j) + d_i * (1 + x_i) = 3 * d_i
# sum_{j in neighbors(i)} x_j + d_i + d_i * x_i = 2 * d_i
# sum_{j in neighbors(i)} x_j + d_i * x_i = d_i
# If d_i is even, sum_{j in neighbors(i)} x_j = d_i (mod 2)
# If d_i is odd, sum_{j in neighbors(i)} x_j + x_i = d_i (mod 2)
# This is a system of linear equations over GF(2).
# Specifically, for each edge (u, v), we have x_u + x_v = 1 (mod 2) if d_u and d_v are both odd.
# If d_i is even, the equation is sum_{j in neighbors(i)} x_j = 0 (mod 2).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    deg = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1
        deg[v] += 1
        idx += 2

    # x_i is 0 or 1. s_i = 1 + x_i.
    # We want lexicographically smallest s_i, so smallest x_i.
    # Equations:
    # If deg[i] is even: sum_{j in adj[i]} x_j = 0
    # If deg[i] is odd: sum_{j in adj[i]} x_j + x_i = 1
    
    # This is a system of linear equations.
    # Let's use BFS/DFS to solve components.
    x = [-1] * (n + 1)
    
    for i in range(1, n + 1):
        if x[i] == -1:
            # Try setting x[i] = 0 first for lexicographical smallest
            component = []
            stack = [i]
            x[i] = 0
            component.append(i)
            while stack:
                u = stack.pop()
                # Equation for u:
                # sum_{v in adj[u]} x_v + (deg[u]%2) * x_u = deg[u]%2
                # sum_{v in adj[u], x_v == -1} x_v = (deg[u]%2) - (deg[u]%2)*x_u - sum_{v in adj[u], x_v != -1} x_v
                
                unknowns = [v for v in adj[u] if x[v] == -1]
                rhs = (deg[u] % 2) - (deg[u] % 2) * x[u]
                for v in adj[u]:
                    if x[v] != -1:
                        rhs -= x[v]
                
                if not unknowns:
                    if rhs % 2 != 0:
                        print("NO")
                        return
                else:
                    # If there's only one unknown, it's forced
                    if len(unknowns) == 1:
                        x[unknowns[0]] = rhs % 2
                        stack.append(unknowns[0])
                        component.append(unknowns[0])
                    else:
                        # If multiple, we can pick one to be 0
                        x[unknowns[0]] = 0
                        stack.append(unknowns[0])
                        component.append(unknowns[0])
            
            # Verify component
            for u in component:
                s = sum(x[v] for v in adj[u]) + (deg[u] % 2) * x[u]
                if s % 2 != deg[u] % 2:
                    print("NO")
                    return

    print("YES")
    print(*(x[i] + 1 for i in range(1, n + 1)))

solve()
