```python
import sys

# The problem asks for an assignment of 1 or 2 to each lens such that for every edge (u, v),
# the sum of the polarization axes is 3.
# Let c_u be the axis for lens u. We need c_u + c_v = 3 for all edges (u, v).
# Since c_u, c_v are in {1, 2}, c_u + c_v = 3 implies that if c_u = 1, then c_v = 2,
# and if c_u = 2, then c_v = 1.
# This means the graph must be bipartite, and for each connected component,
# once we pick the value for one node, all other nodes in that component are determined.
# To get the lexicographically smallest sequence, for each connected component,
# we try setting the first node (in index order) to 1. If that leads to a valid
# 2-coloring, we use it. If not, the configuration is impossible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we prefer 1 over 2.
            # For the first node in the component, we assign 1.
            stack = [(i, 1)]
            assignment[i] = 1
            while stack:
                u, color = stack.pop()
                target = 3 - color
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = target
                        stack.append((v, target))
                    elif assignment[v] != target:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
