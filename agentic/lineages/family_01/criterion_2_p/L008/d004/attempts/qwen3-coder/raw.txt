```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    color = [-1] * n
    queue = deque()
    
    for start in range(n):
        if color[start] == -1:
            color[start] = 0
            queue.append(start)
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        queue.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    # Verify the assignment satisfies all edges
    for u, v in edges:
        # For shift code 1: primary=5, secondary=10
        # For shift code 2: primary=10, secondary=5
        # Worker u with color[u] -> shift = color[u] + 1
        # Worker v with color[v] -> shift = color[v] + 1
        shift_u = color[u] + 1
        shift_v = color[v] + 1
        
        primary_u = 5 if shift_u == 1 else 10
        secondary_v = 10 if shift_v == 2 else 5
        
        if primary_u != secondary_v:
            print("NO")
            return
    
    print("YES")
    result = [str(color[i] + 1) for i in range(n)]
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
