import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n          # 0 or 1
    res = [0] * n

    from collections import deque

    for start in range(n):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        comp = [start]
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                    comp.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
        # decide orientation to get lexicographically smallest
        # find smallest index in component
        min_idx = min(comp)
        # if min_idx currently has color 0, map 0->1,1->2 else flip
        flip = (color[min_idx] == 1)
        for v in comp:
            assigned = 1 if (color[v] ^ flip) == 0 else 2
            res[v] = assigned

    # all edges must connect 1 and 2, which is guaranteed by bipartite check
    print("YES")
    print(' '.join(map(str, res)))

if __name__ == "__main__":
    solve()
