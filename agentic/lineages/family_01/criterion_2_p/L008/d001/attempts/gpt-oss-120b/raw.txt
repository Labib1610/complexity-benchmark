import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)          # 0 or 1 for bipartition, -1 = unvisited
    shift = [0] * (n + 1)           # final values 1 or 2

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS for this component
        q = deque([start])
        color[start] = 0
        comp_nodes = [start]
        ok = True
        while q:
            u = q.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    q.append(v)
                    comp_nodes.append(v)
                elif color[v] == cu:
                    print("NO")
                    return
        # assign shifts: give color 0 the value 1 (lexicographically minimal)
        for node in comp_nodes:
            shift[node] = 1 if color[node] == 0 else 2

    # output
    out = ["YES"]
    out.append(' '.join(str(shift[i]) for i in range(1, n + 1)))
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
