```python
import sys

# The condition "sum to 3" means if component u has voltage v_u, 
# then component v must have voltage 3 - v_u.
# This is equivalent to saying v_u != v_v.
# This is a standard bipartite graph coloring problem.
# We need to check if the graph is bipartite.
# If it is, we have two possible colorings (swapping 1s and 2s).
# We want the lexicographically smallest configuration.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this connected component
            # To get lexicographically smallest, we want node 1 to be 1 if possible.
            # Actually, for each connected component, we have two choices:
            # Choice A: node i has color 1
            # Choice B: node i has color 2
            # We pick the one that makes the first node in the component 1.
            
            stack = [(i, 1)]
            colors[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
            
            # Check if we should flip this component
            # Find the smallest index in this component
            min_node = min(node for node, c in component)
            # If the color of min_node is 2, we flip the whole component
            if colors[min_node] == 2:
                for node, c in component:
                    colors[node] = 3 - c
                    
    print("YES")
    print(*(colors[1:]))

solve()
```
