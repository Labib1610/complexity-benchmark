```python
import sys

# The problem asks to assign each worker a shift (1 or 2) such that for every
# workstation (u, v), the workers u and v have different shifts.
# This is equivalent to 2-coloring a graph.
# A graph is 2-colorable if and only if it is bipartite.
# We need the lexicographically smallest assignment, which means we want
# worker 1 to have shift 1 if possible, then worker 2 to have shift 1 if possible, etc.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate from 1 to n.
    # If a worker is not yet assigned, we assign them shift 1 and perform BFS/DFS.
    for i in range(1, n + 1):
        if assignment[i] == 0:
            assignment[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs if using DFS, 
    # but iterative stack is used here to be safe.
    solve()
```
