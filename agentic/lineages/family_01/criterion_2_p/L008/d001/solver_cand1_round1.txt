```python
import sys

# The problem states that for every link (u, v), s_u + s_v = 3.
# Since s_i is either 1 or 2, this means s_u != s_v.
# This is equivalent to checking if the graph is bipartite.
# If it is bipartite, we want the lexicographically smallest assignment.
# To get the lexicographically smallest assignment, we want s_1 = 1.
# Then for each connected component, we set the first node encountered to 1
# if it hasn't been assigned, and propagate the bipartite coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer 1 over 2.
            # For the first node in a component, assign 1.
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, color = stack.pop()
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - color
                        stack.append((v, assignment[v]))
                    elif assignment[v] == color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    sys.setrecursionlimit(300000)
    solve()
```
