import sys

# The problem states: "the larger of the two assigned transparency modes must be exactly double the smaller one."
# Transparency modes are 1 or 2.
# If mode A = 1, then mode B must be 2.
# If mode A = 2, then mode B must be 4 (impossible) or 1 (if A is the larger one).
# Essentially, for any constraint (u, v), mode(u) != mode(v).
# This is a bipartite graph coloring problem.
# We need to check if the graph is bipartite.
# If it is, we want the lexicographically smallest sequence.
# To get the lexicographically smallest sequence, we want layer 1 to be 1.
# Then we propagate the constraints. For each connected component, we can either
# assign the first node 1 or 2. To be lexicographically smallest, we try to assign 1
# to the smallest index in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we want the first node
            # in the component to be 1 if possible.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a list to keep track of nodes in this component
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if this assignment is the best for this component
            # We already assigned color[i] = 1.
            # If we had assigned color[i] = 2, the colors would be flipped.
            # Since we want the lexicographically smallest sequence, we compare
            # the sequence generated by color[i]=1 vs color[i]=2.
            # Actually, for each component, we just need to ensure the first node
            # encountered (smallest index) is 1.
            
    print("YES")
    print(*(color[1:]))

solve()
