```python
import sys

# The condition is that for every edge (u, v), x_u^2 + x_v^2 - x_u*x_v = 3.
# Given x_i in {1, 2}:
# If x_u = 1, x_v = 1: 1^2 + 1^2 - 1*1 = 1
# If x_u = 1, x_v = 2: 1^2 + 2^2 - 1*2 = 1 + 4 - 2 = 3
# If x_u = 2, x_v = 1: 2^2 + 1^2 - 2*1 = 4 + 1 - 2 = 3
# If x_u = 2, x_v = 2: 2^2 + 2^2 - 2*2 = 4 + 4 - 4 = 4
#
# Thus, for the product to be 3^m, every edge must connect a 1 and a 2.
# This means the graph must be bipartite.
# We need to 2-color the graph. If it's not bipartite, output NO.
# If it is, we want the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for smaller indices.
# For each connected component, we have two possible colorings (swap 1s and 2s).
# We choose the one that makes the first node in the component 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
