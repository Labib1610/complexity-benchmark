```python
import sys

# The interference factor for a tube connecting lenses with axes x and y is x^2 + y^2 - xy.
# If x=1, y=1: 1^2 + 1^2 - 1*1 = 1
# If x=2, y=2: 2^2 + 2^2 - 2*2 = 4
# If x=1, y=2: 1^2 + 2^2 - 1*2 = 3
# If x=2, y=1: 2^2 + 1^2 - 2*1 = 3
# The product of m factors must be 3^m.
# This means every tube must have an interference factor of 3.
# This happens if and only if one lens is 1 and the other is 2.
# This is equivalent to saying the graph must be bipartite.
# We need to 2-color the graph. To get the lexicographically smallest sequence,
# we greedily assign 1 to the first node of each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * (n + 1)
    for i in range(1, n + 1):
        if colors[i] == 0:
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
