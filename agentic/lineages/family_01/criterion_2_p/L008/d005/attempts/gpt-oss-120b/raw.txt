import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    g = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        g[u].append(v)
        g[v].append(u)

    color = [-1] * n          # 0 / 1 bipartite color
    comp = [-1] * n           # component id
    comps = []                # list of vertices per component
    from collections import deque
    cid = 0
    for i in range(n):
        if color[i] != -1:
            continue
        # start BFS
        q = deque([i])
        color[i] = 0
        comp[i] = cid
        cur = [i]
        ok = True
        while q:
            v = q.popleft()
            for to in g[v]:
                if color[to] == -1:
                    color[to] = color[v] ^ 1
                    comp[to] = cid
                    cur.append(to)
                    q.append(to)
                elif color[to] == color[v]:
                    print("NO")
                    return
        comps.append(cur)
        cid += 1

    # decide orientation per component to minimize weighted sum
    flip = [None] * cid   # None = undecided, 0 = keep, 1 = swap
    for i in range(n):
        c = comp[i]
        if flip[c] is None:
            # we want s_i = 1, i.e., color[i] ^ flip == 0  => flip = color[i]
            flip[c] = color[i]

    # build answer
    ans = [0] * n
    for i in range(n):
        f = flip[comp[i]]
        ans[i] = 1 if (color[i] ^ f) == 0 else 2

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
