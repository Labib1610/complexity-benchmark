import sys

# The condition is that for any linked satellites u and v, 
# radius(u)^2 + radius(v)^2 = 5.
# Since radii are 1 or 2, 1^2 + 2^2 = 5.
# This means every link must connect a satellite with radius 1 to one with radius 2.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we assign 1 and 2 to the two partitions.
# To get the lexicographically smallest sequence, we want 1s as much as possible.
# For each connected component, we have two choices of coloring (1,2) or (2,1).
# We choose the one that makes the sequence lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            is_bipartite = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # For this component, we have two colorings:
            # 1. Current color assignment
            # 2. Swapped color assignment (1 becomes 2, 2 becomes 1)
            # We want the lexicographically smallest result.
            # Find the first node in this component.
            first_node = min(component)
            
            # If color[first_node] is 2, we can swap to make it 1.
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
