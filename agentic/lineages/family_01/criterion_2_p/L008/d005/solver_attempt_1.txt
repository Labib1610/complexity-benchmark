import sys

# The problem states that for each restriction (u, v), the XOR of their 
# transparency settings must be 3.
# The possible values for transparency are {1, 2}.
# 1 XOR 1 = 0
# 1 XOR 2 = 3
# 2 XOR 1 = 3
# 2 XOR 2 = 0
# Thus, for every restriction (u, v), u and v must have different values.
# This is equivalent to checking if the graph is bipartite.
# If it is bipartite, we want the lexicographically smallest assignment.
# To get the lexicographically smallest sequence, we want layer 1 to be 1.
# If layer 1 is 1, then all its neighbors must be 2, their neighbors 1, etc.
# We process each connected component. For each component, we can either:
# 1. Assign 1 to the first node encountered (in index order) and propagate.
# 2. Assign 2 to the first node encountered and propagate.
# To make the sequence lexicographically smallest, for each component,
# we check which assignment (1 or 2 for the first node) results in a 
# smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            idx_stack = 0
            while idx_stack < len(component):
                u = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Try setting color[i] = 1
            res1 = [0] * len(component)
            for j, node in enumerate(component):
                res1[j] = color[node]
            
            # Try setting color[i] = 2
            res2 = [0] * len(component)
            for j, node in enumerate(component):
                res2[j] = 3 - color[node]
            
            # Compare lexicographically
            # We need to map back to original indices to compare
            # Actually, we just need to pick the one that makes the 
            # smallest value at the smallest index in the component.
            # Since we process components in order of their smallest index,
            # we just need to ensure the first node in the component gets 1 if possible.
            
            best = res1 if res1[0] < res2[0] else res2
            for j, node in enumerate(component):
                ans[node] = best[j]
                
    print("YES")
    print(*(ans[1:]))

solve()
