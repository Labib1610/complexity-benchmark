```python
import sys

# The diagnostic value formula is 3(c_u + c_v) - 2(c_u * c_v) = 5.
# Possible values for c_u, c_v are {1, 2}.
# If c_u = 1, c_v = 1: 3(1+1) - 2(1*1) = 6 - 2 = 4 != 5.
# If c_u = 1, c_v = 2: 3(1+2) - 2(1*2) = 9 - 4 = 5. (Valid)
# If c_u = 2, c_v = 1: 3(2+1) - 2(2*1) = 9 - 4 = 5. (Valid)
# If c_u = 2, c_v = 2: 3(2+2) - 2(2*2) = 12 - 8 = 4 != 5.
#
# Thus, for every edge (u, v), we must have c_u != c_v.
# This means the graph must be bipartite.
# We need to find a 2-coloring of the graph.
# To get the lexicographically smallest sequence, we want c_i = 1 whenever possible.
# For each connected component, we can pick the color of the first node (smallest index)
# to be 1, then propagate the colors.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we prefer color 1.
            # Since it's bipartite, we can assign color 1 to the first node
            # encountered in the component.
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
