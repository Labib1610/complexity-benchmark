import sys

# The stability condition: (s_u + s_v * i)^4 has real part -7.
# (s_u + s_v * i)^4 = s_u^4 + 4*s_u^3*s_v*i - 6*s_u^2*s_v^2 - 4*s_u*s_v^3*i + s_v^4
# Real part = s_u^4 - 6*s_u^2*s_v^2 + s_v^4 = -7
# If s_u, s_v in {1, 2}:
# (1, 1): 1 - 6 + 1 = -4 != -7
# (1, 2): 1 - 6(4) + 16 = 1 - 24 + 16 = -7 (Valid)
# (2, 1): 16 - 6(4) + 1 = 16 - 24 + 1 = -7 (Valid)
# (2, 2): 16 - 6(16) + 16 = 32 - 96 = -64 != -7
# Thus, s_u != s_v. The graph must be bipartite.

# The harmonic balance condition:
# Centroid of (s_{v_j}, 2*pi*j/k) is (0,0).
# Sum_{j=1}^k s_{v_j} * cos(2*pi*j/k) = 0
# Sum_{j=1}^k s_{v_j} * sin(2*pi*j/k) = 0
# For a bipartite graph, s_v is 1 or 2. Let x_v = s_v - 1.5. x_v is in {-0.5, 0.5}.
# Sum x_{v_j} * exp(i * 2*pi*j/k) = 0.
# This is satisfied if the bipartite sets are balanced in a specific way.
# Actually, for any bipartite graph, if we color it 1 and 2, the condition
# is satisfied if and only if for every cycle, the number of 1s equals the number of 2s.
# This implies that in every connected component, the bipartite coloring is unique
# (up to swapping 1 and 2).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # To minimize sum s_i * 3^(n-i), we want s_i=1 as much as possible for small i.
            # For each component, we have two choices: (1, 2) or (2, 1).
            # Let's check which choice makes the first node in the component 1.
            first_node = min(u for u, c in component)
            # If we set color[first_node] = 1, then all nodes with color[u]==1 get s=1, color[u]==2 get s=2.
            # If we set color[first_node] = 2, then all nodes with color[u]==1 get s=2, color[u]==2 get s=1.
            # We compare the two assignments for this component.
            
    # Re-run to assign based on minimization
    res = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            comp = []
            stack = [i]
            visited[i] = True
            color[i] = 1
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        color[v] = 3 - color[u]
                        stack.append(v)
            
            # Try both
            c1 = [1 if color[u] == 1 else 2 for u in comp]
            c2 = [2 if color[u] == 1 else 1 for u in comp]
            
            # Compare lexicographically by index
            # We want to minimize sum s_i * 3^(n-i)
            # This is equivalent to minimizing the sequence s_1, s_2, ... s_n lexicographically.
            # For each component, pick the one that has 1 at the smallest index.
            best_c = c1 if min(c1) < min(c2) else c2 # This is wrong, need to compare at the first index
            # Correct: find the smallest index in the component, pick the assignment that gives 1 there.
            first = min(comp)
            if color[first] == 1:
                for u in comp: res[u] = color[u]
            else:
                for u in comp: res[u] = 3 - color[u]
                
    print("YES")
    print(*(res[1:]))

solve()
