```python
import sys

# The determinant of the matrix R_{u,v} is:
# det = s_u * (s_v * s_u - 1) - 1 * (1 * s_u - 0) + 0
# det = s_u^2 * s_v - s_u - s_u
# det = s_u^2 * s_v - 2 * s_u
# det = s_u * (s_u * s_v - 2)
# Since s_u is in {1, 2}, s_u is never 0.
# Thus, det = 0 if and only if s_u * s_v = 2.
# If s_u = 1, then s_v must be 2.
# If s_u = 2, then s_v must be 1.
# This means s_u != s_v.
# The condition is equivalent to: for every edge (u, v), s_u != s_v.
# This is exactly the condition for a graph to be bipartite.
# We need to 2-color the graph.
# To minimize sum(s_i * 3^{n-i}), we want s_i to be 1 whenever possible.
# We process nodes in order 1 to n. If a node is not colored, we color it 1.
# Then we propagate the coloring using BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
