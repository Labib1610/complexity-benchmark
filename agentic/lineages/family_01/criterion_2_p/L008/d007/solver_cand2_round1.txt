```python
import sys

# The problem asks for an assignment of 1 or 2 to each lens such that for every lens i,
# the average interference rating is 3.
# Let x_i be the axis of lens i (1 or 2).
# Let y_i = x_i - 1.5. So y_i is -0.5 if x_i=1, and 0.5 if x_i=2.
# The interference rating for a connection between i and j is:
# If x_i=1, x_j=1: rating 2. (y_i=-0.5, y_j=-0.5)
# If x_i=2, x_j=2: rating 4. (y_i=0.5, y_j=0.5)
# If x_i=1, x_j=2: rating 3. (y_i=-0.5, y_j=0.5)
# If x_i=2, x_j=1: rating 3. (y_i=0.5, y_j=-0.5)
# Notice that rating = 3 + 2 * y_i * y_j * 4 = 3 + 8 * y_i * y_j.
# The mean rating for lens i is (1/d_i) * sum_{j in neighbors(i)} (3 + 8 * y_i * y_j)
# = 3 + (8 * y_i / d_i) * sum_{j in neighbors(i)} y_j.
# For the mean to be 3, we need sum_{j in neighbors(i)} y_j = 0.
# Since y_j is either -0.5 or 0.5, this is equivalent to:
# (count of neighbors with axis 2) - (count of neighbors with axis 1) = 0.
# This means each lens must have an equal number of neighbors with axis 1 and axis 2.
# This is only possible if each connected component is bipartite and we color it 2-colorable,
# or more simply, the graph must be bipartite.
# If the graph is bipartite, we can color each component with 1 and 2.
# For a bipartite graph, each node has neighbors only of the opposite color.
# If we assign color 1 to one set and color 2 to the other, then for any node,
# all its neighbors have the opposite color.
# If a node has d_i neighbors, it will have d_i neighbors of the opposite color.
# This satisfies the condition if d_i is even? No, the condition is sum y_j = 0.
# If all neighbors have the opposite color, then all y_j are the same.
# The sum is d_i * y_j. This is 0 only if d_i = 0.
# Wait, the condition is sum_{j in neighbors(i)} y_j = 0.
# This means for every node, the number of neighbors with axis 1 must equal the number of neighbors with axis 2.
# This is possible if and only if the graph is Eulerian (all degrees even) or bipartite?
# Actually, this is a standard problem: a graph has such an assignment iff every connected component
# has an even number of edges or is bipartite? No, it's simpler:
# The condition "number of neighbors with axis 1 = number of neighbors with axis 2"
# is satisfied if we can partition the edges into cycles.
# This is possible if and only if every vertex has an even degree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    deg = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append((v, _))
        adj[v].append((u, _))
        deg[u] += 1
        deg[v] += 1
        idx += 2
    
    for i in range(1, n + 1):
        if deg[i] % 2 != 0:
            print("NO")
            return
            
    ans = [1] * (n + 1)
    used_edge = [False] * m
    
    for i in range(1, n + 1):
        stack = [i]
        while stack:
            u = stack[-1]
            found = False
            while adj[u]:
                v, edge_idx = adj[u].pop()
                if not used_edge[edge_idx]:
                    used_edge[edge_idx] = True
                    ans[v] = 3 - ans[u]
                    stack.append(v)
                    found = True
                    break
            if not found:
                stack.pop()
                
    print("YES")
    print(*(ans[1:]))

solve()
```
