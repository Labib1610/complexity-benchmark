```python
import sys

# The problem asks to assign values 1 or 2 to each node such that for every edge (u, v),
# the product of the values is 2.
# Since the values are 1 or 2, the product is 2 if and only if one node is 1 and the other is 2.
# This is equivalent to checking if the graph is bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest assignment, we want to assign 1 to as many nodes as possible,
# specifically prioritizing smaller indices.
# However, in a bipartite graph, once we fix the color of one node in a connected component,
# the colors of all other nodes in that component are fixed.
# For each connected component, there are two possible colorings (swap 1s and 2s).
# We choose the coloring that results in a lexicographically smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two colorings:
            # 1. Original: color[v] as is
            # 2. Swapped: 3 - color[v]
            # We want the one that is lexicographically smaller.
            # Compare the first node in the component (the one with the smallest index).
            # Actually, we need to compare the whole sequence.
            # Let's pick the coloring that makes the smallest index node 1.
            
            comp_nodes = sorted([node for node, c in component])
            first_node = comp_nodes[0]
            
            # If color[first_node] is 1, coloring 1 is (1, 2, 1, 2...)
            # If color[first_node] is 2, coloring 1 is (2, 1, 2, 1...)
            # We want the sequence to start with 1 if possible.
            
            if color[first_node] == 1:
                for node, c in component:
                    ans[node] = c
            else:
                for node, c in component:
                    ans[node] = 3 - c
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
