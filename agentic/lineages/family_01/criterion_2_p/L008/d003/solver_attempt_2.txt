import sys

# The problem asks if a graph is bipartite. If it is, we need to find the 
# lexicographically smallest 2-coloring.
# A graph is bipartite if and only if it contains no odd cycles.
# To get the lexicographically smallest assignment, we want to assign 1 to 
# as many nodes as possible, prioritizing smaller indices.
# For each connected component, we have two choices for coloring (1, 2) or (2, 1).
# We choose the one that makes the smallest index in the component 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            # to decide the coloring
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring, check if the smallest index in this component
            # has color 1. If not, flip the colors for this component.
            # The smallest index in this component is the first one visited
            # in the BFS, which is 'i'.
            if color[i] == 2:
                for node in q:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
