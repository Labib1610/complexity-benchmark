import sys

# The condition is that for every edge (u, v), 10*s_u + s_v must be divisible by 3.
# s_u, s_v are in {1, 2}.
# Possible values:
# 11: 11 % 3 = 2
# 12: 12 % 3 = 0 (Valid)
# 21: 21 % 3 = 0 (Valid)
# 22: 22 % 3 = 1
# Thus, for every edge (u, v), s_u must be different from s_v.
# This means the graph must be bipartite.
# If bipartite, we can color it with 1 and 2.
# To get the lexicographically smallest assignment, we want s_1 to be 1.
# If the component containing 1 is bipartite, we assign 1 to the partition 
# containing 1 and 2 to the other. For other components, we assign 1 to the 
# partition that allows the smallest values.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we want s_i = 1 whenever possible.
    # For each component, we have two choices of coloring (1,2) or (2,1).
    # We pick the one that makes the smallest index in the component 1.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_stack = 0
            while idx_stack < len(component):
                u = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if we should flip colors in this component
            # We want the smallest index in the component to be 1.
            # Find the smallest index in this component.
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
