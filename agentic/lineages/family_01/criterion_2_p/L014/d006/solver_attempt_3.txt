import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = {}
    components = []
    
    for i in range(1, n + 1):
        if i not in color:
            comp = []
            stack = [(i, 1)]
            color[i] = 1
            comp.append((i, 1))
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if v not in color:
                        color[v] = 3 - c
                        comp.append((v, 3 - c))
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            components.append(comp)
            
    # The deterministic rule:
    # For each asset i, if both 1 and 2 are possible, discard 2.
    # This means for each connected component, we have two possible colorings:
    # C and its complement C'.
    # We want to pick the coloring such that for the smallest index i in the component,
    # the color is 1.
    
    res = [0] * (n + 1)
    for comp in components:
        # Find the smallest index in this component
        min_idx = min(node for node, c in comp)
        
        # Find the color assigned to min_idx in the current BFS/DFS
        c_min = 0
        for node, c in comp:
            if node == min_idx:
                c_min = c
                break
        
        # If c_min is 1, we keep this coloring.
        # If c_min is 2, we flip all colors in this component.
        if c_min == 1:
            for node, c in comp:
                res[node] = c
        else:
            for node, c in comp:
                res[node] = 3 - c
                
    print("YES")
    print(*(res[1:]))

solve()
