import sys

# The problem asks for a 2-coloring of a graph such that every edge connects
# nodes of different colors (bipartite graph).
# The filtering rule: "if both assignments are possible, discard Mode B".
# This means for each connected component, we check if it's bipartite.
# If it is, we have two possible colorings (swapping 1 and 2).
# The rule "discard Mode B if both are possible" implies we should pick the
# coloring that assigns '1' to the smallest index in the component if possible.
# Specifically, for each connected component, we have two colorings:
# C1: {nodes with color A} = S1, {nodes with color B} = S2
# C2: {nodes with color A} = S2, {nodes with color B} = S1
# The rule says: for the smallest index i in the component, if both 1 and 2
# are possible, we must pick the one that assigns 1 to i.
# Since both are always possible in a bipartite component, we always assign
# the smallest index in the component to Mode A (1).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Smallest index in component is i.
            # We want color[i] to be 1 (Mode A).
            # If color[i] is already 1, keep it.
            # If color[i] is 2, flip all colors in component.
            if color[i] == 2:
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
