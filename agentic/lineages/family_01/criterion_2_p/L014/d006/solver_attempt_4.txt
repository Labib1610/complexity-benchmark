```python
import sys

# The problem asks for a deterministic calibration.
# The rule: For each module i from 1 to n, if both polarities 1 and 2 are possible
# in the remaining set of valid configurations, discard all configurations where
# module i is 2.
# This is equivalent to: For each connected component, if it is bipartite,
# there are exactly two ways to color it (swap 1s and 2s).
# The rule says: for the first node i in the component, if both 1 and 2 are possible,
# we must pick 1. This fixes the entire component.
# If the graph is not bipartite, no configuration is possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # The rule says for the first node i in the component,
            # we prefer polarity 1.
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
