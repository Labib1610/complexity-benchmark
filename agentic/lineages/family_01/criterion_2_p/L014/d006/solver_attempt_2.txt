import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color[i] will be 0 (unvisited), 1, or 2
    color = [0] * (n + 1)
    
    # To get the smallest integer, we want instrument 1 to be 1 if possible.
    # We iterate 1 to n. If a component is not visited, we try assigning 1.
    # Since we want the smallest integer, for each component, we check if 
    # assigning the first node to 1 or 2 results in a smaller lexicographical sequence.
    
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # Standard BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a connected component. We have two choices:
            # 1. Keep current colors
            # 2. Flip all colors (1->2, 2->1)
            # We want the smallest lexicographical result.
            # Find the smallest index in this component.
            
            comp_nodes = sorted(component)
            first_node = comp_nodes[0]
            
            if color[first_node] == 1:
                # Option A: color[first_node] is 1
                # Option B: flip all, color[first_node] becomes 2
                # Option A is better because 1 < 2
                for node in component:
                    res[node] = color[node]
            else:
                # Option A: color[first_node] is 2
                # Option B: flip all, color[first_node] becomes 1
                # Option B is better
                for node in component:
                    res[node] = 3 - color[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
