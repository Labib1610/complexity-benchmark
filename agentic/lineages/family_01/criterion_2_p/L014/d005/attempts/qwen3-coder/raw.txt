```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    # Build graph
    graph = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        graph[u].append(v)
        graph[v].append(u)
    
    # Color array: 0 = unvisited, 1 and 2 are the two colors
    color = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    # Check if bipartite
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 1
        visited[start] = True
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            next_color = 3 - current_color  # flips 1->2 and 2->1
            
            for neighbor in graph[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = next_color
                    queue.append(neighbor)
                elif color[neighbor] != next_color:
                    return False
        return True
    
    # Process each connected component
    for i in range(1, n + 1):
        if not visited[i]:
            if not bfs(i):
                print("NO")
                return
    
    # If bipartite, construct the result following the tie-breaking rule:
    # For each node in order 1 to n, if both 1 and 2 are valid, choose 1
    # But we already have a valid coloring. We just need to make sure that
    # for each connected component, we choose the coloring that minimizes
    # the lexicographical order.
    
    # To implement the tie-breaking rule correctly:
    # For each connected component, we can choose to flip the colors or not.
    # We choose the option that gives color 1 to the node with the smallest index in that component.
    
    # Let's re-color properly
    visited = [False] * (n + 1)
    final_color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            # BFS to get all nodes in component
            component = []
            queue = deque()
            queue.append(i)
            visited[i] = True
            component.append(i)
            
            while queue:
                node = queue.popleft()
                for neighbor in graph[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        component.append(neighbor)
                        queue.append(neighbor)
            
            # In this component, decide whether to flip colors
            # We want the smallest node to have color 1
            min_node = min(component)
            current_col = color[min_node]
            if current_col == 2:
                # Flip colors in component
                for node in component:
                    final_color[node] = 3 - color[node]
            else:
                for node in component:
                    final_color[node] = color[node]
    
    print("YES")
    print(' '.join(map(str, final_color[1:])))

if __name__ == "__main__":
    main()
```
