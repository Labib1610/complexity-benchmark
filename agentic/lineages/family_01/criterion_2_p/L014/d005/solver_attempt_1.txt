```python
import sys

# The problem asks for a configuration where linked assets have different values (1 and 2).
# This is equivalent to 2-coloring a graph.
# The tie-breaking rule: for each asset i from 1 to n, if both 1 and 2 are possible,
# choose 1.
# In a connected component, if we fix the value of one node, all other nodes are fixed.
# Let a component have nodes {v1, v2, ...}. If we assign color c to v1,
# then for any node vi, its color is either c or 3-c.
# Let the two possible colorings for a component be C1 and C2 (where C2 is the bitwise flip of C1).
# For each node i, we check if it can be 1 or 2.
# If a component is bipartite, there are exactly two valid colorings.
# Let the two colorings be A and B. For each node i, if A[i] != B[i], then both 1 and 2 are possible.
# The rule says: if both 1 and 2 are possible, pick 1.
# This means for each component, we pick the coloring where the smallest index node is 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # In this component, nodes have colors c or 3-c.
            # The two possible assignments are:
            # Assignment 1: node u has color[u]
            # Assignment 2: node u has 3 - color[u]
            # We want to pick the one where the smallest index node in the component is 1.
            min_node = min(node for node, c in component)
            # Find the color of min_node in the current coloring
            c_min = 0
            for node, c in component:
                if node == min_node:
                    c_min = c
                    break
            
            # If c_min is 1, we keep the current coloring.
            # If c_min is 2, we flip the coloring.
            if c_min == 1:
                for node, c in component:
                    ans[node] = c
            else:
                for node, c in component:
                    ans[node] = 3 - c
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
