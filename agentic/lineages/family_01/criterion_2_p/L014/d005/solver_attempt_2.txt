import sys

# The constraint s_u + s_v = 3 means s_u != s_v.
# This is a 2-coloring problem on a graph.
# The deterministic rule:
# For each connected component, we have two possible colorings (swap 1 and 2).
# The rule says: for i = 1 to n, if both 1 and 2 are possible for s_i, discard s_i = 2.
# This means for each connected component, we pick the coloring that results in s_i = 1
# for the smallest index i in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_idx = min(node for node, c in component)
            # Find the color assigned to min_idx in the current traversal
            for node, c in component:
                if node == min_idx:
                    target_color = c
                    break
            
            # We want s[min_idx] = 1.
            # If target_color is 1, keep as is. If target_color is 2, flip all.
            if target_color == 1:
                for node, c in component:
                    ans[node] = c
            else:
                for node, c in component:
                    ans[node] = 3 - c
                    
    print("YES")
    print(*(ans[1:]))

solve()
