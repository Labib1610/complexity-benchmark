```python
import sys

# The problem asks for a 2-coloring of a graph where each edge (u, v)
# implies that color(u) != color(v).
# Specifically, if color(u) + color(v) = 3, and colors are {1, 2},
# then color(u) must be different from color(v).
# This is a standard bipartite graph check.
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If an asset is not yet colored, we try assigning 1 first (to keep it
# lexicographically smallest), then propagate the constraints using BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the connected component
            # To ensure lexicographical smallest, we prefer color 1
            stack = [(i, 1)]
            colors[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, colors[v]))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
