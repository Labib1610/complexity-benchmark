```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        # No constraints, assign all to fund 1 for lex smallest
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build graph: for each constraint u + v = 3, 
    # if u=1 then v=2, if u=2 then v=1
    # So u and v must have different values.
    adj = defaultdict(list)
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based
        v = int(data[index+1]) - 1
        index += 2
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # Color the graph with 2 colors (1 and 2). Each edge represents that endpoints must differ.
    color = [-1] * n  # -1 means unvisited
    
    # Check all components
    for i in range(n):
        if color[i] == -1:
            # BFS to color this component
            queue = deque()
            queue.append(i)
            color[i] = 1  # Start with color 1
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                next_color = 3 - current_color  # 1->2, 2->1
                
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = next_color
                        queue.append(neighbor)
                    elif color[neighbor] != next_color:
                        print("NO")
                        return
    
    # If we reach here, a valid coloring exists.
    # We want the lexicographically smallest one.
    # That means we prefer color 1 over 2 wherever possible.
    # Our BFS already ensures that: we start with color 1 for the first node,
    # and propagate accordingly.
    
    # But to ensure global lexicographical minimality, we should re-color 
    # by trying to set earlier nodes to 1 if possible.
    # Actually, our BFS from lowest index first with starting color 1
    # already gives the lex smallest coloring for each component.
    # Let's verify this logic by ensuring we process components in order.
    
    # Re-initialize to make sure we get lex smallest
    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            queue = deque()
            queue.append(i)
            color[i] = 1
            while queue:
                node = queue.popleft()
                c = color[node]
                opp_c = 3 - c
                for nei in adj[node]:
                    if color[nei] == -1:
                        color[nei] = opp_c
                        queue.append(nei)
                    elif color[nei] != opp_c:
                        print("NO")
                        return
    
    # Convert colors from 1,2 to string representation
    result = []
    for c in color:
        if c == -1:
            result.append('1')  # Unconstrained nodes get 1 for lex smallest
        else:
            result.append(str(c))
    
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
