```python
import sys

# The constraint "sum of assigned fund numbers must equal 3" means:
# If asset u is 1, asset v must be 2.
# If asset u is 2, asset v must be 1.
# This is equivalent to saying u and v must have different values.
# This is a standard 2-coloring problem on a graph.
# We want the lexicographically smallest assignment (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest, we prefer 1 over 2.
    # We iterate 1 to n. If not colored, start BFS/DFS.
    # To make it lexicographically smallest, we greedily assign 1 to the 
    # first uncolored node in each connected component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
