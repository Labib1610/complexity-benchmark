import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(1000000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] = 0 (unvisited), 1 (mode 1), 2 (mode 2)
    color = [0] * (n + 1)
    result = [0] * (n + 1)
    
    # To get the lexicographically smallest configuration, we iterate 1 to n.
    # For each connected component, we try assigning 1 to the first node.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assigning 1 to i
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                result[u] = c
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
    print("YES")
    print(*(result[1:]))

if __name__ == '__main__':
    solve()
