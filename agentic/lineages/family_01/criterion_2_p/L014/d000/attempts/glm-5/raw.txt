        adj[u].append(v)
        adj[v].append(u)

    parent = [0] * (n + 1)
    visited = [False] * (n + 1)

        stack = [start]
        visited[start] = True
        color[start] = 1
        parent[start] = 0
        while stack:
            u = stack.pop()
            for v in adj[u]:
                if not visited[v]:

                    color[v] = 3 - color[u]
