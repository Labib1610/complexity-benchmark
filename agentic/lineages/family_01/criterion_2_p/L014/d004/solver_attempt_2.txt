```python
import sys

# The constraint is that for each dependency (u, v), the sum of their assigned 
# fund numbers (1 or 2) must be odd.
# 1 + 1 = 2 (even), 2 + 2 = 4 (even), 1 + 2 = 3 (odd), 2 + 1 = 3 (odd).
# Thus, for every dependency (u, v), u and v must have different fund numbers.
# This is a bipartite graph coloring problem.
# We want to minimize the sum of 2^(n-i) for all i in fund 2.
# This is equivalent to minimizing the binary number formed by the assignments,
# where asset 1 is the most significant bit and asset n is the least significant bit.
# To minimize the sum, we want asset 1 to be in fund 1 if possible.
# If asset 1 is in fund 1, then all its neighbors must be in fund 2, and so on.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # We want to minimize the penalty. Penalty = sum_{i in fund 2} 2^(n-i).
    # This is equivalent to minimizing the integer value where bit i is 1 if asset i is in fund 2.
    # We iterate from 1 to n. If asset i is not colored, we try to assign it to fund 1.
    # If we assign i to fund 1, we force its connected component.
    # Since we want to minimize the penalty, for each component, we have two choices:
    # Choice A: Root is 1, Choice B: Root is 2.
    # We compare the penalty of both and pick the smaller one.
    
    ans = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # 1. Keep as is (color[v])
            # 2. Flip (3 - color[v])
            
            penalty1 = 0
            for node, c in component:
                if c == 2:
                    penalty1 += (1 << (n - node))
            
            penalty2 = 0
            for node, c in component:
                if (3 - c) == 2:
                    penalty2 += (1 << (n - node))
            
            if penalty1 <= penalty2:
                for node, c in component:
                    ans[node] = c
            else:
                for node, c in component:
                    ans[node] = 3 - c
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
