```python
import sys

# The problem states: for each dependency (u, v), exactly one of u or v must be 1.
# This is equivalent to saying u != v in terms of their assigned values (1 or 2).
# If u = 1, then v must be 2. If u = 2, then v must be 1.
# This is a standard 2-coloring problem on a graph.
# A graph is 2-colorable if and only if it is bipartite.
# The "deterministic tie-breaking rule" says:
# For each asset i from 1 to n, if both 1 and 2 are possible for i,
# we must choose 1.
# This means for each connected component, we check if we can assign 1 to the
# first node encountered in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To satisfy the rule: "if the pool of still-valid configurations 
            # is split on the index for asset i, discard index 2."
            # This means for the first node in each component, we prefer 1.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            for node, c in component:
                ans[node] = c
                
    print("YES")
    print(*(ans[1:]))

solve()
```
