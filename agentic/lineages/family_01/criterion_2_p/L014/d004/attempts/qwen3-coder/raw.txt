import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 = unvisited, 1 = color 1, 2 = color 2
    color = [0] * (n + 1)
    
    # Check if bipartite and find a valid 2-coloring
    for start in range(1, n + 1):
        if color[start] == 0:
            # BFS to color this component
            queue = deque([start])
            color[start] = 1
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if color[neighbor] == 0:
                        color[neighbor] = 3 - color[node]  # Flip 1->2, 2->1
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        # Not bipartite - no valid configuration
                        print("NO")
                        return
    
    # If we get here, a valid configuration exists
    # Apply the deterministic tie-breaking rule:
    # For each node, if it can be 1 or 2, choose 1
    # Since we already have a valid 2-coloring, we just need to ensure
    # that for each connected component, nodes with color 1 in our
    # bipartite coloring stay as 1 (since that satisfies the tie-breaking)
    
    # Our coloring already satisfies the constraints, and since we start
    # with color 1 for each new component, it follows the tie-breaking rule
    print("YES")
    result = [str(color[i]) for i in range(1, n + 1)]
    print(" ".join(result))

solve()
