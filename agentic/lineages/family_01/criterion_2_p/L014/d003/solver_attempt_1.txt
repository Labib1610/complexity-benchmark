```python
import sys

# The problem states that for each dependency (u, v), the sum of their assigned 
# fund numbers (1 or 2) must be odd.
# 1 + 1 = 2 (even), 2 + 2 = 4 (even), 1 + 2 = 3 (odd), 2 + 1 = 3 (odd).
# Thus, u and v must have different fund numbers. This is a bipartite graph 
# coloring problem where fund 1 and fund 2 are the two colors.
# The tie-breaking rule: for each asset i from 1 to n, if both 1 and 2 are 
# possible, choose 1. This is equivalent to saying that for each connected 
# component, we must pick the assignment that results in the smallest 
# lexicographical sequence of assignments. Since we want to prioritize 1 over 2,
# for each component, we fix the first node encountered (in order 1..n) to 1 
# and propagate the constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            # To satisfy the tie-breaking rule, we always assign the first 
            # node of a component to 1.
            assignment[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
