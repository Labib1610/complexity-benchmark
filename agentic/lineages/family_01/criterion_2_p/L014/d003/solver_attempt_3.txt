```python
import sys

# The problem states that for each pair (u, v), the product of their altitude levels must be 2.
# Altitude levels are 1 or 2.
# The product of two numbers from {1, 2} is 2 if and only if one is 1 and the other is 2.
# This means the graph must be bipartite.
# We need the lexicographically smallest configuration.
# To get the lexicographically smallest, we want to assign 1 to as many early indices as possible.
# For each connected component, we have two possible colorings (swapping 1s and 2s).
# We choose the coloring that results in a smaller value at the first index of the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            idx_stack = 0
            while idx_stack < len(component):
                u, c = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            # We want the color of the smallest index to be 1
            min_node = min(node for node, c in component)
            target_color = color[min_node]
            
            if target_color == 2:
                for node, c in component:
                    color[node] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
```
