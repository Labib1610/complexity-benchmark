We can model this as a bipartite graph problem.  
The condition “sum of fund numbers is odd” means the two assets must be in different funds (one in fund 1, the other in fund 2).  
So we need to check if the graph is bipartite. If not, output `NO`.  

If it is bipartite, we need to apply the tie-breaking rule:  
- Process assets from 1 to n.  
- If both assignments are still possible, choose fund 1.  

We can do this by first finding a valid 2-coloring (bipartition).  
Then, for each connected component, we check which color assignment (flip or not) matches the tie-breaking rule.  

Here’s the implementation:  

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    g = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        g[u].append(v)
        g[v].append(u)
    
    color = [-1] * n
    comps = []
    
    for i in range(n):
        if color[i] == -1:
            stack = [i]
            color[i] = 0
            comp = []
            ok = True
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in g[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        stack.append(v)
                    elif color[v] == color[u]:
                        ok = False
            if not ok:
                print("NO")
                return
            comps.append(comp)
    
    # Now decide for each component which side to assign to fund 1
    assign = [0] * n
    for comp in comps:
        # count how many in comp have color 0 and color 1
        c0 = sum(1 for u in comp if color[u] == 0)
        c1 = len(comp) - c0
        # We want to assign fund 1 to the smaller index set when possible
        # For tie-breaking: process in order, choose fund 1 if possible
        # We'll try both orientations and pick the one that matches tie-break
        # Orientation A: color 0 -> fund 1, color 1 -> fund 2
        # Orientation B: color 1 -> fund 1, color 0 -> fund 2
        # We simulate tie-break for both and pick the one that survives
        
        def simulate(orient):
            # orient: dict mapping color -> fund
            temp = [0]*n
            possible = [True]*n
            # We'll just assign and check tie-break
            for u in comp:
                temp[u] = orient[color[u]]
            # Now simulate tie-break
            # We need to see if at each step, both choices are possible
            # But since we already have a full assignment, we just check if
            # at the first index where the two orientations differ, the chosen one is 1
            # Actually easier: we can just pick orientation that gives fund 1 to the smallest index
            # where they differ.
            return temp
        
        # Find smallest index in comp
        min_idx = min(comp)
        # Determine which orientation gives fund 1 to min_idx
        if color[min_idx] == 0:
            # orientation A gives fund 1 to color 0
            orientA = {0:1, 1:2}
            orientB = {0:2, 1:1}
        else:
            orientA = {0:2, 1:1}
            orientB = {0:1, 1:2}
        
        # Check which orientation matches tie-break
        # We'll just pick the one that gives fund 1 to the smallest index
        # Actually we need to simulate properly:
        # For each asset in order, if both orientations still possible, choose fund 1.
        # Since we only have two orientations, we can just check the first index where they differ.
        tempA = simulate(orientA)
        tempB = simulate(orientB)
        chosen = None
        for u in range(n):
            if u in comp:
                if tempA[u] != tempB[u]:
                    # choose the one with fund 1 at this index
                    if tempA[u] == 1:
                        chosen = tempA
                    else:
                        chosen = tempB
                    break
        if chosen is None:
            chosen = tempA  # identical
        for u in comp:
            assign[u] = chosen[u]
    
    print("YES")
    print(" ".join(map(str, assign)))

if __name__ == "__main__":
    solve()
```
