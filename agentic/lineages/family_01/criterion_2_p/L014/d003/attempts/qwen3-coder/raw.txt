import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    input = sys.stdin.read
    data = input().split()
    
    n, m = int(data[0]), int(data[1])
    
    if m == 0:
        # No dependencies, assign all to fund 1 (deterministic choice)
        print("YES")
        print(' '.join(['1'] * n))
        return
    
    # Build graph
    adj = [[] for _ in range(n+1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # Color array: 0=unvisited, 1=first color, 2=second color
    color = [0] * (n+1)
    
    # Check bipartiteness and find components
    for start in range(1, n+1):
        if color[start] != 0:
            continue
        # BFS to check if component is bipartite
        queue = deque()
        queue.append(start)
        color[start] = 1
        
        while queue:
            u = queue.popleft()
            current_color = color[u]
            next_color = 3 - current_color  # flips 1->2 and 2->1
            
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = next_color
                    queue.append(v)
                elif color[v] != next_color:
                    print("NO")
                    return
    
    # If we reach here, a valid coloring exists.
    # Now apply the tie-breaking rule: for each connected component,
    # among the two possible 2-colorings, choose the one that assigns
    # the lowest-numbered vertex to fund 1.
    
    result = [0] * (n+1)
    
    visited = [False] * (n+1)
    for i in range(1, n+1):
        if visited[i]:
            continue
            
        # Find component via BFS
        component = []
        queue = deque([i])
        visited[i] = True
        component.append(i)
        
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    queue.append(v)
                    component.append(v)
                    
        # In this component, there are two valid colorings:
        # One where the color of min_vertex is 1, and one where it's 2.
        # We want the one where min_vertex gets 1.
        min_vertex = min(component)
        desired_color_min = 1
        actual_color_min = color[min_vertex]
        
        # If actual color of min_vertex is 2, we need to flip all colors in component
        if actual_color_min == 2:
            for node in component:
                result[node] = 3 - color[node]
        else:
            for node in component:
                result[node] = color[node]
                
    print("YES")
    print(' '.join(str(result[i]) for i in range(1, n+1)))

if __name__ == "__main__":
    main()
