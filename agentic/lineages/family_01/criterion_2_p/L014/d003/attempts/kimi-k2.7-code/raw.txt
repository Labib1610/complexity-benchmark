 ```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n  # -1 unvisited, 0 -> fund 1, 1 -> fund 2

    for i in range(n):
        if color[i] != -1:
            continue
        color[i] = 0
        dq = deque([i])
        while dq:
            x = dq.popleft()
            cx = color[x]
            nx = cx ^ 1
            for y in adj[x]:
                if color[y] == -1:
                    color[y] = nx
                    dq.append(y)
                elif color[y] == cx:
                    print("NO")
                    return

    print("YES")
    sys.stdout.write(' '.join(str(c + 1) for c in color))

solve()
```
