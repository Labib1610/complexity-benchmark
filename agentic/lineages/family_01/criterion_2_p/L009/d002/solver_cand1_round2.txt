import sys

# The equation 3xy = 2(x + y) with x, y in {1, 2}:
# If x=1, y=1: 3(1)(1) = 3, 2(1+1) = 4. 3 != 4.
# If x=1, y=2: 3(1)(2) = 6, 2(1+2) = 6. 6 == 6. (Valid)
# If x=2, y=1: 3(2)(1) = 6, 2(2+1) = 6. 6 == 6. (Valid)
# If x=2, y=2: 3(2)(2) = 12, 2(2+2) = 8. 12 != 8.
# Thus, for every link (u, v), we must have x_u != x_v.
# This means the graph must be bipartite.
# We want the lexicographically smallest assignment.
# This means we prefer 1 over 2 for each node.
# We can use BFS/DFS to color each connected component.
# For each component, we have two choices:
# 1. Color nodes in set A as 1 and set B as 2.
# 2. Color nodes in set A as 2 and set B as 1.
# To make it lexicographically smallest, we look at the smallest index node in the component.
# If we set it to 1, we must follow the bipartite coloring.
# If that results in a valid assignment, we check if it's better than setting it to 2.
# Actually, for each component, we just need to pick the coloring that puts 1 at the smallest index.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # Standard BFS/DFS to check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_node = min(component)
            # If color[min_node] is 2, we flip all colors in this component
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
