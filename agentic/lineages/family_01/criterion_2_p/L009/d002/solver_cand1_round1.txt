```python
import sys

# The stability condition: 3 * (x * y) = 2 * (x + y)
# Possible modes are {1, 2}.
# If x=1, y=1: 3*(1*1) = 3, 2*(1+1) = 4. 3 != 4 (Unstable)
# If x=1, y=2: 3*(1*2) = 6, 2*(1+2) = 6. 6 == 6 (Stable)
# If x=2, y=1: 3*(2*1) = 6, 2*(2+1) = 6. 6 == 6 (Stable)
# If x=2, y=2: 3*(2*2) = 12, 2*(2+2) = 8. 12 != 8 (Unstable)
#
# Conclusion: For every linked pair (u, v), x_u != x_v.
# This is a bipartite graph coloring problem.
# We need to color the graph with 2 colors (1 and 2).
# Lexicographically smallest means we prefer 1 over 2 for each node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try color 1 first
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
