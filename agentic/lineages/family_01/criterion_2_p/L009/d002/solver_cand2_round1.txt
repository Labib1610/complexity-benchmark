```python
import sys

# The problem states that for each tool (u, v), exactly one of u or v must be in shift 1,
# and exactly one of u or v must be in shift 2.
# Let s_i be the shift of staff member i (s_i in {1, 2}).
# The condition "exactly one of u or v is in shift 1" means s_u != s_v.
# The condition "exactly one of u or v is in shift 2" means s_u != s_v.
# Both conditions are equivalent to s_u != s_v.
# This is a standard 2-coloring problem on a graph where edges represent constraints.
# If the graph is bipartite, we can color it with 2 colors.
# To get the lexicographically smallest assignment, we greedily assign shift 1
# to the smallest available node in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    shift = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if shift[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we want shift[i] = 1
            stack = [(i, 1)]
            shift[i] = 1
            while stack:
                u, color = stack.pop()
                for v in adj[u]:
                    if shift[v] == 0:
                        shift[v] = 3 - color
                        stack.append((v, 3 - color))
                    elif shift[v] == color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(shift[1:]))

solve()
```
