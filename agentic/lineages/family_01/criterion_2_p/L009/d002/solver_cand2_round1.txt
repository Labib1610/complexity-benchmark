```python
import sys

# The problem states that for every linked pair (u, v), the product of their 
# power modes must be 2. Since modes are in {1, 2}, the only way for 
# mode(u) * mode(v) = 2 is if one is 1 and the other is 2.
# This is a classic 2-coloring problem on a graph.
# If the graph is bipartite, we can assign colors 1 and 2.
# If the graph is not bipartite, no solution exists.
# To get the lexicographically smallest assignment, we want machine 1 to be 1,
# then machine 2 to be as small as possible, and so on.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is not visited, we start a BFS/DFS.
    # To make it lexicographically smallest, we prefer assigning 1 to the 
    # first node of each connected component.
    for i in range(1, n + 1):
        if assignment[i] == 0:
            assignment[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(assignment[1:]))

solve()
```
