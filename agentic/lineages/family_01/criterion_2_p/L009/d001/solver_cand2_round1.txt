```python
import sys

# The constraint s_u^2 + s_v^2 = 5 with s_i in {1, 2}
# 1^2 + 2^2 = 1 + 4 = 5.
# This means for every constraint (u, v), one worker must be 1 and the other must be 2.
# This is equivalent to a 2-coloring problem on a graph.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest assignment, we want s_1 to be 1 if possible.
# If s_1 = 1, then all neighbors of 1 must be 2, their neighbors must be 1, and so on.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try setting color[i] = 1 to keep it lexicographically smallest
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
