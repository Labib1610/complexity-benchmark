import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # shift[i] will store 1 or 2. 0 means unassigned.
    shift = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is unassigned, we assign it to shift 1 and perform BFS/DFS.
    for i in range(1, n + 1):
        if shift[i] == 0:
            # Start BFS/DFS from i, assign shift 1
            stack = [(i, 1)]
            shift[i] = 1
            while stack:
                u, s = stack.pop()
                next_s = 2 if s == 1 else 1
                for v in adj[u]:
                    if shift[v] == 0:
                        shift[v] = next_s
                        stack.append((v, next_s))
                    elif shift[v] == s:
                        print("NO")
                        return
    
    print("YES")
    print(*(shift[1:]))

if __name__ == '__main__':
    solve()
