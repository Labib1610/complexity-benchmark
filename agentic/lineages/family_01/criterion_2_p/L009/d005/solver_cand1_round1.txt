```python
import sys

# The problem requires the product of link stresses (x^2 + y^2) to be 5^m.
# Each link connects two machines with modes x, y in {1, 2}.
# Possible stresses:
# 1^2 + 1^2 = 2
# 1^2 + 2^2 = 5
# 2^2 + 2^2 = 8
# For the product to be 5^m, every link must have a stress of 5.
# This means every link must connect a machine with mode 1 and a machine with mode 2.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we assign 1s and 2s based on the bipartition.
# To get the lexicographically smallest assignment, we want 1s at the smallest indices.
# For each connected component, we have two choices for the bipartition (A, B).
# We choose the one that makes the assignment lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                        stack.append((v, color[v]))
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible assignments for this component:
            # 1. color 1 -> 1, color 2 -> 2
            # 2. color 1 -> 2, color 2 -> 1
            # We want the one that is lexicographically smaller.
            # Find the first node in the component and pick the assignment
            # that gives it a 1 if possible.
            
            first_node = min(component)
            # If we set color[first_node] to 1, then all nodes with color[first_node] get 1
            # If we set color[first_node] to 2, then all nodes with color[first_node] get 2
            # To be lexicographically smallest, we want the smallest index to be 1.
            
            target_color_for_first = color[first_node]
            
            for node in component:
                if color[node] == target_color_for_first:
                    ans[node] = 1
                else:
                    ans[node] = 2
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
