import sys

# The resonance equation is: S_i + 3 * d_i * p_i = 7 * d_i
# Let x_i = p_i - 1, where x_i is in {0, 1}.
# p_i = x_i + 1.
# S_i = sum_{j in neighbors(i)} (p_j)^2 = sum_{j in neighbors(i)} (x_j + 1)^2
# S_i = sum_{j in neighbors(i)} (x_j^2 + 2x_j + 1)
# Since x_j is in {0, 1}, x_j^2 = x_j.
# S_i = sum_{j in neighbors(i)} (x_j + 2x_j + 1) = sum_{j in neighbors(i)} (3x_j + 1)
# S_i = 3 * sum_{j in neighbors(i)} x_j + d_i
# Substitute into resonance equation:
# 3 * sum_{j in neighbors(i)} x_j + d_i + 3 * d_i * (x_i + 1) = 7 * d_i
# 3 * sum_{j in neighbors(i)} x_j + d_i + 3 * d_i * x_i + 3 * d_i = 7 * d_i
# 3 * sum_{j in neighbors(i)} x_j + 3 * d_i * x_i = 3 * d_i
# Divide by 3:
# sum_{j in neighbors(i)} x_j + d_i * x_i = d_i
# If d_i = 0, the equation is 0 = 0, which is always true.
# If d_i > 0:
# If x_i = 0: sum_{j in neighbors(i)} x_j = d_i. This means all neighbors must have x_j = 1.
# If x_i = 1: sum_{j in neighbors(i)} x_j + d_i = d_i => sum_{j in neighbors(i)} x_j = 0. This means all neighbors must have x_j = 0.
# This is exactly the condition for a bipartite graph coloring where x_i is the color.
# If the graph is bipartite, we can color it with 0 and 1.
# To get the lexicographically smallest p_i, we want x_i to be 0 as much as possible.
# For each connected component, we have two choices of coloring (swap 0s and 1s).
# We pick the one that makes the sequence x_1, x_2, ..., x_n lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if this component can be flipped to make it lexicographically smaller
            # Find the first node in this component
            first_node = min(u for u, c in component)
            # If we set color[first_node] = 0, is it better than color[first_node] = 1?
            # We want x_i = 0 (p_i = 1) as much as possible.
            # Let's count how many 0s we get for each choice.
            c0 = [u for u, c in component if c == 0]
            c1 = [u for u, c in component if c == 1]
            
            # We want the smallest index to have x=0.
            # If first_node is in c0, we prefer c0 to be 0.
            # If first_node is in c1, we prefer c1 to be 0.
            # Actually, just compare the two assignments.
            
            # Assignment 1: color[u] = c
            # Assignment 2: color[u] = 1 - c
            # Compare the two assignments lexicographically.
            
            # To be lexicographically smallest, we want the smallest index i to have p_i = 1 (x_i = 0).
            # Find the smallest index in the component.
            min_idx = min(u for u, c in component)
            if color[min_idx] == 1:
                for u, c in component:
                    color[u] = 1 - c
                    
    print("YES")
    print(*(color[i] + 1 for i in range(1, n + 1)))

solve()
