import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # We need to assign 1 or 2 to each container to minimize the sequence lexicographically.
    # This is equivalent to greedily assigning 1 if possible, then 2.
    # Since it's a bipartite check, we can process each connected component.
    # For each component, we have two choices (flip colors).
    # To minimize lexicographically, for each component, we pick the assignment
    # that results in a smaller value for the smallest index in that component.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # New component
            component = []
            stack = [i]
            assignment[i] = 1
            component.append(i)
            
            # BFS/DFS to find all nodes in component and check bipartiteness
            q = [i]
            idx_q = 0
            while idx_q < len(q):
                u = q[idx_q]
                idx_q += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # We can either keep it or flip it (1->2, 2->1).
            # We want to minimize the assignment for the smallest index in the component.
            # Let min_idx be the smallest index in this component.
            # If assignment[min_idx] == 2, we should flip the whole component.
            
            min_idx = min(component)
            if assignment[min_idx] == 2:
                for node in component:
                    assignment[node] = 3 - assignment[node]
                    
    print("YES")
    print(*(assignment[1:]))

solve()
