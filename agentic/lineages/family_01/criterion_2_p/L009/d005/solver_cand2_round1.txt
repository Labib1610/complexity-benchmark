import sys

# The condition "arithmetic mean > geometric mean" for two positive numbers a and b
# is equivalent to (a + b) / 2 > sqrt(a * b).
# This is true if and only if a != b.
# Since the frequency tiers must be positive integers strictly less than 3,
# the only possible tiers are 1 and 2.
# The condition a != b means that for every bridge (u, v),
# the tier of u must be different from the tier of v.
# This is exactly the definition of a bipartite graph.
# We need to check if the graph is bipartite and find the lexicographically
# smallest assignment (1s and 2s).
# To make the assignment lexicographically smallest, we want as many 1s as possible.
# For each connected component, we have two choices for the 2-coloring.
# Let the two sets of the bipartition be A and B.
# We want to assign 1 to the set that contains the smallest index in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # In this component, nodes are colored 1 or 2.
            # We want the smallest index in the component to be 1.
            # Find the smallest index in this component.
            min_node = min(component)
            if color[min_node] == 2:
                # Flip colors for this component
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
