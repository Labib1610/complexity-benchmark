```python
import sys

# The problem states that for each requirement (u, v), the product of their states must be 2.
# Possible states are {1, 2}.
# If x_u * x_v = 2, then:
# - If x_u = 1, then x_v = 2.
# - If x_u = 2, then x_v = 1.
# This is a bipartite coloring problem. Each connected component can be colored in two ways
# (swapping 1s and 2s). To make the resulting number smallest, for each component,
# we want the first device (smallest index) to be 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # x_u * x_u = 2 is impossible since x_u is 1 or 2
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # We have two choices for this component:
            # Choice 1: color[i] = 1 (so x_i = 1)
            # Choice 2: color[i] = 2 (so x_i = 2)
            # To minimize the number, we want the smallest index in the component to be 1.
            # Let min_node be the smallest index in this component.
            # If color[min_node] == 1, we keep it as is.
            # If color[min_node] == 2, we flip all colors in this component.
            
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
```
