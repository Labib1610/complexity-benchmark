```python
import sys

# The problem states that for each constraint (u, v), x_u * x_v = 2.
# Since x_i is in {1, 2}, the only way x_u * x_v = 2 is if one is 1 and the other is 2.
# This is a bipartite coloring problem.
# If u == v, then x_u * x_u = 2, which is impossible since x_u^2 is 1 or 4.
# So if any constraint has u == v, output NO.
# Otherwise, we have a graph where each edge (u, v) implies x_u != x_v.
# This is equivalent to 2-coloring the graph.
# To minimize x_1, then x_2, ..., we greedily assign 1 to the first unvisited node in each component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the component
            # To minimize x_i, we always try to assign 1 first.
            stack = [(i, 1)]
            assignment[i] = 1
            while stack:
                u, color = stack.pop()
                target = 3 - color
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = target
                        stack.append((v, target))
                    elif assignment[v] != target:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
