```python
import sys

# The number of subgrids in a grid of size R x C is given by:
# Sum_{i=1 to R} (R - i + 1) * Sum_{j=1 to C} (C - j + 1)
# = (R * (R + 1) / 2) * (C * (C + 1) / 2)
#
# Possible (c_u, c_v) pairs:
# (1, 1): (1 * 2 / 2) * (1 * 2 / 2) = 1 * 1 = 1
# (1, 2): (1 * 2 / 2) * (2 * 3 / 2) = 1 * 3 = 3
# (2, 1): (2 * 3 / 2) * (1 * 2 / 2) = 3 * 1 = 3
# (2, 2): (2 * 3 / 2) * (2 * 3 / 2) = 3 * 3 = 9
#
# The condition "total number of subgrids is exactly 3" means:
# (c_u, c_v) must be (1, 2) or (2, 1).
# This implies c_u != c_v for every edge (u, v).
# This is equivalent to saying the graph must be bipartite.
#
# To find the lexicographically smallest assignment:
# We want c_i = 1 whenever possible.
# For each connected component, we have two possible 2-colorings.
# Let the two sets of the bipartition be S1 and S2.
# We can either assign 1 to S1 and 2 to S2, or 2 to S1 and 1 to S2.
# We choose the one that makes the sequence lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible assignments:
            # Option 1: nodes with color 0 get 1, nodes with color 1 get 2
            # Option 2: nodes with color 0 get 2, nodes with color 1 get 1
            
            # Find the first node in this component to compare lexicographically
            first_node = min(node for node, c in component)
            
            # Determine which assignment is better
            # We want the smallest value at the smallest index
            # If color[first_node] == 0:
            #   Option 1: ans[first_node] = 1
            #   Option 2: ans[first_node] = 2
            #   Option 1 is better.
            # If color[first_node] == 1:
            #   Option 1: ans[first_node] = 2
            #   Option 2: ans[first_node] = 1
            #   Option 2 is better.
            
            if color[first_node] == 0:
                for node, c in component:
                    ans[node] = 1 if c == 0 else 2
            else:
                for node, c in component:
                    ans[node] = 2 if c == 0 else 1
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
