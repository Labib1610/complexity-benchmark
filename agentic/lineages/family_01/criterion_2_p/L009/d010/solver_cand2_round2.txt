```python
import sys

# The stability condition is 2*c_u^2 + 5*c_u*c_v + 2*c_v^2 = 20
# Possible values for c_u, c_v are {1, 2}.
# If c_u=1, c_v=1: 2(1) + 5(1) + 2(1) = 9 != 20
# If c_u=1, c_v=2: 2(1) + 5(2) + 2(4) = 2 + 10 + 8 = 20 (Valid)
# If c_u=2, c_v=1: 2(4) + 5(2) + 2(1) = 8 + 10 + 2 = 20 (Valid)
# If c_u=2, c_v=2: 2(4) + 5(4) + 2(4) = 8 + 20 + 8 = 36 != 20
# Thus, for every link (u, v), c_u must be different from c_v.
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we prefer 1 over 2.
    # We iterate from 1 to n. If a node is not colored, start BFS/DFS.
    for i in range(1, n + 1):
        if colors[i] == 0:
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
