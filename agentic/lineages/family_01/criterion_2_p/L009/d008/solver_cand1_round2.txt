```python
import sys

# The condition for P(x) = c_u * x^2 + (c_u + c_v) * x + c_v to have two distinct real roots
# is that the discriminant D > 0.
# D = (c_u + c_v)^2 - 4 * c_u * c_v
# D = c_u^2 + 2 * c_u * c_v + c_v^2 - 4 * c_u * c_v
# D = c_u^2 - 2 * c_u * c_v + c_v^2
# D = (c_u - c_v)^2
# For D > 0, we must have (c_u - c_v)^2 > 0, which means c_u != c_v.
# Since c_i can only be 1 or 2, this means for every link (u, v),
# we must have c_u != c_v.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest assignment, we greedily assign 1
# whenever possible in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
