```python
import sys

# The problem states that for each boundary (u, v), p_u * p_v = 2.
# Since p_i is either 1 or 2, the only way for the product to be 2 is:
# If p_u = 1, then p_v = 2.
# If p_u = 2, then p_v = 1.
# This is equivalent to a 2-coloring problem on a graph where each edge
# represents a constraint that the two nodes must have different colors.
# We want the lexicographically smallest assignment (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # colors: 0 = unassigned, 1 = phase 1, 2 = phase 2
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # To get lexicographically smallest, try assigning 1 first
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, colors[v]))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
