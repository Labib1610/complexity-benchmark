```python
import sys

# The problem states:
# Package 1: {A, B}
# Package 2: {B, C}
# A link between u and v requires the union of their libraries to have 3 distinct libraries.
# Union(P1, P1) = {A, B} -> size 2
# Union(P1, P2) = {A, B, C} -> size 3
# Union(P2, P1) = {A, B, C} -> size 3
# Union(P2, P2) = {B, C} -> size 2
#
# Thus, for every link (u, v), u and v must have different packages.
# This is equivalent to 2-coloring a graph.
# The problem asks for the lexicographically smallest assignment.
# We can use a greedy approach with BFS/DFS on each connected component.
# For each component, we try assigning 1 to the first node, then propagate.
# If that fails, we try assigning 2 to the first node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning 1 to the first node of the component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            possible = True
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Reset and try assigning 2
                for node in component:
                    color[node] = 0
                
                stack = [i]
                color[i] = 2
                component = [i]
                head = 0
                while head < len(component):
                    u = component[head]
                    head += 1
                    for v in adj[u]:
                        if color[v] == 0:
                            color[v] = 3 - color[u]
                            component.append(v)
                        elif color[v] == color[u]:
                            print("NO")
                            return
    
    print("YES")
    print(*(color[1:]))

solve()
```
