import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    possible = True
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS for component, try to assign color 1 first
            queue = deque([i])
            color[i] = 1
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break
    
    if not possible:
        print("NO")
        return
    
    # Check if we can improve lexicographically by trying color 2 first for some components
    # We need to check if flipping a component is valid
    # A component can be flipped if it has no edges (isolated node)
    # or if flipping doesn't create same-color edges
    
    # For each component, check if it can be flipped
    # A component can be flipped if all its edges are between different colors
    # which is already satisfied. Flipping preserves the property.
    # But we need to check if flipping creates any same-color edge.
    # Actually, if the component is bipartite, flipping is always valid.
    # So we can flip any component. We want to flip if color[root] == 2 and we can make it 1.
    
    # Re-run BFS, this time trying to assign color 1 first, but if that leads to
    # a conflict, we need to backtrack. Since we want lexicographically smallest,
    # we should try color 1 first for each component.
    
    # The above BFS already tries color 1 first. But we need to verify if
    # assigning color 1 to root is valid for the entire component.
    # If not, we need to try color 2.
    
    # Let's redo the assignment with proper checking
    color = [0] * (n + 1)
    possible = True
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try color 1 first
            color[i] = 1
            queue = deque([i])
            conflict = False
            temp_color = color[:]
            
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if temp_color[v] == 0:
                        temp_color[v] = 3 - temp_color[u]
                        queue.append(v)
                    elif temp_color[v] == temp_color[u]:
                        conflict = True
                        break
                if conflict:
                    break
            
            if conflict:
                # Try color 2
                color[i] = 2
                queue = deque([i])
                conflict = False
                
                while queue:
                    u = queue.popleft()
                    for v in adj[u]:
                        if color[v] == 0:
                            color[v] = 3 - color[u]
                            queue.append(v)
                        elif color[v] == color[u]:
                            conflict = True
                            break
                    if conflict:
                        break
                
                if conflict:
                    possible = False
                    break
            else:
                # Assign temp_color to color
                for j in range(1, n + 1):
                    if temp_color[j] != 0:
                        color[j] = temp_color[j]
    
    if not possible:
        print("NO")
        return
    
    # Verify the solution
    for u in range(1, n + 1):
        for v in adj[u]:
            if color[u] == color[v]:
                print("NO")
                return
    
    print("YES")
    print(' '.join(map(str, color[1:])))

solve()
