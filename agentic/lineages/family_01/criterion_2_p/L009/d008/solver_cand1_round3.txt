```python
import sys

# The resonance score R(u, v) = c_u^{c_v} + c_v^{c_u} must be 3.
# Possible values for c_i are {1, 2}.
# If c_u = 1, c_v = 1: R = 1^1 + 1^1 = 2 (Invalid)
# If c_u = 1, c_v = 2: R = 1^2 + 2^1 = 1 + 2 = 3 (Valid)
# If c_u = 2, c_v = 1: R = 2^1 + 1^2 = 2 + 1 = 3 (Valid)
# If c_u = 2, c_v = 2: R = 2^2 + 2^2 = 4 + 4 = 8 (Invalid)
# Thus, for every link (u, v), c_u must be different from c_v.
# This is a standard 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest assignment, we greedily assign 1
# to the first node of each connected component, then propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the component
            # To be lexicographically smallest, we prefer color 1
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_c
                        stack.append((v, next_c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
