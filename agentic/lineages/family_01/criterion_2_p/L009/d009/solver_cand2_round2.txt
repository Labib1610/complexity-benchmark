import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Potential P_i = (sum of neighbors' channels) / c_i
    # Sum of P_i = sum_{i=1}^n (sum_{j in neighbors(i)} c_j) / c_i = 2.5 * m
    # Let x_i = c_i - 1, where x_i in {0, 1}. Then c_i = x_i + 1.
    # Sum_{i=1}^n (sum_{j in neighbors(i)} (x_j + 1)) / (x_i + 1) = 2.5 * m
    # This is equivalent to:
    # Sum_{i=1}^n (sum_{j in neighbors(i)} (x_j + 1)) / (x_i + 1) = 5/2 * m
    # 2 * Sum_{i=1}^n (sum_{j in neighbors(i)} (x_j + 1)) / (x_i + 1) = 5 * m
    
    # Let's test the simplest case: c_i = 1 for all i.
    # Sum P_i = Sum_{i=1}^n (sum_{j in neighbors(i)} 1) / 1 = Sum_{i=1}^n deg(i) = 2m.
    # We need 2.5m.
    # If we change c_i from 1 to 2, the contribution of node i changes.
    # Let S_i = sum_{j in neighbors(i)} c_j. P_i = S_i / c_i.
    # If c_i = 1, P_i = S_i. If c_i = 2, P_i = S_i / 2.
    # Total = Sum_{i: c_i=1} S_i + Sum_{i: c_i=2} (S_i / 2) = 2.5m.
    
    # Let k be the number of nodes with c_i = 2.
    # Sum_{i=1}^n S_i = 2m.
    # Sum_{i: c_i=1} S_i + Sum_{i: c_i=2} S_i = 2m.
    # Sum_{i: c_i=1} S_i + 0.5 * Sum_{i: c_i=2} S_i = 2.5m.
    # Subtracting: 0.5 * Sum_{i: c_i=2} S_i = -0.5m. Impossible since S_i >= 0.
    
    # Wait, the formula is P_i = (sum of neighbors' channels) / c_i.
    # If c_i = 1, P_i = sum_{j in neighbors(i)} c_j.
    # If c_i = 2, P_i = 0.5 * sum_{j in neighbors(i)} c_j.
    # Let c_i = 1 + x_i, x_i in {0, 1}.
    # Sum P_i = Sum_{i=1}^n (sum_{j in neighbors(i)} (1 + x_j)) / (1 + x_i) = 2.5m.
    
    # Try assigning c_i = 1 for all i: Sum = 2m.
    # Try assigning c_i = 2 for all i: Sum = Sum (deg(i) * 2) / 2 = Sum deg(i) = 2m.
    # The sum is always 2m regardless of the assignment if the graph is regular? No.
    # Let's re-check: Sum P_i = Sum_{i=1}^n (1/c_i) * Sum_{j~i} c_j = Sum_{edges (u,v)} (c_v/c_u + c_u/c_v).
    # For each edge (u,v), the contribution is c_v/c_u + c_u/c_v.
    # If c_u=1, c_v=1: 1/1 + 1/1 = 2.
    # If c_u=1, c_v=2: 2/1 + 1/2 = 2.5.
    # If c_u=2, c_v=2: 2/2 + 2/2 = 2.
    # Total sum = Sum_{edges} (c_v/c_u + c_u/c_v).
    # We need total sum = 2.5 * m.
    # This means we need exactly m edges to have (c_u, c_v) = (1, 2) or (2, 1).
    # Let k be the number of edges with different channels. We need k = m.
    # This is only possible if the graph is bipartite and we color it with 1 and 2.
    
    import collections
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            q = collections.deque([i])
            color[i] = 1
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
