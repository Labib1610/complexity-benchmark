import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Let x_i = c_i - 1. Then x_i is in {0, 1}.
    # c_u + c_v = x_u + x_v + 2
    # c_u * c_v = (x_u + 1)(x_v + 1) = x_u*x_v + x_u + x_v + 1
    # Sum(c_u + c_v) = Sum(x_u + x_v) + 2m = 3m => Sum(x_u + x_v) = m
    # Sum(c_u * c_v) = Sum(x_u*x_v + x_u + x_v + 1) = 2m
    # Sum(x_u*x_v) + Sum(x_u + x_v) + m = 2m
    # Sum(x_u*x_v) + m + m = 2m => Sum(x_u*x_v) = 0
    # Sum(x_u*x_v) = 0 means for every edge (u, v), x_u * x_v = 0.
    # This means at least one of x_u, x_v must be 0.
    # Sum(x_u + x_v) = m means exactly one of x_u, x_v must be 1 for every edge.
    # This is equivalent to saying the graph is bipartite and we must color it 
    # with 0 and 1 such that every edge connects a 0 and a 1.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # Check if the coloring satisfies Sum(x_u + x_v) = m
    # In a bipartite graph, if we color with 0 and 1, x_u + x_v = 1 for all edges.
    # Sum(x_u + x_v) = Sum(1 for all edges) = m.
    # This is always satisfied if the graph is bipartite.
    
    # Lexicographically smallest: we want c_i = 1 (x_i = 0) as much as possible.
    # For each connected component, we have two choices of coloring (0,1) or (1,0).
    # We pick the one that makes the sequence of c_i lexicographically smallest.
    
    res = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Try both colorings for this component
            # Option 1: color[i] = 0
            # Option 2: color[i] = 1
            # We want the smallest c_i = color[i] + 1
            
            # Find the first node in the component
            first_node = min(component)
            # If color[first_node] == 0, we keep it. If 1, we flip.
            if color[first_node] == 1:
                for node in component:
                    color[node] = 1 - color[node]
                    
    print("YES")
    print(*(color[i] + 1 for i in range(1, n + 1)))

solve()
