```python
import sys

# The equation x_u + x_v = 3 with x_i in {1, 2} implies:
# If x_u = 1, then x_v = 2.
# If x_u = 2, then x_v = 1.
# This is equivalent to x_u != x_v.
# If u == v, then x_u + x_u = 3 => 2*x_u = 3, which is impossible for integers.
# So if any equation has u == v, output NO.
# Otherwise, this is a bipartite coloring problem.
# We want to minimize x_1, then x_2, ..., x_n.
# Since x_i can be 1 or 2, we prefer 1 over 2.
# This is equivalent to 2-coloring a graph where edges represent x_u != x_v.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    # To minimize x_1, x_2, ..., x_n lexicographically:
    # We iterate i from 1 to n. If color[i] is not set, we set it to 1 (the smaller value).
    # Then we propagate the constraints using BFS/DFS.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
