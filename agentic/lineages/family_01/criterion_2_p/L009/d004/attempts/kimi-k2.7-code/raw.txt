```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        if u != v:
            adj[v].append(u)

    val = [0] * n
    for i in range(n):
        if val[i]:
            continue
        val[i] = 1
        dq = deque([i])
        ok = True
        while dq:
            u = dq.popleft()
            target = 3 - val[u]
            for w in adj[u]:
                if val[w] == 0:
                    val[w] = target
                    dq.append(w)
                elif val[w] != target:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            sys.stdout.write("NO\n")
            return

    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(map(str, val)) + '\n')

solve()
```
