```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Determine valid combinations
    # We need 3^(x+y) to end with 7
    # Let's check all combinations of x, y in {1, 2}
    valid_pairs = set()
    for x in [1, 2]:
        for y in [1, 2]:
            power = 3**(x + y)
            if power % 10 == 7:
                valid_pairs.add((x, y))
                valid_pairs.add((y, x))
    
    # valid_pairs should contain (1,2) and (2,1) since 3^3 = 27
    # Let's check: 3^(1+1)=9, 3^(1+2)=27, 3^(2+2)=81
    # So only 3^3=27 ends in 7
    # This means adjacent nodes must have different values
    
    # This is a 2-coloring problem (bipartite check)
    color = {}
    
    # Check if graph is bipartite
    for start_node in adj:
        if start_node not in color:
            queue = deque([start_node])
            color[start_node] = 1
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in adj[node]:
                    if neighbor not in color:
                        color[neighbor] = 3 - current_color  # Flip color (1->2, 2->1)
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        print("NO")
                        return
    
    # If bipartite, construct lexicographically smallest solution
    # Assign 1 to all uncolored nodes (isolated vertices)
    result = []
    for i in range(1, n + 1):
        if i not in color:
            result.append("1")
        else:
            result.append(str(color[i]))
    
    print("YES")
    print(" ".join(result))

solve()
```
