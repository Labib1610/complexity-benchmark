```python
import sys

# The condition:
# (s_u & s_v) == 0
# (s_u ^ s_v) == 3
# Given s_i in {1, 2}:
# If s_u = 1 (binary 01), then s_v must be 2 (binary 10).
# If s_u = 2 (binary 10), then s_v must be 1 (binary 01).
# This means s_u != s_v.
# This is a bipartite coloring problem.
# We need to assign 1 or 2 to each node such that connected nodes have different values.
# To get the lexicographically smallest result, we want s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, try assigning 1 first
            stack = [i]
            assignment[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
