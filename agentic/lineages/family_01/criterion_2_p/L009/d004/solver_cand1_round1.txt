import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition is sum(d_i * s_i^2) = 2 * sum_{(u,v)} (s_u * s_v) + m
    # Let s_i = x_i + 1, where x_i in {0, 1}.
    # s_i^2 = (x_i + 1)^2 = x_i^2 + 2x_i + 1 = x_i + 2x_i + 1 = 3x_i + 1
    # sum(d_i * (3x_i + 1)) = 2 * sum_{(u,v)} ((x_u + 1)(x_v + 1)) + m
    # 3 * sum(d_i * x_i) + sum(d_i) = 2 * sum_{(u,v)} (x_u*x_v + x_u + x_v + 1) + m
    # 3 * sum(d_i * x_i) + 2m = 2 * sum_{(u,v)} (x_u*x_v) + 2 * sum(d_i * x_i) + 2m + m
    # sum(d_i * x_i) = 2 * sum_{(u,v)} (x_u*x_v) + m
    # This simplifies to: sum_{(u,v)} (x_u - x_v)^2 = 0
    # Which means x_u = x_v for all connected u, v.
    # So all machines in a connected component must have the same x_i.
    
    visited = [False] * (n + 1)
    comp_id = [-1] * (n + 1)
    comps = []
    
    for i in range(1, n + 1):
        if not visited[i]:
            curr_comp = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                curr_comp.append(u)
                comp_id[u] = len(comps)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            comps.append(curr_comp)
            
    # For each component, x_i must be 0 or 1.
    # Let c_k be the value (0 or 1) for component k.
    # The equation sum(d_i * x_i) = 2 * sum_{(u,v)} (x_u*x_v) + m
    # becomes sum_k (c_k * sum_{i in comp_k} d_i) = 2 * sum_{k} (c_k^2 * m_k) + m
    # where m_k is the number of edges in component k.
    # Since c_k^2 = c_k, sum_k (c_k * sum_{i in comp_k} d_i) = 2 * sum_k (c_k * m_k) + m
    # sum_k (c_k * (sum_{i in comp_k} d_i - 2 * m_k)) = m
    # Note: sum_{i in comp_k} d_i = 2 * m_k.
    # So the equation is sum_k (c_k * 0) = m.
    # This implies 0 = m.
    # If m > 0, no solution exists. If m = 0, any assignment works.
    
    if m == 0:
        print("YES")
        print(*( [1] * n ))
        return

    print("NO")

solve()
