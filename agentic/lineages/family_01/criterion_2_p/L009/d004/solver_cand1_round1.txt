```python
import sys

# The resonance equation is p_u^{p_v} + p_v^{p_u} = 3.
# Possible values for p_i are {1, 2}.
# If p_u = 1, p_v = 1: 1^1 + 1^1 = 2 != 3
# If p_u = 1, p_v = 2: 1^2 + 2^1 = 1 + 2 = 3 (Valid)
# If p_u = 2, p_v = 1: 2^1 + 1^2 = 2 + 1 = 3 (Valid)
# If p_u = 2, p_v = 2: 2^2 + 2^2 = 4 + 4 = 8 != 3
# Thus, the condition is p_u != p_v.
# This is a standard 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # colors: 0 = unvisited, 1 = state 1, 2 = state 2
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # To get lexicographically smallest, we prefer state 1
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_c
                        stack.append((v, next_c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
```
