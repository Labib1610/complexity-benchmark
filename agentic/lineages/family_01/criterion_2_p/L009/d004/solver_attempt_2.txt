```python
import sys

# The problem asks us to assign values {1, 2} to n nodes such that for every edge (u, v),
# c_u * c_v = 2.
# If c_u = 1, then c_v must be 2.
# If c_u = 2, then c_v must be 1.
# This is a classic bipartite coloring problem.
# We want to assign 1 to as many nodes as possible starting from index 1,
# but the constraint c_u * c_v = 2 forces a specific relationship.
# Specifically, if we pick a value for one node in a connected component,
# all other nodes in that component are fixed.
# To minimize the values lexicographically, for each connected component,
# we try assigning 1 to the smallest index node in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we want color[i] = 1
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
