```python
import sys

# The condition: 3^(x+y) ends in 7.
# Powers of 3:
# 3^1 = 3
# 3^2 = 9
# 3^3 = 27 (ends in 7)
# 3^4 = 81
# 3^5 = 243
# 3^6 = 729
# 3^7 = 2187 (ends in 7)
# The exponent (x+y) must be 3, 7, 11, ...
# Since x, y are in {1, 2}, x+y can be 2, 3, or 4.
# The only value that satisfies the condition is x+y = 3.
# This means for every link (u, v), we must have mode(u) + mode(v) = 3.
# This implies mode(u) != mode(v).
# If mode(u) = 1, then mode(v) = 2.
# If mode(u) = 2, then mode(v) = 1.
# This is a bipartite coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, try assigning 1 first
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
