```python
import sys

# The conditions are:
# s_u AND s_v = 0
# s_u XOR s_v = 3
# Given s_u, s_v in {1, 2}:
# 1 is 01 in binary, 2 is 10 in binary.
# If s_u = 1, s_v = 2: 1 AND 2 = 0 (True), 1 XOR 2 = 3 (True).
# If s_u = 2, s_v = 1: 2 AND 1 = 0 (True), 2 XOR 1 = 3 (True).
# If s_u = 1, s_v = 1: 1 AND 1 = 1 (False).
# If s_u = 2, s_v = 2: 2 AND 2 = 2 (False).
# Thus, for every edge (u, v), s_u must be different from s_v.
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # colors: 0 = unvisited, 1 = state 1, 2 = state 2
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # To get lexicographically smallest, we prefer state 1 for the first node in a component
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_c
                        stack.append((v, next_c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
