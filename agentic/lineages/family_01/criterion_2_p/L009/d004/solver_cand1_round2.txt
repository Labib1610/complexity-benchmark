```python
import sys

# The condition is: (x + yi)^2 = x^2 - y^2 + 2xyi.
# The imaginary part is 2xy.
# We need 2xy = 4, which means xy = 2.
# Since x, y are in {1, 2}, the only way xy = 2 is if one is 1 and the other is 2.
# This means for every edge (u, v), x_u != x_v.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest assignment.
# Lexicographically smallest means we want s_1 to be 1 if possible.
# If s_1 = 1, then all neighbors of 1 must be 2, their neighbors 1, and so on.
# We can use BFS/DFS to color the connected components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is not colored, we try to color it 1.
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
