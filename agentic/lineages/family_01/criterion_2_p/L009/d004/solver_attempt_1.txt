import sys

# The problem asks for an assignment of 1 or 2 to each coil such that for every edge (u, v),
# c_u * c_v = 2.
# Since c_i is in {1, 2}, the only way c_u * c_v = 2 is if one is 1 and the other is 2.
# This is equivalent to 2-coloring the graph.
# If the graph is bipartite, we can color it with 1s and 2s.
# To minimize the sequence lexicographically, for each connected component,
# we have two choices:
# 1. Assign 1 to the first node encountered (in index order) and 2 to its neighbors.
# 2. Assign 2 to the first node encountered and 1 to its neighbors.
# We want to pick the choice that makes the sequence lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # Option 1: color[i] = 1, Option 2: color[i] = 2
            # We want to minimize the values at the smallest indices.
            # Let's find the smallest index 'first' in this component.
            first = min(component)
            
            # If we set color[first] = 1, then for all v in component:
            # if color[v] == color[first], new_color[v] = 1
            # else, new_color[v] = 2
            # If we set color[first] = 2, then:
            # if color[v] == color[first], new_color[v] = 2
            # else, new_color[v] = 1
            
            # Compare the two options at the index 'first'
            # Option 1: ans[first] = 1
            # Option 2: ans[first] = 2
            # Clearly Option 1 is better.
            
            c1 = color[first]
            for node in component:
                if color[node] == c1:
                    ans[node] = 1
                else:
                    ans[node] = 2
                    
    print("YES")
    print(*(ans[1:]))

solve()
