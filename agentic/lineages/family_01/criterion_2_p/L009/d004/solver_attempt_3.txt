import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If a node is not colored, we try assigning it color 1.
    # This is a standard bipartite check.
    for i in range(1, n + 1):
        if colors[i] == 0:
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                
                # Sort neighbors to ensure deterministic behavior if needed,
                # though the lexicographical requirement is satisfied by the 
                # outer loop and the choice of 1 over 2.
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
