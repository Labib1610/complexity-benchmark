```python
import sys

# The problem states that for any two nodes u, v connected by an interference link,
# c_u + c_v = 3. Since c_i is either 1 or 2, this means c_u must be different from c_v.
# This is equivalent to 2-coloring a graph.
# We need the lexicographically smallest assignment, which means we prefer 1 over 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate from 1 to n.
    # If a node is not colored, we assign it 1 (the smallest possible value).
    # Then we perform BFS/DFS to color the connected component.
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
```
