```python
import sys

# C_i = p_i^3 - 3p_i^2 + 4p_i - 3
# If p_i = 1: C_i = 1 - 3 + 4 - 3 = -1
# If p_i = 2: C_i = 8 - 12 + 8 - 3 = 1
# M_i = [[i, C_i, 0], [-C_i, i, 0], [0, 0, 1]]
# K = M_u * M_v
# Trace(K) = (u*v - C_u*C_v) + (v*u - C_v*C_u) + 1 = 2*u*v - 2*C_u*C_v + 1
# Stability condition: Trace(K) > 2*u*v
# 2*u*v - 2*C_u*C_v + 1 > 2*u*v
# -2*C_u*C_v + 1 > 0
# 2*C_u*C_v < 1
# Since C_u, C_v are in {-1, 1}, C_u*C_v must be -1.
# This means C_u != C_v, so p_u != p_v.
# The problem reduces to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest assignment (p_1, ..., p_n).
# For each connected component, we have two choices:
# 1. p_root = 1, then all nodes at even distance from root are 1, odd are 2.
# 2. p_root = 2, then all nodes at even distance from root are 2, odd are 1.
# To make the sequence lexicographically smallest, for each component,
# we pick the assignment that results in a smaller value for the smallest index in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            component.append((i, 0))
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        component.append((v, 1 - c))
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Try p_i = 1 for the smallest index i in component
            # If color[i] == 0, then nodes with color 0 get 1, color 1 get 2
            # If color[i] == 1, then nodes with color 1 get 1, color 0 get 2
            # We want the smallest index in component to be 1.
            min_node = min(node for node, c in component)
            c_min = color[min_node]
            
            # Option 1: min_node gets 1
            # If c_min == 0, then color 0 -> 1, color 1 -> 2
            # If c_min == 1, then color 1 -> 1, color 0 -> 2
            
            # Check if this is lexicographically smallest
            # We compare the assignment for the smallest index in the component.
            # Actually, we just need to ensure the smallest index in the component gets 1.
            for node, c in component:
                if c == c_min:
                    res[node] = 1
                else:
                    res[node] = 2
                    
    print("YES")
    print(*(res[1:]))

solve()
```
