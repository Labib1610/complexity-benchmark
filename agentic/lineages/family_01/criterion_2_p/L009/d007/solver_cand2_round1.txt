```python
import sys

# The problem states:
# I1 = [0, 2], I2 = [1, 3]
# Intersection of I1 and I1: [0, 2], length 2
# Intersection of I2 and I2: [1, 3], length 2
# Intersection of I1 and I2: [1, 2], length 1
# Intersection of I2 and I1: [1, 2], length 1
# Stability condition: length of intersection must be 1.
# This means p_u must be different from p_v.
# If p_u = 1, p_v must be 2. If p_u = 2, p_v must be 1.
# This is a standard 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we prefer 1 over 2.
    # We iterate from 1 to n. If a node is uncolored, start BFS/DFS.
    # Since we want lexicographically smallest, we try assigning 1 first.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
