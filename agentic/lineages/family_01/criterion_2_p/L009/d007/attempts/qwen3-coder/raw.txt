import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print("1 " * n)
        return
    
    # Build adjacency list
    graph = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        graph[u].append(v)
        graph[v].append(u)
        edges.append((u, v))
    
    # Color nodes with BFS (2-coloring)
    color = [0] * (n + 1)  # 0: unvisited, 1: color 1, 2: color 2
    visited = [False] * (n + 1)
    
    for start_node in range(1, n + 1):
        if not visited[start_node]:
            queue = deque([start_node])
            color[start_node] = 1
            visited[start_node] = True
            
            while queue:
                node = queue.popleft()
                
                for neighbor in graph[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        color[neighbor] = 3 - color[node]  # Flip color (1->2, 2->1)
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # Verify the solution
    for u, v in edges:
        if color[u] + color[v] != 3:
            print("NO")
            return
    
    # Output the solution (lexicographically smallest: prefer 1 over 2)
    print("YES")
    result = []
    for i in range(1, n + 1):
        result.append(str(color[i]))
    print(" ".join(result))

solve()
