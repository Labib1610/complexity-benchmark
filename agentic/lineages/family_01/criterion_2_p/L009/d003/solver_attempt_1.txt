import sys

# The problem requires that for every edge (u, v), c_u + c_v = 3.
# Since c_i is in {1, 2}, this means c_u must be different from c_v.
# Specifically, if c_u = 1, then c_v = 2, and if c_u = 2, then c_v = 1.
# This is equivalent to checking if the graph is bipartite.
# To minimize the sequence c_1, c_2, ..., c_n lexicographically,
# for each connected component, we should try to set the smallest possible
# value for the smallest index node in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To minimize lexicographically, we want c_i = 1 if possible.
            stack = [(i, 1)]
            colors[i] = 1
            component = []
            
            # Use a stack for DFS to traverse the component
            comp_nodes = []
            q = [i]
            colors[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                comp_nodes.append(u)
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        q.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
