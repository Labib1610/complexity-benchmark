import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition is integral from 0 to 1 of z^(x+y) dz = 0.25
    # [z^(x+y+1) / (x+y+1)] from 0 to 1 = 1 / (x+y+1) = 0.25
    # x + y + 1 = 4 => x + y = 3
    # Since x, y in {1, 2}, the only way to get x + y = 3 is {1, 2}.
    # This means every edge must connect a machine of tier 1 to a machine of tier 2.
    # This is equivalent to checking if the graph is bipartite.
    
    colors = [0] * (n + 1)
    result = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Use BFS/DFS to color components
            # To get lexicographically smallest, we prefer 1 over 2
            stack = [(i, 1)]
            colors[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
            
            # For each component, we have two choices:
            # 1. Start with 1 (colors 1->1, 2->2)
            # 2. Start with 2 (colors 1->2, 2->1)
            # We want the lexicographically smallest sequence.
            # Compare the two options for the current component.
            
            comp_nodes = sorted([node for node, c in component])
            
            # Option 1: node i has color colors[i]
            opt1 = {}
            for node, c in component:
                opt1[node] = colors[node]
            
            # Option 2: node i has color 3 - colors[i]
            opt2 = {}
            for node, c in component:
                opt2[node] = 3 - colors[node]
            
            # Compare lexicographically
            is_opt1_better = False
            for node in comp_nodes:
                if opt1[node] < opt2[node]:
                    is_opt1_better = True
                    break
                elif opt1[node] > opt2[node]:
                    is_opt1_better = False
                    break
            
            best = opt1 if is_opt1_better else opt2
            for node in comp_nodes:
                result[node] = best[node]
                
    print("YES")
    print(*(result[1:]))

solve()
