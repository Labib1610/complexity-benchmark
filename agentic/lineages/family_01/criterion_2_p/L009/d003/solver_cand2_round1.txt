```python
import sys

# The problem states:
# Each workflow (u, v) requires:
# 1. Local cache (Env 1): At least one of {u, v} must be in Env 1.
# 2. Distributed ledger (Env 2): At least one of {u, v} must be in Env 2.
#
# Let e_i be the environment of microservice i (1 or 2).
# Condition 1: (e_u == 1) OR (e_v == 1)
# Condition 2: (e_u == 2) OR (e_v == 2)
#
# This is equivalent to:
# (e_u == 1 or e_v == 1) AND (e_u == 2 or e_v == 2)
# This simplifies to: e_u != e_v.
#
# Proof:
# If e_u == e_v:
#   If e_u = 1, then e_v = 1. Condition 2 fails (no one in Env 2).
#   If e_u = 2, then e_v = 2. Condition 1 fails (no one in Env 1).
# If e_u != e_v:
#   One is 1, one is 2. Both conditions are satisfied.
#
# So the problem is to color the graph with 2 colors such that every edge (u, v)
# has u and v colored differently. This is exactly the definition of a bipartite graph.
# We need to find the lexicographically smallest assignment.
# To get the lexicographically smallest, we greedily assign 1 to the first node
# of each connected component, then propagate the coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for component
            # To be lexicographically smallest, we prefer color 1 for the smallest node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
