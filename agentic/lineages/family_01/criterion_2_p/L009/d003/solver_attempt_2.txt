import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color array: 0 = unassigned, 1 = tier 1, 2 = tier 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is unvisited, we start a BFS/DFS.
    # To ensure lexicographical smallest, we prefer assigning 1 to the 
    # smallest available node in each connected component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign color 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                # Sort neighbors to process in a deterministic way if needed,
                # but here we just need to ensure consistency.
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
