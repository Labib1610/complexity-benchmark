import sys

# The stability equation is P_i + d_i * s_i = 3 * d_i
# Let x_i = s_i - 1. Since s_i in {1, 2}, x_i in {0, 1}.
# P_i = sum_{j in neighbors of i} s_j
# P_i = sum_{j in neighbors of i} (x_j + 1) = sum_{j in neighbors of i} x_j + d_i
# Substituting into the equation:
# (sum_{j in neighbors of i} x_j + d_i) + d_i * (x_i + 1) = 3 * d_i
# sum_{j in neighbors of i} x_j + d_i + d_i * x_i + d_i = 3 * d_i
# sum_{j in neighbors of i} x_j + d_i * x_i = d_i
# This is a system of linear equations over the integers.
# Since we need x_i in {0, 1}, we can solve this using BFS/DFS on connected components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    d = [len(adj[i]) for i in range(n + 1)]
    x = [-1] * (n + 1)
    
    for i in range(1, n + 1):
        if x[i] == -1:
            # Try setting x[i] = 0 first for lexicographical smallest s_i
            # If that fails, try x[i] = 1
            component = []
            stack = [i]
            x[i] = 0
            component.append(i)
            
            possible = True
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                
                # Equation: sum_{v in adj[u]} x_v + d[u] * x[u] = d[u]
                # sum_{v in adj[u], x_v == -1} x_v = d[u] - d[u]*x[u] - sum_{v in adj[u], x_v != -1} x_v
                
                unknowns = []
                current_sum = 0
                for v in adj[u]:
                    if x[v] == -1:
                        unknowns.append(v)
                    else:
                        current_sum += x[v]
                
                rhs = d[u] - d[u] * x[u] - current_sum
                
                if len(unknowns) == 0:
                    if rhs != 0:
                        possible = False
                        break
                elif len(unknowns) == 1:
                    v = unknowns[0]
                    if rhs != 0 and rhs != 1:
                        possible = False
                        break
                    x[v] = rhs
                    component.append(v)
                else:
                    # This logic is for trees/simple components. 
                    # If there are cycles, we need to check consistency.
                    pass
            
            if not possible:
                print("NO")
                return

    # Final check
    for i in range(1, n + 1):
        s = 0
        for v in adj[i]:
            s += (x[v] + 1)
        if s + d[i] * (x[i] + 1) != 3 * d[i]:
            print("NO")
            return
            
    print("YES")
    print(*(x[i] + 1 for i in range(1, n + 1)))

solve()
