```python
import sys

# The problem states that for an interference link between u and v,
# their broadcast intervals (s_u, s_u + 1) and (s_v, s_v + 1) must be disjoint.
# If s_u = 1, interval is (1, 2).
# If s_u = 2, interval is (2, 3).
# These intervals are disjoint if and only if s_u != s_v.
# This is a standard 2-coloring problem on a graph.
# We need to check if the graph is bipartite.
# If it is, we want the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for each node.
# We can iterate from 1 to n, and if a node is not colored,
# perform BFS/DFS to color its connected component.
# To make it lexicographically smallest, we assign 1 to the first node
# of each component, then propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we try to assign 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, color[v]))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
