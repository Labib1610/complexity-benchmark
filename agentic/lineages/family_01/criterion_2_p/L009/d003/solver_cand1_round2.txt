```python
import sys

# The condition for the matrix [[x, 1], [1, y]] to be invertible and have an integer inverse
# is that its determinant must be 1 or -1.
# det = xy - 1.
# So, xy - 1 = 1  => xy = 2
# OR xy - 1 = -1 => xy = 0
# Since x, y are in {1, 2}, xy = 0 is impossible.
# Thus, xy = 2.
# This means one machine must be 1 and the other must be 2.
# This is a bipartite coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for each connected component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    # We want the lexicographically smallest assignment.
    # For each component, we have two choices:
    # 1. Assign color 1 to mode 1, color 2 to mode 2.
    # 2. Assign color 1 to mode 2, color 2 to mode 1.
    # To make it lexicographically smallest, for each component,
    # we look at the smallest index node 'i' in that component.
    # If color[i] == 1, we prefer mode 1. If color[i] == 2, we prefer mode 1.
    # Actually, for each component, we just need to ensure the smallest index
    # node gets mode 1 if possible.
    
    visited = [False] * (n + 1)
    final_modes = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Find the smallest index in this component
            min_node = min(component)
            # If color[min_node] is 1, we keep the current coloring (1->1, 2->2)
            # If color[min_node] is 2, we flip the coloring (1->2, 2->1)
            # This ensures min_node gets mode 1.
            flip = (color[min_node] == 2)
            
            for node in component:
                if flip:
                    final_modes[node] = 3 - color[node]
                else:
                    final_modes[node] = color[node]
                    
    print(*(final_modes[1:]))

solve()
```
