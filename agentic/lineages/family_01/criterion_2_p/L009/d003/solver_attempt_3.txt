```python
import sys

# The problem asks for a 2-coloring of a graph where each edge (u, v)
# must have states s[u] + s[v] = 3. Since s[i] is in {1, 2},
# this means s[u] != s[v].
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest assignment, we want s[i] = 1
# whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # state: 0 = unvisited, 1 = state 1, 2 = state 2
    state = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we prefer state 1 for the root
            stack = [i]
            state[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = 3 - state[u]
                        stack.append(v)
                    elif state[v] == state[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(state[1:]))

solve()
```
