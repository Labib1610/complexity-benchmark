```python
import sys

# The condition c_u + c_v = 3 means:
# If c_u = 1, then c_v = 2.
# If c_u = 2, then c_v = 1.
# This is a standard 2-coloring problem on a graph.
# If a link is (u, u), then c_u + c_u = 3 => 2*c_u = 3, which is impossible.
# So if there is a self-loop, output NO.
# Otherwise, the graph must be bipartite.
# To minimize c_1, then c_2, ..., we greedily assign 1 to the first node 
# of each connected component that hasn't been colored yet.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the component
            # To minimize c_i lexicographically, we assign 1 to the 
            # first uncolored node we encounter.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
