```python
import sys

# The problem defines two permutations:
# A1 = [1, 2, 3] (identity)
# A2 = [2, 1, 3] (swap 1 and 2)
#
# Let V_u = A_{p_u} and V_v = A_{p_v}.
# C[k] = V_u[V_v[k]]
#
# If p_u = 1, V_u = [1, 2, 3], C = [1, 2, 3] (A1). Inversions: 0.
# If p_u = 2, V_u = [2, 1, 3], C = [V_u[V_v[1]], V_u[V_v[2]], V_u[V_v[3]]]
# If p_v = 1, V_v = [1, 2, 3], C = [2, 1, 3] (A2). Inversions: 1.
# If p_v = 2, V_v = [2, 1, 3], C = [1, 2, 3] (A1). Inversions: 0.
#
# Summary of inversions in C:
# p_u=1, p_v=1 => 0
# p_u=1, p_v=2 => 1
# p_u=2, p_v=1 => 1
# p_u=2, p_v=2 => 0
#
# We need exactly 1 inversion for every bridge (u, v).
# This means p_u != p_v.
# This is a bipartite graph coloring problem.
# We need to color the graph with 2 colors (1 and 2) such that no two adjacent
# nodes have the same color.
# To get the lexicographically smallest assignment, we greedily assign 1 to
# the smallest index in each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for component
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
