import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    # components stores list of nodes in each connected component
    # component_id[u] stores which component node u belongs to
    component_id = [0] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == 0:
            comp = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                comp.append((u, c))
                component_id[u] = len(components)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            components.append(comp)
            
    # For each component, there are two possible assignments (flip colors)
    # Let the two assignments be A and B.
    # A: node u has color color[u]
    # B: node u has color 3 - color[u]
    # The protocol: for i = 1 to n, if possible, pick Grid 1.
    # This is equivalent to: for each component, pick the assignment where
    # the first node (smallest index) in that component is 1.
    
    res = [0] * (n + 1)
    for comp in components:
        # Find the smallest index in this component
        min_node = min(node for node, c in comp)
        # Find the color of min_node in the current DFS
        for node, c in comp:
            if node == min_node:
                target_color = c
                break
        
        # We want min_node to be 1.
        # If target_color is 1, we keep the current colors.
        # If target_color is 2, we flip all colors in this component.
        flip = (target_color == 2)
        for node, c in comp:
            if flip:
                res[node] = 3 - c
            else:
                res[node] = c
                
    print("YES")
    print(*(res[1:]))

solve()
