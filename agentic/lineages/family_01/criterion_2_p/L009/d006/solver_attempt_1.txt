```python
import sys

# The condition c_u + c_v = 3 means:
# If c_u = 1, then c_v = 2.
# If c_u = 2, then c_v = 1.
# This is equivalent to saying c_u != c_v.
# This is a standard bipartite graph coloring problem.
# We need to check if the graph is bipartite.
# If it is, we want to minimize c_1, then c_2, ..., c_n.
# Since c_i can be 1 or 2, we want to assign 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To minimize c_1, c_2, ..., c_n lexicographically:
    # We iterate i from 1 to n. If color[i] is not set, we set it to 1.
    # Then we propagate the constraints using BFS/DFS.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
