import sys

# The condition v_x + v_y = 3 means v_x != v_y.
# This is a 2-coloring problem on a graph where edges represent inequality.
# If the graph is bipartite, we can color it with 1 and 2.
# To minimize v_1, v_2, ..., v_n lexicographically:
# For each connected component, we have two possible colorings (swap 1s and 2s).
# Let the two colorings be C1 and C2.
# We want to pick the coloring that makes the sequence (v_1, ..., v_n) lexicographically smallest.
# For a component, let the nodes be {u_1, u_2, ...} where u_1 is the smallest index.
# We must assign v_{u_1} = 1 to minimize the sequence.
# Once v_{u_1} is fixed to 1, all other nodes in the component are determined.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            # to ensure we set the smallest index to 1
            idx_in_comp = 0
            q = [i]
            color[i] = 1
            
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now color[v] is 1 or 2.
            # To minimize lexicographically, we want the smallest index in this
            # component to have value 1.
            # Find the smallest index in this component.
            min_node = min(q)
            target_val = 1
            current_val_of_min = color[min_node]
            
            if current_val_of_min == 1:
                for node in q:
                    ans[node] = color[node]
            else:
                for node in q:
                    ans[node] = 3 - color[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
