import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Total Harmonic Potential = sum(d_i * s_i^2) - sum_{(u,v) in E} (2 * s_u * s_v)
    # Let s_i = 1 + x_i, where x_i in {0, 1}.
    # s_i^2 = 1 + 2x_i + x_i^2 = 1 + 3x_i (since x_i^2 = x_i)
    # Potential = sum(d_i * (1 + 3x_i)) - sum_{(u,v) in E} (2 * (1 + x_u)(1 + x_v))
    # Potential = sum(d_i) + 3*sum(d_i * x_i) - 2 * sum_{(u,v) in E} (1 + x_u + x_v + x_u * x_v)
    # Since sum(d_i) = 2m:
    # Potential = 2m + 3*sum(d_i * x_i) - 2m - 2*sum(x_u + x_v) - 2*sum(x_u * x_v)
    # Potential = 3*sum(d_i * x_i) - 2*sum(d_i * x_i) - 2*sum(x_u * x_v)
    # Potential = sum(d_i * x_i) - 2*sum(x_u * x_v)
    # We want Potential = m.
    # sum(d_i * x_i) - 2*sum(x_u * x_v) = m
    
    # This is equivalent to: sum_{i: x_i=1} d_i - 2 * (number of edges with both endpoints 1) = m
    # Let S be the set of vertices where x_i = 1.
    # Let E(S) be the number of edges with both endpoints in S.
    # sum_{v in S} d_v - 2 * E(S) = m
    # This is exactly the formula for the number of edges between S and V\S.
    # Let cut(S, V\S) be the number of edges between S and V\S.
    # We need cut(S, V\S) = m.
    # Since the total number of edges is m, this means there are no edges within S and no edges within V\S.
    # This means S and V\S must be a bipartition of the graph.
    
    # If the graph is bipartite, we can color it with 2 colors.
    # Let the two partitions be A and B.
    # Possible sets S:
    # 1. S = A (then V\S = B). cut(A, B) = m. This works.
    # 2. S = B (then V\S = A). cut(B, A) = m. This works.
    # 3. S = empty set. cut(empty, V) = 0. Only works if m=0.
    # 4. S = V. cut(V, empty) = 0. Only works if m=0.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, we have two partitions: color 0 and color 1.
    # S can be {i | color[i] == 0} or {i | color[i] == 1}.
    # x_i = 1 means s_i = 2, x_i = 0 means s_i = 1.
    # We want lexicographically smallest s.
    # Option 1: S = {i | color[i] == 0} => s_i = 2 if color[i]==0 else 1
    # Option 2: S = {i | color[i] == 1} => s_i = 2 if color[i]==1 else 1
    
    res1 = [2 if color[i] == 0 else 1 for i in range(1, n + 1)]
    res2 = [2 if color[i] == 1 else 1 for i in range(1, n + 1)]
    
    ans = res1 if res1 < res2 else res2
    print("YES")
    print(*(ans))

solve()
