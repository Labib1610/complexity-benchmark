```python
import sys

# The condition: x^2 - (t_u + t_v)x + 2 = 0 has integer roots.
# Let the roots be r1, r2. Then r1 + r2 = t_u + t_v and r1 * r2 = 2.
# Since r1, r2 are integers, the only pairs (r1, r2) such that r1 * r2 = 2 are:
# (1, 2) or (2, 1) or (-1, -2) or (-2, -1).
# Case 1: r1 + r2 = 1 + 2 = 3. Then t_u + t_v = 3.
# Since t_u, t_v are in {1, 2}, this implies {t_u, t_v} = {1, 2}.
# Case 2: r1 + r2 = -1 + -2 = -3. Then t_u + t_v = -3.
# This is impossible since t_u, t_v >= 1.
# Thus, the condition is t_u + t_v = 3, which means t_u != t_v.
# This is a standard 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If an account is not colored, we assign it 1 (the smallest possible).
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
