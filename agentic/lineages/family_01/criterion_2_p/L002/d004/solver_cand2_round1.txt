import sys

# The resonance condition: |det(M_{u,v})| = 1
# M_{u,v} = [[s_u, s_v, 1], [s_v, s_u, 1], [1, 1, 1]]
# det(M_{u,v}) = s_u(s_u - 1) - s_v(s_v - 1) + 1(s_v - s_u)
# det(M_{u,v}) = s_u^2 - s_u - s_v^2 + s_v + s_v - s_u
# det(M_{u,v}) = s_u^2 - 2s_u - s_v^2 + 2s_v
# det(M_{u,v}) = (s_u - 1)^2 - 1 - ((s_v - 1)^2 - 1)
# det(M_{u,v}) = (s_u - 1)^2 - (s_v - 1)^2
# |(s_u - 1)^2 - (s_v - 1)^2| = 1
# Since s_i >= 1, let x_i = s_i - 1, where x_i >= 0.
# |x_u^2 - x_v^2| = 1
# Since x_i are integers, the only way for the difference of two squares to be 1
# is if {x_u^2, x_v^2} = {0, 1}.
# Thus, {x_u, x_v} = {0, 1}.
# This means {s_u - 1, s_v - 1} = {0, 1}, so {s_u, s_v} = {1, 2}.
# This implies the graph must be bipartite, with one set assigned 1 and the other 2.
# To minimize F = sum s_i * 10^{n-i}, we want smaller values at earlier indices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            # BFS to check bipartiteness and assign colors
            # To minimize F, we greedily assign 1 to the first node in each component
            # if possible, but the constraint is fixed by the bipartite structure.
            # Actually, for each component, we have two choices:
            # Choice A: color[i] = 0 (s_i = 1), neighbors = 1 (s_j = 2)
            # Choice B: color[i] = 1 (s_i = 2), neighbors = 0 (s_j = 1)
            # We compare the contribution to F for both choices.
            
            component = []
            stack = [i]
            color[i] = 0
            component.append(i)
            
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Evaluate cost for both colorings
            cost0 = 0
            cost1 = 0
            for node in component:
                val0 = 1 if color[node] == 0 else 2
                val1 = 2 if color[node] == 0 else 1
                # F = sum s_i * 10^{n-i}
                # We only need to compare the sum of s_i * 10^{n-i}
                cost0 += val0 * (10**(n - node))
                cost1 += val1 * (10**(n - node))
            
            if cost1 < cost0:
                for node in component:
                    color[node] = 1 - color[node]

    print("YES")
    res = []
    for i in range(1, n + 1):
        res.append(str(1 if color[i] == 0 else 2))
    print(" ".join(res))

solve()
