import sys

# The condition p_u | p_v = 3 means that if p_u = 1, then p_v = 2, and if p_u = 2, then p_v = 1.
# This is equivalent to p_u != p_v.
# This is a standard 2-coloring problem on a graph.
# A graph is 2-colorable if and only if it is bipartite.
# We need the lexicographically smallest assignment (1s before 2s).
# For each connected component, we have two choices:
# 1. Assign the first node in the component to 1.
# 2. Assign the first node in the component to 2.
# To get the lexicographically smallest result, for each component, we check which
# assignment makes the first node (by index) 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Component found. Try setting color[i] = 1
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            idx_in_comp = 0
            while idx_in_comp < len(component):
                u = component[idx_in_comp]
                idx_in_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible: break
            
            if not possible:
                print("NO")
                return
            
            # Check if setting color[i] = 1 is better than color[i] = 2
            # We want the smallest value at the smallest index in the component.
            # Let's find the smallest index in this component.
            min_node = min(component)
            if color[min_node] == 2:
                # Flip colors
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
