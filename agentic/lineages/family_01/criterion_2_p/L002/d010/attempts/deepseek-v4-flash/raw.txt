import sys
sys.setrecursionlimit(1 << 20)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    g = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        g[u].append(v)
        g[v].append(u)

    color = [-1] * n
    ans = [0] * n

    for start in range(n):
        if color[start] != -1:
            continue
        stack = [start]
        color[start] = 0
        comp = []
        while stack:
            u = stack.pop()
            comp.append(u)
            for v in g[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    stack.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
        # Now we have a bipartite component with colors 0/1
        # We need to assign phases 1 or 2 to each node.
        # Constraint: p_u | p_v = 3 => (p_u, p_v) in {(1,2),(2,1),(2,2)}
        # So only (1,1) is forbidden.
        # For each edge, we cannot have both endpoints = 1.
        # So nodes with color 0 and color 1: we can set one side to 1 and the other to 2,
        # or both sides to 2. But to get lexicographically smallest, we try to put 1 as early as possible.
        # In a bipartite component, we can choose which color gets 1 and which gets 2.
        # But if we set both to 2, that's also valid but lexicographically larger.
        # So we try: option A: color0 -> 1, color1 -> 2
        # option B: color0 -> 2, color1 -> 1
        # choose lexicographically smaller for the whole component.
        # But we must consider global lexicographic order, so we process components in order of smallest index.
        # We'll store component nodes and decide later.

        # For now, store component and its two possible assignments
        comp.sort()
        # Try option A
        assignA = [0] * n
        for u in comp:
            assignA[u] = 1 if color[u] == 0 else 2
        # Try option B
        assignB = [0] * n
        for u in comp:
            assignB[u] = 2 if color[u] == 0 else 1

        # Compare lexicographically for this component
        # We'll fill into ans later, but we need to compare with existing ans for earlier components.
        # Since we process components in order of smallest index, we can just pick the smaller for this component.
        # But we need to compare only within this component's positions.
        # We'll pick the smaller one.
        # Compare assignA vs assignB on comp positions
        for u in comp:
            if assignA[u] < assignB[u]:
                for x in comp:
                    ans[x] = assignA[x]
                break
            elif assignA[u] > assignB[u]:
                for x in comp:
                    ans[x] = assignB[x]
                break
        else:
            # equal, pick A
            for x in comp:
                ans[x] = assignA[x]

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
