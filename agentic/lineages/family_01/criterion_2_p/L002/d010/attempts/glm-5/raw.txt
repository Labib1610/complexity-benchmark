```python
import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    order = list(range(1, n + 1))
    order.sort(key=lambda x: len(adj[x]), reverse=True)

    for s in order:
        if color[s] != 0:
            continue
        color[s] = 1
        stack = [s]
        while stack:
            u = stack.pop()
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - color[u]
                    stack.append(v)

    for u in range(1, n + 1):
        for v in adj[u]:
            if color[u] | color[v] != 3:
                print("NO")
                return

    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

solve()
```
