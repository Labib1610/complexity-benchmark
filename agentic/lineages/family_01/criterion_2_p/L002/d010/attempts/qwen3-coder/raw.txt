```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list
    adj = defaultdict(list)
    index = 2
    edges = []
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        index += 2
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # If no constraints, assign all 1 (lexicographically smallest)
    if m == 0:
        print("YES")
        print(' '.join(['1'] * n))
        return
    
    # Try to 2-color the graph (check bipartiteness)
    # Each node can be phase 1 or 2. The constraint u OR v = 3 means u != v.
    # So this is equivalent to a 2-coloring problem.
    color = [-1] * n  # -1: unvisited, 0: color 1, 1: color 2
    
    # Check if the graph is bipartite
    visited = [False] * n
    
    for start in range(n):
        if visited[start]:
            continue
        queue = deque()
        queue.append(start)
        color[start] = 0
        visited[start] = True
        
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    color[v] = 1 - color[u]
                    queue.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
    
    # The graph is bipartite. Now we need the lexicographically smallest assignment.
    # We want to assign 1's and 2's such that:
    #   - Each connected component is colored with 0 and 1.
    #   - We choose the coloring that gives lexicographically smallest sequence.
    
    # For each connected component, we have two choices of coloring:
    #   - color 0 -> 1, color 1 -> 2
    #   - color 0 -> 2, color 1 -> 1
    # To get lex smallest, among these two, pick the one that gives smaller value for node 0,
    # then node 1, etc.
    
    # Let's do BFS again to determine components and their colorings
    visited = [False] * n
    final_color = [0] * n  # Will store 1 or 2
    
    for start in range(n):
        if visited[start]:
            continue
        # Collect all nodes in this component
        component_nodes = []
        queue = deque()
        queue.append(start)
        visited[start] = True
        component_nodes.append(start)
        
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    component_nodes.append(v)
                    queue.append(v)
        
        # Determine the best coloring for this component
        # Try assigning color 0 -> 1, 1 -> 2
        option1 = {}
        for node in component_nodes:
            c = color[node]
            option1[node] = 1 if c == 0 else 2
        
        # Try assigning color 0 -> 2, 1 -> 1
        option2 = {}
        for node in component_nodes:
            c = color[node]
            option2[node] = 2 if c == 0 else 1
            
        # Compare lexicographically
        use_option1 = True
        for node in sorted(component_nodes):
            if option1[node] < option2[node]:
                use_option1 = True
                break
            elif option1[node] > option2[node]:
                use_option1 = False
                break
        
        chosen_option = option1 if use_option1 else option2
        for node in component_nodes:
            final_color[node] = chosen_option[node]
    
    # Verify the solution satisfies all constraints
    for u, v in edges:
        if (final_color[u] | final_color[v]) != 3:
            print("NO")
            return
            
    print("YES")
    print(' '.join(map(str, final_color)))

if __name__ == "__main__":
    main()
```
