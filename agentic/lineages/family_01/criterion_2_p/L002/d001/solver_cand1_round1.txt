import sys

# The condition is: priority(u) XOR priority(v) = 3.
# Possible priorities are {1, 2}.
# 1 XOR 2 = 3.
# 1 XOR 1 = 0, 2 XOR 2 = 0.
# So, for every pair (u, v), priority(u) must be different from priority(v).
# This means the graph must be bipartite.
# To get the lexicographically smallest assignment, we want priority(i) = 1
# whenever possible.
# For each connected component, we have two choices:
# 1. Assign 1 to the smallest index node in the component.
# 2. Assign 2 to the smallest index node in the component.
# We compare the resulting assignments lexicographically.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            idx_in_comp = 0
            while idx_in_comp < len(component):
                u = component[idx_in_comp]
                idx_in_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Try two assignments for this component
            # Option 1: color[i] = 1
            # Option 2: color[i] = 2 (which means flipping all colors in component)
            
            # Generate assignment 1
            assign1 = {}
            for node in component:
                assign1[node] = color[node]
            
            # Generate assignment 2
            assign2 = {}
            for node in component:
                assign2[node] = 3 - color[node]
            
            # Compare lexicographically
            # We only care about the smallest index in the component
            # because that determines the order for the whole component.
            # Actually, we need to compare the full sequence.
            # Since we process components independently, we just need to pick
            # the one that makes the smallest index in the component have 1.
            
            # Find the smallest index in this component
            min_node = min(component)
            if assign1[min_node] < assign2[min_node]:
                for node in component:
                    ans[node] = assign1[node]
            else:
                for node in component:
                    ans[node] = assign2[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
