import sys

# The condition: product of clearance levels (1 or 2) must be 2.
# If u and v have product 2, one must be 1 and the other must be 2.
# This is a bipartite graph coloring problem.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we want 1s as early as possible.
# For each connected component, we have two choices:
# 1. Assign 1 to the first node (smallest index) in the component.
# 2. Assign 2 to the first node (smallest index) in the component.
# We should pick the assignment that results in a smaller value at the first node.
# Since we want the lexicographically smallest sequence, we iterate from 1 to n.
# If a node is unvisited, we start a BFS/DFS.
# To make it lexicographically smallest, we should try to assign 1 to the current node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            idx_stack = 0
            while idx_stack < len(component):
                u, c = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if we should flip colors to make the first node 1
            # The first node in the component is i.
            # If color[i] is 2, we flip all colors in this component.
            if color[i] == 2:
                for node, c in component:
                    color[node] = 3 - c

solve()
