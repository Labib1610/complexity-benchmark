```python
import sys

# The problem states that for any pair (u, v) sharing a device, 
# block(u) + block(v) = 3.
# Since block labels are {1, 2}, this means block(u) != block(v).
# This is a classic 2-coloring problem on a graph.
# We need to find the lexicographically smallest assignment.
# To get the lexicographically smallest assignment, we want to assign 1 
# to the smallest possible index whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is unassigned, we assign it 1 (the smallest possible value).
    # Then we perform BFS/DFS to color the connected component.
    for i in range(1, n + 1):
        if assignment[i] == 0:
            assignment[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    solve()
```
