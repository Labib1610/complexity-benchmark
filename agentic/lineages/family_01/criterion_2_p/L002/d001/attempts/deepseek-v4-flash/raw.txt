We can model this as a graph bipartiteness check.  
If the graph is bipartite, we can assign solvents 1 and 2 to the two partitions.  
To get the lexicographically smallest assignment, we try to assign material 1 to solvent 1, then run BFS/DFS coloring, and if that fails, try assigning material 1 to solvent 2.

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

def try_assign(start_color):
    color = [-1] * n
    color[0] = start_color
    q = deque([0])
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = 1 - color[u]
                q.append(v)
            elif color[v] == color[u]:
                return None
    # color any remaining components
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        return None
    return color

# Try starting with 1
res = try_assign(0)
if res is None:
    res = try_assign(1)

if res is None:
    print("NO")
else:
    print("YES")
    print(" ".join(str(c + 1) for c in res))
```
