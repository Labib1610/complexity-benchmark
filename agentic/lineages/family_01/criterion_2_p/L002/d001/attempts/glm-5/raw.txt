import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    possible = True
    
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        # BFS to check bipartiteness
        queue = [start]
        color[start] = 1
        idx = 0
        comp = []
        while idx < len(queue):
            u = queue[idx]
            idx += 1
            comp.append(u)
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = -color[u]
                    queue.append(v)
                elif color[v] == color[u]:
                    possible = False
                    break
            if not possible:
                break
        if not possible:
            break
        # Determine which color should be 1 for lexicographically smallest
        # We want as many small indices as possible to be 1
        # The component is bipartite with colors 1 and -1
        # We need to decide which side gets value 1
        # To minimize lexicographically, we want the first node in the component
        # (which is the smallest index in this component) to be 1
        # Since start is the smallest index in its component (we iterate in order),
        # and color[start] = 1, we keep color 1 as value 1
        # So no flip needed
        for u in comp:
            if color[u] == 1:
                color[u] = 1
            else:
                color[u] = 2
    
    if not possible:
        sys.stdout.write("NO\n")
        return
    
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(str(color[i]) for i in range(1, n + 1)) + "\n")

solve()
