from collections import deque

def solve():
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr += 1
    m = int(input[ptr])
    ptr += 1

        v = int(input[ptr + 1])

                        queue.append(v)
                    elif color[v] == color[u]:

                        return
    
    print("YES")
    print(" ".join(map(str, color[1:])))
