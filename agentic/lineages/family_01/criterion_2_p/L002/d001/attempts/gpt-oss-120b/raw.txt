import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    g = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        g[u].append(v)
        g[v].append(u)

    color = [ -1 ] * (n + 1)          # 0 / 1 parity
    ans = [0] * (n + 1)

    from collections import deque

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        comp = [start]
        min_vertex = start
        while q:
            u = q.popleft()
            if u < min_vertex:
                min_vertex = u
            for v in g[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                    comp.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
        # decide mapping so that min_vertex gets label 1
        # if its parity is 0 -> parity0 -> 1, else parity0 -> 2
        if color[min_vertex] == 0:
            for v in comp:
                ans[v] = 1 if color[v] == 0 else 2
        else:
            for v in comp:
                ans[v] = 2 if color[v] == 0 else 1

    print("YES")
    print(' '.join(str(ans[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
