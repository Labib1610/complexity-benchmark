import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # states: 0 = unassigned, 1 = State 1, 2 = State 2
    states = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a component is unassigned, we assign it State 1 and perform BFS/DFS.
    for i in range(1, n + 1):
        if states[i] == 0:
            # Start BFS/DFS from i, assigning State 1
            stack = [i]
            states[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if states[v] == 0:
                        states[v] = 3 - states[u]
                        stack.append(v)
                    elif states[v] == states[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(states[1:]))

if __name__ == '__main__':
    solve()
