```python
import sys

# The condition: (clearance[u] ^ clearance[v]) == 3
# Possible clearance levels are 1 (binary 01) and 2 (binary 10).
# 1 ^ 2 = 3 (binary 11).
# This means for every interaction (u, v), clearance[u] must be different from clearance[v].
# This is a standard bipartite graph coloring problem.
# We want the lexicographically smallest sequence (1, 2, 1, 2...).
# We iterate from 1 to n, if not visited, assign 1 (the smaller value) and perform BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS, assign 1 to the first node of the component
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```
