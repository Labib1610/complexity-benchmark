```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    parity = [-1] * (n + 1)   # 0 or 1
    res = [0] * (n + 1)

    for start in range(1, n + 1):
        if parity[start] != -1:
            continue
        # start new component, ensure start gets parity 0 -> value 1
        q = deque([start])
        parity[start] = 0
        comp = [start]
        ok = True
        while q:
            u = q.popleft()
            pu = parity[u]
            for v in adj[u]:
                if parity[v] == -1:
                    parity[v] = pu ^ 1
                    q.append(v)
                    comp.append(v)
                elif parity[v] == pu:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            print("NO")
            return
        # assign values: parity 0 -> 1, parity 1 -> 2
        for v in comp:
            res[v] = 1 if parity[v] == 0 else 2

    print("YES")
    print(' '.join(str(res[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
