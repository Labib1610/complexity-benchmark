import sys

# The condition: gcd(a, b) + lcm(a, b) = 3
# Since lcm(a, b) = (a * b) / gcd(a, b),
# let g = gcd(a, b). Then a = g*x, b = g*y with gcd(x, y) = 1.
# g + (g*x*y) = 3 => g(1 + xy) = 3.
# Since g, x, y are positive integers, g must be a divisor of 3.
# Case 1: g = 1. Then 1 + xy = 3 => xy = 2.
# Possible (x, y) pairs: (1, 2) or (2, 1).
# If (x, y) = (1, 2), then a = 1*1 = 1, b = 1*2 = 2.
# If (x, y) = (2, 1), then a = 1*2 = 2, b = 1*1 = 1.
# Case 2: g = 3. Then 3(1 + xy) = 3 => 1 + xy = 1 => xy = 0.
# Not possible since x, y >= 1.
# Thus, for any edge (u, v), the designations must be {1, 2}.
# This means the graph must be bipartite.
# To get the lexicographically smallest sequence, we want 1s as much as possible.
# For each connected component, we have two choices:
# 1. Assign 1 to nodes in set A, 2 to nodes in set B.
# 2. Assign 2 to nodes in set A, 1 to nodes in set B.
# We choose the one that makes the sequence lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Try both assignments for this component
            # Assignment 1: color 1 -> 1, color 2 -> 2
            # Assignment 2: color 1 -> 2, color 2 -> 1
            # We want the smallest value at the smallest index in the component
            comp_nodes = sorted(component)
            first_node, first_color = comp_nodes[0]
            
            # If first_color is 1, assign 1 to 1s, 2 to 2s
            # If first_color is 2, assign 2 to 1s, 1 to 2s
            # This ensures the smallest index gets 1 if possible
            if first_color == 1:
                for node, c in component:
                    res[node] = c
            else:
                for node, c in component:
                    res[node] = 3 - c
                    
    print("YES")
    print(*(res[1:]))

solve()
