```python
import sys

# The condition: gcd(a, b) + lcm(a, b) = 3
# Possible values for a, b are 1 or 2.
# If a=1, b=1: gcd(1,1) + lcm(1,1) = 1 + 1 = 2 != 3
# If a=1, b=2: gcd(1,2) + lcm(1,2) = 1 + 2 = 3 (Valid)
# If a=2, b=1: gcd(2,1) + lcm(2,1) = 1 + 2 = 3 (Valid)
# If a=2, b=2: gcd(2,2) + lcm(2,2) = 2 + 2 = 4 != 3
# Thus, for every edge (u, v), u and v must have different designations.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we prefer 1 over 2.
    # We iterate 1 to n. If not colored, start BFS/DFS.
    # To ensure lexicographical smallest, we try assigning 1 to the first
    # uncolored node in the component.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
