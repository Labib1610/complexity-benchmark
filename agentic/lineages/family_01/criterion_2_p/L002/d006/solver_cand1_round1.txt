```python
import sys

# The condition: t_u^2 + t_v^2 must be a multiple of 5.
# Possible tiers are 1 and 2.
# 1^2 = 1, 2^2 = 4.
# 1+1 = 2 (not multiple of 5)
# 1+4 = 5 (multiple of 5)
# 4+4 = 8 (not multiple of 5)
# Thus, for every edge (u, v), t_u must be different from t_v.
# This is a bipartite graph problem.
# A graph is bipartite if and only if it contains no odd cycles.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
        else:
            print("NO")
            return
        idx += 2

    color = [0] * (n + 1)
    # 0: unassigned, 1: tier 1, 2: tier 2
    
    # We need to check if a valid assignment exists.
    # The sequential locking procedure:
    # For i = 1 to n:
    #   Can we set i=1? (i.e., is there a valid coloring with i=1 and previous fixed?)
    #   If yes, set i=1. Else, set i=2.
    
    # Since it's a bipartite graph, each connected component can be colored in 2 ways.
    # Let a component have nodes {v1, v2, ...}. If we fix one node, all others are fixed.
    # Let the two possible colorings be C and not-C.
    
    comp_id = [-1] * (n + 1)
    components = []
    for i in range(1, n + 1):
        if comp_id[i] == -1:
            curr_comp = []
            stack = [(i, 1)]
            comp_id[i] = len(components)
            color[i] = 1
            curr_comp.append((i, 1))
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if comp_id[v] == -1:
                        comp_id[v] = len(components)
                        color[v] = 3 - c
                        curr_comp.append((v, color[v]))
                        stack.append((v, color[v]))
                    elif color[v] == c:
                        print("NO")
                        return
            components.append(curr_comp)

    # For each component, we have two choices:
    # Choice 0: original coloring
    # Choice 1: flipped coloring
    # We need to decide for each component which choice to make.
    # The sequential locking:
    # For i=1..n, try setting color[i]=1.
    # If i is in a component already decided, we must follow that decision.
    # If i is in a component not yet decided, we can decide it now.
    
    comp_decided = [False] * len(components)
    comp_choice = [0] * len(components)
    final_colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        c_id = comp_id[i]
        if comp_decided[c_id]:
            final_colors[i] = comp_choice[c_id]
        else:
            # Try setting i to 1
            # If we set i to 1, what are the colors of all nodes in this component?
            # If color[i] == 1, we pick choice 0. If color[i] == 2, we pick choice 1.
            target = 1
            if color[i] == target:
                comp_choice[c_id] = 1 # choice 0
            else:
                comp_choice[c_id] = 2 # choice 1
            
            # Apply to all nodes in component
            for node, c in components[c_id]:
                if comp_choice[c_id] == 1:
                    final_colors[node] = c
                else:
                    final_colors[node] = 3 - c
            comp_decided[c_id] = True
            
    print("YES")
    print(*(final_colors[1:]))

solve()
```
