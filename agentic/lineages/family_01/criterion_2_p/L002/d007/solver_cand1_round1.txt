```python
import sys

# The compliance metric C(t_u, t_v) = P(t_u, t_v) + Q(t_u, t_v) = 1.
# Let's evaluate P and Q for t_u, t_v in {1, 2}:
# P(x, y) = 1 if x=1, y=0 else Q(x-1, y)
# Q(x, y) = 1 if x=0, y=1 else P(x, y-1)
#
# P(1, 1): x=1, y=1. P(1, 1) = Q(0, 1) = 1.
# Q(1, 1): x=1, y=1. Q(1, 1) = P(1, 0) = 1.
# C(1, 1) = 1 + 1 = 2. (Invalid)
#
# P(1, 2): x=1, y=2. P(1, 2) = Q(0, 2) = P(0, 1) = Q(-1, 1) = 0.
# Q(1, 2): x=1, y=2. Q(1, 2) = P(1, 1) = 1.
# C(1, 2) = 0 + 1 = 1. (Valid)
#
# P(2, 1): x=2, y=1. P(2, 1) = Q(1, 1) = 1.
# Q(2, 1): x=2, y=1. Q(2, 1) = P(2, 0) = Q(1, 0) = P(1, -1) = 0.
# C(2, 1) = 1 + 0 = 1. (Valid)
#
# P(2, 2): x=2, y=2. P(2, 2) = Q(1, 2) = P(1, 1) = 1.
# Q(2, 2): x=2, y=2. Q(2, 2) = P(2, 1) = Q(1, 1) = 1.
# C(2, 2) = 1 + 1 = 2. (Invalid)
#
# Conclusion: C(t_u, t_v) = 1 iff t_u != t_v.
# This is a standard bipartite graph 2-coloring problem.
# We need the lexicographically smallest assignment (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, try assigning 1 first
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, color[v]))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
