```python
import sys

# The determinant of the matrix M is:
# det(M) = t_u * (t_v * (t_u + t_v)/3 - 1) - 1 * (1 * (t_u + t_v)/3 - 1) + 1 * (1 - t_v)
# det(M) = t_u * t_v * (t_u + t_v)/3 - t_u - (t_u + t_v)/3 + 1 + 1 - t_v
# det(M) = (t_u * t_v * (t_u + t_v) - (t_u + t_v))/3 - t_u - t_v + 2
# det(M) = 0 => (t_u * t_v - 1) * (t_u + t_v) / 3 - (t_u + t_v) + 2 = 0
# Let's test t_u, t_v in {1, 2}:
# If t_u=1, t_v=1: (1-1)*(2)/3 - 2 + 2 = 0. Valid.
# If t_u=1, t_v=2: (2-1)*(3)/3 - 3 + 2 = 1 - 3 + 2 = 0. Valid.
# If t_u=2, t_v=1: (2-1)*(3)/3 - 3 + 2 = 1 - 3 + 2 = 0. Valid.
# If t_u=2, t_v=2: (4-1)*(4)/3 - 4 + 2 = 4 - 4 + 2 = 2 != 0. Invalid.
# So the condition is: t_u and t_v cannot both be 2.
# This is equivalent to: at least one of t_u, t_v must be 1.
# This is a 2-SAT problem where (t_u=1 OR t_v=1) must be true.
# This is equivalent to: if t_u=2, then t_v=1; if t_v=2, then t_u=1.
# To minimize sum(t_i * 2^(n-i)), we want t_i to be 1 as much as possible for small i.
# Since we want to minimize the value, we greedily try to set t_i = 1.
# If we set t_i = 1, it never forces any other node.
# If we are forced to set t_i = 2, then all neighbors must be 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # t_i = 1 is always safe. t_i = 2 forces neighbors to be 1.
    # If a node is 2, all its neighbors must be 1.
    # If two nodes are connected, they cannot both be 2.
    # This is exactly the Independent Set problem on the graph.
    # We want to pick a set of nodes to be 2 such that no two are adjacent.
    # To minimize sum(t_i * 2^(n-i)), we want to pick the largest possible indices
    # to be 2.
    
    t = [1] * (n + 1)
    # We want to set t_i = 2 for as many i as possible, starting from the end,
    # provided it doesn't violate the independent set constraint.
    # Actually, the polynomial is sum(t_i * 2^(n-i)).
    # t_i = 1 is smaller than t_i = 2.
    # To minimize, we want t_i = 1 for as many small i as possible.
    # This means we want t_i = 2 for as many large i as possible.
    
    # Greedy: iterate i from n down to 1. If we can set t_i = 2, do it.
    # We can set t_i = 2 if no neighbor j has t_j = 2.
    
    for i in range(n, 0, -1):
        can_be_2 = True
        for neighbor in adj[i]:
            if t[neighbor] == 2:
                can_be_2 = False
                break
        if can_be_2:
            t[i] = 2
            
    # Verify
    for u in range(1, n + 1):
        for v in adj[u]:
            if t[u] == 2 and t[v] == 2:
                print("NO")
                return
                
    print("YES")
    print(*(t[1:]))

solve()
```
