import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # t^2 - 2 = 0, no integer solution
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # t^2 - 3t + 2 = 0 => (t-1)(t-2) = 0 => t = 1 or t = 2
    # Determinant: t_u * t_v - 2 = 0 => t_u * t_v = 2
    # Possible pairs (t_u, t_v): (1, 2) or (2, 1)
    # This means the graph must be bipartite, and for every edge (u, v),
    # t_u != t_v.
    
    tiers = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if tiers[i] == 0:
            # BFS to color the component
            # To get lexicographically smallest, we want 1 if possible
            # For each component, we have two choices:
            # 1. Start with 1, then neighbors must be 2, etc.
            # 2. Start with 2, then neighbors must be 1, etc.
            # We compare the two assignments for this component and pick the one
            # that results in a smaller sequence.
            
            component = []
            queue = [i]
            tiers[i] = 1
            component.append(i)
            
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                for v in adj[u]:
                    if tiers[v] == 0:
                        tiers[v] = 3 - tiers[u]
                        component.append(v)
                        queue.append(v)
                    elif tiers[v] == tiers[u]:
                        print("NO")
                        return
            
            # Check if swapping 1s and 2s in this component is better
            # The current assignment is 'tiers'. Let's see if swapping is better.
            # We only care about the first node in the component to decide.
            # Actually, for lexicographical smallest, we want the smallest value
            # at the smallest index.
            
            # Let's re-evaluate: for each component, we have two valid colorings.
            # Let C be the set of nodes in the component.
            # Coloring A: {t_v} for v in C
            # Coloring B: {3 - t_v} for v in C
            # We want to pick the one that makes the sequence lexicographically smallest.
            # This is equivalent to picking the coloring such that the first node
            # in the component (by index) has the smaller value.
            
            first_node = min(component)
            if tiers[first_node] == 2:
                for node in component:
                    tiers[node] = 3 - tiers[node]
                    
    print("YES")
    print(*(tiers[1:]))

solve()
