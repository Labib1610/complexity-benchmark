```python
import sys

# The condition: number of pairs (j, k) with 1 <= j <= p_u, 1 <= k <= p_v, j != k is exactly 1.
# If p_u = 1, p_v = 1: pairs (j, k) are (1, 1). j != k is never true. Count = 0.
# If p_u = 1, p_v = 2: pairs are (1, 1), (1, 2). j != k is (1, 2). Count = 1.
# If p_u = 2, p_v = 1: pairs are (1, 1), (2, 1). j != k is (2, 1). Count = 1.
# If p_u = 2, p_v = 2: pairs are (1, 1), (1, 2), (2, 1), (2, 2). j != k are (1, 2), (2, 1). Count = 2.
# Thus, the condition is satisfied if and only if p_u != p_v.
# This is a standard 2-coloring problem on a graph.
# Minimizing E = p_1 * 3^(n-1) + ... + p_n * 3^0 means we want p_1 to be 1 if possible, then p_2 to be 1, etc.
# Since p_i can only be 1 or 2, this is equivalent to lexicographical minimization.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To minimize E, we want p_i = 1 for the smallest i in each component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
