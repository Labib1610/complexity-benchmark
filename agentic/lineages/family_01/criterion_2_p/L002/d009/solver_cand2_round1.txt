```python
import sys

# The condition gcd(p_u, p_v) < (p_u + p_v) / 2
# Possible phases are 1 and 2.
# If p_u = 1, p_v = 1: gcd(1, 1) = 1, (1+1)/2 = 1. 1 < 1 is False.
# If p_u = 1, p_v = 2: gcd(1, 2) = 1, (1+2)/2 = 1.5. 1 < 1.5 is True.
# If p_u = 2, p_v = 1: gcd(2, 1) = 1, (2+1)/2 = 1.5. 1 < 1.5 is True.
# If p_u = 2, p_v = 2: gcd(2, 2) = 2, (2+2)/2 = 2. 2 < 2 is False.
# Thus, p_u and p_v must be different. This is a bipartite graph coloring problem.
# We want to minimize sum(p_i * 3^(n-i)).
# Since 3^(n-i) is much larger for smaller i, we want p_1 to be 1 if possible,
# then p_2 to be 1 if possible, and so on.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    # To minimize the expression, we greedily try to assign 1 to the smallest index
    # in each connected component.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # We want to assign 1 to the first node i
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # The two possible colorings are (color[v]) and (3 - color[v]).
            # We want to pick the one that makes the sequence lexicographically smaller.
            # Actually, we want to minimize sum p_i * 3^(n-i).
            # This is equivalent to minimizing the sequence (p_1, p_2, ..., p_n) lexicographically.
            # For each component, find the smallest index node 'start_node'.
            # We want color[start_node] to be 1.
            start_node = min(component)
            if color[start_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
```
