import sys

# The problem states that for each sensor (u, v), the sum of pressures must be 3.
# Since pressures are 1 or 2, this means p[u] + p[v] = 3.
# This implies p[u] != p[v].
# This is a bipartite coloring problem.
# We want the lexicographically smallest configuration (p1, p2, ..., pn).
# To make it lexicographically smallest, we prefer 1 over 2.
# For each connected component, we have two choices:
# 1. Assign the first node in the component to 1.
# 2. Assign the first node in the component to 2.
# To be lexicographically smallest, we should try to assign 1 to the smallest
# index node in each component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we want the smallest index
            # in this component to be 1.
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard bipartite check
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if we need to flip the colors to make the smallest index 1
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
