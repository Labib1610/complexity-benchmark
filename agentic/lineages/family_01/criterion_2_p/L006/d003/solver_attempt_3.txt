import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want to minimize the sequence t1, t2, ..., tn lexicographically.
    # This means we want t1=1 if possible, then t2=1 if possible, etc.
    # This is a 2-coloring problem.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning color 1 to the current component
            # We use a greedy approach: for each component, try setting the 
            # first node to 1. If that fails, it's impossible.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # The greedy choice for lexicographical minimization is to assign 
            # the smallest possible color (1) to the smallest index in the 
            # component. Since we iterate 1 to n, the first uncolored node 
            # encountered is the smallest in its component.
            
    print("YES")
    print(*(color[1:]))

solve()
