```python
import sys

# The problem states that for each monitor (u, v), one valve must be 
# twice the other. Since states are {1, 2}, this means one must be 1 
# and the other must be 2.
# This is equivalent to a 2-coloring problem on a graph where each 
# edge (u, v) implies s_u != s_v.
# A graph is 2-colorable if and only if it is bipartite.
# To get the lexicographically smallest sequence, we iterate from 1 to n.
# If a node is unvisited, we assign it 1 (the smaller value) and 
# perform BFS/DFS to color its connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    state = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer state 1
            stack = [(i, 1)]
            state[i] = 1
            while stack:
                u, s = stack.pop()
                next_s = 3 - s
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = next_s
                        stack.append((v, next_s))
                    elif state[v] == s:
                        print("NO")
                        return
    
    print("YES")
    print(*(state[1:]))

solve()
```
