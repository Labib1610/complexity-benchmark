import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The filtering process:
    # For i = 1 to n:
    # If there exists a valid 2-coloring where i is 1, force i=1.
    # Else, force i=2.
    
    # This is equivalent to:
    # Try to set i=1. If it leads to a contradiction, i must be 2.
    # If i=2 also leads to a contradiction, then NO.
    
    assignment = [0] * (n + 1)
    
    def get_component(start_node, current_assignment):
        comp = []
        stack = [start_node]
        current_assignment[start_node] = 1
        comp.append(start_node)
        while stack:
            u = stack.pop()
            for v in adj[u]:
                if current_assignment[v] == 0:
                    current_assignment[v] = 3 - current_assignment[u]
                    comp.append(v)
                    stack.append(v)
                elif current_assignment[v] == current_assignment[u]:
                    return None
        return comp

    # We need to maintain the state of the graph coloring.
    # Since we want the lexicographically smallest assignment (1s preferred),
    # we greedily try to set assignment[i] = 1.
    
    # To check if setting i=1 is valid:
    # We need to see if there exists a valid coloring of the remaining graph
    # given the current constraints.
    
    # Actually, the filtering process is simpler:
    # For each connected component, we have two possible colorings (swap 1 and 2).
    # For each component, we pick the coloring that makes the first node in the 
    # component (by index) 1, unless that's impossible.
    
    # Wait, the rule is: "check if any of the currently remaining safe divisions 
    # assign extract i to Formulation 1."
    # This means for each connected component, we have two colorings: C and swap(C).
    # Let the component be S. Let i be the smallest index in S.
    # If in C, i is 1, then in swap(C), i is 2.
    # Since both are valid, we keep both.
    # The rule says: if any remaining division has i=1, discard those with i=2.
    # So we MUST pick the coloring where i=1.
    
    ans = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            # Find component
            comp = []
            stack = [i]
            visited[i] = True
            comp.append(i)
            
            # Standard 2-coloring
            color = {i: 1}
            q = [i]
            possible = True
            while q:
                u = q.pop()
                for v in adj[u]:
                    if v not in color:
                        color[v] = 3 - color[u]
                        visited[v] = True
                        comp.append(v)
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Smallest index in component
            first = min(comp)
            if color[first] == 1:
                for node in comp:
                    ans[node] = color[node]
            else:
                for node in comp:
                    ans[node] = 3 - color[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
