 ```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        stack = [start]
        color[start] = 0
        while stack:
            v = stack.pop()
            cv = color[v]
            for nb in adj[v]:
                if color[nb] == -1:
                    color[nb] = cv ^ 1
                    stack.append(nb)
                elif color[nb] == cv:
                    sys.stdout.write("NO")
                    return

    out = ['YES']
    out.append(' '.join(str(color[i] + 1) for i in range(1, n + 1)))
    sys.stdout.write('\n'.join(out))

solve()
```
