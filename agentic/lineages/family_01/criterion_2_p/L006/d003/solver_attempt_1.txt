import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color array: 0 = unvisited, 1 = method 1, 2 = method 2
    color = [0] * (n + 1)
    
    # To minimize the integer, we want f_1=1, then f_2=1 if possible, etc.
    # We iterate 1 to n. If unvisited, start BFS/DFS.
    # To make f_i as small as possible, we greedily assign 1 if possible.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a component traversal
            stack = [i]
            color[i] = 1
            
            # Use a list to keep track of nodes in this component to check consistency
            component = []
            
            # Standard DFS to check bipartiteness
            # We use a stack to avoid recursion depth issues
            q = [i]
            while q:
                u = q.pop()
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:

                        return
