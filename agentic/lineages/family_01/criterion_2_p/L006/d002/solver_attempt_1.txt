# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():

    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want to minimize the integer formed by f1, f2, ..., fn.
    # This means we want f1=1, then f2=1, etc., greedily.
    # This is a 2-coloring problem.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a BFS/DFS to color the connected component.
            # To minimize the integer, we prefer color 1 for the smallest index.

            idx_comp = 0

                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:

                        component.append(v)

            # the smallest index in this component 1.
