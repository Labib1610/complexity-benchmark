import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want the lexicographically smallest assignment.
    # This means we prefer assigning 1 to node 1, then node 2, etc.
    # This is a 2-coloring problem.
    
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest, we iterate 1 to n.
    # If a node is not colored, we try assigning it 1.
    # If that leads to a conflict, we try 2.
    # Since it's a bipartite check, if a component is connected,
    # picking the color of the first node in the component determines the rest.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To ensure lexicographical smallest, we must pick color 1 for the 
            # smallest node in the component.
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard BFS/DFS to find all nodes in component
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if this assignment is the lexicographically smallest.
            # The current assignment for this component is fixed by color[i]=1.
            # If we swapped colors (1->2, 2->1), would it be smaller?
            # We only swap if the current assignment makes the first node 
            # in the component (which is i) have a 2, but we could have had a 1.
            # But we already set color[i] = 1.
            # Is it possible that a different node j < i in the same component 
            # could have been 1? No, because i is the smallest index in this component.
            
    print("YES")
    print(*(color[1:]))

solve()
