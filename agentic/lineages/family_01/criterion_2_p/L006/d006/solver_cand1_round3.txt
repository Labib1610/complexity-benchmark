import sys

# The problem states that for every interaction (u, v):
# If s_u = 1, then s_v = 2
# If s_u = 2, then s_v = 1
# If s_v = 1, then s_u = 2
# If s_v = 2, then s_u = 1
# This is equivalent to saying s_u != s_v.
# If u == v, then s_u != s_u, which is impossible.
# Thus, the problem is to check if the graph is bipartite.
# If it is, we want the lexicographically smallest assignment of 1s and 2s.
# For each connected component, we have two choices:
# 1. Assign the first node in the component to 1.
# 2. Assign the first node in the component to 2.
# To get the lexicographically smallest result, for each component,
# we pick the assignment that results in a smaller value for the smallest index in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            idx_stack = 0
            while idx_stack < len(component):
                u, c = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # For this component, we have two colorings:
            # Option 1: node i has color 1
            # Option 2: node i has color 2
            # We want the one that makes the smallest index in the component 1.
            # Since we want lexicographical smallest, we want the smallest index
            # in the component to be 1 if possible.
            
            # Find the smallest index in this component
            min_node = min(node for node, c in component)
            
            # Find the color assigned to min_node in the current BFS/DFS
            for node, c in component:
                if node == min_node:
                    target_color = c
                    break
            
            # If target_color is 1, we keep it. If 2, we flip all colors.
            if target_color == 1:
                for node, c in component:
                    res[node] = c
            else:
                for node, c in component:
                    res[node] = 3 - c
                    
    print("YES")
    print(*(res[1:]))

solve()
