import sys

# The condition is: state[u] XOR state[v] = 3.
# Since state values are 1 (binary 01) and 2 (binary 10),
# 1 XOR 2 = 3.
# This means for every bridge (u, v), state[u] must be different from state[v].
# This is equivalent to a 2-coloring problem on a graph.
# To get the lexicographically smallest sequence, for each connected component,
# we try to assign 1 to the smallest index in that component.
# If that leads to a contradiction, the graph is not bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want the smallest index
            # in the component to be 1.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a stack for DFS to avoid recursion depth issues
            # We need to store the component to check consistency
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # color[v] is either 1 or 2.
            # We want the smallest index in this component to have state 1.
            # Let min_node be the smallest index in this component.
            # If color[min_node] == 1, we keep the assignment.
            # If color[min_node] == 2, we flip all colors in this component.
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
