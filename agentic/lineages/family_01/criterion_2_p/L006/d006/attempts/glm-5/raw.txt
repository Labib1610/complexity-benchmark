```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    self_loop = False
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            self_loop = True
        adj[u].append(v)
        adj[v].append(u)
    
    if self_loop:
        sys.stdout.write("NO\n")
        return
    
    color = [0] * (n + 1)
    
    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            stack = [start]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        sys.stdout.write("NO\n")
                        return
    
    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(str(color[i]) for i in range(1, n + 1)) + '\n')

solve()
```
