import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    d = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        d[u] += 1
        d[v] += 1
        
    # Let s_i = 1 + x_i, where x_i in {0, 1}
    # The equation: 2 * sum(s_u * s_v) + 5m = 3 * sum(d_i * s_i)
    # 2 * sum((1+x_u)(1+x_v)) + 5m = 3 * sum(d_i * (1+x_i))
    # 2 * sum(1 + x_u + x_v + x_u*x_v) + 5m = 3 * sum(d_i + d_i*x_i)
    # 2 * (m + sum(d_i*x_i) + sum(x_u*x_v)) + 5m = 3 * (2m + sum(d_i*x_i))
    # 2m + 2*sum(d_i*x_i) + 2*sum(x_u*x_v) + 5m = 6m + 3*sum(d_i*x_i)
    # 2*sum(x_u*x_v) + m = sum(d_i*x_i)
    # sum(d_i*x_i) - 2*sum(x_u*x_v) = m
    # This is equivalent to: sum_{i} x_i * (d_i - 2 * sum_{j in adj(i)} x_j) = m
    # If we set x_i = 1 for all i, LHS = sum(d_i) - 2 * (2m) = 2m - 4m = -2m != m (unless m=0)
    # Actually, the equation simplifies to: sum_{i} x_i * (d_i - 2 * count_neighbors_with_x_j_1) = m
    # This is satisfied if x_i = 1 for all i, but we want lexicographically smallest s_i.
    # s_i = 1 means x_i = 0. s_i = 2 means x_i = 1.
    # Let's test x_i = 1 for all i: sum(d_i) - 2 * (2m) = 2m - 4m = -2m. Not m.
    # Let's test x_i = 0 for all i: 0 = m. Only if m=0.
    # The equation is: sum_{i=1}^n x_i * d_i - 2 * sum_{(u,v) in edges} x_u * x_v = m
    # This is exactly the condition for a cut in the graph.
    # Let S be the set of vertices where x_i = 1.
    # sum_{i in S} d_i - 2 * |edges within S| = m
    # This is the number of edges between S and V\S, which is the cut size.
    # So we need a cut of size m.
    # The total number of edges is m. A cut of size m means all edges must go between S and V\S.
    # This means the graph must be bipartite.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, we have two sets: color 0 and color 1.
    # Let S be the set of vertices with color 0. Cut size is m.
    # Let S be the set of vertices with color 1. Cut size is m.
    # We want lexicographically smallest s_i. s_i = 1 + x_i.
    # x_i = 1 if i in S, else 0.
    # To make s_i smallest, we want x_i = 0 as much as possible.
    # Try S = {i | color[i] == 1} (x_i = 1 if color[i] == 1)
    # Try S = {i | color[i] == 0} (x_i = 1 if color[i] == 0)
    
    def get_s(set_s):
        res = []
        for i in range(1, n + 1):
            res.append(2 if i in set_s else 1)
        return res

    s1 = get_s({i for i in range(1, n + 1) if color[i] == 1})
    s2 = get_s({i for i in range(1, n + 1) if color[i] == 0})
    
    ans = min(s1, s2)
    print("YES")
    print(*(ans))

solve()
