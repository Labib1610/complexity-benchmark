```python
import sys

# M1: 1->2, 2->1, 3->4, 4->3
# M2: 1->3, 2->4, 3->1, 4->2
# Mtarget: 1->4, 2->3, 3->2, 4->1
#
# Let f1(x) be M1(x), f2(x) be M2(x).
# We need f_u(f_v(x)) = Mtarget(x) for all x.
#
# f1(f1(x)) = x (Identity)
# f2(f2(x)) = x (Identity)
# f1(f2(x)) = Mtarget(x)
# f2(f1(x)) = Mtarget(x)
#
# If u and v have the same mode (both 1 or both 2), the result is Identity.
# If u and v have different modes (one 1, one 2), the result is Mtarget.
#
# We need f_u(f_v(x)) = Mtarget(x).
# This means u and v must have different modes.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning 1 to the first node of each component
            # to get lexicographically smallest result
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
