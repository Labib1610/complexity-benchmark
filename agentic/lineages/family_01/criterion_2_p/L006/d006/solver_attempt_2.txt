```python
import sys

# The condition is that for every edge (u, v), c[u] * c[v] = 2.
# Since c[i] is in {1, 2}, the only way for the product to be 2 is if one is 1 and the other is 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with two colors (1 and 2).
# Let the two sets of the bipartition be A and B.
# We have two possible assignments:
# 1. All nodes in A get 1, all nodes in B get 2.
# 2. All nodes in A get 2, all nodes in B get 1.
# We want to maximize sum_{i: c[i]=1} 2^{-i}.
# This is equivalent to comparing the sum of 2^{-i} for i in A vs i in B.
# Since 2^{-i} is strictly decreasing, we compare the smallest index in A vs the smallest index in B.
# If min(A) < min(B), we want A to be the set of 1s.
# If min(B) < min(A), we want B to be the set of 1s.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    set_a = []
    set_b = []
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            set_a.append(i)
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        if next_c == 1:
                            set_a.append(v)
                        else:
                            set_b.append(v)
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # We have two sets A and B. For each connected component, we can swap colors.
    # However, the problem implies a global assignment.
    # Actually, for each connected component, we can choose to assign 1 to A or 1 to B.
    # To maximize sum 2^{-i}, for each component, we pick the set that contains the smallest index.
    
    res = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            comp_a = []
            comp_b = []
            stack = [i]
            visited[i] = True
            color[i] = 1
            comp_a.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        color[v] = 3 - color[u]
                        if color[v] == 1: comp_a.append(v)
                        else: comp_b.append(v)
                        stack.append(v)
            
            # Compare min index in comp_a vs comp_b
            min_a = min(comp_a)
            min_b = min(comp_b) if comp_b else float('inf')
            
            if min_a < min_b:
                for node in comp_a: res[node] = 1
                for node in comp_b: res[node] = 2
            else:
                for node in comp_a: res[node] = 2
                for node in comp_b: res[node] = 1
                
    print("YES")
    print(*(res[1:]))

solve()
```
