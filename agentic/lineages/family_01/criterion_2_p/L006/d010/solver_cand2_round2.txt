import sys

# The problem asks for states s_i in {1, 2}.
# Let the quadrants be (0,0), (0,1), (1,0), (1,1).
# State 1 (row swap): (r, c) -> (r, 1-c)
# State 2 (col swap): (r, c) -> (1-r, c)
#
# If we apply state s_u then state s_v:
# If s_u=1, s_v=1: (r, c) -> (r, 1-c) -> (r, 1-(1-c)) = (r, c). (Identity, invalid)
# If s_u=2, s_v=2: (r, c) -> (1-r, c) -> (1-(1-r), c) = (r, c). (Identity, invalid)
# If s_u=1, s_v=2: (r, c) -> (r, 1-c) -> (1-r, 1-c). (All quadrants move, valid)
# If s_u=2, s_v=1: (r, c) -> (1-r, c) -> (1-r, 1-c). (All quadrants move, valid)
#
# Thus, for every connection (u, v), we must have s_u != s_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (s_1, ..., s_n).
# We can use BFS/DFS to color each connected component.
# For each component, we try assigning s_i = 1 to the first node encountered.
# If that leads to a contradiction, we try s_i = 2.
# To get the lexicographically smallest, we iterate i from 1 to n.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning color 1 to node i
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            possible = True
            idx_comp = 0
            component.append(i)
            
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible: break
            
            if not possible:
                # Try assigning color 2 to node i
                for node in component:
                    color[node] = 0
                
                color[i] = 2
                component = [i]
                idx_comp = 0
                possible = True
                while idx_comp < len(component):
                    u = component[idx_comp]
                    idx_comp += 1
                    for v in adj[u]:
                        if color[v] == 0:
                            color[v] = 3 - color[u]
                            component.append(v)
                        elif color[v] == color[u]:
                            possible = False
                            break
                    if not possible: break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(color[1:]))

solve()
