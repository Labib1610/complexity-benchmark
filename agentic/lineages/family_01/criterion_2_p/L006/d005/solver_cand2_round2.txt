```python
import sys

# The function is W(x) = x^3 - (p_u + p_v)x^2 + (p_u * p_v)x
# W'(x) = 3x^2 - 2(p_u + p_v)x + (p_u * p_v)
# W''(x) = 6x - 2(p_u + p_v)
# The inflection point occurs where W''(x) = 0.
# 6x - 2(p_u + p_v) = 0 => x = (p_u + p_v) / 3
# We require the inflection point to be at x = 1.
# (p_u + p_v) / 3 = 1 => p_u + p_v = 3
# Since p_u, p_v \in {1, 2}, the only way to satisfy p_u + p_v = 3 is
# {p_u, p_v} = {1, 2}.
# This means for every edge (u, v), p_u != p_v.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we greedily assign 1
# to the first node of each connected component, then propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign 1 to i to be lexicographically smallest
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
