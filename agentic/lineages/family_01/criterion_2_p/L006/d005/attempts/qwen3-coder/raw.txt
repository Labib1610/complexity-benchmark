```python
import sys
from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    graph = defaultdict(list)
    
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-indexed
        v = int(data[index + 1]) - 1
        edges.append((u, v))
        graph[u].append(v)
        graph[v].append(u)
        index += 2
    
    # Color array: 0 means unvisited, 1 and 2 are the two colors
    color = [0] * n
    
    # Check if bipartite and assign colors
    for start in range(n):
        if color[start] == 0:
            queue = deque()
            queue.append(start)
            color[start] = 1  # Start with color 1
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                next_color = 3 - current_color  # Switch between 1 and 2
                
                for neighbor in graph[node]:
                    if color[neighbor] == 0:
                        color[neighbor] = next_color
                        queue.append(neighbor)
                    elif color[neighbor] != next_color:
                        print("NO")
                        return
    
    # Verify all edges have product 2
    for u, v in edges:
        if color[u] * color[v] != 2:
            print("NO")
            return
    
    # If we reach here, a valid assignment exists
    # We need to convert colors 1,2 back to configuration codes 1,2
    # Since our coloring already uses 1 and 2, we can use them directly
    print("YES")
    result = []
    for i in range(n):
        result.append(str(color[i]))
    print(' '.join(result))

solve()
```
