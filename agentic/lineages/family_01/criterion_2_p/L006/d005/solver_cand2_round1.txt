```python
import sys

# The number of distinct linear arrangements of a multiset with c_u 'Alpha' tokens
# and c_v 'Beta' tokens is given by the binomial coefficient (c_u + c_v) choose c_u.
# Possible values for c_u, c_v are {1, 2}.
# (1, 1): (1+1)C1 = 2C1 = 2
# (1, 2): (1+2)C1 = 3C1 = 3
# (2, 1): (2+1)C2 = 3C2 = 3
# (2, 2): (2+2)C2 = 4C2 = 6
# The condition "evaluates to exactly 3" means one unit must be 1 and the other must be 2.
# This is a bipartite graph coloring problem where adjacent nodes must have different colors.
# We want to minimize the sequence c_1 c_2 ... c_n lexicographically.
# This means we prefer 1 over 2 for each node in order 1 to n.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS to color the connected component
            # To minimize lexicographically, we try assigning 1 to the first node
            # of each component.
            queue = [i]
            color[i] = 1
            while queue:
                u = queue.pop(0)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
