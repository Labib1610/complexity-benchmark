```python
import sys

# The problem states:
# Mode 1: (x, y) -> (-x, y)
# Mode 2: (x, y) -> (x, -y)
# Combined effect of two modules (u, v) must be (x, y) -> (-x, -y).
# Let T1 be the transformation for Mode 1, T2 for Mode 2.
# T1(T1(x, y)) = (x, y)
# T2(T2(x, y)) = (x, y)
# T1(T2(x, y)) = T1(x, -y) = (-x, -y)
# T2(T1(x, y)) = T2(-x, y) = (-x, -y)
# So, if u and v have different modes (1 and 2), the combined effect is (-x, -y).
# If u and v have the same mode (1 and 1), the effect is (x, y).
# If u and v have the same mode (2 and 2), the effect is (x, y).
# Thus, for the channel to function, u and v must have DIFFERENT modes.
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, we prefer mode 1 (color 1)
            # for the first node in each connected component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
