```python
import sys

# The problem states that for each link (u, v), the sum of projections must be 0.
# Code 1 corresponds to +1/2, Code 2 corresponds to -1/2.
# Sum = 0 means one qubit must be 1 (+1/2) and the other must be 2 (-1/2).
# This is equivalent to a 2-coloring problem on a graph where each edge (u, v)
# implies color(u) != color(v).
# The score S = sum(c_i * 2^(n-i)) is minimized by making c_i as small as possible
# for the smallest i (i.e., i=1, 2, ...).
# Since we want to minimize the score, we greedily assign 1 to the first qubit
# of each connected component, then propagate the constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To minimize score, we want c_i = 1 if possible.
            # Since it's a bipartite graph, once we pick 1 for the first node,
            # all other nodes in the component are fixed.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, colors[v]))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
