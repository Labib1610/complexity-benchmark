```python
import sys

# The transformations are:
# T1(x, y) = (-y, x)
# T2(x, y) = (y, -x)
#
# Composition:
# T1(T1(x, y)) = T1(-y, x) = (-x, -y)
# T1(T2(x, y)) = T1(y, -x) = (x, y)  <- Identity
# T2(T1(x, y)) = T2(-y, x) = (x, y)  <- Identity
# T2(T2(x, y)) = T2(y, -x) = (-x, -y)
#
# For T_u(T_v(x, y)) = (x, y), we need:
# If u=1, v=2 or u=2, v=1, the composition is identity.
# If u=1, v=1 or u=2, v=2, the composition is (-x, -y) != (x, y).
#
# Thus, for every channel (u, v), u and v must have different modes.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we prefer color 1 for the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
