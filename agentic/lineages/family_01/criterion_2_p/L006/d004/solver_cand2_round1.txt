```python
import sys

# The determinant of M = [[p_u, p_v], [p_v, p_u]] is p_u^2 - p_v^2.
# For the matrix to be non-singular, det(M) != 0, so p_u^2 != p_v^2.
# Since p_i is in {1, 2}, p_i^2 is in {1, 4}.
# Thus, p_u^2 != p_v^2 implies p_u != p_v.
# This means for every edge (u, v), we must have p_u != p_v.
# This is equivalent to checking if the graph is bipartite.
# We want the lexicographically smallest assignment (p_1, p_2, ..., p_n).
# To make it lexicographically smallest, we prefer p_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # p[i] = 0 means unassigned, 1 or 2 are the values.
    p = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if p[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try p[i] = 1 first.
            stack = [(i, 1)]
            p[i] = 1
            while stack:
                u, color = stack.pop()
                for v in adj[u]:
                    if p[v] == 0:
                        p[v] = 3 - color
                        stack.append((v, p[v]))
                    elif p[v] == color:
                        print("NO")
                        return
    
    print("YES")
    print(*(p[1:]))

solve()
```
