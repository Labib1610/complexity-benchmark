 ```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * n  # 0 unvisited, 1 or 2 assigned
    for i in range(n):
        if color[i] != 0:
            continue
        color[i] = 1
        q = deque([i])
        while q:
            v = q.popleft()
            other = 3 - color[v]
            for w in adj[v]:
                if color[w] == 0:
                    color[w] = other
                    q.append(w)
                elif color[w] != other:
                    sys.stdout.write("NO")
                    return

    out = ["YES", " ".join(map(str, color))]
    sys.stdout.write("\n".join(out))

solve()
```
