```python
import sys

# The problem states that for each monitor (u, v), s_u + s_v = 3.
# Since s_i is in {1, 2}, this means s_u != s_v.
# This is a bipartite graph coloring problem.
# We need to assign 1 or 2 to each node such that adjacent nodes have different values.
# The objective function f(s) = sum(s_i * 2^-i) is minimized by making s_i as small as possible
# for the smallest i.
# Since we have connected components, for each component, we can pick the state of the
# first node (the one with the smallest index) to be 1, which forces the states of all
# other nodes in that component.
# If we pick s_i = 1 for the smallest index i in a component, we minimize the contribution
# of that node to the sum. Since 2^-i decreases as i increases, prioritizing smaller i
# is equivalent to setting s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    states = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if states[i] == 0:
            # Start BFS/DFS for this component
            # To minimize f(s), we want s_i = 1 for the smallest i in the component.
            states[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if states[v] == 0:
                        states[v] = 3 - states[u]
                        stack.append(v)
                    elif states[v] == states[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(states[1:]))

solve()
```
