```python
import sys

# The problem states:
# Assembler i has cycle duration p_i in {1, 2}.
# In a 2-minute window, it completes 2/p_i cycles.
# For a link (u, v), the sum of cycles must be 3:
# (2/p_u) + (2/p_v) = 3
# Possible values for 2/p_i are 2 (if p_i=1) and 1 (if p_i=2).
# So, 2/p_u + 2/p_v = 3 implies one must be 2 and the other must be 1.
# This means p_u != p_v.
# This is a bipartite coloring problem.
# We need to color the graph with 2 colors (1 and 2) such that adjacent nodes have different colors.
# To get the lexicographically smallest configuration, we want p_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we prefer 1.
    # For each connected component, we have two choices of coloring.
    # We pick the one that results in a smaller sequence.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check which coloring is better for this component
            # Option 1: color[i] = 1
            # Option 2: color[i] = 2
            # We want the smallest sequence.
            # Compare the first node in the component (smallest index)
            # Actually, we just need to ensure the smallest index in the component
            # gets the value 1 if possible.
            
            # Find the smallest index in this component
            min_node = min(component)
            if color[min_node] == 2:
                # Flip all colors in this component
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
```
