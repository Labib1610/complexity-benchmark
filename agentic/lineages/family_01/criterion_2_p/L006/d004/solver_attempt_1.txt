```python
import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition "dosage sum = 3" means if u and v are connected,
    # one must be 1 and the other must be 2.
    # This is a standard 2-coloring problem (bipartite graph).
    # To minimize the integer f1f2...fn, we want f1=1, then f2=1 if possible, etc.
    
    color = [0] * (n + 1) # 0: unvisited, 1: dosage 1, 2: dosage 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To minimize the integer, we greedily assign 1 to the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
