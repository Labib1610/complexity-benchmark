```python
import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # State codes: 1 (+1/2) and 2 (-1/2).
    # Constraint: sum of projections = 0 => c_u != c_v.
    # This is a bipartite coloring problem.
    # To minimize the integer, we want c_i = 1 whenever possible.
    
    color = [0] * (n + 1) # 0: unvisited, 1: code 1, 2: code 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To minimize the integer, we greedily assign 1 to the first node
            stack = [(i, 1)]
            color[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, color[v]))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```
