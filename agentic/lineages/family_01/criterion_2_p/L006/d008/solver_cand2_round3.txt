import sys

# The condition 3p_u + 3p_v - 2p_u p_v = 5
# If p_u = 1: 3(1) + 3p_v - 2(1)p_v = 5 => 3 + p_v = 5 => p_v = 2
# If p_u = 2: 3(2) + 3p_v - 2(2)p_v = 5 => 6 - p_v = 5 => p_v = 1
# This means p_u != p_v.
# The condition is equivalent to p_u XOR p_v = 3 (since 1 XOR 2 = 3).
# This is a standard 2-coloring problem (bipartite graph check).
# Lexicographically smallest means we prefer p_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # p_i can be 1 or 2. 
    # We want to assign 1 to as many nodes as possible, 
    # specifically prioritizing smaller indices.
    # This is a bipartite graph. Each connected component can be 2-colored in 2 ways.
    # To get lexicographically smallest, for each component, we try assigning 1 
    # to the smallest index node and check if it's valid.
    
    color = [0] * (n + 1) # 0: unvisited, 1: phase 1, 2: phase 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_stack = 0
            while idx_stack < len(component):
                u = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if this coloring is the lexicographically smallest for this component.
            # The two possible colorings are:
            # 1. color[v]
            # 2. 3 - color[v]
            # We compare the two assignments for the nodes in this component.
            # We want the one that has 1 at the smallest index node.
            
            # Find the smallest index in this component
            min_node = min(component)
            if color[min_node] == 2:
                # Flip the colors
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
