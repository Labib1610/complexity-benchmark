```python
import sys

# The condition is (p_u^3 * p_v + p_u * p_v^3) % 5 == 0
# p_u, p_v are in {1, 2}
# If p_u = 1, p_v = 1: (1*1 + 1*1) = 2 % 5 != 0
# If p_u = 1, p_v = 2: (1*2 + 1*8) = 10 % 5 == 0 (Valid)
# If p_u = 2, p_v = 1: (8*1 + 2*1) = 10 % 5 == 0 (Valid)
# If p_u = 2, p_v = 2: (8*2 + 2*8) = 16 + 16 = 32 % 5 != 0
# Thus, p_u and p_v must be different.
# This is a bipartite graph coloring problem.
# We want to minimize the sequence as a base-3 integer.
# Since p_i is 1 or 2, we want p_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS to color component
            # To minimize the sequence, we greedily assign 1 to the first node
            # of each component.
            queue = [i]
            color[i] = 1
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
