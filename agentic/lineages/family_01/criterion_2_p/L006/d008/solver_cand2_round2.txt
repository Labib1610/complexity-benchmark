```python
import sys

# The condition is 4*s_u - s_v \in {2, 7} where s_u, s_v \in {1, 2}.
# If s_u = 1: 4(1) - s_v = 4 - s_v.
#   If s_v = 1: 4 - 1 = 3 (Not in {2, 7})
#   If s_v = 2: 4 - 2 = 2 (Valid)
# If s_u = 2: 4(2) - s_v = 8 - s_v.
#   If s_v = 1: 8 - 1 = 7 (Valid)
#   If s_v = 2: 8 - 2 = 6 (Not in {2, 7})
#
# So, s_u = 1 implies s_v = 2, and s_u = 2 implies s_v = 1.
# This means s_u != s_v.
# The condition is simply s_u != s_v for all entanglements (u, v).
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (s_1, ..., s_n).
# To get the smallest, we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try color 1 first
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
