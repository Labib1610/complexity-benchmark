import sys

# The condition p_u^3 + p_v^3 = 9 with p_i in {1, 2}
# 1^3 = 1, 2^3 = 8
# 1 + 8 = 9, so the condition is p_u != p_v.
# This is a bipartite coloring problem.
# We want the lexicographically smallest configuration.
# Lexicographically smallest means we prefer 1 over 2.
# For each connected component, we have two possible colorings (swap 1s and 2s).
# We choose the one that makes the first node in the component 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Component nodes
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            is_bipartite = True
            idx_stack = 0
            while idx_stack < len(component):
                u = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings:
            # 1. color[u] remains as is
            # 2. color[u] becomes 3 - color[u]
            # We want the one that is lexicographically smaller.
            # Compare the first node in the component (which is i).
            # If we set color[i] = 1, we are good.
            
            # Check which coloring makes the first node 1
            # The current coloring has color[i] = 1.
            # If we keep it, i is 1. If we flip, i is 2.
            # Since we want lexicographically smallest, we prefer 1.
            # So we keep the current coloring.
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
