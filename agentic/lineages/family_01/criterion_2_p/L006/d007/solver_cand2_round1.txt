```python
import sys

# The game is a variation of Nim/Subtraction game.
# Let the modes be a and b (a, b in {1, 2}).
# The game ends when the counter exceeds 17.
# This is equivalent to a game where you can subtract a or b from a target 17.
# A position x is losing if all reachable positions (x-a, x-b) are winning.
# A position x is winning if there exists a move to a losing position.
# Let W be winning, L be losing.
# x=0 is L.
# If modes are (1, 1):
# 0:L, 1:W, 2:L, 3:W, ... (x is L iff x is even)
# 17 is W (since 17 is odd). So (1, 1) is unstable.
# If modes are (1, 2):
# 0:L, 1:W, 2:W, 3:L, 4:W, 5:W, 6:L... (x is L iff x % 3 == 0)
# 17 % 3 = 2 != 0. So 17 is W. So (1, 2) is stable.
# If modes are (2, 2):
# 0:L, 1:L, 2:W, 3:W, 4:L, 5:L, 6:W, 7:W... (x is L iff x % 4 == 0 or 1)
# 17 % 4 = 1. So 17 is L. So (2, 2) is unstable.
# Thus, a channel is stable if and only if the modes are (1, 2) or (2, 1).
# This means the graph must be bipartite, and we need to 2-color it.
# Lexicographically smallest: assign 1 to the first node of each component,
# then propagate the coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
