```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # We need to assign modes such that for every edge (u, v),
    # Tr(M_u M_v) = 11.
    # Let x_u be the mode of module u (1 or 2).
    # The condition is x_u + x_v = 3.
    # This is a bipartite graph coloring problem.
    # We want the lexicographically smallest assignment.
    # Lexicographically smallest means we prefer mode 1 (x=1) over mode 2 (x=2).
    # Since x_u + x_v = 3, if x_u = 1 then x_v = 2, and vice versa.
    # To minimize the sequence, we should try to assign 1 to nodes with smaller indices.
    # We process nodes in increasing order of index.
    # If a node is uncolored, we try to color it with 1.
    # This forces its neighbors to be colored with 2.
    # If a conflict arises, it's impossible.

    color = [0] * n  # 0: uncolored, 1: mode 1, 2: mode 2
    
    possible = True
    
    for i in range(n):
        if color[i] == 0:
            # Try to assign mode 1 to this component starting at node i
            # BFS
            queue = deque([i])
            color[i] = 1
            while queue:
                u = queue.popleft()
                cu = color[u]
                # The required color for neighbors
                cv = 3 - cu
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = cv
                        queue.append(v)
                    elif color[v] != cv:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break
            
    if not possible:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, color)))

if __name__ == '__main__':
    solve()
```
