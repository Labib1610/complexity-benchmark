import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [[] for _ in range(n+1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # Define matrices
    M1 = [[1, 2, 0],
          [0, 1, 2],
          [0, 0, 1]]
    
    M2 = [[1, 0, 0],
          [2, 1, 0],
          [0, 2, 1]]
    
    def mat_mult(A, B):
        return [[sum(A[i][k] * B[k][j] for k in range(3)) for j in range(3)] for i in range(3)]
    
    def trace(M):
        return M[0][0] + M[1][1] + M[2][2]
    
    # Precompute traces for all combinations
    traces = {}
    traces[(1,1)] = trace(mat_mult(M1, M1))
    traces[(1,2)] = trace(mat_mult(M1, M2))
    traces[(2,1)] = trace(mat_mult(M2, M1))
    traces[(2,2)] = trace(mat_mult(M2, M2))
    
    # Build constraint graph
    # For each edge (u,v), the trace of M[p[u]] * M[p[v]] must be 11
    # So we need traces[(p[u], p[v])] == 11
    
    allowed = {(1,1), (1,2), (2,1), (2,2)}
    required_trace = 11
    valid_pairs = set()
    for p1 in [1, 2]:
        for p2 in [1, 2]:
            if traces[(p1, p2)] == required_trace:
                valid_pairs.add((p1, p2))
    
    if not valid_pairs:
        print("NO")
        return
    
    # For each edge, we have a constraint that (p[u], p[v]) must be in valid_pairs
    # This is a 2-SAT problem
    
    # We'll model this as a graph where each node represents a variable or its negation
    # var u -> node 2*u and 2*u+1 for mode 1 and 2 respectively
    # But it's easier to think in terms of modes directly
    
    # Let's build an implication graph for 2-SAT
    # Variables: for each module u, we have two boolean variables:
    # x[u] = True means mode 1, x[u] = False means mode 2
    # But we'll use a different approach: for each node, we'll track possible modes
    
    # We can model this as a graph coloring problem with constraints
    # Each node can be colored 1 or 2
    # For each edge, only certain pairs of colors are allowed
    
    # This is a constraint satisfaction problem on a graph
    # We can use BFS/DFS to propagate constraints
    
    # Let's build the constraint graph
    # For each component, we'll try to 2-color it with constraints
    
    # First, check if all edges have at least one valid assignment
    for u, v in edges:
        if not valid_pairs:
            print("NO")
            return
    
    # Now, we need to propagate constraints
    # We'll use a BFS-like approach to assign modes
    
    # Possible modes for each node
    possible = [None] * (n+1)
    for i in range(1, n+1):
        possible[i] = {1, 2}
    
    # Queue of nodes to process
    queue = deque()
    # Initially, all nodes are in possible states
    # But we need to propagate constraints
    
    # Let's build adjacency list for the constraint graph
    changed = True
    while changed:
        changed = False
        for u, v in edges:
            # Constraint: (p[u], p[v]) must be in valid_pairs
            # So p[u] can only be values such that there exists p[v] with (p[u], p[v]) in valid_pairs
            # Similarly for p[v]
            
            # For u: find all p[u] such that there exists p[v] in possible[v] with (p[u], p[v]) valid
            new_pu = set()
            for pu in possible[u]:
                for pv in possible[v]:
                    if (pu, pv) in valid_pairs:
                        new_pu.add(pu)
                        break
            if not new_pu:
                print("NO")
                return
            if new_pu != possible[u]:
                changed = True
                possible[u] = new_pu
            
            # For v: find all p[v] such that there exists p[u] in possible[u] with (p[u], p[v]) valid
            new_pv = set()
            for pv in possible[v]:
                for pu in possible[u]:
                    if (pu, pv) in valid_pairs:
                        new_pv.add(pv)
                        break
            if not new_pv:
                print("NO")
                return
            if new_pv != possible[v]:
                changed = True
                possible[v] = new_pv
    
    # Now, for each node, possible[i] contains valid modes
    # We need to assign modes such that all constraints are satisfied
    # And the result is lexicographically smallest
    
    # We can do this by trying mode 1 first for each node in order
    # This is essentially a backtracking with constraint propagation
    # But since we've already propagated, we can just assign greedily
    
    # For each connected component, we can pick a starting node and try mode 1 first
    visited = [False] * (n+1)
    result = [0] * (n+1)
    
    for start in range(1, n+1):
        if visited[start]:
            continue
        # BFS to assign modes in the component
        component = []
        queue = deque([start])
        visited[start] = True
        component.append(start)
        
        # Try mode 1 first for the start node
        if 1 in possible[start]:
            result[start] = 1
        elif 2 in possible[start]:
            result[start] = 2
        else:
            print("NO")
            return
            
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    component.append(v)
                    # Find a mode for v that's compatible with u
                    found = False
                    for mode_v in [1, 2]:  # Lexicographically smallest first
                        if mode_v in possible[v]:
                            if (result[u], mode_v) in valid_pairs:
                                result[v] = mode_v
                                queue.append(v)
                                found = True
                                break
                    if not found:
                        print("NO")
                        return
    
    # Verify all edges
    for u, v in edges:
        if (result[u], result[v]) not in valid_pairs:
            print("NO")
            return
            
    print("YES")
    print(' '.join(map(str, result[1:])))

if __name__ == "__main__":
    main()
