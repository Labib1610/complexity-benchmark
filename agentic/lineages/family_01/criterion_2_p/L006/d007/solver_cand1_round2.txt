import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # The determinant of M_{u,v} = [[s_u, 1, 0], [0, s_v, 1], [2, 0, 1]]
    # det = s_u * (s_v * 1 - 1 * 0) - 1 * (0 * 1 - 1 * 2) + 0 * (...)
    # det = s_u * s_v + 2
    # We need det = 4, so s_u * s_v = 2.
    # Since s_i in {1, 2}, the only way s_u * s_v = 2 is if one is 1 and the other is 2.
    # This is a bipartite coloring problem: s_u != s_v.
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    # To get lexicographically smallest, we want s_i = 1 whenever possible.
    # We iterate 1 to n. If not colored, color it 1 and perform BFS/DFS.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
