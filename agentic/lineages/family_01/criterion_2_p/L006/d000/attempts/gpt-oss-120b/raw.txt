import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)          # 0/1 parity
    schedule = [0] * (n + 1)        # final 1/2 output
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        comp = [start]
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                    comp.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
        # orient so that the smallest index vertex (start) gets schedule 1
        flip = color[start]          # makes start parity 0 after xor
        for v in comp:
            schedule[v] = (color[v] ^ flip) + 1

    print("YES")
    print(' '.join(str(schedule[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
