```python
import sys

# The compliance matrices are:
# C1 = [[1, 1], [1, 1]]
# C2 = [[1, -1], [-1, 1]]
# Joint compliance J = Cu + Cv
# If u=v, J = 2*Cu, which has det=0, so eigenvalues are 0 and 2. Not positive definite.
# If u != v:
# If s_u=1, s_v=1: J = [[2, 2], [2, 2]], det=0. Not positive definite.
# If s_u=2, s_v=2: J = [[2, -2], [-2, 2]], det=0. Not positive definite.
# If s_u=1, s_v=2: J = [[2, 0], [0, 2]], det=4, trace=4. Eigenvalues are 2, 2. Positive definite.
# If s_u=2, s_v=1: J = [[2, 0], [0, 2]], det=4, trace=4. Eigenvalues are 2, 2. Positive definite.
# Thus, for every connection (u, v), we must have s_u != s_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (s1, s2, ..., sn).
# This means we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we greedily pick 1 if possible
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
