import sys

# The permutations are:
# s=1: (A B)(C D) -> A->B, B->A, C->D, D->C
# s=2: (A C)(B D) -> A->C, C->A, B->D, D->B
#
# Let P1 = (A B)(C D) and P2 = (A C)(B D).
# P1 * P1 = Identity (A->A, B->B, C->C, D->D)
# P2 * P2 = Identity (A->A, B->B, C->C, D->D)
# P1 * P2 = (A B)(C D) * (A C)(B D) = (A D)(B C)
# P2 * P1 = (A C)(B D) * (A B)(C D) = (A B C D) -> A->D, B->A, C->B, D->C
#
# Requirement: For any connection (u, v), the composition must not map any channel to itself.
# If u == v: P1*P1 = I, P2*P2 = I. Both violate the condition.
# If u != v:
# If s_u = 1, s_v = 1: P1*P1 = I (Violates)
# If s_u = 2, s_v = 2: P2*P2 = I (Violates)
# If s_u = 1, s_v = 2: P1*P2 = (A D)(B C) (No channel maps to itself, OK)
# If s_u = 2, s_v = 1: P2*P1 = (A B C D) (No channel maps to itself, OK)
#
# Conclusion: For every connection (u, v), we must have s_u != s_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence s_1, ..., s_n.
# We can use BFS/DFS to color each connected component.
# For each component, we try setting the first node to 1. If that fails, try 2.
# To get lexicographically smallest, we iterate i from 1 to n. If not colored,
# color it 1 and propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we prefer color 1 for the smallest index
            stack = [(i, 1)]

            # Use a list to keep track of nodes in this component to check consistency
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
