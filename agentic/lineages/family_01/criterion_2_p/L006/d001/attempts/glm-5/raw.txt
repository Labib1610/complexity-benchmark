import sys
from sys import setrecursionlimit

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    is_bad = [False] * (n + 1)
    
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        color[start] = 1
        stack = [start]
        while stack:
            u = stack.pop()
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - color[u]
                    stack.append(v)
                elif color[v] == color[u]:
                    is_bad[u] = True
        
        # Check if component is bad
        comp_bad = False
        stack2 = [start]
        visited = {start}
        while stack2:
            u = stack2.pop()
            if is_bad[u]:
                comp_bad = True
                break
            for v in adj[u]:
                if v not in visited:
                    visited.add(v)
                    stack2.append(v)
        
        if comp_bad:
            # Flip colors in this component
            stack3 = [start]
            visited2 = {start}
            while stack3:
                u = stack3.pop()
                color[u] = 3 - color[u]
                for v in adj[u]:
                    if v not in visited2:
                        visited2.add(v)
                        stack3.append(v)
    
    # Verify
    for u in range(1, n + 1):
        for v in adj[u]:
            if color[u] + color[v] != 3:
                print("NO")
                return
    
    print("YES")
    print(' '.join(map(str, color[1:])))

solve()
