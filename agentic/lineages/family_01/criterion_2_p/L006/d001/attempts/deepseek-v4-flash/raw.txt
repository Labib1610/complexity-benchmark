We can model this as a bipartite graph problem: each valve is a node, and each sensor is an edge requiring the two endpoints to have different pressure levels (1 and 2).  

So we need to 2-color the graph. If it’s not bipartite, output `NO`.  
For lexicographically smallest assignment, we assign `1` to the smallest-index node in each connected component when we start coloring, and propagate the alternating colors.

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * n
    
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    # Now color is 0/1, map to 1/2
    res = [1 if c == 0 else 2 for c in color]
    print("YES")
    print(" ".join(map(str, res)))

if __name__ == "__main__":
    solve()
```
