import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unassigned, 1 = cycle 1, 2 = cycle 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If an employee is not colored, we try assigning 1 first.
    # This is a standard bipartite check.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [i]
            color[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process nodes in a way that respects lexicographical order
            # However, standard DFS/BFS doesn't guarantee lexicographical order
            # for the whole sequence unless we are careful.
            # Actually, for each component, we want the smallest assignment.
            # Since we want the lexicographically smallest sequence s1, s2, ..., sn,
            # we must greedily pick 1 for the first uncolored node we encounter.
            
            q = [i]
            color[i] = 1
            while q:
                u = q.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
