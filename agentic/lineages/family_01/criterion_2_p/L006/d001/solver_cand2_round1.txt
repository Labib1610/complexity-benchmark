import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# color(u) != color(v). This is equivalent to checking if the graph is bipartite.
# We want the lexicographically smallest assignment of {1, 2}.
# This means for each connected component, we try to assign 1 to the smallest
# index node in that component, then propagate the constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer color 1 for the
            # smallest node in the component.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a stack for DFS to avoid recursion depth issues
            q = [i]
            color[i] = 1
            while q:
                u = q.pop()
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
    print("YES")
    print(*(color[1:]))

solve()
