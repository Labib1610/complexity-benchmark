import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # assignment: 0 = unassigned, 1 = channel 1, 2 = channel 2
    assignment = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If unassigned, try assigning 1. If that fails, try 2.
    # This is a 2-coloring problem.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [i]
            assignment[i] = 1
            component = [i]
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process nodes in a way that respects lexicographical order
            # Actually, for lexicographical smallest, we must greedily pick 1.
            # If we pick 1 for the first node of a component, the rest is forced.
            
            q = [i]
            while q:
                u = q.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
