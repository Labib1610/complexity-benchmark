from collections import defaultdict
import heapq

def parse_input():
    j, m = map(int, input().split())
    jobs = []
    for _ in range(j):
        line = list(map(int, input().split()))
        o = line[0]
        ops = []
        for i in range(o):
            machine = line[1 + 2*i]
            duration = line[1 + 2*i + 1]
            ops.append((machine, duration))
        jobs.append(ops)
    return j, m, jobs

def get_job_operation_precedence(jobs):
    # For each job, operation index must precede the next operation index
    precedence = []
    for job_idx, job in enumerate(jobs):
        for op_idx in range(len(job) - 1):
            precedence.append(((job_idx, op_idx), (job_idx, op_idx + 1)))
    return precedence

def get_resource_conflicts(jobs):
    # Operations using the same machine cannot overlap
    conflicts = []
    # Map each machine to list of operations using it
    machine_ops = defaultdict(list)
    for job_idx, job in enumerate(jobs):
        for op_idx, (machine, duration) in enumerate(job):
            machine_ops[machine].append((job_idx, op_idx))
    
    # For every pair of operations on the same machine, add a conflict
    for machine, ops in machine_ops.items():
        for i in range(len(ops)):
            for j in range(i+1, len(ops)):
                conflicts.append((ops[i], ops[j]))
    return conflicts

def topological_sort(jobs):
    # Generate all operation nodes
    nodes = set()
    for job_idx, job in enumerate(jobs):
        for op_idx in range(len(job)):
            nodes.add((job_idx, op_idx))
    
    # Get precedence constraints
    precedences = get_job_operation_precedence(jobs)
    
    # Build graph
    in_degree = {node: 0 for node in nodes}
    graph = {node: [] for node in nodes}
    
    for u, v in precedences:
        graph[u].append(v)
        in_degree[v] += 1
    
    # Kahn's algorithm
    queue = [node for node in nodes if in_degree[node] == 0]
    order = []
    
    while queue:
        current = queue.pop(0)
        order.append(current)
        for neighbor in graph[current]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
                
    if len(order) != len(nodes):
        raise Exception("Cycle detected")
        
    return order

def calculate_makespan(schedule, jobs):
    # Given a schedule (list of operations), compute the makespan
    # We'll simulate the execution
    
    # Determine operation durations
    op_durations = {}
    for job_idx, job in enumerate(jobs):
        for op_idx, (machine, duration) in enumerate(job):
            op_durations[(job_idx, op_idx)] = duration
    
    # Machine timelines: for each machine, keep track of when it's free
    machine_free_time = defaultdict(int)
    
    # Job operation completion times
    op_finish_times = {}
    
    # Job last operation finish time (to enforce precedence)
    job_last_finish = [0] * len(jobs)
    
    for job_idx, op_idx in schedule:
        machine, duration = jobs[job_idx][op_idx]
        
        # Earliest start time is max of:
        # 1. When the machine is free
        # 2. When the previous operation of this job finished
        earliest_start = max(machine_free_time[machine], job_last_finish[job_idx])
        finish_time = earliest_start + duration
        
        op_finish_times[(job_idx, op_idx)] = finish_time
        machine_free_time[machine] = finish_time
        job_last_finish[job_idx] = finish_time
        
    return max(machine_free_time.values())
