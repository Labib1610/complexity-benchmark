import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    j = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    
    orders = []
    for _ in range(j):
        o = int(data[idx])
        idx += 1
        phases = []
        for _ in range(o):
            unit = int(data[idx])
            idx += 1
            duration = int(data[idx])
            idx += 1
            phases.append((unit, duration))
        orders.append(phases)
    
    # We'll use backtracking with pruning
    # At any point, we track when each machine is free
    machine_free_time = [0] * (m + 1)  # 1-based indexing
    
    # For each order, we track when it can start its next phase
    order_next_start = [0] * j
    
    def backtrack(order_phase_indices):
        # order_phase_indices[i] = next phase index for order i
        all_done = True
        for i in range(j):
            if order_phase_indices[i] < len(orders[i]):
                all_done = False
                break
        if all_done:
            return 0  # All orders completed
        
        # Try assigning the next phase of each incomplete order
        min_time = float('inf')
        for i in range(j):
            if order_phase_indices[i] >= len(orders[i]):
                continue
            unit, duration = orders[i][order_phase_indices[i]]
            # This phase can start at max(order_next_start[i], machine_free_time[unit])
            start_time = max(order_next_start[i], machine_free_time[unit])
            end_time = start_time + duration
            
            # Make the move
            old_machine_free = machine_free_time[unit]
            old_order_start = order_next_start[i]
            
            machine_free_time[unit] = end_time
            order_next_start[i] = end_time
            
            new_order_phases = list(order_phase_indices)
            new_order_phases[i] += 1
            
            res = backtrack(new_order_phases)
            total_time = max(end_time, res)
            min_time = min(min_time, total_time)
            
            # Undo the move
            machine_free_time[unit] = old_machine_free
            order_next_start[i] = old_order_start
        
        return min_time
    
    initial_phases = [0] * j
    result = backtrack(initial_phases)
    print(result)

if __name__ == "__main__":
    main()
