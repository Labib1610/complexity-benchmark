We can model this as a job-shop scheduling problem. Since the constraints are small (≤10 jobs, ≤10 machines, ≤50 operations total), we can use a search or dynamic programming approach.  

A common approach is to use **branch and bound** or **constraint propagation** with backtracking.  
We’ll represent each operation with its job, machine, and duration, and try to assign start times while respecting machine and job order constraints.

Here’s the Python solution:

```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    m = int(next(it))
    
    jobs = []
    for _ in range(j):
        o = int(next(it))
        ops = []
        for _ in range(o):
            machine = int(next(it)) - 1
            duration = int(next(it))
            ops.append((machine, duration))
        jobs.append(ops)
    
    # Total operations
    total_ops = sum(len(job) for job in jobs)
    
    # Precompute for each job: list of (machine, duration, job_id, op_index)
    all_ops = []
    for job_id, job in enumerate(jobs):
        for op_idx, (machine, duration) in enumerate(job):
            all_ops.append((machine, duration, job_id, op_idx))
    
    # We'll use a simple backtracking with pruning.
    # State: start times for each operation.
    # We'll assign operations in order of some heuristic.
    
    # For each job, track next operation index to schedule
    job_next = [0] * j
    # For each machine, track when it becomes free
    machine_free = [0] * m
    # For each job, track when it becomes free (previous op end)
    job_free = [0] * j
    
    best = float('inf')
    
    # Sort operations by some heuristic: e.g., longest processing time first
    # But we need to respect job order, so we'll do DFS with pruning.
    
    # We'll use a priority: choose operation with smallest possible start time
    # among available next operations.
    
    def dfs(scheduled_count, current_makespan):
        nonlocal best
        if current_makespan >= best:
            return
        if scheduled_count == total_ops:
            best = current_makespan
            return
        
        # Find candidate operations: for each job, the next unscheduled op
        candidates = []
        for job_id in range(j):
            op_idx = job_next[job_id]
            if op_idx < len(jobs[job_id]):
                machine, duration = jobs[job_id][op_idx]
                # Earliest start = max(job_free[job_id], machine_free[machine])
                start = max(job_free[job_id], machine_free[machine])
                candidates.append((start, job_id, op_idx, machine, duration))
        
        # Sort by earliest start time (and maybe by duration descending)
        candidates.sort(key=lambda x: (x[0], -x[4]))
        
        for start, job_id, op_idx, machine, duration in candidates:
            # Try scheduling this operation
            old_job_free = job_free[job_id]
            old_machine_free = machine_free[machine]
            
            job_free[job_id] = start + duration
            machine_free[machine] = start + duration
            job_next[job_id] += 1
            
            new_makespan = max(current_makespan, start + duration)
            
            dfs(scheduled_count + 1, new_makespan)
            
            # Backtrack
            job_free[job_id] = old_job_free
            machine_free[machine] = old_machine_free
            job_next[job_id] -= 1
    
    dfs(0, 0)
    print(best)

if __name__ == "__main__":
    solve()
```
