from collections import defaultdict
import heapq

def solve():
    j, m = map(int, input().split())
    
    documents = []
    for _ in range(j):
        line = list(map(int, input().split()))
        o = line[0]
        reviews = []
        for i in range(o):
            official = line[1 + 2*i]
            duration = line[1 + 2*i + 1]
            reviews.append((official, duration))
        documents.append(reviews)
    
    # Try all possible permutations of processing documents
    # But since j <= 10, we can't enumerate all permutations
    # We need to use a different approach
    
    # Use a greedy scheduling approach with backtracking or branch and bound
    # Let's model this as a state space search
    
    # State: for each document, how many reviews have been completed
    # For each official, when they become free
    
    # Initial state
    doc_progress = [0] * j  # How many reviews done for each document
    official_free_time = [0] * (m + 1)  # Index 0 unused, officials are 1-indexed
    current_time = 0
    
    # We'll use a priority queue to simulate events
    # Event: (time, type, data)
    # Type 0: official becomes free
    # Type 1: document can start next review
    
    # Initially, all documents can start their first review (if they have any)
    events = []
    for doc_id in range(j):
        if documents[doc_id]:  # If document has reviews
            events.append((0, 1, doc_id))  # Document can start at time 0
    
    heapq.heapify(events)
    
    # To avoid recomputing, we can memoize states
    # But let's try a simpler simulation first
    
    # Let's think differently: at each time step, what can happen?
    # We can model this as a discrete event simulation
    
    # Actually, let's use a more direct approach:
    # Keep track of when each official is free
    # Keep track of which document is waiting for which official
    
    # For each document, we know its review sequence
    # When a review finishes, the next one can start (if official is free)
    
    # Let's use a priority queue of events
    # Events: (time, event_type, doc_id, review_index)
    # event_type: 0 = review starts, 1 = review ends
    
    pq = []  # Priority queue of events
    
    # Initialize: all documents can start their first review at time 0
    for doc_id in range(j):
        if documents[doc_id]:  # If document has reviews
            official, duration = documents[doc_id][0]
            heapq.heappush(pq, (0, 0, doc_id, 0))  # Review 0 of doc_id can start at time 0
    
    official_busy_until = [0] * (m + 1)  # When each official will be free
    doc_review_start_time = [[-1] * len(documents[i]) for i in range(j)]  # When each review starts
    
    max_end_time = 0
    
    while pq:
        time, event_type, doc_id, review_idx = heapq.heappop(pq)
        
        if event_type == 0:  # Review starts
            official, duration = documents[doc_id][review_idx]
            
            # Check if official is free at this time
            if time >= official_busy_until[official]:
                # Official is free, start the review
                start_time = time
                end_time = start_time + duration
                
                official_busy_until[official] = end_time
                doc_review_start_time[doc_id][review_idx] = start_time
                
                # Schedule the end of this review
                heapq.heappush(pq, (end_time, 1, doc_id, review_idx))
                
                max_end_time = max(max_end_time, end_time)
            else:
                # Official is busy, delay the start
                # Re-schedule this event for when official is free
                heapq.heappush(pq, (official_busy_until[official], 0, doc_id, review_idx))
                
        else:  # Review ends
            # The next review for this document can start
            if review_idx + 1 < len(documents[doc_id]):
                # There's a next review
                next_review_idx = review_idx + 1
                # It can start earliest when this review ends and the official for next review is free
                next_official, next_duration = documents[doc_id][next_review_idx]
                earliest_start = max(time, official_busy_until[next_official])
                
                heapq.heappush(pq, (earliest_start, 0, doc_id, next_review_idx))
    
    print(max_end_time)

solve()
