import sys
from collections import deque

def solve():
    j, m = map(int, input().split())
    
    # Read subroutines
    subroutines = []
    for _ in range(j):
        line = list(map(int, input().split()))
        o = line[0]
        gates = []
        for i in range(o):
            core = line[1 + 2*i]
            duration = line[1 + 2*i + 1]
            gates.append((core, duration))
        subroutines.append(gates)
    
    # BFS state: (time, [current position in each subroutine], [next available time for each core])
    # We'll use a priority queue to get the minimum time first
    import heapq
    
    # Initial state
    # time = 0
    # positions = [0] * j (all subroutines start at position 0)
    # core_available_time = [0] * (m+1) (index 0 unused, cores 1..m)
    
    # State in queue: (time, positions, core_available_time)
    initial_positions = tuple([0] * j)
    initial_core_times = tuple([0] * (m + 1))
    
    # Priority queue with (time, positions, core_available_time)
    pq = [(0, initial_positions, initial_core_times)]
    
    # Visited set to avoid revisiting states
    # State is (positions, core_available_time_pattern)
    visited = set()
    
    while pq:
        time, positions, core_available_time = heapq.heappop(pq)
        
        # Check if all subroutines are done
        done = True
        for idx in range(j):
            if positions[idx] < len(subroutines[idx]):
                done = False
                break
        
        if done:
            print(time)
            return
        
        # Create a state key for visited set
        # We normalize core times by subtracting the minimum to make equivalent states match
        min_core_time = min(core_available_time[1:])  # cores 1 to m
        normalized_core_times = tuple(ct - min_core_time for ct in core_available_time)
        state_key = (positions, normalized_core_times)
        
        if state_key in visited:
            continue
        visited.add(state_key)
        
        # Try to schedule the next gate from each subroutine
        for i in range(j):
            # If this subroutine has more gates
            if positions[i] < len(subroutines[i]):
                core, duration = subroutines[i][positions[i]]
                
                # When can this gate start?
                # After the previous gate in this subroutine finishes (which is 'time')
                # AND after the core is free
                start_time = max(time, core_available_time[core])
                end_time = start_time + duration
                
                # Create new state
                new_positions = list(positions)
                new_positions[i] += 1
                new_core_times = list(core_available_time)
                new_core_times[core] = end_time
                
                heapq.heappush(pq, (end_time, tuple(new_positions), tuple(new_core_times)))

solve()
