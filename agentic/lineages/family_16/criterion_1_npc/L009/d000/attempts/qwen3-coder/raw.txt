```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    j = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    
    proposals = []
    for _ in range(j):
        o = int(data[idx])
        idx += 1
        stages = []
        for _ in range(o):
            expert = int(data[idx])
            idx += 1
            duration = int(data[idx])
            idx += 1
            stages.append((expert, duration))
        proposals.append(stages)
    
    # For each expert, keep track of when they become free
    expert_free_time = [0] * (m + 1)  # 1-based indexing
    
    # For each proposal, keep track of when each stage finishes
    proposal_times = [0] * j  # When the previous stage of this proposal finished
    
    # Simulate the process
    # We'll process stages in order, but we need to make sure we respect both:
    # 1. Sequential dependence within a proposal
    # 2. Expert availability
    
    # Let's create a list of all stages with their proposal index and stage index
    all_stages = []
    for p_idx, stages in enumerate(proposals):
        for s_idx, (expert, duration) in enumerate(stages):
            all_stages.append((p_idx, s_idx, expert, duration))
    
    # Actually, better approach: process proposals stage by stage
    # Keep track of current stage for each proposal
    current_stage = [0] * j
    proposal_finished = [False] * j
    total_finished = 0
    
    # For tracking when each proposal's stage can start
    proposal_stage_start = [0] * j
    
    while total_finished < j:
        # Find the next stage that can be started
        best_start_time = float('inf')
        best_proposal = -1
        
        for p in range(j):
            if proposal_finished[p]:
                continue
            if current_stage[p] >= len(proposals[p]):
                proposal_finished[p] = True
                total_finished += 1
                continue
                
            stage_expert, stage_duration = proposals[p][current_stage[p]]
            # This stage can start at max(proposal_stage_start[p], expert_free_time[stage_expert])
            start_time = max(proposal_stage_start[p], expert_free_time[stage_expert])
            
            if start_time < best_start_time:
                best_start_time = start_time
                best_proposal = p
            elif start_time == best_start_time and p < best_proposal:  # Tie-breaker
                best_proposal = p
        
        if total_finished >= j:
            break
            
        # Process the best proposal's current stage
        p = best_proposal
        if p == -1 or proposal_finished[p] or current_stage[p] >= len(proposals[p]):
            # Recheck finished proposals
            if not proposal_finished[p]:
                proposal_finished[p] = True
                total_finished += 1
            continue
            
        stage_expert, stage_duration = proposals[p][current_stage[p]]
        start_time = max(proposal_stage_start[p], expert_free_time[stage_expert])
        end_time = start_time + stage_duration
        
        # Update expert's free time
        expert_free_time[stage_expert] = end_time
        
        # Update proposal's next stage start time
        proposal_stage_start[p] = end_time
        
        # Move to next stage
        current_stage[p] += 1
        if current_stage[p] >= len(proposals[p]):
            proposal_finished[p] = True
            total_finished += 1
    
    # The answer is the maximum of all expert free times
    result = max(expert_free_time)
    print(result)

if __name__ == "__main__":
    main()
```
