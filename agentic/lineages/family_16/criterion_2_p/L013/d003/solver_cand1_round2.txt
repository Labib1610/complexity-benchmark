```python
import sys

def solve():
    # The formula is: F(S) = max_{1 <= k <= j} (sum_{i=1}^k p1_i + sum_{i=k}^j p2_i)
    # Let P1_sum = sum_{i=1}^j p1_i and P2_sum = sum_{i=1}^j p2_i.
    # The expression can be rewritten as:
    # sum_{i=1}^k p1_i + (sum_{i=1}^j p2_i - sum_{i=1}^{k-1} p2_i)
    # = sum_{i=1}^j p2_i + sum_{i=1}^k p1_i - sum_{i=1}^{k-1} p2_i
    # = sum_{i=1}^j p2_i + p1_k + sum_{i=1}^{k-1} (p1_i - p2_i)
    
    # This is a variation of Johnson's rule for scheduling.
    # To minimize the maximum stress, we should sort the nodes based on 
    # the relationship between p1 and p2.
    # Specifically, nodes with p1 < p2 should come first (sorted by p1 ascending),
    # and nodes with p1 >= p2 should come last (sorted by p2 descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    nodes = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        nodes.append((p1, p2))
        idx += 2
        
    # Johnson's rule sorting
    group1 = [n for n in nodes if n[0] < n[1]]
    group2 = [n for n in nodes if n[0] >= n[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_nodes = group1 + group2
    
    # Calculate peak stress for the sorted sequence
    # F(S) = max_{1 <= k <= j} (prefix_p1[k] + suffix_p2[k])
    # where prefix_p1[k] = sum_{i=0}^{k-1} p1_i
    # and suffix_p2[k] = sum_{i=k-1}^{j-1} p2_i
    
    n = len(sorted_nodes)
    prefix_p1 = [0] * (n + 1)
    for i in range(n):
        prefix_p1[i+1] = prefix_p1[i] + sorted_nodes[i][0]
        
    suffix_p2 = [0] * (n + 1)
    for i in range(n - 1, -1, -1):
        suffix_p2[i] = suffix_p2[i+1] + sorted_nodes[i][1]
        
    max_stress = 0
    for k in range(1, n + 1):
        # k is 1-indexed pivot
        current_stress = prefix_p1[k] + suffix_p2[k-1]
        if current_stress > max_stress:
            max_stress = current_stress
            
    print(max_stress)

if __name__ == '__main__':
    solve()
```
