import sys

def solve():
    # Let the artifacts be (r_i, s_i).
    # For a sequence S_1, ..., S_j, the interference at split point k is:
    # I_k = sum_{i=1}^k r_i + sum_{i=k}^j s_i
    # I_k = sum_{i=1}^j s_i + sum_{i=1}^k (r_i - s_i)
    # Let D_i = r_i - s_i.
    # I_k = (sum_{i=1}^j s_i) + (sum_{i=1}^k D_i)
    # To minimize the peak interference max(I_k), we need to minimize
    # max_{1 <= k <= j} (sum_{i=1}^k D_i).
    # Let S_total_s = sum_{i=1}^j s_i.
    # We want to minimize (S_total_s + max_{1 <= k <= j} prefix_sum_D_k).
    # This is equivalent to ordering D_i such that the maximum prefix sum is minimized.
    # To minimize the maximum prefix sum of a set of numbers, we should sort them
    # in non-increasing order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    artifacts = []
    idx = 1
    sum_s = 0
    diffs = []
    for _ in range(j):
        r = int(input_data[idx])
        s = int(input_data[idx+1])
        idx += 2
        sum_s += s
        diffs.append(r - s)
    
    # Sort diffs in descending order to minimize the maximum prefix sum
    diffs.sort(reverse=True)
    
    max_prefix_sum = -float('inf')
    current_prefix_sum = 0
    for d in diffs:
        current_prefix_sum += d
        if current_prefix_sum > max_prefix_sum:
            max_prefix_sum = current_prefix_sum
            
    print(sum_s + max_prefix_sum)

if __name__ == '__main__':
    solve()
