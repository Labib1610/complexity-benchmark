```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    fireworks = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        fireworks.append((p1, p2))
        idx += 2
    
    # Let p1_i be ground ignition time, p2_i be aerial display time.
    # Let the sequence be a permutation sigma.
    # Let S_i = sum_{k=1}^i p1_{sigma(k)} be the time ground ignition i finishes.
    # Let E_i be the time aerial display i finishes.
    # E_0 = 0
    # E_i = max(E_{i-1}, S_i) + p2_{sigma(i)}
    # This is a classic scheduling problem (Johnson's Rule variant).
    # To minimize the completion time, we sort fireworks such that:
    # If p1_i < p2_i, put it earlier. If p1_i >= p2_i, put it later.
    # Specifically, sort by min(p1, p2).
    # If p1 < p2, sort by p1 ascending.
    # If p1 >= p2, sort by p2 descending.
    
    group1 = []
    group2 = []
    for p1, p2 in fireworks:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    current_ground_time = 0
    current_aerial_finish = 0
    
    for p1, p2 in sequence:
        current_ground_time += p1
        # Aerial display starts at max(current_ground_time, current_aerial_finish)
        # and lasts p2.
        if current_ground_time > current_aerial_finish:
            current_aerial_finish = current_ground_time + p2
        else:
            current_aerial_finish = current_aerial_finish + p2
            
    print(current_aerial_finish)

if __name__ == '__main__':
    solve()
```
