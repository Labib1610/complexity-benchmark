import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    packets = []
    for _ in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        packets.append((p1, p2))
    
    # Sort by p1 + p2 in ascending order
    packets.sort(key=lambda x: x[0] + x[1])
    
    # Compute prefix sums of p1
    prefix_p1 = [0] * (j + 1)
    for i in range(j):
        prefix_p1[i + 1] = prefix_p1[i] + packets[i][0]
    
    # Compute suffix sums of p2
    suffix_p2 = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suffix_p2[i] = suffix_p2[i + 1] + packets[i][1]
    
    # Find maximum latency
    max_latency = 0
    for k in range(1, j + 1):
        latency = prefix_p1[k] + suffix_p2[k - 1]
        if latency > max_latency:
            max_latency = latency
    
    print(max_latency)

if __name__ == "__main__":
    solve()
