```python
import sys

def solve():
    # The problem asks to minimize Y_j.
    # Let the sequence of layers be (W_1, O_1), (W_2, O_2), ..., (W_j, O_j).
    # X_k = sum_{i=1}^k W_i
    # Y_k = max(X_k, Y_{k-1}) + O_k
    # This recurrence can be expanded:
    # Y_j = max(X_j + O_j, X_{j-1} + O_{j-1} + O_j, ..., X_1 + O_1 + ... + O_j)
    # Y_j = max_{1 <= k <= j} (X_k + sum_{i=k}^j O_i)
    # Let S_k = sum_{i=1}^k W_i and T_k = sum_{i=k}^j O_i.
    # We want to minimize max_{1 <= k <= j} (S_k + T_k).
    # This is a classic scheduling problem.
    # Consider two adjacent layers i and i+1.
    # If we swap them, the terms S_k + T_k for k != i, i+1 remain unchanged.
    # For the original order (W_i, O_i), (W_{i+1}, O_{i+1}):
    # Term i: S_{i-1} + W_i + O_i + O_{i+1} + ...
    # Term i+1: S_{i-1} + W_i + W_{i+1} + O_{i+1} + ...
    # Max is S_{i-1} + O_{i+1} + ... + max(W_i + O_i, W_i + W_{i+1})
    # After swap (W_{i+1}, O_{i+1}), (W_i, O_i):
    # Max is S_{i-1} + O_i + ... + max(W_{i+1} + O_{i+1}, W_{i+1} + W_i)
    # We prefer the first order if max(W_i + O_i, W_i + W_{i+1}) < max(W_{i+1} + O_{i+1}, W_{i+1} + W_i).
    # This simplifies to: if O_i < O_{i+1}, we put i first.
    # If O_i > O_{i+1}, we put i+1 first.
    # So we sort by O in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    layers = []
    idx = 1
    for _ in range(j):
        w = int(input_data[idx])
        o = int(input_data[idx+1])
        layers.append((w, o))
        idx += 2
    
    # Sort by O descending
    layers.sort(key=lambda x: x[1], reverse=True)
    
    current_x = 0
    current_y = 0
    for w, o in layers:
        current_x += w
        current_y = max(current_x, current_y) + o
        
    print(current_y)

if __name__ == '__main__':
    solve()
```
