```python
import sys

def solve():
    # The problem asks to minimize beta_j.
    # Let S = sum(U_i).
    # beta_k = max(alpha_k, beta_{k-1}) + V_k
    # Since alpha_k = sum_{i=1}^k U_i, alpha_k is always increasing.
    # Let's analyze the recurrence:
    # beta_k = max(sum_{i=1}^k U_i, beta_{k-1}) + V_k
    # This is equivalent to beta_k = max(sum_{i=1}^k U_i + V_k, beta_{k-1} + V_k)
    # By induction, beta_j = max_{1 <= k <= j} (sum_{i=1}^k U_i + sum_{i=k}^j V_i)
    # Let S_k = sum_{i=1}^k U_i.
    # beta_j = max_{1 <= k <= j} (S_k + sum_{i=k}^j V_i)
    # beta_j = max_{1 <= k <= j} (sum_{i=1}^j U_i + sum_{i=k}^j V_i - sum_{i=k+1}^j U_i)
    # Actually, let's re-examine:
    # beta_j = max(S_1 + V_1 + V_2 + ... + V_j, S_2 + V_2 + ... + V_j, ..., S_j + V_j)
    # Let T_k = sum_{i=k}^j V_i. Then beta_j = max_{1 <= k <= j} (S_k + T_k).
    # We want to minimize this maximum.
    # This is a classic exchange argument problem.
    # Suppose we have two adjacent elements (U1, V1) and (U2, V2).
    # If we put (U1, V1) then (U2, V2):
    # beta_2 = max(U1 + V1 + V2, U1 + U2 + V2)
    # If we put (U2, V2) then (U1, V1):
    # beta_2 = max(U2 + V2 + V1, U2 + U1 + V1)
    # We prefer (U1, V1) before (U2, V2) if:
    # max(U1 + V1 + V2, U1 + U2 + V2) < max(U2 + V2 + V1, U2 + U1 + V1)
    # This simplifies to comparing V1 and V2.
    # Specifically, if V1 < V2, the first order is better.
    # If V1 > V2, the second order is better.
    # If V1 == V2, the order doesn't matter.
    # So we should sort elements by V in ascending order.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    elements = []
    idx = 1
    for _ in range(j):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        elements.append((u, v))
        idx += 2
    
    # Sort by V ascending
    elements.sort(key=lambda x: x[1])
    
    alpha = 0
    beta = 0
    for u, v in elements:
        alpha += u
        beta = max(alpha, beta) + v
        
    print(beta)

if __name__ == '__main__':
    solve()
```
