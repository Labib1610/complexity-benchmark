```python
import sys

def solve():
    # The problem asks to minimize the total completion time.
    # Let the packets be (p1_i, p2_i).
    # Let the sequence be i_1, i_2, ..., i_j.
    # Let T_k be the time the k-th packet arrives.
    # T_1 = p1_1
    # T_2 = T_1 + p1_2 = p1_1 + p1_2
    # T_k = sum_{m=1}^k p1_m
    # Let W_k be the workload in the queue immediately after the k-th packet arrives.
    # W_1 = p2_1
    # W_2 = max(0, W_1 - p1_2) + p2_2
    # W_k = max(0, W_{k-1} - p1_k) + p2_k
    # The completion time is T_j + W_j.
    # T_j = sum_{m=1}^j p1_m.
    # W_j = max(0, W_{j-1} - p1_j) + p2_j
    #     = max(0, max(0, W_{j-2} - p1_{j-1}) + p2_{j-1} - p1_j) + p2_j
    #     = max(p2_j, p2_{j-1} + p2_j - p1_j, p2_{j-2} + p2_{j-1} + p2_j - p1_{j-1} - p1_j, ...)
    # This is equivalent to:
    # W_j = max_{1 <= k <= j} (sum_{m=k}^j p2_m - sum_{m=k+1}^j p1_m)
    # Total time = sum_{m=1}^j p1_m + max_{1 <= k <= j} (sum_{m=k}^j p2_m - sum_{m=k+1}^j p1_m)
    # Total time = max_{1 <= k <= j} (sum_{m=1}^j p1_m + sum_{m=k}^j p2_m - sum_{m=k+1}^j p1_m)
    # Total time = max_{1 <= k <= j} (sum_{m=1}^k p1_m + sum_{m=k}^j p2_m)
    # This is a classic scheduling problem. To minimize the maximum of these sums,
    # we should sort the packets by p1_i in ascending order.
    # If p1_i = p1_j, the order doesn't matter for the sum.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    packets = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        packets.append((p1, p2))
        idx += 2
    
    # Sort by p1 ascending
    packets.sort(key=lambda x: x[0])
    
    # Calculate completion time
    # W_k = max(0, W_{k-1} - p1_k) + p2_k
    # T_k = T_{k-1} + p1_k
    # Completion time = T_j + W_j
    
    current_w = 0
    for p1, p2 in packets:
        current_w = max(0, current_w - p1) + p2
        
    total_p1 = sum(p[0] for p in packets)
    print(total_p1 + current_w)

if __name__ == '__main__':
    solve()
```
