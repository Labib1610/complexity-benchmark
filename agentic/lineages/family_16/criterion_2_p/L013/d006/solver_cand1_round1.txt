```python
import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    packets = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        packets.append((p1, p2))
        idx += 2
    
    # Let the sequence be (a1, b1), (a2, b2), ..., (aj, bj)
    # Latency at position k: L_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # L_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # Note that L_k = (sum_{i=1}^k a_i + sum_{i=k+1}^j b_i) + b_k
    # Actually, the problem states:
    # L_k = (sum_{i=1}^k p1_i) + (sum_{i=k}^j p2_i)
    # Let S1_k = sum_{i=1}^k p1_i and S2_k = sum_{i=k}^j p2_i
    # L_k = S1_k + S2_k
    # This is equivalent to:
    # L_k = (sum_{i=1}^{k-1} p1_i) + p1_k + p2_k + (sum_{i=k+1}^j p2_i)
    # Let T = sum_{i=1}^j p1_i + sum_{i=1}^j p2_i. This doesn't simplify easily.
    # Let's re-examine: L_k = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # L_k = sum_{i=1}^{k-1} p1_i + (p1_k + p2_k) + sum_{i=k+1}^j p2_i
    # This is a classic scheduling problem. To minimize the maximum of these sums,
    # we should sort packets such that those with smaller p1 are earlier and 
    # those with smaller p2 are later. Specifically, sort by min(p1, p2).
    # Johnson's rule for 2-machine flow shop scheduling:
    # Group 1: p1 < p2, sort by p1 ascending.
    # Group 2: p1 >= p2, sort by p2 descending.
    
    group1 = []
    group2 = []
    for p1, p2 in packets:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate latencies for the sequence
    n = len(sequence)
    p1_arr = [x[0] for x in sequence]
    p2_arr = [x[1] for x in sequence]
    
    prefix_p1 = [0] * (n + 1)
    for i in range(n):
        prefix_p1[i+1] = prefix_p1[i] + p1_arr[i]
        
    suffix_p2 = [0] * (n + 1)
    for i in range(n - 1, -1, -1):
        suffix_p2[i] = suffix_p2[i+1] + p2_arr[i]
        
    max_latency = 0
    for k in range(1, n + 1):
        # L_k = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
        # indices in prefix_p1 are 1-based, so sum_{i=1}^k p1_i is prefix_p1[k]
        # indices in suffix_p2 are 0-based, so sum_{i=k}^j p2_i is suffix_p2[k-1]
        latency = prefix_p1[k] + suffix_p2[k-1]
        if latency > max_latency:
            max_latency = latency
            
    print(max_latency)

solve()
```
