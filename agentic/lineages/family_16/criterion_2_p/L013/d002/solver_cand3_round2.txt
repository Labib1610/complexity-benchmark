import sys

def solve():
    # This is a variation of Johnson's Rule for the two-machine flow shop scheduling problem.
    # We want to minimize the makespan (the time the last job finishes on the second machine).
    # Johnson's Rule:
    # 1. List all jobs.
    # 2. Divide jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 3. Sort Set 1 in ascending order of p1.
    # 4. Sort Set 2 in descending order of p2.
    # 5. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = [job for job in jobs if job[0] < job[1]]
    set2 = [job for job in jobs if job[0] >= job[1]]
    
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    # alpha_time: time when current job finishes Alpha Rift
    # omega_time: time when current job finishes Omega Rift
    alpha_time = 0
    omega_time = 0
    
    for p1, p2 in sequence:
        alpha_time += p1
        # Omega Rift can start only after Alpha Rift finishes AND previous Omega Rift finishes
        if alpha_time > omega_time:
            omega_time = alpha_time + p2
        else:
            omega_time += p2
            
    print(omega_time)

if __name__ == '__main__':
    solve()
