```python
import sys

def solve():
    # The problem asks us to arrange sections (u_i, l_i) to minimize:
    # max_{k=1..j} (sum_{i=1}^k u_i + sum_{i=k}^j l_i)
    # Let S_u = sum(u_i) and S_l = sum(l_i).
    # The expression is sum_{i=1}^k u_i + sum_{i=k}^j l_i
    # = sum_{i=1}^k u_i + (S_l - sum_{i=1}^{k-1} l_i)
    # = S_l + u_k + sum_{i=1}^{k-1} (u_i - l_i)
    # Let d_i = u_i - l_i.
    # The expression is S_l + u_k + sum_{i=1}^{k-1} d_i.
    # To minimize the maximum of these values, we should sort the sections.
    # If we have two adjacent sections (u1, l1) and (u2, l2), swapping them
    # changes the intermediate values.
    # It is a known result for this type of scheduling problem that sorting
    # by the difference (u_i - l_i) or similar criteria is optimal.
    # Specifically, sorting by u_i - l_i in descending order minimizes the max.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sections = []
    idx = 1
    for _ in range(j):
        u = int(input_data[idx])
        l = int(input_data[idx+1])
        sections.append((u, l))
        idx += 2
    
    # Sort by u_i - l_i descending
    sections.sort(key=lambda x: x[0] - x[1], reverse=True)
    
    # Calculate prefix sums of u and suffix sums of l
    u_vals = [s[0] for s in sections]
    l_vals = [s[1] for s in sections]
    
    pref_u = [0] * (j + 1)
    for i in range(j):
        pref_u[i+1] = pref_u[i] + u_vals[i]
        
    suff_l = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suff_l[i] = suff_l[i+1] + l_vals[i]
        
    ans = 0
    for k in range(1, j + 1):
        # Route descending at section k (1-indexed)
        # sum_{i=1}^k u_i + sum_{i=k}^j l_i
        current_effort = pref_u[k] + suff_l[k-1]
        if k == 1:
            ans = current_effort
        else:
            if current_effort > ans:
                ans = current_effort
                
    print(ans)

if __name__ == '__main__':
    solve()
```
