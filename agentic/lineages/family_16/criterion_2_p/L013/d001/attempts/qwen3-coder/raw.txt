import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    j = int(sys.stdin.readline())
    trains = []
    for _ in range(j):
        p1, p2 = map(int, sys.stdin.readline().split())
        trains.append((p1, p2))
    
    # Sort by p1 ascending, then by p2 descending for tie-breaking
    # This is a standard algorithm for this type of scheduling problem
    # (Johnson's rule for 2-machine flow shop scheduling to minimize makespan)
    # But actually, for this specific problem (minimize max completion time),
    # we should sort by (p1 - p2) ascending, or equivalently, by p1 ascending and p2 descending for ties.
    # Actually, let's think carefully.
    # We want to minimize the maximum completion time.
    # This is the classic F2||C_max problem - flow shop scheduling to minimize makespan.
    # The optimal algorithm is Johnson's rule:
    # - Sort jobs so that min(p1, p2) is non-decreasing for the first group (p1 <= p2)
    # - and min(p1, p2) is non-increasing for the second group (p1 > p2)
    # But there's a simpler way for implementation:
    # Sort by (p1 - p2) ascending. This is equivalent for this problem.
    # Or sort by p1 ascending, then p2 descending for ties.
    # Let me recheck the examples.
    # Example 1: (3,2), (2,4), (1,3)
    # If we sort by p1 ascending: (1,3), (2,4), (3,2)
    # Stage 1: 1 -> 2 -> 3
    # Stage 2: can start at max(current_time, finish_time_of_previous_train_on_stage1)
    # Train 1: stage1 [0,1], stage2 [1,4] 
    # Train 2: stage1 [1,3], stage2 starts at max(3,1)=3, so [3,7]
    # Train 3: stage1 [3,6], stage2 starts at max(6,3)=6, so [6,8]
    # Wait, that's not right. Stage 2 must follow the same order.
    # Train 1: stage1 [0,1], stage2 [1,4] 
    # Train 2: stage1 [1,3], stage2 starts at max(3,4)=4, so [4,8]
    # Train 3: stage1 [3,6], stage2 starts at max(6,8)=8, so [8,10]
    # That gives 10. OK.
    # Let's try sorting by (p1-p2) ascending.
    # (3,2): diff=1
    # (2,4): diff=-2
    # (1,3): diff=-2
    # So order: (2,4), (1,3), (3,2)
    # Train 1 (2,4): stage1 [0,2], stage2 [2,6]
    # Train 2 (1,3): stage1 [2,3], stage2 starts at max(3,6)=6, so [6,9]
    # Train 3 (3,2): stage1 [3,6], stage2 starts at max(6,9)=9, so [9,11]
    # That gives 11. Not optimal.
    # So sorting by p1 ascending is better for this example.
    # But let's check the theory.
    # For F2||C_max, the optimal schedule is given by Johnson's rule.
    # Partition jobs into two sets:
    # A = {j : p1_j <= p2_j}
    # B = {j : p1_j > p2_j}
    # In A, schedule by p1 ascending.
    # In B, schedule by p2 descending.
    # Then concatenate A followed by B.
    # Let's implement Johnson's rule.
    
    A = []  # jobs with p1 <= p2
    B = []  # jobs with p1 > p2
    for i in range(j):
        p1, p2 = trains[i]
        if p1 <= p2:
            A.append((p1, p2, i))
        else:
            B.append((p1, p2, i))
    
    # Sort A by p1 ascending
    A.sort(key=lambda x: x[0])
    # Sort B by p2 descending
    B.sort(key=lambda x: x[1], reverse=True)
    
    # The optimal sequence is A followed by B
    sequence = A + B
    
    # Now simulate the schedule
    time_stage1 = 0  # current time on stage 1
    time_stage2 = 0  # current time on stage 2
    
    for p1, p2, idx in sequence:
        # Train starts stage 1
        start_stage1 = time_stage1
        finish_stage1 = start_stage1 + p1
        time_stage1 = finish_stage1
        
        # Train starts stage 2 as soon as both:
        # 1. Stage 2 is free (time_stage2)
        # 2. Train finishes stage 1 (finish_stage1)
        start_stage2 = max(time_stage2, finish_stage1)
        finish_stage2 = start_stage2 + p2
        time_stage2 = finish_stage2
    
    print(time_stage2)

# Use threading to increase recursion limit if needed
threading.Thread(target=main).start()
