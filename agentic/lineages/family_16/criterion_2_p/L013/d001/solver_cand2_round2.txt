import sys

def solve():
    # The problem asks to minimize the completion time of the last train.
    # Let the trains be ordered 1, 2, ..., j.
    # Let p1_i and p2_i be the times for train i.
    # The finish time of train i on stage 1 is S1_i = sum_{k=1}^i p1_k.
    # The finish time of train i on stage 2 is S2_i = max(S2_{i-1}, S1_i) + p2_i.
    # This is a classic scheduling problem known as Johnson's Rule for 2 machines.
    # To minimize the makespan, we sort the trains:
    # 1. Trains with p1 < p2 are sorted by p1 in ascending order.
    # 2. Trains with p1 >= p2 are sorted by p2 in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    trains = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        trains.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in trains:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_trains = group1 + group2
    
    current_s1 = 0
    current_s2 = 0
    
    for p1, p2 in sorted_trains:
        current_s1 += p1
        current_s2 = max(current_s2, current_s1) + p2
        
    print(current_s2)

if __name__ == '__main__':
    solve()
