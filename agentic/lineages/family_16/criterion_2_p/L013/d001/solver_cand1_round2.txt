import sys

def solve():
    # This is a classic Johnson's algorithm problem for the 2-machine flow shop scheduling.
    # To minimize the makespan, we sort the jobs based on the following rule:
    # If min(p1, p2) == p1, the job goes to the front of the sequence (sorted by p1 ascending).
    # If min(p1, p2) == p2, the job goes to the back of the sequence (sorted by p2 descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    # Sort group1 by p1 ascending
    group1.sort(key=lambda x: x[0])
    # Sort group2 by p2 descending
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate makespan
    time_f1 = 0
    time_f2 = 0
    
    for p1, p2 in sequence:
        time_f1 += p1
        if time_f1 > time_f2:
            time_f2 = time_f1 + p2
        else:
            time_f2 += p2
            
    print(time_f2)

if __name__ == '__main__':
    solve()
