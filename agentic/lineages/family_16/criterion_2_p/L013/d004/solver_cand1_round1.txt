```python
import sys

def solve():
    # The grid has 2 rows and j columns.
    # Let the tiles be (u_i, l_i) for i=1 to j.
    # A path consists of moving right through some number of columns,
    # then moving down, then moving right through the remaining columns.
    # If we move down at column k (1 <= k <= j), the path visits:
    # (u_1, u_2, ..., u_k, l_k, l_{k+1}, ..., l_j).
    # The score for a fixed k is: sum(u_1...u_k) + sum(l_k...l_j).
    # This can be rewritten as: sum(u_1...u_j) + sum(l_1...l_j) - sum(u_{k+1}...u_j) - sum(l_1...l_{k-1}).
    # Let S_u = sum(u_i) and S_l = sum(l_i).
    # Score(k) = S_u + S_l - (sum_{i=k+1}^j u_i + sum_{i=1}^{k-1} l_i).
    # To minimize the vulnerability (max score over k), we want to minimize
    # max_k (S_u + S_l - (sum_{i=k+1}^j u_i + sum_{i=1}^{k-1} l_i)).
    # This is equivalent to maximizing min_k (sum_{i=k+1}^j u_i + sum_{i=1}^{k-1} l_i).
    # Let f(k) = sum_{i=k+1}^j u_i + sum_{i=1}^{k-1} l_i.
    # We want to arrange tiles to maximize min_k f(k).
    # Note f(1) = sum_{i=2}^j u_i, f(j) = sum_{i=1}^{j-1} l_i.
    # For any k, f(k+1) - f(k) = l_k - u_{k+1}.
    # To keep f(k) values close to each other, we should sort tiles.
    # If we sort such that l_i - u_i is non-decreasing, we minimize the range of f(k).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        u = int(input_data[idx])
        l = int(input_data[idx+1])
        tiles.append((u, l))
        idx += 2
        
    # Sort by l_i - u_i
    tiles.sort(key=lambda x: x[1] - x[0])
    
    u_vals = [t[0] for t in tiles]
    l_vals = [t[1] for t in tiles]
    
    s_u = sum(u_vals)
    s_l = sum(l_vals)
    
    # f(k) = sum_{i=k+1}^j u_i + sum_{i=1}^{k-1} l_i
    # f(1) = sum_{i=2}^j u_i
    # f(k+1) = f(k) + l_k - u_{k+1}
    
    current_f = sum(u_vals[1:])
    min_f = current_f
    for k in range(1, j):
        current_f = current_f + l_vals[k-1] - u_vals[k]
        if current_f < min_f:
            min_f = current_f
            
    print(s_u + s_l - min_f)

solve()
```
