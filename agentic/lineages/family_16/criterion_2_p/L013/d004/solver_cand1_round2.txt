import sys

def solve():
    # The problem asks to minimize the final peak stress E.
    # Let the sequence be (p1_i, p2_i) for i = 1 to j.
    # C_i = sum_{k=1}^i p1_k
    # E_0 = 0
    # E_i = max(E_{i-1}, C_i) + p2_i
    # This is a classic scheduling problem.
    # Let's analyze the order. Suppose we have two adjacent nodes (p1_a, p2_a) and (p1_b, p2_b).
    # If we process a then b:
    # E_a = max(E_{prev}, C_{prev} + p1_a) + p2_a
    # E_b = max(E_a, C_{prev} + p1_a + p1_b) + p2_b
    #     = max(max(E_{prev}, C_{prev} + p1_a) + p2_a, C_{prev} + p1_a + p1_b) + p2_b
    #
    # This is equivalent to minimizing the completion time in a flow-shop-like problem.
    # Specifically, this is Johnson's Rule or a variation.
    # We sort nodes based on the sign of (p1_i - p2_i).
    # If p1_i < p2_i, the node should come earlier.
    # If p1_i >= p2_i, the node should come later.
    # Sort by p1_i ascending for p1_i < p2_i, and by p2_i descending for p1_i >= p2_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    nodes = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        nodes.append((p1, p2))
        idx += 2
        
    # Group 1: p1 < p2, sort by p1 ascending
    # Group 2: p1 >= p2, sort by p2 descending
    group1 = [n for n in nodes if n[0] < n[1]]
    group2 = [n for n in nodes if n[0] >= n[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    current_c = 0
    current_e = 0
    for p1, p2 in sequence:
        current_c += p1
        current_e = max(current_e, current_c) + p2
        
    print(current_e)

if __name__ == '__main__':
    solve()
