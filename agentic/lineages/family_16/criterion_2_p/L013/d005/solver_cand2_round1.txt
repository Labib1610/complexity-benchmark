import sys

def solve():
    # The problem asks to minimize (Maximum Spike) + (Sum of all p2).
    # Let the sequence of nodes be (p1_1, p2_1), (p1_2, p2_2), ..., (p1_j, p2_j).
    # Baseline after k nodes: B_k = sum_{i=1}^k (p1_i - p2_i).
    # Spike at step k: S_k = B_{k-1} + p1_k = sum_{i=1}^{k-1} (p1_i - p2_i) + p1_k.
    # Maximum Spike = max_{k=1..j} (S_k).
    # Peak Stress = max_{k=1..j} (sum_{i=1}^{k-1} (p1_i - p2_i) + p1_k) + sum_{i=1}^j p2_i.
    # Let D_i = p1_i - p2_i.
    # Peak Stress = max_{k=1..j} (sum_{i=1}^{k-1} D_i + p1_k) + sum_{i=1}^j p2_i.
    # Note that sum_{i=1}^{k-1} D_i + p1_k = sum_{i=1}^{k-1} (p1_i - p2_i) + p1_k
    # = sum_{i=1}^k p1_i - sum_{i=1}^{k-1} p2_i.
    # Peak Stress = max_{k=1..j} (sum_{i=1}^k p1_i - sum_{i=1}^{k-1} p2_i) + sum_{i=1}^j p2_i
    # = max_{k=1..j} (sum_{i=1}^k p1_i + sum_{i=k+1}^j p2_i).
    
    # Let S = sum_{i=1}^j p2_i.
    # Peak Stress = max_{k=1..j} (sum_{i=1}^k p1_i + sum_{i=k+1}^j p2_i).
    # This is equivalent to:
    # We have nodes with (p1, p2). We want to order them to minimize the max prefix sum of p1s and suffix sum of p2s.
    # This is a classic scheduling problem.
    # Split nodes into two sets:
    # Set 1: p1_i <= p2_i. These should come first, sorted by p1 ascending.
    # Set 2: p1_i > p2_i. These should come last, sorted by p2 descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    nodes = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        nodes.append((p1, p2))
        idx += 2
        
    set1 = [n for n in nodes if n[0] <= n[1]]
    set2 = [n for n in nodes if n[0] > n[1]]
    
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    order = set1 + set2
    
    max_spike = 0
    current_baseline = 0
    for p1, p2 in order:
        spike = current_baseline + p1
        if spike > max_spike:
            max_spike = spike
        current_baseline += (p1 - p2)
        
    total_p2 = sum(n[1] for n in nodes)
    print(max_spike + total_p2)

solve()
