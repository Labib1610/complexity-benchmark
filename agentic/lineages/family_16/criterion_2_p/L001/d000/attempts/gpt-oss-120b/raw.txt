```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    front = []  # p1 <= p2
    back = []   # p1 > p2
    for _ in range(n):
        p1 = int(next(it)); p2 = int(next(it))
        if p1 <= p2:
            front.append((p1, p2))
        else:
            back.append((p1, p2))
    front.sort(key=lambda x: x[0])          # increasing p1
    back.sort(key=lambda x: x[1], reverse=True)  # decreasing p2
    order = front + back

    tA = 0
    tB = 0
    for p1, p2 in order:
        tA += p1
        if tA > tB:
            tB = tA + p2
        else:
            tB = tB + p2
    sys.stdout.write(str(tB))

if __name__ == "__main__":
    solve()
```
