```python
import sys

def solve():
    # The problem is equivalent to Johnson's Rule for two-machine scheduling.
    # We want to minimize the makespan on two machines where the second machine
    # cannot start a job until the first machine finishes it.
    # Let p1_i be the Fetcher time and p2_i be the Executor time.
    # Sort jobs into two groups:
    # Group 1: p1_i < p2_i, sorted by p1_i ascending.
    # Group 2: p1_i >= p2_i, sorted by p2_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_jobs = group1 + group2
    
    fetcher_time = 0
    executor_time = 0
    
    for p1, p2 in sorted_jobs:
        fetcher_time += p1
        # Executor starts at max(fetcher_time, current_executor_time)
        if fetcher_time > executor_time:
            executor_time = fetcher_time + p2
        else:
            executor_time += p2
            
    print(executor_time)

if __name__ == '__main__':
    solve()
```
