import sys

def solve():
    # The problem asks to minimize the maximum clearance metric.
    # Let the sequence of dossiers be (p1_i, p2_i) for i = 1 to j.
    # Let S1_k = sum_{i=1}^k p1_i and S2_k = sum_{i=k}^j p2_i.
    # The clearance metric for dossier k is M_k = S1_k + S2_k.
    # Note that S2_k = (sum_{i=1}^j p2_i) - (sum_{i=1}^{k-1} p2_i).
    # Let T = sum_{i=1}^j p2_i.
    # M_k = S1_k + T - sum_{i=1}^{k-1} p2_i.
    # M_k = p1_k + sum_{i=1}^{k-1} (p1_i - p2_i) + T.
    # This is a classic scheduling problem. To minimize the maximum of these values,
    # we should sort the dossiers based on the difference (p1_i - p2_i).
    # Specifically, if we have two adjacent dossiers (p1_a, p2_a) and (p1_b, p2_b),
    # swapping them changes the max of their metrics.
    # The optimal order is to sort by p1_i < p2_i first, then others.
    # Actually, the standard result for this type of problem (Johnson's rule variant)
    # is to sort by min(p1_i, p2_i).
    # Specifically, dossiers with p1_i < p2_i should come first (sorted by p1_i ascending),
    # and dossiers with p1_i >= p2_i should come after (sorted by p2_i descending).

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    dossiers = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        dossiers.append((p1, p2))
        idx += 2

    # Sort: p1 < p2 first (ascending p1), then p1 >= p2 (descending p2)
    group1 = [d for d in dossiers if d[0] < d[1]]
    group2 = [d for d in dossiers if d[0] >= d[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    max_metric = 0
    current_p1_sum = 0
    total_p2_sum = sum(d[1] for d in dossiers)
    
    current_p2_sum_remaining = total_p2_sum
    
    for p1, p2 in sequence:
        current_p1_sum += p1
        metric = current_p1_sum + current_p2_sum_remaining
        if metric > max_metric:
            max_metric = metric
        current_p2_sum_remaining -= p2
        
    print(max_metric)

if __name__ == '__main__':
    solve()
