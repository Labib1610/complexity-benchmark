```python
import sys

def solve():
    # The problem asks to minimize the completion time of the last merchant.
    # Let the sequence of merchants be (a_i, b_i) where a_i is the time at gate 1
    # and b_i is the time at gate 2.
    # This is a classic Johnson's Rule problem for a 2-machine flow shop.
    # To minimize the makespan, we sort the merchants:
    # 1. Merchants with a_i < b_i are sorted by increasing a_i.
    # 2. Merchants with a_i >= b_i are sorted by decreasing b_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    merchants = []
    idx = 1
    for _ in range(n):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        merchants.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for m in merchants:
        if m[0] < m[1]:
            group1.append(m)
        else:
            group2.append(m)
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate completion time
    # time_gate1: time when current merchant finishes gate 1
    # time_gate2: time when current merchant finishes gate 2
    time_gate1 = 0
    time_gate2 = 0
    
    for p1, p2 in order:
        time_gate1 += p1
        # The merchant can start at gate 2 only after finishing gate 1
        # AND after the previous merchant has finished gate 2.
        time_gate2 = max(time_gate1, time_gate2) + p2
        
    print(time_gate2)

if __name__ == '__main__':
    solve()
```
