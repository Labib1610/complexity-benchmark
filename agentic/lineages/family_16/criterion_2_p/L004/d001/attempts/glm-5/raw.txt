```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    p1 = [0]*j
    p2 = [0]*j
    idx = 1
    for i in range(j):
        p1[i] = int(data[idx]); p2[i] = int(data[idx+1])
        idx += 2
    
    # Sort by p1 - p2 ascending (i.e., p1 + (-p2) ascending)
    order = sorted(range(j), key=lambda i: p1[i] - p2[i])
    
    total_p1 = sum(p1)
    total_p2 = sum(p2)
    
    # For each position k (1-indexed), strain = prefix_p1[k] + suffix_p2[k]
    # suffix_p2[k] = total_p2 - prefix_p2[k-1]
    # strain = prefix_p1[k] + total_p2 - prefix_p2[k-1]
    #         = (prefix_p1[k] - prefix_p2[k-1]) + total_p2
    #         = (prefix_p1[k-1] + p1[k] - prefix_p2[k-1]) + total_p2
    #         = (prefix_p1[k-1] - prefix_p2[k-1]) + p1[k] + total_p2
    # Let D[k-1] = prefix_p1[k-1] - prefix_p2[k-1], then strain = D[k-1] + p1[k] + total_p2
    # We want to minimize max over k of (D[k-1] + p1[k]) + total_p2
    
    # D[0] = 0, D[k] = D[k-1] + p1[k] - p2[k]
    # We need max over k of (D[k-1] + p1[k])
    
    max_val = 0
    D = 0
    for k in range(j):
        i = order[k]
        val = D + p1[i]
        if val > max_val:
            max_val = val
        D += p1[i] - p2[i]
    
    ans = max_val + total_p2
    print(ans)

solve()
```
