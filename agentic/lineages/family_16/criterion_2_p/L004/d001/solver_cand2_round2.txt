import sys

def solve():
    # The problem asks to minimize the completion time of the last ritual.
    # Let p1_i be the inscription time and p2_i be the channeling time.
    # If we process rituals in some order, the Inscriber finishes the i-th ritual
    # at time S_i = sum_{k=1}^i p1_k.
    # The Channeler finishes the i-th ritual at time C_i.
    # C_i = max(C_{i-1}, S_i) + p2_i, with C_0 = 0.
    # This is a classic scheduling problem equivalent to Johnson's rule for 
    # two-machine flow shop scheduling, where we want to minimize the makespan.
    # Johnson's rule:
    # 1. Divide rituals into two sets:
    #    Set A: rituals where p1_i < p2_i
    #    Set B: rituals where p1_i >= p2_i
    # 2. Sort Set A in ascending order of p1_i.
    # 3. Sort Set B in descending order of p2_i.
    # 4. The optimal sequence is Set A followed by Set B.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    rituals = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        rituals.append((p1, p2))
        idx += 2
    
    set_a = []
    set_b = []
    for p1, p2 in rituals:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    current_inscriber_time = 0
    current_channeler_time = 0
    
    for p1, p2 in sequence:
        current_inscriber_time += p1
        # Channeler starts at max(when circle is ready, when channeler is free)
        start_channeling = max(current_inscriber_time, current_channeler_time)
        current_channeler_time = start_channeling + p2
        
    print(current_channeler_time)

if __name__ == '__main__':
    solve()
