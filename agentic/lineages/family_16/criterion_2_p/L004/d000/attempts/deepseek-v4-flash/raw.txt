We can solve this by sorting the rituals using a greedy rule:  
- If both rituals have the same relation between inscribe and channel times, we need to decide the order carefully.  
- The optimal order is to put rituals with `p1 <= p2` first, sorted by `p1` ascending, then rituals with `p1 > p2` after, sorted by `p2` descending.

Here’s the implementation:

```python
import sys

def solve():
    input = sys.stdin.readline
    j = int(input())
    rituals = []
    for _ in range(j):
        p1, p2 = map(int, input().split())
        rituals.append((p1, p2))
    
    # Split into two groups
    group1 = []  # p1 <= p2
    group2 = []  # p1 > p2
    
    for p1, p2 in rituals:
        if p1 <= p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
    
    # Sort group1 by p1 ascending
    group1.sort(key=lambda x: x[0])
    # Sort group2 by p2 descending
    group2.sort(key=lambda x: -x[1])
    
    ordered = group1 + group2
    
    inscriber_time = 0
    channeler_time = 0
    
    for p1, p2 in ordered:
        inscriber_time += p1
        if channeler_time < inscriber_time:
            channeler_time = inscriber_time
        channeler_time += p2
    
    print(channeler_time)

if __name__ == "__main__":
    solve()
```
