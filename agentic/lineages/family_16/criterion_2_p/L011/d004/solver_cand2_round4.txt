```python
import sys

def solve():
    # The problem asks to minimize the final energy level E.
    # Let the modules be (d_i, r_i).
    # If we have a sequence of modules, the energy level E_k after k steps is:
    # E_0 = 0
    # E_k = max(0, E_{k-1} - d_k) + r_k
    # This is equivalent to E_k = E_{k-1} - d_k + r_k + max(d_k - E_{k-1}, 0)
    # Summing this up: E_n = sum(r_i - d_i) + sum(max(d_i - E_{i-1}, 0))
    # Let S = sum(r_i - d_i). Then E_n = S + sum(max(d_i - E_{i-1}, 0)).
    # To minimize E_n, we need to minimize the sum of the "lost" depletions.
    # This is a classic scheduling problem.
    # Modules with d_i <= r_i should be processed first in increasing order of d_i.
    # Modules with d_i > r_i should be processed last in decreasing order of r_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    modules = []
    idx = 1
    sum_d = 0
    for _ in range(j):
        d = int(input_data[idx])
        r = int(input_data[idx+1])
        modules.append((d, r))
        sum_d += d
        idx += 2
        
    group1 = [] # d <= r
    group2 = [] # d > r
    for d, r in modules:
        if d <= r:
            group1.append((d, r))
        else:
            group2.append((d, r))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    current_e = 0
    for d, r in sequence:
        current_e = max(0, current_e - d) + r
        
    print(current_e + sum_d)

if __name__ == '__main__':
    solve()
```
