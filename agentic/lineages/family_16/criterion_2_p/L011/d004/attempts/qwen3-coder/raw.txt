import sys
from functools import cmp_to_key

def compare(x, y):
    # x[0] is A_i, x[1] is B_i
    # Compare x then y vs y then x
    A1, B1 = x
    A2, B2 = y
    
    # If we do x then y:
    # S1_x = A1
    # S2_x = max(S1_x + B1, A1 + A2) = max(A1 + B1, A1 + A2)
    # C_x = S2_x + B2 = max(A1 + B1, A1 + A2) + B2
    
    # If we do y then x:
    # S1_y = A2
    # S2_y = max(S1_y + B2, A2 + A1) = max(A2 + B2, A1 + A2)
    # C_y = S2_y + B1 = max(A2 + B2, A1 + A2) + B1
    
    # We want to minimize total time, so we prefer the arrangement with smaller C
    # So if C_x < C_y, we prefer x then y, so x should come before y
    
    C_x = max(A1 + B1, A1 + A2) + B2
    C_y = max(A2 + B2, A1 + A2) + B1
    
    if C_x < C_y:
        return -1
    elif C_x > C_y:
        return 1
    else:
        return 0

def main():
    j = int(sys.stdin.readline())
    performances = []
    
    for _ in range(j):
        p1, p2 = map(int, sys.stdin.readline().split())
        performances.append((p1, p2))
    
    # Sort using our custom comparator
    performances.sort(key=cmp_to_key(compare))
    
    # Calculate the minimum C
    S = 0  # Review room start time
    total_A = 0  # Sum of Main Stage durations up to current performance
    
    for A, B in performances:
        total_A += A
        S = max(S + B_prev if 'B_prev' in locals() else A, total_A)
        B_prev = B
    
    C = S + B_prev if j > 0 else 0
    print(C)

if __name__ == "__main__":
    main()
