```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    performances = []
    for _ in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        performances.append((p1, p2))
    
    # Sort by p1 in descending order
    performances.sort(key=lambda x: -x[0])
    
    # Compute prefix sums of p1
    prefix = [0] * (j + 1)
    for i in range(j):
        prefix[i+1] = prefix[i] + performances[i][0]
    
    # Find max of (prefix[i] + p2_i)
    max_val = 0
    for i in range(j):
        val = prefix[i+1] + performances[i][1]
        if val > max_val:
            max_val = val
    
    # Total time is max of (total_p1 + total_p2, max_val)
    total_p1 = prefix[j]
    total_p2 = sum(p[1] for p in performances)
    result = max(total_p1 + total_p2, max_val)
    print(result)

solve()
```
