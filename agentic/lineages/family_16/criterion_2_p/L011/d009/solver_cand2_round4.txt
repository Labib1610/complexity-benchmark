import sys

def solve():
    # The problem asks to minimize the final completion time T.
    # Let the sequence of trains be (X_1, Y_1), (X_2, Y_2), ..., (X_j, Y_j).
    # R_k = sum_{i=1}^k X_i
    # T_0 = 0
    # T_k = max(T_{k-1}, R_k) + Y_k
    # This is a classic scheduling problem known as Johnson's Rule for 2 machines.
    # To minimize the makespan on two machines where each job must be processed
    # on machine 1 then machine 2:
    # 1. Divide jobs into two sets:
    #    Set 1: X_i < Y_i
    #    Set 2: X_i >= Y_i
    # 2. Sort Set 1 in ascending order of X_i.
    # 3. Sort Set 2 in descending order of Y_i.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    trains = []
    idx = 1
    for _ in range(j):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        trains.append((x, y))
        idx += 2
        
    set1 = []
    set2 = []
    for x, y in trains:
        if x < y:
            set1.append((x, y))
        else:
            set2.append((x, y))
            
    set1.sort(key=lambda p: p[0])
    set2.sort(key=lambda p: p[1], reverse=True)
    
    sequence = set1 + set2
    
    r = 0
    t = 0
    for x, y in sequence:
        r += x
        if t < r:
            t = r
        t += y
        
    print(t)

if __name__ == '__main__':
    solve()
