```python
import sys

def solve():
    # Let the artifacts be (d_k, n_k) for k=1 to j.
    # For a fixed arrangement, the score for focal position i is:
    # S_i = sum_{k=1}^i d_k + sum_{k=i}^j n_k
    # S_i = sum_{k=1}^j n_k + sum_{k=1}^i (d_k - n_k)
    # Let D_k = d_k - n_k.
    # S_i = (sum_{k=1}^j n_k) + (sum_{k=1}^i D_k)
    # Let N_total = sum_{k=1}^j n_k.
    # S_i = N_total + prefix_sum_D[i]
    # The critic chooses i to maximize S_i, so the score is N_total + max_{1 <= i <= j} (prefix_sum_D[i]).
    # We want to minimize this value by reordering the artifacts.
    # Since N_total is constant regardless of order, we need to minimize max_{1 <= i <= j} (prefix_sum_D[i]).
    # Let the values D_k be sorted. To minimize the maximum prefix sum, we should sort D_k in non-increasing order.
    # Wait, if we sort D_k in non-increasing order, the prefix sums will grow as fast as possible.
    # To minimize the maximum prefix sum, we should sort D_k in non-decreasing order.
    # Let the sorted values be D_{(1)} <= D_{(2)} <= ... <= D_{(j)}.
    # The prefix sums are P_i = sum_{k=1}^i D_{(k)}.
    # The maximum prefix sum will be P_j = sum_{k=1}^j D_{(k)} = sum_{k=1}^j (d_k - n_k).
    # Wait, if we sort D_k in non-decreasing order, the maximum prefix sum is always the sum of all D_k?
    # Let's re-check:
    # If D = [1, -2, 3], sorted: [-2, 1, 3]. Prefixes: -2, -1, 2. Max is 2.
    # If D = [1, -2, 3], reverse sorted: [3, 1, -2]. Prefixes: 3, 4, 2. Max is 4.
    # Yes, sorting non-decreasingly makes the prefix sums as small as possible.
    # The maximum prefix sum of a non-decreasing sequence is always the last element (the sum of all).
    # So the minimum possible maximum score is N_total + sum_{k=1}^j (d_k - n_k) = sum_{k=1}^j d_k.
    # Wait, is it always sum(d_k)? Let's check Example 1:
    # (3,2), (2,4), (1,3). D values: 3-2=1, 2-4=-2, 1-3=-2.
    # Sorted D: -2, -2, 1.
    # N_total = 2+4+3 = 9.
    # Prefixes: -2, -4, -3.
    # Max prefix: -2.
    # Score: 9 + (-2) = 7? But the example output is 10.
    # Ah, the critic chooses i such that 1 <= i <= j.
    # S_i = sum_{k=1}^i d_k + sum_{k=i}^j n_k.
    # S_i = (sum_{k=1}^j n_k) + sum_{k=1}^i (d_k - n_k).
    # My formula was correct. Let's re-calculate Example 1 with sorted D:
    # D = [-2, -2, 1].
    # i=1: S_1 = 9 + (-2) = 7.
    # i=2: S_2 = 9 + (-2 + -2) = 5.
    # i=3: S_3 = 9 + (-2 + -2 + 1) = 6.
    # Max is 7. Why is the example 10?
    # Re-reading: "The sum of the daytime appeals of all artifacts from position 1 up to position i (inclusive), plus the sum of the nighttime appeals of all artifacts from position i up to position j (inclusive)."
    # S_i = (d_1 + ... + d_i) + (n_i + ... + n_j).
    # S_i = (d_1 + ... + d_i) + (n_i + n_{i+1} + ... + n_j).
    # This is (d_1 + ... + d_{i-1}) + (d_i + n_i) + (n_{i+1} + ... + n_j).
    # Let A_k = d_k + n_k.
    # S_i = sum_{k=1}^{i-1} d_k + A_i + sum_{k=i+1}^j n_k.
    # This is not a simple prefix sum.
    # Let's use the property: S_i - S_{i-1} = d_i - n_{i-1}.
    # To minimize max(S_i), we should arrange such that d_i - n_{i-1} is negative.
    # This is a known problem: minimize max prefix sum of (d_i - n_{i-1}).
    # Actually, the optimal strategy is to sort by n_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data: return
    j = int(input_data[0])
    artifacts = []
    for k in range(j):
        artifacts.append((int(input_data[2*k+1]), int(input_data[2*k+2])))
    
    # Sort by n_i descending
    artifacts.sort(key=lambda x: x[1], reverse=True)
    
    d = [a[0] for a in artifacts]
    n = [a[1] for a in artifacts]
    
    # S_i = sum_{k=0}^{i-1} d_k + sum_{k=i-1}^{j-1} n_k
    # S_i = sum_{k=0}^{i-1} d_k + n_{i-1} + sum_{k=i}^{j-1} n_k
    # S_i = sum_{k=0}^{i-2} d_k + (d_{i-1} + n_{i-1}) + sum_{k=i}^{j-1} n_k
    
    max_s = 0
    current_d = 0
    total_n = sum(n)
    
    for i in range(j):
        # S_{i+1} = current_d + d_i + (total_n - sum(n[:i]))
        s = current_d + d[i] + (total_n - sum(n[:i]))
        if i == 0: max_s = s
        else: max_s = max(max_s, s)
        current_d += d[i]
        
    print(max_s)

solve()
```
