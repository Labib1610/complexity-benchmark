import sys

def solve():
    # The problem asks to minimize the final Calibration Clock.
    # Let the sequence of maneuvers be (b_i, c_i).
    # Let B_i be the Burn Clock after i maneuvers, C_i be the Calibration Clock.
    # B_i = B_{i-1} + b_i
    # C_i = max(C_{i-1}, B_i) + c_i
    # This is a variation of Johnson's rule or the flow-shop scheduling problem.
    # Specifically, we want to minimize the makespan of a 2-machine flow shop
    # where the first machine is the Burn process and the second is the Calibration.
    # The optimal order is to sort maneuvers such that those with b_i < c_i 
    # come first (sorted by increasing b_i), and those with b_i >= c_i 
    # come last (sorted by decreasing c_i).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    maneuvers = []
    idx = 1
    for _ in range(j):
        b = int(input_data[idx])
        c = int(input_data[idx+1])
        maneuvers.append((b, c))
        idx += 2
        
    # Johnson's rule:
    # Group 1: b_i < c_i, sort by b_i ascending
    # Group 2: b_i >= c_i, sort by c_i descending
    group1 = []
    group2 = []
    for b, c in maneuvers:
        if b < c:
            group1.append((b, c))
        else:
            group2.append((b, c))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    burn_clock = 0
    cal_clock = 0
    
    for b, c in sequence:
        burn_clock += b
        if cal_clock < burn_clock:
            cal_clock = burn_clock
        cal_clock += c
        
    print(cal_clock)

if __name__ == '__main__':
    solve()
