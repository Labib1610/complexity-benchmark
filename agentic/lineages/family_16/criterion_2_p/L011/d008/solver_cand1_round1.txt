```python
import sys

def solve():
    # The problem asks to arrange tiles (a_i, b_i) to minimize the maximum path cost.
    # A path consists of moving right until some column k, then moving down, 
    # then moving right until the end.
    # The cost of a path switching at column k is:
    # sum(a_1...a_k) + sum(b_k...b_j)
    # Let S_a = sum(a_i) and S_b = sum(b_i).
    # The cost is sum(a_1...a_k) + sum(b_k...b_j) = sum(a_1...a_{k-1}) + a_k + b_k + sum(b_{k+1}...b_j)
    # This is equal to (sum(a_1...a_{k-1}) + sum(b_{k+1}...b_j)) + (a_k + b_k).
    # Let P_k = a_k + b_k.
    # The cost is (sum_{i=1}^{k-1} a_i) + P_k + (sum_{i=k+1}^j b_i).
    # Let S_a_prefix[k-1] = sum_{i=1}^{k-1} a_i
    # Let S_b_suffix[k+1] = sum_{i=k+1}^j b_i
    # Cost = S_a_prefix[k-1] + P_k + S_b_suffix[k+1]
    # Note that S_a_prefix[k-1] + S_b_suffix[k+1] = (S_a - sum_{i=k}^j a_i) + (S_b - sum_{i=1}^k b_i)
    # This is equivalent to: Total_Sum + sum_{i=1}^{k-1} (a_i - b_i) - (a_k + b_k) + sum_{i=k+1}^j (b_i - a_i) ...
    # Actually, a simpler observation:
    # Cost = sum_{i=1}^{k-1} a_i + a_k + b_k + sum_{i=k+1}^j b_i
    # Let x_i = a_i - b_i.
    # Cost = sum_{i=1}^j b_i + sum_{i=1}^k a_i - sum_{i=1}^k b_i + b_k
    # Cost = sum(b) + sum_{i=1}^k (a_i - b_i) + b_k
    # To minimize the maximum of these values, we should sort the tiles.
    # The optimal strategy is to sort tiles such that a_i - b_i is non-decreasing.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        tiles.append((a, b))
        idx += 2
        
    # Sort by a_i - b_i
    tiles.sort(key=lambda x: x[0] - x[1])
    
    # Calculate prefix sums of a and suffix sums of b
    a_vals = [t[0] for t in tiles]
    b_vals = [t[1] for t in tiles]
    
    pref_a = [0] * (j + 1)
    for i in range(j):
        pref_a[i+1] = pref_a[i] + a_vals[i]
        
    suff_b = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suff_b[i] = suff_b[i+1] + b_vals[i]
        
    # Max cost for a given ordering is max_{k=0 to j-1} (pref_a[k] + a_vals[k] + suff_b[k+1])
    # which is pref_a[k] + a_vals[k] + suff_b[k+1]
    ans = 0
    for k in range(j):
        current_cost = pref_a[k] + a_vals[k] + suff_b[k+1]
        if k == 0:
            ans = current_cost
        else:
            if current_cost > ans:
                ans = current_cost
                
    print(ans)

solve()
```
