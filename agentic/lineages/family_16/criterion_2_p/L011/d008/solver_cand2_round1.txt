```python
import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    rods = []
    idx = 1
    total_p2 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        rods.append((p1, p2))
        total_p2 += p2
        idx += 2
    
    # Let S be the sum of all p2.
    # Initial pressure = S.
    # When we process rod i (p1_i, p2_i):
    # Current pressure P_curr.
    # Peak = P_curr + p1_i.
    # New pressure = P_curr - p2_i.
    #
    # Let the sequence be r_1, r_2, ..., r_j.
    # Peak_1 = S + p1_1
    # Peak_2 = (S - p2_1) + p1_2
    # Peak_3 = (S - p2_1 - p2_2) + p1_3
    # ...
    # Peak_k = S - sum_{m=1}^{k-1} p2_m + p1_k
    #
    # We want to minimize max(Peak_k).
    # Note that Peak_k = S + p1_k - sum_{m=1}^{k-1} p2_m
    # Peak_k = S + p1_k + p2_k - sum_{m=1}^{k} p2_m
    #
    # This is equivalent to scheduling tasks to minimize maximum lateness.
    # Sort rods by p2 in descending order.
    
    rods.sort(key=lambda x: x[1], reverse=True)
    
    max_peak = 0
    current_pressure = total_p2
    for p1, p2 in rods:
        peak = current_pressure + p1
        if peak > max_peak:
            max_peak = peak
        current_pressure -= p2
        
    print(max_peak)

if __name__ == '__main__':
    solve()
```
