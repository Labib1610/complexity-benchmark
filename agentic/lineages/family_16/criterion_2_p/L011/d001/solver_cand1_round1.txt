```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    requests = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        requests.append((p1, p2))
        idx += 2
    
    # Let the sequence of requests be s_1, s_2, ..., s_j.
    # Let p1_i and p2_i be the durations for request s_i.
    # Activation interval i starts at A_i and ends at A_i + p1_i.
    # Resolution interval i starts at R_i and ends at R_i + p2_i.
    # Constraints:
    # 1. A_{i+1} >= A_i + p1_i
    # 2. R_{i+1} >= R_i + p2_i
    # 3. R_i >= A_i + p1_i
    #
    # To minimize the finish time R_j + p2_j:
    # A_1 = 0
    # A_i = sum_{k=1}^{i-1} p1_k
    # R_1 = A_1 + p1_1 = p1_1
    # R_i = max(R_{i-1} + p2_{i-1}, A_i + p1_i)
    #
    # This is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # To minimize the makespan, sort requests such that:
    # If p1_i < p2_i, put it early. If p1_i >= p2_i, put it late.
    # Specifically, sort by min(p1_i, p2_i), with p1_i < p2_i ascending,
    # and p1_i >= p2_i descending.
    
    group1 = []
    group2 = []
    for p1, p2 in requests:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_requests = group1 + group2
    
    current_a = 0
    current_r = 0
    
    for p1, p2 in sorted_requests:
        current_a += p1
        current_r = max(current_r, current_a) + p2
        
    print(current_r)

if __name__ == '__main__':
    solve()
```
