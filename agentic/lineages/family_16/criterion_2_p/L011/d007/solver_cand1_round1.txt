import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    pulses = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        pulses.append((p1, p2))
        idx += 2
    
    # Let the sequence of pulses be (p1_i, p2_i).
    # Let B_k be the Base Phase after k steps, P_k be the Peak Phase after k steps.
    # B_0 = 0, P_0 = 0.
    # B_k = B_{k-1} + p1_k
    # P_k = max(P_{k-1}, B_k) + p2_k
    #
    # Let S_k = sum_{i=1}^k p1_i. Then B_k = S_k.
    # P_k = max(P_{k-1}, S_k) + p2_k.
    #
    # Expanding this:
    # P_1 = S_1 + p2_1
    # P_2 = max(S_1 + p2_1, S_2) + p2_2
    # P_3 = max(max(S_1 + p2_1, S_2) + p2_2, S_3) + p2_3
    #     = max(S_1 + p2_1 + p2_2 + p2_3, S_2 + p2_2 + p2_3, S_3 + p2_3)
    #
    # In general, P_j = max_{1 <= k <= j} (S_k + sum_{i=k}^j p2_i)
    # P_j = max_{1 <= k <= j} (sum_{i=1}^k p1_i + sum_{i=k}^j p2_i)
    # P_j = sum_{i=1}^j p2_i + max_{1 <= k <= j} (sum_{i=1}^k p1_i - sum_{i=1}^{k-1} p2_i)
    # Let Q_k = p1_k + p2_k.
    # We want to minimize P_j.
    # Note that sum_{i=1}^j p2_i is constant regardless of order.
    # We need to minimize max_{1 <= k <= j} (S_k - sum_{i=1}^{k-1} p2_i).
    # This is equivalent to sorting pulses such that p2_i are in non-decreasing order.
    # If p2_i <= p2_{i+1}, the maximum is minimized.
    
    pulses.sort(key=lambda x: x[1])
    
    current_base = 0
    current_peak = 0
    for p1, p2 in pulses:
        current_base += p1
        if current_base > current_peak:
            current_peak = current_base
        current_peak += p2
        
    print(current_peak)

if __name__ == '__main__':
    solve()
