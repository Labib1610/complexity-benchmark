import sys

def solve():
    # The problem is equivalent to Johnson's Rule for two-machine scheduling.
    # We have j jobs, each with two tasks: Alpha (p1) and Beta (p2).
    # We want to minimize the makespan on two machines in series.
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 2. Sort Set 1 by p1 in ascending order.
    # 3. Sort Set 2 by p2 in descending order.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    # alpha_finish_time: time when current thread finishes Alpha Mutex
    # beta_finish_time: time when current thread finishes Beta Mutex
    alpha_finish_time = 0
    beta_finish_time = 0
    
    for p1, p2 in sequence:
        alpha_finish_time += p1
        # Beta starts at max(alpha_finish_time, previous_beta_finish_time)
        beta_start_time = max(alpha_finish_time, beta_finish_time)
        beta_finish_time = beta_start_time + p2
        
    print(beta_finish_time)

if __name__ == '__main__':
    solve()
