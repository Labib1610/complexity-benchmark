```python
import sys

def solve():
    # Let the sequence of checkpoints be (d_1, n_1), (d_2, n_2), ..., (d_j, n_j).
    # For a fixed switch point k, the total cost is:
    # C(k) = sum_{i=1}^k d_i + sum_{i=k}^j n_i
    # C(k) = sum_{i=1}^j n_i + sum_{i=1}^k (d_i - n_i) + n_k
    # Actually, the formula is:
    # C(k) = (d_1 + ... + d_k) + (n_k + n_{k+1} + ... + n_j)
    # Let D_k = sum_{i=1}^k d_i and N_k = sum_{i=k}^j n_i.
    # C(k) = D_k + N_k.
    # The adversary chooses k to maximize C(k). We want to minimize this maximum.
    # Let S_d = sum d_i and S_n = sum n_i.
    # C(k) = sum_{i=1}^k d_i + sum_{i=k}^j n_i
    # Note that C(k) = sum_{i=1}^j n_i + sum_{i=1}^k (d_i - n_i) + n_k
    # This is equivalent to:
    # C(k) = sum_{i=1}^j d_i + sum_{i=k+1}^j (n_i - d_i) + n_k
    
    # Let's re-examine:
    # C(k) = (d_1 + ... + d_k) + (n_k + ... + n_j)
    # Let x_i = d_i - n_i.
    # C(k) = sum_{i=1}^k (d_i - n_i) + sum_{i=1}^j n_i
    # C(k) = sum_{i=1}^k x_i + S_n
    # This is not quite right because n_k is included in both.
    # Correct: C(k) = sum_{i=1}^{k-1} d_i + (d_k + n_k) + sum_{i=k+1}^j n_i
    # C(k) = sum_{i=1}^j n_i + sum_{i=1}^k (d_i - n_i) + n_k
    # Wait, the formula is:
    # C(k) = sum_{i=1}^k d_i + sum_{i=k}^j n_i
    # Let f_i = d_i + n_i.
    # C(k) = sum_{i=1}^k d_i + sum_{i=k+1}^j n_i + (d_k + n_k)
    # This is equivalent to:
    # C(k) = sum_{i=1}^j n_i + sum_{i=1}^k (d_i - n_i) + n_k
    # Actually, the simplest way to view this:
    # C(k) = sum_{i=1}^k d_i + sum_{i=k}^j n_i
    # Let x_i = d_i - n_i.
    # C(k) = sum_{i=1}^k x_i + sum_{i=1}^j n_i
    # To minimize max_k C(k), we want to arrange x_i such that the prefix sums of x_i are minimized.
    # The prefix sums are minimized by sorting x_i in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    checkpoints = []
    idx = 1
    sum_n = 0
    for _ in range(j):
        d = int(input_data[idx])
        n = int(input_data[idx+1])
        checkpoints.append(d - n)
        sum_n += n
        idx += 2
    
    # Sort x_i = d_i - n_i in descending order to minimize prefix sums
    checkpoints.sort(reverse=True)
    
    max_toll = -float('inf')
    current_prefix_sum = 0
    for x in checkpoints:
        current_prefix_sum += x
        max_toll = max(max_toll, current_prefix_sum + sum_n)
        
    print(max_toll)

solve()
```
