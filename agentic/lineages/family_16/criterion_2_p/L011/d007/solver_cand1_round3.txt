import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (Structural and Cargo) and we want to minimize
    # the makespan (the time the last job finishes on the second machine).
    # Johnson's Rule:
    # 1. Divide the jobs into two sets:
    #    Set 1: jobs where s_i < c_i
    #    Set 2: jobs where s_i >= c_i
    # 2. Sort Set 1 in ascending order of s_i.
    # 3. Sort Set 2 in descending order of c_i.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        s = int(input_data[idx])
        c = int(input_data[idx+1])
        jobs.append((s, c))
        idx += 2
        
    set1 = []
    set2 = []
    for s, c in jobs:
        if s < c:
            set1.append((s, c))
        else:
            set2.append((s, c))
            
    # Sort set1 by s ascending
    set1.sort(key=lambda x: x[0])
    # Sort set2 by c descending
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    a_time = 0
    b_time = 0
    
    for s, c in sequence:
        a_time += s
        b_time = max(a_time, b_time) + c
        
    print(b_time)

if __name__ == '__main__':
    solve()
