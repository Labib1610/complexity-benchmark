```python
import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    blocks = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        blocks.append((p1, p2))
        idx += 2
    
    # Let the sequence of blocks be (u_1, d_1), (u_2, d_2), ..., (u_j, d_j).
    # If the marble drops at block k, the distance is:
    # D_k = (u_1 + ... + u_k) + (d_k + ... + d_j)
    # Let S_u = sum(u_i) and S_d = sum(d_i).
    # D_k = (u_1 + ... + u_k) + (S_d - (d_1 + ... + d_{k-1}))
    # D_k = S_d + (u_1 - d_1) + (u_2 - d_2) + ... + (u_k - d_k) + d_k
    # Actually, simpler: D_k = sum_{i=1}^k u_i + sum_{i=k}^j d_i
    # D_k = sum_{i=1}^j d_i + sum_{i=1}^k (u_i - d_i) + d_k
    # Let x_i = u_i - d_i.
    # D_k = S_d + sum_{i=1}^k x_i + d_k
    
    # We want to minimize max_k (S_d + sum_{i=1}^k x_i + d_k).
    # This is equivalent to minimizing max_k (sum_{i=1}^k x_i + d_k).
    # This is a classic scheduling problem.
    # If we have two adjacent blocks i and i+1, we want to order them to minimize
    # the local maximum.
    # Comparing (x_i + d_i, x_i + x_{i+1} + d_{i+1}) vs (x_{i+1} + d_{i+1}, x_{i+1} + x_i + d_i)
    # The condition for i before i+1 is:
    # max(d_i, x_i + d_{i+1}) < max(d_{i+1}, x_{i+1} + d_i)
    # Substituting x = u - d:
    # max(d_i, u_i - d_i + d_{i+1}) < max(d_{i+1}, u_{i+1} - d_{i+1} + d_i)
    # This is equivalent to sorting by min(u_i, d_{i+1})? No.
    # Let's use the standard Johnson's rule logic:
    # If u_i < d_i, put it early. If u_i > d_i, put it late.
    # Specifically, sort by:
    # If u_i < d_i, sort by u_i ascending.
    # If u_i >= d_i, sort by d_i descending.
    
    def sort_key(block):
        u, d = block
        if u < d:
            return (0, u)
        else:
            return (1, -d)
            
    blocks.sort(key=sort_key)
    
    # Calculate max distance for the sorted sequence
    s_d = sum(b[1] for b in blocks)
    max_dist = 0
    current_u_sum = 0
    current_d_sum = 0
    for u, d in blocks:
        current_u_sum += u
        # Distance if dropped at this block:
        # (sum of u up to here) + (sum of d from here to end)
        # sum of d from here to end = S_d - current_d_sum
        dist = current_u_sum + (s_d - current_d_sum)
        if dist > max_dist:
            max_dist = dist
        current_d_sum += d
        
    print(max_dist)

solve()
```
