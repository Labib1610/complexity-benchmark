```python
import sys

def solve():
    # The problem is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # We want to minimize the makespan (the time the last job finishes on the second machine).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where v1 < v2
    #    Set 2: jobs where v1 >= v2
    # 2. Sort Set 1 in ascending order of v1.
    # 3. Sort Set 2 in descending order of v2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        v1 = int(input_data[idx])
        v2 = int(input_data[idx+1])
        jobs.append((v1, v2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for v1, v2 in jobs:
        if v1 < v2:
            set1.append((v1, v2))
        else:
            set2.append((v1, v2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate the final completion time
    # Primary Validator finishes artifact i at time P_i = sum(v1_1...v1_i)
    # Secondary Validator finishes artifact i at time S_i = max(P_i, S_{i-1}) + v2_i
    
    current_p = 0
    current_s = 0
    
    for v1, v2 in sequence:
        current_p += v1
        if current_p > current_s:
            current_s = current_p + v2
        else:
            current_s += v2
            
    print(current_s)

if __name__ == '__main__':
    solve()
```
