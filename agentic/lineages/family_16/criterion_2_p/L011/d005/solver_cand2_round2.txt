```python
import sys

def solve():
    # The recurrence is:
    # X_k = X_{k-1} + x_k
    # Y_k = max(Y_{k-1}, X_k) + y_k
    # Let S_k = sum_{i=1}^k x_i. Then X_k = S_k.
    # Y_k = max(Y_{k-1}, S_k) + y_k
    # Y_k = max(Y_{k-1} + y_k, S_k + y_k)
    # Expanding this:
    # Y_j = max(S_1 + y_1 + y_2 + ... + y_j,
    #           S_2 + y_2 + y_3 + ... + y_j,
    #           ...,
    #           S_j + y_j)
    # Let T_k = sum_{i=k}^j y_i.
    # Y_j = max_{k=1}^j (S_k + T_k)
    # S_k = sum_{i=1}^k x_i
    # T_k = sum_{i=k}^j y_i
    # S_k + T_k = (sum_{i=1}^k x_i) + (sum_{i=k}^j y_i)
    #           = (sum_{i=1}^j x_i) + (sum_{i=k}^j y_i - sum_{i=k}^j x_i)
    #           = (sum_{i=1}^j x_i) + sum_{i=k}^j (y_i - x_i)
    # To minimize Y_j, we need to minimize max_{k=1}^j (sum_{i=k}^j (y_i - x_i)).
    # Let d_i = y_i - x_i. We want to minimize max_{k=1}^j (sum_{i=k}^j d_i).
    # This is equivalent to the problem of ordering elements to minimize the 
    # maximum suffix sum.
    # If we sort the vectors such that d_i are in non-decreasing order,
    # the maximum suffix sum is minimized.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vectors = []
    idx = 1
    total_x = 0
    for _ in range(j):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        vectors.append((x, y, y - x))
        total_x += x
        idx += 2
        
    # Sort by d_i = y_i - x_i ascending
    vectors.sort(key=lambda v: v[2])
    
    # Calculate max suffix sum of d_i
    # Y_j = total_x + max_{k=1}^j (sum_{i=k}^j d_i)
    max_suffix_sum = -float('inf')
    current_suffix_sum = 0
    for i in range(j - 1, -1, -1):
        current_suffix_sum += vectors[i][2]
        if current_suffix_sum > max_suffix_sum:
            max_suffix_sum = current_suffix_sum
            
    print(total_x + max_suffix_sum)

if __name__ == '__main__':
    solve()
```
