```python
import sys

def solve():
    # The stress at level k is:
    # S_k = (sum of p1 from 1 to k) + (sum of p2 from k to j)
    # Let P1_sum(k) = sum_{i=1}^k p1_i
    # Let P2_sum(k) = sum_{i=k}^j p2_i
    # Let Total_P1 = sum_{i=1}^j p1_i
    # Let Total_P2 = sum_{i=1}^j p2_i
    # S_k = P1_sum(k) + (Total_P2 - P2_sum(1 to k-1))
    # Let C_i = (p1_i, p2_i).
    # If we order containers as c_1, c_2, ..., c_j:
    # S_k = (p1_1 + ... + p1_k) + (p2_k + ... + p2_j)
    # S_k = (p1_1 + ... + p1_k + p2_k + ... + p2_j)
    # Note that S_k - S_{k-1} = p1_k - p2_{k-1}
    # This is a classic scheduling problem. To minimize the maximum of a set of values
    # where the values are prefix/suffix sums, we sort by p1 + p2.
    # Specifically, sorting by (p1 + p2) in ascending order is optimal.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    containers = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        containers.append((p1, p2))
        idx += 2
        
    # Sort by p1 + p2
    containers.sort(key=lambda x: x[0] + x[1])
    
    # Calculate prefix sums of p1 and suffix sums of p2
    p1_prefix = [0] * (j + 1)
    for i in range(j):
        p1_prefix[i+1] = p1_prefix[i] + containers[i][0]
        
    p2_suffix = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        p2_suffix[i] = p2_suffix[i+1] + containers[i][1]
        
    max_stress = 0
    for k in range(1, j + 1):
        # Level k is index k-1
        stress = p1_prefix[k] + p2_suffix[k-1]
        if stress > max_stress:
            max_stress = stress
            
    print(max_stress)

if __name__ == '__main__':
    solve()
```
