import sys

def solve():
    # The problem asks to minimize C = D_j + Y_j + sum(X_k).
    # Let's analyze the recurrence:
    # D_1 = 0
    # D_2 = max(0, D_1 + Y_1 - X_2) = max(0, Y_1 - X_2)
    # D_3 = max(0, D_2 + Y_2 - X_3) = max(0, Y_1 - X_2 + Y_2 - X_3, Y_2 - X_3)
    # In general, D_j = max(0, Y_{j-1} - X_j, Y_{j-1} + Y_{j-2} - X_j - X_{j-1}, ..., sum_{k=1}^{j-1} Y_k - sum_{k=2}^j X_k)
    # C = D_j + Y_j + sum_{k=1}^j X_k
    # C = max(sum_{k=1}^j X_k + Y_j, Y_{j-1} + Y_j + sum_{k=1}^{j-2} X_k, ..., sum_{k=1}^j Y_k)
    # This simplifies to C = max(sum_{k=1}^j X_k + Y_j, sum_{k=1}^j Y_k, sum_{k=1}^{j-1} Y_k + X_j, ...)
    # Actually, the expression simplifies to C = max(sum_{k=1}^j X_k + Y_j, sum_{k=1}^j Y_k).
    # Wait, let's re-evaluate.
    # D_i + Y_i = max(D_{i-1} + Y_{i-1} + Y_i - X_i, Y_i)
    # Let S_i = D_i + Y_i. Then S_i = max(S_{i-1} + Y_i - X_i, Y_i).
    # We want to minimize S_j + sum_{k=1}^{j-1} X_k.
    # This is equivalent to the classic scheduling problem:
    # To minimize the maximum completion time, we sort items.
    # If Y_i < X_i, we want to process them as early as possible.
    # If Y_i >= X_i, we want to process them as late as possible.
    # Sort items with Y_i < X_i by X_i ascending.
    # Sort items with Y_i >= X_i by Y_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    items = []
    idx = 1
    for _ in range(j):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        items.append((x, y))
        idx += 2
        
    group1 = [item for item in items if item[1] < item[0]]
    group2 = [item for item in items if item[1] >= item[0]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    d = 0
    sum_x = 0
    for i in range(j):
        x, y = sequence[i]
        if i > 0:
            d = max(d + prev_y - x, 0)
        sum_x += x
        prev_y = y
        
    print(d + sequence[-1][1] + sum_x)

solve()
