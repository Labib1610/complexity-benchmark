```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(n):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        vessels.append((a, b))
        idx += 2
    
    # Let A_i be Lock Alpha duration and B_i be Lock Beta duration.
    # Let S_i be the start time of vessel i in Lock Alpha.
    # S_1 = 0, S_i = S_{i-1} + A_{i-1} = sum_{j=1}^{i-1} A_j.
    # Let C_i be the completion time of vessel i in Lock Beta.
    # C_i = max(S_i + A_i + B_i, C_{i-1} + B_i)
    # This is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # To minimize the makespan, sort vessels such that:
    # Vessels with A_i < B_i come first (sorted by A_i ascending),
    # Vessels with A_i >= B_i come last (sorted by B_i descending).
    
    group1 = []
    group2 = []
    for a, b in vessels:
        if a < b:
            group1.append((a, b))
        else:
            group2.append((a, b))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_vessels = group1 + group2
    
    current_alpha_completion = 0
    current_beta_completion = 0
    
    for a, b in sorted_vessels:
        current_alpha_completion += a
        # Vessel starts Beta at max(current_alpha_completion, current_beta_completion)
        # Completion is start + b
        current_beta_completion = max(current_alpha_completion, current_beta_completion) + b
        
    print(current_beta_completion)

if __name__ == '__main__':
    solve()
```
