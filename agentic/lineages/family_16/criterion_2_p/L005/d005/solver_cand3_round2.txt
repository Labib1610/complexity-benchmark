```python
import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        tiles.append((t1, t2))
        idx += 2
    
    # Let the arrangement be (a1, b1), (a2, b2), ..., (aj, bj).
    # A path stepping down at tile k has cost:
    # sum(a_i for i=1 to k) + sum(b_i for i=k to j)
    # = sum(a_i for i=1 to j) + sum(b_i for i=k to j) - sum(a_i for i=k+1 to j)
    # Let S_a = sum(a_i), S_b = sum(b_i).
    # Cost(k) = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # Let P_k = sum_{i=1}^k a_i and Q_k = sum_{i=k}^j b_i.
    # We want to minimize max_{k=1..j} (P_k + Q_k).
    
    # Note that P_k + Q_k = (sum_{i=1}^j a_i) + (sum_{i=k}^j b_i - sum_{i=k+1}^j a_i)
    # Let T = sum(a_i). Cost(k) = T + b_k - (a_{k+1} + ... + a_j) + (b_{k+1} + ... + b_j)
    # Actually, simpler:
    # Cost(k) = (a_1 + ... + a_k) + (b_k + ... + b_j)
    # Let A_i = a_i, B_i = b_i.
    # We want to order tiles to minimize max_k (sum_{i=1}^k A_i + sum_{i=k}^j B_i).
    # This is equivalent to minimizing max_k (sum_{i=1}^{k-1} A_i + A_k + B_k + sum_{i=k+1}^j B_i).
    # Let S = sum(A_i) + sum(B_i).
    # Cost(k) = S - sum_{i=k+1}^j A_i - sum_{i=1}^{k-1} B_i.
    # To minimize the maximum cost, we want to maximize the minimum of (sum_{i=k+1}^j A_i + sum_{i=1}^{k-1} B_i).
    # This is a known problem: sort tiles by (a_i - b_i).
    # Specifically, sort such that a_i - b_i is non-decreasing.
    
    tiles.sort(key=lambda x: x[0] - x[1])
    
    a = [t[0] for t in tiles]
    b = [t[1] for t in tiles]
    
    # Prefix sums of a, suffix sums of b
    pref_a = [0] * (j + 1)
    for i in range(j):
        pref_a[i+1] = pref_a[i] + a[i]
        
    suff_b = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suff_b[i] = suff_b[i+1] + b[i]
        
    max_cost = 0
    for k in range(1, j + 1):
        # Path steps down at tile k (1-indexed)
        # Cost = sum(a[0...k-1]) + sum(b[k-1...j-1])
        current_cost = pref_a[k] + suff_b[k-1]
        if current_cost > max_cost:
            max_cost = current_cost
            
    print(max_cost)

if __name__ == '__main__':
    solve()
```
