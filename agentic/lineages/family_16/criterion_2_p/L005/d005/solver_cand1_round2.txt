```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    enchantments = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        enchantments.append((t1, t2))
        idx += 2
    
    # The problem is equivalent to scheduling jobs on a single machine
    # where each job has a processing time t1 and a delivery time t2.
    # This is a classic problem solvable by Smith's Rule (or Johnson's rule variant).
    # To minimize the completion time, we should sort by t2 in descending order.
    # If t2 values are equal, the order doesn't affect the final completion time.
    
    enchantments.sort(key=lambda x: x[1], reverse=True)
    
    current_time = 0
    max_finish_time = 0
    
    for t1, t2 in enchantments:
        current_time += t1
        # The aura finishes at current_time + t2, but it must wait for the 
        # previous aura to finish.
        finish_time = current_time + t2
        if finish_time > max_finish_time:
            max_finish_time = finish_time
            
    print(max_finish_time)

if __name__ == '__main__':
    solve()
```
