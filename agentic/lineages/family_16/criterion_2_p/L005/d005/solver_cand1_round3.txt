```python
import sys

def solve():
    # The total resonance is (sum of all t1) + (final instability level).
    # Let the sequence be (t1_1, t2_1), (t1_2, t2_2), ..., (t1_j, t2_j).
    # Let I_0 = 0.
    # I_k = max(0, I_{k-1} - t1_k) + t2_k.
    # We want to minimize I_j.
    # This is a classic scheduling problem.
    # If we have two adjacent artifacts (a1, b1) and (a2, b2):
    # Order 1: I_mid = max(0, I - a1), I_new = I_mid + b1, I_final = max(0, I_new - a2) + b2
    #          = max(0, max(0, I - a1) + b1 - a2) + b2
    # Order 2: I_mid = max(0, I - a2), I_new = I_mid + b2, I_final = max(0, I_new - a1) + b1
    #          = max(0, max(0, I - a2) + b2 - a1) + b1
    # Comparing the two, the optimal order is to put artifacts with t1 >= t2 first,
    # sorted by t1 descending, and artifacts with t1 < t2 last, sorted by t2 ascending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    artifacts = []
    sum_t1 = 0
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        artifacts.append((t1, t2))
        sum_t1 += t1
        idx += 2
        
    # Group 1: t1 >= t2, sort by t1 descending
    # Group 2: t1 < t2, sort by t2 ascending
    group1 = [a for a in artifacts if a[0] >= a[1]]
    group2 = [a for a in artifacts if a[0] < a[1]]
    
    group1.sort(key=lambda x: x[0], reverse=True)
    group2.sort(key=lambda x: x[1])
    
    sequence = group1 + group2
    
    instability = 0
    for t1, t2 in sequence:
        instability = max(0, instability - t1) + t2
        
    print(sum_t1 + instability)

if __name__ == '__main__':
    solve()
```
