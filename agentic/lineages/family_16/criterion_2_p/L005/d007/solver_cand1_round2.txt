```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    modules = []
    idx = 1
    total_t2 = 0
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        modules.append((t1, t2))
        total_t2 += t2
        idx += 2
    
    # Let f_k be the footprint at step k.
    # f_1 = t1_1 + sum(t2_i)
    # f_k = f_{k-1} + t1_k - t2_{k-1}
    # This implies f_k = sum_{i=1}^k (t1_i - t2_{i-1}) + sum_{i=1}^j t2_i
    # where t2_0 = 0.
    # f_k = sum_{i=1}^k t1_i - sum_{i=1}^{k-1} t2_i + sum_{i=1}^j t2_i
    # f_k = sum_{i=1}^k (t1_i - t2_i) + sum_{i=1}^j t2_i + t2_k
    # Let diff_i = t1_i - t2_i.
    # f_k = sum_{i=1}^k diff_i + total_t2 + t2_k
    
    # We want to minimize max(f_k).
    # This is a variation of the scheduling problem.
    # If we have two adjacent modules i and j, swapping them changes the peak.
    # Standard exchange argument:
    # If diff_i > 0 and diff_j <= 0, i should come before j.
    # If both diff > 0, sort by t2 ascending.
    # If both diff <= 0, sort by t1 descending.
    
    pos = []
    neg = []
    for t1, t2 in modules:
        if t1 - t2 >= 0:
            pos.append((t1, t2))
        else:
            neg.append((t1, t2))
            
    # Sort positive diffs by t2 ascending
    pos.sort(key=lambda x: x[1])
    # Sort negative diffs by t1 descending
    neg.sort(key=lambda x: x[0], reverse=True)
    
    order = pos + neg
    
    max_f = 0
    current_sum_diff = 0
    for t1, t2 in order:
        current_sum_diff += (t1 - t2)
        f_k = current_sum_diff + total_t2 + t2
        if f_k > max_f:
            max_f = f_k
            
    print(max_f)

if __name__ == '__main__':
    solve()
```
