import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    processors = []
    for _ in range(j):
        t1 = int(next(it))
        t2 = int(next(it))
        processors.append((t1, t2))
    
    # Sort by (t1 - t2) descending
    processors.sort(key=lambda x: x[0] - x[1], reverse=True)
    
    # Compute prefix sums
    prefix_t1 = [0] * (j + 1)
    prefix_t2 = [0] * (j + 1)
    for i in range(j):
        prefix_t1[i + 1] = prefix_t1[i] + processors[i][0]
        prefix_t2[i + 1] = prefix_t2[i] + processors[i][1]
    
    total_t2 = prefix_t2[j]
    
    max_spike = 0
    for k in range(1, j + 1):
        spike = total_t2 + prefix_t1[k] - prefix_t2[k - 1]
        if spike > max_spike:
            max_spike = spike
    
    print(max_spike)

if __name__ == "__main__":
    solve()
