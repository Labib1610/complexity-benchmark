```python
import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        vessels.append((t1, t2))
        idx += 2
    
    # Let the sequence be v_1, v_2, ..., v_j.
    # Local stress at k: S_k = sum_{i=1}^k t1_i + sum_{i=k}^j t2_i
    # S_k = sum_{i=1}^j t1_i + sum_{i=k}^j t2_i - sum_{i=1}^{k-1} t1_i (Wait, this is wrong)
    # Let T1 = sum_{i=1}^j t1_i, T2 = sum_{i=1}^j t2_i
    # S_k = (sum_{i=1}^k t1_i) + (sum_{i=k}^j t2_i)
    # S_k = (sum_{i=1}^k t1_i) + (T2 - sum_{i=1}^{k-1} t2_i)
    # S_k = T2 + t1_k + sum_{i=1}^{k-1} (t1_i - t2_i)
    
    # Let d_i = t1_i - t2_i.
    # S_k = T2 + t1_k + sum_{i=1}^{k-1} d_i
    # To minimize max(S_k), we should sort the vessels.
    # If we have two adjacent vessels i and i+1, swapping them changes S_i and S_{i+1}.
    # Standard exchange argument: sort by t1 + t2.
    # Actually, the optimal order is to sort by t1 - t2 ascending.
    
    vessels.sort(key=lambda x: x[0] - x[1])
    
    t1_sum = 0
    t2_sum = sum(v[1] for v in vessels)
    
    # S_k = (sum_{i=1}^k t1_i) + (sum_{i=k}^j t2_i)
    # S_k = (sum_{i=1}^k t1_i) + (T2 - sum_{i=1}^{k-1} t2_i)
    
    current_t1 = 0
    current_t2_prefix = 0
    max_stress = 0
    
    for i in range(j):
        t1, t2 = vessels[i]
        current_t1 += t1
        # S_{i+1} = current_t1 + (T2 - current_t2_prefix)
        stress = current_t1 + (t2_sum - current_t2_prefix)
        if stress > max_stress:
            max_stress = stress
        current_t2_prefix += t2
        
    print(max_stress)

if __name__ == '__main__':
    solve()
```
