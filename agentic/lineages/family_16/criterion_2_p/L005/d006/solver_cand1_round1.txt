import sys

def solve():
    # The problem asks to minimize the maximum chain duration.
    # Let the sequence be v_1, v_2, ..., v_j.
    # Chain duration for pivot k is: sum_{i=1}^k t1_i + sum_{i=k}^j t2_i
    # This can be rewritten as: sum_{i=1}^j t2_i + sum_{i=1}^k (t1_i - t2_i) + t2_k
    # Actually, let's re-examine:
    # D_k = (t1_1 + ... + t1_k) + (t2_k + ... + t2_j)
    # D_k = (t1_1 + ... + t1_k + t2_k + ... + t2_j)
    # This is a classic scheduling problem known as Johnson's Rule.
    # To minimize the maximum completion time (makespan) in a 2-machine flow shop,
    # we sort vessels such that:
    # If t1_i < t2_i, they go in increasing order of t1.
    # If t1_i >= t2_i, they go in decreasing order of t2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        vessels.append((t1, t2))
        idx += 2
        
    group1 = [v for v in vessels if v[0] < v[1]]
    group2 = [v for v in vessels if v[0] >= v[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate the exposure rating for the optimal sequence
    # The exposure rating is max_{k=1..j} (sum_{i=1}^k t1_i + sum_{i=k}^j t2_i)
    
    # Let S1[k] = sum_{i=1}^k t1_i
    # Let S2[k] = sum_{i=k}^j t2_i
    # We want max(S1[k] + S2[k])
    
    n = len(sequence)
    t1_vals = [v[0] for v in sequence]
    t2_vals = [v[1] for v in sequence]
    
    s1 = 0
    s2 = sum(t2_vals)
    
    max_duration = 0
    for i in range(n):
        s1 += t1_vals[i]
        # Current duration for pivot i is s1 + s2
        if s1 + s2 > max_duration:
            max_duration = s1 + s2
        s2 -= t2_vals[i]
        
    print(max_duration)

if __name__ == '__main__':
    solve()
