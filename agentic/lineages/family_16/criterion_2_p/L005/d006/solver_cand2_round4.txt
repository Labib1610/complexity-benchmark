```python
import sys

def solve():
    # The pivot cost at index k is:
    # C(k) = sum_{i=1}^k t1_i + sum_{i=k}^j t2_i
    # C(k) = sum_{i=1}^j t1_i + sum_{i=k}^j t2_i - sum_{i=1}^{k-1} t1_i
    # Let S1 = sum(t1_i) and S2 = sum(t2_i).
    # C(k) = sum_{i=1}^k t1_i + sum_{i=k}^j t2_i
    # Note that C(k) = (sum_{i=1}^{k-1} t1_i + t1_k) + (t2_k + sum_{i=k+1}^j t2_i)
    # Let L_k = sum_{i=1}^{k-1} t1_i and R_k = sum_{i=k+1}^j t2_i.
    # C(k) = L_k + t1_k + t2_k + R_k.
    # Let T = sum(t1_i + t2_i).
    # C(k) = L_k + R_k + t1_k + t2_k.
    # Also, L_k + t1_k + R_k + t2_k = sum(t1) + sum(t2).
    # Actually, the formula is simpler:
    # C(k) = (sum_{i=1}^k t1_i) + (sum_{i=k}^j t2_i)
    # Let S1_k = sum_{i=1}^k t1_i and S2_k = sum_{i=k}^j t2_i.
    # C(k) = S1_k + S2_k.
    # We want to minimize max_k (S1_k + S2_k).
    # Consider two adjacent elements i and j.
    # If we swap them, the pivot costs for k < pos(i) and k > pos(j) don't change.
    # The pivot costs at pos(i) and pos(j) change.
    # It is a known result for this type of scheduling problem that sorting by
    # t1 - t2 is optimal. Specifically, sort such that t1_i - t2_i is increasing.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    elements = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        elements.append((t1, t2))
        idx += 2
    
    # Sort by t1 - t2
    elements.sort(key=lambda x: x[0] - x[1])
    
    # Calculate prefix sums of t1 and suffix sums of t2
    t1_list = [e[0] for e in elements]
    t2_list = [e[1] for e in elements]
    
    s1 = [0] * (j + 1)
    for i in range(j):
        s1[i+1] = s1[i] + t1_list[i]
        
    s2 = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        s2[i] = s2[i+1] + t2_list[i]
        
    # C(k) for k=1 to j is s1[k] + s2[k-1]
    # (using 0-based indexing for s2, where s2[k-1] is sum from k-1 to j-1)
    max_cost = 0
    for k in range(1, j + 1):
        cost = s1[k] + s2[k-1]
        if cost > max_cost:
            max_cost = cost
            
    print(max_cost)

solve()
```
