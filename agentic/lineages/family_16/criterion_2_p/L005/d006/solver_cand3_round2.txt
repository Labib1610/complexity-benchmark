```python
import sys

def solve():
    # Let the sequence of signal zones be (a_1, b_1), (a_2, b_2), ..., (a_n, b_n).
    # Let A_i be the end coordinate of the Alpha installation of zone i.
    # A_i = sum_{k=1}^i a_k.
    # Let B_i be the end coordinate of the Beta installation of zone i.
    # B_0 = 0
    # B_i = max(A_i, B_{i-1}) + b_i
    # We want to minimize B_n.
    # B_n = max(A_n + b_n, B_{n-1} + b_n)
    # B_n = max(A_n + b_n, A_{n-1} + b_{n-1} + b_n, A_{n-2} + b_{n-2} + b_{n-1} + b_n, ..., A_1 + b_1 + ... + b_n)
    # B_n = max_{1 <= i <= n} (A_i + sum_{k=i}^n b_k)
    # B_n = max_{1 <= i <= n} (sum_{k=1}^i a_k + sum_{k=i}^n b_k)
    # B_n = sum_{k=1}^n b_k + max_{1 <= i <= n} (sum_{k=1}^i a_k - sum_{k=1}^{i-1} b_k)
    # Let S_a(i) = sum_{k=1}^i a_k and S_b(i) = sum_{k=1}^i b_k.
    # B_n = S_b(n) + max_{1 <= i <= n} (S_a(i) - S_b(i-1))
    # To minimize B_n, we need to order the zones.
    # This is a classic scheduling problem. If we have two adjacent zones (a1, b1) and (a2, b2):
    # Order 1: (a1, b1), (a2, b2) -> max(a1+b1+b2, a1+a2+b2)
    # Order 2: (a2, b2), (a1, b1) -> max(a2+b2+b1, a2+a1+b1)
    # We prefer Order 1 if max(a1+b1+b2, a1+a2+b2) < max(a2+b2+b1, a2+a1+b1).
    # This simplifies to comparing min(a1, b2) and min(a2, b1).
    # Specifically, sort by: if a_i < b_i, put it earlier (ascending a_i).
    # If a_i >= b_i, put it later (descending b_i).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    zones = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        zones.append((t1, t2))
        idx += 2
        
    # Johnson's rule logic:
    # Group 1: a_i < b_i, sort by a_i ascending
    # Group 2: a_i >= b_i, sort by b_i descending
    group1 = []
    group2 = []
    for a, b in zones:
        if a < b:
            group1.append((a, b))
        else:
            group2.append((a, b))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_zones = group1 + group2
    
    current_a = 0
    current_b = 0
    for a, b in sorted_zones:
        current_a += a
        current_b = max(current_a, current_b) + b
        
    print(current_b)

solve()
```
