```python
import sys

def solve():
    # The total time for a sequence of zones (t1_1, t2_1), ..., (t1_j, t2_j)
    # if we drop down at zone k is:
    # Sum_{i=1}^k t1_i + Sum_{i=k}^j t2_i
    # = Sum_{i=1}^j t1_i + Sum_{i=k}^j (t2_i - t1_i)
    # Let d_i = t2_i - t1_i.
    # Total time = Sum(t1) + Sum_{i=k}^j d_i.
    # The rival chooses k to maximize Sum_{i=k}^j d_i.
    # Let S_k = Sum_{i=k}^j d_i. We want to minimize max_{1 <= k <= j} S_k.
    # Note that S_{j+1} = 0.
    # S_k = d_k + S_{k+1}.
    # We want to arrange the zones to minimize max(S_1, S_2, ..., S_j).
    # This is equivalent to the problem of ordering items to minimize the 
    # maximum suffix sum.
    # If we have two adjacent zones (a1, a2) and (b1, b2) with differences 
    # da = a2-a1 and db = b2-b1:
    # Order (a, b): max(da + db, db)
    # Order (b, a): max(db + da, da)
    # We prefer (a, b) if max(da+db, db) < max(db+da, da).
    # This is a classic sorting problem.
    # If da >= 0 and db >= 0: max(da+db, db) = da+db, max(db+da, da) = da+db. Equal.
    # If da < 0 and db < 0: max(da+db, db) = db, max(db+da, da) = da. 
    #   We want smaller max, so we want max(db, da) to be smaller.
    #   If db < da, then max is da. If da < db, then max is db.
    #   So we want the larger (less negative) one to be at the end.
    #   Sort by d ascending.
    # If da >= 0 and db < 0: max(da+db, db) = da+db, max(db+da, da) = da.
    #   Since da+db < da, (a, b) is better.
    # If da < 0 and db >= 0: max(da+db, db) = db, max(db+da, da) = da+db.
    #   Since db < da+db, (b, a) is better.
    
    # Strategy:
    # 1. Put all zones with d_i >= 0 first, sorted by d_i ascending.
    # 2. Put all zones with d_i < 0 last, sorted by d_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    zones = []
    idx = 1
    sum_t1 = 0
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        sum_t1 += t1
        zones.append(t2 - t1)
        idx += 2
        
    pos = sorted([d for d in zones if d >= 0])
    neg = sorted([d for d in zones if d < 0], reverse=True)
    
    ordered_d = pos + neg
    
    # Now calculate max suffix sum
    max_suffix = -float('inf')
    current_suffix = 0
    for d in reversed(ordered_d):
        current_suffix += d
        if current_suffix > max_suffix:
            max_suffix = current_suffix
            
    print(sum_t1 + max_suffix)

solve()
```
