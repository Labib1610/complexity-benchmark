import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    pavilions = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        pavilions.append((t1, t2))
        idx += 2
    
    # Let S1 be the sum of all t1.
    # The Indoor sections end at S1.
    # Let the sequence of pavilions be P_1, ..., P_j.
    # Let T1_k and T2_k be the lengths of pavilion P_k.
    # Let E_k be the end coordinate of the Outdoor section of P_k.
    # E_1 = T1_1 + T2_1
    # E_k = max(E_{k-1}, sum_{i=1}^k T1_i) + T2_k
    # This recurrence can be rewritten:
    # E_k = max(E_{k-1} + T2_k, sum_{i=1}^k T1_i + T2_k)
    # By induction, E_j = max_{1 <= k <= j} (sum_{i=k}^j T2_i + sum_{i=1}^k T1_i)
    # E_j = sum_{i=1}^j T1_i + max_{1 <= k <= j} (sum_{i=k}^j T2_i - sum_{i=k}^j T1_i)
    # Let D_i = T2_i - T1_i.
    # E_j = sum T1_i + max_{1 <= k <= j} (sum_{i=k}^j D_i)
    # To minimize E_j, we need to minimize the maximum suffix sum of D_i.
    # This is equivalent to arranging D_i such that the maximum suffix sum is minimized.
    # If we sort D_i in ascending order, the maximum suffix sum is minimized.
    
    s1 = sum(p[0] for p in pavilions)
    d = [p[1] - p[0] for p in pavilions]
    d.sort()
    
    max_suffix_sum = 0
    current_suffix_sum = 0
    for val in reversed(d):
        current_suffix_sum += val
        if current_suffix_sum > max_suffix_sum:
            max_suffix_sum = current_suffix_sum
            
    print(s1 + max_suffix_sum)

if __name__ == '__main__':
    solve()
