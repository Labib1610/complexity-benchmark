```python
import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        vessels.append((t1, t2))
        idx += 2
    
    # Let the sequence of vessels be p_1, p_2, ..., p_j.
    # Let t1_i be navigation time and t2_i be mooring time for vessel i.
    # Navigation finish time for vessel p_k:
    # N_k = sum_{i=1}^k t1_{p_i}
    # Mooring finish time for vessel p_k:
    # M_k = max(N_k, M_{k-1}) + t2_{p_k}
    # This is equivalent to Johnson's Rule for a 2-machine flow shop problem.
    # To minimize the makespan, sort vessels:
    # 1. Vessels with t1 < t2 in increasing order of t1.
    # 2. Vessels with t1 >= t2 in decreasing order of t2.
    
    group1 = []
    group2 = []
    for t1, t2 in vessels:
        if t1 < t2:
            group1.append((t1, t2))
        else:
            group2.append((t1, t2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    current_n = 0
    current_m = 0
    for t1, t2 in sequence:
        current_n += t1
        current_m = max(current_n, current_m) + t2
        
    print(current_m)

if __name__ == '__main__':
    solve()
```
