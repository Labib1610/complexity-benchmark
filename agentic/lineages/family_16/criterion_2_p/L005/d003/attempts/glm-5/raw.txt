```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    vessels = []
    for _ in range(j):
        t1 = int(next(it))
        t2 = int(next(it))
        vessels.append((t1, t2))
    
    # Sort by (max(t1, t2), t1, t2) to handle the scheduling optimally
    vessels.sort(key=lambda x: (max(x[0], x[1]), x[0], x[1]))
    
    prev_bow = 0
    prev_stern = 0
    
    for t1, t2 in vessels:
        curr_bow = prev_bow + t1
        curr_stern = max(curr_bow, prev_stern) + t2
        prev_bow = curr_bow
        prev_stern = curr_stern
    
    print(prev_stern)

if __name__ == "__main__":
    solve()
```
