```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        vessels.append((t1, t2))
        idx += 2
    
    # Let the sequence be v_1, v_2, ..., v_j.
    # Pivot cost for k: sum_{i=1}^k t1_i + sum_{i=k}^j t2_i
    # = sum_{i=1}^j t2_i + sum_{i=1}^k (t1_i - t2_i)
    # Let S = sum_{i=1}^j t2_i.
    # Pivot cost for k = S + sum_{i=1}^k (t1_i - t2_i).
    # Let d_i = t1_i - t2_i.
    # Pivot cost for k = S + prefix_sum_d[k].
    # We want to minimize max_{k=1..j} (S + prefix_sum_d[k]).
    # This is equivalent to minimizing max_{k=1..j} (prefix_sum_d[k]).
    # To minimize the maximum prefix sum, we should sort the vessels such that
    # the prefix sums are as small as possible.
    # This is a classic greedy problem: sort by d_i.
    # If we sort d_i in ascending order, the prefix sums will be minimized.
    
    d = [v[0] - v[1] for v in vessels]
    d.sort()
    
    s = sum(v[1] for v in vessels)
    
    max_pivot_cost = -float('inf')
    current_prefix_sum = 0
    for val in d:
        current_prefix_sum += val
        if s + current_prefix_sum > max_pivot_cost:
            max_pivot_cost = s + current_prefix_sum
            
    print(max_pivot_cost)

if __name__ == '__main__':
    solve()
```
