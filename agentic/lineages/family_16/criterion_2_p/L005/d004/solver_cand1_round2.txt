```python
import sys

def solve():
    # The problem asks to minimize the maximum path cost.
    # Let the sequence of vessels be (b_1, s_1), (b_2, s_2), ..., (b_j, s_j).
    # A path moving down at vessel k has cost:
    # sum(b_1...b_k) + sum(s_k...s_j)
    # This can be rewritten as:
    # sum(b_1...b_j) + sum(s_1...s_j) - (sum(b_1...b_j) - sum(b_1...b_k)) + sum(s_k...s_j)
    # Actually, simpler:
    # Cost_k = (sum_{i=1}^k b_i) + (sum_{i=k}^j s_i)
    # Let S_b = sum(b_i) and S_s = sum(s_i).
    # Cost_k = sum_{i=1}^k b_i + sum_{i=k}^j s_i
    # Note that Cost_k = (sum_{i=1}^j b_i) + (sum_{i=1}^j s_i) - (sum_{i=k+1}^j b_i) - (sum_{i=1}^{k-1} s_i)
    # This is not quite right. Let's re-evaluate:
    # Cost_k = (b_1 + ... + b_k) + (s_k + ... + s_j)
    # Let P_k = sum_{i=1}^k b_i and Q_k = sum_{i=k}^j s_i.
    # We want to minimize max_k (P_k + Q_k).
    # Note that P_k + Q_k = (b_1 + ... + b_k) + (s_k + ... + s_j)
    # = (b_1 + ... + b_j) + (s_1 + ... + s_j) - (b_{k+1} + ... + b_j) - (s_1 + ... + s_{k-1})
    # Let T = sum(b_i) + sum(s_i).
    # Cost_k = T - (sum_{i=k+1}^j b_i + sum_{i=1}^{k-1} s_i)
    # To minimize the maximum cost, we want to maximize the minimum of (sum_{i=k+1}^j b_i + sum_{i=1}^{k-1} s_i).
    # This is equivalent to arranging the vessels such that the "gap" is as large as possible.
    # Actually, the optimal strategy is to sort the vessels such that b_i - s_i is non-decreasing.
    # Let d_i = b_i - s_i. If we sort by d_i, the maximum path cost is minimized.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        b = int(input_data[idx])
        s = int(input_data[idx+1])
        vessels.append((b, s))
        idx += 2
    
    # Sort by b_i - s_i
    vessels.sort(key=lambda x: x[0] - x[1])
    
    # Calculate prefix sums of b and suffix sums of s
    b_vals = [v[0] for v in vessels]
    s_vals = [v[1] for v in vessels]
    
    prefix_b = [0] * (j + 1)
    for i in range(j):
        prefix_b[i+1] = prefix_b[i] + b_vals[i]
        
    suffix_s = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suffix_s[i] = suffix_s[i+1] + s_vals[i]
        
    max_cost = 0
    for k in range(j):
        # Path moving down at vessel k (0-indexed)
        # Cost = sum(b_0...b_k) + sum(s_k...s_{j-1})
        cost = prefix_b[k+1] + suffix_s[k]
        if cost > max_cost:
            max_cost = cost
            
    print(max_cost)

solve()
```
