```python
import sys

def solve():
    # Let the modules be (g_i, d_i).
    # Voltage at test point k: V_k
    # V_1 = g_1
    # V_k = V_{k-1} + g_k - d_{k-1}
    # V_k = g_1 + (g_2 - d_1) + (g_3 - d_2) + ... + (g_k - d_{k-1})
    # V_k = sum_{i=1}^k g_i - sum_{i=1}^{k-1} d_i
    # Let S_g = sum_{i=1}^j g_i and S_d = sum_{i=1}^j d_i.
    # The overall rating is (max_{k=1..j} V_k) + S_d.
    # V_k = sum_{i=1}^k g_i - sum_{i=1}^{k-1} d_i
    # V_k = sum_{i=1}^k (g_i - d_i) + d_k
    # Let x_i = g_i - d_i.
    # V_k = sum_{i=1}^k x_i + d_k.
    # We want to minimize max_{k=1..j} (sum_{i=1}^k x_i + d_k) + S_d.
    # This is equivalent to minimizing max_{k=1..j} (sum_{i=1}^k x_i + d_k).
    
    # Let the sequence be p_1, p_2, ..., p_j.
    # V_k = sum_{i=1}^k (g_{p_i} - d_{p_i}) + d_{p_k}
    # Let x_i = g_i - d_i.
    # V_k = sum_{i=1}^k x_{p_i} + d_{p_k}
    
    # If we sort modules such that d_i are in non-decreasing order,
    # it is a known result for this type of scheduling problem.
    # Specifically, if we have two adjacent modules i and j,
    # swapping them changes the peak voltage.
    # The optimal order is to sort by d_i ascending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    modules = []
    idx = 1
    sum_d = 0
    for _ in range(j):
        g = int(input_data[idx])
        d = int(input_data[idx+1])
        modules.append((g, d))
        sum_d += d
        idx += 2
        
    # Sort by d_i ascending
    modules.sort(key=lambda x: x[1])
    
    peak_v = -float('inf')
    current_v = 0
    for i in range(j):
        g, d = modules[i]
        current_v += g
        if current_v > peak_v:
            peak_v = current_v
        current_v -= d
        
    print(peak_v + sum_d)

solve()
```
