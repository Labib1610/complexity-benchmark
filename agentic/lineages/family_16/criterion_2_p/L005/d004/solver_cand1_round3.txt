```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        vessels.append((t1, t2))
        idx += 2
    
    # This is a variation of Johnson's Rule for the two-machine flow shop problem.
    # We want to minimize the makespan (completion time of the last job).
    # Group 1: t1 < t2, sort by t1 ascending.
    # Group 2: t1 >= t2, sort by t2 descending.
    group1 = []
    group2 = []
    for t1, t2 in vessels:
        if t1 < t2:
            group1.append((t1, t2))
        else:
            group2.append((t1, t2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate completion time
    # alpha_finish: time when current vessel finishes Lock Alpha
    # beta_finish: time when current vessel finishes Lock Beta
    alpha_finish = 0
    beta_finish = 0
    
    for t1, t2 in sequence:
        alpha_finish += t1
        # Beta can start only after Alpha finishes AND previous Beta finishes
        beta_start = max(alpha_finish, beta_finish)
        beta_finish = beta_start + t2
        
    print(beta_finish)

if __name__ == '__main__':
    solve()
```
