```python
import sys

def solve():
    # This is a classic application of Johnson's Rule for the 2-machine flow shop problem.
    # The goal is to minimize the makespan (the time the last job finishes).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where checkpoint duration (p1) < scanning duration (p2)
    #    Set B: jobs where checkpoint duration (p1) >= scanning duration (p2)
    # 2. Sort Set A in ascending order of p1.
    # 3. Sort Set B in descending order of p2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for job in jobs:
        if job[0] < job[1]:
            set_a.append(job)
        else:
            set_b.append(job)
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate the completion time
    checkpoint_time = 0
    scanning_time = 0
    
    for p1, p2 in sequence:
        checkpoint_time += p1
        # The scanning tunnel can start only after the checkpoint is done
        # AND the previous scanning job is finished.
        if checkpoint_time > scanning_time:
            scanning_time = checkpoint_time + p2
        else:
            scanning_time += p2
            
    print(scanning_time)

if __name__ == '__main__':
    solve()
```
