```python
import sys

def solve():
    # The problem is a variation of Johnson's Rule for two-machine flow shop scheduling.
    # We want to minimize the makespan (total time to complete all jobs).
    # Let A_i be the time for ship i at Lock Alpha, B_i be the time at Lock Beta.
    # Johnson's Rule:
    # 1. Divide ships into two sets:
    #    Set 1: ships where A_i < B_i
    #    Set 2: ships where A_i >= B_i
    # 2. Sort Set 1 in ascending order of A_i.
    # 3. Sort Set 2 in descending order of B_i.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    ships = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        ships.append((a, b))
        idx += 2
        
    set1 = []
    set2 = []
    for a, b in ships:
        if a < b:
            set1.append((a, b))
        else:
            set2.append((a, b))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    # alpha_finish[i] = sum(A_0...A_i)
    # beta_finish[i] = max(alpha_finish[i], beta_finish[i-1]) + B_i
    
    current_alpha_time = 0
    current_beta_time = 0
    
    for a, b in sequence:
        current_alpha_time += a
        current_beta_time = max(current_alpha_time, current_beta_time) + b
        
    print(current_beta_time)

if __name__ == '__main__':
    solve()
```
