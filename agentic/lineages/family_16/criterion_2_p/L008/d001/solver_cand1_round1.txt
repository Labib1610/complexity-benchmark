```python
import sys

def solve():
    # The total distance for a chosen order and switch point k is:
    # Sum_{i=1 to k} p1_i + Sum_{i=k to j} p2_i
    # = Sum_{i=1 to j} p2_i + Sum_{i=1 to k} (p1_i - p2_i)
    # Let d_i = p1_i - p2_i.
    # The total distance is (Sum p2) + (Sum_{i=1 to k} d_i).
    # To minimize the maximum of this over all k, we want to order the biomes
    # such that the prefix sums of d_i are as small as possible.
    # This is achieved by sorting the biomes by d_i in ascending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    biomes = []
    idx = 1
    sum_p2 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        biomes.append((p1, p2, p1 - p2))
        sum_p2 += p2
        idx += 2
        
    # Sort by d_i = p1 - p2 ascending
    biomes.sort(key=lambda x: x[2])
    
    # Now calculate the prefix sums of d_i
    # The maximum value will be sum_p2 + max(0, prefix_sum_1, prefix_sum_2, ..., prefix_sum_j)
    # Since p1, p2 >= 1, the prefix sums will start at d_1 and end at sum(d_i).
    # The inspector chooses k from 1 to j.
    # Total distance for k is sum_p2 + sum_{i=1 to k} d_i.
    
    max_val = -float('inf')
    current_prefix_sum = 0
    for i in range(j):
        current_prefix_sum += biomes[i][2]
        if current_prefix_sum > max_val:
            max_val = current_prefix_sum
            
    print(sum_p2 + max_val)

if __name__ == '__main__':
    solve()
```
