import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    sectors = []
    for _ in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        sectors.append((p1, p2))
    
    # Johnson's rule: sort by key (min(p1, p2), p1, p2)
    # Sectors with p1 <= p2 sorted by p1 ascending
    # Sectors with p1 > p2 sorted by p2 descending
    sectors.sort(key=lambda x: (min(x[0], x[1]), x[0], x[1]))
    
    # Reorder: first group (p1 <= p2) in sorted order, then second group (p1 > p2) in sorted order
    group1 = [s for s in sectors if s[0] <= s[1]]
    group2 = [s for s in sectors if s[0] > s[1]]
    ordered = group1 + group2
    
    # Simulate the flow shop
    time_radar = 0
    time_sampling = 0
    for p1, p2 in ordered:
        time_radar += p1
        # Sampling starts when both: radar finishes this sector, and sampling finishes previous
        time_sampling = max(time_radar, time_sampling) + p2
    
    print(time_sampling)

if __name__ == "__main__":
    solve()
