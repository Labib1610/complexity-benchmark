import sys

def solve():
    # Let P_i be the primary tracker and S_i be the secondary tracker after step i.
    # Let p1_i, p2_i be the coefficients for the i-th sector in the sequence.
    # P_i = P_{i-1} + p1_i
    # S_i = max(S_{i-1} - p1_i, P_{i-1} + p1_i) + 2 * p2_i
    # Let P_0 = 0, S_0 = 0.
    # P_n = sum(p1_i)
    # S_n = max(S_{n-1} - p1_n, P_{n-1} + p1_n) + 2 * p2_n
    #
    # Let's analyze the recurrence for S_i:
    # S_i = max(S_{i-1} - p1_i, P_{i-1} + p1_i) + 2 * p2_i
    # S_i = max(S_{i-1} - p1_i + 2*p2_i, P_{i-1} + p1_i + 2*p2_i)
    #
    # This looks like we want to minimize S_n.
    # Let's test the order. Suppose we have two sectors (a1, a2) and (b1, b2).
    # Order (a, b):
    # S_a = max(-a1, a1) + 2*a2 = a1 + 2*a2
    # P_a = a1
    # S_b = max(S_a - b1, P_a + b1) + 2*b2 = max(a1 + 2*a2 - b1, a1 + b1) + 2*b2
    # S_b = a1 + 2*b2 + max(2*a2 - b1, b1)
    #
    # Order (b, a):
    # S_a = b1 + 2*a2 + max(2*b2 - a1, a1)
    #
    # Comparing a1 + 2*b2 + max(2*a2 - b1, b1) vs b1 + 2*a2 + max(2*b2 - a1, a1)
    # This simplifies to comparing max(a1+2*b2+2*a2-b1, a1+2*b2+b1) vs max(b1+2*a2+2*b2-a1, b1+2*a2+a1)
    # Which is max(a1-b1+2*a2+2*b2, a1+b1+2*b2) vs max(b1-a1+2*a2+2*b2, b1+a1+2*a2)
    # Subtracting (a1+b1+2*a2+2*b2) from both sides:
    # max(-2*b1, -2*a1) vs max(-2*a1, -2*b1)
    # This suggests the order doesn't matter for the max term?
    # Wait, let's re-evaluate.
    # The metric is (P_n + S_n) / 2. Since P_n is constant, we minimize S_n.
    # S_n = max(S_{n-1} - p1_n, P_{n-1} + p1_n) + 2 * p2_n
    # S_n = max(S_{n-1} - p1_n + 2*p2_n, P_{n-1} + p1_n + 2*p2_n)
    #
    # Actually, the optimal strategy is to sort by p2 descending.
    # Let's check Example 1: (3,2), (2,4), (1,3). Sorted by p2 desc: (2,4), (1,3), (3,2).
    # P=0, S=0
    # (2,4): S = max(-2, 2) + 8 = 10, P = 2
    # (1,3): S = max(10-1, 2+1) + 6 = 9+6 = 15, P = 3
    # (3,2): S = max(15-3, 3+3) + 4 = 12+4 = 16, P = 6. Metric = (6+16)/2 = 11.
    # Example 1 says 10 is possible with (1,3), (2,4), (3,2).
    # (1,3): S = max(-1, 1) + 6 = 7, P = 1
    # (2,4): S = max(7-2, 1+2) + 8 = 5+8 = 13, P = 3
    # (3,2): S = max(13-3, 3+3) + 4 = 10+4 = 14, P = 6. Metric = (6+14)/2 = 10.
    # Sorting by p2 ascending: (3,2), (1,3), (2,4).
    # (3,2): S = 3+4=7, P=3
    # (1,3): S = max(7-1, 3+1)+6 = 6+6=12, P=4
    # (2,4): S = max(12-2, 4+2)+8 = 10+8=18, P=6. Metric = 12.
    # Sorting by p1 ascending: (1,3), (2,4), (3,2). This is the optimal one!
    
    input_data = sys.stdin.read().split()
    if not input_data: return
    j = int(input_data[0])
    sectors = []
    idx = 1
    for _ in range(j):
        sectors.append((int(input_data[idx]), int(input_data[idx+1])))
        idx += 2
    
    sectors.sort()
    
    p_tracker = 0
    s_tracker = 0
    for p1, p2 in sectors:
        baseline = s_tracker - p1
        threshold = p_tracker + p1
        target = max(baseline, threshold)
        p_tracker += p1
        s_tracker = target + 2 * p2
        
    print((p_tracker + s_tracker) // 2)

solve()
