import sys

def solve():
    # Let C_i be the charge before step 1 of sector i.
    # Let C_{i+1} be the charge after step 3 of sector i.
    # C_{i+1} = C_i - min(p1_i, C_i) + p2_i
    # The cost increase for sector i is:
    # Delta_i = p1_i + C_{i+1} - C_i
    # Delta_i = p1_i + (C_i - min(p1_i, C_i) + p2_i) - C_i
    # Delta_i = p1_i + p2_i - min(p1_i, C_i)
    # Total Cost = sum(p1_i + p2_i - min(p1_i, C_i))
    # Total Cost = sum(p1_i + p2_i) - sum(min(p1_i, C_i))
    
    # To minimize the total cost, we must maximize sum(min(p1_i, C_i)).
    # C_1 = 0.
    # C_{i+1} = C_i - min(p1_i, C_i) + p2_i.
    # If p1_i <= C_i, then min(p1_i, C_i) = p1_i, so C_{i+1} = C_i - p1_i + p2_i.
    # If p1_i > C_i, then min(p1_i, C_i) = C_i, so C_{i+1} = p2_i.
    
    # This is a classic scheduling problem.
    # Sectors with p1_i <= p2_i should be processed first in increasing order of p1_i.
    # Sectors with p1_i > p2_i should be processed last in decreasing order of p2_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sectors = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        idx += 2
        
    group1 = [] # p1 <= p2
    group2 = [] # p1 > p2
    
    total_p1_p2 = 0
    for p1, p2 in sectors:
        total_p1_p2 += (p1 + p2)
        if p1 <= p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    current_charge = 0
    total_min = 0
    for p1, p2 in sequence:
        total_min += min(p1, current_charge)
        current_charge = current_charge - min(p1, current_charge) + p2
        
    print(total_p1_p2 - total_min)

if __name__ == '__main__':
    solve()
