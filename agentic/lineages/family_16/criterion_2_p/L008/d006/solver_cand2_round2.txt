import sys

def solve():
    # The problem asks to minimize the final accumulated cost.
    # Let the sequence of sectors be (p1_i, p2_i).
    # Let C_i be the battery charge before processing sector i.
    # C_1 = 0
    # C_{i+1} = C_i - min(p1_i, C_i) + p2_i
    # Cost_i = p1_i + p2_i - min(p1_i, C_i)
    # Total Cost = sum(p1_i + p2_i - min(p1_i, C_i))
    # Total Cost = sum(p1_i + p2_i) - sum(min(p1_i, C_i))
    # To minimize Total Cost, we must maximize sum(min(p1_i, C_i)).
    
    # Let's analyze the greedy strategy.
    # If p1_i <= p2_i, the sector is "net positive" or "neutral".
    # If p1_i > p2_i, the sector is "net negative".
    # It is optimal to process all sectors with p1_i <= p2_i first,
    # sorted by p1_i ascending.
    # Then process all sectors with p1_i > p2_i, sorted by p2_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sectors = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        idx += 2
        
    pos = []
    neg = []
    for p1, p2 in sectors:
        if p1 <= p2:
            pos.append((p1, p2))
        else:
            neg.append((p1, p2))
            
    # Sort positive: p1 ascending
    pos.sort(key=lambda x: x[0])
    # Sort negative: p2 descending
    neg.sort(key=lambda x: x[1], reverse=True)
    
    order = pos + neg
    
    total_cost = 0
    charge = 0
    for p1, p2 in order:
        allocated = min(p1, charge)
        total_cost += (p1 + p2 - allocated)
        charge = charge - allocated + p2
        
    print(total_cost)

if __name__ == '__main__':
    solve()
