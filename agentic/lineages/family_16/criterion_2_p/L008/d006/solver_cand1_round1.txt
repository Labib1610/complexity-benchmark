```python
import sys

def solve():
    # Let C_i be the battery charge at the start of step 1 for sector i.
    # Let C_{i+1} be the battery charge at the end of step 2 for sector i.
    # The cost increment for sector i is:
    # delta_i = p1_i + C_{i+1} - C_i
    # Total cost = sum(delta_i) = sum(p1_i) + sum(C_{i+1} - C_i)
    # Since C_1 = 0, the sum telescopes:
    # Total cost = sum(p1_i) + C_{j+1} - C_1 = sum(p1_i) + C_{j+1}
    #
    # C_{i+1} = max(0, C_i - p1_i) + p2_i
    # We want to minimize C_{j+1}.
    # Let's analyze the recurrence:
    # C_{i+1} = max(p2_i, C_i - p1_i + p2_i)
    # This is equivalent to C_{i+1} = max(p2_i, C_i - (p1_i - p2_i))
    # Let d_i = p1_i - p2_i. Then C_{i+1} = max(p2_i, C_i + d_i)
    #
    # If we have two sectors (p1_a, p2_a) and (p1_b, p2_b):
    # Order a, b: C_2 = max(p2_a, C_0 + d_a), C_3 = max(p2_b, C_2 + d_b)
    # C_3 = max(p2_b, max(p2_a, C_0 + d_a) + d_b)
    # C_3 = max(p2_b, p2_a + d_b, C_0 + d_a + d_b)
    #
    # Order b, a: C_3' = max(p2_a, p2_b + d_a, C_0 + d_b + d_a)
    # Comparing max(p2_b, p2_a + d_b) and max(p2_a, p2_b + d_a):
    # We prefer a, b if max(p2_b, p2_a + p1_b - p2_b) < max(p2_a, p2_b + p1_a - p2_a)
    # This simplifies to sorting by p1_i.
    # If p1_a < p1_b, then p2_a + p1_b - p2_b vs p2_b + p1_a - p2_a
    # Actually, the standard exchange argument for this specific recurrence
    # C_{i+1} = max(p2_i, C_i - p1_i + p2_i) shows that sorting by p1_i ascending
    # is optimal.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sectors = []
    idx = 1
    sum_p1 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        sum_p1 += p1
        idx += 2
    
    # Sort by p1 ascending
    sectors.sort(key=lambda x: x[0])
    
    current_charge = 0
    for p1, p2 in sectors:
        if current_charge < p1:
            current_charge = 0
        else:
            current_charge -= p1
        current_charge += p2
        
    print(sum_p1 + current_charge)

if __name__ == '__main__':
    solve()
```
