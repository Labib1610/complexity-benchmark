import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    anomalies = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        anomalies.append((p1, p2))
        idx += 2
    
    # The problem asks to assign tags 1 to j to minimize the completion time.
    # Let the order of processing be p1_1, p1_2, ..., p1_j and p2_1, p2_2, ..., p2_j.
    # The time an anomaly i finishes State 1 is T1_i = sum_{k=1}^i p1_k.
    # The time an anomaly i finishes State 3 is T3_i = max(T1_i + p2_i, T3_{i-1} + p2_i).
    # This is equivalent to the classic flow shop scheduling problem with 2 machines
    # where the first machine is the State 1 process and the second is State 3.
    # Johnson's Rule minimizes the makespan for a 2-machine flow shop.
    
    # Johnson's Rule:
    # 1. List all p1 and p2 values.
    # 2. Find the minimum value among all p1 and p2.
    # 3. If the minimum is a p1, place that job at the beginning of the sequence.
    # 4. If the minimum is a p2, place that job at the end of the sequence.
    # 5. Remove the job and repeat.
    
    # Implementation of Johnson's Rule:
    # Group 1: p1 < p2 (sort by p1 ascending)
    # Group 2: p1 >= p2 (sort by p2 descending)
    
    group1 = []
    group2 = []
    for p1, p2 in anomalies:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate completion time
    current_t1 = 0
    current_t3 = 0
    for p1, p2 in order:
        current_t1 += p1
        current_t3 = max(current_t1, current_t3) + p2
        
    print(current_t3)

if __name__ == '__main__':
    solve()
