import sys

def solve():
    # The problem asks to minimize: Total Idle Delay + Sum(p2)
    # Let the sequence be (p1_1, p2_1), (p1_2, p2_2), ..., (p1_j, p2_j)
    # Let B_i be the buffer after step 3 of sector i.
    # B_0 = 0
    # For sector i:
    #   temp = B_{i-1} - p1_i
    #   if temp < 0:
    #       delay_i = -temp
    #       B_i = p2_i
    #   else:
    #       delay_i = 0
    #       B_i = temp + p2_i
    # Total Delay = sum(delay_i)
    # Total Metric = sum(delay_i) + sum(p2_i)
    
    # Notice that if we process sectors with p2 >= p1 first, we build up the buffer.
    # If we process sectors with p2 < p1, we deplete the buffer.
    # To minimize delay, we should process sectors that increase the buffer (p2 >= p1)
    # as early as possible, and sectors that decrease the buffer (p2 < p1) as late as possible.
    
    # Let's split into two groups:
    # Group A: p2 >= p1 (buffer increases or stays same)
    # Group B: p2 < p1 (buffer decreases)
    # Within Group A, we want to process them in an order that keeps the buffer high.
    # Within Group B, we want to process them in an order that minimizes the delay.
    
    # Actually, the optimal strategy is:
    # 1. Sort Group A by p1 ascending.
    # 2. Sort Group B by p2 descending.
    # 3. Concatenate A then B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sectors = []
    idx = 1
    sum_p2 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        sum_p2 += p2
        idx += 2
        
    group_a = [s for s in sectors if s[1] >= s[0]]
    group_b = [s for s in sectors if s[1] < s[0]]
    
    group_a.sort(key=lambda x: x[0])
    group_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group_a + group_b
    
    total_delay = 0
    buffer = 0
    for p1, p2 in sequence:
        buffer -= p1
        if buffer < 0:
            total_delay += abs(buffer)
            buffer = 0
        buffer += p2
        
    print(total_delay + sum_p2)

if __name__ == '__main__':
    solve()
