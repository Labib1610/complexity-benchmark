```python
import sys
import heapq

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    sectors = []
    sum_p2 = 0
    for i in range(j):
        p1 = int(data[2*i + 1])
        p2 = int(data[2*i + 2])
        sectors.append((p1, p2))
        sum_p2 += p2
    
    # Sort sectors based on p1 - p2 in descending order
    # This greedy approach minimizes the idle delay
    sectors.sort(key=lambda x: x[0] - x[1], reverse=True)
    
    scanned_buffer = 0
    idle_delay = 0
    
    for p1, p2 in sectors:
        scanned_buffer -= p1
        if scanned_buffer < 0:
            idle_delay += (-scanned_buffer)
            scanned_buffer = 0
        scanned_buffer += p2
    
    result = idle_delay + sum_p2
    print(result)

if __name__ == "__main__":
    main()
```
