```python
import sys

def solve():
    # Let the sequence be (a1, b1), (a2, b2), ..., (aj, bj).
    # The cost for pivot k is C_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i.
    # C_k = sum_{i=1}^j b_i + sum_{i=1}^k (a_i - b_i) + b_k.
    # Let S = sum_{i=1}^j b_i.
    # C_k = S + sum_{i=1}^{k-1} (a_i - b_i) + a_k.
    # We want to minimize max_k C_k.
    # This is equivalent to minimizing max_k (sum_{i=1}^{k-1} (a_i - b_i) + a_k).
    # Let d_i = a_i - b_i.
    # We want to arrange (a_i, b_i) to minimize max(a_1, a_1+d_1+a_2, a_1+d_1+d_2+a_3, ...).
    # This is a classic scheduling problem. The optimal order is to sort by d_i.
    # Specifically, if we sort by d_i = a_i - b_i, we can check the peak cost.
    # Actually, the optimal strategy is to sort by a_i - b_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sectors = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        idx += 2
    
    # Sort by a_i - b_i
    sectors.sort(key=lambda x: x[0] - x[1])
    
    # Calculate peak cost for this sequence
    # C_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # Let prefix_a[k] = sum_{i=1}^k a_i
    # Let suffix_b[k] = sum_{i=k}^j b_i
    # C_k = prefix_a[k] + suffix_b[k]
    
    prefix_a = [0] * (j + 1)
    for i in range(j):
        prefix_a[i+1] = prefix_a[i] + sectors[i][0]
        
    suffix_b = [0] * (j + 2)
    for i in range(j, 0, -1):
        suffix_b[i] = suffix_b[i+1] + sectors[i-1][1]
        
    ans = 0
    for k in range(1, j + 1):
        ans = max(ans, prefix_a[k] + suffix_b[k])
        
    print(ans)

solve()
```
