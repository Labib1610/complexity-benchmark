```python
import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx]); p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
    
    # Johnson's rule: sort by min(p1, p2) ascending
    jobs.sort(key=lambda x: min(x[0], x[1]))
    
    time_a = 0
    time_b = 0
    for p1, p2 in jobs:
        time_a += p1
        if time_a > time_b:
            time_b = time_a + p2
        else:
            time_b += p2
    
    print(time_b)

solve()
```
