import sys

def solve():
    # The problem asks to minimize the total time: sum(p1) + final_backlog.
    # Let the sequence of sectors be (p1_i, p2_i).
    # Let B_i be the backlog after sector i.
    # B_0 = 0
    # B_i = max(0, B_{i-1} - p1_i) + p2_i
    # Total time = sum(p1_i) + B_j
    #
    # This is a classic scheduling problem. If we have two adjacent sectors 
    # (p1_a, p2_a) and (p1_b, p2_b), we want to order them to minimize the final backlog.
    # If we put 'a' then 'b':
    # B_a = p2_a
    # B_b = max(0, p2_a - p1_b) + p2_b
    # If we put 'b' then 'a':
    # B_b = p2_b
    # B_a = max(0, p2_b - p1_a) + p2_a
    #
    # We prefer 'a' then 'b' if max(0, p2_a - p1_b) + p2_b < max(0, p2_b - p1_a) + p2_a.
    # This is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # Sort sectors such that those with p1 < p2 come first (sorted by p1 ascending),
    # and those with p1 >= p2 come last (sorted by p2 descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sectors = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sectors.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    sum_p1 = 0
    for p1, p2 in sectors:
        sum_p1 += p1
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    itinerary = group1 + group2
    
    backlog = 0
    for p1, p2 in itinerary:
        backlog = max(0, backlog - p1) + p2
        
    print(sum_p1 + backlog)

if __name__ == '__main__':
    solve()
