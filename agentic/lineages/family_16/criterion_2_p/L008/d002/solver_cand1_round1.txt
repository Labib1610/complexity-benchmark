import sys

def solve():
    # The problem asks to minimize the completion time of two rovers.
    # Let p1_i be the mapping time and p2_i be the drilling time for sector i.
    # The Radar Rover processes sectors in order of priority (1 to j).
    # The Sampling Rover processes sectors in order of priority (1 to j)
    # as soon as they are mapped.
    # This is equivalent to Johnson's Rule for a 2-machine flow shop problem.
    # Johnson's Rule:
    # 1. List all jobs.
    # 2. Find the minimum processing time among all jobs on both machines.
    # 3. If the minimum is on machine 1 (Radar), place that job at the beginning.
    # 4. If the minimum is on machine 2 (Sampling), place that job at the end.
    # 5. Remove the job and repeat.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Johnson's Rule implementation
    group1 = [] # p1 < p2
    group2 = [] # p1 >= p2
    
    for i in range(j):
        if jobs[i][0] < jobs[i][1]:
            group1.append(jobs[i])
        else:
            group2.append(jobs[i])
            
    # Sort group1 by p1 ascending
    group1.sort(key=lambda x: x[0])
    # Sort group2 by p2 descending
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate completion time
    radar_time = 0
    sampling_time = 0
    
    for p1, p2 in order:
        radar_time += p1
        sampling_time = max(sampling_time, radar_time) + p2
        
    print(sampling_time)

if __name__ == '__main__':
    solve()
