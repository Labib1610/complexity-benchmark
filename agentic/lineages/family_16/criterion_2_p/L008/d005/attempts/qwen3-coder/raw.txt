```python
import heapq
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    sectors = []
    index = 1
    for _ in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        sectors.append((p1, p2))
        index += 2
    
    # Sort sectors by p1 - p2 in ascending order
    # This ensures we process sectors that consume more than they produce first
    # to minimize the accumulated cost
    sectors.sort(key=lambda x: x[0] - x[1])
    
    battery_charge = 0
    accumulated_cost = 0
    
    for p1, p2 in sectors:
        # Step 1: Check if we need to draw from grid
        if battery_charge < p1:
            grid_draw = p1 - battery_charge
        else:
            grid_draw = 0
        
        # Step 2: Update battery charge after consumption
        battery_charge = max(0, battery_charge - p1)
        
        # Step 3: Add energy yield to battery charge
        battery_charge += p2
        
        # Step 4: Update accumulated cost
        accumulated_cost += grid_draw + p2
    
    print(accumulated_cost)

if __name__ == "__main__":
    main()
```
