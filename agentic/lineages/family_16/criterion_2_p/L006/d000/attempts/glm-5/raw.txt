```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx]); p2 = int(data[idx+1])
        idx += 2
        jobs.append((p1, p2))
    
    # Johnson's rule: jobs with p1 <= p2 sorted by p1 ascending,
    # jobs with p1 > p2 sorted by p2 descending.
    group1 = []
    group2 = []
    for p1, p2 in jobs:
        if p1 <= p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    m1_time = 0
    m2_time = 0
    for p1, p2 in order:
        m1_time += p1
        if m2_time < m1_time:
            m2_time = m1_time + p2
        else:
            m2_time += p2
    
    print(m2_time)

solve()
```
