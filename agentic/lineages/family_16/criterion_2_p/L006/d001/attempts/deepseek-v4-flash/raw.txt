We can solve this by sorting the ships by their security scanner time in descending order, then simulating the process.  

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    j = int(data[0])
    ships = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx + 1])
        idx += 2
        ships.append((p1, p2))
    
    # Sort by p1 descending
    ships.sort(key=lambda x: -x[0])
    
    time_scanner = 0
    time_bay = 0
    for p1, p2 in ships:
        time_scanner += p1
        if time_bay < time_scanner:
            time_bay = time_scanner
        time_bay += p2
    
    print(time_bay)

if __name__ == "__main__":
    main()
```
