```python
import sys
from functools import cmp_to_key

def compare_ships(a, b):
    # a[0] is security time, a[1] is departure time
    # For Johnson's rule: 
    # if min(a[0], b[1]) <= min(b[0], a[1]), then a should come before b
    if min(a[0], b[1]) < min(b[0], a[1]):
        return -1
    elif min(a[0], b[1]) > min(b[0], a[1]):
        return 1
    else:
        return 0

def main():
    j = int(sys.stdin.readline())
    ships = []
    
    for _ in range(j):
        p1, p2 = map(int, sys.stdin.readline().split())
        ships.append((p1, p2))
    
    # Sort according to Johnson's rule for 2-machine flow shop
    ships.sort(key=cmp_to_key(compare_ships))
    
    # Simulate the process
    security_finish_time = 0
    departure_finish_time = 0
    
    for security_time, departure_time in ships:
        # When this ship finishes security (it starts as soon as the previous one finishes)
        security_finish_time += security_time
        # This ship can start departure only when both:
        # 1. It finishes security (security_finish_time)
        # 2. The previous ship finishes departure (departure_finish_time)
        # So it starts at max(security_finish_time, departure_finish_time)
        departure_start_time = max(security_finish_time, departure_finish_time)
        departure_finish_time = departure_start_time + departure_time
    
    print(departure_finish_time)

if __name__ == "__main__":
    main()
```
