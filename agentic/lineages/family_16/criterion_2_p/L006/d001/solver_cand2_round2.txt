```python
import sys

def solve():
    # The problem asks to minimize the completion time of the last truck
    # in a two-stage flow shop scheduling problem (Johnson's Rule).
    # Let p1_i be the time at the weigh station and p2_i be the time at the inspection bay.
    # Johnson's Rule:
    # 1. Divide trucks into two sets:
    #    Set A: trucks where p1_i < p2_i
    #    Set B: trucks where p1_i >= p2_i
    # 2. Sort Set A in increasing order of p1_i.
    # 3. Sort Set B in decreasing order of p2_i.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    trucks = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        trucks.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in trucks:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate completion time
    # time_w: finish time at weigh station
    # time_i: finish time at inspection bay
    time_w = 0
    time_i = 0
    
    for p1, p2 in sequence:
        time_w += p1
        # Inspection bay can start only after weigh station finishes
        # and after the previous truck finishes at the inspection bay
        time_i = max(time_w, time_i) + p2
        
    print(time_i)

if __name__ == '__main__':
    solve()
```
