```python
import sys

def solve():
    # The problem is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # We want to minimize the makespan for n jobs on 2 machines (Scanner, Indexer).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where scan_time < index_time
    #    Set 2: jobs where scan_time >= index_time
    # 2. Sort Set 1 in ascending order of scan_time.
    # 3. Sort Set 2 in descending order of index_time.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        s = int(input_data[idx])
        i = int(input_data[idx+1])
        jobs.append((s, i))
        idx += 2
        
    set1 = []
    set2 = []
    for s, i in jobs:
        if s < i:
            set1.append((s, i))
        else:
            set2.append((s, i))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    order = set1 + set2
    
    current_scan_time = 0
    current_index_time = 0
    
    for s, i in order:
        current_scan_time += s
        # Indexer starts at max(when scanning finishes, when indexer is free)
        current_index_time = max(current_scan_time, current_index_time) + i
        
    print(current_index_time)

if __name__ == '__main__':
    solve()
```
