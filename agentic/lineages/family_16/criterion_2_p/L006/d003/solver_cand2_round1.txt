import sys

def solve():
    # The problem asks to minimize the final landing horizon L.
    # Let the sequence of starships be (p1_i, p2_i).
    # E_i = E_{i-1} + p1_i
    # L_i = max(L_{i-1}, E_i) + p2_i
    # This is equivalent to Johnson's rule for the two-machine flow shop scheduling problem
    # (minimizing makespan on two machines).
    # The optimal order is:
    # 1. Starships with p1 < p2, sorted by p1 ascending.
    # 2. Starships with p1 >= p2, sorted by p2 descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    ships = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        ships.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in ships:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    e = 0
    l = 0
    for p1, p2 in order:
        e += p1
        if l < e:
            l = e
        l += p2
        
    print(l)

if __name__ == '__main__':
    solve()
