```python
import sys

def solve():
    input = sys.stdin.buffer.read().split()
    idx = 0
    j = int(input[idx]); idx += 1
    ships = []
    for _ in range(j):
        p1 = int(input[idx]); idx += 1
        p2 = int(input[idx]); idx += 1
        ships.append((p1, p2))
    
    # Sort by p2 descending
    ships.sort(key=lambda x: -x[1])
    
    entry_time = 0
    landing_time = 0
    for p1, p2 in ships:
        entry_time += p1
        start_landing = max(entry_time, landing_time)
        landing_time = start_landing + p2
    
    print(landing_time)

solve()
```
