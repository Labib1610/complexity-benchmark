import sys

def solve():
    # The problem asks to minimize B_j = max(0, B_{i-1} - p1_i) + p2_i.
    # Let S = sum(p1_i).
    # The footprint is S + B_j.
    # We want to minimize B_j.
    # This is a classic scheduling problem.
    # Divide batches into two sets:
    # Set 1: p1_i >= p2_i (Net decrease or neutral)
    # Set 2: p1_i < p2_i (Net increase)
    # For Set 1, we want to process them in descending order of p2_i.
    # For Set 2, we want to process them in ascending order of p1_i.
    # Actually, the standard greedy approach for this specific recurrence:
    # B_i = max(0, B_{i-1} - p1_i) + p2_i
    # is to sort by:
    # 1. Batches with p1_i >= p2_i: sort by p2_i descending.
    # 2. Batches with p1_i < p2_i: sort by p1_i ascending.
    # Wait, let's re-verify:
    # If we have two batches (p1_a, p2_a) and (p1_b, p2_b):
    # Order (a, b): B = max(0, max(0, -p1_a) + p2_a - p1_b) + p2_b = max(0, p2_a - p1_b) + p2_b
    # Order (b, a): B = max(0, p2_b - p1_a) + p2_a
    # We want to minimize this.
    # This is equivalent to Johnson's rule or similar scheduling.
    # The correct sorting criteria:
    # If p1_i >= p2_i and p1_j < p2_j, (i, j) is always better.
    # Within p1 >= p2, sort by p2 descending.
    # Within p1 < p2, sort by p1 ascending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    batches = []
    idx = 1
    sum_p1 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        batches.append((p1, p2))
        sum_p1 += p1
        idx += 2
        
    set1 = [] # p1 >= p2
    set2 = [] # p1 < p2
    for b in batches:
        if b[0] >= b[1]:
            set1.append(b)
        else:
            set2.append(b)
            
    # Sort set1 by p2 descending
    set1.sort(key=lambda x: x[1], reverse=True)
    # Sort set2 by p1 ascending
    set2.sort(key=lambda x: x[0])
    
    order = set1 + set2
    
    b_curr = 0
    for p1, p2 in order:
        b_curr = max(0, b_curr - p1) + p2
        
    print(sum_p1 + b_curr)

if __name__ == '__main__':
    solve()
