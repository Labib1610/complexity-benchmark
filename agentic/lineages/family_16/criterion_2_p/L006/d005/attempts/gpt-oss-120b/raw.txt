```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    good = []   # p2 >= p1
    bad = []    # p2 < p1
    total_p1 = 0
    for _ in range(n):
        p1 = int(next(it)); p2 = int(next(it))
        total_p1 += p1
        if p2 >= p1:
            good.append((p1, p2))
        else:
            bad.append((p1, p2))
    # good: sort by increasing p1
    good.sort(key=lambda x: x[0])
    # bad: sort by decreasing p2
    bad.sort(key=lambda x: x[1], reverse=True)

    battery = 0
    for p1, p2 in good:
        battery = max(battery - p1, 0) + p2
    for p1, p2 in bad:
        battery = max(battery - p1, 0) + p2

    print(total_p1 + battery)

if __name__ == "__main__":
    solve()
```
