import sys

def solve():
    # The recurrence relation is:
    # X_i = X_{i-1} + p1_i
    # Y_i = max(X_i, Y_{i-1}) + p2_i
    # Let S_i = X_i = sum_{k=1}^i p1_k
    # Y_i = max(S_i, Y_{i-1}) + p2_i
    #
    # Expanding this:
    # Y_1 = max(p1_1, 0) + p2_1 = p1_1 + p2_1
    # Y_2 = max(p1_1 + p1_2, Y_1) + p2_2 = max(p1_1 + p1_2, p1_1 + p2_1) + p2_2
    #     = p1_1 + max(p1_2, p2_1) + p2_2
    # Y_3 = max(p1_1 + p1_2 + p1_3, Y_2) + p2_3
    #     = max(p1_1 + p1_2 + p1_3, p1_1 + max(p1_2, p2_1) + p2_2) + p2_3
    #     = p1_1 + max(p1_2 + p1_3, max(p1_2, p2_1) + p2_2) + p2_3
    #     = p1_1 + max(p1_2 + p1_3, p1_2 + p2_2, p2_1 + p2_2) + p2_3
    #
    # This is equivalent to Johnson's Rule for the two-machine flow shop scheduling problem
    # (minimizing makespan on two machines).
    # The rule:
    # 1. Divide components into two sets:
    #    Set A: p1_i < p2_i
    #    Set B: p1_i >= p2_i
    # 2. Sort Set A in ascending order of p1_i.
    # 3. Sort Set B in descending order of p2_i.
    # 4. The optimal sequence is Set A followed by Set B.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    components = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        components.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    for p1, p2 in components:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    order = set_a + set_b
    
    x = 0
    y = 0
    for p1, p2 in order:
        x += p1
        y = max(x, y) + p2
        
    print(y)

if __name__ == '__main__':
    solve()
