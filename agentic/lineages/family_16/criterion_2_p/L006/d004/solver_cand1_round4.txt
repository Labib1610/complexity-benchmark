```python
import sys

def solve():
    # The cost C_k = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # C_k = sum_{i=1}^j p2_i + sum_{i=1}^k (p1_i - p2_i)
    # Let d_i = p1_i - p2_i.
    # C_k = sum(p2) + sum_{i=1}^k d_i
    # To minimize max(C_k), we want to arrange d_i such that the prefix sums
    # of d_i are as small as possible.
    # Let S_k = sum_{i=1}^k d_i. We want to minimize max_{1 <= k <= j} S_k.
    # This is achieved by sorting d_i in non-decreasing order.
    # If we sort d_i such that d_1 <= d_2 <= ... <= d_j, then the prefix sums
    # S_k will be minimized at each step k.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    rows = []
    sum_p2 = 0
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        rows.append(p1 - p2)
        sum_p2 += p2
        idx += 2
        
    rows.sort()
    
    max_c = -float('inf')
    current_prefix_sum = 0
    for d in rows:
        current_prefix_sum += d
        c_k = sum_p2 + current_prefix_sum
        if c_k > max_c:
            max_c = c_k
            
    print(max_c)

if __name__ == '__main__':
    solve()
```
