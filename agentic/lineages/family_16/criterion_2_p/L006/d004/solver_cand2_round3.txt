```python
import sys

def solve():
    # The recurrence is:
    # X_i = X_{i-1} + p1_i
    # Y_i = max(X_i, Y_{i-1}) + p2_i
    # Let S_i = sum_{k=1}^i p1_k. Then X_i = S_i.
    # Y_i = max(S_i, Y_{i-1}) + p2_i
    # Y_i - S_i = max(0, Y_{i-1} - S_i) + p2_i
    # Let D_i = Y_i - S_i.
    # D_i = max(0, Y_{i-1} - S_{i-1} - p1_i) + p2_i
    # D_i = max(0, D_{i-1} - p1_i) + p2_i
    # This is equivalent to Johnson's rule for the two-machine flow shop problem.
    # To minimize the final Y_j, we sort components based on:
    # If p1_i < p2_i, the component should come earlier (ascending order of p1).
    # If p1_i >= p2_i, the component should come later (descending order of p2).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    components = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        components.append((p1, p2))
        idx += 2
        
    # Johnson's rule:
    # Group 1: p1 < p2, sort by p1 ascending
    # Group 2: p1 >= p2, sort by p2 descending
    group1 = [c for c in components if c[0] < c[1]]
    group2 = [c for c in components if c[0] >= c[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    x = 0
    y = 0
    for p1, p2 in order:
        x += p1
        y = max(x, y) + p2
        
    print(y)

if __name__ == '__main__':
    solve()
```
