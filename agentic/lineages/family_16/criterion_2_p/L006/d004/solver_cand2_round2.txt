```python
import sys

def solve():
    # Let the sections be (p1_i, p2_i).
    # Let S1_i = sum_{k=1}^i p1_k.
    # The flexible hose i starts at max(S1_i, end_of_hose_{i-1}).
    # The end of hose i is max(S1_i, end_of_hose_{i-1}) + p2_i.
    # Let E_i be the end of hose i.
    # E_0 = 0.
    # E_i = max(S1_i, E_{i-1}) + p2_i.
    # This recurrence can be expanded:
    # E_j = max(S1_j + p2_j, S1_{j-1} + p2_{j-1} + p2_j, ..., S1_1 + p2_1 + ... + p2_j)
    # E_j = max_{1 <= i <= j} (S1_i + sum_{k=i}^j p2_k)
    # E_j = max_{1 <= i <= j} (S1_i - S2_{i-1} + S2_j) where S2_i = sum_{k=1}^i p2_k.
    # To minimize E_j, we need to minimize max_{1 <= i <= j} (S1_i - S2_{i-1}).
    # Note that S1_i = p1_1 + ... + p1_i and S2_{i-1} = p2_1 + ... + p2_{i-1}.
    # Let f_i = S1_i - S2_{i-1}.
    # We want to order sections to minimize max(f_1, f_2, ..., f_j) + S2_j.
    # Since S2_j is constant regardless of order, we minimize max(f_i).
    # This is a classic scheduling problem.
    # For two adjacent sections (p1_a, p2_a) and (p1_b, p2_b):
    # Order (a, b): f_a = S1_{prev} + p1_a - S2_{prev}, f_b = S1_{prev} + p1_a + p1_b - (S2_{prev} + p2_a)
    # Order (b, a): f_b = S1_{prev} + p1_b - S2_{prev}, f_a = S1_{prev} + p1_b + p1_a - (S2_{prev} + p2_b)
    # We prefer (a, b) if max(p1_a, p1_a + p1_b - p2_a) < max(p1_b, p1_b + p1_a - p2_b).
    # This simplifies to sorting by p2_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    sections = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        sections.append((p1, p2))
        idx += 2
    
    # Sort by p2 descending
    sections.sort(key=lambda x: x[1], reverse=True)
    
    s1 = 0
    s2 = 0
    max_val = -float('inf')
    
    for p1, p2 in sections:
        s1 += p1
        # f_i = S1_i - S2_{i-1}
        val = s1 - s2
        if val > max_val:
            max_val = val
        s2 += p2
        
    print(max_val + s2)

if __name__ == '__main__':
    solve()
```
