```python
import sys

def solve():
    # Let V1_i and V2_i be the volumes in Vat 1 and Vat 2 after step i.
    # Let A_i and B_i be the parameters of the i-th additive in the sequence.
    # V1_0 = 0, V2_0 = 0
    # Step 1: V1_i = V1_{i-1} + A_i
    # Step 2: V2_i_temp = max(V2_{i-1}, V1_i)
    # Step 3: V2_i = V2_i_temp + B_i
    #
    # Let S_i = sum_{k=1}^i A_k. Then V1_i = S_i.
    # V2_i = max(V2_{i-1}, S_i) + B_i
    #
    # This is a classic scheduling problem. We want to minimize the final V2_n.
    # V2_n = max(S_n + B_n, S_{n-1} + B_{n-1} + B_n, S_{n-2} + B_{n-2} + B_{n-1} + B_n, ..., S_1 + B_1 + ... + B_n)
    # V2_n = max_{1 <= i <= n} (S_i + sum_{k=i}^n B_k)
    #
    # To minimize this, we use the Smith's Rule / Exchange Argument.
    # Consider two adjacent additives (A1, B1) and (A2, B2).
    # Order 1, 2: max(S + A1 + B1 + B2, S + A1 + A2 + B2)
    # Order 2, 1: max(S + A2 + B2 + B1, S + A2 + A1 + B1)
    #
    # We want to pick the order that results in a smaller max.
    # This is equivalent to sorting by B_i. If B_i < B_j, put i before j.
    # If B_i == B_j, the order doesn't matter for the max.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    additives = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        additives.append((a, b))
        idx += 2
        
    # Sort by B ascending
    additives.sort(key=lambda x: x[1])
    
    v1 = 0
    v2 = 0
    for a, b in additives:
        v1 += a
        if v1 > v2:
            v2 = v1
        v2 += b
        
    print(v2)

if __name__ == '__main__':
    solve()
```
