```python
import sys

def solve():
    # The inefficiency rating is (sum of p1) + (sum of p2) - (final ILC).
    # Since (sum of p1) and (sum of p2) are constant regardless of order,
    # minimizing the inefficiency rating is equivalent to maximizing the final ILC.
    #
    # Let the sequence be (p1_1, p2_1), (p1_2, p2_2), ..., (p1_j, p2_j).
    # Let TGP_k be the TGP after k applications, TGP_k = sum_{i=1}^k p2_i.
    # Let ILC_k be the ILC after k applications.
    # ILC_0 = 0
    # ILC_k = min(ILC_{k-1} + p1_k, TGP_k)
    #
    # This recurrence implies ILC_j = min(sum_{i=1}^j p1_i, min_{k=1..j} (TGP_k + sum_{i=k+1}^j p1_i))
    # Let S = sum_{i=1}^j p1_i.
    # ILC_j = min(S, min_{k=1..j} (TGP_k + S - sum_{i=1}^k p1_i))
    # ILC_j = S - max(0, max_{k=1..j} (sum_{i=1}^k p1_i - TGP_k))
    #
    # To maximize ILC_j, we need to minimize max_{k=1..j} (sum_{i=1}^k (p1_i - p2_i)).
    # This is a classic scheduling problem. We want to order items to minimize the 
    # maximum prefix sum of (p1_i - p2_i).
    #
    # Split items into two groups:
    # 1. p1_i - p2_i <= 0 (i.e., p1_i <= p2_i): These should come first, sorted by p1_i ascending.
    # 2. p1_i - p2_i > 0 (i.e., p1_i > p2_i): These should come after, sorted by p2_i descending.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    apps = []
    sum_p1 = 0
    sum_p2 = 0
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        apps.append((p1, p2))
        sum_p1 += p1
        sum_p2 += p2
        idx += 2
    
    group1 = [a for a in apps if a[0] <= a[1]]
    group2 = [a for a in apps if a[0] > a[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    ilc = 0
    tgp = 0
    for p1, p2 in order:
        ilc = min(ilc + p1, tgp + p2)
        tgp += p2
        
    print(sum_p1 + sum_p2 - ilc)

if __name__ == '__main__':
    solve()
```
