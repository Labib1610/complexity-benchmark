```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    stages = []
    p1_sum = 0
    
    index = 1
    for _ in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        stages.append((p1, p2))
        p1_sum += p1
        index += 2
    
    # Sort stages by p2 in descending order
    stages.sort(key=lambda x: x[1], reverse=True)
    
    V = 0
    for p1, p2 in stages:
        V = max(p2, V - p1 + p2)
    
    result = p1_sum + V
    print(result)

if __name__ == "__main__":
    main()
```
