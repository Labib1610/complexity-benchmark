```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    modules = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        modules.append((p1, p2))
        idx += 2
    
    # Let the sequence be m_1, m_2, ..., m_j.
    # Score at k: S_k = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # S_k = sum_{i=1}^j p2_i + sum_{i=1}^k (p1_i - p2_i) + p2_k
    # This is equivalent to Johnson's rule logic.
    # Sort modules such that those with p1 < p2 come first (sorted by p1 ascending),
    # and those with p1 >= p2 come last (sorted by p2 descending).
    
    group1 = []
    group2 = []
    for p1, p2 in modules:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate bottleneck rating for this sequence
    # S_k = prefix_p1[k] + suffix_p2[k]
    # prefix_p1[k] = p1_1 + ... + p1_k
    # suffix_p2[k] = p2_k + ... + p2_j
    
    total_p2 = sum(m[1] for m in sequence)
    
    current_p1 = 0
    current_p2_sum = total_p2
    max_score = 0
    
    for p1, p2 in sequence:
        current_p1 += p1
        # Score at this k is current_p1 + current_p2_sum
        score = current_p1 + current_p2_sum
        if score > max_score:
            max_score = score
        current_p2_sum -= p2
        
    print(max_score)

if __name__ == '__main__':
    solve()
```
