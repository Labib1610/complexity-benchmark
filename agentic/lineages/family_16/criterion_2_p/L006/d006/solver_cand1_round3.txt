```python
import sys

def solve():
    # The thermal stress is defined as sum(p1) + final_temp.
    # Let T_i be the temperature after phase i.
    # T_0 = 0
    # T_i = max(T_{i-1} - p1_i + p2_i, p2_i)
    # T_i = max(T_{i-1} - p1_i, 0) + p2_i
    #
    # Let's analyze the final temperature T_j:
    # T_1 = max(0 - p1_1, 0) + p2_1 = p2_1
    # T_2 = max(T_1 - p1_2, 0) + p2_2 = max(p2_1 - p1_2, 0) + p2_2
    # T_3 = max(T_2 - p1_3, 0) + p2_3 = max(max(p2_1 - p1_2, 0) + p2_2 - p1_3, 0) + p2_3
    #
    # This is equivalent to:
    # T_j = max(p2_1 - p1_2 - p1_3 - ... - p1_j, 
    #           p2_2 - p1_3 - ... - p1_j, 
    #           ..., 
    #           p2_j, 
    #           0)
    # Since p1, p2 >= 1, the 0 is redundant.
    # T_j = max(p2_i - sum_{k=i+1}^j p1_k) for i=1 to j.
    # Let S = sum_{k=1}^j p1_k.
    # T_j = max(p2_i - (S - sum_{k=1}^i p1_k))
    # T_j = max(p2_i + sum_{k=1}^i p1_k) - S
    # Thermal Stress = S + T_j = S + max(p2_i + sum_{k=1}^i p1_k) - S
    # Thermal Stress = max_{i=1..j} (p2_i + sum_{k=1}^i p1_k)
    #
    # To minimize this, we want to order the phases such that the prefix sums
    # of p1 are as small as possible when paired with large p2.
    # This is a classic scheduling problem: sort by p1 ascending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    phases = []
    idx = 1
    total_p1 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        phases.append((p1, p2))
        total_p1 += p1
        idx += 2
    
    # Sort by p1 ascending
    phases.sort(key=lambda x: x[0])
    
    # Calculate max(p2_i + sum_{k=1}^i p1_k)
    max_val = 0
    current_prefix_p1 = 0
    for p1, p2 in phases:
        current_prefix_p1 += p1
        val = p2 + current_prefix_p1
        if val > max_val:
            max_val = val
            
    print(max_val)

if __name__ == '__main__':
    solve()
```
