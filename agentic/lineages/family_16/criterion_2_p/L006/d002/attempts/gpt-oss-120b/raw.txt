```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    A = []  # p1 <= p2
    B = []  # p1 > p2
    for _ in range(n):
        p1 = int(next(it)); p2 = int(next(it))
        if p1 <= p2:
            A.append((p1, p2))
        else:
            B.append((p1, p2))
    A.sort(key=lambda x: x[0])          # increasing p1
    B.sort(key=lambda x: -x[1])         # decreasing p2
    seq = A + B

    prefix = 0
    suffix_sum = [0] * (n + 1)
    # compute suffix sums of p2
    for i in range(n - 1, -1, -1):
        suffix_sum[i] = suffix_sum[i + 1] + seq[i][1]

    best = 0
    for i, (p1, _) in enumerate(seq):
        prefix += p1
        t = prefix + suffix_sum[i]   # suffix includes current p2
        if t > best:
            best = t
    print(best)

if __name__ == "__main__":
    solve()
```
