```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    packets = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        packets.append((p1, p2))
        idx += 2
    
    # To minimize the total time, we use Johnson's Rule logic.
    # We want to process packets with smaller p1 first, and packets with 
    # smaller p2 last.
    # Specifically, split into two groups:
    # Group 1: p1 < p2 (sort by p1 ascending)
    # Group 2: p1 >= p2 (sort by p2 descending)
    
    group1 = []
    group2 = []
    for p1, p2 in packets:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_cpu_time = 0
    antenna_finish_time = 0
    
    for p1, p2 in order:
        current_cpu_time += p1
        # Antenna starts at max(current_cpu_time, previous_antenna_finish_time)
        if current_cpu_time > antenna_finish_time:
            antenna_finish_time = current_cpu_time + p2
        else:
            antenna_finish_time += p2
            
    print(antenna_finish_time)

if __name__ == '__main__':
    solve()
```
