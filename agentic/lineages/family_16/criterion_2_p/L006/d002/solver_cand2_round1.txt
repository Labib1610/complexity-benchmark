import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        p = int(input_data[idx])
        s = int(input_data[idx+1])
        vessels.append((p, s))
        idx += 2
    
    # Let P_i be the primary scan duration and S_i be the secondary clearance duration.
    # Let start_p_i be the start time of primary scan for vessel i.
    # Let start_s_i be the start time of secondary clearance for vessel i.
    # Constraints:
    # 1. start_p_{i+1} >= start_p_i + P_i
    # 2. start_s_i >= start_p_i + P_i
    # 3. start_s_{i+1} >= start_s_i + S_i
    #
    # To minimize the finish time of the last vessel (start_s_j + S_j):
    # start_p_1 = 0
    # start_p_i = sum(P_0 ... P_{i-1})
    # start_s_1 = P_1 + S_1 (Wait, no, start_s_1 = P_1 + S_1 is wrong)
    # Actually:
    # start_p_1 = 0
    # start_p_i = start_p_{i-1} + P_{i-1}
    # start_s_1 = start_p_1 + P_1 = P_1
    # start_s_i = max(start_p_i + P_i, start_s_{i-1} + S_{i-1})
    # Finish time = start_s_j + S_j
    
    current_p = 0
    current_s = 0
    
    for i in range(j):
        p, s = vessels[i]
        # Primary scan starts at current_p
        # Secondary clearance starts at max(current_p + p, current_s)
        start_s = max(current_p + p, current_s)
        current_p += p
        current_s = start_s + s
        
    print(current_s)

if __name__ == '__main__':
    solve()
