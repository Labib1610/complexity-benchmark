```python
import sys

def solve():
    # The total resistance for a given arrangement and refraction point k is:
    # R(k) = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # R(k) = sum_{i=1}^j p2_i + sum_{i=1}^k (p1_i - p2_i)
    # Let D_i = p1_i - p2_i.
    # R(k) = (sum_{i=1}^j p2_i) + sum_{i=1}^k D_i
    # To minimize the maximum R(k), we want to minimize the maximum prefix sum of D_i.
    # Let S_k = sum_{i=1}^k D_i. We want to minimize max(S_1, S_2, ..., S_j).
    # This is a classic problem: arrange D_i to minimize the maximum prefix sum.
    # The optimal strategy is to put all negative D_i values first (in any order),
    # followed by all non-negative D_i values (in any order).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    tiles = []
    idx = 1
    sum_p2 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tiles.append(p1 - p2)
        sum_p2 += p2
        idx += 2
        
    neg = [d for d in tiles if d < 0]
    pos = [d for d in tiles if d >= 0]
    
    # Arrange negatives first, then positives
    ordered_d = neg + pos
    
    max_prefix_sum = -float('inf')
    current_sum = 0
    for d in ordered_d:
        current_sum += d
        if current_sum > max_prefix_sum:
            max_prefix_sum = current_sum
            
    print(sum_p2 + max_prefix_sum)

if __name__ == '__main__':
    solve()
```
