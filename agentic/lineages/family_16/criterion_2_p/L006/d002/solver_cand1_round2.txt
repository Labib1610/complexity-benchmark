```python
import sys

def solve():
    # The problem asks to minimize max_{k=1..j} (sum_{i=1}^k p1_i + sum_{i=k}^j p2_i).
    # Let S1_k = sum_{i=1}^k p1_i and S2_k = sum_{i=k}^j p2_i.
    # T(k) = S1_k + S2_k.
    # Note that S1_k + S2_k = (sum_{i=1}^j p1_i) + (sum_{i=k}^j p2_i - sum_{i=k+1}^j p1_i)
    # Actually, a simpler way to look at T(k):
    # T(k) = (sum_{i=1}^{k-1} p1_i) + p1_k + p2_k + (sum_{i=k+1}^j p2_i)
    # Let P = sum_{i=1}^j p1_i and Q = sum_{i=1}^j p2_i.
    # T(k) = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # T(k) = sum_{i=1}^k p1_i + (sum_{i=1}^j p2_i - sum_{i=1}^{k-1} p2_i)
    # T(k) = sum_{i=1}^j p2_i + sum_{i=1}^k p1_i - sum_{i=1}^{k-1} p2_i
    # Let d_i = p1_i - p2_i.
    # T(k) = sum_{i=1}^j p2_i + sum_{i=1}^{k-1} (p1_i - p2_i) + p1_k
    # T(k) = sum_{i=1}^j p2_i + sum_{i=1}^{k-1} d_i + p1_k
    # To minimize the maximum of these values, we should sort the starships.
    # This is a variation of Johnson's rule.
    # If p1_i < p2_i, we want them earlier. If p1_i >= p2_i, we want them later.
    # Specifically, sort by min(p1_i, p2_i).
    # If p1_i < p2_i, sort by p1_i ascending.
    # If p1_i >= p2_i, sort by p2_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    ships = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        ships.append((p1, p2))
        idx += 2
        
    group1 = [s for s in ships if s[0] < s[1]]
    group2 = [s for s in ships if s[0] >= s[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    max_t = 0
    current_p1 = 0
    # Precompute total p2
    total_p2 = sum(s[1] for s in order)
    
    current_p2_sum = 0
    for i in range(j):
        p1, p2 = order[i]
        current_p1 += p1
        # T(k) = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
        # sum_{i=k}^j p2_i = total_p2 - current_p2_sum
        t_k = current_p1 + (total_p2 - current_p2_sum)
        if t_k > max_t:
            max_t = t_k
        current_p2_sum += p2
        
    print(max_t)

if __name__ == '__main__':
    solve()
```
