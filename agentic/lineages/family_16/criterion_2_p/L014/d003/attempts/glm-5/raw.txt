import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    packets = []
    for _ in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        packets.append((p1, p2))
    
    # Johnson's rule for two-machine flow shop
    # Group 1: p1 <= p2, sort by p1 ascending
    # Group 2: p1 > p2, sort by p2 descending
    group1 = []
    group2 = []
    for p1, p2 in packets:
        if p1 <= p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Simulate the flow shop
    t_up = 0
    t_down = 0
    for p1, p2 in order:
        t_up += p1
        t_down = max(t_up, t_down) + p2
    
    print(t_down)

if __name__ == "__main__":
    solve()
