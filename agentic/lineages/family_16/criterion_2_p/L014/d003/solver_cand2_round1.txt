import sys

def solve():
    # This is a classic Johnson's Rule scheduling problem.
    # We have n jobs, each with two stages.
    # Stage 1: p1, Stage 2: p2.
    # We want to minimize the total completion time (makespan).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 2. Sort Set A in ascending order of p1.
    # 3. Sort Set B in descending order of p2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    orders = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        orders.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in orders:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    t1 = 0
    t2 = 0
    for p1, p2 in sequence:
        t1 += p1
        t2 = max(t1, t2) + p2
        
    print(t2)

if __name__ == '__main__':
    solve()
