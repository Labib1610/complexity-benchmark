```python
import sys

def solve():
    # The problem asks to minimize T_out(j) = max(T_in(i), T_out(i-1)) + y_i.
    # This is a classic scheduling problem known as Johnson's Rule for 2-machine flow shop.
    # We want to minimize the makespan on two machines where:
    # Machine 1 (inner delay) processes x_i
    # Machine 2 (outer delay) processes y_i
    # The optimal order is to sort signals such that:
    # If x_i < y_i, the signal should come earlier (sorted by x_i ascending).
    # If x_i >= y_i, the signal should come later (sorted by y_i descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    signals = []
    idx = 1
    for _ in range(j):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        signals.append((x, y))
        idx += 2
        
    # Johnson's Rule:
    # Group 1: x_i < y_i, sort by x_i ascending
    # Group 2: x_i >= y_i, sort by y_i descending
    group1 = []
    group2 = []
    for x, y in signals:
        if x < y:
            group1.append((x, y))
        else:
            group2.append((x, y))
            
    group1.sort(key=lambda s: s[0])
    group2.sort(key=lambda s: s[1], reverse=True)
    
    order = group1 + group2
    
    t_in = 0
    t_out = 0
    for x, y in order:
        t_in += x
        t_out = max(t_in, t_out) + y
        
    print(t_out)

if __name__ == '__main__':
    solve()
```
