```python
import sys

def solve():
    # The problem asks to minimize the final value of TB.
    # Let the sequence of packets be (p1_i, p2_i) for i = 1 to j.
    # TA_k = sum_{i=1}^k p1_i
    # TB_k = max(TB_{k-1}, TA_k) + p2_k
    # This is a classic scheduling problem known as Johnson's Rule for 2-machine flow shop.
    # To minimize the makespan (final TB), we sort the packets:
    # 1. Packets with p1 < p2 should come first, sorted by increasing p1.
    # 2. Packets with p1 >= p2 should come last, sorted by decreasing p2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    packets = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        packets.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in packets:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    ta = 0
    tb = 0
    for p1, p2 in sequence:
        ta += p1
        if tb < ta:
            tb = ta
        tb += p2
        
    print(tb)

if __name__ == '__main__':
    solve()
```
