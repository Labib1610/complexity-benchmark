```python
import sys

def solve():
    # The score for a given arrangement (A_1, B_1), ..., (A_j, B_j) and index k is:
    # Score(k) = sum_{i=1}^k A_i + sum_{i=k}^j B_i
    #          = sum_{i=1}^j B_i + sum_{i=1}^k (A_i - B_i)
    # Let D_i = A_i - B_i.
    # Score(k) = sum(B) + sum_{i=1}^k D_i
    # To minimize the maximum score, we want to minimize the maximum prefix sum of D_i.
    # Let S_k = sum_{i=1}^k D_i. We want to minimize max(0, S_1, S_2, ..., S_j).
    # Note that S_0 = 0 is always a candidate for the max if we consider k=0, 
    # but the problem says 1 <= k <= j.
    # So we want to minimize max(S_1, S_2, ..., S_j).
    # To minimize the maximum prefix sum of a set of numbers, we should sort them
    # such that the negative values come first.
    # If we sort D_i in non-decreasing order, the prefix sums will be as small as possible.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    tiles = []
    sum_b = 0
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        tiles.append(a - b)
        sum_b += b
        idx += 2
        
    # Sort D_i = A_i - B_i in non-decreasing order
    tiles.sort()
    
    # Calculate prefix sums of sorted D_i
    max_prefix_sum = -float('inf')
    current_sum = 0
    for d in tiles:
        current_sum += d
        if current_sum > max_prefix_sum:
            max_prefix_sum = current_sum
            
    print(sum_b + max_prefix_sum)

if __name__ == '__main__':
    solve()
```
