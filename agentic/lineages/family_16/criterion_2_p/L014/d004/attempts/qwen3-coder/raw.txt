import sys
from itertools import accumulate

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    blocks = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx+1])
        blocks.append((p1, p2))
        idx += 2
    
    # Sort blocks according to Johnson's rule:
    # For two jobs i and j, schedule i before j if min(p1_i, p2_j) < min(p1_j, p2_i)
    # This is equivalent to sorting by p1 ascending if p1 <= p2, else by p2 descending
    def sort_key(block):
        p1, p2 = block
        if p1 <= p2:
            return (p1, 0)   # prioritize smaller p1
        else:
            return (p2, 1)   # prioritize larger p2 by using 1 to reverse order
    
    # Actually, let's implement the correct Johnson's rule comparator
    # We'll sort by p1 ascending if p1 <= p2, otherwise by p2 descending
    # But to make it simpler, we use the standard Johnson's rule:
    # job i < job j if min(p1_i, p2_j) < min(p1_j, p2_i)
    # However, there's a simpler way for this specific case:
    # Sort by p1 - p2 ascending (this is known to work for F2||Cmax)
    
    blocks.sort(key=lambda x: x[0] - x[1])
    
    # Now compute the makespan using the crossover point method
    p1_list = [block[0] for block in blocks]
    p2_list = [block[1] for block in blocks]
    
    prefix_p1 = [0] + list(accumulate(p1_list))
    suffix_p2 = list(accumulate(p2_list[::-1]))[::-1] + [0]
    
    max_time = 0
    for k in range(1, j+1):
        time = prefix_p1[k] + suffix_p2[k]
        if time > max_time:
            max_time = time
            
    print(max_time)

if __name__ == "__main__":
    main()
