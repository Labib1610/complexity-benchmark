import sys
from collections import deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    orders = []
    index = 1
    for i in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        orders.append((p1, p2))
        index += 2
    
    # Sort orders by p2 in descending order (Johnson's rule variant for two-stage flow shop)
    # This minimizes the makespan
    orders.sort(key=lambda x: x[1], reverse=True)
    
    # Simulate the process
    mixer_time = 0
    froster_time = 0
    pending = deque()  # (available_time, p2)
    
    i = 0
    time = 0
    
    while i < j or pending:
        # Mixer processes orders
        while i < j:
            p1, p2 = orders[i]
            mixer_time += p1
            pending.append((mixer_time, p2))
            i += 1
            break  # Mixer can only take one order at a time
        
        # Froster takes the earliest ready order from pending
        if pending:
            ready_time, p2 = pending.popleft()
            froster_time = max(froster_time, ready_time) + p2
        
        # Time progresses to the next event
        next_mixer_free = mixer_time if i < j else float('inf')
        next_froster_free = froster_time if pending else float('inf')
        
        if pending:
            time = min(next_mixer_free, next_froster_free)
        else:
            time = next_mixer_free if i < j else froster_time
            if not pending and i >= j:
                break
    
    print(froster_time)

if __name__ == "__main__":
    main()
