import sys

def solve():
    # This is a variation of Johnson's Rule for the two-machine flow shop problem.
    # To minimize the makespan, we sort the jobs based on the following criteria:
    # If p1 < p2, the job should be placed earlier.
    # If p1 >= p2, the job should be placed later.
    # Specifically, sort by min(p1, p2), and for ties or ordering, 
    # jobs with p1 < p2 are sorted by p1 ascending,
    # jobs with p1 >= p2 are sorted by p2 descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Johnson's rule:
    # Group 1: p1 < p2, sort by p1 ascending
    # Group 2: p1 >= p2, sort by p2 descending
    group1 = []
    group2 = []
    for p1, p2 in jobs:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    oven_time = 0
    finish_time = 0
    
    for p1, p2 in order:
        oven_time += p1
        # The decoration counter can start only after the oven finishes this cake
        # AND after the decoration counter finishes the previous cake.
        finish_time = max(finish_time, oven_time) + p2
        
    print(finish_time)

if __name__ == '__main__':
    solve()
