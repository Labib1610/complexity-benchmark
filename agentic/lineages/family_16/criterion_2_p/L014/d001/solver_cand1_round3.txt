```python
import sys

def solve():
    # The problem is a variation of Johnson's Rule for a 2-machine flow shop.
    # Here, we have three stages: Preparation (P), Inspection (I), Finishing (F).
    # Since Inspection (I) takes 0 time for all requests, the problem reduces
    # to a 2-machine flow shop problem with machines P and F.
    # Johnson's Rule:
    # 1. List all jobs.
    # 2. Find the minimum processing time among all P_i and F_i.
    # 3. If the minimum is P_i, place job i at the beginning of the sequence.
    # 4. If the minimum is F_i, place job i at the end of the sequence.
    # 5. Remove job i and repeat.
    
    # Implementation of Johnson's Rule:
    # Sort jobs such that jobs with min(P_i, F_i) = P_i come first (sorted by P_i ascending),
    # and jobs with min(P_i, F_i) = F_i come last (sorted by F_i descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p = int(input_data[idx])
        f = int(input_data[idx+1])
        jobs.append((p, f))
        idx += 2
        
    group1 = [] # P_i <= F_i
    group2 = [] # P_i > F_i
    
    for p, f in jobs:
        if p <= f:
            group1.append((p, f))
        else:
            group2.append((p, f))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate completion time
    # P_time: time when current job finishes preparation
    # F_time: time when current job finishes finishing
    p_time = 0
    f_time = 0
    
    for p, f in sequence:
        p_time += p
        # Finishing can start only after preparation is done and 
        # the assistant is free from the previous job.
        f_time = max(p_time, f_time) + f
        
    print(f_time)

if __name__ == '__main__':
    solve()
```
