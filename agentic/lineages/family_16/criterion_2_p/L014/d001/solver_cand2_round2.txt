import sys

def solve():
    # The cost of a path moving down at column k is:
    # sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # Let S1 = sum(p1_i) and S2 = sum(p2_i).
    # The cost is S1 + S2 - sum_{i=1}^{k-1} p1_i - sum_{i=k+1}^j p2_i
    # Actually, it's simpler:
    # Cost(k) = (sum_{i=1}^j p1_i) + (sum_{i=1}^j p2_i) - (sum_{i=k+1}^j p1_i) - (sum_{i=1}^{k-1} p2_i)
    # Let T = sum(p1) + sum(p2).
    # Cost(k) = T - (sum_{i=k+1}^j p1_i + sum_{i=1}^{k-1} p2_i)
    # To minimize the maximum cost, we want to maximize the minimum value of
    # (sum_{i=k+1}^j p1_i + sum_{i=1}^{k-1} p2_i).
    # Let x_i = p1_i and y_i = p2_i.
    # We want to order pairs (x_i, y_i) such that min_k (sum_{i=k+1}^j x_i + sum_{i=1}^{k-1} y_i) is maximized.
    # This is equivalent to sorting the blocks such that x_i - y_i is non-increasing.
    # If we sort such that x_1 - y_1 >= x_2 - y_2 >= ... >= x_j - y_j,
    # the maximum cost is minimized.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    blocks = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        blocks.append((p1, p2))
        idx += 2
        
    # Sort by p1 - p2 descending
    blocks.sort(key=lambda x: x[0] - x[1], reverse=True)
    
    p1s = [b[0] for b in blocks]
    p2s = [b[1] for b in blocks]
    
    # Prefix sums of p2, suffix sums of p1
    # Cost(k) = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # Let pref1[k] = sum_{i=0}^{k-1} p1_i
    # Let suff2[k] = sum_{i=k}^{j-1} p2_i
    # Cost(k) = pref1[k] + suff2[k-1]
    
    pref1 = [0] * (j + 1)
    for i in range(j):
        pref1[i+1] = pref1[i] + p1s[i]
        
    suff2 = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suff2[i] = suff2[i+1] + p2s[i]
        
    max_cost = 0
    for k in range(1, j + 1):
        # Path moves down at column k (1-indexed)
        # sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
        current_cost = pref1[k] + suff2[k-1]
        if current_cost > max_cost:
            max_cost = current_cost
            
    print(max_cost)

solve()
