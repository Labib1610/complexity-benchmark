```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    blocks = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        blocks.append((p1, p2))
        idx += 2
    
    # This is a variation of Johnson's rule for two-machine scheduling.
    # To minimize the makespan on two machines where each job must go through
    # machine 1 then machine 2, we sort jobs based on the minimum of their
    # processing times.
    # Group 1: p1 < p2, sort by p1 ascending.
    # Group 2: p1 >= p2, sort by p2 descending.
    
    group1 = []
    group2 = []
    for p1, p2 in blocks:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    t_sub = 0
    t_trans = 0
    
    for p1, p2 in order:
        t_sub += p1
        t_trans = max(t_sub, t_trans) + p2
        
    print(t_trans)

if __name__ == '__main__':
    solve()
```
