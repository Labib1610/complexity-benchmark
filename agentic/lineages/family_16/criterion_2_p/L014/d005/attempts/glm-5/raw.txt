```python
def main():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    j = int(input_data[idx]); idx += 1
    group1 = []  # p1 <= p2, sort by p1 asc
```

= max(max_{j<i}(S[j] + sum_{k=j}^{i-1} p2[k]) + p2[i], S[i] + p2[i])
= max(max_{j<i}(S[j] + sum_{k=j}^{i} p2[k]), S[i] + p2[i])
= max_{j≤i}(S[j] + sum_{k=j}^{i} p2[k]). ✓

So H[n-1] = max_{0 ≤ j ≤ n-1} (S[j] + sum_{k=j}^{n-1} p2[k]) = max_j (S[j] + suffix_p2[j]). ✓

Great, the formula is correct.

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    j = int(data[0])
    group1 = []
    group2 = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx]); p2 = int(data[idx+1]); idx += 2
        if p1 <= p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: -x[1])
    order = group1 + group2
    total_p2 = sum(p2 for _, p2 in order)
    prefix_p1 = 0
    suffix_p2 = total_p2
    best = 0
    for p1, p2 in order:
        prefix_p1 += p1
        val = prefix_p1 + suffix_p2
        if val > best:
            best = val
        suffix_p2 -= p2
    print(best)

main()
```
