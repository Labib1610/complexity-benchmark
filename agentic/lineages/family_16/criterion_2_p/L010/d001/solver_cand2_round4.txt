```python
import sys

def solve():
    # The problem asks for the minimum finish time of the last convoy.
    # Let the convoys be c_1, c_2, ..., c_j in the chosen order.
    # Let p1_i and p2_i be the North Gorge and South Pass times for convoy i.
    # The North Gorge finish times are:
    # N_1 = p1_1
    # N_i = N_{i-1} + p1_i
    # The South Pass finish times are:
    # S_1 = N_1 + p2_1
    # S_i = max(S_{i-1}, N_i) + p2_i
    #
    # This is a classic scheduling problem (Johnson's Rule).
    # To minimize the completion time of the last job on two machines:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where p1_i < p2_i
    #    Set B: jobs where p1_i >= p2_i
    # 2. Sort Set A in ascending order of p1_i.
    # 3. Sort Set B in descending order of p2_i.
    # 4. The optimal sequence is Set A followed by Set B.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    convoys = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        convoys.append((p1, p2))
        idx += 2
        
    set_a = [c for c in convoys if c[0] < c[1]]
    set_b = [c for c in convoys if c[0] >= c[1]]
    
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    order = set_a + set_b
    
    current_n = 0
    current_s = 0
    
    for p1, p2 in order:
        current_n += p1
        current_s = max(current_s, current_n) + p2
        
    print(current_s)

if __name__ == '__main__':
    solve()
```
