```python
import sys
from typing import List, Tuple

def read_input() -> List[Tuple[int, int]]:
    lines = sys.stdin.readlines()
    j = int(lines[0])
    orders = []
    for i in range(1, j + 1):
        p1, p2 = map(int, lines[i].split())
        orders.append((p1, p2))
    return orders

def solve(orders: List[Tuple[int, int]]) -> int:
    # Sort orders by p1 ascending, then by p2 descending
    # This is a known heuristic for this type of scheduling problem
    # It's related to the Johnson's rule for 2-machine flow shop scheduling
    # But we're optimizing for makespan (completion time of last job)
    
    # For this specific problem, we want to minimize the maximum completion time
    # A good strategy is to sort by (p1 - p2) ascending
    # If p1 - p2 is negative, it means p2 > p1, so we prefer to do these later
    # If p1 - p2 is positive, it means p1 > p2, so we prefer to do these earlier
    
    orders.sort(key=lambda x: x[0] - x[1])
    
    # Now simulate the schedule
    S = 0  # Preparation start time
    A = 0  # Assembly start time
    
    for p1, p2 in orders:
        S_prev = S
        S += p1  # Start next preparation when current one finishes
        # Assembly can start when both preparation is done AND previous assembly is done
        A = max(S, A + p2) if len(locals()) > 0 else S
        
    # The completion time is when the last assembly finishes
    return A

def main():
    orders = read_input()
    result = solve(orders)
    print(result)

if __name__ == "__main__":
    main()
```
