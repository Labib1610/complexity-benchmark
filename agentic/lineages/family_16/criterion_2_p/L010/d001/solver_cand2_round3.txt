import sys

def solve():
    # The problem asks for the minimum completion time of a two-stage pipeline
    # (North Gorge then South Pass) for j jobs, where each job i has durations
    # p1_i and p2_i. This is a classic Johnson's Rule problem.
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where p1_i < p2_i
    #    Set B: jobs where p1_i >= p2_i
    # 2. Sort Set A in ascending order of p1_i.
    # 3. Sort Set B in descending order of p2_i.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate completion time
    # current_north_end: time when the current job finishes North Gorge
    # current_south_end: time when the current job finishes South Pass
    current_north_end = 0
    current_south_end = 0
    
    for p1, p2 in sequence:
        current_north_end += p1
        # South Pass can start only after North Gorge is done AND 
        # the previous job has cleared the South Pass
        current_south_end = max(current_north_end, current_south_end) + p2
        
    print(current_south_end)

if __name__ == '__main__':
    solve()
