```python
import sys

def solve():
    # The problem is equivalent to Johnson's Rule for a two-machine flow shop.
    # We want to minimize the makespan of a sequence of jobs on two machines.
    # Machine 1: Preparation (p1)
    # Machine 2: Assembly (p2)
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate makespan
    # S_k = sum(p1 of previous jobs)
    # A_k = max(S_k + p1_k, A_{k-1} + p2_{k-1})
    # Completion time = A_j + p2_j
    
    current_p1_sum = 0
    current_a = 0
    
    for p1, p2 in sequence:
        # Preparation for this job finishes at current_p1_sum + p1
        prep_finish = current_p1_sum + p1
        # Assembly for this job can start at max(prep_finish, current_a)
        # Note: current_a is the finish time of the previous assembly
        start_assembly = max(prep_finish, current_a)
        current_a = start_assembly + p2
        current_p1_sum += p1
        
    print(current_a)

if __name__ == '__main__':
    solve()
```
