```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    assets = []
    idx = 1
    for _ in range(j):
        l = int(input_data[idx])
        r = int(input_data[idx+1])
        assets.append((l, r))
        idx += 2
        
    # This is a variation of Johnson's Rule for the two-machine flow shop problem.
    # To minimize the makespan, sort assets such that:
    # Assets with L < R come first, sorted by L ascending.
    # Assets with L >= R come last, sorted by R descending.
    
    group1 = []
    group2 = []
    for l, r in assets:
        if l < r:
            group1.append((l, r))
        else:
            group2.append((l, r))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    editor_time = 0
    animator_time = 0
    
    for l, r in order:
        editor_time += l
        if animator_time < editor_time:
            animator_time = editor_time + r
        else:
            animator_time += r
            
    print(animator_time)

if __name__ == '__main__':
    solve()
```
