import sys

def solve():
    # The problem asks to minimize the completion time of a two-stage pipeline.
    # This is a classic Johnson's Rule problem.
    # Stage 1: Compression (d1), Stage 2: Encryption (d2).
    # Johnson's Rule:
    # 1. Divide files into two sets:
    #    Set A: files where d1 < d2
    #    Set B: files where d1 >= d2
    # 2. Sort Set A in ascending order of d1.
    # 3. Sort Set B in descending order of d2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    files = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        files.append((d1, d2))
        idx += 2
        
    set_a = []
    set_b = []
    for f in files:
        if f[0] < f[1]:
            set_a.append(f)
        else:
            set_b.append(f)
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate completion time
    # X_k is the start time of compression for file k
    # Y_k is the start time of encryption for file k
    # X_1 = 0
    # X_{k+1} = X_k + d1_k
    # Y_k = max(X_k + d1_k, Y_{k-1} + d2_{k-1}) for k > 1
    # Y_1 = X_1 + d1_1
    # Final target = Y_j + d2_j
    
    current_x = 0
    current_y = 0
    
    for i in range(j):
        d1, d2 = sequence[i]
        if i == 0:
            current_x = 0
            current_y = d1
        else:
            current_x += sequence[i-1][0]
            current_y = max(current_x + d1, current_y + sequence[i-1][1])
            
    print(current_y + sequence[-1][1])

if __name__ == '__main__':
    solve()
