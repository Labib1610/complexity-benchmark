```python
import sys
import threading

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    lenses = []
    index = 1
    for i in range(j):
        d1 = int(data[index])
        d2 = int(data[index+1])
        lenses.append((d1, d2))
        index += 2
    
    # Sort lenses by d1 ascending, then by d2 descending
    # This is a known greedy strategy for this type of scheduling problem
    lenses.sort(key=lambda x: (x[0], -x[1]))
    
    X = 0
    Y = 0
    
    for d1, d2 in lenses:
        # Ensure X_k >= 0 (though it will be naturally >= 0)
        # X_{k+1} >= X_k + d1(L_k)
        X_new = X + d1
        # Y_k >= X_k + d1(L_k) 
        Y_min = X + d1
        # Y_k should be at least Y_min and at least previous Y
        Y = max(Y, Y_min)
        # Y_{k+1} >= Y_k + d2(L_k)
        Y_new = Y + d2
        X = X_new
        Y = Y_new
    
    print(Y)

threading.Thread(target=main).start()
```
