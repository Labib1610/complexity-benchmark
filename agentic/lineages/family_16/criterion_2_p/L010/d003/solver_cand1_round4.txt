```python
import sys

def solve():
    # The constraints are:
    # X_{k+1} >= X_k + d1_k
    # Y_k >= X_k + d1_k
    # Y_{k+1} >= Y_k + d2_k
    # Target: Y_j + d2_j
    #
    # Let X_1 = 0.
    # X_k = sum_{i=1}^{k-1} d1_i
    # Y_k >= X_k + d1_k = sum_{i=1}^{k} d1_i
    # Y_{k+1} >= Y_k + d2_k
    #
    # By induction, Y_k >= max_{1 <= i <= k} (sum_{m=1}^{i-1} d1_m + d1_i + sum_{m=i}^{k-1} d2_m)
    # Let S1_i = sum_{m=1}^i d1_m, S2_i = sum_{m=1}^i d2_m.
    # Y_k >= S1_i + (S2_{k-1} - S2_{i-1})
    # Y_k >= S2_{k-1} + S1_i - S2_{i-1}
    # Y_k >= S2_{k-1} + max_{1 <= i <= k} (d1_i + S2_{i-1} - S2_{i-1}) is wrong.
    # Actually, Y_k = max_{1 <= i <= k} (S1_i + S2_{k-1} - S2_{i-1})
    # Target = Y_j + d2_j = max_{1 <= i <= j} (S1_i - S2_{i-1}) + S2_j
    # Target = max_{1 <= i <= j} (S1_i - S2_{i-1} + S2_j)
    # Since S2_j = S2_{j-1} + d2_j,
    # Target = max_{1 <= i <= j} (S1_i - S2_{i-1}) + S2_{j-1} + d2_j
    #
    # To minimize this, we need to order the pairs.
    # This is a classic scheduling problem. If we have two adjacent tasks (d1_a, d2_a) and (d1_b, d2_b),
    # the contribution to the max is determined by the order.
    # The optimal order is to sort by d2 in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    pairs = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        pairs.append((d1, d2))
        idx += 2
    
    # Sort by d2 descending
    pairs.sort(key=lambda x: x[1], reverse=True)
    
    s1 = 0
    s2 = 0
    max_val = -float('inf')
    
    # Target = max_{1 <= i <= j} (S1_i - S2_{i-1}) + S2_j
    # S1_i = S1_{i-1} + d1_i
    # S2_i = S2_{i-1} + d2_i
    # S1_i - S2_{i-1} = S1_{i-1} + d1_i - S2_{i-1}
    
    current_s1 = 0
    current_s2 = 0
    ans = -float('inf')
    
    for d1, d2 in pairs:
        # i-th pair (1-indexed)
        # S1_i = current_s1 + d1
        # S2_{i-1} = current_s2
        val = (current_s1 + d1) - current_s2
        if val > ans:
            ans = val
        current_s1 += d1
        current_s2 += d2
        
    print(ans + current_s2)

solve()
```
