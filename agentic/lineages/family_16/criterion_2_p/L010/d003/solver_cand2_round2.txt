import sys

def solve():
    # The problem asks to minimize Y_j + w(P_j).
    # Let the sequence be (u_1, w_1), (u_2, w_2), ..., (u_j, w_j).
    # The constraints are:
    # X_1 >= 0
    # X_{k+1} >= X_k + u_k
    # Y_k >= X_k + u_k
    # Y_{k+1} >= Y_k + w_k
    #
    # To minimize Y_j + w_j, we should set X_k and Y_k to their minimum possible values:
    # X_1 = 0
    # X_{k+1} = X_k + u_k => X_k = sum_{i=1}^{k-1} u_i
    # Y_1 = X_1 + u_1 = u_1
    # Y_{k+1} = max(Y_k + w_k, X_{k+1} + u_{k+1})
    #
    # Let S_k = sum_{i=1}^k u_i. Then X_{k+1} = S_k.
    # Y_1 = u_1 = S_1
    # Y_2 = max(Y_1 + w_1, S_1 + u_2) = max(S_1 + w_1, S_2)
    # Y_3 = max(Y_2 + w_2, S_2 + u_3) = max(S_1 + w_1 + w_2, S_2 + w_2, S_3)
    # By induction, Y_j = max(S_j, S_{j-1} + w_j, S_{j-2} + w_{j-1} + w_j, ..., S_1 + w_2 + ... + w_j)
    # The target value is Y_j + w_j = max(S_j + w_j, S_{j-1} + w_j + w_j, ..., S_1 + w_2 + ... + w_j + w_j)
    # This is equivalent to max_{1 <= k <= j} (S_{k-1} + sum_{i=k}^j w_i) where S_0 = 0.
    #
    # This is a classic scheduling problem. To minimize the maximum of prefix sums,
    # we sort the pairs based on the values of u and w.
    # Specifically, pairs with u < w should come first (sorted by u ascending),
    # and pairs with u >= w should come last (sorted by w descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    pairs = []
    idx = 1
    for _ in range(j):
        u = int(input_data[idx])
        w = int(input_data[idx+1])
        pairs.append((u, w))
        idx += 2
        
    # Johnson's rule logic for minimizing makespan:
    # Split into two groups: u < w and u >= w
    group1 = [p for p in pairs if p[0] < p[1]]
    group2 = [p for p in pairs if p[0] >= p[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate Y_j + w_j
    # Y_k = max(Y_{k-1} + w_{k-1}, S_k)
    # Target = Y_j + w_j
    
    current_x = 0
    current_y = 0
    for i in range(j):
        u, w = sequence[i]
        current_x += u
        if i == 0:
            current_y = current_x
        else:
            current_y = max(current_y + sequence[i-1][1], current_x)
            
    print(current_y + sequence[-1][1])

solve()
