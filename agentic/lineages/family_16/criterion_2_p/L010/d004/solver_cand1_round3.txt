```python
import sys

def solve():
    # The cost function is Cost(k) = sum_{i=1}^k T_i + sum_{i=k}^j B_i
    # Let S_T = sum_{i=1}^j T_i and S_B = sum_{i=1}^j B_i.
    # Cost(k) = sum_{i=1}^k T_i + sum_{i=k}^j B_i
    # Note that sum_{i=k}^j B_i = S_B - sum_{i=1}^{k-1} B_i
    # So Cost(k) = sum_{i=1}^k T_i + S_B - sum_{i=1}^{k-1} B_i
    # Cost(k) = S_B + T_k + sum_{i=1}^{k-1} (T_i - B_i)
    # Let D_i = T_i - B_i.
    # Cost(k) = S_B + T_k + sum_{i=1}^{k-1} D_i
    # Let P_k = sum_{i=1}^k D_i, with P_0 = 0.
    # Then Cost(k) = S_B + T_k + P_{k-1} = S_B + T_k + P_k - D_k
    # Since D_k = T_k - B_k, Cost(k) = S_B + T_k + P_k - (T_k - B_k) = S_B + P_k + B_k.
    # Also Cost(k) = S_B + P_{k-1} + T_k.
    # To minimize the maximum of these, we should sort the columns based on D_i = T_i - B_i.
    # If we sort such that D_1 <= D_2 <= ... <= D_j, then P_k is the prefix sum of sorted D_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    cols = []
    idx = 1
    s_b = 0
    for _ in range(j):
        t = int(input_data[idx])
        b = int(input_data[idx+1])
        cols.append((t, b, t - b))
        s_b += b
        idx += 2
        
    # Sort by D_i = T_i - B_i
    cols.sort(key=lambda x: x[2])
    
    # Now calculate max(Cost(k))
    # Cost(k) = S_B + P_{k-1} + T_k
    # P_0 = 0, P_k = P_{k-1} + D_k
    
    max_cost = 0
    p_prev = 0
    for i in range(j):
        t, b, d = cols[i]
        cost = s_b + p_prev + t
        if cost > max_cost:
            max_cost = cost
        p_prev += d
        
    print(max_cost)

if __name__ == '__main__':
    solve()
```
