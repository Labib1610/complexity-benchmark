```python
import sys

def solve():
    # The adversary's score for a split point k is:
    # S(k) = sum_{i=1}^k d1(C_i) + sum_{i=k}^j d2(C_i)
    # S(k) = sum_{i=1}^j d2(C_i) + sum_{i=1}^k (d1(C_i) - d2(C_i)) + d2(C_k)
    # Let D_i = d1(C_i) - d2(C_i).
    # S(k) = sum(d2) + sum_{i=1}^k D_i + d2(C_k)
    
    # To minimize the maximum S(k), we should sort the cards.
    # Let's consider the difference D_i = d1_i - d2_i.
    # If we sort cards such that D_i are in non-decreasing order,
    # we can minimize the peak of the prefix sums.
    # Specifically, sorting by d1_i - d2_i is a standard approach for this type of problem.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    cards = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        cards.append((d1, d2))
        idx += 2
        
    # Sort by d1 - d2
    cards.sort(key=lambda x: x[0] - x[1])
    
    # Calculate prefix sums of D_i
    # S(k) = sum(d2) + prefix_D[k] + d2(C_k)
    # where prefix_D[k] = sum_{i=1}^k (d1_i - d2_i)
    
    sum_d2 = sum(c[1] for c in cards)
    
    max_score = -float('inf')
    current_prefix_D = 0
    
    for i in range(j):
        current_prefix_D += (cards[i][0] - cards[i][1])
        score_k = sum_d2 + current_prefix_D + cards[i][1]
        if score_k > max_score:
            max_score = score_k
            
    print(max_score)

if __name__ == '__main__':
    solve()
```
