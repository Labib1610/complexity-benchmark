```python
import sys

def solve():
    # The problem asks to minimize D_j + q(M_j).
    # Let the sequence be M_1, M_2, ..., M_j.
    # C_1 >= 0
    # C_{k+1} >= C_k + p(M_k)
    # D_k >= C_k + p(M_k)
    # D_{k+1} >= D_k + q(M_k)
    #
    # To minimize the final boundary, we set C_1 = 0.
    # Then C_k = sum_{i=1}^{k-1} p(M_i).
    # The condition D_k >= C_k + p(M_k) becomes D_k >= sum_{i=1}^k p(M_i).
    # The condition D_{k+1} >= D_k + q(M_k) implies D_k >= D_1 + sum_{i=1}^{k-1} q(M_i).
    # Combining these, D_k >= max(sum_{i=1}^k p(M_i), D_1 + sum_{i=1}^{k-1} q(M_i)).
    # For k=1, D_1 >= p(M_1).
    # By induction, D_k = max(sum_{i=1}^k p(M_i), D_{k-1} + q(M_{k-1})).
    # Let S_p(k) = sum_{i=1}^k p(M_i) and S_q(k) = sum_{i=1}^k q(M_i).
    # D_k = max(S_p(1) + S_q(k-1), S_p(2) + S_q(k-2), ..., S_p(k)).
    # We want to minimize D_j + q(M_j) = max_{1 <= k <= j} (S_p(k) + S_q(j) - S_q(k-1)).
    # This is equivalent to minimizing max_{1 <= k <= j} (S_p(k) - S_q(k-1)) + S_q(j).
    # Since S_q(j) is constant regardless of the permutation, we minimize max_{1 <= k <= j} (S_p(k) - S_q(k-1)).
    # Let f(k) = S_p(k) - S_q(k-1) = p(M_k) + sum_{i=1}^{k-1} (p(M_i) - q(M_i)).
    # This is a classic scheduling problem. To minimize the maximum prefix sum,
    # we should sort the modules such that those with p_i < q_i come first (sorted by p_i ascending),
    # and those with p_i >= q_i come later (sorted by q_i descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    modules = []
    idx = 1
    for _ in range(j):
        p = int(input_data[idx])
        q = int(input_data[idx+1])
        modules.append((p, q))
        idx += 2
        
    group1 = [m for m in modules if m[0] < m[1]]
    group2 = [m for m in modules if m[0] >= m[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    seq = group1 + group2
    
    current_p_sum = 0
    max_val = 0
    total_q = 0
    
    for p, q in seq:
        # S_p(k) - S_q(k-1)
        val = current_p_sum + p - total_q
        if val > max_val:
            max_val = val
        current_p_sum += p
        total_q += q
        
    print(max_val + total_q)

if __name__ == '__main__':
    solve()
```
