```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    lenses = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        lenses.append((d1, d2))
        idx += 2
    
    # The footprint for a sequence L_1, ..., L_j at shift k is:
    # F(k) = sum_{i=1}^k d1(L_i) + sum_{i=k}^j d2(L_i)
    # F(k) = sum_{i=1}^j d2(L_i) + sum_{i=1}^k (d1(L_i) - d2(L_i))
    # Let S = sum_{i=1}^j d2(L_i).
    # F(k) = S + sum_{i=1}^k (d1(L_i) - d2(L_i))
    # Let diff_i = d1(L_i) - d2(L_i).
    # We want to minimize max_{k=1..j} (S + sum_{i=1}^k diff_i).
    # Since S is constant for any permutation, we want to minimize:
    # max_{k=1..j} (sum_{i=1}^k diff_i)
    # To minimize the prefix sums, we should sort the lenses by diff_i in ascending order.
    # If diff_i < 0, the prefix sums will decrease, keeping the max small.
    # If diff_i > 0, the prefix sums will increase, so we put these at the end.
    
    lenses.sort(key=lambda x: x[0] - x[1])
    
    s = sum(l[1] for l in lenses)
    
    current_prefix_diff = 0
    max_val = -float('inf')
    
    for i in range(j):
        current_prefix_diff += (lenses[i][0] - lenses[i][1])
        max_val = max(max_val, s + current_prefix_diff)
        
    print(max_val)

if __name__ == '__main__':
    solve()
```
