import sys

def solve():
    # The problem asks to minimize the final boundary value Y_j + d_2(L_j).
    # Let the sequence of lenses be L_1, L_2, ..., L_j.
    # Primary Axis:
    # X_1 >= 0
    # X_2 >= X_1 + d_1(L_1)
    # X_3 >= X_2 + d_1(L_2)
    # ...
    # X_k >= X_{k-1} + d_1(L_{k-1})
    # This implies X_k >= sum_{i=1}^{k-1} d_1(L_i).
    # To minimize, we set X_k = sum_{i=1}^{k-1} d_1(L_i).
    
    # Cross-Axis Alignment:
    # For each k, the Primary interval [X_k, X_k + d_1(L_k)] must precede
    # the Secondary interval [Y_k, Y_k + d_2(L_k)].
    # So Y_k >= X_k + d_1(L_k).
    
    # Secondary Axis Sequence:
    # Y_1 >= 0 (implied by context, though not explicitly stated, Y_k must be valid)
    # Y_2 >= Y_1 + d_2(L_1)
    # Y_3 >= Y_2 + d_2(L_2)
    # ...
    # Y_k >= Y_{k-1} + d_2(L_{k-1})
    
    # Combining these:
    # Y_k >= max(X_k + d_1(L_k), Y_{k-1} + d_2(L_{k-1}))
    # Let S_k = sum_{i=1}^k d_1(L_i). Then X_k = S_{k-1} (with S_0 = 0).
    # Y_k >= max(S_{k-1} + d_1(L_k), Y_{k-1} + d_2(L_{k-1}))
    # We want to minimize Y_j + d_2(L_j).
    
    # This is equivalent to finding a permutation that minimizes the final value.
    # Let f(k) = Y_k + d_2(L_k).
    # This structure is a classic scheduling problem.
    # The optimal order is to sort by d_1(L_i) + d_2(L_i) descending? 
    # Actually, let's re-examine:
    # Y_k = max(S_{k-1} + d_1(L_k), Y_{k-1} + d_2(L_{k-1}))
    # Final value = Y_j + d_2(L_j)
    # = max(S_{j-1} + d_1(L_j) + d_2(L_j), Y_{j-1} + d_2(L_{j-1}) + d_2(L_j))
    # This is minimized by sorting lenses by d_1(L_i) + d_2(L_i) in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    lenses = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        lenses.append((d1, d2))
        idx += 2
        
    # Sort by d1 + d2 descending
    lenses.sort(key=lambda x: x[0] + x[1], reverse=True)
    
    current_s = 0
    current_y = 0
    
    for i in range(j):
        d1, d2 = lenses[i]
        # Y_k = max(X_k + d1, Y_{k-1} + d2_{prev})
        # X_k = current_s
        if i == 0:
            current_y = d1 + d2
        else:
            current_y = max(current_s + d1 + d2, current_y + d2)
        current_s += d1
        
    print(current_y)

solve()
