import sys

def solve():
    # The problem asks to minimize T_final + sum(d1_i).
    # Let T_k be the temperature after regulator k.
    # T_k = max(0, T_{k-1} - d1_k) + d2_k
    # T_final = T_j
    # Footprint = T_j + sum(d1_i)
    #
    # Let's analyze the effect of the order.
    # Consider two adjacent regulators (d1_a, d2_a) and (d1_b, d2_b).
    # If we process a then b:
    # T_a = max(0, T - d1_a) + d2_a
    # T_b = max(0, T_a - d1_b) + d2_b
    #     = max(0, max(0, T - d1_a) + d2_a - d1_b) + d2_b
    #
    # This is a classic scheduling problem.
    # Split regulators into two groups:
    # Group 1: d1_i >= d2_i (cooling capacity >= heat output)
    # Group 2: d1_i < d2_i (cooling capacity < heat output)
    #
    # For Group 1, we want to process them in descending order of d1_i.
    # For Group 2, we want to process them in ascending order of d2_i.
    # Group 1 comes before Group 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    regulators = []
    idx = 1
    sum_d1 = 0
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        regulators.append((d1, d2))
        sum_d1 += d1
        idx += 2
        
    group1 = []
    group2 = []
    for d1, d2 in regulators:
        if d1 >= d2:
            group1.append((d1, d2))
        else:
            group2.append((d1, d2))
            
    # Sort group1 by d1 descending
    group1.sort(key=lambda x: x[0], reverse=True)
    # Sort group2 by d2 ascending
    group2.sort(key=lambda x: x[1])
    
    sequence = group1 + group2
    
    t = 0
    for d1, d2 in sequence:
        t = max(0, t - d1) + d2
        
    print(t + sum_d1)

if __name__ == '__main__':
    solve()
