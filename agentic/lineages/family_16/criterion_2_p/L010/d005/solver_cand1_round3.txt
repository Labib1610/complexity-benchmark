```python
import sys

def solve():
    # The problem asks to minimize S_j + w_2(C_j).
    # From the constraints:
    # P_{k+1} >= P_k + w_1(C_k)
    # S_k >= P_k + w_1(C_k)
    # S_{k+1} >= S_k + w_2(C_k)
    #
    # Let P_1 = 0.
    # P_k = sum_{i=1}^{k-1} w_1(C_i)
    # S_k >= P_k + w_1(C_k) = sum_{i=1}^{k} w_1(C_i)
    # Also S_{k+1} >= S_k + w_2(C_k)
    # By induction, S_j >= S_k + sum_{i=k}^{j-1} w_2(C_i)
    # S_j >= sum_{i=1}^{k} w_1(C_i) + sum_{i=k}^{j-1} w_2(C_i)
    # We want to minimize S_j + w_2(C_j).
    # S_j + w_2(C_j) >= sum_{i=1}^{k} w_1(C_i) + sum_{i=k}^{j} w_2(C_i)
    # This must hold for all k in 1..j.
    # Let W1 = sum(w_1), W2 = sum(w_2).
    # The expression is sum_{i=1}^{j} w_1(C_i) + sum_{i=k}^{j} w_2(C_i) - sum_{i=k}^{j} w_1(C_i)
    # Actually, let's simplify:
    # S_j >= sum_{i=1}^{k-1} w_1(C_i) + w_1(C_k) + sum_{i=k}^{j-1} w_2(C_i)
    # S_j + w_2(C_j) >= sum_{i=1}^{k-1} w_1(C_i) + w_1(C_k) + sum_{i=k}^{j} w_2(C_i)
    # Let f(k) = sum_{i=1}^{k-1} w_1(C_i) + w_1(C_k) + sum_{i=k}^{j} w_2(C_i)
    # We want to minimize max_k f(k).
    # This is a classic scheduling problem. To minimize the maximum completion time,
    # we should sort components based on the difference w_1 - w_2.
    # Specifically, if w_1 < w_2, put them earlier.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    comps = []
    idx = 1
    for _ in range(j):
        w1 = int(input_data[idx])
        w2 = int(input_data[idx+1])
        comps.append((w1, w2))
        idx += 2
        
    # Sort by w1 < w2 (ascending) and w1 >= w2 (descending)
    # This is equivalent to sorting by w1 - w2.
    comps.sort(key=lambda x: x[0] - x[1])
    
    # Now calculate the max value of sum_{i=1}^{k-1} w_1(C_i) + w_1(C_k) + sum_{i=k}^{j} w_2(C_i)
    # Let P_sum[k] = sum_{i=1}^k w_1(C_i)
    # Let S_sum[k] = sum_{i=k}^j w_2(C_i)
    # We want max_{k=1..j} (P_sum[k-1] + w_1(C_k) + S_sum[k])
    # Note P_sum[k-1] + w_1(C_k) = P_sum[k]
    # So we want max_{k=1..j} (P_sum[k] + S_sum[k])
    
    p_sum = 0
    s_sum = sum(c[1] for c in comps)
    max_val = 0
    for i in range(j):
        p_sum += comps[i][0]
        # s_sum is sum_{i=k}^j w_2(C_i)
        current_val = p_sum + s_sum
        if current_val > max_val:
            max_val = current_val
        s_sum -= comps[i][1]
        
    print(max_val)

solve()
```
