```python
import sys

def solve():
    # The problem asks to minimize the rightmost coordinate of the Beta track.
    # Let the sequence of brackets be B_1, B_2, ..., B_j.
    # Let d1_i and d2_i be the Alpha and Beta spans of bracket B_i.
    # Let A_i be the start coordinate of the Alpha footprint of B_i.
    # Let B_i be the start coordinate of the Beta footprint of B_i.
    # Constraints:
    # 1. A_{i+1} >= A_i + d1_i
    # 2. B_{i+1} >= B_i + d2_i
    # 3. B_i >= A_i + d1_i
    #
    # To minimize the end of the last Beta footprint (B_j + d2_j):
    # A_1 = 0
    # A_i = sum_{k=1}^{i-1} d1_k
    # B_1 >= A_1 + d1_1 = d1_1
    # B_i >= max(B_{i-1} + d2_{i-1}, A_i + d1_i)
    #
    # Let S1_i = sum_{k=1}^i d1_k and S2_i = sum_{k=1}^i d2_k.
    # B_i >= S1_i + d1_i (Wait, this is wrong. A_i = S1_{i-1})
    # B_i >= S1_{i-1} + d1_i = S1_i
    # Also B_i >= B_{i-1} + d2_{i-1}
    # By induction, B_i >= max_{1 <= k <= i} (S1_k + sum_{m=k}^{i-1} d2_m)
    # The final coordinate is B_j + d2_j >= max_{1 <= k <= j} (S1_k + sum_{m=k}^j d2_m)
    # Let S2_0 = 0. Then sum_{m=k}^j d2_m = S2_j - S2_{k-1}.
    # Final coordinate >= max_{1 <= k <= j} (S1_k - S2_{k-1}) + S2_j.
    # To minimize this, we need to minimize max_{1 <= k <= j} (S1_k - S2_{k-1}).
    # This is equivalent to ordering the brackets to minimize the peak of the prefix sums.
    # This is a classic scheduling problem: sort by d1_i vs d2_i.
    # Specifically, sort by d1_i < d2_i first, then d1_i >= d2_i.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    brackets = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        brackets.append((d1, d2))
        idx += 2
        
    # Sort brackets:
    # If d1 < d2, put it earlier. If d1 >= d2, put it later.
    # Within d1 < d2, sort by increasing d1.
    # Within d1 >= d2, sort by decreasing d2.
    brackets.sort(key=lambda x: (x[0] >= x[1], x[0] if x[0] < x[1] else -x[1]))
    
    s1 = 0
    s2 = 0
    max_val = -float('inf')
    
    for d1, d2 in brackets:
        # Current B_i >= S1_i
        # We want to minimize max(S1_k - S2_{k-1})
        # At each step k, we have S1_k and S2_{k-1}
        val = s1 + d1 - s2
        if val > max_val:
            max_val = val
        s1 += d1
        s2 += d2
        
    print(max_val + s2)

solve()
```
