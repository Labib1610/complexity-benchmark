```python
import sys

def solve():
    # The problem asks to minimize max_{1 <= m <= j} (sum_{i=1}^m w1(v_i) + sum_{i=m}^j w2(v_i))
    # This is equivalent to minimizing (sum_{i=1}^j w1(v_i) + sum_{i=m}^j w2(v_i) - sum_{i=m}^j w1(v_i))
    # Let S1 = sum(w1), S2 = sum(w2).
    # The expression is S1 + sum_{i=m}^j (w2(v_i) - w1(v_i)).
    # To minimize this, we want to order the nodes such that the prefix sums of (w2 - w1)
    # are as small as possible, or more simply, use Johnson's rule logic.
    # The expression is equivalent to:
    # sum(w1) + max(0, max_{m=1..j} (sum_{i=m}^j (w2(v_i) - w1(v_i))))
    # This is a classic scheduling problem (minimizing makespan on two machines).
    # The optimal order is:
    # Nodes with w1 < w2 sorted by w1 ascending.
    # Nodes with w1 >= w2 sorted by w2 descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    nodes = []
    idx = 1
    for _ in range(j):
        w1 = int(input_data[idx])
        w2 = int(input_data[idx+1])
        nodes.append((w1, w2))
        idx += 2
        
    group1 = [n for n in nodes if n[0] < n[1]]
    group2 = [n for n in nodes if n[0] >= n[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate the terminal boundary for the sequence
    # Boundary = max_{m=1..j} (sum_{i=1}^m w1(v_i) + sum_{i=m}^j w2(v_i))
    # Let P1[m] be prefix sum of w1, S2[m] be suffix sum of w2.
    # Boundary = max_{m=1..j} (P1[m] + S2[m])
    
    current_w1_sum = 0
    total_w2_sum = sum(n[1] for n in sequence)
    
    max_boundary = 0
    current_w2_suffix = total_w2_sum
    
    for w1, w2 in sequence:
        current_w1_sum += w1
        # At this position m, the sum is current_w1_sum + current_w2_suffix
        boundary = current_w1_sum + current_w2_suffix
        if boundary > max_boundary:
            max_boundary = boundary
        current_w2_suffix -= w2
        
    print(max_boundary)

if __name__ == '__main__':
    solve()
```
