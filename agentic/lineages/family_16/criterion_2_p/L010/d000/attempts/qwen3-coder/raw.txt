import sys
from functools import cmp_to_key

def compare_jobs(job1, job2):
    # job = (p1, p2, index)
    # This is Johnson's rule comparison
    # If job1 should come before job2, return -1
    # If job2 should come before job1, return 1
    # If equal, return 0
    
    p1_1, p2_1 = job1[0], job1[1]
    p1_2, p2_2 = job2[0], job2[1]
    
    # Compare p1_1 + min(p2_1, p1_2) with p1_2 + min(p2_2, p1_1)
    # Actually, let's use the correct Johnson's rule:
    # Job1 < Job2 if min(p2_1, p1_2) > min(p2_2, p1_1) ? No, that's not right either.
    
    # Correct Johnson's rule for two-machine flow shop:
    # Job1 should come before job2 if min(p2_1, p1_2 + p2_2) <= min(p2_2, p1_1 + p2_1)
    # But actually the standard Johnson's rule is simpler:
    # Partition jobs into two sets:
    # Set1: jobs with p1 <= p2
    # Set2: jobs with p1 > p2
    # Sort Set1 by p1 ascending
    # Sort Set2 by p2 descending
    
    # Let me implement the correct comparison function:
    if p1_1 <= p2_1 and p1_2 <= p2_2:
        # Both in set1, sort by p1 ascending
        return p1_1 - p1_2
    elif p1_1 > p2_1 and p1_2 > p2_2:
        # Both in set2, sort by p2 descending
        return p2_2 - p2_1
    elif p1_1 <= p2_1 and p1_2 > p2_2:
        # job1 in set1, job2 in set2, job1 comes first
        return -1
    elif p1_1 > p2_1 and p1_2 <= p2_2:
        # job1 in set2, job2 in set1, job2 comes first
        return 1
    elif p1_1 <= p2_1:  # both p1_1 <= p2_1 and p1_2 <= p2_2 cannot be true if we're here
        # job1 in set1, job2 in set1
        return p1_1 - p1_2
    else:
        # job1 in set2, job2 in set2
        return p2_2 - p2_1

def solve():
    j = int(sys.stdin.readline())
    jobs = []
    
    for i in range(j):
        p1, p2 = map(int, sys.stdin.readline().split())
        jobs.append((p1, p2))
    
    # Apply Johnson's algorithm
    # Split jobs into two groups:
    set1 = []  # jobs where p1 <= p2
    set2 = []  # jobs where p1 > p2
    
    for i, (p1, p2) in enumerate(jobs):
        if p1 <= p2:
            set1.append((p1, p2, i))
        else:
            set2.append((p1, p2, i))
    
    # Sort set1 by p1 ascending
    set1.sort(key=lambda x: x[0])
    
    # Sort set2 by p2 descending
    set2.sort(key=lambda x: x[1], reverse=True)
    
    # Concatenate set1 and set2
    ordered_jobs = set1 + set2
    
    # Simulate the schedule
    time_m1 = 0  # current time on machine 1
    time_m2 = 0  # current time on machine 2
    
    for p1, p2, idx in ordered_jobs:
        # Start time on machine 1
        start_m1 = time_m1
        
        # Finish time on machine 1
        finish_m1 = start_m1 + p1
        time_m1 = finish_m1
        
        # Machine 2 can start only after machine 1 finishes AND 
        # after the previous job on machine 2 finishes
        start_m2 = max(finish_m1, time_m2)
        finish_m2 = start_m2 + p2
        time_m2 = finish_m2
    
    print(time_m2)

solve()
