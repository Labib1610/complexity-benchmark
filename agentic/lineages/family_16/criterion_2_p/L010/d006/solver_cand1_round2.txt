```python
import sys

def solve():
    # The problem asks to minimize max_{1 <= k <= j} (sum_{i=1}^k d1_i + sum_{i=k}^j d2_i)
    # Let S1_k = sum_{i=1}^k d1_i and S2_k = sum_{i=k}^j d2_i.
    # We want to minimize max_k (S1_k + S2_k).
    # Let D = sum_{i=1}^j d1_i. Then S1_k + S2_k = sum_{i=1}^k d1_i + sum_{i=k}^j d2_i.
    # This is equivalent to (sum_{i=1}^j d1_i) + sum_{i=k}^j d2_i - sum_{i=k+1}^j d1_i.
    # Let's rephrase: we want to minimize max_k (sum_{i=1}^{k-1} d1_i + d1_k + sum_{i=k}^j d2_i).
    # This is a classic scheduling problem. We want to order items to minimize the maximum
    # completion time. The optimal strategy is to sort brackets based on d1 and d2.
    # Specifically, if we have two brackets (a1, b1) and (a2, b2), we should place
    # (a1, b1) before (a2, b2) if a1 + b1 + b2 < a2 + b2 + b1, which simplifies to a1 - b1 < a2 - b2.
    # Wait, the formula is max_k (sum_{i=1}^k d1_i + sum_{i=k}^j d2_i).
    # Let f(k) = sum_{i=1}^k d1_i + sum_{i=k}^j d2_i.
    # f(k) - f(k+1) = (sum_{i=1}^k d1_i + sum_{i=k}^j d2_i) - (sum_{i=1}^{k+1} d1_i + sum_{i=k+1}^j d2_i)
    # = d2_k - d1_{k+1}.
    # To minimize the maximum, we should sort such that d1_i < d2_i come first (sorted by d1 ascending),
    # and d1_i >= d2_i come last (sorted by d2 descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    brackets = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        brackets.append((d1, d2))
        idx += 2
        
    # Sort: d1 < d2 first (ascending d1), then d1 >= d2 (descending d2)
    group1 = sorted([b for b in brackets if b[0] < b[1]], key=lambda x: x[0])
    group2 = sorted([b for b in brackets if b[0] >= b[1]], key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate the max value
    current_s1 = 0
    total_s2 = sum(b[1] for b in sequence)
    
    max_val = 0
    current_s2 = total_s2
    for d1, d2 in sequence:
        current_s1 += d1
        max_val = max(max_val, current_s1 + current_s2)
        current_s2 -= d2
        
    print(max_val)

if __name__ == '__main__':
    solve()
```
