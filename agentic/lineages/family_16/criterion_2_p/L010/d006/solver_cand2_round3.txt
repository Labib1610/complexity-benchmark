```python
import sys

def solve():
    # The pivot sum at index i is:
    # P(i) = sum_{k=1}^i d1(C_k) + sum_{k=i}^j d2(C_k)
    # Let S1 = sum_{k=1}^j d1(C_k) and S2 = sum_{k=1}^j d2(C_k).
    # P(i) = sum_{k=1}^i d1(C_k) + sum_{k=1}^j d2(C_k) - sum_{k=1}^{i-1} d2(C_k)
    # P(i) = S2 + sum_{k=1}^i (d1(C_k) - d2(C_{k-1})) where d2(C_0) = 0.
    # Let x_k = d1(C_k) and y_k = d2(C_k).
    # P(i) = sum_{k=1}^i x_k + sum_{k=i}^j y_k
    # P(i) = (sum_{k=1}^j y_k) + sum_{k=1}^i (x_k - y_k)
    # To minimize the maximum P(i), we want to arrange components such that
    # the prefix sums of (x_k - y_k) are kept as small as possible.
    # This is a classic scheduling problem. If we sort components by (x_k - y_k),
    # we can minimize the peak. Specifically, components with x_k < y_k should
    # come first (sorted by x_k ascending), and components with x_k >= y_k
    # should come last (sorted by y_k descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    comps = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        comps.append((d1, d2))
        idx += 2
        
    # Group 1: d1 < d2, sort by d1 ascending
    # Group 2: d1 >= d2, sort by d2 descending
    g1 = [c for c in comps if c[0] < c[1]]
    g2 = [c for c in comps if c[0] >= c[1]]
    
    g1.sort(key=lambda x: x[0])
    g2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = g1 + g2
    
    # Calculate peak value
    # P(i) = sum_{k=1}^i d1(C_k) + sum_{k=i}^j d2(C_k)
    # Let total_d2 = sum(d2 for all components)
    # P(i) = sum_{k=1}^i d1(C_k) + total_d2 - sum_{k=1}^{i-1} d2(C_k)
    
    total_d2 = sum(c[1] for c in sequence)
    current_d1_sum = 0
    current_d2_sum = 0
    max_p = 0
    
    for i in range(j):
        d1, d2 = sequence[i]
        current_d1_sum += d1
        # P(i+1) = current_d1_sum + (total_d2 - current_d2_sum)
        p_i = current_d1_sum + total_d2 - current_d2_sum
        if p_i > max_p:
            max_p = p_i
        current_d2_sum += d2
        
    print(max_p)

if __name__ == '__main__':
    solve()
```
