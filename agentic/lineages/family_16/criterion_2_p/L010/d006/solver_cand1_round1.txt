```python
import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    shifts = []
    idx = 1
    total_d1 = 0
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        shifts.append((d1, d2))
        total_d1 += d1
        idx += 2
    
    # A_k = T + d2(S_k) + sum_{i=k+1}^j (d2(S_i) - d1(S_i))
    # Let x_i = d2(S_i) - d1(S_i)
    # A_k = T + d2(S_k) + sum_{i=k+1}^j x_i
    # Let S_k = sum_{i=1}^k x_i. Then sum_{i=k+1}^j x_i = S_j - S_k
    # A_k = T + d2(S_k) + S_j - S_k
    # Since S_k = S_{k-1} + x_k, then S_k - x_k = S_{k-1}
    # A_k = T + d2(S_k) + S_j - (S_{k-1} + x_k)
    # A_k = T + S_j + d2(S_k) - d2(S_k) + d1(S_k) - S_{k-1}
    # A_k = T + S_j + d1(S_k) - S_{k-1}
    # Actually, let's simplify:
    # A_k = T + d2(S_k) + (sum_{i=1}^j x_i) - (sum_{i=1}^k x_i)
    # A_k = T + d2(S_k) + (sum_{i=1}^j (d2_i - d1_i)) - (sum_{i=1}^k (d2_i - d1_i))
    # A_k = T + d2(S_k) + (sum_{i=1}^j d2_i - T) - (sum_{i=1}^k (d2_i - d1_i))
    # A_k = sum_{i=1}^j d2_i + d2(S_k) - sum_{i=1}^k (d2_i - d1_i)
    # A_k = sum_{i=1}^j d2_i + d1(S_k) - sum_{i=1}^{k-1} (d2_i - d1_i)
    
    # Let's re-evaluate:
    # A_k = T + d2(S_k) + sum_{i=k+1}^j (d2_i - d1_i)
    # A_k = T + d2(S_k) + (sum_{i=1}^j (d2_i - d1_i)) - (sum_{i=1}^k (d2_i - d1_i))
    # A_k = T + d2(S_k) + sum_{i=1}^j d2_i - T - sum_{i=1}^k d2_i + sum_{i=1}^k d1_i
    # A_k = sum_{i=1}^j d2_i + d1(S_k) - sum_{i=1}^{k-1} (d2_i - d1_i)
    
    # To minimize max(A_k), we should sort shifts such that d1_i - d2_i is increasing.
    # If we sort by (d1_i - d2_i), we minimize the maximum value.
    
    shifts.sort(key=lambda x: x[0] - x[1])
    
    sum_d2 = sum(s[1] for s in shifts)
    
    # A_k = sum_d2 + d1(S_k) - sum_{i=1}^{k-1} (d2_i - d1_i)
    # Let current_diff_sum = sum_{i=1}^{k-1} (d2_i - d1_i)
    # A_k = sum_d2 + d1(S_k) - current_diff_sum
    
    max_a = -float('inf')
    current_diff_sum = 0
    for d1, d2 in shifts:
        a_k = sum_d2 + d1 - current_diff_sum
        if a_k > max_a:
            max_a = a_k
        current_diff_sum += (d2 - d1)
        
    print(max_a)

solve()
```
