import sys

def solve():
    # The problem asks to minimize max(P_k) where P_k = B_{k-1} + d1_k.
    # B_k = B_{k-1} + d1_k - d2_k.
    # Let S = sum(d2_i).
    # B_0 = S.
    # B_1 = S + d1_1 - d2_1
    # B_2 = S + d1_1 - d2_1 + d1_2 - d2_2
    # P_1 = S + d1_1
    # P_2 = S + d1_1 - d2_1 + d1_2
    # P_k = S + sum_{i=1}^{k-1} (d1_i - d2_i) + d1_k
    # Let diff_i = d1_i - d2_i.
    # P_k = S + sum_{i=1}^{k-1} diff_i + d1_k
    
    # We want to minimize max(P_k).
    # This is a classic scheduling problem.
    # Items with d1_i - d2_i >= 0 should be processed first, sorted by d2_i ascending.
    # Items with d1_i - d2_i < 0 should be processed last, sorted by d1_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    items = []
    idx = 1
    s = 0
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        items.append((d1, d2))
        s += d2
        idx += 2
        
    pos = []
    neg = []
    for d1, d2 in items:
        if d1 - d2 >= 0:
            pos.append((d1, d2))
        else:
            neg.append((d1, d2))
            
    # Sort pos by d2 ascending
    pos.sort(key=lambda x: x[1])
    # Sort neg by d1 descending
    neg.sort(key=lambda x: x[0], reverse=True)
    
    order = pos + neg
    
    max_p = 0
    current_b = s
    for d1, d2 in order:
        p = current_b + d1
        if p > max_p:
            max_p = p
        current_b = current_b + d1 - d2
        
    print(max_p)

if __name__ == '__main__':
    solve()
