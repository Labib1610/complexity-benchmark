```python
import sys
from itertools import accumulate

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    brackets = []
    index = 1
    for _ in range(j):
        d1 = int(data[index])
        d2 = int(data[index+1])
        brackets.append((d1, d2))
        index += 2
    
    # According to the note, for a fixed sequence B1,...,Bj,
    # the minimum rightmost coordinate is:
    # max_{1 <= k <= j} ( sum_{i=1}^k d1(B_i) + sum_{i=k}^j d2(B_i) )
    
    # To minimize this, we need to find the optimal sequence.
    # The optimal sequence is to sort the brackets in such a way 
    # that this maximum is minimized. This is achieved by sorting 
    # based on the difference d1 - d2 in non-decreasing order.
    # That is, brackets with smaller (d1 - d2) should come first.
    
    brackets.sort(key=lambda x: x[0] - x[1])
    
    # Now compute the value using the sorted sequence
    d1_list = [b[0] for b in brackets]
    d2_list = [b[1] for b in brackets]
    
    # Precompute prefix sums of d1
    prefix_d1 = [0] + list(accumulate(d1_list))
    # Precompute suffix sums of d2
    suffix_d2 = list(accumulate(d2_list[::-1]))[::-1] + [0]
    
    # Compute the maximum of prefix_d1[k] + suffix_d2[k] for k in 1..j
    max_coordinate = 0
    for k in range(1, j+1):
        coord = prefix_d1[k] + suffix_d2[k]
        if coord > max_coordinate:
            max_coordinate = coord
            
    print(max_coordinate)

if __name__ == "__main__":
    main()
```
