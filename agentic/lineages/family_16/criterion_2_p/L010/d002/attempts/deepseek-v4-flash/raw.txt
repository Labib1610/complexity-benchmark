We can model this as a scheduling problem.  
The key observation is that the optimal order can be found by sorting the jobs by a certain rule, then computing the result in a single pass.

```python
import sys

def solve():
    input = sys.stdin.readline
    j = int(input())
    jobs = []
    for _ in range(j):
        d1, d2 = map(int, input().split())
        jobs.append((d1, d2))
    
    # Sort by d2 descending, then d1 ascending
    jobs.sort(key=lambda x: (-x[1], x[0]))
    
    X = 0
    Y = 0
    for d1, d2 in jobs:
        X += d1
        if Y < X:
            Y = X
        Y += d2
    
    print(Y)

if __name__ == "__main__":
    solve()
```
