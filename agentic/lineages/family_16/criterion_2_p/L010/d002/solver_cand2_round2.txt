import sys

def solve():
    # This is a classic Johnson's Rule problem for a two-machine flow shop.
    # To minimize the makespan, we sort the jobs based on the following criteria:
    # 1. Jobs with p1 < p2 should be processed in increasing order of p1.
    # 2. Jobs with p1 >= p2 should be processed in decreasing order of p2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Group 1: p1 < p2, sorted by p1 ascending
    # Group 2: p1 >= p2, sorted by p2 descending
    group1 = []
    group2 = []
    for p1, p2 in jobs:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate the makespan
    # S_k is the start time on machine 1, A_k is the start time on machine 2
    # S_1 = 0
    # S_{k+1} = S_k + p1_k
    # A_k = max(S_k + p1_k, A_{k-1} + p2_{k-1}) for k > 1
    # A_1 = S_1 + p1_1 = p1_1
    
    current_s = 0
    current_a = 0
    
    for p1, p2 in sequence:
        current_s += p1
        if current_a < current_s:
            current_a = current_s
        current_a += p2
        
    print(current_a)

if __name__ == '__main__':
    solve()
