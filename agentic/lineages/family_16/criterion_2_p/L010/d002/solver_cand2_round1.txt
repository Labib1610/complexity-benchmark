```python
import sys

def solve():
    # The problem asks to minimize Y_j + p_2(C_j).
    # Let's analyze the constraints:
    # X_{k+1} >= X_k + p_1(C_k)
    # Y_{k+1} >= Y_k + p_2(C_k)
    # Y_k >= X_k + p_1(C_k)
    #
    # Let X_1 = 0.
    # Then X_k = sum_{i=1}^{k-1} p_1(C_i) for k > 1.
    # Y_k >= X_k + p_1(C_k) = sum_{i=1}^{k-1} p_1(C_i) + p_1(C_k) = sum_{i=1}^k p_1(C_i).
    # Also Y_{k+1} >= Y_k + p_2(C_k).
    #
    # Let S_k = sum_{i=1}^k p_1(C_i).
    # Y_1 >= S_1
    # Y_2 >= max(Y_1 + p_2(C_1), S_2)
    # Y_3 >= max(Y_2 + p_2(C_2), S_3)
    # ...
    # Y_j >= max(Y_{j-1} + p_2(C_{j-1}), S_j)
    # Terminal value = Y_j + p_2(C_j).
    #
    # This recurrence implies Y_j = max(S_1 + p_2(C_1) + p_2(C_2) + ... + p_2(C_{j-1}),
    #                                  S_2 + p_2(C_2) + ... + p_2(C_{j-1}),
    #                                  ...,
    #                                  S_j)
    # Terminal value = max(S_1 + sum_{i=1}^j p_2(C_i), S_2 + sum_{i=2}^j p_2(C_i), ..., S_j + p_2(C_j))
    # Terminal value = max_{1 <= k <= j} (S_k + sum_{i=k}^j p_2(C_i))
    #
    # Let P1_i = p_1(C_i), P2_i = p_2(C_i).
    # Terminal value = max_{1 <= k <= j} (sum_{i=1}^k P1_i + sum_{i=k}^j P2_i)
    # Terminal value = sum_{i=1}^j P2_i + max_{1 <= k <= j} (sum_{i=1}^k P1_i - sum_{i=1}^{k-1} P2_i)
    #
    # To minimize this, we want to order the components.
    # Let f(k) = P1_k - P2_k.
    # The expression is sum(P2) + max(P1_1, P1_1+P1_2-P2_1, P1_1+P1_2+P1_3-P2_1-P2_2, ...)
    # This is equivalent to finding a permutation that minimizes the prefix sums of (P1_i - P2_i).
    # Actually, the optimal strategy is to sort by P2_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    comps = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        comps.append((p1, p2))
        idx += 2
    
    # Sort by p2 descending
    comps.sort(key=lambda x: x[1], reverse=True)
    
    p1_sum = 0
    p2_sum = sum(c[1] for c in comps)
    
    max_val = 0
    current_p1_sum = 0
    current_p2_sum = 0
    
    for i in range(j):
        p1, p2 = comps[i]
        # S_k = current_p1_sum + p1
        # sum_{i=k}^j P2_i = p2_sum - current_p2_sum
        val = (current_p1_sum + p1) + (p2_sum - current_p2_sum)
        if i == 0 or val > max_val:
            max_val = val
        current_p1_sum += p1
        current_p2_sum += p2
        
    print(max_val)

solve()
```
