import sys

def solve():
    # The problem asks to minimize Y_j + d_2(O_j).
    # Let X_1 = 0.
    # X_{k+1} = X_k + d_1(O_k) => X_k = sum_{i=1}^{k-1} d_1(O_i)
    # Y_1 = X_1 + d_1(O_1) = d_1(O_1)
    # Y_{k+1} = max(Y_k + d_2(O_k), X_{k+1} + d_1(O_{k+1}))
    # This is a classic Johnson's Rule variation.
    # The target value is Y_j + d_2(O_j).
    # Let's re-examine the constraints:
    # Y_k >= X_k + d_1(O_k)
    # Y_{k+1} >= Y_k + d_2(O_k)
    # This is equivalent to the flow shop scheduling problem with 2 machines
    # where we want to minimize the makespan.
    # The makespan for 2 machines is min_{permutation} max_{k} (sum_{i=1}^k d_1(O_i) + sum_{i=k}^j d_2(O_i))
    # This is minimized by Johnson's rule:
    # Sort orders such that those with d_1 < d_2 come first (sorted by d_1 ascending),
    # and those with d_1 >= d_2 come last (sorted by d_2 descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    orders = []
    idx = 1
    for _ in range(j):
        d1 = int(input_data[idx])
        d2 = int(input_data[idx+1])
        orders.append((d1, d2))
        idx += 2
        
    group1 = []
    group2 = []
    for d1, d2 in orders:
        if d1 < d2:
            group1.append((d1, d2))
        else:
            group2.append((d1, d2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate makespan
    # X_k is the completion time on machine 1
    # Y_k is the completion time on machine 2
    x = 0
    y = 0
    for d1, d2 in sequence:
        x += d1
        y = max(y, x) + d2
        
    print(y)

if __name__ == '__main__':
    solve()
