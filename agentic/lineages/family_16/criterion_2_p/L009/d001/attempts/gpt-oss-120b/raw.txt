```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    early = []  # p1 <= p2
    late = []   # p1 > p2
    for _ in range(n):
        p1 = int(next(it)); p2 = int(next(it))
        if p1 <= p2:
            early.append((p1, p2))
        else:
            late.append((p1, p2))
    early.sort(key=lambda x: x[0])          # increasing p1
    late.sort(key=lambda x: x[1], reverse=True)  # decreasing p2
    seq = early + late

    total_p2 = sum(p2 for _, p2 in seq)
    pref1 = 0
    suffix_p2 = total_p2
    ans = 0
    for p1, p2 in seq:
        pref1 += p1
        suffix_p2 -= p2
        # split point k is current position, lighting includes current effect
        cur = pref1 + suffix_p2 + p2
        if cur > ans:
            ans = cur
    print(ans)

if __name__ == "__main__":
    solve()
```
