import sys
import threading

def main():
    import sys
    j = int(sys.stdin.readline())
    effects = []
    for _ in range(j):
        p1, p2 = map(int, sys.stdin.readline().split())
        effects.append((p1, p2))
    
    # Sort effects by p1 - p2 in descending order
    # This is the optimal ordering to minimize the worst-case total time
    effects.sort(key=lambda x: x[0] - x[1], reverse=True)
    
    # Now compute the worst-case total time for this ordering
    # We need to find max over all k of (sum of p1 from 1 to k) + (sum of p2 from k to j)
    
    # Precompute prefix sums of p1
    prefix_p1 = [0]
    for p1, _ in effects:
        prefix_p1.append(prefix_p1[-1] + p1)
    
    # Precompute suffix sums of p2
    suffix_p2 = [0]
    for i in range(j-1, -1, -1):
        _, p2 = effects[i]
        suffix_p2.append(suffix_p2[-1] + p2)
    suffix_p2.reverse()
    
    # Find the maximum total time over all possible split points k
    max_time = 0
    for k in range(1, j+1):
        total_time = prefix_p1[k] + suffix_p2[k-1]
        max_time = max(max_time, total_time)
    
    print(max_time)

# Run the main function in a thread with increased stack size
threading.Thread(target=main).start()
