```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have j jobs, each with two phases (transcription p1, binding p2).
    # We want to minimize the makespan (completion time of the last job).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 2. Sort Set A in ascending order of p1.
    # 3. Sort Set B in descending order of p2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate completion time
    # transcription_time: time when calligrapher finishes current job
    # binding_time: time when bookbinder finishes current job
    transcription_time = 0
    binding_time = 0
    
    for p1, p2 in sequence:
        transcription_time += p1
        # Binding can start only after transcription is done AND 
        # the bookbinder is free from the previous job.
        if binding_time < transcription_time:
            binding_time = transcription_time + p2
        else:
            binding_time += p2
            
    print(binding_time)

if __name__ == '__main__':
    solve()
```
