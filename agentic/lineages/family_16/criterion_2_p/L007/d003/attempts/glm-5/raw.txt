```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    manuscripts = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        manuscripts.append((p1, p2))
        idx += 2
    
    # Sort by conservator time descending, then by historian time ascending
    manuscripts.sort(key=lambda x: (-x[1], x[0]))
    
    historian_time = 0
    completion_time = 0
    
    for p1, p2 in manuscripts:
        historian_time += p1
        # Conservator starts when both historian finishes this manuscript
        # and conservator is free (completion_time)
        start = max(historian_time, completion_time)
        completion_time = start + p2
    
    print(completion_time)

solve()
```
