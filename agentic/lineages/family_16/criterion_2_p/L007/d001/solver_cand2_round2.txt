```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    devices = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        devices.append((p1, p2))
        idx += 2
    
    # Let the sequence be (a1, b1), (a2, b2), ..., (aj, bj).
    # Total radiation for transition k is:
    # S_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # S_k = sum_{i=1}^j b_i + sum_{i=1}^k (a_i - b_i) + b_k
    # Let D_i = a_i - b_i.
    # S_k = sum(b) + sum_{i=1}^{k-1} D_i + a_k
    
    # To minimize the peak, we should sort devices such that those with 
    # smaller a_i - b_i come first.
    # Specifically, sort by (a_i - b_i).
    devices.sort(key=lambda x: x[0] - x[1])
    
    sum_b = sum(d[1] for d in devices)
    
    # We want to minimize max_{k=1..j} (sum_{i=1}^k a_i + sum_{i=k}^j b_i)
    # Let prefix_a[k] = sum_{i=1}^k a_i
    # Let suffix_b[k] = sum_{i=k}^j b_i
    # S_k = prefix_a[k] + suffix_b[k]
    
    prefix_a = 0
    suffix_b = sum_b
    max_exposure = 0
    
    for i in range(j):
        a, b = devices[i]
        prefix_a += a
        # S_k = prefix_a[k] + suffix_b[k]
        # suffix_b[k] is sum of b from i to j
        current_exposure = prefix_a + suffix_b
        if current_exposure > max_exposure:
            max_exposure = current_exposure
        suffix_b -= b
        
    print(max_exposure)

if __name__ == '__main__':
    solve()
```
