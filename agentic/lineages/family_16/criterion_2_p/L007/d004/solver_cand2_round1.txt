```python
import sys

def solve():
    # The burden if we transition at pillar k is:
    # B_k = sum_{i=1}^k S_i + sum_{i=k}^j M_i
    # B_k = sum_{i=1}^j M_i + sum_{i=1}^k (S_i - M_i)
    # Let D_i = S_i - M_i.
    # B_k = sum(M) + sum_{i=1}^k D_i
    # Let P_k = sum_{i=1}^k D_i be the prefix sum of D.
    # B_k = sum(M) + P_k.
    # The pilgrim chooses k to maximize B_k, so the max burden is sum(M) + max_{1<=k<=j} P_k.
    # We want to order the pillars to minimize (sum(M) + max_{1<=k<=j} P_k).
    # Since sum(M) is constant regardless of order, we minimize max_{1<=k<=j} P_k.
    # Let the pillars be (S_i, M_i) and D_i = S_i - M_i.
    # We want to order D_1, D_2, ..., D_j to minimize max(P_1, P_2, ..., P_j).
    # This is a classic problem: to minimize the maximum prefix sum, 
    # we should sort the D_i values in non-increasing order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    pillars = []
    sum_m = 0
    idx = 1
    for _ in range(j):
        s = int(input_data[idx])
        m = int(input_data[idx+1])
        pillars.append(s - m)
        sum_m += m
        idx += 2
        
    # Sort D_i in descending order to minimize the maximum prefix sum
    pillars.sort(reverse=True)
    
    max_p = -float('inf')
    current_p = 0
    for d in pillars:
        current_p += d
        if current_p > max_p:
            max_p = current_p
            
    print(sum_m + max_p)

if __name__ == '__main__':
    solve()
```
