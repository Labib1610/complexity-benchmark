```python
import sys

def solve():
    # The problem asks us to arrange j manuscripts (p1_i, p2_i) to minimize
    # max_{1 <= k <= j} (sum_{i=1}^k p1_i + sum_{i=k}^j p2_i).
    # Let S1_k = sum_{i=1}^k p1_i and S2_k = sum_{i=k}^j p2_i.
    # The expression is S1_k + S2_k = sum_{i=1}^j p2_i + sum_{i=1}^k (p1_i - p2_i) + p2_k.
    # Actually, let's rewrite:
    # Total = sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    #       = sum_{i=1}^j p2_i + sum_{i=1}^k (p1_i - p2_i)
    # Let d_i = p1_i - p2_i.
    # Total = sum(p2) + sum_{i=1}^k d_i.
    # To minimize the maximum of this over k, we want to arrange the manuscripts
    # such that the prefix sums of d_i are as small as possible.
    # This is a classic scheduling problem. If we sort by d_i, we can minimize
    # the maximum prefix sum.
    # Specifically, if we sort by d_i in ascending order, the prefix sums
    # will be minimized.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    manuscripts = []
    idx = 1
    sum_p2 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        manuscripts.append((p1 - p2, p1, p2))
        sum_p2 += p2
        idx += 2
    
    # Sort by d_i = p1 - p2
    manuscripts.sort()
    
    # Now calculate the max prefix sum of d_i
    # The total time for a transition at k is:
    # sum_{i=1}^k p1_i + sum_{i=k}^j p2_i
    # = sum_{i=1}^k p1_i + (sum_{i=1}^j p2_i - sum_{i=1}^{k-1} p2_i)
    # = sum_{i=1}^j p2_i + sum_{i=1}^{k-1} (p1_i - p2_i) + p1_k
    
    current_d_sum = 0
    max_val = 0
    for i in range(j):
        d, p1, p2 = manuscripts[i]
        # The value at transition k (1-indexed) is:
        # sum_{i=1}^j p2_i + current_d_sum + p1
        val = sum_p2 + current_d_sum + p1
        if val > max_val:
            max_val = val
        current_d_sum += d
        
    print(max_val)

if __name__ == '__main__':
    solve()
```
