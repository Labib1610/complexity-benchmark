import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        tiles.append((int(input_data[idx]), int(input_data[idx+1])))
        idx += 2

    # The cost of a path dropping at column k (0-indexed) is:
    # sum(top[0...k]) + sum(bottom[k...j-1])
    # Let T_i be top[i] and B_i be bottom[i].
    # Cost(k) = sum_{i=0}^k T_i + sum_{i=k}^{j-1} B_i
    # Cost(k) = sum_{i=0}^{j-1} B_i + sum_{i=0}^k (T_i - B_i)
    # Let D_i = T_i - B_i.
    # Cost(k) = sum(B) + sum_{i=0}^k D_i
    # We want to minimize max_{k=0..j-1} (sum(B) + prefix_sum_D(k))
    # Since sum(B) is constant for a fixed arrangement, we minimize max_{k=0..j-1} prefix_sum_D(k).
    
    # To minimize the maximum prefix sum of D_i, we should sort D_i in non-decreasing order.
    # If we sort D_i such that D_0 <= D_1 <= ... <= D_{j-1},
    # then the prefix sums are non-decreasing, and the maximum is the total sum of all D_i.
    # Wait, that's not right. If we sort D_i such that D_0 <= D_1 <= ... <= D_{j-1},
    # the prefix sums are S_k = D_0 + ... + D_k.
    # The maximum prefix sum is S_{j-1} = sum(D_i).
    # Actually, to minimize the maximum prefix sum, we should sort D_i in non-increasing order.
    # If D_0 >= D_1 >= ... >= D_{j-1}, then the prefix sums are S_0, S_1, ..., S_{j-1}.
    # The maximum prefix sum will be S_0 = D_0 (if all D_i are negative, this is the max).
    # Actually, the optimal strategy to minimize the maximum prefix sum is to sort D_i 
    # in non-increasing order.
    
    tiles.sort(key=lambda x: x[0] - x[1], reverse=True)
    
    top = [t[0] for t in tiles]
    bottom = [t[1] for t in tiles]
    
    sum_b = sum(bottom)
    current_prefix_t = 0
    max_path_cost = 0
    
    # Path dropping at column k:
    # sum(top[0...k]) + sum(bottom[k...j-1])
    # = sum(top[0...k]) + (sum(bottom) - sum(bottom[0...k-1]))
    
    current_top_sum = 0
    current_bottom_sum = sum_b
    
    for i in range(j):
        current_top_sum += top[i]
        path_cost = current_top_sum + current_bottom_sum
        if i == 0:
            max_path_cost = path_cost
        else:
            max_path_cost = max(max_path_cost, path_cost)
        current_bottom_sum -= bottom[i]
        
    print(max_path_cost)

solve()
