```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tiles.append((p1, p2))
        idx += 2
        
    # Let S1 be the sum of all top values, S2 be the sum of all bottom values.
    # For a sequence of tiles (a1, b1), (a2, b2), ..., (aj, bj),
    # the cost of path k (moving down at column k) is:
    # C_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # C_k = sum_{i=1}^j a_i + sum_{i=k}^j b_i - sum_{i=k+1}^j a_i
    # Let P1_k = sum_{i=1}^k a_i, P2_k = sum_{i=k}^j b_i.
    # C_k = P1_k + P2_k.
    # We want to minimize max(C_k).
    # Note that C_k = (sum of all a_i) + (sum of all b_i) - (sum_{i=1}^{k-1} a_i - sum_{i=k}^j b_i)
    # Actually, simpler: C_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i.
    # Let A = sum(a_i), B = sum(b_i).
    # C_k = sum_{i=1}^k a_i + B - sum_{i=1}^{k-1} b_i.
    # To minimize the maximum, we should sort tiles such that the sequence of 
    # (a_i - b_i) is non-decreasing.
    
    tiles.sort(key=lambda x: x[0] - x[1])
    
    s1 = sum(t[0] for t in tiles)
    s2 = sum(t[1] for t in tiles)
    
    # C_k = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # C_1 = a_1 + sum_{i=1}^j b_i
    # C_{k+1} = C_k + a_{k+1} - b_k
    
    current_c = tiles[0][0] + s2
    max_c = current_c
    
    for i in range(j - 1):
        current_c = current_c + tiles[i+1][0] - tiles[i][1]
        if current_c > max_c:
            max_c = current_c
            
    print(max_c)

if __name__ == '__main__':
    solve()
```
