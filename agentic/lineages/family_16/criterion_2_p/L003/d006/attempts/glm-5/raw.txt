```python
import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tiles.append((p1, p2))
        idx += 2
    
    if j == 1:
        p1, p2 = tiles[0]
        print(p1 + p2)
        return
    
    # Sort tiles by the difference (p1 - p2) in ascending order
    tiles.sort(key=lambda x: x[0] - x[1])
    
    # Compute prefix sums of top and bottom values
    prefix_top = [0] * (j + 1)
    prefix_bot = [0] * (j + 1)
    for i in range(j):
        prefix_top[i+1] = prefix_top[i] + tiles[i][0]
        prefix_bot[i+1] = prefix_bot[i] + tiles[i][1]
    
    total_top = prefix_top[j]
    total_bot = prefix_bot[j]
    
    # For each column k (1-indexed), the path cost is:
    # prefix_top[k] + (total_bot - prefix_bot[k-1])
    # We want the maximum over all k from 1 to j
    max_cost = 0
    for k in range(1, j + 1):
        cost = prefix_top[k] + (total_bot - prefix_bot[k-1])
        if cost > max_cost:
            max_cost = cost
    
    print(max_cost)

solve()
```
