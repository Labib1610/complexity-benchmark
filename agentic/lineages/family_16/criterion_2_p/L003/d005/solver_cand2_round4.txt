```python
import sys

def solve():
    # The pivot sum at index k is:
    # S_k = (sum_{i=1}^k p1_i) + (sum_{i=k}^j p2_i)
    # S_k = (sum_{i=1}^j p1_i) + (sum_{i=k}^j p2_i) - (sum_{i=k+1}^j p1_i)
    # Let P1 = sum(p1_i) and P2 = sum(p2_i).
    # S_k = P1 + p2_k + sum_{i=k+1}^j (p2_i - p1_i)
    #
    # Let d_i = p2_i - p1_i.
    # S_k = P1 + p2_k + sum_{i=k+1}^j d_i
    #
    # To minimize the maximum S_k, we should sort the elements.
    # Consider two adjacent elements (p1_a, p2_a) and (p1_b, p2_b).
    # If we swap them, the pivot sums for indices other than a and b remain unchanged.
    # For index a (before swap): S_a = prefix_p1_a + p2_a + p2_b + suffix_p2
    # For index b (before swap): S_b = prefix_p1_a + p1_b + p2_b + suffix_p2
    # After swap:
    # S'_a = prefix_p1_a + p2_b + p2_a + suffix_p2
    # S'_b = prefix_p1_a + p1_a + p2_a + suffix_p2
    #
    # The max of (S_a, S_b) is max(prefix + p2_a + p2_b + suffix, prefix + p1_b + p2_b + suffix)
    # The max of (S'_a, S'_b) is max(prefix + p2_b + p2_a + suffix, prefix + p1_a + p2_a + suffix)
    #
    # This is a classic scheduling problem. Sorting by p1_i + p2_i is not necessarily correct,
    # but sorting by p1_i - p2_i is. Specifically, we want to arrange elements such that
    # the sequence of p1_i - p2_i is non-decreasing.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    elements = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        elements.append((p1, p2))
        idx += 2
    
    # Sort by p1 - p2 ascending
    elements.sort(key=lambda x: x[0] - x[1])
    
    p1_sum = sum(e[0] for e in elements)
    
    # S_k = (sum_{i=1}^k p1_i) + (sum_{i=k}^j p2_i)
    # Let prefix_p1[k] be sum of p1 from 0 to k-1
    # Let suffix_p2[k] be sum of p2 from k to j-1
    # S_k = prefix_p1[k] + p2[k] + suffix_p2[k+1]
    
    current_p1 = 0
    total_p2 = sum(e[1] for e in elements)
    current_p2_suffix = total_p2
    
    max_pivot = 0
    for i in range(j):
        p1, p2 = elements[i]
        current_p1 += p1
        # S_k (1-indexed) = current_p1 + current_p2_suffix
        pivot_sum = current_p1 + current_p2_suffix
        if pivot_sum > max_pivot:
            max_pivot = pivot_sum
        current_p2_suffix -= p2
        
    print(max_pivot)

solve()
```
