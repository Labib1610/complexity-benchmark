```python
import sys

def solve():
    # Let the tiles be (a_i, b_i) for i = 1 to j.
    # If we arrange them in order p_1, p_2, ..., p_j,
    # the path cost if we move down at column k is:
    # C_k = sum_{i=1}^k a_{p_i} + sum_{i=k}^j b_{p_i}
    # Let S_a = sum a_i and S_b = sum b_i.
    # C_k = sum_{i=1}^k a_{p_i} + (S_b - sum_{i=1}^{k-1} b_{p_i})
    # C_k = S_b + a_{p_k} + sum_{i=1}^{k-1} (a_{p_i} - b_{p_i})
    # Let d_i = a_i - b_i.
    # C_k = S_b + a_{p_k} + sum_{i=1}^{k-1} d_{p_i}
    # We want to minimize max_k C_k.
    # This is equivalent to minimizing max_k (a_{p_k} + sum_{i=1}^{k-1} d_{p_i}).
    # Let x_i = a_i and y_i = b_i. We want to order (x_i, y_i) to minimize
    # max_k (x_{p_k} + sum_{i=1}^{k-1} (x_{p_i} - y_{p_i})).
    # This is a classic scheduling problem. The optimal order is to sort by
    # the difference d_i = x_i - y_i.
    # If d_i < 0, we should put them earlier. If d_i > 0, later.
    # Specifically, sort by d_i ascending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    sum_b = 0
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        tiles.append((a, b, a - b))
        sum_b += b
        idx += 2
        
    # Sort by d_i = a_i - b_i
    tiles.sort(key=lambda x: x[2])
    
    max_cost = 0
    current_sum_d = 0
    for i in range(j):
        a, b, d = tiles[i]
        # C_{i+1} = sum_b + a_i + sum_{k=0}^{i-1} d_k
        cost = sum_b + a + current_sum_d
        if cost > max_cost:
            max_cost = cost
        current_sum_d += d
        
    print(max_cost)

if __name__ == '__main__':
    solve()
```
