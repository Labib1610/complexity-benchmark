import sys
from functools import cmp_to_key

def compare_jobs(job1, job2):
    # job = (p1, p2, index)
    # Sort by p1 - p2 in descending order
    diff1 = job1[0] - job1[1]
    diff2 = job2[0] - job2[1]
    if diff1 > diff2:
        return -1
    elif diff1 < diff2:
        return 1
    else:
        return 0

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    jobs = []
    index = 1
    for i in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        jobs.append((p1, p2))
        index += 2
    
    # Add indices to keep track of original positions if needed
    indexed_jobs = [(jobs[i][0], jobs[i][1], i) for i in range(j)]
    
    # Sort using Johnson's rule: by p1 - p2 descending
    indexed_jobs.sort(key=cmp_to_key(compare_jobs))
    
    # Calculate the completion time
    extraction_end_time = 0
    transformation_end_time = 0
    
    for p1, p2, idx in indexed_jobs:
        extraction_end_time += p1
        transformation_end_time = max(transformation_end_time, extraction_end_time) + p2
    
    print(transformation_end_time)

if __name__ == "__main__":
    main()
