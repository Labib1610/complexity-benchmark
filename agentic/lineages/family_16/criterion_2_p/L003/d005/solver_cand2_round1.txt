```python
import sys

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        tiles.append((p1, p2))
        idx += 2
        
    # Let the tiles be (a_i, b_i).
    # A path moving down at column k has cost:
    # sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # = sum_{i=1}^j a_i + sum_{i=k}^j b_i - sum_{i=k}^j a_i
    # = sum_{i=1}^j a_i + sum_{i=k}^j (b_i - a_i)
    # Let S = sum_{i=1}^j a_i. The cost is S + sum_{i=k}^j (b_i - a_i).
    # Let d_i = b_i - a_i. The cost is S + sum_{i=k}^j d_i.
    # We want to minimize max_{k=1..j} (S + sum_{i=k}^j d_i).
    # This is equivalent to minimizing max_{k=1..j} (sum_{i=k}^j d_i).
    
    # Let D_k = sum_{i=k}^j d_i. We want to minimize max(D_1, D_2, ..., D_j).
    # Note that D_k = d_k + D_{k+1}, with D_{j+1} = 0.
    # To minimize the maximum prefix sum of the reversed sequence d_j, d_{j-1}, ..., d_1,
    # we should sort the d_i values.
    # Specifically, if we have a set of values d_i, we want to arrange them such that
    # the running suffix sums are as small as possible.
    # This is achieved by sorting d_i in ascending order.
    
    d = [t[1] - t[0] for t in tiles]
    d.sort()
    
    # Now we have d_1 <= d_2 <= ... <= d_j.
    # The suffix sums are:
    # S_j = d_j
    # S_{j-1} = d_j + d_{j-1}
    # ...
    # S_1 = d_j + d_{j-1} + ... + d_1
    # The maximum of these is S_1.
    # Wait, the adversary chooses the path. The path is defined by the column k.
    # The cost is S + sum_{i=k}^j d_i.
    # With d_i sorted ascending, the suffix sums are:
    # S_j = d_j
    # S_{j-1} = d_j + d_{j-1}
    # ...
    # S_1 = sum_{i=1}^j d_i
    # The maximum is S_1 if S_1 > 0, or S_j if all are negative.
    # Actually, the adversary chooses k to maximize S + sum_{i=k}^j d_i.
    # If we sort d_i ascending, the suffix sums are increasing as k decreases.
    # The maximum is always the sum of all d_i if the sum is positive, 
    # or the largest d_i if all are negative.
    # Actually, the optimal strategy is to put the largest d_i at the end.
    
    # Let's re-evaluate: we want to minimize max_{k} (sum_{i=k}^j d_i).
    # If we sort d_i such that d_1 <= d_2 <= ... <= d_j,
    # the suffix sums are S_k = d_k + ... + d_j.
    # The maximum is S_1 = sum_{i=1}^j d_i.
    # Is there a better way?
    # If we sort d_i descending: d_1 >= d_2 >= ... >= d_j.
    # S_j = d_j
    # S_{j-1} = d_j + d_{j-1}
    # ...
    # S_1 = sum_{i=1}^j d_i
    # The maximum is still the sum.
    # Actually, the minimum possible maximum is achieved by sorting d_i such that
    # the suffix sums are minimized. This is a known problem:
    # To minimize the maximum suffix sum, sort d_i in ascending order.
    # The maximum suffix sum will be max(0, sum_{i=1}^j d_i) if we can choose an empty set,
    # but here k is 1 to j, so the max is max_{k=1..j} (sum_{i=k}^j d_i).
    
    d.sort()
    current_suffix_sum = 0
    max_suffix_sum = -float('inf')
    for val in reversed(d):
        current_suffix_sum += val
        if current_suffix_sum > max_suffix_sum:
            max_suffix_sum = current_suffix_sum
            
    s_sum = sum(t[0] for t in tiles)
    print(s_sum + max_suffix_sum)

solve()
```
