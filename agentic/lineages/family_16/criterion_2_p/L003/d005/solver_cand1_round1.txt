```python
import sys

def solve():
    # The problem asks to minimize the maximum path cost.
    # A path consists of moving right along the top row until some column k,
    # then moving down to the bottom row, then moving right to the end.
    # Let the tiles be (a_i, b_i) for i=1 to j.
    # The cost of a path switching at column k is:
    # sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # This can be rewritten as:
    # sum_{i=1}^j a_i + sum_{i=k}^j (b_i - a_i)
    # Let S = sum_{i=1}^j a_i. The cost is S + sum_{i=k}^j (b_i - a_i).
    # To minimize the maximum of these costs, we want to arrange the tiles
    # such that the partial sums of (b_i - a_i) are minimized.
    # Let d_i = b_i - a_i. We want to order d_i such that max_{k} sum_{i=k}^j d_i is minimized.
    # This is equivalent to minimizing the maximum suffix sum of the sequence d_i.
    # To minimize the maximum suffix sum, we should sort d_i in non-decreasing order.
    # If d_i are sorted such that d_1 <= d_2 <= ... <= d_j,
    # then the suffix sums are:
    # k=1: sum_{i=1}^j d_i
    # k=2: sum_{i=2}^j d_i
    # ...
    # k=j: d_j
    # Since d_i are sorted non-decreasingly, the maximum suffix sum is the sum of all d_i.
    # Wait, if we sort d_i non-decreasingly, the suffix sums are:
    # S_k = d_k + d_{k+1} + ... + d_j.
    # Since d_i are non-decreasing, the largest suffix sum is S_1 = sum_{i=1}^j d_i.
    # The score is S + sum_{i=1}^j (b_i - a_i) = sum a_i + sum b_i - sum a_i = sum b_i.
    # Actually, let's re-evaluate:
    # Cost_k = (sum_{i=1}^k a_i) + (sum_{i=k}^j b_i)
    # Cost_k = (sum_{i=1}^j a_i) + (sum_{i=k}^j (b_i - a_i))
    # Let d_i = b_i - a_i. We want to minimize max_k (sum_{i=1}^j a_i + sum_{i=k}^j d_i).
    # This is (sum a_i) + max_k (sum_{i=k}^j d_i).
    # To minimize the max suffix sum of d_i, we sort d_i in non-decreasing order.
    # Then the suffix sums are S_k = d_k + ... + d_j.
    # Since d_i are non-decreasing, the maximum is S_1 = sum_{i=1}^j d_i.
    # So the score is (sum a_i) + (sum b_i - sum a_i) = sum b_i.
    # Wait, this logic implies the score is always sum b_i. Let's check Example 1.
    # Example 1: (3,2), (2,4), (1,3). Sum b_i = 2+4+3 = 9. But output is 10.
    # My formula for Cost_k was: sum_{i=1}^k a_i + sum_{i=k}^j b_i.
    # Let's re-check:
    # k=1: a_1 + b_1 + b_2 + ... + b_j
    # k=2: a_1 + a_2 + b_2 + ... + b_j
    # The sum is sum_{i=1}^j b_i + sum_{i=1}^k (a_i - b_i).
    # To minimize the maximum of this, we want to minimize the maximum prefix sum of (a_i - b_i).
    # To minimize the maximum prefix sum, we sort (a_i - b_i) in non-increasing order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    tiles = []
    idx = 1
    for _ in range(j):
        tiles.append((int(input_data[idx]), int(input_data[idx+1])))
        idx += 2
    
    # Sort by (a_i - b_i) descending
    tiles.sort(key=lambda x: x[0] - x[1], reverse=True)
    
    a = [t[0] for t in tiles]
    b = [t[1] for t in tiles]
    
    # Cost_k = sum_{i=0}^{k-1} a_i + sum_{i=k-1}^{j-1} b_i
    # Let P_a[k] be prefix sum of a, S_b[k] be suffix sum of b
    # Cost_k = P_a[k] + S_b[k-1] (using 1-based k)
    
    pref_a = [0] * (j + 1)
    for i in range(j):
        pref_a[i+1] = pref_a[i] + a[i]
        
    suff_b = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suff_b[i] = suff_b[i+1] + b[i]
        
    max_cost = 0
    for k in range(1, j + 1):
        cost = pref_a[k] + suff_b[k-1]
        if cost > max_cost:
            max_cost = cost
            
    print(max_cost)

solve()
```
