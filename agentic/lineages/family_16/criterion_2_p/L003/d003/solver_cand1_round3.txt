```python
import sys

def solve():
    # The problem asks to minimize the completion time of the last task's Beta step.
    # Let the tasks be (a_i, b_i).
    # If we process tasks in order 1, 2, ..., j:
    # Alpha_end_i = sum_{k=1}^i a_k
    # Beta_start_i = max(Alpha_end_i, Beta_end_{i-1})
    # Beta_end_i = Beta_start_i + b_i
    #
    # This is a classic scheduling problem (Johnson's rule variant).
    # The completion time is:
    # sum(a_i) + max_{i=1..j} (b_i + sum_{k=i+1}^j a_k)
    # Let S = sum(a_i). Then the expression is:
    # S + max_{i=1..j} (b_i - sum_{k=1}^i a_k) + sum_{k=1}^j a_k
    # Actually, the formula is:
    # Completion time = sum(a_i) + max_{i=1..j} (b_i + sum_{k=i+1}^j a_k)
    # Let P = sum(a_i). The completion time is P + max_{i=1..j} (b_i + (P - sum_{k=1}^i a_k))
    # This is equivalent to minimizing max_{i=1..j} (b_i - sum_{k=1}^i a_k).
    # This is solved by sorting tasks such that if a_i < b_i, they come first (sorted by a_i ascending),
    # and if a_i >= b_i, they come later (sorted by b_i descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    tasks = []
    idx = 1
    for _ in range(j):
        tasks.append((int(input_data[idx]), int(input_data[idx+1])))
        idx += 2
        
    # Johnson's rule:
    # Group 1: a_i < b_i, sort by a_i ascending
    # Group 2: a_i >= b_i, sort by b_i descending
    group1 = [t for t in tasks if t[0] < t[1]]
    group2 = [t for t in tasks if t[0] >= t[1]]
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_alpha_end = 0
    current_beta_end = 0
    
    for a, b in order:
        current_alpha_end += a
        current_beta_end = max(current_alpha_end, current_beta_end) + b
        
    print(current_beta_end)

if __name__ == '__main__':
    solve()
```
