```python
import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    j = int(sys.stdin.readline())
    tasks = []
    for _ in range(j):
        p1, p2 = map(int, sys.stdin.readline().split())
        tasks.append((p1, p2))
    
    # Sort tasks by p1 - p2 in ascending order (Johnson's rule)
    tasks.sort(key=lambda x: x[0] - x[1])
    
    # Simulate the process
    alpha_end_time = 0
    beta_end_time = 0
    
    for p1, p2 in tasks:
        alpha_end_time += p1
        beta_end_time = max(alpha_end_time, beta_end_time) + p2
    
    print(beta_end_time)

threading.Thread(target=main).start()
```
