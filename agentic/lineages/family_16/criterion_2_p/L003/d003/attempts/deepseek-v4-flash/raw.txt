We can solve this by sorting the tasks using a greedy rule:  
If `p1 < p2`, process earlier; otherwise, process later.  
This is a classic Johnson’s rule for two-machine scheduling.

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    j = int(data[0])
    tasks = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx+1])
        idx += 2
        tasks.append((p1, p2))
    
    # Johnson's rule: sort by min(p1, p2)
    # If p1 < p2, put in first group (sorted by p1 ascending)
    # Else, put in second group (sorted by p2 descending)
    first = []
    second = []
    for p1, p2 in tasks:
        if p1 < p2:
            first.append((p1, p2))
        else:
            second.append((p1, p2))
    
    first.sort(key=lambda x: x[0])
    second.sort(key=lambda x: x[1], reverse=True)
    
    order = first + second
    
    alpha_time = 0
    beta_time = 0
    for p1, p2 in order:
        alpha_time += p1
        if beta_time < alpha_time:
            beta_time = alpha_time
        beta_time += p2
    
    print(beta_time)

if __name__ == "__main__":
    main()
```
