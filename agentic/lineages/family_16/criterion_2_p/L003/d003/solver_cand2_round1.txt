```python
import sys

def solve():
    # The timestamp for the k-th packet in a sequence is:
    # T_k = (sum of p1 from 1 to k) + (sum of p2 from k to j)
    # Let S1_k = sum_{i=1}^k p1_i and S2_k = sum_{i=k}^j p2_i
    # T_k = S1_k + S2_k
    # Let P = sum_{all} p1 and Q = sum_{all} p2.
    # Let S1_{k-1} be the sum of p1 before packet k.
    # Let S2_{k+1} be the sum of p2 after packet k.
    # T_k = S1_{k-1} + p1_k + p2_k + S2_{k+1}
    # T_k = (sum of all p1 and p2) - (sum of p2 before k) + (sum of p1 after k)
    # This is a classic scheduling problem. To minimize the maximum of T_k,
    # we should sort the packets based on the difference (p1 - p2).
    # Specifically, if we have two adjacent packets i and j, swapping them
    # changes the max latency. The optimal order is to sort by p1 - p2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    packets = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        packets.append((p1, p2))
        idx += 2
        
    # Sort by p1 - p2 ascending
    packets.sort(key=lambda x: x[0] - x[1])
    
    # Calculate the timestamps for the sorted sequence
    # T_k = (sum of p1 up to k) + (sum of p2 from k to j)
    # Let prefix_p1[k] be sum of p1[0...k-1]
    # Let suffix_p2[k] be sum of p2[k...j-1]
    
    current_p1_sum = 0
    total_p2_sum = sum(p[1] for p in packets)
    
    max_latency = 0
    current_p2_sum = total_p2_sum
    
    for p1, p2 in packets:
        current_p1_sum += p1
        # T_k = current_p1_sum + current_p2_sum
        latency = current_p1_sum + current_p2_sum
        if latency > max_latency:
            max_latency = latency
        current_p2_sum -= p2
        
    print(max_latency)

if __name__ == '__main__':
    solve()
```
