import sys

def solve():
    # The problem asks to minimize the total time to complete all acts.
    # Let p_i be the preparation time and s_i be the performance time for act i.
    # If we process acts in order 1, 2, ..., j:
    # Preparation finishes at times:
    # T_p(1) = p_1
    # T_p(2) = p_1 + p_2
    # ...
    # T_p(j) = sum(p_1...p_j)
    #
    # Performance starts at:
    # T_s_start(1) = T_p(1) = p_1
    # T_s_start(2) = max(T_p(2), T_s_start(1) + s_1)
    # T_s_start(i) = max(T_p(i), T_s_start(i-1) + s_{i-1})
    #
    # Performance finishes at:
    # T_s_end(i) = T_s_start(i) + s_i
    #
    # This is a classic scheduling problem (Johnson's Rule variant).
    # To minimize the completion time of the last act, we should sort the acts.
    # If p_i < s_i, the act is "preparation-heavy", put it early.
    # If p_i >= s_i, the act is "performance-heavy", put it late.
    # Specifically, sort by min(p_i, s_i) or use the standard Johnson's rule logic:
    # Sort acts with p_i < s_i by p_i ascending,
    # then acts with p_i >= s_i by s_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    acts = []
    idx = 1
    for _ in range(j):
        p = int(input_data[idx])
        s = int(input_data[idx+1])
        acts.append((p, s))
        idx += 2
        
    group1 = []
    group2 = []
    for p, s in acts:
        if p < s:
            group1.append((p, s))
        else:
            group2.append((p, s))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_prep_time = 0
    current_perf_start = 0
    
    for p, s in order:
        current_prep_time += p
        current_perf_start = max(current_prep_time, current_perf_start) + s
        
    print(current_perf_start)

if __name__ == '__main__':
    solve()
