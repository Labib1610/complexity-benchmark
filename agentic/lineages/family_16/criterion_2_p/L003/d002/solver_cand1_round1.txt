```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    orders = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        orders.append((p1, p2))
        idx += 2
    
    # This is a variation of Johnson's Rule for flow shop scheduling.
    # To minimize the completion time (makespan) of the last job on the second machine,
    # we sort jobs based on p1 < p2.
    # Jobs with p1 < p2 should be processed in increasing order of p1.
    # Jobs with p1 >= p2 should be processed in decreasing order of p2.
    
    group1 = []
    group2 = []
    for p1, p2 in orders:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sorted_orders = group1 + group2
    
    # Calculate completion time
    # mainframe_time: time when mainframe finishes current job
    # testing_time: time when testing rig finishes current job
    mainframe_time = 0
    testing_time = 0
    
    for p1, p2 in sorted_orders:
        mainframe_time += p1
        # Testing rig starts when mainframe finishes the job OR when it finishes the previous job
        testing_time = max(mainframe_time, testing_time) + p2
        
    print(testing_time)

if __name__ == '__main__':
    solve()
```
