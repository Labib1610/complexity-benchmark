import sys

def solve():
    # The problem asks to minimize B_j.
    # Let the sequence be (p1_1, p2_1), (p1_2, p2_2), ..., (p1_j, p2_j).
    # A_i = sum_{k=1}^i p1_k
    # B_i = max(A_i, B_{i-1}) + p2_i
    # This can be rewritten as B_j = max(A_1 + p2_1 + p2_2 + ... + p2_j,
    #                                   A_2 + p2_2 + ... + p2_j,
    #                                   ...,
    #                                   A_j + p2_j)
    # Let S = sum_{k=1}^j p2_k.
    # B_j = max_{i=1..j} (A_i + sum_{k=i}^j p2_k)
    # Let P1_i = sum_{k=1}^i p1_k and P2_i = sum_{k=i}^j p2_k.
    # We want to minimize max_{i=1..j} (P1_i + P2_i).
    # This is a classic scheduling problem.
    # Consider two adjacent elements (p1_a, p2_a) and (p1_b, p2_b).
    # If we put 'a' then 'b':
    #   Term 1: A_prev + p1_a + p2_a + p2_b
    #   Term 2: A_prev + p1_a + p1_b + p2_b
    # If we put 'b' then 'a':
    #   Term 1: A_prev + p1_b + p2_b + p2_a
    #   Term 2: A_prev + p1_b + p1_a + p2_a
    # Comparing the max of these, we find that we should sort by p2 in descending order.
    # If p2_a > p2_b, then 'a' should come before 'b'.
    # If p2_a == p2_b, the order doesn't matter for the max.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    elements = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        elements.append((p1, p2))
        idx += 2
        
    # Sort by p2 descending
    elements.sort(key=lambda x: x[1], reverse=True)
    
    a_curr = 0
    b_curr = 0
    for p1, p2 in elements:
        a_curr += p1
        b_curr = max(a_curr, b_curr) + p2
        
    print(b_curr)

if __name__ == '__main__':
    solve()
