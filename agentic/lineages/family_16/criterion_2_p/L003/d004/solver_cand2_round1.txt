```python
import sys

def solve():
    # The problem asks to minimize the maximum path cost.
    # A path consists of moving right across some columns and then down.
    # Let the blocks be (a_i, b_i).
    # A path that goes down at column k has cost:
    # sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # This can be rewritten as:
    # (sum_{i=1}^j a_i + sum_{i=1}^j b_i) - (sum_{i=k+1}^j a_i + sum_{i=1}^{k-1} b_i)
    # Actually, it's simpler:
    # Cost(k) = (sum_{i=1}^k a_i) + (sum_{i=k}^j b_i)
    # Let S_a = sum(a_i) and S_b = sum(b_i).
    # Cost(k) = sum_{i=1}^k a_i + sum_{i=k}^j b_i
    # Let P_k = sum_{i=1}^k a_i and Q_k = sum_{i=k}^j b_i.
    # We want to minimize max_k (P_k + Q_k).
    # Note that P_k + Q_k = sum_{i=1}^j a_i + sum_{i=1}^j b_i - sum_{i=1}^{k-1} a_i - sum_{i=k+1}^j b_i
    # This is equivalent to: Total_Sum - (sum_{i=1}^{k-1} a_i + sum_{i=k+1}^j b_i).
    # To minimize the max, we want to balance the values.
    # It is a known result for this type of problem that sorting the blocks
    # by (a_i - b_i) helps. Specifically, sorting by a_i + b_i or similar
    # doesn't work, but sorting by a_i - b_i is the standard approach for
    # minimizing the maximum prefix/suffix sum.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    blocks = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        blocks.append((a, b))
        idx += 2
    
    # Sort blocks such that a_i - b_i is non-decreasing
    blocks.sort(key=lambda x: x[0] - x[1])
    
    a_vals = [x[0] for x in blocks]
    b_vals = [x[1] for x in blocks]
    
    # Calculate prefix sums of a and suffix sums of b
    pref_a = [0] * (j + 1)
    for i in range(j):
        pref_a[i+1] = pref_a[i] + a_vals[i]
        
    suff_b = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suff_b[i] = suff_b[i+1] + b_vals[i]
        
    # Path cost if we go down at column k (1-indexed, k=1 to j)
    # Path goes through a_1...a_k and b_k...b_j
    # Cost = pref_a[k] + suff_b[k-1]
    max_cost = 0
    for k in range(1, j + 1):
        current_cost = pref_a[k] + suff_b[k-1]
        if current_cost > max_cost:
            max_cost = current_cost
            
    print(max_cost)

solve()
```
