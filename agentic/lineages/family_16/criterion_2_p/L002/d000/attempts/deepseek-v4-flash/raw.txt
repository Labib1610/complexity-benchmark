import sys

def solve():
    data = sys.stdin.buffer.read().split()
    j = int(data[0])
    containers = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx]); p2 = int(data[idx+1])
        idx += 2
        containers.append((p1, p2))
    
    containers.sort(key=lambda x: x[1], reverse=True)
    
    inspection_time = 0
    finish_time = 0
    for p1, p2 in containers:
        inspection_time += p1
        finish_time = max(finish_time, inspection_time + p2)
    
    print(finish_time)

if __name__ == "__main__":
    solve()
