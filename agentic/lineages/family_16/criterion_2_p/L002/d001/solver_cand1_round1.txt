```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    modules = []
    idx = 1
    for _ in range(j):
        p = int(input_data[idx])
        f = int(input_data[idx+1])
        modules.append((p, f))
        idx += 2
    
    # This is a variation of Johnson's Rule for two-machine scheduling.
    # To minimize the makespan on two machines (Origin, Destination),
    # we sort modules into two groups:
    # Group 1: p < f, sort by p ascending
    # Group 2: p >= f, sort by f descending
    
    group1 = []
    group2 = []
    for p, f in modules:
        if p < f:
            group1.append((p, f))
        else:
            group2.append((p, f))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    current_prep_time = 0
    current_finish_time = 0
    
    for p, f in sequence:
        current_prep_time += p
        # The destination bay can start finalization only after:
        # 1. The module has finished preparation (current_prep_time)
        # 2. The previous module has finished finalization (current_finish_time)
        current_finish_time = max(current_prep_time, current_finish_time) + f
        
    print(current_finish_time)

if __name__ == '__main__':
    solve()
```
