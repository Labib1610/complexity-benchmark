```python
import sys

def solve():
    # This is a variation of Johnson's Rule for the two-machine flow shop scheduling problem.
    # We want to minimize the makespan (the time the last ship finishes the Lower Lock).
    # Johnson's Rule:
    # 1. Divide ships into two sets:
    #    Set A: ships where p1 < p2
    #    Set B: ships where p1 >= p2
    # 2. Sort Set A in ascending order of p1.
    # 3. Sort Set B in descending order of p2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    ships = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        ships.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in ships:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate the completion time
    # current_upper_finish: time when the current ship finishes the Upper Lock
    # current_lower_finish: time when the current ship finishes the Lower Lock
    current_upper_finish = 0
    current_lower_finish = 0
    
    for p1, p2 in sequence:
        current_upper_finish += p1
        # The ship can start the Lower Lock only after it finishes the Upper Lock
        # AND the previous ship has finished the Lower Lock.
        current_lower_finish = max(current_upper_finish, current_lower_finish) + p2
        
    print(current_lower_finish)

if __name__ == '__main__':
    solve()
```
