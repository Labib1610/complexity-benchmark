```python
import sys

def solve():
    # The problem asks us to order segments (n_i, s_i) to minimize:
    # max_{k=1..j} (sum_{i=1}^k n_i + sum_{i=k}^j s_i)
    # Let N = sum_{i=1}^j n_i and S = sum_{i=1}^j s_i.
    # The expression is: sum_{i=1}^k n_i + sum_{i=k}^j s_i
    # = sum_{i=1}^j n_i + sum_{i=k}^j s_i - sum_{i=k+1}^j n_i
    # = N + sum_{i=k}^j (s_i - n_i) + n_k (Wait, this is not quite right)
    # Let's re-evaluate:
    # Total time T_k = (n_1 + ... + n_k) + (s_k + ... + s_j)
    # T_k = (n_1 + ... + n_j) + (s_k + ... + s_j) - (n_{k+1} + ... + n_j)
    # Let P_k = n_k - s_k.
    # T_k = sum_{i=1}^j s_i + sum_{i=1}^k (n_i - s_i)
    # T_k = S + sum_{i=1}^k P_i
    # To minimize the maximum of these partial sums, we should sort the segments
    # based on P_i = n_i - s_i.
    # If we sort such that P_1 <= P_2 <= ... <= P_j, then the partial sums
    # sum_{i=1}^k P_i will be minimized at each step k.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    segments = []
    idx = 1
    for _ in range(j):
        n = int(input_data[idx])
        s = int(input_data[idx+1])
        segments.append((n, s))
        idx += 2
        
    # Sort by n_i - s_i
    segments.sort(key=lambda x: x[0] - x[1])
    
    # Calculate the maximum of T_k = sum_{i=1}^k n_i + sum_{i=k}^j s_i
    # T_k = (sum_{i=1}^j s_i) + sum_{i=1}^k (n_i - s_i)
    
    s_sum = sum(s for n, s in segments)
    
    max_time = 0
    current_p_sum = 0
    for n, s in segments:
        current_p_sum += (n - s)
        max_time = max(max_time, s_sum + current_p_sum)
        
    print(max_time)

if __name__ == '__main__':
    solve()
```
