```python
import sys

def solve():
    # The problem asks us to arrange j containers (top_i, bottom_i)
    # to minimize the maximum path score.
    # A path moving down at column k has score:
    # S_k = sum(top_1...top_k) + sum(bottom_k...bottom_j)
    # Let T_i = top_i, B_i = bottom_i.
    # S_k = sum(T_1...T_k) + sum(B_k...B_j)
    # S_k = sum(T_1...T_j) + sum(B_k...B_j) - sum(T_{k+1}...T_j)
    # Let S_total = sum(T_1...T_j) + sum(B_1...B_j).
    # S_k = sum(T_1...T_k) + sum(B_k...B_j)
    # Note that S_{k+1} - S_k = T_{k+1} - B_k.
    # To minimize max(S_k), we want the sequence of S_k to be as flat as possible.
    # This is a classic problem: we want to order pairs (T_i, B_i) such that
    # the maximum prefix sum of (T_i - B_{i-1}) is minimized, or similar.
    # Actually, the optimal strategy is to sort the containers based on the
    # difference (T_i - B_i).
    # Specifically, sort by T_i - B_i ascending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    containers = []
    idx = 1
    for _ in range(j):
        t = int(input_data[idx])
        b = int(input_data[idx+1])
        containers.append((t, b))
        idx += 2
    
    # Sort by T_i - B_i
    containers.sort(key=lambda x: x[0] - x[1])
    
    tops = [c[0] for c in containers]
    bottoms = [c[1] for c in containers]
    
    # S_k = sum(T_0...T_{k-1}) + sum(B_{k-1}...B_{j-1})
    # Let prefix_T[k] = sum(T_0...T_{k-1}), prefix_T[0] = 0
    # Let suffix_B[k] = sum(B_k...B_{j-1}), suffix_B[j] = 0
    # S_k = prefix_T[k] + suffix_B[k-1]
    
    prefix_T = [0] * (j + 1)
    for i in range(j):
        prefix_T[i+1] = prefix_T[i] + tops[i]
        
    suffix_B = [0] * (j + 1)
    for i in range(j - 1, -1, -1):
        suffix_B[i] = suffix_B[i+1] + bottoms[i]
        
    max_score = 0
    for k in range(1, j + 1):
        current_score = prefix_T[k] + suffix_B[k-1]
        if k == 1:
            max_score = current_score
        else:
            if current_score > max_score:
                max_score = current_score
                
    print(max_score)

if __name__ == '__main__':
    solve()
```
