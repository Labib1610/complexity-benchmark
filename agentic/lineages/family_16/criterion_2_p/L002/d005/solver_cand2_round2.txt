import sys

def solve():
    # The expression E(k) = sum_{i=1}^k p1(s_i) + sum_{i=k}^j p2(s_i)
    # Let P1_sum = sum_{i=1}^j p1(s_i) and P2_sum = sum_{i=1}^j p2(s_i).
    # E(k) = sum_{i=1}^k p1(s_i) + sum_{i=k}^j p2(s_i)
    # E(k) = sum_{i=1}^k p1(s_i) + (sum_{i=1}^j p2(s_i) - sum_{i=1}^{k-1} p2(s_i))
    # E(k) = P2_sum + sum_{i=1}^k (p1(s_i) - p2(s_{i-1})) where p2(s_0) = 0.
    # Let d_i = p1(s_i) - p2(s_i).
    # E(k) = sum_{i=1}^k p1(s_i) + sum_{i=k}^j p2(s_i)
    # E(k) = sum_{i=1}^j p2(s_i) + sum_{i=1}^k (p1(s_i) - p2(s_i))
    # Let diff_i = p1_i - p2_i.
    # E(k) = sum(p2) + sum_{i=1}^k diff_i.
    # To minimize the maximum of these partial sums, we should sort the items
    # based on diff_i. If diff_i < 0, we want them early. If diff_i > 0, we want them late.
    # Specifically, sort by diff_i ascending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    items = []
    idx = 1
    sum_p2 = 0
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        items.append((p1, p2))
        sum_p2 += p2
        idx += 2
        
    # Sort by p1 - p2
    items.sort(key=lambda x: x[0] - x[1])
    
    max_e = 0
    current_sum_diff = 0
    for p1, p2 in items:
        current_sum_diff += (p1 - p2)
        max_e = max(max_e, sum_p2 + current_sum_diff)
        
    print(max_e)

if __name__ == '__main__':
    solve()
