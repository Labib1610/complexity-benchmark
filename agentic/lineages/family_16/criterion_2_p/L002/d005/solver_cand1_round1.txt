import sys

def solve():
    # The penalty for a sequence S at transfer point k is:
    # Penalty(k) = sum_{i=1}^k p1(s_i) + sum_{i=k}^j p2(s_i)
    # Penalty(k) = sum_{i=1}^k p1(s_i) + sum_{i=1}^j p2(s_i) - sum_{i=1}^{k-1} p2(s_i)
    # Penalty(k) = sum_{i=1}^j p2(s_i) + sum_{i=1}^k (p1(s_i) - p2(s_{i-1})) where p2(s_0) = 0
    # Let P2_total = sum_{i=1}^j p2(s_i).
    # Penalty(k) = P2_total + sum_{i=1}^k (p1(s_i) - p2(s_{i-1}))
    # To minimize the maximum penalty, we use Johnson's rule logic.
    # Sort vessels such that those with p1 < p2 come first (sorted by p1 ascending),
    # and those with p1 >= p2 come last (sorted by p2 descending).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        vessels.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    for p1, p2 in vessels:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate max penalty for this sequence
    # Penalty(k) = sum_{i=1}^k p1(s_i) + sum_{i=k}^j p2(s_i)
    # Let L[k] = sum_{i=1}^k p1(s_i)
    # Let R[k] = sum_{i=k}^j p2(s_i)
    # Penalty(k) = L[k] + R[k]
    
    n = len(sequence)
    L = [0] * (n + 1)
    for i in range(n):
        L[i+1] = L[i] + sequence[i][0]
        
    R = [0] * (n + 2)
    for i in range(n, 0, -1):
        R[i] = R[i+1] + sequence[i-1][1]
        
    max_penalty = 0
    for k in range(1, n + 1):
        penalty = L[k] + R[k]
        if penalty > max_penalty:
            max_penalty = penalty
            
    print(max_penalty)

if __name__ == '__main__':
    solve()
