```python
import sys

def solve():
    # The score for a split point k is:
    # Score(k) = sum_{i=1}^k red(C_i) + sum_{i=k}^j blue(C_i)
    # Let R_i = red(C_i) and B_i = blue(C_i).
    # Score(k) = sum_{i=1}^k R_i + sum_{i=k}^j B_i
    # Score(k) = sum_{i=1}^j B_i + sum_{i=1}^k (R_i - B_i)
    # Let D_i = R_i - B_i.
    # Score(k) = (sum_{i=1}^j B_i) + sum_{i=1}^k D_i
    # Let S = sum_{i=1}^j B_i.
    # Score(k) = S + prefix_sum_D(k)
    # To minimize the maximum Score(k), we need to minimize the maximum prefix sum of D_i.
    # This is a classic problem: given a set of values D_i, arrange them to minimize
    # the maximum prefix sum.
    # The optimal strategy is to sort D_i in descending order.
    # If we have positive and negative values, we put all positive values first,
    # then all negative values.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    cards = []
    sum_b = 0
    idx = 1
    for _ in range(j):
        r = int(input_data[idx])
        b = int(input_data[idx+1])
        cards.append(r - b)
        sum_b += b
        idx += 2
        
    # To minimize the maximum prefix sum, sort D_i in descending order.
    # Wait, let's re-verify:
    # If we have D = [2, -1, -3], prefix sums are:
    # [2, 1, -2] -> max is 2
    # If we sort descending: [2, -1, -3], max prefix is 2.
    # If we sort ascending: [-3, -1, 2], prefix sums are [-3, -4, -2], max is -2.
    # Yes, sorting ascending minimizes the maximum prefix sum.
    
    cards.sort()
    
    max_prefix_sum = -float('inf')
    current_prefix_sum = 0
    for d in cards:
        current_prefix_sum += d
        if current_prefix_sum > max_prefix_sum:
            max_prefix_sum = current_prefix_sum
            
    print(sum_b + max_prefix_sum)

if __name__ == '__main__':
    solve()
```
