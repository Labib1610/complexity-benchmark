import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (Argus and Midas) and a set of jobs.
    # Each job must be processed on Argus then Midas.
    # We want to minimize the makespan (total time to finish all jobs).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where Argus time (p1) < Midas time (p2)
    #    Set 2: jobs where Argus time (p1) >= Midas time (p2)
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    argus_time = 0
    midas_time = 0
    
    for p1, p2 in sequence:
        argus_time += p1
        # Midas can start only after Argus finishes this job
        # and after Midas finishes the previous job.
        if midas_time < argus_time:
            midas_time = argus_time + p2
        else:
            midas_time += p2
            
    print(midas_time)

if __name__ == '__main__':
    solve()
