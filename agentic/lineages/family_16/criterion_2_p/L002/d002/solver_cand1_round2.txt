import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        vessels.append((p1, p2))
        idx += 2
    
    # This is a classic Johnson's Rule problem variant.
    # To minimize the completion time of the last job in a two-machine flow shop:
    # Sort jobs such that those with p1 < p2 come first (sorted by p1 ascending),
    # and those with p1 >= p2 come last (sorted by p2 descending).
    
    group1 = []
    group2 = []
    for p1, p2 in vessels:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    roster = group1 + group2
    
    current_p1_time = 0
    current_p2_time = 0
    
    for p1, p2 in roster:
        current_p1_time += p1
        if current_p1_time > current_p2_time:
            current_p2_time = current_p1_time + p2
        else:
            current_p2_time += p2
            
    print(current_p2_time)

if __name__ == '__main__':
    solve()
