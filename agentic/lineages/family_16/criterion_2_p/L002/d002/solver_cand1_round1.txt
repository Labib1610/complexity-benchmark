import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        vessels.append((p1, p2))
        idx += 2
    
    # Johnson's Rule for minimizing makespan in a two-machine flow shop:
    # 1. Divide jobs into two sets:
    #    Set 1: p1 < p2 (sort by p1 ascending)
    #    Set 2: p1 >= p2 (sort by p2 descending)
    # 2. Sequence is Set 1 followed by Set 2.
    
    set1 = []
    set2 = []
    for p1, p2 in vessels:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    order = set1 + set2
    
    # Calculate completion time
    # time_p1: time when current vessel finishes at Primary Toll Station
    # time_p2: time when current vessel finishes at Secondary Customs House
    time_p1 = 0
    time_p2 = 0
    
    for p1, p2 in order:
        time_p1 += p1
        # The vessel can start at Secondary Customs House only after:
        # 1. It finishes at Primary Toll Station (time_p1)
        # 2. The previous vessel finishes at Secondary Customs House (time_p2)
        time_p2 = max(time_p1, time_p2) + p2
        
    print(time_p2)

if __name__ == '__main__':
    solve()
