We can model this as a **3-edge-coloring** problem on a **3-regular graph** (since each station has degree 3).  

By **Kőnig’s line coloring theorem**, every bipartite graph has edge chromatic number equal to its maximum degree. But here the graph is not necessarily bipartite. However, for **3-regular graphs**, a 3-edge-coloring exists if and only if the graph is **bridgeless** (by **Tait’s theorem**).  

Since the problem guarantees a uniform degree of 3, we can check if the graph is **bridgeless** (i.e., no bridge). If it has a bridge, no valid 3-edge-coloring exists.  

We can find a 3-edge-coloring using a constructive algorithm:  
- Find a perfect matching, remove it, then the remaining graph is 2-regular (a union of cycles), which can be 2-colored easily.  

Here’s the implementation:  

```python
import sys
from collections import deque

def find_bridges(n, adj, edges):
    visited = [False] * n
    tin = [-1] * n
    low = [-1] * n
    timer = 0
    bridges = set()

    def dfs(v, p_edge):
        nonlocal timer
        visited[v] = True
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in adj[v]:
            if eid == p_edge:
                continue
            if visited[to]:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    bridges.add(eid)

    for i in range(n):
        if not visited[i]:
            dfs(i, -1)
    return bridges

def solve():
    input_data = sys.stdin.read().strip().split()
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    adj = [[] for _ in range(n)]
    idx = 2
    for eid in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        edges.append((u, v))
        adj[u].append((v, eid))
        adj[v].append((u, eid))

    # Check for bridges
    bridges = find_bridges(n, adj, edges)
    if bridges:
        print("NO")
        return

    # Find a perfect matching using simple greedy (since graph is 3-regular and bridgeless)
    # We'll use a standard algorithm: find a perfect matching via augmenting paths
    match = [-1] * n
    def augment(u, visited):
        for v, eid in adj[u]:
            if visited[v]:
                continue
            visited[v] = True
            if match[v] == -1 or augment(match[v], visited):
                match[v] = u
                match[u] = v
                return True
        return False

    for u in range(n):
        if match[u] == -1:
            visited = [False] * n
            visited[u] = True
            augment(u, visited)

    # Now match[] gives a perfect matching
    color = [-1] * m
    matched_edge = [-1] * m
    for u in range(n):
        if match[u] != -1 and u < match[u]:
            for v, eid in adj[u]:
                if v == match[u]:
                    color[eid] = 1
                    matched_edge[eid] = 1
                    break

    # Remaining edges form cycles (2-regular). Color them with 2 and 3.
    # Build adjacency for remaining edges
    rem_adj = [[] for _ in range(n)]
    for eid, (u, v) in enumerate(edges):
        if color[eid] == -1:
            rem_adj[u].append((v, eid))
            rem_adj[v].append((u, eid))

    visited = [False] * n
    for start in range(n):
        if not visited[start] and rem_adj[start]:
            # traverse cycle
            cycle_edges = []
            prev = -1
            cur = start
            while True:
                visited[cur] = True
                nxt = None
                for to, eid in rem_adj[cur]:
                    if eid != prev:
                        nxt = (to, eid)
                        break
                if nxt is None:
                    break
                cycle_edges.append(nxt[1])
                prev = nxt[1]
                cur = nxt[0]
                if cur == start:
                    break
            # color cycle alternately 2,3
            for i, eid in enumerate(cycle_edges):
                color[eid] = 2 if i % 2 == 0 else 3

    print("YES")
    print(" ".join(str(c) for c in color))

if __name__ == "__main__":
    solve()
```
