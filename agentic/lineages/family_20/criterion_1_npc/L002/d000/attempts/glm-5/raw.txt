import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check if 3-regular
    for u in range(1, n + 1):
        if len(adj[u]) != 3:
            print("NO")
            return
    
    # Find a cycle using DFS
    parent = [0] * (n + 1)
    parent_edge = [-1] * (n + 1)
    visited = [0] * (n + 1)
    cycle_nodes = []
    cycle_edges = []
    
    stack = [(1, 0, -1)]
    found = False
    while stack and not found:
        u, p, pe = stack.pop()
        if visited[u] == 0:
            visited[u] = 1
            parent[u] = p
            parent_edge[u] = pe
            for v, eid in adj[u]:
                if visited[v] == 0:
                    stack.append((v, u, eid))
                elif v != p and not found:
                    # Found cycle
                    found = True
                    cycle_edges.append(eid)
                    cur = u
                    while cur != v:
                        cycle_edges.append(parent_edge[cur])
                        cur = parent[cur]
                    cycle_nodes.append(v)
                    cur = u
                    while cur != v:
                        cycle_nodes.append(cur)
                        cur = parent[cur]
    
    if not cycle_edges:
        print("NO")
        return
    
    L = len(cycle_edges)
    # Check if cycle length is even
    if L % 2 == 1:
        print("NO")
        return
    
    # Assign colors to cycle edges: alternating 1, 2
    edge_color = [0] * m
    for idx, eid in enumerate(cycle_edges):
        edge_color[eid] = 1 if idx % 2 == 0 else 2
    
    # For each cycle node, determine the color of its chord edge
    cycle_set = set(cycle_nodes)
    chord_colors = {}
    for node in cycle_nodes:
        # Find the edge not in cycle
        for v, eid in adj[node]:
            if eid not in set(cycle_edges):
                # Determine color for this chord edge
                # Find the two cycle edges incident to node
                ce = []
                for vv, eeid in adj[node]:
                    if eeid in set(cycle_edges):
                        ce.append(edge_color[eeid])
                if len(ce) == 2:
                    chord_colors[eid] = 6 - ce[0] - ce[1]
                    edge_color[eid] = chord_colors[eid]
    
    # BFS to color remaining edges
    visited_node = [False] * (n + 1)
    from collections import deque
    q = deque(cycle_nodes)
    for node in cycle_nodes:
        visited_node[node] = True
    
    while q:
        u = q.popleft()
        # Get colors of edges incident to u
        colors = []
        for v, eid in adj[u]:
            if edge_color[eid] != 0:
                colors.append(edge_color[eid])
        if len(colors) != 3:
            continue  # Skip if not all edges colored yet
        
        for v, eid in adj[u]:
            if edge_color[eid] == 0:
                # This edge is not colored, but u has all 3 colors
                # This shouldn't happen in a proper 3-regular graph
                pass
    
    # Alternative: BFS from cycle nodes, coloring edges as we go
    # Reset edge_color
    edge_color = [0] * m
    for idx, eid in enumerate(cycle_edges):
        edge_color[eid] = 1 if idx % 2 == 0 else 2
    
    # Color chord edges
    cycle_edge_set = set(cycle_edges)
    for node in cycle_nodes:
        for v, eid in adj[node]:
            if eid not in cycle_edge_set:
                ce = []
                for vv, eeid in adj[node]:
                    if eeid in cycle_edge_set:
                        ce.append(edge_color[eeid])
                if len(ce) == 2:
                    edge_color[eid] = 6 - ce[0] - ce[1]
    
    # BFS to propagate colors
    visited_node = [False] * (n + 1)
    q = deque()
    for node in cycle_nodes:
        visited_node[node] = True
        q.append(node)
    
    while q:
        u = q.popleft()
        # Get colored edges at u
        colored = []
        uncolored = []
        for v, eid in adj[u]:
            if edge_color[eid] != 0:
                colored.append(edge_color[eid])
            else:
                uncolored.append((v, eid))
        
        if len(colored) == 3:
            continue  # All edges colored
        
        if len(colored) == 2 and len(uncolored) == 1:
            v, eid = uncolored[0]
            edge_color[eid] = 6 - colored[0] - colored[1]
            if not visited_node[v]:
                visited_node[v] = True
                q.append(v)
    
    # Verify all edges are colored
    for c in edge_color:
        if c == 0:
            print("NO")
            return
    
    # Verify the coloring is valid
    for u in range(1, n + 1):
        colors = set()
        for v, eid in adj[u]:
            colors.add(edge_color[eid])
        if len(colors) != 3:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, edge_color)))

solve()
