import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    edges = []
    for i in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check degree is 3 for all nodes
    for i in range(n):
        if len(adj[i]) != 3:
            print("NO")
            return
    
    answer = [0] * m
    visited = [False] * n
    
    # Process each connected component
    for start in range(n):
        if visited[start]:
            continue
        
        # BFS to collect component
        component_nodes = []
        component_edges = []
        q = [start]
        visited[start] = True
        while q:
            u = q.pop()
            component_nodes.append(u)
            for v, eid in adj[u]:
                component_edges.append(eid)
                if not visited[v]:
                    visited[v] = True
                    q.append(v)
        
        # Remove duplicate edges
        component_edges = list(set(component_edges))
        num_nodes = len(component_nodes)
        num_edges = len(component_edges)
        
        # Check Eulerian: 3-regular graph must have even number of nodes
        if num_nodes % 2 != 0:
            print("NO")
            return
        
        # Build local adjacency for this component
        local_adj = [[] for _ in range(n)]
        for eid in component_edges:
            u, v = edges[eid]
            local_adj[u].append((v, eid))
            local_adj[v].append((u, eid))
        
        # Find a cycle using DFS
        cycle_edges = set()
        parent = [-1] * n
        parent_edge = [-1] * n
        stack = [component_nodes[0]]
        parent[component_nodes[0]] = component_nodes[0]
        found_cycle = False
        
        while stack and not found_cycle:
            u = stack.pop()
            for v, eid in local_adj[u]:
                if v == parent[u]:
                    continue
                if parent[v] == -1:
                    parent[v] = u
                    parent_edge[v] = eid
                    stack.append(v)
                else:
                    # Found cycle: trace back from u and v
                    cycle_edges.add(eid)
                    pu, pv = u, v
                    while pu != pv:
                        if pu != start:
                            cycle_edges.add(parent_edge[pu])
                            pu = parent[pu]
                        if pv != start:
                            cycle_edges.add(parent_edge[pv])
                            pv = parent[pv]
                    found_cycle = True
                    break
        
        # Assign frequencies to cycle edges: alternating 1 and 2
        cycle_list = list(cycle_edges)
        for i, eid in enumerate(cycle_list):
            answer[eid] = 1 if i % 2 == 0 else 2
        
        # Assign remaining edges: frequency 3
        for eid in component_edges:
            if eid not in cycle_edges:
                answer[eid] = 3
    
    # Verify solution
    for u in range(n):
        freqs = []
        for v, eid in adj[u]:
            freqs.append(answer[eid])
        if sorted(freqs) != [1, 2, 3]:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, answer)))

if __name__ == "__main__":
    solve()
