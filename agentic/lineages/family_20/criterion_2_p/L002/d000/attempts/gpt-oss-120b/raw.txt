import sys
sys.setrecursionlimit(1 << 25)

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    nL = int(next(it)); nR = int(next(it)); m = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    # compute degrees and Δ
    degL = [0] * nL
    degR = [0] * nR
    for u, v in edges:
        degL[u] += 1
        degR[v] += 1
    Delta = max(max(degL) if nL else 0, max(degR) if nR else 0)
    if Delta == 0:
        print("YES")
        print()
        return

    # adjacency: for each vertex store list of (neighbor, edge_id)
    adjL = [[] for _ in range(nL)]
    adjR = [[] for _ in range(nR)]
    for idx, (u, v) in enumerate(edges):
        adjL[u].append((v, idx))
        adjR[v].append((u, idx))

    # color of each edge, 0 = uncolored
    col = [0] * m

    # for each vertex, color -> edge id (0 means free)
    # we store as dicts because degree may be much smaller than Delta
    usedL = [dict() for _ in range(nL)]
    usedR = [dict() for _ in range(nR)]

    # helper to find a free color at vertex
    def free_color(used):
        # used is dict of color->edge
        # we need any color in 1..Delta not in used
        # since Delta may be large, we keep a pointer per vertex
        # but here we just scan (degree is small enough)
        for c in range(1, Delta + 1):
            if c not in used:
                return c
        return None  # should never happen

    # BFS to find alternating path to free a color
    from collections import deque
    def bfs(u_start, color_needed):
        # we want to make color_needed free at u_start (left side)
        # BFS over vertices, edges colored with color_needed or some other color
        # state: vertex (side, idx)
        # side 0 = left, 1 = right
        q = deque()
        prev = {}  # (side, v) -> (prev_side, prev_v, edge_id, swapped_color)
        q.append((0, u_start))
        prev[(0, u_start)] = None
        target = None
        while q:
            side, v = q.popleft()
            if side == 0:  # left
                used = usedL[v]
                # if color_needed free here, we reached target
                if color_needed not in used:
                    target = (side, v)
                    break
                # edge with that color
                e_id = used[color_needed]
                u2, v2 = edges[e_id]
                # move to right vertex v2
                nxt = (1, v2)
                if nxt not in prev:
                    prev[nxt] = (side, v, e_id, color_needed)
                    q.append(nxt)
            else:  # right
                used = usedR[v]
                # try any color c that is free at this right vertex
                for c in range(1, Delta + 1):
                    if c not in used:
                        # we can finish by swapping along path with this free color
                        target = (side, v, c)
                        prev[(side, v, c)] = (side, v)  # dummy
                        q.clear()
                        break
                else:
                    # all colors used, explore edges of each used color
                    for c, e_id in used.items():
                        u2, v2 = edges[e_id]
                        nxt = (0, u2)
                        if nxt not in prev:
                            prev[nxt] = (side, v, e_id, c)
                            q.append(nxt)
        if target is None:
            return None
        # reconstruct path and perform swaps
        # target can be (0, v) meaning free at left, or (1, v, c) meaning free at right with color c
        if target[0] == 0:
            # free at left vertex, just recolor along path
            cur = target
            while prev[cur] is not None:
                p = prev[cur]
                side, v = p[0], p[1]
                e_id, col_swap = p[2], p[3]
                # recolor edge e_id to col_swap
                # remove old assignment
                u_e, v_e = edges[e_id]
                if side == 0:
                    # came from left via color_needed
                    del usedL[u_e][col_swap]
                    del usedR[v_e][col_swap]
                else:
                    del usedL[u_e][col_swap]
                    del usedR[v_e][col_swap]
                # assign new color (the one we came from)
                new_c = col_swap if side == 0 else col_swap
                usedL[u_e][new_c] = e_id
                usedR[v_e][new_c] = e_id
                col[e_id] = new_c
                cur = p[:2]
        else:
            # free at right with some color c_free
            _, v_right, c_free = target
            # first recolor the edge that brings us to this right vertex
            cur = (1, v_right)
            while prev[cur] is not None:
                p = prev[cur]
                side, v = p[0], p[1]
                e_id, col_swap = p[2], p[3]
                u_e, v_e = edges[e_id]
                # remove old
                del usedL[u_e][col_swap]
                del usedR[v_e][col_swap]
                # assign new color
                new_c = col_swap if side == 0 else col_swap
                usedL[u_e][new_c] = e_id
                usedR[v_e][new_c] = e_id
                col[e_id] = new_c
                cur = (0, u_e) if side == 0 else (1, v_e)
            # finally assign the original needed color at start vertex
        return True

    # Main coloring loop
    for eid, (u, v) in enumerate(edges):
        if col[eid]:
            continue
        # try to assign a free color directly
        used_u = usedL[u]
        used_v = usedR[v]
        for c in range(1, Delta + 1):
            if c not in used_u and c not in used_v:
                used_u[c] = eid
                used_v[c] = eid
                col[eid] = c
                break
        else:
            # need recoloring, pick any color cu missing at u
            for cu in range(1, Delta + 1):
                if cu not in used_u:
                    break
            # now we need to make cu also missing at v
            bfs(u, cu)

    print("YES")
    print(' '.join(map(str, col)))

if __name__ == "__main__":
    main()
