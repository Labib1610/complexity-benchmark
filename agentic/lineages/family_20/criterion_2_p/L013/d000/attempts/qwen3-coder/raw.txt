import sys
from collections import defaultdict

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n_P, n_S, m = map(int, sys.stdin.readline().split())
    
    edges = []
    deg_primary = [0] * (n_P + 1)
    deg_secondary = [0] * (n_S + 1)
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        deg_primary[u] += 1
        deg_secondary[v] += 1
    
    delta = max(max(deg_primary[1:]), max(deg_secondary[1:])) if m > 0 else 0
    
    if delta == 0:
        print("YES")
        return
    
    adj_primary = [[] for _ in range(n_P + 1)]
    adj_secondary = [[] for _ in range(n_S + 1)]
    
    for idx, (u, v) in enumerate(edges):
        adj_primary[u].append((v, idx))
        adj_secondary[v].append((u, idx))
    
    color = [0] * m
    
    for u in range(1, n_P + 1):
        if not adj_primary[u]:
            continue
        available = [True] * (delta + 1)
        for v, idx in adj_primary[u]:
            if color[idx] != 0:
                available[color[idx]] = False
        
        next_color = 1
        for v, idx in adj_primary[u]:
            if color[idx] == 0:
                while not available[next_color]:
                    next_color += 1
                color[idx] = next_color
                available[next_color] = False
                next_color += 1
    
    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    main()
