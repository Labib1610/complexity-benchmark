import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    import sys
    input = sys.stdin.readline
    nL, nR, m = map(int, input().split())
    edges = [tuple(map(int, input().split())) for _ in range(m)]
    if m == 0:
        print("YES")
        print()
        return

    # degrees
    degL = [0] * (nL + 1)
    degR = [0] * (nR + 1)
    for u, v in edges:
        degL[u] += 1
        degR[v] += 1
    Delta = max(max(degL), max(degR))

    # adjacency lists of edge indices
    adjL = [[] for _ in range(nL + 1)]
    adjR = [[] for _ in range(nR + 1)]
    for idx, (u, v) in enumerate(edges):
        adjL[u].append(idx)
        adjR[v].append(idx)

    # color of each edge, 0 = uncolored
    col = [0] * m

    # for each vertex, map color -> edge index using that color
    usedL = [{} for _ in range(nL + 1)]
    usedR = [{} for _ in range(nR + 1)]

    # helper to try to assign a free color, with possible recoloring via alternating path
    def assign(e_idx):
        u, v = edges[e_idx]
        # try to find a free color directly
        for c in range(1, Delta + 1):
            if c not in usedL[u] and c not in usedR[v]:
                col[e_idx] = c
                usedL[u][c] = e_idx
                usedR[v][c] = e_idx
                return True

        # need recoloring: find an alternating path starting from u
        # BFS over vertices, storing (vertex, side, incoming_color)
        from collections import deque
        q = deque()
        # start from u on left side, with a missing color cu (any not in usedL[u])
        for cu in range(1, Delta + 1):
            if cu not in usedL[u]:
                break
        q.append((u, 0, cu))  # side 0 = left, 1 = right
        parent = {}  # (vertex, side) -> (prev_vertex, prev_side, color_used_on_edge)
        visited = set()
        visited.add((u, 0))

        target = None
        while q and target is None:
            vert, side, needc = q.popleft()
            if side == 0:  # left vertex
                # we can go via edge that currently uses needc on this vertex
                if needc in usedL[vert]:
                    e2 = usedL[vert][needc]
                    u2, v2 = edges[e2]
                    nxt = (v2, 1)  # go to right vertex
                    if nxt not in visited:
                        visited.add(nxt)
                        parent[nxt] = (vert, side, needc)
                        # at right vertex, we look for a free color different from needc
                        for c2 in range(1, Delta + 1):
                            if c2 not in usedR[v2]:
                                target = (nxt, c2)
                                break
                        if target is None:
                            # otherwise continue BFS with the color that is used on right vertex
                            for c2 in range(1, Delta + 1):
                                if c2 in usedR[v2]:
                                    e3 = usedR[v2][c2]
                                    u3, v3 = edges[e3]
                                    nxt2 = (u3, 0)
                                    if nxt2 not in visited:
                                        visited.add(nxt2)
                                        parent[nxt2] = (vert, side, needc)
                                        q.append((u3, 0, c2))
            else:  # right vertex
                # similar
                if needc in usedR[vert]:
                    e2 = usedR[vert][needc]
                    u2, v2 = edges[e2]
                    nxt = (u2, 0)
                    if nxt not in visited:
                        visited.add(nxt)
                        parent[nxt] = (vert, side, needc)
                        for c2 in range(1, Delta + 1):
                            if c2 not in usedL[u2]:
                                target = (nxt, c2)
                                break
                        if target is None:
                            for c2 in range(1, Delta + 1):
                                if c2 in usedL[u2]:
                                    e3 = usedL[u2][c2]
                                    u3, v3 = edges[e3]
                                    nxt2 = (v3, 1)
                                    if nxt2 not in visited:
                                        visited.add(nxt2)
                                        parent[nxt2] = (vert, side, needc)
                                        q.append((v3, 1, c2))

        if target is None:
            return False  # should not happen

        # reconstruct path and recolor
        (end_vert, end_side), free_color = target
        cur = (end_vert, end_side)
        while cur != (u, 0):
            prev_vert, prev_side, col_used = parent[cur]
            # edge between prev_vert and cur_vert uses col_used, we will shift it to free_color
            if prev_side == 0:
                # edge is (prev_vert, cur_vert)
                e_idx2 = usedL[prev_vert][col_used]
                # reassign this edge to free_color
                col[e_idx2] = free_color
                # update maps
                del usedL[prev_vert][col_used]
                del usedR[cur_vert][col_used]
                usedL[prev_vert][free_color] = e_idx2
                usedR[cur_vert][free_color] = e_idx2
            else:
                e_idx2 = usedR[prev_vert][col_used]
                col[e_idx2] = free_color
                del usedR[prev_vert][col_used]
                del usedL[cur_vert][col_used]
                usedR[prev_vert][free_color] = e_idx2
                usedL[cur_vert][free_color] = e_idx2
            # move forward
            cur = (prev_vert, prev_side)
            free_color = col_used

        # finally assign original edge
        col[e_idx] = free_color
        usedL[u][free_color] = e_idx
        usedR[v][free_color] = e_idx
        return True

    for i in range(m):
        if not assign(i):
            print("NO")
            return

    print("YES")
    print(' '.join(map(str, col)))

if __name__ == "__main__":
    solve()
