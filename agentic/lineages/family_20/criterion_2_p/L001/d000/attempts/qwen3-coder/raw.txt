import sys
from collections import defaultdict
import heapq

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n_U = int(data[idx])
    idx += 1
    n_D = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    
    edges = []
    degree_u = [0] * (n_U + 1)
    degree_d = [0] * (n_D + 1)
    
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        edges.append((u, v))
        degree_u[u] += 1
        degree_d[v] += 1
    
    K = max(max(degree_u[1:]), max(degree_d[1:]))
    
    # For each port, maintain a set of used windows
    used_u = [set() for _ in range(n_U + 1)]
    used_d = [set() for _ in range(n_D + 1)]
    
    result = []
    
    # We'll process cables in a specific order to make assignment easier.
    # A greedy approach: process cables in the order they appear,
    # and for each cable, assign the smallest available window.
    
    for u, v in edges:
        # Find the smallest window not used by either u or v
        win = 1
        while win in used_u[u] or win in used_d[v]:
            win += 1
        used_u[u].add(win)
        used_d[v].add(win)
        result.append(str(win))
    
    print("YES")
    print(' '.join(result))

if __name__ == "__main__":
    main()
