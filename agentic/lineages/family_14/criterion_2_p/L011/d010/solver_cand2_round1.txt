```python
import sys

# The problem states that the set of n registers is fully permutable using m = n-1 swaps.
# This means the swaps form a tree structure (a spanning tree of the n registers).
# A subset of registers is fully permutable if and only if the subgraph induced by 
# the remaining registers is connected.
# We want to lock down registers such that the remaining set of registers S 
# contains all t target registers and the induced subgraph is connected.
# To minimize the total fatigue cost, we want to keep the smallest possible 
# connected subgraph that includes all t target registers.
# This is exactly the Steiner Tree problem on a tree, which is the unique path 
# connecting all target nodes in the tree.
# The edges that must remain are those on the unique paths between any two target nodes.
# Any edge not on the path between any two target nodes can be removed by locking 
# the registers in the subtrees that do not contain any target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Identify edges that are part of the Steiner Tree (the minimal subtree 
    # connecting all target nodes).
    # An edge is in the Steiner Tree if and only if both components formed by 
    # removing it contain at least one target node.
    
    has_target = [False] * (n + 1)
    for node in targets:
        has_target[node] = True
        
    subtree_targets = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if has_target[u]:
            subtree_targets[u] += 1
        if parent[u] != 0:
            subtree_targets[parent[u]] += subtree_targets[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # Edge is (i, parent[i])
        # It is in the Steiner Tree if the number of targets in the subtree 
        # of i is > 0 and < t.
        count = subtree_targets[i]
        if count > 0 and count < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
