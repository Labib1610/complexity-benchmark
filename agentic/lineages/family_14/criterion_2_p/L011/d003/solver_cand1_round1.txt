import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    anchors = [int(x) for x in input_data[3:3+t]]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The problem asks for the minimum cost to connect all anchor currencies.
    # Since m = n - 1, the graph is a tree.
    # In a tree, there is a unique path between any two nodes.
    # To connect all anchor nodes, we must include all edges that lie on the 
    # unique paths between any two anchor nodes.
    
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Root the tree at the first anchor
    root = anchors[0]
    parent = [0] * (n + 1)
    edge_to_parent = [0] * (n + 1)
    is_anchor = [False] * (n + 1)
    for a in anchors:
        is_anchor[a] = True
        
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    order = []
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent[v] = w
                stack.append(v)
                
    # subtree_has_anchor[u] is true if the subtree at u contains an anchor
    subtree_has_anchor = [False] * (n + 1)
    for i in range(1, n + 1):
        if is_anchor[i]:
            subtree_has_anchor[i] = True
            
    # Process in reverse topological order (bottom-up)
    total_cost = 0
    for u in reversed(order):
        if u == root:
            continue
        if subtree_has_anchor[u]:
            total_cost += edge_to_parent[u]
            subtree_has_anchor[parent[u]] = True
            
    print(total_cost)

solve()
