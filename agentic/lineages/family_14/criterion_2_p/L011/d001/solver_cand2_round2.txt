```python
import sys

# The problem states there are n sites and exactly n-1 cabling options.
# This means the graph is a tree.
# We need to find the minimum cost to connect a subset of t required sites.
# In a tree, the minimum cost to connect a set of nodes is the sum of the weights
# of all edges that lie on the paths between any two required nodes.
# An edge is part of the path between two required nodes if and only if
# removing that edge splits the tree into two components, each containing at least one required node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    required = set()
    idx = 3
    for _ in range(t):
        required.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many required nodes are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    subtree_required = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in reversed(order):
        if node in required:
            subtree_required[node] += 1
        if parent[node] != 0:
            subtree_required[parent[node]] += subtree_required[node]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the subtree at i contains some required nodes, but not all,
        # then the edge connecting i to its parent must be part of the Steiner tree.
        count = subtree_required[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
