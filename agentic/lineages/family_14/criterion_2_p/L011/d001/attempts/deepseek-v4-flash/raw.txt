 We can solve this by building a **minimum spanning tree (MST)** that connects all core facilities.  
Since the graph is a tree (m = n − 1), we can:

1. Mark core facilities.
2. Remove unnecessary branches that don’t lead to any core facility.
3. Sum the weights of the remaining edges.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    core = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # DFS to find which edges are needed
    visited = [False] * (n + 1)
    needed = [False] * (n + 1)
    
    def dfs(u, parent):
        visited[u] = True
        has_core = u in core
        for v, w in adj[u]:
            if v == parent:
                continue
            if not visited[v]:
                if dfs(v, u):
                    has_core = True
                    needed[v] = True
        return has_core
    
    # Start DFS from any core node
    start = next(iter(core))
    dfs(start, -1)
    
    total = 0
    for u, v, w in edges:
        if needed[u] or needed[v]:
            total += w
    
    print(total)

if __name__ == "__main__":
    solve()
```
