```python
import sys

# The problem states that the graph is a tree (n servers, n-1 channels).
# To connect all critical servers, we must include all edges that lie on the 
# unique path between any two critical servers.
# An edge (u, v) with weight w is part of the required set if and only if 
# removing it splits the tree into two components, each containing at least 
# one critical server.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_servers = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count_critical[u] will store the number of critical servers in the subtree rooted at u
    count_critical = [0] * (n + 1)
    total_expense = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    for u in reversed(order):
        if u in critical_servers:
            count_critical[u] += 1
        if parent[u] != 0:
            p, w = parent[u]
            count_critical[p] += count_critical[u]
            # If the edge (u, p) separates critical servers, it must be included
            if count_critical[u] > 0 and count_critical[u] < t:
                total_expense += w
                
    print(total_expense)

if __name__ == '__main__':
    solve()
```
