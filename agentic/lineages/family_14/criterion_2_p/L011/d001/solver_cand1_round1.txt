```python
import sys

# The problem states m = n - 1, which means the graph is a tree.
# If the graph is a tree, there is only one unique path between any two nodes.
# To connect all core facilities, we must include all edges that lie on the 
# unique paths between any two core facilities.
# This is equivalent to finding the smallest subtree that contains all core nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    core_nodes = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Root the tree at an arbitrary core node
    root = next(iter(core_nodes))
    
    # We need to determine for each edge if it lies on a path between two core nodes.
    # An edge (u, v) where v is a child of u lies on such a path if and only if
    # the subtree rooted at v contains at least one core node AND the rest of the
    # tree (outside the subtree of v) contains at least one core node.
    
    subtree_has_core = [False] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    parent[root] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_core_count = t
    subtree_core_count = [0] * (n + 1)
    for node in core_nodes:
        subtree_core_count[node] = 1
        
    ans = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == root:
            continue
        
        # If the subtree at u has some core nodes but not all, 
        # the edge to its parent must be included.
        if 0 < subtree_core_count[u] < total_core_count:
            ans += edge_to_parent_weight[u]
            
        if parent[u] != -1:
            subtree_core_count[parent[u]] += subtree_core_count[u]
            
    print(ans)

solve()
```
