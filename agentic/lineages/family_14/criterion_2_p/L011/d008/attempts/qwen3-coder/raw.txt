```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    target_vars = set()
    for _ in range(t):
        target_vars.add(int(data[idx]))
        idx += 1
    
    edges = []
    graph = defaultdict(list)
    
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((u, v, w))
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # If all variables are targets, we can't eliminate anything
    if len(target_vars) == n:
        total_cost = sum(w for _, _, w in edges)
        print(total_cost)
        return
    
    # We need to find the minimum spanning tree that connects all target nodes
    # This is essentially the Steiner tree problem on a tree (since m = n-1 and fully coupled)
    # Since the graph is a tree, we can do the following:
    # 1. Find the subtree that contains all target nodes
    # 2. The minimal cost is the sum of edge weights in this subtree
    
    # First, let's build the tree structure
    # Since it's a tree with n nodes and n-1 edges, we can represent it properly
    
    # Let's find which nodes are needed: all nodes that lie on paths between any two target nodes
    # To do that, we can find the union of paths between all pairs of targets
    # But since it's a tree, we can find the minimal subtree that connects all targets
    
    # Efficient approach:
    # 1. Root the tree at any node
    # 2. Do a DFS to compute parent and depth for each node
    # 3. For each target, mark all ancestors
    # 4. The minimal subtree includes all nodes that have at least one marked descendant (including themselves if they are targets)
    # 5. Count the edges in this subtree
    
    # Let's root the tree at node 1
    root = 1
    # Build adjacency list
    adj = defaultdict(list)
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # DFS to compute parent, depth, and subtree info
    parent = {}
    depth = {}
    children = defaultdict(list)
    
    def dfs(node, par, d):
        parent[node] = par
        depth[node] = d
        for nei, _ in adj[node]:
            if nei != par:
                children[node].append(nei)
                dfs(nei, node, d+1)
                
    dfs(root, -1, 0)
    
    # Function to find path between two nodes
    def get_path(u, v):
        path_u = []
        path_v = []
        orig_u, orig_v = u, v
        
        # Bring both nodes to same depth
        while depth[u] > depth[v]:
            path_u.append(u)
            u = parent[u]
        while depth[v] > depth[u]:
            path_v.append(v)
            v = parent[v]
            
        # Now move both up until they meet
        while u != v:
            path_u.append(u)
            path_v.append(v)
            u = parent[u]
            v = parent[v]
            
        path_u.append(u)  # Add LCA
        path = path_u + path_v[::-1]
        return path
    
    # Find all nodes on paths between target nodes
    # But we need a more efficient way
    
    # Better approach: 
    # Mark all target nodes
    # Do BFS from all targets simultaneously to find the minimal subtree
    
    # Actually, since it's a tree, the minimal connected subgraph containing all targets
    # is unique. We can find it by:
    # 1. Pick any target as root
    # 2. Do BFS/DFS to find "steiner tree" nodes
    # 3. A node is in the steiner tree if it has at least two branches leading to targets
    #    or it is a target itself, or it's on the path to a target
    
    # Most efficient: 
    # 1. Root tree arbitrarily
    # 2. For each node, compute if its subtree contains any target
    # 3. An edge is in the minimal solution if both endpoints have targets in their subtrees
    
    # Compute for each node if its subtree contains any target
    has_target = {}
    
    def dfs_subtree(node, par):
        has = node in target_vars
        for child in children[node]:
            if child != par:
                has |= dfs_subtree(child, node)
        has_target[node] = has
        return has
    
    dfs_subtree(root, -1)
    
    # Now count edges where both endpoints have targets in their parts of the tree
    total_cost = 0
    for u, v, w in edges:
        # Check if removing this edge separates targets
        # That happens when one side has targets and the other doesn't
        # So we want edges where both sides have targets
        pu = parent[u] if parent[u] != v else u
        pv = parent[v] if parent[v] != u else v
        
        # More precise check:
        # Edge {u,v} is needed if the component containing u when we remove this edge has targets
        # AND the component containing v has targets
        
        # When we remove edge (u,v), we get two components
        # One containing u (when we think of v as parent of u)  
        # One containing v (when we think of u as parent of v)
        
        # Let's orient edges according to the DFS tree
        if parent[u] == v:
            # v is parent of u
            needed = has_target[u] and (has_target[v] or v in target_vars or any(has_target[c] for c in children[v] if c != u))
            if has_target[u] and (has_target[v] or v in target_vars): 
                total_cost += w
        elif parent[v] == u:
            # u is parent of v
            needed = has_target[v] and (has_target[u] or u in target_vars or any(has_target[c] for c in children[u] if c != v))
            if has_target[v] and (has_target[u] or u in target_vars):
                total_cost += w
                
    # Simpler and more correct approach:
    # The minimal cost is the MST of the metric completion on target nodes
    # But since our graph is already a tree, it's just the cost of the subtree spanning all targets
    
    # Let's use a different method:
    # Find the set of all nodes that must be kept (those on paths between targets)
    # Then sum the costs of edges between these nodes
    
    # Nodes to keep: all nodes that are either targets or on paths between targets
    # In a tree, this is equivalent to finding the minimal subtree connecting all targets
    
    # Algorithm:
    # 1. Root the tree
    # 2. Mark all target nodes
    # 3. Propagate marks up to ancestors
    # 4. A node is in the final tree if it's marked and has at least one marked child OR if it's a target
    
    marked = [False] * (n + 1)
    is_target = [False] * (n + 1)
    for v in target_vars:
        is_target[v] = True
    
    # First pass: mark ancestors of targets
    def mark_ancestors(node):
        if marked[node]:
            return
        marked[node] = True
        if parent[node] != -1:
            mark_ancestors(parent[node])
            
    for v in target_vars:
        mark_ancestors(v)
        
    # Second pass: check which nodes should be kept
    keep = [False] * (n + 1)
    
    def should_keep(node):
        if keep[node]:
            return True
        if is_target[node]:
            keep[node] = True
            return True
            
        # Count how many children lead to targets
        count = 0
        for child in children[node]:
            if should_keep(child):
                count += 1
                
        if count >= 1:  # At least one branch leads to target
            keep[node] = True
            return True
        return False
        
    should_keep(root)
    
    # Sum up the costs of edges where both endpoints are kept
    result = 0
    for u, v, w in edges:
        if keep[u] and keep[v]:
            result += w
            
    print(result)

if __name__ == "__main__":
    main()
```
