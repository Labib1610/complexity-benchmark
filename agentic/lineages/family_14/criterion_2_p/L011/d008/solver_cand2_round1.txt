```python
import sys

# The problem states:
# Initial net liquidity = (number of assets) - (number of pairs) = 1.
# This means m = n - 1.
# A connected graph with n nodes and n-1 edges is a tree.
# When we remove an asset (node) and its incident edges, the net liquidity
# changes by:
#   -1 (asset removed) - (-degree of node) = degree - 1.
# For the net liquidity to remain 1, we must have degree - 1 = 0,
# so degree must be 1.
# Thus, we can only remove a leaf node if it is not a core asset.
# This is equivalent to pruning all non-core nodes that are not part of the
# unique path between core assets.
# The remaining structure will be the minimal subtree containing all core assets.
# The cost is the sum of weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core_assets = set()
    idx = 3
    for _ in range(t):
        core_assets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t == 0:
        print(0)
        return
    if t == 1:
        print(0)
        return

    # We need the sum of weights of edges in the minimal subtree connecting all core assets.
    # This is the sum of weights of all edges that lie on a path between any two core assets.
    # An edge (u, v) with weight w is in the minimal subtree if and only if
    # removing it splits the tree into two components, each containing at least one core asset.
    
    # Count core assets in each subtree formed by removing an edge.
    # Root the tree at an arbitrary core asset.
    root = next(iter(core_assets))
    subtree_core_count = [0] * (n + 1)
    
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in core_assets:
            subtree_core_count[u] += 1
        if parent[u] != 0:
            subtree_core_count[parent[u]] += subtree_core_count[u]
            
    total_cost = 0
    for i in range(1, n + 1):
        if parent[i] != 0:
            # Edge is between i and parent[i]
            # If this edge separates the tree into two parts, one with k cores
            # and one with (t - k) cores, it is part of the minimal subtree
            # if k > 0 and t - k > 0.
            k = subtree_core_count[i]
            if k > 0 and k < t:
                total_cost += edge_to_parent_weight[i]
                
    print(total_cost)

solve()
```
