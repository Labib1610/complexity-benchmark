```python
import sys

# The problem states that the matrix H is symmetric, sparse (K = N-1), 
# and the initial set of N indices is non-separable.
# This implies the graph formed by the non-zero entries is a tree.
# A subset of vertices in a tree is non-separable if and only if the 
# induced subgraph is connected.
# We want to remove nodes such that the remaining set contains all T invariant nodes
# and the induced subgraph remains connected.
# The sum of all elements in the principal submatrix is 2 * sum of weights of edges
# in the induced subgraph. We want to minimize this sum.
# The minimum connected subgraph containing a set of nodes T is the Steiner Tree.
# In a tree, the Steiner Tree is unique: it is the union of all paths between 
# any two nodes in T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    K = int(input_data[1])
    T = int(input_data[2])
    
    invariant = set()
    idx = 3
    for _ in range(T):
        invariant.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(N + 1)]
    for _ in range(K):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if T == 0:
        print(0)
        return
    if T == 1:
        print(0)
        return

    # Find the Steiner Tree:
    # In a tree, the Steiner tree is the minimal subtree connecting all nodes in T.
    # We can find this by pruning all leaves that are not in T, repeatedly.
    
    degree = [0] * (N + 1)
    for i in range(1, N + 1):
        degree[i] = len(adj[i])
        
    is_in_steiner = [True] * (N + 1)
    queue = [i for i in range(1, N + 1) if degree[i] == 1 and i not in invariant]
    
    while queue:
        u = queue.pop()
        is_in_steiner[u] = False
        for v, w in adj[u]:
            if is_in_steiner[v]:
                degree[v] -= 1
                if degree[v] == 1 and v not in invariant:
                    queue.append(v)
                    
    total_weight = 0
    for u in range(1, N + 1):
        if is_in_steiner[u]:
            for v, w in adj[u]:
                if is_in_steiner[v] and u < v:
                    total_weight += w
                    
    print(total_weight)

solve()
```
