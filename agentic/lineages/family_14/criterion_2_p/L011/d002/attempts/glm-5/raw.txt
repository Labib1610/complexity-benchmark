```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1

    is_reserve = bytearray(n + 1)
    for _ in range(t):
        c = int(data[idx]); idx += 1
        is_reserve[c] = 1

    edges = []
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        edges.append((w, u, v))

    edges.sort()

    parent = list(range(n + 1))
    size = [1] * (n + 1)
    reserve_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if is_reserve[i]:
            reserve_count[i] = 1

    def find(x):
        root = x
        while parent[root] != root:
            root = parent[root]
        while parent[x] != root:
            parent[x], x = root, parent[x]
        return root

    total = 0
    for w, u, v in edges:
        ru = find(u)
        rv = find(v)
        if ru == rv:
            continue
        if reserve_count[ru] > 0 and reserve_count[rv] > 0:
            total += w
        if size[ru] < size[rv]:
            ru, rv = rv, ru
        parent[rv] = ru
        size[ru] += size[rv]
        reserve_count[ru] += reserve_count[rv]

    sys.stdout.write(str(total))

solve()
```
