```python
import sys

# The problem states the graph is a tree (N nodes, N-1 edges, connected).
# We need to find the minimum cost to connect all core facilities.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of core facilities, we must include all edges that lie
# on the paths between any two core facilities.
# This is equivalent to finding the minimal subtree containing all core nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    K = int(input_data[2])
    
    if K <= 1:
        print(0)
        return
    
    core_nodes = set(map(int, input_data[3:3+K]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + K
    for _ in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to identify which edges are part of the minimal subtree.
    # An edge is part of the subtree if and only if the subtree formed by
    # removing that edge has core nodes in both components.
    
    # Root the tree at an arbitrary core node
    root = next(iter(core_nodes))
    
    # Count core nodes in each subtree
    core_count = [0] * (N + 1)
    for node in core_nodes:
        core_count[node] = 1
        
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [root]
    visited = [False] * (N + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order to propagate core counts
    for i in range(N - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if core_count[u] > 0 and core_count[u] < K:
            total_cost += edge_to_parent_weight[u]
        core_count[p] += core_count[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
