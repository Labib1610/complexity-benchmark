```python
import sys

# The problem states the catalog contains no redundant links and m = n - 1.
# This means the graph is a tree.
# We need to find the minimum cost to connect all core facilities.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of core facilities, we must include all edges that lie on
# the unique path between any two core facilities.
# An edge is part of the required set if and only if it separates the tree
# into two components, both of which contain at least one core facility.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    core_facilities = set()
    idx = 3
    for _ in range(t):
        core_facilities.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # subtree_cores[u] stores the number of core facilities in the subtree rooted at u
    subtree_cores = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in core_facilities:
            subtree_cores[u] += 1
        
        if parent[u] != 0:
            subtree_cores[parent[u]] += subtree_cores[u]
            # If this edge separates the tree into two parts both containing cores
            if subtree_cores[u] > 0 and subtree_cores[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
