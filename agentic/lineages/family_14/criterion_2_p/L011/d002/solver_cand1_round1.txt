```python
import sys

# The problem states that the core facilities are in a single contiguous segment
# with no redundant links (a tree structure). We need to find the minimum
# investment to connect all core facilities.
# Since the core facilities are already in a tree structure, any subset of
# links that connects all core facilities must include all edges on the
# unique paths between any two core facilities.
# Any edge that is part of a path between two core facilities must be included.
# An edge is part of a path between two core facilities if and only if
# removing that edge splits the tree into two components, each containing
# at least one core facility.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    core_facilities = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many core facilities are in each subtree.
    # Root the tree at an arbitrary core facility (e.g., the first one).
    root = next(iter(core_facilities))
    
    core_count = [0] * (n + 1)
    total_weight = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(root, -1, 0)]
    order = []
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    
    while stack:
        u, p, w = stack.pop()
        order.append(u)
        parent[u] = p
        edge_weight[u] = w
        if u in core_facilities:
            core_count[u] = 1
        for v, weight in adj[u]:
            if v != p:
                stack.append((v, u, weight))
                
    # Process in reverse topological order to propagate core counts
    for u in reversed(order):
        if parent[u] != -1:
            p = parent[u]
            core_count[p] += core_count[u]
            # If this edge separates the tree into two parts both containing
            # at least one core facility, it must be included.
            if core_count[u] > 0 and core_count[u] < t:
                total_weight += edge_weight[u]
                
    print(total_weight)

if __name__ == '__main__':
    solve()
```
