```python
import sys

# The problem states m = n - 1, which means the graph is a tree (or a forest).
# Since all core facilities are guaranteed to be connectable, we are looking for
# the sum of weights of all edges that lie on the unique paths between any two
# core facilities.
# In a tree, an edge is part of the path between two core facilities if and only
# if removing that edge splits the tree into two components, each containing at
# least one core facility.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core_facilities = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Use DFS to count how many core facilities are in each subtree
    # Since n is up to 200,000, we need to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    core_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if i in core_facilities:
            core_count[i] = 1
            
    total_investment = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        p = parent[u]
        core_count[p] += core_count[u]
        # If this subtree contains some but not all core facilities,
        # the edge connecting it to its parent must be included.
        if 0 < core_count[u] < t:
            total_investment += edge_weight[u]
            
    print(total_investment)

solve()
```
