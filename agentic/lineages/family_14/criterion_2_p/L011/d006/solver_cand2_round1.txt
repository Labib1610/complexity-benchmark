```python
import sys

# The problem states m = n - 1 and the region is fully contiguous.
# This means the parcels and borders form a tree structure.
# A subset of nodes in a tree is contiguous if and only if it forms a subtree.
# The smallest contiguous set containing all t heritage sites is the minimal subtree
# connecting all t heritage sites (the Steiner Tree).
# The internal maintenance cost is the sum of weights of all edges where both
# endpoints are in the subtree.
# In a tree, if we select a subtree, the edges "entirely within" the district
# are exactly the edges that belong to the minimal subtree connecting the heritage sites.
# Thus, the cost is simply the sum of weights of all edges in the minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    heritage_sites = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # To find the sum of weights of the minimal subtree:
    # Root the tree at an arbitrary heritage site.
    # A node is in the minimal subtree if it is a heritage site or if it has
    # at least two branches leading to heritage sites.
    # Actually, an edge (u, v) where v is a child of u is in the minimal subtree
    # if and only if the subtree rooted at v contains at least one heritage site
    # AND the rest of the tree (outside the subtree at v) contains at least one heritage site.
    
    root = next(iter(heritage_sites))
    has_heritage = [False] * (n + 1)
    for h in heritage_sites:
        has_heritage[h] = True
        
    total_heritage = t
    subtree_heritage = [0] * (n + 1)
    edge_weights = {}
    
    parent = [0] * (n + 1)
    order = []
    stack = [root]
    parent[root] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_weights[v] = w
                stack.append(v)
                
    for u in reversed(order):
        subtree_heritage[u] += (1 if has_heritage[u] else 0)
        if parent[u] != -1:
            subtree_heritage[parent[u]] += subtree_heritage[u]
            
    total_cost = 0
    for i in range(1, n + 1):
        if parent[i] != -1:
            # Edge (i, parent[i]) is in the minimal subtree if
            # 0 < subtree_heritage[i] < total_heritage
            if 0 < subtree_heritage[i] < total_heritage:
                total_cost += edge_weights[i]
                
    print(total_cost)

solve()
```
