```python
import sys

# The problem states n assets and m = n-1 hedging contracts.
# Since m = n-1 and the graph is fully market-linked, the graph is a tree.
# A subset of assets is market-linked if it is connected.
# We need to select a connected subset of assets that includes all t mandatory assets.
# The minimum connected subgraph containing a set of nodes is the Steiner Tree.
# In a tree, the Steiner Tree is the unique minimal subtree connecting all mandatory nodes.
# This subtree consists of all nodes on the simple paths between any two mandatory nodes.
# The cost is the sum of weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    mandatory = set()
    idx = 3
    for _ in range(t):
        mandatory.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Identify nodes that are part of the minimal subtree.
    # A node is in the minimal subtree if it lies on a path between two mandatory nodes.
    # This is equivalent to saying the subtree rooted at this node contains at least one
    # mandatory node and the rest of the tree contains at least one mandatory node.
    
    has_mandatory = [False] * (n + 1)
    for node in mandatory:
        has_mandatory[node] = True
        
    # Count mandatory nodes in subtree
    subtree_mandatory_count = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if has_mandatory[u]:
            subtree_mandatory_count[u] += 1
        if parent[u] != 0:
            subtree_mandatory_count[parent[u]] += subtree_mandatory_count[u]
            
    total_weight = 0
    for i in range(2, n + 1):
        # Edge to parent is part of the Steiner tree if the subtree at i
        # contains some mandatory nodes and the rest of the tree contains some.
        count = subtree_mandatory_count[i]
        if count > 0 and count < t:
            total_weight += edge_to_parent_weight[i]
            
    print(total_weight)

solve()
```
