```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    anchors = set(map(int, sys.stdin.readline().split()))
    
    # Since m = n - 1 and the graph is connected (cohesive), it's a tree
    adj = defaultdict(list)
    edges = []
    edge_weight = {}
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
        edge_weight[(u,v)] = w
        edge_weight[(v,u)] = w
    
    # We need to find the minimum spanning tree that includes all anchors
    # But since the graph is already a tree, we need to find the minimal subtree that connects all anchors
    # This is the Steiner tree problem on a tree, which can be solved efficiently
    
    # In a tree, the minimal subtree connecting a set of terminals (anchors) 
    # can be found by:
    # 1. Root the tree at any node
    # 2. For each node, compute if the subtree rooted at that node contains any anchor
    # 3. An edge is in the minimal subtree if both its endpoints' subtrees contain at least one anchor
    
    # Let's root the tree at node 1
    root = 1
    
    # Build parent/children relationships with BFS
    parent = [-1] * (n + 1)
    children = defaultdict(list)
    visited = [False] * (n + 1)
    queue = deque([root])
    visited[root] = True
    
    while queue:
        u = queue.popleft()
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                children[u].append(v)
                queue.append(v)
    
    # For each node, determine if its subtree contains any anchor
    # We'll do a post-order traversal
    contains_anchor = [False] * (n + 1)
    
    def dfs(u):
        if u in anchors:
            contains_anchor[u] = True
        for v in children[u]:
            dfs(v)
            if contains_anchor[v]:
                contains_anchor[u] = True
    
    dfs(root)
    
    # Now, count the edges that must be kept
    # An edge {u,v} where parent[v] = u is kept if both u's subtree (including u) 
    # and v's subtree contain at least one anchor
    total_weight = 0
    
    for u in range(1, n+1):
        for v in children[u]:
            # Edge from u to v
            # v's subtree contains an anchor: contains_anchor[v]
            # u's subtree contains an anchor even if we remove v's subtree:
            # This is contains_anchor[u], but we need to be careful
            # Actually, in the rooted view, u is the parent of v
            # The condition is: the subtree of v contains an anchor AND 
            # the rest of the tree (including u and all other branches) contains an anchor
            # Since we're in a tree, if both sides of an edge have at least one anchor, 
            # then this edge is necessary in the minimal subtree
            
            # But we need to think differently:
            # In the minimal subtree connecting all anchors, an edge is included
            # if and only if removing it separates the tree into two components,
            # each containing at least one anchor.
            
            # In a tree, for edge (u,v) where v is child of u:
            # - One component is subtree of v
            # - Other component is rest of tree
            # Edge is needed if both components have at least one anchor
            
            subtree_has_anchor = contains_anchor[v]
            # For the other component, we need to check if there's an anchor 
            # not in subtree of v
            # Since contains_anchor[u] means subtree of u has anchor,
            # and subtree of u includes subtree of v,
            # we need to know if there's an anchor in subtree of u but not in subtree of v
            # This is tricky to compute directly
            
            # Let's recompute: for each node, count anchors in its subtree
            # Then, for each edge (u,v), check:
            # 1. subtree of v has >= 1 anchor
            # 2. (total anchors - subtree of v) >= 1
            
    # Recompute properly:
    # Count anchors in each subtree
    anchor_count = [0] * (n + 1)
    
    def dfs_count(u):
        if u in anchors:
            anchor_count[u] = 1
        for v in children[u]:
            dfs_count(v)
            anchor_count[u] += anchor_count[v]
    
    dfs_count(root)
    
    total_anchors = anchor_count[root]
    
    # Now for each edge, check if it's needed
    for u in range(1, n+1):
        for v in children[u]:
            # Edge from u to v
            # Component 1: subtree of v (anchor_count[v] anchors)
            # Component 2: rest of tree (total_anchors - anchor_count[v] anchors)
            if anchor_count[v] > 0 and (total_anchors - anchor_count[v]) > 0:
                total_weight += edge_weight[(u, v)]
    
    print(total_weight)

if __name__ == "__main__":
    main()
```
