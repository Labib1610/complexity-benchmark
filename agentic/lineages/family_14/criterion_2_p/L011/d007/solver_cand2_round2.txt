```python
import sys

# The problem states that the system has n components and m = n-1 interactions.
# This means the system is a tree.
# An "energy-cohesive" subset in a tree is simply a connected subgraph (a subtree).
# We need to find the minimum weight of a subtree that contains all t critical components.
# The minimum subtree containing a set of nodes is the union of all paths between
# any two critical nodes. This is equivalent to the Steiner Tree problem on a tree,
# which is simply the sum of weights of all edges that lie on the paths between
# any two critical nodes.
# An edge is part of the minimal subtree if and only if removing it splits the
# tree into two components, both of which contain at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count_critical[u] will store the number of critical nodes in the subtree
    # rooted at u (when rooted at 1).
    count_critical = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    # Initialize counts
    for i in range(1, n + 1):
        if i in critical_nodes:
            count_critical[i] = 1
            
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p, w = parent[u]
        if count_critical[u] > 0 and count_critical[u] < t:
            total_cost += w
        count_critical[p] += count_critical[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
