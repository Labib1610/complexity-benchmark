import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); m = int(data[idx+1]); t = int(data[idx+2]); idx += 3
    required = [int(data[idx+i]) for i in range(t)]
    idx += t
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); w = int(data[idx+2]); idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    req_set = [False]*(n+1)
    for r in required:
        req_set[r] = True
    
    parent = [0]*(n+1)
    parent_weight = [0]*(n+1)
    order = []
    stack = [required[0]]
    parent[required[0]] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0 and v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                stack.append(v)
    
    needed = [False]*(n+1)
    for u in reversed(order):
        if req_set[u]:
            needed[u] = True
        if needed[u] and parent[u] != -1:
            needed[parent[u]] = True
    
    total = 0
    for u in range(1, n+1):
        if needed[u] and parent[u] != -1:
            total += parent_weight[u]
    print(total)

solve()
