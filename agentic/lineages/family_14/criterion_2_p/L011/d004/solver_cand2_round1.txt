```python
import sys

# The problem asks for the minimum cost to connect all elements in a core set Y.
# Since the original graph is a tree (n nodes, n-1 edges, connected),
# the unique path between any two nodes in the tree is the only way to connect them.
# To connect all nodes in Y, we need to select all edges that lie on the paths
# between any two nodes in Y.
# This is equivalent to finding the union of all paths between nodes in Y.
# In a tree, the union of paths between nodes in Y is the minimal subtree
# containing all nodes in Y.
# We can solve this by rooting the tree at an arbitrary node (e.g., node 1)
# and marking all nodes in Y. An edge (u, v) (where v is child of u) is in the
# minimal subtree if and only if the subtree rooted at v contains at least one
# node from Y and the complement of the subtree rooted at v contains at least
# one node from Y.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    core_set = set()
    idx = 3
    for _ in range(t):
        core_set.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # DFS to find subtree counts of core nodes
    # parent, edge_weight_to_parent, subtree_core_count
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                weight_to_parent[v] = w
                stack.append(v)
                
    subtree_core_count = [0] * (n + 1)
    for u in core_set:
        subtree_core_count[u] = 1
        
    total_cost = 0
    # Process in reverse topological order
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if subtree_core_count[u] > 0 and subtree_core_count[u] < t:
            total_cost += weight_to_parent[u]
        subtree_core_count[p] += subtree_core_count[u]
        
    print(total_cost)

solve()
```
