```python
import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    required = set(map(int, sys.stdin.readline().split()))
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    # If there's only one required node, we don't need any edges
    if t <= 1:
        print(0)
        return
    
    # Sort edges by weight
    edges.sort()
    
    # Union-Find data structure
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    # Kruskal's algorithm adapted for required nodes
    result = 0
    connected_required = set()
    
    for w, u, v in edges:
        # If both nodes are required and not yet connected
        if u in required and v in required:
            if find(u) != find(v):
                union(u, v)
                result += w
                connected_required.add(u)
                connected_required.add(v)
        # If one node is required and the other isn't
        elif u in required and v not in required:
            if find(u) != find(v):
                union(u, v)
                result += w
                connected_required.add(u)
        elif v in required and u not in required:
            if find(u) != find(v):
                union(u, v)
                result += w
                connected_required.add(v)
        # If neither is required but they can connect required components
        else:
            if find(u) != find(v):
                # Check if this connection helps connect required nodes
                # This is a simplification; we actually need to be more careful
                union(u, v)
    
    # Actually, let's think differently. 
    # We want the minimum spanning tree that connects all required nodes.
    # This is the Steiner tree problem, which is NP-hard in general.
    # But since our graph is a tree (m = n-1 and connected), it's much simpler.
    
    # In a tree, to connect a subset of nodes, we need the minimal subtree
    # that contains all of them. This is found by taking the union of paths
    # between all pairs, but there's a simpler way:
    # Take the edges on the paths between consecutive required nodes in some order.
    
    # However, there's an even simpler approach for trees:
    # The minimum cost is the cost of the subtree induced by the required nodes
    # and their connecting paths.
    
    # Let's restart with the right approach:
    # Since the graph is a tree, we can find the minimal connected subtree
    # containing all required nodes.
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Find all nodes that must be included (the subtree spanning required nodes)
    if t == 1:
        print(0)
        return
    
    # BFS to find the minimal subtree containing all required nodes
    from collections import deque
    
    # First, find one required node to start BFS
    start = next(iter(required))
    
    # BFS to mark all nodes in the subtree
    visited = [False] * (n + 1)
    queue = deque([start])
    visited[start] = True
    subtree_nodes = {start}
    
    while queue:
        node = queue.popleft()
        for neighbor, _ in adj[node]:
            if not visited[neighbor]:
                visited[neighbor] = True
                queue.append(neighbor)
                subtree_nodes.add(neighbor)
    
    # Now we need to find the edges of the minimal subtree containing all required nodes
    # In a tree, this is equivalent to finding the "Steiner tree"
    # For a tree, we can do this by finding the paths between required nodes
    
    # But there's a simpler way: in a tree, the minimal subtree containing a set of nodes
    # is the union of paths between all pairs, which is the same as taking the "closure"
    # under taking paths.
    
    # Even simpler approach:
    # 1. Mark all required nodes
    # 2. Find all nodes that lie on paths between required nodes
    # 3. The cost is the sum of edges in this subtree
    
    # To implement this correctly, we can use the fact that in a tree,
    # the minimal subtree containing a set of nodes S is the subtree induced
    # by all nodes that lie on paths between nodes of S.
    
    # Algorithm:
    # 1. Pick any required node as root
    # 2. Do DFS to compute parents and depths
    # 3. For each required node, mark its path to root
    # 4. The cost is the sum of weights of marked edges
    
    # Let's implement this properly
    
    # Root the tree at the first required node
    root = next(iter(required))
    
    # Compute parent and depth for each node using DFS
    parent_node = [-1] * (n + 1)
    depth = [0] * (n + 1)
    visited_dfs = [False] * (n + 1)
    
    def dfs(node, par, dep):
        parent_node[node] = par
        depth[node] = dep
        visited_dfs[node] = True
        for neighbor, _ in adj[node]:
            if neighbor != par and not visited_dfs[neighbor]:
                dfs(neighbor, node, dep + 1)
    
    dfs(root, -1, 0)
    
    # Function to get path from node to root
    def get_path_to_root(node):
        path = []
        current = node
        while current != -1:
            path.append(current)
            current = parent_node[current]
        return path
    
    # Mark all nodes on paths from required nodes to root
    marked_nodes = set()
    for req_node in required:
        path = get_path_to_root(req_node)
        for node in path:
            marked_nodes.add(node)
    
    # Calculate the cost of edges in the subtree induced by marked_nodes
    total_cost = 0
    edge_visited = set()
    
    # For each marked node (except root), add the edge to its parent if parent is also marked
    for node in marked_nodes:
        if node != root:
            par = parent_node[node]
            if par in marked_nodes:
                # Find the weight of edge (node, par)
                for neighbor, weight in adj[node]:
                    if neighbor == par and (min(node, par), max(node, par)) not in edge_visited:
                        total_cost += weight
                        edge_visited.add((min(node, par), max(node, par)))
                        break
    
    print(total_cost)

threading.Thread(target=main).start()
```
