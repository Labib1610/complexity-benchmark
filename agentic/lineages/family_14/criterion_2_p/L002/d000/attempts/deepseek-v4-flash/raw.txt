We can solve this by pruning the tree: remove all subtrees that don’t contain any vital module, then sum the weights of the remaining edges.  

Here’s the implementation:  

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    vital = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    visited = [False] * (n + 1)
    
    def dfs(u, parent):
        visited[u] = True
        total = 0
        has_vital = u in vital
        for v, w in adj[u]:
            if v == parent:
                continue
            if not visited[v]:
                sub_has_vital, sub_cost = dfs(v, u)
                if sub_has_vital:
                    total += sub_cost + w
                    has_vital = True
        return has_vital, total
    
    # Start from any vital node
    start = next(iter(vital))
    _, ans = dfs(start, -1)
    print(ans)

if __name__ == "__main__":
    solve()
```
