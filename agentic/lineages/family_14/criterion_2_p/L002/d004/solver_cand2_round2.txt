```python
import sys

# The problem asks for the minimum weight of a subtree that connects all t critical nodes.
# In a tree, the unique path connecting all critical nodes is the union of all paths
# between any two critical nodes. This is equivalent to the minimal subtree
# containing all critical nodes.
# The edges in this minimal subtree are exactly those edges that have at least
# one critical node in one of the two components formed by removing the edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many critical nodes are in the subtree of each node
    # when rooted at an arbitrary node (say 1).
    # An edge (u, v) where v is a child of u is part of the minimal subtree
    # if the number of critical nodes in v's subtree is > 0 and < t.
    
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    crit_count = [0] * (n + 1)
    for node in critical_nodes:
        crit_count[node] = 1
        
    total_weight = 0
    # Process in reverse topological order to propagate counts up
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if crit_count[u] > 0 and crit_count[u] < t:
            total_weight += edge_to_parent_w[u]
        crit_count[p] += crit_count[u]
        
    print(total_weight)

if __name__ == '__main__':
    # Increase recursion depth for deep trees
    sys.setrecursionlimit(300000)
    solve()
```
