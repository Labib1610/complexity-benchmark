```python
import sys

# The problem asks for the minimum cost to connect a set of target nodes in a graph.
# Each edge (u, v) with weight w represents the vector v_u - v_v.
# The span of a set of vectors {v_u - v_v} is equivalent to the connectivity in a graph.
# Specifically, v_a - v_b is in the span of the selected edges if and only if
# a and b are in the same connected component formed by the selected edges.
# We need all target indices to be in the same connected component.
# This is equivalent to finding the Minimum Spanning Tree of the graph
# where we contract all target nodes into a single super-node, or more simply,
# finding the Steiner Tree for the set of target nodes.
# Since the graph is a tree (m = n-1 and it spans the space), the Steiner Tree
# is the unique path in the tree connecting all target nodes.
# The cost is the sum of weights of all edges on the unique paths between
# any two target nodes, which is the sum of weights of all edges that
# have at least one target node in each of the two components formed by removing the edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_targets[u] = number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            subtree_targets[u] += 1
        if parent[u] != 0:
            p, w = parent[u]
            subtree_targets[p] += subtree_targets[u]
            
    total_cost = 0
    for u in range(2, n + 1):
        count = subtree_targets[u]
        # If an edge splits the tree into two components with 'count' and 't - count' targets,
        # the edge is part of the Steiner tree if both components have at least one target.
        if count > 0 and count < t:
            total_cost += parent[u][1]
            
    print(total_cost)

solve()
```
