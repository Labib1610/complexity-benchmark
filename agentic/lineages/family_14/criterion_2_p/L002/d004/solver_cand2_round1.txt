```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # subtree_critical[u] stores the number of critical nodes in the subtree rooted at u
    subtree_critical = [0] * (n + 1)
    
    # We need to calculate for each edge (u, v) with weight w, 
    # if it is part of the safe subnetwork.
    # An edge is part of the subnetwork if there is at least one critical node 
    # in the subtree of v (when rooted at u) AND at least one critical node 
    # outside that subtree.
    
    total_power = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Initialize subtree_critical
    for i in range(1, n + 1):
        if i in critical_nodes:
            subtree_critical[i] = 1
            
    # Process in reverse topological order
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            subtree_critical[parent[u]] += subtree_critical[u]
            
    # Check each edge
    for i in range(2, n + 1):
        count = subtree_critical[i]
        if count > 0 and count < t:
            total_power += edge_to_parent_weight[i]
            
    print(total_power)

if __name__ == '__main__':
    solve()
```
