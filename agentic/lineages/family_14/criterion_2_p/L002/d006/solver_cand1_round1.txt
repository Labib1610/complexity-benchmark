```python
import sys

# The problem states that the m constraints form a graph with n nodes and m edges.
# The intersection of x_u - x_v = 0 constraints is the subspace where x_u = x_v
# for all edges (u, v) in the activated set.
# For the intersection to be the 1D subspace x_1 = x_2 = ... = x_n, the graph
# must be connected. Since m = n - 1, the graph is a tree.
# We want to force x_{t1} = x_{t2} = ... = x_{tk} for the target set.
# This is equivalent to saying that in the graph formed by the activated edges,
# all target nodes must belong to the same connected component.
# We want to find a minimum cost subset of edges such that all target nodes
# are in the same connected component.
# This is the Steiner Tree problem on a graph, but since the graph is a tree,
# the unique path between any two nodes is fixed.
# To connect a set of nodes in a tree, we must select all edges on the unique
# paths between them. The union of these paths forms a subtree.
# The cost is the sum of weights of all edges in this subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) - 1 for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Since the graph is a tree, we can root it at targets[0].
    # An edge is part of the Steiner tree if and only if the subtree
    # below it contains at least one target node and the part above it
    # contains at least one target node.
    
    root = targets[0]
    is_target = [False] * n
    for node in targets:
        is_target[node] = True
        
    has_target = [False] * n
    parent_edge_weight = [0] * n
    parent = [-1] * n
    
    order = []
    stack = [root]
    visited = [False] * n
    visited[root] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_target[u]:
            has_target[u] = True
        if has_target[u] and parent[u] != -1:
            has_target[parent[u]] = True
            
    total_cost = 0
    for i in range(n):
        if i == root:
            continue
        # Edge is (i, parent[i])
        # It is in the Steiner tree if the subtree at i has a target
        # AND the rest of the tree (outside subtree i) has a target.
        if has_target[i]:
            # Check if there is a target outside the subtree of i
            # Total targets is t. If count(targets in subtree i) < t, then yes.
            # We can precalculate target counts.
            pass

    # Simpler: An edge is in the Steiner tree if it lies on a path between
    # any two target nodes. This is equivalent to saying the edge separates
    # the set of targets into two non-empty sets.
    target_count_in_subtree = [0] * n
    for u in reversed(order):
        if is_target[u]:
            target_count_in_subtree[u] += 1
        if parent[u] != -1:
            target_count_in_subtree[parent[u]] += target_count_in_subtree[u]
            
    total_cost = 0
    for i in range(n):
        if i == root: continue
        c = target_count_in_subtree[i]
        if c > 0 and c < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
