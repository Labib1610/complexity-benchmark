```python
import sys

# The problem asks for the minimum cost to select a subset of edges such that
# there exists a root r from which all target nodes are reachable.
# Since the graph is a tree (n nodes, n-1 edges), the subset of edges must
# form a connected component (a subtree) that contains all target nodes.
# If we pick a subtree that contains all target nodes, we can always pick
# any node r in that subtree as the root and orient edges away from r.
# Thus, the problem reduces to finding the minimum weight subtree that
# contains all target nodes.
# This is equivalent to the Steiner Tree problem on a tree, which is simply
# the sum of weights of all edges that lie on the unique paths between
# any two target nodes.
# An edge is part of this subtree if and only if removing it splits the
# tree into two components, both of which contain at least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_targets[u] will store the number of target nodes in the subtree
    # rooted at u (when rooted at 1).
    subtree_targets = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in targets:
            subtree_targets[u] = 1
            
    total_cost = 0
    for u in reversed(order):
        if u == 1:
            continue
        if subtree_targets[u] > 0 and subtree_targets[u] < t:
            total_cost += parent_edge_weight[u]
        subtree_targets[parent[u]] += subtree_targets[u]
        
    print(total_cost)

solve()
```
