```python
import sys

# The problem asks for the minimum cost to connect all critical slots in a tree.
# If we have a set of critical slots, any swap of two items at slots a and b
# can be achieved if and only if there is a path between a and b in the graph
# of installed mechanisms. To be able to swap any pair of critical slots,
# all critical slots must belong to the same connected component.
# Furthermore, the "net result" condition implies that we need to be able to
# perform a transposition (a, b). In a tree, a transposition of two nodes
# can be performed if we have a path between them.
# Specifically, if we have a tree connecting all critical nodes, we can swap
# any two nodes a and b by moving them along the path, swapping them, and
# moving them back.
# Thus, the problem reduces to finding the Minimum Spanning Tree of the
# subgraph that connects all critical nodes.
# This is equivalent to finding the Steiner Tree for the set of critical nodes.
# Since the original graph is a tree (n nodes, n-1 edges, connected),
# the Steiner Tree is simply the unique path in the tree that connects all
# critical nodes.
# The cost is the sum of weights of all edges that lie on the unique path
# connecting all critical nodes in the original tree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_slots = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Find the minimal subtree containing all critical nodes.
    # We can do this by pruning the tree: remove all leaves that are not critical.
    # Or, more simply, calculate the sum of weights of all edges that have
    # at least one critical node in both components formed by removing the edge.
    
    is_critical = [False] * (n + 1)
    for s in critical_slots:
        is_critical[s] = True
        
    subtree_has_critical = [False] * (n + 1)
    total_cost = 0
    
    # Use DFS to compute subtree_has_critical and sum weights
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    subtree_has_critical = [is_critical[i] for i in range(n + 1)]
    for u in reversed(order):
        if u == 1: continue
        if subtree_has_critical[u]:
            total_cost += edge_to_parent_weight[u]
            subtree_has_critical[parent[u]] = True
            
    print(total_cost)

solve()
```
