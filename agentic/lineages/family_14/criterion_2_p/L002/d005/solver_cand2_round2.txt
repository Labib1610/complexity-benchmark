```python
import sys

# The problem describes a graph where each constraint (u, v) with weight w
# corresponds to an edge in a graph. The condition Sx = 0 implies x_u = x_v
# for each edge (u, v) included in the selection.
# If we select a set of edges, the condition Sx = 0 implies that for any
# connected component in the graph formed by these edges, all x_i in that
# component must be equal.
# The requirement is that for all critical indices a and b, x_a = x_b.
# This means all critical indices must belong to the same connected component
# in the graph formed by the selected edges.
# We want to find the minimum cost to connect all critical indices.
# This is equivalent to finding the Steiner Tree of the critical indices.
# Since the graph is a tree (n nodes, n-1 edges, and the sum of all edges
# connects all nodes), the Steiner Tree is simply the unique path in the
# tree connecting all critical indices.
# The cost is the sum of weights of all edges on the unique paths between
# any two critical indices. This is the sum of weights of all edges that
# have at least one critical index in each of the two components formed
# by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = set()
    idx = 3
    for _ in range(t):
        critical.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_critical[u] = number of critical nodes in the subtree rooted at u
    subtree_critical = [0] * (n + 1)
    
    # DFS to count critical nodes in subtrees
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in critical:
            subtree_critical[u] += 1
        if parent[u] != -1:
            subtree_critical[parent[u]] += subtree_critical[u]
            
    total_cost = 0
    for u, v, w in edges:
        # Edge (u, v) splits the tree into two components.
        # One component has subtree_critical[child] critical nodes.
        # The other has t - subtree_critical[child].
        child = u if parent[v] == u else v
        count = subtree_critical[child]
        if count > 0 and count < t:
            total_cost += w
            
    print(total_cost)

solve()
```
