```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+T]]
    
    edges = []
    idx = 3 + T
    for _ in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if T <= 1:
        print(0)
        return

    # The problem asks for the minimum cost to connect all target nodes in a tree.
    # Since the given M generators form a tree (N-1 edges, minimal sufficiency),
    # we are looking for the sum of weights of all edges on the unique paths 
    # between all pairs of target nodes.
    # This is equivalent to the sum of weights of all edges that lie on the 
    # unique path between at least one pair of target nodes.
    # An edge (u, v) with weight w is part of the solution if and only if 
    # removing it splits the tree into two components, each containing at least 
    # one target node.
    
    adj = [[] for _ in range(N + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    target_set = set(targets)
    subtree_targets = [0] * (N + 1)
    
    # DFS to count targets in each subtree
    parent = [0] * (N + 1)
    order = []
    stack = [targets[0]]
    visited = [False] * (N + 1)
    visited[targets[0]] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in target_set:
            subtree_targets[u] += 1
        if parent[u] != 0:
            subtree_targets[parent[u]] += subtree_targets[u]
            
    total_cost = 0
    for u in range(1, N + 1):
        if parent[u] != 0:
            # Edge between u and parent[u]
            count = subtree_targets[u]
            if count > 0 and count < T:
                # Find weight of edge (u, parent[u])
                for v, w in adj[u]:
                    if v == parent[u]:
                        total_cost += w
                        break
                        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
