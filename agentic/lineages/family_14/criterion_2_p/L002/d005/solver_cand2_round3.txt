import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to keep all paths between any two critical nodes.
    # This is equivalent to keeping the minimal subtree that contains all critical nodes.
    # A node is part of this subtree if it lies on a path between two critical nodes.
    # This is equivalent to saying the subtree of the minimal Steiner tree for the critical nodes.
    
    is_critical = [False] * (n + 1)
    for node in critical_nodes:
        is_critical[node] = True
        
    # Count how many critical nodes are in the subtree of each node
    subtree_critical_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_critical[u]:
            subtree_critical_count[u] += 1
        if parent[u] != 0:
            subtree_critical_count[parent[u]] += subtree_critical_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the edge to parent is part of the path between any two critical nodes,
        # it must be included. This happens if the subtree contains at least one
        # critical node and the rest of the tree contains at least one critical node.
        count = subtree_critical_count[i]
        if count > 0 and count < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
