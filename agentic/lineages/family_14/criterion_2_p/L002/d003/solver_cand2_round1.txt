```python
import sys

# The problem states the station is a tree (n modules, n-1 tubes).
# We need to find the minimum weight sub-network that connects all override consoles.
# In a tree, the unique path between any two nodes is fixed.
# The union of all paths between all pairs of override consoles forms the minimal sub-network.
# This is equivalent to finding all edges that lie on a path between at least two consoles.
# An edge (u, v) with weight w lies on a path between two consoles if and only if
# the two components formed by removing (u, v) both contain at least one console.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    consoles = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count consoles in subtree
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    console_count = [0] * (n + 1)
    for c in consoles:
        console_count[c] = 1
        
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            console_count[p] += console_count[u]
            # If this subtree contains some but not all consoles, the edge must be included
            if 0 < console_count[u] < t:
                total_weight += edge_weight[u]
                
    print(total_weight)

if __name__ == '__main__':
    solve()
```
