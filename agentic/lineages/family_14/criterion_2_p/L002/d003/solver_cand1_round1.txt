```python
import sys

# The problem asks for the minimum weight of a subset of edges that connects all priority sectors.
# Since the graph is a tree (m = n - 1), there is only one unique path between any two nodes.
# The subset of edges that connects all priority sectors is exactly the union of all paths
# between every pair of priority sectors. This is equivalent to the minimal subtree
# containing all priority nodes.
# The total weight is the sum of weights of all edges that lie on at least one path
# between two priority nodes. An edge (u, v) with weight w is part of this subtree
# if and only if the subtree formed by removing (u, v) has priority nodes in both components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_nodes = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many priority nodes are in the subtree of each node
    # We root the tree at node 1
    priority_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in priority_nodes:
        priority_count[node] = 1
        
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            priority_count[p] += priority_count[u]
            # If the subtree at u contains some but not all priority nodes,
            # the edge to its parent must be included.
            if 0 < priority_count[u] < t:
                total_weight += parent_edge_weight[u]
                
    print(total_weight)

solve()
```
