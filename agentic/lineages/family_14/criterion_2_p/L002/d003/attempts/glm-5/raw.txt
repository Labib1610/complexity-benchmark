import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    critical = [False] * (n + 1)
    for _ in range(t):
        critical[int(next(it))] = True
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = [1]
    parent[1] = -1
    idx = 0
    while idx < len(order):
        u = order[idx]
        idx += 1
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                parent_weight[v] = w
                order.append(v)
    
    keep = [False] * (n + 1)
    for u in order[::-1]:
        if critical[u]:
            keep[u] = True
        if keep[u] and parent[u] != -1:
            keep[parent[u]] = True
    
    total = 0
    for u in range(2, n + 1):
        if keep[u]:
            total += parent_weight[u]
    
    print(total)

if __name__ == "__main__":
    solve()
