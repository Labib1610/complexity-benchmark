```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return

    vital_list = []
    for _ in range(t):
        vital_list.append(int(next(iterator)))
    
    vital_set = set(vital_list)
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        w = int(next(iterator))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Prim's algorithm variant for Steiner Tree on trees
    # Start with all vital nodes in the MST set
    in_mst = [False] * (n + 1)
    min_edge = [float('inf')] * (n + 1)
    heap = []
    
    for u in vital_list:
        in_mst[u] = True
        for v, w in adj[u]:
            if not in_mst[v]:
                if w < min_edge[v]:
                    min_edge[v] = w
                    heapq.heappush(heap, (w, v))
    
    total_power = 0
    nodes_in_mst = t
    
    # We need to connect all t vital nodes.
    # In a tree, the minimal subgraph connecting a set of nodes is a tree.
    # This tree has at most t-1 "branching" edges that connect two components of vital nodes.
    # We iterate t-1 times to find these edges.
    # Note: The loop condition `nodes_in_mst < n` is for general MST.
    # Here we just need to ensure all vital nodes are connected.
    # Since the graph is a tree, connecting all vital nodes requires exactly t-1 "connection" steps
    # where we merge a component containing a vital node with another.
    # However, the Prim's loop naturally handles this: we stop when we have processed enough edges.
    # Actually, we just need to run the loop t-1 times.
    
    for _ in range(t - 1):
        # Find the minimum weight edge connecting a node in MST to a node outside
        while heap:
            w, u = heapq.heappop(heap)
            if in_mst[u]:
                continue
            # This is the minimum edge to add
            total_power += w
            in_mst[u] = True
            nodes_in_mst += 1
            # Add edges from the newly added node
            for v, weight in adj[u]:
                if not in_mst[v]:
                    if weight < min_edge[v]:
                        min_edge[v] = weight
                        heapq.heappush(heap, (weight, v))
            break # Proceed to next iteration of outer loop
            
    print(total_power)

if __name__ == '__main__':
    solve()
```
