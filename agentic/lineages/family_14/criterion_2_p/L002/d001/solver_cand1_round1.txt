```python
import sys

# The problem states the station is a tree (minimum number of corridors to connect all modules).
# We need to find the minimum power to connect all critical modules.
# This is equivalent to finding the sum of weights of all edges on the unique paths
# between all critical modules.
# In a tree, the set of edges forming the minimal subtree connecting a set of nodes
# is the union of paths from an arbitrary root to each critical node, excluding
# edges that are not part of any path between two critical nodes.
# More simply: an edge is part of the minimal subtree if and only if the two
# components it creates both contain at least one critical module.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_critical[u] stores the number of critical nodes in the subtree rooted at u
    subtree_critical = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in critical_nodes:
            subtree_critical[u] = 1
            
    total_power = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        subtree_critical[p] += subtree_critical[u]
        
        # If this edge separates the tree into two parts, both containing
        # at least one critical node, it must be included.
        if 0 < subtree_critical[u] < t:
            total_power += edge_weight[u]
            
    print(total_power)

if __name__ == '__main__':
    solve()
```
