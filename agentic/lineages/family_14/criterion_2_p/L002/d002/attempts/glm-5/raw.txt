```python
import sys
import heapq

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    critical = set()
    for _ in range(t):
        critical.add(int(next(it)))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Prim's algorithm starting from all critical nodes
    in_mst = [False] * (n + 1)
    min_edge = [float('inf')] * (n + 1)
    heap = []
    
    for node in critical:
        min_edge[node] = 0
        heapq.heappush(heap, (0, node))
    
    total_weight = 0
    nodes_used = 0
    
    while heap and nodes_used < t:
        w, u = heapq.heappop(heap)
        if in_mst[u]:
            continue
        in_mst[u] = True
        total_weight += w
        nodes_used += 1
        
        for v, weight in adj[u]:
            if not in_mst[v] and weight < min_edge[v]:
                min_edge[v] = weight
                heapq.heappush(heap, (weight, v))
    
    print(total_weight)

if __name__ == "__main__":
    solve()
```
