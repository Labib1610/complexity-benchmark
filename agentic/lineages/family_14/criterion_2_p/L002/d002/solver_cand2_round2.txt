```python
import sys

# The problem describes a tree structure (n nodes, n-1 edges, connected).
# We need to find the minimum weight of the subtree that connects all t critical nodes.
# This is equivalent to the sum of weights of all edges that lie on the paths
# between any two critical nodes.
# An edge is part of this subtree if and only if the two components it creates
# both contain at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Use iterative DFS to avoid recursion depth issues
    # count[u] will store the number of critical nodes in the subtree rooted at u
    count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in critical_nodes:
        count[node] = 1
        
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if count[u] > 0 and count[u] < t:
            total_weight += edge_to_parent_weight[u]
        count[p] += count[u]
        
    print(total_weight)

if __name__ == '__main__':
    solve()
```
