```python
import sys

# The problem asks for the sum of weights of edges that lie on the unique paths 
# between any two vital modules. If t=1, no paths exist between distinct vital 
# modules, so the answer is 0. If t > 1, we need the union of all paths 
# between all pairs of vital modules. This is equivalent to the set of edges 
# that have at least one vital module in one component and at least one 
# vital module in the other component when the edge is removed.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    vital_modules = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many vital modules are in the subtree of each node
    vital_count = [0] * (n + 1)
    for v in vital_modules:
        vital_count[v] = 1
        
    # DFS to compute subtree vital counts and identify edges
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    total_power = 0
    for u in reversed(order):
        if parent[u] != 0:
            vital_count[parent[u]] += vital_count[u]
            # If an edge separates the tree into two parts, and both parts
            # contain at least one vital module, the edge is part of the union.
            if 0 < vital_count[u] < t:
                total_power += edge_weight_to_parent[u]
                
    print(total_power)

if __name__ == '__main__':
    solve()
```
