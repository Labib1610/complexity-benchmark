```python
import sys

# The problem states the station is a tree (n modules, n-1 tubes, connected).
# We need to find the minimum weight of the subtree that connects all t vital modules.
# This is equivalent to the sum of weights of all edges that lie on the path
# between any two vital modules.
# An edge is part of this subtree if and only if the subtree formed by removing
# the edge splits the vital modules into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    vital_modules = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count[u] will store the number of vital modules in the subtree rooted at u
    count = [0] * (n + 1)
    total_weight = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in vital_modules:
            count[u] = 1
            
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            p = parent[u]
            count[p] += count[u]
            # If this edge separates vital modules, it must be included
            if count[u] > 0 and count[u] < t:
                total_weight += edge_to_parent_weight[u]
                
    print(total_weight)

solve()
```
