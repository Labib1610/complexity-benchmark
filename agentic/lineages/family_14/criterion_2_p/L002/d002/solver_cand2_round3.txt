import sys

# The problem states m = n - 1 and the graph is connected, 
# meaning the facility is a tree.
# To connect a subset of t nodes in a tree, we must include all edges 
# that lie on the unique paths between any two nodes in the subset.
# This is equivalent to the sum of weights of all edges that have at least 
# one indispensable node in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    indispensable = set()
    idx = 3
    for _ in range(t):
        indispensable.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # count[u] will store the number of indispensable nodes in the subtree of u
    count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(1, 0)]
    order = []
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        parent[u] = p
        for v, w in adj[u]:
            if v != p:
                edge_to_parent_weight[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        if u in indispensable:
            count[u] += 1
        if parent[u] != 0:
            count[parent[u]] += count[u]
            # If an edge splits the tree such that both sides have at least one 
            # indispensable node, it must be included.
            # The number of indispensable nodes in the subtree is count[u].
            # The number of indispensable nodes outside is t - count[u].
            if count[u] > 0 and count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
