```python
import sys

# The problem describes a set of n variables and n-1 generators (x_u - x_v).
# This structure forms a tree where variables are nodes and generators are edges.
# The span of a set of generators (edges) contains x_a - x_b if and only if
# there is a path between a and b in the graph formed by the chosen edges.
# We need to select a subset of edges such that all t critical variables are
# in the same connected component.
# This is equivalent to finding the minimum weight Steiner Tree for the t nodes.
# Since the graph is a tree (n nodes, n-1 edges, connected), the Steiner Tree
# is simply the unique path in the tree that connects all t critical nodes.
# This path is the union of all paths between any two critical nodes.
# In a tree, this is the minimal subtree containing all t nodes.
# We can find this by rooting the tree at an arbitrary node (e.g., node 1)
# and calculating the sum of weights of all edges that lie on the path
# between any two critical nodes.
# An edge (u, v) is part of the Steiner Tree if and only if the subtree
# rooted at the child node contains at least one critical node and at least
# one critical node is outside that subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = [False] * (n + 1)
    for i in range(t):
        critical[int(input_data[3 + i])] = True
        
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_critical[u] stores the number of critical nodes in the subtree of u
    subtree_critical = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if critical[u]:
            subtree_critical[u] += 1
        if parent[u] != -1:
            subtree_critical[parent[u]] += subtree_critical[u]
            
    total_weight = 0
    for i in range(2, n + 1):
        # Edge (i, parent[i]) is in the Steiner Tree if the subtree at i
        # contains some but not all critical nodes.
        if 0 < subtree_critical[i] < t:
            total_weight += edge_to_parent_weight[i]
            
    print(total_weight)

solve()
```
