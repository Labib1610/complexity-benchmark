import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    critical = [False] * (n + 1)
    for _ in range(t):
        v = int(data[idx]); idx += 1
        critical[v] = True
    
    edges = []
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        edges.append((u, v, w))
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    for i, (u, v, w) in enumerate(edges):
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Find the path between the two closest critical nodes
    # BFS from each critical node, but that's O(n*t) which is too slow
    # Instead, use multi-source BFS to find the closest pair of critical nodes
    
    # Multi-source BFS to find the minimum weight path between any two critical nodes
    # Using Dijkstra with all critical nodes as sources
    
    import heapq
    
    INF = float('inf')
    dist = [INF] * (n + 1)
    parent_edge = [-1] * (n + 1)
    parent_node = [0] * (n + 1)
    source = [0] * (n + 1)
    
    pq = []
    for v in range(1, n + 1):
        if critical[v]:
            dist[v] = 0
            source[v] = v
            heapq.heappush(pq, (0, v))
    
    best = INF
    best_edge = -1
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, eid in adj[u]:
            w = edges[eid][2]
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                source[v] = source[u]
                parent_edge[v] = eid
                parent_node[v] = u
                heapq.heappush(pq, (nd, v))
            elif source[v] != source[u] and source[v] != 0 and source[u] != 0:
                # Found a path connecting two different critical components
                total = d + w + dist[v]
                if total < best:
                    best = total
                    best_edge = eid
    
    if t <= 1:
        print(0)
        return
    
    # The answer is the minimum weight path between any two critical nodes
    # But we need all critical nodes to be connected, so we need a Steiner tree
    # However, since the graph is a tree, the Steiner tree is the union of paths
    # between all pairs of critical nodes, which is the minimal subtree spanning
    # all critical nodes.
    
    # For a tree, the minimal Steiner tree spanning a set of terminals is the
    # union of all paths between consecutive terminals in some ordering.
    # This equals: sum of all edge weights - sum of weights of edges not needed
    # The edges not needed are those in subtrees with no critical nodes.
    
    # So we need to find the minimal subtree containing all critical nodes.
    # This is equivalent to: total weight - weight of edges that can be removed
    # An edge can be removed if one of the two components (after removing it) has no critical nodes.
    
    # Build tree structure and compute for each edge whether removing it disconnects any critical nodes
    
    # Root the tree at a critical node
    root = 1
    for v in range(1, n + 1):
        if critical[v]:
            root = v
            break
    
    # BFS/DFS to build parent relationships
    parent = [0] * (n + 1)
    parent_eid = [-1] * (n + 1)
    visited = [False] * (n + 1)
    visited[root] = True
    stack = [root]
    order = []
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, eid in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_eid[v] = eid
                stack.append(v)
    
    # Count critical nodes in each subtree
    cnt = [0] * (n + 1)
    for u in reversed(order):
        cnt[u] = 1 if critical[u] else 0
        for v, eid in adj[u]:
            if v != parent[u]:
                cnt[u] += cnt[v]
    
    total = 0
    for u in range(1, n + 1):
        if parent_eid[u] != -1:
            # Edge parent[u] - u with id parent_eid[u]
            # The subtree rooted at u has cnt[u] critical nodes
            # If cnt[u] == 0, this edge is not needed
            # If cnt[u] == t, then the other side has 0 critical nodes, not needed
            if 0 < cnt[u] < t:
                total += edges[parent_eid[u]][2]
    
    print(total)

solve()
