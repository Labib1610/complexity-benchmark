```python
import sys

# The problem asks for the minimum cost to connect all critical variables in a tree.
# The equations x_u - x_v = 0 represent edges in a graph.
# Activating a set of equations means selecting a subset of edges.
# The condition x_i = x_j is entailed if there is a path between i and j in the selected edges.
# We need all critical variables to be in the same connected component.
# This is equivalent to finding the minimum weight Steiner Tree for the set of critical variables.
# Since the graph is a tree (n nodes, n-1 edges, connected), the Steiner Tree is simply
# the unique path in the tree connecting all critical variables.
# The cost is the sum of weights of all edges on the paths between any two critical variables.
# This is equivalent to the sum of weights of all edges that have at least one critical variable
# in one of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = [False] * (n + 1)
    for i in range(t):
        critical[int(input_data[3 + i])] = True
        
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many critical nodes are in each subtree
    subtree_critical = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    total_critical = t
    ans = 0
    for u in reversed(order):
        if critical[u]:
            subtree_critical[u] += 1
        
        if u != 1:
            # If this edge separates some critical nodes from others, it must be included
            if 0 < subtree_critical[u] < total_critical:
                ans += edge_to_parent_w[u]
            
            subtree_critical[parent[u]] += subtree_critical[u]
            
    print(ans)

solve()
```
