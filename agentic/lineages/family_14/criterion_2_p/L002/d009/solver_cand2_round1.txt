```python
import sys

def solve():
    # The problem asks for the minimum weight subset of edges (constraints)
    # such that in the resulting graph, all critical nodes are in the same
    # connected component.
    # The constraint e_u - e_v = 0 implies x_u = x_v.
    # If we select a set of constraints, the null space N(C) consists of
    # vectors x where x_u = x_v for all edges (u, v) in the selected set.
    # This means all nodes in the same connected component of the graph
    # formed by the selected edges must have equal coordinates.
    # We want all t critical nodes to have equal coordinates, so they must
    # all belong to the same connected component.
    # This is equivalent to finding the minimum weight set of edges that
    # connects all t critical nodes.
    # This is the Steiner Tree problem, but since we are given that the
    # original graph is a tree (m = n-1 and it's minimally sufficient),
    # the Steiner Tree is simply the unique path in the tree connecting
    # all critical nodes.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Since the graph is a tree, the Steiner Tree is the union of paths
    # between all critical nodes. This is the set of edges that lie on
    # any path between two critical nodes.
    # An edge is in the Steiner Tree if and only if the two components
    # formed by removing the edge both contain at least one critical node.
    
    total_weight = 0
    
    # Count critical nodes in each subtree
    subtree_critical_count = [0] * (n + 1)
    
    def dfs_count(u, p):
        count = 1 if u in critical_nodes else 0
        for v, w in adj[u]:
            if v != p:
                count += dfs_count(v, u)
        subtree_critical_count[u] = count
        return count
    
    dfs_count(1, -1)
    
    # For each edge (u, v), if we root at 1, let v be the child.
    # The edge is part of the Steiner Tree if 0 < subtree_critical_count[v] < t
    def dfs_sum(u, p):
        nonlocal total_weight
        for v, w in adj[u]:
            if v != p:
                if 0 < subtree_critical_count[v] < t:
                    total_weight += w
                dfs_sum(v, u)
                
    dfs_sum(1, -1)
    print(total_weight)

solve()
```
