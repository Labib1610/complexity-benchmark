```python
import sys

def solve():
    # Read n, m, t
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_indices = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem is equivalent to finding the minimum cost to connect all nodes 
    # in the target set T in the graph formed by the constraints.
    # Since the graph is a tree (m = n-1 and it connects all nodes), 
    # the minimum cost to force x_i = x_j for all i, j in T is the sum of weights 
    # of all edges on the unique paths between nodes in T.
    
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Root the tree at target_indices[0]
    root = target_indices[0]
    
    # parent_edge[i] stores the weight of the edge from i to its parent
    parent_edge = [0] * (n + 1)
    is_target = [False] * (n + 1)
    for x in target_indices:
        is_target[x] = True
        
    # needs_connection[i] is true if the subtree at i contains at least one target node
    needs_connection = [False] * (n + 1)
    
    stack = [(root, -1, 0)]
    order = []
    while stack:
        u, p, w = stack.pop()
        order.append((u, p, w))
        for v, weight in adj[u]:
            if v != p:
                parent_edge[v] = weight
                stack.append((v, u, weight))
                
    # Process in reverse topological order to mark nodes that need connection
    total_cost = 0
    for u, p, w in reversed(order):
        if is_target[u]:
            needs_connection[u] = True
        if needs_connection[u] and p != -1:
            total_cost += w
            needs_connection[p] = True
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
