```python
import sys

# The problem states that the m = n-1 constraints form a tree structure
# because they connect n variables and imply all are equal.
# We need to find the minimum cost to connect all t target variables.
# This is equivalent to finding the sum of weights of all edges on the
# unique paths between all target variables in the tree.
# This is the sum of weights of all edges that have at least one target
# variable in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many target variables are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    target_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in targets:
            target_count[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If this edge separates some targets from others, it must be included
            if 0 < target_count[u] < t:
                total_cost += edge_to_parent_weight[u]
            target_count[p] += target_count[u]
            
    print(total_cost)

solve()
```
