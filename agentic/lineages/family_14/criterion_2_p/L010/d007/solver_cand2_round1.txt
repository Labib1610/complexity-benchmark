```python
import sys

# The problem asks for the minimum cost to connect all t output registers
# in a tree structure (since m = n-1, the graph is a tree).
# This is equivalent to finding the sum of weights of all edges on the
# unique paths between all output registers in the tree.
# In a tree, the set of edges that connect a subset of nodes S is the
# union of all paths between nodes in S. This is exactly the set of edges
# that separate the tree into components such that each component contains
# either zero or at least one node from S, and we only care about edges
# that lie on the path between at least two nodes in S.
# An edge (u, v) with weight w is part of the path between two output
# nodes if and only if removing (u, v) splits the tree into two components,
# each containing at least one output node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    output_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count[u] will store the number of output nodes in the subtree rooted at u
    count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    for u in reversed(order):
        if u in output_nodes:
            count[u] += 1
        if parent[u] != 0:
            p, w = parent[u]
            count[p] += count[u]
            # If the edge (u, p) separates the tree into two parts,
            # both containing at least one output node, it must be included.
            if count[u] > 0 and count[u] < t:
                total_cost += w
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
