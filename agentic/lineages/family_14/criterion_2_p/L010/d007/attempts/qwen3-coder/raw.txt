import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def main():
    import sys

    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1

    idx += 1

    critical = list(map(int, data[idx:idx+t]))
    idx += t

    critical = [x - 1 for x in critical]

    for _ in range(m):
        u = int(data[idx]) - 1
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((w, u, v))
    
    # Sort edges by weight
    edges.sort()
    
    # We need to connect all critical nodes
    # Use Kruskal's algorithm on the induced subgraph
    parent = list(range(n))
    rank = [0] * n
    
    total_cost = 0
    connected_components = t
    
    for w, u, v in edges:
        # Check if u or v is critical
        u_is_critical = u in critical_set if 'critical_set' in locals() else u in set(critical)
        v_is_critical = v in critical_set if 'critical_set' in locals() else v in set(critical)
        
        if not u_is_critical and not v_is_critical:
            continue
            
        if find(parent, u) != find(parent, v):
            if union(parent, rank, u, v):
                total_cost += w
                connected_components -= 1
                if connected_components == 1:
                    break
    
    # Actually, let's rebuild with a better approach
    # Build the MST of the entire graph, but we only care about connecting critical nodes
    
    parent = list(range(n))
    rank = [0] * n
    total_cost = 0
    critical_set = set(critical)
    connected_critical = len(critical)
    
    # Initially, each critical node is its own component
    critical_parent = {node: node for node in critical}
    
    def find_critical(node):
        if critical_parent[node] != node:
            critical_parent[node] = find_critical(critical_parent[node])
        return critical_parent[node]
    
    def union_critical(u, v):
        pu = find_critical(u)
        pv = find_critical(v)
        if pu != pv:
            critical_parent[pv] = pu
            return True
        return False
    
    # For tracking which components contain critical nodes
    component_critical = {}  # component root -> set of critical nodes in that component
    
    for node in critical:
        root = find(parent, node)
        if root not in component_critical:
            component_critical[root] = set()
        component_critical[root].add(node)
    
    for w, u, v in edges:
        root_u = find(parent, u)
        root_v = find(parent, v)
        
        if root_u != root_v:
            # Get critical nodes in each component
            crit_u = component_critical.get(root_u, set())
            crit_v = component_critical.get(root_v, set())
            
            # Union the components
            union(parent, rank, u, v)
            
            # After union, find new root
            new_root = find(parent, u)
            
            # Merge critical node sets
            merged_crit = crit_u | crit_v
            if merged_crit:
                component_critical[new_root] = merged_crit
            
            # Remove old entries
            if root_u in component_critical:
                del component_critical[root_u]
            if root_v in component_critical:
                del component_critical[root_v]
            
            # If we connected two components that both had critical nodes, we needed this edge
            if crit_u and crit_v:
                total_cost += w
                
            # If all critical nodes are now connected, we're done
            if len(merged_crit) == len(critical):
                # Check if we've connected all critical nodes
                final_root = find_critical(list(merged_crit)[0])
                all_same = all(find_critical(node) == final_root for node in merged_crit)
                if all_same and len(merged_crit) == len(critical):
                    break
    
    print(total_cost)

if __name__ == "__main__":
    main()
