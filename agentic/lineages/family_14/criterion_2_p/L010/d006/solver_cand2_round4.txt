```python
import sys

# The problem asks for the sum of weights of all edges that lie on the unique 
# paths between any two core modules. Since the graph is a tree, the set of 
# edges that must be activated is exactly the set of edges that lie on the 
# path between at least one pair of core modules.
# An edge (u, v) with weight w splits the tree into two components of size 
# S and (n - S). This edge is part of a path between two core modules if and 
# only if both components contain at least one core module.
# Let 'count' be the number of core modules in one of the components.
# The edge is needed if 0 < count < t.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    core_modules = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    subtree_core_count = [0] * (n + 1)
    for node in core_modules:
        subtree_core_count[node] = 1
        
    edge_weights = {}
    # Map each node to the weight of the edge connecting it to its parent
    parent_edge_weight = [0] * (n + 1)
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if parent[v] == u:
                parent_edge_weight[v] = w
            elif parent[u] == v:
                parent_edge_weight[u] = w
                
    total_cost = 0
    # Process in reverse topological order
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != -1:
            subtree_core_count[p] += subtree_core_count[u]
            count = subtree_core_count[u]
            if count > 0 and count < t:
                total_cost += parent_edge_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
