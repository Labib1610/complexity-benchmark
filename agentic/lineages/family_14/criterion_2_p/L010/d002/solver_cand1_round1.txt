```python
import sys

# The problem asks for the minimum cost to connect all spore-containing vesicles.
# Since the graph is a tree (m = n-1), there is a unique path between any two nodes.
# To connect a set of nodes in a tree, we must include all edges that lie on the 
# unique paths between any pair of these nodes.
# This is equivalent to finding the smallest subtree containing all spore nodes.
# An edge (u, v) with weight w is part of this subtree if and only if the 
# subtree rooted at u (when the edge is removed) contains at least one spore 
# and the other side also contains at least one spore.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    spores = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # spore_count[u] will store the number of spores in the subtree rooted at u
    spore_count = [0] * (n + 1)
    for s in spores:
        spore_count[s] = 1
        
    total_cost = 0
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        spore_count[p] += spore_count[u]
        # If the subtree at u contains some spores but not all, 
        # the edge to its parent must be included.
        if 0 < spore_count[u] < t:
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
