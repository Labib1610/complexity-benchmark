```python
import sys

# The problem states that the organism is fully contiguous and m < n.
# Since the graph is connected and m < n, it must be a tree.
# We need to find the minimum cost to connect all t spore-containing vesicles.
# In a tree, the set of edges connecting a subset of nodes is the union of 
# paths between them, which is the minimal subtree containing all these nodes.
# This is equivalent to the sum of weights of all edges that have at least one 
# spore-containing node in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    spores = set()
    idx = 3
    for _ in range(t):
        spores.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # spore_count[u] will store the number of spores in the subtree rooted at u
    spore_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in spores:
            spore_count[u] = 1
            
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        spore_count[p] += spore_count[u]
        # If this edge separates some spores from others, it must be included
        if spore_count[u] > 0 and spore_count[u] < t:
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
