```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes in a tree.
# Since the graph is a tree (n nodes, n-1 edges, fully contiguous), there is 
# exactly one unique path between any two nodes.
# To connect a set of nodes {s1, s2, ..., st} in a tree, we must activate all 
# edges that lie on the unique paths between any pair of these nodes.
# This is equivalent to finding the minimal subtree containing all nodes in the set.
# An edge (u, v) with weight w is part of this subtree if and only if the 
# subtree rooted at u (when the edge is removed) contains at least one spore 
# and the subtree rooted at v contains at least one spore.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    spores = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to count how many spores are in the subtree of each node.
    # Since it's a tree, we can root it at an arbitrary node (e.g., 1).
    # For an edge (u, v) where v is a child of u, the edge is included if
    # the number of spores in v's subtree is > 0 and < t.
    
    has_spore = [0] * (n + 1)
    for s in spores:
        has_spore[s] = 1
        
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order (leaves to root)
    subtree_spores = [0] * (n + 1)
    for i in range(1, n + 1):
        subtree_spores[i] = has_spore[i]
        
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        subtree_spores[p] += subtree_spores[u]
        
        # If the subtree at u has spores and is not the whole tree, 
        # the edge to its parent must be included.
        if 0 < subtree_spores[u] < t:
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

solve()
```
