```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    master_keys = set()
    idx = 3
    for _ in range(t):
        master_keys.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count_master[u] stores how many master keys are in the subtree of u
    count_master = [0] * (n + 1)
    total_memory = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u in master_keys:
            count_master[u] += 1
        
        for v, w in adj[u]:
            if v == parent[u]:
                # This is the edge connecting u to its parent
                # If this edge is part of a path between two master keys,
                # it must be included.
                # It is included if count_master[u] > 0 AND (t - count_master[u]) > 0
                if count_master[u] > 0 and (t - count_master[u]) > 0:
                    total_memory += w
            else:
                count_master[u] += count_master[v]
                
    print(total_memory)

if __name__ == '__main__':
    solve()
```
