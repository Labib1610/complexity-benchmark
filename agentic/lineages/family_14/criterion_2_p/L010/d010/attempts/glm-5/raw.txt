import sys
from sys import setrecursionlimit

def main():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    critical = []
    for _ in range(t):
        critical.append(int(data[idx]) - 1)
        idx += 1
    
    edges = []
    for _ in range(m):
        u = int(data[idx]) - 1; idx += 1
        v = int(data[idx]) - 1; idx += 1
        w = int(data[idx]); idx += 1
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    parent = list(range(n))
    rank = [0] * n
    
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    
    def union(x, y):
        rx = find(x)
        ry = find(y)
        if rx == ry:
            return False
        if rank[rx] < rank[ry]:
            parent[rx] = ry
        elif rank[rx] > rank[ry]:
            parent[ry] = rx
        else:
            parent[ry] = rx
            rank[rx] += 1
        return True
    
    adj = [[] for _ in range(n)]
    for i, (u, v, w) in enumerate(edges):
        if union(u, v):
            adj[u].append((v, i))
            adj[v].append((u, i))
    
    LOG = max(1, (n - 1).bit_length())
    up = [[-1] * n for _ in range(LOG)]
    edge_idx = [[-1] * n for _ in range(LOG)]
    depth = [0] * n
    
    stack = [(0, -1, -1, 0)]
    while stack:
        node, par, eidx, d = stack.pop()
        up[0][node] = par
        edge_idx[0][node] = eidx
        depth[node] = d
        for child, e in adj[node]:
            if child != par:
                stack.append((child, node, e, d + 1))
    
    for j in range(1, LOG):
        for i in range(n):
            if up[j-1][i] != -1:
                up[j][i] = up[j-1][up[j-1][i]]
                edge_idx[j][i] = edge_idx[j-1][i]
    
    def get_path_edges(a, b):
        result = set()
        if depth[a] < depth[b]:
            a, b = b, a
        diff = depth[a] - depth[b]
        j = 0
        while diff:
            if diff & 1:
                if edge_idx[j][a] != -1:
                    result.add(edge_idx[j][a])
                a = up[j][a]
            diff >>= 1
            j += 1
        
        if a == b:
            return result
        
        for j in range(LOG - 1, -1, -1):
            if up[j][a] != up[j][b] and up[j][a] != -1:
                result.add(edge_idx[j][a])
                result.add(edge_idx[j][b])
                a = up[j][a]
                b = up[j][b]
        
        result.add(edge_idx[0][a])
        result.add(edge_idx[0][b])
        return result
    
    required = set()
    for i in range(t - 1):
        a = critical[i]
        b = critical[i + 1]
        path_edges = get_path_edges(a, b)
        required.update(path_edges)
    
    total_cost = 0
    for e in required:
        total_cost += edges[e][2]
    
    print(total_cost)

if __name__ == "__main__":
    main()
