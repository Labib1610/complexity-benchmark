```python
import sys

# The problem asks for the minimum cost to span the differences between all critical variables.
# The constraints form a tree (n nodes, n-1 edges).
# The difference x_u - x_v is in the span of a subset of edges if and only if
# u and v are in the same connected component formed by the subset of edges.
# To have x_{c_a} - x_{c_b} in the span for all critical variables c_a, c_b,
# all critical variables must belong to the same connected component.
# We need to find the minimum weight subset of edges that connects all critical variables.
# This is the Steiner Tree problem on a tree, which is simply the sum of weights
# of all edges that lie on the unique paths between any two critical variables.
# An edge lies on such a path if and only if removing it splits the tree into two
# components, each containing at least one critical variable.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_vars = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count critical variables in subtree
    subtree_critical = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    for u in order:
        if u in critical_vars:
            subtree_critical[u] += 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if parent[u] != -1:
            p = parent[u]
            # Find weight of edge (u, p)
            weight = 0
            for v, w in adj[u]:
                if v == p:
                    weight = w
                    break
            
            # If this edge separates critical variables, it must be included
            if 0 < subtree_critical[u] < t:
                total_cost += weight
            
            subtree_critical[p] += subtree_critical[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
