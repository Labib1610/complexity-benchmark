```python
import sys

# The problem asks for the minimum weight of a set of edges such that all 
# quality control stations are in the same connected component.
# Since the graph is a tree, the set of edges that connects a subset of 
# vertices is the union of the paths between all pairs of these vertices.
# This is equivalent to the sum of weights of all edges that lie on at least 
# one path between two quality control stations.
# An edge (u, v) with weight w lies on a path between two QC stations if and 
# only if the removal of this edge splits the tree into two components, 
# each containing at least one QC station.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    qc_stations = set()
    idx = 3
    for _ in range(t):
        qc_stations.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to count how many QC stations are in the subtree of each node.
    # Since n is up to 200,000, we use an iterative DFS to avoid recursion depth issues.
    
    qc_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in order:
        if u in qc_stations:
            qc_count[u] = 1
            
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            qc_count[p] += qc_count[u]
            # If the subtree at u contains some but not all QC stations,
            # the edge to the parent must be included.
            if 0 < qc_count[u] < t:
                total_weight += edge_weight_to_parent[u]
                
    print(total_weight)

if __name__ == '__main__':
    solve()
```
