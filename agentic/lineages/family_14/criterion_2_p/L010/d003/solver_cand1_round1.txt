```python
import sys

# The problem states the network is a tree (n nodes, n-1 edges, fully integrated).
# To connect a subset of nodes in a tree, we must include all edges that lie
# on the unique paths between any two nodes in the subset.
# This is equivalent to the sum of weights of all edges that have at least one
# quality control station in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    qc_stations = set()
    idx = 3
    for _ in range(t):
        qc_stations.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use DFS to count how many QC stations are in the subtree of each node
    # Since n is up to 200,000, use iterative DFS to avoid recursion depth issues
    qc_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for node in order:
        if node in qc_stations:
            qc_count[node] += 1
            
    total_energy = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        qc_count[p] += qc_count[u]
        
        # If the edge (u, p) separates the tree into two parts,
        # and both parts contain at least one QC station, the edge must be included.
        if qc_count[u] > 0 and qc_count[u] < t:
            total_energy += edge_weight_to_parent[u]
            
    print(total_energy)

if __name__ == '__main__':
    solve()
```
