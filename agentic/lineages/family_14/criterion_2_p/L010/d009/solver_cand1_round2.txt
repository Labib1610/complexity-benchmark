```python
import sys

# The problem asks for the minimum cost to connect all target variables in a tree.
# Since the input is a tree (n nodes, n-1 edges), the set of edges that connect
# a subset of nodes is the unique path between them.
# The union of all paths between any two target variables is the minimal subtree
# that contains all target variables.
# We can find this by rooting the tree arbitrarily (e.g., at node 1) and
# marking all nodes that are on a path between any two target variables.
# A node is in the minimal subtree if it is a target variable or if it has
# at least two branches (in the rooted sense) that contain target variables.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Root the tree at an arbitrary target node
    root = next(iter(targets))
    
    # has_target[u] is true if the subtree at u contains at least one target
    has_target = [False] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in targets:
            has_target[u] = True
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to propagate has_target
    total_cost = 0
    for u in reversed(order):
        if u == root:
            continue
        if has_target[u]:
            total_cost += parent_edge_weight[u]
            if parent[u] != 0:
                has_target[parent[u]] = True
                
    print(total_cost)

solve()
```
