import sys

def solve():
    # Read n, m, t
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m, t = map(int, line1)
        
        # Read target variables
        targets = list(map(int, sys.stdin.readline().split()))
        
        # Read edges
        edges = []
        for _ in range(m):
            u, v, w = map(int, sys.stdin.readline().split())
            edges.append((u, v, w))
    except ValueError:
        return

    if t <= 1:
        print(0)
        return

    # The problem states that the m=n-1 edges form a tree (since they connect n variables
    # and enforce x1=x2=...=xn).
    # We need to find the sum of weights of all edges on the unique path between
    # any two nodes in the target set.
    # Since all target nodes must be equal, they must all belong to the same connected
    # component in the graph formed by the chosen edges.
    # Because the original graph is a tree, the only way to make all target nodes
    # equal is to include all edges on the unique path connecting all target nodes.
    
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Find the minimal subtree containing all target nodes.
    # The total weight is the sum of weights of all edges in this subtree.
    # We can use DFS to find the subtree.
    
    is_target = [False] * (n + 1)
    for x in targets:
        is_target[x] = True
        
    # subtree_has_target[u] is true if the subtree rooted at u contains a target node
    subtree_has_target = [False] * (n + 1)
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    order = []
    stack = [targets[0]]
    visited = [False] * (n + 1)
    visited[targets[0]] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_target[u]:
            subtree_has_target[u] = True
        if subtree_has_target[u] and parent[u] != 0:
            subtree_has_target[parent[u]] = True
            total_weight += edge_to_parent_w[u]
            
    print(total_weight)

solve()
