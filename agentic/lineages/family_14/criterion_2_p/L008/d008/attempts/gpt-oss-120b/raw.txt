```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    R = int(next(it))
    S = int(next(it))  # = R-1
    P = int(next(it))
    required = [0] * (R + 1)
    for _ in range(P):
        required[int(next(it))] = 1

    adj = [[] for _ in range(R + 1)]
    edges = []
    for _ in range(S):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    parent = [0] * (R + 1)
    pw = [0] * (R + 1)          # weight to parent
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v == parent[u]:
                continue
            parent[v] = u
            pw[v] = w
            stack.append(v)

    cnt = [0] * (R + 1)
    for u in range(1, R + 1):
        cnt[u] = required[u]

    ans = 0
    for u in reversed(order):
        if parent[u] == -1:
            continue
        cnt[parent[u]] += cnt[u]
        if 0 < cnt[u] < P:
            ans += pw[u]

    print(ans)

if __name__ == "__main__":
    solve()
```
