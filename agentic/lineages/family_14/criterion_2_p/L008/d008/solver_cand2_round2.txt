```python
import sys

# The problem asks for the minimum weight of internal edges in a connected subgraph
# that contains all P core rights. Since the graph is a tree (R nodes, R-1 edges),
# the unique path between any two core rights must be included in the subgraph.
# The union of all paths between all pairs of core rights forms the minimal
# subtree containing all core rights. Any additional nodes/edges would only
# increase the total weight of internal edges.
# Thus, the answer is the sum of weights of all edges that lie on the path
# between at least two core rights.
# An edge (u, v) with weight w is part of this minimal subtree if and only if
# the subtree formed by removing (u, v) contains at least one core right in
# both components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    core_rights = set(map(int, input_data[3:3+P]))
    
    adj = [[] for _ in range(R + 1)]
    idx = 3 + P
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many core rights are in the subtree of each node
    # We root the tree at an arbitrary node, say 1.
    subtree_core_count = [0] * (R + 1)
    parent = [0] * (R + 1)
    edge_to_parent_weight = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in core_rights:
        subtree_core_count[u] = 1
        
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        subtree_core_count[p] += subtree_core_count[u]
        # If the subtree at u contains some core rights, and the rest of the
        # tree also contains some, then the edge (u, p) must be included.
        if 0 < subtree_core_count[u] < P:
            total_weight += edge_to_parent_weight[u]
            
    print(total_weight)

solve()
```
