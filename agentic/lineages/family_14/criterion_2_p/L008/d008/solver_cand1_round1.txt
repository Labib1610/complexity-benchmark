```python
import sys

# The problem states the system is minimally cohesive (a tree) and we need to 
# activate a set of entities that includes all P mandated entities.
# The set of activated entities must be synchronous (connected).
# The cost is the sum of weights of edges where both endpoints are activated.
# If we activate a connected subgraph containing all P mandated entities,
# the cost is the sum of weights of all edges within that subgraph.
# To minimize this, we should pick the smallest connected subgraph containing
# all P mandated entities, which is the Steiner Tree of the P entities.
# In a tree, the Steiner Tree is the unique minimal subtree connecting all P nodes.
# The cost is the sum of weights of all edges on the paths between the P nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set()
    idx = 3
    for _ in range(P):
        mandated.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # Find the Steiner Tree: the union of paths between all mandated nodes.
    # An edge is in the Steiner Tree if and only if the subtree on one side
    # of the edge contains at least one mandated node and the other side
    # also contains at least one mandated node.
    
    has_mandated = [False] * (R + 1)
    for m in mandated:
        has_mandated[m] = True
        
    total_cost = 0
    
    # Use DFS to count mandated nodes in subtrees
    # Since R is up to 200,000, use iterative DFS to avoid recursion limit
    parent = [0] * (R + 1)
    edge_to_parent_w = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    count = [0] * (R + 1)
    for i in range(1, R + 1):
        if has_mandated[i]:
            count[i] = 1
            
    for u in reversed(order):
        if u == 1: continue
        p = parent[u]
        if count[u] > 0 and count[u] < P:
            total_cost += edge_to_parent_w[u]
        count[p] += count[u]
        
    print(total_cost)

solve()
```
