```python
import sys

# The problem states S = R - 1 and the network is connected.
# This means the network is a tree.
# We need to select a set of servers S' such that all P mandated servers are in S'.
# The condition "number of active cables = number of selected servers - 1"
# means the subgraph induced by S' must be a tree (connected and acyclic).
# Since the original graph is a tree, any induced subgraph is a forest.
# For it to be a tree, it must be connected.
# Thus, we need to select a connected subgraph (a subtree) that contains all P mandated servers.
# The smallest such subtree is the unique minimal subtree containing all P mandated servers.
# Any additional servers selected must be connected to this subtree to maintain the tree property.
# However, adding any node to the subtree increases the number of nodes by 1 and the number of edges by 1.
# The cost of the active cables is the sum of weights of edges in the induced subgraph.
# If we select a subtree, the cost is the sum of weights of all edges in that subtree.
# To minimize the cost, we should select exactly the minimal subtree containing all P mandated servers.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set(map(int, input_data[3:3+P]))
    
    adj = [[] for _ in range(R + 1)]
    idx = 3 + P
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # Find the minimal subtree containing all mandated nodes.
    # A node is in the minimal subtree if it lies on the path between some two mandated nodes.
    # This is equivalent to saying the subtree is formed by pruning all branches that do not contain a mandated node.
    
    is_mandated = [False] * (R + 1)
    for m in mandated:
        is_mandated[m] = True
        
    has_mandated_in_subtree = [False] * (R + 1)
    
    # Root the tree at an arbitrary mandated node
    root = next(iter(mandated))
    
    parent = [0] * (R + 1)
    edge_to_parent_weight = [0] * (R + 1)
    order = []
    stack = [root]
    visited = [False] * (R + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if is_mandated[u]:
            has_mandated_in_subtree[u] = True
            
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == root:
            continue
        if has_mandated_in_subtree[u]:
            total_cost += edge_to_parent_weight[u]
            has_mandated_in_subtree[parent[u]] = True
            
    print(total_cost)

solve()
```
