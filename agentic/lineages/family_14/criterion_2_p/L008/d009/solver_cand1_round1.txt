```python
import sys

# The problem states that the system is a tree (R entities, R-1 links, connected).
# Let A be the set of Active entities. Let S_A be the set of links where both endpoints are in A.
# The condition is |S_A| = |A| - 1.
# In a forest, if we have a set of vertices A, the number of edges in the induced subgraph
# is |A| - (number of connected components in the induced subgraph).
# Thus, |S_A| = |A| - (number of connected components in the induced subgraph).
# The condition |S_A| = |A| - 1 implies that the induced subgraph of A must be connected.
# So we need to find a connected subgraph containing all P mandated entities
# such that the sum of weights of edges in the subgraph is minimized.
# This is exactly the Steiner Tree problem on a tree, which is the sum of weights
# of all edges on the unique paths between the mandated entities.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set(map(int, input_data[3:3+P]))
    
    adj = [[] for _ in range(R + 1)]
    idx = 3 + P
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # To find the Steiner Tree in a tree, we identify all edges that lie on
    # the path between any two mandated nodes. An edge is part of the Steiner
    # tree if and only if the subtree formed by removing the edge contains
    # at least one mandated node on both sides.
    
    has_mandated = [False] * (R + 1)
    for m in mandated:
        has_mandated[m] = True
        
    total_weight = 0
    
    # Use DFS to count mandated nodes in subtrees
    # Since R is up to 200,000, use iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    edge_to_parent_w = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    count_mandated = [0] * (R + 1)
    for i in range(1, R + 1):
        if has_mandated[i]:
            count_mandated[i] = 1
            
    for u in reversed(order):
        if u == 1: continue
        p = parent[u]
        count_mandated[p] += count_mandated[u]
        if count_mandated[u] > 0 and count_mandated[u] < P:
            total_weight += edge_to_parent_w[u]
            
    print(total_weight)

solve()
```
