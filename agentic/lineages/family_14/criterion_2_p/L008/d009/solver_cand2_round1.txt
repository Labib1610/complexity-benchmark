```python
import sys

# The problem describes a system of R entities and S links formed by a 
# sequential growth process where each link adds one new entity.
# This implies the system is a tree (R = S + 1).
# The balance condition: (number of saturated links) = (number of activated entities) - 1.
# In a forest, if we have k activated entities and m saturated links, 
# the condition m = k - 1 implies that the activated entities form a single connected component (a tree).
# We need to find a subtree that contains all P mandated entities and minimizes the sum of weights of the edges in that subtree.
# This is the Steiner Tree problem on a tree, which is equivalent to finding the minimal subtree 
# containing all P nodes. The minimal subtree is the union of all paths between the P nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = [int(x) for x in input_data[3:3+P]]
    
    adj = [[] for _ in range(R + 1)]
    idx = 3 + P
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # To find the minimal subtree containing all mandated nodes:
    # 1. Root the tree at mandated[0].
    # 2. An edge is in the minimal subtree if and only if the subtree 
    #    defined by that edge contains at least one mandated node 
    #    and the rest of the tree contains at least one mandated node.
    
    is_mandated = [False] * (R + 1)
    for m in mandated:
        is_mandated[m] = True
        
    subtree_has_mandated = [False] * (R + 1)
    total_mandated = P
    
    parent = [0] * (R + 1)
    edge_to_parent_weight = [0] * (R + 1)
    order = []
    stack = [mandated[0]]
    visited = [False] * (R + 1)
    visited[mandated[0]] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Count mandated nodes in subtree of each node
    count = [0] * (R + 1)
    for i in range(1, R + 1):
        if is_mandated[i]:
            count[i] = 1
            
    total_weight = 0
    for u in reversed(order):
        if u == mandated[0]:
            continue
        if count[u] > 0 and count[u] < total_mandated:
            total_weight += edge_to_parent_weight[u]
        if parent[u] != 0:
            count[parent[u]] += count[u]
            
    print(total_weight)

solve()
```
