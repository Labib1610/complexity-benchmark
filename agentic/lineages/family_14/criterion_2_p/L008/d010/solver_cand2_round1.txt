```python
import sys

# The problem asks for a subset of nodes including all P mandated nodes
# such that (number of nodes) - (number of internal links) = 1.
# In a tree, this condition is equivalent to saying the chosen nodes
# form a connected subtree.
# Proof: For any forest with V nodes and E edges, V - E = number of connected components.
# If V - E = 1, the chosen nodes must form a single connected component.
# Thus, we must choose the smallest subtree that contains all P mandated nodes.
# The cost is the sum of weights of all edges in this minimal subtree.
# The minimal subtree containing a set of nodes is the union of all paths
# between them, which is the subtree spanned by the minimal Steiner tree.
# In a tree, this is simply the set of all edges that lie on a path between
# any two mandated nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set(map(int, input_data[3:3+P]))
    
    adj = [[] for _ in range(R + 1)]
    idx = 3 + P
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the sum of weights of the minimal subtree:
    # Root the tree at an arbitrary mandated node (e.g., the first one).
    # An edge (u, v) where v is a child of u is in the minimal subtree
    # if and only if the subtree rooted at v contains at least one mandated node.
    
    root = next(iter(mandated))
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    edge_to_parent_w = [0] * (R + 1)
    order = []
    stack = [root]
    visited = [False] * (R + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    has_mandated = [False] * (R + 1)
    for m in mandated:
        has_mandated[m] = True
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(R - 1, 0, -1):
        u = order[i]
        if has_mandated[u]:
            p = parent[u]
            if p != 0:
                total_cost += edge_to_parent_w[u]
                has_mandated[p] = True
                
    print(total_cost)

solve()
```
