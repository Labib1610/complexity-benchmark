import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    R = int(next(it))
    S = int(next(it))  # = R-1
    P = int(next(it))
    required = [0] * (R + 1)
    for _ in range(P):
        required[int(next(it))] = 1
    adj = [[] for _ in range(R + 1)]
    edges = []
    for _ in range(S):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    parent = [0] * (R + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, _ in adj[u]:
            if v != parent[u]:
                parent[v] = u
                stack.append(v)

    sub_cnt = [0] * (R + 1)
    for u in reversed(order):
        cnt = required[u]
        for v, _ in adj[u]:
            if parent[v] == u:
                cnt += sub_cnt[v]
        sub_cnt[u] = cnt

    total = 0
    for u in range(2, R + 1):
        cnt = sub_cnt[u]
        if 0 < cnt < P:
            # edge between u and parent[u] is needed
            # find its weight
            pu = parent[u]
            # retrieve weight (could store during traversal)
            # we can store weight in parent array
            # Let's build weight_parent array earlier
            pass

    # Better: store weight to parent during DFS
    weight_to_parent = [0] * (R + 1)
    # redo traversal to fill weight_to_parent
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                weight_to_parent[v] = w
                stack.append(v)

    total = 0
    for v in range(2, R + 1):
        cnt = sub_cnt[v]
        if 0 < cnt < P:
            total += weight_to_parent[v]

    print(total)

if __name__ == "__main__":
    solve()
