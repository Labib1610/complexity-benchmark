```python
import sys

# The problem states that there are R rooms and S = R - 1 segments.
# This means the graph is a tree.
# We need to find the minimum cost to connect all P high-priority rooms.
# In a tree, there is exactly one path between any two nodes.
# The set of edges that connect all P high-priority rooms is the union of all
# paths between these rooms. This is equivalent to the minimal subtree
# containing all P high-priority rooms.
# An edge belongs to this subtree if and only if the removal of the edge
# splits the tree into two components, each containing at least one high-priority room.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    if P <= 1:
        print(0)
        return
    
    priority_rooms = set(map(int, input_data[3:3+P]))
    
    adj = [[] for _ in range(R + 1)]
    idx = 3 + P
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many priority rooms are in the subtree rooted at each node
    # We can use DFS. Since R is up to 200,000, we need to avoid recursion depth issues.
    count = [0] * (R + 1)
    parent = [0] * (R + 1)
    edge_to_parent_weight = [0] * (R + 1)
    
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in priority_rooms:
            count[node] = 1
            
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(R - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        count[p] += count[u]
        
        # If this subtree contains some but not all priority rooms,
        # the edge to the parent must be part of the minimal subtree.
        if count[u] > 0 and count[u] < P:
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    sys.setrecursionlimit(300000)
    solve()
```
