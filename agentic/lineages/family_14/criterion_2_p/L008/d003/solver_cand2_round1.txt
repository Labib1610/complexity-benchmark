```python
import sys

# The problem asks for the minimum cost to connect a set of high-priority rooms.
# The input describes a graph where the number of edges equals the number of nodes minus 1.
# This means the graph is a forest (a collection of trees).
# Since the problem states that all high-priority rooms are in a portion of the network
# that contains no redundant paths (a tree structure), we are essentially looking for
# the sum of weights of all edges on the unique paths between all high-priority rooms.
# This is equivalent to finding the Steiner Tree in a tree, which is simply the sum
# of weights of all edges that lie on at least one path between two high-priority nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    high_priority = set(map(int, input_data[3:3+k]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + k
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if k <= 1:
        print(0)
        return

    # To find edges on the path between any two high-priority nodes,
    # we can root the tree at an arbitrary high-priority node and
    # count how many high-priority nodes are in the subtree of each node.
    # An edge (u, v) where v is a child of u is part of the Steiner tree
    # if the subtree at v contains at least one high-priority node
    # AND the rest of the tree contains at least one high-priority node.
    
    root = next(iter(high_priority))
    subtree_hp = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in reversed(order):
        if node in high_priority:
            subtree_hp[node] += 1
        if parent[node] != 0:
            subtree_hp[parent[node]] += subtree_hp[node]
            
    total_cost = 0
    for node in range(1, n + 1):
        if node == root:
            continue
        # If subtree has some HP nodes and the rest of the tree has some HP nodes
        if subtree_hp[node] > 0 and subtree_hp[node] < k:
            total_cost += parent_edge_weight[node]
            
    print(total_cost)

solve()
```
