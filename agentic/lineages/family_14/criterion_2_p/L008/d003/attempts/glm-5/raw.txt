```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        R = int(next(iterator))
        S = int(next(iterator))
        P = int(next(iterator))
    except StopIteration:
        return
    
    compromised = set()
    for _ in range(P):
        compromised.add(int(next(iterator)))
    
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(next(iterator))
        v = int(next(iterator))
        w = int(next(iterator))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Prim's algorithm starting from all compromised vaults
    total_cost = 0
    visited = [False] * (R + 1)
    pq = []
    
    for node in compromised:
        visited[node] = True
        for neighbor, weight in adj[node]:
            if not visited[neighbor]:
                heapq.heappush(pq, (weight, neighbor))
    
    while pq:
        weight, u = heapq.heappop(pq)
        if visited[u]:
            continue
        visited[u] = True
        total_cost += weight
        for v, w in adj[u]:
            if not visited[v]:
                heapq.heappush(pq, (w, v))
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
