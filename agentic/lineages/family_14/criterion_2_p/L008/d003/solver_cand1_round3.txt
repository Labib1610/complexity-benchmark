```python
import sys

# The problem states that the tunnels form a tree (R vaults, R-1 tunnels, 
# connected, unique path between any two).
# We need to find the minimum cost to connect all P compromised vaults.
# In a tree, the minimum set of edges connecting a subset of nodes is the 
# union of all paths between every pair of nodes in that subset.
# This is equivalent to the Steiner Tree problem on a tree, which is simply 
# the sum of weights of all edges that lie on at least one path between 
# two compromised nodes.
# An edge (u, v) with weight w lies on a path between two compromised nodes 
# if and only if removing that edge splits the tree into two components, 
# each containing at least one compromised node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    compromised = set()
    idx = 3
    for _ in range(P):
        compromised.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    edges = []
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # subtree_comp[u] stores the number of compromised nodes in the subtree 
    # rooted at u (when rooted at 1).
    subtree_comp = [0] * (R + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    edge_to_parent_w = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    for u in order:
        if u in compromised:
            subtree_comp[u] = 1
            
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        subtree_comp[p] += subtree_comp[u]
        # If this edge separates the tree into two parts, both having 
        # at least one compromised node, it must be included.
        if subtree_comp[u] > 0 and subtree_comp[u] < P:
            total_cost += edge_to_parent_w[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
