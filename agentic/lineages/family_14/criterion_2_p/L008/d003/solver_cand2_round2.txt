```python
import sys

# The problem states that the facility has R vaults and S = R-1 corridors,
# forming a tree structure (exactly one path between any two nodes).
# We need to find the minimum cost to connect a subset of P critical vaults.
# In a tree, the minimum cost to connect a set of nodes is the sum of the
# weights of all edges that lie on the unique paths between any two critical nodes.
# An edge is part of the Steiner Tree if and only if it splits the tree into
# two components, both of which contain at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    critical_vaults = set()
    idx = 3
    for _ in range(P):
        critical_vaults.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # We need to count how many critical nodes are in each subtree.
    # We can use DFS to compute this.
    # Since R can be up to 200,000, we need to avoid recursion depth issues.
    
    subtree_critical_count = [0] * (R + 1)
    parent = [0] * (R + 1)
    edge_to_parent_weight = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in critical_vaults:
            subtree_critical_count[node] = 1
            
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(R - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        count = subtree_critical_count[u]
        # If this subtree contains some but not all critical nodes,
        # the edge to the parent must be included.
        if 0 < count < P:
            total_cost += edge_to_parent_weight[u]
        
        subtree_critical_count[p] += count
        
    print(total_cost)

solve()
```
