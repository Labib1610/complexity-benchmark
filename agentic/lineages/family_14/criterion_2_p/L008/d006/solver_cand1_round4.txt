```python
import sys

# The problem asks for the minimum cost to connect a set of P mandated components.
# Since the graph is a tree (R nodes, R-1 edges), the unique contiguous cluster 
# containing all P mandated components is the minimal subtree that spans all 
# mandated nodes. The cost of this cluster is the sum of the weights of all 
# edges that lie on the paths between any two mandated nodes.
# An edge is part of this minimal subtree if and only if it separates the 
# set of mandated nodes into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    mandated = set()
    idx = 3
    for _ in range(P):
        mandated.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # count[u] will store the number of mandated nodes in the subtree rooted at u
    count = [0] * (R + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    stack = [(1, 0, 0)] # node, parent, edge_weight_to_parent
    order = []
    parent = [0] * (R + 1)
    edge_to_parent = [0] * (R + 1)
    
    dfs_stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while dfs_stack:
        u = dfs_stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent[v] = w
                dfs_stack.append(v)
                
    for u in reversed(order):
        if u in mandated:
            count[u] += 1
        if parent[u] != 0:
            count[parent[u]] += count[u]
            # If this edge separates mandated nodes, it must be included
            if count[u] > 0 and count[u] < P:
                total_cost += edge_to_parent[u]
                
    print(total_cost)

solve()
```
