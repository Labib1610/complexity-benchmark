```python
import sys

# The problem states that the graph is a tree (R nodes, R-1 edges, connected).
# A subset Z is cohesive if it contains K and is connected.
# In a tree, the smallest connected subset containing a set of nodes K is the 
# minimal subtree spanning all nodes in K.
# Any cohesive subset Z must contain this minimal subtree.
# To minimize the total weight of internalized edges, we should choose Z to be 
# exactly this minimal subtree.
# The total weight is the sum of weights of all edges in the minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    K = set()
    idx = 3
    for _ in range(P):
        K.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # To find the minimal subtree containing K:
    # 1. Root the tree at an arbitrary node in K (e.g., list(K)[0]).
    # 2. A node u is in the minimal subtree if it is in K or if it has 
    #    descendants in K in at least two different subtrees, or if it is on 
    #    the path between two nodes in K.
    # 3. More simply: an edge (u, v) where v is a child of u is in the 
    #    minimal subtree if and only if the subtree rooted at v contains 
    #    at least one node from K, and the rest of the tree (outside v's subtree)
    #    contains at least one node from K.
    
    root = next(iter(K))
    has_k = [False] * (R + 1)
    for node in K:
        has_k[node] = True
        
    subtree_has_k = [False] * (R + 1)
    parent_edge_weight = [0] * (R + 1)
    parent = [0] * (R + 1)
    order = []
    stack = [root]
    visited = [False] * (R + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if has_k[u]:
            subtree_has_k[u] = True
        for v, w in adj[u]:
            if v != parent[u]:
                if subtree_has_k[v]:
                    subtree_has_k[u] = True
                    
    total_weight = 0
    # Count nodes in K
    count_k = P
    for i in range(1, R + 1):
        if i == root:
            continue
        # Edge (parent[i], i) is in the minimal subtree if:
        # 1. The subtree at i contains at least one K
        # 2. The rest of the tree contains at least one K
        if subtree_has_k[i] and (count_k - (sum(1 for x in range(1, R+1) if subtree_has_k[x] and x == i) if False else True)):
            # Actually, simpler: edge is in if subtree_has_k[i] is True 
            # AND (count_k_in_subtree_i < count_k)
            # We can precalculate count_k_in_subtree
            pass

    # Correct approach:
    count_k_in_subtree = [0] * (R + 1)
    for u in reversed(order):
        count_k_in_subtree[u] += (1 if u in K else 0)
        if parent[u] != 0:
            count_k_in_subtree[parent[u]] += count_k_in_subtree[u]
            
    total_weight = 0
    for i in range(1, R + 1):
        if i == root: continue
        # Edge (parent[i], i) is in if 0 < count_k_in_subtree[i] < P
        if 0 < count_k_in_subtree[i] < P:
            total_weight += parent_edge_weight[i]
            
    print(total_weight)

solve()
```
