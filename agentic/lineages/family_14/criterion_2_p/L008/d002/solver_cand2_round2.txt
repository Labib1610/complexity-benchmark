```python
import sys

# The problem states the infrastructure is a tree (n rooms, n-1 segments, no loops).
# To connect a subset of nodes in a tree, the minimum cost is the sum of the weights
# of all edges that lie on the unique paths between any two high-priority nodes.
# An edge is part of such a path if and only if the subtree formed by removing
# that edge contains at least one high-priority node on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    high_priority = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        high_priority[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many high-priority nodes are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    subtree_hp = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if high_priority[u]:
            subtree_hp[u] += 1
        if parent[u] != 0:
            subtree_hp[parent[u]] += subtree_hp[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # Edge connects node i to its parent.
        # It is needed if the subtree at i has some HP nodes,
        # and the rest of the tree (total_hp - subtree_hp[i]) also has some.
        count_in_subtree = subtree_hp[i]
        if count_in_subtree > 0 and count_in_subtree < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
