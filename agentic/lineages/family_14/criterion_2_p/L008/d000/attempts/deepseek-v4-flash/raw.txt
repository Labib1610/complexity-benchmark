We can solve this by finding the minimal subtree that connects all high-priority rooms.  
Since the graph is a tree, the minimal connected subgraph containing all terminals is the union of paths between them, which is the **Steiner tree** in a tree — and that’s simply the minimal subtree spanning all terminals.

We can compute it by:

1. Marking all high-priority nodes.
2. Doing a DFS to find the minimal subtree that contains all marked nodes.
3. Summing the edge weights in that subtree.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    terminals = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # DFS to find the minimal subtree containing all terminals
    visited = [False] * (n + 1)
    
    def dfs(u, parent):
        visited[u] = True
        needed = u in terminals
        total_cost = 0
        for v, w in adj[u]:
            if v == parent:
                continue
            child_needed, child_cost = dfs(v, u)
            if child_needed:
                total_cost += child_cost + w
                needed = True
        return needed, total_cost
    
    # Start from any terminal
    start = next(iter(terminals))
    _, cost = dfs(start, -1)
    print(cost)

if __name__ == "__main__":
    solve()
```
