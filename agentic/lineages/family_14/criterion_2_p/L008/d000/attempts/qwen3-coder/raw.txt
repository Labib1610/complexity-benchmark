import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1000000)
    
    n, m, t = map(int, sys.stdin.readline().split())
    if t <= 1:
        print(0)
        return
        
    terminals = set(map(int, sys.stdin.readline().split()))
    
    # Build the tree
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # Step 1: Find the subtree that connects all terminals
    # We can do this by finding all nodes that are on paths between terminals
    
    # First, find one terminal to start BFS
    start = next(iter(terminals))
    
    # BFS to find distances and parents
    dist = [-1] * (n+1)
    parent = [-1] * (n+1)
    dist[start] = 0
    queue = deque([start])
    
    while queue:
        u = queue.popleft()
        for v, w in adj[u]:
            if dist[v] == -1:
                dist[v] = dist[u] + 1
                parent[v] = u
                queue.append(v)
    
    # Now, we need to find the minimal subtree connecting all terminals
    # This is essentially finding the Steiner tree for the terminals in a tree
    
    # In a tree, the minimal Steiner tree can be found by:
    # 1. Take all paths between pairs of terminals
    # 2. The union of these paths forms the Steiner tree
    # But there's a more efficient way:
    # The minimal Steiner tree in a tree is the subtree induced by the terminals
    # and their lowest common ancestors.
    
    # A simpler approach for trees:
    # Root the tree at some node, then the minimal Steiner tree is the subtree
    # that includes all terminals and all their ancestors up to their LCA.
    # Even simpler: take all nodes that lie on paths between any two terminals.
    
    # Efficient approach:
    # 1. Root the tree at an arbitrary node
    # 2. For each terminal, mark all ancestors
    # 3. The minimal Steiner tree consists of marked nodes and edges between them
    
    # Let's root at node 'start'
    # Compute depth and ancestors for LCA preprocessing
    depth = [0] * (n+1)
    ancestors = [[-1]*20 for _ in range(n+1)]  # up to 2^19
    
    def dfs(u, p, d):
        parent[u] = p
        depth[u] = d
        ancestors[u][0] = p
        for k in range(1, 20):
            if ancestors[u][k-1] != -1:
                ancestors[u][k] = ancestors[ancestors[u][k-1]][k-1]
            else:
                ancestors[u][k] = -1
        for v, w in adj[u]:
            if v != p:
                dfs(v, u, d+1)
                
    dfs(start, -1, 0)
    
    def lift(u, k):
        for i in range(19, -1, -1):
            if k >= (1 << i) and ancestors[u][i] != -1:
                u = ancestors[u][i]
                k -= (1 << i)
        return u
    
    def lca(u, v):
        if depth[u] < depth[v]:
            u, v = v, u
        u = lift(u, depth[u] - depth[v])
        if u == v:
            return u
        for i in range(19, -1, -1):
            if ancestors[u][i] != ancestors[v][i]:
                u = ancestors[u][i]
                v = ancestors[v][i]
        return ancestors[u][0]
    
    # Mark all nodes that are terminals or on paths between terminals
    # Actually, we just need to find the minimal connected subgraph containing all terminals
    # In a tree, this is equivalent to finding the union of paths between all pairs
    # But even better, it's the subtree that spans all terminals when we look at 
    # the tree rooted appropriately.
    
    # Simpler idea: 
    # 1. Root the tree at one terminal
    # 2. Find the farthest terminal from root (in terms of edges)
    # 3. Find the farthest terminal from that terminal - this gives us a diameter path
    # 4. All other terminals' paths to this diameter will form the Steiner tree
    
    # Even simpler for implementation:
    # Use the fact that in a tree, the minimal Steiner tree can be computed by:
    # 1. Rooting the tree
    # 2. Finding the minimal subtree that contains all terminals
    
    # Let's use a different approach:
    # Since it's a tree, we can compute the minimal subtree containing all terminals
    # by taking the union of all paths between terminal pairs.
    # But that's O(t^2) paths.
    
    # Better approach:
    # 1. Root the tree at an arbitrary node
    # 2. Find the set of nodes that are needed: all terminals and their LCAs
    # 3. The minimal Steiner tree is the subtree induced by these nodes
    
    # But actually, there's a linear-time algorithm for Steiner tree in trees:
    # 1. Root the tree at any node
    # 2. Do a DFS, and for each node, determine if its subtree contains any terminal
    # 3. An edge is included in the Steiner tree iff both its endpoints' subtrees contain terminals
    
    visited = [False] * (n+1)
    contains_terminal = [False] * (n+1)
    total_cost = 0
    
    def dfs_steiner(u, parent_node):
        nonlocal total_cost
        contains_terminal[u] = (u in terminals)
        for v, w in adj[u]:
            if v == parent_node:
                continue
            dfs_steiner(v, u)
            # If subtree of v contains a terminal, and current node will eventually contain a terminal,
            # then we need the edge (u,v)
            if contains_terminal[v]:
                total_cost += w
                contains_terminal[u] = True
                
    dfs_steiner(start, -1)
    print(total_cost)

if __name__ == "__main__":
    main()
