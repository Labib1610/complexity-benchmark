```python
import sys

# The problem states m = n - 1 and the graph is connected, 
# meaning the graph is a tree.
# We need to find the minimum cost to connect all t high-priority rooms.
# In a tree, the minimum set of edges connecting a subset of nodes is the 
# union of all paths between these nodes. This is equivalent to the 
# Steiner Tree problem, but since the graph is a tree, the Steiner Tree 
# is simply the minimal subtree containing all high-priority nodes.
# This subtree consists of all edges that lie on the path between at least 
# two high-priority nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    high_priority = set()
    idx = 3
    for _ in range(t):
        high_priority.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that are part of the minimal subtree.
    # An edge (u, v) with weight w is in the subtree if and only if 
    # the subtree formed by removing (u, v) has high-priority nodes 
    # in both components.
    
    # Count high-priority nodes in each subtree using DFS
    subtree_hp_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in high_priority:
            subtree_hp_count[u] += 1
        if parent[u] != 0:
            subtree_hp_count[parent[u]] += subtree_hp_count[u]
            # If this edge separates the tree into two parts both containing 
            # at least one high-priority node, it must be in the minimal subtree.
            if 0 < subtree_hp_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
