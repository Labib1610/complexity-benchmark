```python
import sys

# The problem states that the network is a tree (n rooms, n-1 segments, fully connected).
# To connect a subset of nodes in a tree, the minimum cost is the sum of the weights
# of all edges that lie on the unique path between any two high-priority nodes.
# An edge belongs to this set if and only if the subtree formed by removing that edge
# contains at least one high-priority node on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_rooms = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use iterative DFS to avoid recursion depth issues
    # We need to count how many priority nodes are in each subtree
    subtree_priority_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in priority_rooms:
            subtree_priority_count[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If this subtree contains some but not all priority nodes,
            # the edge to the parent must be included.
            if 0 < subtree_priority_count[u] < t:
                total_cost += edge_to_parent_weight[u]
            subtree_priority_count[p] += subtree_priority_count[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
