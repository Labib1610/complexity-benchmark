```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    R = int(next(it))
    S = int(next(it))  # = R-1
    P = int(next(it))
    specials = set(int(next(it)) for _ in range(P))
    adj = [[] for _ in range(R + 1)]
    edges = []
    for _ in range(S):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    if P <= 1:
        print(0)
        return

    # root the tree at 1 (any node)
    parent = [0] * (R + 1)
    pw = [0] * (R + 1)          # weight to parent
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        node = stack.pop()
        order.append(node)
        for nb, w in adj[node]:
            if nb == parent[node]:
                continue
            parent[nb] = node
            pw[nb] = w
            stack.append(nb)

    # count specials in subtree
    cnt = [0] * (R + 1)
    for v in specials:
        cnt[v] = 1

    total = 0
    # process in reverse order (post‑order)
    for node in reversed(order):
        if node == 1:
            continue
        sub = cnt[node]
        if 0 < sub < P:
            total += pw[node]
        cnt[parent[node]] += sub

    print(total)

if __name__ == "__main__":
    solve()
```
