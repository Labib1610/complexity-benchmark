```python
import sys

# The problem states that the graph is a tree (R nodes, R-1 links, connected).
# We need to find the minimum cost to connect a subset of P nodes.
# This is the Steiner Tree problem on a tree, which is simply the sum of weights
# of all edges that lie on the unique paths between the P nodes.
# An edge is part of the Steiner Tree if and only if the subtree formed by 
# removing that edge contains at least one of the P nodes on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    injected_nodes = set(map(int, input_data[3:3+P]))
    
    adj = [[] for _ in range(R + 1)]
    idx = 3 + P
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # Use DFS to count how many injected nodes are in each subtree
    # Since R is up to 200,000, use iterative DFS to avoid recursion depth issues
    count = [0] * (R + 1)
    parent = [0] * (R + 1)
    edge_weight = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for node in injected_nodes:
        count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(R - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if count[u] > 0 and count[u] < P:
            total_cost += edge_weight[u]
        count[p] += count[u]
        
    print(total_cost)

solve()
```
