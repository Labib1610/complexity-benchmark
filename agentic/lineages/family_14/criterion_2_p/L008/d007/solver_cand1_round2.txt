```python
import sys

# The problem states the network is a tree (R modules, R-1 linkages, unique path).
# We need to select a connected subgraph (a subtree) that contains all P priority modules.
# To minimize the total synchronization overhead (sum of weights of edges in the subtree),
# we must include all edges that lie on the paths between any two priority modules.
# This is equivalent to finding the minimal Steiner Tree in a tree, which is simply
# the union of all paths between the priority modules.
# The edges in this minimal subtree are exactly those edges that have at least one
# priority module in one of the two components formed by removing the edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    priority_modules = set()
    idx = 3
    for _ in range(P):
        priority_modules.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    edges = []
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_size[u] will store the number of priority modules in the subtree rooted at u
    subtree_priority_count = [0] * (R + 1)
    
    # DFS to count priority modules in each subtree
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in priority_modules:
            subtree_priority_count[u] += 1
        if parent[u] != 0:
            subtree_priority_count[parent[u]] += subtree_priority_count[u]
            
    total_overhead = 0
    for u, v, w in edges:
        # Edge connects u and v. One is parent of other.
        child = u if parent[v] == u else v
        count = subtree_priority_count[child]
        # If the edge separates the tree such that both sides have at least one priority module,
        # the edge must be part of the minimal subtree.
        if count > 0 and count < P:
            total_overhead += w
            
    print(total_overhead)

solve()
```
