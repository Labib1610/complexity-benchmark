```python
import sys

# The problem states: nodes R, links S.
# Structural invariant: R - S = 1.
# This means the graph is a tree.
# We need to connect a set of P nodes to a common meeting node.
# In a tree, the set of edges connecting a set of nodes P to a meeting node M
# is the union of paths from each node in P to M.
# The union of paths from nodes in P to a meeting node M is the minimal subtree
# containing all nodes in P and M.
# The total weight of this subtree is the sum of weights of all edges in the
# minimal subtree containing all nodes in P, plus the path from that subtree to M.
# To minimize this, M should be a node already in the minimal subtree containing P.
# Thus, the answer is simply the sum of weights of all edges that lie on at least
# one path between any two nodes in P.
# An edge lies on a path between two nodes in P if and only if removing that edge
# splits the tree into two components, both of which contain at least one node from P.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    injected = set(map(int, input_data[3:3+P]))
    
    adj = [[] for _ in range(R + 1)]
    idx = 3 + P
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # Count how many injected nodes are in the subtree of each node
    subtree_injected = [0] * (R + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (R + 1)
    edge_to_parent_w = [0] * (R + 1)
    order = []
    stack = [1]
    visited = [False] * (R + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    for node in injected:
        subtree_injected[node] = 1
        
    total_weight = 0
    for u in reversed(order):
        if u == 1:
            continue
        if subtree_injected[u] > 0 and subtree_injected[u] < P:
            total_weight += edge_to_parent_w[u]
        subtree_injected[parent[u]] += subtree_injected[u]
        
    print(total_weight)

solve()
```
