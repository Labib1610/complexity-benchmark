```python
import sys

# The problem states that the graph is a tree (R nodes, R-1 links, connected).
# We need to find the minimum cost to form a connected subgraph containing all P injected nodes.
# In a tree, the unique connected subgraph containing a set of nodes is the minimal subtree
# spanning those nodes. This subtree consists of all edges that lie on the simple path
# between any two injected nodes.
# Therefore, the total cost is simply the sum of the weights of all edges that lie on
# at least one path between two injected nodes.
# An edge (u, v) with weight w lies on a path between two injected nodes if and only if
# the removal of that edge splits the tree into two components, each containing at least one injected node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    injected = set()
    idx = 3
    for _ in range(P):
        injected.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(R + 1)]
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # Use DFS to count how many injected nodes are in each subtree
    # An edge (u, v) where v is child of u is part of the minimal subtree
    # if the subtree at v contains > 0 and < P injected nodes.
    
    injected_count = [0] * (R + 1)
    for node in injected:
        injected_count[node] = 1
        
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(1, 0)]
    order = []
    parent = [0] * (R + 1)
    edge_to_parent_weight = [0] * (R + 1)
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        parent[u] = p
        for v, w in adj[u]:
            if v != p:
                edge_to_parent_weight[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        if parent[u] != 0:
            injected_count[parent[u]] += injected_count[u]
            if 0 < injected_count[u] < P:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
