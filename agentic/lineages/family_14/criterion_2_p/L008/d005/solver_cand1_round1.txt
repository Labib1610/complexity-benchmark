```python
import sys

# The problem asks for the minimum cost to connect all target indices in a graph
# where each edge (u, v) with weight w represents the vector e_u - e_v.
# The span of a set of vectors {e_u - e_v} contains e_i - e_j if and only if
# i and j are in the same connected component in the graph formed by the chosen vectors.
# We need all target indices to be in the same connected component.
# This is equivalent to finding the Minimum Spanning Tree of the target indices
# in the graph where edges are the given vectors.
# Since we have R nodes and R-1 edges forming a tree (given as a basis),
# we can use Kruskal's or simply find the unique path between all target nodes.
# The union of all paths between target nodes in the tree forms a subtree.
# The cost is the sum of weights of all edges in this subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    S = int(input_data[1])
    P = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+P]]
    
    adj = [[] for _ in range(R + 1)]
    idx = 3 + P
    for _ in range(S):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # Root the tree at targets[0]
    parent = [0] * (R + 1)
    edge_weight = [0] * (R + 1)
    depth = [0] * (R + 1)
    
    stack = [(targets[0], 0, 0, 0)]
    visited = [False] * (R + 1)
    visited[targets[0]] = True
    
    while stack:
        u, p, w, d = stack.pop()
        parent[u] = p
        edge_weight[u] = w
        depth[u] = d
        for v, weight in adj[u]:
            if not visited[v]:
                visited[v] = True
                stack.append((v, u, weight, d + 1))
                
    # Mark all edges on the paths between targets
    in_subtree = [False] * (R + 1)
    for t in targets:
        curr = t
        while curr != 0 and not in_subtree[curr]:
            in_subtree[curr] = True
            curr = parent[curr]
            
    total_cost = 0
    for i in range(1, R + 1):
        if in_subtree[i] and parent[i] != 0:
            total_cost += edge_weight[i]
            
    print(total_cost)

solve()
```
