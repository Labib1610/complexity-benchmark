```python
import sys

# The problem asks for the minimum cost to ensure that for any two target indices a and b,
# (A^n)_{a,b} > 0.
# A is the identity matrix plus a subset of the given m updates.
# The updates form a tree structure (since m = n-1 and the graph is connected).
# (A^n)_{a,b} > 0 if and only if there is a path between a and b in the graph
# formed by the selected edges.
# Since we need (A^n)_{a,b} > 0 for all pairs of target indices, all target indices
# must belong to the same connected component in the graph formed by the selected edges.
# This is equivalent to finding the minimum cost to connect all target indices.
# This is the Steiner Tree problem on a tree, which is simply the sum of weights
# of all edges on the unique paths between the target nodes in the original tree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the minimal subtree containing all target nodes.
    # This is the union of all paths between target nodes.
    # An edge is in this subtree if and only if removing it splits the tree
    # into two components, both of which contain at least one target node.
    
    subtree_size = [0] * (n + 1)
    target_count = [0] * (n + 1)
    
    for node in targets:
        target_count[node] = 1
        
    # Use iterative DFS to avoid recursion depth issues
    order = []
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    for u in reversed(order):
        if u == 1: continue
        if target_count[u] > 0 and target_count[u] < t:
            total_cost += edge_to_parent_weight[u]
        target_count[parent[u]] += target_count[u]
        
    print(total_cost)

solve()
```
