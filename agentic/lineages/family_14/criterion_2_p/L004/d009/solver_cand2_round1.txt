```python
import sys

# The problem states that m = n - 1 and the graph is fully connected.
# This means the given operators form a tree.
# We need to find the minimum cost to connect a subset of t target nodes.
# In a tree, there is exactly one unique path between any two nodes.
# To connect a set of target nodes, we must include all edges that lie on the
# unique paths between any pair of target nodes.
# This is equivalent to finding the minimal subtree that contains all target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find all edges that are part of the path between any two target nodes.
    # An edge (u, v) with weight w is part of the minimal subtree if and only if
    # the subtree rooted at v (when the edge is cut) contains at least one target node
    # AND the rest of the tree contains at least one target node.
    
    # Let's root the tree at an arbitrary target node (e.g., the first one).
    root = next(iter(targets))
    
    # count_targets[u] will store the number of target nodes in the subtree of u
    count_targets = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            count_targets[u] += 1
        if parent[u] != 0:
            count_targets[parent[u]] += count_targets[u]
            
    # An edge connecting u to parent[u] is needed if 0 < count_targets[u] < t
    ans = 0
    for i in range(1, n + 1):
        if i != root:
            if 0 < count_targets[i] < t:
                ans += edge_weight[i]
                
    print(ans)

solve()
```
