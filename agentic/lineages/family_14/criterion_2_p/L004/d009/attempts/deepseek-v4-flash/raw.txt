We can solve this by treating the network as a tree (since m = n - 1) and finding the minimal subtree that connects all priority nodes.  

We can do this by:
1. Marking priority nodes.
2. Doing a DFS from any priority node to count how many priority nodes are in each subtree.
3. An edge is needed if both sides contain at least one priority node.
4. Sum the weights of all such edges.

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    priority = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # DFS to count priority nodes in subtree
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    order = []
    stack = [next(iter(priority))]
    parent[stack[0]] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                weight_to_parent[v] = w
                stack.append(v)
    
    count = [0] * (n + 1)
    for u in priority:
        count[u] = 1
    
    total_cost = 0
    for u in reversed(order):
        if u != next(iter(priority)):
            if count[u] > 0 and (t - count[u]) > 0:
                total_cost += weight_to_parent[u]
            if parent[u] != -1:
                count[parent[u]] += count[u]
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
