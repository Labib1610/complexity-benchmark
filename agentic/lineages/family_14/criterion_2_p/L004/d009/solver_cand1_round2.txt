```python
import sys

# The problem states that the given m = n - 1 corridors form a tree.
# We need to find the minimum cost to connect a subset of t priority facilities.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of nodes in a tree, we must include all edges that lie on
# the paths between any two priority nodes.
# This is equivalent to the union of all paths between the priority nodes.
# An edge is part of this union if and only if the subtree formed by removing
# that edge contains at least one priority node on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_nodes = set()
    idx = 3
    for _ in range(t):
        priority_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many priority nodes are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    subtree_priority_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in priority_nodes:
            subtree_priority_count[node] += 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        subtree_priority_count[p] += subtree_priority_count[u]
        
        # If the edge (u, p) separates the tree such that both sides have
        # at least one priority node, it must be included.
        count = subtree_priority_count[u]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

solve()
```
