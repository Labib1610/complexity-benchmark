```python
import sys

# The problem states that the potential welds form a tree (N components, N-1 welds, connected).
# We need to select a subset of these welds such that all load-bearing components are connected,
# there are no cycles, and there is a unique path between any two load-bearing components.
# This is equivalent to finding the minimal subtree that contains all load-bearing components.
# In a tree, the minimal subtree containing a set of nodes is the union of all paths
# between every pair of load-bearing nodes.
# This is equivalent to the set of all edges that lie on a path between at least two
# load-bearing nodes.
# An edge (u, v) is part of this subtree if and only if the two components formed by
# removing (u, v) both contain at least one load-bearing node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    n = int(input_data[idx])
    m = int(input_data[idx+1])
    k = int(input_data[idx+2])
    idx += 3
    
    load_bearing = set()
    for _ in range(k):
        load_bearing.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        
    if k <= 1:
        print(0)
        return

    # subtree_load[u] stores the number of load-bearing nodes in the subtree rooted at u
    subtree_load = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    # DFS to compute subtree counts
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in load_bearing:
            subtree_load[u] += 1
        if parent[u] != 0:
            subtree_load[parent[u]] += subtree_load[u]
            
    total_weight = 0
    for i in range(2, n + 1):
        # Edge (i, parent[i]) is part of the minimal subtree if both sides
        # have at least one load-bearing node.
        # One side has subtree_load[i], the other has k - subtree_load[i].
        if subtree_load[i] > 0 and (k - subtree_load[i]) > 0:
            total_weight += parent_edge_weight[i]
            
    print(total_weight)

solve()
```
