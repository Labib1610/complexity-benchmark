```python
import sys

# The problem asks for the minimum cost to connect a set of target nodes in a tree.
# Since the input is a tree (n nodes, n-1 edges), the path between any two nodes is unique.
# To connect a set of target nodes, we need to find the smallest subtree that contains all of them.
# The cost is the sum of weights of all edges in this minimal subtree.
# This is equivalent to finding the union of all paths between the target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Root the tree at targets[0]
    root = targets[0]
    parent = [0] * (n + 1)
    edge_to_parent = [0] * (n + 1)
    depth = [0] * (n + 1)
    
    # BFS to build tree structure
    queue = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    order = []
    while queue:
        u = queue.pop(0)
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent[v] = w
                depth[v] = depth[u] + 1
                queue.append(v)
                
    # Mark nodes that are part of the minimal subtree
    # A node is in the subtree if it is a target or has a descendant that is a target
    is_target = [False] * (n + 1)
    for node in targets:
        is_target[node] = True
        
    in_subtree = [False] * (n + 1)
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if is_target[u]:
            in_subtree[u] = True
        if in_subtree[u] and parent[u] != 0:
            in_subtree[parent[u]] = True
            
    total_cost = 0
    for i in range(1, n + 1):
        if in_subtree[i] and parent[i] != 0:
            total_cost += edge_to_parent[i]
            
    print(total_cost)

solve()
```
