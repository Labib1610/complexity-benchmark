```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the minimum cost to connect all target nodes.
    # This is equivalent to finding the Steiner Tree for the target nodes.
    # Since the graph is a tree (n nodes, n-1 edges), the Steiner Tree is 
    # the unique path in the tree connecting all target nodes.
    # The cost is the sum of weights of all edges on the paths between targets.
    
    # Root the tree at the first target node
    root = next(iter(targets))
    
    # parent_edge_weight[u] stores the weight of the edge between u and parent[u]
    parent_edge_weight = [0] * (n + 1)
    has_target_in_subtree = [False] * (n + 1)
    
    for target in targets:
        has_target_in_subtree[target] = True
        
    stack = [(root, -1)]
    order = []
    while stack:
        u, p = stack.pop()
        order.append((u, p))
        for v, w in adj[u]:
            if v != p:
                parent_edge_weight[v] = w
                stack.append((v, u))
                
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u, p in reversed(order):
        if p != -1:
            if has_target_in_subtree[u]:
                total_cost += parent_edge_weight[u]
                has_target_in_subtree[p] = True
                
    print(total_cost)

solve()
```
