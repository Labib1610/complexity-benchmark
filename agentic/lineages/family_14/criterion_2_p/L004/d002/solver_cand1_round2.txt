```python
import sys

# The problem states the network is a tree (n nodes, n-1 edges, connected).
# We need to find the minimum weight subset of edges that connects all t flagged units.
# This is equivalent to finding the Steiner Tree in a tree, which is simply the sum
# of weights of all edges that lie on the unique path between any two flagged units.
# An edge belongs to the Steiner Tree if and only if the subtree formed by removing
# that edge contains at least one flagged unit on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    flagged = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        flagged[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count flagged nodes in subtree using DFS
    # Since n is up to 200,000, use iterative DFS to avoid recursion depth issues
    subtree_flagged = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if flagged[u]:
            subtree_flagged[u] += 1
        if parent[u] != -1:
            subtree_flagged[parent[u]] += subtree_flagged[u]
            
    total_energy = 0
    for i in range(2, n + 1):
        # If the edge to parent separates the tree such that both sides have flagged nodes
        count = subtree_flagged[i]
        if count > 0 and count < t:
            total_energy += edge_to_parent_weight[i]
            
    print(total_energy)

if __name__ == '__main__':
    solve()
```
