```python
import sys

# The problem asks for the minimum cost to connect all t critical propositions
# in a graph where edges are axioms with weights.
# Since m = n - 1 and the graph is a tree (or forest), we are looking for the
# minimum weight of the edges that form the unique path between all critical nodes.
# This is equivalent to finding the Steiner Tree in a tree, which is simply the
# sum of weights of all edges that lie on the paths between any two critical nodes.
# In a tree, an edge is part of the path between two critical nodes if and only if
# removing that edge splits the tree into two components, each containing at least
# one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many critical nodes are in the subtree of each node
    # We root the tree at node 1
    subtree_critical_count = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for node in order:
        if node in critical_nodes:
            subtree_critical_count[node] += 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        # If this subtree contains some but not all critical nodes,
        # the edge to its parent must be included.
        count = subtree_critical_count[u]
        if 0 < count < t:
            total_cost += edge_weight_to_parent[u]
        
        # Propagate count to parent
        subtree_critical_count[parent[u]] += count
        
    print(total_cost)

solve()
```
