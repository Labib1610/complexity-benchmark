```python
import sys

# The problem asks for the minimum cost to connect a set of target qubits
# in a tree structure. Since the links form a tree (n nodes, n-1 edges),
# the path between any two nodes is unique.
# To ensure all t target qubits are correlated, we must activate all links
# on the unique paths connecting all target qubits.
# This is equivalent to finding the minimal subtree that contains all target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the union of all paths between target nodes.
    # A node is part of the minimal subtree if it lies on a path between two targets.
    # This is equivalent to saying a node is in the subtree if its subtree
    # (when rooted arbitrarily) contains at least one target and at least one
    # target is outside its subtree.
    
    # Root the tree at the first target
    root = next(iter(targets))
    
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    has_target = [False] * (n + 1)
    for node in targets:
        has_target[node] = True
        
    # Traverse in reverse topological order to propagate target existence
    total_cost = 0
    for i in range(n - 1, 0, -1):
        u = order[i]
        if has_target[u]:
            total_cost += edge_to_parent_weight[u]
            has_target[parent[u]] = True
            
    print(total_cost)

solve()
```
