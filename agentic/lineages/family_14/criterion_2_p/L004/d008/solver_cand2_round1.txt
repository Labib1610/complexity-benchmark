```python
import sys

# The problem asks for the minimum cost to connect all critical facilities
# such that they form a connected component in the graph of built pipelines.
# If we build a set of pipelines, the condition that any zero-sum distribution
# among critical facilities can be balanced is equivalent to saying that all
# critical facilities must belong to the same connected component in the
# subgraph formed by the chosen pipelines.
# Since we are given n-1 pipelines (a tree structure), the unique path between
# any two nodes in the tree must be included if we want to connect them.
# To connect all t critical facilities, we must include all edges that lie
# on the unique paths between any two critical facilities.
# This is equivalent to finding the minimal subtree that contains all t
# critical nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the sum of weights of all edges that are part of the
    # minimal subtree connecting the critical nodes.
    # An edge is in the minimal subtree if and only if both components
    # formed by removing the edge contain at least one critical node.
    
    # Root the tree at an arbitrary critical node
    root = next(iter(critical_nodes))
    
    subtree_critical_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(root, -1, 0)]
    order = []
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    
    dfs_stack = [(root, -1)]
    while dfs_stack:
        u, p = dfs_stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != p:
                parent[v] = u
                edge_to_parent_weight[v] = w
                dfs_stack.append((v, u))
                
    for node in order:
        if node in critical_nodes:
            subtree_critical_count[node] = 1
            
    for node in reversed(order):
        if node == root:
            continue
        p = parent[node]
        if subtree_critical_count[node] > 0:
            # If this subtree contains at least one critical node,
            # and the rest of the tree contains at least one (t - count > 0),
            # then the edge to parent must be included.
            if subtree_critical_count[node] < t:
                total_cost += edge_to_parent_weight[node]
            subtree_critical_count[p] += subtree_critical_count[node]
            
    print(total_cost)

solve()
```
