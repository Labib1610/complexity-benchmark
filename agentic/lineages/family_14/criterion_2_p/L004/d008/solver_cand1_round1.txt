```python
import sys

# The problem asks for the minimum cost to connect a set of t critical nodes
# in a tree (since m = n-1 and the rules form a tree).
# This is equivalent to finding the Steiner Tree for the set of t nodes.
# In a tree, the Steiner Tree is the unique minimal subtree containing all
# the critical nodes. The cost is the sum of weights of all edges in this
# unique subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the Steiner Tree in a tree:
    # 1. Root the tree arbitrarily (e.g., at node 1).
    # 2. For each node, determine if its subtree contains any critical nodes.
    # 3. An edge (u, v) where v is a child of u is in the Steiner Tree if and only if
    #    the subtree at v contains at least one critical node AND the rest of the
    #    tree (outside the subtree at v) contains at least one critical node.
    
    has_critical = [False] * (n + 1)
    for node in critical_nodes:
        has_critical[node] = True
        
    subtree_has_critical = [False] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if has_critical[u]:
            subtree_has_critical[u] = True
        for v, w in adj[u]:
            if v != parent[u]:
                if subtree_has_critical[v]:
                    subtree_has_critical[u] = True
                    
    total_cost = 0
    for i in range(2, n + 1):
        # Edge is (i, parent[i])
        # It is in the Steiner Tree if subtree at i has critical nodes
        # AND the rest of the tree has critical nodes.
        if subtree_has_critical[i] and (subtree_has_critical[i] != subtree_has_critical[1]):
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
