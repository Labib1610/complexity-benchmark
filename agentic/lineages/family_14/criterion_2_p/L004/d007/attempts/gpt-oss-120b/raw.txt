import sys
sys.setrecursionlimit(1_000_000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))  # equals n-1
    t = int(next(it))
    terminals = set(int(next(it)) - 1 for _ in range(t))  # zero‑based
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    if t <= 1:
        print(0)
        return

    # root the tree at 0 (any node)
    parent = [-1] * n
    pw = [0] * n          # weight to parent
    order = []
    stack = [0]
    parent[0] = 0
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == -1:
                parent[v] = u
                pw[v] = w
                stack.append(v)

    # post‑order count terminals in subtree
    cnt = [0] * n
    for u in reversed(order):
        cnt[u] = 1 if u in terminals else 0
        for v, _ in adj[u]:
            if parent[v] == u:   # v is child
                cnt[u] += cnt[v]

    total = 0
    for v in range(1, n):
        # edge between v and parent[v]
        if 0 < cnt[v] < t:
            total += pw[v]

    print(total)

if __name__ == "__main__":
    solve()
