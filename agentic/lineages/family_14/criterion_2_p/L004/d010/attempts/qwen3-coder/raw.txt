```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1000000)
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    priority = set()
    idx = 3
    for _ in range(t):
        priority.add(int(data[idx]))
        idx += 1
    
    edges = []
    graph = defaultdict(list)
    
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        idx += 3
        edges.append((u, v, w))
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Since the graph is a tree (n nodes, n-1 edges, connected), we can use DFS/BFS
    # to find the paths between priority nodes.
    # However, since we need to find all edges that lie on paths between any two priority nodes,
    # we can do the following:
    # 1. Root the tree at any node.
    # 2. For each node, compute its parent and depth.
    # 3. Precompute LCA (Lowest Common Ancestor) to find paths.
    # 4. Count how many times each edge is used in paths between priority nodes.
    
    # But there's a simpler way:
    # Since it's a tree, there's exactly one path between any two nodes.
    # We want to find the union of all paths between pairs of priority nodes.
    # This is equivalent to finding the minimal subtree that connects all priority nodes.
    # This is called the "Steiner tree" problem on trees, and the cost is the sum of edges
    # in the minimal subtree connecting all priority nodes.
    
    # Efficient approach:
    # 1. Root the tree at an arbitrary node (say node 1).
    # 2. Compute parent and depth for each node via BFS/DFS.
    # 3. For each node, compute the lowest priority ancestor in its subtree.
    # 4. An edge (u,v) where u is parent of v is needed iff the subtree of v contains
    #    at least one priority node and at least one priority node is outside the subtree.
    
    # Even simpler:
    # Run a multi-source BFS from all priority nodes.
    # The edges in the resulting BFS tree that connect different components during BFS
    # will form the minimal connecting structure.
    # But again, since it's a tree, the minimal subgraph connecting all priority nodes
    # is unique.
    
    # Best approach:
    # Use the fact that in a tree, the minimal subgraph connecting a set of nodes
    # is obtained by taking all edges on paths between pairs.
    # But since we don't know all pairs, we can:
    # 1. Pick any priority node as root.
    # 2. Do BFS from that root to get parent relationships.
    # 3. Mark all nodes that are ancestors of at least one priority node.
    # 4. Count edges where both endpoints are marked.
    
    # Actually, let's implement the correct algorithm:
    # Algorithm:
    # 1. Root the tree at some node.
    # 2. Compute for each node whether its subtree contains a priority node.
    # 3. An edge is needed if it separates priority nodes, i.e., 
    #    for edge (u,v) where u is parent of v, if subtree of v has >=1 priority nodes
    #    and the rest of the tree has >=1 priority nodes.
    
    # Let's implement this.
    
    # Step 1: Root the tree at node 1
    root = 1
    # Build adjacency list with edge indices
    adj = defaultdict(list)  # node -> [(neighbor, weight, edge_index)]
    for i in range(m):
        u, v, w = edges[i]
        adj[u].append((v, w, i))
        adj[v].append((u, w, i))
        
    # Step 2: BFS to build parent relationships and depths
    parent = [-1] * (n + 1)
    depth = [0] * (n + 1)
    visited = [False] * (n + 1)
    queue = deque([root])
    visited[root] = True
    parent[root] = -1
    
    bfs_order = []
    
    while queue:
        u = queue.popleft()
        bfs_order.append(u)
        for v, w, idx_edge in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                depth[v] = depth[u] + 1
                queue.append(v)
                
    # Step 3: Count priority nodes in each subtree
    # Process nodes in reverse BFS order (post-order)
    has_priority = [0] * (n + 1)
    
    # Initialize leaves
    for node in priority:
        has_priority[node] = 1
        
    # Propagate up the tree
    for u in reversed(bfs_order):
        if parent[u] != -1:
            has_priority[parent[u]] += has_priority[u]
            
    # Total priority nodes
    total_priority = len(priority)
    
    # Step 4: Sum weights of edges that are needed
    result = 0
    for u in range(1, n + 1):
        if parent[u] != -1:
            p = parent[u]
            # Edge between u and p
            # Check if this edge is needed
            # It's needed if:
            #   - subtree of u has at least 1 priority AND 
            #   - rest of tree (including p's other subtrees) has at least 1 priority
            count_in_subtree = has_priority[u]
            count_outside_subtree = total_priority - count_in_subtree
            
            if count_in_subtree > 0 and count_outside_subtree > 0:
                # Find the edge weight
                for v, w, idx_edge in adj[u]:
                    if v == p:
                        result += w
                        break
                        
    print(result)

if __name__ == "__main__":
    main()
```
