```python
import sys

# The problem asks for the minimum cost to ensure that for all priority indices a, b,
# e_a - e_b is in the span of the purchased vectors.
# The vectors are of the form e_u - e_v. This is equivalent to an edge (u, v) in a graph.
# The span of a set of vectors {e_u - e_v} is the set of all vectors sum(c_i * e_i)
# such that sum(c_i) = 0, restricted to the connected components of the graph formed
# by the edges.
# Specifically, e_a - e_b is in the span if and only if a and b are in the same
# connected component.
# Since we are given m = n - 1 edges and the span has dimension n - 1, the graph
# is a tree.
# We need all t priority indices to be in the same connected component.
# In a tree, there is a unique path between any two nodes.
# If we remove an edge, the tree splits into two components.
# We need to find the minimum cost to keep all priority nodes in the same component.
# This is equivalent to finding the minimum cost to cut edges such that all priority
# nodes end up in the same component, but wait: we are *purchasing* a subset.
# Actually, the problem says we have m = n - 1 edges forming a tree.
# We want to select a subset of edges such that all priority nodes are connected.
# But since the original graph is a tree, there is only one way to connect any
# two nodes. To connect all priority nodes, we must include all edges on the
# unique paths between them.
# The union of all paths between priority nodes forms a subtree.
# We must purchase all edges in this subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_nodes = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Mark priority nodes
    is_priority = [False] * (n + 1)
    for p in priority_nodes:
        is_priority[p] = True
        
    # Count how many priority nodes are in the subtree rooted at u
    subtree_priority_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    
    order = []
    stack = [priority_nodes[0]]
    visited = [False] * (n + 1)
    visited[priority_nodes[0]] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        subtree_priority_count[u] += (1 if is_priority[u] else 0)
        if parent[u] != 0:
            subtree_priority_count[parent[u]] += subtree_priority_count[u]
            
    total_cost = 0
    for i in range(1, n + 1):
        # If an edge connects a component with some priority nodes to the rest,
        # and both sides have at least one priority node, the edge must be included.
        count = subtree_priority_count[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

solve()
```
