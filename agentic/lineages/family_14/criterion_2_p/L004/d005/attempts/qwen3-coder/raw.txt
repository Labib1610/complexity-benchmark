import sys
from collections import defaultdict

class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        px, py = self.find(x), self.find(y)
        if px == py:
            return
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1

        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((w, u, v))

    # Sort edges by weight
    edges.sort()
    
    # First, connect all critical nodes with MST
    uf = UnionFind(n)
    cost = 0
    critical_set = set(critical)
    connected_critical = [critical[0]]
    remaining_critical = set(critical[1:])
    
    # We need to connect all critical nodes
    while remaining_critical:
        found = False
        for i, (w, u, v) in enumerate(edges):
            if uf.find(u) == uf.find(v):
                continue

            root_u = uf.find(u)
            root_v = uf.find(v)
            
            u_is_critical = u in critical_set
            v_is_critical = v in critical_set
            u_connected = root_u in [uf.find(c) for c in connected_critical]
            v_connected = root_v in [uf.find(c) for c in connected_critical]

            if (u_is_critical and u_connected and v_is_critical and not v_connected and root_v in [uf.find(c) for c in remaining_critical]):
                uf.union(u, v)

                connected_critical.append(u)
                remaining_critical.remove(u)
                found = True
                break

                # Connect a critical component to a non-critical node
                uf.union(u, v)
                cost += w
                found = True
                break
            elif (not u_is_critical and u_connected) or (not v_is_critical and v_connected):
                # Connect through already connected non-critical nodes to critical nodes
                uf.union(u, v)
                cost += w
                if u_is_critical and u in remaining_critical:
                    connected_critical.append(u)
                    remaining_critical.remove(u)
                elif v_is_critical and v in remaining_critical:

                    remaining_critical.remove(v)

            for i, (w, u, v) in enumerate(edges):

                    uf.union(u, v)

                            new_connected.append(c)

                        connected_critical.append(c)

    print(cost)
