import sys

# The problem states that the m = n-1 axioms form a tree structure (since they are 
# minimally complete and connect n nodes).
# We need to ensure that t critical propositions are in the same connected component.
# This is equivalent to finding the minimum weight of edges in the tree that 
# connect all t critical nodes.
# This is the Steiner Tree problem on a tree, which is simply the sum of weights 
# of all edges on the unique paths between the critical nodes.
# In a tree, the union of paths between a set of nodes is the set of edges 
# that lie on the path between at least one pair of these nodes.
# An edge belongs to this set if and only if the two components formed by 
# removing the edge both contain at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many critical nodes are in each subtree
    subtree_critical_count = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for node in order:
        if node in critical_nodes:
            subtree_critical_count[node] += 1
            
    # Process in reverse topological order to propagate counts
    total_cost = 0
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            p = parent[u]
            subtree_critical_count[p] += subtree_critical_count[u]
            
            # Find weight of edge (u, p)
            # Since it's a tree, we can store it during the first pass
            pass

    # Re-calculate total cost using the counts
    # An edge (u, parent[u]) is included if 0 < subtree_critical_count[u] < t
    edge_weights = {}
    for u, v, w in edges:
        edge_weights[tuple(sorted((u, v)))] = w
        
    total_cost = 0
    for i in range(1, n + 1):
        if parent[i] != 0:
            count = subtree_critical_count[i]
            if count > 0 and count < t:
                total_cost += edge_weights[tuple(sorted((i, parent[i])))]
                
    print(total_cost)

solve()
