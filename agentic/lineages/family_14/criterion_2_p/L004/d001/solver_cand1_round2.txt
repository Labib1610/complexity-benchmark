```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    flagged = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        flagged[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_flagged[u] stores the number of flagged nodes in the subtree of u
    subtree_flagged = [0] * (n + 1)
    total_energy = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order to compute subtree counts
    for u in reversed(order):
        if flagged[u]:
            subtree_flagged[u] += 1
        
        for v, w in adj[u]:
            if v != parent[u]:
                subtree_flagged[u] += subtree_flagged[v]
                # If this edge leads to a subtree containing some but not all flagged nodes,
                # it is essential.
                if 0 < subtree_flagged[v] < t:
                    total_energy += w
                    
    print(total_energy)

if __name__ == '__main__':
    solve()
```
