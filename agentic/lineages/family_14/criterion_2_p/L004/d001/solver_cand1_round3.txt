import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    flagged = set()
    idx = 3
    for _ in range(t):
        flagged.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the sum of weights of all edges that lie on the path
    # between any two flagged nodes. An edge is part of the minimal Steiner tree
    # if and only if the subtree formed by removing that edge contains at least
    # one flagged node on both sides.
    
    is_flagged = [False] * (n + 1)
    for f in flagged:
        is_flagged[f] = True
        
    subtree_flagged_count = [0] * (n + 1)
    total_energy = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order to count flagged nodes in subtrees
    edge_weights = {}
    for u in reversed(order):
        subtree_flagged_count[u] += (1 if is_flagged[u] else 0)
        if parent[u] != -1:
            p = parent[u]
            subtree_flagged_count[p] += subtree_flagged_count[u]
            
    # Calculate total energy
    # Find weight of edge (u, parent[u])
    # We can store weights in a dictionary or map
    weight_map = {}
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if v == parent[u]:
                weight_map[u] = w
            elif u == parent[v]:
                weight_map[v] = w
                
    ans = 0
    for u in range(1, n + 1):
        if u == 1: continue
        count = subtree_flagged_count[u]
        if count > 0 and count < t:
            ans += weight_map[u]
            
    print(ans)

solve()
