```python
import sys

# The problem states m = n - 1 and the grid is connected, 
# meaning the structure is a tree.
# To connect a subset of nodes in a tree, the minimum cost is the sum of 
# weights of all edges that lie on the paths between any two nodes in the subset.
# An edge is part of the minimal Steiner tree if and only if it splits the 
# tree into two components, both of which contain at least one of the t nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    out_of_sync = set()
    idx = 3
    for _ in range(t):
        out_of_sync.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_t[u] stores the number of out-of-sync nodes in the subtree rooted at u
    subtree_t = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in out_of_sync:
            subtree_t[u] = 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        subtree_t[p] += subtree_t[u]
        # If the edge (u, p) separates the tree into two parts, 
        # both containing at least one out-of-sync node, add its weight.
        if 0 < subtree_t[u] < t:
            total_cost += edge_weight[u]
            
    print(total_cost)

solve()
```
