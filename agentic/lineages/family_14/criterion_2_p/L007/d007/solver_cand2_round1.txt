```python
import sys

def solve():
    # Read n, m, t
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_elements = set()
    idx = 3
    for _ in range(t):
        target_elements.add(int(input_data[idx]))
        idx += 1
        
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The problem asks for the minimum cost to connect all target elements
    # in the graph formed by the swap operations.
    # Since m = n - 1 and the graph is connected (it can form any permutation),
    # the graph is a tree.
    # We need to find the smallest subtree that contains all target elements.
    # This is equivalent to the sum of weights of all edges that lie on the
    # unique path between any two target elements.
    
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Root the tree at an arbitrary target element
    root = next(iter(target_elements))
    
    # has_target[u] is true if the subtree at u contains at least one target element
    # We need to sum weights of edges (u, v) where both subtrees contain targets.
    
    total_cost = 0
    
    def dfs(u, p):
        nonlocal total_cost
        contains_target = (u in target_elements)
        
        for v, w in adj[u]:
            if v == p:
                continue
            
            subtree_has_target = dfs(v, u)
            if subtree_has_target:
                total_cost += w
                contains_target = True
        
        return contains_target

    dfs(root, -1)
    print(total_cost)

solve()
```
