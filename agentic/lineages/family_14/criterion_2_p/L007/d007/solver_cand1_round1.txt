```python
import sys

# The problem asks for the minimum cost to select a subset of edges such that
# for every pair of nodes in the target set T, there exists a sequence of 
# swaps that swaps the values at those two nodes while leaving all other 
# nodes unchanged.
#
# A swap of two nodes u and v can be performed if there is a path between them.
# If we have a tree, we can swap any two nodes u and v by performing a sequence
# of swaps along the path between them. Specifically, if the path is 
# v_0, v_1, ..., v_k (where v_0=u, v_k=v), we can swap (v_0, v_1), then 
# (v_1, v_2), ..., (v_{k-1}, v_k), then (v_{k-2}, v_{k-1}), ..., (v_0, v_1).
# This sequence swaps u and v and returns all intermediate nodes to their 
# original values.
#
# The condition "for every pair of distinct variables a, b in T, there must 
# exist a sequence of activated instructions whose net effect is exactly the 
# exchange of the values of a and b" is satisfied if and only if all nodes in 
# T are in the same connected component of the graph formed by the selected 
# instructions.
#
# Since m = n - 1 and the graph is guaranteed to be connected, the graph is a tree.
# We need to find the minimum weight subtree that contains all nodes in T.
# This is the Steiner Tree problem on a tree, which is simply the sum of weights
# of all edges that lie on the unique paths between nodes in T.
# An edge is part of the Steiner Tree if and only if the subtree formed by 
# removing that edge contains at least one node from T in both components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = set()
    idx = 3
    for _ in range(t):
        target_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many target nodes are in each subtree
    subtree_targets = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in target_nodes:
            subtree_targets[u] += 1
            
    total_cost = 0
    for i in range(len(order) - 1, 0, -1):
        u = order[i]
        p = parent[u]
        subtree_targets[p] += subtree_targets[u]
        # If the edge (u, p) separates the tree into two parts, both containing
        # at least one target node, then this edge must be included.
        if subtree_targets[u] > 0 and subtree_targets[u] < t:
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

solve()
```
