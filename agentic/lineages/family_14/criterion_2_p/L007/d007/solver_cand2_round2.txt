```python
import sys

# The problem asks for the minimum cost to generate a subgroup of S_n 
# that contains all transpositions (a, b) for a, b in T.
# A set of transpositions generates the symmetric group on a set of elements 
# if and only if the graph formed by these transpositions is connected.
# Specifically, if we have a set of transpositions, they generate the 
# symmetric group on the connected components of the graph formed by these 
# transpositions.
# To have (a, b) in the generated subgroup for all a, b in T, all elements 
# in T must belong to the same connected component in the graph formed by 
# the selected transpositions.
# Since we are given n-1 transpositions and they form a tree (as m=n-1 
# and they generate S_n), we are looking for the minimum weight subset of 
# edges that connects all nodes in T.
# This is the Steiner Tree problem on a tree, which simplifies to the 
# sum of weights of all edges on the unique paths between nodes in T.
# In a tree, the union of paths between all nodes in T is the minimal 
# subtree containing all nodes in T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = set()
    idx = 3
    for _ in range(t):
        target_nodes.add(int(input_data[idx]))
        idx += 1
        
    if t <= 1:
        print(0)
        return
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the sum of weights of all edges that lie on a path 
    # between any two nodes in T.
    # An edge is part of the Steiner tree if and only if the two components 
    # it separates both contain at least one node from T.
    
    subtree_size = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in target_nodes:
            subtree_size[node] = 1
            
    for node in reversed(order):
        if node == 1:
            continue
        p = parent[node]
        if subtree_size[node] > 0 and subtree_size[node] < t:
            total_cost += edge_weight[node]
        subtree_size[p] += subtree_size[node]
        
    print(total_cost)

solve()
```
