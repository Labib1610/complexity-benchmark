```python
import sys

# The problem asks for the minimum cost to span vectors e_a - e_b for all a, b in T.
# This is equivalent to connecting all nodes in T in a graph where edges are (u, v) with weight w.
# The vectors e_u - e_v form a graph where edges are (u, v).
# The span of a set of edges (u_i, v_i) contains e_a - e_b if and only if there is a path
# between a and b in the graph formed by these edges.
# To connect all nodes in T, we need to select a subset of edges such that all nodes in T
# belong to the same connected component.
# This is equivalent to the Steiner Tree problem, but since the graph is a tree (m = n-1),
# the solution is simply the sum of weights of all edges on the unique paths between
# nodes in T.
# In a tree, the set of edges that connect a subset of vertices T is the union of all
# paths between nodes in T. This is the minimal subtree containing all nodes in T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = set()
    idx = 3
    for _ in range(t):
        target_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the sum of weights of all edges that lie on paths between nodes in T.
    # An edge (u, v) is part of the minimal subtree if and only if the two components
    # formed by removing (u, v) both contain at least one node from T.
    
    subtree_size = [0] * (n + 1)
    total_cost = 0
    
    # Root the tree at an arbitrary node, say 1
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for node in target_nodes:
        subtree_size[node] = 1
        
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        subtree_size[p] += subtree_size[u]
        if subtree_size[u] > 0 and subtree_size[u] < t:
            total_cost += edge_weight_to_parent[u]
            
    print(total_cost)

solve()
```
