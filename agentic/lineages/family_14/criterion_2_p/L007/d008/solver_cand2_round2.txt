```python
import sys

# The problem asks for the minimum cost to connect all nodes in the target set T
# such that any pair in T can be swapped.
# In a graph where edges are row-swap modules, swapping any two nodes in a 
# connected component is possible if and only if the component is 2-connected 
# (or more simply, if we can perform transpositions).
# Actually, the condition "for every pair of distinct rows a, b in T, 
# we can swap them while leaving others in original positions" is equivalent 
# to saying that all nodes in T must belong to the same connected component 
# in the graph formed by the chosen modules.
# Since we have n nodes and n-1 edges forming a tree (guaranteed by the problem),
# the minimum cost to connect a subset of nodes T is the weight of the 
# Steiner Tree for T. In a tree, the Steiner Tree is the unique minimal 
# subtree connecting all nodes in T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = set(map(int, input_data[3:3+t]))
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the Steiner Tree in a tree:
    # 1. Root the tree arbitrarily (e.g., at node 1).
    # 2. A node u is part of the Steiner Tree if the subtree rooted at u 
    #    contains at least one node from T, and there is at least one node 
    #    from T outside the subtree of u (or u itself is in T).
    # 3. Equivalently, an edge (u, v) where v is a child of u is in the 
    #    Steiner Tree if the subtree at v contains at least one node from T 
    #    and the rest of the tree (outside v's subtree) contains at least 
    #    one node from T.
    
    has_target = [False] * (n + 1)
    for node in target_nodes:
        has_target[node] = True
        
    subtree_has_target = [False] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    total_weight = 0
    for u in reversed(order):
        if has_target[u]:
            subtree_has_target[u] = True
        if subtree_has_target[u] and parent[u] != 0:
            # Check if there's a target node outside this subtree
            # Total targets = t. If subtree_count[u] < t, then there's one outside.
            pass
            
    # Simpler: count targets in subtree
    subtree_count = [0] * (n + 1)
    for u in order:
        if has_target[u]:
            subtree_count[u] = 1
            
    for u in reversed(order):
        if parent[u] != 0:
            subtree_count[parent[u]] += subtree_count[u]
            
    ans = 0
    for u in range(2, n + 1):
        # Edge (parent[u], u) is in Steiner Tree if 0 < subtree_count[u] < t
        if 0 < subtree_count[u] < t:
            ans += parent_edge_weight[u]
            
    print(ans)

solve()
```
