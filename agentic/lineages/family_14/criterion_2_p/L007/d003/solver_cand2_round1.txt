import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    rented = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        rented[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the minimal subtree containing all rented nodes.
    # The cost is the sum of weights of edges in this minimal subtree.
    # A node is in the minimal subtree if it lies on a path between two rented nodes.
    # This is equivalent to: a node is in the subtree if its subtree (in a rooted sense)
    # contains at least one rented node AND the rest of the tree contains at least one rented node.
    
    has_rented = [False] * (n + 1)
    
    # Root at an arbitrary rented node
    root = -1
    for i in range(1, n + 1):
        if rented[i]:
            root = i
            break
            
    parent_edge_w = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    parent = [0] * (n + 1)
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_w[v] = w
                stack.append(v)
                
    # has_rented[u] will be true if subtree at u contains a rented node
    for u in reversed(order):
        if rented[u]:
            has_rented[u] = True
        for v, w in adj[u]:
            if v != parent[u]:
                if has_rented[v]:
                    has_rented[u] = True
                    
    total_cost = 0
    # An edge (u, parent[u]) is in the minimal subtree if:
    # 1. The subtree at u contains at least one rented node.
    # 2. The rest of the tree (n - count of rented in subtree at u) contains at least one rented node.
    
    count_rented_in_subtree = [0] * (n + 1)
    for u in reversed(order):
        count_rented_in_subtree[u] += (1 if rented[u] else 0)
        if parent[u] != 0:
            count_rented_in_subtree[parent[u]] += count_rented_in_subtree[u]
            
    for u in range(1, n + 1):
        if u == root:
            continue
        # Edge is (u, parent[u])
        c = count_rented_in_subtree[u]
        if c > 0 and c < t:
            total_cost += parent_edge_w[u]
            
    print(total_cost)

solve()
