```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    is_rented = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        is_rented[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0 and v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                stack.append(v)
                
    max_edge_to_root = [0] * (n + 1)
    total_cost = 0
    
    for u in reversed(order):
        if is_rented[u] or max_edge_to_root[u] > 0:
            total_cost += max_edge_to_root[u]
            p = parent[u]
            if p != -1:
                if max_edge_to_root[u] > max_edge_to_root[p]:
                    max_edge_to_root[p] = max_edge_to_root[u]
                    
    print(total_cost)

if __name__ == '__main__':
    solve()
```
