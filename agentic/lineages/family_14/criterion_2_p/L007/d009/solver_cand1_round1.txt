```python
import sys

# The problem asks for the minimum cost to select a subset of vectors {e_u - e_v}
# such that the orthogonal complement of their span is a subspace of W.
# W is defined by x_a = x_b for all a, b in T.
# Let S be the set of vectors {e_u - e_v}. The orthogonal complement of span(S)
# is the set of vectors x such that x . (e_u - e_v) = 0 for all (u, v) in S.
# This is equivalent to x_u = x_v for all (u, v) in S.
# We want the condition "x_u = x_v for all (u, v) in S" to imply "x_a = x_b for all a, b in T".
# This is equivalent to saying that in the graph where edges are (u, v) in S,
# all nodes in T must be in the same connected component.
# We want to connect all nodes in T using a subset of the given edges with minimum cost.
# This is a Steiner Tree problem on a graph, but since we only need to connect
# the set T, we can treat all nodes in T as a single super-node.
# However, the graph is a tree (n nodes, n-1 edges).
# In a tree, there is a unique path between any two nodes.
# To connect all nodes in T, we must include all edges that lie on the unique
# paths between nodes in T.
# An edge (u, v) is part of the Steiner tree if and only if the removal of (u, v)
# splits the tree into two components, each containing at least one node from T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_indices = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many target nodes are in each subtree
    subtree_t_count = [0] * (n + 1)
    
    # Root the tree at an arbitrary node, say 1
    order = []
    parent = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in target_indices:
            subtree_t_count[u] += 1
        if parent[u] != 0:
            subtree_t_count[parent[u]] += subtree_t_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # Edge between i and parent[i] is needed if it separates the tree
        # into two components, both containing at least one target node.
        count = subtree_t_count[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_w[i]
            
    print(total_cost)

solve()
```
