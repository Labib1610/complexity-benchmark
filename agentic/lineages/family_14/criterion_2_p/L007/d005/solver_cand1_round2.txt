import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the minimal subtree that connects all target nodes.
    # This is the Steiner Tree problem on a tree, which is simply the union
    # of all paths between target nodes.
    # We can identify edges that are part of this subtree by checking if
    # the subtree rooted at a node contains at least one target node
    # and at least one target node is outside this subtree.
    
    has_target = [False] * (n + 1)
    for node in targets:
        has_target[node] = True
        
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to propagate target counts
    target_count = [0] * (n + 1)
    for node in targets:
        target_count[node] = 1
        
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            target_count[parent[u]] += target_count[u]
            
    # An edge (u, parent[u]) is part of the Steiner tree if 
    # 0 < target_count[u] < t
    for i in range(1, n + 1):
        if parent[i] != 0:
            if 0 < target_count[i] < t:
                total_cost += edge_to_parent_weight[i]
                
    print(total_cost)

solve()
