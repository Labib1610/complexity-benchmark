import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal_with_targets(n, edges, targets):
    # Sort edges by weight
    edges.sort(key=lambda x: x[2])
    
    # Initialize Union-Find for the entire graph
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # Keep track of component sizes for target nodes
    target_count = defaultdict(int)
    for t in targets:
        target_count[t] = 1
    
    total_cost = 0
    
    for u, v, w in edges:
        pu, pv = find(parent, u), find(parent, v)
        
        # If both components have targets, we need this edge
        if pu != pv:
            # Count how many targets are in each component
            targets_in_pu = target_count[pu]
            targets_in_pv = target_count[pv]
            
            # Union the components
            if rank[pu] < rank[pv]:
                pu, pv = pv, pu
            parent[pv] = pu
            if rank[pu] == rank[pv]:
                rank[pu] += 1
                
            # Update target count in the new root
            target_count[pu] += target_count[pv]
            target_count[pv] = 0
            
            # If we just connected two components that both had targets, we need this edge
            if targets_in_pu > 0 and targets_in_pv > 0:
                total_cost += w
    
    return total_cost

# Read input
line1 = list(map(int, sys.stdin.readline().split()))
n, m, t = line1[0], line1[1], line1[2]

targets = list(map(int, sys.stdin.readline().split()))

edges = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    edges.append((u, v, w))

# Solve
result = kruskal_with_targets(n, edges, targets)
print(result)
