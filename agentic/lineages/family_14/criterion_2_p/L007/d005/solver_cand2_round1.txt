```python
import sys

# The problem asks for the minimum cost to connect all t target variables in a tree.
# This is equivalent to finding the Steiner Tree for the set of t target nodes.
# Since the graph is a tree (n nodes, n-1 edges), the Steiner Tree is the unique
# minimal subtree that contains all t target nodes.
# The cost is the sum of weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    if t <= 1:
        print(0)
        return
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that lie on the path between any two target nodes.
    # An edge (u, v) with weight w is part of the Steiner Tree if and only if
    # the subtree formed by removing (u, v) contains at least one target node
    # in both components.
    
    # Count how many target nodes are in the subtree of each node.
    # Root the tree at an arbitrary node, say 1.
    target_count = [0] * (n + 1)
    for node in targets:
        target_count[node] = 1
        
    total_targets = t
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    # Process in reverse topological order to propagate target counts
    for u in reversed(order):
        if parent[u] != 0:
            p, w = parent[u]
            target_count[p] += target_count[u]
            # If this edge separates the tree into two parts both containing targets
            if target_count[u] > 0 and target_count[u] < total_targets:
                total_cost += w
                
    print(total_cost)

solve()
```
