```python
import sys

# The problem states m = n - 1 and the graph is connected.
# This means the graph is a tree.
# We need to connect all rented racks. In a tree, the unique path between
# any two nodes is fixed. To connect a set of nodes in a tree, we must
# include all edges that lie on the paths between any two rented nodes.
# This is equivalent to the Steiner Tree problem on a tree, which is
# simply the union of all paths between the rented nodes.
# Since the relay fee is 0, we just need to sum the weights of all edges
# that are part of the minimal subtree connecting the rented nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    rented = set()
    idx = 3
    for _ in range(t):
        rented.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # To find the edges in the minimal subtree:
    # Root the tree at an arbitrary rented node.
    # A node is part of the subtree if it is a rented node or if it has
    # at least two branches leading to rented nodes.
    # Alternatively, count how many rented nodes are in the subtree of each node.
    
    root = next(iter(rented))
    subtree_rented_count = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in rented:
            subtree_rented_count[node] += 1
            
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == root:
            continue
        if subtree_rented_count[u] > 0 and subtree_rented_count[u] < t:
            total_weight += edge_to_parent_weight[u]
        subtree_rented_count[parent[u]] += subtree_rented_count[u]
        
    print(total_weight)

solve()
```
