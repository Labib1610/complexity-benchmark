```python
import sys

# The problem asks for the minimum cost to connect a subset of t nodes in a tree.
# Since the relay fee is 0, this is equivalent to finding the weight of the 
# Steiner Tree for the given subset of nodes in the tree.
# In a tree, the Steiner Tree is simply the union of all paths between the 
# leased nodes. This is the minimal subtree containing all leased nodes.
# We can solve this by rooting the tree arbitrarily (e.g., at node 1) and 
# calculating the sum of weights of all edges that lie on a path between 
# any two leased nodes.
# An edge (u, v) where v is a child of u is part of the Steiner Tree if and 
# only if the subtree rooted at v contains at least one leased node AND 
# the rest of the tree (outside the subtree of v) contains at least one 
# leased node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    leased = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        leased[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # DFS to count leased nodes in subtrees
    subtree_leased = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        subtree_leased[u] += (1 if leased[u] else 0)
        if parent[u] != 0:
            subtree_leased[parent[u]] += subtree_leased[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # Edge between i and parent[i] is part of Steiner Tree if
        # subtree at i has leased nodes and the rest of the tree has leased nodes
        count = subtree_leased[i]
        if count > 0 and count < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
