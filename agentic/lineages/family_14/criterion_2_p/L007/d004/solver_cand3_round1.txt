```python
import sys

# The problem states that the facility has n racks and m = n - 1 links.
# This means the graph is a tree.
# We need to connect a subset of t rented racks.
# Since it's a tree, there is exactly one unique path between any two nodes.
# To connect all rented racks, we must include all edges that lie on the paths
# between any two rented racks.
# An edge (u, v) with weight w is part of the required network if and only if
# the subtree formed by removing (u, v) contains at least one rented rack
# on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    rented = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        rented[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # count_rented[u] will store the number of rented nodes in the subtree rooted at u
    count_rented = [0] * (n + 1)
    
    # DFS to compute count_rented
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if rented[u]:
            count_rented[u] += 1
        if parent[u] != 0:
            count_rented[parent[u]] += count_rented[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # Edge connects i to parent[i]
        # Subtree at i has count_rented[i] rented nodes
        # The rest of the tree has t - count_rented[i] rented nodes
        if count_rented[i] > 0 and count_rented[i] < t:
            total_cost += edge_weight[i]
            
    print(total_cost)

solve()
```
