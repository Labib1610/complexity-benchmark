```python
import sys

# The problem states that the constraints form a tree (m = n - 1 and no cycles).
# We need to find the minimum weight of edges that connect all t target nodes.
# In a tree, the set of edges that connect a subset of nodes is exactly the 
# union of all paths between any two nodes in that subset.
# This is equivalent to the edges in the minimal subtree containing all target nodes.
# An edge is part of this minimal subtree if and only if the subtree formed by 
# removing that edge splits the target nodes into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # target_count[u] will store how many target nodes are in the subtree of u
    target_count = [0] * (n + 1)
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            target_count[u] += 1
        if parent[u] != 0:
            target_count[parent[u]] += target_count[u]
            # If this edge separates the targets, it must be included
            if target_count[u] > 0 and target_count[u] < t:
                total_weight += edge_to_parent_weight[u]
                
    print(total_weight)

solve()
```
