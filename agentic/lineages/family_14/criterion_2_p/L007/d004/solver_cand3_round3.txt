```python
import sys

# The problem states that the structure of the m = n-1 constraints forms a tree.
# We need to unify t critical variables.
# In a tree, the unique path between two nodes is the only way to connect them.
# To unify a set of nodes in a tree, we must select all edges that lie on the 
# paths connecting these nodes.
# Specifically, the union of all paths between any two nodes in a set of nodes
# forms a subtree. The edges in this subtree are exactly those that separate
# the tree into components such that each component contains at least one 
# critical node or no critical nodes, but the critical nodes are all connected.
# More simply: an edge is part of the required set if and only if removing it
# splits the tree into two components, both of which contain at least one 
# critical variable.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_vars = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # count_critical[u] will store the number of critical nodes in the subtree
    # rooted at u (when rooted at 1).
    count_critical = [0] * (n + 1)
    parent_edge_cost = [0] * (n + 1)
    
    # DFS to compute subtree critical counts
    stack = [(1, 0)]
    order = []
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u, p = stack.pop()
        order.append((u, p))
        for v, w in adj[u]:
            if v != p:
                visited[v] = True
                parent_edge_cost[v] = w
                stack.append((v, u))
                
    for u in critical_vars:
        count_critical[u] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u, p in reversed(order):
        if p != 0:
            count_critical[p] += count_critical[u]
            # If the subtree at u contains some but not all critical nodes,
            # the edge (u, p) must be included.
            if 0 < count_critical[u] < t:
                total_cost += parent_edge_cost[u]
                
    print(total_cost)

solve()
```
