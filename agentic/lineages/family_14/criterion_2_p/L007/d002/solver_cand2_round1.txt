```python
import sys

# The problem describes a tree structure (n nodes, n-1 edges).
# We need to find the minimum cost to connect a subset of t nodes.
# In a tree, there is exactly one path between any two nodes.
# To connect a set of nodes S, we must include all edges that lie on the 
# unique paths between any pair of nodes in S.
# This is equivalent to the union of all paths from a root (e.g., the first rented node)
# to every other rented node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    rented = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Root the tree at the first rented node
    root = next(iter(rented))
    
    # parent[u] = (p, weight_to_p)
    parent = [None] * (n + 1)
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    order = []
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    # is_needed[u] is true if node u is in the subtree of a rented node
    is_needed = [False] * (n + 1)
    for r in rented:
        is_needed[r] = True
        
    # Process in reverse topological order (leaves to root)
    total_cost = 0
    for u in reversed(order):
        if is_needed[u] and parent[u] is not None:
            p, w = parent[u]
            total_cost += w
            is_needed[p] = True
            
    print(total_cost)

solve()
```
