```python
import sys

# The problem states that the potential links form a tree (N nodes, N-1 links).
# The condition is: for any partition (A, B) where both A and B contain at least
# one priority node, there must be at least one activated link between A and B.
# This is equivalent to saying that in the subgraph formed by the activated links,
# all priority nodes must belong to the same connected component.
# If all priority nodes are in the same connected component, then any partition
# (A, B) that separates two priority nodes will have a path between them,
# and thus at least one edge crossing the cut.
# If there are two priority nodes in different components, we can partition the
# nodes based on the components, creating a cut with no crossing edges.
# Thus, we need to select a subset of edges such that all priority nodes are
# connected, with minimum total weight.
# This is the Steiner Tree problem on a tree, which simplifies to the sum of
# weights of all edges on the unique paths between all priority nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    priority_nodes = set()
    idx = 3
    for _ in range(k):
        priority_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if k <= 1:
        print(0)
        return

    # Find edges that are part of the minimal subtree connecting all priority nodes.
    # An edge is in the minimal subtree if and only if the two components it
    # creates both contain at least one priority node.
    
    subtree_weight = 0
    
    # Root the tree at an arbitrary priority node
    root = next(iter(priority_nodes))
    
    # count_priority[u] = number of priority nodes in the subtree rooted at u
    count_priority = [0] * (n + 1)
    
    def dfs_count(u, p):
        c = 1 if u in priority_nodes else 0
        for v, w in adj[u]:
            if v != p:
                c += dfs_count(v, u)
        count_priority[u] = c
        return c
    
    dfs_count(root, -1)
    
    # An edge (u, v) where v is child of u is in the Steiner tree if
    # 0 < count_priority[v] < k
    def dfs_sum(u, p):
        nonlocal subtree_weight
        for v, w in adj[u]:
            if v != p:
                if 0 < count_priority[v] < k:
                    subtree_weight += w
                dfs_sum(v, u)
                
    dfs_sum(root, -1)
    print(subtree_weight)

solve()
```
