```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes in a tree.
# This is equivalent to finding the weight of the Steiner Tree in a tree,
# which is simply the sum of weights of all edges that lie on the paths
# between any two leased nodes.
# An edge (u, v) with weight w is part of the Steiner Tree if and only if
# the subtree formed by removing (u, v) contains at least one leased node
# on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    leased = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        leased[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # count_leased[u] will store the number of leased nodes in the subtree rooted at u
    count_leased = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    edge_weights = {} # (min(u,v), max(u,v)) -> w
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if u < v:
                edge_weights[(u, v)] = w
                
    for u in reversed(order):
        count_leased[u] += (1 if leased[u] else 0)
        if parent[u] != -1:
            p = parent[u]
            count_leased[p] += count_leased[u]
            # If the edge (u, p) separates the tree into two parts,
            # and both parts contain at least one leased node, add weight.
            if count_leased[u] > 0 and count_leased[u] < t:
                w = edge_weights[(min(u, p), max(u, p))]
                total_cost += w
                
    print(total_cost)

solve()
```
