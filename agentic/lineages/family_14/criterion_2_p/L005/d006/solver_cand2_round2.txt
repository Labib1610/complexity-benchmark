```python
import sys

# The problem states that there are n variables and m = n - 1 identities,
# forming a tree structure. We need to find the minimum cost to connect
# all t target variables. This is equivalent to finding the sum of weights
# of all edges in the minimal subtree that contains all t target variables.
# An edge (u, v) with weight w is part of this subtree if and only if
# the subtree formed by removing (u, v) has target variables in both components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # target_count[u] stores how many target variables are in the subtree rooted at u
    target_count = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            target_count[u] += 1
        if parent[u] != 0:
            target_count[parent[u]] += target_count[u]
            # If the edge (u, parent[u]) separates the tree into two parts
            # both containing at least one target, it must be included.
            if target_count[u] > 0 and target_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
