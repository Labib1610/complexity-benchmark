```python
import sys

# The problem asks for the minimum cost to connect all target scalars in a tree.
# Since the graph is a tree (n variables, n-1 edges, connected), there is a 
# unique path between any two nodes. To make all target scalars equal, we must 
# enforce all edges that lie on the paths between any two target scalars.
# This is equivalent to finding the smallest subtree that contains all target 
# scalars and summing the weights of all edges in that subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Root the tree at the first target scalar
    root = next(iter(targets))
    
    # We need to identify all edges that are part of the minimal subtree 
    # connecting all target scalars. An edge (u, v) is in the subtree if 
    # the subtree rooted at the child node contains at least one target 
    # and the rest of the tree contains at least one target.
    
    has_target = [False] * (n + 1)
    for node in targets:
        has_target[node] = True
        
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to propagate target counts
    for u in reversed(order):
        if u == root:
            continue
        if has_target[u]:
            total_cost += edge_to_parent_weight[u]
            has_target[parent[u]] = True
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
