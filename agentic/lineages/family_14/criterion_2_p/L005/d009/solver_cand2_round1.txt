```python
import sys

# The problem asks for the minimum cost to connect all priority dimensions
# in a graph where each edge (u, v) with weight w represents the vector e_u - e_v.
# The condition "e_a - e_b is in the span of selected vectors" is equivalent
# to saying that in the graph formed by the selected edges, there is a path
# between a and b.
# Since we have M = N - 1 edges and the graph is connected (guaranteed),
# the graph is a tree.
# We need to select a subset of edges such that all priority dimensions
# belong to the same connected component.
# This is equivalent to finding the minimum weight Steiner Tree for the
# set of priority dimensions in the given tree.
# In a tree, the Steiner Tree for a set of nodes is the minimal subtree
# containing all those nodes. The edges in this subtree are exactly those
# edges that lie on the unique path between some pair of priority nodes.
# An edge (u, v) is in the Steiner Tree if and only if removing it splits
# the tree into two components, each containing at least one priority node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    
    if T <= 1:
        print(0)
        return
    
    priority_nodes = set(map(int, input_data[3:3+T]))
    
    adj = [[] for _ in range(N + 1)]
    edges = []
    idx = 3 + T
    for _ in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_priority_count[u] stores how many priority nodes are in the
    # subtree rooted at u (when rooted at node 1).
    subtree_priority_count = [0] * (N + 1)
    parent_edge_weight = [0] * (N + 1)
    parent = [0] * (N + 1)
    
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in priority_nodes:
            subtree_priority_count[u] += 1
        if parent[u] != 0:
            subtree_priority_count[parent[u]] += subtree_priority_count[u]
            
    total_cost = 0
    for i in range(2, N + 1):
        # Edge (i, parent[i]) is part of the Steiner Tree if both components
        # created by removing it contain at least one priority node.
        count = subtree_priority_count[i]
        if count > 0 and count < T:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
