```python
import sys
from collections import defaultdict, deque

sys.setrecursionlimit(1000000)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    m = int(data[idx+1])
    t = int(data[idx+2])
    idx += 3
    
    essential = list(map(int, data[idx:idx+t]))
    idx += t
    
    # If less than 2 essential sites, no paths needed
    if t < 2:
        print(0)
        return
    
    # Build adjacency list for the tree
    adj = [[] for _ in range(n+1)]
    edges = []
    
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # Build a mapping from node to its neighbors with edge weights
    # We'll do BFS from all essential nodes to find which edges are needed
    
    # However, since it's a tree and we want all paths between essential nodes,
    # we can think of it as finding the minimal subtree that connects all essential nodes.
    # This is the Steiner tree problem on a tree, which can be solved by:
    # 1. Finding the union of all paths between pairs of essential nodes.
    # But more efficiently:
    # 1. Root the tree at any node.
    # 2. Find the subtree that contains all essential nodes.
    # 3. The edges in this subtree are exactly those we need to keep.
    
    # Efficient approach:
    # 1. Pick any essential node as root and do BFS/DFS to compute parent and depth
    # 2. For each essential node, mark all nodes on the path from it to the root
    # 3. The minimal connected subgraph containing all essential nodes is the union of these paths
    # Actually, better approach:
    # 1. Root the tree at node 1
    # 2. Do DFS to compute parent and depth for all nodes
    # 3. Find the lowest common ancestor (LCA) for all pairs, but that's too slow
    # Even better:
    # 1. The required edges are those in the minimal subtree spanning all essential nodes
    # 2. In a tree, this is equivalent to: take all edges such that both endpoints' subtrees 
    #    contain at least one essential node when removing the edge
    
    # Most straightforward O(n) method:
    # 1. Root the tree at an arbitrary node (e.g., 1)
    # 2. Do a post-order DFS: for each node, count how many essential nodes are in its subtree
    # 3. An edge from parent to child is needed iff the child's subtree contains at least one 
    #    essential node and the rest of the tree also contains at least one essential node
    
    # Let's implement this
    
    # First, build the tree rooted at node 1
    # We need to orient the tree. Let's use adjacency list with parent pointers
    
    # Root the tree at node 1
    parent = [-1] * (n+1)
    children = [[] for _ in range(n+1)]
    visited = [False] * (n+1)
    
    # BFS to build rooted tree
    queue = deque([1])
    visited[1] = True
    while queue:
        u = queue.popleft()
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                children[u].append((v, w))
                queue.append(v)
    
    # Now, count essential nodes in each subtree
    is_essential = [False] * (n+1)
    for node in essential:
        is_essential[node] = True
        
    count_essential = [0] * (n+1)
    
    # Post-order DFS to count essential nodes in each subtree
    def dfs_count(u):
        count = 1 if is_essential[u] else 0
        for v, w in children[u]:
            count += dfs_count(v)
        count_essential[u] = count
        return count
    
    dfs_count(1)
    
    # Total number of essential nodes
    total_essential = len(essential)
    
    # Now, collect edges that must be included
    # An edge (u,v) where u is parent of v is needed if:
    # count_essential[v] > 0 and (total_essential - count_essential[v]) > 0
    total_cost = 0
    for u in range(1, n+1):
        for v, w in children[u]:
            if count_essential[v] > 0 and (total_essential - count_essential[v]) > 0:
                total_cost += w
                
    print(total_cost)

if __name__ == "__main__":
    main()
```
