import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    prime_habitats = set()
    idx = 3
    for _ in range(t):
        prime_habitats.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # has_prime[u] will be true if the subtree rooted at u contains at least one prime habitat
    has_prime = [False] * (n + 1)
    
    # We need to count how many prime habitats are in the subtree of each node
    # to determine if an edge is essential.
    # An edge (u, v) where v is child of u is essential if:
    # count_prime(v) > 0 AND (total_prime - count_prime(v)) > 0
    
    subtree_prime_count = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in prime_habitats:
            subtree_prime_count[u] += 1
        if parent[u] != 0:
            subtree_prime_count[parent[u]] += subtree_prime_count[u]
            
    total_sum = 0
    # Check each edge (u, v) where v is child
    for v in range(2, n + 1):
        p = parent[v]
        # Find weight of edge (p, v)
        w = 0
        for neighbor, weight in adj[v]:
            if neighbor == p:
                w = weight
                break
        
        count_v = subtree_prime_count[v]
        if count_v > 0 and (t - count_v) > 0:
            total_sum += w
            
    print(total_sum)

if __name__ == '__main__':
    solve()
