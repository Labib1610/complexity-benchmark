```python
import sys

# The problem states that the infrastructure is a tree (n facilities, n-1 connections).
# We need to find the minimum cost to connect all critical facilities.
# In a tree, the set of edges required to connect a subset of nodes is exactly
# the set of edges that lie on the unique paths between any two nodes in the subset.
# This is equivalent to the edges in the minimal subtree containing all critical nodes.
# An edge (u, v) with weight w is part of this subtree if and only if the subtree
# formed by removing (u, v) contains at least one critical node in each component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_critical[u] will store the number of critical nodes in the subtree rooted at u
    subtree_critical = [0] * (n + 1)
    
    # DFS to count critical nodes in subtrees
    # Since n can be up to 200,000, use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in critical_nodes:
        subtree_critical[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if subtree_critical[u] > 0 and subtree_critical[u] < t:
            total_cost += edge_to_parent_weight[u]
        subtree_critical[p] += subtree_critical[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
