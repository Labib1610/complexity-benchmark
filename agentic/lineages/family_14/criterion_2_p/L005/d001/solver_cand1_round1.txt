```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes (priority facilities)
# in a tree structure. Since the input is guaranteed to be a tree (n nodes, n-1 edges, connected),
# the minimum cost to connect a set of nodes is the sum of the weights of all edges
# that lie on the paths between any two priority nodes.
# An edge is part of the Steiner Tree if and only if it splits the tree into two components,
# each containing at least one priority node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_nodes = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many priority nodes are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    subtree_priority_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in priority_nodes:
            subtree_priority_count[node] += 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        p = parent[u]
        subtree_priority_count[p] += subtree_priority_count[u]
        
        # If this edge separates the tree into two parts, both containing at least one priority node
        count = subtree_priority_count[u]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

solve()
```
