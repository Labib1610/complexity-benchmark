```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need the sum of weights of all edges that lie on the unique path
    # between any two target nodes. This is equivalent to the sum of weights
    # of all edges that have at least one target node in both components
    # formed by removing that edge.
    
    # Root the tree at an arbitrary target node
    root = next(iter(targets))
    
    total_weight = 0
    
    # subtree_targets[u] stores whether the subtree at u contains any target nodes
    # count_targets[u] stores the number of target nodes in the subtree at u
    count_targets = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    parent[root] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in targets:
            count_targets[u] = 1
            
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == root:
            continue
        p = parent[u]
        count_targets[p] += count_targets[u]
        # If the subtree at u contains some targets but not all,
        # the edge (u, p) must be preserved.
        if 0 < count_targets[u] < t:
            total_weight += edge_to_parent_weight[u]
            
    print(total_weight)

solve()
```
