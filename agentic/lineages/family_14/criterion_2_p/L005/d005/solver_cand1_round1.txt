```python
import sys

# The problem states m = n - 1 and the graph is connected.
# This means the graph is a tree.
# We need to find the sum of weights of all edges that lie on the unique path
# between any two critical sectors.
# An edge is part of the path between at least two critical sectors if and only if
# the subtree formed by removing that edge contains at least one critical sector
# on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count critical nodes in subtree
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    crit_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if i in critical_nodes:
            crit_count[i] = 1
            
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != -1:
            p = parent[u]
            crit_count[p] += crit_count[u]
            # If this edge separates the tree into two parts, both containing
            # at least one critical node, then this edge must be included.
            if 0 < crit_count[u] < t:
                total_weight += edge_weight[u]
                
    print(total_weight)

solve()
```
