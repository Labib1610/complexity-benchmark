```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_vars = set()
    idx = 3
    for _ in range(t):
        critical_vars.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the smallest subtree that contains all critical variables.
    # The edges in this subtree are exactly the edges on the paths between 
    # any two critical variables.
    
    # Use DFS to find which nodes are part of the minimal subtree.
    # A node is in the subtree if it is a critical variable or if it has 
    # at least two branches leading to critical variables.
    
    has_critical = [False] * (n + 1)
    for v in critical_vars:
        has_critical[v] = True
        
    total_weight = 0
    
    # visited array to keep track of nodes
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    subtree_has_critical = [False] * (n + 1)
    for i in range(1, n + 1):
        if has_critical[i]:
            subtree_has_critical[i] = True
            
    for i in reversed(order):
        if i == 1:
            continue
        if subtree_has_critical[i]:
            total_weight += edge_to_parent_weight[i]
            subtree_has_critical[parent[i]] = True
            
    print(total_weight)

solve()
```
