import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t < 2:
        print(0)
        return

    # subtree_critical[u] stores the number of critical nodes in the subtree rooted at u
    subtree_critical = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Calculate subtree_critical counts
    for u in reversed(order):
        if u in critical_nodes:
            subtree_critical[u] += 1
        if parent[u] != -1:
            subtree_critical[parent[u]] += subtree_critical[u]
            
    total_sum = 0
    # For each edge (u, v), removing it splits the tree into two components.
    # One component has subtree_critical[child] nodes, the other has t - subtree_critical[child].
    # The condition is met if both are > 0.
    for u, v, w in edges:
        # Identify which node is the child in the rooted tree
        child = u if parent[u] == v else v
        count = subtree_critical[child]
        if count > 0 and (t - count) > 0:
            total_sum += w
            
    print(total_sum)

if __name__ == '__main__':
    solve()
