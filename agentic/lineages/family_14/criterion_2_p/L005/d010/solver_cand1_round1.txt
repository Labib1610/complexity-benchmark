```python
import sys

# The problem states m = n - 1. Since the graph is connected and has n nodes 
# and n-1 edges, the graph is a tree.
# If the graph is a tree, there is exactly one path between any two nodes.
# To connect a set of priority outposts, we must include all edges that lie 
# on the unique paths between any two priority outposts.
# This is equivalent to finding the smallest subtree that contains all 
# priority outposts.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Since it's a tree, we can root it at an arbitrary priority node.
    # We need to find all edges that are part of a path between any two 
    # priority nodes. An edge (u, v) where v is a child of u is part of 
    # such a path if and only if the subtree rooted at v contains at least 
    # one priority node and the rest of the tree (outside the subtree of v) 
    # also contains at least one priority node.
    
    root = next(iter(priority_nodes))
    
    # subtree_has_priority[u] = number of priority nodes in subtree of u
    subtree_has_priority = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in priority_nodes:
            subtree_has_priority[u] += 1
        if parent[u] != 0:
            subtree_has_priority[parent[u]] += subtree_has_priority[u]
            
    total_weight = 0
    for i in range(1, n + 1):
        if i == root:
            continue
        # Edge between i and parent[i] is needed if subtree at i has 
        # priority nodes and the rest of the tree has priority nodes.
        count = subtree_has_priority[i]
        if count > 0 and count < t:
            total_weight += parent_edge_weight[i]
            
    print(total_weight)

solve()
```
