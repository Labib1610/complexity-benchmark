import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    essential = set()
    idx = 3
    for _ in range(t):
        essential.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the smallest subtree containing all essential nodes.
    # This is equivalent to removing all edges that do not lie on a path 
    # between any two essential nodes.
    # An edge (u, v) with weight w is part of the Steiner tree if and only if
    # the subtree at u (when rooted at v) contains at least one essential node
    # AND the rest of the tree (everything not in the subtree at u) contains 
    # at least one essential node.
    
    is_essential = [False] * (n + 1)
    for node in essential:
        is_essential[node] = True
        
    subtree_essential_count = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order to count essential nodes in subtrees
    edge_weights = {}
    for u in reversed(order):
        if is_essential[u]:
            subtree_essential_count[u] += 1
        
        if parent[u] != -1:
            p = parent[u]
            # Find weight of edge (u, p)
            for v, w in adj[u]:
                if v == p:
                    edge_weights[u] = w
                    break
            subtree_essential_count[p] += subtree_essential_count[u]
            
    ans = 0
    for u in order:
        if u == 1: continue
        count = subtree_essential_count[u]
        # If this subtree contains some essential nodes but not all, 
        # the edge to its parent must be included.
        if 0 < count < t:
            ans += edge_weights[u]
            
    print(ans)

solve()
