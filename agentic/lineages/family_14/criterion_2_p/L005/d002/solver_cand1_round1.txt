```python
import sys

# The problem states that the candidate conduits form a tree (n nodes, n-1 edges, connected).
# To connect a subset of essential stations, we need the union of all paths between them.
# This is equivalent to the Steiner Tree problem on a tree, which is simply the sum of 
# weights of all edges that lie on the path between at least two essential stations.
# An edge (u, v) with weight w lies on the path between two essential stations if and only if
# the two components formed by removing (u, v) both contain at least one essential station.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    essential = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        essential[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count essential nodes in subtree
    subtree_essential = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if essential[u]:
            subtree_essential[u] += 1
        if parent[u] != -1:
            subtree_essential[parent[u]] += subtree_essential[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # Edge connects subtree at i to the rest of the tree
        count = subtree_essential[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

solve()
```
