```python
import sys

# The problem asks for the minimum cost to connect all target variables in a tree.
# Since m = n - 1 and the graph is guaranteed to be connected, the graph is a tree.
# To determine the relative difference between all pairs of target variables,
# we need to select the edges that form the minimal subtree containing all target variables.
# This is equivalent to the sum of weights of all edges that lie on the paths
# between any two target variables.
# An edge (u, v) with weight w is part of the minimal subtree if and only if
# the removal of this edge splits the tree into two components, each containing
# at least one target variable.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Root the tree at an arbitrary target node
    root = next(iter(targets))
    
    # subtree_targets[u] = number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    parent = [0] * (n + 1)
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in targets:
            subtree_targets[u] += 1
            
    # Process in reverse topological order to propagate counts
    total_cost = 0
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        subtree_targets[p] += subtree_targets[u]
        # If the edge (u, p) separates the tree into two parts both containing targets
        if subtree_targets[u] > 0 and subtree_targets[u] < t:
            total_cost += parent_edge_weight[u]
            
    print(total_cost)

solve()
```
