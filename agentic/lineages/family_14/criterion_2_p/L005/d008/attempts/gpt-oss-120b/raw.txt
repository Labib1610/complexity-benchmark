```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    focus = [int(next(it)) - 1 for _ in range(t)]
    if t <= 1:
        print(0)
        return

    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        adj[u].append((v, w, len(edges)))
        adj[v].append((u, w, len(edges)))
        edges.append((u, v, w))

    # root the tree at 0 (any node)
    parent = [-1] * n
    parent_edge = [-1] * n
    order = [0]
    parent[0] = 0
    for v in order:
        for to, w, idx in adj[v]:
            if parent[to] == -1:
                parent[to] = v
                parent_edge[to] = idx
                order.append(to)

    # count terminals in subtrees
    term_cnt = [0] * n
    for v in focus:
        term_cnt[v] = 1

    total = 0
    # process in reverse order (post-order)
    for v in reversed(order):
        if v == 0:
            continue
        cnt = term_cnt[v]
        if 0 < cnt < t:
            # edge to parent is needed
            u, _, w = edges[parent_edge[v]]
            total += w
        term_cnt[parent[v]] += cnt

    print(total)

if __name__ == "__main__":
    solve()
```
