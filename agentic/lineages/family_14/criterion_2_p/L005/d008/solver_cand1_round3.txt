```python
import sys

# The problem asks for the minimum cost to connect all focus identifiers in a tree.
# Since m = n - 1 and the graph is guaranteed to be connected, the graph is a tree.
# The condition f(a) = f(b) for all focus identifiers a, b is equivalent to saying
# that all focus identifiers must belong to the same connected component.
# In a tree, there is a unique path between any two nodes. To connect a set of nodes,
# we must include all edges that lie on the paths between any pair of focus nodes.
# This is equivalent to the Steiner Tree problem on a tree, which is simply the
# sum of weights of all edges that have at least one focus node in each of the
# two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    focus_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many focus nodes are in each subtree.
    # Root the tree at an arbitrary node, say 1.
    subtree_focus_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in focus_nodes:
            subtree_focus_count[u] += 1
        if parent[u] != 0:
            subtree_focus_count[parent[u]] += subtree_focus_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        count = subtree_focus_count[i]
        # If an edge separates the tree into two components, and both components
        # contain at least one focus node, the edge must be included.
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

solve()
```
