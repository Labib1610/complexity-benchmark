```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    focus_registers = set()
    idx = 3
    for _ in range(t):
        focus_registers.add(int(input_data[idx]))
        idx += 1
        
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to connect all focus registers in a single component.
    # This is equivalent to finding the Minimum Spanning Tree of the graph
    # where we only care about the connectivity of the focus registers.
    # Since m = n-1 (given m is the minimum to connect all n registers),
    # the graph is a tree. We need the sum of weights of all edges on the
    # unique paths between any two focus registers.
    
    adj = [[] for _ in range(n + 1)]
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Root the tree at the first focus register
    root = next(iter(focus_registers))
    
    # parent_edge[u] = (parent, weight_to_parent)
    parent_edge = [None] * (n + 1)
    is_focus = [False] * (n + 1)
    for f in focus_registers:
        is_focus[f] = True
        
    stack = [root]
    order = []
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent_edge[v] = (u, w)
                stack.append(v)
                
    # subtree_focus_count[u] = number of focus registers in subtree of u
    subtree_focus_count = [0] * (n + 1)
    for u in focus_registers:
        subtree_focus_count[u] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == root:
            continue
        
        p, w = parent_edge[u]
        # If this edge separates the set of focus registers, it must be included
        if subtree_focus_count[u] > 0 and subtree_focus_count[u] < t:
            total_cost += w
            
        subtree_focus_count[p] += subtree_focus_count[u]
        
    print(total_cost)

solve()
```
