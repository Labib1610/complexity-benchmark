```python
import sys

# The problem asks for the minimum cost to connect all focus indices in a graph
# where edges represent symmetry constraints. If we have a set of constraints
# (u, v) with cost w, and we want to ensure symmetry for all permutations of
# a set of focus indices, this is equivalent to ensuring that all focus indices
# belong to the same connected component in the graph formed by the chosen constraints.
# Since the graph of all m constraints is a tree (m = n - 1), the minimum cost
# to connect a subset of nodes in a tree is the sum of the weights of all edges
# that lie on the unique paths between the focus nodes.
# This is equivalent to the sum of weights of all edges that, when removed,
# split the tree into two components, each containing at least one focus node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    focus_indices = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # subtree_focus[u] stores the number of focus nodes in the subtree rooted at u
    subtree_focus = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for i in range(1, n + 1):
        if i in focus_indices:
            subtree_focus[i] = 1
            
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        subtree_focus[p] += subtree_focus[u]
        # If this edge separates the tree into two parts both containing focus nodes
        if subtree_focus[u] > 0 and subtree_focus[u] < t:
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
