```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1

    is_hub = [False] * (n + 1)
    for _ in range(t):
        h = int(data[idx]); idx += 1
        is_hub[h] = True

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))

    if t <= 1:
        print(0)
        return

    parent = [0] * (n + 1)
    parent_edge = [0] * (n + 1)
    depth = [0] * (n + 1)
    order = [1]
    parent[1] = 0
    depth[1] = 0
    pos = 0
    while pos < len(order):
        u = order[pos]
        pos += 1
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                parent_edge[v] = w
                depth[v] = depth[u] + 1
                order.append(v)

    hub_count = [0] * (n + 1)
    for u in range(1, n + 1):
        if is_hub[u]:
            hub_count[u] = 1

    total_cost = 0
    for i in range(len(order) - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if hub_count[u] > 0:
            total_cost += parent_edge[u]
            hub_count[p] += hub_count[u]

    print(total_cost)

solve()
```
