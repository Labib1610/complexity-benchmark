```python
import sys

# The problem asks for the minimum weight of edges to cut to separate all vital chambers
# into different components, such that no two vital chambers are in the same component.
# Since the graph is a tree (N nodes, N-1 edges), this is equivalent to finding the
# minimum weight set of edges whose removal disconnects all vital nodes from each other.
# This is equivalent to the Steiner Tree problem on a tree, which simplifies to:
# For every edge, if removing it splits the tree into two components, and both components
# contain at least one vital node, then this edge MUST be included in the selection.
# If an edge is removed, the vital nodes in the two resulting components are separated.
# The condition "every vital split is crossed" means that for any partition of vital
# nodes into two non-empty sets, there must be an edge crossing the cut.
# In a tree, this is satisfied if and only if we select all edges that lie on the
# unique paths between any two vital nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    vital_nodes = set(map(int, input_data[3:3+F]))
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many vital nodes are in the subtree of each node
    # We root the tree at an arbitrary vital node (e.g., the first one)
    root = next(iter(vital_nodes))
    
    vital_count = [0] * (N + 1)
    parent_edge_weight = [0] * (N + 1)
    parent = [0] * (N + 1)
    order = []
    stack = [root]
    visited = [False] * (N + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in vital_nodes:
        vital_count[node] = 1
        
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == root:
            continue
        p = parent[u]
        if vital_count[u] > 0 and vital_count[u] < F:
            total_weight += parent_edge_weight[u]
        vital_count[p] += vital_count[u]
        
    print(total_weight)

solve()
```
