```python
import sys

# The problem states E = N - 1 and the constraints form a tree.
# We need to find the minimum cost to connect all critical variables.
# In a tree, the unique path between any two nodes is the only way to connect them.
# To connect a set of critical nodes, we must include all edges that lie on the 
# paths between any pair of critical nodes.
# This is equivalent to the sum of weights of all edges that, when removed, 
# split the tree into two components, each containing at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    critical_vars = set(map(int, input_data[3:3+F]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # subtree_critical[u] stores the number of critical nodes in the subtree of u
    subtree_critical = [0] * (N + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in critical_vars:
        subtree_critical[u] = 1
        
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        if subtree_critical[u] > 0 and subtree_critical[u] < F:
            total_cost += edge_to_parent_weight[u]
        subtree_critical[p] += subtree_critical[u]
        
    print(total_cost)

solve()
```
