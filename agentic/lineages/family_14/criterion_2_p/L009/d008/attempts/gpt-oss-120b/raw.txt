import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it)); E = int(next(it)); F = int(next(it))
    targets = set(int(next(it)) for _ in range(F))
    adj = [[] for _ in range(N + 1)]
    edges = []
    for _ in range(E):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    # root the tree at 1
    parent = [0] * (N + 1)
    pw = [0] * (N + 1)          # weight to parent
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        node = stack.pop()
        order.append(node)
        for nb, w in adj[node]:
            if nb == parent[node]:
                continue
            parent[nb] = node
            pw[nb] = w
            stack.append(nb)

    # post-order counts
    cnt = [0] * (N + 1)
    for node in reversed(order):
        cnt[node] = 1 if node in targets else 0
        for nb, _ in adj[node]:
            if parent[nb] == node:   # nb is child
                cnt[node] += cnt[nb]

    total_targets = F
    ans = 0
    for v in range(2, N + 1):
        c = cnt[v]
        if 0 < c < total_targets:
            ans += pw[v]

    print(ans)

if __name__ == "__main__":
    solve()
