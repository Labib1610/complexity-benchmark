import sys

# The problem asks for the minimum cost to connect all target nodes in a graph
# where edges represent b_u - b_v.
# The span of a set of vectors {b_u - b_v} contains b_x - b_y if and only if
# x and y are in the same connected component in the graph formed by the edges.
# We need all target nodes to be in the same connected component.
# Since E = N - 1 and the graph is connected (given), the graph is a tree.
# We need to find the minimum cost to connect all target nodes in a tree.
# This is equivalent to the sum of weights of all edges that lie on the paths
# between any two target nodes.
# An edge divides the tree into two components. If both components contain at least
# one target node, the edge must be included.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    targets = set()
    idx = 3
    for _ in range(F):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(N + 1)]
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many targets are in the subtree of each node
    target_count = [0] * (N + 1)
    for t in targets:
        target_count[t] = 1
        
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order
    edge_weights = {}
    for u in reversed(order):
        if parent[u] != -1:
            # Find weight of edge (u, parent[u])
            # We can store it during the first pass
            pass
            
    # Re-calculate with edge weights stored
    edge_to_parent_w = [0] * (N + 1)
    for u in range(1, N + 1):
        for v, w in adj[u]:
            if v == parent[u]:
                edge_to_parent_w[u] = w
                
    for u in reversed(order):
        if parent[u] != -1:
            p = parent[u]
            target_count[p] += target_count[u]
            if target_count[u] > 0 and target_count[u] < F:
                total_cost += edge_to_parent_w[u]
                
    print(total_cost)

solve()
