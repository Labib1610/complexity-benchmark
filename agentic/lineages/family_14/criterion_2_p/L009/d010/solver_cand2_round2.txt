```python
import sys

# The problem asks for the minimum cost to connect all critical elements in a graph
# where edges are transpositions. Since E = N - 1 and the graph is guaranteed to be
# connected (the full set of transpositions acts transitively), the graph is a tree.
# In a tree, the set of edges that connects a subset of vertices is the unique
# minimal subgraph (a Steiner Tree) connecting those vertices.
# For a tree, the Steiner Tree is simply the union of all paths between the
# critical nodes. This is equivalent to the sum of weights of all edges that
# have at least one critical node in each of the two components formed by
# removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    critical_nodes = set()
    idx = 3
    for _ in range(F):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(N + 1)]
    edges = []
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_critical_count[u] stores the number of critical nodes in the subtree rooted at u
    subtree_critical_count = [0] * (N + 1)
    
    # DFS to compute subtree counts
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in critical_nodes:
            subtree_critical_count[u] += 1
        if parent[u] != 0:
            subtree_critical_count[parent[u]] += subtree_critical_count[u]
            
    total_cost = 0
    for i in range(2, N + 1):
        count = subtree_critical_count[i]
        # If an edge separates the tree into two components, and both components
        # contain at least one critical node, the edge must be included.
        if count > 0 and count < F:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

solve()
```
