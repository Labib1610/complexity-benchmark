```python
import sys

# The problem asks for the minimum cost to connect all critical nodes in a tree.
# Since the graph is a tree (N nodes, N-1 edges, connected), the unique path
# between any two nodes is the only way to connect them.
# To make all F critical nodes belong to the same orbit (connected component),
# we must select all edges that lie on the paths between any two critical nodes.
# This is equivalent to finding the minimal subtree that contains all F critical nodes.
# The cost is the sum of weights of all edges that are part of the minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+F]))
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that lie on the path between any two critical nodes.
    # An edge (u, v) with weight w is part of the minimal subtree if and only if
    # the subtree rooted at u (when the edge is removed) contains at least one
    # critical node, and the subtree rooted at v contains at least one critical node.
    
    is_critical = [False] * (N + 1)
    for node in critical_nodes:
        is_critical[node] = True
        
    subtree_critical_count = [0] * (N + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        subtree_critical_count[u] += (1 if is_critical[u] else 0)
        if parent[u] != 0:
            subtree_critical_count[parent[u]] += subtree_critical_count[u]
            # If this edge separates the tree into two parts both containing critical nodes
            if 0 < subtree_critical_count[u] < F:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
