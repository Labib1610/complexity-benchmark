```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    E = int(next(it))
    F = int(next(it))
    targets = [int(next(it)) - 1 for _ in range(F)]
    adj = [[] for _ in range(N)]
    edges = []
    for _ in range(E):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    if F <= 1:
        print(0)
        return

    target_set = [0] * N
    for t in targets:
        target_set[t] = 1
    total_targets = F

    parent = [-1] * N
    pw = [0] * N          # weight to parent
    order = []
    stack = [0]
    parent[0] = 0
    while stack:
        node = stack.pop()
        order.append(node)
        for nb, w in adj[node]:
            if parent[nb] == -1:
                parent[nb] = node
                pw[nb] = w
                stack.append(nb)

    sub_cnt = [0] * N
    for node in reversed(order):
        cnt = target_set[node]
        for nb, _ in adj[node]:
            if parent[nb] == node:
                cnt += sub_cnt[nb]
        sub_cnt[node] = cnt

    ans = 0
    for v in range(1, N):
        cnt = sub_cnt[v]
        if cnt > 0 and total_targets - cnt > 0:
            ans += pw[v]

    print(ans)

if __name__ == "__main__":
    solve()
```
