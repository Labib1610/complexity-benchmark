```python
import sys

# The problem states the network is a tree (n stations, n-1 segments for connectivity).
# We need to connect all freight hubs using the minimum cost of edges.
# In a tree, there is a unique path between any two nodes.
# To connect a set of nodes in a tree, we must include all edges that lie on the 
# unique paths between any pair of these nodes.
# This is equivalent to the union of all paths between the hubs.
# This set of edges forms the minimal subtree containing all hubs.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    hubs = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Root the tree at an arbitrary hub
    root = next(iter(hubs))
    
    # parent[u] = (p, weight_to_p)
    parent = [None] * (n + 1)
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    order = []
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    # has_hub_in_subtree[u] is true if the subtree at u contains at least one hub
    has_hub_in_subtree = [False] * (n + 1)
    for h in hubs:
        has_hub_in_subtree[h] = True
        
    total_cost = 0
    # Process nodes in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == root:
            continue
        
        p, w = parent[u]
        if has_hub_in_subtree[u]:
            total_cost += w
            has_hub_in_subtree[p] = True
            
    print(total_cost)

solve()
```
