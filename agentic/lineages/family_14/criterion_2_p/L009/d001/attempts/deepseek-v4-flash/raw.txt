We can solve this by finding the minimum spanning tree that connects only the freight hubs. Since the graph is a tree (E = N − 1), the minimal subgraph connecting the hubs is the union of paths between them, which can be found by pruning non-hub leaves.

```python
import sys
from collections import deque

input = sys.stdin.readline

N, E, F = map(int, input().split())
hubs = set(map(int, input().split()))

adj = [[] for _ in range(N + 1)]
edges = []
for _ in range(E):
    u, v, w = map(int, input().split())
    adj[u].append((v, w))
    adj[v].append((u, w))
    edges.append((u, v, w))

# If only one hub, no need to reinforce anything
if F == 1:
    print(0)
    sys.exit()

# Degree of each node
deg = [len(adj[i]) for i in range(N + 1)]

# Queue of leaves that are not hubs
q = deque()
for i in range(1, N + 1):
    if deg[i] == 1 and i not in hubs:
        q.append(i)

# Remove non-hub leaves iteratively
removed = [False] * (N + 1)
while q:
    u = q.popleft()
    removed[u] = True
    for v, w in adj[u]:
        if not removed[v]:
            deg[v] -= 1
            if deg[v] == 1 and v not in hubs:
                q.append(v)

# Now sum weights of edges that connect remaining nodes
total = 0
for u, v, w in edges:
    if not removed[u] and not removed[v]:
        total += w

print(total)
```
