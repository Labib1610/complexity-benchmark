```python
import sys

# The problem states m = n - 1, which means the graph is a tree.
# To connect all freight hubs in a tree, we must include all edges that lie
# on the unique paths between any two freight hubs.
# This is equivalent to finding the smallest subtree that contains all freight hubs.
# The edges in this subtree are exactly those that have at least one freight hub
# in one of the two components formed by removing the edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    hubs = set()
    idx = 3
    for _ in range(t):
        hubs.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_hubs[u] will store the number of hubs in the subtree rooted at u
    subtree_hubs = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in hubs:
            subtree_hubs[u] += 1
            
    total_cost = 0
    # Process in reverse topological order to propagate hub counts up
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        subtree_hubs[p] += subtree_hubs[u]
        # If this edge separates some hubs from others, it must be included
        if 0 < subtree_hubs[u] < t:
            total_cost += parent_edge_weight[u]
            
    print(total_cost)

solve()
```
