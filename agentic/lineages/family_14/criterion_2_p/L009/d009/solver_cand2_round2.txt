```python
import sys

# The problem asks for the minimum cost to select a subset of edges such that
# in the resulting graph, all target nodes are in the same connected component.
# The kernel of the operator defined by edges (u, v) is the set of vectors z
# such that z_u = z_v for all selected edges (u, v).
# This means z_u = z_v if u and v are in the same connected component.
# The condition "every vector z in the kernel satisfies z_x = z_y for all target indices x, y"
# is equivalent to saying that all target indices must belong to the same connected component.
# Since E = N - 1 and the full graph is connected, the graph is a tree.
# We need to find the minimum cost to connect all target nodes in a tree.
# This is equivalent to finding the sum of weights of all edges that lie on the
# unique paths between any two target nodes.
# An edge (u, v) lies on a path between two target nodes if and only if
# removing (u, v) splits the tree into two components, each containing at least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+F]))
    edges = []
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many targets are in the subtree of each node
    target_count = [0] * (N + 1)
    for t in targets:
        target_count[t] = 1
        
    # DFS to compute target_count
    parent = [0] * (N + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    total_cost = 0
    for u in reversed(order):
        if parent[u] != -1:
            p = parent[u]
            target_count[p] += target_count[u]
            
    # An edge (u, parent[u]) is needed if 0 < target_count[u] < F
    for u in range(2, N + 1):
        count = target_count[u]
        if count > 0 and count < F:
            # Find weight of edge (u, parent[u])
            for v, w in adj[u]:
                if v == parent[u]:
                    total_cost += w
                    break
                    
    print(total_cost)

solve()
```
