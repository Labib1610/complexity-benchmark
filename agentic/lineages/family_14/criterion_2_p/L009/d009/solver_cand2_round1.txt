```python
import sys

# The problem asks for the minimum cost to connect all target vectors in a graph.
# Each difference vector b_u - b_v acts as an edge between nodes u and v.
# If we have a set of edges, the span of these vectors contains b_x - b_y
# if and only if x and y are in the same connected component.
# To ensure all target vectors {t_1, t_2, ..., t_F} are in the same component,
# we need to find the Minimum Spanning Tree (MST) of the graph formed by the
# target nodes.
# Since we can use any of the E edges to connect these nodes, this is equivalent
# to finding the MST of the graph where we want to connect all target nodes.
# This is a Steiner Tree problem, but since we want to connect ALL target nodes
# and we have E edges forming a connected graph of N nodes, we can simplify:
# We want to find the minimum cost to connect the set of target nodes.
# This is equivalent to finding the MST of the graph where we contract all
# non-target nodes or simply finding the MST of the graph and then considering
# the connectivity of the target nodes.
# Actually, the problem is simpler: we need to connect all target nodes.
# This is equivalent to finding the MST of the graph where we only care about
# the connectivity of the target nodes.
# We can use Kruskal's algorithm on the given E edges.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+F]]
    
    if F <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    edges.sort()
    
    parent = list(range(N + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # We need to connect all target nodes.
    # This is equivalent to finding the MST of the graph, but we only need
    # to connect the target nodes. However, the edges are given and we must
    # use them. The standard approach for "connect a subset of nodes" is
    # Kruskal's: add edges if they connect two previously unconnected components.
    # To connect all targets, we just run Kruskal's until all targets are in
    # the same component.
    
    total_cost = 0
    edges_count = 0
    
    # To ensure all targets are connected, we can treat all non-target nodes
    # as nodes that can be used to bridge targets.
    for w, u, v in edges:
        if find(u) != find(v):
            union(u, v)
            total_cost += w
            
            # Check if all targets are now in the same component
            root = find(targets[0])
            all_connected = True
            for i in range(1, F):
                if find(targets[i]) != root:
                    all_connected = False
                    break
            if all_connected:
                break
                
    print(total_cost)

solve()
```
