```python
import sys

# The problem states E = N - 1 and the graph is connected, so it's a tree.
# To visit all critical rooms, the inspector must traverse all edges on the 
# unique paths between any two critical rooms.
# The set of edges that must be traversed is the union of all paths between 
# any pair of critical rooms. This is equivalent to the edges in the 
# minimal subtree that contains all critical rooms.
# The total cost is the sum of weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    critical_rooms = set(map(int, input_data[3:3+F]))
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the sum of weights of the minimal subtree containing all critical nodes:
    # We can use a post-order traversal (DFS).
    # A subtree rooted at u contains a critical node if:
    # 1. u itself is critical.
    # 2. Any child of u is the root of a subtree containing a critical node.
    # An edge (u, v) with weight w is in the minimal subtree if and only if
    # the subtree at v contains at least one critical node AND the rest of the 
    # tree (outside the subtree at v) contains at least one critical node.
    
    has_critical = [False] * (N + 1)
    for r in critical_rooms:
        has_critical[r] = True
        
    total_weight = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_to_parent_w = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_w[v] = w
                stack.append(v)
                
    # Process in reverse topological order (post-order)
    for u in reversed(order):
        if has_critical[u]:
            if parent[u] != 0:
                has_critical[parent[u]] = True
                total_weight += edge_to_parent_w[u]
        else:
            # If u is not critical, but has a child that is, 
            # then u effectively becomes "critical" for its parent
            if has_critical[u] and parent[u] != 0:
                total_weight += edge_to_parent_w[u]
                
    print(total_weight)

solve()
```
