```python
import sys

# The problem states E = N - 1 and the graph is a tree.
# We need to find the minimum weight of the subtree that connects all vital chambers.
# This is equivalent to the sum of weights of all edges that lie on the path between
# any two vital chambers.
# An edge (u, v) with weight w is part of the required subtree if and only if
# the removal of this edge splits the tree into two components, each containing
# at least one vital chamber.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    vital_chambers = set(map(int, input_data[3:3+F]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many vital chambers are in the subtree rooted at each node
    # We use iterative DFS to avoid recursion depth issues
    vital_count = [0] * (N + 1)
    parent = [0] * (N + 1)
    edge_weight_to_parent = [0] * (N + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in vital_chambers:
            vital_count[u] = 1
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for i in range(N - 1, -1, -1):
        u = order[i]
        if parent[u] != -1:
            p = parent[u]
            vital_count[p] += vital_count[u]
            # If the subtree at u contains some but not all vital chambers,
            # the edge to the parent must be included.
            if 0 < vital_count[u] < F:
                total_weight += edge_weight_to_parent[u]
                
    print(total_weight)

solve()
```
