import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def main():
    N, E, F = map(int, sys.stdin.readline().split())
    vital = set(map(int, sys.stdin.readline().split()))
    
    if F <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(E):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    # Sort edges by weight
    edges.sort()
    
    # Create a mapping from node to component id for vital nodes
    # Initially each vital node is its own component
    vital_nodes = list(vital)
    node_to_vital_index = {node: i for i, node in enumerate(vital_nodes)}
    num_vital = len(vital)
    
    # Union-Find for the original graph
    parent = list(range(N + 1))
    rank = [0] * (N + 1)
    
    total_cost = 0
    components_merged = 0
    
    for w, u, v in edges:
        # Check if u and v are in different components
        if find(parent, u) != find(parent, v):
            # Merge the components
            union(parent, rank, u, v)
            total_cost += w
            
            # Check if we've connected all vital nodes
            # We do this by checking if all vital nodes are in the same component
            # To optimize, we track how many vital pairs we've connected
            
            # A more efficient approach:
            # When we connect two components, if both had vital nodes,
            # we reduce the number of separate vital groups by 1
            
            # But let's use a simpler method:
            # After each union, check if all vital nodes have the same root
            
            roots = set()
            for node in vital:
                roots.add(find(parent, node))
            
            if len(roots) == 1:
                break
    
    print(total_cost)

if __name__ == "__main__":
    main()
