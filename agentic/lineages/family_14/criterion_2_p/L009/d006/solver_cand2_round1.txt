```python
import sys

# The problem asks for the minimum weight of a subset of edges in a tree
# such that all F output variables are in the same connected component.
# This is equivalent to finding the weight of the minimal subtree that
# connects all F output variables.
# In a tree, the minimal subtree connecting a set of nodes is the union
# of all paths between these nodes.
# An edge (u, v) with weight w is part of this minimal subtree if and only if
# the removal of this edge splits the tree into two components, each of which
# contains at least one of the F output variables.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    outputs = set(map(int, input_data[3:3+F]))
    adj = [[] for _ in range(N + 1)]
    edges = []
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # count_outputs[u] will store the number of output variables in the subtree
    # rooted at u when performing a DFS.
    count_outputs = [0] * (N + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in outputs:
            count_outputs[u] = 1
            
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        count_outputs[p] += count_outputs[u]
        # If the subtree at u contains some but not all output variables,
        # the edge to the parent must be included.
        if 0 < count_outputs[u] < F:
            total_weight += edge_to_parent_weight[u]
            
    print(total_weight)

if __name__ == '__main__':
    solve()
```
