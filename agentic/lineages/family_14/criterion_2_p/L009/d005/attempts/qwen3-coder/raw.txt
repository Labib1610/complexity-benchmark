```python
import sys
from collections import defaultdict
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    N, E, F = map(int, sys.stdin.readline().split())
    targets = list(map(int, sys.stdin.readline().split()))
    target_set = set(targets)
    
    edges = []
    for _ in range(E):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    if F <= 1:
        print(0)
        return
    
    # Sort edges by weight
    edges.sort()
    
    # Union-Find data structure
    parent = list(range(N + 1))
    rank = [0] * (N + 1)
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    # Kruskal's algorithm variant: we want to connect all target nodes
    # We'll use a modified approach where we contract target nodes into components
    # and only care about connecting all targets
    
    # First, let's build the graph and see what targets are already connected
    # But more efficient approach: use Kruskal but stop when all targets are connected
    
    total_cost = 0
    components_with_targets = set()
    
    # Initialize: each target node is its own component
    for t in targets:
        components_with_targets.add(find(t))
    
    # If all targets are already in the same component
    if len(components_with_targets) == 1:
        print(0)
        return
    
    for w, u, v in edges:
        pu, pv = find(u), find(v)
        if pu != pv:
            # Check if either component has targets
            had_targets_u = pu in components_with_targets
            had_targets_v = pv in components_with_targets
            
            # Union the components
            union(u, v)
            
            # The new root after union
            new_root = find(u)
            
            # Update our tracking of components with targets
            if had_targets_u or had_targets_v:
                # Remove old components
                if had_targets_u:
                    components_with_targets.discard(pu)
                if had_targets_v:
                    components_with_targets.discard(pv)
                # Add new component
                components_with_targets.add(new_root)
            
            total_cost += w
            
            # If all targets are now connected
            if len(components_with_targets) == 1:
                break
    
    print(total_cost)

threading.Thread(target=main).start()
```
