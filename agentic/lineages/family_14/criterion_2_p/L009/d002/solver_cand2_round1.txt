```python
import sys

# The problem asks for the minimum weight of a subset of edges that connects all critical nodes.
# Since the graph is a tree (N nodes, N-1 edges), there is only one unique path between any two nodes.
# The set of edges that connects all critical nodes is the union of all paths between every pair of critical nodes.
# This is equivalent to the set of edges that lie on the unique path between the "first" and "last" critical node
# in a DFS traversal order, or more simply, the edges that belong to the minimal subtree containing all critical nodes.
# In a tree, an edge belongs to the minimal subtree connecting a set of nodes S if and only if
# removing that edge splits the tree into two components, both of which contain at least one node from S.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+F]))
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if F <= 1:
        print(0)
        return

    # subtree_critical[u] stores the number of critical nodes in the subtree rooted at u
    subtree_critical = [0] * (N + 1)
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in critical_nodes:
            subtree_critical[u] += 1
        if parent[u] != 0:
            subtree_critical[parent[u]] += subtree_critical[u]
            # If this edge separates the tree into two parts both containing critical nodes
            if 0 < subtree_critical[u] < F:
                total_weight += edge_to_parent_weight[u]
                
    print(total_weight)

solve()
```
