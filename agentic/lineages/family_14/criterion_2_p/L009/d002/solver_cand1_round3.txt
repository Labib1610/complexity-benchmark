```python
import sys

# The problem states E = N - 1 and the system is fully connected.
# This means the graph is a tree.
# The problem asks for the minimum cost to connect all F priority hubs.
# In a tree, there is exactly one unique path between any two nodes.
# The union of all paths between all pairs of priority hubs forms a subtree.
# This subtree is the unique minimal connected subgraph containing all F hubs.
# We need to sum the weights of all edges that belong to this subtree.
# An edge belongs to this subtree if and only if removing it splits the tree
# into two components, each containing at least one priority hub.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    hubs = set(map(int, input_data[3:3+F]))
    
    adj = [[] for _ in range(N + 1)]
    edges = []
    idx = 3 + F
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w, _))
        adj[v].append((u, w, _))
        edges.append((u, v, w))
        idx += 3
        
    if F <= 1:
        print(0)
        return

    # subtree_hubs[u] will store the number of hubs in the subtree rooted at u
    subtree_hubs = [0] * (N + 1)
    edge_weights = [0] * E
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (N + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w, edge_idx in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_weights[edge_idx] = w
                stack.append(v)
                
    # Count hubs in each subtree
    for u in order:
        if u in hubs:
            subtree_hubs[u] += 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if parent[u] != -1:
            p = parent[u]
            subtree_hubs[p] += subtree_hubs[u]
            # If this subtree contains some but not all hubs, the edge to parent is needed
            if 0 < subtree_hubs[u] < F:
                # Find the weight of the edge connecting u to parent
                for v, w, edge_idx in adj[u]:
                    if v == p:
                        total_cost += w
                        break
                        
    print(total_cost)

solve()
```
