```python
import sys

# The problem states that the network is a tree (E = N - 1).
# We need to find the sum of weights of all edges that lie on the unique paths
# between any two primary hubs.
# An edge is part of the Steiner Tree if and only if it separates the set of
# primary hubs into two non-empty sets.
# Since it's a tree, removing an edge splits the tree into two components.
# If both components contain at least one primary hub, the edge must be included.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    E = int(input_data[1])
    F = int(input_data[2])
    
    if F <= 1:
        print(0)
        return
    
    primary_hubs = set()
    idx = 3
    for _ in range(F):
        primary_hubs.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(N + 1)]
    edges = []
    for _ in range(E):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many primary hubs are in each subtree
    # Root the tree at node 1
    subtree_hubs = [0] * (N + 1)
    parent = [0] * (N + 1)
    edge_to_parent_weight = [0] * (N + 1)
    order = []
    stack = [1]
    visited = [False] * (N + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in primary_hubs:
            subtree_hubs[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(N - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if subtree_hubs[u] > 0 and subtree_hubs[u] < F:
            total_cost += edge_to_parent_weight[u]
        subtree_hubs[p] += subtree_hubs[u]
        
    print(total_cost)

solve()
```
