```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes (priority languages)
# in a tree structure. Since the original graph is a tree (n nodes, n-1 edges, connected),
# the subset of edges required to connect all priority languages is the unique path
# between every pair of priority languages.
# This is equivalent to the union of all paths between the priority languages.
# In a tree, this is the minimal subtree containing all priority languages.
# We can solve this by rooting the tree at an arbitrary node (e.g., node 1)
# and checking for each edge whether it lies on a path between two priority languages.
# An edge (u, v) where v is a child of u lies on such a path if and only if
# the subtree rooted at v contains at least one priority language AND
# the rest of the tree (outside the subtree at v) contains at least one priority language.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # DFS to find subtree counts of priority nodes
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    subtree_priority_count = [0] * (n + 1)
    for node in priority_nodes:
        subtree_priority_count[node] = 1
        
    # Process in reverse topological order
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            subtree_priority_count[parent[u]] += subtree_priority_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # Edge is between i and parent[i]
        # It is part of the minimal subtree if the subtree at i has > 0 
        # and < t priority nodes.
        count = subtree_priority_count[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
