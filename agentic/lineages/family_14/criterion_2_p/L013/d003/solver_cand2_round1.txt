import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_langs = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    # We need to connect all priority languages.
    # This is equivalent to finding the Steiner Tree for the set of priority languages.
    # However, the problem states m = n - 1 and the graph is a tree (since it connects n nodes with n-1 edges).
    # If the graph is a tree, the Steiner Tree is simply the unique path between all priority nodes.
    
    adj = [[] for _ in range(n + 1)]
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Find the unique path connecting all priority languages in the tree.
    # A node is part of the Steiner Tree if it lies on a path between two priority nodes.
    # This is equivalent to saying the subtree rooted at the node contains at least one priority node
    # and the rest of the tree contains at least one priority node.
    
    has_priority = [False] * (n + 1)
    for p in priority_langs:
        has_priority[p] = True
        
    total_priority = t
    edge_in_steiner = [False] * m
    
    # Count priority nodes in each subtree
    subtree_priority = [0] * (n + 1)
    
    def dfs_count(u, p):
        count = 1 if has_priority[u] else 0
        for v, w in adj[u]:
            if v != p:
                count += dfs_count(v, u)
        subtree_priority[u] = count
        return count
    
    dfs_count(1, -1)
    
    # An edge (u, v) is in the Steiner Tree if the number of priority nodes in the 
    # component formed by removing the edge is neither 0 nor total_priority.
    
    total_weight = 0
    for i in range(m):
        w, u, v = edges[i]
        # Determine which node is the child in the rooted tree
        child = u if subtree_priority[u] < subtree_priority[v] else v
        count = subtree_priority[child]
        if count > 0 and count < total_priority:
            total_weight += w
            
    print(total_weight)

solve()
