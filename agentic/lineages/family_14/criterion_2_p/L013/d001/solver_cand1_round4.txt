```python
import sys

# The problem states the graph is a tree (n nodes, n-1 edges, no alternative routes).
# We need to find the minimum cost to connect all critical workstations.
# In a tree, the set of edges required to connect a subset of nodes is exactly
# the union of all paths between any two nodes in that subset.
# This is equivalent to the edges on the minimal subtree containing all critical nodes.
# An edge is part of this subtree if and only if the two components it separates
# both contain at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        critical[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use DFS to count how many critical nodes are in the subtree of each node
    # Since n is up to 200,000, use iterative DFS to avoid recursion depth issues
    subtree_critical = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if critical[u]:
            subtree_critical[u] += 1
        if parent[u] != 0:
            subtree_critical[parent[u]] += subtree_critical[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # Edge between i and parent[i] is needed if the subtree at i
        # contains some but not all critical nodes.
        count = subtree_critical[i]
        if count > 0 and count < t:
            total_cost += edge_weight[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
