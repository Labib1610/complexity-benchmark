```python
import sys

# The problem states that the translators form a tree structure (n nodes, n-1 edges, 
# connected). We need to find the minimum cost to connect a subset of t target 
# nodes. This is equivalent to finding the weight of the minimal subtree that 
# contains all t target nodes.
# This is a classic problem: the edges that must be included are exactly those 
# that lie on the paths between any two target nodes.
# An edge (u, v) with weight w is part of the minimal subtree if and only if 
# the subtree formed by removing (u, v) has at least one target node in each 
# component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_targets[u] will store the number of target nodes in the subtree 
    # rooted at u (when rooted at 1).
    subtree_targets = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in targets:
            subtree_targets[u] = 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        if subtree_targets[u] > 0 and subtree_targets[u] < t:
            total_cost += parent_edge_weight[u]
        
        p = parent[u]
        subtree_targets[p] += subtree_targets[u]
        
    print(total_cost)

solve()
```
