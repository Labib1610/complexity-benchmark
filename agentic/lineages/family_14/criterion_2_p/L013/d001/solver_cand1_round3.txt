```python
import sys

# The problem states the structure is a tree (n workstations, n-1 links, 
# exactly one path between any two). We need to find the minimum cost to 
# connect all critical workstations. This is equivalent to finding the 
# sum of weights of all edges that lie on the unique path between any two 
# critical workstations. An edge is part of this set if and only if the 
# subtree formed by removing that edge contains at least one critical 
# workstation on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical = set()
    idx = 3
    for _ in range(t):
        critical.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use DFS to count critical nodes in each subtree
    # Since n can be up to 200,000, use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    critical_count = [0] * (n + 1)
    for node in critical:
        critical_count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If this subtree contains some but not all critical nodes,
            # the edge to the parent must be included.
            if 0 < critical_count[u] < t:
                total_cost += edge_to_parent_weight[u]
            critical_count[p] += critical_count[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
