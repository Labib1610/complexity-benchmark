```python
import sys

# The problem states that the pool of m translators forms a tree (n nodes, n-1 edges).
# We need to find the minimum cost to connect a subset of t target languages.
# This is equivalent to finding the weight of the Steiner Tree for the given t nodes.
# In a tree, the Steiner Tree is the unique minimal subtree connecting all target nodes.
# This subtree consists of all edges that lie on the simple path between any two target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the edges in the Steiner Tree:
    # 1. Root the tree at an arbitrary target node.
    # 2. For each node, determine if its subtree contains any target nodes.
    # 3. An edge (u, v) where v is a child of u is in the Steiner Tree if and only if
    #    the subtree rooted at v contains at least one target node AND the rest of the tree
    #    (outside v's subtree) contains at least one target node.
    
    root = next(iter(targets))
    has_target = [False] * (n + 1)
    for node in targets:
        has_target[node] = True
        
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    # Post-order traversal to propagate target existence
    for u in reversed(order):
        if u != root:
            if has_target[u]:
                has_target[parent[u]] = True
                
    total_weight = 0
    # An edge (u, parent[u]) is part of the Steiner Tree if the subtree at u
    # contains at least one target AND the rest of the tree contains at least one target.
    # Since we rooted at a target, the "rest of the tree" contains a target if
    # the total number of targets is greater than the number of targets in u's subtree.
    
    target_count_in_subtree = [0] * (n + 1)
    for node in targets:
        target_count_in_subtree[node] = 1
        
    for u in reversed(order):
        if u != root:
            target_count_in_subtree[parent[u]] += target_count_in_subtree[u]
            
    for u in range(1, n + 1):
        if u != root:
            if target_count_in_subtree[u] > 0 and target_count_in_subtree[u] < t:
                total_weight += parent_edge_weight[u]
                
    print(total_weight)

solve()
```
