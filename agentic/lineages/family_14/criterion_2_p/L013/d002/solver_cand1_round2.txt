```python
import sys

# The problem states that the structure of the available translator pool 
# ensures that for any pair of languages, there exists exactly one valid 
# sequence of translators connecting them. This means the graph is a tree.
# We need to find the minimum cost to connect all t priority languages.
# In a tree, the set of edges required to connect a subset of nodes is 
# exactly the union of all paths between any two nodes in that subset.
# This is equivalent to the edges in the minimal subtree containing all 
# priority nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that lie on a path between any two priority nodes.
    # An edge (u, v) with weight w is part of the minimal subtree if and only if
    # the subtree rooted at v (when considering the edge (u, v)) contains at least
    # one priority node AND the rest of the tree contains at least one priority node.
    
    is_priority = [False] * (n + 1)
    for node in priority_nodes:
        is_priority[node] = True
        
    count_priority = [0] * (n + 1)
    
    # DFS to count priority nodes in subtrees
    order = []
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_priority[u]:
            count_priority[u] += 1
        if parent[u] != 0:
            count_priority[parent[u]] += count_priority[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # Edge is between i and parent[i]
        # It is needed if count_priority[i] > 0 and count_priority[i] < t
        if 0 < count_priority[i] < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

solve()
```
