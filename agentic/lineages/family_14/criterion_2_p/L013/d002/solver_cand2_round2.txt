```python
import sys

# The problem states that the m = n - 1 translators form a tree structure
# (minimally comprehensive). We need to find the minimum cost to connect
# a subset of t target languages.
# This is equivalent to finding the weight of the Steiner Tree for the
# given t nodes in the tree.
# In a tree, the Steiner Tree is simply the union of all paths between
# the target nodes. This is the minimal subtree containing all target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the minimal subtree containing all target nodes:
    # 1. Root the tree arbitrarily (e.g., at node 1).
    # 2. For each node, determine if its subtree contains any target nodes.
    # 3. An edge (u, v) where v is a child of u is part of the Steiner Tree
    #    if and only if the subtree at v contains at least one target node
    #    AND the rest of the tree (outside v's subtree) contains at least one target node.
    
    has_target = [False] * (n + 1)
    for node in targets:
        has_target[node] = True
        
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    subtree_has_target = [False] * (n + 1)
    for i in range(1, n + 1):
        if has_target[i]:
            subtree_has_target[i] = True
            
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            if subtree_has_target[u]:
                subtree_has_target[parent[u]] = True
                
    total_weight = 0
    # An edge (u, parent[u]) is included if subtree_has_target[u] is True
    # AND (total_targets - count_in_subtree_u) > 0
    count_in_subtree = [0] * (n + 1)
    for i in range(1, n + 1):
        if has_target[i]:
            count_in_subtree[i] = 1
            
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            count_in_subtree[parent[u]] += count_in_subtree[u]
            
    for i in range(2, n + 1):
        if subtree_has_target[i] and (t - count_in_subtree[i] > 0):
            total_weight += edge_to_parent_weight[i]
            
    print(total_weight)

solve()
```
