```python
import sys

# The problem states the pool of translators forms a non-redundant structure
# that connects all languages. This means the translators form a tree (a spanning tree).
# We are given a set of target languages and need to find the minimum cost to
# connect all target languages.
# In a tree, the unique path between any two nodes is fixed. To connect a set
# of nodes in a tree, we must include all edges that lie on the paths between
# any pair of target nodes. This is equivalent to the Steiner Tree problem,
# but since the graph is a tree, the Steiner Tree is simply the minimal subtree
# containing all target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n_languages = int(input_data[0])
    n_translators = int(input_data[1])
    n_targets = int(input_data[2])
    
    target_languages = set(map(int, input_data[3:3+n_targets]))
    
    if n_targets <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n_languages + 1)]
    edges = []
    idx = 3 + n_targets
    for _ in range(n_translators):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # To find the minimal subtree containing all target nodes:
    # 1. Calculate the degree of each node in the original tree.
    # 2. Repeatedly prune leaves that are not target nodes.
    
    degree = [0] * (n_languages + 1)
    for i in range(1, n_languages + 1):
        degree[i] = len(adj[i])
        
    is_target = [False] * (n_languages + 1)
    for t in target_languages:
        is_target[t] = True
        
    queue = [i for i in range(1, n_languages + 1) if degree[i] == 1 and not is_target[i]]
    
    removed = [False] * (n_languages + 1)
    edge_removed = [False] * n_translators
    
    # Map edges to their index to mark them as removed
    edge_map = {}
    for i, (u, v, w) in enumerate(edges):
        if u > v: u, v = v, u
        edge_map[(u, v)] = i
        
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        removed[u] = True
        for v, w in adj[u]:
            if not removed[v]:
                # Mark edge as removed
                e_u, e_v = (u, v) if u < v else (v, u)
                edge_removed[edge_map[(e_u, e_v)]] = True
                
                degree[v] -= 1
                if degree[v] == 1 and not is_target[v]:
                    queue.append(v)
                    
    total_fee = 0
    for i in range(n_translators):
        if not edge_removed[i]:
            total_fee += edges[i][2]
            
    print(total_fee)

solve()
```
