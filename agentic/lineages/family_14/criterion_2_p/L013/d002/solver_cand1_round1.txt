import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_languages = set()
    idx = 3
    for _ in range(t):
        priority_languages.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the smallest subtree that contains all priority languages.
    # This is the union of all paths between any two priority languages.
    # A node is in this subtree if it is on a path between two priority nodes.
    # An edge is in this subtree if it lies on a path between two priority nodes.
    
    is_priority = [False] * (n + 1)
    for p in priority_languages:
        is_priority[p] = True
        
    # subtree_has_priority[u] is true if the subtree rooted at u contains a priority node
    subtree_has_priority = [False] * (n + 1)
    total_fee = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_priority[u]:
            subtree_has_priority[u] = True
        
        if subtree_has_priority[u] and parent[u] != 0:
            total_fee += edge_weight[u]
            subtree_has_priority[parent[u]] = True
            
    print(total_fee)

solve()
