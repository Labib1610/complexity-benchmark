```python
import sys

# The problem asks for the minimum cost to connect all target nodes in a tree.
# Since the graph is a tree (n nodes, n-1 edges, connected), the set of edges
# that connect a subset of nodes is the union of paths between them.
# This is equivalent to finding the Steiner Tree for the target set in the given tree.
# In a tree, the Steiner Tree is the minimal subtree containing all target nodes.
# The edges in this subtree are exactly those edges that lie on the path between
# at least one pair of target nodes.
# An edge (u, v) is in the Steiner Tree if and only if the two components formed
# by removing (u, v) both contain at least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_targets[u] stores the number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    # BFS to establish parent-child relationships
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in targets:
            subtree_targets[u] = 1
            
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == 1:
            continue
        # If the subtree at u contains some but not all targets, 
        # the edge to its parent must be included.
        if 0 < subtree_targets[u] < t:
            total_cost += parent_edge_weight[u]
        
        # Propagate target count to parent
        if parent[u] != 0:
            subtree_targets[parent[u]] += subtree_targets[u]
            
    print(total_cost)

solve()
```
