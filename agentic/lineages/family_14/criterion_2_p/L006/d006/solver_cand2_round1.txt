```python
import sys

def solve():
    # The problem asks for the minimum cost to generate all transpositions (i, j)
    # for i, j in the target set T.
    # A set of transpositions generates the symmetric group on a set of elements
    # if and only if the graph formed by these transpositions as edges is connected.
    # Specifically, to generate all transpositions (i, j) for i, j in T,
    # the chosen transpositions must connect all elements in T into a single
    # connected component in the graph where edges are the transpositions.
    # If T has size 1, the cost is 0.
    # If T has size > 1, we need to connect all nodes in T using the given edges.
    # This is equivalent to finding the Minimum Spanning Tree of the graph
    # where we want to connect all nodes in T.
    # This is the Steiner Tree problem, but since m = n-1 and the graph is a tree
    # (because m = n-1 and the whole graph is connected), the Steiner Tree
    # is simply the unique path in the tree connecting all nodes in T.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_elements = set(map(int, input_data[3:3+t]))
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Since the graph is a tree, the Steiner tree is the union of paths
    # between all nodes in T. This is the minimal subtree containing all nodes in T.
    # We can find this by pruning all leaves that are not in T, repeatedly.
    
    degree = [0] * (n + 1)
    for i in range(1, n + 1):
        degree[i] = len(adj[i])
        
    is_in_steiner = [True] * (n + 1)
    queue = [i for i in range(1, n + 1) if degree[i] == 1 and i not in target_elements]
    
    total_cost = 0
    for u, v_w in enumerate(adj):
        for v, w in v_w:
            total_cost += w
    total_cost //= 2
    
    # Pruning
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        is_in_steiner[u] = False
        for v, w in adj[u]:
            if is_in_steiner[v]:
                total_cost -= w
                degree[v] -= 1
                if degree[v] == 1 and v not in target_elements:
                    queue.append(v)
                    
    print(total_cost)

solve()
```
