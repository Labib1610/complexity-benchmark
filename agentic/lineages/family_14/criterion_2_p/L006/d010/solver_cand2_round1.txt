```python
import sys

# The problem asks for the minimum cost to ensure that for all j, k in T,
# b_j - b_k is in the span of the selected vectors.
# Each vector v_i = b_{u_i} - b_{v_i} can be viewed as an edge between nodes u_i and v_i.
# The condition "b_j - b_k is in the span" is equivalent to saying that
# there is a path between j and k in the graph formed by the selected edges.
# Since we need this for all pairs in T, all nodes in T must belong to the same
# connected component in the graph formed by the selected edges.
# We are given m = n - 1 edges, and the graph of all m edges is a tree (since it spans n-1 edges and has dimension n-1).
# We need to find the minimum cost to connect all nodes in T.
# This is the Steiner Tree problem on a tree, which is simply the sum of weights
# of all edges on the unique paths between nodes in T.
# In a tree, the union of paths between all nodes in T is the minimal subtree
# containing all nodes in T.
# An edge is part of this subtree if and only if the edge splits the tree into
# two components, both of which contain at least one node from T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many target nodes are in each subtree
    subtree_t_count = [0] * (n + 1)
    
    # DFS to compute subtree_t_count
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for node in target_nodes:
        subtree_t_count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # Find weight of edge (u, p)
            # Since it's a tree, we can store it during the first pass
            subtree_t_count[p] += subtree_t_count[u]
            
    # Now calculate cost: an edge (u, p) is included if 0 < subtree_t_count[u] < t
    edge_weights = {}
    for u, v, w in edges:
        edge_weights[tuple(sorted((u, v)))] = w
        
    total_cost = 0
    for i in range(2, n + 1):
        u = i
        p = parent[u]
        count = subtree_t_count[u]
        if count > 0 and count < t:
            total_cost += edge_weights[tuple(sorted((u, p)))]
            
    print(total_cost)

solve()
```
