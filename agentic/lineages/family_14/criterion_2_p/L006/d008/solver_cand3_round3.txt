```python
import sys

# The problem states that the network is a tree (n assets, n-1 channels, 
# and licensing all channels makes it a single connected component).
# We need to select a subset of edges such that all core assets are in the 
# same connected component.
# In a tree, the unique path between any two nodes is fixed.
# To connect all core assets, we must include all edges that lie on the 
# unique paths between any two core assets.
# This is equivalent to finding the minimal subtree that contains all core nodes.
# The total cost is the sum of the weights of all edges that are part of 
# the paths connecting the core nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core_assets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find all edges that are on the path between any two core nodes.
    # An edge (u, v) with weight w is part of the minimal subtree if and only if
    # the subtree rooted at u (when the edge is removed) contains at least one 
    # core node, and the subtree rooted at v contains at least one core node.
    
    core_count_in_subtree = [0] * (n + 1)
    
    # DFS to count core nodes in subtrees
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for node in core_assets:
        core_count_in_subtree[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            core_count_in_subtree[p] += core_count_in_subtree[u]
            # If the subtree at u contains some core nodes, and the rest of the 
            # tree also contains some core nodes, this edge must be included.
            if 0 < core_count_in_subtree[u] < t:
                total_cost += edge_weight_to_parent[u]
                
    print(total_cost)

solve()
```
