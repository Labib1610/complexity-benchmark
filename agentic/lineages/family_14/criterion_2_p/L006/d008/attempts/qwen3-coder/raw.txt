import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal(n, edges, must_connect=None):
    parent = list(range(n))
    rank = [0] * n
    cost = 0
    edges_used = 0
    
    if must_connect:
        for u, v, w in edges:
            if (u, v) in must_connect or (v, u) in must_connect:
                if union(parent, rank, u, v):
                    cost += w
                    edges_used += 1
    
    for u, v, w in sorted(edges, key=lambda x: x[2]):
        if union(parent, rank, u, v):
            cost += w
            edges_used += 1
            if edges_used == n - 1:
                break
                
    return cost if edges_used == n - 1 else float('inf')

def main():
    import sys

    t = int(data[idx+2])
    idx += 3
    
    T = []
    for _ in range(t):
        T.append(int(data[idx]) - 1)  # Convert to 0-indexed
        idx += 1
    
    edges = []
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        w = int(data[idx+2])
        idx += 3
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    # For the induced map ι_* : H_0(T; ℝ) → H_0(X; ℝ) to have rank 1,
    # All vertices in T must be connected in X.
    # So we need to find the minimum spanning tree that connects all vertices in T.
    
    # Create a graph with extra node representing the "terminal" node
    # Connect all nodes in T to this terminal node with weight 0
    # Then find MST of this new graph
    
    extended_edges = edges[:]
    terminal = n  # New vertex
    
    # Connect all vertices in T to the terminal with 0 cost
    for v in T:
        extended_edges.append((v, terminal, 0))
    
    # Now find MST of this extended graph
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    cost = 0
    edges_used = 0
    
    for u, v, w in sorted(extended_edges, key=lambda x: x[2]):
        if union(parent, rank, u, v):
            cost += w
            edges_used += 1
            if edges_used == n:  # MST for n+1 vertices needs n edges
                break
    
    print(cost)

if __name__ == "__main__":
    main()
