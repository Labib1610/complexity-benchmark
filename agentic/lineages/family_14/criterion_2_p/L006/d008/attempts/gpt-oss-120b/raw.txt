```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))  # = n-1
    t = int(next(it))
    T = set(int(next(it)) for _ in range(t))
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    # root the tree at 1 (any node)
    parent = [0] * (n + 1)
    pw = [0] * (n + 1)          # weight to parent
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v == parent[u]:
                continue
            parent[v] = u
            pw[v] = w
            stack.append(v)

    cnt = [0] * (n + 1)
    for v in order[::-1]:
        cnt[v] = 1 if v in T else 0
        for nb, _ in adj[v]:
            if parent[nb] == v:   # child
                cnt[v] += cnt[nb]

    total = 0
    for v in range(2, n + 1):
        if cnt[v] > 0 and t - cnt[v] > 0:
            total += pw[v]

    print(total)

if __name__ == "__main__":
    solve()
```
