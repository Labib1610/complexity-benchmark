```python
import sys

# The problem asks for a subset of edges C such that the rank of the image of 
# the induced map i_*: H_0(T) -> H_0(X) is 1.
# H_0(X) is the vector space spanned by the connected components of X.
# The map i_* sends each vertex in T to the component it belongs to in X.
# The rank of the image is 1 if and only if all vertices in T belong to the 
# same connected component in X.
# Since m = n - 1 and the full graph is a tree, any subset C that connects 
# all vertices in T must be a subgraph that contains a path between any two 
# vertices in T.
# To minimize the total weight, we need to find the minimum weight set of edges 
# that connects all vertices in T. This is the Steiner Tree problem.
# For a tree, the Steiner Tree is the unique minimal subtree connecting all 
# vertices in T, which is the union of all paths between vertices in T.
# This can be solved by rooting the tree at an arbitrary node (say T[0]) 
# and including an edge if the subtree below it contains at least one node from T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    T = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Root the tree at an arbitrary node in T
    root = next(iter(T))
    
    # parent_edge[v] = weight of edge to parent
    parent_edge = [0] * (n + 1)
    has_T = [False] * (n + 1)
    for node in T:
        has_T[node] = True
        
    order = []
    stack = [root]
    parent = [0] * (n + 1)
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge[v] = w
                stack.append(v)
                
    total_weight = 0
    for u in reversed(order):
        if u == root:
            continue
        if has_T[u]:
            total_weight += parent_edge[u]
            has_T[parent[u]] = True
            
    print(total_weight)

solve()
```
