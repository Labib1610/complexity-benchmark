```python
import sys

# The problem asks for the minimum cost to select a set of edges such that
# if any two nodes in T have different states, there exists an edge in the set
# with different states. This is equivalent to saying that the set of edges
# must "cut" every path between any two nodes in T.
# In a tree, the set of edges that separates all nodes in T into different
# components is the set of edges that lie on the paths between nodes in T.
# Specifically, if we consider the minimal subtree containing all nodes in T
# (the Steiner Tree of T), any edge in this subtree must be included to
# ensure that if we assign states such that the tree is partitioned, the
# edge is "cut".
# Actually, the condition "if there exist any two critical elements i, j in T
# that are assigned different states, then your chosen subset C must contain
# at least one link where its two endpoints u and v are assigned different states"
# is satisfied if and only if the chosen edges C form a cut that separates
# all nodes in T from each other.
# In a tree, the minimal set of edges whose removal separates all nodes in T
# is exactly the set of edges in the Steiner Tree of T.
# The cost is the sum of weights of all edges in the Steiner Tree of T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = set()
    idx = 3
    for _ in range(t):
        critical.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # has_critical[u] is true if the subtree rooted at u contains at least one critical node
    has_critical = [False] * (n + 1)
    # count_critical[u] is the number of critical nodes in the subtree rooted at u
    count_critical = [0] * (n + 1)
    
    for node in critical:
        has_critical[node] = True
        count_critical[node] = 1
        
    total_cost = 0
    
    # DFS to compute Steiner Tree edges
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u == 1: continue
        p = parent[u]
        if has_critical[u]:
            total_cost += edge_to_parent_weight[u]
            has_critical[p] = True
            count_critical[p] += count_critical[u]
            
    print(total_cost)

solve()
```
