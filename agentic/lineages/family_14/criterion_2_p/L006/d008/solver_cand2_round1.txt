```python
import sys

# The problem states that we have n basis elements and m = n-1 simplices.
# Since m = n-1 and the full set of simplices allows forming the boundary {x, y}
# for any x, y, the graph formed by the simplices is a tree.
# The boundary of a 1-chain is the symmetric difference of the edges in the chain.
# In a tree, there is a unique path between any two nodes u and v.
# The boundary of the set of edges on the unique path between u and v is exactly {u, v}.
# Thus, to satisfy the condition for all pairs {i, j} where i, j in T,
# we must include all edges that lie on the unique path between any two nodes in T.
# This is equivalent to finding the minimal subtree that contains all nodes in T.
# The cost is the sum of weights of all edges in this minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Root the tree at an arbitrary critical node
    root = next(iter(critical_nodes))
    
    # We need to find edges that are part of the path between any two nodes in T.
    # An edge (u, v) is in the minimal subtree if and only if the subtree 
    # rooted at the child node contains at least one critical node, 
    # and the rest of the tree contains at least one critical node.
    
    subtree_has_critical = [False] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in critical_nodes:
        subtree_has_critical[node] = True
        
    total_cost = 0
    # Process in reverse topological order to propagate critical status up
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == root:
            continue
        if subtree_has_critical[u]:
            total_cost += edge_to_parent_weight[u]
            subtree_has_critical[parent[u]] = True
            
    print(total_cost)

solve()
```
