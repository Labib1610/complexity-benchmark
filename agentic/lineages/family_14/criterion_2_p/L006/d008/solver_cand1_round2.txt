```python
import sys

# The problem asks for the minimum total weight of edges in a subset of the given 
# tree (since m = n-1 and the graph is connected) such that all target isotopes 
# are in the same connected component.
# This is equivalent to finding the Steiner Tree for the set of target isotopes.
# In a tree, the Steiner Tree is the unique minimal subtree connecting all target nodes.
# This subtree consists of all edges that lie on the simple path between any two 
# target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find all edges that are part of the path between any two target nodes.
    # An edge (u, v) with weight w is in the Steiner Tree if and only if the 
    # subtree rooted at v (when the edge is removed) contains at least one target 
    # and the rest of the tree contains at least one target.
    
    has_target = [False] * (n + 1)
    for node in targets:
        has_target[node] = True
        
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    # Process nodes in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            p, w = parent[u]
            if has_target[u]:
                total_weight += w
                has_target[p] = True
                
    print(total_weight)

solve()
```
