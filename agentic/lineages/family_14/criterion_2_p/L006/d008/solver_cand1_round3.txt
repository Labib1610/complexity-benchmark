```python
import sys

# The problem asks for the minimum cost to connect all nodes in T.
# Since m = n - 1 and the graph is a tree (implied by m = n - 1 and the 
# guarantee that all nodes are connected if all edges are used), 
# we are looking for the minimum weight subtree that contains all nodes in T.
# This is equivalent to the Steiner Tree problem on a tree, which is 
# simply the sum of weights of all edges that lie on the paths between 
# nodes in T. An edge is part of this set if and only if removing it 
# splits the tree into two components, both of which contain at least one node from T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many critical nodes are in the subtree rooted at each node
    # We use iterative DFS to avoid recursion depth issues
    subtree_critical_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in critical_nodes:
            subtree_critical_count[u] += 1
            
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        p = parent[u]
        subtree_critical_count[p] += subtree_critical_count[u]
        # If this edge separates the tree into two parts both containing critical nodes
        if 0 < subtree_critical_count[u] < t:
            total_weight += edge_to_parent_weight[u]
            
    print(total_weight)

solve()
```
