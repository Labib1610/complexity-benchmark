```python
import sys

# The problem states that the set of m procedures is a tree (m = n - 1, 
# and the graph is connected). We need to find the minimum cost to connect 
# all t critical sensors. In a tree, there is a unique path between any two 
# nodes. To connect a set of nodes in a tree, we must include all edges 
# that lie on the paths between any pair of critical nodes.
# This is equivalent to finding the minimal subtree that contains all 
# critical nodes. An edge belongs to this subtree if and only if the 
# subtree formed by removing that edge splits the critical nodes into 
# two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_sensors = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many critical nodes are in the subtree of each node.
    # Since it's a tree, we can root it at an arbitrary node (e.g., 1).
    # An edge (u, v) where v is a child of u is part of the minimal subtree
    # if the number of critical nodes in the subtree rooted at v is > 0
    # and < t.
    
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    critical_count = [0] * (n + 1)
    for node in critical_sensors:
        critical_count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order to propagate counts up
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != -1:
            p = parent[u]
            critical_count[p] += critical_count[u]
            # If this subtree contains some but not all critical nodes,
            # the edge to the parent must be included.
            if 0 < critical_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
