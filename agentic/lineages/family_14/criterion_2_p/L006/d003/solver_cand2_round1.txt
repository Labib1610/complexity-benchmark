```python
import sys

# The problem states that the m operations form a tree (since m = n-1).
# We need to find the minimum cost to connect all t priority sensors.
# This is equivalent to finding the weight of the minimal subtree that 
# contains all t priority sensors.
# In a tree, the minimal subtree containing a set of nodes is the union of 
# all paths between these nodes.
# An edge is part of this minimal subtree if and only if the two components 
# formed by removing that edge both contain at least one priority sensor.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_sensors = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w, _))
        adj[v].append((u, w, _))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_priority_count[u] stores how many priority sensors are in the 
    # subtree rooted at u (when rooted at an arbitrary node, say 1).
    subtree_priority_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    parent = [0] * (n + 1)
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w, edge_idx in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in priority_sensors:
            subtree_priority_count[u] += 1
        if parent[u] != 0:
            subtree_priority_count[parent[u]] += subtree_priority_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the subtree at i contains some priority sensors, but not all of them,
        # then the edge connecting i to its parent must be part of the minimal subtree.
        count = subtree_priority_count[i]
        if count > 0 and count < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
