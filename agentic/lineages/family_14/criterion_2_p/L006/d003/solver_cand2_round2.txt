```python
import sys

# The problem states that we have n sensors and m = n - 1 operations.
# This means the operations form a tree structure (a spanning tree of the n sensors).
# We are given a set of t priority sensors.
# We need to find the minimum cost to connect all t priority sensors in the tree.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of nodes in a tree, we must include all edges that lie on the 
# unique paths between any pair of these nodes.
# This is equivalent to finding the minimal subtree that contains all t nodes.
# The edges in this minimal subtree are exactly those edges that, when removed,
# split the tree into components such that at least one component contains 
# some priority nodes and another component contains the rest.
# More simply: an edge is part of the minimal subtree if and only if 
# both sides of the edge contain at least one priority node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_sensors = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w, _))
        adj[v].append((u, w, _))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_priority_count[u] will store how many priority nodes are in the subtree rooted at u
    subtree_priority_count = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w, _ in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in priority_sensors:
        subtree_priority_count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If this subtree contains some priority nodes but not all,
            # the edge to the parent must be included.
            if 0 < subtree_priority_count[u] < t:
                total_cost += edge_to_parent_weight[u]
            subtree_priority_count[p] += subtree_priority_count[u]
            
    print(total_cost)

solve()
```
