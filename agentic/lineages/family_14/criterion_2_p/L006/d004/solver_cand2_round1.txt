```python
import sys

# The problem states that the graph is a tree (n variants, n-1 reactions, 
# and the entire catalog reconciles all variants).
# We need to find the minimum cost to connect all priority variants.
# This is equivalent to finding the sum of weights of all edges that lie on 
# the unique path between any two priority variants.
# An edge is part of the path between two priority variants if and only if 
# the edge splits the tree into two components, each containing at least one 
# priority variant.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_variants = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many priority variants are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    subtree_priority_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in priority_variants:
            subtree_priority_count[u] += 1
        if parent[u] != 0:
            subtree_priority_count[parent[u]] += subtree_priority_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the edge (i, parent[i]) separates the tree such that both sides
        # have at least one priority variant, it must be included.
        count = subtree_priority_count[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
