```python
import sys

# The problem states that the m procedures form a minimal complete set (a tree).
# We are given n sensors and m = n-1 procedures, which means the graph is a tree.
# We need to find the minimum cost to connect all t target sensors.
# In a tree, there is a unique path between any two nodes.
# To connect a set of target nodes in a tree, we must include all edges that lie
# on the unique paths between any pair of target nodes.
# This is equivalent to finding the minimal subtree that contains all target nodes.
# The edges in this minimal subtree are exactly those edges that, when removed,
# split the tree into two components such that both components contain at least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w, _))
        adj[v].append((u, w, _))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_targets[u] will store the number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    edge_included = [False] * m
    
    # DFS to count targets in subtrees
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent = [(-1, -1)] * (n + 1) # (parent_node, edge_index)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w, edge_idx in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent[v] = (w, edge_idx)
                stack.append(v)
                
    for u in order:
        if u in targets:
            subtree_targets[u] += 1
            
    # Process in reverse topological order to propagate counts up
    for u in reversed(order):
        if u != 1:
            p = parent[u]
            subtree_targets[p] += subtree_targets[u]
            w, edge_idx = edge_to_parent[u]
            # If this edge separates the tree into two parts both containing targets
            if subtree_targets[u] > 0 and subtree_targets[u] < t:
                edge_included[edge_idx] = True
                
    total_cost = 0
    for i in range(m):
        if edge_included[i]:
            total_cost += edges[i][2]
            
    print(total_cost)

solve()
```
