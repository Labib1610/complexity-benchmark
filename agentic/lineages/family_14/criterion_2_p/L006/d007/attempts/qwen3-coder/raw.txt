def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

    # Since m = n-1 and activating all works, the graph is a tree
    
    # Use Kruskal's algorithm variant

    rank = [0] * (n + 1)

    total_cost = 0
    edges.sort(key=lambda x: x[2])  # Sort by weight
    
    critical_components = t  # Initially each critical node is its own component
    
    for u, v, w in edges:
        pu = u
        pv = v
        # Find roots
        while parent[pu] != pu:
            pu = parent[pu]
        while parent[pv] != pv:
            pv = parent[pv]
            
        if pu != pv:  # Different components
            # If both components contain critical nodes, we need this edge
            if critical_in_component[pu] > 0 and critical_in_component[pv] > 0:
                total_cost += w
                
            # Union the components
            if rank[pu] < rank[pv]:
                pu, pv = pv, pu
            parent[pv] = pu
            critical_in_component[pu] += critical_in_component[pv]
            critical_in_component[pv] = 0
            
            # Update rank
            if rank[pu] == rank[pv]:

    print(total_cost)
