```python
import sys
import heapq
from collections import defaultdict
import sys
sys.setrecursionlimit(1000000)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    target_sensors = set()
    for _ in range(t):
        target_sensors.add(int(data[idx]))
        idx += 1
    
    # Build graph
    graph = [[] for _ in range(n+1)]
    edges = []
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        graph[u].append((v, w))
        graph[v].append((u, w))
        edges.append((w, u, v))
    
    # If only one target sensor, cost is 0
    if t <= 1:
        print(0)
        return
    
    # Compute shortest paths from each target sensor
    # We'll use Dijkstra from each target sensor
    # But that's inefficient. Instead, we can do a multi-source Dijkstra
    # Or even better, since we need the minimum spanning tree that connects all targets,
    # we can use the fact that the optimal solution is the MST of the target sensors
    # in the metric closure (complete graph with shortest path distances).
    
    # However, there's a simpler way:
    # Since the original graph is a tree (m = n-1 and connected), 
    # the MST of the target nodes in the original tree IS the minimal set of edges
    # needed to connect all target nodes.
    # This is because in a tree, there's exactly one path between any two nodes.
    # So the minimal cost to connect all targets is the cost of the minimal subtree
    # that spans all target nodes. This is a classic "Steiner tree" problem on a tree,
    # which can be solved by:
    # 1. Finding the subtree that contains all target nodes
    # 2. The minimal cost is the sum of weights of all edges in that subtree
    
    # Let's find the minimal subtree that contains all target nodes.
    # We can do this by:
    # 1. Marking all target nodes
    # 2. Doing a DFS to find all nodes that are on paths between target nodes
    # 3. The minimal subtree is the set of edges connecting these nodes
    
    # Actually, in a tree, the minimal way to connect a subset of nodes
    # is to take the subtree induced by those nodes.
    # The cost is the sum of edge weights in that subtree.
    
    # Algorithm:
    # 1. Root the tree at any node
    # 2. For each node, compute if it's necessary (i.e., it's a target or has two subtrees containing targets)
    # 3. Count edges in the minimal subtree
    
    # Let's root the tree at node 1
    parent = [-1] * (n+1)
    children = [[] for _ in range(n+1)]
    edge_weight = [-1] * (n+1)  # edge_weight[u] is weight from u to its parent
    
    # Build rooted tree with DFS
    visited = [False] * (n+1)
    stack = [1]
    visited[1] = True
    while stack:
        u = stack.pop()
        for v, w in graph[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                children[u].append(v)
                stack.append(v)
    
    # Mark nodes that are targets
    is_target = [False] * (n+1)
    for x in target_sensors:
        is_target[x] = True
    
    # Mark nodes that are needed (in the minimal subtree)
    # A node is needed if:
    # - it's a target, OR
    # - it has >= 2 children that are needed (in different subtrees)
    needed = [False] * (n+1)
    
    # Post-order DFS to compute 'needed'
    def dfs_needed(u):
        count = 0
        for v in children[u]:
            count += dfs_needed(v)
        # If u is target, it's needed
        if is_target[u]:
            needed[u] = True
            return 1
        # If u has >= 2 needed children, it's needed
        needed_children = 0
        for v in children[u]:
            if needed[v]:
                needed_children += 1
        if needed_children >= 2:
            needed[u] = True
            return 1
        elif needed_children == 1:
            needed[u] = True
            return 1
        else:
            return 0
    
    dfs_needed(1)
    
    # Sum up the weights of edges in the minimal subtree
    total_cost = 0
    for u in range(2, n+1):  # Skip root
        if needed[u]:
            total_cost += edge_weight[u]
    
    print(total_cost)

if __name__ == "__main__":
    main()
```
