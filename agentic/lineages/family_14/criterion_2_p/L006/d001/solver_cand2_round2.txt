```python
import sys

# The problem states m = n - 1 and the graph is connected.
# This means the graph is a tree.
# To connect a subset of nodes in a tree, we must include all edges
# that lie on the unique paths between any two nodes in the subset.
# This is equivalent to the union of all paths between the critical nodes.
# In a tree, this is the minimal subtree containing all critical nodes.
# We can solve this by calculating the total weight of the tree and
# subtracting the weights of edges that are not part of any path between
# two critical nodes.
# An edge is part of the minimal subtree if and only if the two components
# it creates both contain at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_critical[u] stores the number of critical nodes in the subtree rooted at u
    subtree_critical = [0] * (n + 1)
    
    # DFS to count critical nodes in subtrees
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in critical_nodes:
            subtree_critical[u] += 1
            
    total_weight = 0
    for i in range(len(order) - 1, 0, -1):
        u = order[i]
        p = parent[u]
        subtree_critical[p] += subtree_critical[u]
        
        # If the edge (u, p) separates the tree into two parts,
        # and both parts contain at least one critical node,
        # then this edge must be included.
        if subtree_critical[u] > 0 and subtree_critical[u] < t:
            total_weight += edge_to_parent_weight[u]
            
    print(total_weight)

solve()
```
