```python
import sys

# The problem states m = n - 1 and the graph is connected.
# This means the graph is a tree.
# We need to find the minimum cost to connect all t target sensors.
# In a tree, the set of edges that connect a subset of nodes is the union of 
# all paths between any two nodes in that subset.
# This is equivalent to the minimal subtree containing all target nodes.
# Since it's a tree, the edges required are exactly those that lie on the 
# unique paths between any two target nodes.
# An edge (u, v) with weight w is part of this minimal subtree if and only if
# the removal of this edge splits the tree into two components, both of which
# contain at least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_targets[u] will store the number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order to count targets
    for u in reversed(order):
        if u in targets:
            subtree_targets[u] += 1
        if parent[u] != -1:
            subtree_targets[parent[u]] += subtree_targets[u]
            
    # Calculate cost: an edge (u, parent[u]) is included if 
    # 0 < subtree_targets[u] < t
    for u in range(2, n + 1):
        count = subtree_targets[u]
        if 0 < count < t:
            # Find the weight of the edge (u, parent[u])
            for v, w in adj[u]:
                if v == parent[u]:
                    total_cost += w
                    break
                    
    print(total_cost)

solve()
```
