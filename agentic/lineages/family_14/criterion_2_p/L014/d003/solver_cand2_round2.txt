```python
import sys

# The problem states that the graph is a tree (n nodes, n-1 edges, connected).
# We need to find the minimum cost to connect all t critical facilities.
# In a tree, there is a unique path between any two nodes.
# To connect a set of nodes in a tree, we must include all edges that lie
# on the unique paths between any pair of critical nodes.
# This is equivalent to finding the smallest subtree containing all critical nodes.
# An edge (u, v) with weight w is part of this subtree if and only if
# the subtree formed by removing (u, v) contains at least one critical node
# in both components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count_critical[u] will store the number of critical nodes in the subtree
    # rooted at u (when rooted at node 1).
    count_critical = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in critical_nodes:
            count_critical[u] += 1
        if parent[u] != 0:
            count_critical[parent[u]] += count_critical[u]
            # If the subtree at u contains some but not all critical nodes,
            # the edge (parent[u], u) must be included.
            if 0 < count_critical[u] < t:
                total_cost += edge_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
