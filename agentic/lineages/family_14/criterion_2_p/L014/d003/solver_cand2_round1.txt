```python
import sys

# The problem states that there are n facilities and m = n - 1 links,
# and for any pair of facilities, there is at most one path.
# This means the graph is a tree (or a forest if m < n-1, but here m = n-1).
# Since the graph is a tree, there is exactly one unique path between any two nodes.
# To connect all critical facilities, we must include all edges that lie on the
# unique paths between any two critical facilities.
# This is equivalent to finding the smallest subtree that contains all critical nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that are part of the path between any two critical nodes.
    # An edge (u, v) with weight w is part of the Steiner tree if and only if
    # the subtree rooted at u (when the edge is removed) contains at least one
    # critical node, and the subtree rooted at v contains at least one critical node.
    
    is_critical = [False] * (n + 1)
    for node in critical_nodes:
        is_critical[node] = True
        
    subtree_has_critical = [False] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if is_critical[u]:
            subtree_has_critical[u] = True
        
        if subtree_has_critical[u] and parent[u] != 0:
            total_cost += edge_to_parent_weight[u]
            subtree_has_critical[parent[u]] = True
            
    print(total_cost)

solve()
```
