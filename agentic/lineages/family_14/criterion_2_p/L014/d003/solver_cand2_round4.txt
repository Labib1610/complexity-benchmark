import sys

# The problem states that the potential links form a structure where, if all were 
# constructed, there would be exactly one route between any two facilities.
# This means the graph is a tree (or a forest if m < n-1).
# Since it's a tree, the path between any two nodes is unique.
# To connect a set of critical facilities in a tree, we must include all edges 
# that lie on the unique paths between any two critical facilities.
# This is equivalent to finding the minimal subtree containing all critical nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that are part of the Steiner Tree.
    # An edge (u, v) with weight w is in the Steiner Tree if and only if
    # the subtree at u (when rooted at v) contains at least one critical node,
    # AND the subtree at v (when rooted at u) contains at least one critical node.
    
    is_critical = [False] * (n + 1)
    for node in critical_nodes:
        is_critical[node] = True
        
    has_critical = [False] * (n + 1)
    
    # Since the graph is a tree, we can use DFS to count critical nodes in subtrees.
    # To avoid recursion depth issues, use iterative DFS.
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to propagate critical node counts
    subtree_has_critical = [is_critical[i] for i in range(n + 1)]
    total_cost = 0
    
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if subtree_has_critical[u]:
            # If this subtree has a critical node, the edge to parent is needed
            # if there is at least one critical node outside this subtree.
            # Total critical nodes is t.
            count = sum(1 for node in critical_nodes if subtree_has_critical[node]) # Too slow
            # Better: just check if subtree_has_critical[u] is True and 
            # (t - count_in_subtree[u]) > 0
            pass

    # Simplified: An edge is part of the Steiner tree if the number of critical 
    # nodes in one component is > 0 and < t.
    count_in_subtree = [0] * (n + 1)
    for i in range(n + 1):
        if is_critical[i]:
            count_in_subtree[i] = 1
            
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        count_in_subtree[p] += count_in_subtree[u]
        
    ans = 0
    for i in range(2, n + 1):
        c = count_in_subtree[i]
        if c > 0 and c < t:
            ans += edge_to_parent_weight[i]
            
    print(ans)

solve()
