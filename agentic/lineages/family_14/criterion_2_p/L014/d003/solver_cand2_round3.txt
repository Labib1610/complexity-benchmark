```python
import sys

# The problem states that the graph of all m links is a tree (n nodes, n-1 edges, 
# connected). We need to find the minimum cost to connect a subset of t nodes.
# In a tree, the unique path between any two nodes is fixed. To connect a set 
# of nodes, we must include all edges that lie on the paths between any pair 
# of these nodes. This is equivalent to the sum of weights of all edges that 
# have at least one critical node in each of the two components formed by 
# removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count[u] will store the number of critical nodes in the subtree rooted at u
    count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order
    for u in reversed(order):
        if u in critical_nodes:
            count[u] += 1
        for v, w in adj[u]:
            if v != parent[u]:
                count[u] += count[v]
                # If the edge (u, v) separates some critical nodes from others,
                # it must be included.
                # count[v] is the number of critical nodes in v's subtree.
                # t - count[v] is the number of critical nodes outside v's subtree.
                if count[v] > 0 and count[v] < t:
                    total_cost += w
                    
    print(total_cost)

if __name__ == '__main__':
    solve()
```
