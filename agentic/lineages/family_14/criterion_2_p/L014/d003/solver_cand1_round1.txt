```python
import sys

# The problem states that if all links were built, there is exactly one route 
# between any two facilities. This means the graph is a forest (a collection of trees).
# We need to connect all critical facilities. In a tree, the path between any two 
# nodes is unique. To connect a set of nodes in a tree, we must include all edges 
# that lie on the paths between any pair of critical nodes.
# Since the graph is a forest, we can treat each connected component independently.
# For each component, we identify the critical nodes. If a component has 
# critical nodes, we need to find the minimal subtree that spans all of them.
# The edges in this minimal subtree are exactly those that have at least one 
# critical node in one of the two components formed by removing the edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        critical[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w, _))
        adj[v].append((u, w, _))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_critical[u] will store the number of critical nodes in the subtree
    # rooted at u. We need to perform a DFS to count critical nodes.
    # Since it's a forest, we run DFS on each component.
    
    total_critical = t
    subtree_critical = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    ans = 0
    
    # Use iterative DFS to avoid recursion depth issues
    for i in range(1, n + 1):
        if not visited[i]:
            stack = [(i, -1, -1, 0)] # node, parent, edge_idx, state
            order = []
            while stack:
                u, p, e_idx, state = stack.pop()
                if state == 0:
                    visited[u] = True
                    order.append((u, p, e_idx))
                    for v, w, idx_e in adj[u]:
                        if v != p:
                            stack.append((v, u, idx_e, 0))
            
            # Process in post-order to count critical nodes
            for u, p, e_idx in reversed(order):
                subtree_critical[u] += (1 if critical[u] else 0)
                if p != -1:
                    subtree_critical[p] += subtree_critical[u]
                    # If this edge separates some critical nodes from others,
                    # it must be included.
                    count = subtree_critical[u]
                    if count > 0 and count < total_critical:
                        # Find weight of edge e_idx
                        ans += edges[e_idx][2]
                        
    print(ans)

solve()
```
