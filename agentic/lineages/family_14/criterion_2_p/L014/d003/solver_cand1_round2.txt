```python
import sys

# The problem states that the m links form a tree (n nodes, n-1 edges, no loops).
# To connect all t critical facilities, we need to include all edges that lie on
# the unique path between any two critical facilities.
# An edge is part of the Steiner Tree if and only if the subtree formed by
# removing that edge contains at least one critical facility on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count_crit[u] will store the number of critical nodes in the subtree rooted at u
    count_crit = [0] * (n + 1)
    for node in critical_nodes:
        count_crit[node] = 1
        
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        count_crit[p] += count_crit[u]
        # If this edge separates critical nodes, it must be included
        if count_crit[u] > 0 and count_crit[u] < t:
            total_cost += edge_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    # Increase recursion depth just in case, though iterative is used
    sys.setrecursionlimit(300000)
    solve()
```
