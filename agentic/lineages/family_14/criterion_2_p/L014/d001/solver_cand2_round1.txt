```python
import sys

# The problem states that the facility has n locations and m = n - 1 segments.
# This means the infrastructure is a tree.
# We need to find the minimum cost to connect all critical locations.
# In a tree, there is a unique path between any two nodes.
# To connect a set of critical nodes, we must include all edges that lie on the
# unique paths between any pair of critical nodes.
# This is equivalent to finding the smallest subtree containing all critical nodes.
# Since the graph is a tree, we can root it arbitrarily (e.g., at node 1)
# and use DFS to determine which edges are part of this subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # is_critical_subtree[u] will be true if the subtree rooted at u contains
    # at least one critical node.
    has_critical = [False] * (n + 1)
    for c in critical:
        has_critical[c] = True
        
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_cost = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_cost[v] = w
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if has_critical[u] and parent[u] != 0:
            total_cost += edge_to_parent_cost[u]
            has_critical[parent[u]] = True
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
