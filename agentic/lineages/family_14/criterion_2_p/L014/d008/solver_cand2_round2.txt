```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_holders = set()
    idx = 3
    for _ in range(t):
        key_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t < 2:
        print(0)
        return

    # subtree_keys[u] stores the number of key holders in the subtree rooted at u
    subtree_keys = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in key_holders:
            subtree_keys[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if parent[u] != 0:
            subtree_keys[parent[u]] += subtree_keys[u]
            
    total_fee = 0
    for i in range(2, n + 1):
        # An edge is critical if removing it splits the tree into two components,
        # each containing at least one key holder.
        # This happens if subtree_keys[i] > 0 and (t - subtree_keys[i]) > 0.
        if subtree_keys[i] > 0 and (t - subtree_keys[i]) > 0:
            total_fee += edge_to_parent_weight[i]
            
    print(total_fee)

if __name__ == '__main__':
    solve()
```
