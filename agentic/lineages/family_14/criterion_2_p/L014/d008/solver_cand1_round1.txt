```python
import sys

# The problem states that the market structure is a tree (n nodes, n-1 edges, connected).
# An edge is "critical" if its removal splits the tree into two components,
# each containing at least one key patent holder.
# Let T be the tree. For an edge e, let S1 and S2 be the sets of key patent holders
# in the two components formed by removing e.
# e is critical if |S1| > 0 and |S2| > 0.
# Since the total number of key patent holders is t, this is equivalent to
# 0 < |S1| < t.
# We must include all such edges.
# Let the set of critical edges be E_crit.
# The sum of weights of E_crit is the base cost.
# Any other edge e' not in E_crit can be included if it has a negative weight,
# but here all weights are positive (1 <= w <= 1000000).
# Wait, the problem asks for the minimum total legal fee of a valid subset.
# If an edge is not critical, we don't have to include it. Since all weights are positive,
# we should only include the critical edges.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_holders = set()
    idx = 3
    for _ in range(t):
        key_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w, i))
        adj[v].append((u, w, i))
        edges.append((u, v, w))
        idx += 3
        
    if t < 2:
        print(0)
        return

    # subtree_keys[u] stores the number of key holders in the subtree rooted at u
    subtree_keys = [0] * (n + 1)
    parent_edge = [None] * (n + 1) # (parent, weight)
    
    # DFS to compute subtree_keys
    stack = [(1, 0)]
    order = []
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        for v, w, i in adj[u]:
            if v != p:
                visited[v] = True
                parent_edge[v] = (u, w)
                stack.append((v, u))
                
    for u in reversed(order):
        if u in key_holders:
            subtree_keys[u] += 1
        p, w = parent_edge[u] if parent_edge[u] else (0, 0)
        if p != 0:
            subtree_keys[p] += subtree_keys[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        count = subtree_keys[i]
        # Edge between i and parent[i] is critical if 0 < count < t
        if 0 < count < t:
            total_cost += parent_edge[i][1]
            
    print(total_cost)

solve()
```
