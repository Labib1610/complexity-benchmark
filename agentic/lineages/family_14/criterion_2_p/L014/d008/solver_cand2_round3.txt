```python
import sys

# The problem states that the m proposed agreements form a tree (since m = n-1).
# An edge is critical if removing it splits the tree into two components,
# each containing at least one key patent holder.
# Let T be the tree. For an edge e, let S1 and S2 be the two components formed by removing e.
# e is critical if (count of key holders in S1 > 0) AND (count of key holders in S2 > 0).
# Let K be the total number of key patent holders.
# Let k(v) be the number of key patent holders in the subtree rooted at v.
# An edge (u, v) where v is a child of u is critical if k(v) > 0 and (K - k(v)) > 0.
# Since K >= 1, this is equivalent to 0 < k(v) < K.
# We must include all such critical edges.
# The cost is the sum of weights of all critical edges.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_holders = set()
    idx = 3
    for _ in range(t):
        key_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t < 2:
        print(0)
        return

    # Root the tree at an arbitrary node, say 1
    # Perform DFS to calculate subtree key counts
    subtree_key_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in key_holders:
            subtree_key_count[u] += 1
        if parent[u] != 0:
            subtree_key_count[parent[u]] += subtree_key_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        k_v = subtree_key_count[i]
        # Edge is (i, parent[i])
        # It is critical if 0 < k_v < t
        if 0 < k_v < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
