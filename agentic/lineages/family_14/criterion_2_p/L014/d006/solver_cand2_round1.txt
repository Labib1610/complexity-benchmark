import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    key_holders = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the union of all paths between any two key patent holders.
    # This is equivalent to finding the smallest subtree containing all key patent holders.
    # An edge is in this subtree if and only if the subtree rooted at one of its 
    # endpoints contains at least one key patent holder and the other side also 
    # contains at least one key patent holder.
    
    has_key = [False] * (n + 1)
    for k in key_holders:
        has_key[k] = True
        
    total_weight = 0
    
    # DFS to count key holders in subtree
    subtree_key_count = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        subtree_key_count[u] += (1 if has_key[u] else 0)
        if parent[u] != 0:
            subtree_key_count[parent[u]] += subtree_key_count[u]
            
    # An edge (u, parent[u]) is part of the minimal subtree if 
    # 0 < subtree_key_count[u] < t
    ans = 0
    for i in range(2, n + 1):
        count = subtree_key_count[i]
        if count > 0 and count < t:
            ans += edge_weight[i]
            
    print(ans)

solve()
