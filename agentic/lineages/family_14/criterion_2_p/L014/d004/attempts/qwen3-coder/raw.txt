import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    patent_holders = []

        idx += 1
    
    edges = []

        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((w, u, v))
    
    if t <= 1:
        print(0)
        return
    
    # Sort edges by weight
    edges.sort()
    
    # Initialize DSU
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # First, connect all patent holders
    cost = 0
    patent_set = set(patent_holders)
    first_patent = patent_holders[0]
    
    for w, u, v in edges:
        if u in patent_set and v in patent_set:
            if find(parent, u) != find(parent, v):
                union(parent, rank, u, v)
                cost += w
        elif u in patent_set and v not in patent_set:
            if find(parent, u) != find(parent, v):
                union(parent, rank, u, v)
                cost += w
        elif u not in patent_set and v in patent_set:
            if find(parent, u) != find(parent, v):
                union(parent, rank, u, v)
                cost += w
    
    # Check if all patent holders are connected
    root = find(parent, first_patent)
    all_connected = True
    for ph in patent_holders[1:]:
        if find(parent, ph) != root:
            all_connected = False
            break
    
    if all_connected:
        print(cost)
        return
    
    # If not all connected, we need to continue connecting components
    # Reset and do Kruskal's algorithm to connect all patent holders
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    cost = 0
    components = len(patent_holders)
    
    # Connect patent holders first
    for w, u, v in edges:
        if u in patent_set and v in patent_set:
            if find(parent, u) != find(parent, v):
                union(parent, rank, u, v)
                cost += w
                components -= 1
                if components == 1:
                    break
    
    if components == 1:
        print(cost)
        return
    
    # Continue with all edges to connect remaining patent holders
    for w, u, v in edges:
        if find(parent, u) != find(parent, v):
            # Check if this edge helps connect patent holders
            u_root = find(parent, u)
            v_root = find(parent, v)
            
            u_is_patent = u in patent_set
            v_is_patent = v in patent_set
            u_root_is_patent = u_root in patent_set or any(find(parent, x) == u_root for x in patent_holders)
            v_root_is_patent = v_root in patent_set or any(find(parent, x) == v_root for x in patent_holders)
            
            if (u_is_patent or u_root_is_patent) and (v_is_patent or v_root_is_patent):

                cost += w
                components -= 1

                union(parent, rank, u, v)
                cost += w
                components -= 1

                union(parent, rank, u, v)

                # Recalculate components
                roots = set()

                    roots.add(find(parent, ph))
                components = len(roots)
                if components == 1:
                    break
    
    print(cost)

if __name__ == "__main__":
    main()
