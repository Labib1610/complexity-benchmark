```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    patent = [False] * (n + 1)
    for _ in range(t):
        x = int(data[idx]); idx += 1
        patent[x] = True
    
    edges = []
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        edges.append((w, u, v))
    
    if t <= 1:
        print(0)
        return
    
    edges.sort()
    
    parent = list(range(n + 1))
    has_patent = [False] * (n + 1)
    for i in range(1, n + 1):
        has_patent[i] = patent[i]
    
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    
    total = 0
    for w, u, v in edges:
        ru = find(u)
        rv = find(v)
        if ru == rv:
            continue
        if has_patent[ru] and has_patent[rv]:
            total += w
            parent[rv] = ru
            has_patent[ru] = True
        else:
            parent[rv] = ru
            has_patent[ru] = has_patent[ru] or has_patent[rv]
    
    print(total)

solve()
```
