```python
import sys

# The problem asks for the minimum sum of latencies of a subset of edges such that
# for every partition of the t designated registers into two non-empty sets G1 and G2,
# there is a path between some node in G1 and some node in G2.
#
# Given that the graph is a tree (n nodes, n-1 edges), the condition is equivalent
# to saying that the union of all paths between all pairs of designated registers
# must be included in our subset.
#
# If t < 2, the condition is vacuously true, so the minimum latency is 0.
# If t >= 2, we need to include all edges that lie on the unique path between
# any two designated registers. This is equivalent to the Steiner Tree problem
# on a tree, which is simply the minimal subtree connecting all t designated nodes.
#
# In a tree, the minimal subtree connecting a set of nodes is the union of all
# paths between them. This can be found by pruning all branches that do not
# contain any designated nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    designated = set()
    idx = 3
    for _ in range(t):
        designated.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the sum of weights of the minimal subtree:
    # 1. Root the tree at an arbitrary designated node (e.g., the first one).
    # 2. Perform a post-order traversal.
    # 3. A node is part of the minimal subtree if it is a designated node
    #    or if it has at least one child that is part of the minimal subtree.
    
    root = next(iter(designated))
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    is_in_subtree = [False] * (n + 1)
    for node in designated:
        is_in_subtree[node] = True
        
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == root:
            continue
        if is_in_subtree[u]:
            total_weight += edge_to_parent_weight[u]
            is_in_subtree[parent[u]] = True
            
    print(total_weight)

if __name__ == '__main__':
    solve()
```
