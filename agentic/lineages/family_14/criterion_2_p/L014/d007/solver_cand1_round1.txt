```python
import sys

# The problem asks us to identify "critical" edges in a tree.
# An edge (u, v) is non-critical if we can partition the vertices into two sets
# (Tier A and Tier B) such that:
# 1. All key patent holders are in Tier A.
# 2. For all edges (x, y) != (u, v), x and y are in the same tier.
# 3. u and v are in different tiers.
#
# Condition 2 implies that for every edge (x, y) != (u, v), x and y must be in the same tier.
# This means that in the graph formed by removing (u, v), all vertices in each connected
# component must be in the same tier.
# Since (u, v) is an edge in a tree, removing it splits the tree into two components,
# say T_u (containing u) and T_v (containing v).
# All vertices in T_u must be in the same tier, and all vertices in T_v must be in the same tier.
# Condition 3 says u and v are in different tiers, so T_u and T_v must have different tiers.
# Condition 1 says all key patent holders must be in Tier A.
# If T_u contains any key patent holder, T_u must be Tier A.
# If T_v contains any key patent holder, T_v must be Tier A.
# If both T_u and T_v contain key patent holders, they both must be Tier A,
# which contradicts the requirement that they be in different tiers.
# Thus, an edge (u, v) is non-critical if and only if at most one of the two components
# formed by removing (u, v) contains any key patent holders.
# Since the total number of key patent holders is t, this is equivalent to saying
# that either T_u contains 0 key patent holders or T_v contains 0 key patent holders.
# An edge is critical if both components contain at least one key patent holder.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_holders = set()
    idx = 3
    for _ in range(t):
        key_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count key holders in subtree
    subtree_keys = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in key_holders:
            subtree_keys[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if parent[u] != 0:
            subtree_keys[parent[u]] += subtree_keys[u]
            
    total_critical_fee = 0
    for u, v, w in edges:
        # Determine which node is the child in the rooted tree
        child = u if parent[u] == v else v
        keys_in_subtree = subtree_keys[child]
        
        # The two components have keys_in_subtree and (t - keys_in_subtree)
        # The edge is critical if both > 0
        if keys_in_subtree > 0 and (t - keys_in_subtree) > 0:
            total_critical_fee += w
            
    print(total_critical_fee)

solve()
```
