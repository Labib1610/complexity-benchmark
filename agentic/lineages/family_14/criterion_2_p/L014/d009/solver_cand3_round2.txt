```python
import sys

# The problem asks for the sum of costs of all "essential" contracts.
# A contract is essential if it is a bridge in the graph formed by the 
# potential contracts, and it lies on every path between at least one pair 
# of primary stakeholders.
#
# A contract (u, v) is essential if and only if:
# 1. It is a bridge in the graph.
# 2. Removing it splits the graph into two components, each containing 
#    at least one primary stakeholder.
#
# Algorithm:
# 1. Find all bridges in the graph.
# 2. For each bridge, removing it splits the graph into two components.
# 3. Check if both components contain at least one primary stakeholder.
# 4. Sum the costs of such bridges.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    primary = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        primary[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w, i))
        adj[v].append((u, w, i))
        edges.append((u, v, w))
        idx += 3
        
    tin = [-1] * (n + 1)
    low = [-1] * (n + 1)
    subtree_primary = [0] * (n + 1)
    timer = 0
    
    total_primary = t
    ans = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(1, -1, -1)] # node, parent, edge_index
    
    # Pre-calculate subtree primary counts
    # We need to know how many primary nodes are in the component 
    # formed by removing a bridge.
    
    visited = [False] * (n + 1)
    order = []
    parent = [0] * (n + 1)
    edge_to_parent = [-1] * (n + 1)
    
    dfs_stack = [1]
    visited[1] = True
    while dfs_stack:
        u = dfs_stack.pop()
        order.append(u)
        for v, w, i in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent[v] = i
                dfs_stack.append(v)
                
    tin = [0] * (n + 1)
    low = [0] * (n + 1)
    timer = 0
    
    # Re-run bridge finding
    visited = [False] * (n + 1)
    
    def get_bridges():
        nonlocal timer
        timer = 0
        tin = [0] * (n + 1)
        low = [0] * (n + 1)
        subtree_p = [0] * (n + 1)
        
        # Post-order traversal for subtree counts
        for u in reversed(order):
            subtree_p[u] += (1 if primary[u] else 0)
            if parent[u] != 0:
                subtree_p[parent[u]] += subtree_p[u]
        
        # Bridge finding
        timer = 0
        tin = [0] * (n + 1)
        low = [0] * (n + 1)
        
        # Standard bridge finding
        visited = [False] * (n + 1)
        
        def dfs(u, p_edge):
            nonlocal timer
            timer += 1
            tin[u] = low[u] = timer
            visited[u] = True
            for v, w, i in adj[u]:
                if i == p_edge: continue
                if visited[v]:
                    low[u] = min(low[u], tin[v])
                else:
                    dfs(v, i)
                    low[u] = min(low[u], low[v])
                    if low[v] > tin[u]:
                        # Bridge found
                        if subtree_p[v] > 0 and (total_primary - subtree_p[v]) > 0:
                            nonlocal ans
                            ans += w
        dfs(1, -1)

    get_bridges()
    print(ans)

solve()
```
