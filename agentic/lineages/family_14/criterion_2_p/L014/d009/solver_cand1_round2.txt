```python
import sys

# The problem asks us to partition the edges of a tree into two sets:
# Finalized and Rejected.
# An edge can be rejected if and only if its removal does not disconnect
# the set of key patent holders.
# In a tree, removing an edge splits the tree into two components.
# The condition "all t key patent holders remain linked" is equivalent to
# saying that the two components formed by removing the edge do not both
# contain at least one key patent holder.
# That is, one of the two components must contain zero key patent holders.
# This means an edge can be rejected if and only if all key patent holders
# are on one side of the edge.
# This is equivalent to saying that the edge is NOT on the unique path
# between any two key patent holders.
# The set of edges that must be finalized are exactly those that lie on
# the minimal subtree connecting all t key patent holders (the Steiner Tree).
# Since the graph is already a tree, the Steiner Tree is the union of all
# paths between the key patent holders.
# We need to sum the weights of all edges that are part of this Steiner Tree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_holders = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find all edges that lie on the path between any two key holders.
    # An edge is in the Steiner Tree if and only if both components formed
    # by removing it contain at least one key holder.
    # Let's root the tree at an arbitrary key holder.
    # For each node, count how many key holders are in its subtree.
    
    root = next(iter(key_holders))
    subtree_key_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [(root, 0)]
    parent = [0] * (n + 1)
    while stack:
        u, p = stack.pop()
        order.append(u)
        parent[u] = p
        for v, w in adj[u]:
            if v != p:
                parent_edge_weight[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        if u in key_holders:
            subtree_key_count[u] += 1
        if parent[u] != 0:
            subtree_key_count[parent[u]] += subtree_key_count[u]
            
    total_weight = 0
    for i in range(1, n + 1):
        if i == root:
            continue
        # Edge between i and parent[i] is part of the Steiner Tree if
        # 0 < subtree_key_count[i] < t
        if 0 < subtree_key_count[i] < t:
            total_weight += parent_edge_weight[i]
            
    print(total_weight)

solve()
```
