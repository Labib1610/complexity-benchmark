```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_holders = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t < 2:
        print(0)
        return

    # Multi-source BFS to find the closest key holder for each node
    # dist[i] = (distance, source_id)
    # We need to know if there exists a key holder k1 closer to u than v
    # and a key holder k2 closer to v than u.
    # This is equivalent to:
    # Let f(u) be the set of closest key holders to u.
    # An edge (u, v) is critical if there exists k1 in f(u) and k2 in f(v)
    # such that k1 != k2.
    
    # Actually, the condition is:
    # exists k s.t. dist(k, u) < dist(k, v)
    # exists k' s.t. dist(k', v) < dist(k', u)
    # In a tree, removing edge (u, v) splits the tree into two components T_u and T_v.
    # dist(k, u) < dist(k, v) iff k is in T_u.
    # So the condition is: T_u contains at least one key holder AND T_v contains at least one key holder.
    
    # Count key holders in each subtree
    subtree_keys = [0] * (n + 1)
    for k in key_holders:
        subtree_keys[k] = 1
        
    # Root the tree at key_holders[0]
    parent = [0] * (n + 1)
    order = []
    stack = [key_holders[0]]
    visited = [False] * (n + 1)
    visited[key_holders[0]] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if parent[u] != 0:
            subtree_keys[parent[u]] += subtree_keys[u]
            
    total_keys = t
    critical_sum = 0
    for u, v, w in edges:
        # Determine which node is the child in the rooted tree
        child = v if parent[v] == u else u
        keys_in_subtree = subtree_keys[child]
        # If the subtree has keys and the rest of the tree has keys, it's critical
        if keys_in_subtree > 0 and (total_keys - keys_in_subtree) > 0:
            critical_sum += w
            
    print(critical_sum)

solve()
```
