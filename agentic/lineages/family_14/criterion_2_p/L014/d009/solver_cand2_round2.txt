import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    key_holders = set()
    idx = 3
    for _ in range(t):
        key_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the union of all paths between any two key holders.
    # An edge is in the union if and only if the subtree below it (when rooted)
    # contains at least one key holder AND the rest of the tree contains at least one key holder.
    
    has_key = [False] * (n + 1)
    for k in key_holders:
        has_key[k] = True
        
    subtree_has_key = [False] * (n + 1)
    total_weight = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to compute subtree_has_key
    for u in reversed(order):
        if has_key[u]:
            subtree_has_key[u] = True
        for v, w in adj[u]:
            if v != parent[u]:
                if subtree_has_key[v]:
                    subtree_has_key[u] = True
                    
    # An edge (u, parent[u]) is part of the union if:
    # subtree_has_key[u] is True AND (t - count_of_keys_in_subtree_u) > 0
    # We need to count keys in subtree.
    key_count = [0] * (n + 1)
    for u in order:
        if has_key[u]:
            key_count[u] += 1
            
    for u in reversed(order):
        if parent[u] != 0:
            key_count[parent[u]] += key_count[u]
            
    ans = 0
    for u in range(2, n + 1):
        # Edge is between u and parent[u]
        # It is in the union if subtree at u has keys and the rest of the tree has keys
        if key_count[u] > 0 and (t - key_count[u]) > 0:
            ans += edge_to_parent_weight[u]
            
    print(ans)

solve()
