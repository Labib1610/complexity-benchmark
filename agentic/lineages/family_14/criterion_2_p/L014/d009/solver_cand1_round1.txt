import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(1000000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n, m, t = map(int, input_data[:3])
    key_holders = set(map(int, input_data[3:3+t]))
    
    if t < 2:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    # Kruskal's to find the MST
    edges.sort()
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i: return i
        parent[i] = find(parent[i])
        return parent[i]
    
    adj = [[] for _ in range(n + 1)]
    mst_weight = 0
    for w, u, v in edges:
        root_u, root_v = find(u), find(v)
        if root_u != root_v:
            parent[root_u] = root_v
            adj[u].append((v, w))
            adj[v].append((u, w))
            mst_weight += w
            
    # Identify critical edges: edges on the unique path between any two key holders
    # in the MST are critical.
    # A tree edge is critical if the two components it splits both contain at least one key holder.
    
    is_key = [False] * (n + 1)
    for k in key_holders:
        is_key[k] = True
        
    subtree_keys = [0] * (n + 1)
    critical_weight = 0
    
    # DFS to count keys in subtrees
    order = []
    stack = [(1, 0)]
    parent_node = [0] * (n + 1)
    edge_to_parent_w = [0] * (n + 1)
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        parent_node[u] = p
        for v, w in adj[u]:
            if v != p:
                edge_to_parent_w[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        subtree_keys[u] += (1 if is_key[u] else 0)
        if parent_node[u] != 0:
            subtree_keys[parent_node[u]] += subtree_keys[u]
            
    for u in range(2, n + 1):
        # Edge between u and parent_node[u]
        count = subtree_keys[u]
        if count > 0 and count < t:
            critical_weight += edge_to_parent_w[u]
            
    print(critical_weight)

solve()
