```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes (critical terminals)
# in a tree structure. Since the input guarantees m = n - 1 and the graph is connected,
# the graph is a tree.
# In a tree, there is exactly one path between any two nodes.
# To connect a set of critical terminals, we must include all edges that lie on the
# unique paths between any two critical terminals.
# This is equivalent to finding the Steiner Tree in a tree, which is simply the
# union of all paths between the critical terminals.
# An edge is part of this union if and only if the subtree formed by removing that
# edge contains at least one critical terminal on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Root the tree at an arbitrary node (e.g., 1)
    # Count how many critical nodes are in the subtree of each node
    subtree_critical_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in critical_nodes:
            subtree_critical_count[node] += 1
            
    # Process in reverse topological order to propagate counts up
    total_cost = 0
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        subtree_critical_count[p] += subtree_critical_count[u]
        
        # If the subtree at u contains some but not all critical nodes,
        # the edge connecting u to its parent must be part of the Steiner Tree.
        count = subtree_critical_count[u]
        if count > 0 and count < t:
            total_cost += parent_edge_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
