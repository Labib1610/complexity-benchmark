```python
import sys

# The problem states the facility is a tree (n chambers, n-1 corridors, connected, no loops).
# We need to connect a subset of t critical chambers with minimum total weight.
# In a tree, there is a unique path between any two nodes.
# To connect a set of nodes in a tree, we must activate all edges that lie on the 
# unique paths between any pair of these nodes.
# This is equivalent to the union of all paths between the critical nodes.
# An edge is part of this union if and only if the subtree formed by removing 
# that edge contains at least one critical node on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count_critical[u] will store the number of critical nodes in the subtree of u
    count_critical = [0] * (n + 1)
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in critical_nodes:
            count_critical[u] += 1
        if parent[u] != 0:
            count_critical[parent[u]] += count_critical[u]
            # If this edge separates the tree into two parts, both containing 
            # at least one critical node, then this edge must be activated.
            if count_critical[u] > 0 and count_critical[u] < t:
                total_weight += edge_weight_to_parent[u]
                
    print(total_weight)

solve()
```
