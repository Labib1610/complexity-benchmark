```python
import sys

# The problem states that m = n - 1 and the graph is connected.
# This means the graph is a tree.
# If the graph is a tree, there is only one unique path between any two nodes.
# To connect all critical facilities, we must include all edges that lie on the
# unique paths between any two critical facilities.
# Since it's a tree, the set of edges required is the union of paths between
# all critical facilities, which is equivalent to the edges in the minimal
# subtree containing all critical facilities.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Root the tree at the first critical facility
    root = next(iter(critical))
    
    # We need to find which edges are part of the minimal subtree.
    # An edge (u, v) is in the subtree if and only if the subtree rooted at
    # the child node contains at least one critical facility and the rest of
    # the tree (outside that subtree) also contains at least one critical facility.
    
    has_critical = [False] * (n + 1)
    for c in critical:
        has_critical[c] = True
        
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    parent[root] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order to propagate critical counts
    total_critical = t
    count_critical = [0] * (n + 1)
    for i in range(1, n + 1):
        if has_critical[i]:
            count_critical[i] = 1
            
    ans = 0
    for u in reversed(order):
        if u == root:
            continue
        if count_critical[u] > 0 and count_critical[u] < total_critical:
            ans += edge_to_parent_weight[u]
        
        if parent[u] != -1:
            count_critical[parent[u]] += count_critical[u]
            
    print(ans)

solve()
```
