```python
import sys

# The problem states that the market is a tree (n nodes, n-1 edges).
# The condition: If an edge splits the tree into two components, and both
# components contain at least one key patent holder, that edge must be included.
# An edge splits the tree into two components if and only if it is a bridge.
# In a tree, every edge is a bridge.
# So, an edge must be included if and only if both components formed by
# removing that edge contain at least one key patent holder.
# Let T be the total number of key patent holders.
# Let a component have k key patent holders. The other component will have T - k.
# The condition "both factions contain at least one key patent holder"
# is equivalent to: k > 0 and (T - k) > 0.
# Since T >= 1, this is equivalent to 1 <= k <= T - 1.
# If T = 1, no edge can satisfy this, so the sum is 0.
# If T > 1, an edge must be included if the number of key patent holders
# in one of the components is between 1 and T - 1 inclusive.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    key_holders = set()
    idx = 3
    for _ in range(t):
        key_holders.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count key holders in subtree
    subtree_keys = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in key_holders:
            subtree_keys[u] += 1
        if parent[u] != -1:
            subtree_keys[parent[u]] += subtree_keys[u]
            
    total_fee = 0
    for u, v, w in edges:
        # The edge connects u and v. One is the parent of the other.
        child = u if parent[u] == v else v
        k = subtree_keys[child]
        # If 1 <= k <= t - 1, the edge must be included
        if 1 <= k <= t - 1:
            total_fee += w
            
    print(total_fee)

if __name__ == '__main__':
    solve()
```
