import sys
from collections import defaultdict
import heapq

def find_parent(parent, x):
    if parent[x] != x:
        parent[x] = find_parent(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find_parent(parent, x), find_parent(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    required = set()
    for _ in range(t):
        required.add(int(data[idx]))
        idx += 1
    
    edges = []
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((w, u, v))
    
    if t <= 1:
        print(0)
        return
    
    # Sort edges by weight
    edges.sort()
    
    # Initialize Union-Find
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    total_cost = 0
    connected_required = 0
    
    for w, u, v in edges:
        if find_parent(parent, u) != find_parent(parent, v):
            if union(parent, rank, u, v):
                # Check if this edge connects two required sites or 
                # connects a required site to an already connected component
                pu = find_parent(parent, u)
                pv = find_parent(parent, v)
                
                u_required = u in required
                v_required = v in required
                u_component_has_required = any(x in required and find_parent(parent, x) == pu for x in required)
                v_component_has_required = any(x in required and find_parent(parent, x) == pv for x in required)
                
                # If this edge connects two components and at least one has a required site
                if (u_required or u_component_has_required) and (v_required or v_component_has_required):
                    total_cost += w
                    connected_required += 1
                    
                    # If we've connected all required sites, we're done
                    if connected_required == t - 1:
                        break
                else:
                    # Just connect them, but don't count towards required connections
                    pass
    
    print(total_cost)

if __name__ == "__main__":
    main()
