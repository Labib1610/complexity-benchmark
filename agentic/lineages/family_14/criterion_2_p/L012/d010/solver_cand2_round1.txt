```python
import sys

# The problem asks for the minimum cost to bypass sweeps such that no remaining
# sweep separates the set of protected sectors.
# A sweep (u, v) separates the protected sectors if and only if both pools
# created by removing (u, v) contain at least one protected sector.
# In a tree, removing an edge (u, v) splits the tree into two components.
# Let T_u be the component containing u and T_v be the component containing v.
# The edge (u, v) must be bypassed if T_u contains at least one protected sector
# AND T_v contains at least one protected sector.
# This is equivalent to saying that the edge (u, v) lies on the unique path
# between some pair of protected sectors.
# More generally, an edge must be bypassed if it lies on the path between
# any two protected sectors.
# The set of edges that must be bypassed are exactly those edges that lie on
# the minimal subtree connecting all protected sectors.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    protected = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many protected sectors are in the subtree rooted at v
    # when looking from u.
    # We can use DFS to compute this.
    
    protected_count = [0] * (n + 1)
    parent_edge_cost = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_cost[v] = w
                stack.append(v)
                
    for u in order:
        if u in protected:
            protected_count[u] = 1
            
    total_cost = 0
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        protected_count[p] += protected_count[u]
        # If the subtree at u contains some but not all protected sectors,
        # the edge (p, u) must be bypassed.
        if 0 < protected_count[u] < t:
            total_cost += parent_edge_cost[u]
            
    print(total_cost)

solve()
```
