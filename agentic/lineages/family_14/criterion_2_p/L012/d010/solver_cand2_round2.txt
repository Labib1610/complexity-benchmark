```python
import sys

# The problem asks for the minimum cost to connect all target variables in a tree.
# Since the identities form a tree (n nodes, n-1 edges), the path between any two
# nodes is unique. To force all target variables to be equal, we must select
# all edges that lie on the paths connecting the target variables.
# This is equivalent to finding the smallest subtree that contains all target nodes.
# The cost is the sum of weights of all edges that are part of the minimal subtree
# spanning the target variables.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that lie on paths between any two target nodes.
    # An edge (u, v) with weight w is in the minimal subtree if and only if
    # the two components formed by removing (u, v) both contain at least one target node.
    
    # Root the tree at an arbitrary target node (e.g., the first one)
    root = next(iter(targets))
    
    # subtree_targets[u] = number of target nodes in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in targets:
            subtree_targets[u] += 1
        if parent[u] != 0:
            subtree_targets[parent[u]] += subtree_targets[u]
            # If the subtree at u contains some targets but not all,
            # the edge (u, parent[u]) must be included.
            if 0 < subtree_targets[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
