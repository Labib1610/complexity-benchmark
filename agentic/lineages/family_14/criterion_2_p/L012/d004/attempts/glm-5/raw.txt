```python
import sys
sys.setrecursionlimit(1 << 20)

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1

    is_target = [False] * (n + 1)
    for _ in range(t):
        v = int(data[idx]); idx += 1
        is_target[v] = True

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))

    if t <= 1:
        print(0)
        return

    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = [1]
    parent[1] = -1
    pos = 0
    while pos < len(order):
        u = order[pos]
        pos += 1
        for v, w in adj[u]:
            if parent[v] == 0 and v != 1:
                parent[v] = u
                parent_weight[v] = w
                order.append(v)

    size = [0] * (n + 1)
    for u in order:
        if is_target[u]:
            size[u] = 1

    for i in range(len(order) - 1, 0, -1):
        u = order[i]
        p = parent[u]
        size[p] += size[u]

    total = 0
    for i in range(1, len(order)):
        u = order[i]
        if 0 < size[u] < t:
            total += parent_weight[u]

    print(total)

solve()
```
