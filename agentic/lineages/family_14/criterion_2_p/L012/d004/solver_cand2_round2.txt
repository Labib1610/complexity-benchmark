```python
import sys

# The problem asks for the minimum cost to connect all target variables in a tree.
# Since the input is a tree (n nodes, n-1 edges), the unique path between any two
# nodes is the only way to connect them. To connect a set of target variables,
# we must select all edges that lie on the paths between any two target variables.
# This is equivalent to finding the smallest subtree that contains all target variables.
# We can solve this by rooting the tree arbitrarily (e.g., at node 1) and 
# calculating for each edge whether it lies on a path between two target variables.
# An edge (u, v) where v is a child of u lies on such a path if and only if the 
# subtree rooted at v contains at least one target variable AND the rest of the 
# tree (outside the subtree of v) also contains at least one target variable.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # DFS to compute subtree target counts
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    target_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if i in targets:
            target_count[i] = 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if target_count[u] > 0 and target_count[u] < t:
            total_cost += edge_to_parent_weight[u]
        target_count[p] += target_count[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
