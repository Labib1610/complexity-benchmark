import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    auditing_offices = set()
    idx = 3
    for _ in range(t):
        auditing_offices.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the Steiner Tree for the auditing offices.
    # Since the graph is a tree, the Steiner Tree is simply the union of all 
    # paths between any two auditing offices.
    # This is equivalent to the set of all edges that lie on a path between 
    # at least two auditing offices.
    
    # An edge (u, v) with weight w is part of the Steiner Tree if and only if
    # the subtree rooted at the child node contains at least one auditing office
    # AND the rest of the tree contains at least one auditing office.
    
    has_auditing = [False] * (n + 1)
    for office in auditing_offices:
        has_auditing[office] = True
        
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    # Process nodes in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            p, w = parent[u]
            if has_auditing[u]:
                total_cost += w
                has_auditing[p] = True
                
    print(total_cost)

solve()
