```python
import sys

# The problem asks for the minimum cost to connect all t auditing offices in a tree.
# This is equivalent to finding the sum of weights of all edges that lie on the 
# unique path between any two auditing offices.
# An edge is part of the minimal subtree connecting all auditing offices if and 
# only if the subtree formed by removing that edge splits the auditing offices 
# into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    auditing_offices = set()
    idx = 3
    for _ in range(t):
        auditing_offices.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many auditing offices are in the subtree of each node.
    # Since n is up to 200,000, we use an iterative DFS to avoid recursion depth issues.
    
    count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for node in order:
        if node in auditing_offices:
            count[node] = 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If the subtree at u contains some but not all auditing offices,
            # the edge (u, p) must be included.
            if 0 < count[u] < t:
                total_cost += edge_weight_to_parent[u]
            count[p] += count[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
