```python
import sys
from sys import setrecursionlimit

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    is_primary = [False] * (n + 1)
    for _ in range(t):
        hub = int(data[idx]); idx += 1
        is_primary[hub] = True
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    setrecursionlimit(n + 10)
    
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = []
    visited = [False] * (n + 1)
    
    stack = [(1, 0, 0, False)]
    while stack:
        node, p, pw, processed = stack.pop()
        if processed:
            order.append(node)
            parent[node] = p
            parent_weight[node] = pw
            continue
        if visited[node]:
            continue
        visited[node] = True
        stack.append((node, p, pw, True))
        for v, w in adj[node]:
            if v != p and not visited[v]:
                stack.append((v, node, w, False))
    
    dp = [0] * (n + 1)
    
    for node in order:
        if node == 1:
            continue
        p = parent[node]
        w = parent_weight[node]
        
        if is_primary[node] or dp[node] > 0:
            dp[p] += w + dp[node]
    
    print(dp[1])

if __name__ == "__main__":
    solve()
```
