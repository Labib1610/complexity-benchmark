```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    hubs = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    # Steiner Tree problem on a graph where we want to connect t hubs.
    # Since the graph is a tree (m = n-1 and connected), the path between 
    # any two nodes is unique.
    # The set of edges required to connect all hubs is the union of all paths
    # between any two hubs.
    
    adj = [[] for _ in range(n + 1)]
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Root the tree at hubs[0]
    parent = [0] * (n + 1)
    edge_to_parent = [0] * (n + 1)
    depth = [0] * (n + 1)
    
    stack = [(hubs[0], 0, 0, 0)]
    while stack:
        u, p, d, w = stack.pop()
        parent[u] = p
        depth[u] = d
        edge_to_parent[u] = w
        for v, weight in adj[u]:
            if v != p:
                stack.append((v, u, d + 1, weight))
                
    # Mark all nodes that are part of the path between any two hubs.
    # A node is in the Steiner tree if it is a hub or an ancestor of a hub
    # that is also an ancestor of another hub (LCA logic).
    # Actually, simpler: an edge is in the Steiner tree if the subtree 
    # below it contains at least one hub and the rest of the tree contains 
    # at least one hub.
    
    has_hub = [False] * (n + 1)
    for h in hubs:
        has_hub[h] = True
        
    # Post-order traversal to propagate hub existence
    nodes_by_depth = sorted(range(1, n + 1), key=lambda x: depth[x], reverse=True)
    
    total_cost = 0
    for u in nodes_by_depth:
        if u == hubs[0]:
            continue
        if has_hub[u]:
            total_cost += edge_to_parent[u]
            has_hub[parent[u]] = True
            
    print(total_cost)

solve()
```
