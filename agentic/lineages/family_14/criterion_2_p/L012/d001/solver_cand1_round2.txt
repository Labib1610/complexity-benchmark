```python
import sys

# The problem states that the available comm-links form a tree (n nodes, n-1 edges, 
# unique path between any two nodes). To connect all primary command hubs, we 
# must include all edges that lie on the paths between any two primary hubs.
# This is equivalent to finding the minimal subtree that contains all primary hubs.
# An edge is part of this subtree if and only if the subtree formed by removing 
# that edge splits the primary hubs into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    hubs = set()
    idx = 3
    for _ in range(t):
        hubs.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many hubs are in the subtree of each node.
    # Since it's a tree, we can root it at an arbitrary node (e.g., 1).
    # An edge (u, v) where v is a child of u is part of the minimal subtree
    # if the number of hubs in v's subtree is > 0 and < t.
    
    hub_count = [0] * (n + 1)
    for h in hubs:
        hub_count[h] = 1
        
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            hub_count[p] += hub_count[u]
            # If this subtree contains some but not all hubs, the edge must be used
            if 0 < hub_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    # Increase recursion depth for deep trees
    sys.setrecursionlimit(300000)
    solve()
```
