```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1000000)
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    if t <= 1:
        print(0)
        return
        
    critical = set()
    for _ in range(t):
        critical.add(int(data[idx]))
        idx += 1
    
    # Build the tree
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # Since it's a tree, we can do a DFS to find the subtree that contains all critical nodes
    # The minimal connected subgraph is the union of paths between all pairs of critical nodes
    # But since it's a tree, the minimal subtree containing all critical nodes is unique
    # This is the Steiner tree problem on a tree, which can be solved efficiently
    
    # Approach:
    # 1. Root the tree at any node
    # 2. For each node, compute whether its subtree contains any critical node
    # 3. An edge should be included in the solution if it lies on the path between two critical nodes
    # 4. In a tree, this is equivalent to including an edge if both its subtrees contain at least one critical node
    
    # Let's root the tree at node 1
    parent = [-1] * (n + 1)
    parent_edge = [-1] * (n + 1)  # stores the weight of edge to parent
    children = defaultdict(list)
    
    # BFS to build rooted tree
    visited = [False] * (n + 1)
    queue = deque([1])
    visited[1] = True
    
    while queue:
        u = queue.popleft()
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge[v] = w
                children[u].append(v)
                queue.append(v)
    
    # Now, for each node, count how many critical nodes are in its subtree
    # We'll do a post-order traversal
    subtree_critical_count = [0] * (n + 1)
    critical_list = list(critical)
    
    # Mark leaves and propagate counts upwards
    def dfs_count(u):
        count = 1 if u in critical else 0
        for v in children[u]:
            count += dfs_count(v)
        subtree_critical_count[u] = count
        return count
    
    dfs_count(1)
    
    # Now, include an edge if it separates critical nodes
    # That is, for each edge (u,v) where v is child of u, 
    # if subtree of v has at least 1 critical node and the rest of the tree also has at least 1
    total_cost = 0
    for u in range(1, n + 1):
        for v in children[u]:
            # Edge from u to v with weight parent_edge[v]
            count_in_v = subtree_critical_count[v]
            count_outside_v = t - count_in_v
            if count_in_v > 0 and count_outside_v > 0:
                total_cost += parent_edge[v]
                
    print(total_cost)

if __name__ == "__main__":
    main()
```
