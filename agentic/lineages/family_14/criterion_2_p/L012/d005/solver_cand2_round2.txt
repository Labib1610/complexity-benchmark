import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary_hubs = set()
    idx = 3
    for _ in range(t):
        primary_hubs.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that lie on the path between any two primary hubs.
    # An edge (u, v) is on such a path if and only if the subtree formed by 
    # removing the edge (u, v) contains at least one primary hub in both components.
    
    is_primary = [False] * (n + 1)
    for h in primary_hubs:
        is_primary[h] = True
        
    count_primary = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order to count primary hubs in subtrees
    edge_weights = {}
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if v != parent[u]:
                edge_weights[(u, v)] = w
                
    for u in reversed(order):
        count_primary[u] += (1 if is_primary[u] else 0)
        if parent[u] != -1:
            count_primary[parent[u]] += count_primary[u]
            
    ans = 0
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if v == parent[u]:
                # Edge is (u, parent[u])
                # Subtree at u has count_primary[u]
                # The other part has t - count_primary[u]
                if count_primary[u] > 0 and (t - count_primary[u]) > 0:
                    ans += w
                    
    print(ans)

if __name__ == '__main__':
    solve()
