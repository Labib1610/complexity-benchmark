import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    auditing_offices = set()
    idx = 3
    for _ in range(t):
        auditing_offices.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The problem states all t Auditing Offices are in the same connected component
    # which is a tree. We need the sum of weights of all edges that lie on the
    # unique path between any two auditing offices.
    # An edge is part of the Steiner Tree if and only if the subtree formed by
    # removing that edge contains at least one auditing office on both sides.
    
    # Pick an arbitrary auditing office as root
    root = next(iter(auditing_offices))
    
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    # stack stores (u, p, edge_weight_to_parent)
    stack = [(root, -1, 0)]
    
    # has_auditing[u] is True if the subtree rooted at u contains an auditing office
    has_auditing = [False] * (n + 1)
    
    # To process post-order, we need to visit children first
    order = []
    parent = [0] * (n + 1)
    edge_to_parent = [0] * (n + 1)
    
    dfs_stack = [root]
    while dfs_stack:
        u = dfs_stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                edge_to_parent[v] = w
                dfs_stack.append(v)
                
    # Initialize has_auditing
    for office in auditing_offices:
        has_auditing[office] = True
        
    # Process in reverse topological order (post-order)
    for u in reversed(order):
        if u == root:
            continue
        p = parent[u]
        if has_auditing[u]:
            total_cost += edge_to_parent[u]
            has_auditing[p] = True
            
    print(total_cost)

solve()
