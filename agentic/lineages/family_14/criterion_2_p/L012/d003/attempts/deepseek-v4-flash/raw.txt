We can solve this by finding the minimal subtree that connects all auditing offices.  
Since the graph is a tree, the minimal connected subgraph containing all required nodes is the union of paths between them.  
We can compute this by marking required nodes and doing a DFS to sum only the edges that lead to at least one required node.

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    offices = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    visited = [False] * (n + 1)
    
    def dfs(u):
        visited[u] = True
        total = 0
        has_office = u in offices
        for v, w in adj[u]:
            if not visited[v]:
                child_has_office = dfs(v)
                if child_has_office:
                    total += w + dfs_cost[v]
                has_office = has_office or child_has_office
        return has_office
    
    dfs_cost = [0] * (n + 1)
    
    def dfs2(u):
        visited[u] = True
        has_office = u in offices
        for v, w in adj[u]:
            if not visited[v]:
                child_has_office = dfs2(v)
                if child_has_office:
                    dfs_cost[u] += w + dfs_cost[v]
                has_office = has_office or child_has_office
        return has_office
    
    visited = [False] * (n + 1)
    dfs2(1)
    
    # Find any office as root to compute minimal subtree
    root = next(iter(offices))
    visited = [False] * (n + 1)
    dfs_cost = [0] * (n + 1)
    
    def dfs3(u):
        visited[u] = True
        has_office = u in offices
        for v, w in adj[u]:

                child_has_office = dfs3(v)
                if child_has_office:
                    dfs_cost[u] += w + dfs_cost[v]
                has_office = has_office or child_has_office
        return has_office
    
    dfs3(root)
    print(dfs_cost[root])
```
