```python
import sys

# The problem asks for the minimum cost to connect all t Auditing Offices in a tree.
# This is equivalent to finding the Steiner Tree in a tree, which is simply the 
# sum of weights of all edges that lie on the unique path between any two 
# Auditing Offices. An edge is part of this set if and only if the subtree 
# formed by removing that edge contains at least one Auditing Office on both sides.

def solve():
    # Increase recursion depth for deep trees
    sys.setrecursionlimit(300000)
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    auditing_offices = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # is_auditing[u] will store the count of auditing offices in the subtree rooted at u
    is_auditing = [0] * (n + 1)
    for office in auditing_offices:
        is_auditing[office] = 1
        
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    edge_weight_to_parent = [0] * (n + 1)
    # We need to store weights associated with the edge to parent
    # Re-traverse to store weights
    temp_adj = [[] for _ in range(n + 1)]
    for u in range(1, n + 1):
        for v, w in adj[u]:
            temp_adj[u].append((v, w))
            
    # Re-calculate weights to parent
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if v == parent[u]:
                edge_weight_to_parent[u] = w
                
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        if is_auditing[u] > 0 and is_auditing[u] < t:
            total_cost += edge_weight_to_parent[u]
        is_auditing[p] += is_auditing[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
