```python
import sys

def solve():
    # Read n, m, t
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_dims = [int(x) for x in input_data[3:3+t]]
    
    # If t=1, the condition is always satisfied with 0 cost
    if t <= 1:
        print(0)
        return
    
    # The operators form a tree structure (m = n-1).
    # We need to select a subset of edges such that all target nodes
    # are in the same connected component.
    # This is equivalent to finding the Minimum Spanning Tree of the 
    # graph where we contract all target nodes into a single super-node.
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # Disjoint Set Union
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # Contract all target dimensions into one component
    for i in range(t - 1):
        union(target_dims[i], target_dims[i+1])
        
    # Kruskal's to connect the remaining components
    edges.sort(key=lambda x: x[2])
    
    total_cost = 0
    for u, v, w in edges:
        if union(u, v):
            total_cost += w
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
