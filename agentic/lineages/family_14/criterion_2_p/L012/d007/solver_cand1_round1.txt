```python
import sys

# The problem asks for the minimum cost to connect all elements in T 
# such that they belong to the same connected component in a graph 
# where edges are the selected transpositions.
# Since the input provides exactly n-1 transpositions that generate S_n,
# the graph formed by all m transpositions is a tree.
# We need to find the minimum cost to connect all nodes in T in this tree.
# This is equivalent to finding the Steiner Tree for the set T in the given tree.
# In a tree, the Steiner Tree is the unique minimal subtree containing all nodes in T.
# The cost is the sum of weights of all edges that lie on the paths between 
# nodes in T. An edge is in the Steiner Tree if and only if it separates 
# the tree into two components, both of which contain at least one node from T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    t_elements = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # count_t[u] will store the number of nodes from T in the subtree rooted at u
    count_t = [0] * (n + 1)
    for node in t_elements:
        count_t[node] = 1
        
    total_cost = 0
    
    # Perform DFS to calculate subtree counts and identify edges in the Steiner Tree
    # Since n is up to 200,000, use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if parent[u] != 0:
            count_t[parent[u]] += count_t[u]
            # If this edge separates T into two non-empty sets, it must be included
            if count_t[u] > 0 and count_t[u] < t:
                total_cost += edge_weight_to_parent[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
