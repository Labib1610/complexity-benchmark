```python
import sys

# The problem asks us to select a subset of edges (updates) such that in the 
# resulting graph, all target indices belong to the same connected component.
# If a set of indices are in the same connected component, the nullspace 
# vectors x satisfy x_i = x_j for all i, j in that component.
# If they are in different components, we can construct a nullspace vector 
# where x_i != x_j.
# Thus, we need to connect all target indices into a single component.
# This is equivalent to finding the Minimum Spanning Tree of the graph 
# where we contract all target indices into a single super-node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    if t <= 1:
        print(0)
        return
    
    # DSU to find MST
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # Contract all target indices into one
    target_root = targets[0]
    for i in range(1, t):
        union(target_root, targets[i])
        
    edges.sort()
    
    total_cost = 0
    for w, u, v in edges:
        if union(u, v):
            total_cost += w
            
    print(total_cost)

solve()
```
