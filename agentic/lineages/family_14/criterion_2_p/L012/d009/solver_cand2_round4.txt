```python
import sys

# The problem asks for the minimum cost to connect all benchmark commodities.
# Since the graph is a tree, the set of edges required to connect a subset of 
# nodes is exactly the set of edges that lie on the paths between any two 
# benchmark nodes. This is equivalent to the union of all paths between 
# benchmark nodes, which is the minimal subtree containing all benchmark nodes.
# We can find this by pruning all branches of the tree that do not contain 
# any benchmark nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    benchmarks = set()
    idx = 3
    for _ in range(t):
        benchmarks.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the minimal subtree, we can use a post-order traversal (DFS).
    # A node is part of the minimal subtree if it is a benchmark node or 
    # if it has at least two branches that contain benchmark nodes.
    # Alternatively, we can count how many benchmark nodes are in each subtree.
    
    is_benchmark = [False] * (n + 1)
    for b in benchmarks:
        is_benchmark[b] = True
        
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    subtree_benchmarks = [0] * (n + 1)
    for i in range(1, n + 1):
        if is_benchmark[i]:
            subtree_benchmarks[i] = 1
            
    # Process in reverse topological order
    for u in reversed(order):
        if u == 1:
            continue
        if subtree_benchmarks[u] > 0:
            total_cost += edge_to_parent_weight[u]
            subtree_benchmarks[parent[u]] += subtree_benchmarks[u]
            
    print(total_cost)

solve()
```
