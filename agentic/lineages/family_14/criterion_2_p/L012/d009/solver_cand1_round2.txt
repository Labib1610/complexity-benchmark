```python
import sys

# The problem states that the m = n - 1 constraints form a tree (a fully connected network with n-1 edges).
# We need to select a subset of edges such that all target variables are in the same connected component.
# Since the original graph is a tree, there is a unique path between any two nodes.
# To connect all target variables, we must select all edges that lie on the unique paths 
# between any pair of target variables.
# This is equivalent to selecting all edges that lie on the unique path between the 
# "first" target variable and every other target variable.
# More simply, an edge is part of the required set if and only if removing it 
# splits the tree into two components, each of which contains at least one target variable.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_targets[u] will store the number of target variables in the subtree rooted at u
    subtree_targets = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in targets:
            subtree_targets[u] += 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_targets[p] += subtree_targets[u]
            # If this edge separates the tree into two parts, both containing at least one target
            if 0 < subtree_targets[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
