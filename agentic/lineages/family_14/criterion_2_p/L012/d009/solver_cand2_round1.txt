```python
import sys

# The problem asks for the minimum cost to connect all benchmark commodities
# in a tree structure formed by the candidate regulations.
# Since the full set of regulations forms a tree (n nodes, n-1 edges),
# any subset of regulations also forms a forest.
# The condition "all benchmark commodities share the same price" means
# that in the forest formed by the selected regulations, all benchmark
# commodities must belong to the same connected component.
# This is equivalent to finding the minimum cost to connect all benchmark
# nodes in the original tree.
# This is a classic problem: the cost is the sum of weights of all edges
# that lie on the unique paths between any two benchmark nodes.
# An edge (u, v) with weight w is part of the solution if and only if
# the removal of this edge splits the tree into two components, each
# containing at least one benchmark node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    benchmarks = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # count_bench[u] will store the number of benchmark nodes in the subtree
    # rooted at u (when rooted at an arbitrary node, say 1).
    count_bench = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in benchmarks:
            count_bench[u] += 1
        if parent[u] != 0:
            count_bench[parent[u]] += count_bench[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the edge to parent separates the tree such that both sides
        # have at least one benchmark node, it must be included.
        # count_bench[i] is the number of benchmarks in the subtree of i.
        # t - count_bench[i] is the number of benchmarks outside.
        if count_bench[i] > 0 and (t - count_bench[i]) > 0:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
