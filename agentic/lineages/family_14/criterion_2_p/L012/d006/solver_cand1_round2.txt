```python
import sys

# The problem asks for the minimum cost to select a subset of edges in a tree
# such that all target nodes are in the same connected component.
# If F(x) = 0, then for every selected edge (u, v), x_u = x_v.
# This means all nodes in the same connected component of the selected edges
# must have the same value.
# To ensure all target nodes have the same value, they must all belong to the
# same connected component in the graph formed by the selected edges.
# Since the original graph is a tree (n nodes, n-1 edges, connected),
# the unique path between any two nodes in the tree must be included in the
# selected subset to connect them.
# To connect all target nodes, we must select all edges that lie on the
# unique paths between any pair of target nodes.
# This is equivalent to selecting all edges that lie on the minimal subtree
# spanning all target nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the sum of weights of all edges that are part of the
    # minimal subtree connecting all target nodes.
    # An edge is in this subtree if and only if both components formed by
    # removing the edge contain at least one target node.
    
    subtree_target_count = [0] * (n + 1)
    total_cost = 0
    
    # Root the tree at an arbitrary target node
    root = next(iter(targets))
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in targets:
            subtree_target_count[node] += 1
            
    # Process in reverse order to propagate counts up
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_target_count[p] += subtree_target_count[u]
            # If this subtree contains some but not all targets, the edge must be included
            if 0 < subtree_target_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
