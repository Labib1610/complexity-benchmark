```python
import sys

# The problem asks for the minimum cost to ensure that all t critical indicators
# are in the same connected component in the graph formed by the selected constraints.
# Since the constraints form a tree (n nodes, n-1 edges, linearly independent),
# we are looking for the minimum cost to connect all t critical nodes in this tree.
# This is equivalent to the Steiner Tree problem on a tree, which is simply the
# sum of the weights of all edges that lie on the paths between any two critical nodes.
# An edge is part of the Steiner tree if and only if the subtree formed by removing
# that edge contains at least one critical node on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_indices = set()
    idx = 3
    for _ in range(t):
        critical_indices.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count how many critical nodes are in each subtree
    subtree_critical_count = [0] * (n + 1)
    
    # Since it's a tree, we can use DFS to compute subtree counts
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for i in range(1, n + 1):
        if i in critical_indices:
            subtree_critical_count[i] = 1
            
    total_cost = 0
    # Process in reverse topological order
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        # Find weight of edge (u, p)
        weight = 0
        for v, w in adj[u]:
            if v == p:
                weight = w
                break
        
        # If the subtree at u contains some but not all critical nodes,
        # the edge (u, p) must be included.
        count = subtree_critical_count[u]
        if count > 0 and count < t:
            total_cost += weight
        
        subtree_critical_count[p] += count
        
    print(total_cost)

solve()
```
