```python
import sys

# The structural index I(x) = sum(x_i) - sum(x_u * x_v) for a tree is equal to the number of connected components of active nodes.
# We need I(x) = 1, meaning the active nodes must form a single connected subtree.
# Since all critical nodes must be active, the subtree must contain all critical nodes.
# The smallest such subtree is the Steiner Tree of the critical nodes.
# Any valid configuration must be a connected subtree containing all critical nodes.
# Let S be the set of nodes in the Steiner Tree. If we add any node v not in S to the configuration,
# we must add it such that the configuration remains connected.
# If we add a node v adjacent to the current subtree, the number of active nodes increases by 1,
# and the number of edges (x_u * x_v) increases by 1.
# Thus, I(x) remains 1.
# The cost increases by the weight of the edge connecting v to the subtree.
# To minimize cost, we should only include nodes in the Steiner Tree.
# The Steiner Tree is the union of all paths between any two critical nodes.
# This is equivalent to the minimal subtree containing all critical nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # To find the Steiner Tree:
    # 1. Root the tree at an arbitrary critical node.
    # 2. A node is in the Steiner Tree if it is a critical node or if it lies on a path between two critical nodes.
    # 3. This is equivalent to: a node is in the Steiner Tree if its subtree (in the rooted sense) contains at least one critical node,
    #    AND it is an ancestor of some critical node (or is one).
    
    root = next(iter(critical_nodes))
    is_critical = [False] * (n + 1)
    for node in critical_nodes:
        is_critical[node] = True
        
    has_critical_in_subtree = [False] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_critical[u]:
            has_critical_in_subtree[u] = True
        if has_critical_in_subtree[u] and parent[u] != 0:
            has_critical_in_subtree[parent[u]] = True
            
    total_cost = 0
    for i in range(1, n + 1):
        if has_critical_in_subtree[i] and parent[i] != 0:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
