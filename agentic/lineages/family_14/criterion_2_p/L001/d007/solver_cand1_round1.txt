```python
import sys

# The problem asks for the minimum cost to activate a subset of nodes such that
# the induced subgraph is connected and contains all primary nodes.
# The diagnostic test condition is equivalent to saying the induced subgraph
# must be a tree (connected and acyclic). Since the original graph is a tree,
# any connected subgraph is a tree.
# Thus, we need the minimum weight Steiner Tree containing all primary nodes.
# For a tree, the Steiner Tree is the minimal subtree connecting all primary nodes.
# The cost is the sum of weights of all edges in the minimal subtree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    primary_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # To find the Steiner Tree in a tree:
    # The edges in the Steiner Tree are exactly those that lie on the unique
    # path between some pair of primary nodes.
    # We can use a post-order traversal (or DFS) to mark nodes that are part
    # of the Steiner Tree.
    
    is_primary = [False] * (n + 1)
    for p in primary_nodes:
        is_primary[p] = True
        
    subtree_has_primary = [False] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_primary[u]:
            subtree_has_primary[u] = True
        
        if subtree_has_primary[u] and parent[u] != 0:
            total_cost += edge_to_parent_weight[u]
            subtree_has_primary[parent[u]] = True
            
    print(total_cost)

solve()
```
