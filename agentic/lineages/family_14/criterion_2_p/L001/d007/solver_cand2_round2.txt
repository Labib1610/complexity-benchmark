import sys

# The problem asks for the minimum cost to activate a subset of nodes including all primary nodes.
# A configuration is valid if the activated nodes form a subtree (since the graph is a tree)
# and the condition about the sequence is satisfied.
# The condition "for every index i from 1 to k-1, the module v_i is directly connected 
# to exactly one module that appears later in the sequence" is equivalent to saying 
# that the activated nodes form a connected subgraph (a subtree) and every node 
# in the subtree has degree at most 2 in the induced subgraph, except for the last node.
# Actually, the condition implies that the activated nodes must form a path.
# Wait, let's re-read: "v_i is connected to exactly one module that appears later".
# This means the induced subgraph must be a path.
# If the induced subgraph is a path, then every node except the last one has 
# exactly one neighbor later in the sequence.
# So the problem is to find a path in the tree that contains all primary nodes 
# such that the sum of weights of edges in the path is minimized.
# Wait, the cost is the sum of weights of all energized channels.
# If the activated nodes form a path, all edges in the path are energized.
# So we need to find the smallest subtree that contains all primary nodes.
# The smallest subtree containing a set of nodes is the union of all paths 
# between the primary nodes. This is the Steiner Tree in a tree, which is 
# simply the minimal subtree connecting all primary nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    primary = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Find the minimal subtree containing all primary nodes.
    # This is done by pruning all leaves that are not primary nodes.
    degree = [0] * (n + 1)
    for i in range(1, n + 1):
        degree[i] = len(adj[i])
        
    queue = [i for i in range(1, n + 1) if degree[i] == 1 and i not in primary]
    
    is_removed = [False] * (n + 1)
    total_weight = 0
    for u, v, w in []: pass # dummy
    
    # We need to sum the weights of all edges that are part of the minimal subtree.
    # An edge is in the minimal subtree if it lies on a path between two primary nodes.
    # Let's use DFS to find the subtree.
    
    # Root the tree at an arbitrary primary node.
    root = next(iter(primary))
    
    # subtree_has_primary[u] is true if the subtree rooted at u contains a primary node.
    subtree_has_primary = [False] * (n + 1)
    
    def dfs(u, p):
        has_p = (u in primary)
        weight_sum = 0
        for v, w in adj[u]:
            if v == p:
                continue
            child_has_p, child_weight = dfs(v, u)
            if child_has_p:
                has_p = True
                weight_sum += child_weight + w
        return has_p, weight_sum

    _, result = dfs(root, -1)
    print(result)

solve()
