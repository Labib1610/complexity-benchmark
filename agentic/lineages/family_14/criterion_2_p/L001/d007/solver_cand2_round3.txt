```python
import sys

# The problem asks for the minimum cost to connect all primary modules.
# A set of modules is valid if they form a connected component in the tree
# where all edges in the component are "energized".
# If we activate a set of modules, all edges on the unique paths between
# any two activated modules must be energized.
# This implies that if we activate a set of modules, we must activate all
# modules on the paths between them to keep the configuration valid.
# Thus, the set of activated modules must be a subtree that contains all
# primary modules. The cost is the sum of weights of all edges in this subtree.
# This is equivalent to the Steiner Tree problem on a tree, which is simply
# the sum of weights of all edges on the unique paths connecting all primary nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    primary_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the sum of weights of all edges that lie on the path
    # between any two primary nodes. An edge is part of this set if and only if
    # the subtree formed by removing the edge splits the primary nodes into
    # two non-empty sets.
    
    # Count how many primary nodes are in the subtree of each node (rooted at 1)
    subtree_primary_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in primary_nodes:
            subtree_primary_count[u] += 1
        if parent[u] != 0:
            subtree_primary_count[parent[u]] += subtree_primary_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        count = subtree_primary_count[i]
        # If the edge to parent separates primary nodes, it must be included
        if count > 0 and count < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
