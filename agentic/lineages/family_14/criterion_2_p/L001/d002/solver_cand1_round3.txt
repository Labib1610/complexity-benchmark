```python
import sys

# The problem states the network is a tree (n nodes, n-1 edges, no loops).
# We need to connect all primary locations.
# In a tree, there is a unique path between any two nodes.
# To connect a set of primary nodes, we must activate all edges that lie on
# the paths between any two primary nodes.
# This is equivalent to finding the smallest subtree that contains all primary nodes.
# Since the graph is already a tree, we just need to identify which edges
# are part of the paths connecting the primary nodes.
# An edge (u, v) is part of the required set if and only if the subtree
# formed by removing (u, v) contains at least one primary node in both components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary_nodes = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many primary nodes are in the subtree of each node.
    # Since it's a tree, we can root it at an arbitrary node (e.g., 1).
    # For each edge (u, v) where v is a child of u, the edge is included
    # if the number of primary nodes in v's subtree is > 0 and < t.
    
    primary_count = [0] * (n + 1)
    for p in primary_nodes:
        primary_count[p] = 1
        
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order to propagate primary counts
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != 0:
            # If this subtree contains some but not all primary nodes,
            # the edge to the parent must be activated.
            if 0 < primary_count[u] < t:
                total_cost += edge_to_parent_weight[u]
            primary_count[parent[u]] += primary_count[u]
            
    print(total_cost)

if __name__ == '__main__':
    # Increase recursion depth for deep trees
    sys.setrecursionlimit(300000)
    solve()
```
