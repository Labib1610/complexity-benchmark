```python
import sys

# The problem states the network is a tree (n nodes, n-1 edges, connected).
# We need to find the minimum weight of the subtree that connects all command outposts.
# This is equivalent to the sum of weights of all edges that lie on the unique path
# between any two command outposts.
# An edge is part of this subtree if and only if the subtree formed by removing
# that edge splits the command outposts into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    commands = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many command nodes are in the subtree of each node.
    # Since n is up to 200,000, we use iterative DFS to avoid recursion depth issues.
    
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    command_count = [0] * (n + 1)
    for node in commands:
        command_count[node] = 1
        
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If this subtree contains some but not all command nodes,
            # the edge to the parent must be included.
            if 0 < command_count[u] < t:
                total_weight += edge_weight_to_parent[u]
            command_count[p] += command_count[u]
            
    print(total_weight)

solve()
```
