```python
import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    primary = set(map(int, sys.stdin.readline().split()))
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    if t <= 1:
        print(0)
        return
    
    # Sort edges by weight
    edges.sort()
    
    # Union-Find data structure
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    # Count how many primary nodes are connected
    primary_components = {}
    for p in primary:
        primary_components[p] = p
    
    total_cost = 0
    connected_primaries = 1  # Start with one component
    
    for w, u, v in edges:
        if union(u, v):
            # Check if we connected two different primary components
            pu = find(u)
            pv = find(v)
            
            # Get the representative primary node for u and v if they exist
            rep_u = None
            rep_v = None
            
            for p in primary:
                if find(p) == pu:
                    rep_u = p
                if find(p) == pv:
                    rep_v = p
                    
            # If both u and v are in primary nodes and they belong to different components
            if rep_u is not None and rep_v is not None and find(rep_u) != find(rep_v):
                connected_primaries += 1
                # Merge the components
                union(rep_u, rep_v)
                
            if connected_primaries == t:
                print(total_cost + w)
                return
                
    print(0)

threading.Thread(target=main).start()
```
