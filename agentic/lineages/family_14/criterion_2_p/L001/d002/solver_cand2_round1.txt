```python
import sys

# The problem states m = n - 1, meaning the graph is a tree.
# If the graph is a tree, there is exactly one path between any two nodes.
# To connect a set of primary hubs, we must include all edges that lie on the
# unique paths between any two primary hubs.
# An edge (u, v) with weight w is part of the required set if and only if
# removing it splits the tree into two components, each containing at least one primary hub.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary_hubs = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w, _))
        adj[v].append((u, w, _))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_primary_count[u] stores how many primary hubs are in the subtree rooted at u
    subtree_primary_count = [0] * (n + 1)
    
    # DFS to calculate subtree_primary_count
    # Since n can be up to 200,000, use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w, _ in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in primary_hubs:
        subtree_primary_count[u] = 1
        
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if subtree_primary_count[u] > 0 and subtree_primary_count[u] < t:
            total_weight += edge_to_parent_weight[u]
        subtree_primary_count[p] += subtree_primary_count[u]
        
    print(total_weight)

solve()
```
