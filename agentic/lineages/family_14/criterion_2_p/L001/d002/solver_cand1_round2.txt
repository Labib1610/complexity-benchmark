```python
import sys

# The problem asks for the sum of weights of all edges that lie on the unique 
# paths between any two primary locations.
# In a tree, the union of paths between a set of nodes S is the minimal 
# subtree containing all nodes in S (the Steiner Tree).
# An edge belongs to this subtree if and only if removing it splits the 
# tree into two components, each containing at least one primary location.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        primary[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many primary nodes are in the subtree rooted at each node
    # when rooted at an arbitrary node (e.g., 1).
    count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if primary[u]:
            count[u] += 1
        if parent[u] != 0:
            count[parent[u]] += count[u]
            
    total_cost = 0
    # An edge (u, parent[u]) is part of the Steiner tree if the number of 
    # primary nodes in the subtree of u is > 0 and < t.
    for i in range(2, n + 1):
        if count[i] > 0 and count[i] < t:
            total_cost += parent_edge_weight[i]
            
    print(total_cost)

solve()
```
