```python
import sys

# The problem states the network is non-redundant and m = n - 1.
# This means the campus is a tree.
# We need to find the minimum cost to connect all mandatory sites.
# In a tree, the set of edges that connect a subset of nodes is the union
# of all paths between any two mandatory nodes.
# This is equivalent to the sum of weights of all edges that have at least
# one mandatory node in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    mandatory = set()
    idx = 3
    for _ in range(t):
        mandatory.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count_mandatory[u] will store the number of mandatory nodes in the subtree of u
    count_mandatory = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in mandatory:
            count_mandatory[u] = 1
            
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        count_mandatory[p] += count_mandatory[u]
        # If this edge separates some mandatory nodes from others, it must be included
        if count_mandatory[u] > 0 and count_mandatory[u] < t:
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
