```python
import sys

# The problem asks for a minimum weight connected subgraph that contains all primary modules.
# The condition "exactly one activated module has 0 outgoing energized channels, 
# and every other activated module has 1 outgoing energized channel" is equivalent 
# to saying the energized channels form a tree (a connected structure with no cycles).
# Since the original graph is a tree, any subset of nodes that includes all primary 
# modules and forms a connected component will satisfy the condition.
# The minimum weight connected subgraph containing a set of nodes in a tree is 
# the Steiner Tree, which is the union of all paths between the primary nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    primary_modules = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the sum of weights of all edges that lie on the path 
    # between any two primary nodes. An edge is part of the Steiner tree 
    # if and only if the subtree formed by removing the edge contains 
    # at least one primary node on both sides.
    
    is_primary = [False] * (n + 1)
    for p in primary_modules:
        is_primary[p] = True
        
    subtree_primary_count = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    order = []
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        subtree_primary_count[u] += (1 if is_primary[u] else 0)
        if parent[u] != 0:
            subtree_primary_count[parent[u]] += subtree_primary_count[u]
            
    total_weight = 0
    for i in range(2, n + 1):
        count = subtree_primary_count[i]
        if count > 0 and count < t:
            total_weight += edge_to_parent_weight[i]
            
    print(total_weight)

solve()
```
