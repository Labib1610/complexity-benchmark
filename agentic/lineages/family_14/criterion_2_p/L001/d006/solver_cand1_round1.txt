```python
import sys

# The problem asks for a subset of nodes S such that the sum of (2 - c(u)) for u in S is 2.
# c(u) is the number of neighbors of u in S.
# Sum_{u in S} (2 - c(u)) = 2 * |S| - Sum_{u in S} c(u).
# Since each energized edge (u, v) contributes 1 to c(u) and 1 to c(v),
# Sum_{u in S} c(u) = 2 * |E_S|, where E_S is the set of energized edges.
# So, 2 * |S| - 2 * |E_S| = 2, which simplifies to |S| - |E_S| = 1.
# In a forest, |S| - |E_S| is the number of connected components in the subgraph induced by S.
# Thus, we need the induced subgraph to have exactly 1 connected component.
# This means S must be a connected subtree containing all primary nodes.
# The minimum cost to connect a set of nodes in a tree is the weight of the Steiner Tree.
# The Steiner Tree for a set of nodes T is the union of all paths between nodes in T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    primary = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t == 0:
        print(0)
        return
    
    # Steiner Tree: find the smallest subtree containing all primary nodes.
    # This is equivalent to the sum of weights of all edges that lie on at least one path
    # between two primary nodes.
    # An edge (u, v) with weight w is in the Steiner Tree if and only if
    # the subtree at u (when rooted at v) contains at least one primary node
    # AND the subtree at v (when rooted at u) contains at least one primary node.
    
    is_primary = [False] * (n + 1)
    for p in primary:
        is_primary[p] = True
        
    has_primary = [False] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [primary[0]]
    visited = [False] * (n + 1)
    visited[primary[0]] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    # Post-order traversal to propagate primary node counts
    for u in reversed(order):
        if is_primary[u]:
            has_primary[u] = True
        if parent[u] != 0:
            p, w = parent[u]
            if has_primary[u]:
                has_primary[p] = True
                total_cost += w
                
    print(total_cost)

solve()
```
