```python
import sys

# The condition "sum of connection counts of deactivated modules = number of deactivated modules + number of links where both endpoints are deactivated"
# is equivalent to saying that the set of deactivated modules must form a forest (a collection of trees)
# where each tree is a connected component in the subgraph induced by deactivated modules.
# In a tree, the sum of degrees equals 2 * (number of edges).
# Let D be the set of deactivated modules, E_D be the set of edges with both endpoints in D.
# Sum_{v in D} deg(v) = 2 * |E_D| + |E_{boundary}|, where E_{boundary} is the set of edges with one endpoint in D and one in A (activated).
# The condition given is Sum_{v in D} deg(v) = |D| + |E_D|.
# Since for a forest, |E_D| = |D| - (number of connected components in D),
# the condition simplifies to: Sum_{v in D} deg(v) = |D| + |D| - k = 2|D| - k.
# This is equivalent to saying that the deactivated modules must form a set of components such that
# each component is a tree and the boundary edges are handled correctly.
# Actually, the condition is equivalent to saying that the set of activated modules must be a connected subtree
# containing all critical modules.
# Let A be the set of activated modules. The cost is the sum of weights of edges in the induced subgraph of A.
# This is equivalent to: Total weight of all edges - weight of edges in the induced subgraph of D.
# To minimize the cost, we maximize the weight of edges in the induced subgraph of D.
# D must be a forest such that all critical modules are in A.
# This is equivalent to finding a Steiner Tree for the critical modules.
# The cost is the sum of weights of all edges minus the weight of the maximum weight forest in D.
# Since D must be the complement of a connected subtree containing all critical nodes,
# we are looking for the minimum weight Steiner Tree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, t = map(int, input_data[:3])
    critical = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    total_weight = 0
    idx = 3 + t
    for _ in range(m):
        u, v, w = map(int, input_data[idx:idx+3])
        adj[u].append((v, w))
        adj[v].append((u, w))
        total_weight += w
        idx += 3

    # DP on tree: dp[u][0] = max weight of forest in subtree u if u is in D
    # dp[u][1] = max weight of forest in subtree u if u is in A
    # If u is in A, it cannot be part of the forest in D.
    # If u is in D, it can be part of a forest.
    
    dp0 = [0] * (n + 1)
    dp1 = [0] * (n + 1)
    
    # Post-order traversal
    stack = [(1, 0)]
    order = []
    while stack:
        u, p = stack.pop()
        order.append((u, p))
        for v, w in adj[u]:
            if v != p:
                stack.append((v, u))
    
    for u, p in reversed(order):
        if u in critical:
            dp0[u] = -float('inf')
            dp1[u] = 0
            for v, w in adj[u]:
                if v != p:
                    dp1[u] += max(dp0[v] + w, dp1[v])
        else:
            # u is in D
            dp0[u] = 0
            for v, w in adj[u]:
                if v != p:
                    dp0[u] += max(dp0[v] + w, dp1[v])
            # u is in A
            dp1[u] = 0
            for v, w in adj[u]:
                if v != p:
                    dp1[u] += max(dp0[v], dp1[v])
                    
    print(total_weight - max(dp0[1], dp1[1]))

solve()
```
