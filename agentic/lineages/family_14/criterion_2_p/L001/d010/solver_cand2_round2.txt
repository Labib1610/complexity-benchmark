```python
import sys

# The problem states that the network is a tree (m = n - 1).
# Let A be the set of active outposts.
# Each active outpost i has a baseline of 2 tokens.
# Each link (u, v) where u, v in A consumes 1 token from u and 1 from v.
# Total surplus = sum_{i in A} (2 - degree_A(i)) = 2 * |A| - sum_{i in A} degree_A(i).
# Since sum_{i in A} degree_A(i) = 2 * |E_A| (where E_A is the set of links between active outposts),
# Surplus = 2 * |A| - 2 * |E_A| = 2.
# This simplifies to |A| - |E_A| = 1.
# In a forest, |A| - |E_A| is the number of connected components formed by A.
# So, the condition "net token surplus = 2" is equivalent to "the active outposts form exactly one connected component".
# Since the original graph is a tree, the set of active outposts A must form a connected subtree.
# We need to find a subtree that contains all critical outposts and minimizes the sum of weights of edges in the subtree.
# The minimum weight subtree containing a set of nodes is the Steiner Tree.
# For a tree, the Steiner Tree is simply the unique path connecting all nodes in the set.
# The edges in the Steiner Tree are all edges on the paths between any two critical outposts.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, t = map(int, input_data[:3])
    critical = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u, v, w = map(int, input_data[idx:idx+3])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3

    # Find the smallest subtree containing all critical nodes.
    # This is the union of all paths between critical nodes.
    # We can identify these edges by checking if a subtree contains at least one critical node
    # and at least one critical node is outside the subtree.
    
    has_critical = [False] * (n + 1)
    for c in critical:
        has_critical[c] = True
        
    subtree_has_critical = [0] * (n + 1)
    
    # Root the tree at an arbitrary critical node (if t > 0)
    root = next(iter(critical))
    
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Count critical nodes in each subtree
    total_critical = t
    count = [0] * (n + 1)
    for i in range(1, n + 1):
        if has_critical[i]:
            count[i] = 1
            
    ans = 0
    for u in reversed(order):
        if u == root:
            continue
        p = parent[u]
        count[p] += count[u]
        # If the edge (u, p) separates the tree into two parts, both containing at least one critical node,
        # then this edge must be part of the Steiner Tree.
        if count[u] > 0 and count[u] < total_critical:
            ans += edge_to_parent_weight[u]
            
    print(ans)

solve()
```
