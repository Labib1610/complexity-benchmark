```python
import sys

# The problem asks to minimize sum(W_j * X_u * X_v) subject to:
# sum((X_u - X_v)^2) + sum((2 - D_i) * X_i) = 2
# Since m = n - 1 and the graph is connected, it's a tree.
# Let S be the set of vertices where X_i = 1.
# sum((X_u - X_v)^2) is the number of edges between S and V\S (the cut size).
# sum((2 - D_i) * X_i) = sum(2 * X_i) - sum(D_i * X_i)
# sum(D_i * X_i) is the sum of degrees of vertices in S.
# In a tree, sum(D_i * X_i) = 2 * |Edges inside S| + |Edges between S and V\S|.
# So the equation becomes:
# CutSize + 2 * |S| - (2 * |Edges inside S| + CutSize) = 2
# 2 * |S| - 2 * |Edges inside S| = 2
# |S| - |Edges inside S| = 1
# This is a property of a forest with |S| vertices and |Edges inside S| edges.
# |S| - |Edges inside S| is the number of connected components in the subgraph induced by S.
# So the condition is: the subgraph induced by S must have exactly 1 connected component.
# Thus, S must be a connected subtree of the original tree.
# We are given that a set of critical vertices T must be in S.
# The smallest subtree containing all vertices in T is the Steiner Tree of T.
# Any valid S must be a connected subtree containing the Steiner Tree of T.
# We want to minimize sum(W_j * X_u * X_v) for all edges (u, v) in the subtree S.
# This is equivalent to minimizing the sum of weights of edges in S.
# Since S must contain the Steiner Tree, and adding edges to S only increases the cost,
# the minimum cost is achieved when S is exactly the Steiner Tree of T.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, t = map(int, input_data[:3])
    critical = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u, v, w = map(int, input_data[idx:idx+3])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3

    # Steiner Tree in a tree is the unique path between all nodes in T.
    # This is the union of all paths between nodes in T.
    # We can find this by pruning the tree:
    # Any leaf node not in T can be removed iteratively.
    
    degree = [0] * (n + 1)
    for i in range(1, n + 1):
        degree[i] = len(adj[i])
    
    is_in_steiner = [False] * (n + 1)
    queue = []
    for i in range(1, n + 1):
        if degree[i] == 1 and i not in critical:
            queue.append(i)
            
    removed = [False] * (n + 1)
    while queue:
        u = queue.pop()
        removed[u] = True
        for v, w in adj[u]:
            if not removed[v]:
                degree[v] -= 1
                if degree[v] == 1 and v not in critical:
                    queue.append(v)
                    
    # The Steiner Tree consists of all nodes not removed.
    # The cost is the sum of weights of edges where both endpoints are not removed.
    total_cost = 0
    for u in range(1, n + 1):
        if not removed[u]:
            for v, w in adj[u]:
                if u < v and not removed[v]:
                    total_cost += w
    print(total_cost)

solve()
```
