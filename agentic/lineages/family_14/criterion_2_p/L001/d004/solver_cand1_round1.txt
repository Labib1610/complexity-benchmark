```python
import sys

# The problem states that the graph is a tree (n variables, n-1 relations, 
# exactly one simple sequence between any two).
# Let S be the set of variables where V_i = 1.
# The condition sum(E_{u,v}) = |S| - 1 means that the subgraph induced by S
# must be a tree (connected and acyclic).
# Since all core variables must be in S, S must be a connected subtree 
# containing all core variables.
# The cost is the sum of weights of all edges (u, v) where both u and v are in S.
# This is equivalent to the sum of weights of all edges in the minimal subtree 
# connecting all core variables.
# Let T_core be the minimal subtree containing all core variables.
# Any node in S must be part of T_core, because if we add a node v not in T_core
# to S, we must add the path from v to T_core to keep S connected.
# Adding an edge (u, v) to the set of activated edges increases the cost by w(u, v).
# The condition sum(E_{u,v}) = |S| - 1 is satisfied if and only if the 
# activated edges form a tree.
# Thus, we need to find a subtree S that contains all core variables such that
# the sum of weights of edges in S is minimized.
# Wait, the cost is the sum of weights of edges (u, v) where u, v in S.
# This is exactly the sum of weights of edges in the minimal subtree connecting 
# all core variables. Any additional nodes added to S would only increase the cost.
# So the answer is the sum of weights of all edges in the minimal subtree 
# connecting all core variables.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core_vars = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Find the minimal subtree containing all core variables.
    # This is the union of paths between all pairs of core variables.
    # We can use post-order traversal to mark nodes that are part of this subtree.
    
    is_core = [False] * (n + 1)
    for c in core_vars:
        is_core[c] = True
        
    has_core_in_subtree = [False] * (n + 1)
    total_cost = 0
    
    # Root the tree at an arbitrary core variable
    root = next(iter(core_vars))
    
    def dfs(u, p):
        nonlocal total_cost
        subtree_has_core = is_core[u]
        for v, w in adj[u]:
            if v == p:
                continue
            child_has_core = dfs(v, u)
            if child_has_core:
                total_cost += w
                subtree_has_core = True
        return subtree_has_core

    dfs(root, -1)
    print(total_cost)

solve()
```
