```python
import sys

# The problem states that the m equivalences form a tree (since m = n-1 for a connected graph).
# We are given a tree of n nodes and m = n-1 edges.
# We need to find the minimum weight subset of edges such that all t core nodes are in the same connected component.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of nodes {c1, c2, ..., ct} in a tree, we must include all edges that lie on the paths
# between any pair of these nodes.
# This is equivalent to finding the minimal subtree that contains all t core nodes.
# The edges in this minimal subtree are exactly those edges that, when removed, split the tree into
# components such that at least one component contains some core nodes and another contains the rest.
# More simply: an edge (u, v) is part of the minimal subtree if and only if the two components
# formed by removing (u, v) both contain at least one core node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    core_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_core_count[u] will store the number of core nodes in the subtree rooted at u
    subtree_core_count = [0] * (n + 1)
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    for u in reversed(order):
        if u in core_nodes:
            subtree_core_count[u] += 1
        if parent[u] != -1:
            subtree_core_count[parent[u]] += subtree_core_count[u]
            
    total_weight = 0
    for u, v, w in edges:
        # The edge connects u and v. One is the parent of the other.
        child = u if parent[u] == v else v
        count = subtree_core_count[child]
        # If the subtree at 'child' contains some core nodes (but not all),
        # then this edge is necessary.
        if 0 < count < t:
            total_weight += w
            
    print(total_weight)

solve()
```
