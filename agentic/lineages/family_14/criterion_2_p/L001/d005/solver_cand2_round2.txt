```python
import sys

# The condition X^T A X = 2 * X^T X - 2 can be rewritten.
# X^T A X is the sum of 2 * (number of energized edges).
# Let k be the number of activated modules (X^T X = k).
# Let E be the set of energized edges.
# The condition is 2 * |E| = 2 * k - 2, which simplifies to |E| = k - 1.
# In a tree, a set of k vertices forms a connected component if and only if
# the number of edges between them is k - 1.
# Thus, the condition is equivalent to: the set of activated modules must
# form a single connected component in the tree.
# Since all primary modules must be activated, the set of activated modules
# must be a connected subgraph containing all primary modules.
# To minimize the sum of weights of edges in this subgraph, we must choose
# the smallest connected subgraph (a subtree) that contains all primary modules.
# This is the Steiner Tree problem on a tree, which is simply the union of
# all paths between the primary modules.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    primary_modules = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need the sum of weights of edges that lie on paths between any two primary modules.
    # An edge is part of the Steiner tree if and only if both components formed by
    # removing the edge contain at least one primary module.
    
    subtree_primary_count = [0] * (n + 1)
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    for u in reversed(order):
        if u in primary_modules:
            subtree_primary_count[u] += 1
        if parent[u] != 0:
            p, w = parent[u]
            subtree_primary_count[p] += subtree_primary_count[u]
            # If this edge separates the tree into two parts, both having primary modules
            if 0 < subtree_primary_count[u] < t:
                total_weight += w
                
    print(total_weight)

solve()
```
