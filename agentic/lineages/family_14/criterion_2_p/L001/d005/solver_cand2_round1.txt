```python
import sys

# The problem states m = n - 1 and the graph is a tree.
# The condition sum(V_i * d_i) = 2 * (sum(V_i) - 1) is equivalent to saying
# that the subgraph induced by the active components is a connected tree.
# If sum(V_i) = 1, the condition is 0 = 2(1) - 2 = 0, which is always true.
# If sum(V_i) > 1, the condition is satisfied if and only if the active nodes
# form a connected subtree.
# We need to find a subtree that contains all primary nodes and minimizes the
# sum of weights of edges within the subtree.
# This is the Steiner Tree problem on a tree, which is simply the unique path
# connecting all primary nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    primary = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Find the smallest subtree containing all primary nodes.
    # This is the union of paths between all primary nodes.
    # We can identify this by pruning all leaves that are not primary nodes
    # and are not on a path between primary nodes.
    
    is_primary = [False] * (n + 1)
    for p in primary:
        is_primary[p] = True
        
    degree = [0] * (n + 1)
    for i in range(1, n + 1):
        degree[i] = len(adj[i])
        
    queue = []
    for i in range(1, n + 1):
        if degree[i] == 1 and not is_primary[i]:
            queue.append(i)
            
    removed = [False] * (n + 1)
    total_weight = 0
    
    # Calculate total weight of all edges first
    all_edges_weight = 0
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if u < v:
                all_edges_weight += w
                
    # Prune the tree
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        removed[u] = True
        for v, w in adj[u]:
            if not removed[v]:
                total_weight += w
                degree[v] -= 1
                if degree[v] == 1 and not is_primary[v]:
                    queue.append(v)
                    
    print(all_edges_weight - total_weight)

solve()
```
