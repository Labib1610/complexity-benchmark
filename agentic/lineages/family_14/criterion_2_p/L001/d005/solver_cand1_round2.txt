```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    primary_nodes = set()
    idx = 3
    for _ in range(t):
        primary_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # The problem asks for the sum of weights of all edges that lie on the path
    # between any two primary nodes. An edge is part of such a path if and only if
    # the subtree formed by removing the edge contains at least one primary node
    # on both sides.
    
    # Let's root the tree at an arbitrary node, say 1.
    # For each edge (u, v) where v is a child of u, the edge is active if
    # the subtree at v contains at least one primary node AND the rest of the
    # tree (outside the subtree at v) contains at least one primary node.
    
    has_primary = [False] * (n + 1)
    for p in primary_nodes:
        has_primary[p] = True
        
    subtree_primary_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        subtree_primary_count[u] += (1 if has_primary[u] else 0)
        if parent[u] != 0:
            subtree_primary_count[parent[u]] += subtree_primary_count[u]
            
    for i in range(2, n + 1):
        count_in_subtree = subtree_primary_count[i]
        if count_in_subtree > 0 and count_in_subtree < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

solve()
```
