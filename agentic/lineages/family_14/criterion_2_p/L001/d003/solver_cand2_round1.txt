```python
import sys

# The problem states m = n - 1. This means the graph is a tree.
# In a tree, there is exactly one path between any two nodes.
# To connect all core sites, we must include all edges that lie on the 
# unique paths between any two core sites.
# Since the graph is a tree, the union of all paths between core sites 
# forms a subtree that contains all core sites and the minimum number of 
# edges to connect them.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    core_sites = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that are part of the minimal subtree 
    # connecting all core sites.
    # An edge (u, v) with weight w is in the minimal subtree if and only if
    # the subtree rooted at u (when the edge is removed) contains at least 
    # one core site, and the subtree rooted at v contains at least one core site.
    
    # Since it's a tree, we can root it at an arbitrary core site.
    root = next(iter(core_sites))
    
    has_core = [False] * (n + 1)
    for c in core_sites:
        has_core[c] = True
        
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == root:
            continue
        
        p, w = parent[u]
        if has_core[u]:
            total_cost += w
            has_core[p] = True
            
    print(total_cost)

solve()
```
