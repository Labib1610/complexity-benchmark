```python
import sys

# The problem states m = n - 1 and the graph is connected, so it's a tree.
# A conduit (edge) is indispensable if removing it disconnects at least one pair
# of critical processors. In a tree, removing an edge splits the tree into two
# components. The edge is indispensable if and only if both components contain
# at least one critical processor.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Use iterative DFS to avoid recursion depth issues
    subtree_critical_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in critical_nodes:
            subtree_critical_count[u] += 1
        if parent[u] != -1:
            subtree_critical_count[parent[u]] += subtree_critical_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        count = subtree_critical_count[i]
        # If the subtree at i contains some critical nodes and the rest of the
        # tree also contains some, the edge to parent is indispensable.
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
