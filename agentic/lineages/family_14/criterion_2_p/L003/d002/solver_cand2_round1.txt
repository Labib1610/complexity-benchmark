```python
import sys

# The problem describes a tree structure (N modules, N-1 conduits, connected).
# We need to connect a subset of critical processors.
# In a tree, the path between any two nodes is unique.
# To connect all critical processors, we must include all edges that lie on the 
# unique paths between any two critical processors.
# This is equivalent to finding the smallest subtree containing all critical nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    K = int(input_data[2])
    
    if K <= 1:
        print(0)
        return
    
    critical = set(map(int, input_data[3:3+K]))
    adj = [[] for _ in range(N + 1)]
    
    idx = 3 + K
    for _ in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to identify which edges are part of the minimal subtree.
    # An edge is part of the subtree if and only if both components formed by 
    # removing the edge contain at least one critical processor.
    
    # Root the tree at an arbitrary critical processor (e.g., the first one)
    root = next(iter(critical))
    
    # Perform DFS to count critical nodes in each subtree
    subtree_critical_count = [0] * (N + 1)
    parent_edge_weight = [0] * (N + 1)
    parent = [0] * (N + 1)
    
    order = []
    stack = [root]
    visited = [False] * (N + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in critical:
            subtree_critical_count[node] = 1
            
    total_cost = 0
    # Process in reverse topological order to propagate counts up
    for i in range(N - 1, 0, -1):
        u = order[i]
        p = parent[u]
        subtree_critical_count[p] += subtree_critical_count[u]
        
        # If the subtree at u contains some but not all critical nodes,
        # the edge (u, p) must be included.
        if 0 < subtree_critical_count[u] < K:
            total_cost += parent_edge_weight[u]
            
    print(total_cost)

solve()
```
