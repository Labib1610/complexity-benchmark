```python
import sys

# The problem states there are n modules and m = n - 1 conduits.
# This means the graph is a tree.
# We need to connect all t critical processors.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of nodes in a tree, we must activate all edges that lie
# on the paths between any pair of critical nodes.
# An edge is part of the required set if and only if it splits the tree
# into two components, each containing at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_critical[u] stores the number of critical nodes in the subtree of u
    subtree_critical = [0] * (n + 1)
    
    # DFS to count critical nodes in subtrees
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                stack.append(v)
                
    for u in order:
        if u in critical_nodes:
            subtree_critical[u] += 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        subtree_critical[p] += subtree_critical[u]
        
        # Find the weight of the edge (u, p)
        # Since it's a tree, we can find it by looking at the edge list
        # or storing it during DFS. Let's store it in a map.
    
    edge_weights = {}
    for u, v, w in edges:
        edge_weights[tuple(sorted((u, v)))] = w
        
    for u in range(2, n + 1):
        count = subtree_critical[u]
        # If the edge splits the tree such that both sides have critical nodes
        if count > 0 and count < t:
            p = parent[u]
            total_cost += edge_weights[tuple(sorted((u, p)))]
            
    print(total_cost)

solve()
```
