```python
import sys
import heapq

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    critical = [False] * (n + 1)
    for _ in range(t):
        critical[int(next(it))] = True
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Prim's algorithm starting from any critical node
    # Find first critical node
    start = 1
    while start <= n and not critical[start]:
        start += 1
    
    if t <= 1:
        print(0)
        return
    
    in_mst = [False] * (n + 1)
    min_edge = [float('inf')] * (n + 1)
    min_edge[start] = 0
    
    # Priority queue: (weight, node)
    pq = [(0, start)]
    
    total_cost = 0
    nodes_in_mst = 0
    
    while pq and nodes_in_mst < n:
        w, u = heapq.heappop(pq)
        if in_mst[u]:
            continue
        in_mst[u] = True
        total_cost += w
        nodes_in_mst += 1
        
        for v, weight in adj[u]:
            if not in_mst[v] and weight < min_edge[v]:
                min_edge[v] = weight
                heapq.heappush(pq, (weight, v))
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
