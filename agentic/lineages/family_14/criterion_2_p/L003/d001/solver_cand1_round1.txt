```python
import sys

# The problem states m = n - 1 and the graph is a tree.
# We need to find the sum of weights of all edges that lie on the paths
# between any two critical processors.
# In a tree, an edge is part of the path between two critical processors
# if and only if the edge splits the tree into two components,
# each containing at least one critical processor.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        critical[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Count critical nodes in subtree
    subtree_critical = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if critical[u]:
            subtree_critical[u] += 1
        if parent[u] != -1:
            subtree_critical[parent[u]] += subtree_critical[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the edge splits the tree such that both sides have critical nodes
        count = subtree_critical[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

solve()
```
