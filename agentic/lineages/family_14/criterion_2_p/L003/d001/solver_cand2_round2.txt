```python
import sys

# The problem states that the graph is a tree (n facilities, n-1 links, 
# exactly one path between any two nodes).
# To connect all priority centers, we must activate all edges that lie on 
# the unique path between any two priority centers.
# An edge is part of the path between at least two priority centers if and 
# only if the subtree formed by removing that edge contains at least one 
# priority center on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_centers = set()
    idx = 3
    for _ in range(t):
        priority_centers.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # subtree_priority_count[u] stores how many priority centers are in the 
    # subtree rooted at u (when rooted at 1)
    subtree_priority_count = [0] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in priority_centers:
            subtree_priority_count[u] += 1
        if parent[u] != 0:
            subtree_priority_count[parent[u]] += subtree_priority_count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the edge (i, parent[i]) separates the tree into two parts,
        # and both parts contain at least one priority center, we must include it.
        count = subtree_priority_count[i]
        if count > 0 and count < t:
            total_cost += edge_weight_to_parent[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
