```python
import sys

# The problem states the graph is a tree (n nodes, n-1 edges, connected).
# To connect all priority targets, we need to include all edges that lie on
# the unique paths between any two priority targets.
# This is equivalent to finding the minimal subtree that contains all priority targets.
# An edge (u, v) with weight w is part of this subtree if and only if
# the subtree rooted at u (when the edge is removed) contains at least one
# priority target, AND the subtree rooted at v contains at least one priority target.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # has_priority[u] will store whether the subtree rooted at u contains a priority target
    has_priority = [False] * (n + 1)
    for node in priority_nodes:
        has_priority[node] = True
        
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    # parent_edge_weight stores the weight of the edge connecting node to its parent
    parent = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    # Process nodes in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        if has_priority[u]:
            total_cost += parent_edge_weight[u]
            has_priority[p] = True
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
