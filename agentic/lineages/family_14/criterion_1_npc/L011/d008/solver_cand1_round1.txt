```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_set = set()
    idx = 3
    for _ in range(t):
        target_set.add(int(input_data[idx]))
        idx += 1
        
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem describes a graph where nodes are indices 1..n.
    # A vector b_u - b_v is an edge between u and v.
    # A set of vectors C forms a forest in this graph.
    # |C| = |S| - 1 means the selected edges form a tree spanning the set S.
    # Non-target participation: every node in S \ T must have degree >= 2.
    # This means every node in S \ T must not be a leaf in the tree.
    # This is equivalent to saying that all leaves of the tree must be in T.
    
    # We want to find a tree that spans some set S such that T is a subset of S,
    # and all leaves of the tree are in T.
    # This is the Steiner Tree problem where we want to connect all nodes in T,
    # but we can include extra nodes (S \ T) as long as they are not leaves.
    # Actually, the condition "all leaves are in T" is exactly the definition
    # of a Steiner Tree for the set T in the graph.
    
    # Since n is small (100), we can use the Dreyfus-Wagner algorithm or 
    # simply realize that for a fixed set of nodes S, the minimum cost to 
    # connect them is the Minimum Spanning Tree of the subgraph induced by S.
    # However, we need to ensure nodes in S \ T are not leaves.
    # If we pick a set S such that T <= S, the MST of S will have leaves.
    # If a leaf is not in T, we can remove it and its incident edge, 
    # reducing the cost and the set S. We repeat this until all leaves are in T.
    # This means the optimal S is exactly the set of nodes in the Steiner Tree of T.
    
    # For small T (up to 10), we can use the Steiner Tree algorithm.
    # dp[mask][i] = min cost to connect the subset of T represented by mask, 
    # with i being a node in the tree.
    
    inf = float('inf')
    dist = [[inf] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
    for u, v, w in edges:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
        
    # Floyd-Warshall
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    if t <= 1:
        print(0)
        return

    targets = list(target_set)
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    for mask in range(1, 1 << t):
        for i in range(1, n + 1):
            submask = (mask - 1) & mask
            while submask > 0:
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
                submask = (submask - 1) & mask
        
        for i in range(1, n + 1):
            if dp[mask][i] == inf: continue
            for j in range(1, n + 1):
                dp[mask][j] = min(dp[mask][j], dp[mask][i] + dist[i][j])
                
    print(min(dp[(1 << t) - 1]))

solve()
```
