```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    essential = set()
    idx = 3
    for _ in range(t):
        essential.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t == 1:
        print(0)
        return

    # The conditions describe a Steiner Tree problem where non-essential nodes
    # (Steiner points) must have degree >= 2.
    # However, the condition "rank = |S|-1" and "|C| = |S|-1" implies the 
    # selected couplings form a tree spanning S.
    # The condition "non-essential nodes have degree >= 2" means that in the 
    # resulting tree, no non-essential node can be a leaf.
    # This is equivalent to finding a Steiner Tree where all leaves are essential nodes.
    
    # This is the Steiner Tree problem. Since t is small (up to 10), we can use
    # the Dreyfus-Wagner algorithm.
    
    # dp[mask][i] = min cost to connect the subset of essential nodes represented 
    # by mask, with node i being part of the tree.
    
    inf = float('inf')
    ess_list = list(essential)
    num_ess = len(ess_list)
    
    dp = [[inf] * (n + 1) for _ in range(1 << num_ess)]
    
    for i in range(num_ess):
        dp[1 << i][ess_list[i]] = 0
        
    for mask in range(1, 1 << num_ess):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relax edges using Dijkstra
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << num_ess) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
