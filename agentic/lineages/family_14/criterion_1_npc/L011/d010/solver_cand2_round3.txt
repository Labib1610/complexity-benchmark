import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_set = set()
    idx = 3
    for _ in range(t):
        target_set.add(int(input_data[idx]))
        idx += 1
        
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The conditions imply that S must be a connected component in the graph of chosen edges.
    # Minimal Assertions (|C| = |S| - 1) means C must be a spanning tree of S.
    # Indispensability: For any k in S \ T, removing k must leave at least two components,
    # each containing at least one element of T.
    # This is equivalent to saying that k is a cut-vertex in the tree C,
    # and every component formed by removing k contains at least one element of T.
    # This is exactly the definition of a Steiner Tree where the Steiner nodes are 
    # the nodes in S \ T, and the condition ensures that the tree is "minimal" 
    # in the sense that no leaf is a non-target node.
    # Actually, the condition simplifies to: S is a set of nodes such that T is a subset of S,
    # and the Steiner Tree of T in the graph G is the unique valid C.
    # Since n is small (100), we can use the property that the optimal C is a Steiner Tree
    # where all leaves are in T.
    
    # If t=1, cost is 0.
    if t == 1:
        print(0)
        return

    # For small t, we can use the Dreyfus-Wagner algorithm or simply iterate over all subsets
    # of nodes that could be in S. But wait, the condition "every leaf is in T" is key.
    # A tree where every leaf is in T is a Steiner Tree.
    # The problem asks for the minimum cost to connect T.
    
    # Let's use the standard Steiner Tree DP:
    # dp[mask][i] = min cost to connect subset of T represented by mask, with i as root.
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    target_list = list(target_set)
    for i in range(t):
        dp[1 << i][target_list[i]] = 0
        
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    for mask in range(1, 1 << t):
        # Combine submasks
        for submask in range((mask - 1) & mask, 0, -1):
            submask &= mask
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
        
        # Relax with edges (Dijkstra)
        pq = []
        import heapq
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
