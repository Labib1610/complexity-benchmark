```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_vars = [int(x) - 1 for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "projected assignment space size is 2" means that
    # all variables in T must be forced to be equal to each other.
    # If T is empty or has 1 element, the size is always 2 (unless n=0, but n>=1).
    # If T has multiple elements, they must all belong to the same connected component
    # in the graph formed by the selected edges.
    
    if t <= 1:
        print(0)
        return
    
    # We need to connect all nodes in target_vars into a single component.
    # This is a Steiner Tree problem variant, but since we only care about
    # connectivity of the set T, we can use the property that we want to
    # connect all nodes in T using a subset of edges.
    # This is equivalent to finding a Minimum Spanning Tree in a graph
    # where we can contract nodes.
    # Actually, this is exactly the Steiner Tree problem on a graph where
    # we want to connect all nodes in T. Since t is small (<= 10),
    # we can use the Dreyfus-Wagner algorithm.
    
    inf = float('inf')
    dist = [[inf] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        dist[u][v] = dist[v][u] = min(dist[u][v], w)
        
    # Floyd-Warshall to get all-pairs shortest paths
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    # dp[mask][i] is the min cost to connect the subset of T represented by mask,
    # with node i being part of the connected component.
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][target_vars[i]] = 0
        
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relax with edges
        for i in range(n):
            for j in range(n):
                if dp[mask][j] + dist[j][i] < dp[mask][i]:
                    dp[mask][i] = dp[mask][j] + dist[j][i]
                    
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
