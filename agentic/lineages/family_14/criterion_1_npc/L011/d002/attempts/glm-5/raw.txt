import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n, m, t = map(int, input_data[:3])
    idx = 3
    
    critical = set(map(int, input_data[idx:idx+t]))
    idx += t
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input_data[idx:idx+3])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    if t <= 1:
        print(0)
        return

    # Dijkstra from all critical nodes
    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    
    for s in critical:
        d = dist[s]
        d[s] = 0
        pq = [(0, s)]
        while pq:
            du, u = pq.pop()
            if du > d[u]:
                continue
            for v, w in adj[u]:
                if d[v] > du + w:
                    d[v] = du + w
                    pq.append((d[v], v))

    # Steiner Tree DP
    # dp[mask][v]: min cost to connect critical nodes in `mask` and end at `v`
    dp = [[INF] * (n + 1) for _ in range(1 << t)]
    crit_list = list(critical)
    
    for i in range(t):
        s = crit_list[i]
        dp[1 << i][s] = 0
        
    for mask in range(1, 1 << t):
        # Transition 1: merge two disjoint subsets
        sub = (mask - 1) & mask
        while sub > 0:
            comp = mask ^ sub
            if comp > sub:
                sub = (sub - 1) & mask
                continue
            for v in range(1, n + 1):
                if dp[mask][v] > dp[sub][v] + dp[comp][v]:
                    dp[mask][v] = dp[sub][v] + dp[comp][v]
            sub = (sub - 1) & mask
            
        # Transition 2: relax edges (shortest path extension)
        pq = []
        for v in range(1, n + 1):
            if dp[mask][v] < INF:
                pq.append((dp[mask][v], v))
        pq.sort()
        
        while pq:
            du, u = pq.pop()
            if du > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > du + w:
                    dp[mask][v] = du + w
                    pq.append((dp[mask][v], v))
                    
    full_mask = (1 << t) - 1
    ans = min(dp[full_mask][v] for v in range(1, n + 1))
    print(ans)

if __name__ == '__main__':
    solve()
