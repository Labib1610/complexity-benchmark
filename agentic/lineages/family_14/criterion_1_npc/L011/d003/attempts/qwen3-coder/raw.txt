import sys
import heapq
from collections import defaultdict

def find_mst_cost(n, edges, essential_nodes):
    if len(essential_nodes) <= 1:
        return 0
    
    # Create a graph with only essential nodes using Steiner tree approach
    # We'll use a modified Prim's algorithm to find the minimum spanning tree
    # that connects all essential nodes
    
    # First, we need to compute shortest paths between all pairs of essential nodes
    # Using Dijkstra from each essential node
    
    # Build adjacency list
    graph = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Function to run Dijkstra from a source node
    def dijkstra(src):
        dist = [float('inf')] * (n + 1)
        dist[src] = 0
        pq = [(0, src)]
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue
                
            for v, w in graph[u]:
                if dist[u] + w < dist[v]:
                    dist[v] = dist[u] + w
                    heapq.heappush(pq, (dist[v], v))
        
        return dist
    
    # Compute shortest distances from each essential node to all other nodes
    distances = {}
    for node in essential_nodes:
        distances[node] = dijkstra(node)
    
    # Now create a complete graph between essential nodes with edge weights
    # equal to shortest path distances
    essential_graph = []
    ess_nodes = list(essential_nodes)
    for i in range(len(ess_nodes)):
        for j in range(i + 1, len(ess_nodes)):
            u, v = ess_nodes[i], ess_nodes[j]
            weight = distances[u][v]
            essential_graph.append((weight, u, v))
    
    # Now find MST of this essential graph using Kruskal's algorithm
    def find(parent, x):
        if parent[x] != x:
            parent[x] = find(parent, parent[x])
        return parent[x]
    
    def union(parent, rank, x, y):
        px, py = find(parent, x), find(parent, y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    # Sort edges by weight
    essential_graph.sort()
    
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    total_cost = 0
    edges_used = 0
    
    for weight, u, v in essential_graph:
        if union(parent, rank, u, v):
            total_cost += weight
            edges_used += 1
            if edges_used == len(essential_nodes) - 1:
                break
    
    return total_cost

# Read input
line = sys.stdin.readline().split()
n, m, t = int(line[0]), int(line[1]), int(line[2])

essential_nodes = set(map(int, sys.stdin.readline().split()))

edges = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    edges.append((u, v, w))

# Solve and print result
result = find_mst_cost(n, edges, essential_nodes)
print(result)
