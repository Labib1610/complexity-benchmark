```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = [int(x) for x in input_data[3:3+t]]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem is equivalent to finding a Steiner Tree in a graph where
    # nodes are dimensions 1..n and edges are the generator vectors.
    # We need to connect all target nodes.
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm.
    
    dist = [[float('inf')] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
            
    # Floyd-Warshall to get all-pairs shortest paths
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    # dp[mask][i] is the min cost to connect the subset of targets represented by mask,
    # with i being a node in the Steiner tree.
    dp = [[float('inf')] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dp[mask][i] using submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Update dp[mask][i] using shortest paths
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dp[mask][j] + dist[j][i] < dp[mask][i]:
                    dp[mask][i] = dp[mask][j] + dist[j][i]
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
