import sys
from itertools import combinations

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    essential = [int(next(it)) for _ in range(t)]
    essential_set = set(essential)
    
    # Build adjacency matrix for edge weights
    INF = float('inf')
    weight = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        weight[i][i] = 0
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        if w < weight[u][v]:
            weight[u][v] = w
            weight[v][u] = w
    
    # Special case: t == 1, no constraints needed
    if t == 1:
        print(0)
        return
    
    # For each non-essential node, compute min cost to connect to essential set
    # This is the minimum edge weight from that node to any essential node
    non_essential = [i for i in range(1, n + 1) if i not in essential_set]
    
    # Precompute: for each non-essential node, min edge to essential
    min_to_essential = {}
    for node in non_essential:
        min_cost = INF
        for e in essential:
            if weight[node][e] < min_cost:
                min_cost = weight[node][e]
        min_to_essential[node] = min_cost
    
    # Precompute: for each pair of non-essential nodes, min edge between them
    min_between_non_essential = {}
    for i, u in enumerate(non_essential):
        for v in non_essential[i+1:]:
            min_between_non_essential[(u, v)] = weight[u][v]
    
    # Try all subsets of non-essential nodes
    num_non_essential = len(non_essential)
    best_cost = INF
    
    # Precompute MST for essential nodes only
    essential_list = list(essential)
    essential_mst_cost = 0
    if t > 1:
        # Prim's algorithm for essential nodes
        in_mst = set()
        min_edge = {e: INF for e in essential}
        start = essential_list[0]
        min_edge[start] = 0
        
        for _ in range(t):
            # Find minimum edge node not in MST
            u = None
            min_val = INF
            for e in essential:
                if e not in in_mst and min_edge[e] < min_val:
                    min_val = min_edge[e]
                    u = e
            if u is None:
                break
            in_mst.add(u)
            essential_mst_cost += min_val
            for v in essential:
                if v not in in_mst and weight[u][v] < min_edge[v]:
                    min_edge[v] = weight[u][v]
    
    best_cost = essential_mst_cost
    
    # Try adding subsets of non-essential nodes
    for k in range(1, num_non_essential + 1):
        for subset in combinations(non_essential, k):
            subset_set = set(subset)
            all_nodes = essential_list + list(subset)
            num_nodes = t + k
            
            # Compute MST cost for all_nodes
            mst_cost = 0
            in_mst = set()
            min_edge = {node: INF for node in all_nodes}
            start = all_nodes[0]
            min_edge[start] = 0
            
            for _ in range(num_nodes):
                u = None
                min_val = INF
                for node in all_nodes:
                    if node not in in_mst and min_edge[node] < min_val:
                        min_val = min_edge[node]
                        u = node
                if u is None:
                    break
                in_mst.add(u)
                mst_cost += min_val
                for v in all_nodes:
                    if v not in in_mst and weight[u][v] < min_edge[v]:
                        min_edge[v] = weight[u][v]
            
            if mst_cost < best_cost:
                best_cost = mst_cost
    
    print(best_cost)

if __name__ == "__main__":
    solve()
