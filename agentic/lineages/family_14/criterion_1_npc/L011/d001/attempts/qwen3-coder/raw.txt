```python
import sys
import heapq
from collections import defaultdict

def find_mst(n, edges, primary_set):
    # Use Prim's algorithm to find MST
    if len(primary_set) <= 1:
        return 0
    
    # Build adjacency list
    adj = defaultdict(list)
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Find minimum spanning tree that connects all primary nodes
    # We'll use a modified Prim's algorithm
    
    # First, compute shortest paths from each primary node to all other nodes
    dist = {}
    for p in primary_set:
        dist[p] = dijkstra(n, adj, p)
    
    # Create a complete graph between primary nodes with weights as shortest paths
    primary_edges = []
    primary_list = list(primary_set)
    for i in range(len(primary_list)):
        for j in range(i+1, len(primary_list)):
            u = primary_list[i]
            v = primary_list[j]
            w = dist[u][v]  # This is the shortest path distance
            primary_edges.append((w, u, v))
    
    # Now find MST of this complete graph using Kruskal's algorithm
    parent = list(range(n + 1))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    primary_edges.sort()
    cost = 0
    edges_used = 0
    
    for w, u, v in primary_edges:
        if union(u, v):
            cost += w
            edges_used += 1
            if edges_used == len(primary_set) - 1:
                break
    
    return cost

def dijkstra(n, adj, start):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
            
        for v, w in adj[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    
    return dist

# Read input
n, m, t = map(int, input().split())
primary = list(map(int, input().split()))
primary_set = set(primary)

edges = []
for _ in range(m):
    u, v, w = map(int, input().split())
    edges.append((u, v, w))

# Solve
result = find_mst(n, edges, primary_set)
print(result)
```
