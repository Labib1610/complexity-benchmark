```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(input_data[i + 3]) for i in range(t)]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The condition pi(e_a) = pi(e_b) for all targets a, b is equivalent to
    # saying that e_a - e_b is in W for all targets a, b.
    # This is equivalent to saying that all targets belong to the same connected
    # component in a graph where edges are the vectors e_u - e_v.
    # We want to find a minimum cost set of edges such that all targets are in
    # the same connected component.
    # This is the Steiner Tree problem on a graph where the terminals are the targets.
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm.
    
    inf = float('inf')
    dist = [[inf] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
            
    # Floyd-Warshall to get all-pairs shortest paths
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    # dp[mask][i] is the min cost to connect the subset of targets represented by mask,
    # with i being a node in the resulting tree.
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        for node in range(1, n + 1):
            dp[1 << i][node] = dist[targets[i]][node]
            
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for node in range(1, n + 1):
                dp[mask][node] = min(dp[mask][node], dp[submask][node] + dp[mask ^ submask][node])
            submask = (submask - 1) & mask
            
        # Relax with edges
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dp[mask][j] + dist[j][i] < dp[mask][i]:
                    dp[mask][i] = dp[mask][j] + dist[j][i]
                    
    print(min(dp[(1 << t) - 1]))

solve()
```
