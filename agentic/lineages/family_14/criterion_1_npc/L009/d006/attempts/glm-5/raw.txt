import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    targets = []
    for _ in range(t):
        targets.append(int(next(it)))
    
    # Build adjacency list for Dijkstra
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Compute all-pairs shortest paths using Dijkstra from each target
    INF = float('inf')
    dist = {}
    for s in targets:
        d = [INF] * (n + 1)
        d[s] = 0
        pq = [(0, s)]
        while pq:
            du, u = heapq.heappop(pq)
            if du > d[u]:
                continue
            for v, w in adj[u]:
                if d[u] + w < d[v]:
                    d[v] = d[u] + w
                    heapq.heappush(pq, (d[v], v))
        dist[s] = d
    
    # Build complete graph on targets with edge weights = shortest path distance
    # Then find MST of this complete graph
    if t <= 1:
        print(0)
        return
    
    # Prim's algorithm for MST
    in_mst = set()
    min_edge = [INF] * t
    min_edge[0] = 0
    total_cost = 0
    
    for _ in range(t):
        # Find minimum edge vertex not in MST
        u = -1
        for i in range(t):
            if i not in in_mst and (u == -1 or min_edge[i] < min_edge[u]):
                u = i
        in_mst.add(u)
        total_cost += min_edge[u]
        
        # Update minimum edges
        for v in range(t):
            if v not in in_mst:
                w = dist[targets[u]][targets[v]]
                if w < min_edge[v]:
                    min_edge[v] = w
    
    print(total_cost)

if __name__ == "__main__":
    solve()
