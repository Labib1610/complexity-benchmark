```python
import sys
import math

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]
    INF = 10**18
    # distance matrix
    dist = [[INF]*n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for _ in range(m):
        u = int(next(it))-1
        v = int(next(it))-1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            ik = di[k]
            if ik == INF: continue
            for j in range(n):
                nd = ik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    full = (1<<t) - 1
    dp = [[INF]*n for _ in range(1<<t)]
    # init single terminals
    for i, node in enumerate(terminals):
        mask = 1<<i
        for v in range(n):
            dp[mask][v] = dist[node][v]
    # DP over subsets
    for mask in range(1, full+1):
        # skip singletons
        sub = (mask-1) & mask
        while sub:
            other = mask ^ sub
            # combine at same root v
            for v in range(n):
                val = dp[sub][v] + dp[other][v]
                if val < dp[mask][v]:
                    dp[mask][v] = val
            sub = (sub-1) & mask
        # after combine, improve via moving root using distances
        # essentially run a relaxation: dp[mask][u] = min_v dp[mask][v] + dist[v][u]
        # we can do a simple Floyd-like relaxation
        for v in range(n):
            dv = dp[mask][v]
            if dv == INF: continue
            dvrow = dist[v]
            for u in range(n):
                nd = dv + dvrow[u]
                if nd < dp[mask][u]:
                    dp[mask][u] = nd
    ans = min(dp[full])
    print(ans)

if __name__ == "__main__":
    solve()
```
