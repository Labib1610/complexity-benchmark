import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def minimum_cost_to_connect_priority_coordinates(n, m, t, priority_coords, edges):
    if t <= 1:
        return 0
    
    # Create a mapping for priority coordinates
    coord_index = {coord: i for i, coord in enumerate(priority_coords)}
    
    # We need to connect all pairs of priority coordinates
    # This is equivalent to finding a minimum spanning tree on the complete graph
    # where edge weights are the shortest path distances between priority coords
    
    # First, compute shortest paths between all pairs of priority coordinates
    # Using Dijkstra's algorithm for each priority node
    dist = defaultdict(lambda: defaultdict(lambda: float('inf')))
    
    # Build adjacency list
    adj = defaultdict(list)
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Run Dijkstra from each priority coordinate
    for start in priority_coords:
        pq = [(0, start)]
        dist[start][start] = 0
        visited = set()
        
        while pq:
            d, u = heapq.heappop(pq)
            if u in visited:
                continue
            visited.add(u)
            dist[start][u] = d
            
            for v, w in adj[u]:
                if v not in visited and d + w < dist[start][v]:
                    dist[start][v] = d + w
                    heapq.heappush(pq, (d + w, v))
    
    # Now create edges for the complete graph on priority coordinates
    priority_edges = []
    for i in range(t):
        for j in range(i+1, t):
            u, v = priority_coords[i], priority_coords[j]
            if dist[u][v] < float('inf'):
                priority_edges.append((dist[u][v], i, j))
    
    # Sort edges by weight
    priority_edges.sort()
    
    # Kruskal's algorithm to find MST
    parent = list(range(t))
    rank = [0] * t
    total_cost = 0
    edges_used = 0
    
    for weight, u, v in priority_edges:
        if union(parent, rank, u, v):
            total_cost += weight
            edges_used += 1
            if edges_used == t - 1:
                break
    
    return total_cost

# Read input
line = sys.stdin.readline().split()
n, m, t = int(line[0]), int(line[1]), int(line[2])
priority_coords = list(map(int, sys.stdin.readline().split()))
edges = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    edges.append((u, v, w))

# Solve and print result
result = minimum_cost_to_connect_priority_coordinates(n, m, t, priority_coords, edges)
print(result)
