```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    essential = []
    idx = 3
    for _ in range(t):
        essential.append(int(input_data[idx]) - 1)
        idx += 1
        
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Dijkstra to find shortest paths from each essential facility to all nodes
    dist = [[float('inf')] * n for _ in range(t)]
    for i in range(t):
        d = dist[i]
        d[essential[i]] = 0
        pq = [(0, essential[i])]
        while pq:
            curr_d, u = heapq.heappop(pq)
            if curr_d > d[u]:
                continue
            for v, w in adj[u]:
                if d[v] > d[u] + w:
                    d[v] = d[u] + w
                    heapq.heappush(pq, (d[v], v))
                    
    # Steiner Tree DP
    # dp[mask][i] is min cost to connect subset of essential facilities represented by mask,
    # with node i being part of the tree.
    dp = [[float('inf')] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][essential[i]] = 0
        
    for mask in range(1, 1 << t):
        # Combine submasks
        for i in range(n):
            submask = (mask - 1) & mask
            while submask > 0:
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
                submask = (submask - 1) & mask
        
        # Relax edges using Dijkstra
        pq = []
        for i in range(n):
            if dp[mask][i] != float('inf'):
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > dp[mask][u] + w:
                    dp[mask][v] = dp[mask][u] + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    print(min(dp[(1 << t) - 1]))

solve()
```
