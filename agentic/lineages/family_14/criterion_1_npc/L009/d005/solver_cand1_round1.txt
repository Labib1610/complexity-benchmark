```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) - 1 for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "every entry M^*_{i,j} where both i and j are target indices evaluates to 1"
    # means all target indices must belong to the same connected component.
    # This is a Steiner Tree problem variant, but since we only need to connect
    # a set of nodes in an undirected graph, it is exactly the Steiner Tree problem.
    # Given t is small (up to 10), we can use the Dreyfus-Wagner algorithm.
    
    # dist[i][j] is the shortest path between i and j
    inf = float('inf')
    dist = [[inf] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    adj = [[] for _ in range(n)]
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
            
    # Floyd-Warshall for all-pairs shortest paths
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    # dp[mask][i] is the min cost to connect the subset of targets represented by mask,
    # with i being a node in the Steiner tree.
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        for v in range(n):
            dp[1 << i][v] = dist[targets[i]][v]
            
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for v in range(n):
                dp[mask][v] = min(dp[mask][v], dp[submask][v] + dp[mask ^ submask][v])
            submask = (submask - 1) & mask
            
        # Relax with edges
        pq = []
        for v in range(n):
            if dp[mask][v] != inf:
                heapq.heappush(pq, (dp[mask][v], v))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v in range(n):
                if dp[mask][v] > d + dist[u][v]:
                    dp[mask][v] = d + dist[u][v]
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    print(min(dp[(1 << t) - 1]))

solve()
```
