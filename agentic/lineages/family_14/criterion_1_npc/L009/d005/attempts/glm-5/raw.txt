import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    targets = [int(next(it)) for _ in range(t)]
    
    # Build adjacency list for all n nodes
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # If t <= 1, no vectors needed
    if t <= 1:
        print(0)
        return
    
    # Compute shortest paths between all pairs of targets
    import heapq
    
    INF = float('inf')
    target_set = set(targets)
    total_cost = 0
    
    # For each target, compute shortest paths to all other nodes
    # Then sum up the cost of MST on targets using these distances
    
    # Compute pairwise distances between targets
    target_dist = [[INF] * (n + 1) for _ in range(t)]
    
    for i in range(t):
        start = targets[i]
        dist = [INF] * (n + 1)
        dist[start] = 0
        pq = [(0, start)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue
            for v, w in adj[u]:
                nd = d + w
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(pq, (nd, v))
        target_dist[i] = dist
    
    # Build MST on targets using Prim's algorithm
    # Edge weight between targets i and j is target_dist[i][targets[j]]
    in_mst = [False] * t
    min_edge = [INF] * t
    min_edge[0] = 0
    mst_cost = 0
    
    for _ in range(t):
        # Find minimum edge vertex not in MST
        u = -1
        for v in range(t):
            if not in_mst[v] and (u == -1 or min_edge[v] < min_edge[u]):
                u = v
        
        in_mst[u] = True
        mst_cost += min_edge[u]
        
        # Update minimum edges
        for v in range(t):
            if not in_mst[v]:
                w = target_dist[u][targets[v]]
                if w < min_edge[v]:
                    min_edge[v] = w
    
    print(mst_cost)

solve()
