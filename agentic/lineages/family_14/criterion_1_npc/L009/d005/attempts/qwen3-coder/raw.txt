import sys
from itertools import combinations

def find_min_cost(n, m, t, targets, edges):
    # If there's only one target, we need nothing
    if t <= 1:
        return 0
    
    # Create a mapping from target indices to 0-based indices for easier handling
    target_indices = {targets[i]: i for i in range(t)}
    
    # We need to ensure all pairwise differences between targets are in the span
    # This is equivalent to making sure all targets are connected in the graph
    # formed by the selected edges - so we need a spanning tree on the targets
    
    # But since we can use linear combinations, we actually need to make sure
    # the targets are connected in some component, and we want the minimum cost
    # to connect all targets.
    
    # This becomes a Steiner Tree problem, but since t is small (≤ 10),
    # we can use dynamic programming with bitmasks.
    
    # However, let's think differently. We want to ensure that for every pair
    # of targets, their difference is in the span of our chosen vectors.
    # This means all targets should be in the same connected component.
    
    # So we need the minimum cost to connect all targets - this is like
    # finding a minimum spanning tree that connects all targets.
    
    # Since t is small, we can:
    # 1. Find shortest paths between all pairs of targets
    # 2. Then find the minimum spanning tree on the complete graph of targets
    #    where edge weights are the shortest path distances
    
    # First, let's implement Floyd-Warshall to get shortest paths
    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    
    # Initialize distances
    for i in range(1, n + 1):
        dist[i][i] = 0
    
    # Add edges (both directions)
    for u, v, w in edges:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
    
    # Floyd-Warshall
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    # Now we have shortest paths between all nodes
    # We need to connect all targets with minimum cost
    # This is a Steiner Tree problem, but we'll solve it with enumeration
    # since t is small
    
    # For t targets, we can try all possible subsets and compute MST
    # But let's use a simpler approach since t ≤ 10
    
    # Actually, let's think of it this way:
    # We want the dimension of the span of {x_{a_i} - x_{a_j}} for all target pairs
    # to be (t-1). This happens when the targets are connected in our graph.
    
    # The minimum cost is the minimum cost to make all targets connected.
    # This is exactly the Steiner Tree problem.
    
    # Let's solve it properly with DP:
    # dp[mask] = minimum cost to connect targets in the mask
    
    # But let's simplify: 
    # The answer is the minimum cost of a subtree that spans all targets.
    # We can use the fact that we only care about connecting targets.
    
    # Use Prim's algorithm variant or solve the metric closure MST:
    # Compute all pairwise distances between targets, then find MST of this
    
    # Get target-target distances
    target_dist = [[0] * t for _ in range(t)]
    for i in range(t):
        for j in range(t):
            target_dist[i][j] = dist[targets[i]][targets[j]]
    
    # Now find MST of targets using Prim's algorithm
    # vertices: 0 to t-1
    if t == 1:
        return 0
        
    # Prim's algorithm
    in_tree = [False] * t
    min_edge = [float('inf')] * t
    min_edge[0] = 0
    total_cost = 0
    
    for _ in range(t):
        # Find vertex with minimum edge weight not yet in tree
        u = -1
        for v in range(t):
            if not in_tree[v] and (u == -1 or min_edge[v] < min_edge[u]):
                u = v
        
        # Add vertex u to tree
        in_tree[u] = True
        total_cost += min_edge[u]
        
        # Update min_edge values for vertices adjacent to u
        for v in range(t):
            if not in_tree[v] and target_dist[u][v] < min_edge[v]:
                min_edge[v] = target_dist[u][v]
    
    return total_cost

# Read input
line = sys.stdin.readline().split()
n, m, t = int(line[0]), int(line[1]), int(line[2])

targets = list(map(int, sys.stdin.readline().split()))

edges = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    edges.append((u, v, w))

print(find_min_cost(n, m, t, targets, edges))
