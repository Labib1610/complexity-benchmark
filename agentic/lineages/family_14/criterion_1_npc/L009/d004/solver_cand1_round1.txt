```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    observables = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "impossible to have observables not uniform" means
    # that all observables must be in the same connected component.
    # We want to partition the set of observables into two non-empty sets S1 and S2,
    # and find the minimum cost to separate them (i.e., make sure no path exists
    # between any node in S1 and any node in S2).
    # This is equivalent to the min-cut problem.
    # Since t is small (up to 10), we can iterate over all partitions of 
    # observables into two sets S1 and S2.
    
    min_total_cost = float('inf')
    
    # There are 2^t possible subsets of observables.
    # We only care about partitions where both S1 and S2 are non-empty.
    # We can fix the first observable to be in S1 to avoid double counting.
    
    for i in range(1, 1 << (t - 1)):
        s1 = {observables[0]}
        s2 = set()
        for j in range(1, t):
            if (i >> (j - 1)) & 1:
                s1.add(observables[j])
            else:
                s2.add(observables[j])
        
        # Min-cut between S1 and S2:
        # Create a graph where all nodes in S1 are connected to a source S
        # and all nodes in S2 are connected to a sink T with infinite capacity.
        # The edges are the bonds.
        
        source = 0
        sink = n + 1
        capacity = [[0] * (n + 2) for _ in range(n + 2)]
        
        for u, v, w in edges:
            capacity[u][v] += w
            capacity[v][u] += w
            
        for obs in s1:
            capacity[source][obs] = float('inf')
        for obs in s2:
            capacity[obs][sink] = float('inf')
            
        # Edmonds-Karp
        flow = 0
        while True:
            parent = {node: None for node in range(n + 2)}
            queue = [source]
            parent[source] = source
            while queue:
                curr = queue.pop(0)
                if curr == sink: break
                for neighbor in range(n + 2):
                    if parent[neighbor] is None and capacity[curr][neighbor] > 0:
                        parent[neighbor] = curr
                        queue.append(neighbor)
                else: continue
                break
            else: break
            
            path_flow = float('inf')
            s = sink
            while s != source:
                path_flow = min(path_flow, capacity[parent[s]][s])
                s = parent[s]
            flow += path_flow
            v = sink
            while v != source:
                u = parent[v]
                capacity[u][v] -= path_flow
                capacity[v][u] += path_flow
                v = u
        
        min_total_cost = min(min_total_cost, flow)
        
    print(min_total_cost)

solve()
```
