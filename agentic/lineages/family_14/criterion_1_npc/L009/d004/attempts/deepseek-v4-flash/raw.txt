We can solve this by treating it as a minimum Steiner tree problem, but since **t ≤ 10**, we can use a bitmask DP approach over the critical nodes.

Here’s the Python solution:  

```python
import sys
import heapq

input = sys.stdin.readline
INF = 10**18

n, m, t = map(int, input().split())
critical = list(map(int, input().split()))
critical = [x - 1 for x in critical]  # 0-based

# Build graph
graph = [[] for _ in range(n)]
for _ in range(m):
    u, v, w = map(int, input().split())
    u -= 1
    v -= 1
    graph[u].append((v, w))
    graph[v].append((u, w))

# If only one critical point, cost is 0
if t == 1:
    print(0)
    sys.exit()

# Precompute shortest distances between all pairs of critical nodes
# We'll run Dijkstra from each critical node
dist = [[INF] * n for _ in range(t)]
for i, s in enumerate(critical):
    d = dist[i]
    d[s] = 0
    pq = [(0, s)]
    while pq:
        du, u = heapq.heappop(pq)
        if du != d[u]:
            continue
        for v, w in graph[u]:
            nd = du + w
            if nd < d[v]:
                d[v] = nd
                heapq.heappush(pq, (nd, v))

# DP over subsets of critical nodes
# dp[mask][i] = min cost to connect nodes in mask, ending at critical node i
full_mask = (1 << t) - 1
dp = [[INF] * t for _ in range(1 << t)]
for i in range(t):
    dp[1 << i][i] = 0

# Steiner DP
for mask in range(1 << t):
    # Combine subsets
    sub = (mask - 1) & mask
    while sub:
        other = mask ^ sub
        if sub < other:  # avoid duplicate work
            for i in range(t):
                if dp[sub][i] < INF:
                    for j in range(t):
                        if dp[other][j] < INF:
                            cost = dp[sub][i] + dp[other][j] + dist[i][critical[j]]
                            if cost < dp[mask][i]:
                                dp[mask][i] = cost
        sub = (sub - 1) & mask

    # Relax via shortest paths
    for i in range(t):
        if dp[mask][i] < INF:
            for j in range(t):
                if i != j:
                    nd = dp[mask][i] + dist[i][critical[j]]
                    if nd < dp[mask][j]:
                        dp[mask][j] = nd

ans = min(dp[full_mask])
print(ans)
```
