import sys
import heapq
from collections import defaultdict

def dijkstra(graph, start, nodes):
    dist = {node: float('inf') for node in nodes}
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
            
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
                
    return dist

def solve():
    n, m, t = map(int, input().split())
    required = list(map(int, input().split()))
    
    if t <= 1:
        print(0)
        return
    
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, input().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Compute shortest paths between all required nodes
    # We'll create a complete graph on required nodes with edge weights as shortest paths
    required_dist = {}
    
    for req in required:
        distances = dijkstra(graph, req, required)
        required_dist[req] = distances
    
    # Now we have a complete graph on required nodes, find MST
    # Use Prim's algorithm
    visited = set()
    min_cost = 0
    pq = [(0, required[0])]  # (cost, node)
    
    while pq and len(visited) < len(required):
        cost, node = heapq.heappop(pq)
        
        if node in visited:
            continue
            
        visited.add(node)
        min_cost += cost
        
        for neighbor in required:
            if neighbor not in visited:
                heapq.heappush(pq, (required_dist[node][neighbor], neighbor))
    
    print(min_cost)

solve()
