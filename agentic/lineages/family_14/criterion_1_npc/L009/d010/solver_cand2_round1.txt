```python
import sys

def solve():
    # Read n, m, t
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_loops = [int(x) for x in input_data[3:3+t]]
    
    # If t <= 1, the condition is vacuously satisfied with cost 0
    if t <= 1:
        print(0)
        return
    
    # The condition pi(b_x) = pi(b_y) is equivalent to b_x + b_y in W.
    # This must hold for all pairs of critical loops.
    # If we have critical loops c_1, c_2, ..., c_t, we need
    # b_{c_1} + b_{c_2} in W, b_{c_2} + b_{c_3} in W, ..., b_{c_{t-1}} + b_{c_t} in W.
    # This is equivalent to finding a minimum cost set of edges such that
    # all c_i are in the same connected component in the graph formed by the edges.
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    # Kruskal's algorithm to connect all critical loops
    edges.sort()
    
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    total_cost = 0
    # We need to connect all critical_loops[i] to critical_loops[0]
    # This is equivalent to finding the MST of the graph where we want
    # all critical nodes to be in the same component.
    # We can treat all critical nodes as a single super-node or just
    # connect them sequentially.
    
    # Standard approach: Kruskal's, but only care about connecting the critical set.
    # Actually, the problem is exactly: find a forest such that all critical nodes
    # are in the same connected component.
    
    # To do this, we can use Kruskal's:
    # 1. Sort all edges by weight.
    # 2. Add edges if they connect two components that contain at least one critical node,
    #    or if they connect a critical node to a non-critical node, etc.
    # Actually, simpler: just run Kruskal's normally, but stop when all critical nodes
    # are in the same component.
    
    def all_critical_connected():
        root = find(critical_loops[0])
        for i in range(1, t):
            if find(critical_loops[i]) != root:
                return False
        return True
    
    for w, u, v in edges:
        if all_critical_connected():
            break
        if union(u, v):
            total_cost += w
            
    print(total_cost)

solve()
```
