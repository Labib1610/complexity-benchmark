import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(input_data[i + 3]) for i in range(t)]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The condition "K remains unchanged" means that the connectivity of the graph
    # formed by the chosen rules must be such that the number of connected components
    # is the same whether or not we add an edge between all targets.
    # Adding an edge between all targets merges the components containing the targets.
    # Let C be the set of components. Let C_t be the set of components that contain
    # at least one target. The number of components changes if and only if
    # |C_t| > 1.
    # Thus, we need to ensure that all targets are already in the same component.
    
    # This is equivalent to finding the Minimum Spanning Tree of the targets
    # in the graph where edge weights are the costs.
    # We can use Kruskal's or Prim's. Since n is small (100), we can use
    # Kruskal's on the graph.
    
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # We need to connect all targets. This is a Steiner Tree problem,
    # but since we only have t <= 10 targets, we can use the Dreyfus-Wagner
    # algorithm or simply observe that we want to connect all targets.
    # Actually, the condition is: all targets must be in the same component.
    # This is exactly the Steiner Tree problem for the set of targets.
    
    # DP for Steiner Tree:
    # dp[mask][i] = min cost to connect subset of targets 'mask' such that i is in the component
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    for mask in range(1, 1 << t):
        # Submask combination
        for submask in range(mask):
            if (submask & mask) == submask:
                for i in range(1, n + 1):
                    dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
        
        # Dijkstra-like update
        changed = True
        while changed:
            changed = False
            for u, v, w in edges:
                if dp[mask][u] + w < dp[mask][v]:
                    dp[mask][v] = dp[mask][u] + w
                    changed = True
                if dp[mask][v] + w < dp[mask][u]:
                    dp[mask][u] = dp[mask][v] + w
                    changed = True
                    
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
