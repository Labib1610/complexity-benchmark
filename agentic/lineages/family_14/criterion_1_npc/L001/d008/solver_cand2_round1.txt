```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical = [int(x) - 1 for x in input_data[3:3+t]]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "for every partition of critical nodes into two non-empty sets,
    # there is an edge between the two sets" is equivalent to saying that
    # all critical nodes must be in the same connected component.
    # Proof: If they are in the same connected component, any cut separating
    # some critical nodes from others must cut at least one edge in the path
    # connecting them.
    
    # Kruskal's algorithm to find the MST of the graph where all critical nodes
    # are merged into a single super-node.
    
    parent = list(range(n))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # Merge all critical nodes into one component
    for i in range(t - 1):
        union(critical[i], critical[i+1])
        
    edges.sort(key=lambda x: x[2])
    
    total_cost = 0
    for u, v, w in edges:
        if union(u, v):
            total_cost += w
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
