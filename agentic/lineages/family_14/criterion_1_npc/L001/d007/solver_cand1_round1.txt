import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_formats = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # The problem asks for a Steiner Tree where the set of terminals is {c1, ..., ct}
    # plus one additional node 'r' which acts as the native format.
    # Since n is small (100), we can iterate over all possible native formats 'r'.
    # For a fixed 'r', we need the minimum cost to connect {c1, ..., ct, r}.
    # This is equivalent to the Steiner Tree problem on {c1, ..., ct, r}.
    
    # Dreyfus-Wagner algorithm for Steiner Tree:
    # dp[mask][i] is the min cost to connect the subset of terminals represented by mask,
    # with i being a node in the tree.
    
    terminals = critical_formats
    num_terminals = len(terminals)
    
    # Precompute all-pairs shortest paths
    dist = [[float('inf')] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
        pq = [(0, i)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[i][u]:
                continue
            for v, w in adj[u]:
                if dist[i][v] > d + w:
                    dist[i][v] = d + w
                    heapq.heappush(pq, (dist[i][v], v))
                    
    min_total_cost = float('inf')
    
    # Try each node as the native format
    for r in range(1, n + 1):
        # Terminals are {c1, ..., ct, r}
        # If r is already in critical_formats, it's just {c1, ..., ct}
        current_terminals = list(set(terminals + [r]))
        k = len(current_terminals)
        
        dp = [[float('inf')] * (n + 1) for _ in range(1 << k)]
        
        for i, node in enumerate(current_terminals):
            dp[1 << i][node] = 0
            
        for mask in range(1, 1 << k):
            for i in range(1, n + 1):
                submask = (mask - 1) & mask
                while submask > 0:
                    dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
                    submask = (submask - 1) & mask
            
            # Relax using shortest paths
            pq = []
            for i in range(1, n + 1):
                if dp[mask][i] != float('inf'):
                    heapq.heappush(pq, (dp[mask][i], i))
            
            while pq:
                d, u = heapq.heappop(pq)
                if d > dp[mask][u]:
                    continue
                for v in range(1, n + 1):
                    if dp[mask][v] > d + dist[u][v]:
                        dp[mask][v] = d + dist[u][v]
                        heapq.heappush(pq, (dp[mask][v], v))
        
        min_total_cost = min(min_total_cost, dp[(1 << k) - 1][r])
        
    print(min_total_cost)

solve()
