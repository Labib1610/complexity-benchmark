```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    mandatory = [int(x) - 1 for x in input_data[3:3+t]]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "any zero-sum distribution can be routed" is equivalent to
    # saying that all mandatory sites must belong to the same connected component.
    # This is a Steiner Tree problem.
    # Since t is small (t <= 10), we can use the Dreyfus-Wagner algorithm.
    
    inf = float('inf')
    # dp[mask][i] is the min cost to connect the subset of mandatory sites 
    # represented by 'mask' such that site 'i' is in the resulting tree.
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][mandatory[i]] = 0
        
    # Precompute all-pairs shortest paths for the base case
    dist = [[inf] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
        
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][j] > dist[i][k] + dist[k][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relax using edges (Dijkstra-like)
        for i in range(n):
            for j in range(n):
                dp[mask][i] = min(dp[mask][i], dp[mask][j] + dist[j][i])
                
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
