We can model this as a **Steiner tree** problem, but since the number of critical modules is small (t ≤ 10), we can solve it using **bitmask DP over subsets of terminals** combined with shortest paths.

Here’s the plan:

1. Run Dijkstra from each critical node to get shortest distances to all nodes.
2. Use DP where `dp[mask][v]` = minimum cost to connect the terminals in `mask` with the tree rooted at node `v`.
3. Combine subsets using the standard Steiner tree DP:
   - First, for each node, combine two submasks.
   - Then, run a shortest-path relaxation (like Dijkstra) over the DP states to propagate costs.
4. The answer is `min_v dp[(1<<t)-1][v]`.

```python
import sys
import heapq

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    terminals = list(map(int, input().split()))
    terminals = [x-1 for x in terminals]
    
    graph = [[] for _ in range(n)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # If only one terminal, no links needed
    if t == 1:
        print(0)
        return
    
    # Dijkstra from each terminal
    dist = [[float('inf')] * n for _ in range(t)]
    for i, src in enumerate(terminals):
        d = dist[i]
        d[src] = 0
        pq = [(0, src)]
        while pq:
            du, u = heapq.heappop(pq)
            if du != d[u]:
                continue
            for v, w in graph[u]:
                nd = du + w
                if nd < d[v]:
                    d[v] = nd
                    heapq.heappush(pq, (nd, v))
    
    # DP over subsets
    INF = 10**18
    dp = [[INF] * n for _ in range(1 << t)]
    for i in range(t):
        for v in range(n):
            dp[1 << i][v] = dist[i][v]
    
    # Combine subsets
    for mask in range(1, 1 << t):
        # Split mask into submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for v in range(n):
                    dp[mask][v] = min(dp[mask][v], dp[sub][v] + dp[other][v])
            sub = (sub - 1) & mask
        
        # Dijkstra relaxation for this mask
        pq = [(dp[mask][v], v) for v in range(n) if dp[mask][v] < INF]
        heapq.heapify(pq)
        while pq:
            du, u = heapq.heappop(pq)
            if du != dp[mask][u]:
                continue
            for v, w in graph[u]:
                nd = du + w
                if nd < dp[mask][v]:
                    dp[mask][v] = nd
                    heapq.heappush(pq, (nd, v))
    
    full_mask = (1 << t) - 1
    ans = min(dp[full_mask][v] for v in range(n))
    print(ans)

if __name__ == "__main__":
    solve()
```
