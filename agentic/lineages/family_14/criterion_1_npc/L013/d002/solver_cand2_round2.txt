```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    controllers = [int(input_data[i + 3]) for i in range(t)]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        edges.append((w, u, v))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Steiner Tree problem for small t (t <= 10)
    # DP state: dp[mask][v] is min cost to connect subset of controllers 'mask'
    # such that they are connected to node 'v'.
    # mask is a bitmask of length t.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][controllers[i]] = 0
        
    adj = [[] for _ in range(n + 1)]
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    for mask in range(1, 1 << t):
        # Update dp[mask][v] using submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for v in range(1, n + 1):
                dp[mask][v] = min(dp[mask][v], dp[submask][v] + dp[mask ^ submask][v])
            submask = (submask - 1) & mask
            
        # Dijkstra to update dp[mask][v] for all v
        import heapq
        pq = []
        for v in range(1, n + 1):
            if dp[mask][v] != inf:
                heapq.heappush(pq, (dp[mask][v], v))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][v] for v in range(1, n + 1))
    print(ans)

solve()
```
