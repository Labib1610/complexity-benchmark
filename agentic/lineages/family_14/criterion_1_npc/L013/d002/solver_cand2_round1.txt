```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    controllers = [int(x) for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    if t <= 1:
        print(0)
        return
    
    # Kruskal's algorithm to find the Minimum Spanning Tree
    # Since we only care about connecting the t controllers,
    # we can use Kruskal's on the whole graph.
    # The MST of the whole graph connects all nodes, including controllers.
    # However, we only need to connect the controllers.
    # This is the Steiner Tree problem, but with n <= 100 and t <= 10,
    # we can use the property that the optimal solution is a tree.
    
    # Actually, for n=100, t=10, the Steiner Tree problem is NP-hard.
    # But wait, the constraints are small enough for the Dreyfus-Wagner algorithm
    # or simply realizing that with t=10, we can use dynamic programming.
    
    # DP state: dp[mask][i] is the min cost to connect the subset of controllers
    # represented by 'mask', with node 'i' being part of the connected component.
    
    adj = [[] for _ in range(n + 1)]
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Dijkstra to find shortest paths between all pairs
    dist = [[float('inf')] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
        import heapq
        pq = [(0, i)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[i][u]:
                continue
            for v, w in adj[u]:
                if dist[i][v] > d + w:
                    dist[i][v] = d + w
                    heapq.heappush(pq, (dist[i][v], v))
                    
    dp = [[float('inf')] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][controllers[i]] = 0
        
    for mask in range(1, 1 << t):
        for i in range(1, n + 1):
            submask = (mask - 1) & mask
            while submask > 0:
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
                submask = (submask - 1) & mask
        
        for i in range(1, n + 1):
            if dp[mask][i] == float('inf'): continue
            for j in range(1, n + 1):
                dp[mask][j] = min(dp[mask][j], dp[mask][i] + dist[i][j])
                
    print(min(dp[(1 << t) - 1]))

solve()
```
