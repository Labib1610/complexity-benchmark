```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    P = int(input_data[2])
    
    mandatory = []
    idx = 3
    for _ in range(P):
        mandatory.append(int(input_data[idx]) - 1)
        idx += 1
        
    adj = [[] for _ in range(N)]
    for _ in range(M):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if P <= 1:
        print(0)
        return

    # Steiner Tree problem: N <= 100, P <= 10
    # dp[mask][i] is the min cost to connect the subset of mandatory sites 
    # represented by mask, with i being part of the tree.
    
    inf = float('inf')
    dp = [[inf] * N for _ in range(1 << P)]
    
    for i in range(P):
        dp[1 << i][mandatory[i]] = 0
        
    for mask in range(1, 1 << P):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(N):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Dijkstra to update dp[mask][i]
        pq = []
        for i in range(N):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << P) - 1])
    print(ans)

solve()
```
