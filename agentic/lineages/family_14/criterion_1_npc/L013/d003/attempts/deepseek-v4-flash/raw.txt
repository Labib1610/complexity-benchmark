We need to connect all primary workstations, but we can use any other nodes as well.  
This is essentially a **Steiner tree** problem, but with a small number of terminals (t ≤ 10), so we can solve it using **DP over subsets** and **Dijkstra** for shortest paths.

Here’s the Python solution:

```python
import sys
import heapq

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    terminals = list(map(int, input().split()))
    terminals = [x - 1 for x in terminals]  # 0-based

    # Build graph
    graph = [[] for _ in range(n)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        graph[u].append((v, w))
        graph[v].append((u, w))

    # If only one terminal, cost is 0
    if t == 1:
        print(0)
        return

    # Precompute shortest distances between all pairs of terminals
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for s in range(n):
        d = [INF] * n
        d[s] = 0
        pq = [(0, s)]
        while pq:
            du, u = heapq.heappop(pq)
            if du != d[u]:
                continue
            for v, w in graph[u]:
                nd = du + w
                if nd < d[v]:
                    d[v] = nd
                    heapq.heappush(pq, (nd, v))
        dist[s] = d

    # DP over subsets of terminals
    # dp[mask][i] = min cost to connect terminals in mask, ending at terminal i
    full_mask = 1 << t
    dp = [[INF] * t for _ in range(full_mask)]
    for i in range(t):
        dp[1 << i][i] = 0

    for mask in range(full_mask):
        for i in range(t):
            if dp[mask][i] == INF:
                continue
            # Try to add another terminal
            for j in range(t):
                if not (mask & (1 << j)):
                    nmask = mask | (1 << j)
                    cost = dp[mask][i] + dist[terminals[i]][terminals[j]]
                    if cost < dp[nmask][j]:
                        dp[nmask][j] = cost
            # Try to merge two subsets
            sub = (mask - 1) & mask
            while sub:
                other = mask ^ sub
                if sub < other:  # avoid double work
                    for i in range(t):
                        if dp[sub][i] == INF:
                            continue
                        for j in range(t):
                            if dp[other][j] == INF:
                                continue
                            cost = dp[sub][i] + dp[other][j]
                            if cost < dp[mask][i]:
                                dp[mask][i] = cost
                            if cost < dp[mask][j]:
                                dp[mask][j] = cost
                sub = (sub - 1) & mask

    ans = min(dp[full_mask - 1])
    print(ans)

if __name__ == "__main__":
    solve()
```
