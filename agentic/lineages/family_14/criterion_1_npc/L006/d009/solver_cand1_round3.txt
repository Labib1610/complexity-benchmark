```python
import sys
import collections

# The problem asks for the minimum cost to partition the n variables such that
# all t critical variables are in the same connected component.
# This is equivalent to the "Multiway Cut" problem where we want to group
# all t terminals into a single component.
# This can be solved by picking one terminal as a reference and finding the
# minimum cut separating it from all other terminals, but that's for a different
# variation. Here, we want all t terminals to be in the same component.
# This is equivalent to the "Steiner Tree" problem if we were looking for a tree,
# but here we can use any edges. Actually, this is the "Minimum Cut" problem
# where we want to partition the graph into two sets S and V\S such that
# all terminals are in S or all are in V\S.
# This is equivalent to finding the minimum cut that separates the set of
# terminals into two non-empty sets, and doing this recursively.
# Or more simply: for any partition of the t terminals into two non-empty sets A and B,
# find the min-cut separating A from B. The answer is the minimum over all such partitions.
# Since t <= 10, we can iterate over all partitions of the t terminals into two sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, t = map(int, input_data[:3])
    critical = list(map(int, input_data[3:3+t]))
    edges = []
    idx = 3 + t
    for _ in range(m):
        u, v, w = map(int, input_data[idx:idx+3])
        edges.append((u - 1, v - 1, w))
        idx += 3

    if t <= 1:
        print(0)
        return

    # Min-cut using Edmonds-Karp or Dinic
    def get_min_cut(source_set, sink_set):
        # Build graph
        adj = collections.defaultdict(list)
        for u, v, w in edges:
            adj[u].append([v, w, len(adj[v])])
            adj[v].append([u, w, len(adj[u]) - 1])
        
        # Super source and sink
        S, T = n, n + 1
        for s in source_set:
            adj[S].append([s, float('inf'), len(adj[s])])
            adj[s].append([S, 0, len(adj[S]) - 1])
        for t_node in sink_set:
            adj[t_node].append([T, float('inf'), len(adj[T])])
            adj[T].append([t_node, 0, len(adj[t_node]) - 1])
            
        flow = 0
        while True:
            parent = {S: None}
            edge_from = {}
            queue = collections.deque([S])
            while queue:
                u = queue.popleft()
                if u == T: break
                for i, (v, cap, rev_idx) in enumerate(adj[u]):
                    if cap > 0 and v not in parent:
                        parent[v] = u
                        edge_from[v] = i
                        queue.append(v)
                else: continue
                break
            else: break
            
            path_flow = float('inf')
            curr = T
            while curr != S:
                prev = parent[curr]
                idx = edge_from[curr]
                path_flow = min(path_flow, adj[prev][idx][1])
                curr = prev
            flow += path_flow
            curr = T
            while curr != S:
                prev = parent[curr]
                idx = edge_from[curr]
                rev_idx = adj[prev][idx][2]
                adj[prev][idx][1] -= path_flow
                adj[curr][rev_idx][1] += path_flow
                curr = prev
        return flow

    # We want to partition the t terminals into two non-empty sets A and B
    # and find the min cut between them.
    min_total_cost = float('inf')
    # Iterate through all 2^t - 2 partitions
    for i in range(1, 1 << (t - 1)):
        A = []
        B = []
        for j in range(t):
            if (i >> j) & 1:
                A.append(critical[j])
            else:
                B.append(critical[j])
        min_total_cost = min(min_total_cost, get_min_cut(A, B))
        
    print(min_total_cost)

solve()
```
