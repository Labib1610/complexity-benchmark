```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) - 1 for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition B_{x,y} > 0 for B = A^n where A is the adjacency matrix 
    # (plus identity) means there is a path of length at most n between x and y.
    # Since A has 1s on the diagonal, a path of length k <= n exists if and only if
    # x and y are in the same connected component.
    # We need all target indices to be in the same connected component.
    
    if t <= 1:
        print(0)
        return
    
    # Kruskal's or Prim's to connect all target nodes.
    # This is a Steiner Tree problem, but since we only need to connect 
    # a set of terminals, and the graph is undirected, we can use the 
    # Dreyfus-Wagner algorithm or simply realize that for small t, 
    # we can use dynamic programming on subsets.
    
    # dp[mask][i] = min cost to connect the subset of targets represented by mask,
    # with node i being part of the connected component.
    
    inf = float('inf')
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    adj = [[] for _ in range(n)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    for mask in range(1, 1 << t):
        # 1. Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # 2. Relax edges using Dijkstra
        pq = []
        for i in range(n):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    print(min(dp[(1 << t) - 1]))

solve()
```
