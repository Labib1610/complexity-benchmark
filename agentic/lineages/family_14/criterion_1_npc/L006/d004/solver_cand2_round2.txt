```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_set = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # Steiner Tree problem on a graph with n nodes.
    # Since t is small (up to 10), we use the Dreyfus-Wagner algorithm.
    # dp[mask][i] is the min cost to connect the subset of critical nodes 
    # represented by 'mask' such that node 'i' is in the connected component.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][critical_set[i]] = 0
        
    # Dijkstra for single-source shortest paths
    def dijkstra(mask):
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v1, v2, w in edges:
                if v1 == u:
                    if dp[mask][v2] > d + w:
                        dp[mask][v2] = d + w
                        heapq.heappush(pq, (dp[mask][v2], v2))
                elif v2 == u:
                    if dp[mask][v1] > d + w:
                        dp[mask][v1] = d + w
                        heapq.heappush(pq, (dp[mask][v1], v1))

    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
        
        # Relax edges
        dijkstra(mask)
        
    print(min(dp[(1 << t) - 1]))

solve()
```
