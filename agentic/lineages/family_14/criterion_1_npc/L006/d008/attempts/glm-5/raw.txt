import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    critical = [int(next(it)) - 1 for _ in range(t)]
    
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        edges.append((u, v, w))
    
    if t == 1:
        print(0)
        return
    
    # For each critical vertex, find min cost to connect it to any non-critical vertex
    INF = 10**18
    min_to_noncrit = [INF] * n
    
    for u, v, w in edges:
        if u in critical_set or v in critical_set:
            pass
    
    critical_set = set(critical)
    for u, v, w in edges:
        if u in critical_set and v not in critical_set:
            if w < min_to_noncrit[u]:
                min_to_noncrit[u] = w
        if v in critical_set and u not in critical_set:
            if w < min_to_noncrit[v]:
                min_to_noncrit[v] = w
    
    # Option 1: All critical vertices connect to non-critical vertices
    option1 = sum(min_to_noncrit[c] for c in critical)
    if any(min_to_noncrit[c] == INF for c in critical):
        option1 = INF
    
    # Option 2: Critical vertices form a connected component among themselves
    # Find MST on the subgraph induced by critical vertices
    # Use Kruskal's algorithm
    parent = list(range(n))
    rank = [0] * n
    
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    
    def union(x, y):
        rx, ry = find(x), find(y)
        if rx == ry:
            return False
        if rank[rx] < rank[ry]:
            parent[rx] = ry
        elif rank[rx] > rank[ry]:
            parent[ry] = rx
        else:
            parent[ry] = rx
            rank[rx] += 1
        return True
    
    # Sort edges by weight
    sorted_edges = sorted(edges, key=lambda e: e[2])
    
    mst_cost = 0
    edges_used = 0
    for u, v, w in sorted_edges:
        if u in critical_set and v in critical_set:
            if union(u, v):
                mst_cost += w
                edges_used += 1
                if edges_used == t - 1:
                    break
    
    option2 = mst_cost if edges_used == t - 1 else INF
    
    ans = min(option1, option2)
    print(ans)

solve()
