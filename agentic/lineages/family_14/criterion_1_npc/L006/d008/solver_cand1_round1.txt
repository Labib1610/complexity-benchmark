```python
import sys
import heapq

def solve():
    # The matrix exponential E = exp(M) where M is symmetric.
    # Since M is symmetric, E = exp(M) has the property that E_{i,j} > 0
    # if and only if there is a path between i and j in the graph defined by M.
    # Initially M = I, so M_{i,i} = 1.
    # Adding an update (u, v) adds 1 to M_{u,v} and M_{v,u}.
    # This is equivalent to adding an edge between u and v in a graph.
    # The condition E_{i,j} > 0 for all critical indices i, j means that
    # all critical indices must belong to the same connected component.
    # Since M_{i,i} = 1 initially, the diagonal entries of E are always > 0.
    # Thus, we only need to ensure all critical indices are in the same connected component.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = [int(x) - 1 for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # We need to connect all critical nodes. This is a Steiner Tree problem.
    # Since t is small (t <= 10), we can use the Dreyfus-Wagner algorithm.
    
    # dist[mask][i] is the min cost to connect the subset of critical nodes
    # represented by 'mask' such that node 'i' is in the component.
    inf = float('inf')
    dist = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dist[1 << i][critical[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dist[mask][i] using submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dist[mask][i] = min(dist[mask][i], dist[submask][i] + dist[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Dijkstra to update dist[mask][i] for all i
        pq = []
        for i in range(n):
            if dist[mask][i] != inf:
                heapq.heappush(pq, (dist[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[mask][u]:
                continue
            for v1, v2, w in edges:
                if v1 == u:
                    if dist[mask][v2] > d + w:
                        dist[mask][v2] = d + w
                        heapq.heappush(pq, (dist[mask][v2], v2))
                elif v2 == u:
                    if dist[mask][v1] > d + w:
                        dist[mask][v1] = d + w
                        heapq.heappush(pq, (dist[mask][v1], v1))
                        
    ans = min(dist[(1 << t) - 1])
    print(ans)

solve()
```
