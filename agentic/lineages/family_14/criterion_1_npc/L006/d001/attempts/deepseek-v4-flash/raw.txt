We need the minimum cost to connect all primary currencies, but we may also use non-primary nodes as intermediates.  
This is essentially a **Steiner tree** problem on a graph, with at most 10 terminals (primary currencies).  

We can solve it using **dynamic programming over subsets**:

1. First, compute all-pairs shortest paths.
2. Let `dp[mask][v]` be the minimum cost to connect the set of primary currencies in `mask`, ending at node `v`.
3. Initialize with shortest paths from root (primary currency) to all nodes.
4. Combine subsets using standard Steiner DP.

```python
import sys
import heapq

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    terminals = list(map(lambda x: int(x) - 1, input().split()))
    terminals_set = set(terminals)
    
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append((v, w))
        adj[v].append((u, w))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            dik = dist[i][k]
            if dik == INF:
                continue
            di = dist[i]
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    # Steiner DP
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]
    
    # Initialize: connecting a single terminal i to node v costs dist[terminal_i][v]
    for i, term in enumerate(terminals):
        for v in range(n):
            dp[1 << i][v] = dist[term][v]
    
    for mask in range(1, full_mask):
        # Combine two submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:
                for v in range(n):
                    if dp[sub][v] + dp[other][v] < dp[mask][v]:
                        dp[mask][v] = dp[sub][v] + dp[other][v]
            sub = (sub - 1) & mask
        
        # Relax using Dijkstra on this mask
        pq = [(dp[mask][v], v) for v in range(n) if dp[mask][v] < INF]
        heapq.heapify(pq)
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                nd = d + w
                if nd < dp[mask][v]:
                    dp[mask][v] = nd
                    heapq.heappush(pq, (nd, v))
    
    ans = min(dp[full_mask - 1][v] for v in range(n))
    print(ans)

if __name__ == "__main__":
    solve()
```
