```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # The problem asks for the minimum cost to connect a starting node 's'
    # to all nodes in 'targets'. This is a Steiner Tree problem.
    # Since t is small (t <= 10), we can use the Dreyfus-Wagner algorithm.
    
    # dist[mask][i] is the min cost to connect all nodes in subset 'mask'
    # such that node 'i' is part of the connected component.
    inf = float('inf')
    dist = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dist[1 << i][targets[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dist[mask][i] using edges (Dijkstra)
        pq = []
        for i in range(1, n + 1):
            if dist[mask][i] != inf:
                heapq.heappush(pq, (dist[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[mask][u]:
                continue
            for v, w in adj[u]:
                if dist[mask][v] > d + w:
                    dist[mask][v] = d + w
                    heapq.heappush(pq, (dist[mask][v], v))
        
        # Update dist[mask][i] using submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dist[mask][i] = min(dist[mask][i], dist[submask][i] + dist[mask ^ submask][i])
            submask = (submask - 1) & mask
            
    # The answer is min_{start_node} (dist[all_targets][start_node])
    # If start_node is one of the targets, the cost is just the Steiner tree cost.
    # If start_node is not a target, we can draft in start_node and pay to connect to all targets.
    ans = min(dist[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
