```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) - 1 for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem asks for the minimum cost to connect all target nodes in a graph.
    # This is equivalent to the Steiner Tree problem.
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm.
    
    # dist[i][j] is the shortest path between i and j
    dist = [[float('inf')] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
        
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][j] > dist[i][k] + dist[k][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    # dp[mask][i] is the min cost to connect the subset of targets represented by mask,
    # with i being a node in the Steiner tree.
    dp = [[float('inf')] * n for _ in range(1 << t)]
    
    for i in range(t):
        for v in range(n):
            dp[1 << i][v] = dist[targets[i]][v]
            
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for v in range(n):
                dp[mask][v] = min(dp[mask][v], dp[submask][v] + dp[mask ^ submask][v])
            submask = (submask - 1) & mask
            
        # Relax edges
        for v in range(n):
            for u in range(n):
                dp[mask][v] = min(dp[mask][v], dp[mask][u] + dist[u][v])
                
    print(min(dp[(1 << t) - 1]))

solve()
```
