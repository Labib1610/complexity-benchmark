import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) - 1 for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # Steiner Tree problem on a graph with n <= 100 and t <= 10.
    # Since t is very small, we use the Dreyfus-Wagner algorithm.
    # dp[mask][i] is the min cost to connect the subset of targets represented by mask,
    # with i being a node in the tree.
    
    inf = float('inf')
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    # Precompute all-pairs shortest paths using Floyd-Warshall
    dist = [[inf] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
            
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relax using shortest paths
        pq = []
        for i in range(n):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v in range(n):
                if dp[mask][v] > dp[mask][u] + dist[u][v]:
                    dp[mask][v] = dp[mask][u] + dist[u][v]
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    print(min(dp[(1 << t) - 1]))

solve()
