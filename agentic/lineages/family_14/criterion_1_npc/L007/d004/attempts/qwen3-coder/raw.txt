import sys
from itertools import combinations

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal(n, edges):
    parent = list(range(n))
    rank = [0] * n
    cost = 0
    for u, v, w in sorted(edges, key=lambda x: x[2]):
        if union(parent, rank, u, v):
            cost += w
    return cost

def connected_components(n, edges):
    parent = list(range(n))
    rank = [0] * n
    for u, v, w in edges:
        union(parent, rank, u, v)
    components = {}
    for i in range(n):
        root = find(parent, i)
        if root not in components:
            components[root] = []
        components[root].append(i)
    return list(components.values())

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    targets = list(map(int, sys.stdin.readline().split()))
    targets = [x - 1 for x in targets]  # Convert to 0-indexed
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u-1, v-1, w))  # Convert to 0-indexed
    
    # Get all connected components
    components = connected_components(n, edges)
    
    # Find which component each target belongs to
    target_component = {}
    for i, comp in enumerate(components):
        for target in targets:
            if target in comp:
                target_component[target] = i
    
    # Group targets by component
    targets_by_component = {}
    for target, comp_id in target_component.items():
        if comp_id not in targets_by_component:
            targets_by_component[comp_id] = []
        targets_by_component[comp_id].append(target)
    
    # For each component with multiple targets, we need to connect them
    total_cost = 0
    for comp_id, comp_targets in targets_by_component.items():
        if len(comp_targets) > 1:
            # Extract edges within this component
            comp_edges = [(u, v, w) for u, v, w in edges 
                         if u in components[comp_id] and v in components[comp_id]]
            
            # Create a mapping from original node ids to local ids
            node_map = {node: i for i, node in enumerate(components[comp_id])}
            reverse_map = {i: node for node, i in node_map.items()}
            local_edges = [(node_map[u], node_map[v], w) for u, v, w in comp_edges]
            
            # Use Prim's algorithm to find MST connecting all targets in this component
            local_n = len(components[comp_id])
            local_targets = [node_map[target] for target in comp_targets]
            
            # Initialize
            in_tree = [False] * local_n
            min_edge = [float('inf')] * local_n
            prev = [-1] * local_n
            
            # Start with first target
            start = local_targets[0]
            min_edge[start] = 0
            cost = 0
            tree_nodes = set()
            
            # Prim's algorithm modified to stop when all targets are connected
            while len(tree_nodes) < len(local_targets):
                # Find node with minimum edge weight not yet in tree
                u = -1
                for v in range(local_n):
                    if v not in tree_nodes and (u == -1 or min_edge[v] < min_edge[u]):
                        u = v
                
                # Add node to tree
                tree_nodes.add(u)
                cost += min_edge[u]
                
                # Update min_edge values for adjacent nodes
                for v1, v2, w in local_edges:
                    if v1 == u and v2 not in tree_nodes and w < min_edge[v2]:
                        min_edge[v2] = w
                        prev[v2] = u
                    elif v2 == u and v1 not in tree_nodes and w < min_edge[v1]:
                        min_edge[v1] = w
                        prev[v1] = u
            
            total_cost += cost
    
    print(total_cost)

solve()
