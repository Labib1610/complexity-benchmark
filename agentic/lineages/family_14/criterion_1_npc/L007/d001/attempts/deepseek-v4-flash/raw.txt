We can solve this by treating it as a **Steiner tree** problem, but since the number of priority locations is small (≤ 10), we can use a **bitmask dynamic programming** approach over subsets of terminals.

Here’s the Python solution:

```python
import sys
import heapq

input = sys.stdin.readline

def solve():
    n, m, t = map(int, input().split())
    terminals = list(map(int, input().split()))
    terminals = [x - 1 for x in terminals]  # 0-based

    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append((v, w))
        adj[v].append((u, w))

    # All-pairs shortest paths (Floyd-Warshall)
    for u in range(n):
        for v, w in adj[u]:
            dist[u][v] = min(dist[u][v], w)

    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]

    # DP over subsets of terminals
    # dp[mask][v] = min cost to connect terminals in mask, ending at v
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]

    # Initialize: each terminal alone
    for i in range(t):
        dp[1 << i][terminals[i]] = 0

    # For each subset
    for mask in range(full_mask):
        # Combine two submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid duplicate work
                for v in range(n):
                    dp[mask][v] = min(dp[mask][v], dp[sub][v] + dp[other][v])
            sub = (sub - 1) & mask

        # Relax via shortest paths
        for v in range(n):
            if dp[mask][v] < INF:
                for to in range(n):
                    if dist[v][to] < INF:
                        if dp[mask][v] + dist[v][to] < dp[mask][to]:
                            dp[mask][to] = dp[mask][v] + dist[v][to]

    ans = min(dp[full_mask - 1])
    print(ans)

if __name__ == "__main__":
    solve()
```
