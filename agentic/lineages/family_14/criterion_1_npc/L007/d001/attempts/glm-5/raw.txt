import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    terminals = [int(data[idx+i]) for i in range(t)]
    idx += t
    edges = []
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); w = int(data[idx+2])
        idx += 3
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    INF = float('inf')
    dist = [[INF]*(n+1) for _ in range(n+1)]
    for i in range(1, n+1):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    for k in range(1, n+1):
        dk = dist[k]
        for i in range(1, n+1):
            dik = dist[i][k]
            if dik == INF:
                continue
            di = dist[i]
            for j in range(1, n+1):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    K = t
    dp = [[INF]*K for _ in range(1 << K)]
    for i in range(K):
        dp[1 << i][i] = 0
    
    for mask in range(1 << K):
        for i in range(K):
            if not (mask & (1 << i)):
                continue
            if dp[mask][i] == INF:
                continue
            for j in range(K):
                if mask & (1 << j):
                    continue
                nm = mask | (1 << j)
                cost = dp[mask][i] + dist[terminals[i]][terminals[j]]
                if cost < dp[nm][j]:
                    dp[nm][j] = cost
    
    full = (1 << K) - 1
    ans = min(dp[full][i] for i in range(K))
    print(ans)

solve()
