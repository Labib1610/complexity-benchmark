```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_vars = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition that x_a - x_b is in the ideal generated by {x_u - x_v}
    # is equivalent to saying that in the graph where edges are (u, v),
    # all target variables a, b are in the same connected component.
    # We need to find the Minimum Spanning Tree (or Steiner Tree) 
    # that connects all target variables.
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm
    # or simply realize this is a Steiner Tree problem.
    
    # Steiner Tree for small t:
    # dp[mask][i] = min cost to connect subset of targets 'mask' with root 'i'
    
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][target_vars[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dp[mask][i] using submasks
        for submask in range((mask - 1) & mask, 0, -1):
            submask &= mask
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
        
        # Dijkstra to update dp[mask][i] for fixed mask
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
