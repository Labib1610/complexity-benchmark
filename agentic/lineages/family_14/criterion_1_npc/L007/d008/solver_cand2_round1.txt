```python
import sys

def solve():
    # Read n, m, t
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_states = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    # The condition that (e_a - e_b) is in V for all a, b in priority_states
    # is equivalent to saying all priority states must be in the same connected 
    # component of the graph formed by the chosen edges.
    # This is a Minimum Spanning Tree problem on a graph where we want to 
    # connect all priority states.
    # We can use Kruskal's algorithm or Prim's.
    # Since we need to connect t nodes, we can use a Disjoint Set Union (DSU).
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    edges.sort()
    
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # We only care about connecting the priority states.
    # This is equivalent to finding the MST of the graph where we contract 
    # all non-priority nodes or simply using Kruskal's and checking connectivity.
    # Actually, the standard way to connect a subset of nodes is to find the 
    # Steiner Tree, but here the graph is undirected and we want to connect 
    # all priority nodes. This is equivalent to finding the MST of the 
    # component containing the priority nodes.
    
    # Wait, the condition is that all priority states are in the same component.
    # This is exactly the MST problem on the graph where we want to connect 
    # the set of priority nodes.
    
    total_cost = 0
    edges_count = 0
    
    # Kruskal's
    for w, u, v in edges:
        if find(u) != find(v):
            union(u, v)
            total_cost += w
            
    # Check if all priority states are connected
    root = find(priority_states[0])
    for i in range(1, t):
        if find(priority_states[i]) != root:
            # This case shouldn't happen based on problem constraints
            break
            
    print(total_cost)

solve()
```
