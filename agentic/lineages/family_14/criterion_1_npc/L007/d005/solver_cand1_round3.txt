```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_registers = [int(x) for x in input_data[3:3+t]]
    
    links = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        links.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to connect all priority registers into a single connected component.
    # This is equivalent to finding a Steiner Tree in the graph where the 
    # terminals are the priority registers.
    # Since n is small (100) and t is very small (10), we can use the 
    # Dreyfus-Wagner algorithm or a simpler DP approach.
    
    # dp[mask][i] = min cost to connect the subset of priority registers 
    # represented by 'mask', with register 'i' being in the component.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][priority_registers[i]] = 0
        
    # Precompute shortest paths between all pairs using Dijkstra or Floyd-Warshall
    dist = [[inf] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
    for u, v, w in links:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
        
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    for mask in range(1, 1 << t):
        # Split mask
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relax with edges
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dp[mask][j] + dist[j][i] < dp[mask][i]:
                    dp[mask][i] = dp[mask][j] + dist[j][i]
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
