```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    target_indices = [int(x) for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The condition "y orthogonal to all v in S implies y_a = y_b for all a, b in T"
    # is equivalent to saying that in the graph where edges are vectors in S,
    # all nodes in T must be in the same connected component.
    # This is a classic Steiner Tree problem variant, but since we only need
    # to connect a set of nodes T in a graph, it's equivalent to finding the
    # Minimum Spanning Tree of the graph where we contract all nodes in T
    # into a single super-node.
    
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # Contract all target indices into one
    first_target = target_indices[0]
    for i in range(1, t):
        union(first_target, target_indices[i])
        
    # Kruskal's algorithm
    edges.sort(key=lambda x: x[2])
    
    total_cost = 0
    for u, v, w in edges:
        if union(u, v):
            total_cost += w
            
    print(total_cost)

solve()
```
