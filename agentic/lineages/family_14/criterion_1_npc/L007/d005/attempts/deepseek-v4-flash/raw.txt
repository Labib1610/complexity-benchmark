 We need to connect the critical parameters using the cheapest possible set of edges, but we can also use non-critical nodes as intermediate connectors.  

This is essentially a **minimum Steiner tree** problem, but with a small number of terminals (t ≤ 10), so we can solve it using **bitmask DP over subsets of terminals** after computing all-pairs shortest paths.

Here’s the Python solution:

```python
import sys
import heapq

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    terminals = list(map(int, input().split()))
    terminals = [x - 1 for x in terminals]  # 0-based indexing

    # Build graph
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    # All-pairs shortest paths (Floyd-Warshall)
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            dik = dist[i][k]
            if dik == INF:
                continue
            di = dist[i]
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd

    # DP over subsets of terminals
    # dp[mask][v] = min cost to connect terminals in mask, ending at node v
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]

    # Initialize: single terminal
    for i in range(t):
        dp[1 << i][terminals[i]] = 0

    # For each mask, try to combine two submasks
    for mask in range(1, full_mask):
        # Try splitting mask into two non-empty submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if other < sub:  # avoid duplicate work
                sub = (sub - 1) & mask
                continue
            # combine
            for v in range(n):
                if dp[sub][v] + dp[other][v] < dp[mask][v]:
                    dp[mask][v] = dp[sub][v] + dp[other][v]
            sub = (sub - 1) & mask

        # Run Dijkstra from all nodes in this mask to relax distances
        # (since we can move from any node to any other using shortest paths)
        pq = []
        for v in range(n):
            if dp[mask][v] < INF:
                heapq.heappush(pq, (dp[mask][v], v))
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v in range(n):
                nd = d + dist[u][v]
                if nd < dp[mask][v]:
                    dp[mask][v] = nd
                    heapq.heappush(pq, (nd, v))

    # Answer is min over all nodes for full mask
    ans = min(dp[full_mask - 1])
    print(ans)

if __name__ == "__main__":
    solve()
```
