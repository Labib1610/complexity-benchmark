```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # We need to connect all critical nodes in the same component.
    # This is equivalent to the Steiner Tree problem, but since we only care
    # about connecting a set of terminals, and the graph is undirected,
    # we can use the Dreyfus-Wagner algorithm or simply realize that for 
    # small t, we can use dynamic programming on subsets.
    
    # dp[mask][i] = min cost to connect the subset of critical nodes represented by mask,
    # with node i being in the connected component.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][critical[i]] = 0
        
    # Dijkstra-like updates for each mask
    for mask in range(1, 1 << t):
        # Submask combination
        for submask in range(1, mask):
            if (submask & mask) == submask:
                for i in range(1, n + 1):
                    dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
        
        # Edge relaxation (Dijkstra)
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                pq.append((dp[mask][i], i))
        
        pq.sort()
        while pq:
            d, u = pq.pop(0)
            if d > dp[mask][u]:
                continue
            for v1, v2, w in edges:
                if v1 == u:
                    if dp[mask][v2] > d + w:
                        dp[mask][v2] = d + w
                        pq.append((dp[mask][v2], v2))
                        pq.sort()
                elif v2 == u:
                    if dp[mask][v1] > d + w:
                        dp[mask][v1] = d + w
                        pq.append((dp[mask][v1], v1))
                        pq.sort()
                        
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
