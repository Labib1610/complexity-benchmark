```python
import sys

# The problem asks for the minimum cost to ensure that in the nullspace of the 
# Laplacian matrix L, all priority nodes have the same displacement.
# The Laplacian matrix L of a graph is such that Lx = 0 if and only if 
# x_u = x_v for all nodes u, v in the same connected component.
# Thus, the condition "x_c1 = x_c2 = ... = x_ct" for all x in the nullspace
# is equivalent to saying that all priority nodes must belong to the same 
# connected component in the graph formed by the selected edges.
# This is a classic Minimum Spanning Tree problem variant:
# We want to connect all nodes in the set {c1, c2, ..., ct} into a single 
# connected component using the available edges with minimum cost.
# This is the Steiner Tree problem, but since we only need to connect 
# the set of priority nodes, and we can use any other nodes as intermediate 
# nodes, this is exactly the Steiner Tree problem on a graph.
# Given the constraints (n <= 100, t <= 10), we can use the Dreyfus-Wagner 
# algorithm or a simple BFS/Dijkstra approach for Steiner Tree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_nodes = [int(input_data[i + 3]) - 1 for i in range(t)]
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        w = int(input_data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Steiner Tree using DP: dp[mask][i] is min cost to connect nodes in mask 
    # with i being part of the component.
    import heapq
    
    inf = float('inf')
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][priority_nodes[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dp[mask][i] using submasks
        for submask in range((mask - 1) & mask, 0, -1):
            submask &= mask
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
        
        # Dijkstra to update dp[mask][i] for fixed mask
        pq = [(dp[mask][i], i) for i in range(n) if dp[mask][i] != inf]
        heapq.heapify(pq)
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
