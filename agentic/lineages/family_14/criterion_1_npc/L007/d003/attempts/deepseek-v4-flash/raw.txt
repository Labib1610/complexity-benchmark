We can solve this by finding the minimum spanning tree (MST) that connects all the priority access points.  
Since we can use non-priority points as intermediate nodes, the problem reduces to finding the minimum cost to connect all terminals (priority points) in a graph — this is the **Steiner tree** problem, but with a small number of terminals (t ≤ 10), we can solve it using **DP over subsets** and **shortest paths**.

Here’s the implementation:

```python
import sys
import heapq

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    terminals = list(map(int, input().split()))
    terminals = [x-1 for x in terminals]
    
    # Build adjacency list
    graph = [[] for _ in range(n)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # If only one terminal, cost is 0
    if t == 1:
        print(0)
        return
    
    # Precompute shortest distances between all pairs of terminals
    # We'll run Dijkstra from each terminal
    INF = 10**18
    dist = [[INF]*n for _ in range(t)]
    for i, s in enumerate(terminals):
        d = [INF]*n
        d[s] = 0
        pq = [(0, s)]
        while pq:
            du, u = heapq.heappop(pq)
            if du != d[u]:
                continue
            for v, w in graph[u]:
                nd = du + w
                if nd < d[v]:
                    d[v] = nd
                    heapq.heappush(pq, (nd, v))
        dist[i] = d
    
    # DP over subsets of terminals
    # dp[mask][i] = min cost to connect terminals in mask, ending at terminal i
    full_mask = (1 << t) - 1
    dp = [[INF]*t for _ in range(1 << t)]
    
    # Initialize: single terminal
    for i in range(t):
        dp[1 << i][i] = 0
    
    # For each subset
    for mask in range(1 << t):
        # Try to extend by merging two subsets
        # Submask enumeration
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid duplicate work
                for i in range(t):
                    if dp[sub][i] == INF:
                        continue
                    for j in range(t):
                        if dp[other][j] == INF:
                            continue
                        # Connect terminal i to terminal j via shortest path
                        cost = dp[sub][i] + dp[other][j] + dist[i][terminals[j]]
                        if cost < dp[mask][i]:
                            dp[mask][i] = cost
                        if cost < dp[mask][j]:
                            dp[mask][j] = cost
            sub = (sub - 1) & mask
        
        # Also try to extend by adding one terminal to the subset
        for i in range(t):
            if dp[mask][i] == INF:
                continue
            for j in range(t):
                if not (mask & (1 << j)):
                    new_mask = mask | (1 << j)
                    cost = dp[mask][i] + dist[i][terminals[j]]
                    if cost < dp[new_mask][j]:
                        dp[new_mask][j] = cost
    
    ans = min(dp[full_mask])
    print(ans)

if __name__ == "__main__":
    solve()
```
