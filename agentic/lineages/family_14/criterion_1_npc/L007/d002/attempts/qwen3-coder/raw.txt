import sys
import heapq
from collections import defaultdict

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal(nodes, edges):
    parent = {node: node for node in nodes}
    rank = {node: 0 for node in nodes}
    edges.sort()
    cost = 0
    for w, u, v in edges:
        if union(parent, rank, u, v):
            cost += w
    return cost

def dijkstra(graph, start, nodes):
    dist = {node: float('inf') for node in nodes}
    dist[start] = 0
    pq = [(0, start)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            if dist[v] > d + w:
                dist[v] = d + w
                heapq.heappush(pq, (dist[v], v))
    return dist

def main():
    n, m, t = map(int, sys.stdin.readline().split())
    priorities = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Compute shortest paths between all pairs of priority nodes
    # We can use Dijkstra from each priority node
    dists = {}
    for p in priorities:
        dists[p] = dijkstra(graph, p, range(1, n+1))
    
    # Create a complete graph on priority nodes with edge weights as shortest path distances
    priority_edges = []
    for i in range(t):
        for j in range(i+1, t):
            u, v = priorities[i], priorities[j]
            priority_edges.append((dists[u][v], u, v))
    
    # Run Kruskal's algorithm on this complete graph
    print(kruskal(priorities, priority_edges))

if __name__ == "__main__":
    main()
