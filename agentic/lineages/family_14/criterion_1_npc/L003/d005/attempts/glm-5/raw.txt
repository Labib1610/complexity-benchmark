import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    target = [int(next(it)) for _ in range(t)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    # Build graph with minimum cost between every pair of vertices.
    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    for k in range(1, n + 1):
        dk = dist[k]
        for i in range(1, n + 1):
            dik = dist[i][k]
            if dik == INF:
                continue
            di = dist[i]
            for j in range(1, n + 1):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    # Steiner tree on target vertices using Dreyfus-Wagner DP.
    # dp[mask][v] = min cost of Steiner tree connecting target vertices in mask
    #               plus vertex v (v is included in the tree).
    K = t
    dp = [[INF] * (n + 1) for _ in range(1 << K)]
    for i in range(K):
        dp[1 << i][target[i]] = 0
    
    for mask in range(1, 1 << K):
        # Merge submasks.
        sub = (mask - 1) & mask
        while sub > 0:
            other = mask ^ sub
            if sub < other:
                break
            dpm = dp[mask]
            dps = dp[sub]
            dpo = dp[other]
            for v in range(1, n + 1):
                val = dps[v] + dpo[v]
                if val < dpm[v]:
                    dpm[v] = val
            sub = (sub - 1) & mask
        
        # Relax: extend the tree from some vertex u to vertex v using shortest path.
        dpm = dp[mask]
        for v in range(1, n + 1):
            best = INF
            dv = dist[v]
            for u in range(1, n + 1):
                val = dpm[u] + dv[u]
                if val < best:
                    best = val
            if best < dpm[v]:
                dpm[v] = best
    
    full = (1 << K) - 1
    ans = min(dp[full][v] for v in range(1, n + 1))
    print(ans)

solve()
