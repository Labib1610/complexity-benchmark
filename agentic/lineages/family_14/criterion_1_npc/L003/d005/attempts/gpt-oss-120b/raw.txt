```python
import sys
import math

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    targets = [int(next(it)) - 1 for _ in range(t)]
    INF = 10**18
    # distance matrix
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            ik = di[k]
            if ik == INF:
                continue
            dj = dist[k]
            for j in range(n):
                nd = ik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    if t <= 1:
        print(0)
        return
    T = t
    full = (1 << T) - 1
    dp = [[INF] * n for _ in range(1 << T)]
    # init singletons
    for i, node in enumerate(targets):
        mask = 1 << i
        for v in range(n):
            dp[mask][v] = dist[node][v]
    # DP over subsets
    for mask in range(1, full + 1):
        # skip singletons
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            # combine only once per partition (avoid duplicate work)
            if sub < other:
                for v in range(n):
                    val = dp[sub][v] + dp[other][v]
                    if val < dp[mask][v]:
                        dp[mask][v] = val
            sub = (sub - 1) & mask
        # propagate via shortest paths
        # dp[mask][v] = min_u dp[mask][u] + dist[u][v]
        # simple O(n^2) relaxation
        improved = True
        # Since distances satisfy triangle inequality, one pass suffices
        for v in range(n):
            best = dp[mask][v]
            for u in range(n):
                cand = dp[mask][u] + dist[u][v]
                if cand < best:
                    best = cand
            dp[mask][v] = best
    ans = min(dp[full])
    print(ans)

if __name__ == "__main__":
    solve()
```
