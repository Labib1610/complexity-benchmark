```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    target_indices = []
    for _ in range(t):
        target_indices.append(int(next(it)))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Special case: if t <= 1, no constraints needed
    if t <= 1:
        print(0)
        return
    
    # Compute pairwise shortest paths between target nodes
    # dist[i][j] = minimum cost to connect target i and target j
    INF = float('inf')
    dist = [[INF] * t for _ in range(t)]
    
    for i in range(t):
        start = target_indices[i]
        d = [INF] * (n + 1)
        d[start] = 0
        pq = [(0, start)]
        while pq:
            current_dist, u = heapq.heappop(pq)
            if current_dist > d[u]:
                continue
            for v, w in adj[u]:
                new_dist = current_dist + w
                if new_dist < d[v]:
                    d[v] = new_dist
                    heapq.heappush(pq, (new_dist, v))
        
        for j in range(t):
            node_j = target_indices[j]
            dist[i][j] = d[node_j]
    
    # DP over subsets: dp[mask] = minimum cost to connect nodes in mask
    # mask is a bitmask of target indices
    dp = [INF] * (1 << t)
    for i in range(t):
        dp[1 << i] = 0
    
    for mask in range(1, 1 << t):
        # Try to extend current mask by adding one more node
        # Find the best edge to add
        for i in range(t):
            if not (mask & (1 << i)):
                # Node i is not in mask, try to add it
                best_cost = INF
                for j in range(t):
                    if mask & (1 << j):
                        # Connect i to j
                        best_cost = min(best_cost, dist[i][j])
                new_mask = mask | (1 << i)
                dp[new_mask] = min(dp[new_mask], dp[mask] + best_cost)
    
    print(dp[(1 << t) - 1])

if __name__ == "__main__":
    solve()
```
