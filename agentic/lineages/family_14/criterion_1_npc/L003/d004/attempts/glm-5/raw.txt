import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    focals = [int(next(it)) for _ in range(t)]
    
    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    for k in range(1, n + 1):
        dk = dist[k]
        for i in range(1, n + 1):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(1, n + 1):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    if t == 1:
        print(0)
        return
    
    # DP over subsets of focals, using each node as Steiner point
    size = 1 << t
    dp = [[INF] * (n + 1) for _ in range(size)]
    for i in range(t):
        f = focals[i]
        dp[1 << i][f] = 0
    
    for mask in range(1, size):
        dpm = dp[mask]
        # Merge submasks
        sub = (mask - 1) & mask
        while sub > 0:
            other = mask ^ sub
            if sub <= other:
                dps = dp[sub]
                dpo = dp[other]
                for v in range(1, n + 1):
                    val = dps[v] + dpo[v]
                    if val < dpm[v]:
                        dpm[v] = val
            sub = (sub - 1) & mask
        # Relax via shortest path from any node to v
        best = INF
        for v in range(1, n + 1):
            if dpm[v] < best:
                best = dpm[v]
        if best < INF:
            for v in range(1, n + 1):
                nd = best + dist[v][v]  # dist[v][v] is 0, but keep general
                # Actually we want best + dist[u][v] for some u, but since best is min over u of dp[mask][u],
                # and dist[u][v] is the shortest path, we can do:
                pass
            # Proper relaxation: for each v, dp[mask][v] = min over u of dp[mask][u] + dist[u][v]
            new_vals = [INF] * (n + 1)
            for v in range(1, n + 1):
                mn = INF
                dv = dist[v]
                for u in range(1, n + 1):
                    val = dpm[u] + dv[u]
                    if val < mn:
                        mn = val
                new_vals[v] = mn
            for v in range(1, n + 1):
                if new_vals[v] < dpm[v]:
                    dpm[v] = new_vals[v]
    
    full = size - 1
    ans = min(dp[full][v] for v in range(1, n + 1))
    print(ans)

solve()
