```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    flagged = []
    idx = 3
    for _ in range(t):
        flagged.append(int(input_data[idx]))
        idx += 1
        
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Steiner Tree problem on a graph with n <= 100 and t <= 10.
    # DP state: dp[mask][i] is the min cost to connect the subset of flagged 
    # accounts represented by mask, with i being part of the tree.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][flagged[i]] = 0
        
    for mask in range(1, 1 << t):
        # Combine submasks
        for submask in range((mask - 1) & mask, 0, -1):
            submask2 = mask ^ submask
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[submask2][i])
        
        # Dijkstra to update dp[mask]
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
