```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    cores = [int(x) - 1 for x in input_data[3:3+t]]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "core-cohesive" means all core modules must be in the same 
    # connected component. If we partition the modules into A and B such that 
    # both contain at least one core module, there must be an edge between A and B.
    # To make it impossible to partition such that no edge crosses, we must 
    # ensure all core modules are in the same connected component.
    # This is equivalent to the Steiner Tree problem, but since n is small (100),
    # we can use the Dreyfus-Wagner algorithm or simply realize that for 
    # t <= 10, we can use min-cut.
    # Actually, the problem asks for the minimum cost to connect all core modules.
    # This is exactly the Steiner Tree problem.
    
    # Since n=100 and t=10, we can use the DP approach for Steiner Tree:
    # dp[mask][i] = min cost to connect the subset of cores represented by mask,
    # with i being one of the nodes in the tree.
    
    inf = float('inf')
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][cores[i]] = 0
        
    for mask in range(1, 1 << t):
        # Submask combination
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Dijkstra to update dp[mask][i]
        pq = []
        import heapq
        for i in range(n):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
            
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
