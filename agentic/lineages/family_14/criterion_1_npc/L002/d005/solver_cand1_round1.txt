```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_indices = [int(x) - 1 for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The condition "joint kernel is a subspace of the kernel of x_a - x_b"
    # is equivalent to saying that in the graph formed by the activated functionals,
    # all critical indices must be in the same connected component.
    # This is a Steiner Tree problem where the terminals are the critical indices.
    # Since t <= 10, we can use the Dreyfus-Wagner algorithm.
    
    # Distance matrix using Floyd-Warshall
    dist = [[float('inf')] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
            
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][j] > dist[i][k] + dist[k][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    # dp[mask][i] is the min cost to connect the subset of critical indices
    # represented by mask, with i being a node in the resulting tree.
    num_masks = 1 << t
    dp = [[float('inf')] * n for _ in range(num_masks)]
    
    for i in range(t):
        dp[1 << i][critical_indices[i]] = 0
        
    for mask in range(1, num_masks):
        # Submask splitting
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relax with edges
        for i in range(n):
            for j in range(n):
                if dp[mask][j] + dist[j][i] < dp[mask][i]:
                    dp[mask][i] = dp[mask][j] + dist[j][i]
                    
    print(min(dp[num_masks - 1]))

solve()
```
