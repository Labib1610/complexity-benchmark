import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    target_elements = [int(x) for x in input_data[3:3+t]]
    target_set = set(target_elements)
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "Any subset Y that splits T must split at least one pair in S"
    # is equivalent to saying that the graph formed by S must connect all elements of T
    # into a single connected component, or more precisely, that the partition of E
    # induced by the connected components of S must be a refinement of the partition
    # {T, E \ T}.
    # Actually, the condition is equivalent to: for any partition of T into two non-empty
    # sets T1 and T2, there must be an edge in S connecting a node in T1 to a node in T2.
    # This is equivalent to saying that all elements of T must be in the same connected
    # component in the graph formed by S.
    
    # Kruskal's algorithm to find the MST of the target elements.
    # We want to connect all elements in T.
    # We can use a Disjoint Set Union (DSU) to keep track of components.
    
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False
    
    # We need to connect all nodes in T.
    # This is a Steiner Tree problem, but since we only care about connecting
    # the set T, and we can use any nodes in E as intermediate nodes,
    # this is exactly the Steiner Tree problem on the graph G=(V, E) with terminals T.
    # Since |T| <= 10, we can use the Dreyfus-Wagner algorithm.
    
    inf = float('inf')
    # dp[mask][i] is the min cost to connect the subset of T represented by mask,
    # with i being in the connected component.
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][target_elements[i]] = 0
        
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    for mask in range(1, 1 << t):
        # Submask update
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Dijkstra update
        import heapq
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
