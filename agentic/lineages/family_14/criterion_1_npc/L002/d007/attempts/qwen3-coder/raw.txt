from itertools import combinations

def solve():
    n, m, t = map(int, input().split())
    target_coords = set(map(int, input().split()))
    
    candidates = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        candidates.append((u, v, w))
    
    # If there's only one target coordinate, L is all of R^n, so S can be empty
    if t <= 1:
        print(0)
        return
    
    # For S⊥ ⊆ L, we need that any vector x with x·y = 0 for all y in S
    # must satisfy x_a = x_b for all target coordinates a,b.
    #
    # This is equivalent to saying that for any pair of target coordinates (a,b),
    # there must exist a vector in the span of S with 1 in position a and -1 in position b.
    # (This forces x_a = x_b in any vector orthogonal to S.)
    #
    # So we need to ensure that for every pair of target coordinates,
    # there's a way to combine selected candidate vectors to get a vector
    # with 1 at one target coord and -1 at another target coord.
    
    # This is essentially a graph connectivity problem.
    # Each candidate vector (u,v) is an edge in a graph.
    # We want to ensure that the target coordinates form a connected component
    # in the graph formed by selected edges.
    
    # We can model this as a minimum spanning tree problem on the target coordinates.
    # For each pair of target coordinates, we compute the minimum cost to connect them
    # (using Dijkstra or similar to find shortest path).
    # Then we find the minimum spanning tree on the target coordinates with these distances.
    
    # But there's a simpler approach: we can use a graph-based method.
    # Create a graph where nodes are target coordinates.
    # For each candidate vector (u,v,w), if both u and v are target coordinates,
    # add an edge between them with weight w.
    # Then find the minimum spanning tree of this graph.
    
    # Actually, we need to be more careful. 
    # The condition is that S⊥ ⊆ L.
    # L is the space where all target coordinates are equal.
    # S⊥ is the space orthogonal to all selected vectors.
    # For S⊥ ⊆ L, it means that if x is orthogonal to all vectors in S,
    # then x has all target coordinates equal.
    
    # This is equivalent to: for every pair of target coordinates a,b,
    # there exists a linear combination of vectors in S that has value 1 at a and -1 at b.
    # In other words, the target coordinates must be connected in the graph formed by S.
    
    # So we need to find the minimum cost set of edges such that 
    # all target coordinates are in the same connected component.
    # This is the minimum spanning tree problem on the target coordinates.
    
    # But we need to compute the minimum cost to connect each pair of target coordinates.
    # This can be done with a shortest path algorithm.
    
    # Let's build a graph and compute all-pairs shortest paths among target coordinates.
    
    import heapq
    from collections import defaultdict
    
    # Build adjacency list for the full graph
    graph = defaultdict(list)
    for u, v, w in candidates:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Function to compute shortest path between two nodes
    def dijkstra(start, end):
        if start == end:
            return 0
        dist = {i: float('inf') for i in range(1, n+1)}
        dist[start] = 0
        pq = [(0, start)]
        
        while pq:
            d, u = heapq.heappop(pq)
            if u == end:
                return d
            if d > dist[u]:
                continue
            for v, w in graph[u]:
                if dist[u] + w < dist[v]:
                    dist[v] = dist[u] + w
                    heapq.heappush(pq, (dist[v], v))
        return dist[end] if dist[end] != float('inf') else -1
    
    # Compute distances between all pairs of target coordinates
    target_list = list(target_coords)
    target_indices = {coord: i for i, coord in enumerate(target_list)}
    k = len(target_list)
    
    # Distance matrix for target coordinates
    dist_matrix = [[float('inf')] * k for _ in range(k)]
    for i in range(k):
        dist_matrix[i][i] = 0
    
    # For each candidate that connects two target coordinates directly
    for u, v, w in candidates:
        if u in target_coords and v in target_coords:
            i, j = target_indices[u], target_indices[v]
            dist_matrix[i][j] = min(dist_matrix[i][j], w)
            dist_matrix[j][i] = min(dist_matrix[j][i], w)
    
    # If t <= 1, no connections needed
    if t <= 1:
        print(0)
        return
    
    # For pairs not directly connected, compute shortest paths
    for i in range(k):
        for j in range(i+1, k):
            if dist_matrix[i][j] == float('inf'):
                # Compute shortest path between target_list[i] and target_list[j]
                cost = dijkstra(target_list[i], target_list[j])
                if cost != -1:
                    dist_matrix[i][j] = cost
                    dist_matrix[j][i] = cost
    
    # Now find minimum spanning tree on target coordinates
    # Prim's algorithm
    visited = [False] * k
    min_edge = [float('inf')] * k
    min_edge[0] = 0
    total_cost = 0
    
    for _ in range(k):
        # Find the vertex with minimum edge weight not yet included
        u = -1
        for v in range(k):
            if not visited[v] and (u == -1 or min_edge[v] < min_edge[u]):
                u = v
        
        if min_edge[u] == float('inf'):
            # Graph is not connected, but problem guarantees solution exists
            # This shouldn't happen
            pass
            
        visited[u] = True
        total_cost += min_edge[u]
        
        # Update min_edge values for adjacent vertices
        for v in range(k):
            if not visited[v] and dist_matrix[u][v] < min_edge[v]:
                min_edge[v] = dist_matrix[u][v]
    
    print(total_cost)

solve()
