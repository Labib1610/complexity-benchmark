```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The condition S^perp subset L is equivalent to L^perp subset S.
    # L = {x | x_a = x_b for all a, b in targets}.
    # L^perp is spanned by vectors e_a - e_b for all a, b in targets.
    # This is equivalent to saying that in the graph where edges are S,
    # all target nodes must be in the same connected component.
    # We want to select a subset of edges S such that all target nodes are connected,
    # minimizing the total cost.
    # This is the Steiner Tree problem, but since we only have t <= 10,
    # we can use the Dreyfus-Wagner algorithm or simply treat it as a 
    # Minimum Spanning Tree problem on a graph where we contract nodes.
    # Actually, since we only need to connect the set of target nodes,
    # we can use the Steiner Tree DP.
    
    # Steiner Tree DP:
    # dp[mask][i] = min cost to connect the subset of targets represented by mask,
    # with node i being part of the tree.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][targets[i]] = 0
        
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    for mask in range(1, 1 << t):
        # Combine submasks
        for submask in range((mask - 1) & mask, 0, -1):
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
        
        # Dijkstra to update dp[mask][i]
        import heapq
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
