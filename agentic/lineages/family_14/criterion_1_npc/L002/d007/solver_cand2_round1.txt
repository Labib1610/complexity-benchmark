```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    essential = [int(input_data[i + 3]) - 1 for i in range(t)]
    
    # D is diagonal, let D = diag(d_1, ..., d_n).
    # L_{u,v}(D) = D E_{u,v} - E_{u,v} D
    # (D E_{u,v})_{ij} = d_i if i=u, j=v else 0
    # (E_{u,v} D)_{ij} = d_j if i=u, j=v else 0
    # So L_{u,v}(D) = (d_u - d_v) E_{u,v}
    # Kernel condition: L_{u,v}(D) = 0 iff d_u = d_v.
    # If we activate (u, v), we force d_u = d_v.
    # This is equivalent to saying u and v are in the same connected component.
    
    # The condition: for all D in K, Tr(D E_{a,a}) = Tr(D E_{b,b}) for all a, b in essential.
    # Tr(D E_{a,a}) = d_a.
    # So we need d_a = d_b for all a, b in essential.
    # This means all essential indices must be in the same connected component.
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        w = int(input_data[idx + 2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to connect all essential nodes in a graph where edges have costs.
    # This is the Steiner Tree problem, but with only t nodes to connect.
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm.
    
    # dist[mask][i] = min cost to connect subset of essential nodes 'mask' 
    # such that they are all connected to node i.
    
    inf = float('inf')
    dist = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dist[1 << i][essential[i]] = 0
        
    # Precompute all-pairs shortest paths for the graph
    adj = [[inf] * n for _ in range(n)]
    for i in range(n): adj[i][i] = 0
    for u, v, w in edges:
        adj[u][v] = min(adj[u][v], w)
        adj[v][u] = min(adj[v][u], w)
        
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if adj[i][k] + adj[k][j] < adj[i][j]:
                    adj[i][j] = adj[i][k] + adj[k][j]
                    
    for mask in range(1, 1 << t):
        # Update using submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dist[mask][i] = min(dist[mask][i], dist[submask][i] + dist[mask ^ submask][i])
            submask = (submask - 1) & mask
        
        # Update using shortest paths
        for i in range(n):
            for j in range(n):
                if dist[mask][j] + adj[j][i] < dist[mask][i]:
                    dist[mask][i] = dist[mask][j] + adj[j][i]
                    
    print(min(dist[(1 << t) - 1]))

solve()
```
