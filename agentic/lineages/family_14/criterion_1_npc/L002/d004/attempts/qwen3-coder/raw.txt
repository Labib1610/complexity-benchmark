import sys
from itertools import combinations

def find_min_cost_to_span_targets(n, m, t, targets, vectors):
    # If there's only one target, no vectors are needed
    if t <= 1:
        return 0
    
    # Create a mapping from target indices to a smaller range for easier handling
    target_set = set(targets)
    target_index_map = {targets[i]: i for i in range(t)}
    
    # Generate all required target pairs (edges in a complete graph of targets)
    required_pairs = list(combinations(targets, 2))
    
    # Build a graph where nodes are target indices and edges are the minimum cost
    # to connect them (directly or indirectly)
    # We'll use Union-Find (Disjoint Set Union) to track connected components
    
    class UnionFind:
        def __init__(self, size):
            self.parent = list(range(size))
            self.rank = [0] * size
        
        def find(self, x):
            if self.parent[x] != x:
                self.parent[x] = self.find(self.parent[x])  # Path compression
            return self.parent[x]
        
        def union(self, x, y):
            px, py = self.find(x), self.find(y)
            if px == py:
                return False
            # Union by rank
            if self.rank[px] < self.rank[py]:
                px, py = py, px
            self.parent[py] = px
            if self.rank[px] == self.rank[py]:
                self.rank[px] += 1
            return True
    
    # Sort vectors by weight (cost)
    vectors.sort(key=lambda x: x[2])
    
    # Use Kruskal's algorithm to find the minimum spanning tree that connects all targets
    uf = UnionFind(t)
    total_cost = 0
    edges_added = 0
    
    for u, v, w in vectors:
        # Check if both u and v are relevant (at least one is a target)
        u_is_target = u in target_set
        v_is_target = v in target_set
        
        if not u_is_target and not v_is_target:
            continue  # Neither endpoint is a target, skip
            
        # Get indices in the target mapping
        u_idx = target_index_map.get(u, -1)
        v_idx = target_index_map.get(v, -1)
        
        # If one endpoint is not a target, we need to check if adding this edge
        # helps connect targets. To handle this properly, we model the problem
        # as finding the MST in a graph where targets are nodes and the cost
        # between two targets is the shortest path between them in the original graph.
        
        # However, a simpler approach is to build a graph with all nodes,
        # then find the minimum cost to connect all targets.
        
        # Let's restart with a cleaner approach:
        # Model this as a Steiner Tree problem on graphs, but since t <= 10,
        # we can use the following approach:
        # 1. Compute shortest paths between all pairs of targets
        # 2. Find the minimum spanning tree in the complete graph of targets
        #    with edge weights equal to shortest path distances
        
    # For the clean implementation, let's use a different approach:
    # Since t is small (<= 10), we can:
    # 1. Compute shortest path between every pair of targets
    # 2. Create a complete graph on targets with these distances
    # 3. Find the MST of this graph
    
    # But even simpler: use the fact that we want to connect all targets,
    # so we can run a modified Kruskal on the original graph focusing only
    # on connecting target nodes.
    
    # Restart with clear algorithm:
    # Use Union-Find on target nodes. For each edge in the original graph,
    # if it connects two components that contain targets, add it if it merges components.
    
    # Track which components contain targets
    component_has_target = [False] * (n + 1)
    for target in targets:
        component_has_target[target] = True
    
    parent_nodes = list(range(n + 1))
    rank_nodes = [0] * (n + 1)
    
    def find_node(x):
        if parent_nodes[x] != x:
            parent_nodes[x] = find_node(parent_nodes[x])
        return parent_nodes[x]
    
    def union_nodes(x, y):
        px, py = find_node(x), find_node(y)
        if px == py:
            return False
        if rank_nodes[px] < rank_nodes[py]:
            px, py = py, px
        parent_nodes[py] = px
        component_has_target[px] = component_has_target[px] or component_has_target[py]
        if rank_nodes[px] == rank_nodes[py]:
            rank_nodes[px] += 1
        return True
    
    # Count how many targets are connected
    targets_connected = 1  # Start with first target as "connected"
    uf_targets = UnionFind(t)
    
    total_cost = 0
    for u, v, w in vectors:
        # Check if this edge connects two different components
        if union_nodes(u, v):
            # Check how many new target-to-target connections this enables
            u_root_targets = set()
            v_root_targets = set()
            
            # Find which targets are in u's component and v's component
            u_root = find_node(u)
            v_root = find_node(v)
            
            # This is inefficient; let's think differently.
            
            # Simpler correct approach:
            # Run Kruskal's algorithm on the full graph, but only count cost
            # when it helps connect target nodes.
            
    # Correct and simple solution:
    # Use Union-Find on the full set of nodes (1..n)
    # Process edges in order of increasing weight
    # When processing an edge, if it connects two different components,
    # check if this connection brings us closer to connecting all targets
    # Actually, just run standard Kruskal and stop when all targets are connected
    
    parent_full = list(range(n + 1))
    rank_full = [0] * (n + 1)
    
    def find_full(x):
        if parent_full[x] != x:
            parent_full[x] = find_full(parent_full[x])
        return parent_full[x]
    
    def union_full(x, y):
        px, py = find_full(x), find_full(y)
        if px == py:
            return False
        if rank_full[px] < rank_full[py]:
            px, py = py, px
        parent_full[py] = px
        if rank_full[px] == rank_full[py]:
            rank_full[px] += 1
        return True
    
    # Check initial number of connected target components
    target_components = {}
    for target in targets:
        root = find_full(target)
        if root not in target_components:
            target_components[root] = []
        target_components[root].append(target)
    
    components_with_targets = len(target_components)
    
    total_cost = 0
    for u, v, w in vectors:
        if union_full(u, v):  # If this edge connects two different components
            # Recalculate number of target components
            target_components_new = {}
            for target in targets:
                root = find_full(target)
                if root not in target_components_new:
                    target_components_new[root] = []
                target_components_new[root].append(target)
            
            new_components_with_targets = len(target_components_new)
            
            # If we reduced the number of components, we made progress
            if new_components_with_targets < components_with_targets:
                total_cost += w
                components_with_targets = new_components_with_targets
                
            # If all targets are in one component, we're done
            if components_with_targets == 1:
                break
    
    return total_cost

vectors = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    vectors.append((u, v, w))
