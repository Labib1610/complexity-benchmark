```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    anchors = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    edges.sort()
    
    # Steiner Tree problem for small number of terminals (t <= 10)
    # dp[mask][i] is the min cost to connect terminals in mask, with i as the root
    # mask is a bitmask of length t
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][anchors[i]] = 0
        
    for mask in range(1, 1 << t):
        # Submask combination
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Dijkstra to update dp[mask]
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                pq.append((dp[mask][i], i))
        
        pq.sort()
        
        # Adjacency list
        adj = [[] for _ in range(n + 1)]
        for w, u, v in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
            
        while pq:
            d, u = pq.pop(0)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > dp[mask][u] + w:
                    dp[mask][v] = dp[mask][u] + w
                    pq.append((dp[mask][v], v))
                    pq.sort()
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
