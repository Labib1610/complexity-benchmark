import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]

    INF = 10**15
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w

    # Floyd‑Warshall for all‑pairs shortest paths
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            dik = dist[i][k]
            if dik == INF:
                continue
            di = dist[i]
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd

    if t == 1:
        print(0)
        return

    full = 1 << t
    dp = [[INF] * n for _ in range(full)]

    # base masks: single terminal
    for i, term in enumerate(terminals):
        mask = 1 << i
        for v in range(n):
            dp[mask][v] = dist[term][v]

    # DP over subsets
    for mask in range(1, full):
        # skip singletons
        if mask & (mask - 1):
            # combine two non‑empty proper submasks
            sub = (mask - 1) & mask
            while sub:
                other = mask ^ sub
                # to avoid duplicate work, ensure sub < other (optional)
                if sub < other:
                    for v in range(n):
                        val = dp[sub][v] + dp[other][v]
                        if val < dp[mask][v]:
                            dp[mask][v] = val
                sub = (sub - 1) & mask
        # after merging, improve via moving root using distances
        # classic relaxation: dp[mask][v] = min_u dp[mask][u] + dist[u][v]
        # we can run a simple Floyd‑like relaxation (since n small)
        improved = True
        while improved:
            improved = False
            for u in range(n):
                du = dp[mask][u]
                if du == INF:
                    continue
                for v in range(n):
                    nd = du + dist[u][v]
                    if nd < dp[mask][v]:
                        dp[mask][v] = nd
                        improved = True

    ans = min(dp[full - 1])
    print(ans)

if __name__ == "__main__":
    solve()
