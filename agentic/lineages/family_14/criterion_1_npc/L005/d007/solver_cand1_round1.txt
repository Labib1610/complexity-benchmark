```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    critical_flags = [int(x) - 1 for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem is equivalent to finding a subset of edges such that 
    # for every pair of critical nodes (c_i, c_j), there exists a path 
    # between them in the graph formed by the chosen edges.
    # This is because if there is a path between c_i and c_j, we can toggle 
    # all edges along the path to flip c_i and c_j while leaving intermediate 
    # nodes unchanged (they are toggled twice).
    # This is equivalent to saying all critical nodes must be in the same 
    # connected component.
    
    # We need to find the minimum cost to connect all critical nodes.
    # This is the Steiner Tree problem, but since we only have one component 
    # containing all critical nodes, it's equivalent to finding a Minimum 
    # Spanning Tree in a graph where we can contract nodes or use 
    # shortest paths.
    # Actually, since we just need all critical nodes to be connected, 
    # we can use the Dreyfus-Wagner algorithm or simply realize that 
    # for small t, we can use dynamic programming on subsets.
    
    # dist[i][j] is the shortest path between node i and j
    inf = float('inf')
    dist = [[inf] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
            
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    # dp[mask][i] = min cost to connect the subset of critical nodes represented by mask,
    # with i being a node in the resulting tree.
    dp = [[inf] * n for _ in range(1 << t)]
    for i in range(t):
        dp[1 << i][critical_flags[i]] = 0
        
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(n):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
        
        # Relax with edges
        for i in range(n):
            for j in range(n):
                if dp[mask][j] + dist[j][i] < dp[mask][i]:
                    dp[mask][i] = dp[mask][j] + dist[j][i]
                    
    print(min(dp[(1 << t) - 1]))

solve()
```
