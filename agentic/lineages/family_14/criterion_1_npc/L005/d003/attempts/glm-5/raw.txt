import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    delegated = [int(data[idx + i]) for i in range(t)]
    idx += t

    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0

    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    for k in range(1, n + 1):
        dk = dist[k]
        for i in range(1, n + 1):
            dik = dist[i][k]
            if dik == INF:
                continue
            di = dist[i]
            for j in range(1, n + 1):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd

    T = t
    dp = [[INF] * T for _ in range(1 << T)]
    for i in range(T):
        dp[1 << i][i] = 0

    for mask in range(1 << T):
        for last in range(T):
            if not (mask & (1 << last)):
                continue
            cur = dp[mask][last]
            if cur == INF:
                continue
            for nxt in range(T):
                if mask & (1 << nxt):
                    continue
                cost = dist[delegated[last]][delegated[nxt]]
                if cost == INF:
                    continue
                nm = mask | (1 << nxt)
                if cur + cost < dp[nm][nxt]:
                    dp[nm][nxt] = cur + cost

    full = (1 << T) - 1
    ans = min(dp[full])
    print(ans)

solve()
