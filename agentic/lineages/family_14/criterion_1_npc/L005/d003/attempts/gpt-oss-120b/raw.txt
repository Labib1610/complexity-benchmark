```python
import sys
import heapq

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]
    # build undirected graph with minimal weight per pair
    adj = [[] for _ in range(n)]
    edge_min = {}
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if u > v:
            u, v = v, u
        key = (u, v)
        if key not in edge_min or w < edge_min[key]:
            edge_min[key] = w
    for (u, v), w in edge_min.items():
        adj[u].append((v, w))
        adj[v].append((u, w))

    if t <= 1:
        print(0)
        return

    INF = 10**18
    full_mask = (1 << t) - 1
    dp = [[INF] * n for _ in range(1 << t)]

    # initialize single-terminal masks
    term_index = {terminals[i]: i for i in range(t)}
    for v in range(n):
        if v in term_index:
            mask = 1 << term_index[v]
            dp[mask][v] = 0

    # for each mask, combine submasks
    for mask in range(1, full_mask + 1):
        # skip singletons
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            # combine only once per partition (sub < other to avoid duplicate)
            if sub < other:
                for v in range(n):
                    val = dp[sub][v] + dp[other][v]
                    if val < dp[mask][v]:
                        dp[mask][v] = val
            sub = (sub - 1) & mask

        # run Dijkstra to propagate through edges for this mask
        heap = [(dp[mask][v], v) for v in range(n) if dp[mask][v] < INF]
        heapq.heapify(heap)
        visited = [False] * n
        while heap:
            d, u = heapq.heappop(heap)
            if visited[u]:
                continue
            visited[u] = True
            if d != dp[mask][u]:
                continue
            for v, w in adj[u]:
                nd = d + w
                if nd < dp[mask][v]:
                    dp[mask][v] = nd
                    heapq.heappush(heap, (nd, v))

    answer = min(dp[full_mask])
    print(answer)

if __name__ == "__main__":
    solve()
```
