import sys
import heapq
from collections import defaultdict

def dijkstra(graph, start, nodes):
    dist = {node: float('inf') for node in nodes}
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
                
    return dist

        w = int(data[index+2])
        index += 3
        edges.append((u, v, w))
        graph[u].append((v, w))
        graph[v].append((u, w))  # Since we get reverse for free
    
    if t <= 1:

    # For each delegated language, compute shortest paths to all other nodes
    distances = {}
    for lang in delegated:
        distances[lang] = dijkstra(graph, lang, range(1, n+1))

    # Actually, better approach: create a complete graph where weight between 
    # two delegated languages is their shortest path distance, then find MST
    
    # Compute all pairwise shortest paths between delegated languages
    pairwise_distances = []
    for i in range(t):
        for j in range(i+1, t):
            d1 = delegated[i]
            d2 = delegated[j]
            # Distance is the shortest path between d1 and d2
            dist = distances[d1][d2]
            if dist != float('inf'):
                pairwise_distances.append((dist, d1, d2))
    
    # Sort edges by weight
    pairwise_distances.sort()
    
    # Union-Find for Kruskal's algorithm
    parent = {lang: lang for lang in delegated}
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # Kruskal's algorithm to find MST
    total_cost = 0
    edges_used = 0
    
    for weight, u, v in pairwise_distances:
        if union(u, v):
            total_cost += weight
            edges_used += 1
            if edges_used == t - 1:
                break
                
    print(total_cost)

if __name__ == "__main__":
    main()
