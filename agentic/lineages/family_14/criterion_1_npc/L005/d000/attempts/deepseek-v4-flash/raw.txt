We can solve this by treating it as a **Steiner tree** problem, but since the number of terminals (delegated languages) is at most 10, we can use **bitmask DP** over subsets of terminals.

Here’s the plan:

1. Run **Floyd–Warshall** to get shortest paths between all pairs of nodes.
2. Let `dp[mask][v]` = minimum cost to connect the terminals in `mask` and end at node `v`.
3. Initialize `dp[1 << i][terminal[i]] = 0` for each terminal.
4. Combine subsets: for each mask, try merging two submasks at the same node.
5. Relax using shortest paths: from `dp[mask][u]`, go to any `v` with cost `dist[u][v]`.
6. Answer is `min(dp[(1<<t)-1][v])` over all `v`.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]
    
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            if dist[i][k] == INF:
                continue
            for j in range(n):
                if dist[k][j] == INF:
                    continue
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    # DP over subsets
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]
    
    for i in range(t):
        dp[1 << i][terminals[i]] = 0
    
    for mask in range(full_mask):
        # Combine subsets
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for v in range(n):
                    if dp[sub][v] + dp[other][v] < dp[mask][v]:
                        dp[mask][v] = dp[sub][v] + dp[other][v]
            sub = (sub - 1) & mask
        
        # Relax via shortest paths
        for u in range(n):
            if dp[mask][u] == INF:
                continue
            for v in range(n):
                if dist[u][v] == INF:
                    continue
                if dp[mask][u] + dist[u][v] < dp[mask][v]:
                    dp[mask][v] = dp[mask][u] + dist[u][v]
    
    ans = min(dp[full_mask - 1])
    print(ans)

if __name__ == "__main__":
    solve()
```
