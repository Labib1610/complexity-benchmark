import sys
import heapq
from itertools import combinations

def dijkstra(graph, start, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    
    return dist

def main():
    n, m, t = map(int, sys.stdin.readline().split())
    delegated = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
    
    # Build graph
    graph = [[] for _ in range(n + 1)]
    edges = []
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
        edges.append((u, v, w))
    
    # Compute shortest paths between all pairs of delegated languages
    # We only need distances between delegated languages
    dists = {}
    for lang in delegated:
        dists[lang] = dijkstra(graph, lang, n)
    
    # Create a complete graph of delegated languages with shortest path weights
    delegated_graph = []
    for i in range(t):
        for j in range(i + 1, t):
            u, v = delegated[i], delegated[j]
            weight = dists[u][v]
            delegated_graph.append((u, v, weight))
    
    # Kruskal's algorithm to find MST of delegated languages
    delegated_graph.sort(key=lambda x: x[2])
    
    parent = {lang: lang for lang in delegated}
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    total_cost = 0
    edges_used = 0
    
    for u, v, w in delegated_graph:
        if union(u, v):
            total_cost += w
            edges_used += 1
            if edges_used == t - 1:
                break
    
    print(total_cost)

if __name__ == "__main__":
    main()
