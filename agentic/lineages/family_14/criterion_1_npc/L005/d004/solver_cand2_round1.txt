```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    core_departments = [int(x) for x in input_data[3:3+k]]
    
    if k <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + k
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((w, u, v))
        idx += 3
        
    # We need to connect all core departments. This is a Steiner Tree problem.
    # Since the number of core departments is small (<= 10), we can use the 
    # Dreyfus-Wagner algorithm or a variation.
    # Given the constraints and the nature of the problem, we can use 
    # Dijkstra on the state space (mask, u) where mask is the set of core 
    # departments connected to u.
    
    adj = [[] for _ in range(n + 1)]
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # dp[mask][u] = min cost to connect core departments in mask, with u in the component
    # mask is a bitmask of length k
    num_masks = 1 << k
    dp = [[float('inf')] * (n + 1) for _ in range(num_masks)]
    
    for i in range(k):
        dp[1 << i][core_departments[i]] = 0
        
    for mask in range(1, num_masks):
        # 1. Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for u in range(1, n + 1):
                dp[mask][u] = min(dp[mask][u], dp[submask][u] + dp[mask ^ submask][u])
            submask = (submask - 1) & mask
            
        # 2. Dijkstra to update costs for the current mask
        pq = []
        for u in range(1, n + 1):
            if dp[mask][u] != float('inf'):
                heapq.heappush(pq, (dp[mask][u], u))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[num_masks - 1][u] for u in range(1, n + 1))
    print(ans)

solve()
```
