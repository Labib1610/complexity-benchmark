```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_assets = [int(input_data[i + 3]) for i in range(t)]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        edges.append((w, u, v))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Kruskal's to find MST
    edges.sort()
    parent = list(range(n + 1))
    def find(i):
        if parent[i] == i:
            return i
        parent[i] = find(parent[i])
        return parent[i]
    
    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j:
            parent[root_i] = root_j
            return True
        return False

    # Steiner Tree problem for small t (t <= 10)
    # dp[mask][i] = min cost to connect subset of key assets 'mask' 
    # such that they are all connected to node 'i'
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][key_assets[i]] = 0
        
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Dijkstra-like update for current mask
        changed = True
        while changed:
            changed = False
            for w, u, v in edges:
                if dp[mask][u] + w < dp[mask][v]:
                    dp[mask][v] = dp[mask][u] + w
                    changed = True
                if dp[mask][v] + w < dp[mask][u]:
                    dp[mask][u] = dp[mask][v] + w
                    changed = True
                    
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
