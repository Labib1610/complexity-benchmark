import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    critical_bays = [int(x) - 1 for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem asks for a set of swaps such that for any pair of critical bays (i, j),
    # we can perform a swap of crates at i and j.
    # This is equivalent to saying that the group generated by the chosen swaps
    # must contain all transpositions (i, j) for all critical bays i, j.
    # This is equivalent to saying that all critical bays must belong to the same
    # connected component in the graph formed by the chosen swaps, and that
    # the component must be "2-transitive" or simply that we can perform any
    # permutation of the critical bays.
    # Actually, the condition "can swap any pair of critical bays" is equivalent
    # to saying that all critical bays are in the same connected component of the
    # graph formed by the chosen edges.
    # Wait, the example 1 shows that we can use an auxiliary node (bay 4) to swap
    # (1, 2) using (1, 4) and (2, 4). This is a standard result: if we have a
    # connected graph, we can perform any transposition (i, j) by moving i to j's
    # position and back.
    # Thus, the condition is simply that all critical bays must be in the same
    # connected component of the graph formed by the chosen edges.
    
    # We need to find the minimum cost to connect all critical bays.
    # This is the Steiner Tree problem, but with t <= 10, it's solvable.
    # Since we only need to connect a set of terminals, and n is small (100),
    # we can use the Dreyfus-Wagner algorithm or simply realize that for t <= 10,
    # we can use the Steiner Tree approach.
    
    # Actually, for t <= 10, we can use the following DP:
    # dp[mask][i] = min cost to connect the subset of critical bays represented by mask,
    # with i being a node in the tree.
    
    inf = float('inf')
    dist = [[inf] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
        
    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    dp = [[inf] * n for _ in range(1 << t)]
    for i in range(t):
        dp[1 << i][critical_bays[i]] = 0
        
    for mask in range(1, 1 << t):
        for i in range(n):
            submask = (mask - 1) & mask
            while submask > 0:
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
                submask = (submask - 1) & mask
        
        # Relax with edges
        pq = []
        for i in range(n):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]: continue
            for v in range(n):
                if dist[u][v] != inf and dp[mask][v] > d + dist[u][v]:
                    dp[mask][v] = d + dist[u][v]
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    print(min(dp[(1 << t) - 1]))

solve()
