```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t < 2:
        print(0)
        return
    
    priority_bays = [int(input_data[i + 3]) for i in range(t)]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        edges.append((u, v, w))
        idx += 3
        
    # The problem asks for a set of tracks such that for every pair of high-priority bays (i, j),
    # we can swap their contents while leaving others fixed.
    # This is possible if and only if all high-priority bays belong to the same connected component
    # in the graph formed by the purchased tracks.
    # If they are in the same component, we can perform any transposition (i, j)
    # using a spanning tree of that component.
    # Specifically, if we have a spanning tree, we can swap any two nodes by moving
    # the contents along the path between them.
    # Thus, we need to find the Minimum Spanning Tree that connects all high-priority bays.
    # This is the Steiner Tree problem, but since we only need to connect the t high-priority bays,
    # and t is small (t < 10), we can use the Dreyfus-Wagner algorithm or simply
    # realize that for small t, we can use a simpler approach.
    # Actually, for t <= 10, the Steiner Tree problem is solvable.
    
    # Steiner Tree for small t:
    # dp[mask][v] = min cost to connect the subset of high-priority bays represented by mask,
    # with v being a node in the tree.
    
    inf = float('inf')
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][priority_bays[i]] = 0
        
    for mask in range(1, 1 << t):
        # Submask update
        for submask in range((mask - 1) & mask, 0, -1):
            submask = (submask & mask)
            for v in range(1, n + 1):
                dp[mask][v] = min(dp[mask][v], dp[submask][v] + dp[mask ^ submask][v])
        
        # Dijkstra update
        pq = []
        for v in range(1, n + 1):
            if dp[mask][v] != inf:
                heapq.heappush(pq, (dp[mask][v], v))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][v] for v in range(1, n + 1))
    print(ans)

solve()
```
