```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core = [int(x) - 1 for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The condition "Any autonomous subset must contain all or none of the core"
    # is equivalent to saying that for any partition of the core into two non-empty
    # sets C1 and C2, there must be at least one purchased edge crossing the cut
    # between C1 and C2.
    # This is a classic min-cut problem. We want to ensure that for every 
    # partition of the core set into two non-empty sets, the cut capacity is > 0.
    # This is equivalent to saying that in the graph of purchased edges, 
    # all core nodes must be in the same connected component.
    # Wait, that's not quite right. The condition is that no autonomous set 
    # can contain a proper non-empty subset of the core.
    # An autonomous set is a set S such that no edges leave S.
    # If we want no autonomous set to contain a proper non-empty subset of core,
    # it means for any subset S of {1..n}, if S contains some but not all core,
    # then there must be an edge crossing the boundary of S.
    # This is equivalent to saying that in the graph of purchased edges,
    # all core nodes must be in the same connected component.
    # If all core nodes are in the same connected component, then any autonomous
    # set containing one core node must contain all nodes in that component,
    # which includes all core nodes.
    
    # So we need to connect all core nodes together with minimum cost.
    # This is the Steiner Tree problem, but since we only have 10 core nodes,
    # we can use the Dreyfus-Wagner algorithm or simply realize that for 
    # small t, we can use dynamic programming on subsets of core nodes.
    
    # dp[mask][i] = min cost to connect the subset of core nodes 'mask' 
    # such that node 'i' is in the resulting component.
    
    inf = float('inf')
    dp = [[inf] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][core[i]] = 0
        
    for mask in range(1, 1 << t):
        # Relax edges
        for _ in range(n):
            changed = False
            for u, v, w in edges:
                for i in range(n):
                    if dp[mask][u] + w < dp[mask][v]:
                        dp[mask][v] = dp[mask][u] + w
                        changed = True
                    if dp[mask][v] + w < dp[mask][u]:
                        dp[mask][u] = dp[mask][v] + w
                        changed = True
            if not changed: break
            
        # Combine masks
        for submask in range(1, mask):
            if (submask & mask) == submask:
                for i in range(n):
                    dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
                    
    print(min(dp[(1 << t) - 1]))

solve()
```
