```python
import sys
import heapq

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); t = int(next(it))
    core = [int(next(it)) - 1 for _ in range(t)]
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))

    if t == 1:
        print(0)
        return

    INF = 10**18
    full = (1 << t) - 1
    dp = [[INF] * n for _ in range(1 << t)]

    # base masks: shortest paths from each terminal
    for i, src in enumerate(core):
        mask = 1 << i
        dist = [INF] * n
        dist[src] = 0
        pq = [(0, src)]
        while pq:
            d, v = heapq.heappop(pq)
            if d != dist[v]:
                continue
            for to, w in adj[v]:
                nd = d + w
                if nd < dist[to]:
                    dist[to] = nd
                    heapq.heappush(pq, (nd, to))
        for v in range(n):
            dp[mask][v] = dist[v]

    # DP over subsets
    for mask in range(1, full + 1):
        # skip singletons (already done)
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            # combine at same vertex
            for v in range(n):
                val = dp[sub][v] + dp[other][v]
                if val < dp[mask][v]:
                    dp[mask][v] = val
            sub = (sub - 1) & mask

        # run Dijkstra to propagate through edges
        dist = dp[mask][:]
        pq = [(dist[v], v) for v in range(n) if dist[v] < INF]
        heapq.heapify(pq)
        while pq:
            d, v = heapq.heappop(pq)
            if d != dist[v]:
                continue
            for to, w in adj[v]:
                nd = d + w
                if nd < dist[to]:
                    dist[to] = nd
                    heapq.heappush(pq, (nd, to))
        dp[mask] = dist

    ans = min(dp[full])
    print(ans)

if __name__ == "__main__":
    solve()
```
