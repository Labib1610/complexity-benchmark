```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    focal = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    # The condition "span contains e_i - e_j for all focal i, j"
    # is equivalent to saying that all focal nodes must be in the same
    # connected component in the graph formed by the chosen edges.
    # This is a Steiner Tree problem on a graph with n nodes,
    # where we want to connect the set of focal nodes.
    # Since t is small (up to 10), we can use the Dreyfus-Wagner algorithm.
    
    inf = float('inf')
    dist = [[inf] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
            
    # Floyd-Warshall for all-pairs shortest paths
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    # dp[mask][i] is the min cost to connect the subset of focal nodes
    # represented by mask, with node i being part of the connected structure.
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][focal[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dp[mask][i] using submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Update dp[mask][i] using shortest paths
        for i in range(1, n + 1):
            if dp[mask][i] == inf: continue
            for j in range(1, n + 1):
                if dp[mask][j] > dp[mask][i] + dist[i][j]:
                    dp[mask][j] = dp[mask][i] + dist[i][j]
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
