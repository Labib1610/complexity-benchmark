We can model the expressions as nodes in a graph, where each subroutine is an edge with a cost.  
To make all target expressions equivalent, we need to connect them into one connected component using the minimum total edge cost.  
This is exactly the **Steiner tree** problem, but since the number of targets is small (≤ 10), we can solve it with a bitmask DP over the target nodes.

We’ll:
1. Compute all-pairs shortest paths (Floyd–Warshall) since n ≤ 100.
2. Use DP over subsets of target nodes:  
   `dp[mask][v]` = minimum cost to connect the targets in `mask` with the tree rooted at node `v`.
3. Combine subsets and relax using shortest paths.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    targets = [int(next(it)) - 1 for _ in range(t)]
    
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            if dist[i][k] == INF:
                continue
            for j in range(n):
                if dist[k][j] == INF:
                    continue
                nd = dist[i][k] + dist[k][j]
                if nd < dist[i][j]:
                    dist[i][j] = nd
    
    # DP over subsets of targets
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]
    
    # Initialize: single target at its own node
    for i, node in enumerate(targets):
        dp[1 << i][node] = 0
    
    # For each mask, try to combine two submasks
    for mask in range(1, full_mask):
        # Combine submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for v in range(n):
                    if dp[sub][v] == INF or dp[other][v] == INF:
                        continue
                    val = dp[sub][v] + dp[other][v]
                    if val < dp[mask][v]:
                        dp[mask][v] = val
            sub = (sub - 1) & mask
        
        # Relax via shortest paths
        for v in range(n):
            if dp[mask][v] == INF:
                continue
            for u in range(n):
                if dist[v][u] == INF:
                    continue
                nd = dp[mask][v] + dist[v][u]
                if nd < dp[mask][u]:
                    dp[mask][u] = nd
    
    ans = min(dp[full_mask - 1][v] for v in range(n))
    print(ans)

if __name__ == "__main__":
    solve()
```
