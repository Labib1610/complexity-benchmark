```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    s_indices = [int(x) for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The condition "subspace spanned by T contains e_i - e_j for all i, j in S"
    # is equivalent to saying that all vertices in S must be in the same connected
    # component of the graph formed by the edges in T.
    # This is a Steiner Tree problem variant: finding the Minimum Spanning Forest
    # that connects all vertices in S. Since we only care about connecting S,
    # we can contract all nodes not in S or treat it as a Steiner Tree problem.
    # Given t is very small (t <= 10), we can use the Dreyfus-Wagner algorithm
    # or simply realize this is a Minimum Spanning Tree problem on a graph
    # where we want to connect a subset of vertices.
    # Actually, for t <= 10, we can use dynamic programming on subsets.
    
    # Let dp[mask][i] be the min cost to connect the subset of S represented by mask,
    # with node i being in the connected component.
    
    inf = float('inf')
    # dp[mask][i]
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][s_indices[i]] = 0
        
    for mask in range(1, 1 << t):
        # Relax edges
        for _ in range(n):
            changed = False
            for u, v, w in edges:
                for i in range(1, n + 1):
                    if dp[mask][u] + w < dp[mask][v]:
                        dp[mask][v] = dp[mask][u] + w
                        changed = True
                    if dp[mask][v] + w < dp[mask][u]:
                        dp[mask][u] = dp[mask][v] + w
                        changed = True
            if not changed:
                break
        
        # Combine masks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
    ans = min(dp[(1 << t) - 1])
    print(ans)

solve()
```
