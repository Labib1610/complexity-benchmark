```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]
    INF = 10**18
    # distance matrix
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            ik = di[k]
            if ik == INF:
                continue
            for j in range(n):
                nd = ik + dk[j]
                if nd < di[j]:
                    di[j] = nd

    if t <= 1:
        print(0)
        return

    full = 1 << t
    dp = [[INF] * n for _ in range(full)]

    # initialize single terminals
    for idx, node in enumerate(terminals):
        mask = 1 << idx
        for v in range(n):
            dp[mask][v] = dist[node][v]

    # DP over subsets
    for mask in range(1, full):
        # skip singletons
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            # combine at same root v
            for v in range(n):
                val = dp[sub][v] + dp[other][v]
                if val < dp[mask][v]:
                    dp[mask][v] = val
            sub = (sub - 1) & mask
        # after combine, propagate via shortest paths
        # dp[mask][v] = min_w dp[mask][w] + dist[w][v]
        nd = [INF] * n
        for w in range(n):
            dw = dp[mask][w]
            if dw == INF:
                continue
            drow = dist[w]
            for v in range(n):
                cand = dw + drow[v]
                if cand < nd[v]:
                    nd[v] = cand
        dp[mask] = nd

    ans = min(dp[full - 1])
    print(ans)

if __name__ == "__main__":
    solve()
```
