We can model this as a minimum spanning tree problem on the target set, where the cost between two target nodes is the shortest path distance in the original graph (with edge weights as given).  

Here’s the code:  

```python
import sys
import heapq

input = sys.stdin.readline

n, m, t = map(int, input().split())
targets = list(map(int, input().split()))
target_set = set(targets)

# Build graph
graph = [[] for _ in range(n + 1)]
for _ in range(m):
    u, v, w = map(int, input().split())
    graph[u].append((v, w))
    graph[v].append((u, w))

# If only one target, cost is 0
if t == 1:
    print(0)
    sys.exit()

# Dijkstra from each target to get pairwise distances
INF = 10**18
dist_matrix = {}
for s in targets:
    dist = [INF] * (n + 1)
    dist[s] = 0
    pq = [(0, s)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(pq, (nd, v))
    dist_matrix[s] = dist

# Build complete graph among targets with shortest path costs
edges = []
for i in range(t):
    for j in range(i + 1, t):
        u = targets[i]
        v = targets[j]
        cost = dist_matrix[u][v]
        edges.append((cost, u, v))

# Kruskal's MST
parent = {x: x for x in targets}
rank = {x: 0 for x in targets}

def find(x):
    while parent[x] != x:
        parent[x] = parent[parent[x]]
        x = parent[x]
    return x

def union(x, y):
    rx, ry = find(x), find(y)
    if rx == ry:
        return False
    if rank[rx] < rank[ry]:
        parent[rx] = ry
    elif rank[rx] > rank[ry]:
        parent[ry] = rx
    else:
        parent[ry] = rx
        rank[rx] += 1
    return True

edges.sort()
total_cost = 0
for cost, u, v in edges:
    if union(u, v):
        total_cost += cost

print(total_cost)
```
