import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    targets = [int(next(it)) for _ in range(t)]
    
    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    for k in range(1, n + 1):
        dk = dist[k]
        for i in range(1, n + 1):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(1, n + 1):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    if t == 1:
        print(0)
        return
    
    K = 1 << t
    dp = [[INF] * (n + 1) for _ in range(K)]
    for i in range(t):
        dp[1 << i][targets[i]] = 0
    
    for mask in range(K):
        dpm = dp[mask]
        sub = (mask - 1) & mask
        while sub > 0:
            comp = mask ^ sub
            dps = dp[sub]
            dpc = dp[comp]
            for v in range(1, n + 1):
                if dps[v] < INF:
                    for u in range(1, n + 1):
                        nd = dps[v] + dpc[u] + dist[u][v]
                        if nd < dpm[v]:
                            dpm[v] = nd
            sub = (sub - 1) & mask
        
        best = INF
        for v in range(1, n + 1):
            if dpm[v] < best:
                best = dpm[v]
        for v in range(1, n + 1):
            if best < dpm[v]:
                dpm[v] = best
    
    print(dp[K - 1][1])

solve()
