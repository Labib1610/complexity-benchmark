import sys
import heapq
from collections import defaultdict

def dijkstra(graph, start, nodes):
    dist = {node: float('inf') for node in nodes}
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
            
        for v, w in graph[u]:
            if v in nodes and dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
                
    return dist

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    critical = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
    
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Compute shortest paths between all critical nodes
    # We'll create a complete graph on critical nodes with edge weights as shortest paths
    critical_graph = []
    
    for i, start_node in enumerate(critical):
        distances = dijkstra(graph, start_node, set(critical))
        for j, end_node in enumerate(critical):
            if i != j:
                critical_graph.append((start_node, end_node, distances[end_node]))
    
    # Now find MST of this complete graph on critical nodes
    # Using Kruskal's algorithm
    def find(parent, x):
        if parent[x] != x:
            parent[x] = find(parent, parent[x])
        return parent[x]
    
    def union(parent, rank, x, y):
        px, py = find(parent, x), find(parent, y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    # Sort edges by weight
    edges = sorted(critical_graph, key=lambda x: x[2])
    
    parent = {node: node for node in critical}
    rank = {node: 0 for node in critical}
    
    total_cost = 0
    edges_used = 0
    
    for u, v, w in edges:
        if union(parent, rank, u, v):
            total_cost += w
            edges_used += 1
            if edges_used == t - 1:
                break
                
    print(total_cost)

solve()
