```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core = [int(x) for x in input_data[3:3+t]]
    
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # The condition "if at least one core proposition is true, then all t core 
    # propositions must be true" is equivalent to saying that for any two 
    # core propositions c_i and c_j, they must be in the same connected 
    # component in the graph formed by the chosen axioms.
    # This is because if they are in the same component, they are equivalent.
    # If they are not, there exists a theory where c_i is true and c_j is false.
    
    # We need to select a subset of edges such that all core nodes are in the 
    # same connected component. This is the Steiner Tree problem for the 
    # set of core nodes. Since t is small (up to 10), we can use the 
    # Dreyfus-Wagner algorithm or a simpler approach for small t.
    
    # For t <= 10, we can use dynamic programming on subsets of core nodes.
    # dp[mask][i] = min cost to connect the subset of core nodes represented 
    # by 'mask', with node 'i' being in the resulting component.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][core[i]] = 0
        
    for mask in range(1, 1 << t):
        # Update dp[mask][i] using Steiner points
        for i in range(1, n + 1):
            submask = (mask - 1) & mask
            while submask > 0:
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
                submask = (submask - 1) & mask
        
        # Dijkstra to update dp[mask][i] using edges
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        adj = [[] for _ in range(n + 1)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
            
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][i] for i in range(1, n + 1))
    print(ans)

solve()
```
