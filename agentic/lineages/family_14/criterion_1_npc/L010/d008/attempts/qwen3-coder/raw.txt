import sys
from itertools import combinations
from collections import defaultdict
import heapq

def dijkstra(graph, start, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
            
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
                
    return dist

def solve():
    n, m, t = map(int, input().split())
    centers = list(map(int, input().split()))
    
    if t <= 1:
        print(0)
        return
    
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, input().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Compute shortest paths from each center
    distances = {}
    for center in centers:
        distances[center] = dijkstra(graph, center, n)
    
    # For each node, compute the cost to connect it to all centers
    min_cost = float('inf')
    
    for node in range(1, n + 1):
        cost = 0
        for center in centers:
            cost += distances[center][node]
        min_cost = min(min_cost, cost)
    
    print(min_cost)

solve()
