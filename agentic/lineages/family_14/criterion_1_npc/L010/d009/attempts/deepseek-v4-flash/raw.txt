We need to connect all critical chambers so that they end up in the same connected component, regardless of initial pressures.  
This is essentially a **Steiner tree** problem, but with a small number of terminals (t ≤ 10), so we can solve it with **DP over subsets** and **shortest paths**.

We’ll:
1. Compute all-pairs shortest paths (Floyd–Warshall) since n ≤ 100.
2. Use DP: `dp[mask][v]` = minimum cost to connect the terminals in `mask` with the component ending at node `v`.
3. Initialize with single terminals.
4. Combine subsets and relax via shortest paths.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    critical = [int(next(it)) - 1 for _ in range(t)]
    
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            if dist[i][k] == INF:
                continue
            for j in range(n):
                if dist[k][j] == INF:
                    continue
                nd = dist[i][k] + dist[k][j]
                if nd < dist[i][j]:
                    dist[i][j] = nd
    
    # DP over subsets
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]
    
    for i, node in enumerate(critical):
        dp[1 << i][node] = 0
    
    for mask in range(full_mask):
        # Combine subsets
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for v in range(n):
                    if dp[sub][v] == INF:
                        continue
                    for u in range(n):
                        if dp[other][u] == INF:
                            continue
                        nd = dp[sub][v] + dp[other][u] + dist[v][u]
                        if nd < dp[mask][v]:
                            dp[mask][v] = nd
            sub = (sub - 1) & mask
        
        # Relax via shortest paths
        for v in range(n):
            if dp[mask][v] == INF:
                continue
            for u in range(n):
                if dist[v][u] == INF:
                    continue
                nd = dp[mask][v] + dist[v][u]
                if nd < dp[mask][u]:
                    dp[mask][u] = nd
    
    ans = min(dp[full_mask - 1])
    print(ans)

if __name__ == "__main__":
    solve()
```
