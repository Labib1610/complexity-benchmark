```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    n = int(input_data[idx])
    m = int(input_data[idx+1])
    k = int(input_data[idx+2])
    idx += 3
    
    master_vaults = []
    for _ in range(k):
        master_vaults.append(int(input_data[idx]))
        idx += 1
        
    if k <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Steiner Tree problem: find min cost to connect k terminals in a graph.
    # Since k <= 10, we can use the Dreyfus-Wagner algorithm.
    # dp[mask][i] is the min cost to connect the subset of terminals 'mask'
    # such that vault 'i' is in the connected component.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << k)]
    
    for i in range(k):
        dp[1 << i][master_vaults[i]] = 0
        
    for mask in range(1, 1 << k):
        # Update dp[mask][i] using submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Dijkstra to update dp[mask][i] for all i
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << k) - 1][i] for i in range(1, n + 1))
    print(ans)

if __name__ == '__main__':
    solve()
```
