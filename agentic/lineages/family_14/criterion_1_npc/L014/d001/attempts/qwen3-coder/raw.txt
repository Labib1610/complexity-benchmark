import sys
import heapq
from collections import defaultdict

def find_mst_with_terminals(n, edges, terminals):
    if len(terminals) <= 1:
        return 0
    
    # Use Prim's algorithm to find the minimum Steiner tree
    # We'll use a modified version that only cares about connecting terminals
    
    # Build adjacency list
    adj = defaultdict(list)
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Convert terminals to a set for fast lookup
    terminal_set = set(terminals)
    
    # We'll use a modified Prim's algorithm
    # Start with one terminal
    start = terminals[0]
    visited = set()
    visited.add(start)
    
    # Priority queue: (weight, node)
    pq = []
    
    # Add all edges from the starting node
    for neighbor, weight in adj[start]:
        heapq.heappush(pq, (weight, neighbor))
    
    total_cost = 0
    terminals_connected = 1  # We've connected the first terminal to itself (cost 0)
    
    while pq and terminals_connected < len(terminals):
        weight, node = heapq.heappop(pq)
        
        if node in visited:
            continue
            
        visited.add(node)
        total_cost += weight
        
        # If this node is a terminal, we've connected one more terminal
        if node in terminal_set:
            terminals_connected += 1
        
        # Add all edges from this node
        for neighbor, edge_weight in adj[node]:
            if neighbor not in visited:
                heapq.heappush(pq, (edge_weight, neighbor))
    
    return total_cost

def find_min_cost_to_connect_terminals(n, edges, terminals):
    if len(terminals) <= 1:
        return 0
    
    # Use a different approach: create a complete graph between terminals
    # where edge weights are shortest paths in the original graph
    terminal_indices = {term: i for i, term in enumerate(terminals)}
    
    # Floyd-Warshall to find all-pairs shortest paths
    INF = float('inf')
    dist = [[INF] * n for _ in range(n)]
    
    # Initialize distances
    for i in range(n):
        dist[i][i] = 0
    
    for u, v, w in edges:
        dist[u-1][v-1] = min(dist[u-1][v-1], w)
        dist[v-1][u-1] = min(dist[v-1][u-1], w)
    
    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    # Create a complete graph of terminals
    terminal_edges = []
    for i in range(len(terminals)):
        for j in range(i+1, len(terminals)):
            u = terminals[i]
            v = terminals[j]
            w = dist[u-1][v-1]
            terminal_edges.append((i, j, w))
    
    # Find MST of this complete graph using Kruskal's algorithm
    def find(parent, x):
        if parent[x] != x:
            parent[x] = find(parent, parent[x])
        return parent[x]
    
    def union(parent, rank, x, y):
        px, py = find(parent, x), find(parent, y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    # Sort edges by weight
    terminal_edges.sort(key=lambda x: x[2])
    
    parent = list(range(len(terminals)))
    rank = [0] * len(terminals)
    
    total_cost = 0
    edges_used = 0
    
    for u, v, w in terminal_edges:
        if union(parent, rank, u, v):
            total_cost += w
            edges_used += 1
            if edges_used == len(terminals) - 1:
                break
    
    return total_cost

# Read input
line = sys.stdin.readline().split()
n, m, t = int(line[0]), int(line[1]), int(line[2])

terminals = list(map(int, sys.stdin.readline().split()))

edges = []
for _ in range(m):
    line = sys.stdin.readline().split()
    u, v, w = int(line[0]), int(line[1]), int(line[2])
    edges.append((u, v, w))

# Solve and output
result = find_min_cost_to_connect_terminals(n, edges, terminals)
print(result)
