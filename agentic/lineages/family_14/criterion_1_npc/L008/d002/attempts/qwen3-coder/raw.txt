import sys
import heapq
from collections import defaultdict

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def main():
    n, m, t = map(int, sys.stdin.readline().split())
    primaries = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    edges.sort()
    
    # We need to connect all primary turbines to the same component
    # We can use a modified Kruskal's algorithm where we track which primary turbines are connected
    
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # Group primaries by their initial component (initially each is its own component)
    primary_groups = {p: p for p in primaries}
    
    total_cost = 0
    
    for w, u, v in edges:
        # Check if this edge connects two different components
        pu, pv = find(parent, u), find(parent, v)
        if pu != pv:
            # This edge will merge two components
            # Check how many primary groups this affects
            groups_to_merge = set()
            for p in primaries:
                group_id = find(parent, p)
                if group_id == pu or group_id == pv:
                    groups_to_merge.add(primary_groups[p])
            
            # Perform the union
            union(parent, rank, u, v)
            
            # If this edge connected two groups that had primaries, we need to merge their group IDs
            if len(groups_to_merge) > 1:
                total_cost += w
                # Pick one group ID to be the new group ID for all
                new_group_id = min(groups_to_merge)
                for p in primaries:
                    if find(parent, p) == find(parent, u) or find(parent, p) == find(parent, v):
                        primary_groups[p] = new_group_id
            
            # Check if all primaries are now in the same group
            final_groups = set()
            for p in primaries:
                final_groups.add(primary_groups[p])
            
            if len(final_groups) == 1:
                break
    
    # Let me re-think this approach - it's simpler to think of it as connecting components
    
    # Better approach: 
    # Use Union-Find and keep track of how many primary turbines are in each component
    # When we unite two components, we check if we've connected all primaries
    
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    primary_count = [0] * (n + 1)
    
    for p in primaries:
        primary_count[p] = 1
        
    total_cost = 0
    primaries_connected = 1  # Initially each primary is in its own component
    
    for w, u, v in edges:
        pu = find(parent, u)
        pv = find(parent, v)
        
        if pu != pv:
            # Count how many primaries would be connected by this edge
            primaries_before = primary_count[pu] + primary_count[pv]
            
            # Merge the components
            union(parent, rank, u, v)
            
            # Update the primary count in the new root
            new_root = find(parent, u)
            primary_count[new_root] = primaries_before
            
            if primaries_before > 0:  # This edge connects at least one primary
                total_cost += w
                
            # Check if all primaries are now connected
            if primary_count[new_root] == t:
                break
    
    print(total_cost)

if __name__ == "__main__":
    main()
