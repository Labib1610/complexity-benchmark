```python
import sys

def solve():
    data = sys.stdin.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    terminals = []
    for _ in range(t):
        terminals.append(int(data[idx]) - 1)
        idx += 1
    edges = []
    for _ in range(m):
        u = int(data[idx]) - 1; idx += 1
        v = int(data[idx]) - 1; idx += 1
        w = int(data[idx]); idx += 1
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    INF = float('inf')
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    K = t
    full = (1 << K) - 1
    dp = [[INF] * n for _ in range(1 << K)]
    for i, term in enumerate(terminals):
        dp[1 << i][term] = 0
    
    for mask in range(1, 1 << K):
        sub = (mask - 1) & mask
        while sub > 0:
            other = mask ^ sub
            for v in range(n):
                a = dp[sub][v]
                if a == INF:
                    continue
                for u in range(n):
                    b = dp[other][u]
                    if b == INF:
                        continue
                    d = dist[v][u]
                    if d == INF:
                        continue
                    val = a + b + d
                    if val < dp[mask][v]:
                        dp[mask][v] = val
            sub = (sub - 1) & mask
        
        for v in range(n):
            if dp[mask][v] == INF:
                continue
            for u in range(n):
                d = dist[v][u]
                if d == INF:
                    continue
                val = dp[mask][v] + d
                if val < dp[mask][u]:
                    dp[mask][u] = val
    
    ans = min(dp[full])
    print(ans)

solve()
```
