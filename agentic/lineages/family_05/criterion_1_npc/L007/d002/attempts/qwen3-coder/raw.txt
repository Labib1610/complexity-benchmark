from collections import defaultdict
import sys

def solve():
    k, t = map(int, sys.stdin.readline().split())
    
    # If k is 0, special case
    if k == 0:
        print("YES")
        return
    
    triples = []
    for _ in range(t):
        c, n, i = map(int, sys.stdin.readline().split())
        triples.append((c, n, i))
    
    # We need to find a perfect matching where each chassis, nav, and image is used exactly once
    # This is essentially finding a perfect matching in a 3-dimensional matching problem
    
    # Try all possible subsets of k triples
    # But this is too slow for k=22, so we need a smarter approach
    
    # Let's model this as a bipartite matching problem
    # But actually, it's a 3D matching which is NP-hard in general
    # However, since k <= 22, we might be able to use backtracking with pruning
    
    # Build mappings for faster lookup
    chassis_to_triples = defaultdict(list)
    nav_to_triples = defaultdict(list)
    image_to_triples = defaultdict(list)
    
    for idx, (c, n, i) in enumerate(triples):
        chassis_to_triples[c].append(idx)
        nav_to_triples[n].append(idx)
        image_to_triples[i].append(idx)
    
    # Try to build a solution using backtracking
    used_triples = [False] * t
    selected = []
    used_chassis = [False] * (k + 1)
    used_nav = [False] * (k + 1)
    used_image = [False] * (k + 1)
    
    def backtrack(pos):
        if pos == k:
            return True
            
        # Try each triple
        for i in range(t):
            if used_triples[i]:
                continue
                
            c, n, i_val = triples[i]
            
            # Check if this triple can be used
            if used_chassis[c] or used_nav[n] or used_image[i_val]:
                continue
                
            # Use this triple
            used_triples[i] = True
            used_chassis[c] = True
            used_nav[n] = True
            used_image[i_val] = True
            selected.append((c, n, i_val))
            
            # Recurse
            if backtrack(pos + 1):
                return True
                
            # Backtrack
            used_triples[i] = False
            used_chassis[c] = False
            used_nav[n] = False
            used_image[i_val] = False
            selected.pop()
            
        return False
    
    # Optimization: start with chassis that have fewest options
    # This helps prune the search space early
    
    if backtrack(0):
        print("YES")
        for c, n, i in selected:
            print(c, n, i)
    else:
        print("NO")

solve()
