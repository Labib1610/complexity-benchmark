import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Store triples as (c, n, i)
    triples = []
    idx = 2
    for _ in range(t):
        c = int(input_data[idx]) - 1
        n = int(input_data[idx+1]) - 1
        i = int(input_data[idx+2]) - 1
        triples.append((c, n, i))
        idx += 3
        
    # Group triples by chassis
    by_chassis = [[] for _ in range(k)]
    for c, n, i in triples:
        by_chassis[c].append((n, i))
        
    # To avoid the "prohibitive number of dead-end paths" mentioned in the note,
    # we use a heuristic: process chassis with fewer options first (MRV).
    # Also, keep track of used navigation modules and imaging sensors.
    
    used_n = [False] * k
    used_i = [False] * k
    result = []
    
    # Sort chassis by number of available options
    order = sorted(range(k), key=lambda x: len(by_chassis[x]))
    
    def backtrack(idx):
        if idx == k:
            return True
        
        c = order[idx]
        for n, i in by_chassis[c]:
            if not used_n[n] and not used_i[i]:
                used_n[n] = True
                used_i[i] = True
                result.append((c + 1, n + 1, i + 1))
                
                if backtrack(idx + 1):
                    return True
                
                result.pop()
                used_n[n] = False
                used_i[i] = False
        return False

    if backtrack(0):
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
