import sys
from itertools import combinations

def solve():
    k, t = map(int, sys.stdin.readline().split())
    configs = []
    for _ in range(t):
        p1, p2, p3 = map(int, sys.stdin.readline().split())
        configs.append((p1, p2, p3))
    
    # Try all combinations of k configurations
    for combo in combinations(range(t), k):
        # Check if this combination uses each value 1..k exactly once for each parameter
        p1_values = set()
        p2_values = set()
        p3_values = set()
        
        selected_configs = []
        for i in combo:
            p1, p2, p3 = configs[i]
            p1_values.add(p1)
            p2_values.add(p2)
            p3_values.add(p3)
            selected_configs.append((p1, p2, p3))
        
        # Check if we have exactly k distinct values for each parameter
        if (len(p1_values) == k and len(p2_values) == k and len(p3_values) == k and
            all(1 <= v <= k for v in p1_values) and
            all(1 <= v <= k for v in p2_values) and
            all(1 <= v <= k for v in p3_values)):
            
            # Verify that the sets are exactly {1, 2, ..., k}
            if (p1_values == set(range(1, k+1)) and 
                p2_values == set(range(1, k+1)) and 
                p3_values == set(range(1, k+1))):
                print("YES")
                for config in selected_configs:
                    print(config[0], config[1], config[2])
                return
    
    print("NO")

solve()
