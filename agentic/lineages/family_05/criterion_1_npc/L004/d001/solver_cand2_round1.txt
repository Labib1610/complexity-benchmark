import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    sites = []
    idx = 2
    for _ in range(t):
        lon = int(input_data[idx]) - 1
        lat = int(input_data[idx+1]) - 1
        depth = int(input_data[idx+2]) - 1
        sites.append((lon, lat, depth))
        idx += 3

    # We need to select k sites such that each lon, lat, depth is used exactly once.
    # This is equivalent to finding a perfect matching in a 3D bipartite-like structure,
    # or more simply, a 3D matching problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    # Group sites by lon
    by_lon = [[] for _ in range(k)]
    for s in sites:
        by_lon[s[0]].append(s)
        
    used_lat = [False] * k
    used_depth = [False] * k
    result = []

    def backtrack(lon_idx):
        if lon_idx == k:
            return True
        
        for s in by_lon[lon_idx]:
            lat, depth = s[1], s[2]
            if not used_lat[lat] and not used_depth[depth]:
                used_lat[lat] = True
                used_depth[depth] = True
                result.append(s)
                
                if backtrack(lon_idx + 1):
                    return True
                
                result.pop()
                used_lat[lat] = False
                used_depth[depth] = False
        return False

    # Sort by number of options to prune search space
    if backtrack(0):
        print("YES")
        for s in result:
            print(f"{s[0]+1} {s[1]+1} {s[2]+1}")
    else:
        print("NO")

solve()
