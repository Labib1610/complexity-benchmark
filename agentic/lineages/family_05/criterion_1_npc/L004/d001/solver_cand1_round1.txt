import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    adj = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx+1]) - 1
        c = int(input_data[idx+2]) - 1
        adj.append((a, b, c))
        idx += 3
        
    # We need to select k triples such that each ID 0..k-1 
    # in each department appears exactly once.
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    
    used_a = [False] * k
    used_b = [False] * k
    used_c = [False] * k
    result = []
    
    def backtrack(idx_t):
        if len(result) == k:
            return True
        
        # Optimization: find the first unused Alpha employee
        first_a = -1
        for i in range(k):
            if not used_a[i]:
                first_a = i
                break
        
        if first_a == -1:
            return True
            
        # Try all combinations involving first_a
        for i in range(len(adj)):
            a, b, c = adj[i]
            if a == first_a and not used_b[b] and not used_c[c]:
                used_a[a] = used_b[b] = used_c[c] = True
                result.append((a + 1, b + 1, c + 1))
                
                if backtrack(i + 1):
                    return True
                
                result.pop()
                used_a[a] = used_b[b] = used_c[c] = False
        return False

    # Pre-filter combinations to speed up
    # Group by Alpha ID
    by_a = [[] for _ in range(k)]
    for a, b, c in adj:
        by_a[a].append((b, c))
        
    def backtrack_fast(a_idx):
        if a_idx == k:
            return True
        
        for b, c in by_a[a_idx]:
            if not used_b[b] and not used_c[c]:
                used_b[b] = used_c[c] = True
                result.append((a_idx + 1, b + 1, c + 1))
                if backtrack_fast(a_idx + 1):
                    return True
                result.pop()
                used_b[b] = used_c[c] = False
        return False

    if backtrack_fast(0):
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

solve()
