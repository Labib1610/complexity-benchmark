import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        p1 = int(input_data[idx]) - 1
        p2 = int(input_data[idx+1]) - 1
        p3 = int(input_data[idx+2]) - 1
        configs.append((p1, p2, p3))
        idx += 3

    # We need to select k configurations such that each p1, p2, p3 is used exactly once.
    # This is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph.
    # Since k <= 22, we can use backtracking with pruning.
    
    # Group configs by p1
    by_p1 = [[] for _ in range(k)]
    for c in configs:
        by_p1[c[0]].append(c)
        
    used_p2 = [False] * k
    used_p3 = [False] * k
    result = []

    def backtrack(p1_idx):
        if p1_idx == k:
            return True
        
        for c in by_p1[p1_idx]:
            _, p2, p3 = c
            if not used_p2[p2] and not used_p3[p3]:
                used_p2[p2] = True
                used_p3[p3] = True
                result.append(c)
                
                if backtrack(p1_idx + 1):
                    return True
                
                result.pop()
                used_p2[p2] = False
                used_p3[p3] = False
        return False

    if backtrack(0):
        print("YES")
        for c in result:
            print(f"{c[0]+1} {c[1]+1} {c[2]+1}")
    else:
        print("NO")

solve()
