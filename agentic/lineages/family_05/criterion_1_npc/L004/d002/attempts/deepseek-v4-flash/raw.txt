We can model this as a matching problem. Since k ≤ 22, we can use bitmask DP to check if we can cover all values in all three dimensions with exactly k configurations.

We’ll use a DP over masks for the first two dimensions, and check if the third can be covered.

```python
k, t = map(int, input().split())
configs = []
for _ in range(t):
    a, b, c = map(int, input().split())
    configs.append((a - 1, b - 1, c - 1))

full_mask = (1 << k) - 1

# dp[mask1][mask2] = mask3 achievable
dp = [[-1] * (1 << k) for _ in range(1 << k)]
dp[0][0] = 0

# For reconstruction
parent = {}

for a, b, c in configs:
    bit1 = 1 << a
    bit2 = 1 << b
    bit3 = 1 << c
    for m1 in range(1 << k):
        if m1 & bit1:
            continue
        for m2 in range(1 << k):
            if m2 & bit2:
                continue
            m3 = dp[m1][m2]
            if m3 == -1:
                continue
            if m3 & bit3:
                continue
            nm1 = m1 | bit1
            nm2 = m2 | bit2
            nm3 = m3 | bit3
            if dp[nm1][nm2] == -1 or (nm3 | dp[nm1][nm2]) != dp[nm1][nm2]:
                # store better or first found
                if dp[nm1][nm2] == -1 or bin(nm3).count('1') > bin(dp[nm1][nm2]).count('1'):
                    dp[nm1][nm2] = nm3
                    parent[(nm1, nm2)] = (m1, m2, (a, b, c))

if dp[full_mask][full_mask] == full_mask:
    print("YES")
    # reconstruct
    m1, m2 = full_mask, full_mask
    chosen = []
    while (m1, m2) != (0, 0):
        pm1, pm2, cfg = parent[(m1, m2)]
        chosen.append(cfg)
        m1, m2 = pm1, pm2
    for a, b, c in chosen:
        print(a + 1, b + 1, c + 1)
else:
    print("NO")
```
