import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    points = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        points.append((x, y, z))
        idx += 3
        
    # The condition "no zero components" means for any two points (x1, y1, z1) 
    # and (x2, y2, z2), x1 != x2, y1 != y2, and z1 != z2.
    # This implies that if we select k points, the set of x-coordinates must be 
    # a permutation of {1, ..., k}, the set of y-coordinates must be a 
    # permutation of {1, ..., k}, and the set of z-coordinates must be a 
    # permutation of {1, ..., k}.
    
    # This is equivalent to finding a system of distinct representatives or 
    # a matching in a hypergraph, but since k is small (<= 22), we can use 
    # backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    # Group points by x to speed up search
    points_by_x = [[] for _ in range(k + 1)]
    for p in points:
        points_by_x[p[0]].append(p)
        
    result = []
    
    def backtrack(x_idx):
        if x_idx > k:
            return True
        
        for p in points_by_x[x_idx]:
            _, y, z = p
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(p)
                
                if backtrack(x_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
