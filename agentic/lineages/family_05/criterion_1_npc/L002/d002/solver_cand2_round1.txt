import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # We need to select k triples (u_i, v_i, w_i) such that
    # {u_1, ..., u_k} = {1, ..., k}
    # {v_1, ..., v_k} = {1, ..., k}
    # {w_1, ..., w_k} = {1, ..., k}
    # This is equivalent to finding a transversal in a 3D Latin Square problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    sculptures = []
    idx = 2
    for _ in range(t):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2]) - 1
        sculptures.append((u, v, w))
        idx += 3
        
    # Group by material u
    by_u = [[] for _ in range(k)]
    for s in sculptures:
        by_u[s[0]].append(s)
        
    used_v = [False] * k
    used_w = [False] * k
    result = []
    
    def backtrack(u_idx):
        if u_idx == k:
            return True
        
        for s in by_u[u_idx]:
            v, w = s[1], s[2]
            if not used_v[v] and not used_w[w]:
                used_v[v] = True
                used_w[w] = True
                result.append(s)
                
                if backtrack(u_idx + 1):
                    return True
                
                result.pop()
                used_v[v] = False
                used_w[w] = False
        return False

    # Optimization: sort by number of options to fail fast
    by_u.sort(key=len)
    
    if backtrack(0):
        print("YES")
        for s in result:
            print(f"{s[0]+1} {s[1]+1} {s[2]+1}")
    else:
        print("NO")

solve()
