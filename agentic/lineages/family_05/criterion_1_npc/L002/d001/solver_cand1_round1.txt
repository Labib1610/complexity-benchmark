import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        configs.append((x, y, z))
        idx += 3
        
    # We need to select k configurations (x_i, y_i, z_i) such that
    # all x_i are distinct, all y_i are distinct, and all z_i are distinct.
    # Since 1 <= x, y, z <= k, this means {x_i} = {1..k}, {y_i} = {1..k}, {z_i} = {1..k}.
    # This is equivalent to finding a transversal in a 3D Latin Square or 
    # solving an Exact Cover problem.
    # Given k <= 22, we can use backtracking with pruning.
    
    # Group configs by x
    by_x = [[] for _ in range(k + 1)]
    for c in configs:
        by_x[c[0]].append(c)
        
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    result = []
    
    def backtrack(x):
        if x > k:
            return True
        
        for c in by_x[x]:
            _, y, z = c
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(c)
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for c in result:
            print(f"{c[0]} {c[1]} {c[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
