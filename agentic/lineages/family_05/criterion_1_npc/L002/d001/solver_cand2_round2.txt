```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    pathways = []
    idx = 2
    for _ in range(t):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2]) - 1
        pathways.append((u, v, w))
        idx += 3
        
    # We need to select k pathways (u_i, v_i, w_i) such that
    # {u_i} = {0..k-1}, {v_i} = {0..k-1}, {w_i} = {0..k-1}
    # This is equivalent to finding a perfect matching in a 3D matching problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    # Group pathways by u
    adj = [[] for _ in range(k)]
    for u, v, w in pathways:
        adj[u].append((v, w))
        
    used_v = [False] * k
    used_w = [False] * k
    result = []
    
    def backtrack(u):
        if u == k:
            return True
        
        for v, w in adj[u]:
            if not used_v[v] and not used_w[w]:
                used_v[v] = True
                used_w[w] = True
                result.append((u + 1, v + 1, w + 1))
                
                if backtrack(u + 1):
                    return True
                
                result.pop()
                used_v[v] = False
                used_w[w] = False
        return False

    if backtrack(0):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
