```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    pathways = []
    idx = 2
    for _ in range(t):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2]) - 1
        pathways.append((u, v, w))
        idx += 3

    # We need to select k pathways such that:
    # - Each input port 0..k-1 is used exactly once.
    # - Each output port 0..k-1 is used exactly once.
    # - Each frequency 0..k-1 is used exactly once.
    # This is equivalent to finding a transversal in a 3D k x k x k binary tensor
    # where we select k cells (u, v, w) such that no two share a coordinate.
    
    # Since k <= 22, we can use backtracking with pruning.
    # Sort pathways to improve search.
    pathways.sort()
    
    used_u = [False] * k
    used_v = [False] * k
    used_w = [False] * k
    result = []

    def backtrack(idx, count):
        if count == k:
            return True
        if idx == t or (k - count) > (t - idx):
            return False
        
        # Try including pathways[idx]
        u, v, w = pathways[idx]
        if not used_u[u] and not used_v[v] and not used_w[w]:
            used_u[u] = used_v[v] = used_w[w] = True
            result.append((u + 1, v + 1, w + 1))
            if backtrack(idx + 1, count + 1):
                return True
            result.pop()
            used_u[u] = used_v[v] = used_w[w] = False
            
        # Try excluding pathways[idx]
        if backtrack(idx + 1, count):
            return True
            
        return False

    if backtrack(0, 0):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
```
