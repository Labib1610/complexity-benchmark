import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    configs = []
    idx = 2
    for _ in range(t):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2]) - 1
        configs.append((u, v, w))
        idx += 3

    # We need to select k configurations such that each column has 0..k-1 exactly once.
    # This is equivalent to finding a set of k triples (u_i, v_i, w_i) such that
    # {u_0, ..., u_{k-1}} = {0, ..., k-1}, {v_0, ..., v_{k-1}} = {0, ..., k-1},
    # and {w_0, ..., w_{k-1}} = {0, ..., k-1}.
    
    # Since k <= 22, we can use backtracking with pruning.
    # To optimize, group configs by u.
    by_u = [[] for _ in range(k)]
    for c in configs:
        by_u[c[0]].append(c)
        
    used_v = [False] * k
    used_w = [False] * k
    result = []

    def backtrack(u_idx):
        if u_idx == k:
            return True
        
        for c in by_u[u_idx]:
            v, w = c[1], c[2]
            if not used_v[v] and not used_w[w]:
                used_v[v] = True
                used_w[w] = True
                result.append(c)
                if backtrack(u_idx + 1):
                    return True
                result.pop()
                used_v[v] = False
                used_w[w] = False
        return False

    if backtrack(0):
        print("YES")
        for c in result:
            print(f"{c[0]+1} {c[1]+1} {c[2]+1}")
    else:
        print("NO")

solve()
