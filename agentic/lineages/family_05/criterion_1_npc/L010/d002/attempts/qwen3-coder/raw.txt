from itertools import permutations

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
        
    configs = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        configs.append((x, y, z))
    
    # Try all possible combinations of k configurations
    from itertools import combinations
    
    for combo in combinations(range(t), k):
        selected = [configs[i] for i in combo]
        
        # Check if all technicians, hardware units, and shifts are used exactly once
        techs = [config[0] for config in selected]
        hardwares = [config[1] for config in selected]
        shifts = [config[2] for config in selected]
        
        if (len(set(techs)) == k and len(set(hardwares)) == k and 
            len(set(shifts)) == k and len(techs) == k):
            print("YES")
            for config in selected:
                print(config[0], config[1], config[2])
            return
    
    print("NO")

solve()
