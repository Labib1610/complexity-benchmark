import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3

    # Group triples by supervisor
    by_sup = [[] for _ in range(k)]
    for x, y, z in triples:
        by_sup[x].append((y, z))

    # State: (supervisor_index, used_facilities_mask, used_shifts_mask)
    # Since k <= 22, we use bitmasks.
    # To optimize, we use backtracking with pruning.
    
    memo = {}

    def backtrack(sup_idx, f_mask, s_mask):
        if sup_idx == k:
            return []
        
        state = (sup_idx, f_mask, s_mask)
        if state in memo:
            return None
        
        for f, s in by_sup[sup_idx]:
            if not (f_mask & (1 << f)) and not (s_mask & (1 << s)):
                res = backtrack(sup_idx + 1, f_mask | (1 << f), s_mask | (1 << s))
                if res is not None:
                    return [(sup_idx + 1, f + 1, s + 1)] + res
        
        memo[state] = False
        return None

    # Heuristic: Sort supervisors by number of available options
    # But since we must use each supervisor exactly once, we just process 0 to k-1.
    # To speed up, we can reorder supervisors by the number of available triples.
    
    result = backtrack(0, 0, 0)
    
    if result is not None:
        print("YES")
        for unit in result:
            print(f"{unit[0]} {unit[1]} {unit[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
