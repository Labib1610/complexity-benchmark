from itertools import combinations
import sys

def determinant(matrix):
    n = len(matrix)
    if n == 1:
        return matrix[0][0]
    
    det = 0
    for col in range(n):
        # Create submatrix
        submatrix = []
        for i in range(1, n):
            row = []
            for j in range(n):
                if j != col:
                    row.append(matrix[i][j])
            submatrix.append(row)
        
        sign = (-1) ** col
        det += sign * matrix[0][col] * determinant(submatrix)
    
    return det

def is_valid_selection(relays, k):
    # Check primary channel matrix
    primary_matrix = [[min(relay[0], j+1) for j in range(k)] for relay in relays]
    if determinant(primary_matrix) == 0:
        return False
    
    # Check secondary channel matrix
    secondary_matrix = [[min(relay[1], j+1) for j in range(k)] for relay in relays]
    if determinant(secondary_matrix) == 0:
        return False
    
    # Check tertiary channel matrix
    tertiary_matrix = [[min(relay[2], j+1) for j in range(k)] for relay in relays]
    if determinant(tertiary_matrix) == 0:
        return False
    
    return True

def solve():
    line = input().split()
    k, t = int(line[0]), int(line[1])
    
    if t == 0:
        if k == 0:
            print("YES")
        else:
            print("NO")
        return
    
    relays = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        relays.append((a, b, c))
    
    if k == 0:
        print("YES")
        return
    
    # Try all combinations of k relays
    for combo in combinations(relays, k):
        if is_valid_selection(combo, k):
            print("YES")
            for relay in combo:
                print(relay[0], relay[1], relay[2])
            return
    
    print("NO")

solve()
