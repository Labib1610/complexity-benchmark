import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    triples = []
    by_a = [[] for _ in range(k)]  # 0-indexed a-1
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        c = int(next(it)) - 1
        triples.append((a, b, c))
        by_a[a].append(len(triples) - 1)

    # quick impossibility check: each value 0..k-1 must appear at least once in each dimension
    if any(len(by_a[i]) == 0 for i in range(k)):
        print("NO")
        return
    # similarly for b and c
    cnt_b = [0] * k
    cnt_c = [0] * k
    for a, b, c in triples:
        cnt_b[b] += 1
        cnt_c[c] += 1
    if any(v == 0 for v in cnt_b) or any(v == 0 for v in cnt_c):
        print("NO")
        return

    used_a = 0
    used_b = 0
    used_c = 0
    selected = []

    # precompute bit for each index
    bit = [1 << i for i in range(k)]

    # order of a's to try: we will pick the unused a with fewest candidate rows that are still viable
    def dfs():
        nonlocal used_a, used_b, used_c
        if len(selected) == k:
            return True
        # choose an unused a with minimal viable options
        best_a = -1
        best_options = None
        for a in range(k):
            if used_a & bit[a]:
                continue
            opts = []
            for idx in by_a[a]:
                _, b, c = triples[idx]
                if (used_b & bit[b]) or (used_c & bit[c]):
                    continue
                opts.append(idx)
            if not opts:
                return False
            if best_options is None or len(opts) < len(best_options):
                best_a = a
                best_options = opts
                if len(best_options) == 1:
                    break
        # try each option
        for idx in best_options:
            a, b, c = triples[idx]
            # mark
            used_a ^= bit[a]
            used_b ^= bit[b]
            used_c ^= bit[c]
            selected.append(idx)
            if dfs():
                return True
            # undo
            selected.pop()
            used_a ^= bit[a]
            used_b ^= bit[b]
            used_c ^= bit[c]
        return False

    if dfs():
        print("YES")
        for idx in selected:
            a, b, c = triples[idx]
            print(a + 1, b + 1, c + 1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
