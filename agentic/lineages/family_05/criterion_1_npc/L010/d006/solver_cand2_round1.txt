import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    relays = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        relays.append((a, b, c))
        idx += 3
        
    # The condition is sum(2^a_i) = sum(2^j) for j=1..k,
    # sum(2^b_i) = sum(2^j) for j=1..k,
    # sum(2^c_i) = sum(2^j) for j=1..k.
    # Since 1 <= a, b, c <= k, this is equivalent to saying that
    # the multiset of {a_i} must be {1, 2, ..., k},
    # the multiset of {b_i} must be {1, 2, ..., k},
    # and the multiset of {c_i} must be {1, 2, ..., k}.
    
    # This is an exact cover problem. Since k is small (up to 22),
    # we can use backtracking with pruning.
    
    target = (1 << k) - 1
    
    # State: (index in relays, mask_a, mask_b, mask_c, count_selected)
    # Using bitmasks to track which values 1..k have been used for a, b, c.
    
    chosen = []
    
    def backtrack(idx, mask_a, mask_b, mask_c, count):
        if count == k:
            return mask_a == target and mask_b == target and mask_c == target
        
        if idx == t:
            return False
        
        # Pruning: if remaining relays are not enough to fill the masks
        if count + (t - idx) < k:
            return False
            
        # Try including relays[idx]
        a, b, c = relays[idx]
        bit_a = 1 << (a - 1)
        bit_b = 1 << (b - 1)
        bit_c = 1 << (c - 1)
        
        if not (mask_a & bit_a) and not (mask_b & bit_b) and not (mask_c & bit_c):
            chosen.append(relays[idx])
            if backtrack(idx + 1, mask_a | bit_a, mask_b | bit_b, mask_c | bit_c, count + 1):
                return True
            chosen.pop()
            
        # Try excluding relays[idx]
        if backtrack(idx + 1, mask_a, mask_b, mask_c, count):
            return True
            
        return False

    if backtrack(0, 0, 0, 0, 0):
        print("YES")
        for r in chosen:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
