import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    relays = []
    idx = 2
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx+1]); c = int(data[idx+2])
        relays.append((a, b, c))
        idx += 3
    
    if k == 1:
        if t >= 1:
            print("YES")
            print(f"{relays[0][0]} {relays[0][1]} {relays[0][2]}")
        else:
            print("NO")
        return
    
    # Build adjacency: for each (a,b), list of c values
    from collections import defaultdict
    ab_to_c = defaultdict(list)
    for (a, b, c) in relays:
        ab_to_c[(a, b)].append(c)
    
    # For each a, list of (b, c) pairs
    a_to_bc = defaultdict(list)
    for (a, b), cs in ab_to_c.items():
        for c in cs:
            a_to_bc[a].append((b, c))
    
    A_list = sorted(a_to_bc.keys())
    
    # For each a, build b -> set of c
    a_b_c = {}
    for a in A_list:
        b_c = defaultdict(set)
        for (b, c) in a_to_bc[a]:
            b_c[b].add(c)
        a_b_c[a] = b_c
    
    # Try all permutations of A
    from itertools import permutations
    
    for A_perm in permutations(A_list):
        if len(A_perm) < k:
            continue
        
        # Greedy: for each a in order, pick b not used, c not used
        used_b = set()
        used_c = set()
        chosen = []
        ok = True
        
        for i in range(k):
            a = A_perm[i]
            b_c = a_b_c[a]
            found = False
            # Try b in sorted order for determinism
            for b in sorted(b_c.keys()):
                if b in used_b:
                    continue
                cs = b_c[b]
                for c in cs:
                    if c not in used_c:
                        chosen.append((a, b, c))
                        used_b.add(b)
                        used_c.add(c)
                        found = True
                        break
                if found:
                    break
            if not found:
                ok = False
                break
        
        if ok:
            print("YES")
            for (a, b, c) in chosen:
                print(f"{a} {b} {c}")
            return
    
    print("NO")

solve()
