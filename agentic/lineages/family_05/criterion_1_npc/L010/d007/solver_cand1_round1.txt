import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    relays = []
    idx = 2
    for _ in range(t):
        relays.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    # The condition sum_{j=1}^3 sum_{v=1}^k (c_{j,v})^2 = 3k
    # Since sum_{v=1}^k c_{j,v} = k for each j, the sum of squares is at least k.
    # The minimum value of sum_{v=1}^k (c_{j,v})^2 is k (when all c_{j,v} = 1).
    # Thus, the total sum is at least 3k.
    # The condition sum = 3k is satisfied if and only if for each j,
    # sum_{v=1}^k (c_{j,v})^2 = k.
    # This happens if and only if c_{j,1} = c_{j,2} = ... = c_{j,k} = 1.
    # This means for each position j, every value from 1 to k must appear exactly once.
    # This is equivalent to saying the set of relays forms a Latin Square-like structure:
    # each column j is a permutation of {1, ..., k}.
    
    # Since k <= 22, we can use backtracking with pruning.
    
    chosen = []
    used_a = [False] * (k + 1)
    used_b = [False] * (k + 1)
    used_c = [False] * (k + 1)
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for i in range(start_idx, t):
            a, b, c = relays[i]
            if not used_a[a] and not used_b[b] and not used_c[c]:
                used_a[a] = used_b[b] = used_c[c] = True
                chosen.append(relays[i])
                if backtrack(i + 1, count + 1):
                    return True
                chosen.pop()
                used_a[a] = used_b[b] = used_c[c] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for r in chosen:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
