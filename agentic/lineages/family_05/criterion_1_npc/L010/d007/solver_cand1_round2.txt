import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    relays = []
    idx = 2
    for _ in range(t):
        relays.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    # The condition sum(c1(x)^2 + c2(x)^2 + c3(x)^2) = 3k
    # Since sum(c1(x)) = k, sum(c2(x)) = k, sum(c3(x)) = k,
    # the minimum value of sum(ci(x)^2) is k (when all ci(x) are 1).
    # Thus, the sum of squares is at least k + k + k = 3k.
    # The condition is satisfied if and only if c1(x) = 1, c2(x) = 1, c3(x) = 1 for all x in 1..k.
    # This means the set of chosen relays must form a Latin Square-like structure
    # where each coordinate 1..k appears exactly once in each position.
    
    # This is equivalent to finding a subset of size k such that
    # every value 1..k appears exactly once in the first column,
    # exactly once in the second column, and exactly once in the third column.
    
    # This is a variation of the Exact Cover problem.
    # We need to pick k relays such that:
    # 1. Each a in {1..k} appears exactly once.
    # 2. Each b in {1..k} appears exactly once.
    # 3. Each c in {1..k} appears exactly once.
    
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    chosen = []
    used_a = [False] * (k + 1)
    used_b = [False] * (k + 1)
    used_c = [False] * (k + 1)
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for i in range(start_idx, t):
            a, b, c = relays[i]
            if not used_a[a] and not used_b[b] and not used_c[c]:
                used_a[a] = used_b[b] = used_c[c] = True
                chosen.append(relays[i])
                if backtrack(i + 1, count + 1):
                    return True
                chosen.pop()
                used_a[a] = used_b[b] = used_c[c] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for r in chosen:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
