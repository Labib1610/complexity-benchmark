import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    relays = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        relays.append((a, b, c))
        idx += 3
        
    # The condition "product > 0" means for any two chosen relays (a1, b1, c1) and (a2, b2, c2),
    # a1 != a2, b1 != b2, and c1 != c2.
    # This is equivalent to finding a subset of size k such that all a's are distinct,
    # all b's are distinct, and all c's are distinct.
    # Since coordinates are in [1, k], this means the set of a's must be {1, ..., k},
    # the set of b's must be {1, ..., k}, and the set of c's must be {1, ..., k}.
    # This is exactly the problem of finding a transversal in a 3D Latin Square or
    # a matching in a 3-partite hypergraph.
    
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    used_a = [False] * (k + 1)
    used_b = [False] * (k + 1)
    used_c = [False] * (k + 1)
    
    chosen = []
    
    def backtrack(idx):
        if len(chosen) == k:
            return True
        
        if idx >= t:
            return False
        
        # Try including relays[idx]
        a, b, c = relays[idx]
        if not used_a[a] and not used_b[b] and not used_c[c]:
            used_a[a] = used_b[b] = used_c[c] = True
            chosen.append(relays[idx])
            if backtrack(idx + 1):
                return True
            chosen.pop()
            used_a[a] = used_b[b] = used_c[c] = False
            
        # Try skipping relays[idx]
        # Pruning: if remaining relays are not enough to fill the rest
        if (t - (idx + 1)) >= (k - len(chosen)):
            if backtrack(idx + 1):
                return True
        
        return False

    if backtrack(0):
        print("YES")
        for r in chosen:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
