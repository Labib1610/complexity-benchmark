from itertools import combinations

def read_input():
    k, t = map(int, input().split())
    relays = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        relays.append((a, b, c))
    return k, t, relays

def is_valid_selection(relays):
    n = len(relays)
    if n <= 1:
        return True
    
    for i in range(n):
        for j in range(i+1, n):
            a_i, b_i, c_i = relays[i]
            a_j, b_j, c_j = relays[j]
            if a_i == a_j or b_i == b_j or c_i == c_j:
                return False
    return True

def solve():
    k, t, relays = read_input()
    
    if k > t:
        print("NO")
        return
        
    # Try all combinations of k relays
    for combo in combinations(relays, k):
        if is_valid_selection(combo):
            print("YES")
            for relay in combo:
                print(relay[0], relay[1], relay[2])
            return
    
    print("NO")

solve()
