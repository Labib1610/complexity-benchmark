import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    candidates = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        candidates.append((a, b, c))
        idx += 3

    # The condition "total stabilized length is k" means that for each dimension,
    # the set of chosen stabilizers must cover all k unit intervals [0,1], [1,2], ..., [k-1,k].
    # This is equivalent to saying that for each dimension, the set of values {a_i}, {b_i}, {c_i}
    # must contain every integer from 1 to k at least once.
    
    # Since k is small (up to 22), we can use backtracking with pruning.
    # We need to pick k stabilizers such that the set of a's is {1..k}, b's is {1..k}, c's is {1..k}.
    
    n = len(candidates)
    
    def backtrack(idx, count, mask_a, mask_b, mask_c, chosen):
        if count == k:
            if mask_a == (1 << k) - 1 and mask_b == (1 << k) - 1 and mask_c == (1 << k) - 1:
                return chosen
            return None
        
        if idx == n or (k - count) > (n - idx):
            return None
        
        # Try including candidates[idx]
        a, b, c = candidates[idx]
        res = backtrack(idx + 1, count + 1, mask_a | (1 << (a - 1)), mask_b | (1 << (b - 1)), mask_c | (1 << (c - 1)), chosen + [candidates[idx]])
        if res is not None:
            return res
        
        # Try excluding candidates[idx]
        res = backtrack(idx + 1, count, mask_a, mask_b, mask_c, chosen)
        if res is not None:
            return res
        
        return None

    # Optimization: Sort candidates to try to fill masks faster
    result = backtrack(0, 0, 0, 0, 0, [])
    
    if result:
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
