import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    templates = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        templates.append((a, b, c))
        idx += 3
        
    # The condition is sum(2^(x-1)) = 2^k - 1.
    # This is equivalent to saying that in binary representation, 
    # the sum of 2^(x-1) must have bits 0 to k-1 all set to 1.
    # Since each template contributes 2^(a-1), 2^(b-1), 2^(c-1),
    # we can represent each template as a triplet of bitmasks:
    # (1 << (a-1), 1 << (b-1), 1 << (c-1)).
    # We need to choose k templates such that the sum of masks in each 
    # dimension is (1 << k) - 1.
    
    target = (1 << k) - 1
    
    # Meet-in-the-middle approach
    # Split t templates into two halves
    mid = t // 2
    left = templates[:mid]
    right = templates[mid:]
    
    def get_sums(arr):
        # sums[maskA][maskB][maskC] = list of indices
        # Since k <= 22, we can't use a 3D array.
        # Use a dictionary to store reachable states.
        # state: (sumA, sumB, sumC) -> list of templates
        res = {}
        n = len(arr)
        for i in range(1 << n):
            sa, sb, sc = 0, 0, 0
            count = 0
            subset = []
            for j in range(n):
                if (i >> j) & 1:
                    a, b, c = arr[j]
                    sa += (1 << (a - 1))
                    sb += (1 << (b - 1))
                    sc += (1 << (c - 1))
                    count += 1
                    subset.append(arr[j])
            if count <= k:
                res[(sa, sb, sc, count)] = subset
        return res

    left_sums = get_sums(left)
    right_sums = get_sums(right)
    
    for (sa, sb, sc, count), subset_l in left_sums.items():
        needed_count = k - count
        needed_a = target - sa
        needed_b = target - sb
        needed_c = target - sc
        
        if (needed_a, needed_b, needed_c, needed_count) in right_sums:
            print("YES")
            for item in subset_l:
                print(f"{item[0]} {item[1]} {item[2]}")
            for item in right_sums[(needed_a, needed_b, needed_c, needed_count)]:
                print(f"{item[0]} {item[1]} {item[2]}")
            return

    print("NO")

solve()
