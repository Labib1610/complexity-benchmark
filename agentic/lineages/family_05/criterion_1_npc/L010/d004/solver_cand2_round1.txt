import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    protocols = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        protocols.append((x, y, z))
        idx += 3

    # The condition c1 + c2 + c3 >= 3k with c1, c2, c3 <= k
    # implies c1 = k, c2 = k, c3 = k.
    # This means we need to pick k protocols such that all x are distinct,
    # all y are distinct, and all z are distinct.
    # This is equivalent to finding a transversal of size k in a 3D matching problem,
    # or more simply, finding a permutation p and q such that (i, p[i], q[i])
    # are all in the set of protocols for i in 0..k-1.

    # Since k is small (up to 22), we can use backtracking with pruning.
    adj = [[] for _ in range(k)]
    for x, y, z in protocols:
        adj[x].append((y, z))

    used_y = [False] * k
    used_z = [False] * k
    res = []

    def backtrack(x):
        if x == k:
            return True
        for y, z in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                res.append((x + 1, y + 1, z + 1))
                if backtrack(x + 1):
                    return True
                res.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for p in res:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
