from collections import defaultdict

def solve():
    k, t = map(int, input().split())
    
    if k == 0:
        print("YES")
        return
    
    triplets = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        triplets.append((x, y, z))
    
    # Try to find a perfect matching using backtracking with pruning
    # We'll use a recursive approach with bitmasks for efficiency
    
    # Preprocess: for each specialist, store which triplets they appear in
    spec_to_triplets = defaultdict(list)
    mach_to_triplets = defaultdict(list)
    time_to_triplets = defaultdict(list)
    
    for i, (x, y, z) in enumerate(triplets):
        spec_to_triplets[x].append(i)
        mach_to_triplets[y].append(i)
        time_to_triplets[z].append(i)
    
    # We'll use backtracking to find a valid assignment
    # State: which specialists, machines, and time intervals are used
    used_spec = [False] * (k + 1)
    used_mach = [False] * (k + 1)
    used_time = [False] * (k + 1)
    selected = []
    
    def backtrack():
        if len(selected) == k:
            return True
            
        # Find the next unused specialist (smallest index for deterministic behavior)
        next_spec = -1
        for i in range(1, k + 1):
            if not used_spec[i]:
                next_spec = i
                break
        
        if next_spec == -1:
            return False
            
        # Try each triplet with this specialist
        for triplet_idx in spec_to_triplets[next_spec]:
            x, y, z = triplets[triplet_idx]
            if not used_spec[x] and not used_mach[y] and not used_time[z]:
                # Use this triplet
                used_spec[x] = True
                used_mach[y] = True
                used_time[z] = True
                selected.append(triplet_idx)
                
                if backtrack():
                    return True
                    
                # Backtrack
                used_spec[x] = False
                used_mach[y] = False
                used_time[z] = False
                selected.pop()
        
        return False
    
    # Optimization: check if it's even possible
    # Each resource must appear in at least one triplet
    for i in range(1, k + 1):
        if not spec_to_triplets[i] or not mach_to_triplets[i] or not time_to_triplets[i]:
            print("NO")
            return
    
    if backtrack():
        print("YES")
        for idx in selected:
            x, y, z = triplets[idx]
            print(x, y, z)
    else:
        print("NO")

solve()
