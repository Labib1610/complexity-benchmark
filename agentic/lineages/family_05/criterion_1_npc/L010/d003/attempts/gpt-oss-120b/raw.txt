import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    rows = []          # list of (mask, (x,y,z))
    col_rows = [[] for _ in range(3 * k)]   # indices of rows covering each column

    for idx in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        mask = (1 << x) | (1 << (k + y)) | (1 << (2 * k + z))
        rows.append((mask, (x + 1, y + 1, z + 1)))
        col_rows[x].append(idx)
        col_rows[k + y].append(idx)
        col_rows[2 * k + z].append(idx)

    full_mask = (1 << (3 * k)) - 1
    sys.setrecursionlimit(10000)

    from functools import lru_cache

    @lru_cache(maxsize=None)
    def dfs(covered_mask):
        if covered_mask == full_mask:
            return ()
        # choose uncovered column with fewest compatible rows
        uncovered = (~covered_mask) & full_mask
        best_col = -1
        best_cnt = 10**9
        bits = uncovered
        while bits:
            b = bits & -bits
            col = b.bit_length() - 1
            cnt = 0
            for r in col_rows[col]:
                if rows[r][0] & covered_mask == 0:
                    cnt += 1
            if cnt == 0:
                return None
            if cnt < best_cnt:
                best_cnt = cnt
                best_col = col
                if cnt == 1:
                    break
            bits -= b

        for r in col_rows[best_col]:
            mask, triple = rows[r]
            if mask & covered_mask:
                continue
            nxt = dfs(covered_mask | mask)
            if nxt is not None:
                return (triple,) + nxt
        return None

    result = dfs(0)
    out = []
    if result is None:
        out.append("NO")
    else:
        out.append("YES")
        for triple in result:
            out.append(f"{triple[0]} {triple[1]} {triple[2]}")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
