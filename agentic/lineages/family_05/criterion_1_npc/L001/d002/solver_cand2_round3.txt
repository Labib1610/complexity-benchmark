```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triplets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triplets.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking
    # with pruning or Dancing Links. Given the constraints and the nature of 
    # the problem, a recursive backtracking approach with bitmasks is efficient.
    
    # State: (index in triplets, mask_a, mask_b, mask_c, current_roster)
    # Since we need to pick exactly k triplets, we can use recursion.
    
    chosen = []
    
    def backtrack(start_idx, mask_a, mask_b, mask_c):
        if len(chosen) == k:
            return True
        
        for i in range(start_idx, t):
            x, y, z = triplets[i]
            
            bit_a = 1 << x
            bit_b = 1 << y
            bit_c = 1 << z
            
            if not (mask_a & bit_a) and not (mask_b & bit_b) and not (mask_c & bit_c):
                chosen.append(triplets[i])
                if backtrack(i + 1, mask_a | bit_a, mask_b | bit_b, mask_c | bit_c):
                    return True
                chosen.pop()
        return False

    # Optimization: If k is small, simple backtracking works.
    # For k up to 22, we need to be careful.
    # Let's sort triplets to potentially find a solution faster.
    if backtrack(0, 0, 0, 0):
        print("YES")
        for tpl in chosen:
            print(f"{tpl[0]+1} {tpl[1]+1} {tpl[2]+1}")
    else:
        print("NO")

solve()
```
