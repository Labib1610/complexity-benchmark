from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    combinations = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        combinations.append((x, y, z))
        index += 3
    
    # Try to find a matching of size k using backtracking
    # This is essentially finding a maximum matching in a tripartite 3-uniform hypergraph
    # We can model this as a bipartite matching problem by considering operators on one side 
    # and (hardware, phase) pairs on the other side, but since we need to match each dimension exactly once,
    # it's a 3D matching problem.
    
    # We'll use backtracking with pruning
    def backtrack(pos, used_ops, used_hw, used_phase, current_match):
        if len(current_match) == k:
            return True, current_match[:]
        
        if pos == t:
            return False, []
            
        # Try not taking this combination
        found, result = backtrack(pos + 1, used_ops, used_hw, used_phase, current_match)
        if found:
            return True, result
            
        # Try taking this combination if possible
        op, hw, phase = combinations[pos]
        if op not in used_ops and hw not in used_hw and phase not in used_phase:
            used_ops.add(op)
            used_hw.add(hw)
            used_phase.add(phase)
            current_match.append(combinations[pos])
            
            found, result = backtrack(pos + 1, used_ops, used_hw, used_phase, current_match)
            
            if found:
                return True, result
                
            # Backtrack
            used_ops.remove(op)
            used_hw.remove(hw)
            used_phase.remove(phase)
            current_match.pop()
            
        return False, []
    
    # Optimization: if any operator/hw/phase doesn't appear at all, impossible
    ops_used = set()
    hw_used = set()
    phase_used = set()
    
    for op, hw, phase in combinations:
        ops_used.add(op)
        hw_used.add(hw)
        phase_used.add(phase)
        
    if len(ops_used) < k or len(hw_used) < k or len(phase_used) < k:
        print("NO")
        return
        
    found, result = backtrack(0, set(), set(), set(), [])
    
    if found:
        print("YES")
        for combo in result:
            print(combo[0], combo[1], combo[2])
    else:
        print("NO")

solve()
