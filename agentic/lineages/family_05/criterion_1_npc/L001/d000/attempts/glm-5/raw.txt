import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    idx = 2
    triples = []
    for _ in range(t):
        x = int(data[idx]) - 1
        y = int(data[idx+1]) - 1
        z = int(data[idx+2]) - 1
        idx += 3
        triples.append((x, y, z))

    # For each (x, y), collect list of z's
    xy_to_z = {}
    for (x, y, z) in triples:
        key = x * k + y
        if key not in xy_to_z:
            xy_to_z[key] = []
        xy_to_z[key].append(z)

    # For each x, list of y's that have at least one z
    x_to_y = [[] for _ in range(k)]
    for (x, y, z) in triples:
        x_to_y[x].append(y)

    # Remove duplicates and sort for determinism
    for x in range(k):
        x_to_y[x] = sorted(set(x_to_y[x]))

    # We need a perfect matching in bipartite graph (X-Y) where edge (x,y) exists
    # if there's at least one z. Then for matched (x,y), pick any z.
    # But we also need z's to be distinct! So it's a 3D matching.
    # 
    # Actually we need distinct z's too. So it's a 3-dimensional matching.
    # Let's do: match X to Y (perfect matching), then check if we can assign distinct z's.
    # But that's not sufficient because z assignment might fail.
    # 
    # Better: think of it as matching X to (Y,Z) pairs where each (y,z) used at most once
    # and each y used exactly once, each z used exactly once.
    # 
    # Since k <= 22, we can use DP over subsets.
    # 
    # Let's reformulate: we need to select k triples, one per x, covering all y and all z.
    # For each x, we have a set of (y,z) pairs. We need to pick one (y,z) per x
    # such that all y's are distinct and all z's are distinct.
    # 
    # DP over x (0..k-1) and bitmask of used y's and used z's.
    # State: (x, maskY, maskZ) -> but that's 22 * 2^22 * 2^22 too large.
    # 
    # Alternative: DP over x with maskZ only, and track maskY implicitly?
    # No, we need both.
    # 
    # Let's try: DP[maskZ] = set of achievable maskY? Still too large.
    # 
    # Since k <= 22, maybe we can do DP over x with maskZ, and for each maskZ
    # store the maskY. But there could be multiple maskY per maskZ.
    # 
    # Actually, let's think differently. We process x from 0 to k-1.
    # At step i, we've assigned x=0..i-1. State = (maskY, maskZ) of used y's and z's.
    # Number of states at step i: C(k, i)^2 roughly. For k=22, i=11, that's ~ 7e5^2 = 5e11. Too large.
    # 
    # Let's try meet-in-the-middle or other approach.
    # 
    # Actually, let's just try a backtracking with pruning. k <= 22 but with good
    # pruning it might work. Let's order x by number of options (fewest first).
    # 
    # Alternatively, since k <= 22, we can use bitmask DP where we iterate over
    # x and track maskZ, and for each maskZ we track which maskY values are reachable.
    # But we can compress: for each maskZ, we only need one maskY (any valid one).
    # Wait, no—different maskY for same maskZ could lead to different outcomes.
    # 
    # Hmm, but actually if we fix the assignment of z's to x's (a permutation),
    # then we need a perfect matching in the bipartite graph where x connects to y
    # if (x, y, z_x) is allowed. So:
    # 
    # For each permutation z of 0..k-1 (assigning z[i] to x[i]), check if there's
    # a perfect matching between x and y using only edges (x, y, z[x]).
    # But k! permutations is too many for k=22.
    # 
    # Let's try the backtracking approach with good pruning.

    # Order x's by increasing number of (y,z) options
    x_options = [[] for _ in range(k)]
    for (x, y, z) in triples:
        x_options[x].append((y, z))

    # Sort x indices by number of options
    x_order = sorted(range(k), key=lambda x: len(x_options[x]))

    used_y = 0
    used_z = 0
    assignment = [None] * k  # assignment[x] = (y, z)

    def backtrack(pos):
        nonlocal used_y, used_z
        if pos == k:
            return True
        x = x_order[pos]
        for (y, z) in x_options[x]:
            yb = 1 << y
            zb = 1 << z
            if (used_y & yb) or (used_z & zb):
                continue
            used_y |= yb
            used_z |= zb
            assignment[x] = (y, z)
            if backtrack(pos + 1):
                return True
            used_y &= ~yb
            used_z &= ~zb
            assignment[x] = None
        return False

    if backtrack(0):
        out = ["YES"]
        for x in range(k):
            y, z = assignment[x]
            out.append(f"{x+1} {y+1} {z+1}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")

solve()
