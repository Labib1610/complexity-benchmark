import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        triples.append((x - 1, y - 1, z - 1))
        idx += 3
        
    # This is an Exact Cover problem. Since k <= 22, we use backtracking with pruning.
    # Represent the state as bitmasks for used Hardware and Mission phases.
    # We iterate through Operators 0 to k-1.
    
    adj = [[] for _ in range(k)]
    for x, y, z in triples:
        adj[x].append((y, z))
        
    roster = []
    used_y = 0
    used_z = 0
    
    def backtrack(op_idx):
        if op_idx == k:
            return True
        
        for y, z in adj[op_idx]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                nonlocal used_y, used_z
                used_y |= (1 << y)
                used_z |= (1 << z)
                roster.append((op_idx + 1, y + 1, z + 1))
                
                if backtrack(op_idx + 1):
                    return True
                
                roster.pop()
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    # Optimization: Sort operators by number of available choices
    # But for k=22, simple backtracking with bitmasks is usually sufficient.
    if backtrack(0):
        print("YES")
        for triple in roster:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

solve()
