import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem.
    # We have 3*k constraints:
    # 1. Each specialist used exactly once (k constraints)
    # 2. Each machine used exactly once (k constraints)
    # 3. Each time slot used exactly once (k constraints)
    # Total constraints: 3*k. Each triple covers 3 constraints.
    
    # Using Algorithm X (Dancing Links approach)
    # Since k <= 22, we can use a backtracking approach with bitmasks or simple sets.
    
    used_s = [False] * k
    used_m = [False] * k
    used_t = [False] * k
    schedule = []
    
    def backtrack(idx):
        if len(schedule) == k:
            return True
        
        if idx == len(triples):
            return False
        
        # Optimization: if remaining triples are not enough to fill k
        if len(schedule) + (len(triples) - idx) < k:
            return False
            
        # Try including triples[idx]
        s, m, t_val = triples[idx]
        if not used_s[s] and not used_m[m] and not used_t[t_val]:
            used_s[s] = used_m[m] = used_t[t_val] = True
            schedule.append(triples[idx])
            if backtrack(idx + 1):
                return True
            schedule.pop()
            used_s[s] = used_m[m] = used_t[t_val] = False
            
        # Try excluding triples[idx]
        if backtrack(idx + 1):
            return True
            
        return False

    # Heuristic: sort triples to potentially find solution faster
    # For k=22, simple backtracking might be slow, but constraints are tight.
    if backtrack(0):
        print("YES")
        for s, m, t_val in schedule:
            print(f"{s+1} {m+1} {t_val+1}")
    else:
        print("NO")

solve()
