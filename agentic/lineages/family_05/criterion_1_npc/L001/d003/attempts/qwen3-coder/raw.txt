from itertools import permutations

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
    
    groupings = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        groupings.append((x, y, z))
    
    # Try all permutations of k groupings from the list
    from itertools import combinations
    
    # For each combination of k groupings
    for combo in combinations(groupings, k):
        # Extract technicians, bays, and cores
        techs = [g[0] for g in combo]
        bays = [g[1] for g in combo]
        cores = [g[2] for g in combo]
        
        # Check if all are distinct
        if len(set(techs)) == k and len(set(bays)) == k and len(set(cores)) == k:
            print("YES")
            for g in combo:
                print(g[0], g[1], g[2])
            return
    
    print("NO")

solve()
