from collections import defaultdict

def solve():
    k, t = map(int, input().split())
    
    if k == 0:
        print("YES")
        return
    
    if t < k:
        print("NO")
        return
    
    configs = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        configs.append((x, y, z))
    
    # Try to find a valid assignment using backtracking
    # We need to assign each specialist exactly once
    # This is essentially finding a matching in a 3D bipartite graph
    # We can model this as finding a perfect matching where each specialist,
    # machine, and time slot is used exactly once
    
    def backtrack(pos, used_spec, used_mach, used_time):
        if pos == k:
            return []
        
        for i in range(t):
            x, y, z = configs[i]
            if x in used_spec or y in used_mach or z in used_time:
                continue
            
            used_spec.add(x)
            used_mach.add(y)
            used_time.add(z)
            
            result = backtrack(pos + 1, used_spec, used_mach, used_time)
            if result is not None:
                return [(x, y, z)] + result
                
            used_spec.remove(x)
            used_mach.remove(y)
            used_time.remove(z)
        
        return None
    
    # Optimization: pre-check if it's possible
    spec_count = [0] * (k + 1)
    mach_count = [0] * (k + 1)
    time_count = [0] * (k + 1)
    
    for x, y, z in configs:
        spec_count[x] += 1
        mach_count[y] += 1
        time_count[z] += 1
    
    # Each specialist, machine, and time slot must appear at least once
    for i in range(1, k + 1):
        if spec_count[i] == 0 or mach_count[i] == 0 or time_count[i] == 0:
            print("NO")
            return
    
    result = backtrack(0, set(), set(), set())
    
    if result is None:
        print("NO")
    else:
        print("YES")
        for x, y, z in result:
            print(x, y, z)

solve()
