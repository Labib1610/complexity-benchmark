import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    voxels = []
    idx = 2
    for _ in range(t):
        voxels.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    # The condition |X(V)| + |Y(V)| + |Z(V)| = 3k with |V| = k
    # implies |X(V)| = k, |Y(V)| = k, and |Z(V)| = k.
    # This means all x-coordinates must be distinct, all y-coordinates distinct,
    # and all z-coordinates distinct.
    # Since coordinates are in [1, k], this means the set of x-coords is {1..k},
    # the set of y-coords is {1..k}, and the set of z-coords is {1..k}.
    # This is equivalent to finding a transversal in a 3D grid, or a Latin Square
    # representation where each (x, y, z) is a cell.
    
    # Since k is small (up to 22), we can use backtracking with pruning.
    # We need to pick k voxels such that all x, y, z are distinct.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    selected = []
    
    def backtrack(idx):
        if len(selected) == k:
            return True
        if idx == t:
            return False
        
        # Try including voxels[idx]
        x, y, z = voxels[idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            selected.append(voxels[idx])
            if backtrack(idx + 1):
                return True
            selected.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try skipping voxels[idx]
        # Pruning: if remaining voxels are not enough to reach k
        if (t - 1 - idx) >= (k - len(selected)):
            if backtrack(idx + 1):
                return True
        
        return False

    if backtrack(0):
        print("YES")
        for v in selected:
            print(f"{v[0]} {v[1]} {v[2]}")
    else:
        print("NO")

solve()
