import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    models = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        models.append((a, b, c))
        idx += 3
        
    # The condition U_alpha + U_beta + U_gamma = 3k with k models
    # implies U_alpha = k, U_beta = k, U_gamma = k.
    # This means all k models must have distinct alpha, distinct beta, and distinct gamma.
    # This is equivalent to finding a transversal of size k in a 3D grid,
    # or specifically, a set of k points (a_i, b_i, c_i) such that all a_i are distinct,
    # all b_i are distinct, and all c_i are distinct.
    
    # Since k <= 22, we can use backtracking with pruning.
    
    used_a = [False] * (k + 1)
    used_b = [False] * (k + 1)
    used_c = [False] * (k + 1)
    
    res = []
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for i in range(start_idx, t):
            a, b, c = models[i]
            if not used_a[a] and not used_b[b] and not used_c[c]:
                used_a[a] = True
                used_b[b] = True
                used_c[c] = True
                res.append(models[i])
                
                if backtrack(i + 1, count + 1):
                    return True
                
                res.pop()
                used_a[a] = False
                used_b[b] = False
                used_c[c] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for m in res:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

solve()
