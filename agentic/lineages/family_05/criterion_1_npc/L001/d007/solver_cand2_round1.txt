```python
import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    models = []
    idx = 2
    for _ in range(t):
        models.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    if k > t:
        print("NO")
        return

    # The condition for a unique solution to the system:
    # U_1 * x_i^1 + U_2 * x_i^2 + ... + U_k * x_i^k = i
    # is that the matrix M where M_{ij} = x_i^j is invertible.
    # This is a Vandermonde-like matrix.
    # The determinant is product(x_i) * product_{1 <= i < j <= k} (x_j - x_i).
    # For this to be non-zero:
    # 1. All x_i must be non-zero (given as 1 to k, so this is satisfied).
    # 2. All x_i must be distinct.
    # Thus, the condition is simply that all k chosen coefficients must be distinct.
    
    # We need to pick k models (a_i, b_i, c_i) such that:
    # {a_1, ..., a_k} = {1, ..., k}
    # {b_1, ..., b_k} = {1, ..., k}
    # {c_1, ..., c_k} = {1, ..., k}
    
    # This is equivalent to finding a system of distinct representatives or 
    # a matching in a hypergraph, but since k is small (up to 22), 
    # we can use backtracking with pruning.
    
    def backtrack(idx, current_models, used_a, used_b, used_c):
        if len(current_models) == k:
            return current_models
        
        if idx == t:
            return None
        
        # Pruning: if remaining models are not enough to fill the sets
        if len(current_models) + (t - idx) < k:
            return None
            
        # Try including models[idx]
        a, b, c = models[idx]
        if not (used_a & (1 << (a - 1))) and not (used_b & (1 << (b - 1))) and not (used_c & (1 << (c - 1))):
            res = backtrack(idx + 1, current_models + [models[idx]], 
                            used_a | (1 << (a - 1)), 
                            used_b | (1 << (b - 1)), 
                            used_c | (1 << (c - 1)))
            if res:
                return res
        
        # Try excluding models[idx]
        return backtrack(idx + 1, current_models, used_a, used_b, used_c)

    # For k=22, simple backtracking might be slow. 
    # However, the constraints on a, b, c being a permutation of 1..k 
    # are very strong.
    result = backtrack(0, [], 0, 0, 0)
    
    if result:
        print("YES")
        for m in result:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

solve()
```
