import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    slots = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        i = int(input_data[idx+1])
        e = int(input_data[idx+2])
        slots.append((a, i, e))
        idx += 3
        
    # The condition V(S1, S2) > 0 means |a1-a2| > 0 AND |i1-i2| > 0 AND |e1-e2| > 0.
    # This implies that for any two selected slots, all three coordinates must be distinct.
    # This is equivalent to saying that in the set of k selected slots:
    # - All 'a' coordinates are distinct (must be a permutation of 1..k)
    # - All 'i' coordinates are distinct (must be a permutation of 1..k)
    # - All 'e' coordinates are distinct (must be a permutation of 1..k)
    # This is exactly the definition of a Latin Square of order k, 
    # but here we are selecting k points from a 3D grid such that no two share 
    # the same a, i, or e coordinate. This is equivalent to finding a 
    # transversal in a 3D Latin Cube, or more simply, a matching in a 
    # hypergraph or a constraint satisfaction problem.
    
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    used_a = [False] * (k + 1)
    used_i = [False] * (k + 1)
    used_e = [False] * (k + 1)
    
    result = []
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for j in range(start_idx, t):
            a, i, e = slots[j]
            if not used_a[a] and not used_i[i] and not used_e[e]:
                used_a[a] = True
                used_i[i] = True
                used_e[e] = True
                result.append(slots[j])
                
                if backtrack(j + 1, count + 1):
                    return True
                
                result.pop()
                used_a[a] = False
                used_i[i] = False
                used_e[e] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for s in result:
            print(f"{s[0]} {s[1]} {s[2]}")
    else:
        print("NO")

solve()
