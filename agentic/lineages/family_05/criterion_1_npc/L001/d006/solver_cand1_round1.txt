import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    models = []
    idx = 2
    for _ in range(t):
        models.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    # We need to select k models such that for any two models (f1, p1, a1) and (f2, p2, a2),
    # f1 != f2, p1 != p2, and a1 != a2.
    # This is equivalent to finding a clique of size k in a graph where nodes are models
    # and edges exist between isolated models.
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    adj = [[] for _ in range(t)]
    for i in range(t):
        for j in range(i + 1, t):
            m1 = models[i]
            m2 = models[j]
            if m1[0] != m2[0] and m1[1] != m2[1] and m1[2] != m2[2]:
                adj[i].append(j)
                adj[j].append(i)
                
    def backtrack(current_selection, candidates):
        if len(current_selection) == k:
            return current_selection
        
        if not candidates:
            return None
        
        # Heuristic: pick the candidate that has the most connections to other candidates
        # Or simply iterate
        for i in range(len(candidates)):
            node = candidates[i]
            new_candidates = [c for c in candidates[i+1:] if c in adj[node]]
            
            res = backtrack(current_selection + [node], new_candidates)
            if res:
                return res
        return None

    result = backtrack([], list(range(t)))
    
    if result:
        print("YES")
        for idx in result:
            print(f"{models[idx][0]} {models[idx][1]} {models[idx][2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
