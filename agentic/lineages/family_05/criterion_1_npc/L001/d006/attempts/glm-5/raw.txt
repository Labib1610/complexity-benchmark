```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    catalog = set()
    idx = 2
    for _ in range(t):
        f = int(input_data[idx])
        p = int(input_data[idx+1])
        a = int(input_data[idx+2])
        catalog.add((f, p, a))
        idx += 3
        
    if k == 1:
        if (1, 1, 1) in catalog:
            print("YES")
            print("1 1 1")
        else:
            print("NO")
        return

    # Try all k! permutations of (1..k) for the phase shift and amplitude scale
    # The frequency multiplier is always (1, 2, ..., k)
    base_perms = list(range(1, k + 1))
    
    # To avoid redundant checks, we can fix the first element of phase to 1
    # since any valid assignment can be re-indexed.
    # This reduces the search space from (k!)^2 to (k-1)! * k!
    
    # Generate all permutations for phase shift
    from itertools import permutations
    
    phase_perms = []
    for perm in permutations(base_perms):
        if perm[0] == 1:
            phase_perms.append(perm)
            
    amp_perms = list(permutations(base_perms))
    
    for phase in phase_perms:
        for amp in amp_perms:
            valid = True
            for i in range(k):
                if (i + 1, phase[i], amp[i]) not in catalog:
                    valid = False
                    break
            if valid:
                print("YES")
                for i in range(k):
                    print(f"{i+1} {phase[i]} {amp[i]}")
                return
                
    print("NO")

if __name__ == '__main__':
    solve()
```
