from itertools import combinations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    models = []
    idx = 2
    for i in range(t):
        f = int(data[idx])
        p = int(data[idx+1])
        a = int(data[idx+2])
        idx += 3
        models.append((f, p, a))
    
    # Try all combinations of k models
    for combo in combinations(models, k):
        valid = True
        # Check all pairs in this combination
        for i in range(k):
            for j in range(i+1, k):
                m1 = combo[i]
                m2 = combo[j]
                # Models must differ in all three parameters
                if m1[0] == m2[0] or m1[1] == m2[1] or m1[2] == m2[2]:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            print("YES")
            for model in combo:
                print(model[0], model[1], model[2])
            return
    
    print("NO")

solve()
