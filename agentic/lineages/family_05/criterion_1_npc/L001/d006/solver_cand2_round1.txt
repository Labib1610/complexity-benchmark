import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    candidates = []
    idx = 2
    for _ in range(t):
        candidates.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    def get_hamming(t1, t2):
        dist = 0
        if t1[0] != t2[0]: dist += 1
        if t1[1] != t2[1]: dist += 1
        if t1[2] != t2[2]: dist += 1
        return dist

    # Backtracking to find k templates with pairwise Hamming distance 3
    # Since k is small (up to 22), we can use a simple backtracking approach
    # with pruning.
    
    selected = []
    
    def backtrack(start_idx):
        if len(selected) == k:
            return True
        
        for i in range(start_idx, t):
            cand = candidates[i]
            
            # Check if cand is compatible with all currently selected
            possible = True
            for s in selected:
                if get_hamming(s, cand) != 3:
                    possible = False
                    break
            
            if possible:
                selected.append(cand)
                if backtrack(i + 1):
                    return True
                selected.pop()
        return False

    if backtrack(0):
        print("YES")
        for s in selected:
            print(f"{s[0]} {s[1]} {s[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
