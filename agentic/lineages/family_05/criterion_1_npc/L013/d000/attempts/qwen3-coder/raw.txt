from itertools import permutations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    combinations = []
    index = 2
    for _ in range(t):
        d = int(data[index])
        f = int(data[index+1])
        m = int(data[index+2])
        combinations.append((d, f, m))
        index += 3
    
    # We need to find a permutation of k demographics such that we can match 
    # each with a unique formulation and facility
    
    # Try all permutations of demographics (there are k! of them, max 22! which is too large)
    # Actually, let's think differently - this is a tripartite matching problem
    
    # Let's model this as finding a perfect matching in a tripartite hypergraph
    # But since k <= 22, and we're looking for a set of k triples from the t triples
    # such that each of d, f, m from 1 to k appears exactly once
    
    # This is equivalent to finding a subset of k triples from the given t triples
    # such that when we look at all first elements, they are 1..k each once,
    # all second elements are 1..k each once, and all third elements are 1..k each once
    
    # Since k <= 22, we can't try all subsets of k triples (C(t,k) is too large)
    # But we can try all permutations of [1..k] for one part, and see if we can extend
    
    # Let's fix a permutation of demographics and see if we can find matching f and m
    # Actually, even better approach: try all possible k-tuples of the given combinations
    
    # But t <= k^3 = 22^3 = 10648, so trying all subsets of k combinations might be feasible for small k
    # For larger k it's still intractable
    
    # Better approach: This is essentially a 3D matching problem.
    # We can reduce it to a bipartite matching by creating a graph where:
    # On one side we have all combinations, on the other side we have positions 1..k
    # But that doesn't directly work either.
    
    # Let's think of it as selecting exactly k triples (d_i, f_i, m_i) from the t available ones
    # such that {d_1, ..., d_k} = {1, ..., k}, same for f's and m's
    
    # For small k, we can do backtracking search
    def backtrack(selected, used_d, used_f, used_m):
        if len(selected) == k:
            return selected
        
        for i, (d, f, m) in enumerate(combinations):
            if d not in used_d and f not in used_f and m not in used_m:
                used_d.add(d)
                used_f.add(f)
                used_m.add(m)
                selected.append(i)
                
                result = backtrack(selected, used_d, used_f, used_m)
                if result is not None:
                    return result
                
                selected.pop()
                used_d.remove(d)
                used_f.remove(f)
                used_m.remove(m)
        
        return None
    
    # Special case optimization for small k
    if k == 0:
        print("YES")
        return
    
    result_indices = backtrack([], set(), set(), set())
    
    if result_indices is None:
        print("NO")
    else:
        print("YES")
        for idx in result_indices:
            d, f, m = combinations[idx]
            print(d, f, m)

solve()
