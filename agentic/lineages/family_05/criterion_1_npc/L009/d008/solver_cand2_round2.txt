import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    paths = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        paths.append((x, y, z))
        idx += 3
        
    # The target profile is sum_{i=1}^{3k} 1/(w-i).
    # Each path (x, y, z) contributes 1/(w-x) + 1/(w-(k+y)) + 1/(w-(2k+z)).
    # Let c_i be the number of times value i appears in the chosen paths.
    # We need sum_{j=1}^k (1/(w-x_j) + 1/(w-(k+y_j)) + 1/(w-(2k+z_j))) = sum_{i=1}^{3k} 1/(w-i).
    # This means for each i in {1, ..., 3k}, the value i must appear exactly once 
    # across all x_j, (k+y_j), and (2k+z_j) for the chosen k paths.
    # This is equivalent to:
    # The set of {x_1, ..., x_k} must be {1, ..., k}.
    # The set of {y_1, ..., y_k} must be {1, ..., k}.
    # The set of {z_1, ..., z_k} must be {1, ..., k}.
    
    # This is a variation of the Exact Cover problem or 3D Matching.
    # Since k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    result = []
    
    def backtrack(idx, count):
        if count == k:
            return True
        if idx == t:
            return False
        
        # Try including paths[idx]
        x, y, z = paths[idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append(paths[idx])
            if backtrack(idx + 1, count + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try skipping paths[idx]
        # Pruning: if remaining paths are not enough to reach k
        if (t - 1 - idx) >= (k - count):
            if backtrack(idx + 1, count):
                return True
        
        return False

    if backtrack(0, 0):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
