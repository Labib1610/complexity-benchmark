import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    paths = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        paths.append((x, y, z))
        idx += 3

    # The energy signature of path (x, y, z) is p_x * p_{k+y} * p_{2k+z}.
    # We need to choose k paths (x_i, y_i, z_i) such that the product of their
    # signatures equals the product of the first 3k primes.
    # This means the set of indices {x_i} union {k+y_i} union {2k+z_i}
    # must be exactly {1, 2, ..., 3k}.
    # This is equivalent to saying that in the chosen k paths:
    # 1. Each x_i must be distinct (1 to k).
    # 2. Each y_i must be distinct (1 to k).
    # 3. Each z_i must be distinct (1 to k).
    # This is exactly the definition of a Latin Square of size k.
    
    # We need to find a subset of k paths that form a Latin Square.
    # This is equivalent to finding a perfect matching in a 3-partite hypergraph
    # or, more simply, a set of k triples (x, y, z) such that all x, y, z are distinct.
    
    # Since k is small (3k <= 66, so k <= 22), we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    result = []
    
    def backtrack(idx, count):
        if count == k:
            return True
        if idx == t:
            return False
        
        # Option 1: Include paths[idx]
        x, y, z = paths[idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append(paths[idx])
            if backtrack(idx + 1, count + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Option 2: Skip paths[idx]
        # Pruning: if remaining paths are not enough to reach k
        if (t - 1 - idx) >= (k - count):
            if backtrack(idx + 1, count):
                return True
        
        return False

    if backtrack(0, 0):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
