import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    paths = []
    idx = 2
    for _ in range(t):
        paths.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    # The operator is the product of partial derivatives.
    # Applying it to Phi = product(v_1...v_{3k}) results in 1 if and only if
    # each variable v_i is differentiated exactly once.
    # Since we choose k paths, each path (x, y, z) contributes to 
    # v_x, v_{k+y}, v_{2k+z}.
    # Total variables: 3k. Each path uses 3 variables. k paths use 3k variables.
    # For the result to be 1, each variable must be used exactly once.
    # This is equivalent to saying:
    # {x_1, ..., x_k} must be a permutation of {1, ..., k}
    # {y_1, ..., y_k} must be a permutation of {1, ..., k}
    # {z_1, ..., z_k} must be a permutation of {1, ..., k}
    # This is the Exact Cover problem, specifically a 3D matching problem.
    
    # Since k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    res = []
    
    def backtrack(idx, count):
        if count == k:
            return True
        if idx == t:
            return False
        
        # Try including paths[idx]
        x, y, z = paths[idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            res.append(paths[idx])
            if backtrack(idx + 1, count + 1):
                return True
            res.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try skipping paths[idx]
        # Pruning: if remaining paths are not enough
        if t - (idx + 1) >= k - count:
            if backtrack(idx + 1, count):
                return True
        return False

    if backtrack(0, 0):
        print("YES")
        for p in res:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
