import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    paths = []
    idx = 2
    for _ in range(t):
        paths.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    # M = (k+1)I + N, where N is the nilpotent matrix with 1s on the superdiagonal.
    # M^p = sum_{j=0}^2 comb(p, j) * (k+1)^(p-j) * N^j
    # Since we only care about the sum of matrices, we can represent each M^p
    # as a vector of 3 coefficients (c0, c1, c2) corresponding to I, N, N^2.
    # M^p = c0*I + c1*N + c2*N^2
    # c0 = (k+1)^p
    # c1 = p * (k+1)^(p-1)
    # c2 = p*(p-1)/2 * (k+1)^(p-2)
    
    def get_coeffs(p):
        c0 = pow(k + 1, p)
        c1 = p * pow(k + 1, p - 1)
        c2 = (p * (p - 1) // 2) * pow(k + 1, p - 2) if p >= 2 else 0
        return (c0, c1, c2)

    def add_coeffs(a, b):
        return (a[0] + b[0], a[1] + b[1], a[2] + b[2])

    target = (0, 0, 0)
    for i in range(1, 3 * k + 1):
        target = add_coeffs(target, get_coeffs(i))

    path_coeffs = []
    for x, y, z in paths:
        s1 = get_coeffs(x)
        s2 = get_coeffs(k + y)
        s3 = get_coeffs(2 * k + z)
        path_coeffs.append(add_coeffs(s1, add_coeffs(s2, s3)))

    # Meet-in-the-middle: k is up to 22. k/2 is 11.
    # t can be up to k^3 = 10648.
    # We need to pick k paths. If k is small, we can use recursion with pruning.
    
    memo = {}

    def find_subset(idx, count, current_sum):
        if count == k:
            return [] if current_sum == target else None
        if idx == t or count + (t - idx) < k:
            return None
        
        # Try including paths[idx]
        res = find_subset(idx + 1, count + 1, add_coeffs(current_sum, path_coeffs[idx]))
        if res is not None:
            return [paths[idx]] + res
        
        # Try excluding paths[idx]
        res = find_subset(idx + 1, count, current_sum)
        if res is not None:
            return res
        
        return None

    # Since k is small and t is large, use a randomized approach or meet-in-the-middle
    # Given the constraints and time limit, a simple backtracking with pruning might work
    # if the search space is constrained.
    
    result = find_subset(0, 0, (0, 0, 0))
    
    if result:
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
