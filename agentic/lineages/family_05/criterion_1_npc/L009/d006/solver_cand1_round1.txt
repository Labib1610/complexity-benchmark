import sys

def get_primes(n):
    primes = []
    num = 23
    while len(primes) < n:
        is_prime = True
        for i in range(2, int(num**0.5) + 1):
            if num % i == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(num)
        num += 1
    return primes

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    paths = []
    idx = 2
    for _ in range(t):
        paths.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    primes = get_primes(3 * k)
    
    # The condition is sum_{i=1}^k (1/P_{x_i} + 1/P_{k+y_i} + 1/P_{2k+z_i}) = sum_{j=1}^{3k} 1/P_j
    # This is equivalent to saying that the multiset of indices {x_1, ..., x_k, k+y_1, ..., k+y_k, 2k+z_1, ..., 2k+z_k}
    # must be exactly {1, 2, ..., 3k}.
    
    target_indices = set(range(1, 3 * k + 1))
    
    def backtrack(idx, current_paths, used_indices):
        if len(current_paths) == k:
            if not used_indices:
                return current_paths
            return None
        
        if idx == t:
            return None
        
        # Try including paths[idx]
        x, y, z = paths[idx]
        indices = {x, k + y, 2 * k + z}
        
        if indices.issubset(used_indices):
            res = backtrack(idx + 1, current_paths + [paths[idx]], used_indices - indices)
            if res is not None:
                return res
        
        # Try excluding paths[idx]
        res = backtrack(idx + 1, current_paths, used_indices)
        if res is not None:
            return res
        
        return None

    result = backtrack(0, [], target_indices)
    
    if result:
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
