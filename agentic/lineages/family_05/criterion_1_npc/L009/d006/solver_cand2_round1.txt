import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    paths = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        paths.append((x, y, z))
        idx += 3

    # The target sum is sum_{i=1}^k 100^{i-1} + sum_{i=1}^k 100^{k+i-1} + sum_{i=1}^k 100^{2k+i-1}
    # This means in base 100, the digits at positions 0 to k-1, k to 2k-1, and 2k to 3k-1
    # must all be 1.
    # Each path (x, y, z) contributes 1 to position x-1, 1 to position k+y-1, and 1 to position 2k+z-1.
    # Since we need exactly k paths, and each path contributes exactly one 1 to each of the three
    # blocks of positions, this is equivalent to saying:
    # The set of x-coordinates must be {1, ..., k}
    # The set of y-coordinates must be {1, ..., k}
    # The set of z-coordinates must be {1, ..., k}
    # This is exactly the definition of a Latin Square constraint if we view it as a 3D matching problem.
    
    # This is a 3D matching problem, which is NP-hard. However, k is small (up to 22).
    # We can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    result = []
    
    def backtrack(idx, count):
        if count == k:
            return True
        if idx == t:
            return False
        
        # Pruning: if remaining paths are not enough to fill the remaining slots
        if count + (t - idx) < k:
            return False
            
        x, y, z = paths[idx]
        
        # Try including paths[idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append(paths[idx])
            if backtrack(idx + 1, count + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try excluding paths[idx]
        if backtrack(idx + 1, count):
            return True
            
        return False

    if backtrack(0, 0):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
