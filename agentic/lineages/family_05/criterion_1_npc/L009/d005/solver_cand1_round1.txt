import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    paths = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        paths.append((x, y, z))
        idx += 3

    # The condition: AND_{i=1}^k (C - 2^{x_i} - 2^{k+y_i} - 2^{2k+z_i}) = 1
    # C = 2^{3k+1} - 1.
    # Let M_i = C - (2^{x_i} + 2^{k+y_i} + 2^{2k+z_i}).
    # The bitwise AND is 1 if and only if for every bit position j in [0, 3k]:
    # If j=0, the j-th bit of all M_i must be 1.
    # If j > 0, the j-th bit of at least one M_i must be 0.
    
    # The j-th bit of M_i is 1 if the j-th bit of (2^{x_i} + 2^{k+y_i} + 2^{2k+z_i}) is 0.
    # Since j ranges from 0 to 3k, and the sum has only 3 bits set (at x_i, k+y_i, 2k+z_i),
    # the j-th bit of the sum is 1 if j is one of {x_i, k+y_i, 2k+z_i}.
    # So the j-th bit of M_i is 0 if j in {x_i, k+y_i, 2k+z_i}, and 1 otherwise.
    
    # Condition:
    # For j=0: 0 is never in {x_i, k+y_i, 2k+z_i} (since x,y,z >= 1), so bit 0 is always 1.
    # For j in {1, ..., 3k}:
    # The j-th bit of AND M_i is 1 iff for all i, the j-th bit of M_i is 1.
    # This means for all i, j is NOT in {x_i, k+y_i, 2k+z_i}.
    # We want the AND to be 1, so for all j in {1, ..., 3k}, the j-th bit must be 0.
    # This means for every j in {1, ..., 3k}, there must exist at least one i such that j is in {x_i, k+y_i, 2k+z_i}.
    
    # This is equivalent to saying the union of sets {x_i, k+y_i, 2k+z_i} for i=1..k must be {1, 2, ..., 3k}.
    # Since there are 3k elements in the union and we have k paths each contributing 3 elements,
    # this is only possible if all 3k elements are distinct.
    # That is, {x_1, ..., x_k} = {1, ..., k}, {y_1, ..., y_k} = {1, ..., k}, {z_1, ..., z_k} = {1, ..., k}.
    # And for each j in {1..k}, the j-th frequency, j-th polarization, and j-th phase must appear exactly once.
    
    # This is exactly the definition of a Latin Square or a set of k triples forming a transversal.
    # We need to select k paths such that each x, y, z appears exactly once.
    
    import itertools
    
    # Backtracking to find a set of k paths that form a permutation of 1..k for x, y, and z.
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    res = []
    def backtrack(idx, count):
        if count == k:
            return True
        if idx == t:
            return False
        
        # Try including paths[idx]
        x, y, z = paths[idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            res.append(paths[idx])
            if backtrack(idx + 1, count + 1):
                return True
            res.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try skipping
        if backtrack(idx + 1, count):
            return True
        return False

    if backtrack(0, 0):
        print("YES")
        for p in res:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
