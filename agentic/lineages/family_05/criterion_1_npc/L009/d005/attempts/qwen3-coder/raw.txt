def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    paths = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        paths.append((x, y, z))
        index += 3
    
    if k == 0:
        if t >= 0:
            print("YES")
        else:
            print("NO")
        return
    
    C = (1 << (3 * k + 1)) - 1
    
    masks = []
    for x, y, z in paths:
        mask = C - (1 << x) - (1 << (k + y)) - (1 << (2 * k + z))
        masks.append(mask)
    
    target = 1
    
    # Try all combinations of k paths
    from itertools import combinations
    
    for indices in combinations(range(t), k):
        combined_mask = C
        selected_paths = []
        for i in indices:
            combined_mask &= masks[i]
            selected_paths.append(paths[i])
        
        if combined_mask == target:
            print("YES")
            for x, y, z in selected_paths:
                print(x, y, z)
            return
    
    print("NO")

solve()
