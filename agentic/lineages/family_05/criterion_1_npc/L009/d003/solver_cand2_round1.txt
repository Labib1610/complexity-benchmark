import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    events = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        events.append((x, y, z))
        idx += 3
        
    # We need to select k events such that:
    # {x_i} = {1..k}, {y_i} = {1..k}, {z_i} = {1..k}
    # This is equivalent to finding a transversal in a 3D matching problem
    # or a system of distinct representatives.
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    # Group events by x to make backtracking faster
    by_x = [[] for _ in range(k + 1)]
    for e in events:
        by_x[e[0]].append(e)
        
    result = []
    
    def backtrack(x_val):
        if x_val > k:
            return True
        
        for e in by_x[x_val]:
            _, y, z = e
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(e)
                
                if backtrack(x_val + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for e in result:
            print(f"{e[0]} {e[1]} {e[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
