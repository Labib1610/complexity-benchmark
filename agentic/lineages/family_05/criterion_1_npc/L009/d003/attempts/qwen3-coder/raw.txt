from itertools import permutations

def solve():
    k, t = map(int, input().split())
    paths = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        paths.append((x, y, z))
    
    if k == 0:
        print("YES")
        return
    
    # Try all permutations of k paths
    for perm in permutations(range(t), k):
        selected_paths = [paths[i] for i in perm]
        
        # Extract components
        inputs = [p[0] for p in selected_paths]
        encryptors = [p[1] for p in selected_paths]
        outputs = [p[2] for p in selected_paths]
        
        # Check if each component is used exactly once
        if (sorted(inputs) == list(range(1, k+1)) and 
            sorted(encryptors) == list(range(1, k+1)) and 
            sorted(outputs) == list(range(1, k+1))):
            print("YES")
            for path in selected_paths:
                print(path[0], path[1], path[2])
            return
    
    print("NO")

solve()
