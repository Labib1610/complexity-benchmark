We need exactly one path for each input port, encryptor, and output channel;  
this is equivalent to finding a perfect matching in a **3D matching** problem.  

Since `k ≤ 22`, we can use **backtracking with bitmasks** and pruning.  
We can also use **Dancing Links / Algorithm X** style, but simple recursion  
with bitmasks and sorting edges works well.

Here’s the Python solution:  

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    k = int(next(it))
    t = int(next(it))
    paths = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        paths.append((x, y, z))
    
    # Sort by "most constrained" – we can just sort by encryption/out for better pruning
    paths.sort(key=lambda p: (p[1], p[2], p[0]))
    used_input = [False] * k
    used_enc = [False] * k
    used_out = [False] * k
    chosen = []
    n = len(paths)
    
    # Precompute for each index, which paths are available if we haven't used certain indices
    # Simple backtracking with constraint checks is enough for k<=22 and t<=k^3
    from functools import lru_cache
    
    # We need to pick exactly k paths, one for each input, enc, out.
    # We will match input ports first: for each input i, try a path (i,y,z).
    # Group paths by input
    by_input = [[] for _ in range(k)]
    for idx, (x,y,z) in enumerate(paths):
        by_input[x].append((y,z,idx))
    
    # Sort each by degree of (y,z) remaining – for better DFS
    for i in range(k):
        by_input[i].sort(key=lambda t: (t[0], t[1]))
    
    # Backtracking with memo on used masks (not necessary but speeds)
    # masks: used_input, used_enc, used_out
    memo = {}
    
    def dfs(i, used_in_mask, used_enc_mask, used_out_mask):
        if i == k:
            return True
        key = (i, used_in_mask, used_enc_mask, used_out_mask)
        if key in memo:
            return memo[key]
        # try all paths from input i
        for y,z,pidx in by_input[i]:
            if (used_enc_mask >> y) & 1:
                continue
            if (used_out_mask >> z) & 1:
                continue
            chosen.append(pidx)
            # mark
            if dfs(i+1, used_in_mask | (1<<i), used_enc_mask | (1<<y), used_out_mask | (1<<z)):
                memo[key] = True
                return True
            chosen.pop()
        memo[key] = False
        return False
    
    if dfs(0, 0, 0, 0):
        print("YES")
        for pidx in chosen:
            x,y,z = paths[pidx]
            print(x+1, y+1, z+1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
