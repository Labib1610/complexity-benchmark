import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        configs.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem.
    # We have 3*k elements to cover:
    # Alpha ports 1..k, Beta ports 1..k, Gamma ports 1..k
    # Each configuration (x, y, z) covers:
    # Alpha_x, Beta_y, Gamma_z
    
    # Since k is small (max 11), we can use backtracking with pruning.
    
    alpha_used = [False] * (k + 1)
    beta_used = [False] * (k + 1)
    gamma_used = [False] * (k + 1)
    
    result = []
    
    def backtrack(config_idx):
        if len(result) == k:
            return True
        
        if config_idx == t:
            return False
        
        # Try including configs[config_idx]
        x, y, z = configs[config_idx]
        if not alpha_used[x] and not beta_used[y] and not gamma_used[z]:
            alpha_used[x] = True
            beta_used[y] = True
            gamma_used[z] = True
            result.append((x, y, z))
            
            if backtrack(config_idx + 1):
                return True
            
            result.pop()
            alpha_used[x] = False
            beta_used[y] = False
            gamma_used[z] = False
            
        # Try excluding configs[config_idx]
        if backtrack(config_idx + 1):
            return True
            
        return False

    if backtrack(0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
