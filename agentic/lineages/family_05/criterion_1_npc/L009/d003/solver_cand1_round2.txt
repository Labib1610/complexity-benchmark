```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    paths = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        paths.append((x, y, z))
        idx += 3

    # We need to select k paths such that:
    # x_i are distinct (0 to k-1)
    # y_i are distinct (0 to k-1)
    # z_i are distinct (0 to k-1)
    # This is equivalent to finding a set of k triples (x, y, z)
    # such that x, y, z are all permutations of 0..k-1.
    
    # Group paths by x
    by_x = [[] for _ in range(k)]
    for p in paths:
        by_x[p[0]].append(p)
        
    # Use backtracking with bitmasks
    # mask_y: bitmask of used y
    # mask_z: bitmask of used z
    
    ans = [None] * k
    
    def backtrack(x, mask_y, mask_z):
        if x == k:
            return True
        
        for p in by_x[x]:
            y, z = p[1], p[2]
            if not (mask_y & (1 << y)) and not (mask_z & (1 << z)):
                ans[x] = p
                if backtrack(x + 1, mask_y | (1 << y), mask_z | (1 << z)):
                    return True
        return False

    if backtrack(0, 0, 0):
        print("YES")
        for p in ans:
            print(f"{p[0]+1} {p[1]+1} {p[2]+1}")
    else:
        print("NO")

solve()
```
