```python
import sys
from itertools import combinations

def solve():
    k, t = map(int, sys.stdin.readline().split())
    
    if t < k:
        print("NO")
        return
        
    combos = []
    for _ in range(t):
        x, y, z = map(int, sys.stdin.readline().split())
        combos.append((x, y, z))
    
    # Try all combinations of k triples from the available ones
    for selected in combinations(range(t), k):
        # Check if this selection is valid
        operators = set()
        apparatuses = set()
        shifts = set()
        
        valid = True
        for i in selected:
            x, y, z = combos[i]
            if x in operators or y in apparatuses or z in shifts:
                valid = False
                break
            operators.add(x)
            apparatuses.add(y)
            shifts.add(z)
        
        if valid:
            print("YES")
            for i in selected:
                print(combos[i][0], combos[i][1], combos[i][2])
            return
    
    print("NO")

solve()
```
