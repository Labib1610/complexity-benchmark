import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    paths = []
    idx = 2
    for _ in range(t):
        paths.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    c = [0] * (3 * k + 1)
    c[1] = 1
    for i in range(2, 3 * k + 1):
        c[i] = (k + 1) * c[i - 1] + 1
    
    target = sum(c[1:])
    
    # Each path (x, y, z) uses levels x, k+y, 2k+z.
    # We need to pick k paths such that the sum of their levels' capacities equals target.
    # Let A_i be the number of times level i is used.
    # Sum_{i=1}^{3k} A_i * C_i = Sum_{i=1}^{3k} C_i
    # Since C_i grows very fast (C_i = (k+1)C_{i-1} + 1), this is essentially a base-(k+1) representation.
    # Specifically, Sum_{i=1}^n C_i = (C_{n+1} - 1) / k.
    # The condition is equivalent to saying that each level i must be used exactly once.
    # Because if we use level i, we contribute C_i. If we use it more than once, 
    # the sum will exceed the target because C_i > sum_{j=1}^{i-1} C_j.
    # So we need to select k paths such that each level from 1 to 3k is used exactly once.
    
    # This is an exact cover problem.
    # Levels: 1..k (freq), k+1..2k (pol), 2k+1..3k (phase)
    # Each path (x, y, z) covers levels x, k+y, 2k+z.
    
    adj = [[] for _ in range(3 * k + 1)]
    for i, (x, y, z) in enumerate(paths):
        adj[x].append(i)
        adj[k + y].append(i)
        adj[2 * k + z].append(i)
        
    chosen = []
    used_paths = [False] * t
    used_levels = [False] * (3 * k + 1)
    
    def backtrack(level_idx, count):
        if count == k:
            return True
        
        # Find the first unused level
        while level_idx <= 3 * k and used_levels[level_idx]:
            level_idx += 1
        
        if level_idx > 3 * k:
            return count == k
        
        for path_idx in adj[level_idx]:
            if not used_paths[path_idx]:
                x, y, z = paths[path_idx]
                levels = [x, k + y, 2 * k + z]
                
                if not used_levels[levels[0]] and not used_levels[levels[1]] and not used_levels[levels[2]]:
                    used_paths[path_idx] = True
                    for l in levels: used_levels[l] = True
                    chosen.append(paths[path_idx])
                    
                    if backtrack(level_idx + 1, count + 1):
                        return True
                    
                    chosen.pop()
                    for l in levels: used_levels[l] = False
                    used_paths[path_idx] = False
        return False

    if backtrack(1, 0):
        print("YES")
        for p in chosen:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
