import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    triples = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        triples.append((x, y, z))
    
    if k == 0:
        print("YES")
        return
    
    # Precompute masks for each operator
    op_masks = [0] * k
    for idx, (x, y, z) in enumerate(triples):
        op_masks[x] |= (1 << idx)
    
    full = (1 << k) - 1
    
    # DP over operators: dp[mask] = (prev_mask, triple_index)
    dp = {0: (-1, -1)}
    for op in range(k):
        if op_masks[op] == 0:
            print("NO")
            return
        new_dp = {}
        for mask, (prev_mask, prev_idx) in dp.items():
            m = op_masks[op]
            while m:
                low = m & (-m)
                idx = low.bit_length() - 1
                new_mask = mask | (1 << triples[idx][1]) | (1 << (triples[idx][2] + k))
                if new_mask not in new_dp:
                    new_dp[new_mask] = (mask, idx)
                m ^= low
        dp = new_dp
        if not dp:
            print("NO")
            return
    
    # Find a mask covering all apparatuses and shifts
    target = full | (full << k)
    final_mask = None
    for mask in dp:
        if (mask & target) == target:
            final_mask = mask
            break
    
    if final_mask is None:
        print("NO")
        return
    
    # Reconstruct
    result = []
    mask = final_mask
    op = k - 1
    while op >= 0:
        prev_mask, idx = dp[mask]
        result.append(triples[idx])
        mask = prev_mask
        op -= 1
    
    print("YES")
    for x, y, z in reversed(result):
        print(x + 1, y + 1, z + 1)

solve()
