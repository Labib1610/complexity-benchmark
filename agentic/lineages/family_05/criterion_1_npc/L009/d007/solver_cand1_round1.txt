import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    triplets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        triplets.append((x - 1, y - 1, z - 1))
        idx += 3

    # The target sum is 2^(3k+1) - 2.
    # In binary, this is a sequence of 3k ones followed by a zero: 11...110.
    # Each triplet (x, y, z) contributes 2^x + 2^(k+y) + 2^(2k+z).
    # We need exactly k triplets such that:
    # Sum_{i=1}^k 2^{x_i} = 2^k - 1 (all alpha channels 0 to k-1 used exactly once)
    # Sum_{i=1}^k 2^{k+y_i} = 2^{2k} - 2^k (all beta channels 0 to k-1 used exactly once)
    # Sum_{i=1}^k 2^{2k+z_i} = 2^{3k} - 2^{2k} (all gamma channels 0 to k-1 used exactly once)
    # This is equivalent to saying the set of x_i is {0..k-1}, y_i is {0..k-1}, z_i is {0..k-1}.
    # This is an Exact Cover problem. Since k <= 22, we use backtracking with pruning.

    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(triplets):
        adj[x].append(i)

    used_y = [False] * k
    used_z = [False] * k
    res = []

    def backtrack(x_idx):
        if x_idx == k:
            return True
        
        for triplet_idx in adj[x_idx]:
            _, y, z = triplets[triplet_idx]
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                res.append(triplets[triplet_idx])
                
                if backtrack(x_idx + 1):
                    return True
                
                res.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for x, y, z in res:
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")

solve()
