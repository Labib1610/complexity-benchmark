import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    paths = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        paths.append((x, y, z))
        idx += 3
        
    # The target sum is (1 << (3k + 1)) - 2.
    # Binary representation: 1 followed by 3k zeros, minus 2 (binary 10).
    # This is 011...110 (3k ones, then a zero at the end).
    # Specifically, bits 1 to k are 1, bits k+1 to 2k are 1, bits 2k+1 to 3k are 1.
    # Bit 0 is 0.
    # Since each path (x, y, z) contributes 1 to bit x, 1 to bit k+y, and 1 to bit 2k+z,
    # and we pick k paths, the sum of bits x must be k, bits k+y must be k, bits 2k+z must be k.
    # This means each frequency 1..k must appear exactly once, each polarization 1..k exactly once,
    # and each phase shift 1..k exactly once.
    
    # This is equivalent to finding a system of distinct representatives or a perfect matching
    # in a 3-partite hypergraph, but specifically, it's a 3D matching problem.
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    res = []
    
    def backtrack(idx, count):
        if count == k:
            return True
        if idx == t:
            return False
        
        # Try including paths[idx]
        x, y, z = paths[idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            res.append(paths[idx])
            if backtrack(idx + 1, count + 1):
                return True
            res.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try skipping paths[idx]
        # Pruning: if remaining paths are not enough to reach k
        if (t - 1 - idx) >= (k - count):
            if backtrack(idx + 1, count):
                return True
        
        return False

    if backtrack(0, 0):
        print("YES")
        for p in res:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
