```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    paths = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        paths.append((x - 1, y - 1, z - 1))
        idx += 3

    # The condition sum(2^x) = 2^(k+1) - 2 means that for each dimension,
    # the set of chosen paths must contain exactly one path for each value 0 to k-1.
    # This is equivalent to finding a set of k paths (x_i, y_i, z_i) such that
    # {x_0, ..., x_{k-1}} = {0, ..., k-1}, {y_0, ..., y_{k-1}} = {0, ..., k-1},
    # and {z_0, ..., z_{k-1}} = {0, ..., k-1}.
    # This is exactly the problem of finding a transversal in a 3D Latin square,
    # or more simply, finding a perfect matching in a 3-partite hypergraph.
    # Since k <= 22, we can use backtracking with pruning.

    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(paths):
        adj[x].append((y, z, i))

    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        # Try all paths with frequency x
        for y, z, i in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(paths[i])
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for x, y, z in result:
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")

solve()
```
