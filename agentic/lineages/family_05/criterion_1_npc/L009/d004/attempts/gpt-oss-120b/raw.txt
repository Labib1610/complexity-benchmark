```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    rows = []
    col_rows = [[] for _ in range(3 * k)]
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        mask = (1 << x) | (1 << (k + y)) | (1 << (2 * k + z))
        rows.append((mask, (x + 1, y + 1, z + 1)))
        idx = len(rows) - 1
        col_rows[x].append(idx)
        col_rows[k + y].append(idx)
        col_rows[2 * k + z].append(idx)

    full_mask = (1 << (3 * k)) - 1
    solution = []

    # heuristic: choose column with fewest remaining rows
    def choose_column(used_mask):
        min_c = -1
        min_len = 10**9
        for c in range(3 * k):
            if not (used_mask >> c) & 1:
                cnt = 0
                for r in col_rows[c]:
                    if rows[r][0] & used_mask == 0:
                        cnt += 1
                if cnt == 0:
                    return -1  # dead end
                if cnt < min_len:
                    min_len = cnt
                    min_c = c
        return min_c

    def dfs(used_mask, depth):
        if used_mask == full_mask:
            return True
        if depth == k:
            return False
        col = choose_column(used_mask)
        if col == -1:
            return False
        for r in col_rows[col]:
            row_mask, triple = rows[r]
            if row_mask & used_mask:
                continue
            solution.append(triple)
            if dfs(used_mask | row_mask, depth + 1):
                return True
            solution.pop()
        return False

    if dfs(0, 0):
        out = ["YES"]
        out += ["{} {} {}".format(*tr) for tr in solution]
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
```
