```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    tasks = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        tasks.append((x, y, z))
        idx += 3
        
    # We need to select k tasks such that all x are distinct, all y are distinct,
    # and all z are distinct. This is equivalent to finding a transversal in a 
    # 3D Latin Square or a matching in a 3-partite hypergraph.
    # Since k <= 22, we can use backtracking with pruning.
    
    # Group tasks by processor x
    by_x = [[] for _ in range(k)]
    for task in tasks:
        by_x[task[0]].append(task)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x):
        if x == k:
            return True
        
        for task in by_x[x]:
            _, y, z = task
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(task)
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for task in result:
            print(f"{task[0] + 1} {task[1] + 1} {task[2] + 1}")
    else:
        print("NO")

solve()
```
