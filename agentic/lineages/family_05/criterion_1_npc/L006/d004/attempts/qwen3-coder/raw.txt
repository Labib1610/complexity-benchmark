from collections import defaultdict
import sys

def solve():
    k, t = map(int, sys.stdin.readline().split())
    
    # If no records, impossible unless k=0, but k >= 1
    if t == 0:
        if k == 0:
            print("YES")
        else:
            print("NO")
        return
    
    # Store compatibility records
    records = []
    # For each module, store which (row,col) pairs it can be placed at
    module_to_positions = defaultdict(list)
    # For each position (row,col), store which modules can be placed there
    position_to_modules = defaultdict(list)
    
    for _ in range(t):
        m, c, f = map(int, sys.stdin.readline().split())
        records.append((m, c, f))
        module_to_positions[m].append((c, f))
        position_to_modules[(c, f)].append(m)
    
    # We need to find a perfect matching: each module used once, each row once, each column once
    # This is equivalent to finding a permutation of rows and columns such that we can assign
    # each module to a unique (row, col) pair where it's compatible
    
    # This is essentially a bipartite matching problem, but with weights/assignments.
    # More precisely, we want to find a permutation π of {1..k} such that for each i,
    # module i can be placed at (i, π(i)).
    
    # Let's model this as a matching problem:
    # Left side: modules 1..k
    # Right side: positions (row, col) where 1 <= row, col <= k
    # Edge exists if module can be placed at that position
    # We want a perfect matching that uses each row and each column exactly once.
    
    # Actually, let's think differently:
    # We want to select k triples (m,c,f) such that:
    # - All m values are distinct and in 1..k
    # - All c values are distinct and in 1..k  
    # - All f values are distinct and in 1..k
    # - Each selected triple is from the input records
    
    # This is a tripartite matching problem. It's NP-hard in general, but since k<=22,
    # we might be able to use backtracking with pruning.
    
    # Let's try backtracking:
    # Try assigning each module to a position (c,f) such that:
    # - Module m can go to (c,f) according to records
    # - No other module is assigned to row c
    # - No other module is assigned to column f
    # - Each module used exactly once
    
    # To make it faster, for each module, precompute valid (c,f) positions
    # Then do backtracking search
    
    # available_rows and available_cols track which rows/cols haven't been used yet
    available_rows = set(range(1, k+1))
    available_cols = set(range(1, k+1))
    used_modules = [False] * (k+1)  # 1-indexed
    solution = []
    
    def backtrack(module):
        # Try to assign 'module'
        if module > k:
            return True  # All modules assigned
            
        # Get valid positions for this module
        for (c, f) in module_to_positions[module]:
            if c in available_rows and f in available_cols:
                # Try this assignment
                available_rows.remove(c)
                available_cols.remove(f)
                solution.append((module, c, f))
                
                if backtrack(module + 1):
                    return True
                    
                # Backtrack
                available_rows.add(c)
                available_cols.add(f)
                solution.pop()
        
        return False
    
    # Optimization: start with modules that have fewer options
    # But for simplicity, let's just go in order 1..k
    
    if backtrack(1):
        print("YES")
        for (m, c, f) in solution:
            print(m, c, f)
    else:
        print("NO")

solve()
