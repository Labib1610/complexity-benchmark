import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    W = int(input_data[0])
    N = int(input_data[1])
    
    vectors = []
    idx = 2
    for _ in range(N):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        vectors.append((x, y, z))
        idx += 3
        
    # We need to select W vectors such that:
    # 1. All X are distinct (1 to W)
    # 2. All Y are distinct (1 to W)
    # 3. All Z are distinct (1 to W)
    # This is equivalent to finding a transversal in a 3D Latin Square or 
    # a set of W disjoint cells in a WxWxW grid.
    
    # Since W <= 22, we can use backtracking with pruning.
    # Sort vectors to try to fill X=1, X=2, ...
    vectors.sort()
    
    used_y = [False] * (W + 1)
    used_z = [False] * (W + 1)
    
    # Group vectors by X
    by_x = [[] for _ in range(W + 1)]
    for v in vectors:
        by_x[v[0]].append(v)
        
    result = []
    
    def backtrack(x):
        if x > W:
            return True
        
        for v in by_x[x]:
            _, y, z = v
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(v)
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for v in result:
            print(f"{v[0]} {v[1]} {v[2]}")
    else:
        print("NO")

solve()
