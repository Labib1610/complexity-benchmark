import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # We need to select k triples (m, c, f) such that:
    # - All m are distinct (1 to k)
    # - All c are distinct (1 to k)
    # - All f are distinct (1 to k)
    # This is equivalent to finding a perfect matching in a 3D bipartite-like structure
    # or more simply, an exact cover problem.
    # Since k <= 22, we can use backtracking with pruning (Dancing Links style).
    
    records = []
    idx = 2
    for _ in range(t):
        m = int(input_data[idx]) - 1
        c = int(input_data[idx+1]) - 1
        f = int(input_data[idx+2]) - 1
        records.append((m, c, f))
        idx += 3
        
    # Group records by module m
    adj = [[] for _ in range(k)]
    for m, c, f in records:
        adj[m].append((c, f))
        
    used_c = [False] * k
    used_f = [False] * k
    result = []
    
    def backtrack(m):
        if m == k:
            return True
        
        # Heuristic: sort by number of available options to prune faster
        options = []
        for c, f in adj[m]:
            if not used_c[c] and not used_f[f]:
                options.append((c, f))
        
        # Sort options by how much they restrict future choices (MRV)
        # For simplicity, just iterate
        for c, f in options:
            used_c[c] = True
            used_f[f] = True
            result.append((m + 1, c + 1, f + 1))
            
            if backtrack(m + 1):
                return True
            
            result.pop()
            used_c[c] = False
            used_f[f] = False
            
        return False

    if backtrack(0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

solve()
