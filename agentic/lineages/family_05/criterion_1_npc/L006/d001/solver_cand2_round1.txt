```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        modules.append((x, y, z))
        idx += 3

    # This is an Exact Cover problem.
    # We have k^3 possible triples, but only t are available.
    # We need to pick k triples such that each (grid, port) is used exactly once.
    # This is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph.
    # Since k <= 22, we use backtracking with pruning (Dancing Links style).
    
    adj = [[] for _ in range(k + 1)]
    for i in range(t):
        x, y, z = modules[i]
        adj[x].append(i)
        
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    result = []

    def backtrack(x_idx):
        if x_idx > k:
            return True
        
        for mod_idx in adj[x_idx]:
            _, y, z = modules[mod_idx]
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(modules[mod_idx])
                
                if backtrack(x_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for m in result:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

solve()
```
