import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        modules.append((x, y, z))
        idx += 3

    # We need to select k modules such that each zone, freq, and protocol is used exactly once.
    # This is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph,
    # or more simply, an exact cover problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    # Group modules by zone
    by_zone = [[] for _ in range(k)]
    for m in modules:
        by_zone[m[0]].append(m)
        
    used_freq = [False] * k
    used_prot = [False] * k
    result = [None] * k
    
    def backtrack(zone_idx):
        if zone_idx == k:
            return True
        
        for m in by_zone[zone_idx]:
            f, p = m[1], m[2]
            if not used_freq[f] and not used_prot[p]:
                used_freq[f] = True
                used_prot[p] = True
                result[zone_idx] = m
                
                if backtrack(zone_idx + 1):
                    return True
                
                used_freq[f] = False
                used_prot[p] = False
        return False

    if backtrack(0):
        print("YES")
        for m in result:
            print(f"{m[0]+1} {m[1]+1} {m[2]+1}")
    else:
        print("NO")

solve()
