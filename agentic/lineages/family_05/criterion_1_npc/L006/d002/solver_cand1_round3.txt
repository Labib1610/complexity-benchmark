import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for _ in range(t):
        u = int(input_data[idx])
        d = int(input_data[idx+1])
        s = int(input_data[idx+2])
        modules.append((u, d, s))
        idx += 3
        
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    # To optimize, we use the Dancing Links approach or simple backtracking with pruning.
    # Given the constraints and the nature of the problem (Latin Square completion),
    # simple backtracking with bitmasks is efficient enough for k=22.
    
    # Represent constraints as bitmasks
    # u_mask: bit i is set if Uplink ID i+1 is used
    # d_mask: bit i is set if Downlink ID i+1 is used
    # s_mask: bit i is set if Sideband ID i+1 is used
    
    # Sort modules to potentially find a solution faster
    modules.sort()
    
    chosen = []
    
    def backtrack(idx, u_mask, d_mask, s_mask):
        if len(chosen) == k:
            return True
        
        if idx == t:
            return False
        
        # Pruning: if remaining modules are not enough to fill the rest
        if len(chosen) + (t - idx) < k:
            return False
            
        for i in range(idx, t):
            u, d, s = modules[i]
            u_bit = 1 << (u - 1)
            d_bit = 1 << (d - 1)
            s_bit = 1 << (s - 1)
            
            if not (u_mask & u_bit) and not (d_mask & d_bit) and not (s_mask & s_bit):
                chosen.append(modules[i])
                if backtrack(i + 1, u_mask | u_bit, d_mask | d_bit, s_mask | s_bit):
                    return True
                chosen.pop()
        return False

    if backtrack(0, 0, 0, 0):
        print("YES")
        for m in chosen:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
