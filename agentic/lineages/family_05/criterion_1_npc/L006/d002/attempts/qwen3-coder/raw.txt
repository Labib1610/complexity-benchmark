        m = int(data[index])
        c = int(data[index+1])
        f = int(data[index+2])
        protocols.append((m, c, f))
        index += 3

    # - All materials are distinct
    # - All catalysts are distinct
    # - All treatments are distinct
    
    # This is equivalent to finding a matching of size k in a 3-dimensional bipartite graph
    # We can model this as finding a perfect matching in a tripartite hypergraph
    # Since k <= 22, we can try all combinations of k protocols from t protocols
    # But that's C(t, k) which could be too large
    # Better approach: since we need exactly k protocols and k materials/catalysts/treatments,
    # this is equivalent to finding a permutation of k elements such that each protocol matches

    # We want to select k protocols where the materials, catalysts, and treatments are all permutations of [1..k]

    # Create a graph where each protocol is an edge connecting (material, catalyst, treatment)

    # Since k <= 22, and we're looking for a 3D matching, we can use backtracking or try all subsets

    # this is like finding a Latin square or a 3D matching.
    
    # A more practical approach for small k:
    # Try all k! permutations of [1..k] for materials, and for each check if there exists
    # a corresponding set of protocols. But we also need to match catalysts and treatments.
    
    # Even better: model as exact cover and use Algorithm X (Dancing Links), but that's complex.
    
    # Simple approach for k<=22: try all subsets of k protocols and check if they form a valid matching
    # C(t, k) for t up to k^3 = 22^3 = 10648, and k=22, C(10648, 22) is way too large.
    
    # Let's try a different approach:
    # Build a mapping from (m,c,f) to see if protocol exists
    # Then try all permutations of [1..k] for materials, [1..k] for catalysts, [1..k] for treatments
    # Check if the k protocols exist
    
    # Number of permutations is k! which is manageable for k<=22 (22! is huge though)
    # But we can do better: since we need a 3D matching, we can fix one dimension and permute others
    
    protocol_set = set(protocols)
    
    # Try all permutations of catalysts and treatments for a fixed material order [1..k]
    # Actually, let's not fix anything, just try to find a valid combination
    
    # More efficient approach:
    # Since we need exactly k protocols with distinct materials, catalysts, and treatments,
    # we can think of this as finding a bijection between materials, catalysts, and treatments
    # such that for each i, the protocol (materials[i], catalysts[i], treatments[i]) exists
    
    # So we can:
    # 1. Generate all permutations of [1..k] for catalysts
    # 2. Generate all permutations of [1..k] for treatments  
    # 3. For each pair, check if all k protocols (i, catalysts[i], treatments[i]) exist
    # 4. If found, we have our solution
    
    # But k! * k! = (22!)^2 is still too large
    
    # Wait, there's a smarter way:
    # Since materials must be 1..k, catalysts must be a permutation of 1..k, 
    # treatments must be a permutation of 1..k,
    # we just need to find if there exist permutations p1, p2 of [1..k] such that
    # protocols (1, p1[1], p2[1]), (2, p1[2], p2[2]), ..., (k, p1[k], p2[k]) all exist
    
    # Number of checks: k! * k! which is too much for k=22
    
    # Let's use a backtracking approach instead:
    # Try to build the matching incrementally
    
    from collections import defaultdict
    import sys
    
    # Map each (m,c,f) to check existence quickly
    protocol_exists = {}
    for m, c, f in protocols:
        protocol_exists[(m, c, f)] = True
        
    # For each material, precompute which (c, f) pairs exist
    mat_to_cf = defaultdict(list)
    cat_to_mf = defaultdict(list)
    fin_to_mc = defaultdict(list)
    
    for m, c, f in protocols:
        mat_to_cf[m].append((c, f))
        cat_to_mf[c].append((m, f))
        fin_to_mc[f].append((m, c))
        
    # Backtracking approach
    def backtrack(selected, used_m, used_c, used_f):
        if len(selected) == k:
            return selected
            
        # Find the next unused material
        next_m = None
        for m in range(1, k + 1):
            if m not in used_m:
                next_m = m
                break
                
        if next_m is None:
            return None
            
        # Try all protocols with this material
        for c, f in mat_to_cf.get(next_m, []):
            if c not in used_c and f not in used_f:
                used_m.add(next_m)
                used_c.add(c)
                used_f.add(f)
                selected.append((next_m, c, f))
                
                result = backtrack(selected, used_m, used_c, used_f)
                if result is not None:
                    return result
                    
                selected.pop()
                used_m.remove(next_m)
                used_c.remove(c)
                used_f.remove(f)
                
        return None
        
    # Start backtracking
    solution = backtrack([], set(), set(), set())
    
    if solution is None:
        print("NO")
    else:
        print("YES")
        for m, c, f in solution:
            print(m, c, f)

solve()
