import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for _ in range(t):
        u = int(input_data[idx]) - 1
        d = int(input_data[idx+1]) - 1
        s = int(input_data[idx+2]) - 1
        modules.append((u, d, s))
        idx += 3

    # This is an Exact Cover problem (3D matching).
    # Since k <= 22, we use backtracking with pruning.
    # Represent constraints as bitmasks.
    
    # Group modules by uplink to simplify selection
    by_uplink = [[] for _ in range(k)]
    for m in modules:
        by_uplink[m[0]].append(m)
        
    chosen = []
    used_d = 0
    used_s = 0
    
    def backtrack(u_idx):
        if u_idx == k:
            return True
        
        for m in by_uplink[u_idx]:
            d_bit = 1 << m[1]
            s_bit = 1 << m[2]
            
            if not (used_d & d_bit) and not (used_s & s_bit):
                nonlocal used_d, used_s
                used_d |= d_bit
                used_s |= s_bit
                chosen.append(m)
                
                if backtrack(u_idx + 1):
                    return True
                
                chosen.pop()
                used_d &= ~d_bit
                used_s &= ~s_bit
        return False

    if backtrack(0):
        print("YES")
        for m in chosen:
            print(f"{m[0]+1} {m[1]+1} {m[2]+1}")
    else:
        print("NO")

solve()
