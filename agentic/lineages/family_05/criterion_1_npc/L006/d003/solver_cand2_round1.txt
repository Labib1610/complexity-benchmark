import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    points = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        points.append((x, y, z))
        idx += 3
        
    # We need to select k points such that all x, y, z are distinct.
    # This is equivalent to finding a perfect matching in a 3D bipartite-like structure
    # or a Latin Square completion problem. Since k <= 22, we use backtracking.
    
    used_x = [False] * k
    used_y = [False] * k
    used_z = [False] * k
    
    # Group points by x to simplify backtracking
    by_x = [[] for _ in range(k)]
    for p in points:
        by_x[p[0]].append(p)
        
    result = []
    
    def backtrack(x_idx):
        if x_idx == k:
            return True
        
        for p in by_x[x_idx]:
            _, y, z = p
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(p)
                
                if backtrack(x_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for p in result:
            print(f"{p[0]+1} {p[1]+1} {p[2]+1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
