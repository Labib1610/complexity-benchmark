import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    protocols = []
    idx = 2
    for _ in range(t):
        m = int(input_data[idx])
        c = int(input_data[idx+1])
        f = int(input_data[idx+2])
        protocols.append((m, c, f))
        idx += 3
        
    # This is a 3D matching problem, which is NP-complete.
    # Given k <= 22, we can use backtracking with pruning.
    
    used_m = [False] * (k + 1)
    used_c = [False] * (k + 1)
    used_f = [False] * (k + 1)
    
    result = []
    
    def backtrack(p_idx):
        if len(result) == k:
            return True
        if p_idx == t:
            return False
        
        # Pruning: if remaining protocols are not enough
        if len(result) + (t - p_idx) < k:
            return False
            
        m, c, f = protocols[p_idx]
        
        # Try including this protocol
        if not used_m[m] and not used_c[c] and not used_f[f]:
            used_m[m] = True
            used_c[c] = True
            used_f[f] = True
            result.append((m, c, f))
            
            if backtrack(p_idx + 1):
                return True
            
            result.pop()
            used_m[m] = False
            used_c[c] = False
            used_f[f] = False
            
        # Try excluding this protocol
        if backtrack(p_idx + 1):
            return True
            
        return False

    # Sort protocols to potentially hit a solution faster
    # Using a simple heuristic: try to pick protocols that are "rare"
    if backtrack(0):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
