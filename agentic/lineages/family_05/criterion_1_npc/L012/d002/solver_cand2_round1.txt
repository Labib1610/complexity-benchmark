import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    pockets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        pockets.append((x, y, z))
        idx += 3

    # Interference condition:
    # dist3d^2 = (x1-x2)^2 + (y1-y2)^2 + (z1-z2)^2
    # dist_xy^2 = (x1-x2)^2 + (y1-y2)^2
    # dist_xz^2 = (x1-x2)^2 + (z1-z2)^2
    # dist_yz^2 = (y1-y2)^2 + (z1-z2)^2
    # Interference if dist3d^2 == dist_xy^2 OR dist3d^2 == dist_xz^2 OR dist3d^2 == dist_yz^2
    # This simplifies to:
    # (z1-z2)^2 == 0 OR (y1-y2)^2 == 0 OR (x1-x2)^2 == 0
    # Which means: x1 == x2 OR y1 == y2 OR z1 == z2

    adj = [[] for _ in range(t)]
    for i in range(t):
        for j in range(i + 1, t):
            p1 = pockets[i]
            p2 = pockets[j]
            if p1[0] == p2[0] or p1[1] == p2[1] or p1[2] == p2[2]:
                adj[i].append(j)
                adj[j].append(i)

    # We need to find an independent set of size k
    # Since k is small (up to 22), we can use backtracking with pruning
    
    selected = []
    
    def backtrack(start_idx, current_selection):
        if len(current_selection) == k:
            return current_selection
        
        for i in range(start_idx, t):
            # Check if pockets[i] interferes with any in current_selection
            can_add = True
            for s_idx in current_selection:
                p1 = pockets[i]
                p2 = pockets[s_idx]
                if p1[0] == p2[0] or p1[1] == p2[1] or p1[2] == p2[2]:
                    can_add = False
                    break
            
            if can_add:
                res = backtrack(i + 1, current_selection + [i])
                if res is not None:
                    return res
        return None

    result = backtrack(0, [])
    
    if result:
        print("YES")
        for idx in result:
            print(f"{pockets[idx][0]} {pockets[idx][1]} {pockets[idx][2]}")
    else:
        print("NO")

solve()
