import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    k = int(next(it))
    t = int(next(it))
    
    pockets = []
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        pockets.append((x, y, z))
        
    # We need to select exactly k pockets such that all x, y, z are distinct.
    # This is equivalent to finding a perfect matching in a 3-partite 3-uniform hypergraph.
    # We can reduce this to a bipartite matching problem between x and (y, z).
    
    adj = [[] for _ in range(k + 1)]
    for x, y, z in pockets:
        adj[x].append((y, z))
        
    match_yz = {}
    match_x = [0] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if v not in visited:
                visited.add(v)
                if v not in match_yz or dfs(match_yz[v], visited):
                    match_yz[v] = u
                    match_x[u] = v
                    return True
        return False

    for u in range(1, k + 1):
        visited = set()
        if not dfs(u, visited):
            print("NO")
            return
            
    print("YES")
    for x in range(1, k + 1):
        y, z = match_x[x]
        print(x, y, z)

if __name__ == '__main__':
    solve()
