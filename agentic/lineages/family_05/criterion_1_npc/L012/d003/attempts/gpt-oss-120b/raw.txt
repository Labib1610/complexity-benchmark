```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    triples = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        triples.append((x, y, z))

    # Exact cover: columns 0..k-1 for X, k..2k-1 for Y, 2k..3k-1 for Z
    ncol = 3 * k
    # Build column lists
    col_size = [0] * ncol
    rows_of_col = [[] for _ in range(ncol)]
    for idx, (x, y, z) in enumerate(triples):
        cols = (x, k + y, 2 * k + z)
        for c in cols:
            rows_of_col[c].append(idx)
            col_size[c] += 1

    # If any column has zero rows, impossible
    if any(sz == 0 for sz in col_size):
        print("NO")
        return

    # DLX structures (simplified)
    # For each row store its three columns
    row_cols = [ (x, k + y, 2 * k + z) for (x, y, z) in triples ]

    used_row = [False] * t
    solution = []

    # heuristic: choose column with minimal remaining rows
    def choose_column(active_cols):
        # active_cols is set of uncovered columns
        min_c = None
        min_sz = 10**9
        for c in active_cols:
            sz = 0
            for r in rows_of_col[c]:
                if not used_row[r]:
                    sz += 1
            if sz < min_sz:
                min_sz = sz
                min_c = c
                if sz == 0:
                    break
        return min_c, min_sz

    def search(uncovered):
        if not uncovered:
            return True
        c, sz = choose_column(uncovered)
        if sz == 0:
            return False
        # try each row covering column c
        for r in rows_of_col[c]:
            if used_row[r]:
                continue
            # check if row's columns are all uncovered
            cols = row_cols[r]
            if any(col not in uncovered for col in cols):
                continue
            # choose row
            used_row[r] = True
            solution.append(r)
            new_uncovered = uncovered - set(cols)
            if search(new_uncovered):
                return True
            # backtrack
            solution.pop()
            used_row[r] = False
        return False

    all_cols = set(range(ncol))
    if search(all_cols):
        print("YES")
        out = []
        for r in solution:
            x, y, z = triples[r]
            out.append(f"{x+1} {y+1} {z+1}")
        sys.stdout.write("\n".join(out))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
