import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    pockets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        pockets.append((x, y, z))
        idx += 3

    # We need to pick k pockets such that all x, y, z are covered.
    # This is equivalent to finding a transversal of size k in a 3D grid.
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    selected = []
    
    def backtrack(start_idx, x_mask, y_mask, z_mask, count):
        if count == k:
            if x_mask == (1 << k) - 1 and y_mask == (1 << k) - 1 and z_mask == (1 << k) - 1:
                return True
            return False
        
        # Pruning: if remaining pockets are not enough to cover missing dimensions
        rem = k - count
        missing_x = bin(((1 << k) - 1) ^ x_mask).count('1')
        missing_y = bin(((1 << k) - 1) ^ y_mask).count('1')
        missing_z = bin(((1 << k) - 1) ^ z_mask).count('1')
        if max(missing_x, missing_y, missing_z) > rem:
            return False
        
        for i in range(start_idx, t):
            x, y, z = pockets[i]
            new_x = x_mask | (1 << x)
            new_y = y_mask | (1 << y)
            new_z = z_mask | (1 << z)
            
            selected.append(pockets[i])
            if backtrack(i + 1, new_x, new_y, new_z, count + 1):
                return True
            selected.pop()
            
        return False

    if backtrack(0, 0, 0, 0, 0):
        print("YES")
        for p in selected:
            print(f"{p[0]+1} {p[1]+1} {p[2]+1}")
    else:
        print("NO")

solve()
