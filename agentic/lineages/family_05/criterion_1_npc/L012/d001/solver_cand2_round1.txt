import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem is equivalent to finding a perfect matching in a 3-partite 
    # hypergraph, or more simply, a Latin Square of size k given a set of 
    # allowed cells. This is equivalent to finding a system of distinct 
    # representatives or an exact cover.
    # Since k <= 22, we can model this as an exact cover problem.
    # We need to select k triples (x, y, z) such that all x are distinct,
    # all y are distinct, and all z are distinct.
    
    # This is exactly the problem of finding a perfect matching in a 3-partite 
    # graph, which is NP-complete (3D matching). However, with k <= 22, 
    # we can use backtracking with pruning.
    
    pockets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        pockets.append((x, y, z))
        idx += 3
        
    # Group pockets by x
    by_x = [[] for _ in range(k + 1)]
    for p in pockets:
        by_x[p[0]].append(p)
        
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    result = []
    
    def backtrack(x):
        if x > k:
            return True
        
        for p in by_x[x]:
            _, y, z = p
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(p)
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
