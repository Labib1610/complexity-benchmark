import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    anomalies = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        anomalies.append((x, y, z))
        idx += 3
        
    # We need to pick k anomalies such that no two share x, y, or z.
    # This is equivalent to finding a matching of size k in a 3D bipartite-like structure.
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    # Group anomalies by x to simplify backtracking
    by_x = [[] for _ in range(k + 1)]
    for a in anomalies:
        by_x[a[0]].append(a)
        
    result = []
    
    def backtrack(x_idx, count):
        if count == k:
            return True
        if x_idx > k:
            return False
        
        # Try skipping this x
        if backtrack(x_idx + 1, count):
            return True
            
        # Try picking an anomaly with this x
        for a in by_x[x_idx]:
            y, z = a[1], a[2]
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(a)
                if backtrack(x_idx + 1, count + 1):
                    return True
                result.pop()
                used_y[y] = False
                used_z[z] = False
        
        return False

    if backtrack(1, 0):
        print("YES")
        for a in result:
            print(f"{a[0]} {a[1]} {a[2]}")
    else:
        print("NO")

solve()
