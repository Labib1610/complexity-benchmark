```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    pockets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        pockets.append((x, y, z))
        idx += 3
        
    # This is a 3D matching problem, which is NP-hard.
    # Given k <= 22, we can use backtracking with pruning.
    # We need to pick k points such that all x, y, z are distinct.
    # This is equivalent to finding a transversal in a 3D grid.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    # Sort pockets to potentially improve pruning
    pockets.sort()
    
    result = []
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        # Pruning: if remaining pockets are not enough to reach k
        if (t - start_idx) < (k - count):
            return False
            
        for i in range(start_idx, t):
            x, y, z = pockets[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = used_y[y] = used_z[z] = True
                result.append((x, y, z))
                
                if backtrack(i + 1, count + 1):
                    return True
                
                result.pop()
                used_x[x] = used_y[y] = used_z[z] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
