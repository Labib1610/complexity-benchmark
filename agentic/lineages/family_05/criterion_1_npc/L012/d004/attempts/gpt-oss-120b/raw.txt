import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    points = {}
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        points.setdefault(x, []).append((y, z))

    # each x must be used exactly once, so we need exactly k distinct x's.
    # If any x missing, impossible.
    if len(points) < k:
        print("NO")
        return

    # Prepare list of x's to assign
    xs = list(range(k))
    # sort xs by number of candidates to improve backtracking
    xs.sort(key=lambda x: len(points.get(x, [])))
    used_y = 0
    used_z = 0
    assign = [None] * k  # index by position in xs list

    # Precompute bit masks for quick check
    full_mask = (1 << k) - 1

    def dfs(idx: int) -> bool:
        nonlocal used_y, used_z
        if idx == k:
            return True
        x = xs[idx]
        cand = points.get(x, [])
        # try candidates where y and z are free
        for y, z in cand:
            if (used_y >> y) & 1:  # y already used
                continue
            if (used_z >> z) & 1:  # z already used
                continue
            # assign
            used_y |= 1 << y
            used_z |= 1 << z
            assign[idx] = (x, y, z)
            if dfs(idx + 1):
                return True
            # backtrack
            used_y ^= 1 << y
            used_z ^= 1 << z
        return False

    if dfs(0):
        print("YES")
        # output in original 1-based coordinates, any order
        for x, y, z in assign:
            print(x + 1, y + 1, z + 1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
