import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    pockets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        pockets.append((x, y, z))
        idx += 3
        
    def is_valid(subset):
        for i in range(len(subset)):
            for j in range(i + 1, len(subset)):
                p1 = subset[i]
                p2 = subset[j]
                
                dx = abs(p1[0] - p2[0])
                dy = abs(p1[1] - p2[1])
                dz = abs(p1[2] - p2[2])
                
                dist_3d = dx + dy + dz
                
                # Conditions:
                # dist_3d > dx + dy  => dz > 0
                # dist_3d > dx + dz  => dy > 0
                # dist_3d > dy + dz  => dx > 0
                
                if not (dz > 0 and dy > 0 and dx > 0):
                    return False
        return True

    # The condition simplifies to: for any two points, 
    # their x, y, and z coordinates must all be distinct.
    # This is equivalent to saying that for any two points (x1, y1, z1) and (x2, y2, z2),
    # x1 != x2, y1 != y2, and z1 != z2.
    
    # This is a variation of the Maximum Clique problem or finding a set of k points
    # such that no two share an x, y, or z coordinate.
    # This is equivalent to finding a matching in a 3-partite hypergraph,
    # or more simply, a set of k points where all x are distinct, all y are distinct,
    # and all z are distinct.
    
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    def backtrack(start_idx, current_subset, used_x, used_y, used_z):
        if len(current_subset) == k:
            return current_subset
        
        for i in range(start_idx, t):
            x, y, z = pockets[i]
            if not (used_x & (1 << x) or used_y & (1 << y) or used_z & (1 << z)):
                res = backtrack(i + 1, current_subset + [pockets[i]], 
                                used_x | (1 << x), used_y | (1 << y), used_z | (1 << z))
                if res:
                    return res
        return None

    result = backtrack(0, [], 0, 0, 0)
    
    if result:
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
