```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    pockets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        pockets.append((x, y, z))
        idx += 3

    # We need to select k pockets such that each x, y, z coordinate 0..k-1 appears exactly once.
    # This is equivalent to finding a system of distinct representatives or a matching in a 3D grid.
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    # Group pockets by x coordinate
    by_x = [[] for _ in range(k)]
    for p in pockets:
        by_x[p[0]].append(p)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x_idx):
        if x_idx == k:
            return True
        
        for p in by_x[x_idx]:
            y, z = p[1], p[2]
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(p)
                
                if backtrack(x_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    # Heuristic: sort by_x to try more constrained x first
    # Actually, just simple backtracking is often enough for k=22 if the search space is sparse.
    if backtrack(0):
        print("YES")
        for p in result:
            print(f"{p[0]+1} {p[1]+1} {p[2]+1}")
    else:
        print("NO")

solve()
```
