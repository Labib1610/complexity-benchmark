```python
import sys

sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    triples = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        triples.append((x, y, z))

    # columns: 0..k-1 -> x, k..2k-1 -> y, 2k..3k-1 -> z
    ncol = 3 * k
    # DLX structures
    max_nodes = t * 3 + ncol + 5
    U = [0] * max_nodes
    D = [0] * max_nodes
    L = [0] * max_nodes
    R = [0] * max_nodes
    C = [0] * max_nodes  # column header for each node
    S = [0] * (ncol)     # size of each column
    row_id = [0] * max_nodes  # which input row this node belongs to

    header = ncol  # index of master header
    # init header links
    for i in range(ncol + 1):
        U[i] = D[i] = i
        L[i] = i - 1
        R[i] = i + 1
    L[0] = header
    R[header] = 0
    L[header] = ncol
    R[ncol] = header

    node_cnt = ncol + 1  # next free node index

    def add_row(rid, cols):
        nonlocal node_cnt
        first = -1
        for col in cols:
            nd = node_cnt
            node_cnt += 1
            C[nd] = col
            row_id[nd] = rid
            # vertical links
            U[nd] = U[col]
            D[nd] = col
            D[U[col]] = nd
            U[col] = nd
            S[col] += 1
            # horizontal links
            if first == -1:
                first = nd
                L[nd] = R[nd] = nd
            else:
                L[nd] = L[first]
                R[nd] = first
                R[L[first]] = nd
                L[first] = nd

    for idx, (x, y, z) in enumerate(triples):
        add_row(idx, [x, k + y, 2 * k + z])

    solution_rows = []

    def cover(col):
        # remove column header
        L[R[col]] = L[col]
        R[L[col]] = R[col]
        # remove rows
        i = D[col]
        while i != col:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(col):
        i = U[col]
        while i != col:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[col]] = col
        R[L[col]] = col

    found = False

    def search(k_depth):
        nonlocal found
        if R[header] == header:
            # all columns covered
            found = True
            return
        # choose column with minimal S
        c = R[header]
        min_s = S[c]
        col = c
        j = R[c]
        while j != header:
            if S[j] < min_s:
                min_s = S[j]
                col = j
                if min_s == 0:
                    break
            j = R[j]
        if min_s == 0:
            return
        cover(col)
        r = D[col]
        while r != col:
            solution_rows.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            search(k_depth + 1)
            if found:
                return
            # backtrack
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            solution_rows.pop()
            r = D[r]
        uncover(col)

    search(0)

    if not found:
        print("NO")
    else:
        print("YES")
        for rid in solution_rows:
            x, y, z = triples[rid]
            print(x + 1, y + 1, z + 1)

if __name__ == "__main__":
    solve()
```
