import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    pockets = []
    idx = 2
    for _ in range(t):
        pockets.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    if k > t:
        print("NO")
        return

    def check(subset):
        n = len(subset)
        for i in range(n):
            for j in range(i + 1, n):
                p1 = subset[i]
                p2 = subset[j]
                dx = abs(p1[0] - p2[0])
                dy = abs(p1[1] - p2[1])
                dz = abs(p1[2] - p2[2])
                
                s = dx + dy + dz
                q = dx*dx + dy*dy + dz*dz
                c = dx*dx*dx + dy*dy*dy + dz*dz*dz
                
                if not (s**3 + 2 * c > 3 * s * q):
                    return False
        return True

    # The condition S^3 + 2C > 3SQ is equivalent to:
    # (dx+dy+dz)^3 + 2(dx^3+dy^3+dz^3) > 3(dx+dy+dz)(dx^2+dy^2+dz^2)
    # This is a known inequality related to Schur's Inequality.
    # It holds if and only if at most one of dx, dy, dz is non-zero,
    # or if the values are such that they don't violate the strict inequality.
    # Actually, for non-negative dx, dy, dz, S^3 + 2C - 3SQ = 6*dx*dy*dz.
    # So the condition is 6*dx*dy*dz > 0, meaning dx > 0, dy > 0, and dz > 0.
    
    # We need to pick k pockets such that for every pair, 
    # |x1-x2| > 0, |y1-y2| > 0, and |z1-z2| > 0.
    
    def check_fast(subset):
        n = len(subset)
        for i in range(n):
            for j in range(i + 1, n):
                if subset[i][0] == subset[j][0] or \
                   subset[i][1] == subset[j][1] or \
                   subset[i][2] == subset[j][2]:
                    return False
        return True

    # Backtracking to find a subset of size k
    def backtrack(start, current_subset):
        if len(current_subset) == k:
            return current_subset
        
        for i in range(start, t):
            p = pockets[i]
            valid = True
            for existing in current_subset:
                if p[0] == existing[0] or p[1] == existing[1] or p[2] == existing[2]:
                    valid = False
                    break
            if valid:
                res = backtrack(i + 1, current_subset + [p])
                if res:
                    return res
        return None

    result = backtrack(0, [])
    
    if result:
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
