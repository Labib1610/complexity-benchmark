from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    combinations = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        combinations.append((x, y, z))
        index += 3
    
    # Build a graph where each combination is an edge
    # This is essentially a tripartite matching problem
    # We can model it as a bipartite matching by considering (x,y) -> z
    # But with k<=22, we can use backtracking with pruning
    
    # For each compute node, store viable (image, shard) pairs
    compat = defaultdict(list)
    for x, y, z in combinations:
        compat[x].append((y, z))
    
    # Try to find a perfect matching
    # Use backtracking with bitmasks for efficiency
    
    def backtrack(node, used_img, used_shard):
        if node > k:
            return []
        
        for img, shard in compat[node]:
            if not (used_img & (1 << img)) and not (used_shard & (1 << shard)):
                result = backtrack(node + 1, used_img | (1 << img), used_shard | (1 << shard))
                if result is not None:
                    return [(node, img, shard)] + result
        
        return None
    
    # Special case when k=1
    if k == 1:
        if (1, 1, 1) in combinations:
            print("YES")
            print("1 1 1")
            return
        else:
            print("NO")
            return
    
    # For small k, try all permutations approach might work but let's use smarter backtracking
    # Actually, let's think differently - build a graph and find maximum matching
    # Since k<=22, we can afford exponential algorithm with pruning
    
    # Let's try a different approach: model as finding a system of distinct representatives
    # This is equivalent to finding a perfect matching in a tripartite hypergraph
    
    # Use recursive backtracking with memoization
    from functools import lru_cache
    
    # Convert combinations to a set for O(1) lookup
    combo_set = set(combinations)
    
    # Precompute for each compute node, what (img, shard) pairs are valid
    valid_pairs = defaultdict(list)
    for x, y, z in combinations:
        valid_pairs[x].append((y, z))
    
    # Try all assignments using backtracking
    def search(pos, used_img, used_shard, current_assignment):
        if pos > k:
            return current_assignment[:]
        
        # Pruning: check if remaining resources can satisfy remaining positions
        remaining_positions = k - pos + 1
        available_imgs = 0
        available_shards = 0
        for i in range(1, k+1):
            if not (used_img & (1 << i)):
                available_imgs += 1
            if not (used_shard & (1 << i)):
                available_shards += 1
                
        if available_imgs < remaining_positions or available_shards < remaining_positions:
            return None
            
        for img, shard in valid_pairs[pos]:
            if not (used_img & (1 << img)) and not (used_shard & (1 << shard)):
                current_assignment.append((pos, img, shard))
                result = search(pos + 1, used_img | (1 << img), used_shard | (1 << shard), current_assignment)
                if result is not None:
                    return result
                current_assignment.pop()
                
        return None
    
    assignment = search(1, 0, 0, [])
    
    if assignment is not None:
        print("YES")
        for triple in assignment:
            print(triple[0], triple[1], triple[2])
    else:
        print("NO")

solve()
