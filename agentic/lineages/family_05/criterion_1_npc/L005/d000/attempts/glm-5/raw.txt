import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    triples = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        triples.append((x, y, z))
    
    # Build adjacency: for each (x, y), list of z values
    adj = {}
    for x, y, z in triples:
        key = (x, y)
        if key not in adj:
            adj[key] = []
        adj[key].append(z)
    
    # DP with bitmask for z-assignment
    # dp[mask] = (prev_mask, x, y, z) or None
    dp = {0: None}
    
    for x in range(k):
        new_dp = {}
        for mask in dp:
            for y in range(k):
                key = (x, y)
                if key not in adj:
                    continue
                for z in adj[key]:
                    if mask & (1 << z):
                        continue
                    new_mask = mask | (1 << z)
                    if new_mask not in new_dp:
                        new_dp[new_mask] = (mask, x, y, z)
        dp = new_dp
        if not dp:
            print("NO")
            return
    
    full_mask = (1 << k) - 1
    if full_mask not in dp:
        print("NO")
        return
    
    print("YES")
    result = []
    mask = full_mask
    for _ in range(k):
        prev_mask, x, y, z = dp[mask]
        result.append((x + 1, y + 1, z + 1))
        mask = prev_mask
    
    for triple in result:
        print(triple[0], triple[1], triple[2])

solve()
