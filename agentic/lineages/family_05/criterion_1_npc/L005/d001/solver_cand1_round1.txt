```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3
        
    # Group triples by compute node
    by_node = [[] for _ in range(k)]
    for x, y, z in triples:
        by_node[x].append((y, z))
        
    # memo[node_idx][mask_y][mask_z]
    # Since k <= 22, we use bitmasks.
    # However, 2^22 * 2^22 is too large.
    # But we only need to pick k triples.
    # This is an Exact Cover problem (3D matching).
    # With k=22, we can use backtracking with pruning.
    
    assignment = [None] * k
    used_y = 0
    used_z = 0
    
    def backtrack(node_idx):
        nonlocal used_y, used_z
        if node_idx == k:
            return True
        
        for y, z in by_node[node_idx]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                used_y |= (1 << y)
                used_z |= (1 << z)
                assignment[node_idx] = (y, z)
                
                if backtrack(node_idx + 1):
                    return True
                
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    # Heuristic: sort nodes by number of available choices
    # But simple backtracking might pass given the constraints and structure
    if backtrack(0):
        print("YES")
        for i in range(k):
            y, z = assignment[i]
            print(f"{i + 1} {y + 1} {z + 1}")
    else:
        print("NO")

solve()
```
