```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        modules.append((x, y, z))
        idx += 3

    # This is an Exact Cover problem.
    # We need to select k triples (x, y, z) such that:
    # - Each x in {0..k-1} appears once.
    # - Each y in {0..k-1} appears once.
    # - Each z in {0..k-1} appears once.
    # This is equivalent to finding a transversal in a 3D Latin Square or 
    # a set of k disjoint edges in a 3-partite hypergraph.
    # Since k <= 22, we use backtracking with pruning (Dancing Links style).

    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(modules):
        adj[x].append((y, z, i))

    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        # Heuristic: pick the x with the fewest available options
        # But for k=22, simple backtracking with state tracking is usually enough
        for y, z, i in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(modules[i])
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for x, y, z in result:
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")

solve()
```
