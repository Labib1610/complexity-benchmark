import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    ops = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        ops.append((x, y, z))
        idx += 3

    # Each operation (x, y, z) contributes x to Alpha[x], y to Beta[y], z to Gamma[z].
    # We need Alpha[i] = i, Beta[i] = i, Gamma[i] = i for all i in 1..k.
    # This means for each i, the set of chosen operations must have exactly one
    # operation with x=i, exactly one with y=i, and exactly one with z=i.
    # This is equivalent to finding a set of k operations that form a 
    # 3-dimensional matching in a k x k x k grid.
    
    # Since k is small (up to 22), we can use backtracking with pruning.
    # We need to pick k operations (x_i, y_i, z_i) such that:
    # {x_1, ..., x_k} = {1, ..., k}
    # {y_1, ..., y_k} = {1, ..., k}
    # {z_1, ..., z_k} = {1, ..., k}
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    res = []
    
    def backtrack(op_idx, count):
        if count == k:
            return True
        if op_idx == t:
            return False
        
        # Option 1: Include ops[op_idx]
        x, y, z = ops[op_idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            res.append(ops[op_idx])
            if backtrack(op_idx + 1, count + 1):
                return True
            res.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Option 2: Skip ops[op_idx]
        # Pruning: if remaining ops are not enough to reach k
        if (t - 1 - op_idx) >= (k - count):
            if backtrack(op_idx + 1, count):
                return True
        
        return False

    if backtrack(0, 0):
        print("YES")
        for op in res:
            print(f"{op[0]} {op[1]} {op[2]}")
    else:
        print("NO")

solve()
