from collections import defaultdict
import sys

def find_matching(k, configs):
    # Build a graph where each configuration is an edge
    # We want to find a perfect matching in a 3D assignment problem
    # This is equivalent to finding a permutation of arms, sensors, modules
    # such that each triple (arm[i], sensor[p[i]], module[q[i]]) is viable
    
    # configs is a list of tuples (arm, sensor, module)
    
    # Create a mapping from (arm, sensor, module) to index
    config_set = set(configs)
    
    # Try all permutations of sensors and modules for arms 1..k
    # Since k <= 22, we need a smarter approach
    
    # This is a tripartite matching problem
    # Let's model it as a maximum bipartite matching problem
    
    # But actually, since we need exactly k matches and use each item once,
    # it's a 3D matching problem. For small k, we can try backtracking.
    
    def backtrack(arm_idx, used_arms, used_sensors, used_modules, selected):
        if arm_idx == k:
            return len(selected) == k
        
        for i in range(len(configs)):
            arm, sensor, module = configs[i]
            if arm == arm_idx + 1 and sensor not in used_sensors and module not in used_modules:
                used_sensors.add(sensor)
                used_modules.add(module)
                selected.append((arm, sensor, module))
                
                if backtrack(arm_idx + 1, used_arms, used_sensors, used_modules, selected):
                    return True
                    
                used_sensors.remove(sensor)
                used_modules.remove(module)
                selected.pop()
        
        return False
    
    # Better approach: build adjacency for each arm
    adj = [[] for _ in range(k + 1)]
    for i, (arm, sensor, module) in enumerate(configs):
        adj[arm].append((sensor, module, i))
    
    def solve():
        # Try to assign each arm to a unique (sensor, module) pair
        match_sensor = [-1] * (k + 1)  # match_sensor[s] = arm that uses sensor s
        match_module = [-1] * (k + 1)  # match_module[m] = arm that uses module m
        result = []
        
        def dfs(arm, visited):
            for sensor, module, idx in adj[arm]:
                if sensor in visited:
                    continue
                visited.add(sensor)
                
                # If sensor or module is free
                if match_sensor[sensor] == -1 and match_module[module] == -1:
                    match_sensor[sensor] = arm
                    match_module[module] = arm
                    result.append((arm, sensor, module))
                    return True
                elif match_sensor[sensor] != -1:
                    # Try to reassign the current owner of sensor
                    prev_arm = match_sensor[sensor]
                    if dfs(prev_arm, visited):
                        match_sensor[sensor] = arm
                        match_module[module] = arm
                        # Need to remove previous entry from result and add new one
                        result[:] = [(a,s,m) for (a,s,m) in result if a != arm]
                        result.append((arm, sensor, module))
                        return True
            return False
        
        # Actually, let's use a simpler backtracking with better pruning
        selected = []
        used_sensors = set()
        used_modules = set()
        
        def backtrack_v2(arm_idx):
            if arm_idx > k:
                return len(selected) == k
            
            if arm_idx == k + 1:
                return len(selected) == k
                
            # For current arm, try all its valid configs
            found = False
            for arm, sensor, module in configs:
                if arm == arm_idx and sensor not in used_sensors and module not in used_modules:
                    used_sensors.add(sensor)
                    used_modules.add(module)
                    selected.append((arm, sensor, module))
                    
                    if backtrack_v2(arm_idx + 1):
                        return True
                        
                    used_sensors.remove(sensor)
                    used_modules.remove(module)
                    selected.pop()
            
            # If no valid config for this arm, but we haven't placed all yet
            return False
        
        # Even simpler: just try all combinations
        from itertools import permutations
        
        # Group configs by arm
        arm_configs = defaultdict(list)
        for arm, sensor, module in configs:
            arm_configs[arm].append((sensor, module))
        
        # Try all permutations of arms 1..k
        for perm in permutations(range(1, k + 1)):
            used_sensors = set()
            used_modules = set()
            selected_config = []
            valid = True
            
            for i, arm_id in enumerate(perm):
                found_match = False
                for sensor, module in arm_configs[arm_id]:
                    if sensor not in used_sensors and module not in used_modules:
                        used_sensors.add(sensor)
                        used_modules.add(module)
                        selected_config.append((arm_id, sensor, module))
                        found_match = True
                        break
                if not found_match:
                    valid = False
                    break
            
            if valid and len(selected_config) == k:
                return selected_config
        
        return None
    
    # Simple recursive backtracking
    def backtrack_simple(index, used_sensors, used_modules, selected):
        if len(selected) == k:
            return selected[:]
        
        if index >= len(configs):
            return None
            
        # Try including current config
        arm, sensor, module = configs[index]
        if sensor not in used_sensors and module not in used_modules:
            used_sensors.add(sensor)
            used_modules.add(module)
            selected.append((arm, sensor, module))
            
            res = backtrack_simple(index + 1, used_sensors, used_modules, selected)
            if res is not None:
                return res
                
            used_sensors.remove(sensor)
            used_modules.remove(module)
            selected.pop()
        
        # Try not including current config
        return backtrack_simple(index + 1, used_sensors, used_modules, selected)
    
    result = backtrack_simple(0, set(), set(), [])
    return result

line = sys.stdin.readline().split()

configs = []
for _ in range(t):

# Solve using backtracking with bitmasks for efficiency
def solve_with_bitmask():
    # Precompute for each arm, which (sensor, module) pairs are valid
    valid_pairs = [set() for _ in range(k + 1)]
    pair_to_configs = {}
    
    for arm, sensor, module in configs:
        valid_pairs[arm].add((sensor, module))
        pair_to_configs[(arm, sensor, module)] = True
    
    # Use backtracking with bitmasks
    def backtrack(arm, sensor_mask, module_mask, selected):
        if arm > k:
            return selected if len(selected) == k else None
        
        # Try all valid pairs for this arm
        for sensor, module in valid_pairs[arm]:
            # Check if sensor and module are not used
            if not (sensor_mask & (1 << sensor)) and not (module_mask & (1 << module)):
                # Mark as used
                new_s_mask = sensor_mask | (1 << sensor)
                new_m_mask = module_mask | (1 << module)
                selected.append((arm, sensor, module))
                
                result = backtrack(arm + 1, new_s_mask, new_m_mask, selected)
                if result is not None:
                    return result
                    
                selected.pop()
        
        return None
    
    # Start backtracking
    return backtrack(1, 0, 0, [])

result = solve_with_bitmask()
