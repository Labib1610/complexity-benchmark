import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    sensors = []
    idx = 2
    for _ in range(t):
        sensors.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    def is_stable(freqs):
        # The condition is that the Vandermonde matrix is non-singular.
        # This happens if and only if all k frequencies are distinct.
        return len(set(freqs)) == k

    # We need to pick k sensors (x_i, y_i, z_i) such that:
    # {x_1, ..., x_k} are distinct, {y_1, ..., y_k} are distinct, {z_1, ..., z_k} are distinct.
    # Since 1 <= x, y, z <= k, this means {x_1, ..., x_k} must be a permutation of {1, ..., k}.
    # This is equivalent to saying the set of x's is {1, ..., k}, same for y and z.
    
    # Backtracking search
    def backtrack(idx, current_sensors, used_x, used_y, used_z):
        if len(current_sensors) == k:
            return current_sensors
        
        for i in range(idx, t):
            x, y, z = sensors[i]
            if not (used_x & (1 << (x - 1))) and not (used_y & (1 << (y - 1))) and not (used_z & (1 << (z - 1))):
                res = backtrack(i + 1, current_sensors + [sensors[i]], 
                                used_x | (1 << (x - 1)), 
                                used_y | (1 << (y - 1)), 
                                used_z | (1 << (z - 1)))
                if res:
                    return res
        return None

    result = backtrack(0, [], 0, 0, 0)
    
    if result:
        print("YES")
        for s in result:
            print(f"{s[0]} {s[1]} {s[2]}")
    else:
        print("NO")

solve()
