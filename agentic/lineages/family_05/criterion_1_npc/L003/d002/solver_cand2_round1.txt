import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        configs.append((x, y, z))
        idx += 3
        
    # We need to select k triples (x_i, y_i, z_i) such that
    # {x_0, ..., x_{k-1}} = {0, ..., k-1}
    # {y_0, ..., y_{k-1}} = {0, ..., k-1}
    # {z_0, ..., z_{k-1}} = {0, ..., k-1}
    
    # This is an Exact Cover problem.
    # Since k <= 22, we can use backtracking with pruning.
    # State: (left_mask, right_mask, keystone_mask, index_in_configs)
    
    # Group configs by left pillar to simplify
    by_left = [[] for _ in range(k)]
    for c in configs:
        by_left[c[0]].append(c)
        
    used_right = [False] * k
    used_key = [False] * k
    result = []
    
    def backtrack(left_idx):
        if left_idx == k:
            return True
        
        for c in by_left[left_idx]:
            _, y, z = c
            if not used_right[y] and not used_key[z]:
                used_right[y] = True
                used_key[z] = True
                result.append(c)
                
                if backtrack(left_idx + 1):
                    return True
                
                result.pop()
                used_right[y] = False
                used_key[z] = False
        return False

    if backtrack(0):
        print("YES")
        for c in result:
            print(f"{c[0]+1} {c[1]+1} {c[2]+1}")
    else:
        print("NO")

solve()
