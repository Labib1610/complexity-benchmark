import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    blueprints = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        blueprints.append((x, y, z))
        idx += 3
        
    # This is a 3D matching problem, which is NP-complete.
    # Given k <= 22, we use backtracking with pruning.
    
    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(blueprints):
        adj[x].append(i)
        
    used_y = [False] * k
    used_z = [False] * k
    result = [None] * k
    
    def backtrack(x):
        if x == k:
            return True
        
        for idx_bp in adj[x]:
            _, y, z = blueprints[idx_bp]
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result[x] = (x + 1, y + 1, z + 1)
                
                if backtrack(x + 1):
                    return True
                
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
