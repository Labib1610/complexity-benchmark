import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    edges = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        edges.append((x, y, z))
        idx += 3
        
    # This is a 3D matching problem, which is NP-complete.
    # Given k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * k
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    # Sort edges to potentially find a solution faster
    # Or just use them as is.
    
    def backtrack(idx_edge, count):
        if count == k:
            return True
        if idx_edge >= t:
            return False
        
        # Optimization: if remaining edges are not enough to fill k
        if (t - idx_edge) < (k - count):
            return False
            
        # Try including edges[idx_edge]
        x, y, z = edges[idx_edge]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append((x + 1, y + 1, z + 1))
            if backtrack(idx_edge + 1, count + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try skipping edges[idx_edge]
        if backtrack(idx_edge + 1, count):
            return True
            
        return False

    if backtrack(0, 0):
        print("YES")
        for line in result:
            print(f"{line[0]} {line[1]} {line[2]}")
    else:
        print("NO")

solve()
