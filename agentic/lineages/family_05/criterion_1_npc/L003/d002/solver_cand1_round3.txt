```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    blueprints = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        blueprints.append((x, y, z))
        idx += 3
        
    # We need to select k blueprints such that each pump, filter, and valve
    # is used exactly once. This is an Exact Cover problem.
    # Since k <= 22, we can use backtracking with bitmasks.
    # However, 22 is too large for simple bitmasking (2^22 is ~4M, but we need 3 sets).
    # We can use Dancing Links or optimized backtracking.
    
    # Group blueprints by pump
    by_pump = [[] for _ in range(k)]
    for b in blueprints:
        by_pump[b[0]].append(b)
        
    used_filter = 0
    used_valve = 0
    result = []
    
    def backtrack(pump_idx):
        if pump_idx == k:
            return True
        
        for b in by_pump[pump_idx]:
            f, v = b[1], b[2]
            if not (used_filter & (1 << f)) and not (used_valve & (1 << v)):
                # Place
                nonlocal used_filter, used_valve
                used_filter |= (1 << f)
                used_valve |= (1 << v)
                result.append(b)
                
                if backtrack(pump_idx + 1):
                    return True
                
                # Backtrack
                result.pop()
                used_filter &= ~(1 << f)
                used_valve &= ~(1 << v)
        return False

    if backtrack(0):
        print("YES")
        for b in result:
            print(f"{b[0]+1} {b[1]+1} {b[2]+1}")
    else:
        print("NO")

solve()
```
