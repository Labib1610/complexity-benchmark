```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # configurations[x][y] = list of z
    # We need to pick (x, y, z) for x=1..k such that all y are distinct and all z are distinct.
    # This is equivalent to finding a permutation p of {1..k} and a permutation q of {1..k}
    # such that (i, p[i], q[i]) is a valid configuration for all i in {1..k}.
    
    adj = [[[] for _ in range(k + 1)] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x][y].append(z)
        idx += 3
        
    # State: (x, mask_y, mask_z)
    # x: current foundation block index (1 to k)
    # mask_y: bitmask of used column shafts
    # mask_z: bitmask of used decorative capitals
    
    memo = {}

    def backtrack(x, mask_y, mask_z):
        if x > k:
            return []
        
        state = (x, mask_y, mask_z)
        if state in memo:
            return memo[state]
        
        for y in range(1, k + 1):
            if not (mask_y & (1 << (y - 1))):
                for z in adj[x][y]:
                    if not (mask_z & (1 << (z - 1))):
                        res = backtrack(x + 1, mask_y | (1 << (y - 1)), mask_z | (1 << (z - 1)))
                        if res is not None:
                            memo[state] = [(x, y, z)] + res
                            return memo[state]
        
        memo[state] = None
        return None

    result = backtrack(1, 0, 0)
    
    if result is not None:
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
```
