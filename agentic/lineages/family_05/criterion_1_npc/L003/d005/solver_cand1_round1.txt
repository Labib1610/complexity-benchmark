import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        configs.append((x, y, z))
        idx += 3
        
    # The requirement is that if we pick k configurations (x_i, y_i, z_i),
    # the set {x_i} must be {1, 2, ..., k}, {y_i} must be {1, 2, ..., k},
    # and {z_i} must be {1, 2, ..., k}.
    # This is equivalent to finding a system of distinct representatives or
    # a perfect matching in a 3D bipartite-like structure (3-dimensional matching).
    # Since k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    # Pre-filter: only keep configs where x, y, z <= k
    valid_configs = [c for c in configs if c[0] <= k and c[1] <= k and c[2] <= k]
    
    result = []
    
    def backtrack(idx_in_configs):
        if len(result) == k:
            return True
        
        if idx_in_configs >= len(valid_configs):
            return False
        
        # Try including valid_configs[idx_in_configs]
        x, y, z = valid_configs[idx_in_configs]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append((x, y, z))
            if backtrack(idx_in_configs + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try skipping
        if backtrack(idx_in_configs + 1):
            return True
            
        return False

    # Optimization: Sort configs to help pruning or use a more efficient search
    # Given k=22, simple backtracking might be slow, but the constraints on 
    # x, y, z being a permutation make it a 3D matching problem.
    if backtrack(0):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
