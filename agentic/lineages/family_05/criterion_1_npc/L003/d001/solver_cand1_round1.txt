import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    a = list(map(int, input_data[1:]))
    
    # We need to find a subsequence of length n/2 such that the sum is maximized.
    # If there are multiple, we need the lexicographically smallest one.
    # This is a classic DP problem.
    # Let dp[i][j] be the maximum sum using j elements from the first i elements.
    # To handle the lexicographical requirement, we can work backwards.
    
    target_count = n // 2
    
    # suffix_max[i][j] = max sum using j elements from a[i:]
    # To reconstruct the lexicographically smallest, we need to know if a[i]
    # can be part of an optimal solution.
    
    dp = [[-float('inf')] * (target_count + 1) for _ in range(n + 1)]
    dp[n][0] = 0
    
    for i in range(n - 1, -1, -1):
        for j in range(target_count + 1):
            # Option 1: Don't include a[i]
            dp[i][j] = dp[i+1][j]
            # Option 2: Include a[i]
            if j > 0 and dp[i+1][j-1] != -float('inf'):
                dp[i][j] = max(dp[i][j], a[i] + dp[i+1][j-1])
                
    # Reconstruct
    res = []
    curr_sum = dp[0][target_count]
    curr_count = target_count
    
    for i in range(n):
        if curr_count == 0:
            break
        
        # Can we pick a[i]?
        # If we pick a[i], we need the remaining sum to be curr_sum - a[i]
        # using curr_count - 1 elements from a[i+1:]
        can_pick = False
        if curr_count > 0 and dp[i+1][curr_count-1] != -float('inf'):
            if a[i] + dp[i+1][curr_count-1] == curr_sum:
                can_pick = True
        
        # Can we skip a[i]?
        can_skip = False
        if dp[i+1][curr_count] == curr_sum:
            can_skip = True
            
        # To get lexicographically smallest, we prefer smaller values.
        # If we can pick a[i] and skip a[i], we compare a[i] with the 
        # first element of the sequence if we skip a[i].
        # Actually, the greedy choice is: if we can pick a[i] and it's 
        # part of an optimal solution, we check if picking it is better 
        # than skipping it.
        
        if can_pick and can_skip:
            # If we pick a[i], the sequence starts with a[i].
            # If we skip a[i], the sequence starts with something from a[i+1:].
            # We need to find the first element of the sequence if we skip.
            # This is complex. Simpler: compare a[i] with the first element 
            # of the optimal sequence starting from i+1.
            
            # Find the first element of the optimal sequence if we skip a[i]
            # This is the smallest index k > i such that a[k] is part of an optimal sum.
            
            # Actually, just compare a[i] with the value we would pick if we skip.
            # Let's find the first element if we skip.
            temp_idx = i + 1
            temp_count = curr_count
            temp_sum = curr_sum
            while temp_idx < n:
                if temp_count > 0 and dp[temp_idx+1][temp_count-1] != -float('inf') and a[temp_idx] + dp[temp_idx+1][temp_count-1] == temp_sum:
                    first_if_skip = a[temp_idx]
                    break
                temp_idx += 1
            
            if a[i] <= first_if_skip:
                res.append(a[i])
                curr_sum -= a[i]
                curr_count -= 1
            else:
                curr_sum = dp[i+1][curr_count]
        elif can_pick:
            res.append(a[i])
            curr_sum -= a[i]
            curr_count -= 1
        else:
            curr_sum = dp[i+1][curr_count]
            
    print(*(res))

solve()
