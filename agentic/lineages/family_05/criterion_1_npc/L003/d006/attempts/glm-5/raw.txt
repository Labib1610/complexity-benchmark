import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    k = int(next(it))
    t = int(next(it))
    
    components = []
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        components.append((x - 1, y - 1, z - 1))
        
    if k == 1:
        if t >= 1:
            print("YES")
            print("1 1 1")
        else:
            print("NO")
        return

    # dp[mask] = list of (prev_mask, component_index)
    dp = [None] * (1 << k)
    dp[0] = []
    
    for i, comp in enumerate(components):
        x, y, z = comp
        for mask in range((1 << k) - 1, -1, -1):
            if dp[mask] is not None:
                new_mask = mask | (1 << x) | (1 << y) | (1 << z)
                if dp[new_mask] is None:
                    dp[new_mask] = dp[mask] + [(mask, i)]
                    if new_mask == (1 << k) - 1:
                        break
        if dp[(1 << k) - 1] is not None:
            break

    full_mask = (1 << k) - 1
    if dp[full_mask] is None:
        print("NO")
        return

    # Reconstruct chosen components
    chosen_indices = []
    curr_mask = full_mask
    for prev_mask, idx in reversed(dp[full_mask]):
        chosen_indices.append(idx)
        curr_mask = prev_mask
        
    # We have k components, but we need to ensure exactly k are selected.
    # The problem guarantees we can pick at most one of each, and we need exactly k.
    # The components found might be less than k if some components didn't add new bits.
    # We can add any components that don't change the full_mask.
    for i, comp in enumerate(components):
        if len(chosen_indices) == k:
            break
        if i not in chosen_indices:
            x, y, z = comp
            if (full_mask | (1 << x) | (1 << y) | (1 << z)) == full_mask:
                chosen_indices.append(i)

    if len(chosen_indices) < k:
        print("NO")
        return

    print("YES")
    for idx in chosen_indices:
        x, y, z = components[idx]
        print(x + 1, y + 1, z + 1)

if __name__ == '__main__':
    solve()
