```python
import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    components = []
    idx = 2
    for _ in range(t):
        components.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    if t < k:
        print("NO")
        return

    # The condition is: (Alpha_div) * (Beta_div) * (Gamma_div) >= k^3
    # Since each diversity is at most k, the only way to satisfy this is if
    # Alpha_div = k, Beta_div = k, and Gamma_div = k.
    # This means we need to pick k components such that all Alpha ratings 1..k are present,
    # all Beta ratings 1..k are present, and all Gamma ratings 1..k are present.
    
    # This is a variation of the Exact Cover problem or hitting set, but with k <= 22.
    # Given the constraints and the nature of the problem, we can use a randomized approach
    # or backtracking. Since we need to pick exactly k components, we are looking for a 
    # subset of size k that covers all values.
    
    import random
    
    # If k is small, we can try to find a subset.
    # If k is large, the problem might be constrained.
    # Let's try a randomized search (Las Vegas/Monte Carlo).
    
    for _ in range(20000):
        if t == k:
            subset = components
        else:
            subset = random.sample(components, k)
            
        alpha_set = set()
        beta_set = set()
        gamma_set = set()
        for c in subset:
            alpha_set.add(c[0])
            beta_set.add(c[1])
            gamma_set.add(c[2])
            
        if len(alpha_set) == k and len(beta_set) == k and len(gamma_set) == k:
            print("YES")
            for c in subset:
                print(f"{c[0]} {c[1]} {c[2]}")
            return

    print("NO")

solve()
```
