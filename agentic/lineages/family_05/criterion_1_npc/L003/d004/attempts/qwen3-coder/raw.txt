from itertools import permutations

def solve():
    k, t = map(int, input().split())
    
    configs = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        configs.append((x, y, z))
    
    # We need to select k configurations such that all x, y, z values are unique across selected configs
    # This is essentially finding a matching in a tripartite graph
    # We can model this as a bipartite matching problem by considering x, y, z as separate sets
    
    # Try all combinations of k configurations
    # Since k <= 22, and we're looking for a permutation, we can try all permutations
    # But t could be up to k^3, so t could be large
    # Better approach: try all possible subsets of size k and check if they form a valid matching
    
    # For small k, we can enumerate all possible ways to choose k configurations
    # and check if their x, y, z values are all distinct
    
    # Actually, since we want each x, y, z to be unique, this is like finding a perfect matching
    # in a 3-dimensional matching problem
    
    # Let's think of it differently:
    # We have t triples (x, y, z)
    # We want to select k of them such that all x's are distinct, all y's are distinct, all z's are distinct
    # This means we're looking for a subset of k triples where the x-coordinates are a permutation of 1..k,
    # y-coordinates are a permutation of 1..k, and z-coordinates are a permutation of 1..k
    
    # So we're looking for a permutation of 1..k for x's, 1..k for y's, and 1..k for z's
    # such that each (x[i], y[i], z[i]) is in our config list
    
    # Generate all possible permutations for y and z coordinates
    # For each permutation of y and z, check if there exists corresponding x values
    
    ids = list(range(1, k+1))
    
    # Create a set of available configs for fast lookup
    config_set = set(configs)
    
    # Try all permutations for y coordinates
    for p_y in permutations(ids):
        # Try all permutations for z coordinates
        for p_z in permutations(ids):
            # Check if we can find corresponding x coordinates
            found = True
            selected_configs = []
            
            # For each position i, we need a config (x_i, p_y[i], p_z[i])
            for i in range(k):
                y_val = p_y[i]
                z_val = p_z[i]
                
                # Find if there exists an x such that (x, y_val, z_val) is in configs
                # and x hasn't been used yet
                found_config = False
                for x_val in ids:
                    if (x_val, y_val, z_val) in config_set:
                        selected_configs.append((x_val, y_val, z_val))
                        found_config = True
                        break
                
                if not found_config:
                    found = False
                    break
            
            # Check if all x values in selected_configs are distinct
            if found:
                x_values = [cfg[0] for cfg in selected_configs]
                if len(set(x_values)) == k:
                    print("YES")
                    for cfg in selected_configs:
                        print(cfg[0], cfg[1], cfg[2])
                    return
    
    # More efficient approach - try all possible combinations
    # Since k <= 22, but we only need k items, let's try a different method
    
    # Use backtracking to find a valid assignment
    def backtrack(pos, used_x, used_y, used_z, selected):
        if pos == k:
            return True
        
        for config in configs:
            x, y, z = config
            if x not in used_x and y not in used_y and z not in used_z:
                used_x.add(x)
                used_y.add(y)
                used_z.add(z)
                selected.append(config)
                
                if backtrack(pos + 1, used_x, used_y, used_z, selected):
                    return True
                    
                used_x.remove(x)
                used_y.remove(y)
                used_z.remove(z)
                selected.pop()
        
        return False
    
    used_x = set()
    used_y = set()
    used_z = set()
    selected = []
    
    if backtrack(0, used_x, used_y, used_z, selected):
        print("YES")
        for config in selected:
            print(config[0], config[1], config[2])
    else:
        print("NO")

solve()
