import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    profiles = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        profiles.append((x, y, z))
        idx += 3

    # We need to select k profiles (x_i, y_i, z_i) such that all x_i are distinct,
    # all y_i are distinct, and all z_i are distinct.
    # This is equivalent to finding a perfect matching in a 3-partite hypergraph,
    # or more simply, a set of k triples (x, y, z) such that x, y, z are permutations of 0..k-1.
    # Since k <= 22, we can use dynamic programming with bitmask.
    # dp[mask_y][mask_z] = the index of the profile that completes this state, or -1.
    # We iterate through x = 0, 1, ..., k-1.
    
    # dp[mask_y][mask_z] stores the last profile used to reach this state
    # To reconstruct, we store (prev_mask_y, prev_mask_z, profile_index)
    dp = {}
    dp[(0, 0)] = None
    
    # Group profiles by x
    by_x = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(profiles):
        by_x[x].append((y, z, i))
        
    for x in range(k):
        new_dp = {}
        for (mask_y, mask_z), _ in dp.items():
            for y, z, i in by_x[x]:
                if not (mask_y & (1 << y)) and not (mask_z & (1 << z)):
                    new_state = (mask_y | (1 << y), mask_z | (1 << z))
                    if new_state not in new_dp:
                        new_dp[new_state] = (mask_y, mask_z, i)
        dp = new_dp
        if not dp:
            print("NO")
            return

    final_state = ((1 << k) - 1, (1 << k) - 1)
    if final_state in dp:
        print("YES")
        res = []
        curr = final_state
        while dp[curr] is not None:
            prev_y, prev_z, idx = dp[curr]
            res.append(profiles[idx])
            curr = (prev_y, prev_z)
        for p in res:
            print(f"{p[0]+1} {p[1]+1} {p[2]+1}")
    else:
        print("NO")

solve()
