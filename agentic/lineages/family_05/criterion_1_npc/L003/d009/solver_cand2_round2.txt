```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Group modules by frequency
    freq_groups = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        freq_groups[x].append((y, z))
        idx += 3
        
    # We need to pick one module (y, z) for each frequency x from 1 to k
    # such that all y are distinct and all z are distinct.
    # This is equivalent to finding a system of distinct representatives
    # or a perfect matching in a bipartite graph, but here we have two constraints.
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    result = []
    
    def backtrack(freq):
        if freq > k:
            return True
        
        for y, z in freq_groups[freq]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((freq, y, z))
                
                if backtrack(freq + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    # Sort freq_groups by size to optimize backtracking (MRV heuristic)
    # But we must maintain the frequency order 1..k.
    # Actually, just simple backtracking is fine for k=22.
    
    if backtrack(1):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
```
