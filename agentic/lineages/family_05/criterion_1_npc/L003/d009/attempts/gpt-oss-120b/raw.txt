```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    modules = []
    by_x = [[] for _ in range(k + 1)]
    for idx in range(t):
        x = int(next(it)); y = int(next(it)); z = int(next(it))
        modules.append((x, y, z))
        by_x[x].append((y - 1, z - 1, idx))   # zero‑based for bit masks

    # if any stage has no candidates, impossible immediately
    for i in range(1, k + 1):
        if not by_x[i]:
            print("NO")
            return

    # order stages by fewest candidates to improve backtracking
    stages = list(range(1, k + 1))
    stages.sort(key=lambda i: len(by_x[i]))

    used_amp = 0
    used_phase = 0
    chosen = [None] * k  # index by original stage number-1

    sys.setrecursionlimit(10000)

    def dfs(pos: int) -> bool:
        nonlocal used_amp, used_phase
        if pos == k:
            return True
        stage = stages[pos]
        for a, p, idx in by_x[stage]:
            mask_a = 1 << a
            mask_p = 1 << p
            if (used_amp & mask_a) or (used_phase & mask_p):
                continue
            # choose
            used_amp ^= mask_a
            used_phase ^= mask_p
            chosen[stage - 1] = idx
            if dfs(pos + 1):
                return True
            # backtrack
            used_amp ^= mask_a
            used_phase ^= mask_p
            chosen[stage - 1] = None
        return False

    if dfs(0):
        print("YES")
        out = []
        for idx in chosen:
            x, y, z = modules[idx]
            out.append(f"{x} {y} {z}")
        sys.stdout.write("\n".join(out))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
