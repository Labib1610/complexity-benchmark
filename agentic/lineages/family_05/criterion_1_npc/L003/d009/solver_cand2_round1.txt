import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    modules = []
    idx = 2
    for _ in range(t):
        modules.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3
        
    target = (5**(k + 1) - 5) // 4
    
    # The condition sum(5^v_i) = (5^(k+1) - 5) / 4 is equivalent to
    # sum(5^(v_i - 1)) = (5^k - 1) / 4 = 1 + 5 + 25 + ... + 5^(k-1)
    # Since 1 <= v_i <= k, this is a base-5 representation problem.
    # Specifically, if we count how many modules have value v, say c_v,
    # then sum(c_v * 5^(v-1)) = sum(1 * 5^(v-1)) for v=1 to k.
    # This implies c_v = 1 for all v in {1, ..., k}.
    # So we need exactly one module with Frequency=v, one with Amplitude=v,
    # and one with Phase=v for each v in {1, ..., k}.
    
    # This is equivalent to finding a system of distinct representatives or 
    # a matching in a specific graph. Since we need to pick k modules such that
    # the set of frequencies is {1..k}, amplitudes {1..k}, and phases {1..k}.
    
    # This is a 3D matching problem, which is NP-hard. However, k is small (22).
    # We can use backtracking with pruning or randomized search.
    
    def find_solution():
        # We need to select k modules (x_i, y_i, z_i) such that
        # {x_1, ..., x_k} = {1, ..., k}
        # {y_1, ..., y_k} = {1, ..., k}
        # {z_1, ..., z_k} = {1, ..., k}
        
        # Use backtracking with bitmasks
        used_x = 0
        used_y = 0
        used_z = 0
        selected = []
        
        def backtrack(idx_m):
            if len(selected) == k:
                return True
            if idx_m >= t:
                return False
            
            # Try including modules[idx_m]
            x, y, z = modules[idx_m]
            if not (used_x & (1 << (x-1))) and not (used_y & (1 << (y-1))) and not (used_z & (1 << (z-1))):
                selected.append(modules[idx_m])
                nonlocal used_x, used_y, used_z
                used_x |= (1 << (x-1))
                used_y |= (1 << (y-1))
                used_z |= (1 << (z-1))
                if backtrack(idx_m + 1):
                    return True
                used_x &= ~(1 << (x-1))
                used_y &= ~(1 << (y-1))
                used_z &= ~(1 << (z-1))
                selected.pop()
            
            # Try skipping
            if backtrack(idx_m + 1):
                return True
            return False

        # Shuffle to avoid adversarial cases
        random.shuffle(modules)
        return backtrack(0), selected

    # Given the constraints and the nature of the problem, 
    # a simple backtracking might be slow. Let's use a more robust approach.
    # Actually, the condition c_v = 1 for all v is necessary.
    
    found, result = find_solution()
    if found:
        print("YES")
        for m in result:
            print(f"{m[0]} {m[1]} {m[2]}")
    else:
        print("NO")

solve()
