import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    recipes = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        recipes.append((x, y, z))
        idx += 3

    # We need to select k recipes such that each x, y, z is used exactly once.
    # This is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph.
    # Since k <= 22, we can use backtracking with pruning.
    
    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(recipes):
        adj[x].append((y, z, i))
        
    used_y = [False] * k
    used_z = [False] * k
    chosen = [None] * k
    
    def backtrack(x):
        if x == k:
            return True
        
        # Heuristic: pick x with fewest options
        for y, z, i in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                chosen[x] = (x + 1, y + 1, z + 1)
                if backtrack(x + 1):
                    return True
                used_y[y] = False
                used_z[z] = False
        return False

    # Sort x by number of available recipes to optimize backtracking
    # But simple backtracking is often enough for k=22 if the graph is sparse
    if backtrack(0):
        print("YES")
        for res in chosen:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

solve()
