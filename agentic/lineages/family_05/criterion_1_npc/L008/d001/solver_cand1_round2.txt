import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    pathways = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        pathways.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem.
    # We need to select k triplets (x, y, z) such that:
    # - Each x in {0..k-1} appears exactly once.
    # - Each y in {0..k-1} appears exactly once.
    # - Each z in {0..k-1} appears exactly once.
    
    # Since k <= 22, we can use backtracking with pruning.
    # Group pathways by origin x
    by_origin = [[] for _ in range(k)]
    for p in pathways:
        by_origin[p[0]].append(p)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(origin_idx):
        if origin_idx == k:
            return True
        
        for p in by_origin[origin_idx]:
            _, y, z = p
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(p)
                
                if backtrack(origin_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for p in result:
            print(f"{p[0]+1} {p[1]+1} {p[2]+1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
