from collections import defaultdict
import sys

def find_matching(k, recipes):
    # Build bipartite graph: base liquids -> (catalyst, essence) pairs
    # This is essentially a 3D matching problem, but we can reduce it
    
    # We'll model this as a matching problem:
    # Create a graph where we need to match each base liquid to a unique (catalyst, essence) pair
    # And each catalyst and essence can only be used once
    
    # Use backtracking / brute force since k <= 22, but let's try max flow approach for efficiency
    
    # Actually, let's think differently:
    # This is equivalent to finding a perfect matching in a 3-partite 3-uniform hypergraph
    # But since k is small, we can use backtracking
    
    # Let's represent the problem as finding a subset of recipes of size k
    # such that all x, y, z values are distinct across the selected recipes
    
    def backtrack(index, selected, used_x, used_y, used_z):
        if len(selected) == k:
            return selected[:]
        
        if index >= len(recipes):
            return None
            
        # Pruning: if not enough recipes left to complete selection
        if len(selected) + (len(recipes) - index) < k:
            return None
            
        x, y, z = recipes[index]
        
        # Try including current recipe
        if x not in used_x and y not in used_y and z not in used_z:
            selected.append((x, y, z))
            used_x.add(x)
            used_y.add(y)
            used_z.add(z)
            
            result = backtrack(index + 1, selected, used_x, used_y, used_z)
            if result is not None:
                return result
                
            selected.pop()
            used_x.remove(x)
            used_y.remove(y)
            used_z.remove(z)
        
        # Try not including current recipe
        return backtrack(index + 1, selected, used_x, used_y, used_z)
    
    # Sort recipes to potentially improve pruning
    result = backtrack(0, [], set(), set(), set())
    return result

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    recipes = []
    idx = 2
    for _ in range(t):
        x = int(data[idx])
        y = int(data[idx+1])
        z = int(data[idx+2])
        recipes.append((x, y, z))
        idx += 3
    
    # Special case: if k=0, which shouldn't happen based on constraints, but just in case
    if k == 0:
        print("YES")
        return
        
    selection = find_matching(k, recipes)
    
    if selection is None:
        print("NO")
    else:
        print("YES")
        for x, y, z in selection:
            print(x, y, z)

if __name__ == "__main__":
    main()
