We need to find a perfect matching in a 3-partite hypergraph where each vertex (ingredient) must be used exactly once.  
Since k ≤ 22, we can use a bitmask DP approach.

We can represent the state by the set of used base liquids and used catalysts, and then check if we can assign essences accordingly.

Here’s the implementation:

```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # Store recipes by (base, catalyst) -> essence
    recipes = {}
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        recipes[(x, y)] = z
    
    # DP: dp[mask_bases][mask_cats] = mask_essences or None
    # We'll store chosen recipe for reconstruction
    dp = [[None] * (1 << k) for _ in range(1 << k)]
    dp[0][0] = 0  # mask of used essences
    
    # Iterate over all base masks
    for mask_b in range(1 << k):
        for mask_c in range(1 << k):
            if dp[mask_b][mask_c] is None:
                continue
            # Try to add a new base
            for b in range(k):
                if mask_b & (1 << b):
                    continue
                # Try to add a new catalyst
                for c in range(k):
                    if mask_c & (1 << c):
                        continue
                    if (b, c) in recipes:
                        z = recipes[(b, c)]
                        if dp[mask_b][mask_c] & (1 << z):
                            continue
                        new_mask_b = mask_b | (1 << b)
                        new_mask_c = mask_c | (1 << c)
                        new_mask_e = dp[mask_b][mask_c] | (1 << z)
                        if dp[new_mask_b][new_mask_c] is None:
                            dp[new_mask_b][new_mask_c] = new_mask_e
                        # We don't need to store more than one way; any works
    
    full = (1 << k) - 1
    if dp[full][full] is None:
        print("NO")
        return
    
    # Reconstruct
    chosen = []
    mask_b = full
    mask_c = full
    mask_e = dp[full][full]
    
    while mask_b:
        found = False
        for b in range(k):
            if not (mask_b & (1 << b)):
                continue
            for c in range(k):
                if not (mask_c & (1 << c)):
                    continue
                if (b, c) in recipes:
                    z = recipes[(b, c)]
                    if mask_e & (1 << z):
                        # Check if removing b,c,z leads to a valid previous state
                        prev_mask_b = mask_b ^ (1 << b)
                        prev_mask_c = mask_c ^ (1 << c)
                        prev_mask_e = mask_e ^ (1 << z)
                        if dp[prev_mask_b][prev_mask_c] == prev_mask_e:
                            chosen.append((b+1, c+1, z+1))
                            mask_b = prev_mask_b
                            mask_c = prev_mask_c
                            mask_e = prev_mask_e
                            found = True
                            break
            if found:
                break
    
    print("YES")
    for x, y, z in reversed(chosen):
        print(x, y, z)

if __name__ == "__main__":
    solve()
```
