import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    recipes = []
    idx = 2
    for _ in range(t):
        x = int(data[idx]) - 1
        y = int(data[idx+1]) - 1
        z = int(data[idx+2]) - 1
        idx += 3
        recipes.append((x, y, z))
    
    # Group recipes by base liquid x
    by_x = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(recipes):
        by_x[x].append((y, z, i))
    
    # DP over base liquids 0..k-1
    # State: bitmask of used catalysts (y), value: (bitmask of used essences (z), recipe indices)
    # Since k <= 22, bitmask fits in int.
    # We need to track both y-mask and z-mask, but y-mask is the DP key.
    # z-mask is stored alongside.
    
    # dp: dict mapping y_mask -> (z_mask, list_of_recipe_indices)
    dp = {0: (0, [])}
    
    for x in range(k):
        new_dp = {}
        # For each existing state, try adding one recipe with base x
        for y_mask, (z_mask, chosen) in dp.items():
            for y, z, rec_idx in by_x[x]:
                bit_y = 1 << y
                bit_z = 1 << z
                if (y_mask & bit_y) == 0 and (z_mask & bit_z) == 0:
                    ny = y_mask | bit_y
                    nz = z_mask | bit_z
                    if ny not in new_dp:
                        new_dp[ny] = (nz, chosen + [rec_idx])
        dp = new_dp
        if not dp:
            break
    
    # After processing all k base liquids, any state in dp has used all k catalysts and k essences
    # (since we used exactly one recipe per base liquid, and no duplicates in y or z)
    # We need a state where y_mask has all k bits set (which means z_mask also has all k bits set)
    full_mask = (1 << k) - 1
    if full_mask in dp:
        z_mask, chosen = dp[full_mask]
        out = ["YES"]
        for rec_idx in chosen:
            x, y, z = recipes[rec_idx]
            out.append(f"{x+1} {y+1} {z+1}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

solve()
