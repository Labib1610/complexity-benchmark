import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # We need to select k points (x_i, y_i, z_i) such that all x_i are distinct,
    # all y_i are distinct, and all z_i are distinct.
    # This is equivalent to finding a perfect matching in a 3-partite hypergraph,
    # or more simply, a set of k edges in a 3D grid.
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    points = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        points.append((x, y, z))
        idx += 3
        
    # Group points by x-coordinate to simplify the search
    by_x = [[] for _ in range(k)]
    for p in points:
        by_x[p[0]].append(p)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x):
        if x == k:
            return True
        
        for p in by_x[x]:
            _, y, z = p
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(p)
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for p in result:
            print(f"{p[0] + 1} {p[1] + 1} {p[2] + 1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
