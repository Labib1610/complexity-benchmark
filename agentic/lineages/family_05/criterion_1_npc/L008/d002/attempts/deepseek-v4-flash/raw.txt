We need to select exactly one sensor per X, Y, and Z plane.  
This is equivalent to finding a perfect matching in a 3D matching problem, which is NP-hard in general.  
However, since k ≤ 22, we can use a backtracking approach with pruning, or a bitmask DP.

We can model it as:  
We need to assign each X-plane a unique Y and Z such that the triple exists in the given list.

We can use recursion with bitmasks for Y and Z planes used.

Here’s the implementation:

```python
import sys

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    points = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        points.append((x-1, y-1, z-1))  # 0-indexed

    # Group points by x
    by_x = [[] for _ in range(k)]
    for x, y, z in points:
        by_x[x].append((y, z))

    # Sort each group to help pruning
    for i in range(k):
        by_x[i].sort()

    # We'll do backtracking
    used_y = 0
    used_z = 0
    chosen = [None] * k

    # Precompute for each x, the list of (y,z) pairs
    # We'll try to assign each x a unique y and z

    # To speed up, we can use recursion with memoization on (x, used_y, used_z)
    # But k <= 22, so state space is 2^22 * 2^22 too big.
    # Instead, we use backtracking with pruning.

    # We'll order x by number of options (smallest first) to reduce branching
    order = sorted(range(k), key=lambda i: len(by_x[i]))

    def dfs(idx):
        if idx == k:
            return True
        x = order[idx]
        for y, z in by_x[x]:
            if not (used_y >> y) & 1 and not (used_z >> z) & 1:
                chosen[x] = (y, z)
                nonlocal used_y, used_z
                used_y |= (1 << y)
                used_z |= (1 << z)
                if dfs(idx + 1):
                    return True
                used_y ^= (1 << y)
                used_z ^= (1 << z)
                chosen[x] = None
        return False

    # We need nonlocal in nested function, so define used_y, used_z as lists or use a class
    # Simpler: use a list for used_y and used_z
    used_y = [0]
    used_z = [0]

    def dfs(idx):
        if idx == k:
            return True
        x = order[idx]
        for y, z in by_x[x]:
            if not (used_y[0] >> y) & 1 and not (used_z[0] >> z) & 1:
                chosen[x] = (y, z)
                used_y[0] |= (1 << y)
                used_z[0] |= (1 << z)
                if dfs(idx + 1):
                    return True
                used_y[0] ^= (1 << y)
                used_z[0] ^= (1 << z)
                chosen[x] = None
        return False

    if dfs(0):
        print("YES")
        for x in range(k):
            y, z = chosen[x]
            print(x+1, y+1, z+1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
