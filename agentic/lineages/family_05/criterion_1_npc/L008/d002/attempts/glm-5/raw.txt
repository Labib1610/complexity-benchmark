import sys
sys.setrecursionlimit(100000)

def solve():
    input_data = sys.stdin.read().split()
    idx = 0
    k = int(input_data[idx]); idx += 1
    t = int(input_data[idx]); idx += 1
    
    adj = [[[] for _ in range(k+1)] for _ in range(k+1)]
    for _ in range(t):
        x = int(input_data[idx]); idx += 1
        y = int(input_data[idx]); idx += 1
        z = int(input_data[idx]); idx += 1
        adj[x][y].append(z)
    
    if k == 0:
        print("YES")
        return
    
    y_used = [False] * (k+1)
    z_used = [False] * (k+1)
    result = [None] * k
    
    def backtrack(x):
        if x == k + 1:
            return True
        for y in range(1, k + 1):
            if y_used[y]:
                continue
            if not adj[x][y]:
                continue
            y_used[y] = True
            for z in adj[x][y]:
                if z_used[z]:
                    continue
                z_used[z] = True
                result[x - 1] = (x, y, z)
                if backtrack(x + 1):
                    return True
                z_used[z] = False
            y_used[y] = False
        return False
    
    if backtrack(1):
        print("YES")
        for x, y, z in result:
            print(f"{x} {y} {z}")
    else:
        print("NO")
