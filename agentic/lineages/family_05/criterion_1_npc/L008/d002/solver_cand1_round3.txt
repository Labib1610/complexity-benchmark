```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    coords = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        coords.append((x, y, z))
        idx += 3
        
    # We need to select k coordinates (x_i, y_i, z_i) such that
    # {x_i} = {0..k-1}, {y_i} = {0..k-1}, {z_i} = {0..k-1}.
    # This is equivalent to finding a perfect matching in a 3D bipartite-like structure.
    # Since k <= 22, we can use backtracking with pruning.
    
    adj = [[] for _ in range(k)]
    for x, y, z in coords:
        adj[x].append((y, z))
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x):
        if x == k:
            return True
        
        for y, z in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    # Optimization: sort by degree or use MRV (Minimum Remaining Values)
    # But for k=22, simple backtracking might be slow. 
    # However, the constraints on x, y, z are very restrictive.
    if backtrack(0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

solve()
```
