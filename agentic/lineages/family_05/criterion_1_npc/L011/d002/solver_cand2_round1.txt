import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    if k > 22:
        print("NO")
        return
    
    profiles = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        profiles.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem (3D matching).
    # Since k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    result = []
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for i in range(start_idx, t):
            x, y, z = profiles[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = True
                used_y[y] = True
                used_z[z] = True
                result.append(profiles[i])
                
                if backtrack(i + 1, count + 1):
                    return True
                
                result.pop()
                used_x[x] = False
                used_y[y] = False
                used_z[z] = False
        return False

    # Optimization: sort profiles or use bitmasks if k was larger,
    # but for k=22, simple backtracking with state tracking is sufficient
    # given the constraints and the nature of 3D matching.
    if backtrack(0, 0):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
