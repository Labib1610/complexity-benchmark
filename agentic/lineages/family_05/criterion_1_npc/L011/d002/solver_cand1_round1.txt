```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    vectors = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        vectors.append((x, y, z))
        idx += 3

    # We need to select k vectors such that all x are distinct, all y are distinct,
    # and all z are distinct. This is equivalent to finding a transversal of size k
    # in a 3D matching problem, which is NP-hard. Given k <= 22, we use backtracking
    # with pruning.
    
    # Group vectors by x to simplify the search
    by_x = [[] for _ in range(k)]
    for v in vectors:
        by_x[v[0]].append(v)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x_idx):
        if x_idx == k:
            return True
        
        for v in by_x[x_idx]:
            _, y, z = v
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(v)
                
                if backtrack(x_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for v in result:
            print(f"{v[0]+1} {v[1]+1} {v[2]+1}")
    else:
        print("NO")

solve()
```
