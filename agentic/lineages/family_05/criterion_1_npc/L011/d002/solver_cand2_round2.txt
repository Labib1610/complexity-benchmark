import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    vectors = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        vectors.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    # To optimize, we use bitmasks for the three rings.
    
    used_x = 0
    used_y = 0
    used_z = 0
    
    # Group vectors by x to speed up search
    by_x = [[] for _ in range(k)]
    for v in vectors:
        by_x[v[0]].append(v)
        
    result = []
    
    def backtrack(x_idx):
        if x_idx == k:
            return True
        
        # If x_idx is already used, skip to next
        if (used_x >> x_idx) & 1:
            return backtrack(x_idx + 1)
        
        for v in by_x[x_idx]:
            _, y, z = v
            if not ((used_y >> y) & 1) and not ((used_z >> z) & 1):
                # Place
                used_y |= (1 << y)
                used_z |= (1 << z)
                result.append(v)
                
                if backtrack(x_idx + 1):
                    return True
                
                # Backtrack
                result.pop()
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        
        return False

    if backtrack(0):
        print("YES")
        for v in result:
            print(f"{v[0]+1} {v[1]+1} {v[2]+1}")
    else:
        print("NO")

solve()
