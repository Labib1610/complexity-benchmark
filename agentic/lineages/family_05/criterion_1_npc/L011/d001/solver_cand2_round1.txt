import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    blueprints = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        blueprints.append((x, y, z))
        idx += 3
        
    # This is a 3D matching problem, which is NP-complete.
    # Given k <= 22, we can use backtracking with pruning.
    
    adj = [[] for _ in range(k + 1)]
    for i, (x, y, z) in enumerate(blueprints):
        adj[x].append(i)
        
    used_beta = [False] * (k + 1)
    used_gamma = [False] * (k + 1)
    result = []
    
    def backtrack(alpha_idx):
        if alpha_idx > k:
            return True
        
        for b_idx in adj[alpha_idx]:
            _, y, z = blueprints[b_idx]
            if not used_beta[y] and not used_gamma[z]:
                used_beta[y] = True
                used_gamma[z] = True
                result.append(blueprints[b_idx])
                
                if backtrack(alpha_idx + 1):
                    return True
                
                result.pop()
                used_beta[y] = False
                used_gamma[z] = False
        return False

    if backtrack(1):
        print("YES")
        for b in result:
            print(f"{b[0]} {b[1]} {b[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
