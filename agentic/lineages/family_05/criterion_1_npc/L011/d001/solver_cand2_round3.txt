```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    cells = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        cells.append((x - 1, y - 1, z - 1))
        idx += 3
        
    # This is a 3D matching problem, which is NP-hard.
    # Given k <= 22, we can use backtracking with pruning.
    # Represent cells as adjacency list: x -> list of (y, z)
    adj = [[] for _ in range(k)]
    for x, y, z in cells:
        adj[x].append((y, z))
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x):
        if x == k:
            return True
        
        for y, z in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    # Heuristic: sort x by number of available choices
    # But for k=22, simple backtracking might be slow.
    # Let's try standard backtracking first.
    if backtrack(0):
        print("YES")
        for cell in result:
            print(f"{cell[0]} {cell[1]} {cell[2]}")
    else:
        print("NO")

solve()
```
