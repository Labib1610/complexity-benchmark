```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    profiles = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        profiles.append((x, y, z))
        idx += 3
        
    # We need to select k triples (x_i, y_i, z_i) such that
    # {x_i} = {0..k-1}, {y_i} = {0..k-1}, {z_i} = {0..k-1}
    # This is equivalent to finding a perfect matching in a 3D matching problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(profiles):
        adj[x].append((y, z, i))
        
    used_y = [False] * k
    used_z = [False] * k
    result = [None] * k
    
    def backtrack(x):
        if x == k:
            return True
        
        for y, z, i in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result[x] = (x + 1, y + 1, z + 1)
                
                if backtrack(x + 1):
                    return True
                
                used_y[y] = False
                used_z[z] = False
        return False

    # Sort adj to try to find a solution faster
    for i in range(k):
        adj[i].sort()
        
    if backtrack(0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
