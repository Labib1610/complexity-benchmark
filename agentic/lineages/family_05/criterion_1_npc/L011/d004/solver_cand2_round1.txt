import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    elements = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        elements.append((x, y, z))
        idx += 3

    # We need to select k elements (x_i, y_i, z_i) such that:
    # {x_0, ..., x_{k-1}} = {0, ..., k-1}
    # {y_0, ..., y_{k-1}} = {0, ..., k-1}
    # {z_0, ..., z_{k-1}} = {0, ..., k-1}
    # And all x_i are distinct, all y_i are distinct, all z_i are distinct.
    # This is equivalent to finding a set of k elements such that
    # x_i = i, y_i = p[i], z_i = q[i] for some permutations p and q.
    
    # Group elements by x
    by_x = [[] for _ in range(k)]
    for x, y, z in elements:
        by_x[x].append((y, z))
        
    # Backtracking with pruning
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        for y, z in by_x[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
