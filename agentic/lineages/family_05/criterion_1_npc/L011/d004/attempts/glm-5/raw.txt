import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    triples = []
    idx = 2
    for _ in range(t):
        x = int(data[idx]); y = int(data[idx+1]); z = int(data[idx+2])
        idx += 3
        triples.append((x, y, z))
    
    # Build adjacency: for each x, list of (y, z)
    adj = [[] for _ in range(k+1)]
    for (x, y, z) in triples:
        adj[x].append((y, z))
    
    # We need to pick exactly one triple for each x in 1..k
    # such that the chosen y's are a permutation of 1..k
    # and the chosen z's are a permutation of 1..k.
    # This is like finding a perfect matching in a 3-partite 3-uniform hypergraph,
    # but since x must cover 1..k exactly once, we can do DP over x with
    # state = (used_y_mask, used_z_mask).
    # k <= 22, so 2^22 is too large for both masks simultaneously (2^44).
    # However, we can use meet-in-the-middle or backtracking with pruning.
    # Given constraints (t <= k^3, k<=22), backtracking might be too slow in worst case.
    # But let's try a smarter approach: since we need a permutation for y and z,
    # we can think of it as: for each x, choose a pair (y,z) with y and z both unused.
    # This is equivalent to finding a perfect matching in a bipartite graph between
    # the "y" side and "z" side, but with the constraint that each x contributes one edge
    # and each x must be used exactly once. Actually, it's a 3D matching problem.
    # 
    # Alternative: since k <= 22, we can use DP with memoization on (x_index, used_y_mask)
    # and for each state, track which z assignments are possible. But that's still large.
    #
    # Let's use backtracking with pruning: sort x's by number of available options,
    # and use bitmasks for used y and z. With k=22, worst case is bad, but typical
    # competitive programming inputs might be okay with pruning.
    
    # Order x's by increasing number of options (MRV heuristic)
    order = sorted(range(1, k+1), key=lambda x: len(adj[x]))
    
    # Precompute for each x the list of (y, z) as 0-indexed
    options = []
    for x in order:
        opts = [(y-1, z-1) for (y, z) in adj[x]]
        options.append(opts)
    
    full_mask = (1 << k) - 1
    chosen = [None] * k  # chosen[i] = (y, z) for order[i]
    
    # Check if any x has no options
    for opts in options:
        if not opts:
            print("NO")
            return
    
    # Backtracking
    import sys
    sys.setrecursionlimit(10000)
    
    found = [False]
    
    def backtrack(i, used_y, used_z):
        if found[0]:
            return
        if i == k:
            found[0] = True
            return
        for (y, z) in options[i]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                chosen[i] = (y, z)
                backtrack(i+1, used_y | (1 << y), used_z | (1 << z))
                if found[0]:
                    return
                chosen[i] = None
    
    backtrack(0, 0, 0)
    
    if not found[0]:
        print("NO")
        return
    
    print("YES")
    # Reconstruct: for each x in 1..k, find its chosen (y, z)
    result = [None] * (k+1)
    for i, x in enumerate(order):
        y, z = chosen[i]
        result[x] = (y+1, z+1)
    for x in range(1, k+1):
        y, z = result[x]
        print(f"{x} {y} {z}")

solve()
