We can model this as finding a perfect matching in a 3-partite hypergraph, but since k ≤ 22, we can use a backtracking approach with pruning.

Here’s the Python solution:  

```python
import sys

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    configs = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        configs.append((x-1, y-1, z-1))
    
    # Build adjacency: for each (x,y) pair, list of z's
    adj = [[[] for _ in range(k)] for _ in range(k)]
    for x, y, z in configs:
        adj[x][y].append(z)
    
    used_x = [False]*k
    used_y = [False]*k
    used_z = [False]*k
    chosen = []
    
    def backtrack(idx):
        if idx == k:
            return True
        # Try to assign frequency band idx
        for y in range(k):
            if used_y[y]:
                continue
            for z in adj[idx][y]:
                if used_z[z]:
                    continue
                used_y[y] = True
                used_z[z] = True
                chosen.append((idx+1, y+1, z+1))
                if backtrack(idx+1):
                    return True
                chosen.pop()
                used_y[y] = False
                used_z[z] = False
        return False
    
    if backtrack(0):
        print("YES")
        for x, y, z in chosen:
            print(x, y, z)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
