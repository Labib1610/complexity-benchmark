import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    profiles = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        profiles.append((x, y, z))
        idx += 3
        
    # The target sum is sum(2^j for j in 1..k).
    # In binary, this is a number with k ones: 11...1 (k times).
    # Since each profile (x, y, z) contributes 2^x, 2^y, 2^z,
    # we need the sum of 2^x_i to be 2^1 + 2^2 + ... + 2^k.
    # This is only possible if the multiset {x_1, ..., x_k} is exactly {1, 2, ..., k}.
    # The same must hold for {y_1, ..., y_k} and {z_1, ..., z_k}.
    
    # This is a variation of the Exact Cover problem or a 3D matching problem.
    # Given the constraints and the nature of the problem, we can use backtracking
    # with pruning or bitmasking if k is small.
    # k is 22 (11 players per team).
    
    target = (1 << (k + 1)) - 2
    
    # We need to pick k profiles such that:
    # sum(1 << x_i) == target
    # sum(1 << y_i) == target
    # sum(1 << z_i) == target
    
    # Since k is small (22), we can use backtracking.
    
    selected = []
    
    def backtrack(idx, current_x, current_y, current_z, count):
        if count == k:
            return current_x == target and current_y == target and current_z == target
        
        if idx == t:
            return False
        
        # Pruning: if remaining profiles are not enough
        if count + (t - idx) < k:
            return False
            
        # Try including profiles[idx]
        x, y, z = profiles[idx]
        
        # Check if adding this profile exceeds the target bits
        if (current_x & (1 << x)) == 0 and (current_y & (1 << y)) == 0 and (current_z & (1 << z)) == 0:
            selected.append(profiles[idx])
            if backtrack(idx + 1, current_x | (1 << x), current_y | (1 << y), current_z | (1 << z), count + 1):
                return True
            selected.pop()
            
        # Try excluding profiles[idx]
        if backtrack(idx + 1, current_x, current_y, current_z, count):
            return True
            
        return False

    if backtrack(0, 0, 0, 0, 0):
        print("YES")
        for p in selected:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
