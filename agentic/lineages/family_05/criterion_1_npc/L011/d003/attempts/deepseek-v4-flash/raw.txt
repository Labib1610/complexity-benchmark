import sys
from sys import stdin

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # Store configurations grouped by mold
    # Each entry: (dye, packaging)
    by_mold = [[] for _ in range(k)]
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        by_mold[x].append((y, z))
    
    # We need to find a perfect matching in a 3-partite hypergraph.
    # k <= 22, so we can use backtracking with bitmasks for dyes and packagings.
 
    # Pre-sort to help pruning
    for i in range(k):
        by_mold[i].sort()
    
    # We'll try to assign each mold a unique dye and packaging.

    # Use recursive backtracking with bitmasks.

    used_dye = 0
    used_pack = 0
    assignment = [None] * k  # assignment[mold] = (dye, pack)
 
    # Order molds by number of options (smallest domain first)
    order = sorted(range(k), key=lambda i: len(by_mold[i]))
 
    # For each mold, we can try its options.

    def dfs(pos):
        nonlocal used_dye, used_pack
        if pos == k:
            return True
        mold = order[pos]
        for y, z in by_mold[mold]:
            if (used_dye >> y) &  or (used_pack >> z) & :
                continue
            used_dye |= 1 << y
            used_pack |= 1 << z
            assignment[mold] = (y, z)
            if dfs(pos + 1):
                return True
            # backtrack
            used_dye &= ~(1 << y)
            used_pack &= ~(1 << z)
            assignment[mold] = None
        return False
 
    if not dfs(0):
        print("NO")
        return
 
    print("YES")
    out = []
    for mold in range(k):
        y, z = assignment[mold]
        out.append(f"{mold+1} {y+1} {z+1}")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
