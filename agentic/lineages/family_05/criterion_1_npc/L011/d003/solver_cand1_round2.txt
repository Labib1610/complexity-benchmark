```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        configs.append((x, y, z))
        idx += 3

    # We need to select k triples (x_i, y_i, z_i) such that all x_i are distinct,
    # all y_i are distinct, and all z_i are distinct.
    # This is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph,
    # or more simply, an exact cover problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(configs):
        adj[x].append((y, z, i))
        
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        # Optimization: pick the x with the fewest available options
        # But for k=22, simple backtracking with bitmasks or just ordering might work.
        for y, z, i in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(configs[i])
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for triple in result:
            print(f"{triple[0]+1} {triple[1]+1} {triple[2]+1}")
    else:
        print("NO")

solve()
```
