```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    setups = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        setups.append((x, y, z))
        idx += 3
        
    # We need to select k triplets (x_i, y_i, z_i) such that
    # {x_1, ..., x_k} = {1, ..., k}
    # {y_1, ..., y_k} = {1, ..., k}
    # {z_1, ..., z_k} = {1, ..., k}
    
    # This is equivalent to finding a set of k disjoint edges in a 3D hypergraph
    # or a Latin Square construction problem. Since k <= 22, we can use backtracking.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    result = []
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for i in range(start_idx, t):
            x, y, z = setups[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = used_y[y] = used_z[z] = True
                result.append((x, y, z))
                
                if backtrack(i + 1, count + 1):
                    return True
                
                result.pop()
                used_x[x] = used_y[y] = used_z[z] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for triplet in result:
            print(f"{triplet[0]} {triplet[1]} {triplet[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
