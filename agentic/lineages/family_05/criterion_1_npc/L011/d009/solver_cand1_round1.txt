```python
import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        configs.append((x, y, z))
        idx += 3
        
    if t < k:
        print("NO")
        return

    # The determinant of a Vandermonde matrix is non-zero if and only if
    # all elements in the set are distinct.
    # We need to select k configurations (x_i, y_i, z_i) such that:
    # {x_1, ..., x_k} are distinct, {y_1, ..., y_k} are distinct,
    # and {z_1, ..., z_k} are distinct.
    # Since x, y, z are in [1, k], this means each set must be a permutation of {1, ..., k}.
    
    # This is equivalent to finding a system of distinct representatives or
    # a matching in a hypergraph, but since k is small (up to 22), we can use backtracking.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    selected = []

    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for i in range(start_idx, t):
            x, y, z = configs[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = True
                used_y[y] = True
                used_z[z] = True
                selected.append(configs[i])
                
                if backtrack(i + 1, count + 1):
                    return True
                
                selected.pop()
                used_x[x] = False
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for item in selected:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
```
