import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        configs.append((x, y, z))
        idx += 3
        
    # The condition R(w) = B_k(w) for all w means the multiset of values
    # {x_1, ..., x_k} must be {1, 2, ..., k}.
    # Same for y and z.
    # This is an exact cover problem where we need to pick k items such that
    # the counts of each x, y, z are exactly 1.
    
    # State: (index_in_configs, current_x_mask, current_y_mask, current_z_mask, count)
    # Since k <= 22, we can use backtracking with pruning.
    
    target_mask = (1 << k) - 1
    
    memo = {}

    def backtrack(idx, x_mask, y_mask, z_mask, count):
        if count == k:
            return [] if (x_mask == target_mask and y_mask == target_mask and z_mask == target_mask) else None
        
        if idx == t or count + (t - idx) < k:
            return None
        
        state = (idx, x_mask, y_mask, z_mask, count)
        if state in memo:
            return memo[state]
        
        # Option 1: Include configs[idx]
        x, y, z = configs[idx]
        if not (x_mask & (1 << (x - 1))) and not (y_mask & (1 << (y - 1))) and not (z_mask & (1 << (z - 1))):
            res = backtrack(idx + 1, x_mask | (1 << (x - 1)), y_mask | (1 << (y - 1)), z_mask | (1 << (z - 1)), count + 1)
            if res is not None:
                memo[state] = [configs[idx]] + res
                return memo[state]
        
        # Option 2: Skip configs[idx]
        res = backtrack(idx + 1, x_mask, y_mask, z_mask, count)
        memo[state] = res
        return res

    result = backtrack(0, 0, 0, 0, 0)
    
    if result is not None:
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
