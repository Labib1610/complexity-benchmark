import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))

    pairs_by_x = [set() for _ in range(k + 1)]
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        pairs_by_x[x].add((y, z))

    # Every value 1..k must appear as an x-coordinate in the selected set.
    for x in range(1, k + 1):
        if not pairs_by_x[x]:
            print("NO")
            return

    # Process the most constrained x first.
    order = list(range(1, k + 1))
    order.sort(key=lambda x: (len(pairs_by_x[x]), x))

    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    chosen = [None] * (k + 1)

    def feasible(idx):
        """Prune: necessary condition for remaining x's."""
        rem = order[idx:]
        m = len(rem)
        if m == 0:
            return True

        unused_y = [y for y in range(1, k + 1) if not used_y[y]]
        unused_z = [z for z in range(1, k + 1) if not used_z[z]]

        # Each remaining x needs at least one free pair.
        for x in rem:
            ok = False
            for y, z in pairs_by_x[x]:
                if not used_y[y] and not used_z[z]:
                    ok = True
                    break
            if not ok:
                return False

        # 1) Remaining x must be matchable with distinct unused y's.
        y_pos = {y: i for i, y in enumerate(unused_y)}
        adj_y = []
        for x in rem:
            edges = []
            for y in unused_y:
                for z in unused_z:
                    if (y, z) in pairs_by_x[x]:
                        edges.append(y_pos[y])
                        break
            adj_y.append(edges)

        match_y = [-1] * len(unused_y)

        def bpm_y(u, seen):
            for v in adj_y[u]:
                if not seen[v]:
                    seen[v] = True
                    if match_y[v] == -1 or bpm_y(match_y[v], seen):
                        match_y[v] = u
                        return True
            return False

        cnt_y = 0
        for u in range(m):
            seen = [False] * len(unused_y)
            if bpm_y(u, seen):
                cnt_y += 1
            else:
                return False
        if cnt_y < m:
            return False

        # 2) Similarly for unused z's.
        z_pos = {z: i for i, z in enumerate(unused_z)}
        adj_z = []
        for x in rem:
            edges = []
            for z in unused_z:
                for y in unused_y:
                    if (y, z) in pairs_by_x[x]:
                        edges.append(z_pos[z])
                        break
            adj_z.append(edges)

        match_z = [-1] * len(unused_z)

        def bpm_z(u, seen):
            for v in adj_z[u]:
                if not seen[v]:
                    seen[v] = True
                    if match_z[v] == -1 or bpm_z(match_z[v], seen):
                        match_z[v] = u
                        return True
            return False

        cnt_z = 0
        for u in range(m):
            seen = [False] * len(unused_z)
            if bpm_z(u, seen):
                cnt_z += 1
            else:
                return False
        if cnt_z < m:
            return False

        return True

    def dfs(idx):
        if idx == k:
            return True
        if not feasible(idx):
            return False

        x = order[idx]
        for y, z in pairs_by_x[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                chosen[x] = (y, z)
                if dfs(idx + 1):
                    return True
                chosen[x] = None
                used_y[y] = False
                used_z[z] = False
        return False

    if dfs(0):
        out = ["YES"]
        for x in range(1, k + 1):
            y, z = chosen[x]
            out.append(f"{x} {y} {z}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
