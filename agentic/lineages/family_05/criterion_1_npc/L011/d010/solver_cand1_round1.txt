import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    configs = []
    idx = 2
    for _ in range(t):
        configs.append((int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])))
        idx += 3

    def get_divisors_count(val, j):
        count = 0
        for d in range(1, min(val, j) + 1):
            if val % d == 0 and j % d == 0:
                count += 1
        return count

    # Precompute matrix rows for each possible value 1..k
    # row_val[v][j] = number of common divisors of v and j+1
    row_vals = [[get_divisors_count(v, j + 1) for j in range(k)] for v in range(k + 1)]

    def is_det_nonzero(vals):
        # vals is a list of k integers
        # Construct matrix
        matrix = [row_vals[v] for v in vals]
        # Gaussian elimination to check determinant
        mat = [row[:] for row in matrix]
        for i in range(k):
            pivot = -1
            for r in range(i, k):
                if mat[r][i] != 0:
                    pivot = r
                    break
            if pivot == -1:
                return False
            mat[i], mat[pivot] = mat[pivot], mat[i]
            for r in range(i + 1, k):
                factor = mat[r][i] / mat[i][i]
                for c in range(i, k):
                    mat[r][c] -= factor * mat[i][c]
        return True

    # Since k is small (up to 22), but we need to pick k out of t,
    # and t can be up to k^3, we use backtracking with pruning.
    
    def backtrack(start_idx, current_configs):
        if len(current_configs) == k:
            xs = [c[0] for c in current_configs]
            ys = [c[1] for c in current_configs]
            zs = [c[2] for c in current_configs]
            if is_det_nonzero(xs) and is_det_nonzero(ys) and is_det_nonzero(zs):
                return current_configs
            return None
        
        for i in range(start_idx, t):
            res = backtrack(i + 1, current_configs + [configs[i]])
            if res:
                return res
        return None

    result = backtrack(0, [])
    if result:
        print("YES")
        for c in result:
            print(f"{c[0]} {c[1]} {c[2]}")
    else:
        print("NO")

solve()
