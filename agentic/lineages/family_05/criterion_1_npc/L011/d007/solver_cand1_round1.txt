import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        configs.append((x, y, z))
        idx += 3

    target = sum(10**j for j in range(1, k + 1))
    
    # We need to pick k configurations such that:
    # sum(10^x_i) = target
    # sum(10^y_i) = target
    # sum(10^z_i) = target
    # Since 10^j are powers of 10, this is equivalent to saying that
    # for each j in {1..k}, the count of x_i == j is exactly 1,
    # the count of y_i == j is exactly 1,
    # and the count of z_i == j is exactly 1.
    
    # This is an exact cover problem.
    # Each configuration (x, y, z) covers:
    # - Frequency j: x == j
    # - Modulation j: y == j
    # - Encryption j: z == j
    # We need to pick k configurations to cover each of the 3k constraints exactly once.
    
    # Constraints:
    # F1, F2, ..., Fk
    # M1, M2, ..., Mk
    # E1, E2, ..., Ek
    
    num_constraints = 3 * k
    adj = []
    for i, (x, y, z) in enumerate(configs):
        # Constraint indices: 0..k-1 (F), k..2k-1 (M), 2k..3k-1 (E)
        adj.append([x-1, k + y-1, 2*k + z-1])
        
    selected = []
    covered = [False] * num_constraints
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        # Heuristic: pick the first uncovered constraint and try to cover it
        first = -1
        for i in range(num_constraints):
            if not covered[i]:
                first = i
                break
        
        if first == -1:
            return True
            
        for i in range(start_idx, t):
            c = adj[i]
            if first in c:
                if not (covered[c[0]] or covered[c[1]] or covered[c[2]]):
                    covered[c[0]] = covered[c[1]] = covered[c[2]] = True
                    selected.append(configs[i])
                    if backtrack(i + 1, count + 1):
                        return True
                    selected.pop()
                    covered[c[0]] = covered[c[1]] = covered[c[2]] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for s in selected:
            print(f"{s[0]} {s[1]} {s[2]}")
    else:
        print("NO")

solve()
