We can model this as a matching problem: we need to pick exactly one configuration for each frequency band, each modulation scheme, and each encryption key, all simultaneously.

Since \( k \leq 22 \), we can use bitmask DP over the frequency and modulation dimensions, and check the encryption dimension as we go.

Here’s the implementation:  

```python
import sys

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    configs = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        configs.append((x-1, y-1, z-1))  # 0-indexed

    # dp[mask_freq][mask_mod] = mask_enc or None
    # We'll store the chosen configs to reconstruct
    dp = [[None] * (1 << k) for _ in range(1 << k)]
    dp[0][0] = (0, [])  # (mask_enc, list of configs)

    for x, y, z in configs:
        bx = 1 << x
        by = 1 << y
        bz = 1 << z
        # iterate masks in reverse to avoid reuse
        for mf in range((1 << k) - 1, -1, -1):
            if mf & bx:
                continue
            for mm in range((1 << k) - 1, -1, -1):
                if mm & by:
                    continue
                if dp[mf][mm] is not None:
                    enc_mask, chosen = dp[mf][mm]
                    if not (enc_mask & bz):
                        new_mf = mf | bx
                        new_mm = mm | by
                        new_enc = enc_mask | bz
                        if dp[new_mf][new_mm] is None:
                            dp[new_mf][new_mm] = (new_enc, chosen + [(x+1, y+1, z+1)])

    full_mask = (1 << k) - 1
    if dp[full_mask][full_mask] is not None:
        enc_mask, chosen = dp[full_mask][full_mask]
        if enc_mask == full_mask:
            print("YES")
            for c in chosen:
                print(*c)
            return
    print("NO")

if __name__ == "__main__":
    solve()
```
