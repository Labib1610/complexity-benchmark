import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        configs.append((x, y, z))
        idx += 3
        
    # Group by x to facilitate backtracking
    adj = [[] for _ in range(k)]
    for x, y, z in configs:
        adj[x].append((y, z))
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x):
        if x == k:
            return True
        
        for y, z in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
