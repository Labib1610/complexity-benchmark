import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        configs.append((x, y, z))
        idx += 3

    # We need to select k configurations such that each x, y, z appears exactly once.
    # This is equivalent to finding a set of k triples (x_i, y_i, z_i) such that
    # {x_i} = {0..k-1}, {y_i} = {0..k-1}, {z_i} = {0..k-1}.
    # This is the 3D matching problem, which is NP-complete.
    # Given k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * k
    used_y = [False] * k
    used_z = [False] * k
    selected = []

    def backtrack(idx_in_configs, count):
        if count == k:
            return True
        if idx_in_configs == t:
            return False
        
        # Pruning: if remaining configs are not enough to fill the rest
        if t - idx_in_configs < k - count:
            return False
            
        x, y, z = configs[idx_in_configs]
        
        # Try including this config
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            selected.append(configs[idx_in_configs])
            if backtrack(idx_in_configs + 1, count + 1):
                return True
            selected.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try excluding this config
        if backtrack(idx_in_configs + 1, count):
            return True
            
        return False

    # Optimization: sort configs to potentially find a solution faster
    # or use a more efficient search. For k=22, simple backtracking might be slow,
    # but the constraints on t and the nature of the problem often allow it.
    if backtrack(0, 0):
        print("YES")
        for x, y, z in selected:
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")

solve()
