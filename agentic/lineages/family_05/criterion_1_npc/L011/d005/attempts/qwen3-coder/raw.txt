from collections import defaultdict
import sys

def find_matching(k, approved):
    # Create a graph where each configuration is an edge between frequency and (modulation, key) pair
    # This is essentially a bipartite matching problem
    
    # We need to find a perfect matching in a 3D assignment problem
    # We can model this as finding a perfect matching in a tripartite hypergraph
    # But let's use a backtracking approach with pruning
    
    # Try all possible combinations using backtracking
    def backtrack(pos, freq_used, mod_used, key_used, selected):
        if pos == k:
            return len(selected) == k
        
        # Try each approved configuration
        for i in range(len(approved)):
            x, y, z = approved[i]
            x -= 1  # Convert to 0-indexed
            y -= 1
            z -= 1
            
            if x in freq_used or y in mod_used or z in key_used:
                continue
                
            freq_used.add(x)
            mod_used.add(y)
            key_used.add(z)
            selected.append(i)
            
            if backtrack(pos + 1, freq_used, mod_used, key_used, selected):
                return True
                
            freq_used.remove(x)
            mod_used.remove(y)
            key_used.remove(z)
            selected.pop()
            
        return False
    
    # Optimization: check if each frequency, modulation and key appears in at least one configuration
    freq_count = [0] * k
    mod_count = [0] * k
    key_count = [0] * k
    
    for x, y, z in approved:
        freq_count[x-1] += 1
        mod_count[y-1] += 1
        key_count[z-1] += 1
    
    for i in range(k):
        if freq_count[i] == 0 or mod_count[i] == 0 or key_count[i] == 0:
            return None
    
    selected = []
    if backtrack(0, set(), set(), set(), selected):
        return [approved[i] for i in selected]
    
    return None

def solve():
    line = input().split()
    k, t = int(line[0]), int(line[1])
    
    approved = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        approved.append((x, y, z))
    
    result = find_matching(k, approved)
    
    if result is None:
        print("NO")
    else:
        print("YES")
        for x, y, z in result:
            print(x, y, z)

solve()
