```python
import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        configs.append((x, y, z))
        idx += 3

    if t < k:
        print("NO")
        return

    # We need to pick k configurations such that:
    # OR of x_i = (1<<k)-1, OR of y_i = (1<<k)-1, OR of z_i = (1<<k)-1
    # This is an exact cover-like problem. Since k is small (up to 22),
    # we can use a randomized approach (Color Coding / Randomized Search).
    
    target = (1 << k) - 1
    
    # Try randomized greedy/local search
    # Since we need exactly k, we can try picking random subsets of size k
    # or use a hill-climbing approach.
    
    for _ in range(200000):
        subset = random.sample(configs, k)
        
        mask_x = 0
        mask_y = 0
        mask_z = 0
        for x, y, z in subset:
            mask_x |= (1 << x)
            mask_y |= (1 << y)
            mask_z |= (1 << z)
            
        if mask_x == target and mask_y == target and mask_z == target:
            print("YES")
            for x, y, z in subset:
                print(f"{x+1} {y+1} {z+1}")
            return

    # If random sampling fails, try a simple hill climbing
    current_subset = random.sample(configs, k)
    for _ in range(500000):
        mask_x = 0
        mask_y = 0
        mask_z = 0
        for x, y, z in current_subset:
            mask_x |= (1 << x)
            mask_y |= (1 << y)
            mask_z |= (1 << z)
            
        if mask_x == target and mask_y == target and mask_z == target:
            print("YES")
            for x, y, z in current_subset:
                print(f"{x+1} {y+1} {z+1}")
            return
        
        # Swap one element
        idx_to_replace = random.randrange(k)
        new_config = random.choice(configs)
        current_subset[idx_to_replace] = new_config

    print("NO")

solve()
```
