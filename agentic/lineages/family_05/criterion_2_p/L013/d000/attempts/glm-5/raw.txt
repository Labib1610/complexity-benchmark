import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    k = int(next(it))
    t = int(next(it))
    
    adj = [[] for _ in range(k + 1)]
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        adj[x].append(z)
        
    match_to_agent = [0] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_to_agent[v] == 0 or dfs(match_to_agent[v], visited):
                    match_to_agent[v] = u
                    return True
        return False

    match_count = 0
    for u in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(u, visited):
            match_count += 1
            
    if match_count < k:
        print("NO")
    else:
        print("YES")
        for z in range(1, k + 1):
            x = match_to_agent[z]
            print(x, x, z)

if __name__ == '__main__':
    solve()
