from collections import defaultdict, deque

def solve():
    k, t = map(int, input().split())
    
    # Since each pilot is paired with exactly one starfighter,
    # and all pilots use different starfighters,
    # we can think of this as a matching problem where
    # we need to assign missions to the pilot-starfighter pairs.
    
    # Create a mapping from pilot to starfighter
    pilot_to_starfighter = {}
    # For each pilot-starfighter pair, store the list of possible missions
    ps_to_missions = defaultdict(list)
    
    for _ in range(t):
        p, s, m = map(int, input().split())
        if p not in pilot_to_starfighter:
            pilot_to_starfighter[p] = s
        ps_to_missions[(p, s)].append(m)
    
    # Check if we have exactly k pilots, each with a unique starfighter
    if len(pilot_to_starfighter) != k:
        print("NO")
        return
        
    # Build a bipartite graph: left side is pilot-starfighter pairs (which correspond to pilots),
    # right side is missions. An edge exists if the pair can do that mission.
    # This is actually a matching problem: match each of the k pairs to a distinct mission.
    
    # Create adjacency list for missions for each pilot
    adj = [[] for _ in range(k + 1)]  # 1-indexed
    
    for (p, s), missions in ps_to_missions.items():
        for m in missions:
            adj[p].append(m)
    
    # Try to find a maximum matching using Hungarian algorithm approach or DFS
    # We want a perfect matching of size k
    
    match_mission = [0] * (k + 1)  # which pilot is assigned to each mission
    match_pilot = [0] * (k + 1)    # which mission is assigned to each pilot
    
    def dfs(u, visited):
        for v in adj[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            if match_mission[v] == 0 or dfs(match_mission[v], visited):
                match_mission[v] = u
                match_pilot[u] = v
                return True
        return False
    
    matched = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            matched += 1
    
    if matched != k:
        print("NO")
        return
    
    print("YES")
    for i in range(1, k + 1):
        p = i
        s = pilot_to_starfighter[p]
        m = match_pilot[p]
        print(p, s, m)

solve()
