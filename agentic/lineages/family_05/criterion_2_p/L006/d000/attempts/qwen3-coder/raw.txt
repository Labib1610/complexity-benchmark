from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    index = 2
    allowed = []
    pilot_aircraft = {}
    
    # Read allowed combinations and build pilot->aircraft mapping
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        allowed.append((x, y, z))
        if x not in pilot_aircraft:
            pilot_aircraft[x] = y
        index += 3
    
    # Build bipartite graph: pilots on one side, routes on the other
    # Edge exists if (pilot, aircraft_for_pilot, route) is allowed
    graph = defaultdict(list)
    pilot_to_routes = defaultdict(list)
    
    for x, y, z in allowed:
        if pilot_aircraft[x] == y:
            graph[x].append(z)
            pilot_to_routes[x].append(z)
    
    # Now we need to find a perfect matching in this bipartite graph
    # Left side: pilots (1 to k)
    # Right side: routes (1 to k)
    # Edges: pilot -> route if they can be paired
    
    # Build adjacency list for graph where pilots are connected to routes
    adj = defaultdict(list)
    for x, y, z in allowed:
        if pilot_aircraft[x] == y:
            adj[x].append(z)
    
    # Maximum bipartite matching using Ford-Fulkerson with BFS (Edmonds-Karp)
    def bfs_match(match_route_to_pilot, match_pilot_to_route):
        queue = deque()
        parent = {}
        visited = set()
        
        # Find unmatched pilots as starting points
        for pilot in range(1, k+1):
            if pilot not in match_pilot_to_route:
                queue.append(('p', pilot))
                parent[('p', pilot)] = None
                visited.add(('p', pilot))
        
        while queue:
            side, u = queue.popleft()
            
            if side == 'p':  # pilot side
                for route in adj.get(u, []):
                    if ('r', route) not in visited:
                        parent[('r', route)] = ('p', u)
                        visited.add(('r', route))
                        if route not in match_route_to_pilot:
                            # Found an augmenting path
                            current = ('r', route)
                            path = []
                            while current is not None:
                                path.append(current)
                                current = parent[current]
                            path.reverse()
                            return path
                        queue.append(('r', route))
                        visited.add(('r', route))
            else:  # route side
                pilot = match_route_to_pilot[u]
                if ('p', pilot) not in visited:
                    parent[('p', pilot)] = ('r', u)
                    visited.add(('p', pilot))
                    queue.append(('p', pilot))
        return None
    
    match_route_to_pilot = {}
    match_pilot_to_route = {}
    
    while True:
        path = bfs_match(match_route_to_pilot, match_pilot_to_route)
        if not path:
            break
        
        # Augment along the path
        for i in range(len(path) - 1):
            if path[i][0] == 'p':
                pilot = path[i][1]
                route = path[i+1][1]
                match_pilot_to_route[pilot] = route
                match_route_to_pilot[route] = pilot
    
    # Check if we have a perfect matching
    if len(match_pilot_to_route) != k:
        print("NO")
        return
    
    print("YES")
    for pilot in range(1, k+1):
        route = match_pilot_to_route[pilot]
        aircraft = pilot_aircraft[pilot]
        print(pilot, aircraft, route)

solve()
