from collections import defaultdict

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
    
    reactions = []
    catalyst_to_solvent = {}
    solvent_to_catalyst = {}
    temp_reactions = defaultdict(list)
    
    # Read reactions and build mappings
    for _ in range(t):
        x, y, z = map(int, input().split())
        reactions.append((x, y, z))
        
        # Check consistency of catalyst-solvent mapping
        if x in catalyst_to_solvent:
            if catalyst_to_solvent[x] != y:
                print("NO")
                return
        else:
            catalyst_to_solvent[x] = y
            
        if y in solvent_to_catalyst:
            if solvent_to_catalyst[y] != x:
                print("NO")
                return
        else:
            solvent_to_catalyst[y] = x
            
        temp_reactions[z].append((x, y, z))
    
    # Check if we have exactly k catalysts and k solvents
    if len(catalyst_to_solvent) != k or len(solvent_to_catalyst) != k:
        print("NO")
        return
    
    # Build bipartite graph for matching catalysts/solvents to temperatures
    # Each reaction is an edge between catalyst and temperature
    catalyst_temp_edges = defaultdict(list)
    temp_catalyst_edges = defaultdict(list)
    
    for x, y, z in reactions:
        catalyst_temp_edges[x].append(z)
        temp_catalyst_edges[z].append(x)
    
    # Try to find a perfect matching using Hungarian algorithm approach
    # or simpler approach: since each catalyst has exactly one solvent,
    # and each solvent has exactly one catalyst, we just need to match
    # each catalyst-solvent pair with a unique temperature
    
    # For each temperature, we can use at most one reaction
    # For each catalyst, we must use exactly one reaction
    
    # This is a bipartite matching problem: match catalysts to temperatures
    # such that the reaction exists in our log
    
    # Use a greedy approach with backtracking simulation
    match_catalyst_to_temp = [-1] * (k + 1)  # catalyst -> temperature
    match_temp_to_catalyst = [-1] * (k + 1)  # temperature -> catalyst
    
    def dfs(catalyst, visited):
        for temp in catalyst_temp_edges[catalyst]:
            if temp in visited:
                continue
            visited.add(temp)
            
            # If temp is unmatched or we can find augmenting path
            if match_temp_to_catalyst[temp] == -1 or dfs(match_temp_to_catalyst[temp], visited):
                match_catalyst_to_temp[catalyst] = temp
                match_temp_to_catalyst[temp] = catalyst
                return True
        return False
    
    # Try to match all catalysts
    matched = 0
    for catalyst in range(1, k + 1):
        if dfs(catalyst, set()):
            matched += 1
    
    if matched != k:
        print("NO")
        return
    
    print("YES")
    for catalyst in range(1, k + 1):
        temp = match_catalyst_to_temp[catalyst]
        solvent = catalyst_to_solvent[catalyst]
        print(catalyst, solvent, temp)

solve()
