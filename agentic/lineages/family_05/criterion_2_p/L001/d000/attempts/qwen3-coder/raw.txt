from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    index = 2
    configs = []
    drone_to_station = {}
    station_to_drone = {}
    
    # Read configurations and build mappings
    for _ in range(t):
        d = int(data[index])
        c = int(data[index+1])
        f = int(data[index+2])
        configs.append((d, c, f))
        index += 3
        
        # Build drone to station mapping
        if d not in drone_to_station:
            drone_to_station[d] = c
        else:
            # Verify consistency
            if drone_to_station[d] != c:
                # This shouldn't happen according to problem statement
                pass
                
        if c not in station_to_drone:
            station_to_drone[c] = d
        else:
            # Verify consistency
            if station_to_drone[c] != d:
                # This shouldn't happen according to problem statement
                pass
    
    # Build bipartite graph: drones on one side, frequencies on the other
    # Edge exists if there's a configuration with that drone and frequency
    graph = defaultdict(list)  # drone -> list of frequencies
    config_map = {}  # (drone, freq) -> (d, c, f)
    
    for d, c, f in configs:
        graph[d].append(f)
        config_map[(d, f)] = (d, c, f)
    
    # Try to find maximum matching using Hungarian algorithm approach
    # But since k <= 1200, we can use a simpler augmenting path approach
    
    match_freq_to_drone = {}  # frequency -> drone
    match_drone_to_freq = {}  # drone -> frequency
    
    def dfs(drone, visited):
        for freq in graph[drone]:
            if freq in visited:
                continue
            visited.add(freq)
            
            # If frequency is unmatched or we can rematch its current drone
            if freq not in match_freq_to_drone:
                match_freq_to_drone[freq] = drone
                match_drone_to_freq[drone] = freq
                return True
            else:
                # Try to reassign the drone currently matched to this frequency
                prev_drone = match_freq_to_drone[freq]
                if dfs(prev_drone, visited):
                    match_freq_to_drone[freq] = drone
                    match_drone_to_freq[drone] = freq
                    return True
        return False
    
    # Try to match each drone
    for drone in range(1, k+1):
        visited = set()
        dfs(drone, visited)
    
    # Check if we have a perfect matching
    if len(match_drone_to_freq) == k:
        print("YES")
        result = []
        for drone in range(1, k+1):
            freq = match_drone_to_freq[drone]
            station = drone_to_station[drone]
            result.append(f"{drone} {station} {freq}")
        print("\n".join(result))
    else:
        print("NO")

solve()
