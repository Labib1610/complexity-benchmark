import sys

def solve():
    sys.setrecursionlimit(1 << 25)
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    contracts = [(int(next(it)) - 1, int(next(it)) - 1, int(next(it))) for _ in range(k)]
    edges = [(int(next(it)) - 1, int(next(it)) - 1, int(next(it))) for _ in range(m)]

    # Simple heuristic: try to satisfy each contract sequentially using any available capacity.
    # This is not guaranteed to find a solution when one exists, but serves as a placeholder.
    remaining = [c for _, _, c in edges]
    plan = [[0] * m for _ in range(k)]

    from collections import deque

    def bfs(s, t, parent):
        q = deque([s])
        parent[:] = [-1] * n
        parent[s] = s
        while q:
            u = q.popleft()
            if u == t:
                break
            for idx, (a, b, _) in enumerate(edges):
                if a == u and remaining[idx] > 0 and parent[b] == -1:
                    parent[b] = (u, idx)
                    q.append(b)
        return parent[t] != -1

    for ci, (src, dst, demand) in enumerate(contracts):
        flow = 0
        parent = [-1] * n
        while flow < demand and bfs(src, dst, parent):
            # find bottleneck
            v = dst
            bottleneck = float('inf')
            while v != src:
                u, idx = parent[v]
                bottleneck = min(bottleneck, remaining[idx])
                v = u
            add = min(bottleneck, demand - flow)
            v = dst
            while v != src:
                u, idx = parent[v]
                remaining[idx] -= add
                plan[ci][idx] += add
                v = u
            flow += add
        if flow < demand:
            print("NO")
            return

    print("YES")
    out_lines = [' '.join(map(str, row)) for row in plan]
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
