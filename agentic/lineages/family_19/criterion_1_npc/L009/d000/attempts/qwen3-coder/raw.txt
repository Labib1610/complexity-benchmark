from collections import defaultdict, deque
import sys

def solve():
    n, m, k = map(int, input().split())
    
    categories = []
    for i in range(k):
        s, t, d = map(int, input().split())
        categories.append((s-1, t-1, d))  # Convert to 0-indexed
    
    chutes = []
    chute_edges = []
    for i in range(m):
        u, v, cap = map(int, input().split())
        chutes.append(cap)
        chute_edges.append((u-1, v-1))  # Convert to 0-indexed
    
    # For each category, find a feasible flow
    category_flows = []
    
    for cat_idx, (source, sink, demand) in enumerate(categories):
        # Build graph for this category
        adj = defaultdict(list)
        edge_index = {}
        for i, (u, v) in enumerate(chute_edges):
            adj[u].append((v, i))
            adj[v].append((u, i))  # For residual graph
            edge_index[(u, v, i)] = i
        
        # Use Ford-Fulkerson with BFS (Edmonds-Karp) to find flow
        flow = [0] * m
        
        # Create residual graph
        residual_capacity = [cap for cap in chutes]
        residual_graph = defaultdict(list)
        for i, (u, v) in enumerate(chute_edges):
            residual_graph[u].append((v, i, 1))      # forward edge
            residual_graph[v].append((u, i, -1))     # backward edge
        
        # Find max flow
        total_flow = 0
        
        while total_flow < demand:
            # BFS to find augmenting path
            parent = [-1] * n
            parent_edge = [-1] * n
            queue = deque([source])
            parent[source] = source
            
            found = False
            while queue and not found:
                u = queue.popleft()
                for v, edge_idx, direction in residual_graph[u]:
                    if parent[v] == -1 and residual_capacity[edge_idx] > 0:
                        parent[v] = u
                        parent_edge[v] = edge_idx
                        if v == sink:
                            found = True
                            break
                        queue.append(v)
            
            if not found:
                break
            
            # Find minimum capacity along path
            path_flow = demand - total_flow
            s = sink
            while s != source:
                edge_idx = parent_edge[s]
                path_flow = min(path_flow, residual_capacity[edge_idx])
                s = parent[s]
            
            # Update residual capacities and flow
            s = sink
            while s != source:
                edge_idx = parent_edge[s]
                residual_capacity[edge_idx] -= path_flow
                # The reverse edge has index edge_idx as well
                flow[edge_idx] += path_flow if chute_edges[edge_idx][0] == parent[s] else -path_flow
                s = parent[s]
            
            total_flow += path_flow
        
        if total_flow < demand:
            print("NO")
            return
            
        category_flows.append(flow[:])
    
    # Check if the combined flow satisfies all constraints
    for j in range(m):
        total = sum(category_flows[i][j] for i in range(k))
        if total > chutes[j]:
            print("NO")
            return
    
    print("YES")
    for flow in category_flows:
        print(' '.join(map(str, flow)))

solve()
