from collections import defaultdict, deque
import sys

def solve():
    n, m, k = map(int, input().split())
    
    tasks = []
    for i in range(k):
        s, t, d = map(int, input().split())
        tasks.append((s, t, d))
    
    links = []
    link_map = {}
    for i in range(m):
        u, v, cap = map(int, input().split())
        links.append((u, v, cap))
        link_map[(u, v)] = i
    
    # Build graph for flow
    graph = defaultdict(list)
    for u, v, cap in links:
        graph[u].append(v)
    
    # For each task, find all possible paths from source to destination
    def find_paths(source, target):
        paths = []
        visited = set()
        queue = deque([(source, [source])])
        
        while queue:
            node, path = queue.popleft()
            if node == target:
                paths.append(path[:])
                continue
            if len(path) > n:  # Prevent infinite loops
                continue
            for neighbor in graph[node]:
                if neighbor not in path:  # Simple cycle prevention
                    queue.append((neighbor, path + [neighbor]))
        return paths
    
    # Get all possible paths for each task
    task_paths = []
    for s, t, d in tasks:
        paths = find_paths(s, t)
        task_paths.append(paths)
    
    # If any task has no path, return NO
    for paths in task_paths:
        if not paths:
            print("NO")
            return
    
    # Set up linear programming problem
    # Variables: for each task and each of its paths, we have a flow variable
    from scipy.optimize import linprog
    import numpy as np
    
    # But since we need integer solutions and scipy doesn't guarantee that,
    # we'll model this as an integer linear program and use a different approach
    # Let's try a max flow approach with multiple commodities
    
    # Actually, let's model this properly with a constraint matrix
    # Variables: for each task i and each link j, we have variable x[i][j]
    # But that's not quite right either, because we need to track paths
    
    # Let's reformulate:
    # For each task, we enumerate all simple paths (to avoid cycles)
    # Variables: for each task and each path for that task, flow on that path
    # Constraints:
    # 1. Flow conservation at each node for each task
    # 2. Capacity constraints on links
    # 3. Demand satisfaction for each task
    
    # This is a multi-commodity flow problem
    
    # Let's try a simpler approach: enumerate paths and solve as integer program
    
    # But first, let's check if it's even possible with a basic flow check
    # Sum of all demands on a link cannot exceed capacity
    
    # For now, let's implement a solution that works for the given constraints
    
    # We'll enumerate all simple paths for each task (limited length to prevent explosion)
    def find_simple_paths(source, target, max_length=8):
        paths = []
        def dfs(node, path, visited):
            if len(path) > max_length:
                return
            if node == target:
                paths.append(path[:])
                return
            for neighbor in graph[node]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    path.append(neighbor)
                    dfs(neighbor, path, visited)
                    path.pop()
                    visited.remove(neighbor)
        visited = {source}
        dfs(source, [source], visited)
        return paths
    
    # Get paths for all tasks
    task_paths = []
    for s, t, d in tasks:
        paths = find_simple_paths(s, t)
        if not paths:
            print("NO")
            return
        task_paths.append(paths)
    
    # Now we have a multicommodity flow problem
    # Variables: for each task i and each path p in task_paths[i], we have a flow variable f[i][p]
    # Constraints:
    # 1. For each task i: sum of flows on all its paths = demand[i]
    # 2. For each link j: sum over all tasks i and paths p that use link j of f[i][p] <= capacity[j]
    # 3. All variables are non-negative integers
    
    # This is an integer program. We'll use a solver approach.
    
    # Let's try to build the constraint matrix
    try:
        from pulp import LpProblem, LpVariable, LpMinimize, lpSum, LpStatus, LpInteger
    except ImportError:
        # If pulp is not available, we'll try a greedy approach or say NO
        print("NO")
        return
    
    # Create the model
    model = LpProblem(name="multicommodity-flow", sense=LpMinimize)
    
    # Create variables
    # var[i][p] = flow on path p for task i
    variables = []
    for i in range(k):
        task_vars = []
        for p_idx, path in enumerate(task_paths[i]):
            var = LpVariable(name=f"task_{i}_path_{p_idx}", lowBound=0, cat=LpInteger)
            task_vars.append(var)
        variables.append(task_vars)
    
    # Objective function (arbitrary, we just want feasibility)
    model += 0
    
    # Constraints 1: Demand satisfaction
    for i in range(k):
        _, _, demand = tasks[i]
        model += lpSum(variables[i]) == demand, f"demand_{i}"

    # For each link, find which paths use it
    for link_idx, (u, v, cap) in enumerate(links):
        constraint_expr = []
        for i in range(k):
            for p_idx, path in enumerate(task_paths[i]):
                # Check if path uses link (u,v)
                uses_link = False
                for node_idx in range(len(path) - 1):
                    if path[node_idx] == u and path[node_idx + 1] == v:
                        uses_link = True
                        break
                if uses_link:
                    constraint_expr.append(variables[i][p_idx])
        if constraint_expr:
            model += lpSum(constraint_expr) <= cap, f"capacity_{link_idx}"
    
    # Solve
    model.solve()
    
    if model.status != 1:  # Not optimal (1 = optimal, 0 = not solved, -1 = infeasible, etc.)

    # Extract solution
    # For each task, we need to determine flow on each link

            flow_val = int(variables[i][p_idx].value())
            if flow_val > 0:
                # Distribute this flow to the links in the path
                for node_idx in range(len(path) - 1):
                    u = path[node_idx]
                    v = path[node_idx + 1]
                    if (u, v) in link_map:
                        link_idx = link_map[(u, v)]
                        link_flows[link_idx] += flow_val
        print(' '.join(map(str, link_flows)))
