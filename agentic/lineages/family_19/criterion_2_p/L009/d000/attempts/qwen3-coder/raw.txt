from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    _ = int(data[idx])
    idx += 1
    
    s = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    d = int(data[idx])
    idx += 1
    
    edges = []
    graph = defaultdict(list)
    capacity = []
    
    for i in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        cap = int(data[idx])
        idx += 1
        edges.append((u, v))
        graph[u].append((v, len(capacity)))
        capacity.append(cap)
        graph[v].append((u, len(capacity)))  # reverse edge for residual graph
        capacity.append(0)  # reverse edge has 0 initial capacity
    
    # We need to find a flow of value d from s to t
    # This is a standard max flow problem
    
    def bfs(source, sink, parent):
        visited = [False] * (n + 1)
        queue = deque([source])
        visited[source] = True
        parent[source] = -1
        
        while queue:
            u = queue.popleft()
            
            for v, edge_idx in graph[u]:
                if not visited[v] and capacity[edge_idx] > 0:
                    queue.append(v)
                    visited[v] = True
                    parent[v] = edge_idx
                    
        return visited[sink]
    
    def ford_fulkerson(source, sink, target_flow):
        parent = [-1] * (n + 1)
        max_flow = 0
        
        while max_flow < target_flow and bfs(source, sink, parent):
            path_flow = target_flow - max_flow
            s_idx = sink
            
            # Find minimum capacity in the path
            while s_idx != source:
                edge_idx = parent[s_idx]
                path_flow = min(path_flow, capacity[edge_idx])
                # Find the node from which we came to s_idx
                for v, idx in graph[s_idx]:
                    if idx == (edge_idx ^ 1):  # Find reverse edge
                        s_idx = v
                        break
                else:
                    break
            
            # Update capacities
            v_idx = sink
            while v_idx != source:
                edge_idx = parent[v_idx]
                capacity[edge_idx] -= path_flow
                capacity[edge_idx ^ 1] += path_flow
                # Find the node from which we came to v_idx
                for v, idx in graph[v_idx]:
                    if idx == (edge_idx ^ 1):
                        v_idx = v
                        break
                else:
                    break
            
            max_flow += path_flow
            
        return max_flow
    
    # Run Ford-Fulkerson to send exactly d units of flow
    flow_value = ford_fulkerson(s, t, d)
    
    if flow_value < d:
        print("NO")
        return
    
    # Reconstruct the flow on original edges
    result = []
    for i in range(0, len(capacity), 2):
        # Original edge capacity was at i, reverse at i+1
        # Flow on original edge = original capacity - residual capacity
        original_capacity = capacity[i] + capacity[i+1]  # Since we added reverse edge with 0 capacity
        flow = capacity[i+1]  # Flow is what's pushed to reverse edge
        result.append(flow)
    
    print("YES")
    print(" ".join(map(str, result)))

solve()
