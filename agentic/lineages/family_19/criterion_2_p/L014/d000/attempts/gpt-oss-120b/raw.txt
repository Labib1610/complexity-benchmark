```python
import sys
from collections import deque

def solve() -> None:
    data = list(map(int, sys.stdin.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it); m = next(it); _ = next(it)  # the third is always 1
    s = next(it)-1; t = next(it)-1; d = next(it)
    edges_input = []
    for _ in range(m):
        u = next(it)-1; v = next(it)-1; cap = next(it)
        edges_input.append((u, v, cap))

    # Dinic
    N = n
    g = [[] for _ in range(N)]
    class Edge:
        __slots__ = ('to','rev','cap','orig')
        def __init__(self, to, rev, cap, orig=False):
            self.to = to
            self.rev = rev
            self.cap = cap
            self.orig = orig

    edge_refs = []  # reference to original edges in order
    for idx, (u, v, cap) in enumerate(edges_input):
        fwd = Edge(v, len(g[v]), cap, True)
        rev = Edge(u, len(g[u]), 0)
        g[u].append(fwd)
        g[v].append(rev)
        edge_refs.append((u, len(g[u])-1))  # store location of forward edge

    level = [0]*N
    it_ptr = [0]*N

    def bfs():
        for i in range(N):
            level[i] = -1
        q = deque([s])
        level[s] = 0
        while q:
            v = q.popleft()
            for e in g[v]:
                if e.cap > 0 and level[e.to] < 0:
                    level[e.to] = level[v] + 1
                    q.append(e.to)
        return level[t] >= 0

    def dfs(v, f):
        if v == t:
            return f
        for i in range(it_ptr[v], len(g[v])):
            e = g[v][i]
            if e.cap > 0 and level[v] < level[e.to]:
                ret = dfs(e.to, min(f, e.cap))
                if ret:
                    e.cap -= ret
                    g[e.to][e.rev].cap += ret
                    return ret
            it_ptr[v] += 1
        return 0

    flow = 0
    INF = 10**18
    while bfs():
        it_ptr = [0]*N
        while True:
            pushed = dfs(s, INF)
            if not pushed:
                break
            flow += pushed
            if flow >= d:
                break
        if flow >= d:
            break

    if flow < d:
        print("NO")
        return

    # We have at least d flow, but maybe more. Need to adjust to exactly d.
    # Since we stopped early, flow == d.
    # Retrieve flows on original edges.
    res = []
    for (u, idx) in edge_refs:
        e = g[u][idx]
        used = edges_input[edge_refs.index((u, idx))][2] - e.cap  # original cap - remaining
        res.append(str(used))
    print("YES")
    print(" ".join(res))

if __name__ == "__main__":
    solve()
```
