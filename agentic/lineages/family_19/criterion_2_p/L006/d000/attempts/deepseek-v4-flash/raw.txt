We can solve this by finding a single path from source to destination that can carry the required flow, since only one stream is being routed.  
We can use a simple DFS/BFS to find a path where every edge has capacity at least `d`.  
If found, we assign `d` to each edge on that path and `0` to all others.  

```python
import sys
from collections import deque

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))  # skip 1
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    for idx in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u, v, cap))
        adj[u].append((v, idx, cap))
    
    # BFS to find a path with all capacities >= d
    parent = [-1] * (n + 1)
    parent_edge = [-1] * (n + 1)
    visited = [False] * (n + 1)
    q = deque([s])
    visited[s] = True
    
    while q:
        u = q.popleft()
        if u == t:
            break
        for v, idx, cap in adj[u]:
            if not visited[v] and cap >= d:
                visited[v] = True
                parent[v] = u
                parent_edge[v] = idx
                q.append(v)
    
    if not visited[t]:
        print("NO")
        return
    
    # Trace path
    path_edges = []
    cur = t
    while cur != s:
        path_edges.append(parent_edge[cur])
        cur = parent[cur]
    
    flow = [0] * m
    for idx in path_edges:
        flow[idx] = d
    
    print("YES")
    print(" ".join(map(str, flow)))

if __name__ == "__main__":
    main()
```
