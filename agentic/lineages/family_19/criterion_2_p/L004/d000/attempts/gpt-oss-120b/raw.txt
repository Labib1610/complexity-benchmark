import sys
sys.setrecursionlimit(1 << 25)

class Dinic:
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]

    def add_edge(self, fr, to, cap):
        fwd = [to, cap, None]   # to, cap, rev index placeholder
        rev = [fr, 0, None]
        fwd[2] = len(self.g[to])
        rev[2] = len(self.g[fr])
        self.g[fr].append(fwd)
        self.g[to].append(rev)
        return fwd  # reference to forward edge

    def bfs(self, s, t, level):
        from collections import deque
        q = deque([s])
        level[:] = [-1] * self.n
        level[s] = 0
        while q:
            v = q.popleft()
            for to, cap, rev in self.g[v]:
                if cap > 0 and level[to] < 0:
                    level[to] = level[v] + 1
                    q.append(to)
        return level[t] >= 0

    def dfs(self, v, t, f, level, it):
        if v == t:
            return f
        for i in range(it[v], len(self.g[v])):
            it[v] = i
            to, cap, rev = self.g[v][i]
            if cap > 0 and level[v] < level[to]:
                ret = self.dfs(to, t, min(f, cap), level, it)
                if ret:
                    self.g[v][i][1] -= ret
                    self.g[to][rev][1] += ret
                    return ret
        return 0

    def max_flow(self, s, t, limit=None):
        flow = 0
        level = [0]*self.n
        INF = 10**18
        while self.bfs(s, t, level):
            it = [0]*self.n
            while True:
                f = self.dfs(s, t, INF if limit is None else limit - flow, level, it)
                if not f:
                    break
                flow += f
                if limit is not None and flow >= limit:
                    return flow
        return flow

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); _ = next(it)  # asset type ignored
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    d = int(next(it))
    edges = []
    caps = []
    dinic = Dinic(n)
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        fwd = dinic.add_edge(u, v, cap)
        edges.append(fwd)
        caps.append(cap)

    flow = dinic.max_flow(s, t, d)
    if flow != d:
        print("NO")
        return
    # retrieve flows
    res = []
    for fwd, cap in zip(edges, caps):
        used = cap - fwd[1]   # original - remaining capacity
        res.append(str(used))
    print("YES")
    print(' '.join(res))

if __name__ == "__main__":
    solve()
