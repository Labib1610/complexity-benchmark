from collections import defaultdict, deque

def solve():
    n, m, _ = map(int, input().split())
    s, t, d = map(int, input().split())
    
    edges = []
    graph = defaultdict(list)
    capacity = []
    
    for i in range(m):
        u, v, c = map(int, input().split())
        edges.append((u, v))
        graph[u].append((v, len(capacity)))
        capacity.append(c)
        graph[v].append((u, len(capacity)))
        capacity.append(0)  # Reverse edge with 0 capacity initially
    
    # Add super source and super sink
    super_s = 0
    super_t = n + 1
    
    # Connect super source to s with capacity d
    graph[super_s].append((s, len(capacity)))
    capacity.append(d)
    graph[s].append((super_s, len(capacity)))
    capacity.append(0)
    
    # Connect t to super sink with capacity d
    graph[t].append((super_t, len(capacity)))
    capacity.append(d)
    graph[super_t].append((t, len(capacity)))
    capacity.append(0)
    
    # For each intermediate node, add constraint: flow in = flow out
    for node in range(1, n + 1):
        if node != s and node != t:
            # We need to ensure flow conservation
            # This is automatically handled by the flow algorithm
    
    def bfs():
        parent = [-1] * (n + 2)
        visited = [False] * (n + 2)
        queue = deque([super_s])
        visited[super_s] = True
        
        while queue:
            u = queue.popleft()
            for v, edge_idx in graph[u]:
                if not visited[v] and capacity[edge_idx] > 0:
                    visited[v] = True
                    parent[v] = (u, edge_idx)
                    queue.append(v)
                    if v == super_t:
                        return parent
        return None
    
    def edmonds_karp():
        max_flow = 0
        flows = [0] * len(capacity)
        
        while True:
            parent = bfs()
            if parent is None:
                break
                
            # Find minimum capacity along the path
            path_flow = float('inf')
            s_node = super_t
            while s_node != super_s:
                u, edge_idx = parent[s_node]
                path_flow = min(path_flow, capacity[edge_idx])
                s_node = u
            
            # Update capacities and flows
            v = super_t
            while v != super_s:
                u, edge_idx = parent[v]
                capacity[edge_idx] -= path_flow
                capacity[edge_idx ^ 1] += path_flow
                flows[edge_idx] += path_flow
                flows[edge_idx ^ 1] -= path_flow
                v = u
            
            max_flow += path_flow
        
        return max_flow, flows
    
    max_flow, flows = edmonds_karp()
    
    if max_flow != d:
        print("NO")
        return
    
    # Extract the flow for original edges (even indices)
    result = []
    for i in range(0, 2*m, 2):
        result.append(flows[i])
    
    print("YES")
    print(" ".join(map(str, result)))

solve()
