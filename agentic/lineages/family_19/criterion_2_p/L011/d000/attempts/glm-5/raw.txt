```python
import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    idx = 0
    n = int(data[idx]); m = int(data[idx+1]); _ = int(data[idx+2]); idx += 3
    s = int(data[idx]); t = int(data[idx+1]); d = int(data[idx+2]); idx += 3
    edges = []
    adj = [[] for _ in range(n+1)]
    for j in range(m):
        u = int(data[idx]); v = int(data[idx+1]); w = int(data[idx+2]); idx += 3
        edges.append((u, v, w))
        adj[u].append((v, j))
        adj[v].append((u, j))
    
    # BFS to find shortest path from s to t
    parent_node = [0]*(n+1)
    parent_edge = [-1]*(n+1)
    visited = [False]*(n+1)
    q = deque([s])
    visited[s] = True
    while q:
        u = q.popleft()
        if u == t:
            break
        for (v, j) in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent_node[v] = u
                parent_edge[v] = j
                q.append(v)
    
    if not visited[t]:
        print("NO")
        return
    
    # Reconstruct path
    path_edges = []
    cur = t
    while cur != s:
        ej = parent_edge[cur]
        path_edges.append(ej)
        cur = parent_node[cur]
    path_edges.reverse()
    
    # Check min capacity along path
    min_cap = min(edges[ej][2] for ej in path_edges)
    if min_cap < d:
        print("NO")
        return
    
    ans = [0]*m
    for ej in path_edges:
        ans[ej] = d
    
    print("YES")
    print(' '.join(map(str, ans)))

solve()
```
