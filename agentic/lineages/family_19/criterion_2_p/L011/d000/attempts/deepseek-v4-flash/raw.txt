We can model this as a **max flow problem** on a directed graph where each edge has a capacity equal to the number of wagons available. We need to check if we can send at least `d` units from `s` to `t`. If yes, we output the flow on each edge.

We'll use Dinic's algorithm for max flow, which is efficient for this size.

```python
import sys
from collections import deque

input = sys.stdin.readline

class Dinic:
    def __init__(self, n):
        self.n = n
        self.graph = [[] for _ in range(n)]
    
    def add_edge(self, fr, to, cap):
        forward = [to, cap, None]
        backward = [fr, 0, forward]
        forward[2] = backward
        self.graph[fr].append(forward)
        self.graph[to].append(backward)
    
    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = deque()
        q.append(s)
        self.level[s] = 0
        while q:
            v = q.popleft()
            for to, cap, rev in self.graph[v]:
                if cap > 0 and self.level[to] < 0:
                    self.level[to] = self.level[v] + 1
                    q.append(to)
        return self.level[t] >= 0
    
    def dfs(self, v, t, f):
        if v == t:
            return f
        for i in range(self.it[v], len(self.graph[v])):
            self.it[v] = i
            to, cap, rev = self.graph[v][i]
            if cap > 0 and self.level[v] < self.level[to]:
                d = self.dfs(to, t, min(f, cap))
                if d > 0:
                    self.graph[v][i][1] -= d
                    rev[1] += d
                    return d
        return 0
    
    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if f == 0:
                    break
                flow += f
        return flow

def main():
    n, m, _ = map(int, input().split())
    s, t, d = map(int, input().split())
    s -= 1
    t -= 1
    
    edges = []
    dinic = Dinic(n)
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v, w))
        dinic.add_edge(u, v, w)
    
    flow = dinic.max_flow(s, t)
    if flow < d:
        print("NO")
        return
    
    # Extract flow on each edge
    result = []
    for u, v, w in edges:
        # Find the forward edge in dinic graph
        for to, cap, rev in dinic.graph[u]:
            if to == v and cap <= w:  # cap is remaining capacity
                used = w - cap
                result.append(str(used))
                break
        else:
            result.append("0")
    
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
