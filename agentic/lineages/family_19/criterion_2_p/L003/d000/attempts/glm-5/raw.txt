def main():
    data = sys.stdin.buffer.read().split()
    ptr = 0
    n = int(data[ptr]); ptr += 1

        return

    SS = n + 1
    TT = n + 2

    graph = [[] for _ in range(N)]
    to = []
    cap = []
    edge_ids = []

    for _ in range(m):
