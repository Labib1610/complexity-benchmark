We can model this as a **max flow** problem:

- States are nodes.
- Microservices are directed edges with capacity `cap`.
- We need to send exactly `d` units of flow from `s` to `t`.
- If max flow from `s` to `t` is at least `d`, we can output a valid flow assignment.

We’ll use **Dinic’s algorithm** to compute the max flow and also retrieve the flow on each edge.

```python
import sys
from collections import deque

input = sys.stdin.readline

class Dinic:
    def __init__(self, n):
        self.n = n
        self.graph = [[] for _ in range(n)]
        self.edges = []

    def add_edge(self, u, v, cap):
        self.graph[u].append(len(self.edges))
        self.edges.append([v, cap, len(self.graph[v])])
        self.graph[v].append(len(self.edges))
        self.edges.append([u, 0, len(self.graph[u]) - 1])

    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = deque([s])
        self.level[s] = 0
        while q:
            u = q.popleft()
            for ei in self.graph[u]:
                v, cap, _ = self.edges[ei]
                if cap > 0 and self.level[v] < 0:
                    self.level[v] = self.level[u] + 1
                    q.append(v)
        return self.level[t] >= 0

    def dfs(self, u, t, f):
        if u == t:
            return f
        for i in range(self.it[u], len(self.graph[u])):
            self.it[u] = i
            ei = self.graph[u][i]
            v, cap, rev = self.edges[ei]
            if cap > 0 and self.level[v] == self.level[u] + 1:
                ret = self.dfs(v, t, min(f, cap))
                if ret > 0:
                    self.edges[ei][1] -= ret
                    self.edges[rev][1] += ret
                    return ret
        return 0

    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if f == 0:
                    break
                flow += f
        return flow

def main():
    n, m, _ = map(int, input().split())
    s, t, d = map(int, input().split())
    s -= 1
    t -= 1

    dinic = Dinic(n)
    edge_indices = []
    for _ in range(m):
        u, v, cap = map(int, input().split())
        u -= 1
        v -= 1
        dinic.add_edge(u, v, cap)
        edge_indices.append(len(dinic.edges) - 2)  # forward edge index

    maxflow = dinic.max_flow(s, t)
    if maxflow < d:
        print("NO")
        return

    # Extract flow on each original edge
    result = []
    for idx in edge_indices:
        # original capacity is cap + current reverse flow
        # forward edge capacity after flow = original_cap - flow
        # so flow = original_cap - current_forward_capacity
        # original_cap = forward_cap + reverse_cap (since reverse starts 0)
        forward_cap = dinic.edges[idx][1]
        reverse_cap = dinic.edges[idx ^ 1][1]
        original_cap = forward_cap + reverse_cap
        flow = reverse_cap  # flow sent = amount in reverse edge
        result.append(flow)

    print("YES")
    print(" ".join(map(str, result)))

if __name__ == "__main__":
    main()
```
