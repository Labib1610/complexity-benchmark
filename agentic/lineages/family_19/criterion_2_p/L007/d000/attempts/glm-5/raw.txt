import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    radj = [[] for _ in range(n + 1)]
    
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        L = int(next(it))
        edges.append((u, v, L))
        adj[u].append(i)
        radj[v].append(i)
    
    # Build flow network for max flow
    # Node indices: 0 to m-1 for edges, m to m+n-1 for facilities
    # Source: m+s-1, Sink: m+t-1
    # Super source: m+n, Super sink: m+n+1
    
    N_nodes = m + n + 2
    SRC = m + n
    SNK = m + n + 1
    cap = [[] for _ in range(N_nodes)]
    
    # Edge structure: [to, cap, rev_idx]
    def add_edge(u, v, c):
        cap[u].append([v, c, len(cap[v])])
        cap[v].append([u, 0, len(cap[u]) - 1])
    
    # Edge nodes: 0 to m-1
    for i in range(m):
        u, v, L = edges[i]
        # Edge node i has capacity L
        add_edge(i, m + v - 1, L)
        # Incoming flow to edge node from facility u
        add_edge(m + u - 1, i, L)
    
    # Source and Sink connections
    # Source s: supply d
    add_edge(SRC, m + s - 1, d)
    # Sink t: demand d
    add_edge(m + t - 1, SNK, d)
    
    # Dinic's algorithm
    level = [0] * N_nodes
    ptr = [0] * N_nodes
    
    def bfs():
        for i in range(N_nodes):
            level[i] = -1
        level[SRC] = 0
        q = deque([SRC])
        while q:
            u = q.popleft()
            for v, c, rev in cap[u]:
                if c > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[SNK] != -1
    
    def dfs(u, flow):
        if u == SNK:
            return flow
        while ptr[u] < len(cap[u]):
            v, c, rev = cap[u][ptr[u]]
            if c > 0 and level[v] == level[u] + 1:
                pushed = dfs(v, min(flow, c))
                if pushed:
                    cap[u][ptr[u]][1] -= pushed
                    cap[v][rev][1] += pushed
                    return pushed
            ptr[u] += 1
        return 0
    
    total_flow = 0
    while bfs():
        ptr = [0] * N_nodes
        while True:
            pushed = dfs(SRC, float('inf'))
            if not pushed:
                break
            total_flow += pushed
    
    if total_flow != d:
        print("NO")
        return
    
    print("YES")
    # Extract flow values from edge nodes
    # Flow on edge i is the flow from node i to facility node m+v-1
    result = []
    for i in range(m):
        u, v, L = edges[i]
        # Find the edge from i to m+v-1
        flow_val = 0
        for e in cap[i]:
            if e[0] == m + v - 1:
                # Original capacity was L, current capacity is remaining
                flow_val = L - e[1]
                break
        result.append(str(flow_val))
    print(" ".join(result))

if __name__ == "__main__":
    solve()
