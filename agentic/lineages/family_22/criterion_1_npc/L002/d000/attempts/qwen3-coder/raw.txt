from collections import defaultdict, deque

def main():
    n, M, T, m = map(int, input().split())
    
    # Build graph
    graph = defaultdict(list)
    indegree = [0] * (n + 1)
    
    for _ in range(m):
        a, b = map(int, input().split())
        graph[a].append(b)
        indegree[b] += 1
    
    # Topological sort with level assignment
    queue = deque()
    for i in range(1, n + 1):
        if indegree[i] == 0:
            queue.append(i)
    
    # levels[i] will store the stage number for dish i
    levels = [0] * (n + 1)
    current_level = 1
    count_in_current_level = 0
    next_queue = deque()
    
    # Process nodes level by level respecting topological order and capacity constraint
    while queue:
        while queue and count_in_current_level < M:
            node = queue.popleft()
            levels[node] = current_level
            count_in_current_level += 1
            
            # Reduce indegree of neighbors
            for neighbor in graph[node]:
                indegree[neighbor] -= 1
                if indegree[neighbor] == 0:
                    next_queue.append(neighbor)
        
        # Move to next level
        current_level += 1
        count_in_current_level = 0
        queue = next_queue
        next_queue = deque()
    
    # Check if all nodes were processed
    for i in range(1, n + 1):
        if levels[i] == 0:
            print("NO")
            return
    
    # Check if we used more stages than allowed
    if current_level - 1 > T:
        print("NO")
        return
    
    # Pad stages to exactly T if needed (though not required by problem)
    # But we need to make sure no dish is assigned to a stage > T
    max_stage = max(levels[1:])
    if max_stage > T:
        print("NO")
        return
    
    print("YES")
    print(' '.join(map(str, levels[1:])))

if __name__ == "__main__":
    main()
