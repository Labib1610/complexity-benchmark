import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n+1)]
    radj = [[] for _ in range(n+1)]
    indeg = [0]*(n+1)
    outdeg = [0]*(n+1)
    
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        radj[b].append(a)
        indeg[b] += 1
        outdeg[a] += 1
    
    # Earliest stage: longest path from sources
    earliest = [1]*(n+1)
    q = deque()
    for i in range(1, n+1):
        if indeg[i] == 0:
            q.append(i)
    
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    
    if len(topo) != n:
        print("NO")
        return
    
    # Process in topological order to compute earliest
    for u in topo:
        for v in adj[u]:
            if earliest[v] < earliest[u] + 1:
                earliest[v] = earliest[u] + 1
    
    # Check earliest constraint
    for i in range(1, n+1):
        if earliest[i] > T:
            print("NO")
            return
    
    # Latest stage: longest path to sinks
    latest = [T]*(n+1)
    q = deque()
    for i in range(1, n+1):
        if outdeg[i] == 0:
            q.append(i)
    
    # Process in reverse topological order
    for u in reversed(topo):
        for v in radj[u]:
            if latest[v] > latest[u] - 1:
                latest[v] = latest[u] - 1
    
    # Check latest constraint
    for i in range(1, n+1):
        if latest[i] < 1:
            print("NO")
            return
    
    # Build potential stages and check earliest[i] <= latest[i]
    potential = [[] for _ in range(T+1)]
    for i in range(1, n+1):
        if earliest[i] > latest[i]:
            print("NO")
            return
        for t in range(earliest[i], latest[i]+1):
            potential[t].append(i)
    
    # Check Hall's condition: for any k consecutive stages, capacity must be sufficient
    # Simplified check: for consecutive stages, outer stage must cover inner
    prefix_count = [0]*(T+1)
    for t in range(1, T+1):
        prefix_count[t] = prefix_count[t-1] + len(potential[t])
    
    for left in range(1, T+1):
        cnt = prefix_count[left] - prefix_count[left-1]
        capacity = M
        for right in range(left, T+1):
            if right > left:
                capacity += M
            need = prefix_count[right] - prefix_count[left-1]
            if need > capacity:
                print("NO")
                return
    
    # Greedy assignment: assign each dish as early as possible if stage not full
    assignment = [0]*(n+1)
    stage_count = [0]*(T+1)
    
    for u in topo:
        assigned = False
        for t in range(earliest[u], latest[u]+1):
            if stage_count[t] < M:
                assignment[u] = t
                stage_count[t] += 1
                assigned = True
                break
        if not assigned:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, assignment[1:])))

solve()
