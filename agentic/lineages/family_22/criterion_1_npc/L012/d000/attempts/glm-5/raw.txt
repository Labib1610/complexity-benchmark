import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        indeg[b] += 1
    
    # Kahn's algorithm for topological sort
    q = deque([i for i in range(n) if indeg[i] == 0])
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    
    if len(topo) != n:
        print("NO")
        return
    
    # Earliest day assignment
    day = [1] * n
    for u in topo:
        for v in adj[u]:
            if day[v] < day[u] + 1:
                day[v] = day[u] + 1
    
    # Check if earliest days fit within T
    if max(day) > T:
        print("NO")
        return
    
    # Check capacity constraints
    cnt = [0] * (T + 1)
    for i in range(n):
        cnt[day[i]] += 1
    
    if any(c > M for c in cnt[1:]):
        # Need to spread out injections
        # Greedy assignment: process in topological order, assign earliest valid day
        day = [1] * n
        cnt = [0] * (T + 2)
        
        for u in topo:
            # Find earliest valid day for u
            d = 1
            while d <= T and cnt[d] >= M:
                d += 1
            if d > T:
                print("NO")
                return
            day[u] = d
            cnt[d] += 1
            
            # Update successors' earliest possible day
            for v in adj[u]:
                if day[v] < day[u] + 1:
                    day[v] = day[u] + 1
        
        if max(day) > T:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, day)))

if __name__ == "__main__":
    solve()
