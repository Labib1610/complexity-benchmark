import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    try:
        n = int(next(it))
        M = int(next(it))
        T = int(next(it))
        m = int(next(it))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        in_degree[b] += 1
    
    # Calculate longest path to each node (minimum shift required)
    # We process in topological order
    topo_order = []
    q = deque()
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            q.append(i)
    
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in adj[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                q.append(v)
    
    if len(topo_order) != n:
        print("NO")
        return
    
    # Calculate earliest possible shift for each component
    # Earliest shift is 1 + max(earliest shift of prerequisites)
    earliest = [1] * (n + 1)
    for u in topo_order:
        for v in adj[u]:
            if earliest[v] < earliest[u] + 1:
                earliest[v] = earliest[u] + 1
    
    # Check if any component requires a shift beyond T
    for i in range(1, n + 1):
        if earliest[i] > T:
            print("NO")
            return
    
    # Now we need to assign shifts such that:
    # 1. shift[i] >= earliest[i]
    # 2. shift[i] <= T
    # 3. At most M components per shift
    
    # Strategy: Process shifts from T down to 1
    # For each shift t, we must schedule all components with earliest == t
    # Plus we can schedule components with earliest < t if capacity allows
    
    # We'll use a greedy approach: for each shift t from T down to 1,
    # we schedule as many components as possible, prioritizing those with earliest == t
    
    result = [0] * (n + 1)
    # Group components by their earliest shift
    components_by_earliest = [[] for _ in range(T + 1)]
    for i in range(1, n + 1):
        components_by_earliest[earliest[i]].append(i)
    
    # Pending components that can be scheduled but haven't been assigned yet
    pending = []
    
    for t in range(T, 0, -1):
        # Add components that must be scheduled at shift t or earlier
        # Components with earliest == t must be scheduled at shift t or later
        # But since we're processing from T down, they must be scheduled now or earlier
        # Actually, components with earliest == t can be scheduled at any shift >= t
        # But we're processing from T down, so we should schedule them as late as possible
        # to leave room for components with higher earliest values
        
        # Add components with earliest == t to pending
        pending.extend(components_by_earliest[t])
        
        # Schedule up to M components from pending for this shift
        capacity = M
        newly_scheduled = []
        for comp in pending:
            if capacity == 0:
                newly_scheduled.append(comp)
            else:
                result[comp] = t
                capacity -= 1
        
        pending = newly_scheduled
    
    # Check if all components are scheduled
    if pending:
        print("NO")
        return
    
    # Verify the solution
    # Check capacity constraint
    shift_count = [0] * (T + 1)
    for i in range(1, n + 1):
        shift_count[result[i]] += 1
    for t in range(1, T + 1):
        if shift_count[t] > M:
            print("NO")
            return
    
    # Check prerequisite constraint
    for u in range(1, n + 1):
        for v in adj[u]:
            if result[u] >= result[v]:
                print("NO")
                return
    
    print("YES")
    print(' '.join(map(str, result[1:])))

if __name__ == "__main__":
    solve()
