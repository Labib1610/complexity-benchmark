import sys
from collections import deque

def solve():

    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    deadline = int(next(it))
    r = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)
    for _ in range(r):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        indeg[b] += 1
    
    # Kahn's algorithm to compute earliest possible slot respecting precedence
    # We also need to respect the per-slot capacity m.

    # We'll do a topological sort with a priority queue by earliest slot.

    # First compute the earliest slot for each node based on predecessors:
    # earliest[v] = 1 + max(earliest[u]) for all edges u->v
    # But we also need to ensure no slot has more than m tasks.
 
    # Approach: process nodes in topological order.
 For each node, its earliest slot is at least 1 + max(earliest[predecessors]).
    # Additionally, if that slot already has m tasks assigned, we may need to push it later.

    # We'll simulate assigning slots in topological order, using a min-heap of available nodes
    # sorted by their current earliest possible slot (which depends on predecessors' assigned slots).
    # When we pop a node, we assign it the smallest slot >= its earliest possible that has capacity < m.
    # Then update successors' earliest possible slot.

 
    # We'll maintain earliest_possible[v] = the smallest slot it could be placed based on predecessors.

    earliest_possible = [1] * (n + 1)
    # For nodes with no predecessors, earliest_possible = 1.

 
    # We'll use a priority queue: (earliest_possible[v], v) for nodes whose indegree has become 0.
    # But earliest_possible may increase when a predecessor is assigned a later slot than its earliest.

    # So we need to update successors when a node is assigned its actual slot.

    # We'll do a standard topological sort with a min-heap keyed by earliest_possible.

    pq = []
    for i in range(1, n+1):
        if indeg[i] == 0:
            heapq.heappush(pq, (earliest_possible[i], i)))
    
    import heapq
    assigned = [0] * (n + 1)
    slot_count = {}  # slot -> number of tasks assigned
    processed = 0
    while pq:
        earliest, u = heapq.heappop(pq)
        # Find the smallest slot >= earliest that has capacity < m
        slot = earliest
        while slot_count.get(slot, 0) >= m:
            slot += 1
        if slot > deadline:
            print("NO")
            return
        assigned[u] = slot
        slot_count[slot] = slot_count.get(slot, 0) + 1
        processed += 1
        # Update successors
        for v in adj[u]:
            indeg[v] -= 1
            # The earliest possible for v is at least assigned[u] + 1
            if assigned[u] + 1 > earliest_possible[v]:
                earliest_possible[v] = assigned[u] + 1
            if indeg[v] == 0:
                heapq.heappush(pq, (earliest_possible[v], v))
    
    if processed != n:
        print("NO")
        return
    
    # Check if any slot exceeds capacity (shouldn't happen but just in case)
    for cnt in slot_count.values():
        if cnt > m:
            print("NO")
            return
    
    print("YES")
    print(' '.join(str(assigned[i]) for i in range(1, n+1))))

    solve()
