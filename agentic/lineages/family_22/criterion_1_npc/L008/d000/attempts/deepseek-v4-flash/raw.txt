We can model this as a scheduling problem on a DAG (directed acyclic graph) where each node is a compound, and edges represent ingredient dependencies.  

We need to assign each node a stage from 1 to T such that:
- If there is an edge a → b, then stage(a) < stage(b).
- No stage has more than M nodes.

We can solve this using a greedy topological ordering with capacity constraints.  
A common approach is to process nodes in topological order, assigning the smallest possible stage that respects dependencies and capacity.

Here’s the implementation:

```python
from collections import deque
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        indeg[b] += 1
    
    # Kahn's algorithm with stage assignment
    stage = [0] * n
    # queue holds (node, earliest_stage_possible)
    q = deque()
    for i in range(n):
        if indeg[i] == 0:
            q.append((i, 1))
    
    processed = 0
    # current stage fill count
    stage_count = [0] * (T + 1)
    
    while q:
        # sort by earliest stage to fill capacity properly
        q = deque(sorted(q, key=lambda x: x[1]))
        node, earliest = q.popleft()
        
        # find the smallest stage >= earliest with capacity
        s = earliest
        while s <= T and stage_count[s] >= M:
            s += 1
        if s > T:
            print("NO")
            return
        
        stage[node] = s
        stage_count[s] += 1
        processed += 1
        
        for nb in adj[node]:
            indeg[nb] -= 1
            if indeg[nb] == 0:
                q.append((nb, s + 1))
    
    if processed != n:
        print("NO")
        return
    
    print("YES")
    print(" ".join(str(stage[i]) for i in range(n)))

if __name__ == "__main__":
    solve()
```
