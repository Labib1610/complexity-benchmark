from collections import defaultdict, deque

def solve():
    n, M, T, m = map(int, input().split())
    
    # Build graph and in-degree count
    graph = defaultdict(list)
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        a, b = map(int, input().split())
        graph[a].append(b)
        in_degree[b] += 1
    
    # Topological sort with slot assignment
    # We use a greedy approach: assign each task to the earliest possible slot
    # considering both precedence and station capacity constraints
    
    # Initialize slots array
    slots = [0] * (n + 1)
    
    # For each time slot, keep track of how many tasks are assigned
    slot_count = [0] * (T + 1)
    
    # Queue for topological sort (tasks with no predecessors)
    queue = deque()
    
    # Find all tasks with no dependencies
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            queue.append(i)
    
    # Process tasks in topological order
    while queue:
        # For each slot from 1 to T, try to schedule as many tasks as possible
        # But we process tasks in topological order
        task = queue.popleft()
        
        # Find the earliest slot where this task can be scheduled
        # It must be after all its predecessors
        earliest_slot = 1
        for pred in graph:
            if task in graph[pred]:
                earliest_slot = max(earliest_slot, slots[pred] + 1)
        
        # Try to assign task to the earliest available slot
        assigned = False
        for slot in range(earliest_slot, T + 1):
            if slot_count[slot] < M:
                slots[task] = slot
                slot_count[slot] += 1
                assigned = True
                break
        
        if not assigned:
            print("NO")
            return
        
        # Check successors
        for succ in graph[task]:
            in_degree[succ] -= 1
            if in_degree[succ] == 0:
                queue.append(succ)
    
    # Check if all tasks were scheduled
    for i in range(1, n + 1):
        if slots[i] == 0:
            print("NO")
            return
    
    print("YES")
    print(*slots[1:])

solve()
