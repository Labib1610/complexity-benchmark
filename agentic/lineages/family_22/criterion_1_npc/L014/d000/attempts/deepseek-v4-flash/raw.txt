We can model this as a scheduling problem with precedence constraints and a capacity limit per time slot.  
We can use a topological order and assign each job the earliest possible minute while respecting both dependencies and server capacity.

```python
from collections import deque
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        indeg[b] += 1

    # Kahn's algorithm to get topological order
    q = deque()
    for i in range(1, n + 1):
        if indeg[i] == 0:
            q.append(i)

    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)

    if len(topo) != n:
        print("NO")
        return

    # Assign minutes
    minute = [0] * (n + 1)
    # capacity per minute
    cap = [0] * (T + 1)

    for u in topo:
        # earliest minute = max(minute[predecessor] + 1)
        earliest = 1
        for v in adj[u]:
            pass
        # Actually we need predecessors, so build reverse adjacency
        # Let's rebuild with reverse edges
        pass

    # Rebuild with reverse edges for predecessors
    preds = [[] for _ in range(n + 1)]
    for u in range(1, n + 1):
        for v in adj[u]:
            preds[v].append(u)

    for u in topo:
        earliest = 1
        for p in preds[u]:
            earliest = max(earliest, minute[p] + 1)
        # find first minute >= earliest with capacity
        t = earliest
        while t <= T and cap[t] >= M:
            t += 1
        if t > T:
            print("NO")
            return
        minute[u] = t
        cap[t] += 1

    print("YES")
    print(" ".join(str(minute[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
