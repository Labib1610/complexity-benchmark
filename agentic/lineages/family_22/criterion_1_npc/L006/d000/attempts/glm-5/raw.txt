import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    in_deg = [0] * n
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        in_deg[b] += 1
    
    # Earliest cycle: topological order, can we assign all <= T?
    # Must have critical path length <= T.
    earliest = [1] * n
    q = deque()
    for i in range(n):
        if in_deg[i] == 0:
            q.append(i)
    
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            if earliest[v] < earliest[u] + 1:
                earliest[v] = earliest[u] + 1
            in_deg[v] -= 1
            if in_deg[v] == 0:
                q.append(v)
    
    if len(topo) != n:
        print("NO")
        return
    
    max_earliest = max(earliest)
    if max_earliest > T:
        print("NO")
        return
    
    # Latest cycle: reverse topological order
    latest = [T] * n
    for u in reversed(topo):
        for v in adj[u]:
            if latest[u] > latest[v] - 1:
                latest[u] = latest[v] - 1
    
    if min(latest) < 1:
        print("NO")
        return
    
    # Groups by earliest, handles longest paths first in each group
    groups = [[] for _ in range(T + 1)]
    for i in range(n):
        groups[earliest[i]].append(i)
    
    # Sort each group by latest ascending (so critical items get earlier slots)
    for g in range(1, T + 1):
        groups[g].sort(key=lambda x: latest[x])
    
    cycles = [0] * n
    for c in range(1, T + 1):
        positions = list(range(M))
        for idx, u in enumerate(groups[c]):
            cycles[u] = c  # all in same cycle if earliest[u] is enforced
    
    # Prepare answer list sorted by step number
    ans = [0] * n
    for i in range(n):
        ans[i] = cycles[i]
    
    print("YES")
    print(" ".join(map(str, ans)))

if __name__ == "__main__":
    solve()
