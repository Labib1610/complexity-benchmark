We need to separate two nodes in an undirected weighted graph by removing edges, with a total cost limit. This is essentially finding a minimum cut between `s` and `t` and checking if its cost is ≤ `B`.

We can use a max-flow/min-cut algorithm (e.g., Dinic’s) on the graph with edge capacities equal to the suspension costs. If the min-cut cost ≤ `B`, we output the edges in that cut.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

class Dinic:
    def __init__(self, n):
        self.n = n
        self.graph = [[] for _ in range(n)]
    
    def add_edge(self, u, v, cap):
        self.graph[u].append([v, cap, len(self.graph[v])])
        self.graph[v].append([u, 0, len(self.graph[u]) - 1])
    
    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = deque([s])
        self.level[s] = 0
        while q:
            u = q.popleft()
            for v, cap, rev in self.graph[u]:
                if cap > 0 and self.level[v] < 0:
                    self.level[v] = self.level[u] + 1
                    q.append(v)
        return self.level[t] >= 0
    
    def dfs(self, u, t, f):
        if u == t:
            return f
        for i in range(self.it[u], len(self.graph[u])):
            self.it[u] = i
            v, cap, rev = self.graph[u][i]
            if cap > 0 and self.level[u] + 1 == self.level[v]:
                ret = self.dfs(v, t, min(f, cap))
                if ret > 0:
                    self.graph[u][i][1] -= ret
                    self.graph[v][rev][1] += ret
                    return ret
        return 0
    
    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if f == 0:
                    break
                flow += f
        return flow

def main():
    n, m, s, t, B = map(int, input().split())
    s -= 1
    t -= 1
    edges = []
    dinic = Dinic(n)
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v, w))
        dinic.add_edge(u, v, w)
        dinic.add_edge(v, u, w)  # undirected: add both directions with capacity w
    
    min_cut_cost = dinic.max_flow(s, t)
    if min_cut_cost > B:
        print("NO")
        return
    
    # Find reachable nodes from s in residual graph
    reachable = [False] * n
    q = deque([s])
    reachable[s] = True
    while q:
        u = q.popleft()
        for v, cap, rev in dinic.graph[u]:
            if cap > 0 and not reachable[v]:
                reachable[v] = True
                q.append(v)
    
    cut_edges = []
    for u, v, w in edges:
        if reachable[u] != reachable[v]:
            cut_edges.append((u+1, v+1))
    
    print("YES")
    print(len(cut_edges))
    for u, v in cut_edges:
        print(u, v)

if __name__ == "__main__":
    main()
```
