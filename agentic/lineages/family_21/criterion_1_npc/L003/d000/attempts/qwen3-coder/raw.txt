import sys

def find_reachable(graph, start, blocked_edges, n):
    """Find all nodes reachable from start using BFS, avoiding blocked edges"""
    visited = [False] * (n + 1)
    queue = deque([start])
    visited[start] = True
    reachable = {start}
    
    while queue:
        node = queue.popleft()
        for neighbor, edge_id in graph[node]:
            # Check if this edge is blocked
            if (node, neighbor) in blocked_edges or (neighbor, node) in blocked_edges:
                continue
            if not visited[neighbor]:
                visited[neighbor] = True
                reachable.add(neighbor)
                queue.append(neighbor)
    
    return reachable

def is_secure(graph, reactive_sites, blocked_edges, n):
    """Check if no two reactive sites can reach each other"""
    reachable_from_each = {}
    
    for site in reactive_sites:
        reachable_from_each[site] = find_reachable(graph, site, blocked_edges, n)
    
    # Check if any two reactive sites can reach each other
    for i in range(len(reactive_sites)):
        for j in range(i + 1, len(reactive_sites)):
            site1, site2 = reactive_sites[i], reactive_sites[j]
            if site2 in reachable_from_each[site1] or site1 in reachable_from_each[site2]:
                return False
    
    return True
