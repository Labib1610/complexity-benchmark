We can solve this using a backtracking approach with pruning, since **n ≤ 46** is small enough for a meet-in-the-middle or branch-and-bound search.

Here’s the Python solution:

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Try to find an independent set of size k
    # Use backtracking with pruning
    chosen = []
    def backtrack(start, chosen_mask, chosen_count):
        if chosen_count == k:
            return True
        if start >= n:
            return False
        # Prune: if even taking all remaining vertices can't reach k
        if chosen_count + (n - start) < k:
            return False
        # Try to include start
        if (chosen_mask & adj[start]) == 0:
            chosen.append(start + 1)
            if backtrack(start + 1, chosen_mask | (1 << start), chosen_count + 1):
                return True
            chosen.pop()
        # Try to exclude start
        if backtrack(start + 1, chosen_mask, chosen_count):
            return True
        return False

    if backtrack(0, 0, 0):
        print("YES")
        print(" ".join(map(str, chosen)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
