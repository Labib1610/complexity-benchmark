import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)
    
    # Greedy reduction: remove vertex with highest degree if too many vertices
    S = set(range(n))
    while len(S) > k:
        # Find vertex in S with max degree within S
        best = -1
        best_deg = -1
        for u in S:
            deg = len(adj[u] & S)
            if deg > best_deg:
                best_deg = deg
                best = u
        S.remove(best)
    
    # Check if remaining set is independent
    for u in S:
        if adj[u] & S:
            print("NO")
            return
    
    print("YES")
    result = [str(x + 1) for x in S]
    print(' '.join(result))

solve()
