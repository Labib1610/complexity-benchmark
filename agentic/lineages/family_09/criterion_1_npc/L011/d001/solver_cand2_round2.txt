import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1] |= (1 << (v-1))
        adj[v-1] |= (1 << (u-1))

    # The problem is to find an Independent Set of size k.
    # Since n <= 46, we use a meet-in-the-middle approach.
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, adj_list):
        # nodes is a list of indices
        size = len(nodes)
        res = {} # size -> list of bitmasks
        for i in range(1 << size):
            mask = 0
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    if adj_list[nodes[j]] & mask:
                        valid = False
                        break
                    mask |= (1 << nodes[j])
            if valid:
                count = bin(i).count('1')
                if count not in res: res[count] = []
                res[count].append(mask)
        return res

    adj1 = get_independent_sets(list(range(n1)), adj)
    adj2 = get_independent_sets(list(range(n1, n)), adj)

    for s1 in range(k + 1):
        s2 = k - s1
        if s1 in adj1 and s2 in adj2:
            # Pre-calculate constraints for second half
            # For each mask in adj2, check if it conflicts with any mask in adj1
            # To optimize, we need to check if mask2 is compatible with mask1
            # mask2 is compatible with mask1 if (adj[v] & mask1) == 0 for all v in mask2
            # This is equivalent to: mask1 & (union of adj[v] for v in mask2) == 0
            
            # Precompute union of adjacencies for each mask in adj2
            adj2_unions = []
            for m2 in adj2[s2]:
                u = 0
                for i in range(n2):
                    if (m2 >> (n1 + i)) & 1:
                        u |= adj[n1 + i]
                adj2_unions.append((m2, u))
            
            for m1 in adj1[s1]:
                for m2, u in adj2_unions:
                    if not (m1 & u):
                        # Found it
                        res = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                res.append(i + 1)
                        print("YES")
                        print(*(res))
                        return

    print("NO")

solve()
