```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    comps = []          # list of (nodes_in_comp, (c0_nodes, c1_nodes))
    ok = True
    for i in range(n):
        if color[i] != -1:
            continue
        stack = [i]
        color[i] = 0
        nodes = [i]
        c0 = [i]
        c1 = []
        while stack:
            v = stack.pop()
            for to in adj[v]:
                if color[to] == -1:
                    color[to] = color[v] ^ 1
                    if color[to] == 0:
                        c0.append(to)
                    else:
                        c1.append(to)
                    stack.append(to)
                    nodes.append(to)
                elif color[to] == color[v]:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            break
        comps.append((c0, c1))

    if not ok:
        print("NO")
        return

    # DP for possible sizes
    dp = {0: []}  # size -> list of choices (0 means take c0, 1 means take c1) per component processed
    for idx, (c0, c1) in enumerate(comps):
        sz0, sz1 = len(c0), len(c1)
        ndp = {}
        for cur_sz, path in dp.items():
            # take c0
            ns = cur_sz + sz0
            if ns <= k and ns not in ndp:
                ndp[ns] = path + [0]
            # take c1
            ns = cur_sz + sz1
            if ns <= k and ns not in ndp:
                ndp[ns] = path + [1]
        dp = ndp
        if not dp:
            break

    if k not in dp:
        print("NO")
        return

    # reconstruct selection
    choices = dp[k]
    result = []
    for (c0, c1), ch in zip(comps, choices):
        result.extend(c0 if ch == 0 else c1)

    print("YES")
    print(' '.join(str(v + 1) for v in result))

if __name__ == "__main__":
    solve()
```
