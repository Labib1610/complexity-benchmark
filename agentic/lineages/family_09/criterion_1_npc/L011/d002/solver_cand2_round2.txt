import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle for Maximum Independent Set of size k
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for both halves
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        sets = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            is_independent = True
            mask = 0
            count = 0
            for j in range(size):
                if (i >> j) & 1:
                    u = nodes[j]
                    if (adj_list[u] & mask) != 0:
                        is_independent = False
                        break
                    mask |= (1 << u)
                    count += 1
            if is_independent:
                sets[count].append(mask)
        return sets

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Check compatibility
    for c1 in range(min(k + 1, n1 + 1)):
        c2 = k - c1
        if 0 <= c2 <= n2:
            for m1 in sets1[c1]:
                # Find nodes in second half compatible with m1
                forbidden = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        forbidden |= adj[i]
                
                for m2 in sets2[c2]:
                    if (m2 & forbidden) == 0:
                        # Found it
                        res = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                res.append(i + 1)
                        print("YES")
                        print(*(res))
                        return
    print("NO")

solve()
