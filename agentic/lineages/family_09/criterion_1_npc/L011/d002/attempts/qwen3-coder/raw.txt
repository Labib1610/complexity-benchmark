from itertools import combinations

def solve():
    n, m, k = map(int, input().split())
    
    # Read interactions
    interactions = []
    for _ in range(m):
        u, v = map(int, input().split())
        interactions.append((u, v))
    
    # If no interactions, any combination works
    if m == 0:
        print("YES")
        print(*range(1, k+1))
        return
    
    # Try all possible combinations of k members for the focus panel
    all_members = set(range(1, n+1))
    
    # For each combination of k members as focus panel
    for focus_panel in combinations(range(1, n+1), k):
        focus_set = set(focus_panel)
        control_group = all_members - focus_set
        
        # Check if control group contains at least one participant from each interaction
        valid = True
        for u, v in interactions:
            if u not in control_group and v not in control_group:
                valid = False
                break
        
        if valid:
            print("YES")
            print(*focus_panel)
            return
    
    print("NO")

solve()
