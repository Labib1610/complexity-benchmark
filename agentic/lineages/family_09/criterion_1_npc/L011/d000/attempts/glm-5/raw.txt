import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    full = (1 << (n + 1)) - 1
    base = (1 << 1) - 1
    mask_all = (1 << (n + 1)) - 2
    
    def find_clique(target, available, chosen):
        if chosen == target:
            return True
        rem = target - chosen
        avail = available
        cnt = bin(avail).count('1')
        if cnt < rem:
            return False
        while avail:
            low = avail & (-avail)
            v = low.bit_length() - 1
            new_avail = available & ~(1 << v)
            new_chosen = chosen | (1 << v)
            if find_clique(target, new_avail, new_chosen):
                return True
            available &= ~(1 << v)
            avail = available
            cnt -= 1
            if cnt < rem:
                return False
        return False
    
    if find_clique(k, mask_all, 0):
        chosen = []
        for v in range(1, n + 1):
            if find_clique(k, mask_all, 0):
                pass
        # Reconstruct one valid clique
        chosen = []
        avail = mask_all
        for _ in range(k):
            for v in range(1, n + 1):
                if avail & (1 << v):
                    new_avail = avail & ~(1 << v)
                    if find_clique(k - len(chosen) - 1, new_avail, 0):
                        chosen.append(v)
                        avail = new_avail
                        break
            else:
                break
        print("YES")
        print(' '.join(map(str, chosen)))
    else:
        print("NO")

solve()
