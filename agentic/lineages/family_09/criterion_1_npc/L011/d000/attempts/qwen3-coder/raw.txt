from collections import defaultdict
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    index = 3
    conflicts = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        conflicts[u].add(v)
        conflicts[v].add(u)
        index += 2
    
    # Generate all possible subsets of size k using bit manipulation
    # Since n <= 46, we need to optimize. We'll use meet-in-the-middle approach.
    
    if k > n:
        print("NO")
        return
    
    # For small n, we can check all combinations
    # But since n can be up to 46, we need a better approach.
    # Let's use backtracking with pruning.
    
    # However, given the constraints and that we only need one solution,
    # let's try a greedy approach combined with checking validity.
    
    # Better approach: Use bitmasks and meet-in-the-middle since n<=46
    # Split the set into two halves
    mid = n // 2
    first_half = list(range(mid))
    second_half = list(range(mid, n))
    
    # Precompute conflict masks for each half
    # For each subset of first half, precompute its conflict set
    first_conflicts = {}
    first_valid = []
    
    for mask in range(1 << mid):
        subset = [i for i in range(mid) if mask & (1 << i)]
        if len(subset) > k:
            continue
        conflict_set = set()
        valid = True
        for i in subset:
            if conflicts[i] & conflict_set:
                valid = False
                break
            conflict_set.add(i)
            conflict_set.update(conflicts[i])
        if valid:
            first_valid.append((mask, frozenset(conflict_set), len(subset)))
            first_conflicts[mask] = frozenset(conflict_set)
    
    # Similarly for second half
    second_conflicts = {}
    second_valid = []
    
    for mask in range(1 << (n - mid)):
        subset = [i + mid for i in range(n - mid) if mask & (1 << i)]
        if len(subset) > k:
            continue
        conflict_set = set()
        valid = True
        for i in subset:
            if conflicts[i] & conflict_set:
                valid = False
                break
            conflict_set.add(i)
            conflict_set.update(conflicts[i])
        if valid:
            second_valid.append((mask, frozenset(conflict_set), len(subset)))
            second_conflicts[mask] = frozenset(conflict_set)
    
    # Now check all pairs
    for mask1, conflict1, size1 in first_valid:
        if size1 > k:
            continue
        needed = k - size1
        if needed < 0:
            continue
        for mask2, conflict2, size2 in second_valid:
            if size2 != needed:
                continue
            # Check if there's conflict between the two sets
            subset1 = [i for i in range(mid) if mask1 & (1 << i)]
            subset2 = [i + mid for i in range(n - mid) if mask2 & (1 << i)]
            
            # Check internal validity is already done
            # Check cross validity
            total_conflict = conflict1 | conflict2
            valid = True
            for i in subset1:
                if i in total_conflict:
                    valid = False
                    break
            if not valid:
                continue
            for i in subset2:
                if i in total_conflict:
                    valid = False
                    break
            if valid:
                result = subset1 + subset2
                print("YES")
                print(' '.join(str(x + 1) for x in result))
                return
    
    # If k=1, just pick any element that doesn't conflict with itself (which is impossible)
    # Actually, a member never conflicts with themselves based on the problem description
    # So for k=1, we can always pick any member unless there's a self-conflict which isn't allowed
    
    # Special case: k == 1
    if k == 1:
        for i in range(n):
            # A member can be selected alone if they don't conflict with anyone already selected
            # Since the panel is empty initially, we just need to ensure we can add them
            # According to rule 3, when we add a member, others who share events with them become incompatible
            # But for k=1, we only select one member, so no issue
            print("YES")
            print(i + 1)
            return
            
    print("NO")

if __name__ == "__main__":
    main()
