import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The control group (size n-k) must cover all m edges.
    # This means no edge can have both endpoints in the focus panel (size k).
    # Let S be the set of members in the focus panel.
    # Condition: For every edge (u, v), it is NOT the case that (u in S and v in S).
    # This is equivalent to saying S must be an Independent Set of size k.
    
    # Since n <= 46, we use meet-in-the-middle to find an independent set of size k.
    n1 = n // 2
    n2 = n - n1
    
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    def get_independent_sets(nodes, offset):
        size = len(nodes)
        res = {} # size -> list of bitmasks
        for i in range(1 << size):
            mask = 0
            valid = True
            count = 0
            for j in range(size):
                if (i >> j) & 1:
                    if mask & adj[nodes[j]]:
                        valid = False
                        break
                    mask |= (1 << nodes[j])
                    count += 1
            if valid:
                if count not in res: res[count] = []
                res[count].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, 0)
    sets2 = get_independent_sets(nodes2, n1)
    
    # Check if any combination of sets1[c1] and sets2[c2] where c1+c2 == k
    # forms an independent set.
    # Two masks m1 and m2 are compatible if (m1 & adj_mask_for_m2) == 0
    # Precompute adjacency for each mask in sets2
    adj_masks2 = {}
    for c2, masks in sets2.items():
        adj_masks2[c2] = []
        for m2 in masks:
            a_m2 = 0
            for i in range(n2):
                if (m2 >> (i + n1)) & 1:
                    a_m2 |= adj[i + n1]
            adj_masks2[c2].append((m2, a_m2))

    for c1 in range(k + 1):
        c2 = k - c1
        if c1 in sets1 and c2 in sets2:
            for m1 in sets1[c1]:
                a_m1 = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        a_m1 |= adj[i]
                
                for m2, a_m2 in adj_masks2[c2]:
                    if not (m1 & a_m2) and not (m2 & a_m1):
                        # Found it
                        res_mask = m1 | m2
                        ans = []
                        for i in range(n):
                            if (res_mask >> i) & 1:
                                ans.append(i + 1)
                        print("YES")
                        print(*(ans))
                        return
    print("NO")

solve()
