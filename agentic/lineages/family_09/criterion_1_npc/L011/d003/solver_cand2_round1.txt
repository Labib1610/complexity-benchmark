import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
        
    # A set of k members is valid if and only if no edge is contained entirely 
    # within the set of members NOT in the panel.
    # Let S be the set of members in the panel. |S| = k.
    # Let V \ S be the set of members NOT in the panel. |V \ S| = n - k.
    # The condition "zero events are assigned to any member of the focus panel"
    # means that for every edge (u, v), at least one of u or v must be in V \ S.
    # This is equivalent to saying that the set V \ S must be a Vertex Cover 
    # of the graph.
    # So we need to find a Vertex Cover of size n - k.
    
    target_vc_size = n - k
    
    # We need to find a vertex cover of size at most target_vc_size.
    # Since n <= 46, we can use a search algorithm for Vertex Cover.
    
    def get_vc(current_edges, k_rem):
        if not current_edges:
            return []
        if k_rem <= 0:
            return None
        
        u, v = current_edges[0]
        
        # Try picking u
        res = get_vc([e for e in current_edges if e[0] != u and e[1] != u], k_rem - 1)
        if res is not None:
            return [u] + res
            
        # Try picking v
        res = get_vc([e for e in current_edges if e[0] != v and e[1] != v], k_rem - 1)
        if res is not None:
            return [v] + res
            
        return None

    vc = get_vc(edges, target_vc_size)
    
    if vc is not None:
        # vc is a vertex cover of size <= target_vc_size.
        # We need exactly n-k members in the complement.
        # If |vc| < n-k, we can add arbitrary members to it.
        vc_set = set(vc)
        remaining = [i for i in range(n) if i not in vc_set]
        while len(vc_set) < target_vc_size and remaining:
            vc_set.add(remaining.pop())
            
        if len(vc_set) == target_vc_size:
            # The panel is the complement of the vertex cover
            panel = [i + 1 for i in range(n) if i not in vc_set]
            print("YES")
            print(*(panel))
        else:
            print("NO")
    else:
        print("NO")

solve()
