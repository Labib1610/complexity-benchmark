import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # The condition "at least one of the two participating users is in the background"
    # is equivalent to saying the primary focus panel must be an Independent Set.
    # We need to find an independent set of size exactly k.
    
    # Since n <= 46, we use meet-in-the-middle.
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        # mask represents subset of nodes
        for i in range(1 << size):
            is_independent = True
            count = 0
            for j in range(size):
                if (i >> j) & 1:
                    count += 1
                    if adj_list[nodes[j]] & i:
                        is_independent = False
                        break
            if is_independent:
                # Store the actual bitmask relative to the original graph
                mask = 0
                for j in range(size):
                    if (i >> j) & 1:
                        mask |= (1 << nodes[j])
                res[count].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Precompute adjacency for sets2 to check compatibility
    # A set m1 from sets1 and m2 from sets2 are compatible if no edges between them
    for c1 in range(n1 + 1):
        for m1 in sets1[c1]:
            # Find nodes in m1 that have neighbors in the second half
            forbidden_in_2 = 0
            for i in range(n1):
                if (m1 >> i) & 1:
                    forbidden_in_2 |= adj[i]
            
            needed = k - c1
            if 0 <= needed <= n2:
                for m2 in sets2[needed]:
                    if not (m2 & forbidden_in_2):
                        # Found it
                        res = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                res.append(i + 1)
                        print("YES")
                        print(*(res))
                        return
    print("NO")

solve()
