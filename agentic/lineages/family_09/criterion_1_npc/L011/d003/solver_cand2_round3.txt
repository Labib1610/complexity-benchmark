import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The problem asks for an independent set of size k.
    # Since n <= 46, we can use a meet-in-the-middle approach or a simple backtracking
    # with pruning. Given the constraints and the nature of the problem,
    # a backtracking search with bitmasking is efficient enough.
    
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    def get_independent_set(candidates, current_set, k_needed):
        if k_needed == 0:
            return current_set
        if not candidates or len(candidates) < k_needed:
            return None
        
        # Heuristic: pick the vertex with the most neighbors in candidates to prune
        v = candidates[0]
        
        # Option 1: Include v in the set
        new_candidates = [c for c in candidates[1:] if not (adj[v] & (1 << c))]
        res = get_independent_set(new_candidates, current_set + [v], k_needed - 1)
        if res is not None:
            return res
        
        # Option 2: Exclude v
        res = get_independent_set(candidates[1:], current_set, k_needed)
        return res

    # For n up to 46, simple backtracking might be slow in worst case,
    # but the constraints on m and the nature of independent set usually allow it.
    # Let's use a slightly more optimized search.
    
    def search(candidates_mask, k_needed, current_set):
        if k_needed == 0:
            return current_set
        if bin(candidates_mask).count('1') < k_needed:
            return None
        
        # Pick vertex with highest degree in the subgraph
        v = (candidates_mask & -candidates_mask).bit_length() - 1
        
        # Try including v
        res = search(candidates_mask & ~adj[v] & ~(1 << v), k_needed - 1, current_set + [v + 1])
        if res is not None:
            return res
        
        # Try excluding v
        res = search(candidates_mask & ~(1 << v), k_needed, current_set)
        return res

    result = search((1 << n) - 1, k, [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
