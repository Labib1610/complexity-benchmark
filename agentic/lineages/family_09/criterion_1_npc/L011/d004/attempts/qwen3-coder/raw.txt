```python
import sys
from itertools import combinations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    idx = 3
    edges = []
    adj = [set() for _ in range(n+1)]
    
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Precompute degrees
    deg = [0] * (n+1)
    for u in range(1, n+1):
        deg[u] = len(adj[u])
    
    # Function to check if a subset satisfies x^T L x = sum D_{i,i} x_i
    def is_valid(subset):
        x = [0] * (n+1)
        for i in subset:
            x[i] = 1
        
        # Compute x^T L x
        # L = D - A
        # x^T L x = x^T D x - x^T A x
        # x^T D x = sum_{i in S} deg[i]
        # x^T A x = sum_{i,j in S, i~j} 2 = 2 * number of edges within S
        lhs = 0
        edge_count = 0
        for u in subset:
            lhs += deg[u]
            for v in adj[u]:
                if v in subset and v > u:  # To count each edge once
                    edge_count += 1
        
        rhs = lhs
        lhs = lhs - 2 * edge_count
        return lhs == rhs
    
    # Try all combinations of size k
    from itertools import combinations
    candidates = list(range(1, n+1))
    
    for combo in combinations(candidates, k):
        if is_valid(combo):
            print("YES")
            print(' '.join(map(str, combo)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
```
