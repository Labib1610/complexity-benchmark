import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # Bron-Kerbosch with pivoting to find max clique, 
    # but we only need a clique of size k.
    # Since n <= 46, we can use a backtracking search with pruning.
    
    def find_clique(candidates, current_clique):
        if len(current_clique) == k:
            return current_clique
        
        if not candidates:
            return None
        
        # Pruning: if remaining candidates + current_clique < k, return
        if len(current_clique) + bin(candidates).count('1') < k:
            return None
            
        # Heuristic: pick vertex with most neighbors in candidates
        v = (candidates & -candidates).bit_length() - 1
        
        # Try including v
        res = find_clique(candidates & ~adj[v] & ~(1 << v), current_clique + [v + 1])
        if res:
            return res
            
        # Try excluding v
        return find_clique(candidates & ~(1 << v), current_clique)

    # Optimization: use bitmask for candidates
    def search(candidates, current_clique):
        if len(current_clique) == k:
            return current_clique
        if len(current_clique) + bin(candidates).count('1') < k:
            return None
        
        while candidates:
            # Pick vertex with highest index
            v = candidates.bit_length() - 1
            res = search(candidates & ~adj[v] & ~(1 << v), current_clique + [v + 1])
            if res:
                return res
            candidates &= ~(1 << v)
        return None

    result = search((1 << n) - 1, [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
