import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    # Precompute independent sets of size up to 3
    ind3 = []
    for i in range(1, n + 1):
        if k == 1:
            print("YES")
            print(i)
            return
        ind3.append((1 << i, i))
    
    if k == 2:
        for mask, u in ind3:
            avail = ((1 << (n + 1)) - 2) & ~mask & ~adj[u]
            v = (avail & -avail).bit_length() - 1
            if v > 0:
                print("YES")
                print(u, v)
                return
        print("NO")
        return
        
    if k == 3:
        for mask, u in ind3:
            avail1 = ((1 << (n + 1)) - 2) & ~mask & ~adj[u]
            v = (avail1 & -avail1).bit_length() - 1
            while v > 0:
                mask2 = mask | (1 << v)
                avail2 = avail1 & ~((1 << v) | adj[v])
                w = (avail2 & -avail2).bit_length() - 1
                if w > 0:
                    print("YES")
                    print(u, v, w)
                    return
                avail1 ^= 1 << v
                v = (avail1 & -avail1).bit_length() - 1
        print("NO")
        return

    # For k >= 4, use meet-in-the-middle
    half = n // 2
    left_nodes = list(range(1, half + 1))
    right_nodes = list(range(half + 1, n + 1))
    
    left_masks = [1 << i for i in left_nodes]
    right_masks = [1 << i for i in right_nodes]
    
    # Precompute left independent sets
    left_ind = []
    for i in range(1 << len(left_nodes)):
        is_ind = True
        for j in range(len(left_nodes)):
            if i & (1 << j):
                if i & adj[left_nodes[j]]:
                    is_ind = False
                    break
        if is_ind:
            left_ind.append(i)
            
    # Precompute right independent sets
    right_ind = []
    for i in range(1 << len(right_nodes)):
        is_ind = True
        for j in range(len(right_nodes)):
            if i & (1 << j):
                if i & adj[right_nodes[j]]:
                    is_ind = False
                    break
        if is_ind:
            right_ind.append(i)
            
    # Group right independent sets by size
    right_by_size = [[] for _ in range(k + 1)]
    for r in right_ind:
        sz = bin(r).count('1')
        if sz <= k:
            right_by_size[sz].append(r)
            
    # For each left independent set, find a compatible right independent set
    for l in left_ind:
        l_sz = bin(l).count('1')
        if l_sz > k:
            continue
        need = k - l_sz
        if not right_by_size[need]:
            continue
            
        # Nodes in left set
        l_nodes = []
        for j in range(len(left_nodes)):
            if l & (1 << j):
                l_nodes.append(left_nodes[j])
                
        # Forbidden nodes in right part (adjacent to any node in left set)
        forbidden = 0
        for u in l_nodes:
            forbidden |= adj[u]
        forbidden >>= (half + 1)
        
        for r in right_by_size[need]:
            if r & forbidden:
                continue
                
            # Found a valid combination
            ans = l_nodes
            for j in range(len(right_nodes)):
                if r & (1 << j):
                    ans.append(right_nodes[j])
            print("YES")
            print(*ans)
            return
            
    print("NO")

if __name__ == '__main__':
    solve()
