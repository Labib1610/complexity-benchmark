import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to retain k markers such that no two retained markers interfere.
    # This is equivalent to finding an Independent Set of size k.
    # The discarded markers must cover all edges (Vertex Cover of size n-k).
    # Finding an Independent Set of size k is equivalent to finding a 
    # Vertex Cover of size n-k.
    
    target_is = k
    
    # Bron-Kerbosch or simple backtracking with pruning for Independent Set
    # Since n <= 46, we can use a meet-in-the-middle or optimized search.
    # Given the constraints and the nature of the problem, a simple recursive
    # search with bitmasking is efficient enough for n=46.
    
    def find_is(candidates, current_is):
        if len(current_is) == target_is:
            return current_is
        if len(current_is) + bin(candidates).count('1') < target_is:
            return None
        
        if candidates == 0:
            return None
            
        # Heuristic: pick vertex with max degree in subgraph
        v = (candidates & -candidates).bit_length() - 1
        
        # Option 1: Include v
        res = find_is(candidates & ~adj[v] & ~(1 << v), current_is + [v + 1])
        if res:
            return res
            
        # Option 2: Exclude v
        res = find_is(candidates & ~(1 << v), current_is)
        return res

    # To optimize, sort vertices by degree
    nodes = sorted(range(n), key=lambda x: bin(adj[x]).count('1'))
    
    # Re-map nodes to simplify
    mapping = {old: new for new, old in enumerate(nodes)}
    rev_mapping = {new: old for new, old in enumerate(nodes)}
    new_adj = [0] * n
    for u in range(n):
        for v in range(n):
            if (adj[rev_mapping[u]] >> rev_mapping[v]) & 1:
                new_adj[u] |= (1 << v)
                
    def find_is_fast(candidates, current_is):
        if len(current_is) == target_is:
            return current_is
        if len(current_is) + bin(candidates).count('1') < target_is:
            return None
        if candidates == 0:
            return None
            
        # Pick vertex with highest index (last in sorted list)
        v = candidates.bit_length() - 1
        
        # Include v
        res = find_is_fast(candidates & ~new_adj[v] & ~(1 << v), current_is + [rev_mapping[v] + 1])
        if res:
            return res
        
        # Exclude v
        return find_is_fast(candidates & ~(1 << v), current_is)

    result = find_is_fast((1 << n) - 1, [])
    
    if result:
        print("YES")
        print(*(sorted(result)))
    else:
        print("NO")

solve()
