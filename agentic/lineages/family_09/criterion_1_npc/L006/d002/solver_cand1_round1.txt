import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # Meet-in-the-middle for Maximum Independent Set of size k
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the two halves
    # sets1[size] = list of bitmasks of size 'size'
    sets1 = [[] for _ in range(n1 + 1)]
    for i in range(1 << n1):
        valid = True
        for j in range(n1):
            if (i >> j) & 1:
                if adj[j] & i:
                    valid = False
                    break
        if valid:
            sets1[bin(i).count('1')].append(i)
            
    sets2 = [[] for _ in range(n2 + 1)]
    for i in range(1 << n2):
        valid = True
        for j in range(n2):
            if (i >> j) & 1:
                if (adj[j + n1] >> n1) & i:
                    valid = False
                    break
        if valid:
            sets2[bin(i).count('1')].append(i)

    # Check compatibility between halves
    # For each mask in sets1, find which nodes in the second half are forbidden
    for s1 in range(max(0, k - n2), min(k, n1) + 1):
        s2 = k - s1
        for mask1 in sets1[s1]:
            # Forbidden nodes in the second half
            forbidden = 0
            for i in range(n1):
                if (mask1 >> i) & 1:
                    forbidden |= (adj[i] >> n1)
            
            for mask2 in sets2[s2]:
                if not (mask2 & forbidden):
                    # Found it
                    res = []
                    for i in range(n1):
                        if (mask1 >> i) & 1:
                            res.append(i + 1)
                    for i in range(n2):
                        if (mask2 >> i) & 1:
                            res.append(i + n1 + 1)
                    print("YES")
                    print(*(res))
                    return

    print("NO")

solve()
