import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    # Bron-Kerbosch or simple recursive search with bitmasks.
    
    def find_is(candidates, current_set, target_size):
        if len(current_set) == target_size:
            return current_set
        if not candidates or len(current_set) + bin(candidates).count('1') < target_size:
            return None
        
        # Heuristic: pick the candidate with the most neighbors in the candidate set
        # to prune the search space (or just pick the lowest index)
        v = (candidates & -candidates).bit_length() - 1
        
        # Option 1: Include v
        res = find_is(candidates & ~adj[v] & ~(1 << v), current_set + [v + 1], target_size)
        if res:
            return res
        
        # Option 2: Exclude v
        res = find_is(candidates & ~(1 << v), current_set, target_size)
        if res:
            return res
            
        return None

    # Optimization: For n up to 46, simple recursion might be slow.
    # Use a more efficient search.
    def search(candidates, current_set, k):
        if k == 0:
            return current_set
        if not candidates or bin(candidates).count('1') < k:
            return None
        
        # Pivot to reduce branching
        pivot = (candidates).bit_length() - 1
        # Candidates that are not neighbors of pivot
        remaining = candidates & ~adj[pivot]
        
        # Try including pivot
        res = search(candidates & ~adj[pivot] & ~(1 << pivot), current_set + [pivot + 1], k - 1)
        if res: return res
        
        # Try excluding pivot
        res = search(candidates & ~(1 << pivot), current_set, k)
        return res

    result = search((1 << n) - 1, [], k)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
