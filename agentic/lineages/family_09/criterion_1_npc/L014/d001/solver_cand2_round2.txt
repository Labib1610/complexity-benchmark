import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    
    def find_independent_set(candidates, current_set):
        if len(current_set) == k:
            return current_set
        
        if not candidates:
            return None
        
        # Heuristic: pick the candidate with the fewest neighbors in the remaining candidates
        # to prune the search space (Bron-Kerbosch style optimization)
        while candidates:
            # Pick vertex with highest degree in the subgraph induced by candidates
            v = -1
            max_deg = -1
            for c in candidates:
                deg = bin(adj[c] & candidates).count('1')
                if deg > max_deg:
                    max_deg = deg
                    v = c
            
            # Try including v
            res = find_independent_set(candidates & ~adj[v] & ~(1 << v), current_set | {v + 1})
            if res is not None:
                return res
            
            # Try excluding v
            candidates &= ~(1 << v)
            
            # Pruning: if remaining candidates are not enough to reach size k
            if len(current_set) + bin(candidates).count('1') < k:
                return None
                
        return None

    result = find_independent_set((1 << n) - 1, set())
    
    if result:
        print("YES")
        print(*(sorted(list(result))))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
