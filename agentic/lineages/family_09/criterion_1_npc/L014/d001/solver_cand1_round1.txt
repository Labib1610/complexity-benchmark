```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    k = int(input_data[1])
    a = [int(x) for x in input_data[2:]]
    
    # The problem asks for a group of size k such that the sum of elements
    # is divisible by k.
    # This is a classic application of the Erdős–Ginzburg–Ziv theorem,
    # which states that any sequence of 2k-1 integers contains a subsequence
    # of length k whose sum is divisible by k.
    # Since n >= 2k-1, such a group always exists.
    
    # We can use dynamic programming to find one such group.
    # dp[i][j][rem] = a subset of size j using first i elements with sum % k == rem.
    # To reconstruct the subset, we store the choice.
    
    # dp[count][remainder] = list of indices
    # Since we only need one valid group, we can use a dictionary or array.
    # dp[count][remainder] = (last_index_used, previous_remainder)
    
    dp = [[None for _ in range(k)] for _ in range(k + 1)]
    dp[0][0] = True
    
    parent = [[None for _ in range(k)] for _ in range(k + 1)]
    
    for i in range(n):
        val = a[i] % k
        for count in range(min(i + 1, k - 1), -1, -1):
            for rem in range(k):
                if dp[count][rem] is not None:
                    new_count = count + 1
                    new_rem = (rem + val) % k
                    if dp[new_count][new_rem] is None:
                        dp[new_count][new_rem] = True
                        parent[new_count][new_rem] = (i, rem)
    
    # Reconstruct
    res = []
    curr_rem = 0
    curr_count = k
    while curr_count > 0:
        idx, prev_rem = parent[curr_count][curr_rem]
        res.append(a[idx])
        curr_rem = prev_rem
        curr_count -= 1
        
    print(*(res))

if __name__ == '__main__':
    solve()
```
