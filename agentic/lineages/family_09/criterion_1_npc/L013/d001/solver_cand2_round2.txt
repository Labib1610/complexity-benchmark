import sys

def solve():
    # The problem asks for an independent set of size k in a graph where
    # edges represent hardware locks. If two processes share a lock, they
    # cannot both be selected. This is exactly the Maximum Independent Set problem.
    # Since n <= 46, we can use a meet-in-the-middle approach.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the two halves
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        # sets_by_size[s] = list of (mask, bitmask_of_nodes)
        sets_by_size = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            is_independent = True
            count = 0
            mask = 0
            for j in range(size):
                if (i >> j) & 1:
                    count += 1
                    u = nodes[j]
                    if adj_list[u] & mask:
                        is_independent = False
                        break
                    mask |= (1 << u)
            if is_independent:
                sets_by_size[count].append(mask)
        return sets_by_size

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Check combinations
    for s1 in range(len(sets1)):
        s2 = k - s1
        if 0 <= s2 < len(sets2):
            for m1 in sets1[s1]:
                # Find nodes in second half compatible with m1
                forbidden = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        forbidden |= adj[i]
                
                for m2 in sets2[s2]:
                    if not (m2 & forbidden):
                        # Found it
                        res = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                res.append(i + 1)
                        print("YES")
                        print(*(res))
                        return
    print("NO")

solve()
