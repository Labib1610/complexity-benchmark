We need to select exactly **k** vertices from an undirected graph such that no edge has both endpoints selected.  
That means the selected set must be an **independent set** of size exactly **k**.

Since **n ≤ 46**, we can use a **meet-in-the-middle** approach:

- Split vertices into two halves.
- For each half, enumerate all independent sets and store their sizes and bitmasks.
- Then combine sets from both halves to check if we can reach total size **k**.

Here’s the implementation:

```python
import sys
from itertools import combinations

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Split into two halves
    n1 = n // 2
    n2 = n - n1
    left = list(range(n1))
    right = list(range(n1, n))

    # Enumerate independent sets in left half
    left_sets = []  # list of (size, mask)
    for mask in range(1 << n1):
        ok = True
        for i in range(n1):
            if mask & (1 << i):
                # check conflict with other selected in left
                if (adj[left[i]] & mask & ((1 << n1) - 1)) & ~(1 << i):
                    ok = False
                    break
        if ok:
            size = bin(mask).count("1")
            left_sets.append((size, mask))

    # Enumerate independent sets in right half
    right_sets = []
    for mask in range(1 << n2):
        ok = True
        for i in range(n2):
            if mask & (1 << i):
                if (adj[right[i]] & (mask << n1)) & ~(1 << (n1 + i)):
                    ok = False
                    break
        if ok:
            size = bin(mask).count("1")
            right_sets.append((size, mask))

    # For each left set, find a right set that is compatible
    # Compatibility: no edge between selected left and selected right
    # Precompute for each left mask the forbidden right mask
    forbidden = {}
    for size_l, mask_l in left_sets:
        forb = 0
        for i in range(n1):
            if mask_l & (1 << i):
                # neighbors in right half
                forb |= (adj[left[i]] >> n1)
        forbidden[mask_l] = forb

    # Group right sets by size for quick lookup
    right_by_size = {}
    for size_r, mask_r in right_sets:
        right_by_size.setdefault(size_r, []).append(mask_r)

    # Try to find combination
    for size_l, mask_l in left_sets:
        need = k - size_l
        if need < 0 or need > n2:
            continue
        if need not in right_by_size:
            continue
        forb = forbidden[mask_l]
        for mask_r in right_by_size[need]:
            if (mask_r & forb) == 0:
                # Found
                selected = []
                for i in range(n1):
                    if mask_l & (1 << i):
                        selected.append(left[i] + 1)
                for i in range(n2):
                    if mask_r & (1 << i):
                        selected.append(right[i] + 1)
                print("YES")
                print(" ".join(map(str, selected)))
                return

    print("NO")

if __name__ == "__main__":
    solve()
```
