import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    n1 = n // 2
    n2 = n - n1
    
    # Meet-in-the-middle: find all independent sets in both halves
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            mask = 0
            valid = True
            count = 0
            for j in range(size):
                if (i >> j) & 1:
                    if adj_list[nodes[j]] & mask:
                        valid = False
                        break
                    mask |= (1 << nodes[j])
                    count += 1
            if valid:
                res[count].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Precompute adjacency for sets2 to check compatibility
    # adj_mask[i] is the mask of nodes in the first half that conflict with node i in the second half
    adj_mask = [0] * n2
    for i in range(n2):
        for j in range(n1):
            if (adj[n1 + i] >> j) & 1:
                adj_mask[i] |= (1 << j)
                
    for c1 in range(n1 + 1):
        c2 = k - c1
        if 0 <= c2 <= n2:
            for m1 in sets1[c1]:
                # Find nodes in second half compatible with m1
                allowed_nodes = []
                for i in range(n2):
                    if not (adj_mask[i] & m1):
                        allowed_nodes.append(i)
                
                # Check if any subset of size c2 in allowed_nodes is an independent set
                # Since n2 is small (max 23), we can use recursion or bitmask
                def find_is(idx, count, current_mask, current_indices):
                    if count == c2:
                        return current_indices
                    if idx == len(allowed_nodes):
                        return None
                    
                    # Try including allowed_nodes[idx]
                    node = allowed_nodes[idx]
                    if not (adj[n1 + node] & current_mask):
                        res = find_is(idx + 1, count + 1, current_mask | (1 << (n1 + node)), current_indices + [node + n1 + 1])
                        if res: return res
                    
                    # Try excluding
                    return find_is(idx + 1, count, current_mask, current_indices)

                res = find_is(0, 0, m1, [])
                if res:
                    print("YES")
                    ans = [i + 1 for i in range(n1) if (m1 >> i) & 1] + res
                    print(*(ans))
                    return

    print("NO")

solve()
