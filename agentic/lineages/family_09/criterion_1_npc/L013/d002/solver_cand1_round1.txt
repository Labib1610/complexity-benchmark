import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * (n + 1)
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    # To optimize, we can use the Bron-Kerbosch style logic or simple bitmask recursion.
    
    found_set = None
    
    def backtrack(candidates, current_set):
        nonlocal found_set
        if found_set is not None:
            return
        
        if len(current_set) == k:
            found_set = list(current_set)
            return
        
        if len(current_set) + bin(candidates).count('1') < k:
            return
        
        while candidates:
            # Pick the lowest bit
            v_bit = candidates & -candidates
            v = (v_bit).bit_length()
            
            # New candidates must not be neighbors of v
            new_candidates = candidates & ~v_bit & ~adj[v]
            
            backtrack(new_candidates, current_set + [v])
            if found_set:
                return
            
            candidates &= ~v_bit

    # Initial candidates: all processes 1 to n
    initial_candidates = (1 << n) - 1
    backtrack(initial_candidates, [])
    
    if found_set:
        print("YES")
        print(*(found_set))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
