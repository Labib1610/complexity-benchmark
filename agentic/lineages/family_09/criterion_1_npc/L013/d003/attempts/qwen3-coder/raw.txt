from collections import defaultdict
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    edges = []
    index = 3
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # If no edges, any k nodes work
    if m == 0:
        print("YES")
        print(' '.join(str(i+1) for i in range(k)))
        return
    
    # Build adjacency list
    adj = [set() for _ in range(n)]
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # If k >= n/2, we use meet-in-the-middle on the complement (independent set perspective)
    # Else, we do direct enumeration on smaller side
    
    if k <= n // 2:
        # Enumerate all subsets of size k directly
        # But since n up to 46, we need smarter way
        # Use randomized or heuristic method won't work due to adversarial test cases
        # So we proceed with meet-in-the-middle
        
        # Split vertices into two halves
        mid = n // 2
        left = list(range(mid))
        right = list(range(mid, n))
        
        # Precompute all valid subsets for left half
        left_valid = {}
        for mask in range(1 << len(left)):
            subset = [left[i] for i in range(len(left)) if mask & (1 << i)]
            # Check if subset is independent
            valid = True
            for u in subset:
                for v in adj[u]:
                    if v in subset:
                        valid = False
                        break
                if not valid:
                    break
            if valid:
                size = bin(mask).count('1')
                if size not in left_valid:
                    left_valid[size] = []
                left_valid[size].append((mask, set(subset)))
        
        # Now iterate over right half
        for mask in range(1 << len(right)):
            subset_r = [right[i] for i in range(len(right)) if mask & (1 << i)]
            # Check if subset_r is independent
            valid = True
            for u in subset_r:
                for v in adj[u]:
                    if v in subset_r:
                        valid = False
                        break
                if not valid:
                    break
            if not valid:
                continue
            
            size_r = bin(mask).count('1')
            needed = k - size_r
            if needed < 0:
                continue
                
            if needed in left_valid:
                subset_r_set = set(subset_r)
                for mask_l, subset_l_set in left_valid[needed]:
                    # Check if union is independent
                    conflict = False
                    for u in subset_l_set:
                        for v in adj[u]:
                            if v in subset_r_set:
                                conflict = True
                                break
                        if conflict:
                            break
                    if not conflict:
                        result = sorted([x+1 for x in subset_l_set | subset_r_set])
                        if len(result) == k:
                            print("YES")
                            print(' '.join(map(str, result)))
                            return
        
        print("NO")
    else:
        # k > n/2, so we look for an independent set of size n-k
        target = n - k
        # Again, split and meet-in-the-middle
        mid = n // 2
        left = list(range(mid))
        right = list(range(mid, n))
        
        # Valid subsets of left
        left_valid = {}
        for mask in range(1 << len(left)):
            subset = [left[i] for i in range(len(left)) if mask & (1 << i)]
            valid = True
            for u in subset:
                for v in adj[u]:
                    if v in subset:
                        valid = False
                        break
                if not valid:
                    break
            if valid:
                size = bin(mask).count('1')
                if size not in left_valid:
                    left_valid[size] = []
                left_valid[size].append((mask, set(subset)))
        
        # Iterate over right
        for mask in range(1 << len(right)):
            subset_r = [right[i] for i in range(len(right)) if mask & (1 << i)]
            valid = True
            for u in subset_r:
                for v in adj[u]:
                    if v in subset_r:
                        valid = False
                        break
                if not valid:
                    break
            if not valid:
                continue
            
            size_r = bin(mask).count('1')
            needed = target - size_r
            if needed < 0:
                continue
                
            if needed in left_valid:
                subset_r_set = set(subset_r)
                for mask_l, subset_l_set in left_valid[needed]:
                    conflict = False
                    for u in subset_l_set:
                        for v in adj[u]:
                            if v in subset_r_set:
                                conflict = True
                                break
                        if conflict:
                            break
                    if not conflict:
                        # Found independent set of size target
                        ind_set = subset_l_set | subset_r_set
                        # Take complement
                        all_nodes = set(range(n))
                        audit_set = sorted(list(all_nodes - ind_set))
                        if len(audit_set) == k:
                            print("YES")
                            print(' '.join(str(x+1) for x in audit_set))
                            return
        
        print("NO")

if __name__ == "__main__":
    main()
