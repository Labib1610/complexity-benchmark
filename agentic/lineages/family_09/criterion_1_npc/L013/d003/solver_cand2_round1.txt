import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * (n + 1)
    idx = 3
    for _ in range(m):
        u, v = map(int, input_data[idx:idx+2])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2

    # The problem is equivalent to finding an Independent Set of size k.
    # This is the Maximum Independent Set problem, which is NP-hard.
    # Given n <= 46, we use a backtracking approach with pruning.
    
    nodes = list(range(1, n + 1))
    # Sort nodes by degree to improve pruning
    nodes.sort(key=lambda x: bin(adj[x]).count('1'), reverse=True)
    
    memo = {}

    def backtrack(idx, current_k, current_mask):
        if current_k == 0:
            return []
        if idx == n:
            return None
        
        state = (idx, current_k, current_mask)
        if state in memo:
            return None
        
        # Try including nodes[idx]
        u = nodes[idx]
        if not (current_mask & (1 << (u - 1))):
            # Check if u is compatible with already selected nodes
            if not (adj[u] & current_mask):
                res = backtrack(idx + 1, current_k - 1, current_mask | (1 << (u - 1)))
                if res is not None:
                    return [u] + res
        
        # Try excluding nodes[idx]
        # Pruning: if remaining nodes are not enough to reach k
        if (n - 1 - idx) >= current_k:
            res = backtrack(idx + 1, current_k, current_mask)
            if res is not None:
                return res
        
        memo[state] = False
        return None

    result = backtrack(0, k, 0)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
