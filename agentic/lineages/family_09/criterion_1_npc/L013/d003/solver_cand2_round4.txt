import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Meet-in-the-middle: split n into n1 and n2
    n1 = n // 2
    n2 = n - n1

    def get_subsets(start, end, adj_list):
        # Returns a dictionary: count -> list of bitmasks
        size = end - start
        res = {}
        for i in range(1 << size):
            mask = 0
            valid = True
            count = 0
            for j in range(size):
                if (i >> j) & 1:
                    node = start + j
                    if (mask & adj_list[node]) != 0:
                        valid = False
                        break
                    mask |= (1 << node)
                    count += 1
            if valid:
                if count not in res:
                    res[count] = []
                res[count].append(mask)
        return res

    sub1 = get_subsets(0, n1, adj)
    sub2 = get_subsets(n1, n, adj)

    # Pre-calculate adjacency constraints for sub2 relative to sub1
    # For each mask in sub2, we need to know which nodes in sub1 it conflicts with
    # Actually, it's easier: for a mask1 in sub1, we need a mask2 in sub2 such that
    # (mask2 & adj[node_in_mask1]) == 0 and (mask1 & adj[node_in_mask2]) == 0
    # Since adj is symmetric, this is just (mask1 & adj_mask2) == 0
    
    # Optimization: pre-calculate adj_mask for each mask in sub2
    # adj_mask is the union of adj[v] for all v in mask2
    sub2_processed = {}
    for count, masks in sub2.items():
        sub2_processed[count] = []
        for m2 in masks:
            m2_adj = 0
            for i in range(n2):
                if (m2 >> (n1 + i)) & 1:
                    m2_adj |= adj[n1 + i]
            sub2_processed[count].append((m2, m2_adj))

    for c1 in sub1:
        c2 = k - c1
        if c2 in sub2_processed:
            for m1 in sub1[c1]:
                m1_adj = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        m1_adj |= adj[i]
                
                for m2, m2_adj in sub2_processed[c2]:
                    if (m1 & m2_adj) == 0 and (m2 & m1_adj) == 0:
                        # Found it
                        res = []
                        combined = m1 | m2
                        for i in range(n):
                            if (combined >> i) & 1:
                                res.append(i + 1)
                        print("YES")
                        print(*(res))
                        return

    print("NO")

solve()
