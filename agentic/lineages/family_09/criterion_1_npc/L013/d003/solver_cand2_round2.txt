import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1] |= (1 << (v-1))
        adj[v-1] |= (1 << (u-1))

    n1 = n // 2
    n2 = n - n1
    
    def get_subsets(nodes, adj_list):
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        
        def backtrack(idx, current_mask, count):
            if idx == size:
                res[count].append(current_mask)
                return
            
            # Option 1: Don't include nodes[idx]
            backtrack(idx + 1, current_mask, count)
            
            # Option 2: Include nodes[idx] if safe
            u = nodes[idx]
            if not (current_mask & adj_list[u]):
                backtrack(idx + 1, current_mask | (1 << u), count + 1)
        
        backtrack(0, 0, 0)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    subsets1 = get_subsets(nodes1, adj)
    subsets2 = get_subsets(nodes2, adj)
    
    # Precompute adjacency constraints between sets
    adj2_for_1 = [0] * n1
    for i in range(n1):
        adj2_for_1[i] = adj[i] >> n1
        
    for c1 in range(n1 + 1):
        c2 = k - c1
        if 0 <= c2 <= n2:
            for mask1 in subsets1[c1]:
                # Filter mask2s that are compatible with mask1
                forbidden_in_2 = 0
                for i in range(n1):
                    if (mask1 >> i) & 1:
                        forbidden_in_2 |= adj2_for_1[i]
                
                for mask2 in subsets2[c2]:
                    if not (mask2 & forbidden_in_2):
                        # Found it
                        res = []
                        for i in range(n):
                            if ((mask1 | mask2) >> i) & 1:
                                res.append(i + 1)
                        print("YES")
                        print(*(res))
                        return

    print("NO")

solve()
