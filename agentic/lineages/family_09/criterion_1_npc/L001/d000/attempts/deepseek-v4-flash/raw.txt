Here is a Python 3 solution using a meet-in-the-middle approach to handle the maximum n = 46 efficiently.  

```python
import sys
from collections import defaultdict

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    if k == 1:
        print("YES")
        print(1)
        return
    
    # Split vertices into two halves
    left_n = n // 2
    right_n = n - left_n
    
    left_adj = [adj[i] & ((1 << left_n) - 1) for i in range(left_n)]
    right_adj = [ (adj[i + left_n] >> left_n) for i in range(right_n)]
    
    # Enumerate all independent sets in left half
    left_sets = []  # list of (mask, size)
    for mask in range(1 << left_n):
        ok = True
        for i in range(left_n):
            if mask & (1 << i):
                if left_adj[i] & mask:
                    ok = False
                    break
        if ok:
            left_sets.append((mask, bin(mask).count("1")))
    
    # Group left sets by size
    left_by_size = defaultdict(list)
    for mask, sz in left_sets:
        left_by_size[sz].append(mask)
    
    # Enumerate all independent sets in right half
    right_sets = []
    for mask in range(1 << right_n):
        ok = True
        for i in range(right_n):
            if mask & (1 << i):
                if right_adj[i] & mask:
                    ok = False
                    break
        if ok:
            right_sets.append((mask, bin(mask).count("1")))
    
    right_by_size = defaultdict(list)
    for mask, sz in right_sets:
        right_by_size[sz].append(mask)
    
    # For each left set, find compatible right sets
    # Compatibility: no edge between left and right selected vertices
    # Precompute for each left mask the forbidden right mask
    forbidden = {}
    for lmask, _ in left_sets:
        f = 0
        for i in range(left_n):
            if lmask & (1 << i):
                f |= (adj[i] >> left_n)
        forbidden[lmask] = f
    
    # Try to find combination
    for lmask, lsz in left_sets:
        need = k - lsz
        if need < 0 or need > right_n:
            continue
        f = forbidden[lmask]
        # We need a right independent set of size 'need' that has no overlap with f
        for rmask in right_by_size.get(need, []):
            if rmask & f == 0:
                # Found
                res = []
                for i in range(left_n):
                    if lmask & (1 << i):
                        res.append(i + 1)
                for i in range(right_n):
                    if rmask & (1 << i):
                        res.append(i + left_n + 1)
                print("YES")
                print(" ".join(map(str, res)))
                return
    
    print("NO")

if __name__ == "__main__":
    solve()
```
