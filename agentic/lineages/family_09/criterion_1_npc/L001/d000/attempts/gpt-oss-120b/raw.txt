import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    all_vertices = (1 << n) - 1
    target = k
    result_mask = None

    # order vertices by degree descending for better pruning
    order = sorted(range(n), key=lambda x: bin(adj[x]).count("1"), reverse=True)
    # map original index to position in ordered list
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i
    # rebuild adjacency in ordered space
    adj2 = [0] * n
    for i in range(n):
        v = order[i]
        mask = 0
        for u in range(n):
            if (adj[v] >> u) & 1:
                mask |= 1 << pos[u]
        adj2[i] = mask

    # recursive search
    def dfs(cur_mask: int, cand_mask: int, cur_size: int) -> bool:
        nonlocal result_mask
        if cur_size == target:
            result_mask = cur_mask
            return True
        # prune
        if cur_size + cand_mask.bit_count() < target:
            return False
        # choose a vertex (heuristic: first)
        # pick vertex with maximum degree in candidates to branch first
        # simple: take least significant bit
        v = (cand_mask & -cand_mask).bit_length() - 1
        # branch: include v
        new_cand = cand_mask & ~adj2[v] & ~(1 << v)
        if dfs(cur_mask | (1 << v), new_cand, cur_size + 1):
            return True
        # branch: exclude v
        cand_mask &= ~(1 << v)
        return dfs(cur_mask, cand_mask, cur_size)

    initial_cand = (1 << n) - 1
    dfs(0, initial_cand, 0)

    if result_mask is None:
        print("NO")
    else:
        ans = []
        for i in range(n):
            if (result_mask >> i) & 1:
                ans.append(str(order[i] + 1))
        print("YES")
        print(" ".join(ans))

if __name__ == "__main__":
    solve()
