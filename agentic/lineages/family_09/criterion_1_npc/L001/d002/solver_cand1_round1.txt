import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # Meet-in-the-middle approach for Maximum Independent Set / k-clique
    # n <= 46 is small enough for a optimized search
    n1 = n // 2
    n2 = n - n1
    
    # adj1: adjacency for first n1 nodes, adj2: for remaining n2 nodes
    # mask1: conflicts between first n1 and last n2 nodes
    adj1 = [adj[i] & ((1 << n1) - 1) for i in range(n1)]
    adj2 = [adj[i + n1] >> n1 for i in range(n2)]
    cross = [(adj[i] >> n1) for i in range(n1)]

    def get_subsets(nodes_adj, num_nodes):
        # returns dict: size -> list of bitmasks
        res = [[] for _ in range(num_nodes + 1)]
        res[0].append(0)
        for i in range(num_nodes):
            for size in range(i, -1, -1):
                for mask in res[size]:
                    if not (mask & nodes_adj[i]):
                        res[size + 1].append(mask | (1 << i))
        return res

    sub1 = get_subsets(adj1, n1)
    sub2 = get_subsets(adj2, n2)

    for s1 in range(max(0, k - n2), min(k, n1) + 1):
        s2 = k - s1
        for m1 in sub1[s1]:
            # Find nodes in n2 compatible with m1
            forbidden = 0
            for i in range(n1):
                if (m1 >> i) & 1:
                    forbidden |= cross[i]
            
            for m2 in sub2[s2]:
                if not (m2 & forbidden):
                    # Found it
                    res = []
                    for i in range(n1):
                        if (m1 >> i) & 1:
                            res.append(i + 1)
                    for i in range(n2):
                        if (m2 >> i) & 1:
                            res.append(i + n1 + 1)
                    print("YES")
                    print(*(res))
                    return

    print("NO")

solve()
