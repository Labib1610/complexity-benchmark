import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Bron-Kerbosch with pivoting and size constraint
    # We want to find a clique of size k in the complement graph
    # The complement graph has edges where there were no destructive pairings
    comp_adj = [(( (1 << n) - 1 ) ^ (1 << i) ^ adj[i]) for i in range(n)]
    
    def get_clique(candidates, current_clique):
        if not candidates:
            if len(current_clique) == k:
                return current_clique
            return None
        
        if len(current_clique) + bin(candidates).count('1') < k:
            return None
            
        # Pivot to reduce branches
        pivot = (candidates | (candidates & 0)).bit_length() - 1
        remaining = candidates & ~comp_adj[pivot]
        
        # Sort candidates by degree in remaining graph to prune faster
        nodes = []
        temp = remaining
        while temp:
            v = (temp & -temp).bit_length() - 1
            nodes.append(v)
            temp &= temp - 1
            
        for v in nodes:
            res = get_clique(candidates & comp_adj[v], current_clique + [v + 1])
            if res:
                return res
            candidates &= ~(1 << v)
        return None

    # Optimization: If k is large, search for independent set in original graph
    # which is equivalent to clique in complement graph.
    # For n=46, standard BK is usually fast enough.
    result = get_clique((1 << n) - 1, [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
