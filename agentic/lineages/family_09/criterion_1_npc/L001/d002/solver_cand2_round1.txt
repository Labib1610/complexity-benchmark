import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * (n + 1)
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    # Since n <= 46, we use bitmasking and backtracking with pruning.
    # We need to find an independent set of size k.
    
    def get_independent_set(candidates, current_set, target_size):
        if len(current_set) == target_size:
            return current_set
        
        if not candidates:
            return None
        
        # Heuristic: pick the node with the fewest available neighbors (MRV)
        # or simply pick the first one. Given n=46, simple backtracking with 
        # bitmasking is efficient enough.
        
        # Convert bitmask to list
        nodes = []
        temp = candidates
        while temp:
            node = (temp & -temp).bit_length()
            nodes.append(node)
            temp &= temp - 1
            
        # Sort nodes by degree in the subgraph induced by candidates
        nodes.sort(key=lambda x: bin(adj[x] & candidates).count('1'))
        
        for node in nodes:
            res = get_independent_set(candidates & ~(adj[node] | (1 << (node - 1))), 
                                      current_set + [node], target_size)
            if res is not None:
                return res
        return None

    initial_candidates = (1 << n) - 1
    result = get_independent_set(initial_candidates, [], k)
    
    if result:
        print("YES")
        print(*(sorted(result)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
