import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # Meet-in-the-middle approach for Maximum Independent Set variant
    # n <= 46 is too large for 2^(n/2), but we need exactly k.
    # Since n is small enough for meet-in-the-middle (n=46, 23+23),
    # we can split into two halves.
    n1 = n // 2
    n2 = n - n1
    
    adj1 = [adj[i] & ((1 << n1) - 1) for i in range(n1)]
    adj2 = [adj[i + n1] >> n1 for i in range(n2)]
    cross = [(adj[i] >> n1) for i in range(n1)]
    
    def get_subsets(n_size, adj_list):
        # returns dict: count -> list of masks
        res = [[] for _ in range(n_size + 1)]
        for i in range(1 << n_size):
            valid = True
            count = 0
            for j in range(n_size):
                if (i >> j) & 1:
                    count += 1
                    if adj_list[j] & i:
                        valid = False
                        break
            if valid:
                res[count].append(i)
        return res

    sub1 = get_subsets(n1, adj1)
    sub2 = get_subsets(n2, adj2)
    
    for c1 in range(n1 + 1):
        c2 = k - c1
        if 0 <= c2 <= n2:
            for m1 in sub1[c1]:
                # Precompute forbidden mask in second half
                forbidden = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        forbidden |= cross[i]
                
                for m2 in sub2[c2]:
                    if not (m2 & forbidden):
                        print("YES")
                        ans = []
                        for i in range(n1):
                            if (m1 >> i) & 1: ans.append(i + 1)
                        for i in range(n2):
                            if (m2 >> i) & 1: ans.append(i + n1 + 1)
                        print(*(ans))
                        return
    print("NO")

solve()
