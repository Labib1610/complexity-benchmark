import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle for Maximum Independent Set of size k
    n1 = n // 2
    n2 = n - n1
    
    # Precompute valid subsets for both halves
    def get_subsets(start, size):
        subsets = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            valid = True
            count = 0
            mask = 0
            for j in range(size):
                if (i >> j) & 1:
                    if adj[start + j] & (i << start):
                        valid = False
                        break
                    mask |= (1 << (start + j))
                    count += 1
            if valid:
                subsets[count].append((mask, i))
        return subsets

    sub1 = get_subsets(0, n1)
    sub2 = get_subsets(n1, n2)
    
    # Check combinations
    for k1 in range(min(k + 1, n1 + 1)):
        k2 = k - k1
        if k2 < 0 or k2 > n2:
            continue
        
        for m1, bits1 in sub1[k1]:
            # Filter sub2[k2] for compatibility with m1
            forbidden_in_2 = 0
            for i in range(n1):
                if (m1 >> i) & 1:
                    forbidden_in_2 |= adj[i]
            
            for m2, bits2 in sub2[k2]:
                if not (m2 & forbidden_in_2):
                    # Found it
                    res = []
                    for i in range(n1):
                        if (bits1 >> i) & 1:
                            res.append(i + 1)
                    for i in range(n2):
                        if (bits2 >> i) & 1:
                            res.append(n1 + i + 1)
                    print("YES")
                    print(*(res))
                    return

    print("NO")

solve()
