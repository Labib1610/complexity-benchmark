import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Max Independent Set using Bron-Kerbosch style backtracking
    # Since n <= 46, we use a meet-in-the-middle or optimized search
    # Given the constraints and the nature of the problem, a simple 
    # backtracking with pruning is often sufficient for n=46.
    
    def find_subset(candidates, current_subset, target_size):
        if len(current_subset) == target_size:
            return current_subset
        if not candidates or len(current_subset) + len(candidates) < target_size:
            return None
        
        # Heuristic: pick candidate with most conflicts to prune faster
        v = candidates[0]
        
        # Option 1: Include v
        new_candidates = [c for c in candidates[1:] if not (adj[v] & (1 << c))]
        res = find_subset(new_candidates, current_subset + [v], target_size)
        if res:
            return res
            
        # Option 2: Exclude v
        res = find_subset(candidates[1:], current_subset, target_size)
        if res:
            return res
            
        return None

    # For n=46, simple recursion might be slow. 
    # Use bitmask-based search for efficiency.
    memo = {}

    def solve_recursive(mask, count):
        if count == 0:
            return []
        if mask == 0:
            return None
        
        state = (mask, count)
        if state in memo:
            return memo[state]
        
        # Pick the lowest bit
        v = (mask & -mask).bit_length() - 1
        
        # Try including v
        new_mask = mask & ~adj[v] & ~(1 << v)
        res = solve_recursive(new_mask, count - 1)
        if res is not None:
            memo[state] = [v + 1] + res
            return memo[state]
        
        # Try excluding v
        res = solve_recursive(mask & ~(1 << v), count)
        memo[state] = res
        return res

    result = solve_recursive((1 << n) - 1, k)
    
    if result:
        print("YES")
        print(*(sorted(result)))
    else:
        print("NO")

solve()
