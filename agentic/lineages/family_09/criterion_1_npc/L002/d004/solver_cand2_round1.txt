import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2
        
    # Kinetic Metric: sum over links of (3^(Su+Sv) - 1)
    # Potential Metric: sum over links of 2*(Su+Sv)
    # Equilibrium: sum(3^(Su+Sv) - 1) == sum(2*(Su+Sv))
    # Let x be the number of links with (Su+Sv) = 1
    # Let y be the number of links with (Su+Sv) = 2
    # (Su+Sv) = 0 links contribute 3^0 - 1 = 0 and 2*0 = 0
    # (Su+Sv) = 1 links contribute 3^1 - 1 = 2 and 2*1 = 2
    # (Su+Sv) = 2 links contribute 3^2 - 1 = 8 and 2*2 = 4
    # Equilibrium: 2x + 8y = 2x + 4y
    # This implies 4y = 0, so y = 0.
    # y is the number of links where both nodes are activated.
    # So we need to choose k nodes such that no two activated nodes are connected.
    # This is equivalent to finding an independent set of size k.
    
    if k > n:
        print("NO")
        return

    # Meet-in-the-middle for Independent Set of size k
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, edges):
        # returns dict: size -> list of bitmasks
        adj_mask = [0] * len(nodes)
        node_map = {node: i for i, node in enumerate(nodes)}
        for u, v in edges:
            if u in node_map and v in node_map:
                adj_mask[node_map[u]] |= (1 << node_map[v])
                adj_mask[node_map[v]] |= (1 << node_map[u])
        
        res = [[] for _ in range(len(nodes) + 1)]
        for i in range(1 << len(nodes)):
            is_indep = True
            cnt = 0
            for j in range(len(nodes)):
                if (i >> j) & 1:
                    cnt += 1
                    if (adj_mask[j] & i):
                        is_indep = False
                        break
            if is_indep:
                res[cnt].append(i)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Check cross-edges
    cross_adj = [0] * n1
    for u, v in adj:
        if u < n1 and v >= n1:
            cross_adj[u] |= (1 << (v - n1))
        elif v < n1 and u >= n1:
            cross_adj[v] |= (1 << (u - n1))
            
    for k1 in range(k + 1):
        k2 = k - k1
        if k1 < len(sets1) and k2 < len(sets2):
            for mask1 in sets1[k1]:
                # Precompute forbidden mask2
                forbidden = 0
                for i in range(n1):
                    if (mask1 >> i) & 1:
                        forbidden |= cross_adj[i]
                
                for mask2 in sets2[k2]:
                    if not (mask2 & forbidden):
                        print("YES")
                        ans = []
                        for i in range(n1):
                            if (mask1 >> i) & 1: ans.append(i + 1)
                        for i in range(n2):
                            if (mask2 >> i) & 1: ans.append(i + n1 + 1)
                        print(*(ans))
                        return
    print("NO")

solve()
