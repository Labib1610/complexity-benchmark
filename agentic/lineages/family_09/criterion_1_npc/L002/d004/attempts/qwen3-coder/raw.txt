from itertools import combinations

def read_input():
    n, m, k = map(int, input().split())
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u-1, v-1))  # Convert to 0-indexed
    return n, m, k, edges

def calculate_metrics(nodes, edges):
    kinetic = 0
    potential = 0
    for u, v in edges:
        su = 1 if u in nodes else 0
        sv = 1 if v in nodes else 0
        activation_sum = su + sv
        kinetic += 3**activation_sum - 1
        potential += 2 * activation_sum
    return kinetic, potential

def solve_small(n, m, k, edges):
    # For small n, we can check all combinations
    if n <= 25:  # Use bitmask approach for smaller n
        for mask in range(1 << n):
            if bin(mask).count('1') != k:
                continue
            active_nodes = set()
            for i in range(n):
                if mask & (1 << i):
                    active_nodes.add(i)
            
            kinetic, potential = calculate_metrics(active_nodes, edges)
            if kinetic == potential:
                return True, [str(i+1) for i in sorted(active_nodes)]
        return False, []
    
    # For larger n, split into two halves and use meet-in-the-middle
    mid = n // 2
    left_indices = list(range(mid))
    right_indices = list(range(mid, n))
    
    # Precompute all possible combinations for the left half
    left_combinations = {}
    for comb in combinations(left_indices, min(k, len(left_indices))):
        active_set = set(comb)
        kinetic_part = 0
        potential_part = 0
        for u, v in edges:
            su = 1 if u in active_set else 0
            sv = 1 if v in active_set else 0
            activation_sum = su + sv
            kinetic_part += 3**activation_sum - 1
            potential_part += 2 * activation_sum
        
        count = len(active_set)
        key = (kinetic_part - potential_part, count)
        if key not in left_combinations:
            left_combinations[key] = []
        left_combinations[key].append(set(comb))
    
    # Check all combinations for the right half
    for r_count in range(min(k+1, len(right_indices)+1)):
        if r_count > k:
            break
        for right_comb in combinations(right_indices, r_count):
            active_set = set(right_comb)
            kinetic_part = 0
            potential_part = 0
            for u, v in edges:
                su = 1 if u in active_set else 0
                sv = 1 if v in active_set else 0
                activation_sum = su + sv
                kinetic_part += 3**activation_sum - 1
                potential_part += 2 * activation_sum
            
            remaining = k - r_count
            target_diff = -(kinetic_part - potential_part)
            
            # Look for matching left combination
            key = (target_diff, remaining)
            if key in left_combinations:
                for left_set in left_combinations[key]:
                    if len(left_set) == remaining:
                        result = sorted(list(left_set) + list(active_set))
                        return True, [str(i+1) for i in result]
    
    return False, []

def main():
    n, m, k, edges = read_input()
    
    # Special case when there are no edges
    if m == 0:
        # Any k nodes will work since both metrics will be 0
        print("YES")
        print(" ".join(str(i) for i in range(1, k+1)))
        return
    
    # Try to solve
    found, nodes = solve_small(n, m, k, edges)
    
    if found:
        print("YES")
        print(" ".join(nodes))
    else:
        print("NO")

if __name__ == "__main__":
    main()
