```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use backtracking with pruning.
    
    selected = []
    
    def is_safe(node, current_selection):
        for neighbor in adj[node]:
            if neighbor in current_selection:
                return False
        return True

    def backtrack(start_node, count):
        if count == k:
            return True
        
        # Pruning: if remaining nodes are not enough to reach k
        if (n - start_node + 1) < (k - count):
            return False
            
        for i in range(start_node, n + 1):
            if is_safe(i, selected):
                selected.append(i)
                if backtrack(i + 1, count + 1):
                    return True
                selected.pop()
        return False

    if backtrack(1, 0):
        print("YES")
        print(*(selected))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
