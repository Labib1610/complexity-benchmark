import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle approach for Maximum Independent Set variant
    # n <= 46, split into n1 = 23, n2 = 23
    n1 = n // 2
    n2 = n - n1
    
    # Precompute valid subsets for each half
    def get_subsets(nodes_range, offset):
        size = len(nodes_range)
        subsets = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            mask = 0
            count = 0
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    if (adj[nodes_range[j]] & mask) != 0:
                        valid = False
                        break
                    mask |= (1 << nodes_range[j])
                    count += 1
            if valid:
                subsets[count].append((mask, i << offset))
        return subsets

    sub1 = get_subsets(range(n1), 0)
    sub2 = get_subsets(range(n1, n), n1)
    
    # Check combinations
    for c1 in range(min(k + 1, n1 + 1)):
        c2 = k - c1
        if c2 < 0 or c2 > n2:
            continue
        
        for m1, original_mask1 in sub1[c1]:
            # Filter m2 such that no edges exist between m1 and m2
            # m2 must not contain any node connected to any node in m1
            forbidden = 0
            for i in range(n1):
                if (m1 >> i) & 1:
                    forbidden |= adj[i]
            
            for m2, original_mask2 in sub2[c2]:
                if (m2 & forbidden) == 0:
                    # Found it
                    res = []
                    combined = original_mask1 | original_mask2
                    for i in range(n):
                        if (combined >> i) & 1:
                            res.append(str(i + 1))
                    print("YES")
                    print(" ".join(res))
                    return

    print("NO")

solve()
