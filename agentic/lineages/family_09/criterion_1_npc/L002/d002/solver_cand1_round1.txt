import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The rule: "number of its connected servers designated for the workload 
    # must never exceed the number of its connected servers that remain in standard mode."
    # For a transformer connected to u and v:
    # Let x_u = 1 if u is selected, 0 otherwise.
    # Let x_v = 1 if v is selected, 0 otherwise.
    # Standard mode count = (1-x_u) + (1-x_v)
    # Workload count = x_u + x_v
    # Rule: x_u + x_v <= (1-x_u) + (1-x_v)
    # x_u + x_v <= 2 - x_u - x_v
    # 2(x_u + x_v) <= 2
    # x_u + x_v <= 1
    # This means for every edge (u, v), we cannot have both u and v selected.
    # This is exactly the Independent Set problem.
    # We need to find an independent set of size exactly k.
    
    # Since n <= 46, we use meet-in-the-middle.
    n1 = n // 2
    n2 = n - n1
    
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    def get_independent_sets(nodes, offset):
        size = len(nodes)
        res = {} # count -> list of bitmasks
        for i in range(1 << size):
            mask = 0
            valid = True
            count = 0
            for j in range(size):
                if (i >> j) & 1:
                    node = nodes[j]
                    if mask & adj[node]:
                        valid = False
                        break
                    mask |= (1 << node)
                    count += 1
            if valid:
                if count not in res: res[count] = []
                res[count].append(mask)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, 0)
    sets2 = get_independent_sets(nodes2, n1)
    
    for c1, masks1 in sets1.items():
        c2 = k - c1
        if c2 in sets2:
            for m1 in masks1:
                for m2 in sets2[c2]:
                    # Check if m1 and m2 are compatible
                    # They are compatible if no edge exists between any node in m1 and any node in m2
                    compatible = True
                    for i in range(n1):
                        if (m1 >> i) & 1:
                            if adj[i] & m2:
                                compatible = False
                                break
                    if compatible:
                        print("YES")
                        res = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                res.append(str(i + 1))
                        print(" ".join(res))
                        return
    print("NO")

solve()
