from itertools import combinations

def solve():
    n, m, k = map(int, input().split())
    
    # Read transformer connections
    transformers = []
    for _ in range(m):
        u, v = map(int, input().split())
        transformers.append((u-1, v-1))  # Convert to 0-indexed
    
    # Try all possible combinations of k servers
    for selected in combinations(range(n), k):
        # Check if this selection violates any transformer constraint
        valid = True
        selected_set = set(selected)
        
        for u, v in transformers:
            # Count how many of the connected servers are selected
            count_selected = 0
            if u in selected_set:
                count_selected += 1
            if v in selected_set:
                count_selected += 1
            
            # The number of selected servers must not exceed non-selected servers
            # Since there are exactly 2 servers per transformer, this means:
            # count_selected <= 2 - count_selected
            # Which simplifies to: count_selected <= 1
            if count_selected > 1:
                valid = False
                break
        
        if valid:
            print("YES")
            # Convert back to 1-indexed and print
            result = [str(x + 1) for x in selected]
            print(" ".join(result))
            return
    
    print("NO")

solve()
