from itertools import combinations

def solve():
    n, m, k = map(int, input().split())
    
    # Read edges
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u-1, v-1))  # Convert to 0-indexed
    
    # Try all combinations of k nodes to be positive
    nodes = list(range(n))
    
    for positive_nodes in combinations(nodes, k):
        # Create a set for O(1) lookup
        positive_set = set(positive_nodes)
        
        # Check if this assignment is valid
        valid = True
        for u, v in edges:
            # Count polarities: +1 for positive, -1 for negative
            polarity_sum = 0
            if u in positive_set:
                polarity_sum += 1
            else:
                polarity_sum -= 1
                
            if v in positive_set:
                polarity_sum += 1
            else:
                polarity_sum -= 1
            
            # If any edge has positive sum, this assignment is invalid
            if polarity_sum > 0:
                valid = False
                break
        
        if valid:
            # Found a valid assignment
            print("YES")
            result = [str(node + 1) for node in positive_nodes]  # Convert back to 1-indexed
            print(" ".join(result))
            return
    
    # No valid assignment found
    print("NO")

solve()
