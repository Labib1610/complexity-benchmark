import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition: if both servers in a loop are high-intensity, 
    # heat = 100 + 100 = 200 > 100. So, no two high-intensity servers 
    # can be connected by a loop.
    # This means the set of k servers must be an Independent Set.
    
    # Since n <= 46, we can use backtracking with pruning or bitmasking.
    # Given the constraints and the nature of the problem, 
    # we look for an independent set of size k.
    
    selected = []
    
    def is_independent(node, current_set):
        for neighbor in adj[node]:
            if neighbor in current_set:
                return False
        return True

    def backtrack(start_node, current_set):
        if len(current_set) == k:
            return True
        
        # Pruning: if remaining nodes are not enough to reach k
        if len(current_set) + (n - start_node + 1) < k:
            return False
            
        for i in range(start_node, n + 1):
            if is_independent(i, current_set):
                current_set.append(i)
                if backtrack(i + 1, current_set):
                    return True
                current_set.pop()
        return False

    if backtrack(1, selected):
        print("YES")
        print(*(selected))
    else:
        print("NO")

solve()
