import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem is equivalent to finding an Independent Set of size k in the graph
    # where edges represent cross-reactivity profiles.
    # Since n <= 46, we can use a meet-in-the-middle approach or a backtracking search.
    
    nodes = list(range(1, n + 1))
    
    # Sort nodes by degree to optimize backtracking
    nodes.sort(key=lambda x: len(adj[x]))
    
    adj_mask = [0] * (n + 1)
    for i in range(1, n + 1):
        for neighbor in adj[i]:
            adj_mask[i] |= (1 << (neighbor - 1))
            
    def backtrack(idx, current_set, current_mask):
        if len(current_set) == k:
            return current_set
        
        if idx == n:
            return None
        
        # Try including nodes[idx]
        u = nodes[idx]
        if not (current_mask & (1 << (u - 1))):
            res = backtrack(idx + 1, current_set + [u], current_mask | adj_mask[u] | (1 << (u - 1)))
            if res is not None:
                return res
        
        # Try excluding nodes[idx]
        # Pruning: if remaining nodes are not enough to reach k
        if (n - 1 - idx) >= (k - len(current_set)):
            res = backtrack(idx + 1, current_set, current_mask)
            if res is not None:
                return res
        
        return None

    result = backtrack(0, [], 0)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
