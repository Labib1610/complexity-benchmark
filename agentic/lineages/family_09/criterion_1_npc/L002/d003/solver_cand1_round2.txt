import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Metabolic load condition:
    # For a link (u, v):
    # If both u, v activated: load = 1 + 1 = 2 (Invalid)
    # If one activated, one not: load = 1 - 1 = 0 (Valid)
    # If both not activated: load = -1 - 1 = -2 (Valid)
    # Thus, we cannot have both u and v activated.
    # This means the set of activated traits must be an Independent Set.
    
    # We need an independent set of size exactly k.
    # Since n <= 46, we use a meet-in-the-middle approach.
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        # mask represents subset of nodes
        for i in range(1 << size):
            is_independent = True
            count = 0
            for j in range(size):
                if (i >> j) & 1:
                    count += 1
                    if adj_list[nodes[j]] & i_to_mask(i, nodes):
                        is_independent = False
                        break
            if is_independent:
                res[count].append(i_to_mask(i, nodes))
        return res

    def i_to_mask(i, nodes):
        m = 0
        for j in range(len(nodes)):
            if (i >> j) & 1:
                m |= (1 << nodes[j])
        return m

    # Optimized search for independent set of size k
    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = [[] for _ in range(n1 + 1)]
    for i in range(1 << n1):
        valid = True
        for j in range(n1):
            if (i >> j) & 1:
                if adj[j] & i:
                    valid = False
                    break
        if valid:
            sets1[bin(i).count('1')].append(i)
            
    sets2 = [[] for _ in range(n2 + 1)]
    for i in range(1 << n2):
        valid = True
        mask = i << n1
        for j in range(n2):
            if (i >> j) & 1:
                if adj[j + n1] & mask:
                    valid = False
                    break
        if valid:
            sets2[bin(i).count('1')].append(mask)

    for c1 in range(n1 + 1):
        c2 = k - c1
        if 0 <= c2 <= n2:
            for s1 in sets1[c1]:
                for s2 in sets2[c2]:
                    if not (adj[bin(s1).bit_length()-1] & s2) and not (adj[n1] & s1 and False): # dummy
                        # Check if s1 and s2 are compatible
                        compatible = True
                        for i in range(n1):
                            if (s1 >> i) & 1:
                                if adj[i] & s2:
                                    compatible = False
                                    break
                        if compatible:
                            res = []
                            combined = s1 | s2
                            for i in range(n):
                                if (combined >> i) & 1:
                                    res.append(i + 1)
                            print("YES")
                            print(*(res))
                            return
    print("NO")

solve()
