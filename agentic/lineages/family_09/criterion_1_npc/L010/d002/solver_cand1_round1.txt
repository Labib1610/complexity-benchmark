import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [set() for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # The problem is equivalent to finding an Independent Set of size k.
    # Since n is small (up to 46), we can use a backtracking approach with pruning.
    
    compounds = list(range(1, n + 1))
    # Sort by degree to prune search space (heuristic)
    compounds.sort(key=lambda x: len(adj[x]))
    
    result = []
    
    def backtrack(index, current_set, forbidden_mask):
        if len(current_set) == k:
            return current_set
        
        if index == n:
            return None
        
        # Remaining elements to pick
        remaining = k - len(current_set)
        if n - index < remaining:
            return None
        
        # Try including compounds[index]
        u = compounds[index]
        if not (forbidden_mask & (1 << u)):
            new_forbidden = forbidden_mask
            for neighbor in adj[u]:
                new_forbidden |= (1 << neighbor)
            
            res = backtrack(index + 1, current_set + [u], new_forbidden)
            if res is not None:
                return res
        
        # Try excluding compounds[index]
        res = backtrack(index + 1, current_set, forbidden_mask)
        if res is not None:
            return res
            
        return None

    # Use a bitmask for forbidden compounds for speed
    # Since n <= 46, bitmask fits in a 64-bit integer
    final_res = backtrack(0, [], 0)
    
    if final_res:
        print("YES")
        print(*(final_res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
