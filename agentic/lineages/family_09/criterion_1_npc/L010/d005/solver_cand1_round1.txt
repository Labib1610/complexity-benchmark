import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
        idx += 2
        
    degrees = [len(adj[i]) for i in range(n + 1)]
    
    # Let S be the set of standard modules (size n-k)
    # Let U be the set of upgraded modules (size k)
    # Stability condition:
    # (Number of edges between two modules in S) = (Sum of degrees of modules in S) - m
    # Let E(S) be the number of edges with both endpoints in S.
    # Let D(S) be the sum of degrees of modules in S.
    # Condition: E(S) = D(S) - m
    
    # Note: D(S) = sum_{v in S} deg(v)
    # Also, sum_{v in S} deg(v) = 2 * E(S) + E(S, U), where E(S, U) is the number of edges between S and U.
    # So E(S) = 2 * E(S) + E(S, U) - m
    # m - E(S, U) = E(S)
    # This is equivalent to saying the number of edges with at least one endpoint in U is m - E(S) = E(S, U) + E(U)
    # Actually, the condition E(S) = D(S) - m is equivalent to:
    # m = D(S) - E(S)
    # Since D(S) = 2*E(S) + E(S, U),
    # m = 2*E(S) + E(S, U) - E(S) = E(S) + E(S, U)
    # This is just the definition of m (total edges = edges within S + edges between S and U + edges within U).
    # Wait, the condition is E(S) = D(S) - m.
    # D(S) = sum_{v in S} deg(v) = 2*E(S) + E(S, U)
    # So E(S) = 2*E(S) + E(S, U) - m
    # m = E(S) + E(S, U)
    # This means there are no edges within U.
    # So the condition is equivalent to: E(U) = 0.
    
    # Check if there exists a subset U of size k such that no two modules in U are connected.
    # This is the Independent Set problem.
    
    if k == 0:
        # If k=0, S is all modules. E(S) = m, D(S) = 2m.
        # Condition: m = 2m - m, which is always true.
        print("YES")
        print("")
        return

    # Since n <= 46, we can use meet-in-the-middle or simple backtracking for Independent Set.
    # Given the constraints and the nature of the problem, a simple search is likely sufficient.
    
    def get_independent_set(k, current_nodes, forbidden):
        if k == 0:
            return []
        if not current_nodes:
            return None
        
        v = current_nodes[0]
        # Try including v
        if v not in forbidden:
            new_forbidden = forbidden.union(adj[v])
            res = get_independent_set(k - 1, [n for n in current_nodes[1:] if n not in new_forbidden], new_forbidden)
            if res is not None:
                return [v] + res
        
        # Try excluding v
        res = get_independent_set(k, current_nodes[1:], forbidden)
        return res

    # The problem asks for k upgraded modules.
    # We need to find an independent set of size k.
    
    import itertools
    
    # Simple backtracking with pruning
    def find_is(idx, count, current_set, forbidden_mask):
        if count == k:
            return current_set
        if idx > n or (n - idx + 1) < (k - count):
            return None
        
        # Try including idx
        if not (forbidden_mask & (1 << idx)):
            new_mask = forbidden_mask
            for neighbor in adj[idx]:
                new_mask |= (1 << neighbor)
            res = find_is(idx + 1, count + 1, current_set + [idx], new_mask)
            if res: return res
            
        # Try excluding idx
        res = find_is(idx + 1, count, current_set, forbidden_mask)
        return res

    # Use a bitmask for forbidden nodes
    adj_mask = [0] * (n + 1)
    for i in range(1, n + 1):
        for neighbor in adj[i]:
            adj_mask[i] |= (1 << neighbor)
            
    def find_is_fast(idx, count, current_set, forbidden_mask):
        if count == k:
            return current_set
        if idx > n:
            return None
        
        # Pruning
        if (n - idx + 1) < (k - count):
            return None
            
        # Try including idx
        if not (forbidden_mask & (1 << idx)):
            res = find_is_fast(idx + 1, count + 1, current_set + [idx], forbidden_mask | adj_mask[idx])
            if res: return res
            
        # Try excluding idx
        res = find_is_fast(idx + 1, count, current_set, forbidden_mask)
        return res

    result = find_is_fast(1, 0, [], 0)
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
