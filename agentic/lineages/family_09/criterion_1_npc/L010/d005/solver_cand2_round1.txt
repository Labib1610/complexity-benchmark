import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    deg = [0] * (n + 1)
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1
        deg[v] += 1
        edges.append((u, v))
        idx += 2
        
    # Let S be the set of upgraded modules, |S| = k.
    # Power factor p_i = 2 if i in S, else 1.
    # Link load for (u, v) is p_u * p_v.
    # Total load = sum_{(u,v) in E} p_u * p_v
    # p_u * p_v = (1 + I(u in S)) * (1 + I(v in S)) = 1 + I(u in S) + I(v in S) + I(u in S)*I(v in S)
    # Total load = sum_{(u,v) in E} (1 + I(u in S) + I(v in S) + I(u in S)*I(v in S))
    # Total load = m + sum_{u in S} deg(u) + sum_{(u,v) in E, u in S, v in S} 1
    # Stability condition: Total load = m + sum_{u in S} deg(u)
    # This implies sum_{(u,v) in E, u in S, v in S} 1 = 0.
    # This means no two upgraded modules can be connected by a critical link.
    # S must be an independent set of size k.
    
    # Since n <= 46, we can use meet-in-the-middle to find an independent set of size k.
    # Split into two halves: n1 = n // 2, n2 = n - n1
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes):
        # returns dict: size -> list of bitmasks
        res = {i: [] for i in range(len(nodes) + 1)}
        res[0].append(0)
        for i, u in enumerate(nodes):
            # neighbors of u in nodes
            mask = 0
            for j, v in enumerate(nodes):
                if v in adj[u]:
                    mask |= (1 << j)
            
            for size in range(i, -1, -1):
                for msk in res[size]:
                    if not (msk & (1 << i)):
                        if not (msk & mask):
                            res[size + 1].append(msk | (1 << i))
        return res

    nodes1 = list(range(1, n1 + 1))
    nodes2 = list(range(n1 + 1, n + 1))
    
    sets1 = get_independent_sets(nodes1)
    sets2 = get_independent_sets(nodes2)
    
    # Precompute adjacency between nodes1 and nodes2
    adj_cross = [0] * n1
    for i in range(n1):
        u = nodes1[i]
        for j in range(n2):
            v = nodes2[j]
            if v in adj[u]:
                adj_cross[i] |= (1 << j)
                
    for s1 in range(k + 1):
        s2 = k - s1
        if s1 in sets1 and s2 in sets2:
            for msk1 in sets1[s1]:
                # Filter msk2 that are compatible with msk1
                forbidden = 0
                for i in range(n1):
                    if (msk1 >> i) & 1:
                        forbidden |= adj_cross[i]
                
                for msk2 in sets2[s2]:
                    if not (msk2 & forbidden):
                        print("YES")
                        ans = []
                        for i in range(n1):
                            if (msk1 >> i) & 1: ans.append(nodes1[i])
                        for i in range(n2):
                            if (msk2 >> i) & 1: ans.append(nodes2[i])
                        print(*(ans))
                        return
    print("NO")

solve()
