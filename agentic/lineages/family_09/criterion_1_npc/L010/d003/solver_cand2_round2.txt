import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle: n <= 46, split into n1 = 23, n2 = 23
    n1 = n // 2
    n2 = n - n1
    
    # Precompute valid subsets for each half
    # valid_subsets[count] = list of bitmasks
    def get_valid(num_nodes, offset):
        valid = [[] for _ in range(num_nodes + 1)]
        for i in range(1 << num_nodes):
            mask = 0
            count = 0
            possible = True
            for j in range(num_nodes):
                if (i >> j) & 1:
                    if adj[j + offset] & (i << offset):
                        possible = False
                        break
                    mask |= (1 << (j + offset))
                    count += 1
            if possible:
                valid[count].append(mask)
        return valid

    v1 = get_valid(n1, 0)
    v2 = get_valid(n2, n1)
    
    # Check if any combination of v1[c1] and v2[c2] works where c1 + c2 = k
    for c1 in range(n1 + 1):
        c2 = k - c1
        if 0 <= c2 <= n2:
            for m1 in v1[c1]:
                # Check conflicts between m1 and m2
                # m2 must not have any node connected to a node in m1
                forbidden = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        forbidden |= adj[i]
                
                for m2 in v2[c2]:
                    if not (m2 & forbidden):
                        # Found it
                        res = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                res.append(i + 1)
                        print("YES")
                        print(*(res))
                        return
    print("NO")

solve()
