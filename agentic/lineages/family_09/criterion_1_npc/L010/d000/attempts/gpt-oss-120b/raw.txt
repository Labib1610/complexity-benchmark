import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    all_vertices_mask = (1 << n) - 1
    solution = []

    # simple branch and bound DFS
    def dfs(cand_mask: int, cur_mask: int, cur_list: list) -> bool:
        nonlocal solution
        cur_size = len(cur_list)
        if cur_size == k:
            solution = cur_list[:]
            return True
        # prune if even taking all remaining cannot reach k
        if cur_size + cand_mask.bit_count() < k:
            return False

        # choose a vertex with maximum degree in the candidate set (heuristic)
        # to possibly reduce branching
        # extract candidates into list
        # pick first vertex
        # we can pick the vertex with highest degree among candidates
        # compute degrees quickly
        # simple: pick the least significant bit
        u = (cand_mask & -cand_mask).bit_length() - 1

        # Branch: include u
        new_cand = cand_mask & ~adj[u] & ~(1 << u)
        if dfs(new_cand, cur_mask | (1 << u), cur_list + [u + 1]):
            return True

        # Branch: exclude u
        cand_mask &= ~(1 << u)
        if dfs(cand_mask, cur_mask, cur_list):
            return True

        return False

    # start DFS
    if dfs(all_vertices_mask, 0, []):
        print("YES")
        print(" ".join(map(str, solution)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
