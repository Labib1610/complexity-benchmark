We can solve this by using a backtracking search with pruning, since n ≤ 46 is small enough for a branch-and-bound approach.

```python
import sys

def main():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        edges.append((u, v))
    
    # adjacency bitmask
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    # For pruning: if a vertex has too many neighbors, skip early
    # We'll do a simple backtracking with ordering by degree
    order = sorted(range(n), key=lambda x: bin(adj[x]).count('1'))
    
    chosen = []
    
    def backtrack(start, chosen_count):
        if chosen_count == k:
            return True
        if start >= n:
            return False
        # Prune: not enough remaining vertices
        if chosen_count + (n - start) < k:
            return False
        
        # Try including order[start]
        v = order[start]
        # Check if v conflicts with chosen
        conflict = False
        mask = 0
        for c in chosen:
            if (adj[v] >> c) & 1:
                conflict = True
                break
        if not conflict:
            chosen.append(v)
            if backtrack(start + 1, chosen_count + 1):
                return True
            chosen.pop()
        
        # Try excluding order[start]
        if backtrack(start + 1, chosen_count):
            return True
        
        return False
    
    if backtrack(0, 0):
        print("YES")
        print(" ".join(str(x + 1) for x in chosen))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
