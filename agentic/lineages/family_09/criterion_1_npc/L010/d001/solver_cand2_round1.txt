import sys

def solve():
    # Palladium atomic number is 46
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * (n + 1)
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    # We need to find the lexicographically smallest subset of size k
    # that is an independent set.
    # Since n <= 46, we can use backtracking with pruning.
    
    result = []
    
    def find_set(start_node, current_set, forbidden_mask):
        if len(current_set) == k:
            return current_set
        
        for i in range(start_node, n + 1):
            if not (forbidden_mask & (1 << (i - 1))):
                # Try including i
                res = find_set(i + 1, current_set + [i], forbidden_mask | adj[i] | (1 << (i - 1)))
                if res:
                    return res
        return None

    ans = find_set(1, [], 0)
    
    if ans:
        print("YES")
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
