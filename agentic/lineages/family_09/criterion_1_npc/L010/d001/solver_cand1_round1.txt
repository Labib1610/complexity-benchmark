import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle for Maximum Independent Set of size k
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the two halves
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        # sets_by_size[s] = list of bitmasks of size s
        sets_by_size = [[] for _ in range(size + 1)]
        sets_by_size[0].append(0)
        
        for i in range(size):
            u = nodes[i]
            u_adj = adj_list[u]
            for s in range(i, -1, -1):
                for mask in sets_by_size[s]:
                    if not (mask & (1 << u)) and not (u_adj & mask):
                        sets_by_size[s+1].append(mask | (1 << u))
        return sets_by_size

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Precompute adjacency for second half relative to first half
    adj2_to_1 = [0] * n2
    for i in range(n2):
        u = nodes2[i]
        adj2_to_1[i] = adj[u] & ((1 << n1) - 1)
        
    for s1 in range(k + 1):
        s2 = k - s1
        if s1 < len(sets1) and s2 < len(sets2):
            for mask1 in sets1[s1]:
                # Find all nodes in second half compatible with mask1
                allowed_mask2 = 0
                for i in range(n2):
                    if not (adj[nodes2[i]] & mask1):
                        allowed_mask2 |= (1 << i)
                
                for mask2 in sets2[s2]:
                    if (mask2 & allowed_mask2) == mask2:
                        # Found it
                        print("YES")
                        res = []
                        for i in range(n1):
                            if (mask1 >> i) & 1:
                                res.append(i + 1)
                        for i in range(n2):
                            if (mask2 >> i) & 1:
                                res.append(nodes2[i] + 1)
                        print(*(sorted(res)))
                        return
    
    print("NO")

solve()
