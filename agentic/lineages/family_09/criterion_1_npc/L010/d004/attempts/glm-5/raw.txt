import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[False] * n for _ in range(n)]
    deg = [0] * n
    
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u][v] = True
        adj[v][u] = True
        deg[u] += 1
        deg[v] += 1
    
    # We need to find a subset S of size k such that:
    # 2 * |E(S, V\S)| = sum_{v in S} deg(v)
    # This is equivalent to: sum_{v in S} deg(v) - 2*|E(S)| = 2*|E(S, V\S)|
    # Which simplifies to: sum_{v in S} deg(v) = 2*|E(S)| + 2*|E(S, V\S)|
    # Since total edges incident to S = 2*|E(S)| + |E(S, V\S)|
    # The condition becomes: sum_{v in S} deg(v) = 2 * (edges incident to S) - 2*|E(S)|
    # Or: |E(S)| = (sum deg - edges_incident) = edges_incident - |E(S, V\S)|
    # Actually, let's derive properly:
    # Condition: |E(S, V\S)| = sum_{v in S} deg(v)
    # LHS = edges_incident_to_S - 2*|E(S)|
    # RHS = sum_{v in S} deg(v) = edges_incident_to_S
    # So condition is: edges_incident_to_S - 2*|E(S)| = edges_incident_to_S
    # This gives: |E(S)| = 0
    # So S must be an independent set!
    
    # Use meet-in-the-middle for n <= 46
    half = n // 2
    left_size = half
    right_size = n - half
    
    # Precompute for left half
    left_nodes = list(range(left_size))
    left_indep_sets = {}  # (size, edges_to_right) -> list of bitmasks
    
    for mask in range(1 << left_size):
        # Check if it's an independent set
        valid = True
        for i in range(left_size):
            if mask & (1 << i):
                for j in range(i + 1, left_size):
                    if mask & (1 << j):
                        if adj[i][j]:
                            valid = False
                            break
                if not valid:
                    break
        
        if valid:
            size = bin(mask).count('1')
            # Count edges to right half
            edges_to_right = 0
            for i in range(left_size):
                if mask & (1 << i):
                    for j in range(left_size, n):
                        if adj[i][j]:
                            edges_to_right += 1
            
            key = (size, edges_to_right)
            if key not in left_indep_sets:
                left_indep_sets[key] = []
            left_indep_sets[key].append(mask)
    
    # Precompute for right half
    right_nodes = list(range(left_size, n))
    right_indep_sets = {}  # (size, edges_to_left) -> list of bitmasks
    
    for mask in range(1 << right_size):
        # Check if it's an independent set
        valid = True
        for i in range(right_size):
            if mask & (1 << i):
                for j in range(i + 1, right_size):
                    if mask & (1 << j):
                        if adj[left_size + i][left_size + j]:
                            valid = False
                            break
                if not valid:
                    break
        
        if valid:
            size = bin(mask).count('1')
            # Count edges to left half
            edges_to_left = 0
            for i in range(right_size):
                if mask & (1 << i):
                    node = left_size + i
                    for j in range(left_size):
                        if adj[node][j]:
                            edges_to_left += 1
            
            key = (size, edges_to_left)
            if key not in right_indep_sets:
                right_indep_sets[key] = []
            right_indep_sets[key].append(mask)
    
    # Try to find a combination with total size k
    for left_key in left_indep_sets:
        left_size_subset, left_edges = left_key
        needed_right_size = k - left_size_subset
        
        if needed_right_size < 0:
            continue
        
        # We need: edges from left to right + edges from right to left = 0
        # Since left_edges = edges from left to right, we need right_edges = 0
        # and left_edges = 0 (no edges between left and right parts)
        
        right_key = (needed_right_size, 0)
        
        if left_edges == 0 and right_key in right_indep_sets:
            # Found a valid combination
            left_mask = left_indep_sets[left_key][0]
            right_mask = right_indep_sets[right_key][0]
            
            result = []
            for i in range(left_size):
                if left_mask & (1 << i):
                    result.append(i + 1)
            for i in range(right_size):
                if right_mask & (1 << i):
                    result.append(left_size + i + 1)
            
            print("YES")
            print(" ".join(map(str, result)))
            return
    
    print("NO")

if __name__ == "__main__":
    solve()
