import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    degree = [0] * (n + 1)
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
        edges.append((u, v))
        idx += 2
        
    # Let S be the set of upgraded modules, |S| = k.
    # Let A be the set of asymmetric links.
    # A link (u, v) is in A if (u in S and v not in S) or (u not in S and v in S).
    # Let d(u) be the degree of module u.
    # Sum of capacities of upgraded modules = sum_{u in S} d(u).
    # Number of asymmetric links |A| = sum_{u in S} d(u) - 2 * (number of links between two upgraded modules).
    # The condition |A| = sum_{u in S} d(u) implies:
    # sum_{u in S} d(u) - 2 * (links within S) = sum_{u in S} d(u)
    # This simplifies to: 2 * (links within S) = 0, so (links within S) = 0.
    # Thus, we need to find an independent set of size k.
    
    # Since n <= 46, we use meet-in-the-middle to find an independent set of size k.
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes):
        size = len(nodes)
        res = {} # {count: [bitmasks]}
        for i in range(1 << size):
            subset = [nodes[j] for j in range(size) if (i & (1 << j))]
            is_independent = True
            for u in subset:
                for v in subset:
                    if v in adj[u]:
                        is_independent = False
                        break
                if not is_independent: break
            if is_independent:
                count = len(subset)
                if count not in res: res[count] = []
                res[count].append(i)
        return res

    nodes1 = list(range(1, n1 + 1))
    nodes2 = list(range(n1 + 1, n + 1))
    
    sets1 = get_independent_sets(nodes1)
    sets2 = get_independent_sets(nodes2)
    
    for c1 in sets1:
        c2 = k - c1
        if c2 in sets2:
            for mask1 in sets1[c1]:
                subset1 = [nodes1[j] for j in range(n1) if (mask1 & (1 << j))]
                for mask2 in sets2[c2]:
                    subset2 = [nodes2[j] for j in range(n2) if (mask2 & (1 << j))]
                    # Check if any edge between subset1 and subset2
                    possible = True
                    for u in subset1:
                        for v in subset2:
                            if v in adj[u]:
                                possible = False
                                break
                        if not possible: break
                    if possible:
                        print("YES")
                        print(*(subset1 + subset2))
                        return
    print("NO")

solve()
