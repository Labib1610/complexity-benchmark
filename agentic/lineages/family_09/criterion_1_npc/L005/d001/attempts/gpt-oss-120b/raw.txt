import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n  # undirected conflict mask
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    ALL = (1 << n) - 1
    solution_mask = 0
    found = False

    sys.setrecursionlimit(10000)

    # simple recursive backtracking with pruning
    def dfs(cur_mask: int, cand_mask: int) -> bool:
        nonlocal solution_mask, found
        cur_sz = cur_mask.bit_count()
        if cur_sz == k:
            solution_mask = cur_mask
            return True
        # not enough vertices left
        if cur_sz + cand_mask.bit_count() < k:
            return False
        # choose a vertex (least significant)
        v = (cand_mask & -cand_mask).bit_length() - 1
        # branch: include v
        new_cand = cand_mask & ~adj[v] & ~(1 << v)
        if dfs(cur_mask | (1 << v), new_cand):
            return True
        # branch: exclude v
        cand_mask &= ~(1 << v)
        if dfs(cur_mask, cand_mask):
            return True
        return False

    dfs(0, ALL)

    if not solution_mask:
        print("NO")
    else:
        print("YES")
        res = [str(i + 1) for i in range(n) if (solution_mask >> i) & 1]
        print(" ".join(res))

if __name__ == "__main__":
    solve()
