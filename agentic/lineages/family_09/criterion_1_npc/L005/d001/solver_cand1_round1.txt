import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # We need to find a clique of size k in the complement graph.
    # The graph is undirected because if u has a grievance against v,
    # they cannot both be in the delegation.
    # This is the Maximum Clique problem, which is NP-hard,
    # but n <= 46 is small enough for a Bron-Kerbosch variant or bitmask search.
    
    comp_adj = [((1 << n) - 1) ^ (1 << i) ^ adj[i] for i in range(n)]
    
    def find_clique(candidates, current_clique):
        if len(current_clique) == k:
            return current_clique
        if not candidates or len(current_clique) + bin(candidates).count('1') < k:
            return None
        
        # Pivot optimization
        pivot = (candidates | (candidates >> 0)).bit_length() - 1
        if pivot < 0: return None
        
        remaining = candidates & ~comp_adj[pivot]
        
        # Iterate over candidates
        while remaining:
            v = (remaining & -remaining).bit_length() - 1
            res = find_clique(candidates & comp_adj[v], current_clique + [v + 1])
            if res:
                return res
            remaining &= ~(1 << v)
            candidates &= ~(1 << v)
        return None

    result = find_clique((1 << n) - 1, [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
