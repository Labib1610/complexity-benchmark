```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition x . v_j <= 1 means we cannot pick both indices u and v
    # This is equivalent to finding an independent set of size k in the graph
    # where edges represent the constraints.
    
    # Since n <= 46, we can use a backtracking approach with pruning.
    # For small n, this is efficient.
    
    selected = []
    
    def backtrack(curr_idx, count):
        if count == k:
            return True
        if curr_idx > n or (n - curr_idx + 1) < (k - count):
            return False
        
        # Try including curr_idx
        can_include = True
        for neighbor in adj[curr_idx]:
            if neighbor in selected:
                can_include = False
                break
        
        if can_include:
            selected.append(curr_idx)
            if backtrack(curr_idx + 1, count + 1):
                return True
            selected.pop()
            
        # Try excluding curr_idx
        if backtrack(curr_idx + 1, count):
            return True
            
        return False

    if backtrack(1, 0):
        print("YES")
        print(*(selected))
    else:
        print("NO")

solve()
```
