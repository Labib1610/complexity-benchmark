import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The stability equation 3^(s_u + s_v) - 2^(s_u + s_v) = s_u + s_v
    # If s_u + s_v = 0: 3^0 - 2^0 = 1 - 1 = 0. (Valid)
    # If s_u + s_v = 1: 3^1 - 2^1 = 3 - 2 = 1. (Valid)
    # If s_u + s_v = 2: 3^2 - 2^2 = 9 - 4 = 5 != 2. (Invalid)
    # Thus, no two connected qubits can both be in state 1.
    # This is equivalent to finding an independent set of size k.
    
    # Since n <= 46, we can use a backtracking approach with pruning.
    # To optimize, we can use the Bron-Kerbosch style or simple bitmasking/backtracking.
    
    nodes = list(range(1, n + 1))
    # Sort nodes by degree to prune faster
    nodes.sort(key=lambda x: len(adj[x]), reverse=True)
    
    res = []
    
    def backtrack(idx, current_k, current_set, forbidden_mask):
        if current_k == k:
            return True
        if idx == n or (n - idx) < (k - current_k):
            return False
        
        u = nodes[idx]
        
        # Try including u if not forbidden
        if not (forbidden_mask & (1 << u)):
            new_forbidden = forbidden_mask
            for v in adj[u]:
                new_forbidden |= (1 << v)
            
            current_set.append(u)
            if backtrack(idx + 1, current_k + 1, current_set, new_forbidden):
                return True
            current_set.pop()
            
        # Try excluding u
        if backtrack(idx + 1, current_k, current_set, forbidden_mask):
            return True
            
        return False

    if backtrack(0, 0, res, 0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
