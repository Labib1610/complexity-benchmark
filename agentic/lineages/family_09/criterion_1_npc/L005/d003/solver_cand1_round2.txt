import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * (n + 1)
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    
    def find_clique(candidates, current_group):
        if len(current_group) == k:
            return current_group
        
        if len(current_group) + len(candidates) < k:
            return None
        
        while candidates:
            # Heuristic: pick the candidate with the fewest neighbors in the remaining set
            # Or simply pick the first one.
            v = candidates[0]
            
            new_candidates = [c for c in candidates[1:] if not (adj[v] & (1 << (c - 1)))]
            
            res = find_clique(new_candidates, current_group + [v])
            if res:
                return res
            
            candidates = candidates[1:]
            if len(current_group) + len(candidates) < k:
                break
        return None

    result = find_clique(list(range(1, n + 1)), [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
