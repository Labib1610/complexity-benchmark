```python
import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    if k == 0:
        print("YES")
        print()
        return
    
    # The polynomial evaluates to 0 if and only if no two variables 
    # set to 1 are connected by an edge in the graph defined by the terms.
    # This is equivalent to finding an independent set of size k.
    
    nodes = list(range(1, n + 1))
    
    # For n <= 46, we can use a randomized approach or backtracking.
    # Since we need an independent set of size k, we can use a simple 
    # randomized greedy search with restarts.
    
    for _ in range(200000):
        random.shuffle(nodes)
        res = []
        forbidden = [False] * (n + 1)
        
        for node in nodes:
            if not forbidden[node]:
                res.append(node)
                if len(res) == k:
                    print("YES")
                    print(*(sorted(res)))
                    return
                forbidden[node] = True
                for neighbor in adj[node]:
                    forbidden[neighbor] = True
                    
    print("NO")

solve()
```
