We need to find an independent set of size exactly `k` in a graph with up to 46 vertices.  
A brute force over all subsets is too large, but we can use a meet-in-the-middle or branch-and-bound approach.

Here’s a possible solution using a recursive backtracking with pruning:

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Try to find an independent set of size k
    # Use recursive backtracking with pruning
    # We'll try to pick vertices in order, skipping conflicts

    # Precompute for each vertex the set of vertices that conflict with it
    # We'll use bitmasks for efficiency

    # Recursive function: idx = current vertex index to consider
    # chosen = bitmask of chosen vertices
    # count = number chosen so far
    # available = bitmask of vertices that can still be chosen (no conflicts with chosen)
    def dfs(idx, chosen, count, available):
        if count == k:
            return chosen
        if idx >= n:
            return None
        # Prune: if even taking all remaining possible vertices can't reach k
        if count + bin(available).count("1") < k:
            return None

        # Option 1: skip vertex idx
        res = dfs(idx + 1, chosen, count, available)
        if res is not None:
            return res

        # Option 2: take vertex idx if possible
        if (available >> idx) & 1:
            new_available = available & ~adj[idx]
            # remove idx itself from available
            new_available &= ~(1 << idx)
            res = dfs(idx + 1, chosen | (1 << idx), count + 1, new_available)
            if res is not None:
                return res
        return None

    # Initial available: all vertices
    initial_available = (1 << n) - 1
    result = dfs(0, 0, 0, initial_available)

    if result is None:
        print("NO")
    else:
        print("YES")
        selected = [str(i + 1) for i in range(n) if (result >> i) & 1]
        print(" ".join(selected))

if __name__ == "__main__":
    solve()
```
