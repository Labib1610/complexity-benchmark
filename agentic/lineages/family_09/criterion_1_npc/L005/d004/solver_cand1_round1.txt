import sys
import itertools
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    patterns = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        patterns.append((u, v))
        idx += 2
        
    # Divergence score of C and P = {u, v} is:
    # |(C \ P) U (P \ C)| = |C| + |P| - 2 * |C intersect P|
    # = k + 2 - 2 * |C intersect {u, v}|
    # Condition: k + 2 - 2 * |C intersect {u, v}| >= k
    # 2 - 2 * |C intersect {u, v}| >= 0
    # 1 >= |C intersect {u, v}|
    # This means for every pattern {u, v}, the cluster C cannot contain both u and v.
    # This is exactly the Independent Set problem: find an independent set of size k.
    
    # Since n <= 46, we can use a randomized approach or backtracking.
    # Given the constraints and the nature of the problem, a simple randomized search
    # or a greedy approach with backtracking is effective.
    
    adj = [0] * n
    for u, v in patterns:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    def is_independent(subset_mask):
        for i in range(n):
            if (subset_mask >> i) & 1:
                if adj[i] & subset_mask:
                    return False
        return True

    # Try randomized search
    sensors = list(range(n))
    for _ in range(200000):
        subset = random.sample(sensors, k)
        mask = 0
        for s in subset:
            mask |= (1 << s)
        
        if is_independent(mask):
            print("YES")
            print(*(s + 1 for s in subset))
            return

    # If random search fails, try a simple backtracking for small k or specific cases
    def backtrack(idx, current_mask, count):
        if count == k:
            print("YES")
            res = []
            for i in range(n):
                if (current_mask >> i) & 1:
                    res.append(i + 1)
            print(*(res))
            return True
        if idx == n or (n - idx) < (k - count):
            return False
        
        # Try including idx
        if not (adj[idx] & current_mask):
            if backtrack(idx + 1, current_mask | (1 << idx), count + 1):
                return True
        
        # Try excluding idx
        if backtrack(idx + 1, current_mask, count):
            return True
        return False

    if backtrack(0, 0, 0):
        return
    
    print("NO")

solve()
