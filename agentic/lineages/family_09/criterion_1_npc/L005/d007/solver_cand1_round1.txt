import sys
from itertools import combinations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Let S be the set of selected items.
    # T = sum_{i in S} deg(i)
    # For each edge (u, v):
    # c = (1 if u in S else 0) + (1 if v in S else 0)
    # h = c ^ 2
    # Target: sum_{edges} (c ^ 2) = 2m + T
    #
    # Let x_i = 1 if i in S else 0.
    # T = sum_{i=1}^n x_i * deg(i)
    # c = x_u + x_v
    # h = (x_u + x_v) ^ 2
    # If c=0, h=2. If c=1, h=3. If c=2, h=0.
    # sum h = sum_{edges} ( (x_u + x_v == 0)*2 + (x_u + x_v == 1)*3 + (x_u + x_v == 2)*0 )
    # sum h = sum_{edges} ( 2(1-x_u)(1-x_v) + 3(x_u + x_v - 2x_u x_v) )
    # sum h = sum_{edges} ( 2 - 2x_u - 2x_v + 2x_u x_v + 3x_u + 3x_v - 6x_u x_v )
    # sum h = sum_{edges} ( 2 + x_u + x_v - 4x_u x_v )
    # sum h = 2m + sum_{edges} x_u + sum_{edges} x_v - 4 * (number of edges with both u, v in S)
    # sum h = 2m + sum_{i in S} deg(i) - 4 * (number of edges with both u, v in S)
    # Target: 2m + T = 2m + sum_{i in S} deg(i)
    # This implies 4 * (number of edges with both u, v in S) = 0
    # So, no edge can have both endpoints in S. S must be an independent set.

    def check(subset):
        s_set = set(subset)
        for u, v in edges:
            if u in s_set and v in s_set:
                return False
        return True

    if n <= 24:
        for combo in combinations(range(n), k):
            if check(combo):
                print("YES")
                print(*(i + 1 for i in combo))
                return
    else:
        # Meet-in-the-middle for n up to 46
        n1 = n // 2
        n2 = n - n1
        
        adj = [[] for _ in range(n)]
        for u, v in edges:
            adj[u].append(v)
            adj[v].append(u)
            
        def get_independent_sets(nodes):
            res = {}
            size = len(nodes)
            for i in range(1 << size):
                subset = [nodes[j] for j in range(size) if (i & (1 << j))]
                valid = True
                for u_idx in range(len(subset)):
                    for v_idx in range(u_idx + 1, len(subset)):
                        if subset[v_idx] in adj[subset[u_idx]]:
                            valid = False; break
                    if not valid: break
                if valid:
                    res[len(subset)] = res.get(len(subset), []) + [set(subset)]
            return res

        left_sets = get_independent_sets(list(range(n1)))
        right_sets = get_independent_sets(list(range(n1, n)))
        
        for k1 in range(k + 1):
            k2 = k - k1
            if k1 in left_sets and k2 in right_sets:
                for s1 in left_sets[k1]:
                    for s2 in right_sets[k2]:
                        valid = True
                        for u in s1:
                            for v in adj[u]:
                                if v in s2:
                                    valid = False; break
                            if not valid: break
                        if valid:
                            print("YES")
                            print(*(i + 1 for i in list(s1) + list(s2)))
                            return
    print("NO")

solve()
