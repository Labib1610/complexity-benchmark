import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
        
    # We need to find an independent set of size k in the graph of grievances.
    # The problem is equivalent to finding an independent set of size k in the 
    # graph where edges represent grievances.
    # Since n <= 46, we can use a meet-in-the-middle approach or a simple 
    # backtracking with pruning. Given the 5s time limit, backtracking with 
    # bitmasking or Bron-Kerbosch style pruning is efficient.
    
    adj = [0] * n
    for u, v in edges:
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    # We want to find a subset S of size k such that for all u, v in S, 
    # (u, v) is not an edge. This is exactly the Independent Set problem.
    
    def find_is(candidates, current_size):
        if current_size == k:
            return []
        if not candidates or current_size + bin(candidates).count('1') < k:
            return None
        
        # Heuristic: pick vertex with highest degree in the remaining subgraph
        v = (candidates & -candidates).bit_length() - 1
        
        # Try including v
        res = find_is(candidates & ~adj[v] & ~(1 << v), current_size + 1)
        if res is not None:
            return [v + 1] + res
            
        # Try excluding v
        res = find_is(candidates & ~(1 << v), current_size)
        if res is not None:
            return res
        
        return None

    result = find_is((1 << n) - 1, 0)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
