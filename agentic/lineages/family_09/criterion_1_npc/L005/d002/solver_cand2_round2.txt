import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # Meet-in-the-middle for Maximum Independent Set of size k
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for both halves
    def get_isets(nodes, offset):
        size = len(nodes)
        isets = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            mask = 0
            valid = True
            count = 0
            for j in range(size):
                if (i >> j) & 1:
                    if adj[nodes[j]] & mask:
                        valid = False
                        break
                    mask |= (1 << nodes[j])
                    count += 1
            if valid:
                isets[count].append((mask, i << offset))
        return isets

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    isets1 = get_isets(nodes1, 0)
    isets2 = get_isets(nodes2, n1)
    
    # Check combinations
    for c1 in range(min(k + 1, n1 + 1)):
        c2 = k - c1
        if c2 < 0 or c2 > n2:
            continue
        
        for m1, raw1 in isets1[c1]:
            # Find nodes in second half compatible with m1
            forbidden = 0
            for i in range(n1):
                if (m1 >> i) & 1:
                    forbidden |= adj[i]
            
            for m2, raw2 in isets2[c2]:
                if not (m2 & forbidden):
                    # Found it
                    res = []
                    combined = raw1 | raw2
                    for i in range(n):
                        if (combined >> i) & 1:
                            res.append(i + 1)
                    print("YES")
                    print(*(res))
                    return

    print("NO")

solve()
