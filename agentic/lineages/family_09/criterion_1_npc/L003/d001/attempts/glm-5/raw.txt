```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)
    
    # Greedy selection: pick vertex with smallest degree first
    vertices = list(range(n))
    vertices.sort(key=lambda x: len(adj[x]))
    
    selected = []
    available = [True] * n
    
    for u in vertices:
        if len(selected) == k:
            break
        if available[u]:
            # Check if u is compatible with all already selected vertices
            ok = True
            for v in selected:
                if v in adj[u]:
                    ok = False
                    break
            if ok:
                selected.append(u)
                # Mark neighbors as unavailable
                for v in adj[u]:
                    available[v] = False
    
    if len(selected) == k:
        print("YES")
        print(' '.join(str(x + 1) for x in selected))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
