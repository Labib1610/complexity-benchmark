import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # A configuration is valid if no two pins with elevated voltage are connected.
    # This is equivalent to finding an independent set of size k in the graph.
    # Since n <= 46, we can use a backtracking approach with pruning.
    
    nodes = list(range(1, n + 1))
    # Sort nodes by degree to improve pruning
    nodes.sort(key=lambda x: len(adj[x]))
    
    selected = []
    
    def backtrack(index, current_k, forbidden_mask):
        if current_k == k:
            return True
        if index == n or (n - index) < (k - current_k):
            return False
        
        # Try including nodes[index]
        u = nodes[index]
        if not (forbidden_mask & (1 << u)):
            new_forbidden = forbidden_mask
            for neighbor in adj[u]:
                new_forbidden |= (1 << neighbor)
            
            selected.append(u)
            if backtrack(index + 1, current_k + 1, new_forbidden):
                return True
            selected.pop()
            
        # Try excluding nodes[index]
        if backtrack(index + 1, current_k, forbidden_mask):
            return True
            
        return False

    if backtrack(0, 0, 0):
        print("YES")
        print(*(sorted(selected)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
