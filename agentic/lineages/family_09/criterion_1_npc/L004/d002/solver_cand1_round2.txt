import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1] |= (1 << (v-1))
        adj[v-1] |= (1 << (u-1))

    n1 = n // 2
    n2 = n - n1
    
    # Map: (count of set bits, bitmask) -> True/False (is independent set)
    # We need to find two masks m1 (size i) and m2 (size k-i)
    # such that m1 is independent, m2 is independent, and no edge between m1 and m2.
    
    def get_independent_sets(nodes):
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            mask = 0
            cnt = 0
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    if adj[nodes[j]] & mask:
                        valid = False
                        break
                    mask |= (1 << nodes[j])
                    cnt += 1
            if valid:
                res[cnt].append(mask)
        return res

    sets1 = get_independent_sets(list(range(n1)))
    sets2 = get_independent_sets(list(range(n1, n)))

    for i in range(max(0, k - n2), min(k, n1) + 1):
        j = k - i
        for m1 in sets1[i]:
            # Pre-filter m2s that don't conflict with m1
            forbidden = 0
            for bit in range(n1):
                if (m1 >> bit) & 1:
                    forbidden |= adj[bit]
            
            for m2 in sets2[j]:
                if not (m2 & forbidden):
                    print("YES")
                    res = []
                    for b in range(n):
                        if ((m1 | m2) >> b) & 1:
                            res.append(str(b + 1))
                    print(" ".join(res))
                    return

    print("NO")

solve()
