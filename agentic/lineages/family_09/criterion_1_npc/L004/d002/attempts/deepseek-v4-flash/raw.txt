We need to find an independent set of size exactly `k` in a graph with `n ≤ 46` vertices.  
A direct brute force over all subsets is impossible, but we can use a meet-in-the-middle approach.

We split the vertices into two halves, compute all independent sets in each half, and then combine them.

```python
import sys
from collections import defaultdict

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    # Split into two halves
    left_n = n // 2
    right_n = n - left_n
    
    left_adj = [0] * left_n
    right_adj = [0] * right_n
    for i in range(left_n):
        for j in range(left_n):
            if (adj[i] >> j) & 1:
                left_adj[i] |= (1 << j)
    for i in range(right_n):
        for j in range(right_n):
            if (adj[left_n + i] >> (left_n + j)) & 1:
                right_adj[i] |= (1 << j)
    
    # Generate all independent sets for left half
    left_sets = defaultdict(list)  # size -> list of (mask, vertices)
    for mask in range(1 << left_n):
        ok = True
        for i in range(left_n):
            if (mask >> i) & 1:
                if (mask & left_adj[i]) != 0:
                    ok = False
                    break
        if ok:
            size = bin(mask).count("1")
            left_sets[size].append(mask)
    
    # Generate all independent sets for right half
    right_sets = defaultdict(list)
    for mask in range(1 << right_n):
        ok = True
        for i in range(right_n):
            if (mask >> i) & 1:
                if (mask & right_adj[i]) != 0:
                    ok = False
                    break
        if ok:
            size = bin(mask).count("1")
            right_sets[size].append(mask)
    
    # For each left set, we need to know which right vertices are forbidden
    # forbidden_right[mask] = bitmask of right vertices adjacent to any vertex in left mask
    forbidden_right = [0] * (1 << left_n)
    for mask in range(1 << left_n):
        f = 0
        for i in range(left_n):
            if (mask >> i) & 1:
                # add right neighbors of i
                for j in range(right_n):
                    if (adj[i] >> (left_n + j)) & 1:
                        f |= (1 << j)
        forbidden_right[mask] = f
    
    # For each right set, we need to know which left vertices are forbidden
    forbidden_left = [0] * (1 << right_n)
    for mask in range(1 << right_n):
        f = 0
        for j in range(right_n):
            if (mask >> j) & 1:
                for i in range(left_n):
                    if (adj[left_n + j] >> i) & 1:
                        f |= (1 << i)
        forbidden_left[mask] = f
    
    # Now combine
    for lsize in left_sets:
        for lmask in left_sets[lsize]:
            need = k - lsize
            if need < 0 or need > right_n:
                continue
            # We need a right independent set of size 'need' that does not intersect forbidden_right[lmask]
            allowed_right_mask = ((1 << right_n) - 1) ^ forbidden_right[lmask]
            # Check if there is a right set of size 'need' that is subset of allowed_right_mask
            # We can precompute for each allowed mask the max size independent set inside it
            # But simpler: iterate over right sets of size 'need' and check subset condition
            for rmask in right_sets.get(need, []):
                if (rmask & forbidden_right[lmask]) == 0:
                    # Found
                    ans = []
                    for i in range(left_n):
                        if (lmask >> i) & 1:
                            ans.append(i + 1)
                    for j in range(right_n):
                        if (rmask >> j) & 1:
                            ans.append(left_n + j + 1)
                    print("YES")
                    print(" ".join(map(str, ans)))
                    return
    
    print("NO")

if __name__ == "__main__":
    solve()
```
