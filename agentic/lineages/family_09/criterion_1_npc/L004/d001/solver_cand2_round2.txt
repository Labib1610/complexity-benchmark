import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle approach for Maximum Independent Set
    # n <= 46 is too large for simple backtracking, but we only need size k.
    # Split into two halves: n1 = n // 2, n2 = n - n1
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the second half
    # subsets2[size] = list of bitmasks of size 'size' that are independent
    subsets2 = [[] for _ in range(n2 + 1)]
    for i in range(1 << n2):
        valid = True
        mask = 0
        count = 0
        for j in range(n2):
            if (i >> j) & 1:
                if (adj[n1 + j] >> n1) & i:
                    valid = False
                    break
                mask |= adj[n1 + j]
                count += 1
        if valid:
            subsets2[count].append((i, mask))
            
    # Search in the first half
    for i in range(1 << n1):
        valid = True
        mask = 0
        count = 0
        for j in range(n1):
            if (i >> j) & 1:
                if adj[j] & i:
                    valid = False
                    break
                mask |= adj[j]
                count += 1
        
        if valid and count <= k:
            needed = k - count
            if needed <= n2:
                # mask contains neighbors of selected nodes in first half
                # We need to pick 'needed' nodes from second half such that
                # they are not in 'mask' and are independent among themselves
                for cand_mask, cand_adj in subsets2[needed]:
                    if not (cand_mask & (mask >> n1)):
                        # Check if any node in cand_mask is connected to nodes in i
                        if not (cand_adj & i):
                            # Found it
                            res = []
                            for j in range(n1):
                                if (i >> j) & 1:
                                    res.append(j + 1)
                            for j in range(n2):
                                if (cand_mask >> j) & 1:
                                    res.append(n1 + j + 1)
                            print("YES")
                            print(*(res))
                            return

    print("NO")

solve()
