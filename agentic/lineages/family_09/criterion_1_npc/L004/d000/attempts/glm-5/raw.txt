import sys

def main():
    input = sys.stdin.read().split()
    if not input:
        return
    it = iter(input)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)
    
    unknown = list(range(n))
    
    while len(unknown) > 2 * k:
        u = unknown[0]
        conflict_count = 0
        for v in unknown[1:]:
            if v in adj[u]:
                conflict_count += 1
                if conflict_count == 2:
                    break
        if conflict_count >= 2:
            unknown.remove(u)
        else:
            v1 = next(v for v in unknown if v != u and v in adj[u])
            unknown.remove(v1)
    
    u_len = len(unknown)
    
    for i in range(1 << u_len):
        if bin(i).count('1') == k:
            group = [unknown[j] for j in range(u_len) if (i >> j) & 1]
            valid = True
            for a in range(k):
                for b in range(a + 1, k):
                    if group[b] in adj[group[a]]:
                        valid = False
                        break
                if not valid:
                    break
            if valid:
                print("YES")
                print(' '.join(str(x + 1) for x in group))
                return
    
    print("NO")

if __name__ == "__main__":
    main()
