import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n+1)
    components = []
    
    for start in range(1, n+1):
        if color[start] == 0:
            comp = [[], []]
            color[start] = 1
            comp[0].append(start)
            q = deque([start])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        comp[color[v]-1].append(v)
                        q.append(v)
            components.append(comp)
    
    total = n
    if k > total:
        print("NO")
        return
    
    dp = [False] * (k+1)
    dp[0] = True
    parent = [-1] * (k+1)
    
    for idx, comp in enumerate(components):
        a = len(comp[0])
        b = len(comp[1])
        for j in range(k, -1, -1):
            if dp[j]:
                if j + a <= k and not dp[j+a]:
                    dp[j+a] = True
                    parent[j+a] = idx
                if j + b <= k and not dp[j+b]:
                    dp[j+b] = True
                    parent[j+b] = idx
    
    if not dp[k]:
        print("NO")
        return
    
    choice = [0] * len(components)
    cur = k
    while cur > 0:
        idx = parent[cur]
        comp = components[idx]
        a = len(comp[0])
        b = len(comp[1])
        if cur >= a and dp[cur-a] and (choice[idx] == 0):
            choice[idx] = 1
            cur -= a
        else:
            choice[idx] = 2
            cur -= b
    
    result = []
    for idx, comp in enumerate(components):
        if choice[idx] == 1:
            result.extend(comp[0])
        else:
            result.extend(comp[1])
    
    print("YES")
    print(' '.join(map(str, result)))

solve()
