```python
import sys

# The problem states that the graph is bipartite because shared volumes only
# exist between a producer and a consumer. Two producers never share a volume,
# and two consumers never share a volume.
# Thus, we can partition the graph into two sets, U and V, such that all edges
# connect a node in U to a node in V.
# We need to find a subset of size k that is an independent set.
# In a bipartite graph, the maximum independent set is n - (size of maximum matching).
# However, we just need to find *any* independent set of size k.
# Since the graph is bipartite, we can color it with 2 colors (0 and 1).
# Let S0 be the set of nodes with color 0, and S1 be the set of nodes with color 1.
# Both S0 and S1 are independent sets.
# If |S0| >= k, we can pick any k nodes from S0.
# If |S1| >= k, we can pick any k nodes from S1.
# If both are < k, is it possible to have an independent set of size k?
# Yes, but the problem asks for *a* selection.
# Actually, in a bipartite graph, any independent set is a subset of the union
# of the two partitions. The largest independent set is at least max(|S0|, |S1|).
# If max(|S0|, |S1|) < k, then no independent set of size k exists.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    s0 = []
    s1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            s0.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            s0.append(v)
                        else:
                            s1.append(v)
                        stack.append((v, color[v]))
    
    if len(s0) >= k:
        print("YES")
        print(*(s0[:k]))
    elif len(s1) >= k:
        print("YES")
        print(*(s1[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
