import sys

# The problem states that the graph contains no odd cycles.
# This means the graph is bipartite.
# A bipartite graph can be partitioned into two sets, say A and B,
# such that all edges connect a vertex in A to a vertex in B.
# We can pick any number of vertices from A and any number from B
# as long as we don't pick two vertices connected by an edge.
# Actually, in a bipartite graph, we can pick any subset of A and any subset of B
# independently, because there are no edges within A and no edges within B.
# Wait, that's not correct. We can pick any subset of A and any subset of B
# such that no edge exists between the chosen vertices.
# Since all edges are between A and B, if we pick a subset S_A from A and S_B from B,
# we must ensure that for every u in S_A and v in S_B, there is no edge (u, v).
# However, the problem is simpler: we just need to find an independent set of size k.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# But we don't need the maximum independent set, just any set of size k.
# Actually, in a bipartite graph, the two sets A and B are themselves independent sets.
# Let |A| = n1 and |B| = n2, where n1 + n2 = n.
# Any subset of A is an independent set, and any subset of B is an independent set.
# So we can pick any size s1 <= n1 and s2 <= n2 such that s1 + s2 = k.
# This is possible if and only if k <= max(n1, n2) is NOT the condition.
# The condition is: can we pick k vertices such that no two are connected?
# Since A is an independent set and B is an independent set,
# we can pick any k vertices from A if k <= |A|, or any k vertices from B if k <= |B|.
# Is it possible to pick k vertices if k > max(|A|, |B|)?
# Yes, if we pick some from A and some from B such that no edges exist between them.
# But the problem is simpler: the union of all vertices in A is an independent set,
# and the union of all vertices in B is an independent set.
# If we pick all vertices in A, we have |A| independent vertices.
# If we pick all vertices in B, we have |B| independent vertices.
# Any subset of A is independent, any subset of B is independent.
# The largest independent set in a bipartite graph is at least max(|A|, |B|).
# Actually, the size of the maximum independent set is n - (size of maximum matching).
# By Konig's theorem, in any bipartite graph, the number of edges in a maximum matching
# equals the number of vertices in a minimum vertex cover.
# Let M be the size of max matching. Max independent set size = n - M.
# We need to check if n - M >= k.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    A, B = [], []
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: A.append(u)
                else: B.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # We need an independent set of size k.
    # The set A is an independent set, B is an independent set.
    # We can pick any subset of A or any subset of B.
    # The maximum independent set size is n - matching_size.
    # But we can just use the fact that we can pick any subset of A or B.
    # Actually, we can pick any subset of A and any subset of B as long as no edges exist.
    # This is equivalent to picking a subset of vertices that doesn't contain any edge.
    # Since the graph is bipartite, we can just pick all of A or all of B.
    # If k <= len(A), we pick k from A. If k <= len(B), we pick k from B.
    # Is it possible that k > len(A) and k > len(B) but a solution exists?
    # Yes, e.g., a graph with no edges, A and B can be any partition.
    # Actually, if we take the partition (A, B), any subset of A is independent,
    # any subset of B is independent. The union of a subset of A and a subset of B
    # is independent if no edges exist between them.
    # This is only possible if we pick vertices such that no edge (u, v) exists.
    # The simplest way: the maximum independent set is at least max(|A|, |B|).
    # Actually, we can just output A if k <= len(A) or B if k <= len(B).
    
    if k <= len(A):
        print("YES")
        print(*(A[:k]))
    elif k <= len(B):
        print("YES")
        print(*(B[:k]))
    else:
        print("NO")

solve()
