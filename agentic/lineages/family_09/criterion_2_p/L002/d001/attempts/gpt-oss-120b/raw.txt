```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [ -1 ] * (n + 1)
    selected = []          # vertices from the larger side of each component
    from collections import deque

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        side = [[], []]      # vertices of color 0 and 1
        side[0].append(start)
        while q:
            u = q.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    side[color[v]].append(v)
                    q.append(v)
                # else: graph is guaranteed bipartite, no need to check conflict
        # choose larger side
        if len(side[0]) >= len(side[1]):
            selected.extend(side[0])
        else:
            selected.extend(side[1])

    if len(selected) < k:
        print("NO")
        return

    print("YES")
    print(' '.join(map(str, selected[:k])))

if __name__ == "__main__":
    solve()
```
