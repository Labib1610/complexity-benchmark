```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    k = int(input_data[idx]); idx += 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx]); idx += 1
        v = int(input_data[idx]); idx += 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    components = []

    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        color[start] = 1
        comp = [start]
        q = deque([start])
        while q:
            u = q.popleft()
            for w in adj[u]:
                if color[w] == 0:
                    color[w] = 3 - color[u]
                    comp.append(w)
                    q.append(w)
        components.append(comp)

    total = 0
    for comp in components:
        total += max(len(comp) // 2, 1)

    if k > total:
        print("NO")
        return

    chosen = []
    for comp in components:
        if k == 0:
            break
        if len(comp) == 1:
            chosen.append(comp[0])
            k -= 1
        else:
            cnt = min((len(comp) + 1) // 2, k)
            for v in comp:
                if color[v] == 1:
                    chosen.append(v)
                    cnt -= 1
                    if cnt == 0:
                        break
            k -= cnt

    print("YES")
    print(' '.join(map(str, chosen)))

solve()
```
