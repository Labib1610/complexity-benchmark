import sys

def solve():
    # The problem states that there exists a non-zero voltage configuration 
    # such that for every edge (u, v), voltage(u) + voltage(v) = 0.
    # This implies that for every connected component, the graph is bipartite.
    # If a graph is bipartite, we can partition the vertices into two sets,
    # say A and B, such that all edges are between A and B.
    # A safe subset is an independent set. In a bipartite graph, the two
    # partitions A and B are independent sets.
    # The problem asks for an independent set of size k.
    # Since the graph is bipartite, we can find the bipartition for each 
    # connected component. Let the partitions be (A_i, B_i) for component i.
    # We want to pick a subset of size k from the union of all A_i and B_i
    # such that no two vertices are connected.
    # Actually, the condition "voltage(u) + voltage(v) = 0" implies that
    # the graph is bipartite. In any bipartite graph, the maximum independent
    # set is at least n/2. However, we need exactly k.
    # Since it's bipartite, we can collect all A_i and B_i sets.
    # Let the sizes of the partitions be (a_1, b_1), (a_2, b_2), ..., (a_c, b_c).
    # We need to choose a subset of these partitions such that the sum of 
    # sizes is k. This is a variation of the subset sum problem.
    # Since we have n components, we can use dynamic programming.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    partitions = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_a, comp_b = [], []
            stack = [(i, 0)]
            color[i] = 0
            comp_a.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0: comp_a.append(v)
                        else: comp_b.append(v)
                        stack.append((v, color[v]))
            partitions.append((comp_a, comp_b))
            
    # dp[i][j] = which partition (0 or 1) was used for component i to get sum j
    dp = {0: []}
    for i, (p0, p1) in enumerate(partitions):
        new_dp = {}
        len0, len1 = len(p0), len(p1)
        for current_sum, path in dp.items():
            if current_sum + len0 <= k:
                new_dp[current_sum + len0] = path + [(i, 0)]
            if current_sum + len1 <= k:
                new_dp[current_sum + len1] = path + [(i, 1)]
        dp = new_dp
        
    if k in dp:
        print("YES")
        res = []
        for comp_idx, part_idx in dp[k]:
            if part_idx == 0:
                res.extend(partitions[comp_idx][0])
            else:
                res.extend(partitions[comp_idx][1])
        print(*(res[:k]))
    else:
        print("NO")

solve()
