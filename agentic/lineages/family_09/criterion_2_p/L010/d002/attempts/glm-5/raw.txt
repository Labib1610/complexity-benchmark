import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    # BFS to 2-color each connected component
    # Since the graph is bipartite (guaranteed by voltage property), this works.
    for start in range(1, n + 1):
        if not visited[start]:
            queue = [start]
            visited[start] = True
            color[start] = 1
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        color[v] = 3 - color[u]
                        queue.append(v)
    
    # Count nodes of each color per component
    # We need to track component membership to count color0/1 counts per component
    comp_id = [0] * (n + 1)
    comp_color_counts = []  # list of (count_color1, count_color2)
    curr_comp = 0
    
    visited2 = [False] * (n + 1)
    for start in range(1, n + 1):
        if not visited2[start]:
            curr_comp += 1
            queue = [start]
            visited2[start] = True
            comp_id[start] = curr_comp
            c1 = 0
            c2 = 0
            if color[start] == 1:
                c1 += 1
            else:
                c2 += 1
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                for v in adj[u]:
                    if not visited2[v]:
                        visited2[v] = True
                        comp_id[v] = curr_comp
                        if color[v] == 1:
                            c1 += 1
                        else:
                            c2 += 1
                        queue.append(v)
            comp_color_counts.append((c1, c2))
    
    # DP to find achievable subset sizes
    # dp[i] = True if size i is achievable
    # We need to track which color choice was made for each component
    num_comp = curr_comp
    # dp[j] = component index used to achieve size j, or -1 if not achievable
    # We'll use a dictionary or list for DP
    # Since n <= 100000, O(n * num_comp) might be too slow if num_comp is large
    # But total work is O(n) because each node contributes to exactly one component
    # We can use a bitset or boolean array approach
    
    # dp[j] stores the component index that was last added to achieve size j
    # We need to reconstruct the path, so we store predecessor info
    # dp[j] = (prev_size, color_choice) where color_choice is 0 or 1 indicating which color was taken
    # Or we can store for each component, which sizes it can contribute
    
    # Let's use a list of sets or a more efficient method
    # Since we need to reconstruct, we'll store for each component and each possible size
    
    # Actually, we can do DP where dp[j] = component index that achieved size j
    # But we need to know which color was chosen for that component
    
    # Let's use: dp[j] = (prev_j, color_chosen) where prev_j is the size before adding this component
    # Initialize dp[0] = (0, 0) meaning size 0 is achievable with no components
    
    INF = -1
    dp = [INF] * (n + 1)
    dp[0] = 0  # size 0 is achievable
    
    # For reconstruction, we need to know for each component, which color was chosen
    # choice[comp][size] = color chosen (1 or 2) or 0 if not used
    # But this is too memory intensive
    
    # Alternative: after DP, we know the final size k is achievable
    # We can backtrack: for each component from last to first, check if we chose color 1 or 2
    
    # We'll store for each component, the sizes it can contribute and the color
    # Actually, let's store for each component, the two possible contributions
    
    # Let's use a different approach:
    # dp[j] = bitmask or list of (component, color) pairs that achieve size j
    # But this is also memory intensive
    
    # Better: store for each component, which sizes it was used for
    # We'll use a 2D array: used[comp][j] = color (1 or 2) if component 'comp' was used with that color to achieve size j
    
    # Since n can be 100000 and num_comp can be up to n, this is O(n^2) memory which is too much
    
    # Alternative: use a dictionary for each component
    # Or, we can do DP and store only the necessary info
    
    # Let's use: for each component, we store the sizes it contributes to
    # We'll use a list of dictionaries
    
    # Actually, the standard approach for subset sum reconstruction:
    # dp[j] = component index that was added to achieve size j
    # We also need to know which color was chosen
    
    # Let's use: dp[j] = (prev_size, color_chosen, comp_idx)
    # where prev_size is the size before adding this component
    
    dp_info = [None] * (n + 1)
    dp_info[0] = (0, 0, 0)  # (prev_size, color_chosen, comp_idx)
    
    for comp_idx in range(1, num_comp + 1):
        c1, c2 = comp_color_counts[comp_idx - 1]
        # We need to process in reverse to avoid using the same component twice
        for j in range(n, -1, -1):
            if dp_info[j] is not None:
                # Try adding color 1
                new_size = j + c1
                if new_size <= n and dp_info[new_size] is None:
                    dp_info[new_size] = (j, 1, comp_idx)
                # Try adding color 2
                new_size = j + c2
                if new_size <= n and dp_info[new_size] is None:
                    dp_info[new_size] = (j, 2, comp_idx)
    
    if dp_info[k] is None:
        print("NO")
        return
    
    print("YES")
    
    # Reconstruct the solution
    # We need to find which color was chosen for each component
    comp_chosen_color = [0] * (num_comp + 1)  # 0 = not chosen, 1 or 2 = chosen color
    
    curr_size = k
    while curr_size > 0:
        prev_size, color_chosen, comp_idx = dp_info[curr_size]
        comp_chosen_color[comp_idx] = color_chosen
        curr_size = prev_size
    
    # Now collect all nodes with the chosen color in their component
    result = []
    for node in range(1, n + 1):
        ci = comp_id[node]
        if comp_chosen_color[ci] != 0:
            if color[node] == comp_chosen_color[ci]:
                result.append(node)
    
    print(' '.join(map(str, result)))

solve()
