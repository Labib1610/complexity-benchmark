```python
import sys

# The problem states that any cycle has an even length.
# This implies the graph is bipartite.
# In a bipartite graph, the vertices can be partitioned into two sets,
# say L and R, such that all edges connect a vertex in L to a vertex in R.
# An all-strangers group is an independent set.
# In a bipartite graph, the largest independent set is the complement of the
# minimum vertex cover. By Konig's theorem, in any bipartite graph,
# the number of edges in a maximum matching equals the number of vertices
# in a minimum vertex cover.
# Thus, the size of the maximum independent set is n - (size of max matching).
# Since the graph is bipartite, we can color it with 2 colors (0 and 1).
# Let V0 be the set of vertices with color 0, and V1 be the set with color 1.
# Any subset of V0 is an independent set, and any subset of V1 is an independent set.
# The maximum independent set size is max(|V0|, |V1|).
# Wait, that's only if the graph is connected.
# For a general bipartite graph, the maximum independent set is the union of
# the larger color partition from each connected component.
# Let the components be C_1, C_2, ..., C_p.
# For each component C_i, let L_i and R_i be the two color sets.
# We can pick either L_i or R_i from each component to form an independent set.
# We want to pick a subset of size k. This is a variation of the subset sum problem.
# Since we have many components, we can use dynamic programming.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_nodes = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp_nodes[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append((len(comp_nodes[0]), len(comp_nodes[1]), comp_nodes[0], comp_nodes[1]))
            
    # dp[i][j] = which choice (0 or 1) was made for component i to get sum j
    # To reconstruct, store the choice.
    dp = {0: []}
    for i, (s0, s1, nodes0, nodes1) in enumerate(components):
        new_dp = {}
        for current_sum, path in dp.items():
            if current_sum + s0 <= k:
                new_dp[current_sum + s0] = path + [(0, nodes0)]
            if current_sum + s1 <= k:
                new_dp[current_sum + s1] = path + [(1, nodes1)]
        dp = new_dp
        
    if k not in dp:
        print("NO")
    else:
        print("YES")
        res = []
        for choice, nodes in dp[k]:
            res.extend(nodes)
        print(*(res))

solve()
```
