```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)
    comp_vertices = []          # list of (list0, list1) per component
    total_max = 0

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        q = deque([start])
        color[start] = 0
        part = [[], []]
        part[0].append(start)
        while q:
            u = q.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    part[color[v]].append(v)
                    q.append(v)
                # graph is guaranteed bipartite, no need to check conflict
        comp_vertices.append(part)
        total_max += max(len(part[0]), len(part[1]))

    if total_max < k:
        print("NO")
        return

    ans = []
    for part0, part1 in comp_vertices:
        # choose the larger side
        if len(part0) >= len(part1):
            chosen = part0
        else:
            chosen = part1
        need = k - len(ans)
        if need <= 0:
            break
        if len(chosen) <= need:
            ans.extend(chosen)
        else:
            ans.extend(chosen[:need])
            break

    # ans size is exactly k
    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
```
