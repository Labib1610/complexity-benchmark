import sys

# The problem states that there exists a non-zero voltage assignment such that
# for every edge (u, v), V_u + V_v = 0.
# This implies V_u = -V_v.
# If we have a path u -> v -> w, then V_v = -V_u and V_w = -V_v = V_u.
# This means the graph must be bipartite.
# In a bipartite graph, the vertices can be partitioned into two sets,
# say S1 and S2, such that all edges connect a vertex in S1 to a vertex in S2.
# Any subset of S1 is an independent set (stable group), and any subset of S2
# is an independent set.
# To find a stable group of size k, we check if |S1| >= k or |S2| >= k.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    s1 = []
    s2 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            s1.append(i)
            while stack:
                u, c = stack.pop()
                next_c = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = next_c
                        if next_c == 0:
                            s1.append(v)
                        else:
                            s2.append(v)
                        stack.append((v, next_c))
    
    if len(s1) >= k:
        print("YES")
        print(*(s1[:k]))
    elif len(s2) >= k:
        print("YES")
        print(*(s2[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
