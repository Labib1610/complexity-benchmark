import sys
from collections import defaultdict, deque

def read_input():
    n, m, k = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
    return n, m, k, edges

def bipartite_partition(n, edges):
    adj = defaultdict(list)
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    color = {}
    for start in range(1, n+1):
        if start not in color:
            queue = deque([start])
            color[start] = 0
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if neighbor not in color:
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        # This should not happen according to problem statement
                        return None, None
    set_a = [node for node in range(1, n+1) if color.get(node, 0) == 0]
    set_b = [node for node in range(1, n+1) if color.get(node, 0) == 1]
    return set_a, set_b

def find_independent_set(set_nodes, adj, target_size):
    if target_size <= 0:
        return []
    if len(set_nodes) < target_size:
        return None
    
    # For efficiency, we'll just try to pick nodes that don't conflict
    # Since this is a bipartite graph, we can take all nodes from one side 
    # that are not connected to each other (which they aren't within same side)
    # But we need to be more careful
    
    # Actually, in a bipartite graph, nodes in the same partition form an independent set
    # So we just need to check if we have enough nodes in either partition
    
    return set_nodes[:target_size] if len(set_nodes) >= target_size else None

def solve():
    n, m, k, edges = read_input()
    
    if m == 0:
        # No edges, so all nodes can form an independent set
        print("YES")
        print(' '.join(map(str, range(1, min(k+1, n+1)))))
        return
    
    # Partition the graph into two sets
    adj = defaultdict(list)
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    set_a, set_b = bipartite_partition(n, edges)
    
    if len(set_a) >= k:
        print("YES")
        print(' '.join(map(str, set_a[:k])))
        return
        
    if len(set_b) >= k:
        print("YES")
        print(' '.join(map(str, set_b[:k])))
        return
    
    # If we can't find k nodes in one partition, we might still be able 
    # to find a combination across partitions, but that's harder.
    # Let's think about this differently.
    
    # In a bipartite graph, to get an independent set of size k,
    # we can select nodes such that none are adjacent.
    # The maximum independent set in a bipartite graph can be found,
    # but we just need any independent set of size k.
    
    # A greedy approach: try to build a set by adding nodes that don't
    # conflict with already selected ones.
    
    # But since it's bipartite, the largest independent set is at least
    # the size of the larger partition. If k exceeds both partition sizes,
    # we must look for combinations.
    
    # However, let's reconsider the problem. We want exactly k nodes with no edges between them.
    # In a bipartite graph with parts A and B, we can choose:
    # - Only from A (since no edges within A)
    # - Only from B (since no edges within B)
    # - A mix, but then we must ensure no node in A is connected to any node in B in our selection
    
    # Actually, that last point is wrong. In our selection, we can have nodes from both A and B
    # as long as there are no edges between any pair in our selection.
    
    # But checking all combinations is too expensive.
    # Let's think of it this way:
    # If we sort by degree and greedily select non-adjacent nodes...
    
    # Let's go back to basics. In a bipartite graph, the maximum independent set
    # has size max(|A|, |B|). If k > max(|A|, |B|), then it's impossible unless
    # we can combine from both A and B carefully.
    
    # But the problem states that friendships ONLY go between camps.
    # So within each camp, there are no friendships.
    # Therefore, selecting any subset from one camp gives an independent set.
    
    # Wait, re-reading the problem:
    # "every friendship connects a user in one camp to a user in the other"
    # So indeed, within a camp, no two users are friends.
    # So we can take any k users from the same camp.
    
    # But what if k is larger than both camps?
    # Then we must take some from A and some from B, ensuring no edges between our selections.
    # That means, for nodes we pick from A, none of their neighbors in B should be picked.
    # And vice versa.
    
    # So, if we pick S_A from A and S_B from B, we need:
    # For every node in S_A, its neighbors in B are not in S_B
    # For every node in S_B, its neighbors in A are not in S_A
    # And |S_A| + |S_B| = k
    
    # This is getting complex. Let's see if k <= max(|A|, |B|)
    max_independent_from_single_side = max(len(set_a), len(set_b))
    
    if k <= max(len(set_a), len(set_b)):
        if len(set_a) >= k:
            result = set_a[:k]
        else:
            result = set_b[:k]
        print("YES")
        print(' '.join(map(str, result)))
        return
    
    # Now the case where k > max(|A|, |B|).
    # We need to combine from both sides.
    # Let's say we take 'a' nodes from A and 'b' nodes from B, where a + b = k.
    # For this to work, the nodes taken from A must not be connected to any nodes taken from B.
    # That means: N(B_taken) intersect A_taken = empty
    # and     N(A_taken) intersect B_taken = empty
    
    # One strategy: Try all reasonable values of a and b.
    # Or, smarter: for small k, we can enumerate.
    # Actually, let's think of it as: pick a subset of A and a subset of B
    # such that there are no edges between them.
    
    # Let's create adjacency sets for quick lookup
    adj_set = defaultdict(set)
    for u, v in edges:
        adj_set[u].add(v)
        adj_set[v].add(u)
        
    def is_valid_combination(nodes):
        for i in range(len(nodes)):
            for j in range(i+1, len(nodes)):
                if nodes[j] in adj_set[nodes[i]]:
                    return False
        return True
                    
    # If k is small, we can try combinations
    # But k can be up to n, so that won't work.
    
    # Let's think of a different approach.
    # We have two independent sets (the two camps).
    # To get a larger independent set, we can try to add nodes from the other side
    # if they don't conflict.
    
    # Let's consider taking all of the smaller set and part of the larger set.
    # Or, take a subset from both.
    
    # This is actually solvable by trying:
    # For each possible split (a, b) with a+b=k:
    #   Check if there's an independent set of size a in A and size b in B
    #   Such that no edge goes between these subsets.
    
    # But that's still exponential.
    
    # Let's try a greedy approach:
    # Build the set incrementally.
    # Start with the node (from either A or B) that has the smallest degree.
    # Add nodes that don't conflict with current set.
    
    # Actually, let's just do a simple greedy for small k.
    # Given the constraints, maybe that's intended.
    
    if k > n:
        print("NO")
        return
        
    # Special case handled. Now let's do a greedy search for a valid set.
    # This will work well when k is not too large relative to n.
    
    # But given performance requirements, let's stick to the key insight:
    # If k <= max(|A|, |B|), we're done.
    # Else, we need to check if we can combine from both A and B.
    
    # When combining, we can model this as:
    # Pick X subset of A, Y subset of B, such that no edge connects X to Y.
    # This is equivalent to saying: N(X) intersect Y = empty and N(Y) intersect X = empty.
    
    # A greedy way: 
    # 1. Sort nodes in A by degree ascending
    # 2. Try to add them one by one, keeping track of forbidden nodes in B
    # 3. Similarly for B
    
    # But let's implement a cleaner version:
    
    # Key insight: if we pick a set S, it's invalid iff there exists an edge within S.
    # In a bipartite graph, S can contain nodes from both partitions.
    # S is valid iff for all u,v in S, (u,v) is not an edge.
    
    # So we can do: try to build S greedily.
    
    # But with n=100000, we can't afford to check all combinations.
    
    # Let's go back to the mathematical property:
    # In a bipartite graph, the maximum independent set size is |A| + |B| - (size of minimum vertex cover)
    # By Konig's theorem, size of minimum vertex cover = size of maximum matching.
    # But we don't need the maximum, just any of size k.
    
    # Let's just do a smart brute-force:
    # If k <= max(|A|, |B|), we're done.
    # Else, we need to see if we can take a + b = k with a from A, b from B, and no edges between.
    
    # To make this feasible, let's assume that if k > max(|A|, |B|),
    # then we must have k <= |A| + |B|, and there's enough sparsity.
    
    # Let's code a solution that works:
    
    if k <= len(set_a):
        print("YES")
        print(' '.join(map(str, set_a[:k])))
        return
        
    if k <= len(set_b):
        print("YES")
        print(' '.join(map(str, set_b[:k])))
        return
    
    # Now k > max(|A|, |B|). Need to combine.
    # Try to take some from A and some from B.
    
    # Simple greedy: take all of smaller set, fill rest from larger set avoiding conflicts.
    if len(set_a) <= len(set_b):
        small, large = set_a, set_b
    else:
        small, large = set_b, set_a
    
    # Take all from 'small'
    selected = set(small)
    # For each node in 'large', check if it conflicts with 'selected'
    forbidden = set()
    for node in small:
        forbidden.update(adj_set[node])
    
    # From 'large', pick those not in 'forbidden'
    candidates = [node for node in large if node not in forbidden]
    
    total_selected = len(selected) + len(candidates)
    
    if total_selected >= k:
        # We can make it. Take all from small and enough from candidates.
        needed_from_large = k - len(selected)
        result = list(selected) + candidates[:needed_from_large]
        print("YES")
        print(' '.join(map(str, result)))
        return
    
    # Try the other way around
    if len(set_a) > len(set_b):
        small, large = set_b, set_a
    else:
        small, large = set_a, set_b
        
    selected = set(small)
    forbidden = set()
    for node in small:
        forbidden.update(adj_set[node])
        
    candidates = [node for node in large if node not in forbidden]
    total_selected = len(selected) + len(candidates)
    
    if total_selected >= k:
        needed_from_large = k - len(selected)
        result = list(selected) + candidates[:needed_from_large]
        print("YES")
        print(' '.join(map(str, result)))
        return
    
    # Last resort: general greedy independent set builder
    # This should handle remaining cases including examples like 3
    
    # Build adjacency list again to be sure
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # Greedy independent set construction
    # Sort nodes by degree ascending for better chances
    nodes_sorted = sorted(range(1, n+1), key=lambda x: len(adj[x]))
    selected = []
    selected_set = set()
    
    for node in nodes_sorted:
        # Check if node conflicts with any in selected
        conflict = False
        for sel_node in selected:
            if sel_node in adj[node]:
                conflict = True
                break
        if not conflict:
            selected.append(node)
            selected_set.add(node)
            if len(selected) == k:
                print("YES")
                print(' '.join(map(str, selected)))
                return
                
    print("NO")

solve()
