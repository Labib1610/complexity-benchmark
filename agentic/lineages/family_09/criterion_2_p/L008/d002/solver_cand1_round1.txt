```python
import sys

def solve():
    # The problem states that the graph is bipartite because it can be 
    # partitioned into transmitters and receivers.
    # In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
    # However, the problem is simpler: the graph is bipartite, so we can 
    # find the two sets of the bipartition (let's call them S1 and S2).
    # Any subset of S1 is an independent set, and any subset of S2 is an independent set.
    # The maximum independent set size is max(|S1|, |S2|).
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    s1 = []
    s2 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            s1.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            s1.append(v)
                        else:
                            s2.append(v)
                        stack.append((v, color[v]))
    
    if len(s1) >= k:
        print("YES")
        print(*(s1[:k]))
    elif len(s2) >= k:
        print("YES")
        print(*(s2[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
