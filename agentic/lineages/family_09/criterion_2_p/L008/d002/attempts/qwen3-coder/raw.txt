```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    index = 3
    edges = []
    adj = defaultdict(list)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # The key insight from the problem's constraint is that the graph is bipartite.
    # This means we can color it with two colors such that no two adjacent nodes have the same color.
    # In a bipartite graph, one color class forms an independent set.
    
    # Let's build the graph and check if it's bipartite, and find the larger independent set.
    color = {}
    visited = [False] * (n + 1)
    
    # We'll use BFS to color the graph
    from collections import deque
    
    set1 = []
    set2 = []
    
    def bfs(start):
        queue = deque([start])
        color[start] = 0
        set1.append(start) if color[start] == 0 else set2.append(start)
        visited[start] = True
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            
            for neighbor in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = 1 - current_color
                    if color[neighbor] == 0:
                        set1.append(neighbor)
                    else:
                        set2.append(neighbor)
                    queue.append(neighbor)
                elif color[neighbor] == current_color:
                    # This should not happen according to problem statement
                    return False
        return True
    
    # Process all components
    for i in range(1, n + 1):
        if not visited[i]:
            if not bfs(i):
                # According to problem statement, this shouldn't happen
                pass
    
    # Choose the larger set
    if len(set1) >= k:
        print("YES")
        print(' '.join(map(str, set1[:k])))
    elif len(set2) >= k:
        print("YES")
        print(' '.join(map(str, set2[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
