We can solve this by noticing the graph is bipartite.  
A mutually disconnected set means we can only pick elements from the same side of the bipartition.  
So the largest such set is the size of the larger side.  
If `k` is larger than that, answer is `NO`.  
Otherwise, we can pick any `k` elements from that side.

```python
import sys
from collections import deque

def main():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * n
    side0 = []
    side1 = []
    
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            side0.append(i)
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        if color[v] == 0:
                            side0.append(v)
                        else:
                            side1.append(v)
                        q.append(v)
    
    if len(side0) >= k:
        chosen = side0[:k]
    elif len(side1) >= k:
        chosen = side1[:k]
    else:
        print("NO")
        return
    
    print("YES")
    print(" ".join(str(x + 1) for x in chosen))

if __name__ == "__main__":
    main()
```
