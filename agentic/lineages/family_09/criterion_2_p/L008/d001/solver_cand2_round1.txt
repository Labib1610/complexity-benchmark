```python
import sys

# The problem states that any cycle has an even length.
# This is the definition of a bipartite graph.
# A mutually disconnected collection of elements is an independent set.
# In a bipartite graph, the vertices can be partitioned into two sets,
# say L and R, such that all edges connect a vertex in L to a vertex in R.
# An independent set can be formed by taking all vertices in L or all vertices in R.
# However, the problem asks for a collection of size k.
# Since the graph is bipartite, we can find the two partitions L and R.
# Any subset of L is an independent set, and any subset of R is an independent set.
# Thus, if |L| >= k, we can pick k elements from L.
# If |R| >= k, we can pick k elements from R.
# If both |L| < k and |R| < k, then no independent set of size k exists.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    set_l = []
    set_r = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            set_l.append(i)
            while stack:
                u, c = stack.pop()
                next_c = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = next_c
                        if next_c == 0:
                            set_l.append(v)
                        else:
                            set_r.append(v)
                        stack.append((v, next_c))
    
    if len(set_l) >= k:
        print("YES")
        print(*(set_l[:k]))
    elif len(set_r) >= k:
        print("YES")
        print(*(set_r[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
