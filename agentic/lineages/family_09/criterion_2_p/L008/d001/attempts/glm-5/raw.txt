import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    sizes = [0, 0]
    order = [1, 2]

    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            sizes[1] += 1
            stack = [start]
            while stack:
                u = stack.pop()
                cu = color[u]
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - cu
                        sizes[3 - cu] += 1
                        stack.append(v)

    if sizes[1] >= sizes[2]:
        target = 1
    else:
        target = 2
        order = [2, 1]

    if sizes[target] >= k:
        out = []
        for u in range(1, n + 1):
            if color[u] == target:
                out.append(str(u))
                if len(out) == k:
                    break
        sys.stdout.write("YES\n" + " ".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")

solve()
