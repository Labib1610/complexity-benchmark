```python
import sys
from collections import defaultdict, deque

def read_input():
    n, m, k = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u-1, v-1))  # Convert to 0-indexed
    return n, m, k, edges

def build_graph(n, edges):
    graph = [[] for _ in range(n)]
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    return graph

def bfs_bipartite_check(graph, n):
    """
    Check if the graph is bipartite and return the partition if it is.
    Returns (is_bipartite, partition0, partition1)
    """
    color = [-1] * n  # -1: unvisited, 0: color0, 1: color1
    partition0 = []
    partition1 = []
    
    for start in range(n):
        if color[start] == -1:  # Unvisited node
            queue = deque([start])
            color[start] = 0
            partition0.append(start)
            
            while queue:
                u = queue.popleft()
                current_color = color[u]
                next_color = 1 - current_color
                
                for v in graph[u]:
                    if color[v] == -1:
                        color[v] = next_color
                        if next_color == 0:
                            partition0.append(v)
                        else:
                            partition1.append(v)
                        queue.append(v)
                    elif color[v] == current_color:
                        return False, [], []  # Not bipartite
    
    return True, partition0, partition1

def solve():
    n, m, k, edges = read_input()
    graph = build_graph(n, edges)
    
    is_bipartite, partition0, partition1 = bfs_bipartite_check(graph, n)
    
    if not is_bipartite:
        print("NO")
        return
    
    # Since the graph is bipartite, we can choose all nodes from one side
    # as they won't have conflicts among themselves
    if len(partition0) >= k:
        selected = [str(x + 1) for x in partition0[:k]]  # Convert back to 1-indexed
        print("YES")
        print(" ".join(selected))
    elif len(partition1) >= k:
        selected = [str(x + 1) for x in partition1[:k]]  # Convert back to 1-indexed
        print("YES")
        print(" ".join(selected))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
