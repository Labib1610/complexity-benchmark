import sys

def solve():
    # The problem states that conflicts only occur between relays with odd Hamming distance.
    # This implies the conflict graph is bipartite.
    # Let the two partitions be A and B.
    # Any subset of A is an independent set, and any subset of B is an independent set.
    # We need to pick k vertices such that no two are connected.
    # Since it's bipartite, we can find the two partitions using BFS/DFS.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    partition0 = []
    partition1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            partition0.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            partition0.append(v)
                        else:
                            partition1.append(v)
                        stack.append((v, color[v]))
    
    # We have two independent sets: partition0 and partition1.
    # We need to pick k vertices.
    # If len(partition0) >= k, we can pick k from partition0.
    # If len(partition1) >= k, we can pick k from partition1.
    # If neither, it's impossible because any independent set in a bipartite graph
    # is a subset of one of the partitions or a mix, but the maximum independent set
    # size is max(|P0|, |P1|).
    
    if len(partition0) >= k:
        print("YES")
        print(*(partition0[:k]))
    elif len(partition1) >= k:
        print("YES")
        print(*(partition1[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
