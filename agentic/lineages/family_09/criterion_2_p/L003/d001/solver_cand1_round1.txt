```python
import sys

# The problem states that conflicts only occur between relays with different parity 
# of phase frequencies. This means the conflict graph is bipartite.
# In a bipartite graph, we can partition the vertices into two sets (say, set A and set B)
# such that all edges connect a vertex in A to a vertex in B.
# Within set A, there are no edges. Within set B, there are no edges.
# We want to pick k vertices such that no two are connected.
# This is equivalent to picking some number of vertices from A and some from B
# such that no edge is violated.
# Actually, since it's bipartite, any independent set is a subset of vertices.
# But we need exactly k.
# Let the connected components be C_1, C_2, ..., C_p.
# Each component C_i is bipartite with partitions (A_i, B_i).
# We can pick either all vertices in A_i or all vertices in B_i (or any subset, 
# but picking the whole partition is always safe).
# Actually, for each component, we can pick any independent set.
# The maximum independent set of a bipartite graph is |V| - |Maximum Matching|.
# However, we need exactly k.
# Since we can pick any subset of A_i or B_i, we can pick any number of vertices 
# from 0 to |A_i| or 0 to |B_i|? No, that's not right.
# We can pick any subset of A_i (size 0 to |A_i|) or any subset of B_i (size 0 to |B_i|).
# Wait, if we pick a subset of A_i, we can't pick any vertex in B_i that is connected to it.
# Actually, for each component, we can pick either a subset of A_i or a subset of B_i.
# This is still not quite right. The simplest way:
# For each component, we can pick all of A_i or all of B_i.
# Let a_i = |A_i| and b_i = |B_i|. We want to choose x_i in {a_i, b_i} such that sum(x_i) = k.
# This is a variation of the subset sum problem (knapsack).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_a, comp_b = [], []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: comp_a.append(u)
                else: comp_b.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append((comp_a, comp_b))
            
    # dp[i][j] = which set (0 for A, 1 for B) was chosen for component i to get sum j
    dp = {0: []}
    for i, (a, b) in enumerate(components):
        new_dp = {}
        len_a, len_b = len(a), len(b)
        for current_sum, path in dp.items():
            if current_sum + len_a <= k:
                new_dp[current_sum + len_a] = path + [(i, 0)]
            if current_sum + len_b <= k:
                new_dp[current_sum + len_b] = path + [(i, 1)]
        dp = new_dp
        
    if k not in dp:
        print("NO")
    else:
        print("YES")
        res = []
        for comp_idx, set_type in dp[k]:
            if set_type == 0:
                res.extend(components[comp_idx][0])
            else:
                res.extend(components[comp_idx][1])
        print(*(res))

solve()
```
