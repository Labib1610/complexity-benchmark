```python
import sys

# The problem states that the conflict graph is bipartite because all cycles are even.
# We need to find a subset of size k that is an independent set.
# In a bipartite graph, we can partition the vertices into two sets, L and R,
# such that all edges connect a vertex in L to a vertex in R.
# Any subset of L is an independent set, and any subset of R is an independent set.
# However, we need exactly k vertices.
# Let the connected components be C_1, C_2, ..., C_p.
# Each component C_i is bipartite with partitions (L_i, R_i).
# We can choose either L_i or R_i from each component to form an independent set.
# Let l_i = |L_i| and r_i = |R_i|. We want to choose a subset of components
# and for each component choose either L_i or R_i such that the total size is k.
# This is a variation of the subset sum problem (or knapsack).
# Since we have n <= 100,000, we use dynamic programming with bitsets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_l = []
            comp_r = []
            stack = [(i, 0)]
            color[i] = 0
            comp_l.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0: comp_l.append(v)
                        else: comp_r.append(v)
                        stack.append((v, 1 - c))
            components.append((comp_l, comp_r))

    # dp[i] stores a way to get sum i. 
    # To reconstruct, we store which choice was made.
    # dp[sum] = (component_index, choice_type)
    # choice_type: 0 for L, 1 for R
    dp = {0: []}
    
    for idx, (L, R) in enumerate(components):
        new_dp = dp.copy()
        l_size, r_size = len(L), len(R)
        for s, path in dp.items():
            if s + l_size <= k and (s + l_size) not in new_dp:
                new_dp[s + l_size] = path + [(idx, 0)]
            if s + r_size <= k and (s + r_size) not in new_dp:
                new_dp[s + r_size] = path + [(idx, 1)]
        dp = new_dp
        
    if k not in dp:
        print("NO")
    else:
        print("YES")
        res = []
        for comp_idx, choice in dp[k]:
            if choice == 0:
                res.extend(components[comp_idx][0])
            else:
                res.extend(components[comp_idx][1])
        print(*(res))

solve()
```
