import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append(comp)

    # dp[i][j] = can we pick j nodes from first i components
    # Since we need to reconstruct, we store which choice was made
    # dp[j] stores the number of nodes from camp 0 in the current sum
    # Actually, for each component, we can pick either len(comp[0]) or len(comp[1]) nodes.
    # Let c0 = len(comp[0]), c1 = len(comp[1]).
    # We need to pick exactly k nodes.
    
    # dp[i][j] = True if j nodes can be picked from first i components
    # To reconstruct, store choice: dp[i][j] = (choice_made, prev_j)
    dp = {0: None}
    
    for idx, comp in enumerate(components):
        c0, c1 = len(comp[0]), len(comp[1])
        new_dp = {}
        for j in dp:
            if j + c0 <= k:
                new_dp[j + c0] = (0, j)
            if j + c1 <= k:
                new_dp[j + c1] = (1, j)
        dp = new_dp
        
    if k not in dp:
        print("NO")
        return

    print("YES")
    res = []
    curr_k = k
    for i in range(len(components) - 1, -1, -1):
        choice, prev_k = dp[curr_k]
        res.extend(components[i][choice])
        curr_k = prev_k
        
    print(*(res))

solve()
