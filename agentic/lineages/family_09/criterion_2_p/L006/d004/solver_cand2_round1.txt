import sys

# The problem states that for any connected component, if we pick a reference gear,
# no engagement point exists between gears at the same distance from the reference.
# This implies that the graph is bipartite.
# Specifically, if we perform a BFS from a reference node, all nodes at distance d
# can only have edges to nodes at distance d-1 or d+1.
# This means the graph is bipartite, where the two sets are:
# Set A: {nodes at even distance from reference}
# Set B: {nodes at odd distance from reference}
# Since the graph is bipartite, the maximum independent set size is n - (size of maximum matching).
# However, the problem asks for exactly k gears.
# In a bipartite graph, we can pick any subset of nodes from one partition,
# but we must ensure no two are connected.
# Actually, the property "no edge between nodes at the same distance" means
# the graph is bipartite. Let the two partitions be V0 and V1.
# Any independent set can be formed by picking some nodes from V0 and some from V1.
# But wait, the property is stronger: it's a collection of bipartite components.
# For each connected component, we have two sets V0 and V1.
# We can pick any number of nodes from V0 and V1 such that no edges are violated.
# Since it's bipartite, we can pick all nodes in V0, or all nodes in V1,
# or any subset of V0 and V1 that doesn't contain an edge.
# Actually, the simplest way to get an independent set in a bipartite graph
# is to pick a subset of V0 and a subset of V1.
# Since there are no edges within V0 and no edges within V1,
# any subset of V0 is an independent set, and any subset of V1 is an independent set.
# The union of a subset of V0 and a subset of V1 is an independent set
# if and only if no edges exist between the chosen nodes.
# Given the structure, we can just use the two partitions V0 and V1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = map(int, input_data[idx:idx+2])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    v0, v1 = [], []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: v0.append(u)
                else: v1.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # We have two independent sets: V0 and V1.
    # We can pick any number of nodes from V0 and any number from V1.
    # But we must ensure no edges between them.
    # Actually, the problem is simpler: the graph is bipartite.
    # Any independent set is a subset of nodes.
    # The maximum independent set size is |V0| + |V1| - (max matching).
    # But we just need to find *any* independent set of size k.
    # Since the graph is bipartite, we can use the fact that we can pick
    # nodes from V0 and V1.
    
    # Let's try to find an independent set of size k.
    # A simple greedy approach:
    # The graph is bipartite. We can pick nodes from V0 and V1.
    # Let's use max flow to find the maximum independent set.
    # Or just use the fact that we can pick any subset of V0 and V1.
    # Actually, the problem is equivalent to finding an independent set of size k.
    # For bipartite graphs, this is still NP-hard in general,
    # but the structure here is special.
    # Wait, the property "no edge between nodes at same distance"
    # means the graph is a collection of bipartite components.
    # Let's just use a simple backtracking or greedy with bitmask if n is small,
    # but n is large. Let's use the property:
    # The max independent set in a bipartite graph is n - max_matching.
    
    import collections
    match = [-1] * (n + 1)
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    matching_size = 0
    for u in v0:
        visited = [False] * (n + 1)
        if dfs(u, visited):
            matching_size += 1
            
    # Max independent set size is n - matching_size.
    # If k > n - matching_size, output NO.
    if k > n - matching_size:
        print("NO")
        return

    # To get an independent set of size k:
    # This is a standard construction:
    # 1. Find max matching.
    # 2. Min vertex cover = (V0 \ U) U (V1 \cap N(U)) where U is reachable from unmatched V0.
    # 3. Max independent set = V \ Min vertex cover.
    
    unmatched_v0 = [u for u in v0 if not any(match[v] == u for v in adj[u])]
    # This is not quite right. Let's use the standard algorithm.
    # The set of nodes in the max independent set:
    # Let M be the matching.
    # Z = set of vertices reachable from unmatched nodes in V0 via alternating paths.
    # MIS = (V0 \ Z) U (V1 \cap Z).
    
    z = set()
    def dfs_z(u):
        z.add(u)
        for v in adj[u]:
            if color[u] == 0: # u in V0, edge is not in matching
                if v not in z and match[u] != v:
                    dfs_z(v)
            else: # u in V1, edge is in matching
                if v not in z and match[v] == u:
                    dfs_z(v)
                    
    for u in v0:
        if all(match[v] != u for v in adj[u]): # unmatched
            dfs_z(u)
            
    mis = [u for u in v0 if u not in z] + [u for u in v1 if u in z]
    print("YES")
    print(*(mis[:k]))

solve()
