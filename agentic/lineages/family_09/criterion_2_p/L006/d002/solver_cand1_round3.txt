```python
import sys

# The problem states that any cycle has an even number of edges.
# This is the definition of a bipartite graph.
# In a bipartite graph, we can partition the vertices into two sets,
# say S0 and S1, such that all edges connect a vertex in S0 to one in S1.
# We need to pick k vertices such that no two are connected.
# This means we can pick any number of vertices from S0 and any number from S1,
# provided we don't pick two connected vertices.
# Actually, the condition "no two selected gears are connected" means
# we can pick any subset of S0 and any subset of S1 independently.
# Let |S0| = c0 and |S1| = c1. We can pick any i vertices from S0 (0 <= i <= c0)
# and any j vertices from S1 (0 <= j <= c1) such that i + j = k.
# This is possible if and only if min(c0, c1) <= k <= max(c0, c1) is NOT necessarily true.
# Actually, we can pick any number of vertices from S0 and S1 as long as
# the total count is k. Since there are no edges within S0 and no edges within S1,
# we can pick any subset of S0 and any subset of S1.
# The total number of available vertices is n = c0 + c1.
# We can pick any number of vertices from S0 (up to c0) and any from S1 (up to c1).
# So we can pick any k such that 0 <= k <= n.
# Wait, the condition is: we can pick any subset of S0 and any subset of S1.
# The total number of vertices we can pick is any k such that 0 <= k <= n.
# Is it always possible to pick k? Yes, because we can pick i from S0 and j from S1
# such that i+j=k, provided 0 <= i <= c0 and 0 <= j <= c1.
# This is possible if and only if max(0, k - c1) <= i <= min(k, c0).
# Such an i exists if max(0, k - c1) <= min(k, c0).
# This is equivalent to:
# 1) 0 <= k
# 2) k - c1 <= k (always true since c1 >= 0)
# 3) 0 <= c0 (always true)
# 4) k - c1 <= c0 => k <= c0 + c1 = n.
# So yes, any k from 0 to n is possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    s0, s1 = [], []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: s0.append(u)
                else: s1.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # We need to pick k vertices. We have |s0| and |s1|.
    # We need i from s0 and j from s1 such that i+j=k, 0<=i<=|s0|, 0<=j<=|s1|.
    # Let i = max(0, k - len(s1)). Then j = k - i.
    # Check if j <= len(s1).
    i = max(0, k - len(s1))
    j = k - i
    
    if i <= len(s0) and j <= len(s1):
        print("YES")
        res = s0[:i] + s1[:j]
        print(*(res))
    else:
        print("NO")

solve()
```
