```python
import sys

# The problem states the graph is bipartite.
# We need to find an independent set of size k.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# However, we don't need the maximum independent set, just any independent set of size k.
# Since the graph is bipartite, we can partition the vertices into two sets, A and B.
# Any subset of A is an independent set, and any subset of B is an independent set.
# Let |A| = nA and |B| = nB. We can pick any number of vertices from A (0 to nA)
# and any number of vertices from B (0 to nB).
# Wait, this is only true if we pick *only* from A or *only* from B.
# Actually, we can pick any subset of A and any subset of B, and the union will be
# an independent set because there are no edges within A and no edges within B.
# Thus, we can form an independent set of any size s where 0 <= s <= nA + nB = n.
# Wait, is that true? Yes, because the graph is bipartite, the set of all vertices
# in A is an independent set, and the set of all vertices in B is an independent set.
# Any subset of A is independent, and any subset of B is independent.
# The union of any subset of A and any subset of B is independent because
# all edges only go between A and B.
# Therefore, we can pick any number of vertices from A and any number from B.
# The total number of vertices we can pick is any k such that 0 <= k <= n.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    set_a = []
    set_b = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0:
                    set_a.append(u)
                else:
                    set_b.append(u)
                
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                        
    if k > n:
        print("NO")
        return
    
    # We need to pick k vertices. We can pick min(k, len(set_a)) from set_a
    # and the rest from set_b.
    res = []
    count = 0
    for node in set_a:
        if count < k:
            res.append(node)
            count += 1
        else:
            break
    for node in set_b:
        if count < k:
            res.append(node)
            count += 1
        else:
            break
            
    if len(res) == k:
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
```
