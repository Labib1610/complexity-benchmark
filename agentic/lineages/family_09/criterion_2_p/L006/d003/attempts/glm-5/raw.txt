import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            queue = deque([i])
            comp_a = []
            comp_b = []
            while queue:
                u = queue.popleft()
                if color[u] == 1:
                    comp_a.append(u)
                else:
                    comp_b.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
            components.append((comp_a, comp_b))
    
    # DP: dp[j] = max size achievable with j components, -1 if impossible
    # We track the choice (0 or 1) for each component
    num_comp = len(components)
    dp = [-1] * (num_comp + 1)
    dp[0] = 0
    choice = [0] * num_comp
    
    for i in range(num_comp):
        a, b = components[i]
        new_dp = [-1] * (num_comp + 1)
        for j in range(i + 1):
            if dp[j] >= 0:
                # Choose side a
                if new_dp[j + 1] < dp[j] + len(a):
                    new_dp[j + 1] = dp[j] + len(a)
                # Choose side b
                if new_dp[j + 1] < dp[j] + len(b):
                    new_dp[j + 1] = dp[j] + len(b)
        dp = new_dp
    
    # Find valid selection
    result = []
    found = False
    
    # Try all possible numbers of components where we pick side a
    for cnt_a in range(num_comp + 1):
        if dp[cnt_a] >= k:
            # Reconstruct: we need cnt_a components with side a, rest with side b
            # Greedy: pick smallest sides first for the remaining
            indices = list(range(num_comp))
            # Sort by (len(a) - len(b)) ascending, so smallest a first for b-picks
            indices.sort(key=lambda x: len(components[x][0]) - len(components[x][1]))
            
            result = []
            for idx in indices:
                a, b = components[idx]
                if cnt_a > 0:
                    result.extend(a)
                    cnt_a -= 1
                else:
                    result.extend(b)
            
            if len(result) >= k:
                result = result[:k]
                found = True
                break
    
    if found:
        print("YES")
        print(' '.join(map(str, result)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
