import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_nodes = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp_nodes[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append(comp_nodes)
            
    # Each component i has two independent sets: S_i0 and S_i1
    # We need to pick either S_i0 or S_i1 for each component to sum to k
    # This is a variation of the subset sum problem
    # dp[i][j] = can we get sum j using first i components?
    # Since we need to reconstruct, store choices.
    
    # Optimization: use bitset-like approach or simple DP
    # dp[j] = (prev_j, choice)
    dp = {0: None}
    
    for i, (s0, s1) in enumerate(components):
        new_dp = {}
        len0, len1 = len(s0), len(s1)
        for current_sum in dp:
            if current_sum + len0 <= k:
                new_dp[current_sum + len0] = (current_sum, 0)
            if current_sum + len1 <= k:
                new_dp[current_sum + len1] = (current_sum, 1)
        dp = new_dp
        
    if k not in dp:
        print("NO")
        return
    
    print("YES")
    res = []
    curr = k
    for i in range(len(components) - 1, -1, -1):
        prev_sum, choice = dp[curr]
        res.extend(components[i][choice])
        curr = prev_sum
        
    print(*(res))

solve()
