import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp0 = []
            comp1 = []
            stack = [(i, 0)]
            color[i] = 0
            comp0.append(i)
            
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            comp0.append(v)
                        else:
                            comp1.append(v)
                        stack.append((v, 1 - c))
            components.append((comp0, comp1))
            
    # dp[i][j] = can we pick j nodes from first i components?
    # Since we need to pick exactly k, and each component offers two sets (comp0, comp1)
    # This is a variation of the subset sum problem.
    # dp[j] stores the choice made for each component to reach sum j.
    # To reconstruct, we store which set (0 or 1) was used.
    
    # dp[j] = (prev_j, choice_index)
    dp = {0: None}
    
    for idx, (c0, c1) in enumerate(components):
        new_dp = {}
        s0, s1 = len(c0), len(c1)
        for j in dp:
            if j + s0 <= k:
                new_dp[j + s0] = (j, 0)
            if j + s1 <= k:
                new_dp[j + s1] = (j, 1)
        dp = new_dp
        
    if k not in dp:
        print("NO")
        return
    
    print("YES")
    res = []
    curr = k
    for i in range(len(components) - 1, -1, -1):
        prev_j, choice = dp[curr]
        if choice == 0:
            res.extend(components[i][0])
        else:
            res.extend(components[i][1])
        curr = prev_j
        
    print(*(res))

solve()
