import sys

# The problem asks for an independent set of size k in a graph.
# The problem statement guarantees the mechanism is "fully operational",
# which implies the graph is bipartite.
# A bipartite graph can be partitioned into two sets, say A and B,
# such that all edges connect a vertex in A to a vertex in B.
# Any subset of A is an independent set, and any subset of B is an independent set.
# To find an independent set of size k, we can use the fact that
# the maximum independent set in a bipartite graph is n - (size of maximum matching).
# However, we just need *any* independent set of size k.
# Since the graph is bipartite, we can color it with 2 colors.
# Let S1 be the set of nodes with color 0, and S2 be the set of nodes with color 1.
# Any subset of S1 is independent, and any subset of S2 is independent.
# We can pick k nodes from S1 U S2 if we can find a subset of size k.
# Actually, the maximum independent set in a bipartite graph is not just S1 or S2.
# But for this problem, we can use the property that the graph is bipartite.
# We can find the maximum independent set using Konig's theorem:
# Max Independent Set = Total Vertices - Minimum Vertex Cover.
# Min Vertex Cover = Max Matching in bipartite graphs.
# So Max Independent Set size = n - |Max Matching|.
# If k <= Max Independent Set size, we can pick k vertices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    s1, s2 = [], []
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: s1.append(u)
                else: s2.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # In a bipartite graph, the maximum independent set can be found by
    # finding the max matching and using the construction from Konig's theorem.
    # However, a simpler approach: the size of the maximum independent set is
    # at least max(|S1|, |S2|). If k <= max(|S1|, |S2|), we can just pick k from the larger set.
    # Is it possible that k > max(|S1|, |S2|) but a solution exists? Yes.
    # We need the actual maximum independent set.
    
    # Hopcroft-Karp or simple DFS matching
    match = [-1] * (n + 1)
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    matching_size = 0
    for u in s1:
        visited = [False] * (n + 1)
        if dfs(u, visited):
            matching_size += 1
            
    # Max Independent Set construction:
    # Let U be the set of unmatched vertices in S1.
    # Let Z be the set of vertices reachable from U via alternating paths.
    # Max Independent Set = (S1 \ Z) U (S2 ∩ Z)
    z = [False] * (n + 1)
    def dfs_z(u):
        z[u] = True
        for v in adj[u]:
            if color[u] == 0: # u in S1, edge to S2 is not in matching
                if match[v] != u and not z[v]:
                    dfs_z(v)
            else: # u in S2, edge to S1 is in matching
                if match[u] == v and not z[v]:
                    dfs_z(v)
                    
    unmatched_s1 = [u for u in s1 if match[u] == -1] # Wait, match stores S1 nodes for S2 keys
    # Correct matching: match[v] = u where u in S1, v in S2
    match = [-1] * (n + 1)
    for u in s1:
        visited = [False] * (n + 1)
        if dfs(u, visited): matching_size += 1
        
    z = [False] * (n + 1)
    def dfs_z(u):
        z[u] = True
        for v in adj[u]:
            if color[u] == 0:
                if match[v] != u and not z[v]: dfs_z(v)
            else:
                if match[u] == v and not z[v]: dfs_z(v)
    
    for u in s1:
        if match[u] == -1: dfs_z(u)
        
    mis = [u for u in s1 if not z[u]] + [u for u in s2 if z[u]]
    
    if len(mis) >= k:
        print("YES")
        print(*(mis[:k]))
    else:
        print("NO")

solve()
