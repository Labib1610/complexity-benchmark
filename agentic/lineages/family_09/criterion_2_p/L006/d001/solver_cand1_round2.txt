import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    set0 = []
    set1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            set0.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            set0.append(v)
                        else:
                            set1.append(v)
                        stack.append((v, 1 - c))
    
    len0 = len(set0)
    len1 = len(set1)
    
    # We need to pick k gears such that no two are connected.
    # Since the graph is bipartite, we can pick any subset of set0 or any subset of set1.
    # However, we can pick a mix if we pick independent sets.
    # Actually, the problem asks for *any* independent set of size k.
    # In a bipartite graph, the maximum independent set is n - matching.
    # But we just need *any* independent set of size k.
    # Any subset of set0 is an independent set. Any subset of set1 is an independent set.
    # If k <= len0, we can pick k from set0.
    # If k <= len1, we can pick k from set1.
    # If k > len0 and k > len1, is it possible?
    # The problem guarantees the graph is bipartite.
    # The maximum independent set size is at least max(len0, len1).
    # If k <= max(len0, len1), we can definitely pick k.
    # If k > max(len0, len1), it might be impossible.
    
    if k <= len0:
        print("YES")
        print(*(set0[:k]))
    elif k <= len1:
        print("YES")
        print(*(set1[:k]))
    else:
        print("NO")

solve()
