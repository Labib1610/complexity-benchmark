import sys

# The problem states that the graph is bipartite.
# We can partition the vertices into two sets: Transmitters (T) and Receivers (R).
# Any edge (u, v) must connect a vertex in T to a vertex in R.
# This is a standard bipartite graph.
# We need to find an independent set of size k.
# In a bipartite graph, the size of the maximum independent set is n - (size of maximum matching).
# However, we need to find if there exists an independent set of size k.
# Actually, the problem is simpler: we can pick any subset of vertices that are not connected.
# Since it's bipartite, we can find the two partitions using BFS/DFS.
# Let the partitions be V1 and V2.
# Any subset of V1 is an independent set, and any subset of V2 is an independent set.
# But we can pick vertices from both.
# Wait, the problem is equivalent to finding an independent set of size k.
# In a bipartite graph, finding the maximum independent set is solvable,
# but finding an independent set of size k is generally NP-hard.
# Re-reading: "it is possible to assign a direction... every frequency acts exclusively as a transmitter... or receiver".
# This means the graph is bipartite.
# Is it possible that the graph is a collection of disjoint complete bipartite graphs?
# The example 1: 1-3, 1-4, 2-3, 2-4. This is K_{2,2}.
# Max independent set size is 2 (e.g., {1, 2} or {3, 4}).
# The problem asks for exactly k.
# Since it's bipartite, we can use the property that the maximum independent set size is n - max_matching.
# But we need exactly k.
# Actually, in a bipartite graph, if we have a maximum independent set of size M,
# can we always find an independent set of size k for any k <= M?
# Yes, because we can just remove vertices from the independent set.
# So we just need to check if k <= n - max_matching.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    v1, v2 = [], []
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: v1.append(u)
                else: v2.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # Max independent set in bipartite graph:
    # Max Independent Set = V \ Minimum Vertex Cover
    # By Konig's theorem, Min Vertex Cover = Max Matching
    # So Max Independent Set = n - Max Matching
    
    match = [-1] * (n + 1)
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    matching_size = 0
    for u in v1:
        visited = [False] * (n + 1)
        if dfs(u, visited):
            matching_size += 1
            
    if k <= n - matching_size:
        # We need to find an independent set of size k.
        # The set of all vertices not in the min vertex cover is a max independent set.
        # Min vertex cover can be found from max matching.
        # Let U be the set of unmatched vertices in V1, plus vertices reachable from U via alternating paths.
        # This is standard.
        
        # Simplified: just pick vertices from V1 and V2 greedily? No.
        # Actually, the set of all vertices in V1 that are not in the matching
        # plus vertices in V2 that are matched is not necessarily an independent set.
        # Let's use the construction:
        # S = (V1 \ Z) U (V2 \cap Z) where Z is the set of vertices reachable from unmatched V1.
        # This is the Max Independent Set.
        
        matched_v1 = {match[v] for v in v2 if match[v] != -1}
        unmatched_v1 = [u for u in v1 if u not in matched_v1]
        
        z = set()
        stack = unmatched_v1
        for u in stack: z.add(u)
        while stack:
            u = stack.pop()
            for v in adj[u]:
                if v not in z and v in v2 and match[v] != u:
                    z.add(v)
                    if match[v] != -1:
                        z.add(match[v])
                        stack.append(match[v])
        
        indep_set = [u for u in v1 if u not in z] + [u for u in v2 if u in z]
        print("YES")
        print(*(indep_set[:k]))
    else:
        print("NO")

solve()
