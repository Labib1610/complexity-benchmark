import sys

def solve():
    # Read n, m, k
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem states the graph is bipartite.
    # We can find the two partitions using BFS/DFS.
    color = [-1] * (n + 1)
    part0 = []
    part1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            part0.append(i)
            while stack:
                u, c = stack.pop()
                next_c = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = next_c
                        if next_c == 0:
                            part0.append(v)
                        else:
                            part1.append(v)
                        stack.append((v, next_c))
    
    # We need to pick k nodes such that no two are connected.
    # Since it's bipartite, any subset of part0 is independent,
    # and any subset of part1 is independent.
    # We can pick x nodes from part0 and y nodes from part1 such that x + y = k.
    # However, we can only pick from part0 if we don't pick any neighbors,
    # but in a bipartite graph, all nodes in part0 are independent of each other.
    # So we can pick any subset of size x from part0 and any subset of size y from part1.
    
    len0 = len(part0)
    len1 = len(part1)
    
    # We need to find x, y such that 0 <= x <= len0, 0 <= y <= len1, x + y = k
    # This is possible if and only if len0 + len1 >= k, which is always true (n >= k).
    # Wait, the condition is simply that we can pick k nodes from the union of independent sets.
    # Actually, any subset of part0 is an independent set, and any subset of part1 is an independent set.
    # But we cannot mix them arbitrarily if there are edges between them.
    # Wait, the problem says "A batch is safe if absolutely no two microservices within it are listed as conflicting."
    # In a bipartite graph, the maximum independent set is the larger of the two partitions.
    # But we don't need the maximum independent set, we just need *any* independent set of size k.
    # Any subset of part0 is independent. Any subset of part1 is independent.
    # So if k <= len0, we can just pick k nodes from part0.
    # If k <= len1, we can just pick k nodes from part1.
    # If k > len0 and k > len1, can we pick k nodes?
    # Not necessarily. But the problem asks for *a* safe batch.
    # Actually, in a bipartite graph, the union of any subset of part0 and any subset of part1 is safe
    # ONLY IF there are no edges between the chosen nodes.
    # This is only guaranteed if we pick nodes from ONLY ONE partition.
    # Is it possible that we need to pick from both?
    # If we pick a set S, it is safe if no two nodes in S have an edge.
    # This means S must be an independent set.
    # The maximum independent set size is at least max(len0, len1).
    # If k <= max(len0, len1), we can pick k nodes from the larger partition.
    # If k > max(len0, len1), it is impossible.
    
    if k <= len0:
        print("YES")
        print(*(part0[:k]))
    elif k <= len1:
        print("YES")
        print(*(part1[:k]))
    else:
        print("NO")

solve()
