import sys
from collections import defaultdict, deque

def read_input():
    n, m, k = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
    return n, m, k, edges

def build_graph(n, edges):
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    return graph

def bfs_2coloring(graph, n):
    # Returns True if bipartite, and color array (or None if not bipartite)
    # Since input guarantees no odd cycles, it's always bipartite
    color = [-1] * (n + 1)  # -1: unvisited, 0: color A, 1: color B
    for i in range(1, n + 1):
        if color[i] == -1:
            queue = deque()
            queue.append(i)
            color[i] = 0
            while queue:
                node = queue.popleft()
                current_color = color[node]
                next_color = 1 - current_color
                for neighbor in graph[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = next_color
                        queue.append(neighbor)
                    elif color[neighbor] != next_color:
                        # This should never happen due to problem guarantee
                        return False, None
    return True, color

def solve():
    n, m, k, edges = read_input()
    graph = build_graph(n, edges)
    
    # Check if graph is bipartite (should always be true per problem statement)
    is_bipartite, colors = bfs_2coloring(graph, n)
    
    # Separate nodes by color
    set_a = []
    set_b = []
    for i in range(1, n + 1):
        if colors[i] == 0:
            set_a.append(i)
        elif colors[i] == 1:
            set_b.append(i)
        else:  # Unconnected node (no edges), assign to set_a
            set_a.append(i)
    
    # Try to take k nodes from the larger set first for better chance
    if len(set_a) >= k:
        selected = set_a[:k]
        print("YES")
        print(" ".join(map(str, selected)))
    elif len(set_b) >= k:
        selected = set_b[:k]
        print("YES")
        print(" ".join(map(str, selected)))
    else:
        # Combine both sets if neither alone has enough
        total_available = len(set_a) + len(set_b)
        if total_available >= k:
            # Take from both sets
            take_from_a = min(len(set_a), k)
            take_from_b = k - take_from_a
            selected = set_a[:take_from_a] + set_b[:take_from_b]
            print("YES")
            print(" ".join(map(str, selected)))
        else:
            print("NO")

if __name__ == "__main__":
    solve()
