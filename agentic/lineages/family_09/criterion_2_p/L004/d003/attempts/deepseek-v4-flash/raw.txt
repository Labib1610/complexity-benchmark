import sys
sys.setrecursionlimit(1_000_000)

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)
    comps = []  # list of (list0, list1) for each component
    possible = True

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        color[start] = 0
        stack = [start]
        part0 = []
        part1 = []
        while stack:
            u = stack.pop()
            if color[u] == 0:
                part0.append(u)
            else:
                part1.append(u)
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    stack.append(v)
                elif color[v] == color[u]:
                    # This should not happen due to no odd cycles, but keep safe
                    possible = False
        comps.append((part0, part1))

    if not possible:
        print("NO")
        return

    # For each component, we can take either side.
    # We need to see if we can reach total size k.
    # This is a subset sum problem with n up to 1e5, but each component contributes
    # one of two sizes: len(part0) or len(part1).
    # We only need to know if k is achievable, and reconstruct one solution.
)
    # Use bitset DP for feasibility.
 n up to 1e5, bitset of size n+1 is fine.
)
    bits = 1  # bit i set means size i achievable
    choices = []  # store (a,b) for each component
    for p0, p1 in comps:
        a = len(p0)
        b = len(p1)
        choices.append((a,b)
        bits = (bits << a) | (bits << b)
        # mask to n bits to avoid huge growth
        bits &= (1 << (n + 1)) - 1

    if not (bits >> k) & 1:
        print("NO")
        return

    # Reconstruct: go backwards
    selected = []
    remaining = k
    # We need to know which components were used.
 We can recompute DP backwards.
)
    # Store DP states after each component.

    dp_states = []
    cur = 1
    dp_states.append(cur)
    for a,b in choices:
        cur = (cur << a) | (cur << b)
        cur &= (1 << (n + 1)) - 1
        dp_states.append(cur)

    for idx in range(len(choices) -1, -1, -1):
        a, b = choices[idx]
        prev = dp_states[idx]
        # Check if we can take a from this component
        if remaining >= a and (prev >> (remaining - a)) & 1:
            selected.extend(comps[idx][0])
            remaining -= a
        else:
            # must take b
            selected.extend(comps[idx][1])
            remaining -= b

    # selected should have size k
    print("YES")
    print(' '.join(map(str, selected[:k])))

if __name__ == "__main__":
    main()
