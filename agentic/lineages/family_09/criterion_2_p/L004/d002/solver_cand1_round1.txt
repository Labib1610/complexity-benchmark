```python
import sys

# The problem states that the conflict graph is bipartite because all cycles are even.
# We need to find an independent set of size k.
# In a bipartite graph, the vertices can be partitioned into two sets, L and R,
# such that all edges connect a vertex in L to a vertex in R.
# Any subset of L is an independent set, and any subset of R is an independent set.
# Thus, we can form an independent set of size k if |L| >= k or |R| >= k.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    set0 = []
    set1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            set0.append(i)
            while stack:
                u, c = stack.pop()
                next_c = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = next_c
                        if next_c == 0:
                            set0.append(v)
                        else:
                            set1.append(v)
                        stack.append((v, next_c))
    
    if len(set0) >= k:
        print("YES")
        print(*(set0[:k]))
    elif len(set1) >= k:
        print("YES")
        print(*(set1[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
