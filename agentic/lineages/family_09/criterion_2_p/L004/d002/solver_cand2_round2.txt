```python
import sys

# The problem states that the graph is bipartite.
# A bipartite graph can be partitioned into two independent sets (color 0 and color 1).
# Any subset of an independent set is also an independent set (a "safe batch").
# Therefore, we can find the two independent sets by 2-coloring the graph.
# Let the sizes of the two independent sets be S0 and S1.
# We can form a safe batch of size k if k <= S0 or k <= S1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    group0 = []
    group1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            group0.append(i)
            while stack:
                u, c = stack.pop()
                next_c = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = next_c
                        if next_c == 0:
                            group0.append(v)
                        else:
                            group1.append(v)
                        stack.append((v, next_c))
    
    if len(group0) >= k:
        print("YES")
        print(*(group0[:k]))
    elif len(group1) >= k:
        print("YES")
        print(*(group1[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
