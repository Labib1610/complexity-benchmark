```python
import sys

# The problem states that the graph is bipartite.
# We need to find a subset of size k that is an independent set.
# In a bipartite graph, the vertices can be partitioned into two sets,
# say A and B, such that all edges connect a vertex in A to a vertex in B.
# Any subset of A is an independent set, and any subset of B is an independent set.
# Thus, if we can partition the graph into two sets A and B,
# we can form a safe batch of size k if |A| >= k or |B| >= k.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    camp1 = []
    camp2 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            camp1.append(i)
            while stack:
                u, c = stack.pop()
                next_c = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = next_c
                        if next_c == 0:
                            camp1.append(v)
                        else:
                            camp2.append(v)
                        stack.append((v, next_c))
    
    if len(camp1) >= k:
        print("YES")
        print(*(camp1[:k]))
    elif len(camp2) >= k:
        print("YES")
        print(*(camp2[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
