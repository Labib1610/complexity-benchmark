Here’s the Python solution for the problem:

```python
import sys
from collections import deque

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    comps = []  # each comp: (size0, size1, list0, list1)
    
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            q = deque([i])
            nodes0 = []
            nodes1 = []
            cnt0 = 0
            cnt1 = 0
            while q:
                u = q.popleft()
                if color[u] == 0:
                    cnt0 += 1
                    nodes0.append(u)
                else:
                    cnt1 += 1
                    nodes1.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        q.append(v)
            comps.append((cnt0, cnt1, nodes0, nodes1))

    # DP to see if we can pick exactly k nodes with no conflicts
    # Each component: we can pick either from side0 or side1
    dp = [False] * (k + 1)
    dp[0] = True
    choice = [-1] * (k + 1)  # store which component and side chosen
    parent = [-1] * (k + 1)
    
    for idx, (s0, s1, _, _) in enumerate(comps):
        new_dp = [False] * (k + 1)
        new_choice = [-1] * (k + 1)
        new_parent = [-1] * (k + 1)
        for cur in range(k + 1):
            if not dp[cur]:
                continue
            # choose side 0
            if cur + s0 <= k and not new_dp[cur + s0]:
                new_dp[cur + s0] = True
                new_choice[cur + s0] = 0
                new_parent[cur + s0] = cur
            # choose side 1
            if cur + s1 <= k and not new_dp[cur + s1]:
                new_dp[cur + s1] = True
                new_choice[cur + s1] = 1
                new_parent[cur + s1] = cur
        dp = new_dp
        choice = new_choice
        parent = new_parent

    if not dp[k]:
        print("NO")
        return

    # reconstruct
    selected = []
    cur = k
    for idx in range(len(comps) - 1, -1, -1):
        if choice[cur] == -1:
            continue
        side = choice[cur]
        prev = parent[cur]
        if side == 0:
            selected.extend(comps[idx][2])
        else:
            selected.extend(comps[idx][3])
        cur = prev

    print("YES")
    print(" ".join(str(x + 1) for x in selected))

if __name__ == "__main__":
    solve()
```
