import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_nodes = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp_nodes[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append(comp_nodes)

    # dp[i][j] = can we pick j nodes from first i components?
    # To reconstruct, store which choice (0 or 1) was made
    # dp[i][j] stores the choice (0 or 1) for component i to reach sum j
    # Using a dictionary or bitset-like approach for DP
    # Since n is 10^5, we use a list of sets or a boolean array
    
    dp = {0: []}
    for comp in components:
        new_dp = {}
        c0 = len(comp[0])
        c1 = len(comp[1])
        for current_sum, path in dp.items():
            if current_sum + c0 <= k:
                new_dp[current_sum + c0] = path + [0]
            if current_sum + c1 <= k:
                new_dp[current_sum + c1] = path + [1]
        dp = new_dp
    
    if k not in dp:
        print("NO")
    else:
        print("YES")
        choices = dp[k]
        result = []
        for i, choice in enumerate(choices):
            result.extend(components[i][choice])
        print(*(result))

solve()
