```python
import sys
from collections import defaultdict, deque

def read_input():
    n, m, k = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
    return n, m, k, edges

def solve(n, m, k, edges):
    # Build adjacency list
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # If no edges, we can pick any k nodes
    if m == 0:
        return "YES", list(range(1, k+1))
    
    # Try to 2-color the graph (bipartite check)
    color = {}
    partitions = {0: [], 1: []}
    
    # Process all components
    visited = set()
    
    for start_node in range(1, n+1):
        if start_node in visited:
            continue
            
        # BFS to color component
        queue = deque([start_node])
        color[start_node] = 0
        partitions[0].append(start_node)
        visited.add(start_node)
        
        bipartite = True
        while queue and bipartite:
            u = queue.popleft()
            u_color = color[u]
            neighbor_color = 1 - u_color
            
            for v in adj[u]:
                if v not in color:
                    color[v] = neighbor_color
                    partitions[neighbor_color].append(v)
                    visited.add(v)
                    queue.append(v)
                elif color[v] != neighbor_color:
                    bipartite = False
                    break
        
        if not bipartite:
            return "NO", []
    
    # Get the two partitions
    part0 = partitions[0]
    part1 = partitions[1]
    
    # Check if we can take k nodes
    if k <= len(part0):
        return "YES", part0[:k]
    elif k <= len(part1):
        return "YES", part1[:k]
    else:
        # Try to take from both partitions
        # We can take at most len(part0) + len(part1) = n nodes
        # But we need exactly k
        if len(part0) + len(part1) < k:
            return "NO", []
        else:
            # We can take all from one partition and some from the other
            if len(part0) >= k:
                return "YES", part0[:k]
            elif len(part1) >= k:
                return "YES", part1[:k]
            else:
                # Take all from smaller partition and rest from larger
                if len(part0) > len(part1):
                    if len(part0) + len(part1) >= k:
                        needed = k - len(part1)
                        return "YES", part1 + part0[:needed]
                    else:
                        return "NO", []
                else:
                    if len(part0) + len(part1) >= k:
                        needed = k - len(part0)
                        return "YES", part0 + part1[:needed]
                    else:
                        return "NO", []

def main():
    n, m, k, edges = read_input()
    result, components = solve(n, m, k, edges)
    print(result)
    if result == "YES":
        print(' '.join(map(str, components)))

if __name__ == "__main__":
    main()
```
