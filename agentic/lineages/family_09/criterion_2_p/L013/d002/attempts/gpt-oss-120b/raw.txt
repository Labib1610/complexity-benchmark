```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [[] for _ in range(n+1)]
    edges = []
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))

    # bipartition via BFS coloring
    color = [-1]*(n+1)
    left_nodes = []
    right_nodes = []
    for start in range(1, n+1):
        if color[start] != -1:
            continue
        q = deque([start])
        color[start] = 0
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u]^1
                    q.append(v)
    for v in range(1, n+1):
        if color[v]==0:
            left_nodes.append(v)
        else:
            right_nodes.append(v)

    # build adjacency from left to right
    G = {u: [] for u in left_nodes}
    for u,v in edges:
        if color[u]==0:
            G[u].append(v)
        else:
            G[v].append(u)

    # Hopcroft-Karp
    INF = 10**9
    pairU = {u:0 for u in left_nodes}
    pairV = {v:0 for v in right_nodes}
    dist = {}

    def bfs():
        q = deque()
        for u in left_nodes:
            if pairU[u]==0:
                dist[u]=0
                q.append(u)
            else:
                dist[u]=INF
        d = INF
        while q:
            u = q.popleft()
            if dist[u] < d:
                for v in G[u]:
                    if pairV[v]==0:
                        d = dist[u]+1
                    elif dist[pairV[v]]==INF:
                        dist[pairV[v]] = dist[u]+1
                        q.append(pairV[v])
        return d!=INF

    def dfs(u):
        for v in G[u]:
            if pairV[v]==0 or (dist[pairV[v]]==dist[u]+1 and dfs(pairV[v])):
                pairU[u]=v
                pairV[v]=u
                return True
        dist[u]=INF
        return False

    matching = 0
    while bfs():
        for u in left_nodes:
            if pairU[u]==0 and dfs(u):
                matching+=1

    # find minimum vertex cover
    # BFS from unmatched left vertices on alternating graph
    L_vis = set()
    R_vis = set()
    q = deque()
    for u in left_nodes:
        if pairU[u]==0:
            q.append(u)
            L_vis.add(u)
    while q:
        u = q.popleft()
        for v in G[u]:
            if v not in R_vis and pairU[u]!=v:  # edge not in matching
                R_vis.add(v)
                if pairV[v]!=0 and pairV[v] not in L_vis:
                    L_vis.add(pairV[v])
                    q.append(pairV[v])

    # minimum vertex cover = (Left - L_vis) ∪ (Right ∩ R_vis)
    # independent set = (Left ∩ L_vis) ∪ (Right - R_vis)
    indep = []
    for u in left_nodes:
        if u in L_vis:
            indep.append(u)
    for v in right_nodes:
        if v not in R_vis:
            indep.append(v)

    if len(indep) < k:
        print("NO")
        return
    print("YES")
    print(' '.join(map(str, indep[:k])))

if __name__ == "__main__":
    solve()
```
