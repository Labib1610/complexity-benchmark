```python
import sys

# The problem states that resonance occurs if the area of a triangle with legs
# |f_u + f_v| and |f_u - f_v| is not a whole number.
# Area = 0.5 * |f_u + f_v| * |f_u - f_v| = 0.5 * |f_u^2 - f_v^2|.
# For the area to be a whole number, |f_u^2 - f_v^2| must be even.
# This means f_u^2 and f_v^2 must have the same parity.
# If they have different parity, the area is X.5, which is not a whole number.
# The problem states resonance ONLY occurs if the area is NOT a whole number.
# Thus, resonance only occurs if f_u^2 and f_v^2 have different parity.
# This implies the graph is bipartite: one set has even f^2, the other odd f^2.
# We need to find an independent set of size k in a bipartite graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append(comp)

    # dp[i][j] = can we pick j nodes from first i components?
    # Since we need to pick k nodes, and each component offers two sets (size A, size B),
    # we want to pick one set from each component such that sum of sizes is k.
    # This is a variation of the subset sum problem.
    
    dp = {0: []}
    for i, (c0, c1) in enumerate(components):
        new_dp = {}
        s0, s1 = len(c0), len(c1)
        for current_k, path in dp.items():
            if current_k + s0 <= k:
                new_dp[current_k + s0] = path + c0
            if current_k + s1 <= k:
                new_dp[current_k + s1] = path + c1
        dp = new_dp
        
    if k in dp:
        print("YES")
        print(*(dp[k]))
    else:
        print("NO")

solve()
```
