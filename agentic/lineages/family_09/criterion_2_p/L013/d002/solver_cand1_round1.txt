```python
import sys

# The problem states that resonance only occurs between modules with 
# positive and negative temperature offsets. This implies the graph is bipartite.
# We can color the graph using 2 colors (0 and 1).
# Let S0 be the set of nodes with color 0, and S1 be the set of nodes with color 1.
# Any subset of S0 is an independent set, and any subset of S1 is an independent set.
# We need to pick k nodes such that no two are connected.
# This is equivalent to picking x nodes from S0 and y nodes from S1 such that x + y = k,
# where x <= |S0| and y <= |S1|.
# Since we can pick any subset of S0 and any subset of S1, we can form an independent set
# of size k if and only if there exist x, y such that 0 <= x <= |S0|, 0 <= y <= |S1|,
# and x + y = k.
# This is possible if and only if k <= |S0| + |S1| = n, which is always true,
# but we must satisfy the constraints x <= |S0| and y <= |S1|.
# The range of possible sizes for an independent set formed by picking x from S0 and y from S1
# is [0, |S0| + |S1|] is not quite right; we need to pick exactly k.
# The maximum size of an independent set we can form this way is |S0| + |S1| = n.
# Wait, the condition is simply: can we pick x from S0 and y from S1 such that x+y=k?
# Yes, if max(0, k - |S1|) <= x <= min(k, |S0|).
# Such an x exists if max(0, k - |S1|) <= min(k, |S0|).
# This simplifies to: k <= |S0| + |S1| (always true) and k >= 0 (always true).
# Actually, the constraint is just that we need to pick x nodes from S0 and k-x from S1.
# We need 0 <= x <= |S0| and 0 <= k-x <= |S1|.
# This means x <= |S0| and x >= k - |S1|.
# So we need an x such that max(0, k - |S1|) <= x <= min(k, |S0|).
# Since |S0| + |S1| = n and k <= n, such an x always exists.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    s0, s1 = [], []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: s0.append(u)
                else: s1.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # We need x from s0, k-x from s1
    # 0 <= x <= len(s0)
    # 0 <= k-x <= len(s1) => x <= k and x >= k - len(s1)
    lower = max(0, k - len(s1))
    upper = min(k, len(s0))
    
    if lower <= upper:
        x = lower
        res = s0[:x] + s1[:k-x]
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
```
