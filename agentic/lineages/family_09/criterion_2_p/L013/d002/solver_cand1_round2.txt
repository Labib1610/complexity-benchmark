```python
import sys

# The problem states that the graph is bipartite.
# We can find the connected components of the graph.
# For each connected component, we can 2-color it into two sets: A and B.
# Let a component have size A_i and B_i.
# We need to pick a subset of components and for each, pick either set A_i or B_i
# such that the total number of picked modules is k.
# This is a variation of the subset sum problem (knapsack).
# Since we need to pick exactly k, we can use dynamic programming.
# To optimize, we use bitsets (in Python, large integers).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_a = []
            comp_b = []
            stack = [(i, 0)]
            color[i] = 0
            comp_a.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0: comp_a.append(v)
                        else: comp_b.append(v)
                        stack.append((v, 1 - c))
            components.append((comp_a, comp_b))

    # dp[i] stores a way to reach sum i. 
    # To reconstruct, we store which choice was made.
    # dp[sum] = (component_index, choice_0_or_1)
    dp = {0: None}
    
    for i, (a, b) in enumerate(components):
        new_dp = dp.copy()
        len_a, len_b = len(a), len(b)
        for s, _ in dp.items():
            if s + len_a <= k and (s + len_a) not in new_dp:
                new_dp[s + len_a] = (i, 0)
            if s + len_b <= k and (s + len_b) not in new_dp:
                new_dp[s + len_b] = (i, 1)
        dp = new_dp
        
    if k not in dp:
        print("NO")
        return

    print("YES")
    res = []
    curr = k
    while dp[curr] is not None:
        comp_idx, choice = dp[curr]
        comp_a, comp_b = components[comp_idx]
        if choice == 0:
            res.extend(comp_a)
            curr -= len(comp_a)
        else:
            res.extend(comp_b)
            curr -= len(comp_b)
            
    print(*(res))

solve()
```
