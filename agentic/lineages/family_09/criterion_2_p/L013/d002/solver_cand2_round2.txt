```python
import sys

# The problem states that the graph is bipartite (resonance only between different vendors).
# We need to find a subset of size k that is an independent set.
# In a bipartite graph, each connected component can be 2-colored.
# Let a component have size A of color 0 and size B of color 1.
# We can pick either all A nodes or all B nodes from this component.
# This is a variation of the subset sum problem (knapsack).
# Since we need to pick exactly k, we can use dynamic programming.
# To optimize, we use bitsets (in Python, large integers).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append(comp)

    # dp[i] stores a bitmask where the j-th bit is 1 if sum j is possible
    # To reconstruct, we store which choice was made
    dp = {0: []}
    
    for comp in components:
        new_dp = {}
        size0 = len(comp[0])
        size1 = len(comp[1])
        
        for s, path in dp.items():
            # Option 1: pick color 0
            if s + size0 <= k:
                new_dp[s + size0] = path + [(comp[0], 0)]
            # Option 2: pick color 1
            if s + size1 <= k:
                new_dp[s + size1] = path + [(comp[1], 1)]
        dp = new_dp
        
    if k not in dp:
        print("NO")
    else:
        print("YES")
        res = []
        for nodes, c in dp[k]:
            res.extend(nodes)
        print(*(res))

solve()
```
