import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    sets = [[], []]
    possible = True
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            while stack:
                u, c = stack.pop()
                if color[u] != 0:
                    continue
                color[u] = c
                sets[c - 1].append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break
    
    if not possible:
        print("NO")
        return
    
    set0 = sets[0]
    set1 = sets[1]
    
    if len(set0) > len(set1):
        set0, set1 = set1, set0
    
    if k <= len(set0):
        print("YES")
        print(' '.join(map(str, set0[:k])))
    elif k <= len(set1):
        print("YES")
        print(' '.join(map(str, set1[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
