```python
import sys

# The problem states that any sequence of consecutive interference links that 
# starts and ends at the same component consists of an even number of links.
# This is the definition of a bipartite graph.
# In a bipartite graph, we can partition the vertices into two sets (colors) 
# such that no two vertices in the same set are connected by an edge.
# Let the two sets be S1 and S2 with sizes |S1| = c1 and |S2| = c2.
# We need to pick k vertices such that no two are connected.
# This means we can pick x vertices from S1 and y vertices from S2 
# such that x + y = k, and we don't violate the independent set condition.
# Actually, since S1 and S2 are independent sets, any subset of S1 is 
# independent, and any subset of S2 is independent.
# So we can pick any x vertices from S1 and y vertices from S2 as long as 
# x <= c1 and y <= c2, and x + y = k.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    s1 = []
    s2 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0:
                    s1.append(u)
                else:
                    s2.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    c1 = len(s1)
    c2 = len(s2)
    
    # We need to find x, y such that x + y = k, 0 <= x <= c1, 0 <= y <= c2
    # This is possible if and only if k <= c1 + c2 (which is n)
    # and there exists x such that max(0, k - c2) <= x <= min(k, c1)
    
    found = False
    res = []
    for x in range(max(0, k - c2), min(k, c1) + 1):
        y = k - x
        res = s1[:x] + s2[:y]
        found = True
        break
        
    if found:
        print("YES")
        print(*(res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
