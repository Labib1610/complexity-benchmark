```python
import sys

# The problem states that resonance only occurs between modules with an odd frequency difference.
# This implies the graph is bipartite. We can color the graph with two colors (0 and 1).
# Any independent set can be formed by picking some nodes from the set of nodes colored 0
# and some from the set of nodes colored 1.
# However, the constraint is that we cannot pick two nodes connected by an edge.
# In a bipartite graph, this means we can pick any subset of nodes from each connected component
# such that no two connected nodes are picked.
# Actually, the problem is simpler: since it's bipartite, we can partition the graph into
# connected components. For each component, we can pick either the set of nodes in one partition
# or the other, or subsets thereof. But wait, the problem is just finding an independent set of size k.
# In a bipartite graph, the maximum independent set is n - minimum vertex cover.
# Actually, for each connected component, we can pick either the set of nodes in partition A
# or partition B. Let the component have partitions A_i and B_i.
# We can pick any subset of A_i or any subset of B_i? No, that's not right.
# We can pick any independent set. In a bipartite graph, any subset of A_i is an independent set,
# and any subset of B_i is an independent set.
# We want to pick exactly k nodes. Let the components be C_1, ..., C_p.
# Each C_j has partitions A_j and B_j. We can pick any subset of A_j or any subset of B_j.
# This is equivalent to saying we have a collection of sets A_j and B_j, and we want to pick
# a total of k elements from these such that we don't pick both a node and its neighbor.
# Actually, we can just pick all nodes in A_j or all nodes in B_j for each component.
# This is a variation of the subset sum problem (knapsack).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_a, comp_b = [], []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: comp_a.append(u)
                else: comp_b.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append((comp_a, comp_b))
            
    # dp[i][j] = can we pick j nodes using first i components
    # To reconstruct, store which choice was made
    dp = {0: []}
    for i, (a, b) in enumerate(components):
        new_dp = {}
        len_a, len_b = len(a), len(b)
        for current_k, path in dp.items():
            if current_k + len_a <= k:
                new_dp[current_k + len_a] = path + [('a', i)]
            if current_k + len_b <= k:
                new_dp[current_k + len_b] = path + [('b', i)]
        dp = new_dp
        
    if k not in dp:
        print("NO")
    else:
        print("YES")
        res = []
        for choice, comp_idx in dp[k]:
            if choice == 'a':
                res.extend(components[comp_idx][0])
            else:
                res.extend(components[comp_idx][1])
        print(*(res))

solve()
```
