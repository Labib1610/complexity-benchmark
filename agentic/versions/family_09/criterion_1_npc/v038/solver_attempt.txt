import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle for Maximum Independent Set
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        res = {}
        for i in range(1 << size):
            subset = []
            valid = True
            mask = 0
            for j in range(size):
                if (i >> j) & 1:
                    u = nodes[j]
                    if (mask & adj_list[u]):
                        valid = False
                        break
                    mask |= (1 << u)
                    subset.append(u)
            if valid:
                count = len(subset)
                if count not in res or mask > res[count]:
                    res[count] = mask
        return res

    # Since n <= 46, 2^(n/2) is ~2^23, which is manageable.
    # However, we need to find if there exists a set of size k.
    # We can use a simple backtracking with pruning (Bron-Kerbosch style)
    
    nodes = list(range(n))
    
    def find_k(candidates, current_set, k_needed):
        if k_needed == 0:
            return current_set
        if not candidates or len(candidates) < k_needed:
            return None
        
        # Heuristic: pick node with most conflicts
        v = candidates[0]
        
        # Try including v
        new_candidates = [u for u in candidates[1:] if not (adj[v] & (1 << u))]
        res = find_k(new_candidates, current_set + [v + 1], k_needed - 1)
        if res: return res
        
        # Try excluding v
        res = find_k(candidates[1:], current_set, k_needed)
        return res

    # The constraints and the problem type suggest a search.
    # Given n=46, simple backtracking might be slow, but the graph is sparse
    # or has specific properties.
    result = find_k(list(range(n)), [], k)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
