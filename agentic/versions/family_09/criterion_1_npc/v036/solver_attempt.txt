import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a meet-in-the-middle approach or a 
    # backtracking search with pruning. Given the constraints and 
    # the nature of the problem, a simple backtracking with bitmask 
    # pruning is efficient enough.
    
    def find_is(candidates, current_set, size_needed):
        if size_needed == 0:
            return current_set
        if not candidates or bin(candidates).count('1') < size_needed:
            return None
        
        # Heuristic: pick the node with the fewest candidates to branch on
        # Or just pick the lowest bit
        v = (candidates & -candidates).bit_length() - 1
        
        # Option 1: Include v
        res = find_is(candidates & ~adj[v] & ~(1 << v), current_set | (1 << v), size_needed - 1)
        if res is not None:
            return res
        
        # Option 2: Exclude v
        res = find_is(candidates & ~(1 << v), current_set, size_needed)
        return res

    result_mask = find_is((1 << n) - 1, 0, k)
    
    if result_mask is not None:
        print("YES")
        ans = []
        for i in range(n):
            if (result_mask >> i) & 1:
                ans.append(str(i + 1))
        print(" ".join(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
