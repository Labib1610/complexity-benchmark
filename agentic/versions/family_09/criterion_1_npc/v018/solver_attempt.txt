import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle approach for Maximum Independent Set
    # n <= 46 is too large for 2^(n/2) if we just brute force,
    # but we only need to find a set of size k.
    
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the two halves
    def get_independent_sets(nodes, adj_list):
        size = len(nodes)
        # sets_by_size[s] = list of (mask, bitmask_of_neighbors_in_other_half)
        sets_by_size = [[] for _ in range(size + 1)]
        sets_by_size[0].append((0, 0))
        
        for i in range(size):
            u = nodes[i]
            u_adj = adj_list[u]
            for s in range(i, -1, -1):
                for mask, neighbors in sets_by_size[s]:
                    if not (mask & (1 << i)) and not (neighbors & (1 << i)):
                        new_mask = mask | (1 << i)
                        new_neighbors = neighbors | (u_adj >> nodes[0])
                        sets_by_size[s+1].append((new_mask, new_neighbors))
        return sets_by_size

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Map neighbors in the other half to the actual bitmask
    # For sets1, neighbors are in nodes2.
    # For sets2, neighbors are in nodes1.
    
    for s1 in range(k + 1):
        s2 = k - s1
        if s1 < len(sets1) and s2 < len(sets2):
            for m1, n1_mask in sets1[s1]:
                # n1_mask is relative to nodes2
                for m2, n2_mask in sets2[s2]:
                    # n2_mask is relative to nodes1
                    if not (m1 & n2_mask) and not (m2 & n1_mask):
                        # Found it
                        res = []
                        for i in range(n1):
                            if (m1 >> i) & 1:
                                res.append(i + 1)
                        for i in range(n2):
                            if (m2 >> i) & 1:
                                res.append(i + n1 + 1)
                        print("YES")
                        print(*(res))
                        return
    print("NO")

solve()
