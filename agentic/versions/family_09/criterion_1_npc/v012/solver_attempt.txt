import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1] |= (1 << (v-1))
        adj[v-1] |= (1 << (u-1))

    # The problem asks for an independent set of size k.
    # Since n <= 46, we use a meet-in-the-middle approach.
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the two halves
    def get_independent_sets(nodes, adj_list):
        # nodes is a list of indices
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        
        # Map original indices to 0..size-1
        mapping = {node: i for i, node in enumerate(nodes)}
        local_adj = [0] * size
        for i in range(size):
            for j in range(size):
                if (adj_list[nodes[i]] >> nodes[j]) & 1:
                    local_adj[i] |= (1 << j)
        
        def backtrack(idx, current_mask, count):
            if idx == size:
                res[count].append(current_mask)
                return
            # Option 1: Don't include nodes[idx]
            backtrack(idx + 1, current_mask, count)
            # Option 2: Include nodes[idx] if possible
            if not (current_mask & (1 << idx)):
                backtrack(idx + 1, current_mask | (1 << idx) | local_adj[idx], count + 1)
        
        backtrack(0, 0, 0)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Precompute cross-constraints
    # cross_adj[i] is a bitmask of nodes in nodes1 that are connected to node i+n1
    cross_adj = [0] * n2
    for i in range(n2):
        for j in range(n1):
            if (adj[i + n1] >> j) & 1:
                cross_adj[i] |= (1 << j)

    for k1 in range(k + 1):
        k2 = k - k1
        if k1 < len(sets1) and k2 < len(sets2):
            for mask1 in sets1[k1]:
                # Find nodes in nodes2 compatible with mask1
                forbidden_mask2 = 0
                for i in range(n1):
                    if (mask1 >> i) & 1:
                        forbidden_mask2 |= (adj[i] >> n1)
                
                for mask2 in sets2[k2]:
                    if not (mask2 & forbidden_mask2):
                        # Found it
                        res = []
                        for i in range(n1):
                            if (mask1 >> i) & 1: res.append(i + 1)
                        for i in range(n2):
                            if (mask2 >> i) & 1: res.append(i + n1 + 1)
                        print("YES")
                        print(*(res))
                        return

    print("NO")

solve()
