import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # Meet-in-the-middle for Maximum Independent Set of size k
    n1 = n // 2
    n2 = n - n1
    
    # Precompute subsets for the two halves
    def get_subsets(nodes, offset):
        size = len(nodes)
        subsets = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            mask = 0
            count = 0
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    u = nodes[j]
                    if mask & adj[u]:
                        valid = False
                        break
                    mask |= (1 << u)
                    count += 1
            if valid:
                subsets[count].append((mask, i << offset))
        return subsets

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sub1 = get_subsets(nodes1, 0)
    sub2 = get_subsets(nodes2, n1)
    
    for c1 in range(min(k, n1) + 1):
        c2 = k - c1
        if c2 < 0 or c2 > n2:
            continue
        
        for mask1, raw1 in sub1[c1]:
            # Find compatible masks in sub2
            # A mask2 is compatible if (mask2 & adj[u]) == 0 for all u in mask1
            forbidden = 0
            for i in range(n1):
                if (mask1 >> i) & 1:
                    forbidden |= adj[i]
            
            for mask2, raw2 in sub2[c2]:
                if not (mask2 & forbidden):
                    print("YES")
                    res = []
                    combined = raw1 | raw2
                    for i in range(n):
                        if (combined >> i) & 1:
                            res.append(str(i + 1))
                    print(" ".join(res))
                    return

    print("NO")

solve()
