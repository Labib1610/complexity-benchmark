import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Maximum Independent Set problem for n <= 46.
    # Use Bron-Kerbosch or simple backtracking with pruning.
    # Since we need exactly k, we can use a recursive search.
    
    def find_independent_set(candidates, current_set, target_k):
        if len(current_set) == target_k:
            return current_set
        if not candidates or len(current_set) + bin(candidates).count('1') < target_k:
            return None
        
        # Heuristic: pick vertex with highest degree in the subgraph
        v = (candidates & -candidates).bit_length() - 1
        
        # Option 1: Include v
        res = find_independent_set(candidates & ~adj[v] & ~(1 << v), current_set + [v + 1], target_k)
        if res:
            return res
        
        # Option 2: Exclude v
        res = find_independent_set(candidates & ~(1 << v), current_set, target_k)
        return res

    # Optimization: split into two halves if n is large, but n=46 is small enough
    # for a well-pruned backtracking.
    result = find_independent_set((1 << n) - 1, [], k)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
