import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find a clique of size k in the complement graph.
    # The complement graph has an edge between u and v if they are NOT dangerous.
    # Let comp_adj[u] be the bitmask of nodes NOT dangerous to u.
    comp_adj = [((1 << n) - 1) ^ (1 << u) ^ adj[u] for u in range(n)]
    
    # Bron-Kerbosch with pivoting to find cliques in the complement graph.
    # Since n <= 46, we can use a meet-in-the-middle or optimized search.
    # Given the time limit of 5s, a standard optimized backtracking is sufficient.
    
    def find_clique(candidates, current_clique):
        if not candidates:
            if len(current_clique) == k:
                return current_clique
            return None
        
        if len(current_clique) + bin(candidates).count('1') < k:
            return None
            
        # Pivot to prune search space
        pivot = (candidates & -candidates).bit_length() - 1
        if pivot < 0: pivot = candidates.bit_length() - 1
        
        remaining = candidates & ~comp_adj[pivot]
        
        # Iterate over candidates
        while remaining:
            v = (remaining & -remaining).bit_length() - 1
            res = find_clique(candidates & comp_adj[v], current_clique + [v + 1])
            if res:
                return res
            remaining &= ~(1 << v)
            candidates &= ~(1 << v)
        return None

    result = find_clique((1 << n) - 1, [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
