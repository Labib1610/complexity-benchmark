import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find a clique of size k in the complement graph.
    # The complement graph has edges where there is no resonance.
    comp_adj = [(( (1 << n) - 1 ) ^ (1 << i) ^ adj[i]) for i in range(n)]
    
    # Bron-Kerbosch with pivoting or simple backtracking for max clique
    # Since n <= 46, we can use a meet-in-the-middle or optimized search.
    # Given the constraints and the nature of the problem, a simple recursive
    # search with pruning is often sufficient for n=46.
    
    def find_clique(candidates, current_clique):
        if len(current_clique) == k:
            return current_clique
        if not candidates or len(current_clique) + bin(candidates).count('1') < k:
            return None
        
        # Heuristic: pick vertex with most candidates
        v = (candidates & -candidates).bit_length() - 1
        
        # Try including v
        res = find_clique(candidates & comp_adj[v], current_clique + [v + 1])
        if res:
            return res
        
        # Try excluding v
        return find_clique(candidates ^ (1 << v), current_clique)

    result = find_clique((1 << n) - 1, [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
