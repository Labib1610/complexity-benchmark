import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    
    def find_independent_set(candidates, current_set):
        if len(current_set) == k:
            return current_set
        
        if not candidates:
            return None
        
        # Heuristic: pick the candidate with the fewest neighbors in the remaining candidates
        # Or simply pick the first one.
        while candidates:
            # Pruning: if remaining candidates + current_set size < k, return None
            if len(current_set) + len(candidates) < k:
                return None
            
            v = candidates[0]
            
            # Try including v
            new_candidates = [c for c in candidates[1:] if not (adj[v] & (1 << c))]
            res = find_independent_set(new_candidates, current_set + [v + 1])
            if res:
                return res
            
            # Try excluding v
            candidates = candidates[1:]
            
        return None

    result = find_independent_set(list(range(n)), [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
