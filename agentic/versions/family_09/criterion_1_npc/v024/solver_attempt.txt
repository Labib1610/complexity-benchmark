import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    # Bron-Kerbosch or simple recursive search with bitmasks.
    
    def find_is(candidates, current_set, needed):
        if needed == 0:
            return current_set
        if not candidates or bin(candidates).count('1') < needed:
            return None
        
        # Heuristic: pick the vertex with the most conflicts in candidates
        v = (candidates & -candidates).bit_length() - 1
        
        # Option 1: Include v
        res = find_is(candidates & ~adj[v] & ~(1 << v), current_set + [v + 1], needed - 1)
        if res:
            return res
        
        # Option 2: Exclude v
        res = find_is(candidates & ~(1 << v), current_set, needed)
        return res

    # To optimize, sort vertices by degree
    nodes = sorted(range(n), key=lambda x: bin(adj[x]).count('1'))
    
    # Re-map adj to sorted order
    new_adj = [0] * n
    for i in range(n):
        for j in range(n):
            if (adj[nodes[i]] >> nodes[j]) & 1:
                new_adj[i] |= (1 << j)
                
    def backtrack(idx_in_nodes, current_mask, needed):
        if needed == 0:
            return []
        if idx_in_nodes == n:
            return None
        
        # Try including nodes[idx_in_nodes]
        if not (current_mask & (1 << idx_in_nodes)):
            # Check if it conflicts with current_mask
            if not (new_adj[idx_in_nodes] & current_mask):
                res = backtrack(idx_in_nodes + 1, current_mask | (1 << idx_in_nodes), needed - 1)
                if res is not None:
                    return [nodes[idx_in_nodes]] + res
        
        # Try excluding
        # Pruning: if remaining nodes are not enough
        if (n - 1 - idx_in_nodes) >= needed:
            res = backtrack(idx_in_nodes + 1, current_mask, needed)
            if res is not None:
                return res
        return None

    result = backtrack(0, 0, k)
    
    if result:
        print("YES")
        print(*(sorted(result)))
    else:
        print("NO")

solve()
