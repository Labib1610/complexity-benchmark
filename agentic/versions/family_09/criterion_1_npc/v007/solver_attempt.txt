import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a meet-in-the-middle approach or a simple backtracking
    # with pruning. Given the constraints and the nature of the problem,
    # a backtracking search with bitmasking is efficient enough.
    
    def find_is(candidates, current_set, target_size):
        if len(current_set) == target_size:
            return current_set
        
        if not candidates:
            return None
        
        # Heuristic: pick the candidate with the fewest neighbors in the remaining candidates
        # to prune the search tree.
        while candidates:
            # Pick the last element
            v = (candidates & -candidates).bit_length() - 1
            res = find_is(candidates & ~adj[v] & ~(1 << v), current_set + [v + 1], target_size)
            if res:
                return res
            candidates &= ~(1 << v)
        return None

    # Optimization: Bron-Kerbosch style or simple backtracking
    # For n=46, simple backtracking with bitmask is usually fast.
    result = find_is((1 << n) - 1, [], k)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
