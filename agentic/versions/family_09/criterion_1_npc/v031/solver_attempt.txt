import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle: split n into n1 and n2
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the two halves
    # sets1[size] = list of bitmasks of size 'size'
    sets1 = [[] for _ in range(n1 + 1)]
    for i in range(1 << n1):
        valid = True
        count = 0
        for j in range(n1):
            if (i >> j) & 1:
                count += 1
                if adj[j] & i:
                    valid = False
                    break
        if valid:
            sets1[count].append(i)
            
    sets2 = [[] for _ in range(n2 + 1)]
    for i in range(1 << n2):
        valid = True
        count = 0
        mask = i << n1
        for j in range(n2):
            if (i >> j) & 1:
                count += 1
                if adj[j + n1] & mask:
                    valid = False
                    break
        if valid:
            sets2[count].append(i << n1)
            
    # Try to combine
    for c1 in range(n1 + 1):
        c2 = k - c1
        if 0 <= c2 <= n2:
            for m1 in sets1[c1]:
                # Find nodes in the second half that don't conflict with m1
                forbidden = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        forbidden |= adj[i]
                
                for m2 in sets2[c2]:
                    if not (m2 & forbidden):
                        # Found it
                        res = []
                        for i in range(n):
                            if ((m1 | m2) >> i) & 1:
                                res.append(str(i + 1))
                        print("YES")
                        print(" ".join(res))
                        return
    
    print("NO")

solve()
