import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle for Maximum Independent Set
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes_mask, offset):
        # Returns a dictionary: count -> list of masks
        # nodes_mask is the bitmask of nodes in this half
        nodes = [i for i in range(n) if (nodes_mask & (1 << i))]
        size = len(nodes)
        res = {}
        for i in range(1 << size):
            mask = 0
            valid = True
            count = 0
            for j in range(size):
                if (i >> j) & 1:
                    u = nodes[j]
                    if (mask & adj[u]):
                        valid = False
                        break
                    mask |= (1 << u)
                    count += 1
            if valid:
                if count not in res:
                    res[count] = []
                res[count].append(mask)
        return res

    sets1 = get_independent_sets(((1 << n1) - 1), 0)
    sets2 = get_independent_sets(((1 << n2) - 1) << n1, n1)
    
    for c1 in sets1:
        c2 = k - c1
        if c2 in sets2:
            for m1 in sets1[c1]:
                # Precompute forbidden nodes in the second half
                forbidden = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        forbidden |= adj[i]
                
                for m2 in sets2[c2]:
                    if not (m2 & forbidden):
                        # Found it
                        result = []
                        combined = m1 | m2
                        for i in range(n):
                            if (combined >> i) & 1:
                                result.append(i + 1)
                        print("YES")
                        print(*(result))
                        return
    print("NO")

solve()
