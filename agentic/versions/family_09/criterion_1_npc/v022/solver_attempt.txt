import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle for Maximum Independent Set of size k
    # n <= 46 is too large for simple 2^(n/2), but we only need size k.
    # Since we need a specific size k, we use a backtracking search with pruning.
    
    def find_clique(candidates, current_set, k_needed):
        if k_needed == 0:
            return current_set
        if not candidates or len(candidates) < k_needed:
            return None
        
        # Heuristic: pick the node with the most constraints to prune faster
        # Or simply pick the first one
        v = candidates[0]
        
        # Option 1: Include v
        new_candidates = [c for c in candidates[1:] if not (adj[v] & (1 << c))]
        res = find_clique(new_candidates, current_set + [v + 1], k_needed - 1)
        if res:
            return res
        
        # Option 2: Exclude v
        res = find_clique(candidates[1:], current_set, k_needed)
        return res

    # Sort vertices by degree to improve pruning
    nodes = sorted(range(n), key=lambda x: bin(adj[x]).count('1'))
    
    result = find_clique(nodes, [], k)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
