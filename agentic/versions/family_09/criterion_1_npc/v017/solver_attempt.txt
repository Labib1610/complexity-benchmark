import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a meet-in-the-middle approach or a 
    # backtracking algorithm with pruning.
    
    # Split into two halves for meet-in-the-middle
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for both halves
    def get_independent_sets(nodes, adj_list):
        # returns a dict: size -> list of bitmasks
        res = [[] for _ in range(len(nodes) + 1)]
        res[0].append(0)
        
        for i, node in enumerate(nodes):
            new_res = [list(x) for x in res]
            mask = 0
            for j in range(i):
                if (adj_list[node] >> nodes[j]) & 1:
                    mask |= (1 << j)
            
            for size in range(len(nodes)):
                for prev_mask in res[size]:
                    if not (prev_mask & mask):
                        new_res[size + 1].append(prev_mask | (1 << i))
            res = new_res
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    sets1 = get_independent_sets(nodes1, adj)
    sets2 = get_independent_sets(nodes2, adj)
    
    # Precompute adjacency between halves
    adj_cross = [0] * n1
    for i in range(n1):
        for j in range(n2):
            if (adj[i] >> (n1 + j)) & 1:
                adj_cross[i] |= (1 << j)
                
    for s1 in range(k + 1):
        s2 = k - s1
        if s1 < len(sets1) and s2 < len(sets2):
            for mask1 in sets1[s1]:
                # Calculate forbidden mask in set2
                forbidden = 0
                for i in range(n1):
                    if (mask1 >> i) & 1:
                        forbidden |= adj_cross[i]
                
                for mask2 in sets2[s2]:
                    if not (mask2 & forbidden):
                        # Found it
                        res = []
                        for i in range(n1):
                            if (mask1 >> i) & 1:
                                res.append(i + 1)
                        for i in range(n2):
                            if (mask2 >> i) & 1:
                                res.append(n1 + i + 1)
                        print("YES")
                        print(*(res))
                        return
    print("NO")

solve()
