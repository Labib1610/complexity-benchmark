import sys

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1] |= (1 << (v-1))
        adj[v-1] |= (1 << (u-1))

    # Meet-in-the-middle approach for Maximum Independent Set of size k
    # n <= 46, split into n1 = 23, n2 = 23
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the first half
    # sets1[size] = list of bitmasks of size 'size'
    sets1 = [[] for _ in range(n1 + 1)]
    for i in range(1 << n1):
        is_independent = True
        count = 0
        for j in range(n1):
            if (i >> j) & 1:
                count += 1
                if adj[j] & i:
                    is_independent = False
                    break
        if is_independent:
            sets1[count].append(i)
            
    # Precompute independent sets for the second half
    # We need to store masks that are compatible with a mask from the first half
    # For a mask m1 from first half, we need m2 from second half such that:
    # 1. m2 is independent
    # 2. No node in m2 is connected to any node in m1
    
    # Precompute compatibility: for each mask m1, which nodes in second half are forbidden?
    # forbidden[m1] = bitmask of nodes in second half connected to any node in m1
    
    def get_forbidden(m1):
        f = 0
        for j in range(n1):
            if (m1 >> j) & 1:
                f |= (adj[j] >> n1)
        return f

    for s1 in range(min(k + 1, n1 + 1)):
        s2 = k - s1
        if s2 < 0 or s2 > n2:
            continue
        
        # Group second half masks by their forbidden mask
        # Actually, just iterate through all independent sets of size s2
        # and check if they are compatible with any m1 in sets1[s1]
        
        # Optimization: iterate through all masks of size s2
        for i2 in range(1 << n2):
            count2 = bin(i2).count('1')
            if count2 != s2:
                continue
            
            is_independent2 = True
            for j in range(n2):
                if (i2 >> j) & 1:
                    if (adj[j + n1] >> n1) & i2:
                        is_independent2 = False
                        break
            if not is_independent2:
                continue
            
            # Check compatibility with any m1 in sets1[s1]
            forbidden_by_i2 = 0
            for j in range(n2):
                if (i2 >> j) & 1:
                    forbidden_by_i2 |= (adj[j + n1] & ((1 << n1) - 1))
            
            for m1 in sets1[s1]:
                if not (m1 & forbidden_by_i2):
                    # Found it
                    res = []
                    for j in range(n1):
                        if (m1 >> j) & 1: res.append(j + 1)
                    for j in range(n2):
                        if (i2 >> j) & 1: res.append(j + n1 + 1)
                    print("YES")
                    print(*(res))
                    return

    print("NO")

solve()
