import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle for Maximum Independent Set of size k
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes, offset):
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            subset = []
            valid = True
            for j in range(size):
                if (i >> j) & 1:
                    u = nodes[j]
                    for v in subset:
                        if (adj[u + offset] >> (v + offset)) & 1:
                            valid = False
                            break
                    if not valid: break
                    subset.append(u + offset)
            if valid:
                res[len(subset)].append((i, set(subset)))
        return res

    left_nodes = list(range(n1))
    right_nodes = list(range(n1, n))
    
    left_sets = get_independent_sets(left_nodes, 0)
    right_sets = get_independent_sets(right_nodes, n1)
    
    # Precompute adjacency masks for right side
    right_masks = []
    for i in range(1 << n2):
        mask = 0
        for j in range(n2):
            if (i >> j) & 1:
                mask |= adj[j + n1]
        right_masks.append(mask)

    for i in range(max(0, k - n2), min(k, n1) + 1):
        j = k - i
        for l_mask, l_set in left_sets[i]:
            # Check which right sets are compatible with l_set
            l_adj_mask = 0
            for node in l_set:
                l_adj_mask |= adj[node]
            
            for r_mask, r_set in right_sets[j]:
                if not (l_adj_mask & (r_mask << n1)):
                    print("YES")
                    print(*(sorted([x + 1 for x in list(l_set) + list(r_set)])))
                    return

    print("NO")

solve()
