import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find a clique of size k in the complement graph.
    # The complement graph has an edge between u and v if they are NOT in adj.
    # Let's use Bron-Kerbosch or a simple backtracking with pruning.
    # Since n <= 46, we can use a bitmask-based backtracking.
    
    comp_adj = [((1 << n) - 1) ^ (1 << i) ^ adj[i] for i in range(n)]
    
    def find_clique(candidates, current_clique):
        if len(current_clique) == k:
            return current_clique
        
        if not candidates:
            return None
        
        # Heuristic: pick the node with the most candidates to prune faster
        while candidates:
            # Pick node with highest index to explore
            v = (candidates & -candidates).bit_length() - 1
            
            res = find_clique(candidates & comp_adj[v], current_clique + [v + 1])
            if res:
                return res
            
            candidates &= ~(1 << v)
        return None

    result = find_clique((1 << n) - 1, [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
