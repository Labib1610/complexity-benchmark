import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle approach for Maximum Independent Set
    # Split n into two halves: n1 = n // 2, n2 = n - n1
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the second half
    # subsets2[count] stores a list of bitmasks of size 'count'
    subsets2 = [[] for _ in range(n2 + 1)]
    for i in range(1 << n2):
        valid = True
        count = 0
        for j in range(n2):
            if (i >> j) & 1:
                count += 1
                if (adj[n1 + j] >> n1) & i:
                    valid = False
                    break
        if valid:
            # Store the mask and the forbidden mask in the first half
            forbidden_in_first = 0
            for j in range(n2):
                if (i >> j) & 1:
                    forbidden_in_first |= adj[n1 + j]
            subsets2[count].append((i, forbidden_in_first & ((1 << n1) - 1)))
            
    # Search in the first half
    for i in range(1 << n1):
        valid = True
        count1 = 0
        forbidden_in_second = 0
        for j in range(n1):
            if (i >> j) & 1:
                count1 += 1
                if adj[j] & i:
                    valid = False
                    break
                forbidden_in_second |= adj[j]
        
        if not valid:
            continue
            
        needed = k - count1
        if 0 <= needed <= n2:
            forbidden_mask = (forbidden_in_second >> n1)
            for mask2, forbidden_in_first in subsets2[needed]:
                if not (mask2 & forbidden_mask) and not (i & forbidden_in_first):
                    # Found it
                    res = []
                    for j in range(n1):
                        if (i >> j) & 1:
                            res.append(j + 1)
                    for j in range(n2):
                        if (mask2 >> j) & 1:
                            res.append(n1 + j + 1)
                    print("YES")
                    print(*(res))
                    return

    print("NO")

solve()
