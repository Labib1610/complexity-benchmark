import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle: split n into n1 and n2
    n1 = n // 2
    n2 = n - n1
    
    # Precompute independent sets for the two halves
    # sets1[size] = list of bitmasks of size 'size'
    sets1 = [[] for _ in range(n1 + 1)]
    for i in range(1 << n1):
        valid = True
        for j in range(n1):
            if (i >> j) & 1:
                if adj[j] & i:
                    valid = False
                    break
        if valid:
            sets1[bin(i).count('1')].append(i)
            
    sets2 = [[] for _ in range(n2 + 1)]
    for i in range(1 << n2):
        valid = True
        for j in range(n2):
            if (i >> j) & 1:
                if (adj[j + n1] >> n1) & i:
                    valid = False
                    break
        if valid:
            sets2[bin(i).count('1')].append(i)
            
    # Precompute adjacency constraints for the second half
    # adj_mask[i] is the mask of nodes in the first half that interfere with node i+n1
    adj_mask = [(adj[i + n1] & ((1 << n1) - 1)) for i in range(n2)]
    
    for s1 in range(k + 1):
        s2 = k - s1
        if s1 <= n1 and s2 <= n2:
            for mask1 in sets1[s1]:
                # Find nodes in the second half that don't interfere with mask1
                allowed_mask2 = (1 << n2) - 1
                for j in range(n1):
                    if (mask1 >> j) & 1:
                        allowed_mask2 &= ~(adj[j] >> n1)
                
                for mask2 in sets2[s2]:
                    if (mask2 & allowed_mask2) == mask2:
                        # Found it
                        res = []
                        for i in range(n1):
                            if (mask1 >> i) & 1:
                                res.append(i + 1)
                        for i in range(n2):
                            if (mask2 >> i) & 1:
                                res.append(i + n1 + 1)
                        print("YES")
                        print(*(res))
                        return
    print("NO")

solve()
