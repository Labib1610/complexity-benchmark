import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Meet-in-the-middle for Maximum Independent Set / k-clique
    # n <= 46 is small enough for a meet-in-the-middle approach
    n1 = n // 2
    n2 = n - n1
    
    # adj1: adjacency for first n1 nodes, adj2: for remaining n2 nodes
    # mask1: bitmask of nodes in first half, mask2: bitmask of nodes in second half
    
    def get_subsets(nodes, adj_list):
        # Returns a dictionary: size -> list of bitmasks
        size = len(nodes)
        res = [[] for _ in range(size + 1)]
        
        def backtrack(idx, current_mask, current_size):
            if idx == size:
                res[current_size].append(current_mask)
                return
            
            # Try including nodes[idx]
            node = nodes[idx]
            if not (current_mask & adj_list[node]):
                backtrack(idx + 1, current_mask | (1 << node), current_size + 1)
            
            # Try excluding nodes[idx]
            backtrack(idx + 1, current_mask, current_size)
            
        backtrack(0, 0, 0)
        return res

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    subsets1 = get_subsets(nodes1, adj)
    subsets2 = get_subsets(nodes2, adj)
    
    # Precompute cross-collisions
    # cross_adj[i] is the mask of nodes in nodes2 that collide with node i in nodes1
    cross_adj = [0] * n1
    for i in range(n1):
        cross_adj[i] = adj[i] >> n1
        
    for s1 in range(k + 1):
        s2 = k - s1
        if s1 < len(subsets1) and s2 < len(subsets2):
            for m1 in subsets1[s1]:
                # Find nodes in nodes2 that don't collide with m1
                forbidden = 0
                for i in range(n1):
                    if (m1 >> i) & 1:
                        forbidden |= cross_adj[i]
                
                for m2 in subsets2[s2]:
                    if not (m2 & (forbidden << n1)):
                        # Found it
                        res = []
                        for i in range(n1):
                            if (m1 >> i) & 1: res.append(i + 1)
                        for i in range(n2):
                            if (m2 >> (i + n1)) & 1: res.append(i + n1 + 1)
                        print("YES")
                        print(*(res))
                        return

    print("NO")

solve()
