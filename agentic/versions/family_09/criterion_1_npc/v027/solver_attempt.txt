import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    n1 = n // 2
    n2 = n - n1
    
    def get_subsets(nodes, adj_list):
        size = len(nodes)
        subsets = [[] for _ in range(size + 1)]
        for i in range(1 << size):
            mask = 0
            valid = True
            count = 0
            for j in range(size):
                if (i >> j) & 1:
                    if mask & adj_list[nodes[j]]:
                        valid = False
                        break
                    mask |= (1 << nodes[j])
                    count += 1
            if valid:
                subsets[count].append(mask)
        return subsets

    nodes1 = list(range(n1))
    nodes2 = list(range(n1, n))
    
    subsets1 = get_subsets(nodes1, adj)
    subsets2 = get_subsets(nodes2, adj)
    
    for c1 in range(min(k + 1, n1 + 1)):
        c2 = k - c1
        if c2 < 0 or c2 > n2:
            continue
        
        for m1 in subsets1[c1]:
            # Pre-filter m2 candidates
            forbidden = 0
            for i in range(n1):
                if (m1 >> i) & 1:
                    forbidden |= adj[i]
            
            for m2 in subsets2[c2]:
                if not (m2 & forbidden):
                    print("YES")
                    res = []
                    for i in range(n):
                        if ((m1 | m2) >> i) & 1:
                            res.append(str(i + 1))
                    print(" ".join(res))
                    return
                    
    print("NO")

solve()
