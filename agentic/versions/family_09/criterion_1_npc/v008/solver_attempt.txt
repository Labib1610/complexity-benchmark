import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a meet-in-the-middle approach or 
    # a simple backtracking with pruning.
    
    memo = {}

    def find_is(candidates, target_k):
        if target_k == 0:
            return []
        if not candidates:
            return None
        
        state = (candidates, target_k)
        if state in memo:
            return memo[state]
        
        # Heuristic: pick the node with the most neighbors in candidates to prune
        v = (candidates & -candidates).bit_length() - 1
        
        # Option 1: Include v
        new_candidates = candidates & ~adj[v] & ~(1 << v)
        res = find_is(new_candidates, target_k - 1)
        if res is not None:
            memo[state] = [v + 1] + res
            return memo[state]
        
        # Option 2: Exclude v
        res = find_is(candidates & ~(1 << v), target_k)
        memo[state] = res
        return res

    # For n=46, simple backtracking might be slow, but the constraints 
    # on k and the nature of independent sets often allow it.
    # Let's use a more efficient Bron-Kerbosch style approach.
    
    def get_independent_set(candidates, k):
        if k == 0:
            return []
        if not candidates:
            return None
        
        # Pivot to reduce branching
        pivot = (candidates & -candidates).bit_length() - 1
        
        # Try nodes not connected to pivot
        nodes = []
        temp = candidates & ~adj[pivot]
        while temp:
            v = (temp & -temp).bit_length() - 1
            nodes.append(v)
            temp &= ~(1 << v)
            
        for v in nodes:
            res = get_independent_set(candidates & ~adj[v] & ~(1 << v), k - 1)
            if res is not None:
                return [v + 1] + res
        return None

    result = get_independent_set((1 << n) - 1, k)
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
