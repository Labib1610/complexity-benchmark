import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a backtracking approach with pruning.
    
    def find_independent_set(candidates, current_set):
        if len(current_set) == k:
            return current_set
        
        if len(current_set) + bin(candidates).count('1') < k:
            return None
        
        while candidates:
            # Pick the node with the fewest candidates to branch on (heuristic)
            v = (candidates & -candidates).bit_length() - 1
            
            res = find_independent_set(candidates & ~adj[v] & ~(1 << v), current_set + [v + 1])
            if res:
                return res
            
            candidates &= ~(1 << v)
            
        return None

    result = find_independent_set((1 << n) - 1, [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
