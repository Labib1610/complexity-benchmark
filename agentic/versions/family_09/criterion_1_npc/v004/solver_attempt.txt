import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We need to find an independent set of size k.
    # Since n <= 46, we can use a meet-in-the-middle approach or 
    # a simple backtracking with pruning.
    
    def find_is(candidates, current_set):
        if len(current_set) == k:
            return current_set
        
        if not candidates:
            return None
        
        # Heuristic: pick the node with the most neighbors in candidates to prune
        # Or just pick the first one
        v = (candidates & -candidates).bit_length() - 1
        
        # Option 1: Include v
        new_candidates = candidates & ~adj[v] & ~(1 << v)
        res = find_is(new_candidates, current_set + [v + 1])
        if res:
            return res
            
        # Option 2: Exclude v
        res = find_is(candidates & ~(1 << v), current_set)
        return res

    # Optimization: Bron-Kerbosch style or simple backtracking
    # Given n=46, simple backtracking with bitmask is efficient enough.
    
    def backtrack(candidates, current_set):
        if len(current_set) == k:
            return current_set
        if len(current_set) + bin(candidates).count('1') < k:
            return None
        
        while candidates:
            v = (candidates & -candidates).bit_length() - 1
            res = backtrack(candidates & ~adj[v] & ~(1 << v), current_set + [v + 1])
            if res:
                return res
            candidates &= ~(1 << v)
        return None

    result = backtrack((1 << n) - 1, [])
    
    if result:
        print("YES")
        print(*(result))
    else:
        print("NO")

solve()
