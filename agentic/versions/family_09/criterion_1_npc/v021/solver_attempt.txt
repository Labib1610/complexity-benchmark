import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # A NAND gate (u, v) outputs HIGH unless both u and v are HIGH.
    # This means we cannot have both u and v in our set of k pins.
    # This is equivalent to finding an Independent Set of size k in the graph.
    
    # Since n <= 46, we can use a meet-in-the-middle approach for Independent Set.
    n1 = n // 2
    n2 = n - n1
    
    def get_independent_sets(nodes):
        size = len(nodes)
        res = {} # mask -> count
        for i in range(1 << size):
            subset = []
            for j in range(size):
                if (i >> j) & 1:
                    subset.append(nodes[j])
            
            valid = True
            for idx1 in range(len(subset)):
                for idx2 in range(idx1 + 1, len(subset)):
                    if subset[idx2] in adj[subset[idx1]]:
                        valid = False
                        break
                if not valid: break
            
            if valid:
                res[i] = len(subset)
        return res

    nodes1 = list(range(1, n1 + 1))
    nodes2 = list(range(n1 + 1, n + 1))
    
    sets1 = get_independent_sets(nodes1)
    sets2 = get_independent_sets(nodes2)
    
    # Precompute adjacency for cross-checking
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for mask1, count1 in sets1.items():
        if count1 > k: continue
        needed = k - count1
        
        # Check if mask1 is compatible with mask2
        subset1 = [nodes1[j] for j in range(n1) if (mask1 >> j) & 1]
        
        for mask2, count2 in sets2.items():
            if count2 == needed:
                subset2 = [nodes2[j] for j in range(n2) if (mask2 >> j) & 1]
                
                # Check cross edges
                possible = True
                for u in subset1:
                    for v in subset2:
                        if v in adj_sets[u]:
                            possible = False
                            break
                    if not possible: break
                
                if possible:
                    print("YES")
                    result = subset1 + subset2
                    print(*(result))
                    return

    print("NO")

solve()
