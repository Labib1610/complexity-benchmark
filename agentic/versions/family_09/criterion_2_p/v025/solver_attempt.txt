import sys

# The problem states that the graph is bipartite.
# We can find the two partitions using BFS/DFS.
# Let the two partitions be A and B with sizes |A| and |B|.
# We need to select k modules such that no two are incompatible.
# This means we can pick some x modules from A and y modules from B
# such that x + y = k, and no edges exist between the chosen x and y.
# Since the graph is bipartite, any subset of A is an independent set,
# and any subset of B is an independent set.
# However, if we pick a subset of A and a subset of B, we must ensure
# there are no edges between them.
# Actually, the problem is simpler: we want to pick k vertices such that
# no two are connected. This is the Maximum Independent Set problem.
# For a bipartite graph, the size of the Maximum Independent Set is n - (size of Maximum Matching).
# But we need exactly k.
# Since the graph is bipartite, we can pick any subset of A and any subset of B
# as long as there are no edges between them.
# Actually, the simplest independent sets are just A itself and B itself.
# Any subset of A is an independent set, and any subset of B is an independent set.
# If we pick x from A and y from B, we need no edges between them.
# This is equivalent to saying we pick a subset of A and a subset of B
# such that no edges exist between them.
# The largest independent set is either A or B or some combination.
# Actually, in a bipartite graph, the maximum independent set is found by
# taking all vertices in one partition and the vertices in the other
# that are not covered by the maximum matching.
# Let the partitions be V1 and V2.
# Any subset of V1 is independent. Any subset of V2 is independent.
# We can pick any k vertices from V1 if |V1| >= k.
# We can pick any k vertices from V2 if |V2| >= k.
# If we need to pick from both, we must ensure no edges.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    v1, v2 = [], []
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: v1.append(u)
                else: v2.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # We can always pick up to |V1| vertices from V1 or |V2| from V2.
    # The maximum independent set size is n - max_matching.
    # But we just need to check if we can pick k.
    # Actually, any subset of V1 is independent, any subset of V2 is independent.
    # If we pick k vertices, we can just pick them all from V1 if |V1| >= k,
    # or all from V2 if |V2| >= k.
    
    if len(v1) >= k:
        print("YES")
        print(*(v1[:k]))
    elif len(v2) >= k:
        print("YES")
        print(*(v2[:k]))
    else:
        print("NO")

solve()
