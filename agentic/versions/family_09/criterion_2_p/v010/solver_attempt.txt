import sys

# The problem states that interference links only exist between nodes of different channels.
# This means the graph is bipartite. Let the two sets of the bipartition be A and B.
# An interference-free set is an independent set.
# In a bipartite graph, any subset of A is an independent set, and any subset of B is an independent set.
# Since there are no edges within A and no edges within B, we can pick any number of nodes from A
# and any number of nodes from B, provided we don't pick two nodes connected by an edge.
# However, the problem asks for an interference-free set of size k.
# Since all edges are between A and B, any subset of A is independent, and any subset of B is independent.
# Let |A| = nA and |B| = nB. We can form an independent set of size k if k <= max(nA, nB).
# Wait, that's not quite right. We can pick any subset of A (size up to nA) OR any subset of B (size up to nB).
# Actually, we can pick any subset of A and any subset of B such that no edges exist between them.
# But since all edges are between A and B, if we pick a subset S_A from A and S_B from B,
# we must ensure no node in S_A is connected to any node in S_B.
# This is equivalent to saying we can pick all of A or all of B.
# Actually, the largest independent set in a bipartite graph is the size of the larger partition.
# Wait, the problem says "interference links only ever exist between nodes configured to different channels".
# This means the graph is bipartite with partitions A and B.
# Any subset of A is independent. Any subset of B is independent.
# Can we pick some from A and some from B? Only if there are no edges between the chosen nodes.
# But the problem asks for *an* interference-free set.
# The simplest interference-free sets are just subsets of A or subsets of B.
# If k <= |A|, we can pick k nodes from A. If k <= |B|, we can pick k nodes from B.
# If k > max(|A|, |B|), can we pick more? No, because the maximum independent set size in a bipartite graph
# is n - (size of maximum matching). By Konig's theorem, this is n - (size of min vertex cover).
# But we don't need the maximum independent set, just any set of size k.
# If k <= max(|A|, |B|), we can definitely pick k nodes from the larger partition.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    part_a = []
    part_b = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0:
                    part_a.append(u)
                else:
                    part_b.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    if k <= len(part_a):
        print("YES")
        print(*(part_a[:k]))
    elif k <= len(part_b):
        print("YES")
        print(*(part_b[:k]))
    else:
        print("NO")

solve()
