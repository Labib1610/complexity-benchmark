```python
import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_nodes = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            comp_nodes[0].append(i)
            
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        comp_nodes[1 - c].append(v)
                        stack.append((v, 1 - c))
            components.append(comp_nodes)
            
    # dp[i][j] = can we get j antennas from first i components
    # Since we need to reconstruct, we store which choice was made
    # dp[j] stores a list of (count_from_set0, count_from_set1)
    # Actually, we can use a bitset-like approach or simple DP
    # Since n is 10^5, we use a dictionary or array to track reachable sums
    
    # dp[j] = (c0, c1) where c0 is count from set0, c1 from set1
    # To optimize, we only need to know if sum is possible
    # dp[sum] = (index_of_component, choice_0_or_1)
    
    dp = {0: []}
    for i, (s0, s1) in enumerate(components):
        new_dp = dp.copy()
        len0, len1 = len(s0), len(s1)
        for current_sum in dp:
            if len0 > 0:
                new_sum = current_sum + len0
                if new_sum not in new_dp:
                    new_dp[new_sum] = (i, 0)
            if len1 > 0:
                new_sum = current_sum + len1
                if new_sum not in new_dp:
                    new_dp[new_sum] = (i, 1)
        dp = new_dp
        
    if k not in dp:
        print("NO")
        return
    
    # Reconstruct
    res = []
    curr = k
    while curr > 0:
        comp_idx, choice = dp[curr]
        comp = components[comp_idx]
        res.extend(comp[choice])
        curr -= len(comp[choice])
        # Remove to avoid re-processing
        del dp[curr + len(comp[choice])]
        
    print("YES")
    print(*(res))

solve()
```
