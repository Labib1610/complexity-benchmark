```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The graph is bipartite. Let the two camps be C1 and C2.
    # A group is all-strangers if it contains only nodes from C1 or only nodes from C2.
    # We need to find if |C1| >= k or |C2| >= k.
    
    color = [-1] * (n + 1)
    c1 = []
    c2 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            c1.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            c1.append(v)
                        else:
                            c2.append(v)
                        stack.append((v, 1 - c))
    
    if len(c1) >= k:
        print("YES")
        print(*(c1[:k]))
    elif len(c2) >= k:
        print("YES")
        print(*(c2[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
