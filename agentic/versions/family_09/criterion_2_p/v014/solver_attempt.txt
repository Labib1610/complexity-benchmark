import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The graph is bipartite. We can find the two partitions using BFS/DFS.
    # Since the graph might be disconnected, we process each component.
    color = [-1] * (n + 1)
    partition0 = []
    partition1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            partition0.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            partition0.append(v)
                        else:
                            partition1.append(v)
                        stack.append((v, 1 - c))
    
    # We need to pick k nodes such that no two are connected.
    # This means we can pick any subset of partition0 OR any subset of partition1.
    # Wait, that's not entirely true. We can pick a subset of partition0 AND 
    # a subset of partition1 as long as no edges exist between them.
    # But the problem says "no two accounts within it share an interaction".
    # Since all edges are between partition0 and partition1, any subset of 
    # partition0 is an independent set, and any subset of partition1 is an independent set.
    # Can we mix them? Only if there are no edges between the chosen nodes.
    # Actually, the problem is equivalent to finding an independent set of size k.
    # In a bipartite graph, the maximum independent set is n - (size of min vertex cover).
    # However, the problem guarantees the graph is bipartite.
    # Any independent set is a subset of nodes such that no two are connected.
    # In a bipartite graph, the largest independent set is the union of the 
    # larger partition and any nodes from the other partition that have no edges 
    # to the chosen nodes.
    
    # Actually, the simplest approach:
    # The set of all nodes in partition0 is an independent set.
    # The set of all nodes in partition1 is an independent set.
    # Any subset of partition0 is an independent set.
    # Any subset of partition1 is an independent set.
    # Can we pick nodes from both? Only if there are no edges between them.
    # But the graph is bipartite, so edges only exist between P0 and P1.
    # If we pick a set S = S0 U S1 where S0 subset P0 and S1 subset P1,
    # S is independent iff there are no edges between any u in S0 and v in S1.
    
    # Let's simplify: The maximum independent set in a bipartite graph is 
    # found by taking all nodes in the larger partition, plus nodes in the 
    # smaller partition that are not connected to any nodes in the larger partition.
    # Actually, just check if len(partition0) >= k or len(partition1) >= k.
    # Is it possible to have a larger independent set by mixing?
    # Yes, if there are isolated vertices.
    
    # Let's use the property: Max Independent Set = N - Min Vertex Cover.
    # For bipartite, Min Vertex Cover = Max Matching (Konig's theorem).
    # But we just need *any* independent set of size k.
    # Let's find the maximum independent set.
    
    # Actually, the problem is simpler:
    # Any subset of P0 is independent. Any subset of P1 is independent.
    # If we take all nodes in P0 and all nodes in P1 that have no edges to P0,
    # that is an independent set.
    
    p0_set = set(partition0)
    p1_set = set(partition1)
    
    # Nodes in P1 that have no edges to P0
    p1_isolated = [v for v in p1_set if not adj[v]]
    # Nodes in P0 that have no edges to P1
    p0_isolated = [u for u in p0_set if not adj[u]]
    
    # Independent set 1: P0 + p1_isolated
    indep1 = partition0 + p1_isolated
    # Independent set 2: P1 + p0_isolated
    indep2 = partition1 + p0_isolated
    
    if len(indep1) >= k:
        print("YES")
        print(*(indep1[:k]))
    elif len(indep2) >= k:
        print("YES")
        print(*(indep2[:k]))
    else:
        print("NO")

solve()
