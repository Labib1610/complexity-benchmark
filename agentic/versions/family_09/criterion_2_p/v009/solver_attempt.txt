```python
import sys

# The problem states the graph is bipartite.
# An all-strangers group is an independent set.
# In a bipartite graph, the two partitions (let's call them A and B) are 
# independent sets. Any subset of A is an independent set, and any subset of B 
# is an independent set.
# Since the graph is bipartite, any independent set is a subset of one of the 
# two partitions if the graph is connected. If the graph has multiple connected 
# components, each component has two partitions.
# Let the components be C_1, C_2, ..., C_p.
# Each component C_i has two partitions A_i and B_i.
# We can choose either A_i or B_i from each component to form an independent set.
# We want to pick a collection of sets {S_1, S_2, ..., S_p} where S_i is either 
# A_i or B_i, such that the sum of sizes is k.
# This is a variation of the subset sum problem (specifically, the partition 
# problem variant). Since we have many components, we can use dynamic 
# programming with bitsets or a frequency array to solve it efficiently.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_a = []
            comp_b = []
            stack = [(i, 0)]
            color[i] = 0
            comp_a.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0: comp_a.append(v)
                        else: comp_b.append(v)
                        stack.append((v, 1 - c))
            components.append((comp_a, comp_b))
            
    # dp[i] stores which choice was made to reach sum i
    # dp[i] = (component_index, choice_0_or_1)
    dp = {0: None}
    
    for i, (a, b) in enumerate(components):
        new_dp = dp.copy()
        len_a, len_b = len(a), len(b)
        for s in dp:
            if s + len_a <= k and (s + len_a) not in new_dp:
                new_dp[s + len_a] = (i, 0)
            if s + len_b <= k and (s + len_b) not in new_dp:
                new_dp[s + len_b] = (i, 1)
        dp = new_dp
        
    if k not in dp:
        print("NO")
        return
    
    print("YES")
    res = []
    curr = k
    while curr > 0:
        comp_idx, choice = dp[curr]
        comp_a, comp_b = components[comp_idx]
        if choice == 0:
            res.extend(comp_a)
            curr -= len(comp_a)
        else:
            res.extend(comp_b)
            curr -= len(comp_b)
    print(*(res))

solve()
```
