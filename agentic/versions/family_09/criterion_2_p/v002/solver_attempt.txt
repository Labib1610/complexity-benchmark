import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The graph is bipartite. We need to find the two sets.
    color = [0] * (n + 1)
    sets = [[], []]
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                sets[c - 1].append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
    
    # In a bipartite graph, an interference-free group is a subset of vertices
    # such that no two are connected. Since all edges are between set 0 and set 1,
    # any subset of set 0 is interference-free, and any subset of set 1 is interference-free.
    # However, we can pick some from set 0 and some from set 1 as long as no edges exist.
    # But wait, the problem says "every interference link connects a device of one type to a device of the other".
    # This means the graph is bipartite. Any independent set in a bipartite graph
    # can be found by picking vertices from both sides.
    # Actually, the maximum independent set in a bipartite graph is n - (size of min vertex cover).
    # By Konig's theorem, min vertex cover = max matching.
    # So max independent set = n - max matching.
    
    # Let's use Hopcroft-Karp or simple DFS for max matching
    match = [0] * (n + 1)
    vis = [False] * (n + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] == 0 or dfs(match[v], visited):
                    match[v] = u
                    match[u] = v
                    return True
        return False

    # Max matching
    matching_size = 0
    for u in sets[0]:
        visited = [False] * (n + 1)
        if dfs(u, visited):
            matching_size += 1
            
    # Max independent set construction:
    # 1. Find min vertex cover:
    #    Let U be set 0, V be set 1.
    #    Z = set of unmatched vertices in U + vertices reachable from them via alternating paths.
    #    Min Vertex Cover K = (U \ Z) U (V intersect Z)
    #    Max Independent Set I = (U intersect Z) U (V \ Z)
    
    z = [False] * (n + 1)
    def dfs_z(u):
        z[u] = True
        for v in adj[u]:
            if match[u] != v and not z[v]:
                dfs_z(v)
                
    unmatched_u = [u for u in sets[0] if match[u] == 0]
    for u in unmatched_u:
        if not z[u]:
            dfs_z(u)
            
    independent_set = []
    for u in sets[0]:
        if z[u]: independent_set.append(u)
    for v in sets[1]:
        if not z[v]: independent_set.append(v)
        
    if len(independent_set) >= k:
        print("YES")
        print(*(independent_set[:k]))
    else:
        print("NO")

solve()
