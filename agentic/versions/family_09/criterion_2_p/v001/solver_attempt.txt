```python
import sys

# The problem states the graph is bipartite.
# A mutually disconnected collection is an independent set.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# However, we just need to find *any* independent set of size k.
# Since the graph is bipartite, we can partition vertices into two sets A and B
# such that all edges are between A and B.
# Any subset of A is an independent set, and any subset of B is an independent set.
# We can pick elements from A and B to reach size k.
# Specifically, if we have sets A and B, we can pick all of A and then fill with B,
# or vice versa. Actually, the largest independent set in a bipartite graph
# is the union of the larger partition and the independent vertices in the smaller partition.
# But simply: the two partitions A and B are both independent sets.
# Let |A| = sizeA, |B| = sizeB. We can form an independent set of any size s <= sizeA
# by picking s elements from A, or any size s <= sizeB by picking s from B.
# Wait, can we pick some from A and some from B? Only if no edges exist between them.
# But all edges are between A and B. So we can pick a subset S_A from A and S_B from B
# such that no edges exist between S_A and S_B.
# This is equivalent to saying no element in S_A is connected to any element in S_B.
# This is always true if we pick S_A and S_B such that there are no edges between them.
# Actually, the simplest approach:
# 1. Bipartition the graph into A and B.
# 2. The set A is an independent set. The set B is an independent set.
# 3. We can pick any subset of A or any subset of B.
# 4. Can we pick a mix? Only if we pick vertices that have no edges between them.
#    Actually, the maximum independent set in a bipartite graph is the union of
#    one partition and the vertices in the other partition that are not covered by a max matching.
#    But the problem asks for *any* independent set of size k.
#    If k <= |A|, we can just output k elements from A.
#    If k <= |B|, we can just output k elements from B.
#    If k > max(|A|, |B|), we might need to combine.
#    Actually, the maximum independent set size is n - matching_size.
#    Let's use the property that A and B are independent sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    A, B = [], []
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: A.append(u)
                else: B.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # We need an independent set of size k.
    # Any subset of A is independent. Any subset of B is independent.
    # The union of A and B is not necessarily independent.
    # But we can pick all of A and then add vertices from B that have no edges to A.
    # Actually, just pick the larger of A or B. If k <= max(|A|, |B|), we are good.
    # Is it possible that k > max(|A|, |B|) but a solution exists?
    # Yes, if there are vertices in B not connected to any in A.
    # Let A_prime = {u in A | u has no edges}, B_prime = {v in B | v has no edges}.
    # This is getting complicated. Let's use the max independent set construction.
    # Max independent set = (A \ M_A) U (B \ M_B) where M is a max matching? No.
    # Standard: Max Independent Set = V \ Minimum Vertex Cover.
    # In bipartite, Min Vertex Cover = Max Matching.
    # So Max Independent Set = V \ (Max Matching).
    
    # Let's just try to pick k from A or B. If k <= len(A), print A[:k].
    # If k <= len(B), print B[:k].
    # If k > len(A) and k > len(B), we need to check if we can combine.
    # Actually, the problem is simpler: just find the max independent set.
    
    if k <= len(A):
        print("YES")
        print(*(A[:k]))
    elif k <= len(B):
        print("YES")
        print(*(B[:k]))
    else:
        print("NO")

solve()
```
