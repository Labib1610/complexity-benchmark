```python
import sys

# The problem states the graph is bipartite (tier 0 and tier 1).
# A safe cluster is an independent set.
# In a bipartite graph, the two sets of the partition (let's call them S0 and S1)
# are both independent sets.
# Any subset of S0 is an independent set, and any subset of S1 is an independent set.
# We need to find if we can pick k nodes such that they form an independent set.
# Since the graph is bipartite, we can partition the vertices into two sets V0 and V1.
# Any subset of V0 is an independent set, and any subset of V1 is an independent set.
# If |V0| >= k, we can pick any k nodes from V0.
# If |V1| >= k, we can pick any k nodes from V1.
# If both |V0| < k and |V1| < k, can we pick k nodes?
# Actually, the problem asks for *a* safe cluster. If we can pick k nodes from V0 or V1,
# we are done. If not, is it possible to pick k nodes that are not entirely in V0 or V1?
# Yes, but the problem is simpler: if we can pick k nodes from V0 or V1, we satisfy the condition.
# Is it possible that we cannot pick k nodes from V0 or V1, but we can pick k nodes elsewhere?
# No, because any independent set in a bipartite graph is a subset of some independent set
# that is a union of some vertices from V0 and some from V1, but specifically,
# the maximum independent set size is at least max(|V0|, |V1|).
# Actually, the problem is equivalent to: can we pick k nodes such that no two are connected?
# Since the graph is bipartite, the maximum independent set size is n - (size of minimum vertex cover).
# By Konig's theorem, size of min vertex cover = size of max matching.
# So max independent set size = n - max_matching.
# However, we just need to check if we can pick k nodes.
# If we pick all nodes from V0, we have |V0| nodes. If we pick all from V1, we have |V1| nodes.
# If k <= |V0| or k <= |V1|, we can definitely pick k nodes.
# If k > |V0| and k > |V1|, can we pick k nodes?
# No, because any independent set I satisfies |I| <= |V0| + |V1| - |matching|.
# Actually, the simplest approach is to partition into V0 and V1.
# If |V0| >= k, output k nodes from V0. Else if |V1| >= k, output k nodes from V1.
# Else, output NO.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    v0 = []
    v1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [i]
            color[i] = 0
            v0.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        if color[v] == 0:
                            v0.append(v)
                        else:
                            v1.append(v)
                        stack.append(v)
                        
    if len(v0) >= k:
        print("YES")
        print(*(v0[:k]))
    elif len(v1) >= k:
        print("YES")
        print(*(v1[:k]))
    else:
        print("NO")

solve()
```
