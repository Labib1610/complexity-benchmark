```python
import sys

def solve():
    # The problem states the graph is bipartite.
    # A completely disconnected group is an independent set.
    # In a bipartite graph, the two partitions (let's call them S0 and S1)
    # are independent sets.
    # Any subset of S0 is an independent set, and any subset of S1 is an independent set.
    # Thus, we can form an independent set of size k if and only if
    # |S0| >= k or |S1| >= k.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    s0 = []
    s1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [i]
            color[i] = 0
            s0.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        if color[v] == 0:
                            s0.append(v)
                        else:
                            s1.append(v)
                        stack.append(v)
    
    if len(s0) >= k:
        print("YES")
        print(*(s0[:k]))
    elif len(s1) >= k:
        print("YES")
        print(*(s1[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
