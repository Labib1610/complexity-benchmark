```python
import sys

def solve():
    # The problem states the graph is bipartite (Proposer/Acceptor).
    # A fully disconnected group is an independent set.
    # In a bipartite graph, the two sets of the bipartition are independent sets.
    # We can find the bipartition using BFS/DFS.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    set0 = []
    set1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [i]
            color[i] = 0
            set0.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        if color[v] == 0:
                            set0.append(v)
                        else:
                            set1.append(v)
                        stack.append(v)
    
    if len(set0) >= k:
        print("YES")
        print(*(set0[:k]))
    elif len(set1) >= k:
        print("YES")
        print(*(set1[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
