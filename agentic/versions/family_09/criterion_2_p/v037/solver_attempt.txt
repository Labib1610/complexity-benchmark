```python
import sys

# The problem states the graph is bipartite.
# In a bipartite graph, the two camps are the two partitions of the graph.
# An all-strangers group is an independent set.
# Since the graph is bipartite, any subset of one of the two partitions is an independent set.
# Let the two partitions be A and B.
# Any subset of A is an independent set, and any subset of B is an independent set.
# To find an independent set of size k, we just need to check if |A| >= k or |B| >= k.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    camp0 = []
    camp1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            camp0.append(i)
            while stack:
                u, c = stack.pop()
                next_c = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = next_c
                        if next_c == 0:
                            camp0.append(v)
                        else:
                            camp1.append(v)
                        stack.append((v, next_c))
    
    if len(camp0) >= k:
        print("YES")
        print(*(camp0[:k]))
    elif len(camp1) >= k:
        print("YES")
        print(*(camp1[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
