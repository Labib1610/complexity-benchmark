import sys

# The problem states that the graph is bipartite (conflicts only between different shifts).
# A harmonious team is an independent set.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# However, we need a team of exactly size k.
# Since the graph is bipartite, we can decompose it into connected components.
# For each component, we have two partitions (let's call them A and B).
# Any subset of A is an independent set, and any subset of B is an independent set.
# Actually, for each connected component, we can pick either all of partition A or all of partition B
# to be part of our independent set.
# Wait, that's not quite right. For each component, we can pick any subset of A or any subset of B?
# No, the largest independent set in a bipartite graph is the union of one partition from each component.
# Let the components be C_1, C_2, ..., C_m. Each C_i has partitions A_i and B_i.
# We can pick either A_i or B_i for each component to form an independent set.
# This is equivalent to the subset sum problem: can we pick sizes s_i from {|A_i|, |B_i|} such that sum(s_i) = k?
# Since n is up to 100,000, we can use bitsets (or large integers in Python) to solve this.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_a = []
            comp_b = []
            stack = [(i, 0)]
            color[i] = 0
            comp_a.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0: comp_a.append(v)
                        else: comp_b.append(v)
                        stack.append((v, 1 - c))
            components.append((comp_a, comp_b))
            
    # dp[i] is a bitset where the j-th bit is 1 if a team of size j is possible
    dp = 1
    # To reconstruct, we store which choice was made
    choices = []
    
    for comp_a, comp_b in components:
        size_a = len(comp_a)
        size_b = len(comp_b)
        new_dp = (dp << size_a) | (dp << size_b)
        choices.append((size_a, size_b))
        dp = new_dp
        
    if not (dp & (1 << k)):
        print("NO")
        return
    
    print("YES")
    # Backtrack to find which partition was chosen for each component
    curr_k = k
    result = []
    for i in range(len(components) - 1, -1, -1):
        size_a, size_b = choices[i]
        comp_a, comp_b = components[i]
        if curr_k >= size_a and (dp >> (curr_k - size_a)) & 1:
            # Check if this path was possible
            # We need to re-verify the DP state or store it
            # Actually, we can just check if the previous state had the bit
            # But we need to be careful. Let's re-run DP or store it.
            pass
            
    # Simpler: store the DP states
    dp_states = [1]
    curr = 1
    for size_a, size_b in choices:
        curr = (curr << size_a) | (curr << size_b)
        dp_states.append(curr)
        
    curr_k = k
    res = []
    for i in range(len(components) - 1, -1, -1):
        size_a, size_b = choices[i]
        comp_a, comp_b = components[i]
        if curr_k >= size_a and (dp_states[i] & (1 << (curr_k - size_a))):
            res.extend(comp_a)
            curr_k -= size_a
        else:
            res.extend(comp_b)
            curr_k -= size_b
            
    print(*(res))

solve()
