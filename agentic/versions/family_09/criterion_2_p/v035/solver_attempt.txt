```python
import sys

# The problem states the graph is bipartite.
# A set of users is "all-strangers" if it is an independent set.
# In a bipartite graph, the two camps (let's call them A and B) are independent sets.
# Any subset of A is an independent set, and any subset of B is an independent set.
# We need to find a subset of size k.
# If |A| >= k, we can pick any k nodes from A.
# If |B| >= k, we can pick any k nodes from B.
# If both |A| < k and |B| < k, then no independent set of size k exists
# because any independent set in a bipartite graph is a subset of A union B,
# and the maximum independent set size is max(|A|, |B|).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    camp0 = []
    camp1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0:
                    camp0.append(u)
                else:
                    camp1.append(u)
                
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    if len(camp0) >= k:
        print("YES")
        print(*(camp0[:k]))
    elif len(camp1) >= k:
        print("YES")
        print(*(camp1[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
