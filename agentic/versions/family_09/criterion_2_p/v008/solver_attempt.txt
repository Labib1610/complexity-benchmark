```python
import sys

# The problem states the graph is bipartite (two camps).
# In a bipartite graph, the largest independent set is not necessarily 
# just one of the two partitions, but the problem guarantees that 
# every friendship connects a user in one camp to a user in the other.
# This means the graph is bipartite.
# An independent set is a set of vertices where no two are connected.
# In a bipartite graph with partitions A and B, any subset of A is an 
# independent set, and any subset of B is an independent set.
# However, we can pick vertices from both A and B as long as no edges exist.
# Since the graph is bipartite, the maximum independent set size is 
# n - (size of minimum vertex cover). By Konig's theorem, in any bipartite 
# graph, the number of edges in a maximum matching equals the number of 
# vertices in a minimum vertex cover.
# Thus, max independent set size = n - (max matching size).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [i]
            color[i] = 0
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        stack.append(v)
    
    camp0 = [i for i in range(1, n + 1) if color[i] == 0]
    camp1 = [i for i in range(1, n + 1) if color[i] == 1]
    
    # Max independent set in bipartite graph:
    # We need to find the maximum matching to find the max independent set.
    # Hopcroft-Karp or simple DFS based matching.
    match = [-1] * (n + 1)
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    matching_size = 0
    for u in camp0:
        visited = [False] * (n + 1)
        if dfs(u, visited):
            matching_size += 1
            
    if n - matching_size < k:
        print("NO")
        return

    # To find the max independent set:
    # 1. Find min vertex cover:
    #    Let U be the set of unmatched vertices in camp0.
    #    Let Z be the set of vertices reachable from U via alternating paths.
    #    Min Vertex Cover K = (camp0 \ Z) U (camp1 ∩ Z)
    #    Max Independent Set = V \ K
    
    matched_in_camp0 = {match[v] for v in camp1 if match[v] != -1}
    unmatched_camp0 = [u for u in camp0 if u not in matched_in_camp0]
    
    z = set()
    stack = unmatched_camp0
    for u in stack: z.add(u)
    while stack:
        u = stack.pop()
        if color[u] == 0:
            for v in adj[u]:
                if v != match[u] and v not in z:
                    z.add(v)
                    stack.append(v)
        else:
            if match[u] != -1 and match[u] not in z:
                z.add(match[u])
                stack.append(match[u])
                
    mvc = [u for u in camp0 if u not in z] + [u for u in camp1 if u in z]
    mvc_set = set(mvc)
    independent_set = [u for u in range(1, n + 1) if u not in mvc_set]
    
    print("YES")
    print(*(independent_set[:k]))

solve()
```
