```python
import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem states the graph is bipartite.
    # We can find the two sets (departments) using BFS/DFS.
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_nodes = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp_nodes[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append(comp_nodes)
            
    # For each connected component, we have two independent sets (departments).
    # Let S_i = (A_i, B_i) be the two sets of component i.
    # We need to pick either A_i or B_i from each component such that the total size is k.
    # This is a variation of the subset sum problem / knapsack.
    # Since we need to pick exactly k, we use DP.
    
    # dp[i][j] = True if it's possible to pick j nodes from the first i components.
    # To reconstruct, we store which choice was made.
    
    dp = {0: []}
    for comp in components:
        new_dp = {}
        size_a = len(comp[0])
        size_b = len(comp[1])
        
        for current_sum, path in dp.items():
            # Option 1: pick set A
            if current_sum + size_a <= k:
                new_dp[current_sum + size_a] = path + [(comp[0])]
            # Option 2: pick set B
            if current_sum + size_b <= k:
                new_dp[current_sum + size_b] = path + [(comp[1])]
        dp = new_dp
        
    if k in dp:
        print("YES")
        result = []
        for group in dp[k]:
            result.extend(group)
        print(*(result))
    else:
        print("NO")

solve()
```
