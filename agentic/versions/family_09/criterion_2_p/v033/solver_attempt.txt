```python
import sys

def solve():
    # The problem states the graph is bipartite.
    # In a bipartite graph, the set of all users can be partitioned into two sets,
    # say A and B, such that all edges connect a node in A to a node in B.
    # An "all-strangers" group is an independent set.
    # In a bipartite graph, any subset of A is an independent set,
    # and any subset of B is an independent set.
    # Thus, the maximum independent set size is max(|A|, |B|).
    # If k <= max(|A|, |B|), we can pick k nodes from the larger set.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    set_a = []
    set_b = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            set_a.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            set_a.append(v)
                        else:
                            set_b.append(v)
                        stack.append((v, color[v]))
    
    if len(set_a) >= k:
        print("YES")
        print(*(set_a[:k]))
    elif len(set_b) >= k:
        print("YES")
        print(*(set_b[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
