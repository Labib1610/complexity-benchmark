```python
import sys

# The problem states the graph is bipartite.
# An all-strangers group is an independent set.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# However, we can simply partition the graph into two sets A and B (the two roles).
# Any subset of A is an independent set, and any subset of B is an independent set.
# Let |A| = size_A and |B| = size_B.
# We can form an independent set of size k if k <= size_A or k <= size_B.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    set_a = []
    set_b = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0:
                    set_a.append(u)
                else:
                    set_b.append(u)
                
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    if len(set_a) >= k:
        print("YES")
        print(*(set_a[:k]))
    elif len(set_b) >= k:
        print("YES")
        print(*(set_b[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
