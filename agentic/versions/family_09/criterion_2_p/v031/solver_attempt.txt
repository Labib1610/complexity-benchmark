```python
import sys

# The problem states the graph is bipartite.
# An interference-free group is an independent set.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# However, we need an independent set of size exactly k.
# In a bipartite graph, the vertices can be partitioned into two sets L and R.
# Any subset of L is an independent set, and any subset of R is an independent set.
# Let |L| = n1 and |R| = n2.
# We can pick any number of vertices from L and any number from R,
# provided no edges exist between them.
# Since it's bipartite, we can pick all of L or all of R.
# Actually, the maximum independent set in a bipartite graph is the union of
# all vertices in L not in the matching and all vertices in R not in the matching?
# No, that's for vertex cover.
# The maximum independent set size is n - |Maximum Matching|.
# To find an independent set of size k:
# Let the two partitions be L and R.
# Any subset of L is independent, any subset of R is independent.
# We can pick x vertices from L and y vertices from R such that no edges exist between them.
# This is equivalent to picking an independent set in the bipartite graph.
# Since we just need *any* independent set of size k, we can use the fact that
# the graph is bipartite. Let the components be C1, C2, ..., Cm.
# Each component has two partitions (L_i, R_i).
# We can pick either L_i or R_i from each component to form an independent set.
# Let s_i = max(|L_i|, |R_i|). The max independent set size is sum(s_i).
# If k <= sum(s_i), we can greedily pick vertices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_l = []
            comp_r = []
            stack = [(i, 0)]
            color[i] = 0
            comp_l.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0: comp_l.append(v)
                        else: comp_r.append(v)
                        stack.append((v, 1 - c))
            components.append((comp_l, comp_r))
            
    # dp[i][j] = can we pick j vertices from first i components
    # Since we need to reconstruct, store choices
    # dp[j] = list of (comp_idx, side)
    # Actually, just need to know if k is reachable.
    # Max possible is sum(max(|L_i|, |R_i|)).
    # Min possible is sum(min(|L_i|, |R_i|)).
    # Wait, we can pick *any* subset of L_i or R_i? No, that's not true.
    # We can pick any subset of L_i OR any subset of R_i.
    # So for each component, we can contribute any size from 0 to |L_i| or 0 to |R_i|.
    # This is a variation of subset sum.
    
    dp = {0: []}
    for i, (L, R) in enumerate(components):
        new_dp = {}
        size_l, size_r = len(L), len(R)
        for current_k, path in dp.items():
            if current_k + size_l <= k:
                new_dp[current_k + size_l] = path + [(i, 0)]
            if current_k + size_r <= k:
                new_dp[current_k + size_r] = path + [(i, 1)]
        dp = new_dp
        
    if k not in dp:
        print("NO")
    else:
        print("YES")
        res = []
        for comp_idx, side in dp[k]:
            if side == 0:
                res.extend(components[comp_idx][0])
            else:
                res.extend(components[comp_idx][1])
        print(*(res))

solve()
```
