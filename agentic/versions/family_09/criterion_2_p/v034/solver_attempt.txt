```python
import sys

# The problem states that any cycle has an even number of reactive pairs.
# This is the definition of a bipartite graph.
# A stable mixture is an independent set in this graph.
# In a bipartite graph, we can partition the vertices into two sets, L and R,
# such that all edges are between L and R.
# Any subset of L is an independent set, and any subset of R is an independent set.
# We want to pick k vertices such that no two are connected.
# This means we need to pick x vertices from L and y vertices from R such that
# x + y = k, and no edges exist between the chosen x and y.
# Since all edges are between L and R, this is always satisfied if we pick
# any subset of size x from L and any subset of size y from R.
# However, we must ensure that no edges exist between the chosen x and y.
# Actually, in a bipartite graph, any subset of L is independent, and any subset of R is independent.
# But if we pick some from L and some from R, we must ensure no edges exist between them.
# This is only possible if we pick vertices from only one side, OR if we pick
# vertices from both sides such that no edges exist between the chosen subsets.
# Wait, the simplest way to form an independent set in a bipartite graph is to
# pick a subset of one of the partitions. But we can also pick vertices from both
# partitions if there are no edges between them.
# Actually, for each connected component, we have two sets L_i and R_i.
# We can pick any subset of L_i or any subset of R_i.
# To get exactly k, we can use dynamic programming.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_l = []
            comp_r = []
            stack = [(i, 0)]
            color[i] = 0
            comp_l.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0: comp_l.append(v)
                        else: comp_r.append(v)
                        stack.append((v, color[v]))
            components.append((comp_l, comp_r))
            
    # dp[i][j] = possible to pick j vertices from first i components
    # Since we can pick either comp_l or comp_r from each component,
    # this is a variation of the subset sum problem.
    # dp[j] = set of (count_l, count_r)
    dp = {0: []}
    
    for l_nodes, r_nodes in components:
        new_dp = {}
        len_l, len_r = len(l_nodes), len(r_nodes)
        for current_k, path in dp.items():
            # Option 1: pick from L
            for take in range(len_l + 1):
                if current_k + take <= k:
                    new_dp[current_k + take] = path + [('L', l_nodes[:take])]
            # Option 2: pick from R
            for take in range(len_r + 1):
                if current_k + take <= k:
                    new_dp[current_k + take] = path + [('R', r_nodes[:take])]
        dp = new_dp
        
    if k in dp:
        print("YES")
        res = []
        for _, nodes in dp[k]:
            res.extend(nodes)
        print(*(res))
    else:
        print("NO")

solve()
```
