```python
import sys

# The problem states the graph is bipartite.
# A harmonious delegation is an independent set.
# In a bipartite graph, the two partitions (let's call them A and B) are independent sets.
# Any subset of A is an independent set, and any subset of B is an independent set.
# We need to find if there exists a subset of size k that is an independent set.
# Since the graph is bipartite, we can find the two partitions using BFS/DFS.
# Let the sizes of the partitions be |A| and |B|.
# Any subset of A of size k is a valid delegation.
# Any subset of B of size k is a valid delegation.
# Thus, a delegation of size k exists if and only if k <= |A| or k <= |B|.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    partition_a = []
    partition_b = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            partition_a.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            partition_a.append(v)
                        else:
                            partition_b.append(v)
                        stack.append((v, color[v]))
    
    if len(partition_a) >= k:
        print("YES")
        print(*(partition_a[:k]))
    elif len(partition_b) >= k:
        print("YES")
        print(*(partition_b[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
