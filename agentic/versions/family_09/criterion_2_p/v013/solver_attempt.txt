```python
import sys

# The problem states that interference only occurs between components of opposite polarities.
# This means the interference graph is bipartite.
# We can find the connected components of the graph. For each component, we can 2-color it
# into two sets (say, set A and set B).
# Let a connected component have size S_i, with |A_i| = a_i and |B_i| = b_i.
# We need to pick a total of k components such that no two are connected.
# This means for each connected component, we can pick either a subset of A_i or a subset of B_i.
# Actually, we can pick any subset of A_i (size x <= a_i) OR any subset of B_i (size y <= b_i).
# Wait, the condition is: we cannot pick two connected components.
# If we pick a subset of A_i, we cannot pick any from B_i.
# So for each connected component i, we can contribute either x elements (0 <= x <= a_i)
# from set A_i, or y elements (0 <= y <= b_i) from set B_i.
# This is a variation of the subset sum problem (knapsack).
# Since we need exactly k, and we have many components, we can use DP.
# Given the constraints, we can use bitsets or a frequency array to optimize.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_a = []
            comp_b = []
            stack = [(i, 0)]
            color[i] = 0
            comp_a.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0: comp_a.append(v)
                        else: comp_b.append(v)
                        stack.append((v, 1 - c))
            components.append((comp_a, comp_b))
            
    # dp[i] stores which choice was made to reach sum i
    # dp[sum] = (comp_idx, choice_type, prev_sum)
    # choice_type: 0 for comp_a, 1 for comp_b
    dp = {0: None}
    
    for i, (a, b) in enumerate(components):
        new_dp = dp.copy()
        len_a, len_b = len(a), len(b)
        for s, _ in dp.items():
            if s + len_a <= k and (s + len_a) not in new_dp:
                new_dp[s + len_a] = (i, 0, s)
            if s + len_b <= k and (s + len_b) not in new_dp:
                new_dp[s + len_b] = (i, 1, s)
        dp = new_dp
        
    if k not in dp:
        print("NO")
        return
    
    print("YES")
    res = []
    curr = k
    while curr > 0:
        comp_idx, choice, prev = dp[curr]
        comp_a, comp_b = components[comp_idx]
        if choice == 0:
            res.extend(comp_a[:curr - prev])
        else:
            res.extend(comp_b[:curr - prev])
        curr = prev
        
    print(*(res))

solve()
```
