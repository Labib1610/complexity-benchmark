```python
import sys

# The problem states that the graph is bipartite because every interference
# occurs between a routing module and a storage module.
# We need to find an independent set of size k.
# In a bipartite graph, the size of the maximum independent set is n - (size of maximum matching).
# We can find the two partitions of the bipartite graph using BFS/DFS.
# Let the partitions be L and R.
# The maximum independent set can be constructed by taking all vertices in L 
# that are not in the minimum vertex cover, and all vertices in R that are not in the minimum vertex cover.
# Alternatively, for bipartite graphs, we can just find the two sets of the bipartition.
# Let the two sets be V1 and V2.
# Any subset of V1 is an independent set, and any subset of V2 is an independent set.
# However, we need a set of size k.
# The maximum independent set size is n - |max_matching|.
# We can use Hopcroft-Karp or simple DFS-based matching to find the max matching.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    v1, v2 = [], []
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: v1.append(u)
                else: v2.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # Max matching to find max independent set
    match = [-1] * (n + 1)
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    matching_size = 0
    for u in v1:
        visited = [False] * (n + 1)
        if dfs(u, visited):
            matching_size += 1
            
    # Max independent set size is n - matching_size
    if k > (n - matching_size):
        print("NO")
        return

    # Konig's theorem: Min vertex cover size = Max matching size
    # To find min vertex cover:
    # 1. Find max matching
    # 2. Let U be unmatched vertices in V1
    # 3. Z = set of vertices connected to U by alternating paths
    # 4. Min vertex cover K = (V1 \ Z) U (V2 ∩ Z)
    # Max independent set = V \ K = (V1 ∩ Z) U (V2 \ Z)
    
    matched_in_v1 = {match[v] for v in v2 if match[v] != -1}
    unmatched_v1 = [u for u in v1 if u not in matched_in_v1]
    
    z = set(unmatched_v1)
    stack = unmatched_v1[:]
    while stack:
        u = stack.pop()
        for v in adj[u]:
            if color[u] == 0 and v not in z and match[v] != u:
                z.add(v)
                if match[v] != -1:
                    z.add(match[v])
                    stack.append(match[v])
                    
    independent_set = []
    for u in v1:
        if u in z: independent_set.append(u)
    for u in v2:
        if u not in z: independent_set.append(u)
        
    print("YES")
    print(*(independent_set[:k]))

solve()
```
