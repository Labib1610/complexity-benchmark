```python
import sys

# The problem states that the graph is bipartite.
# We need to find if there exists a subset of size k that is an independent set.
# In a bipartite graph, the components are independent.
# Let the connected components be C_1, C_2, ..., C_p.
# Each component C_i is bipartite, so it has two partitions (A_i, B_i).
# We can pick either A_i or B_i to be part of our independent set.
# Let a_i = |A_i| and b_i = |B_i|.
# We need to choose one partition from each component such that the sum of sizes is k.
# This is a variation of the subset sum problem (or knapsack).
# Since we have many components, we can use dynamic programming with bitsets
# or a frequency array to optimize.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_nodes = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp_nodes[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append((comp_nodes[0], comp_nodes[1]))

    # dp[i] stores which choice (0 or 1) was made for component i to reach sum i
    # Using a dictionary to store reachable sums and the path to reconstruct
    dp = {0: []}
    
    for i, (part0, part1) in enumerate(components):
        new_dp = {}
        s0, s1 = len(part0), len(part1)
        for current_sum, path in dp.items():
            if current_sum + s0 <= k:
                new_dp[current_sum + s0] = path + [(i, 0)]
            if current_sum + s1 <= k:
                new_dp[current_sum + s1] = path + [(i, 1)]
        dp = new_dp
        
    if k not in dp:
        print("NO")
    else:
        print("YES")
        result = []
        for comp_idx, choice in dp[k]:
            result.extend(components[comp_idx][choice])
        print(*(result))

solve()
```
