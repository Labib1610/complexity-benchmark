```python
import sys

# The problem states the graph is bipartite.
# We need to find an independent set of size k.
# In a bipartite graph, any subset of one of the two partitions is an independent set.
# Let the two partitions be A and B.
# Any subset of A is an independent set, and any subset of B is an independent set.
# We can pick any size s from A (0 <= s <= |A|) and any size k-s from B (0 <= k-s <= |B|).
# If there exists an s such that 0 <= s <= |A| and 0 <= k-s <= |B|, then YES.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    part0 = []
    part1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0:
                    part0.append(u)
                else:
                    part1.append(u)
                
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    len0 = len(part0)
    len1 = len(part1)
    
    # We need to find s such that 0 <= s <= len0 and 0 <= k-s <= len1
    # This is equivalent to max(0, k - len1) <= s <= min(k, len0)
    
    low = max(0, k - len1)
    high = min(k, len0)
    
    if low <= high:
        s = low
        res = part0[:s] + part1[:k-s]
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
```
