import sys

def solve():
    # The problem states the graph is bipartite because races only occur between
    # an initiator and a processor. We can find the two sets of the bipartite graph
    # using BFS or DFS.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    set0 = []
    set1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            set0.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            set0.append(v)
                        else:
                            set1.append(v)
                        stack.append((v, color[v]))
    
    # We have two independent sets: set0 and set1.
    # Any subset of set0 is race-free, and any subset of set1 is race-free.
    # We need to pick exactly k nodes from either set0 or set1.
    
    if len(set0) >= k:
        print("YES")
        print(*(set0[:k]))
    elif len(set1) >= k:
        print("YES")
        print(*(set1[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
