```python
import sys

# The problem states the gear network is "physically workable", 
# which implies the graph is bipartite.
# We need to find an independent set of size k.
# In a bipartite graph, the set of all vertices in one partition 
# is an independent set, and the set of all vertices in the other 
# partition is also an independent set.
# Let the two partitions be A and B. Any subset of A is an independent set,
# and any subset of B is an independent set.
# We can pick any subset of size k from A or B if |A| >= k or |B| >= k.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    part0 = []
    part1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [i]
            color[i] = 0
            part0.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        if color[v] == 0:
                            part0.append(v)
                        else:
                            part1.append(v)
                        stack.append(v)
    
    if len(part0) >= k:
        print("YES")
        print(*(part0[:k]))
    elif len(part1) >= k:
        print("YES")
        print(*(part1[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
