```python
import sys

# The problem states that the graph is bipartite.
# A set of mutually disconnected users is an independent set.
# In a bipartite graph, the two cohorts (let's call them A and B) are 
# themselves independent sets.
# Any subset of A is an independent set, and any subset of B is an independent set.
# Therefore, if we can find a subset of size k from either A or B, we are done.
# If |A| >= k, we can pick any k nodes from A.
# If |B| >= k, we can pick any k nodes from B.
# If both |A| < k and |B| < k, then no independent set of size k exists 
# because the maximum independent set in a bipartite graph is the larger of 
# the two partitions (if we consider the partitions as the only candidates).
# Wait, is the maximum independent set always one of the partitions?
# No, but for a bipartite graph, the size of the maximum independent set is 
# n - (size of maximum matching).
# However, the problem asks for *any* independent set of size k.
# If we have a bipartite graph with partitions A and B, any subset of A is 
# independent, and any subset of B is independent.
# If we can't find k nodes in A or B, can we find k nodes elsewhere?
# Actually, in a bipartite graph, the maximum independent set is indeed 
# the union of the larger partition and some nodes from the other side.
# But the problem is simpler: if we have a bipartite graph, we can 
# color it with 2 colors. Let the sets be C0 and C1.
# Any subset of C0 is an independent set. Any subset of C1 is an independent set.
# If |C0| >= k, we output k nodes from C0.
# If |C1| >= k, we output k nodes from C1.
# If both |C0| < k and |C1| < k, is it possible to have an independent set of size k?
# Yes, e.g., a path 1-2-3. C0={1, 3}, C1={2}. k=2. Independent set {1, 3} works.
# So we need to find the maximum independent set.
# For bipartite graphs, Max Independent Set = n - Max Matching.
# But we don't need the *maximum* independent set, just *any* of size k.
# Actually, the problem is simpler: if we have a bipartite graph, 
# the two partitions are the largest independent sets we can easily identify.
# Let's use BFS/DFS to 2-color the graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    c0 = []
    c1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            c0.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            c0.append(v)
                        else:
                            c1.append(v)
                        stack.append((v, 1 - c))
                        
    if len(c0) >= k:
        print("YES")
        print(*(c0[:k]))
    elif len(c1) >= k:
        print("YES")
        print(*(c1[:k]))
    else:
        print("NO")

solve()
```
