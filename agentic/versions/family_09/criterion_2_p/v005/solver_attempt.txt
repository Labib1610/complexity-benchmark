import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The graph is bipartite. We can find the two camps using BFS/DFS.
    # camp[i] will be 0 or 1.
    camp = [-1] * (n + 1)
    camps = [[], []]
    
    for i in range(1, n + 1):
        if camp[i] == -1:
            stack = [(i, 0)]
            camp[i] = 0
            while stack:
                u, c = stack.pop()
                camps[c].append(u)
                for v in adj[u]:
                    if camp[v] == -1:
                        camp[v] = 1 - c
                        stack.append((v, 1 - c))
    
    # In a bipartite graph, an all-strangers group is a set of vertices
    # such that no two are connected.
    # Since all edges are between camp 0 and camp 1, any subset of camp 0
    # is an all-strangers group, and any subset of camp 1 is an all-strangers group.
    # The largest all-strangers group is the larger of the two camps.
    
    c0 = camps[0]
    c1 = camps[1]
    
    if len(c0) >= k:
        print("YES")
        print(*(c0[:k]))
    elif len(c1) >= k:
        print("YES")
        print(*(c1[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
