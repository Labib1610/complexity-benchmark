import sys

# The problem states the graph is bipartite.
# We need to find if we can pick k vertices such that no two are connected.
# This is equivalent to finding an independent set of size k.
# In a bipartite graph, the size of the maximum independent set is n - (size of maximum matching).
# However, we need to pick exactly k.
# Let the two partitions be L and R. Let |L| = n1, |R| = n2.
# Any subset of L is an independent set, and any subset of R is an independent set.
# We can pick any number of vertices from L and any number from R as long as no edges exist.
# But wait, the problem asks for *any* k vertices that are independent.
# In a bipartite graph, any subset of L is independent, and any subset of R is independent.
# If we pick x vertices from L and y vertices from R, they are independent if no edges exist between them.
# Actually, the condition is simply that the chosen set is an independent set.
# In a bipartite graph, the maximum independent set size is n - max_matching.
# But we can pick *any* independent set of size k.
# Since L is an independent set and R is an independent set, we can pick any subset of L or any subset of R.
# If k <= max(|L|, |R|), we can just pick k vertices from the larger partition.
# Is it possible to pick k > max(|L|, |R|)?
# Yes, if the graph is disconnected, we can pick independent sets from each component.
# Let the components be C_1, C_2, ..., C_m.
# Each component C_i is bipartite with partitions L_i and R_i.
# We can pick either L_i or R_i from each component.
# This is a variation of the subset sum problem / knapsack problem.
# We want to choose s_i in {0, 1} for each component such that sum(s_i * |L_i| + (1-s_i) * |R_i|) = k.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = map(int, input_data[idx:idx+2])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    components = []
    for i in range(1, n + 1):
        if color[i] == -1:
            comp_nodes = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp_nodes[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append((len(comp_nodes[0]), len(comp_nodes[1]), comp_nodes[0], comp_nodes[1]))
            
    # dp[i][j] = can we get sum j using first i components
    # To reconstruct, store which choice was made
    dp = {0: []}
    for i, (c0, c1, nodes0, nodes1) in enumerate(components):
        new_dp = {}
        for s, path in dp.items():
            if s + c0 <= k:
                new_dp[s + c0] = path + [(0, nodes0)]
            if s + c1 <= k:
                new_dp[s + c1] = path + [(1, nodes1)]
        dp = new_dp
        
    if k in dp:
        print("YES")
        res = []
        for choice, nodes in dp[k]:
            res.extend(nodes)
        print(*(res))
    else:
        print("NO")

solve()
