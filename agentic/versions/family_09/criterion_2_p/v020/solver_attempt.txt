import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            comp = [[], []]
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                comp[c].append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
            components.append(comp)
            
    # dp[i][j] = can we pick j proteins from first i components
    # Since we need to reconstruct, we store which choice was made
    # dp[i][j] = (choice_for_comp_i, prev_j)
    # choice: 0 for picking comp[i][0], 1 for picking comp[i][1]
    
    dp = {0: None}
    choices = []
    
    for comp in components:
        new_dp = {}
        c0 = len(comp[0])
        c1 = len(comp[1])
        for j, _ in dp.items():
            if j + c0 <= k:
                new_dp[j + c0] = (0, j)
            if j + c1 <= k:
                new_dp[j + c1] = (1, j)
        dp = new_dp
        choices.append(dp)
        
    if k not in dp:
        print("NO")
        return
    
    print("YES")
    res = []
    curr_k = k
    for i in range(len(components) - 1, -1, -1):
        choice, prev_k = choices[i][curr_k]
        res.extend(components[i][choice])
        curr_k = prev_k
        
    print(*(res))

solve()
