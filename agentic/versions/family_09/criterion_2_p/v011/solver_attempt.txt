```python
import sys

# The problem states that the graph is bipartite.
# We need to find a stable set of size k.
# A stable set is an independent set.
# In a bipartite graph, the maximum independent set size is n - (size of maximum matching).
# However, we just need any independent set of size k.
# Since the graph is bipartite, we can partition the vertices into two sets,
# say A and B, such that all edges are between A and B.
# Any subset of A is an independent set, and any subset of B is an independent set.
# We can pick any subset of size k from A or B if possible.
# If we can't pick k from A or B alone, we might need to pick some from A and some from B.
# But wait, if we pick a vertex from A and a vertex from B, they might be connected.
# Actually, the problem is simpler: we have two independent sets A and B.
# We can pick any number of vertices from A and any number from B,
# but we must ensure no edges exist between the chosen vertices.
# Since all edges are between A and B, if we pick a subset S_A from A and S_B from B,
# the shipment is stable if and only if there are no edges between S_A and S_B.
# This means for every v in S_B, none of its neighbors are in S_A.
# This is equivalent to saying S_A must be a subset of (A \ Neighbors(S_B)).
# To maximize the size of the independent set, we should pick all of A or all of B?
# No, the largest independent set in a bipartite graph is found by taking
# all vertices in one partition and the vertices in the other partition that are not matched.
# Actually, the simplest approach:
# 1. Partition the graph into two sets A and B using BFS/DFS.
# 2. Let |A| = n1, |B| = n2.
# 3. If k <= n1, output any k from A.
# 4. If k <= n2, output any k from B.
# 5. If k > n1 and k > n2, it might still be possible.
#    Wait, the problem asks for *any* stable set of size k.
#    If we take all of A, we can't take any from B that are connected to any in A.
#    Actually, the maximum independent set size is n - (max matching).
#    Let M be the max matching. The set of vertices not in M is an independent set? No.
#    The standard construction: Independent set = (A \ M_A) U (B \ M_B) where M_A, M_B are matched vertices.
#    Actually, just use the property: A is an independent set, B is an independent set.
#    If k <= |A|, output A[:k]. If k <= |B|, output B[:k].
#    If k > |A| and k > |B|, it is impossible? No, that's not true.
#    Example: 1-2, 3-4. A={1,3}, B={2,4}. |A|=2, |B|=2. k=3.
#    We can pick {1, 3, 2} - NO, 1-2 is an edge.
#    We can pick {1, 3, 4} - NO, 3-4 is an edge.
#    Actually, in a bipartite graph, the maximum independent set size is indeed n - |matching|.
#    If k <= n - |matching|, we can always find one.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m, k = map(int, input_data[:3])
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    A, B = [], []
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                if c == 0: A.append(u)
                else: B.append(u)
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
    
    if len(A) >= k:
        print("YES")
        print(*(A[:k]))
    elif len(B) >= k:
        print("YES")
        print(*(B[:k]))
    else:
        print("NO")

solve()
```
