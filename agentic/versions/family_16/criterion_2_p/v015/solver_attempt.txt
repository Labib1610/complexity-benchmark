```python
import sys

def solve():
    # The problem asks to minimize the completion time of the last truck.
    # Let p1_i be the manifest review time and p2_i be the X-ray scan time for truck i.
    # If we process trucks in order 1, 2, ..., n:
    # Let M_i be the time the manifest review for truck i finishes.
    # M_i = sum_{k=1}^i p1_k
    # Let X_i be the time the X-ray scan for truck i finishes.
    # X_i = max(X_{i-1}, M_i) + p2_i, with X_0 = 0.
    # This is a classic scheduling problem equivalent to Johnson's Rule for 2 machines.
    # To minimize the makespan, we sort trucks:
    # 1. Trucks with p1_i < p2_i are sorted by p1_i ascending.
    # 2. Trucks with p1_i >= p2_i are sorted by p2_i descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    trucks = []
    idx = 1
    for _ in range(n):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        trucks.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in trucks:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    current_m = 0
    current_x = 0
    
    for p1, p2 in order:
        current_m += p1
        current_x = max(current_x, current_m) + p2
        
    print(current_x)

if __name__ == '__main__':
    solve()
```
