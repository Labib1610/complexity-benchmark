import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # To minimize the total completion time (makespan) for a two-machine
    # flow shop, we sort the jobs based on the following criteria:
    # 1. Jobs with tD < tA should be processed first, sorted by tD ascending.
    # 2. Jobs with tD >= tA should be processed last, sorted by tA descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        td = int(input_data[idx])
        ta = int(input_data[idx+1])
        jobs.append((td, ta))
        idx += 2
        
    group1 = []
    group2 = []
    
    for td, ta in jobs:
        if td < ta:
            group1.append((td, ta))
        else:
            group2.append((td, ta))
            
    # Sort group1 by td ascending
    group1.sort(key=lambda x: x[0])
    # Sort group2 by ta descending
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate completion time
    # time_d: time when Decryptor finishes current packet
    # time_a: time when Analyzer finishes current packet
    time_d = 0
    time_a = 0
    
    for td, ta in order:
        time_d += td
        # Analyzer starts at max(time_d, time_a)
        if time_d > time_a:
            time_a = time_d + ta
        else:
            time_a += ta
            
    print(time_a)

if __name__ == '__main__':
    solve()
