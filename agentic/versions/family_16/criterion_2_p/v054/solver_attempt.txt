```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    batches = []
    idx = 1
    for _ in range(n):
        p = int(input_data[idx])
        f = int(input_data[idx+1])
        batches.append((p, f))
        idx += 2
        
    # This is a classic Johnson's Rule problem.
    # To minimize the makespan of a two-stage flow shop:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where preparation time < finishing time
    #    Set 2: jobs where preparation time >= finishing time
    # 2. Sort Set 1 in ascending order of preparation time.
    # 3. Sort Set 2 in descending order of finishing time.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    set1 = []
    set2 = []
    for p, f in batches:
        if p < f:
            set1.append((p, f))
        else:
            set2.append((p, f))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate total time
    prep_time = 0
    finish_time = 0
    
    for p, f in sequence:
        prep_time += p
        if finish_time < prep_time:
            finish_time = prep_time + f
        else:
            finish_time += f
            
    print(finish_time)

if __name__ == '__main__':
    solve()
```
