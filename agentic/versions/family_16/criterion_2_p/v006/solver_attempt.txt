```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (customs and crane) and we want to minimize the
    # makespan.
    # Johnson's Rule:
    # 1. Divide the jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate the completion time
    # customs_finish_time tracks when the current job finishes inspection
    # crane_finish_time tracks when the current job finishes loading
    customs_finish_time = 0
    crane_finish_time = 0
    
    for p1, p2 in sequence:
        customs_finish_time += p1
        # The crane can start loading only after inspection is done
        # AND the crane is free from the previous job.
        if customs_finish_time > crane_finish_time:
            crane_finish_time = customs_finish_time + p2
        else:
            crane_finish_time += p2
            
    print(crane_finish_time)

if __name__ == '__main__':
    solve()
```
