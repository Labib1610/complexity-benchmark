```python
import sys

def solve():
    # The problem is a variation of Johnson's Rule for the two-machine flow shop scheduling problem.
    # Johnson's Rule minimizes the makespan by sorting jobs:
    # 1. Jobs with p1 < p2 are sorted in increasing order of p1.
    # 2. Jobs with p1 >= p2 are sorted in decreasing order of p2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    # Johnson's Rule sorting
    # Group 1: p1 < p2, sort by p1 ascending
    # Group 2: p1 >= p2, sort by p2 descending
    group1 = []
    group2 = []
    for p1, p2 in jobs:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate makespan
    # m1_finish: time when machine 1 finishes the current job
    # m2_finish: time when machine 2 finishes the current job
    m1_finish = 0
    m2_finish = 0
    
    for p1, p2 in order:
        m1_finish += p1
        # Machine 2 can start only after machine 1 finishes the job
        # and after machine 2 finishes the previous job.
        if m2_finish < m1_finish:
            m2_finish = m1_finish + p2
        else:
            m2_finish += p2
            
    print(m2_finish)

if __name__ == '__main__':
    solve()
```
