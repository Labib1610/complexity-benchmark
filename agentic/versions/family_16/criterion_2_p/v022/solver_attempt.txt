```python
import sys

def solve():
    # This is a variation of Johnson's Rule for the two-machine flow shop scheduling problem.
    # We want to minimize the makespan (the time the last block finishes decryption).
    # Let d_i be the download time and c_i be the decryption time for block i.
    # Johnson's Rule states that we should sort the blocks based on min(d_i, c_i).
    # Specifically:
    # 1. Divide blocks into two sets:
    #    Set 1: blocks where d_i < c_i
    #    Set 2: blocks where d_i >= c_i
    # 2. Sort Set 1 in ascending order of d_i.
    # 3. Sort Set 2 in descending order of c_i.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    blocks = []
    idx = 1
    for _ in range(j):
        d = int(input_data[idx])
        c = int(input_data[idx+1])
        blocks.append((d, c))
        idx += 2
        
    set1 = []
    set2 = []
    for d, c in blocks:
        if d < c:
            set1.append((d, c))
        else:
            set2.append((d, c))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate the completion time
    # current_download_time tracks when the DSN finishes the current block
    # current_decrypt_time tracks when the Core finishes the current block
    current_download_time = 0
    current_decrypt_time = 0
    
    for d, c in sequence:
        current_download_time += d
        # Decryption starts at max(download_finish_time, previous_decrypt_finish_time)
        start_decrypt = max(current_download_time, current_decrypt_time)
        current_decrypt_time = start_decrypt + c
        
    print(current_decrypt_time)

if __name__ == '__main__':
    solve()
```
