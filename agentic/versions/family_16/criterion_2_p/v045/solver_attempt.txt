```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    canisters = []
    idx = 1
    for _ in range(j):
        p = int(input_data[idx])
        c = int(input_data[idx+1])
        canisters.append((p, c))
        idx += 2
        
    # Johnson's Rule for two-machine flow shop scheduling:
    # 1. Divide jobs into two sets: 
    #    Set 1: jobs where p < c
    #    Set 2: jobs where p >= c
    # 2. Sort Set 1 in ascending order of p.
    # 3. Sort Set 2 in descending order of c.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    set1 = [x for x in canisters if x[0] < x[1]]
    set2 = [x for x in canisters if x[0] >= x[1]]
    
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    # p_time: time when current canister finishes particulate filter
    # c_time: time when current canister finishes chemical scrubber
    p_time = 0
    c_time = 0
    
    for p, c in sequence:
        p_time += p
        if c_time < p_time:
            c_time = p_time + c
        else:
            c_time += c
            
    print(c_time)

if __name__ == '__main__':
    solve()
```
