import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (wash and bake) and a set of jobs.
    # Each job must go through wash then bake.
    # To minimize the total completion time (makespan), we use Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where wash_time < bake_time
    #    Set B: jobs where wash_time >= bake_time
    # 2. Sort Set A in ascending order of wash_time.
    # 3. Sort Set B in descending order of bake_time.
    # 4. The optimal order is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    order = set_a + set_b
    
    wash_time = 0
    bake_time = 0
    
    for p1, p2 in order:
        wash_time += p1
        # The bake cabin can start only after the wash is done
        # AND the previous bake is done.
        if bake_time < wash_time:
            bake_time = wash_time + p2
        else:
            bake_time += p2
            
    print(bake_time)

if __name__ == '__main__':
    solve()
