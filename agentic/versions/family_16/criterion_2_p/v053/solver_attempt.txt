```python
import sys

def solve():
    # This is a classic Johnson's Rule problem for a 2-machine flow shop.
    # We want to minimize the makespan (total completion time).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for job in jobs:
        if job[0] < job[1]:
            set1.append(job)
        else:
            set2.append(job)
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate makespan for the optimal sequence
    # time_solver: time when current job finishes on solver
    # time_tracer: time when current job finishes on tracer
    time_solver = 0
    time_tracer = 0
    
    for p1, p2 in sequence:
        time_solver += p1
        # Tracer starts at max(time_solver, time_tracer)
        if time_solver > time_tracer:
            time_tracer = time_solver + p2
        else:
            time_tracer = time_tracer + p2
            
    print(time_tracer)

if __name__ == '__main__':
    solve()
```
