```python
import sys

def solve():
    # This is a classic application of Johnson's Rule for the two-machine 
    # flow shop scheduling problem to minimize the makespan.
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where forming time (f_i) < finishing time (s_i)
    #    Set 2: jobs where forming time (f_i) >= finishing time (s_i)
    # 2. Sort Set 1 in ascending order of f_i.
    # 3. Sort Set 2 in descending order of s_i.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        f = int(input_data[idx])
        s = int(input_data[idx+1])
        jobs.append((f, s))
        idx += 2
        
    set1 = []
    set2 = []
    
    for f, s in jobs:
        if f < s:
            set1.append((f, s))
        else:
            set2.append((f, s))
            
    # Sort set1 by f_i ascending
    set1.sort(key=lambda x: x[0])
    # Sort set2 by s_i descending
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate makespan
    # current_former_time: time when the former finishes the current job
    # current_finisher_time: time when the finisher finishes the current job
    current_former_time = 0
    current_finisher_time = 0
    
    for f, s in sequence:
        current_former_time += f
        # Finisher can start only after former finishes the current job
        # AND after the finisher finishes the previous job
        if current_former_time > current_finisher_time:
            current_finisher_time = current_former_time + s
        else:
            current_finisher_time += s
            
    print(current_finisher_time)

if __name__ == '__main__':
    solve()
```
