```python
import sys

def solve():
    # The problem asks to minimize B_j.
    # Let the sequence be E_1, E_2, ..., E_j.
    # A_k = sum_{i=1}^k X_i
    # B_k = max(A_k, B_{k-1}) + Y_k
    # This recurrence can be expanded:
    # B_j = max(A_1 + Y_1 + Y_2 + ... + Y_j,
    #           A_2 + Y_2 + Y_3 + ... + Y_j,
    #           ...,
    #           A_j + Y_j)
    # Let S_X = sum(X_i) and S_Y = sum(Y_i).
    # Let P_k = sum_{i=1}^k X_i.
    # Let Q_k = sum_{i=k}^j Y_i.
    # Then B_j = max_{1 <= k <= j} (P_k + Q_k).
    # Note that P_k + Q_k = (sum_{i=1}^k X_i) + (sum_{i=k}^j Y_i)
    # = (sum_{i=1}^j X_i) + (sum_{i=k}^j Y_i) - (sum_{i=k+1}^j X_i)
    # This is equivalent to:
    # B_j = sum(X) + sum(Y) - min_{1 <= k <= j} (sum_{i=k+1}^j X_i + sum_{i=1}^{k-1} Y_i)
    # Actually, a simpler observation:
    # B_j = max(A_k + sum_{i=k}^j Y_i)
    # Let's re-examine the condition.
    # If we have two elements (x1, y1) and (x2, y2):
    # Order 1, 2: B = max(x1+y1+y2, x1+x2+y2)
    # Order 2, 1: B = max(x2+y2+y1, x2+x1+y1)
    # Comparing max(x1+y1+y2, x1+x2+y2) and max(x2+y2+y1, x2+x1+y1)
    # This is equivalent to sorting by Y.
    # If Y_i < Y_j, we should put the one with smaller Y first?
    # Let's test Example 1: (3,2), (2,4), (1,3). Sorted by Y: (3,2), (1,3), (2,4).
    # Order (3,2), (1,3), (2,4):
    # A1=3, B1=3+2=5
    # A2=4, B2=max(4,5)+3=8
    # A3=6, B3=max(6,8)+4=12
    # Example 1 says optimal is (1,3), (2,4), (3,2) -> B3=10.
    # Sorting by Y descending: (2,4), (1,3), (3,2)
    # A1=2, B1=2+4=6
    # A2=3, B2=max(3,6)+3=9
    # A3=6, B3=max(6,9)+2=11
    # Sorting by X ascending: (1,3), (2,4), (3,2) -> B3=10.
    # Sorting by X descending: (3,2), (2,4), (1,3) -> B3=11.
    # The optimal order is to sort by X ascending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    j = int(input_data[0])
    elements = []
    idx = 1
    for _ in range(j):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        elements.append((x, y))
        idx += 2
    
    # Sort by X ascending
    elements.sort()
    
    a = 0
    b = 0
    for x, y in elements:
        a += x
        if a > b:
            b = a + y
        else:
            b = b + y
            
    print(b)

solve()
```
