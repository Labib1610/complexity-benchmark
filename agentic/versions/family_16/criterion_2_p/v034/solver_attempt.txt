```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    assets = []
    idx = 1
    for _ in range(n):
        e = int(input_data[idx])
        u = int(input_data[idx+1])
        assets.append((e, u))
        idx += 2
    
    # This is a variation of Johnson's Rule for a two-machine flow shop.
    # To minimize the makespan, we sort assets into two groups:
    # Group 1: assets where encoding time < uploading time (sort by encoding time ascending)
    # Group 2: assets where encoding time >= uploading time (sort by uploading time descending)
    
    group1 = []
    group2 = []
    for e, u in assets:
        if e < u:
            group1.append((e, u))
        else:
            group2.append((e, u))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    current_encoding_time = 0
    current_uploading_time = 0
    
    for e, u in sequence:
        current_encoding_time += e
        # The uploading module can start only after encoding is done
        # and the uploading module is free.
        current_uploading_time = max(current_encoding_time, current_uploading_time) + u
        
    print(current_uploading_time)

if __name__ == '__main__':
    solve()
```
