import sys

def solve():
    # This is a classic Johnson's Rule problem for the 2-machine flow shop scheduling.
    # The goal is to minimize the makespan.
    # Johnson's Rule:
    # 1. List all jobs.
    # 2. Find the minimum processing time among all jobs in both stations.
    # 3. If the minimum is in the Fabrication station, place that job at the beginning of the sequence.
    # 4. If the minimum is in the Polishing station, place that job at the end of the sequence.
    # 5. Remove the job and repeat.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for i in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2, i))
        idx += 2
        
    # Group 1: p1 < p2, sorted by p1 ascending
    # Group 2: p1 >= p2, sorted by p2 descending
    group1 = []
    group2 = []
    for job in jobs:
        if job[0] < job[1]:
            group1.append(job)
        else:
            group2.append(job)
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate makespan
    fab_time = 0
    pol_time = 0
    for p1, p2, _ in sequence:
        fab_time += p1
        if pol_time < fab_time:
            pol_time = fab_time + p2
        else:
            pol_time += p2
            
    print(pol_time)

if __name__ == '__main__':
    solve()
