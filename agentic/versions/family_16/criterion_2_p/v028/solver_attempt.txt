```python
import sys

def solve():
    # The problem is a variation of Johnson's Rule for two-machine flow shop scheduling.
    # We have j artifacts, each with a Channeler time (c_i) and a Binder time (b_i).
    # The goal is to minimize the makespan (total time until the last artifact is sealed).
    # Johnson's Rule states:
    # 1. Divide artifacts into two sets:
    #    Set 1: artifacts where c_i < b_i
    #    Set 2: artifacts where c_i >= b_i
    # 2. Sort Set 1 in ascending order of c_i.
    # 3. Sort Set 2 in descending order of b_i.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    artifacts = []
    idx = 1
    for _ in range(j):
        c = int(input_data[idx])
        b = int(input_data[idx+1])
        artifacts.append((c, b))
        idx += 2
        
    set1 = []
    set2 = []
    for c, b in artifacts:
        if c < b:
            set1.append((c, b))
        else:
            set2.append((c, b))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate makespan
    # channeler_time: time when Channeler finishes current artifact
    # binder_time: time when Binder finishes current artifact
    channeler_time = 0
    binder_time = 0
    
    for c, b in sequence:
        channeler_time += c
        # Binder can start only after Channeler finishes this artifact
        # and after Binder finishes the previous artifact.
        binder_time = max(channeler_time, binder_time) + b
        
    print(binder_time)

if __name__ == '__main__':
    solve()
```
