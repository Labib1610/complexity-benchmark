import sys

def solve():
    # The problem is a variation of Johnson's Rule for the two-machine flow shop scheduling problem.
    # We have j containers, each with a scan time (p1) and a load time (p2).
    # The goal is to minimize the makespan.
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = [job for job in jobs if job[0] < job[1]]
    set2 = [job for job in jobs if job[0] >= job[1]]
    
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate makespan
    # scan_finish_time[i] = sum(p1 for k in 0..i)
    # load_finish_time[i] = max(scan_finish_time[i], load_finish_time[i-1]) + p2[i]
    
    current_scan_time = 0
    current_load_time = 0
    
    for p1, p2 in sequence:
        current_scan_time += p1
        if current_load_time < current_scan_time:
            current_load_time = current_scan_time + p2
        else:
            current_load_time += p2
            
    print(current_load_time)

if __name__ == '__main__':
    solve()
