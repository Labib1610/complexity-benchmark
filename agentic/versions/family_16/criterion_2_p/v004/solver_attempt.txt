import sys

def solve():
    # This is a classic Johnson's Rule problem.
    # To minimize the makespan on two machines, we sort jobs:
    # 1. Jobs with p1 < p2 are sorted by p1 in ascending order.
    # 2. Jobs with p1 >= p2 are sorted by p2 in descending order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    # Sort group1 by p1 ascending
    group1.sort(key=lambda x: x[0])
    # Sort group2 by p2 descending
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate makespan
    time_m1 = 0
    time_m2 = 0
    
    for p1, p2 in sequence:
        time_m1 += p1
        if time_m1 > time_m2:
            time_m2 = time_m1 + p2
        else:
            time_m2 += p2
            
    print(time_m2)

if __name__ == '__main__':
    solve()
