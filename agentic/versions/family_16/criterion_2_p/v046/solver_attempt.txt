```python
import sys

def solve():
    # The problem is a variation of Johnson's Rule for two-machine flow shop scheduling.
    # We want to minimize the makespan (total time until the last container finishes).
    # Let S_i be the scanner time and I_i be the inspector time for container i.
    # Johnson's Rule:
    # 1. Divide containers into two sets:
    #    Set 1: S_i < I_i
    #    Set 2: S_i >= I_i
    # 2. Sort Set 1 in ascending order of S_i.
    # 3. Sort Set 2 in descending order of I_i.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    containers = []
    idx = 1
    for _ in range(j):
        s = int(input_data[idx])
        i = int(input_data[idx+1])
        containers.append((s, i))
        idx += 2
        
    set1 = []
    set2 = []
    for s, i in containers:
        if s < i:
            set1.append((s, i))
        else:
            set2.append((s, i))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate makespan
    # scanner_time: time when current container finishes scanning
    # inspector_time: time when current container finishes inspection
    scanner_time = 0
    inspector_time = 0
    
    for s, i in sequence:
        scanner_time += s
        # Inspector can start only after scanner finishes this container
        # and after inspector finishes previous container
        inspector_time = max(scanner_time, inspector_time) + i
        
    print(inspector_time)

if __name__ == '__main__':
    solve()
```
