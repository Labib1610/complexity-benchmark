```python
import sys

def solve():
    # This is a variation of Johnson's Rule for the two-machine flow shop scheduling problem.
    # We want to minimize the makespan (total time).
    # Let p1 be the time for the Inscriber and p2 be the time for the Channeler.
    # Johnson's Rule:
    # 1. Divide the jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 2. Sort Set A in ascending order of p1.
    # 3. Sort Set B in descending order of p2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate makespan
    # Inscriber finishes ritual i at time T_inscriber_i
    # Channeler finishes ritual i at time T_channeler_i
    # T_inscriber_i = T_inscriber_{i-1} + p1_i
    # T_channeler_i = max(T_inscriber_i, T_channeler_{i-1}) + p2_i
    
    t_inscriber = 0
    t_channeler = 0
    
    for p1, p2 in sequence:
        t_inscriber += p1
        t_channeler = max(t_inscriber, t_channeler) + p2
        
    print(t_channeler)

if __name__ == '__main__':
    solve()
```
