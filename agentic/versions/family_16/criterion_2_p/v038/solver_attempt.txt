import sys

def solve():
    # The problem is a variation of Johnson's Rule for two-machine flow shop scheduling.
    # We want to minimize the makespan (completion time of the last job).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where g < r
    #    Set 2: jobs where g >= r
    # 2. Sort Set 1 in ascending order of g.
    # 3. Sort Set 2 in descending order of r.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        g = int(input_data[idx])
        r = int(input_data[idx+1])
        jobs.append((g, r))
        idx += 2
        
    set1 = []
    set2 = []
    for g, r in jobs:
        if g < r:
            set1.append((g, r))
        else:
            set2.append((g, r))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    # telescope_finish[i] = sum(g_0 ... g_i)
    # mainframe_finish[i] = max(mainframe_finish[i-1], telescope_finish[i]) + r_i
    
    current_g_time = 0
    current_r_time = 0
    
    for g, r in sequence:
        current_g_time += g
        if current_r_time < current_g_time:
            current_r_time = current_g_time + r
        else:
            current_r_time += r
            
    print(current_r_time)

if __name__ == '__main__':
    solve()
