import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    packets = []
    idx = 1
    for _ in range(j):
        u = int(input_data[idx])
        d = int(input_data[idx+1])
        packets.append((u, d))
        idx += 2
    
    # This is a variation of Johnson's Rule for two-machine scheduling.
    # We want to minimize the makespan on a flow shop with 2 machines.
    # Machine 1: Earth to Relay (Upload)
    # Machine 2: Relay to Mars (Download)
    # The optimal sequence is found by sorting:
    # Group 1: packets where U < D, sorted by U ascending.
    # Group 2: packets where U >= D, sorted by D descending.
    
    group1 = []
    group2 = []
    for u, d in packets:
        if u < d:
            group1.append((u, d))
        else:
            group2.append((u, d))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate the completion time
    # current_u_finish: time when Earth finishes uploading current packet to relay
    # current_d_finish: time when Relay finishes downloading current packet to Mars
    current_u_finish = 0
    current_d_finish = 0
    
    for u, d in sequence:
        current_u_finish += u
        # Relay can start downloading at max(current_u_finish, current_d_finish)
        # and finishes at that start time + d
        if current_u_finish > current_d_finish:
            current_d_finish = current_u_finish + d
        else:
            current_d_finish = current_d_finish + d
            
    print(current_d_finish)

if __name__ == '__main__':
    solve()
