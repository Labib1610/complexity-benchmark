import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (Exciter and Dampener) and a set of jobs.
    # Each job must go through the Exciter then the Dampener.
    # To minimize the total makespan, we sort the jobs:
    # 1. Jobs where p1 < p2 are sorted by p1 in ascending order.
    # 2. Jobs where p1 >= p2 are sorted by p2 in descending order.
    # 3. Place group 1 before group 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    cores = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        cores.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in cores:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    order = group1 + group2
    
    # Calculate the completion time
    # exciter_time: time when the current core finishes excitation
    # dampener_time: time when the current core finishes dampening
    exciter_time = 0
    dampener_time = 0
    
    for p1, p2 in order:
        exciter_time += p1
        # The dampener can start only after the exciter finishes this core
        # AND after the dampener finishes the previous core.
        dampener_time = max(exciter_time, dampener_time) + p2
        
    print(dampener_time)

if __name__ == '__main__':
    solve()
