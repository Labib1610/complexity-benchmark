```python
import sys

def solve():
    # This is a classic application of Johnson's Rule for the 2-machine flow shop problem.
    # To minimize the makespan:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 2. Sort Set A in ascending order of p1.
    # 3. Sort Set B in descending order of p2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate completion time
    # time_on_solver: current time when the current job finishes on the solver
    # time_on_tracer: current time when the current job finishes on the tracer
    time_on_solver = 0
    time_on_tracer = 0
    
    for p1, p2 in sequence:
        time_on_solver += p1
        # The tracer can start the current job only after the solver finishes it
        # AND the tracer finishes the previous job.
        time_on_tracer = max(time_on_solver, time_on_tracer) + p2
        
    print(time_on_tracer)

if __name__ == '__main__':
    solve()
```
