import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (oven and applicator) and we want to minimize
    # the makespan (total time to complete all jobs).
    # Johnson's Rule:
    # 1. Divide the jobs into two sets:
    #    Set 1: jobs where oven_time < applicator_time
    #    Set 2: jobs where oven_time >= applicator_time
    # 2. Sort Set 1 in ascending order of oven_time.
    # 3. Sort Set 2 in descending order of applicator_time.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(n):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate makespan
    oven_time = 0
    applicator_time = 0
    
    for p1, p2 in sequence:
        oven_time += p1
        if applicator_time < oven_time:
            applicator_time = oven_time + p2
        else:
            applicator_time += p2
            
    print(applicator_time)

if __name__ == '__main__':
    solve()
