```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (decryption and analysis).
    # To minimize the makespan, we split the jobs into two sets:
    # Set 1: jobs where decryption time (p1) < analysis time (p2)
    # Set 2: jobs where decryption time (p1) >= analysis time (p2)
    # Sort Set 1 by p1 ascending.
    # Sort Set 2 by p2 descending.
    # The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate the completion time
    # current_decrypt_time tracks when the decryption module finishes the current job
    # current_analysis_time tracks when the analysis module finishes the current job
    current_decrypt_time = 0
    current_analysis_time = 0
    
    for p1, p2 in sequence:
        current_decrypt_time += p1
        # Analysis can start only after decryption is done AND previous analysis is done
        current_analysis_time = max(current_decrypt_time, current_analysis_time) + p2
        
    print(current_analysis_time)

if __name__ == '__main__':
    solve()
```
