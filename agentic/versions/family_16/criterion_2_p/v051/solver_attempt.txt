```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # To minimize the makespan of a two-machine flow shop:
    # 1. Divide the jobs into two sets:
    #    Set 1: jobs where transcription time < analysis time
    #    Set 2: jobs where transcription time >= analysis time
    # 2. Sort Set 1 in ascending order of transcription time.
    # 3. Sort Set 2 in descending order of analysis time.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    scrolls = []
    idx = 1
    for _ in range(j):
        t = int(input_data[idx])
        a = int(input_data[idx+1])
        scrolls.append((t, a))
        idx += 2
        
    set1 = []
    set2 = []
    
    for t, a in scrolls:
        if t < a:
            set1.append((t, a))
        else:
            set2.append((t, a))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    current_transcribe_time = 0
    current_analyze_time = 0
    
    for t, a in sequence:
        current_transcribe_time += t
        # The historian starts at max(when transcription finishes, when they finish previous analysis)
        current_analyze_time = max(current_transcribe_time, current_analyze_time) + a
        
    print(current_analyze_time)

if __name__ == '__main__':
    solve()
```
