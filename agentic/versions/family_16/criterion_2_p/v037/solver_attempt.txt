import sys

def solve():
    # This is a classic application of Johnson's Rule for the 2-machine flow shop problem.
    # To minimize the makespan, we sort jobs into two groups:
    # Group 1: jobs where p1 < p2, sorted by p1 in ascending order.
    # Group 2: jobs where p1 >= p2, sorted by p2 in descending order.
    # The optimal sequence is Group 1 followed by Group 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(n):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate the makespan for the optimal sequence
    m1_time = 0
    m2_time = 0
    
    for p1, p2 in sequence:
        m1_time += p1
        if m1_time > m2_time:
            m2_time = m1_time + p2
        else:
            m2_time += p2
            
    print(m2_time)

if __name__ == '__main__':
    solve()
