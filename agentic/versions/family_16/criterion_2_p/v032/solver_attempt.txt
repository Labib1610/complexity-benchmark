import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have j jobs, each with extraction time p1 and encoding time p2.
    # We want to minimize the total completion time (makespan).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 2. Sort Set A in ascending order of p1.
    # 3. Sort Set B in descending order of p2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate completion time
    # extraction_time_sum tracks when the current job finishes extraction
    # encoding_finish_time tracks when the current job finishes encoding
    extraction_time_sum = 0
    encoding_finish_time = 0
    
    for p1, p2 in sequence:
        extraction_time_sum += p1
        # The encoding server starts at max(extraction_finish_time, previous_encoding_finish_time)
        if extraction_time_sum > encoding_finish_time:
            encoding_finish_time = extraction_time_sum + p2
        else:
            encoding_finish_time += p2
            
    print(encoding_finish_time)

if __name__ == '__main__':
    solve()
