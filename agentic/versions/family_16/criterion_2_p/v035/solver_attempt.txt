import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # To minimize the makespan on two machines, we partition the jobs into two sets:
    # Set 1: jobs where p1 < p2 (sort by p1 ascending)
    # Set 2: jobs where p1 >= p2 (sort by p2 descending)
    # The optimal sequence is all jobs in Set 1 followed by all jobs in Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    # Sort set1 by p1 ascending
    set1.sort(key=lambda x: x[0])
    # Sort set2 by p2 descending
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    m1_time = 0
    m2_time = 0
    
    for p1, p2 in sequence:
        m1_time += p1
        if m1_time > m2_time:
            m2_time = m1_time + p2
        else:
            m2_time += p2
            
    print(m2_time)

if __name__ == '__main__':
    solve()
