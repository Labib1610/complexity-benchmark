```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (upload and download).
    # We want to minimize the makespan (total time).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where upload_time < download_time
    #    Set 2: jobs where upload_time >= download_time
    # 2. Sort Set 1 in ascending order of upload_time.
    # 3. Sort Set 2 in descending order of download_time.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    # upload_finish[i] = upload_finish[i-1] + p1[i]
    # download_finish[i] = max(upload_finish[i], download_finish[i-1]) + p2[i]
    
    upload_time = 0
    download_time = 0
    
    for p1, p2 in sequence:
        upload_time += p1
        if download_time < upload_time:
            download_time = upload_time + p2
        else:
            download_time += p2
            
    print(download_time)

if __name__ == '__main__':
    solve()
```
