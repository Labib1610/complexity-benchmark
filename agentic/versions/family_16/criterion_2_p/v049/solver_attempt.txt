```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # We have two machines (transcription desk and binding table).
    # Each job must go through machine 1 then machine 2.
    # To minimize the total completion time (makespan), we use Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where p1 < p2
    #    Set B: jobs where p1 >= p2
    # 2. Sort Set A in ascending order of p1.
    # 3. Sort Set B in descending order of p2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set_a = []
    set_b = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set_a.append((p1, p2))
        else:
            set_b.append((p1, p2))
            
    # Sort set_a by p1 ascending
    set_a.sort(key=lambda x: x[0])
    # Sort set_b by p2 descending
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    order = set_a + set_b
    
    # Calculate completion time
    # time_m1: time when machine 1 finishes current job
    # time_m2: time when machine 2 finishes current job
    time_m1 = 0
    time_m2 = 0
    
    for p1, p2 in order:
        time_m1 += p1
        # Machine 2 can only start after machine 1 finishes the job
        # and after machine 2 finishes the previous job.
        time_m2 = max(time_m1, time_m2) + p2
        
    print(time_m2)

if __name__ == '__main__':
    solve()
```
