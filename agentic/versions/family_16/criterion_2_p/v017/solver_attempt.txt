```python
import sys

def solve():
    # This is a classic application of Johnson's Rule for the 2-machine flow shop scheduling problem.
    # The goal is to minimize the makespan (the time the last job finishes on the second machine).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where Molder time (p1) < Welder time (p2)
    #    Set 2: jobs where Molder time (p1) >= Welder time (p2)
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    # Sort set1 by p1 ascending
    set1.sort(key=lambda x: x[0])
    # Sort set2 by p2 descending
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate makespan
    # molder_time tracks when the current job finishes on the Molder
    # welder_time tracks when the current job finishes on the Welder
    molder_time = 0
    welder_time = 0
    
    for p1, p2 in sequence:
        molder_time += p1
        # The Welder can start the job only after the Molder finishes it
        # AND after the Welder finishes the previous job.
        if welder_time < molder_time:
            welder_time = molder_time + p2
        else:
            welder_time += p2
            
    print(welder_time)

if __name__ == '__main__':
    solve()
```
