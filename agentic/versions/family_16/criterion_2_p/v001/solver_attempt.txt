```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # To minimize the makespan of two-machine flow shop scheduling:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = [job for job in jobs if job[0] < job[1]]
    set2 = [job for job in jobs if job[0] >= job[1]]
    
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    time_a = 0
    time_b = 0
    
    for p1, p2 in sequence:
        time_a += p1
        if time_a > time_b:
            time_b = time_a + p2
        else:
            time_b += p2
            
    print(time_b)

if __name__ == '__main__':
    solve()
```
