```python
import sys

def solve():
    # This is a variation of Johnson's Rule for the two-machine flow shop problem.
    # We want to minimize the makespan (total time).
    # Johnson's Rule:
    # 1. Divide artifacts into two sets:
    #    Set 1: c < r (cleaning time < repair time)
    #    Set 2: c >= r (cleaning time >= repair time)
    # 2. Sort Set 1 in ascending order of c.
    # 3. Sort Set 2 in descending order of r.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    artifacts = []
    idx = 1
    for _ in range(j):
        c = int(input_data[idx])
        r = int(input_data[idx+1])
        artifacts.append((c, r))
        idx += 2
        
    set1 = []
    set2 = []
    
    for c, r in artifacts:
        if c < r:
            set1.append((c, r))
        else:
            set2.append((c, r))
            
    # Sort set1 by c ascending
    set1.sort(key=lambda x: x[0])
    # Sort set2 by r descending
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    # chemist_finish_time tracks when the chemist finishes the current artifact
    # sculptor_finish_time tracks when the sculptor finishes the current artifact
    chemist_finish_time = 0
    sculptor_finish_time = 0
    
    for c, r in sequence:
        chemist_finish_time += c
        # Sculptor can start only after chemist finishes this artifact
        # AND after sculptor finishes the previous artifact
        sculptor_start_time = max(chemist_finish_time, sculptor_finish_time)
        sculptor_finish_time = sculptor_start_time + r
        
    print(sculptor_finish_time)

if __name__ == '__main__':
    solve()
```
