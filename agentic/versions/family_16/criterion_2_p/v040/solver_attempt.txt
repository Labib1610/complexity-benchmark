```python
import sys

def solve():
    # Read all input at once for speed
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    vessels = []
    idx = 1
    for _ in range(j):
        c = int(input_data[idx])
        t = int(input_data[idx+1])
        vessels.append((c, t))
        idx += 2
    
    # The problem asks to minimize the completion time of the last vessel.
    # Let the sequence be v_1, v_2, ..., v_j.
    # Let C_i be the calibration time and T_i be the telemetry time for vessel i.
    # The completion time of vessel i at the Drydock is D_i = sum_{k=1}^i C_k.
    # The completion time of vessel i at the Uplink is U_i = max(U_{i-1}, D_i) + T_i.
    # This is a classic scheduling problem. The order does not actually affect 
    # the final completion time because the total time is:
    # U_j = max_{1 <= i <= j} (sum_{k=1}^i C_k + sum_{k=i}^j T_k)
    # U_j = sum_{k=1}^j C_k + sum_{k=1}^j T_k - min_{1 <= i <= j} (sum_{k=1}^{i-1} C_k + sum_{k=i+1}^j T_k)
    # Actually, the total time is simply the sum of all C_i plus the T_i of the last vessel,
    # but wait, the constraint is that the Uplink can only start after the Drydock finishes.
    # The total time is:
    # Let S_C = sum(C_i), S_T = sum(T_i).
    # The completion time is max_{1 <= i <= j} (sum_{k=1}^i C_k + sum_{k=i}^j T_k).
    # This value is invariant regardless of the order of vessels.
    
    current_drydock_time = 0
    current_uplink_time = 0
    
    for c, t in vessels:
        current_drydock_time += c
        current_uplink_time = max(current_uplink_time, current_drydock_time) + t
        
    print(current_uplink_time)

if __name__ == '__main__':
    solve()
```
