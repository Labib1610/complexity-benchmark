```python
import sys

def solve():
    # This is a classic application of Johnson's Rule for the two-machine 
    # flow shop scheduling problem.
    # To minimize the makespan, we sort the jobs into two groups:
    # Group 1: jobs where p1 < p2, sorted by p1 in ascending order.
    # Group 2: jobs where p1 >= p2, sorted by p2 in descending order.
    # The optimal sequence is Group 1 followed by Group 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    group1 = []
    group2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate the makespan for the optimal sequence
    # time_p: time when primary unit finishes current job
    # time_s: time when secondary unit finishes current job
    time_p = 0
    time_s = 0
    
    for p1, p2 in sequence:
        time_p += p1
        # Secondary unit starts when primary finishes or when it's free
        time_s = max(time_p, time_s) + p2
        
    print(time_s)

if __name__ == '__main__':
    solve()
```
