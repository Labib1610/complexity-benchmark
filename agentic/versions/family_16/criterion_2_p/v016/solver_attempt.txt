```python
import sys

def solve():
    # This is a classic Johnson's Rule problem for two-machine flow shop scheduling.
    # To minimize the makespan, we sort the jobs into two groups:
    # Group 1: Jobs where Extractor time (e) < Analyzer time (a), sorted by e ascending.
    # Group 2: Jobs where Extractor time (e) >= Analyzer time (a), sorted by a descending.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        e = int(input_data[idx])
        a = int(input_data[idx+1])
        jobs.append((e, a))
        idx += 2
        
    group1 = []
    group2 = []
    
    for e, a in jobs:
        if e < a:
            group1.append((e, a))
        else:
            group2.append((e, a))
            
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    # Calculate makespan
    # extractor_time: time when current job finishes on Extractor
    # analyzer_time: time when current job finishes on Analyzer
    extractor_time = 0
    analyzer_time = 0
    
    for e, a in sequence:
        extractor_time += e
        # Analyzer can start only after Extractor finishes current job
        # and after Analyzer finishes previous job
        analyzer_time = max(extractor_time, analyzer_time) + a
        
    print(analyzer_time)

if __name__ == '__main__':
    solve()
```
