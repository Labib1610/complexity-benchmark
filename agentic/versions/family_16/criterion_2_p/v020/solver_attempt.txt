```python
import sys

def solve():
    # This is a classic application of Johnson's Rule for the 2-machine flow shop problem.
    # To minimize the makespan:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where crusher time (c) < smelter time (s)
    #    Set 2: jobs where crusher time (c) >= smelter time (s)
    # 2. Sort Set 1 in ascending order of crusher time (c).
    # 3. Sort Set 2 in descending order of smelter time (s).
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        c = int(input_data[idx])
        s = int(input_data[idx+1])
        jobs.append((c, s))
        idx += 2
        
    set1 = []
    set2 = []
    
    for job in jobs:
        if job[0] < job[1]:
            set1.append(job)
        else:
            set2.append(job)
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    # crusher_time: time when current batch finishes crushing
    # smelter_time: time when current batch finishes smelting
    crusher_time = 0
    smelter_time = 0
    
    for c, s in sequence:
        crusher_time += c
        # Smelter can start only after crusher finishes current batch
        # and after smelter finishes previous batch
        if smelter_time < crusher_time:
            smelter_time = crusher_time + s
        else:
            smelter_time += s
            
    print(smelter_time)

if __name__ == '__main__':
    solve()
```
