import sys

def solve():
    # The problem is equivalent to Johnson's Rule for two-machine flow shop scheduling.
    # We want to minimize the makespan for n jobs on 2 machines (Surveyor, Harvester).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    order = set1 + set2
    
    # Calculate makespan
    # Surveyor time: current_p1_sum
    # Harvester time: max(surveyor_finish_time, previous_harvester_finish_time) + p2
    
    current_p1_sum = 0
    harvester_finish_time = 0
    
    for p1, p2 in order:
        current_p1_sum += p1
        harvester_finish_time = max(current_p1_sum, harvester_finish_time) + p2
        
    print(harvester_finish_time)

if __name__ == '__main__':
    solve()
