```python
import sys

def solve():
    # The problem asks to minimize the completion time of a sequence of tasks
    # where each task has a Scan time (S) and a Decontamination time (D).
    # This is a classic scheduling problem known as Johnson's Rule for 2 machines.
    # However, the problem states: "The containers are processed by the Scan 
    # apparatus in the order of this lineup, and they are processed by the 
    # Decontamination apparatus in the exact same chosen order."
    # This constraint simplifies the problem significantly: the order does not 
    # affect the total completion time.
    # Let S_i be the scan time and D_i be the decontamination time for container i.
    # Let C_i be the completion time of the i-th container.
    # C_1 = S_1 + D_1
    # C_i = max(C_{i-1}, sum_{k=1}^i S_k) + D_i
    
    # Let's verify:
    # If we have two containers (S1, D1) and (S2, D2):
    # Order 1, 2:
    # Finish 1: S1 + D1
    # Finish 2: max(S1 + D1, S1 + S2) + D2 = max(S1 + D1 + D2, S1 + S2 + D2)
    # Order 2, 1:
    # Finish 2: S2 + D2
    # Finish 1: max(S2 + D2, S2 + S1) + D1 = max(S2 + D2 + D1, S2 + S1 + D1)
    
    # Since the order is fixed to be the same for both, we just need to 
    # calculate the completion time based on the input order.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    containers = []
    idx = 1
    for _ in range(j):
        s = int(input_data[idx])
        d = int(input_data[idx+1])
        containers.append((s, d))
        idx += 2
        
    current_scan_time = 0
    current_decon_finish_time = 0
    
    for s, d in containers:
        current_scan_time += s
        # The decontamination can start at max(current_scan_time, current_decon_finish_time)
        # But wait, the problem says "A container can begin its Decontamination 
        # only after its Scan is completely finished."
        # And "The containers are processed by the Decontamination apparatus 
        # in the exact same chosen order."
        # This means the Decontamination apparatus is free only after the 
        # previous container's Decontamination is finished.
        
        start_decon = max(current_scan_time, current_decon_finish_time)
        current_decon_finish_time = start_decon + d
        
    print(current_decon_finish_time)

if __name__ == '__main__':
    solve()
```
