import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # To minimize the makespan on two machines, we partition the jobs into two sets:
    # Set 1: jobs where p1 < p2
    # Set 2: jobs where p1 >= p2
    # Sort Set 1 by p1 in ascending order.
    # Sort Set 2 by p2 in descending order.
    # The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    order = set1 + set2
    
    time_m1 = 0
    time_m2 = 0
    
    for p1, p2 in order:
        time_m1 += p1
        if time_m1 > time_m2:
            time_m2 = time_m1 + p2
        else:
            time_m2 += p2
            
    print(time_m2)

if __name__ == '__main__':
    solve()
