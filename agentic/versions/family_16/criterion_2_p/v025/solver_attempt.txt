import sys

def solve():
    # This is a classic Johnson's Rule problem for a two-machine flow shop.
    # We want to minimize the makespan of a sequence of jobs on two machines.
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where A_i < B_i
    #    Set 2: jobs where A_i >= B_i
    # 2. Sort Set 1 in ascending order of A_i.
    # 3. Sort Set 2 in descending order of B_i.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        jobs.append((a, b))
        idx += 2
        
    set1 = []
    set2 = []
    for a, b in jobs:
        if a < b:
            set1.append((a, b))
        else:
            set2.append((a, b))
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    # time_a: completion time of current job on Alignment Bay
    # time_b: completion time of current job on Sync Bay
    time_a = 0
    time_b = 0
    
    for a, b in sequence:
        time_a += a
        # Sync Bay can start only after Alignment Bay finishes current job
        # and after Sync Bay finishes previous job
        if time_a > time_b:
            time_b = time_a + b
        else:
            time_b = time_b + b
            
    print(time_b)

if __name__ == '__main__':
    solve()
