```python
import sys

def solve():
    # This is a classic scheduling problem known as Johnson's Rule.
    # To minimize the makespan of two-stage flow shop scheduling:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where wash_time < dry_time
    #    Set 2: jobs where wash_time >= dry_time
    # 2. Sort Set 1 in ascending order of wash_time.
    # 3. Sort Set 2 in descending order of dry_time.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    v = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(v):
        w = int(input_data[idx])
        d = int(input_data[idx+1])
        jobs.append((w, d))
        idx += 2
        
    set1 = []
    set2 = []
    
    for w, d in jobs:
        if w < d:
            set1.append((w, d))
        else:
            set2.append((w, d))
            
    # Sort set1 by wash time ascending
    set1.sort(key=lambda x: x[0])
    # Sort set2 by dry time descending
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate completion time
    wash_finish = 0
    dry_finish = 0
    
    for w, d in sequence:
        wash_finish += w
        # Drying starts at max(wash_finish, previous_dry_finish)
        if wash_finish > dry_finish:
            dry_finish = wash_finish + d
        else:
            dry_finish += d
            
    print(dry_finish)

if __name__ == '__main__':
    solve()
```
