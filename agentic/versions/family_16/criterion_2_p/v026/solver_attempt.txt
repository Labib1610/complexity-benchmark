import sys

def solve():
    # The problem is a variation of Johnson's Rule for a 2-machine flow shop.
    # We have j jobs, each with processing times t_B and t_C.
    # We want to minimize the makespan (the time the last job finishes on machine C).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where t_B < t_C
    #    Set 2: jobs where t_B >= t_C
    # 2. Sort Set 1 in ascending order of t_B.
    # 3. Sort Set 2 in descending order of t_C.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        t_b = int(input_data[idx])
        t_c = int(input_data[idx+1])
        jobs.append((t_b, t_c))
        idx += 2
        
    set1 = []
    set2 = []
    
    for job in jobs:
        if job[0] < job[1]:
            set1.append(job)
        else:
            set2.append(job)
            
    set1.sort(key=lambda x: x[0])
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate makespan
    # time_b: completion time of current job on machine B
    # time_c: completion time of current job on machine C
    time_b = 0
    time_c = 0
    
    for t_b, t_c in sequence:
        time_b += t_b
        # Machine C can start only after machine B finishes the current job
        # and after machine C finishes the previous job.
        if time_c < time_b:
            time_c = time_b + t_c
        else:
            time_c += t_c
            
    print(time_c)

if __name__ == '__main__':
    solve()
