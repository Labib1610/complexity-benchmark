```python
import sys

def solve():
    # The problem is a variation of Johnson's Rule for the two-machine flow shop scheduling problem.
    # We have C jobs, each with two tasks: Illumination (time t1) and Extraction (time t2).
    # The goal is to minimize the makespan (total time).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set A: jobs where t1 < t2
    #    Set B: jobs where t1 >= t2
    # 2. Sort Set A in ascending order of t1.
    # 3. Sort Set B in descending order of t2.
    # 4. The optimal sequence is Set A followed by Set B.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    c = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(c):
        t1 = int(input_data[idx])
        t2 = int(input_data[idx+1])
        jobs.append((t1, t2))
        idx += 2
        
    set_a = [j for j in jobs if j[0] < j[1]]
    set_b = [j for j in jobs if j[0] >= j[1]]
    
    set_a.sort(key=lambda x: x[0])
    set_b.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set_a + set_b
    
    # Calculate makespan
    # current_illumination_time tracks when the Lightbearer finishes the current job
    # current_extraction_time tracks when the Miner finishes the current job
    current_illumination_time = 0
    current_extraction_time = 0
    
    for t1, t2 in sequence:
        current_illumination_time += t1
        # Miner can start extraction only after Lightbearer finishes this job
        # AND after Miner finishes the previous job
        current_extraction_time = max(current_illumination_time, current_extraction_time) + t2
        
    print(current_extraction_time)

if __name__ == '__main__':
    solve()
```
