```python
import sys

def solve():
    # This is a classic application of Johnson's Rule for the 2-machine flow shop problem.
    # The goal is to minimize the makespan (total time to complete all jobs).
    # Johnson's Rule:
    # 1. Divide jobs into two sets:
    #    Set 1: jobs where p1 < p2
    #    Set 2: jobs where p1 >= p2
    # 2. Sort Set 1 in ascending order of p1.
    # 3. Sort Set 2 in descending order of p2.
    # 4. The optimal sequence is Set 1 followed by Set 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        jobs.append((p1, p2))
        idx += 2
        
    set1 = []
    set2 = []
    
    for p1, p2 in jobs:
        if p1 < p2:
            set1.append((p1, p2))
        else:
            set2.append((p1, p2))
            
    # Sort set1 by p1 ascending
    set1.sort(key=lambda x: x[0])
    # Sort set2 by p2 descending
    set2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = set1 + set2
    
    # Calculate makespan
    # primary_finish_time tracks when the primary processor finishes the current item
    # secondary_finish_time tracks when the secondary processor finishes the current item
    primary_finish_time = 0
    secondary_finish_time = 0
    
    for p1, p2 in sequence:
        primary_finish_time += p1
        # Secondary processor can start only after primary finishes the current item
        # AND after the secondary processor finishes the previous item.
        secondary_finish_time = max(primary_finish_time, secondary_finish_time) + p2
        
    print(secondary_finish_time)

if __name__ == '__main__':
    solve()
```
