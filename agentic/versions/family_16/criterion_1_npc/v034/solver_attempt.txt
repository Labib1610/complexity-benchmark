import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    s = int(input_data[0])
    c = int(input_data[1])
    
    scrolls = []
    idx = 2
    for _ in range(s):
        p = int(input_data[idx])
        idx += 1
        phases = []
        for _ in range(p):
            chamber = int(input_data[idx])
            duration = int(input_data[idx+1])
            phases.append((chamber, duration))
            idx += 2
        scrolls.append(phases)
    
    # State: (current_time, scroll_indices_tuple, chamber_free_times_tuple)
    # scroll_indices: tuple of current phase index for each scroll
    # chamber_free_times: tuple of time when each chamber becomes free
    
    initial_scroll_indices = tuple([0] * s)
    initial_chamber_free_times = tuple([0] * c)
    
    # Priority queue for Dijkstra: (time, scroll_indices, chamber_free_times)
    pq = [(0, initial_scroll_indices, initial_chamber_free_times)]
    visited = {}
    
    while pq:
        curr_time, scroll_indices, chamber_free_times = heapq.heappop(pq)
        
        if all(idx == len(scrolls[i]) for i, idx in enumerate(scroll_indices)):
            print(curr_time)
            return
        
        state = (scroll_indices, chamber_free_times)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        # Try to advance any scroll that has a next phase
        for i in range(s):
            if scroll_indices[i] < len(scrolls[i]):
                chamber, duration = scrolls[i][scroll_indices[i]]
                chamber_idx = chamber - 1
                
                # Earliest start time for this phase:
                # 1. When the scroll finished its previous phase (if any)
                # 2. When the chamber becomes free
                
                # We need to track when each scroll finished its last phase
                # Let's calculate that from the state
                # Actually, the state needs to track when each scroll is ready
                # Let's refine the state: (scroll_ready_times, chamber_free_times)
                pass

    # Re-implementing with correct state tracking
    # State: (scroll_ready_times, chamber_free_times, scroll_phase_indices)
    
    pq = [(0, tuple([0]*s), tuple([0]*c), tuple([0]*s))]
    visited = {}
    
    while pq:
        time, scroll_ready_times, chamber_free_times, scroll_phase_indices = heapq.heappop(pq)
        
        if all(idx == len(scrolls[i]) for i, idx in enumerate(scroll_phase_indices)):
            print(time)
            return
        
        state = (scroll_ready_times, chamber_free_times, scroll_phase_indices)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        for i in range(s):
            if scroll_phase_indices[i] < len(scrolls[i]):
                chamber, duration = scrolls[i][scroll_phase_indices[i]]
                chamber_idx = chamber - 1
                
                start_time = max(scroll_ready_times[i], chamber_free_times[chamber_idx])
                finish_time = start_time + duration
                
                new_scroll_ready_times = list(scroll_ready_times)
                new_scroll_ready_times[i] = finish_time
                
                new_chamber_free_times = list(chamber_free_times)
                new_chamber_free_times[chamber_idx] = finish_time
                
                new_scroll_phase_indices = list(scroll_phase_indices)
                new_scroll_phase_indices[i] += 1
                
                new_time = max(time, finish_time)
                heapq.heappush(pq, (new_time, tuple(new_scroll_ready_times), tuple(new_chamber_free_times), tuple(new_scroll_phase_indices)))

solve()
