import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    num_jobs = int(input_data[idx])
    num_machines = int(input_data[idx+1])
    idx += 2
    
    jobs = []
    for _ in range(num_jobs):
        num_ops = int(input_data[idx])
        idx += 1
        ops = []
        for _ in range(num_ops):
            m = int(input_data[idx])
            d = int(input_data[idx+1])
            ops.append((m, d))
            idx += 2
        jobs.append(ops)
    
    # State: (tuple of current operation index for each job)
    # We use A* search to find the minimum makespan.
    # Priority queue stores (estimated_total_time, current_time, job_progress_tuple, machine_free_times)
    
    initial_progress = tuple([0] * num_jobs)
    initial_machine_free = tuple([0] * (num_machines + 1))
    
    # pq: (f_score, current_makespan, progress_tuple, machine_free_times)
    pq = [(0, 0, initial_progress, initial_machine_free)]
    visited = {}
    
    while pq:
        f, makespan, progress, machine_free = heapq.heappop(pq)
        
        if progress == tuple(len(job) for job in jobs):
            print(makespan)
            return
        
        state = (progress, machine_free)
        if visited.get(state, float('inf')) <= makespan:
            continue
        visited[state] = makespan
        
        # Try scheduling the next operation of any job that isn't finished
        for i in range(num_jobs):
            if progress[i] < len(jobs[i]):
                m, d = jobs[i][progress[i]]
                
                # Job i's next op can start at max(job_ready_time, machine_ready_time)
                # job_ready_time is the finish time of the previous op of this job
                # We need to track job_ready_times. Let's add it to the state.
                pass

    # The state needs to track job_ready_times as well.
    # Revised state: (progress_tuple, machine_free_times, job_ready_times)
    
    pq = [(0, initial_progress, initial_machine_free, tuple([0]*num_jobs))]
    visited = {}
    
    while pq:
        _, progress, machine_free, job_ready = heapq.heappop(pq)
        
        if progress == tuple(len(job) for job in jobs):
            print(max(job_ready))
            return
        
        state = (progress, machine_free, job_ready)
        if visited.get(state, float('inf')) <= max(job_ready):
            continue
        visited[state] = max(job_ready)
        
        for i in range(num_jobs):
            if progress[i] < len(jobs[i]):
                m, d = jobs[i][progress[i]]
                start_time = max(job_ready[i], machine_free[m])
                finish_time = start_time + d
                
                new_progress = list(progress)
                new_progress[i] += 1
                new_progress = tuple(new_progress)
                
                new_machine_free = list(machine_free)
                new_machine_free[m] = finish_time
                new_machine_free = tuple(new_machine_free)
                
                new_job_ready = list(job_ready)
                new_job_ready[i] = finish_time
                new_job_ready = tuple(new_job_ready)
                
                heapq.heappush(pq, (max(new_job_ready), new_progress, new_machine_free, new_job_ready))

solve()
