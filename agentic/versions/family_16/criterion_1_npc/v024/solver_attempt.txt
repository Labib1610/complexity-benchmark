import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    protocols = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        tests = []
        for _ in range(o):
            inst = int(input_data[idx])
            dur = int(input_data[idx + 1])
            tests.append((inst, dur))
            idx += 2
        protocols.append(tests)
    
    # State: (tuple of current test indices for each sample, 
    #         tuple of next available time for each instrument,
    #         tuple of next available time for each sample)
    
    initial_state = (tuple([0] * j), tuple([0] * (m + 1)), tuple([0] * j))
    
    # Priority Queue stores (time, state)
    pq = [(0, initial_state)]
    visited = {initial_state: 0}
    
    while pq:
        curr_time, (test_indices, inst_free_time, sample_free_time) = heapq.heappop(pq)
        
        if curr_time > visited.get((test_indices, inst_free_time, sample_free_time), float('inf')):
            continue
            
        if all(test_indices[i] == len(protocols[i]) for i in range(j)):
            print(curr_time)
            return
        
        # Try to schedule the next test for each sample that isn't finished
        for i in range(j):
            if test_indices[i] < len(protocols[i]):
                inst, dur = protocols[i][test_indices[i]]
                
                # Earliest start time for this test
                start_time = max(inst_free_time[inst], sample_free_time[i])
                finish_time = start_time + dur
                
                new_test_indices = list(test_indices)
                new_test_indices[i] += 1
                
                new_inst_free_time = list(inst_free_time)
                new_inst_free_time[inst] = finish_time
                
                new_sample_free_time = list(sample_free_time)
                new_sample_free_time[i] = finish_time
                
                new_state = (tuple(new_test_indices), tuple(new_inst_free_time), tuple(new_sample_free_time))
                
                if finish_time < visited.get(new_state, float('inf')):
                    visited[new_state] = finish_time
                    heapq.heappush(pq, (finish_time, new_state))

solve()
