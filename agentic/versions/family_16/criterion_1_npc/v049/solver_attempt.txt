import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    commissions = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            station = int(input_data[idx])
            duration = int(input_data[idx+1])
            steps.append((station, duration))
            idx += 2
        commissions.append(steps)
    
    # State: (tuple of current step indices for each commission, tuple of next available time for each station)
    # Since the total number of steps is small (50), we can use BFS/Dijkstra.
    # However, the state space is large. Given the constraints and the nature of the problem,
    # this is a job shop scheduling problem. With j, m <= 10 and total steps <= 50,
    # we can use a priority queue to explore the state space.
    
    # state: (current_step_indices, station_free_times)
    # We want to minimize the time when all steps are finished.
    # This is equivalent to finding the shortest path in the state space.
    
    initial_steps = tuple([0] * j)
    initial_stations = tuple([0] * (m + 1))
    
    # Priority Queue stores (time_finished, current_steps, station_free_times)
    pq = [(0, initial_steps, initial_stations)]
    visited = {}
    
    while pq:
        time, steps, stations = heapq.heappop(pq)
        
        if all(steps[i] == len(commissions[i]) for i in range(j)):
            print(time)
            return
        
        state = (steps, stations)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to schedule the next available step for each commission
        for i in range(j):
            if steps[i] < len(commissions[i]):
                s_idx = steps[i]
                station, duration = commissions[i][s_idx]
                
                # Earliest start time for this step:
                # 1. When the previous step of this commission finished (tracked by time)
                # 2. When the station becomes free
                
                # We need to track the finish time of each commission separately
                # Let's refine the state: (commission_finish_times, station_free_times, steps)
                pass

    # Re-evaluating: The state needs to track when each commission is free to start its next step.
    # State: (tuple of commission_ready_times, tuple of station_free_times, tuple of step_indices)
    
    pq = [(0, tuple([0]*j), tuple([0]*(m+1)), tuple([0]*j))]
    visited = {}
    
    while pq:
        curr_time, comm_ready, stat_free, steps = heapq.heappop(pq)
        
        if all(steps[i] == len(commissions[i]) for i in range(j)):
            print(curr_time)
            return
        
        state = (comm_ready, stat_free, steps)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        for i in range(j):
            if steps[i] < len(commissions[i]):
                station, duration = commissions[i][steps[i]]
                start_time = max(comm_ready[i], stat_free[station])
                finish_time = start_time + duration
                
                new_steps = list(steps)
                new_steps[i] += 1
                new_comm_ready = list(comm_ready)
                new_comm_ready[i] = finish_time
                new_stat_free = list(stat_free)
                new_stat_free[station] = finish_time
                
                new_time = max(curr_time, finish_time)
                heapq.heappush(pq, (new_time, tuple(new_comm_ready), tuple(new_stat_free), tuple(new_steps)))

solve()
