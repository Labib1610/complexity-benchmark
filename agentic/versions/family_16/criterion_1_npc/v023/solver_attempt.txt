import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    R = int(input_data[0])
    L = int(input_data[1])
    
    idx = 2
    researchers = []
    for _ in range(R):
        num_exp = int(input_data[idx])
        idx += 1
        seq = []
        for _ in range(num_exp):
            lab = int(input_data[idx])
            dur = int(input_data[idx+1])
            seq.append((lab, dur))
            idx += 2
        researchers.append(seq)
    
    # State: (current_time, researcher_idx, experiment_idx, lab_free_times, res_free_times)
    # Since the total number of experiments is small (50), we can use a priority queue
    # to simulate the process. However, the choice of which experiment to start next
    # is not trivial. This is a scheduling problem.
    # Given the constraints (R <= 10, L <= 10, total experiments <= 50),
    # we can use a greedy approach with backtracking or a search.
    # Actually, since we want the *earliest* completion time, and the order of 
    # experiments for each researcher is fixed, we can use a priority queue to 
    # explore the state space.
    
    # State: (time, next_exp_indices, lab_free_times, res_free_times)
    # next_exp_indices: tuple of indices of the next experiment for each researcher
    # lab_free_times: tuple of free times for each lab
    # res_free_times: tuple of free times for each researcher
    
    import heapq
    
    initial_next_exp = tuple([0] * R)
    initial_lab_free = tuple([0] * L)
    initial_res_free = tuple([0] * R)
    
    # Priority queue stores (time, next_exp_indices, lab_free_times, res_free_times)
    pq = [(0, initial_next_exp, initial_lab_free, initial_res_free)]
    visited = {}
    
    min_total_time = float('inf')
    
    while pq:
        curr_time, next_exp, lab_free, res_free = heapq.heappop(pq)
        
        if all(next_exp[i] == len(researchers[i]) for i in range(R)):
            print(curr_time)
            return
        
        state = (next_exp, lab_free, res_free)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        # Try to start any available experiment
        can_start_any = False
        for i in range(R):
            if next_exp[i] < len(researchers[i]):
                lab_id, dur = researchers[i][next_exp[i]]
                lab_idx = lab_id - 1
                
                start_time = max(curr_time, lab_free[lab_idx], res_free[i])
                finish_time = start_time + dur
                
                new_next_exp = list(next_exp)
                new_next_exp[i] += 1
                
                new_lab_free = list(lab_free)
                new_lab_free[lab_idx] = finish_time
                
                new_res_free = list(res_free)
                new_res_free[i] = finish_time
                
                heapq.heappush(pq, (max(curr_time, finish_time), tuple(new_next_exp), tuple(new_lab_free), tuple(new_res_free)))
                can_start_any = True
        
        # If no experiment can be started, jump to the next event
        if not can_start_any:
            continue

solve()
