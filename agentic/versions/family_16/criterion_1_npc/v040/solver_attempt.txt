import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    workpieces = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            bay = int(input_data[idx])
            dur = int(input_data[idx + 1])
            steps.append((bay, dur))
            idx += 2
        workpieces.append(steps)
    
    # State: (current_time, workpiece_index, step_index)
    # We need to track when each bay becomes free and when each workpiece becomes free.
    # Since the total number of steps is small (50), we can use a priority queue
    # to simulate the process. However, the order of execution matters.
    # This is a job shop scheduling problem. With j, m <= 10, we can use backtracking
    # or a greedy approach if the order is fixed. But the problem implies we choose
    # the order. Actually, the problem asks for the earliest time.
    # Given the constraints and the nature of the problem, we can use a search
    # to find the optimal sequence of operations.
    
    # State: (tuple of current step index for each workpiece, 
    #         tuple of finish time for each workpiece,
    #         tuple of finish time for each bay)
    
    import functools
    
    @functools.lru_cache(None)
    def get_earliest(steps_done, wp_finish_times, bay_finish_times):
        if all(steps_done[i] == len(workpieces[i]) for i in range(j)):
            return max(wp_finish_times)
        
        res = float('inf')
        for i in range(j):
            if steps_done[i] < len(workpieces[i]):
                bay, dur = workpieces[i][steps_done[i]]
                # Can start this step at max(wp_finish_times[i], bay_finish_times[bay-1])
                start_time = max(wp_finish_times[i], bay_finish_times[bay-1])
                end_time = start_time + dur
                
                new_steps_done = list(steps_done)
                new_steps_done[i] += 1
                
                new_wp_finish = list(wp_finish_times)
                new_wp_finish[i] = end_time
                
                new_bay_finish = list(bay_finish_times)
                new_bay_finish[bay-1] = end_time
                
                res = min(res, get_earliest(tuple(new_steps_done), tuple(new_wp_finish), tuple(new_bay_finish)))
        return res

    print(get_earliest(tuple([0]*j), tuple([0]*j), tuple([0]*m)))

solve()
