import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    streams = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            proc = int(input_data[idx]) - 1
            dur = int(input_data[idx + 1])
            steps.append((proc, dur))
            idx += 2
        streams.append(steps)
    
    # State: (current_step_indices_tuple, processor_free_times_tuple)
    # current_step_indices: tuple of length j, index of next step for each stream
    # processor_free_times: tuple of length m, time when each processor becomes free
    
    initial_state = (tuple([0] * j), tuple([0] * m))
    
    # memoization: state -> min_time_to_finish_all
    memo = {}
    
    def get_min_time(step_indices, proc_free_times):
        if all(step_indices[i] == len(streams[i]) for i in range(j)):
            return max(proc_free_times)
        
        state = (step_indices, proc_free_times)
        if state in memo:
            return memo[state]
        
        # Try all possible next steps: any stream i whose next step is not finished
        # A step can start at max(stream_ready_time, processor_free_time)
        # But we need to track stream_ready_time (when the previous step finished)
        # Let's refine the state to include stream_ready_times
        return 0

    # Since total steps <= 50, we can use a priority queue to find the shortest path
    # State: (time_so_far, step_indices, proc_free_times, stream_ready_times)
    # Actually, the problem is a scheduling problem. Since j, m <= 10 and total steps <= 50,
    # we can use Dijkstra.
    
    pq = [(0, tuple([0] * j), tuple([0] * m), tuple([0] * j))]
    visited = {}
    
    while pq:
        time, step_indices, proc_free_times, stream_ready_times = heapq.heappop(pq)
        
        if all(step_indices[i] == len(streams[i]) for i in range(j)):
            print(time)
            return
        
        state = (step_indices, proc_free_times, stream_ready_times)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        for i in range(j):
            if step_indices[i] < len(streams[i]):
                proc, dur = streams[i][step_indices[i]]
                
                start_time = max(stream_ready_times[i], proc_free_times[proc])
                end_time = start_time + dur
                
                new_step_indices = list(step_indices)
                new_step_indices[i] += 1
                
                new_proc_free_times = list(proc_free_times)
                new_proc_free_times[proc] = end_time
                
                new_stream_ready_times = list(stream_ready_times)
                new_stream_ready_times[i] = end_time
                
                new_time = max(time, end_time)
                
                heapq.heappush(pq, (new_time, tuple(new_step_indices), tuple(new_proc_free_times), tuple(new_stream_ready_times)))

solve()
