import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    blueprints = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        stages = []
        for _ in range(o):
            f_idx = int(input_data[idx])
            dur = int(input_data[idx + 1])
            stages.append((f_idx, dur))
            idx += 2
        blueprints.append(stages)
    
    # State: (tuple of current stage indices for each blueprint, tuple of next available time for each fabricator)
    # Since the total number of stages is small (50), we can use Dijkstra to find the minimum completion time.
    # However, the state space is potentially large. Given the constraints (j <= 10, m <= 10, total stages <= 50),
    # we can use a priority queue to explore the state space.
    
    # current_stages: tuple of indices of the next stage to be performed for each blueprint
    # fabricator_free_time: tuple of times when each fabricator becomes free
    # blueprint_ready_time: tuple of times when each blueprint's current stage can start
    
    initial_stages = tuple([0] * j)
    initial_fab_free = tuple([0] * (m + 1))
    initial_bp_ready = tuple([0] * j)
    
    # pq stores (time_of_completion, current_stages, fabricator_free_time, blueprint_ready_time)
    pq = [(0, initial_stages, initial_fab_free, initial_bp_ready)]
    visited = {}
    
    while pq:
        time, stages, fab_free, bp_ready = heapq.heappop(pq)
        
        if all(stages[i] == len(blueprints[i]) for i in range(j)):
            print(time)
            return
        
        state = (stages, fab_free, bp_ready)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to schedule any available stage
        for i in range(j):
            if stages[i] < len(blueprints[i]):
                f_idx, dur = blueprints[i][stages[i]]
                
                # Start time is max of when the blueprint is ready and when the fabricator is free
                start_time = max(bp_ready[i], fab_free[f_idx])
                finish_time = start_time + dur
                
                new_stages = list(stages)
                new_stages[i] += 1
                
                new_fab_free = list(fab_free)
                new_fab_free[f_idx] = finish_time
                
                new_bp_ready = list(bp_ready)
                new_bp_ready[i] = finish_time
                
                new_state = (tuple(new_stages), tuple(new_fab_free), tuple(new_bp_ready))
                new_time = max(time, finish_time)
                
                if visited.get(new_state, float('inf')) > new_time:
                    heapq.heappush(pq, (new_time, tuple(new_stages), tuple(new_fab_free), tuple(new_bp_ready)))

solve()
