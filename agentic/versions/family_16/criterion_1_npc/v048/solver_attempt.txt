import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    songs = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        tasks = []
        for _ in range(o):
            tasks.append((int(input_data[idx]), int(input_data[idx+1])))
            idx += 2
        songs.append(tasks)
    
    # State: (current_task_indices_tuple, equipment_free_times_tuple, song_free_times_tuple)
    # Since total tasks <= 50 and j, m <= 10, we use Dijkstra to find the min time.
    # However, the state space is large. Let's use a priority queue to explore states.
    # State: (time, task_indices, equipment_free_times, song_free_times)
    
    initial_task_indices = tuple([0] * j)
    initial_equip_free = tuple([0] * (m + 1))
    initial_song_free = tuple([0] * j)
    
    pq = [(0, initial_task_indices, initial_equip_free, initial_song_free)]
    visited = {}
    
    while pq:
        time, task_indices, equip_free, song_free = heapq.heappop(pq)
        
        if all(task_indices[i] == len(songs[i]) for i in range(j)):
            print(time)
            return
        
        state = (task_indices, equip_free, song_free)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to start any available task
        for i in range(j):
            if task_indices[i] < len(songs[i]):
                equip_id, duration = songs[i][task_indices[i]]
                
                # Earliest start time for this task
                start_time = max(song_free[i], equip_free[equip_id])
                finish_time = start_time + duration
                
                new_task_indices = list(task_indices)
                new_task_indices[i] += 1
                new_task_indices = tuple(new_task_indices)
                
                new_equip_free = list(equip_free)
                new_equip_free[equip_id] = finish_time
                new_equip_free = tuple(new_equip_free)
                
                new_song_free = list(song_free)
                new_song_free[i] = finish_time
                new_song_free = tuple(new_song_free)
                
                heapq.heappush(pq, (max(time, finish_time), new_task_indices, new_equip_free, new_song_free))

solve()
