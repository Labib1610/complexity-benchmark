import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    projects = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            ws = int(input_data[idx])
            dur = int(input_data[idx+1])
            steps.append((ws, dur))
            idx += 2
        projects.append(steps)
    
    # State: (tuple of current step indices for each project, tuple of finish times for each workstation)
    # Since the number of steps is small (50 total), we can use Dijkstra to find the minimum makespan.
    # However, the state space is large. Given the constraints (j, m <= 10), 
    # we can use a priority queue to explore the state space.
    
    # state: (current_step_indices, workstation_free_times, project_ready_times)
    # current_step_indices: tuple of length j
    # workstation_free_times: tuple of length m
    # project_ready_times: tuple of length j
    
    initial_state = (tuple([0]*j), tuple([0]*m), tuple([0]*j))
    pq = [(0, initial_state)]
    visited = {}
    
    while pq:
        makespan, (step_indices, ws_free, proj_ready) = heapq.heappop(pq)
        
        if all(step_indices[i] == len(projects[i]) for i in range(j)):
            print(makespan)
            return
        
        if (step_indices, ws_free, proj_ready) in visited and visited[(step_indices, ws_free, proj_ready)] <= makespan:
            continue
        visited[(step_indices, ws_free, proj_ready)] = makespan
        
        # Try to start any available step
        for i in range(j):
            if step_indices[i] < len(projects[i]):
                ws, dur = projects[i][step_indices[i]]
                ws_idx = ws - 1
                
                start_time = max(proj_ready[i], ws_free[ws_idx])
                finish_time = start_time + dur
                
                new_step_indices = list(step_indices)
                new_step_indices[i] += 1
                
                new_ws_free = list(ws_free)
                new_ws_free[ws_idx] = finish_time
                
                new_proj_ready = list(proj_ready)
                new_proj_ready[i] = finish_time
                
                new_makespan = max(makespan, finish_time)
                
                state = (tuple(new_step_indices), tuple(new_ws_free), tuple(new_proj_ready))
                if state not in visited or visited[state] > new_makespan:
                    heapq.heappush(pq, (new_makespan, state))

solve()
