```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    queries = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            res = int(input_data[idx])
            dur = int(input_data[idx+1])
            steps.append((res, dur))
            idx += 2
        queries.append(steps)
    
    # State: (tuple of current step indices for each query, tuple of finish times for each resource)
    # Since j <= 10 and total steps <= 50, we can use BFS/Dijkstra.
    # However, the state space is large. Given the constraints and the nature of the problem
    # (Job Shop Scheduling), we can use a priority queue to explore the state space.
    
    # state: (current_step_indices, resource_free_times, query_ready_times)
    # current_step_indices: tuple of length j
    # resource_free_times: tuple of length m
    # query_ready_times: tuple of length j (when the next step of query i can start)
    
    initial_state = (tuple([0] * j), tuple([0] * m), tuple([0] * j))
    pq = [(0, initial_state)]
    visited = {}
    
    min_finish = float('inf')
    
    while pq:
        cost, (step_indices, res_free, q_ready) = heapq.heappop(pq)
        
        if all(step_indices[i] == len(queries[i]) for i in range(j)):
            print(cost)
            return
        
        if visited.get((step_indices, res_free, q_ready), float('inf')) <= cost:
            continue
        visited[(step_indices, res_free, q_ready)] = cost
        
        # Try to schedule any available next step
        for i in range(j):
            if step_indices[i] < len(queries[i]):
                res, dur = queries[i][step_indices[i]]
                res_idx = res - 1
                
                start_time = max(q_ready[i], res_free[res_idx])
                finish_time = start_time + dur
                
                new_step_indices = list(step_indices)
                new_step_indices[i] += 1
                
                new_res_free = list(res_free)
                new_res_free[res_idx] = finish_time
                
                new_q_ready = list(q_ready)
                new_q_ready[i] = finish_time
                
                new_state = (tuple(new_step_indices), tuple(new_res_free), tuple(new_q_ready))
                new_cost = max(cost, finish_time)
                
                if new_cost < visited.get(new_state, float('inf')):
                    heapq.heappush(pq, (new_cost, new_state))

solve()
```
