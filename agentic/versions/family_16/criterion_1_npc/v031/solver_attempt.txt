import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    dishes = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            ws = int(input_data[idx])
            dur = int(input_data[idx+1])
            steps.append((ws, dur))
            idx += 2
        dishes.append(steps)
    
    # State: (tuple of current step indices for each dish, tuple of finish times for each workstation)
    # Since the total number of steps is small (50), and j, m <= 10, we can use BFS/Dijkstra.
    # However, the state space for workstation finish times is large.
    # Given the constraints and the nature of the problem (Job Shop Scheduling),
    # we can use a priority queue to explore the state space.
    
    # State: (current_time, dish_progress_indices, workstation_free_times)
    # dish_progress_indices: tuple of length j
    # workstation_free_times: tuple of length m
    
    initial_progress = tuple([0] * j)
    initial_ws_times = tuple([0] * m)
    
    # Priority Queue stores (time, progress, ws_times)
    # We want to minimize the time when all dishes are finished.
    # Actually, we can use a simple search.
    
    pq = [(0, initial_progress, initial_ws_times)]
    visited = {}
    
    min_finish_time = float('inf')
    
    while pq:
        curr_time, progress, ws_times = heapq.heappop(pq)
        
        if all(progress[i] == len(dishes[i]) for i in range(j)):
            print(curr_time)
            return
        
        state = (progress, ws_times)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        # Try to schedule the next step of any dish that isn't finished
        for i in range(j):
            if progress[i] < len(dishes[i]):
                ws, dur = dishes[i][progress[i]]
                ws_idx = ws - 1
                
                # The step can start at max(dish_ready_time, ws_free_time)
                # We need to track dish_ready_time.
                # Let's refine the state to include dish_ready_times.
                pass

    # Re-evaluating: The state needs to track when each dish is ready for its next step.
    # State: (dish_ready_times, ws_free_times, progress)
    
    pq = [(0, tuple([0]*j), tuple([0]*m), tuple([0]*j))]
    visited = {}
    
    while pq:
        time, dish_ready_times, ws_free_times, progress = heapq.heappop(pq)
        
        if all(progress[i] == len(dishes[i]) for i in range(j)):
            print(time)
            return
        
        state = (dish_ready_times, ws_free_times, progress)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        for i in range(j):
            if progress[i] < len(dishes[i]):
                ws, dur = dishes[i][progress[i]]
                ws_idx = ws - 1
                
                start_time = max(dish_ready_times[i], ws_free_times[ws_idx])
                finish_time = start_time + dur
                
                new_progress = list(progress)
                new_progress[i] += 1
                
                new_dish_ready = list(dish_ready_times)
                new_dish_ready[i] = finish_time
                
                new_ws_free = list(ws_free_times)
                new_ws_free[ws_idx] = finish_time
                
                heapq.heappush(pq, (max(time, finish_time), tuple(new_dish_ready), tuple(new_ws_free), tuple(new_progress)))

solve()
