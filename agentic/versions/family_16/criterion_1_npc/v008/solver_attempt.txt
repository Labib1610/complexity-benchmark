import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    num_jobs = int(input_data[idx])
    num_machines = int(input_data[idx+1])
    idx += 2
    
    jobs = []
    for _ in range(num_jobs):
        num_ops = int(input_data[idx])
        idx += 1
        ops = []
        for _ in range(num_ops):
            m = int(input_data[idx])
            d = int(input_data[idx+1])
            ops.append((m, d))
            idx += 2
        jobs.append(ops)
    
    # State: tuple of current operation index for each job
    # memo stores the minimum finish time for a given state
    memo = {}
    
    # job_progress[i] is the index of the next operation to perform for job i
    # machine_free_time[m] is the time when machine m becomes free
    # job_free_time[j] is the time when job j becomes free
    
    def get_min_time(progress, machine_free_time, job_free_time):
        state = (progress, machine_free_time, job_free_time)
        if state in memo:
            return memo[state]
        
        # Check if all jobs are finished
        if all(progress[i] == len(jobs[i]) for i in range(num_jobs)):
            return max(job_free_time)
        
        res = float('inf')
        
        # Try scheduling the next operation of any job that isn't finished
        for i in range(num_jobs):
            if progress[i] < len(jobs[i]):
                m, d = jobs[i][progress[i]]
                # Machine index is 1-based
                m_idx = m - 1
                
                # Start time is max of when machine is free and when job is free
                start_time = max(machine_free_time[m_idx], job_free_time[i])
                finish_time = start_time + d
                
                new_progress = list(progress)
                new_progress[i] += 1
                
                new_m_free = list(machine_free_time)
                new_m_free[m_idx] = finish_time
                
                new_j_free = list(job_free_time)
                new_j_free[i] = finish_time
                
                res = min(res, get_min_time(tuple(new_progress), tuple(new_m_free), tuple(new_j_free)))
        
        memo[state] = res
        return res

    # Since the state space is small enough for the constraints (50 total ops),
    # we can use Dijkstra-like approach or memoized recursion.
    # Given the constraints, simple recursion with memoization is sufficient.
    
    print(get_min_time(tuple([0]*num_jobs), tuple([0]*num_machines), tuple([0]*num_jobs)))

solve()
