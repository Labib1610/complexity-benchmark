import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    sequences = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        seq = []
        for _ in range(o):
            emitter = int(input_data[idx])
            duration = int(input_data[idx + 1])
            seq.append((emitter, duration))
            idx += 2
        sequences.append(seq)
    
    # state: tuple of current phase index for each sequence
    # memo: dictionary to store min time to reach state
    memo = {}
    
    # current_phase_idx: tuple of length j
    # emitter_free_time: tuple of length m
    # sequence_free_time: tuple of length j
    
    # Since the total number of phases is small (50), we can use BFS/Dijkstra
    # However, the state space is defined by the progress in each sequence.
    # Let's use a priority queue to find the earliest completion time.
    
    import heapq
    
    # state: (tuple of progress indices)
    # We want to find the minimum time to reach (o1, o2, ..., oj)
    # But the time depends on when each emitter and sequence becomes free.
    
    # Actually, for a fixed set of progress indices, the earliest time 
    # each sequence can start its next phase is determined by:
    # 1. When the previous phase of that sequence finished.
    # 2. When the required emitter becomes free.
    
    # Let's use Dijkstra:
    # State: (progress_indices)
    # Value: (sequence_free_times, emitter_free_times)
    
    initial_progress = tuple([0] * j)
    initial_seq_free = tuple([0] * j)
    initial_emit_free = tuple([0] * m)
    
    pq = [(0, initial_progress, initial_seq_free, initial_emit_free)]
    visited = {}
    
    while pq:
        time, progress, seq_free, emit_free = heapq.heappop(pq)
        
        if progress == tuple(len(s) for s in sequences):
            print(time)
            return
        
        state = (progress, seq_free, emit_free)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        for i in range(j):
            if progress[i] < len(sequences[i]):
                emitter, duration = sequences[i][progress[i]]
                emitter_idx = emitter - 1
                
                # Earliest start time for this phase
                start_time = max(seq_free[i], emit_free[emitter_idx])
                finish_time = start_time + duration
                
                new_progress = list(progress)
                new_progress[i] += 1
                new_progress = tuple(new_progress)
                
                new_seq_free = list(seq_free)
                new_seq_free[i] = finish_time
                new_seq_free = tuple(new_seq_free)
                
                new_emit_free = list(emit_free)
                new_emit_free[emitter_idx] = finish_time
                new_emit_free = tuple(new_emit_free)
                
                new_time = max(time, finish_time)
                heapq.heappush(pq, (new_time, new_progress, new_seq_free, new_emit_free))

solve()
