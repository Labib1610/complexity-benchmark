import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    petitions = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            mag = int(input_data[idx])
            hours = int(input_data[idx+1])
            steps.append((mag, hours))
            idx += 2
        petitions.append(steps)
    
    # State: (tuple of current step indices for each petition, tuple of next available time for each magistrate)
    # Since j and m are small (up to 10), we can use BFS/Dijkstra.
    # However, the state space is large. Given the constraints and the nature of the problem,
    # we need to find the minimum time. This is a scheduling problem.
    # Since we want to minimize the completion time, we can use a priority queue to explore states.
    
    # state: (current_step_indices, magistrate_free_times)
    # current_step_indices: tuple of length j
    # magistrate_free_times: tuple of length m
    
    initial_steps = tuple([0] * j)
    initial_mags = tuple([0] * m)
    
    # pq stores (time_of_last_completion, current_step_indices, magistrate_free_times)
    pq = [(0, initial_steps, initial_mags)]
    visited = {}
    
    min_total_time = float('inf')
    
    while pq:
        time, steps, mags = heapq.heappop(pq)
        
        if all(steps[i] == len(petitions[i]) for i in range(j)):
            print(time)
            return
        
        state = (steps, mags)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to advance any petition that has a next step
        for i in range(j):
            if steps[i] < len(petitions[i]):
                mag, duration = petitions[i][steps[i]]
                mag_idx = mag - 1
                
                # The petition can start at max(time_petition_ready, time_magistrate_free)
                # But we don't track petition_ready time explicitly, 
                # it's the finish time of the previous step.
                # Let's refine the state: (steps, mags, finish_times_of_petitions)
                pass

    # Re-evaluating: The problem is a classic job shop scheduling problem.
    # Given the small constraints (j, m <= 10, total steps <= 50), 
    # we can use a simple backtracking with pruning or a priority queue.
    
    def get_min_time(steps, mags, finish_times):
        if all(steps[i] == len(petitions[i]) for i in range(j)):
            return max(finish_times)
        
        state = (steps, mags, finish_times)
        if state in memo: return memo[state]
        
        res = float('inf')
        for i in range(j):
            if steps[i] < len(petitions[i]):
                mag, duration = petitions[i][steps[i]]
                mag_idx = mag - 1
                
                start_time = max(finish_times[i], mags[mag_idx])
                end_time = start_time + duration
                
                new_steps = list(steps)
                new_steps[i] += 1
                new_mags = list(mags)
                new_mags[mag_idx] = end_time
                new_finish_times = list(finish_times)
                new_finish_times[i] = end_time
                
                res = min(res, get_min_time(tuple(new_steps), tuple(new_mags), tuple(new_finish_times)))
        
        memo[state] = res
        return res

    memo = {}
    print(get_min_time(tuple([0]*j), tuple([0]*m), tuple([0]*j)))

solve()
