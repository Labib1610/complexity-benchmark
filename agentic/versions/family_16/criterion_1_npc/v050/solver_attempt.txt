import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    K = int(input_data[1])
    
    compounds = []
    idx = 2
    for _ in range(N):
        S = int(input_data[idx])
        idx += 1
        stages = []
        for _ in range(S):
            c = int(input_data[idx]) - 1
            d = int(input_data[idx+1])
            stages.append((c, d))
            idx += 2
        compounds.append(stages)
    
    # State: (tuple of current stage indices for each compound, 
    #         tuple of finish times for each crystal, 
    #         tuple of finish times for each compound)
    # Since N and K are small, we can use A* or Dijkstra.
    # Priority queue stores (time, current_stage_indices, crystal_free_times, compound_free_times)
    
    initial_stages = tuple([0] * N)
    initial_crystal_free = tuple([0] * K)
    initial_compound_free = tuple([0] * N)
    
    pq = [(0, initial_stages, initial_crystal_free, initial_compound_free)]
    visited = {}
    
    while pq:
        time, stages, crystal_free, compound_free = heapq.heappop(pq)
        
        if all(stages[i] == len(compounds[i]) for i in range(N)):
            print(time)
            return
        
        state = (stages, crystal_free, compound_free)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to start any available stage
        for i in range(N):
            if stages[i] < len(compounds[i]):
                c, d = compounds[i][stages[i]]
                
                # Earliest start time for this stage
                start_time = max(compound_free[i], crystal_free[c])
                finish_time = start_time + d
                
                new_stages = list(stages)
                new_stages[i] += 1
                new_stages = tuple(new_stages)
                
                new_crystal_free = list(crystal_free)
                new_crystal_free[c] = finish_time
                new_crystal_free = tuple(new_crystal_free)
                
                new_compound_free = list(compound_free)
                new_compound_free[i] = finish_time
                new_compound_free = tuple(new_compound_free)
                
                new_time = max(time, finish_time)
                heapq.heappush(pq, (new_time, new_stages, new_crystal_free, new_compound_free))

solve()
