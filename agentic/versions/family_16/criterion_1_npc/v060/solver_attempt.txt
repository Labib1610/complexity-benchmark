import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    potions = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        stages = []
        for _ in range(o):
            app = int(input_data[idx])
            dur = int(input_data[idx+1])
            stages.append((app, dur))
            idx += 2
        potions.append(stages)
    
    # State: (tuple of current stage index for each potion, tuple of next available time for each apparatus)
    # Since j <= 10 and total stages <= 50, we can use BFS/Dijkstra.
    # However, the state space is large. Given the constraints and the nature of the problem,
    # this is a Job Shop Scheduling problem. With j, m <= 10, we can use a priority queue
    # to explore the state space.
    
    # current_stages: tuple of indices of the next stage to perform for each potion
    # apparatus_free_time: tuple of times when each apparatus becomes free
    # potion_free_time: tuple of times when each potion becomes free
    
    initial_state = (tuple([0] * j), tuple([0] * (m + 1)), tuple([0] * j))
    pq = [(0, initial_state)]
    visited = {}
    
    min_finish_time = float('inf')
    
    while pq:
        time, (stages_idx, app_free, pot_free) = heapq.heappop(pq)
        
        if all(stages_idx[i] == len(potions[i]) for i in range(j)):
            print(time)
            return
        
        if visited.get((stages_idx, app_free, pot_free), float('inf')) <= time:
            continue
        visited[(stages_idx, app_free, pot_free)] = time
        
        # Try to schedule any available stage
        for i in range(j):
            if stages_idx[i] < len(potions[i]):
                app, dur = potions[i][stages_idx[i]]
                start_time = max(pot_free[i], app_free[app])
                finish_time = start_time + dur
                
                new_stages_idx = list(stages_idx)
                new_stages_idx[i] += 1
                
                new_app_free = list(app_free)
                new_app_free[app] = finish_time
                
                new_pot_free = list(pot_free)
                new_pot_free[i] = finish_time
                
                new_state = (tuple(new_stages_idx), tuple(new_app_free), tuple(new_pot_free))
                heapq.heappush(pq, (max(time, finish_time), new_state))

solve()
