import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    potions = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            app = int(input_data[idx])
            dur = int(input_data[idx+1])
            steps.append((app, dur))
            idx += 2
        potions.append(steps)
    
    # State: (tuple of current step indices for each potion, 
    #         tuple of finish times for each apparatus,
    #         tuple of finish times for each potion)
    # Since j and m are small (up to 10), we can use A* or Dijkstra.
    # However, the state space is potentially large. 
    # Given the constraints and the nature of the problem, 
    # this is a scheduling problem.
    
    # Priority Queue stores: (time_to_finish_all, current_step_indices, apparatus_free_times, potion_free_times)
    # Actually, we can simplify: at any point, we can choose to start the next available step 
    # of any potion that is not currently being processed.
    
    # State: (tuple of current step indices, tuple of apparatus free times, tuple of potion free times)
    initial_steps = tuple([0] * j)
    initial_app_free = tuple([0] * (m + 1))
    initial_pot_free = tuple([0] * j)
    
    pq = [(0, initial_steps, initial_app_free, initial_pot_free)]
    visited = {}
    
    while pq:
        time, steps, app_free, pot_free = heapq.heappop(pq)
        
        if all(steps[i] == len(potions[i]) for i in range(j)):
            print(time)
            return
        
        state = (steps, app_free, pot_free)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to start any available next step
        for i in range(j):
            if steps[i] < len(potions[i]):
                app, dur = potions[i][steps[i]]
                
                # Earliest start time for this step
                start_time = max(app_free[app], pot_free[i])
                finish_time = start_time + dur
                
                new_steps = list(steps)
                new_steps[i] += 1
                
                new_app_free = list(app_free)
                new_app_free[app] = finish_time
                
                new_pot_free = list(pot_free)
                new_pot_free[i] = finish_time
                
                new_time = max(time, finish_time)
                
                heapq.heappush(pq, (new_time, tuple(new_steps), tuple(new_app_free), tuple(new_pot_free)))

solve()
