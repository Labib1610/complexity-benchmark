import sys
from itertools import permutations

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    j = int(input_data[0])
    m = int(input_data[1])
    
    proposals = []
    idx = 2
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        stages = []
        for _ in range(o):
            expert = int(input_data[idx]) - 1
            duration = int(input_data[idx+1])
            stages.append((expert, duration))
            idx += 2
        proposals.append(stages)
    
    # For each expert, we need to decide the order in which they process the stages
    # assigned to them. Since j is small (up to 10), we can try all permutations
    # of stages for each expert.
    
    expert_stages = [[] for _ in range(m)]
    for p_idx, stages in enumerate(proposals):
        for s_idx, (exp, dur) in enumerate(stages):
            expert_stages[exp].append((p_idx, s_idx, dur))
            
    # Generate all permutations of stages for each expert
    import itertools
    
    expert_perms = []
    for e in range(m):
        expert_perms.append(list(itertools.permutations(expert_stages[e])))
        
    min_total_time = float('inf')
    
    # Iterate over all combinations of permutations
    for p in itertools.product(*expert_perms):
        # p[e] is the sequence of stages for expert e
        # We need to find the completion time for each stage
        # stage_finish[p_idx][s_idx]
        stage_finish = [[0] * len(proposals[i]) for i in range(j)]
        
        # expert_next_available[e]
        expert_next_available = [0] * m
        
        # We need to process stages in an order that respects dependencies
        # A stage (p_idx, s_idx) depends on (p_idx, s_idx-1)
        # and the previous stage assigned to the same expert.
        
        # Since we have a fixed order for each expert, we can use a topological sort
        # or simply iterate until convergence. Given the constraints, 
        # we can use a simple simulation.
        
        # Track how many stages of each proposal are finished
        proposal_stage_idx = [0] * j
        # Track how many stages of each expert's permutation are finished
        expert_stage_idx = [0] * m
        
        finished_count = 0
        total_stages = sum(len(p) for p in proposals)
        
        while finished_count < total_stages:
            progress = False
            for e in range(m):
                if expert_stage_idx[e] < len(p[e]):
                    p_idx, s_idx, dur = p[e][expert_stage_idx[e]]
                    # Can this stage start?
                    # 1. Previous stage of this proposal must be finished
                    # 2. Expert must be free
                    prev_finish = stage_finish[p_idx][s_idx - 1] if s_idx > 0 else 0
                    start_time = max(prev_finish, expert_next_available[e])
                    
                    stage_finish[p_idx][s_idx] = start_time + dur
                    expert_next_available[e] = start_time + dur
                    expert_stage_idx[e] += 1
                    finished_count += 1
                    progress = True
            if not progress:
                break
        
        current_max = 0
        for i in range(j):
            current_max = max(current_max, stage_finish[i][-1])
        min_total_time = min(min_total_time, current_max)
        
    print(min_total_time)

solve()
