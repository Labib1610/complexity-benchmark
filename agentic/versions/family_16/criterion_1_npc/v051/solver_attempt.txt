import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    dossiers = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        reviews = []
        for _ in range(o):
            dept = int(input_data[idx])
            dur = int(input_data[idx + 1])
            reviews.append((dept, dur))
            idx += 2
        dossiers.append(reviews)
    
    # State: (tuple of current review indices for each dossier, tuple of next available time for each department)
    # Since j and m are small (up to 10), we can use BFS/Dijkstra.
    # However, the state space is large. But we only care about the order of reviews.
    # Actually, this is a job shop scheduling problem. Given the constraints (j, m <= 10),
    # we can use a search algorithm.
    
    # Priority Queue stores (current_time, progress_tuple, dept_free_times)
    # progress_tuple: (review_index_for_dossier_0, ..., review_index_for_dossier_j-1)
    # dept_free_times: (time_dept_1_free, ..., time_dept_m_free)
    
    initial_progress = tuple([0] * j)
    initial_dept_free = tuple([0] * m)
    
    # Using Dijkstra to find the minimum time to complete all reviews
    pq = [(0, initial_progress, initial_dept_free)]
    visited = {}
    
    while pq:
        curr_time, progress, dept_free = heapq.heappop(pq)
        
        if all(progress[i] == len(dossiers[i]) for i in range(j)):
            print(curr_time)
            return
        
        state = (progress, dept_free)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        # Try to start any available review
        for i in range(j):
            if progress[i] < len(dossiers[i]):
                dept, dur = dossiers[i][progress[i]]
                dept_idx = dept - 1
                
                # When can this review start?
                # 1. After previous review of this dossier finishes
                # 2. After the department is free
                
                # We need to track when each dossier's previous review finished
                # Let's refine the state to include dossier_free_times
                pass

    # Re-evaluating: The state needs to track when each dossier is ready for its next review.
    # State: (dossier_ready_times, dept_free_times, progress_tuple)
    
    pq = [(0, tuple([0]*j), tuple([0]*m), tuple([0]*j))]
    visited = {}
    
    while pq:
        time, dossier_ready, dept_free, progress = heapq.heappop(pq)
        
        if all(progress[i] == len(dossiers[i]) for i in range(j)):
            print(time)
            return
        
        state = (dossier_ready, dept_free, progress)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        for i in range(j):
            if progress[i] < len(dossiers[i]):
                dept, dur = dossiers[i][progress[i]]
                dept_idx = dept - 1
                
                start_time = max(dossier_ready[i], dept_free[dept_idx])
                finish_time = start_time + dur
                
                new_progress = list(progress)
                new_progress[i] += 1
                new_progress = tuple(new_progress)
                
                new_dossier_ready = list(dossier_ready)
                new_dossier_ready[i] = finish_time
                new_dossier_ready = tuple(new_dossier_ready)
                
                new_dept_free = list(dept_free)
                new_dept_free[dept_idx] = finish_time
                new_dept_free = tuple(new_dept_free)
                
                heapq.heappush(pq, (max(time, finish_time), new_dossier_ready, new_dept_free, new_progress))

solve()
