import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    petitions = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        hearings = []
        for _ in range(o):
            mag = int(input_data[idx])
            dur = int(input_data[idx + 1])
            hearings.append((mag, dur))
            idx += 2
        petitions.append(hearings)
    
    # State: (tuple of current hearing indices for each petition, 
    #         tuple of next available time for each magistrate,
    #         tuple of next available time for each petition)
    # Since j and m are small (up to 10), we can use BFS/Dijkstra.
    # However, the state space is large. Given the constraints and the nature
    # of the problem, we can use a priority queue to explore states.
    
    # State: (time_finished, current_hearing_indices, mag_free_times, pet_free_times)
    # To optimize, we only care about the earliest time a petition can start its next hearing.
    
    initial_pet_indices = tuple([0] * j)
    initial_mag_free = tuple([0] * (m + 1))
    initial_pet_free = tuple([0] * j)
    
    # Priority Queue stores (time, pet_indices, mag_free, pet_free)
    pq = [(0, initial_pet_indices, initial_mag_free, initial_pet_free)]
    visited = {}
    
    while pq:
        time, pet_indices, mag_free, pet_free = heapq.heappop(pq)
        
        if all(pet_indices[i] == len(petitions[i]) for i in range(j)):
            print(time)
            return
        
        state = (pet_indices, mag_free, pet_free)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to start any available hearing
        for i in range(j):
            if pet_indices[i] < len(petitions[i]):
                mag, dur = petitions[i][pet_indices[i]]
                start_time = max(pet_free[i], mag_free[mag])
                end_time = start_time + dur
                
                new_pet_indices = list(pet_indices)
                new_pet_indices[i] += 1
                
                new_mag_free = list(mag_free)
                new_mag_free[mag] = end_time
                
                new_pet_free = list(pet_free)
                new_pet_free[i] = end_time
                
                new_state = (tuple(new_pet_indices), tuple(new_mag_free), tuple(new_pet_free))
                heapq.heappush(pq, (max(time, end_time), tuple(new_pet_indices), tuple(new_mag_free), tuple(new_pet_free)))

solve()
