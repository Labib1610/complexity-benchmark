import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    initiatives = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        stages = []
        for _ in range(o):
            s = int(input_data[idx])
            h = int(input_data[idx+1])
            stages.append((s, h))
            idx += 2
        initiatives.append(stages)
    
    # State: (tuple of current stage indices for each initiative, 
    #         tuple of next available time for each station,
    #         tuple of next available time for each initiative)
    # Since the total number of stages is small (50), we can use A* or Dijkstra.
    # However, the state space is large. Given the constraints (j, m <= 10),
    # we can use a priority queue to explore the state space.
    
    # state: (time_finished, current_stage_indices, station_free_times, initiative_free_times)
    # To optimize, we only need to decide which initiative to schedule next.
    # A greedy approach won't work, but we can use a search.
    
    initial_stage_indices = tuple([0] * j)
    initial_station_free = tuple([0] * (m + 1))
    initial_initiative_free = tuple([0] * j)
    
    # Priority Queue stores: (time, stage_indices, station_free, initiative_free)
    pq = [(0, initial_stage_indices, initial_station_free, initial_initiative_free)]
    visited = {}
    
    min_total_time = float('inf')
    
    while pq:
        time, stage_indices, station_free, initiative_free = heapq.heappop(pq)
        
        if all(stage_indices[i] == len(initiatives[i]) for i in range(j)):
            print(time)
            return
        
        state = (stage_indices, station_free, initiative_free)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        for i in range(j):
            if stage_indices[i] < len(initiatives[i]):
                s_idx = stage_indices[i]
                station, duration = initiatives[i][s_idx]
                
                start_time = max(initiative_free[i], station_free[station])
                end_time = start_time + duration
                
                new_stage_indices = list(stage_indices)
                new_stage_indices[i] += 1
                
                new_station_free = list(station_free)
                new_station_free[station] = end_time
                
                new_initiative_free = list(initiative_free)
                new_initiative_free[i] = end_time
                
                heapq.heappush(pq, (max(time, end_time), tuple(new_stage_indices), tuple(new_station_free), tuple(new_initiative_free)))

solve()
