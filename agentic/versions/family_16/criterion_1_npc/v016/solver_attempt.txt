import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    orders = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        tasks = []
        for _ in range(o):
            ws = int(input_data[idx])
            dur = int(input_data[idx + 1])
            tasks.append((ws, dur))
            idx += 2
        orders.append(tasks)
    
    # State: (tuple of current task indices for each order)
    # We use Dijkstra to find the minimum time to reach the state where all tasks are done.
    # However, the state space is too large for simple Dijkstra.
    # Given the constraints (j <= 10, total tasks <= 50), this is a Job Shop Scheduling problem.
    # Since j and m are small, we can use a search with pruning or a heuristic.
    # Actually, with j, m <= 10, we can use a simple state-space search.
    
    memo = {}
    
    def get_min_time(task_indices, order_finish_times, ws_free_times):
        state = (task_indices, order_finish_times, ws_free_times)
        if state in memo:
            return memo[state]
        
        if all(task_indices[i] == len(orders[i]) for i in range(j)):
            return max(order_finish_times)
        
        res = float('inf')
        for i in range(j):
            if task_indices[i] < len(orders[i]):
                ws, dur = orders[i][task_indices[i]]
                start_time = max(order_finish_times[i], ws_free_times[ws-1])
                
                new_task_indices = list(task_indices)
                new_task_indices[i] += 1
                
                new_order_finish_times = list(order_finish_times)
                new_order_finish_times[i] = start_time + dur
                
                new_ws_free_times = list(ws_free_times)
                new_ws_free_times[ws-1] = start_time + dur
                
                res = min(res, get_min_time(tuple(new_task_indices), tuple(new_order_finish_times), tuple(new_ws_free_times)))
        
        memo[state] = res
        return res

    # The state space is still potentially large. Let's use a priority queue (Dijkstra).
    pq = [(0, tuple([0]*j), tuple([0]*j), tuple([0]*m))]
    visited = {}
    
    while pq:
        time, task_indices, order_finish_times, ws_free_times = heapq.heappop(pq)
        
        if all(task_indices[i] == len(orders[i]) for i in range(j)):
            print(time)
            return
        
        state = (task_indices, order_finish_times, ws_free_times)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        for i in range(j):
            if task_indices[i] < len(orders[i]):
                ws, dur = orders[i][task_indices[i]]
                start_time = max(order_finish_times[i], ws_free_times[ws-1])
                
                new_task_indices = list(task_indices)
                new_task_indices[i] += 1
                
                new_order_finish_times = list(order_finish_times)
                new_order_finish_times[i] = start_time + dur
                
                new_ws_free_times = list(ws_free_times)
                new_ws_free_times[ws-1] = start_time + dur
                
                new_time = max(time, start_time + dur)
                heapq.heappush(pq, (new_time, tuple(new_task_indices), tuple(new_order_finish_times), tuple(new_ws_free_times)))

solve()
