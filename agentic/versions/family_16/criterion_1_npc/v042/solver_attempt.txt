import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    songs = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        sessions = []
        for _ in range(o):
            room = int(input_data[idx])
            duration = int(input_data[idx+1])
            sessions.append((room, duration))
            idx += 2
        songs.append(sessions)
    
    # State: (tuple of current session indices for each song, tuple of finish times for each room)
    # Since j <= 10 and total sessions <= 50, we use A* or Dijkstra to find the minimum time.
    # The state is (progress_indices, room_free_times)
    # progress_indices: tuple of length j
    # room_free_times: tuple of length m
    
    initial_progress = tuple([0] * j)
    initial_rooms = tuple([0] * m)
    
    # Priority Queue stores (time, progress_indices, room_free_times)
    pq = [(0, initial_progress, initial_rooms)]
    visited = {}
    
    while pq:
        time, progress, rooms = heapq.heappop(pq)
        
        if all(progress[i] == len(songs[i]) for i in range(j)):
            print(time)
            return
        
        state = (progress, rooms)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to start any available session
        for i in range(j):
            if progress[i] < len(songs[i]):
                room, duration = songs[i][progress[i]]
                room_idx = room - 1
                
                # Earliest start time for this session:
                # 1. When the previous session of this song finished
                # 2. When the room becomes free
                
                # We need to track when each song's previous session finished.
                # This is implicitly the time when the song's progress was updated.
                # To keep state simple, we calculate the finish time of the previous session.
                # Let's store song_finish_times in the state.
                pass

    # Refined approach: State includes (progress_indices, room_free_times, song_free_times)
    # Given the constraints and the nature of the problem, this is a job shop scheduling problem.
    # Since j, m <= 10 and total sessions <= 50, we can use a simple search.
    
    pq = [(0, tuple([0]*j), tuple([0]*m), tuple([0]*j))]
    visited = {}
    
    while pq:
        curr_time, progress, rooms, song_free = heapq.heappop(pq)
        
        if all(progress[i] == len(songs[i]) for i in range(j)):
            print(curr_time)
            return
        
        state = (progress, rooms, song_free)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        for i in range(j):
            if progress[i] < len(songs[i]):
                room, duration = songs[i][progress[i]]
                room_idx = room - 1
                
                start_time = max(song_free[i], rooms[room_idx])
                finish_time = start_time + duration
                
                new_progress = list(progress)
                new_progress[i] += 1
                new_rooms = list(rooms)
                new_rooms[room_idx] = finish_time
                new_song_free = list(song_free)
                new_song_free[i] = finish_time
                
                new_time = max(curr_time, finish_time)
                heapq.heappush(pq, (new_time, tuple(new_progress), tuple(new_rooms), tuple(new_song_free)))

solve()
