import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    orders = []
    for _ in range(j):
        num_phases = int(input_data[idx])
        idx += 1
        phases = []
        for _ in range(num_phases):
            unit = int(input_data[idx])
            duration = int(input_data[idx + 1])
            phases.append((unit, duration))
            idx += 2
        orders.append(phases)
    
    # State: (tuple of current phase indices for each order, tuple of finish times for each unit)
    # Since the total number of phases is small (50), we can use A* or Dijkstra.
    # However, the state space for unit finish times is large.
    # Given the constraints (j, m <= 10, total phases <= 50), we can use a search.
    
    # Actually, this is a Job Shop Scheduling problem, which is NP-hard.
    # With j, m <= 10 and total phases <= 50, we can use a priority queue to explore
    # the state space of (order_progress, unit_availability).
    
    # To keep the state space manageable, we use the fact that we only care about
    # the next available phase for each order.
    
    initial_progress = tuple([0] * j)
    initial_unit_times = tuple([0] * m)
    initial_order_ready_times = tuple([0] * j)
    
    # pq stores (current_max_time, progress, unit_times, order_ready_times)
    pq = [(0, initial_progress, initial_unit_times, initial_order_ready_times)]
    visited = {}
    
    best_finish = float('inf')
    
    while pq:
        curr_time, progress, unit_times, order_ready_times = heapq.heappop(pq)
        
        if curr_time >= best_finish:
            continue
            
        if all(progress[i] == len(orders[i]) for i in range(j)):
            best_finish = min(best_finish, curr_time)
            continue
            
        state = (progress, unit_times, order_ready_times)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        # Try to schedule the next phase of any order that is ready
        for i in range(j):
            if progress[i] < len(orders[i]):
                unit, duration = orders[i][progress[i]]
                unit_idx = unit - 1
                
                start_time = max(order_ready_times[i], unit_times[unit_idx])
                finish_time = start_time + duration
                
                new_progress = list(progress)
                new_progress[i] += 1
                new_progress = tuple(new_progress)
                
                new_unit_times = list(unit_times)
                new_unit_times[unit_idx] = finish_time
                new_unit_times = tuple(new_unit_times)
                
                new_order_ready_times = list(order_ready_times)
                new_order_ready_times[i] = finish_time
                new_order_ready_times = tuple(new_order_ready_times)
                
                new_max_time = max(curr_time, finish_time)
                
                heapq.heappush(pq, (new_max_time, new_progress, new_unit_times, new_order_ready_times))
                
    print(best_finish)

solve()
