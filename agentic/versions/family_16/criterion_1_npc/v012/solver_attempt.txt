import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    compositions = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        segments = []
        for _ in range(o):
            inst = int(input_data[idx])
            dur = int(input_data[idx+1])
            segments.append((inst, dur))
            idx += 2
        compositions.append(segments)
    
    # State: (tuple of current segment indices for each composition, 
    #         tuple of finish times for each instrument)
    # Since we want the earliest finish time, we use Dijkstra.
    # However, the state space is small enough that we can use memoization.
    
    memo = {}
    
    def get_min_time(indices, inst_finish_times):
        state = (indices, inst_finish_times)
        if state in memo:
            return memo[state]
        
        if all(indices[i] == len(compositions[i]) for i in range(j)):
            return max(inst_finish_times)
        
        res = float('inf')
        
        # Try to schedule the next segment of any composition that can be started
        for i in range(j):
            if indices[i] < len(compositions[i]):
                inst, dur = compositions[i][indices[i]]
                
                # Earliest start time for this segment:
                # 1. After the previous segment of this composition finishes
                # 2. After the instrument is free
                
                # We need to track when each composition's last segment finished
                # Let's add comp_finish_times to the state
                pass
        
        # Re-evaluating: The state needs to track:
        # 1. indices: current segment index for each composition
        # 2. comp_finish_times: when each composition's last segment finished
        # 3. inst_finish_times: when each instrument is free
        return res

    # Given the constraints (j, m <= 10, total segments <= 50), 
    # we can use a priority queue to explore states (Dijkstra).
    # State: (current_time, indices, comp_finish_times, inst_finish_times)
    # Actually, we don't need current_time in the state, just the finish times.
    
    pq = [(0, tuple([0]*j), tuple([0]*j), tuple([0]*m))]
    visited = {}
    
    while pq:
        time, indices, comp_finish_times, inst_finish_times = heapq.heappop(pq)
        
        if all(indices[i] == len(compositions[i]) for i in range(j)):
            print(time)
            return
        
        state = (indices, comp_finish_times, inst_finish_times)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        for i in range(j):
            if indices[i] < len(compositions[i]):
                inst, dur = compositions[i][indices[i]]
                inst_idx = inst - 1
                
                start_time = max(comp_finish_times[i], inst_finish_times[inst_idx])
                finish_time = start_time + dur
                
                new_indices = list(indices)
                new_indices[i] += 1
                
                new_comp_finish = list(comp_finish_times)
                new_comp_finish[i] = finish_time
                
                new_inst_finish = list(inst_finish_times)
                new_inst_finish[inst_idx] = finish_time
                
                new_time = max(time, finish_time)
                heapq.heappush(pq, (new_time, tuple(new_indices), tuple(new_comp_finish), tuple(new_inst_finish)))

solve()
