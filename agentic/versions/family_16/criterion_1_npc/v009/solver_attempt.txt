import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    num_jobs = int(input_data[idx])
    num_machines = int(input_data[idx+1])
    idx += 2
    
    jobs = []
    for _ in range(num_jobs):
        num_ops = int(input_data[idx])
        idx += 1
        ops = []
        for _ in range(num_ops):
            m = int(input_data[idx])
            d = int(input_data[idx+1])
            ops.append((m, d))
            idx += 2
        jobs.append(ops)
        
    # State: tuple of current operation index for each job
    # memo stores the minimum time to complete the remaining operations
    # However, the state needs to include machine availability times.
    # Given the constraints (total ops <= 50, j, m <= 10), we can use 
    # a search with pruning or memoization.
    
    memo = {}
    
    def get_min_time(job_progress, machine_free_time, job_free_time):
        state = (job_progress, machine_free_time, job_free_time)
        if state in memo:
            return memo[state]
        
        if all(job_progress[i] == len(jobs[i]) for i in range(num_jobs)):
            return max(job_free_time)
        
        res = float('inf')
        
        # Try scheduling any available next operation
        can_schedule = False
        for i in range(num_jobs):
            if job_progress[i] < len(jobs[i]):
                can_schedule = True
                m, d = jobs[i][job_progress[i]]
                
                # Earliest start time for this operation
                start_time = max(job_free_time[i], machine_free_time[m-1])
                end_time = start_time + d
                
                new_job_progress = list(job_progress)
                new_job_progress[i] += 1
                
                new_machine_free_time = list(machine_free_time)
                new_machine_free_time[m-1] = end_time
                
                new_job_free_time = list(job_free_time)
                new_job_free_time[i] = end_time
                
                res = min(res, get_min_time(tuple(new_job_progress), 
                                            tuple(new_machine_free_time), 
                                            tuple(new_job_free_time)))
        
        if not can_schedule:
            return max(job_free_time)
            
        memo[state] = res
        return res

    # The state space is too large for simple memoization.
    # Use A* or simple backtracking with pruning.
    # Given the constraints and the nature of the problem (Job Shop Scheduling),
    # we can use a simple recursive search with a heuristic.
    
    best_time = float('inf')
    
    def backtrack(job_progress, machine_free_time, job_free_time, current_max):
        nonlocal best_time
        if current_max >= best_time:
            return
        
        if all(job_progress[i] == len(jobs[i]) for i in range(num_jobs)):
            best_time = min(best_time, current_max)
            return
        
        for i in range(num_jobs):
            if job_progress[i] < len(jobs[i]):
                m, d = jobs[i][job_progress[i]]
                start_time = max(job_free_time[i], machine_free_time[m-1])
                end_time = start_time + d
                
                old_m_time = machine_free_time[m-1]
                old_j_time = job_free_time[i]
                
                job_progress[i] += 1
                machine_free_time[m-1] = end_time
                job_free_time[i] = end_time
                
                backtrack(job_progress, machine_free_time, job_free_time, max(current_max, end_time))
                
                job_progress[i] -= 1
                machine_free_time[m-1] = old_m_time
                job_free_time[i] = old_j_time

    backtrack([0]*num_jobs, [0]*num_machines, [0]*num_jobs, 0)
    print(best_time)

solve()
