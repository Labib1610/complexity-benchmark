import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    orders = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        stages = []
        for _ in range(o):
            station = int(input_data[idx])
            duration = int(input_data[idx + 1])
            stages.append((station, duration))
            idx += 2
        orders.append(stages)
    
    # State: tuple of current stage index for each order
    # We use Dijkstra to find the minimum time to reach the state where all stages are done.
    # However, the state needs to include when each station becomes free and when each order becomes free.
    # Given the constraints (j, m <= 10, total stages <= 50), we can use A* or Dijkstra.
    # State: (tuple of current_stage_indices_for_each_order, tuple of free_time_for_each_station, tuple of free_time_for_each_order)
    
    initial_order_progress = tuple([0] * j)
    initial_station_free = tuple([0] * m)
    initial_order_free = tuple([0] * j)
    
    # Priority Queue: (time, order_progress, station_free, order_free)
    pq = [(0, initial_order_progress, initial_station_free, initial_order_free)]
    visited = {}
    
    while pq:
        time, order_progress, station_free, order_free = heapq.heappop(pq)
        
        if all(order_progress[i] == len(orders[i]) for i in range(j)):
            print(time)
            return
        
        state = (order_progress, station_free, order_free)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to start any available next stage
        for i in range(j):
            if order_progress[i] < len(orders[i]):
                station, duration = orders[i][order_progress[i]]
                station_idx = station - 1
                
                # Earliest start time for this stage
                start_time = max(order_free[i], station_free[station_idx])
                finish_time = start_time + duration
                
                new_order_progress = list(order_progress)
                new_order_progress[i] += 1
                new_order_progress = tuple(new_order_progress)
                
                new_station_free = list(station_free)
                new_station_free[station_idx] = finish_time
                new_station_free = tuple(new_station_free)
                
                new_order_free = list(order_free)
                new_order_free[i] = finish_time
                new_order_free = tuple(new_order_free)
                
                new_time = max(time, finish_time)
                
                heapq.heappush(pq, (new_time, new_order_progress, new_station_free, new_order_free))

solve()
