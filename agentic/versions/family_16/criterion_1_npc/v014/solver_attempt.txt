import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    commissions = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            w = int(input_data[idx])
            d = int(input_data[idx + 1])
            steps.append((w, d))
            idx += 2
        commissions.append(steps)
    
    # commission_next_step_idx[c] stores the index of the next step to be processed for commission c
    commission_next_step_idx = [0] * j
    # commission_ready_time[c] stores the time when the previous step of commission c finished
    commission_ready_time = [0] * j
    # workstation_free_time[w] stores the time when workstation w becomes free
    workstation_free_time = [0] * (m + 1)
    
    # We need to schedule steps. A step can start at max(commission_ready_time[c], workstation_free_time[w])
    # Since we want to minimize the final completion time, this is a classic job shop scheduling problem.
    # Given the constraints (50 steps total), we can use a greedy approach with a priority queue
    # to always pick the next available step that can start earliest.
    
    # However, the problem asks for the earliest possible completion time.
    # In job shop scheduling, the greedy approach (earliest start time) is a heuristic.
    # Given the small number of steps (50), we can simulate the process.
    
    finished_steps = 0
    total_steps = sum(len(c) for c in commissions)
    
    while finished_steps < total_steps:
        best_start_time = float('inf')
        best_c = -1
        
        for c in range(j):
            if commission_next_step_idx[c] < len(commissions[c]):
                w, d = commissions[c][commission_next_step_idx[c]]
                start_time = max(commission_ready_time[c], workstation_free_time[w])
                if start_time < best_start_time:
                    best_start_time = start_time
                    best_c = c
        
        c = best_c
        w, d = commissions[c][commission_next_step_idx[c]]
        start_time = best_start_time
        finish_time = start_time + d
        
        commission_ready_time[c] = finish_time
        workstation_free_time[w] = finish_time
        commission_next_step_idx[c] += 1
        finished_steps += 1
        
    print(max(commission_ready_time))

solve()
