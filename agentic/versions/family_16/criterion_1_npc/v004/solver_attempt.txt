import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    jobs = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        ops = []
        for _ in range(o):
            machine = int(input_data[idx])
            duration = int(input_data[idx+1])
            ops.append((machine, duration))
            idx += 2
        jobs.append(ops)
    
    # State: tuple of current operation index for each job
    # memo stores the minimum finish time for a given state
    memo = {}
    
    # job_progress[i] is the index of the next operation to perform for job i
    # machine_free_time[m] is the time when machine m becomes free
    # job_free_time[j] is the time when job j becomes free
    
    def get_min_finish_time(progress, machine_free_time, job_free_time):
        state = (progress, machine_free_time, job_free_time)
        if state in memo:
            return memo[state]
        
        # Check if all jobs are finished
        if all(progress[i] == len(jobs[i]) for i in range(j)):
            return max(job_free_time)
        
        res = float('inf')
        
        # Try scheduling the next operation of any job that isn't finished
        for i in range(j):
            if progress[i] < len(jobs[i]):
                m_idx, duration = jobs[i][progress[i]]
                
                # Start time is max of when machine is free and when job is free
                start_time = max(machine_free_time[m_idx-1], job_free_time[i])
                finish_time = start_time + duration
                
                new_progress = list(progress)
                new_progress[i] += 1
                
                new_machine_free_time = list(machine_free_time)
                new_machine_free_time[m_idx-1] = finish_time
                
                new_job_free_time = list(job_free_time)
                new_job_free_time[i] = finish_time
                
                res = min(res, get_min_finish_time(tuple(new_progress), tuple(new_machine_free_time), tuple(new_job_free_time)))
        
        memo[state] = res
        return res

    # Since the state space is large, use A* or Dijkstra
    # Priority Queue stores (current_max_finish_time, progress, machine_free_time, job_free_time)
    pq = [(0, tuple([0]*j), tuple([0]*m), tuple([0]*j))]
    visited = {}
    
    while pq:
        curr_time, progress, m_free, j_free = heapq.heappop(pq)
        
        if all(progress[i] == len(jobs[i]) for i in range(j)):
            print(curr_time)
            return
        
        state = (progress, m_free, j_free)
        if state in visited and visited[state] <= curr_time:
            continue
        visited[state] = curr_time
        
        for i in range(j):
            if progress[i] < len(jobs[i]):
                m_idx, duration = jobs[i][progress[i]]
                start_time = max(m_free[m_idx-1], j_free[i])
                finish_time = start_time + duration
                
                new_progress = list(progress)
                new_progress[i] += 1
                
                new_m_free = list(m_free)
                new_m_free[m_idx-1] = finish_time
                
                new_j_free = list(j_free)
                new_j_free[i] = finish_time
                
                new_time = max(curr_time, finish_time)
                heapq.heappush(pq, (new_time, tuple(new_progress), tuple(new_m_free), tuple(new_j_free)))

solve()
