import sys
from itertools import product

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    dossiers = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            clerk = int(input_data[idx])
            duration = int(input_data[idx+1])
            steps.append((clerk, duration))
            idx += 2
        dossiers.append(steps)
    
    # Each dossier i has a sequence of steps. Let s[i] be the current step index.
    # The state is (s[0], s[1], ..., s[j-1]).
    # Since total steps <= 50 and j <= 10, we can use BFS or Dijkstra.
    # However, the constraint is that a clerk can only work on one dossier at a time.
    # This is a scheduling problem. Since j and m are small, we can use the fact
    # that the order of dossiers at each clerk is a permutation.
    # But wait, the problem asks for the minimum time. This is equivalent to
    # finding a schedule. Given the constraints, we can use recursion with memoization.
    
    memo = {}

    def get_min_time(steps_done, clerk_free_time):
        # steps_done: tuple of current step index for each dossier
        # clerk_free_time: tuple of time each clerk becomes free
        state = (steps_done, clerk_free_time)
        if state in memo:
            return memo[state]
        
        if all(steps_done[i] == len(dossiers[i]) for i in range(j)):
            return max(clerk_free_time)
        
        res = float('inf')
        
        # Try to pick a dossier to advance
        for i in range(j):
            if steps_done[i] < len(dossiers[i]):
                clerk, duration = dossiers[i][steps_done[i]]
                # The clerk is 1-indexed
                c_idx = clerk - 1
                
                # The dossier can start at max(clerk_free_time[c_idx], time_dossier_finished_prev_step)
                # We need to track when each dossier finished its previous step.
                # Let's refine the state.
                pass
        return 0

    # Actually, the state needs to track when each dossier is ready for its next step.
    # Let dossier_ready_time[i] be the time dossier i is ready for its next step.
    
    memo = {}
    def solve_recursive(steps_done, dossier_ready_time, clerk_free_time):
        state = (steps_done, dossier_ready_time, clerk_free_time)
        if state in memo: return memo[state]
        
        if all(steps_done[i] == len(dossiers[i]) for i in range(j)):
            return max(clerk_free_time)
        
        best = float('inf')
        for i in range(j):
            if steps_done[i] < len(dossiers[i]):
                clerk, duration = dossiers[i][steps_done[i]]
                c_idx = clerk - 1
                
                start_time = max(dossier_ready_time[i], clerk_free_time[c_idx])
                end_time = start_time + duration
                
                new_steps = list(steps_done)
                new_steps[i] += 1
                new_dossier_ready = list(dossier_ready_time)
                new_dossier_ready[i] = end_time
                new_clerk_free = list(clerk_free_time)
                new_clerk_free[c_idx] = end_time
                
                best = min(best, solve_recursive(tuple(new_steps), tuple(new_dossier_ready), tuple(new_clerk_free)))
        
        memo[state] = best
        return best

    print(solve_recursive(tuple([0]*j), tuple([0]*j), tuple([0]*m)))

solve()
