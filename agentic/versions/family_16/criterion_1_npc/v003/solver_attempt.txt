import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    docs = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        reviews = []
        for _ in range(o):
            official = int(input_data[idx])
            duration = int(input_data[idx+1])
            reviews.append((official, duration))
            idx += 2
        docs.append(reviews)
    
    # State: (tuple of current review index for each doc, tuple of finish times for each official)
    # Since j <= 10 and total reviews <= 50, we can use BFS/Dijkstra.
    # However, the state space is large. Given the constraints and the nature of the problem,
    # we can use a priority queue to explore states (current_progress, official_free_times).
    
    # progress: tuple of indices representing how many reviews completed for each doc
    # official_free_times: tuple of times when each official becomes free
    
    initial_progress = tuple([0] * j)
    initial_free_times = tuple([0] * m)
    
    # pq stores (max_finish_time, progress, official_free_times)
    pq = [(0, initial_progress, initial_free_times)]
    visited = {}
    
    min_total_finish = float('inf')
    
    while pq:
        curr_max, progress, free_times = heapq.heappop(pq)
        
        if curr_max >= min_total_finish:
            continue
            
        if all(progress[i] == len(docs[i]) for i in range(j)):
            min_total_finish = min(min_total_finish, curr_max)
            continue
            
        state = (progress, free_times)
        if visited.get(state, float('inf')) <= curr_max:
            continue
        visited[state] = curr_max
        
        # Try to schedule the next review for each document that isn't finished
        for i in range(j):
            if progress[i] < len(docs[i]):
                off, dur = docs[i][progress[i]]
                off_idx = off - 1
                
                # The review can start at max(time doc is ready, time official is free)
                # We need to track when each document is ready.
                # To do this, we need to store doc_ready_times in the state.
                pass

    # Re-evaluating: The state needs to include doc_ready_times.
    # State: (progress, official_free_times, doc_ready_times)
    
    initial_doc_ready = tuple([0] * j)
    pq = [(0, initial_progress, initial_free_times, initial_doc_ready)]
    visited = {}
    
    while pq:
        curr_max, progress, free_times, doc_ready = heapq.heappop(pq)
        
        if all(progress[i] == len(docs[i]) for i in range(j)):
            print(curr_max)
            return
            
        state = (progress, free_times, doc_ready)
        if visited.get(state, float('inf')) <= curr_max:
            continue
        visited[state] = curr_max
        
        for i in range(j):
            if progress[i] < len(docs[i]):
                off, dur = docs[i][progress[i]]
                off_idx = off - 1
                
                start_time = max(doc_ready[i], free_times[off_idx])
                finish_time = start_time + dur
                
                new_progress = list(progress)
                new_progress[i] += 1
                new_progress = tuple(new_progress)
                
                new_free_times = list(free_times)
                new_free_times[off_idx] = finish_time
                new_free_times = tuple(new_free_times)
                
                new_doc_ready = list(doc_ready)
                new_doc_ready[i] = finish_time
                new_doc_ready = tuple(new_doc_ready)
                
                heapq.heappush(pq, (max(curr_max, finish_time), new_progress, new_free_times, new_doc_ready))

solve()
