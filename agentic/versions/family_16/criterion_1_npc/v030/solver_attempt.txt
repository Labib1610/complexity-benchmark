```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    artifacts = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            spec = int(input_data[idx])
            dur = int(input_data[idx+1])
            steps.append((spec, dur))
            idx += 2
        artifacts.append(steps)
    
    # State: (tuple of current step indices for each artifact, tuple of next available time for each specialist)
    # Since the state space can be large, we use Dijkstra to find the minimum time.
    # However, the constraints are small enough that we can use a priority queue to explore states.
    
    # current_steps: tuple of indices of the next step to perform for each artifact
    # spec_free_time: tuple of times when each specialist becomes free
    # artifact_free_time: tuple of times when each artifact becomes free
    
    initial_state = (tuple([0] * j), tuple([0] * m), tuple([0] * j))
    pq = [(0, initial_state)]
    visited = {}
    
    while pq:
        time, (curr_steps, spec_free, art_free) = heapq.heappop(pq)
        
        if all(curr_steps[i] == len(artifacts[i]) for i in range(j)):
            print(time)
            return
        
        if visited.get((curr_steps, spec_free, art_free), float('inf')) <= time:
            continue
        visited[(curr_steps, spec_free, art_free)] = time
        
        # Try to start any available step
        for i in range(j):
            if curr_steps[i] < len(artifacts[i]):
                spec, dur = artifacts[i][curr_steps[i]]
                spec_idx = spec - 1
                
                # Earliest start time for this step
                start_time = max(spec_free[spec_idx], art_free[i])
                finish_time = start_time + dur
                
                new_curr_steps = list(curr_steps)
                new_curr_steps[i] += 1
                
                new_spec_free = list(spec_free)
                new_spec_free[spec_idx] = finish_time
                
                new_art_free = list(art_free)
                new_art_free[i] = finish_time
                
                new_state = (tuple(new_curr_steps), tuple(new_spec_free), tuple(new_art_free))
                heapq.heappush(pq, (max(time, finish_time), new_state))

solve()
```
