import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    vessels = []
    for _ in range(j):
        p = int(input_data[idx])
        idx += 1
        phases = []
        for _ in range(p):
            f = int(input_data[idx])
            d = int(input_data[idx + 1])
            phases.append((f, d))
            idx += 2
        vessels.append(phases)
    
    # State: (tuple of current phase indices for each vessel, tuple of facility free times)
    # Since j <= 10 and total phases <= 50, we use Dijkstra to find the minimum time.
    # However, the state space is large. Given the constraints and the nature of the problem,
    # we can use a priority queue to explore states.
    
    # state: (time_finished, current_phase_indices, facility_free_times)
    # current_phase_indices: tuple of length j
    # facility_free_times: tuple of length m
    
    initial_phases = tuple([0] * j)
    initial_facilities = tuple([0] * m)
    
    # pq stores (time, phase_indices, facility_free_times)
    pq = [(0, initial_phases, initial_facilities)]
    visited = {}
    
    while pq:
        time, phases_idx, fac_free = heapq.heappop(pq)
        
        if all(phases_idx[i] == len(vessels[i]) for i in range(j)):
            print(time)
            return
        
        state = (phases_idx, fac_free)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to start the next phase for each vessel that isn't finished
        for i in range(j):
            if phases_idx[i] < len(vessels[i]):
                f_idx, duration = vessels[i][phases_idx[i]]
                f_idx -= 1 # 0-indexed
                
                # The vessel can start this phase at max(time_vessel_ready, fac_free[f_idx])
                # But we need to track when each vessel is ready.
                # Let's refine the state: include vessel_ready_times
                pass

    # Re-evaluating: The state needs to track when each vessel is ready for its next phase.
    # state: (vessel_ready_times, facility_free_times, phase_indices)
    
    pq = [(0, tuple([0]*j), tuple([0]*m), tuple([0]*j))]
    visited = {}
    
    while pq:
        curr_time, v_ready, f_free, p_idx = heapq.heappop(pq)
        
        if all(p_idx[i] == len(vessels[i]) for i in range(j)):
            print(curr_time)
            return
        
        state = (v_ready, f_free, p_idx)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        for i in range(j):
            if p_idx[i] < len(vessels[i]):
                f_req, dur = vessels[i][p_idx[i]]
                f_req -= 1
                
                start_time = max(v_ready[i], f_free[f_req])
                end_time = start_time + dur
                
                new_v_ready = list(v_ready)
                new_v_ready[i] = end_time
                
                new_f_free = list(f_free)
                new_f_free[f_req] = end_time
                
                new_p_idx = list(p_idx)
                new_p_idx[i] += 1
                
                heapq.heappush(pq, (max(curr_time, end_time), tuple(new_v_ready), tuple(new_f_free), tuple(new_p_idx)))

solve()
