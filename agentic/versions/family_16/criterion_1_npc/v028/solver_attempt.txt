import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    c = int(input_data[0])
    p = int(input_data[1])
    
    canvases = []
    idx = 2
    for _ in range(c):
        l = int(input_data[idx])
        idx += 1
        layers = []
        for _ in range(l):
            pigment = int(input_data[idx])
            duration = int(input_data[idx+1])
            layers.append((pigment, duration))
            idx += 2
        canvases.append(layers)
    
    # State: (tuple of current layer indices for each canvas, 
    #         tuple of finish times for each pigment)
    # Since c <= 10 and total layers <= 50, we can use BFS/Dijkstra.
    # However, the state space is large. Given the constraints and the nature
    # of the problem (Job Shop Scheduling), we can use a priority queue
    # to explore the state space.
    
    # State: (current_time, layer_indices, pigment_free_times)
    # layer_indices: tuple of length c
    # pigment_free_times: tuple of length p
    
    initial_layers = tuple([0] * c)
    initial_pigments = tuple([0] * p)
    
    # Priority Queue stores (time, layer_indices, pigment_free_times)
    pq = [(0, initial_layers, initial_pigments)]
    visited = {}
    
    min_finish_time = float('inf')
    
    while pq:
        curr_time, layer_indices, pigment_free_times = heapq.heappop(pq)
        
        if all(layer_indices[i] == len(canvases[i]) for i in range(c)):
            print(curr_time)
            return
        
        state = (layer_indices, pigment_free_times)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        # Try to start the next layer for each canvas
        for i in range(c):
            if layer_indices[i] < len(canvases[i]):
                pigment, duration = canvases[i][layer_indices[i]]
                pigment_idx = pigment - 1
                
                # The layer can start at max(curr_time_of_canvas, pigment_free_time)
                # But we don't track canvas_free_time explicitly.
                # We can derive it: it's the finish time of the previous layer.
                # Let's refine the state to include canvas_free_times.
                pass

    # Re-evaluating: The state needs to be (layer_indices, pigment_free_times, canvas_free_times)
    # Given the constraints, let's use a simpler approach:
    # Since c is small, we can use recursion with memoization.
    
    memo = {}

    def get_min_time(layer_indices, pigment_free_times, canvas_free_times):
        state = (layer_indices, pigment_free_times, canvas_free_times)
        if state in memo: return memo[state]
        
        if all(layer_indices[i] == len(canvases[i]) for i in range(c)):
            return max(canvas_free_times)
        
        res = float('inf')
        for i in range(c):
            if layer_indices[i] < len(canvases[i]):
                pigment, duration = canvases[i][layer_indices[i]]
                p_idx = pigment - 1
                
                start_time = max(canvas_free_times[i], pigment_free_times[p_idx])
                finish_time = start_time + duration
                
                new_layer_indices = list(layer_indices)
                new_layer_indices[i] += 1
                new_pigment_free = list(pigment_free_times)
                new_pigment_free[p_idx] = finish_time
                new_canvas_free = list(canvas_free_times)
                new_canvas_free[i] = finish_time
                
                res = min(res, get_min_time(tuple(new_layer_indices), tuple(new_pigment_free), tuple(new_canvas_free)))
        
        memo[state] = res
        return res

    print(get_min_time(tuple([0]*c), tuple([0]*p), tuple([0]*c)))

solve()
