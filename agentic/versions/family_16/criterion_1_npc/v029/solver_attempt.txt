import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    projects = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        phases = []
        for _ in range(o):
            f = int(input_data[idx])
            d = int(input_data[idx+1])
            phases.append((f, d))
            idx += 2
        projects.append(phases)
    
    # State: (tuple of current phase indices for each project, 
    #         tuple of finish times for each project,
    #         tuple of available times for each facility)
    # Since we want to minimize the max finish time, we use a priority queue
    # to explore states. However, the state space is large.
    # Given the constraints (j, m <= 10, total phases <= 50), 
    # we can use a greedy approach with backtracking or a search.
    # Actually, this is a job shop scheduling problem.
    # With small constraints, we can use a search with pruning.
    
    best_finish = float('inf')
    
    # memoization
    memo = {}

    def get_min_finish(phase_indices, project_finish_times, facility_available_times):
        state = (phase_indices, project_finish_times, facility_available_times)
        if state in memo:
            return memo[state]
        
        if all(phase_indices[i] == len(projects[i]) for i in range(j)):
            return max(project_finish_times)
        
        res = float('inf')
        
        # Try to schedule any available next phase
        can_schedule = False
        for i in range(j):
            if phase_indices[i] < len(projects[i]):
                can_schedule = True
                f, d = projects[i][phase_indices[i]]
                
                # Start time is max of project ready time and facility ready time
                start_time = max(project_finish_times[i], facility_available_times[f-1])
                finish_time = start_time + d
                
                new_phase_indices = list(phase_indices)
                new_phase_indices[i] += 1
                
                new_project_finish_times = list(project_finish_times)
                new_project_finish_times[i] = finish_time
                
                new_facility_available_times = list(facility_available_times)
                new_facility_available_times[f-1] = finish_time
                
                res = min(res, get_min_finish(tuple(new_phase_indices), 
                                              tuple(new_project_finish_times), 
                                              tuple(new_facility_available_times)))
        
        memo[state] = res
        return res

    # The search space is still potentially large. 
    # Given the time limit, let's use a simple priority-based simulation 
    # or a heuristic search.
    # Actually, for this problem, a simple greedy search with state 
    # exploration is sufficient.
    
    print(get_min_finish(tuple([0]*j), tuple([0]*j), tuple([0]*m)))

solve()
