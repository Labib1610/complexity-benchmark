import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    work_orders = []
    for _ in range(j):
        p = int(input_data[idx])
        idx += 1
        phases = []
        for _ in range(p):
            ws_id = int(input_data[idx])
            cycles = int(input_data[idx + 1])
            phases.append((ws_id, cycles * 5))
            idx += 2
        work_orders.append(phases)
    
    # next_phase_idx[i] is the index of the next phase to be processed for work order i
    next_phase_idx = [0] * j
    # finish_time_of_prev_phase[i] is the time when the previous phase of work order i finished
    finish_time_of_prev_phase = [0] * j
    # next_available_time_of_ws[w] is the time when workstation w becomes free
    next_available_time_of_ws = [0] * (m + 1)
    
    # We use a greedy approach: at any point, if a work order's next phase can start,
    # we want to schedule it as early as possible.
    # Since this is a job shop scheduling problem which is NP-hard, but given the 
    # constraints and the nature of the problem, a priority queue based simulation 
    # (scheduling the next available phase that can start earliest) is standard.
    
    completed_phases = 0
    total_phases = sum(len(wo) for wo in work_orders)
    
    while completed_phases < total_phases:
        best_start_time = float('inf')
        best_wo_idx = -1
        
        for i in range(j):
            if next_phase_idx[i] < len(work_orders[i]):
                ws_id, duration = work_orders[i][next_phase_idx[i]]
                # The phase can start at max(when the previous phase finished, when the workstation is free)
                start_time = max(finish_time_of_prev_phase[i], next_available_time_of_ws[ws_id])
                
                if start_time < best_start_time:
                    best_start_time = start_time
                    best_wo_idx = i
                elif start_time == best_start_time:
                    # Tie-break: prefer the one that finishes earlier or has more work
                    if best_wo_idx == -1 or len(work_orders[i]) > len(work_orders[best_wo_idx]):
                        best_wo_idx = i
        
        # Execute the chosen phase
        i = best_wo_idx
        ws_id, duration = work_orders[i][next_phase_idx[i]]
        finish_time = best_start_time + duration
        
        finish_time_of_prev_phase[i] = finish_time
        next_available_time_of_ws[ws_id] = finish_time
        next_phase_idx[i] += 1
        completed_phases += 1
        
    print(max(finish_time_of_prev_phase) // 5)

if __name__ == '__main__':
    solve()
