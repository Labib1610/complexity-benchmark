```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    pipelines = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        tasks = []
        for _ in range(o):
            res = int(input_data[idx])
            dur = int(input_data[idx + 1])
            tasks.append((res, dur))
            idx += 2
        pipelines.append(tasks)
        
    # State: (tuple of progress indices for each pipeline)
    # We want to find the minimum time to reach (len(p1), len(p2), ..., len(pj))
    # Since we need to track resource availability, we use Dijkstra.
    # State: (current_time, progress_tuple, resource_free_times)
    
    initial_progress = tuple([0] * j)
    initial_resources = tuple([0] * (m + 1))
    initial_pipeline_ready_times = tuple([0] * j)
    
    # pq stores (time, progress, resource_free_times, pipeline_ready_times)
    pq = [(0, initial_progress, initial_resources, initial_pipeline_ready_times)]
    visited = {}
    
    while pq:
        curr_time, progress, res_free, pipe_ready = heapq.heappop(pq)
        
        if progress == tuple(len(p) for p in pipelines):
            print(curr_time)
            return
        
        state = (progress, res_free, pipe_ready)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        # Try starting any available task
        for i in range(j):
            if progress[i] < len(pipelines[i]):
                res_id, dur = pipelines[i][progress[i]]
                
                # Task can start at max(pipe_ready[i], res_free[res_id])
                start_time = max(pipe_ready[i], res_free[res_id])
                finish_time = start_time + dur
                
                new_progress = list(progress)
                new_progress[i] += 1
                new_progress = tuple(new_progress)
                
                new_res = list(res_free)
                new_res[res_id] = finish_time
                new_res = tuple(new_res)
                
                new_pipe = list(pipe_ready)
                new_pipe[i] = finish_time
                new_pipe = tuple(new_pipe)
                
                heapq.heappush(pq, (max(curr_time, finish_time), new_progress, new_res, new_pipe))

solve()
```
