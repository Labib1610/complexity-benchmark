import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    elixirs = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        phases = []
        for _ in range(o):
            app = int(input_data[idx])
            dur = int(input_data[idx+1])
            phases.append((app, dur))
            idx += 2
        elixirs.append(phases)
    
    # State: (tuple of current phase indices for each elixir, tuple of finish times for each apparatus)
    # Since j <= 10 and total phases <= 50, we can use BFS/Dijkstra.
    # However, the state space is large. Given the constraints and the nature of the problem,
    # this is a job-shop scheduling problem. With j, m <= 10, we can use a search.
    
    # To keep it simple and efficient, we use a priority queue to explore states.
    # State: (time, current_phase_indices, apparatus_free_times)
    # current_phase_indices: tuple of length j
    # apparatus_free_times: tuple of length m
    
    initial_state = (0, tuple([0] * j), tuple([0] * m))
    pq = [initial_state]
    visited = {}
    
    min_finish_time = float('inf')
    
    while pq:
        time, phase_indices, app_free = heapq.heappop(pq)
        
        if all(phase_indices[i] == len(elixirs[i]) for i in range(j)):
            print(time)
            return
        
        state = (phase_indices, app_free)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to advance each elixir that has a next phase
        for i in range(j):
            if phase_indices[i] < len(elixirs[i]):
                app, dur = elixirs[i][phase_indices[i]]
                # The phase can start at max(time_when_previous_phase_finished, app_free[app-1])
                # But we don't track previous phase finish time directly in state.
                # We can calculate it by re-simulating or storing it.
                # Actually, let's store the finish time of the last phase of each elixir.
                pass

    # The above approach is complex. Given the constraints, a simple backtracking 
    # with pruning or a greedy approach with limited search is often sufficient.
    # Let's use a simple simulation-based search.
    
    def get_finish_time(order):
        app_free = [0] * (m + 1)
        elixir_free = [0] * j
        for elixir_idx in order:
            for app, dur in elixirs[elixir_idx]:
                start = max(elixir_free[elixir_idx], app_free[app])
                finish = start + dur
                elixir_free[elixir_idx] = finish
                app_free[app] = finish
        return max(elixir_free)

    import itertools
    best = float('inf')
    # Try all permutations of elixir sequences
    for p in itertools.permutations(range(j)):
        best = min(best, get_finish_time(p))
    print(best)

solve()
