import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    docs = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        reviews = []
        for _ in range(o):
            official = int(input_data[idx])
            duration = int(input_data[idx+1])
            reviews.append((official, duration))
            idx += 2
        docs.append(reviews)
    
    # State: (tuple of current review index for each doc, tuple of finish time for each official)
    # Since j <= 10 and total reviews <= 50, we can use BFS/Dijkstra.
    # However, the state space is large. Given the constraints and the nature of the problem,
    # we can use a priority queue to find the minimum finish time.
    
    # current_pos: tuple of indices (which review is next for each doc)
    # official_free_time: tuple of times when each official becomes free
    # doc_ready_time: tuple of times when each doc becomes ready for its next review
    
    initial_state = (tuple([0] * j), tuple([0] * (m + 1)), tuple([0] * j))
    pq = [(0, initial_state)]
    visited = {}
    
    min_finish_time = float('inf')
    
    while pq:
        time, (pos, off_free, doc_ready) = heapq.heappop(pq)
        
        if all(pos[i] == len(docs[i]) for i in range(j)):
            print(time)
            return
        
        if visited.get((pos, off_free, doc_ready), float('inf')) <= time:
            continue
        visited[(pos, off_free, doc_ready)] = time
        
        # Try to schedule any available review
        for i in range(j):
            if pos[i] < len(docs[i]):
                off, dur = docs[i][pos[i]]
                start_time = max(doc_ready[i], off_free[off])
                finish_time = start_time + dur
                
                new_pos = list(pos)
                new_pos[i] += 1
                new_off_free = list(off_free)
                new_off_free[off] = finish_time
                new_doc_ready = list(doc_ready)
                new_doc_ready[i] = finish_time
                
                new_state = (tuple(new_pos), tuple(new_off_free), tuple(new_doc_ready))
                heapq.heappush(pq, (max(time, finish_time), new_state))

solve()
