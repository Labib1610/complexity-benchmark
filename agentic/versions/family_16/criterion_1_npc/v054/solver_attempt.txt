import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    num_orders = int(input_data[0])
    num_bays = int(input_data[1])
    
    orders = []
    idx = 2
    for _ in range(num_orders):
        num_steps = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(num_steps):
            bay = int(input_data[idx])
            duration = int(input_data[idx + 1])
            steps.append((bay, duration))
            idx += 2
        orders.append(steps)
    
    # order_step_idx[i] is the index of the next step to be processed for order i
    order_step_idx = [0] * num_orders
    # order_ready_time[i] is the time when order i finishes its previous step
    order_ready_time = [0] * num_orders
    # bay_free_time[b] is the time when bay b becomes free
    bay_free_time = [0] * (num_bays + 1)
    
    # We need to schedule all steps. Since the number of orders is small (<= 10),
    # we can use a greedy approach: at any point, for each order, the next step
    # is available to be scheduled as soon as the order is ready AND the bay is free.
    # However, this is a job shop scheduling problem which is NP-hard.
    # Given the constraints (orders <= 10, bays <= 10, total steps <= 50),
    # we can simulate the process. Since we want the minimum completion time,
    # and the constraints are very small, we can use a priority queue to pick
    # the next available step that can start earliest.
    
    total_steps = sum(len(o) for o in orders)
    completed_steps = 0
    
    while completed_steps < total_steps:
        best_start_time = float('inf')
        best_order = -1
        
        for i in range(num_orders):
            if order_step_idx[i] < len(orders[i]):
                bay, duration = orders[i][order_step_idx[i]]
                # Earliest this step can start is max(order_ready_time, bay_free_time)
                start_time = max(order_ready_time[i], bay_free_time[bay])
                if start_time < best_start_time:
                    best_start_time = start_time
                    best_order = i
                elif start_time == best_start_time:
                    # Tie-break: pick the one that finishes earliest or just any
                    pass
        
        # Execute the best step
        i = best_order
        bay, duration = orders[i][order_step_idx[i]]
        finish_time = best_start_time + duration
        
        order_ready_time[i] = finish_time
        bay_free_time[bay] = finish_time
        order_step_idx[i] += 1
        completed_steps += 1
        
    print(max(order_ready_time))

if __name__ == '__main__':
    solve()
