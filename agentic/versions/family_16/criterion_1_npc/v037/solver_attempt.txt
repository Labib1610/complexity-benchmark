import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    num_jobs = int(input_data[idx])
    num_machines = int(input_data[idx+1])
    idx += 2
    
    jobs = []
    for _ in range(num_jobs):
        num_ops = int(input_data[idx])
        idx += 1
        ops = []
        for _ in range(num_ops):
            m = int(input_data[idx])
            d = int(input_data[idx+1])
            ops.append((m, d))
            idx += 2
        jobs.append(ops)
    
    # State: tuple of current operation index for each job
    # memo stores the minimum finish time for a given state
    memo = {}
    
    # To handle machine constraints, we need to track when each machine is free.
    # However, the state space is small enough that we can use a search.
    # Since we need the optimal schedule, this is a variation of the Job Shop Scheduling problem.
    # Given the constraints (J, M <= 10, total ops <= 50), we can use A* or BFS.
    
    # State: (tuple of current op indices, tuple of machine free times)
    # But machine free times are not fixed; they depend on the order.
    # Actually, for a fixed sequence of operations, we can determine the finish time.
    # This is a permutation problem.
    
    from itertools import permutations
    
    # Generate all possible topological orderings of operations
    # Total operations <= 50. This is too large for simple permutations.
    # But we only need to decide the order of operations on each machine.
    
    # Given the small constraints, we can use a priority queue to explore the state space.
    # State: (job_progress_indices, machine_free_times)
    
    initial_progress = tuple([0] * num_jobs)
    initial_machine_free = tuple([0] * (num_machines + 1))
    
    # Priority Queue stores (current_max_time, progress, machine_free_times)
    pq = [(0, initial_progress, initial_machine_free)]
    visited = {}
    
    best_finish = float('inf')
    
    while pq:
        curr_time, progress, machine_free = heapq.heappop(pq)
        
        if curr_time >= best_finish:
            continue
            
        if all(progress[i] == len(jobs[i]) for i in range(num_jobs)):
            best_finish = min(best_finish, curr_time)
            continue
            
        state = (progress, machine_free)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        for i in range(num_jobs):
            if progress[i] < len(jobs[i]):
                m, d = jobs[i][progress[i]]
                
                # Job i's next op can start at max(job_ready_time, machine_free_time)
                # We need to track job_ready_time.
                # Let's refine the state to include job_ready_times.
                pass

    # Re-evaluating: The problem is small enough for a simple backtracking with pruning.
    job_ready_times = [0] * num_jobs
    machine_free_times = [0] * (num_machines + 1)
    progress = [0] * num_jobs
    
    def backtrack(curr_progress, curr_job_ready, curr_machine_free, current_max):
        nonlocal best_finish
        if current_max >= best_finish:
            return
        
        if all(curr_progress[i] == len(jobs[i]) for i in range(num_jobs)):
            best_finish = min(best_finish, current_max)
            return
        
        for i in range(num_jobs):
            if curr_progress[i] < len(jobs[i]):
                m, d = jobs[i][curr_progress[i]]
                start_time = max(curr_job_ready[i], curr_machine_free[m])
                finish_time = start_time + d
                
                new_progress = list(curr_progress)
                new_progress[i] += 1
                new_job_ready = list(curr_job_ready)
                new_job_ready[i] = finish_time
                new_machine_free = list(curr_machine_free)
                new_machine_free[m] = finish_time
                
                backtrack(tuple(new_progress), tuple(new_job_ready), tuple(new_machine_free), max(current_max, finish_time))

    backtrack(tuple(progress), tuple(job_ready_times), tuple(machine_free_times), 0)
    print(best_finish)

solve()
