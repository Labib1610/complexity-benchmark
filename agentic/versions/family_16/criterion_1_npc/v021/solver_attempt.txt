import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    potions = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        phases = []
        for _ in range(o):
            f = int(input_data[idx])
            d = int(input_data[idx+1])
            phases.append((f, d))
            idx += 2
        potions.append(phases)
    
    # State: (tuple of current phase indices for each potion, 
    #         tuple of finish times for each potion, 
    #         tuple of available times for each focus)
    # Since j is small (up to 10), we can use BFS/Dijkstra.
    # However, the state space is large. Given the constraints and the nature
    # of the problem (Job Shop Scheduling), we can use a priority queue
    # to explore the state space.
    
    # State: (current_time, phase_indices, potion_ready_times, focus_ready_times)
    # But wait, the problem is to minimize the makespan.
    # This is a classic Job Shop Scheduling problem.
    # With j, m <= 10, we can use a search with pruning.
    
    memo = {}

    def get_min_time(phase_indices, potion_ready_times, focus_ready_times):
        state = (phase_indices, potion_ready_times, focus_ready_times)
        if state in memo:
            return memo[state]
        
        if all(phase_indices[i] == len(potions[i]) for i in range(j)):
            return max(potion_ready_times)
        
        res = float('inf')
        
        # Try to schedule the next available phase for any potion
        can_schedule = False
        for i in range(j):
            if phase_indices[i] < len(potions[i]):
                can_schedule = True
                f, d = potions[i][phase_indices[i]]
                
                # Earliest start time for this phase
                start_time = max(potion_ready_times[i], focus_ready_times[f-1])
                
                new_phase_indices = list(phase_indices)
                new_phase_indices[i] += 1
                
                new_potion_ready_times = list(potion_ready_times)
                new_potion_ready_times[i] = start_time + d
                
                new_focus_ready_times = list(focus_ready_times)
                new_focus_ready_times[f-1] = start_time + d
                
                res = min(res, get_min_time(tuple(new_phase_indices), 
                                            tuple(new_potion_ready_times), 
                                            tuple(new_focus_ready_times)))
        
        if not can_schedule:
            res = max(potion_ready_times)
            
        memo[state] = res
        return res

    print(get_min_time(tuple([0]*j), tuple([0]*j), tuple([0]*m)))

solve()
