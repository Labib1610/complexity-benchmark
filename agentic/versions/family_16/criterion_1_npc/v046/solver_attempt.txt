import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    products = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            ws = int(input_data[idx])
            dur = int(input_data[idx + 1])
            steps.append((ws, dur))
            idx += 2
        products.append(steps)
    
    # State: (tuple of current step indices for each product, tuple of next available time for each workstation)
    # Since the total number of steps is small (50), we can use BFS/Dijkstra.
    # However, the state space for workstation availability is continuous.
    # But notice that for a fixed sequence of steps, the start time of a step is 
    # max(finish time of previous step of same product, finish time of workstation).
    # This is a classic Job Shop Scheduling problem. Given the constraints (j, m <= 10, total steps <= 50),
    # we can use a priority queue to explore the state space of which product's next step to schedule.
    
    # Actually, with j <= 10, we can use a simple search.
    # State: (tuple of progress indices, tuple of workstation free times)
    
    initial_progress = tuple([0] * j)
    initial_ws_free = tuple([0] * m)
    initial_prod_free = tuple([0] * j)
    
    # Priority Queue stores: (time_to_finish_all, progress, ws_free, prod_free)
    # But we want to minimize the makespan.
    # Since we need to decide the order of steps, this is equivalent to finding a permutation
    # of all steps that respects the product order constraints.
    
    import itertools
    
    # The number of steps is small enough that we can use memoization on the progress state.
    memo = {}

    def get_min_time(progress, ws_free, prod_free):
        state = (progress, ws_free, prod_free)
        if state in memo:
            return memo[state]
        
        if all(progress[i] == len(products[i]) for i in range(j)):
            return max(prod_free)
        
        res = float('inf')
        for i in range(j):
            if progress[i] < len(products[i]):
                ws, dur = products[i][progress[i]]
                ws_idx = ws - 1
                
                start_time = max(prod_free[i], ws_free[ws_idx])
                finish_time = start_time + dur
                
                new_progress = list(progress)
                new_progress[i] += 1
                
                new_ws_free = list(ws_free)
                new_ws_free[ws_idx] = finish_time
                
                new_prod_free = list(prod_free)
                new_prod_free[i] = finish_time
                
                res = min(res, get_min_time(tuple(new_progress), tuple(new_ws_free), tuple(new_prod_free)))
        
        memo[state] = res
        return res

    # The state space might be too large for simple recursion.
    # Given the constraints and the nature of the problem, we use a priority queue
    # to perform a search similar to Dijkstra.
    
    pq = [(0, initial_progress, initial_ws_free, initial_prod_free)]
    visited = {}
    
    while pq:
        curr_time, progress, ws_free, prod_free = heapq.heappop(pq)
        
        if all(progress[i] == len(products[i]) for i in range(j)):
            print(curr_time)
            return
        
        state = (progress, ws_free, prod_free)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        for i in range(j):
            if progress[i] < len(products[i]):
                ws, dur = products[i][progress[i]]
                ws_idx = ws - 1
                start_time = max(prod_free[i], ws_free[ws_idx])
                finish_time = start_time + dur
                
                new_progress = list(progress)
                new_progress[i] += 1
                new_ws_free = list(ws_free)
                new_ws_free[ws_idx] = finish_time
                new_prod_free = list(prod_free)
                new_prod_free[i] = finish_time
                
                heapq.heappush(pq, (max(curr_time, finish_time), tuple(new_progress), tuple(new_ws_free), tuple(new_prod_free)))

solve()
