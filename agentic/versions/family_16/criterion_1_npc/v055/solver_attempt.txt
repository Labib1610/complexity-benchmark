import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    recipes = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        steps = []
        for _ in range(o):
            s = int(input_data[idx])
            d = int(input_data[idx+1])
            steps.append((s, d))
            idx += 2
        recipes.append(steps)
    
    # State: (tuple of current step indices for each recipe, tuple of finish times for each station)
    # Since j <= 10 and total steps <= 50, we can use BFS/Dijkstra.
    # However, the state space is large. Given the constraints and the nature of the problem,
    # this is a job shop scheduling problem. With j, m <= 10, we can use a priority queue
    # to explore the state space.
    
    # state: (tuple of current step index for each recipe, tuple of next available time for each station, tuple of finish time for each recipe)
    # Actually, we just need to decide the order of tasks for each station.
    # But since we want the earliest completion time, we can use a greedy approach with backtracking
    # or simply realize that for small constraints, we can simulate the choices.
    
    # Given the constraints and the nature of the problem, we can use a search
    # where at each step we pick a recipe that is ready to proceed.
    
    import itertools
    
    # The state is defined by (current_step_index_for_each_recipe, station_free_time)
    # Since we want to minimize the makespan, we can use a priority queue to explore
    # the order in which tasks are assigned to stations.
    
    memo = {}

    def get_min_time(steps_done, station_free_times, recipe_ready_time):
        state = (steps_done, station_free_times, recipe_ready_time)
        if state in memo:
            return memo[state]
        
        if all(steps_done[i] == len(recipes[i]) for i in range(j)):
            return max(recipe_ready_time)
        
        res = float('inf')
        
        # Try to pick any recipe that has a next step
        for i in range(j):
            if steps_done[i] < len(recipes[i]):
                s, d = recipes[i][steps_done[i]]
                # The task can start at max(recipe_ready_time[i], station_free_times[s-1])
                start_time = max(recipe_ready_time[i], station_free_times[s-1])
                finish_time = start_time + d
                
                new_steps_done = list(steps_done)
                new_steps_done[i] += 1
                
                new_station_free_times = list(station_free_times)
                new_station_free_times[s-1] = finish_time
                
                new_recipe_ready_time = list(recipe_ready_time)
                new_recipe_ready_time[i] = finish_time
                
                res = min(res, get_min_time(tuple(new_steps_done), tuple(new_station_free_times), tuple(new_recipe_ready_time)))
        
        memo[state] = res
        return res

    print(get_min_time(tuple([0]*j), tuple([0]*m), tuple([0]*j)))

solve()
