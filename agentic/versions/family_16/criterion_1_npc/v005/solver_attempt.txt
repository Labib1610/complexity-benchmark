import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    num_jobs = int(input_data[idx])
    num_machines = int(input_data[idx+1])
    idx += 2
    
    jobs = []
    for _ in range(num_jobs):
        num_ops = int(input_data[idx])
        idx += 1
        ops = []
        for _ in range(num_ops):
            m = int(input_data[idx])
            d = int(input_data[idx+1])
            ops.append((m, d))
            idx += 2
        jobs.append(ops)
    
    # State: (tuple of current operation index for each job, tuple of machine free times)
    # Since total operations <= 50 and jobs <= 10, we can use BFS/Dijkstra.
    # However, the state space is large. Given the constraints (j, m <= 10),
    # we can use a priority queue to find the minimum makespan.
    
    # state: (current_job_op_indices, machine_free_times)
    # But wait, the order of operations is fixed per job.
    # We need to decide which job to schedule next.
    
    # Since this is a classic Job Shop Scheduling problem which is NP-hard,
    # and constraints are small (j, m <= 10, total ops <= 50),
    # we can use a search with pruning or a simple heuristic.
    
    # Given the time limit of 5s, we can use a priority queue to explore states.
    # State: (time, job_op_indices, machine_free_times)
    
    initial_indices = tuple([0] * num_jobs)
    initial_machines = tuple([0] * (num_machines + 1))
    
    # pq stores (makespan, job_op_indices, machine_free_times)
    pq = [(0, initial_indices, initial_machines)]
    visited = {}
    
    best_makespan = float('inf')
    
    while pq:
        curr_time, indices, machines = heapq.heappop(pq)
        
        if all(indices[i] == len(jobs[i]) for i in range(num_jobs)):
            print(curr_time)
            return
        
        state = (indices, machines)
        if visited.get(state, float('inf')) <= curr_time:
            continue
        visited[state] = curr_time
        
        for i in range(num_jobs):
            if indices[i] < len(jobs[i]):
                m, d = jobs[i][indices[i]]
                
                # Earliest start time for this operation:
                # 1. When the job's previous operation finished
                # 2. When the machine is free
                
                # We need to track when each job's last operation finished
                # Let's refine the state to include job_finish_times
                pass

    # Re-evaluating: The state needs to track when each job is ready for its next op.
    # State: (job_ready_times, machine_free_times, job_op_indices)
    
    pq = [(0, tuple([0]*num_jobs), tuple([0]*(num_machines+1)), tuple([0]*num_jobs))]
    visited = {}
    
    while pq:
        makespan, job_ready_times, machine_free_times, indices = heapq.heappop(pq)
        
        if all(indices[i] == len(jobs[i]) for i in range(num_jobs)):
            print(makespan)
            return
        
        state = (indices, machine_free_times)
        if visited.get(state, float('inf')) <= makespan:
            continue
        visited[state] = makespan
        
        for i in range(num_jobs):
            if indices[i] < len(jobs[i]):
                m, d = jobs[i][indices[i]]
                start_time = max(job_ready_times[i], machine_free_times[m])
                finish_time = start_time + d
                
                new_indices = list(indices)
                new_indices[i] += 1
                new_indices = tuple(new_indices)
                
                new_job_ready = list(job_ready_times)
                new_job_ready[i] = finish_time
                new_job_ready = tuple(new_job_ready)
                
                new_machines = list(machine_free_times)
                new_machines[m] = finish_time
                new_machines = tuple(new_machines)
                
                heapq.heappush(pq, (max(makespan, finish_time), new_job_ready, new_machines, new_indices))

solve()
