import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    batches = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        phases = []
        for _ in range(o):
            w = int(input_data[idx])
            d = int(input_data[idx + 1])
            phases.append((w, d))
            idx += 2
        batches.append(phases)
    
    # State: tuple of current phase index for each batch
    # memo stores the minimum completion time for a given state
    memo = {}
    
    # We need to track the finish time of each workstation and the finish time of each batch
    # Since the number of phases is small (50), we can use Dijkstra/BFS on the state
    # State: (tuple of current phase indices for each batch)
    # To calculate the earliest finish time, we need to know when each workstation is free
    # and when each batch is free to start its next phase.
    
    # Actually, this is a classic Job Shop Scheduling problem.
    # Given the constraints (total phases <= 50), we can use a search.
    
    initial_state = tuple([0] * j)
    # queue stores (time, workstation_free_times, batch_free_times, current_phase_indices)
    # But wait, the state space is small enough that we can just use recursion with memoization.
    
    memo = {}

    def get_min_time(phase_indices, ws_free, batch_free):
        state = (phase_indices, ws_free, batch_free)
        if state in memo:
            return memo[state]
        
        # Check if all batches are finished
        if all(phase_indices[i] == len(batches[i]) for i in range(j)):
            return max(batch_free)
        
        res = float('inf')
        # Try to schedule any available next phase
        for i in range(j):
            if phase_indices[i] < len(batches[i]):
                w, d = batches[i][phase_indices[i]]
                # Start time is max of when workstation is free and when batch is free
                start_time = max(ws_free[w-1], batch_free[i])
                finish_time = start_time + d
                
                new_phase_indices = list(phase_indices)
                new_phase_indices[i] += 1
                
                new_ws_free = list(ws_free)
                new_ws_free[w-1] = finish_time
                
                new_batch_free = list(batch_free)
                new_batch_free[i] = finish_time
                
                res = min(res, get_min_time(tuple(new_phase_indices), tuple(new_ws_free), tuple(new_batch_free)))
        
        memo[state] = res
        return res

    print(get_min_time(initial_state, tuple([0]*m), tuple([0]*j)))

solve()
