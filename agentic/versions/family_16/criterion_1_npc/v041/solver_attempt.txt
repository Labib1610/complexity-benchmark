import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    campaigns = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        segments = []
        for _ in range(o):
            f = int(input_data[idx])
            d = int(input_data[idx+1])
            segments.append((f, d))
            idx += 2
        campaigns.append(segments)
    
    # State: (tuple of current segment indices for each campaign, 
    #         tuple of finish times for each facility,
    #         tuple of finish times for each campaign)
    # Since the total number of segments is small (50), we can use A* or Dijkstra.
    # However, the state space is large. Given the constraints and the nature of 
    # the problem (Job Shop Scheduling), we can use a priority queue to explore 
    # the state space.
    
    # State: (current_time, progress_indices, facility_free_times, campaign_free_times)
    # progress_indices: tuple of length j
    # facility_free_times: tuple of length m
    # campaign_free_times: tuple of length j
    
    initial_progress = tuple([0] * j)
    initial_facility_free = tuple([0] * m)
    initial_campaign_free = tuple([0] * j)
    
    # Priority Queue stores (time, progress, facility_free, campaign_free)
    pq = [(0, initial_progress, initial_facility_free, initial_campaign_free)]
    visited = {}
    
    min_total_time = float('inf')
    
    while pq:
        time, progress, fac_free, camp_free = heapq.heappop(pq)
        
        if all(progress[i] == len(campaigns[i]) for i in range(j)):
            print(time)
            return
        
        state = (progress, fac_free, camp_free)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to start any available segment
        for i in range(j):
            if progress[i] < len(campaigns[i]):
                f, d = campaigns[i][progress[i]]
                f_idx = f - 1
                
                # Earliest start time for this segment
                start_time = max(fac_free[f_idx], camp_free[i])
                finish_time = start_time + d
                
                new_progress = list(progress)
                new_progress[i] += 1
                
                new_fac_free = list(fac_free)
                new_fac_free[f_idx] = finish_time
                
                new_camp_free = list(camp_free)
                new_camp_free[i] = finish_time
                
                heapq.heappush(pq, (max(time, finish_time), tuple(new_progress), tuple(new_fac_free), tuple(new_camp_free)))

solve()
