import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2
    
    proposals = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        stages = []
        for _ in range(o):
            expert = int(input_data[idx])
            duration = int(input_data[idx + 1])
            stages.append((expert, duration))
            idx += 2
        proposals.append(stages)
    
    # state: (tuple of current stage indices for each proposal)
    # We want to find the minimum time to reach the state where all stages are done.
    # Since the number of stages is small (total <= 50), we can use Dijkstra.
    # However, the state space is potentially large.
    # Given the constraints (j <= 10, total stages <= 50), we can use a priority queue
    # to explore the state space.
    
    # current_stage_indices: tuple of length j
    # expert_free_time: tuple of length m
    # proposal_ready_time: tuple of length j
    
    # Priority Queue stores: (time, current_stage_indices, expert_free_time, proposal_ready_time)
    # Actually, we only need to track the next available time for each expert and each proposal.
    
    initial_stage_indices = tuple([0] * j)
    initial_expert_free_time = tuple([0] * m)
    initial_proposal_ready_time = tuple([0] * j)
    
    pq = [(0, initial_stage_indices, initial_expert_free_time, initial_proposal_ready_time)]
    visited = {}
    
    while pq:
        time, stage_indices, expert_free_time, proposal_ready_time = heapq.heappop(pq)
        
        if all(stage_indices[i] == len(proposals[i]) for i in range(j)):
            print(time)
            return
        
        state = (stage_indices, expert_free_time, proposal_ready_time)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        # Try to schedule the next stage for any proposal that isn't finished
        for i in range(j):
            if stage_indices[i] < len(proposals[i]):
                expert, duration = proposals[i][stage_indices[i]]
                expert_idx = expert - 1
                
                start_time = max(proposal_ready_time[i], expert_free_time[expert_idx])
                finish_time = start_time + duration
                
                new_stage_indices = list(stage_indices)
                new_stage_indices[i] += 1
                new_stage_indices = tuple(new_stage_indices)
                
                new_expert_free_time = list(expert_free_time)
                new_expert_free_time[expert_idx] = finish_time
                new_expert_free_time = tuple(new_expert_free_time)
                
                new_proposal_ready_time = list(proposal_ready_time)
                new_proposal_ready_time[i] = finish_time
                new_proposal_ready_time = tuple(new_proposal_ready_time)
                
                heapq.heappush(pq, (max(time, finish_time), new_stage_indices, new_expert_free_time, new_proposal_ready_time))

solve()
