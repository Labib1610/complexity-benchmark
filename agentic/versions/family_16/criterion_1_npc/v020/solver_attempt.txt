import sys
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    decrees = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        reviews = []
        for _ in range(o):
            mag = int(input_data[idx])
            dur = int(input_data[idx+1])
            reviews.append((mag, dur))
            idx += 2
        decrees.append(reviews)
    
    # Each decree has a sequence of reviews. Let's say decree i has o_i reviews.
    # We need to choose an ordering of all reviews for each magistrate.
    # However, the constraint is that for each decree, review k must finish before k+1 starts.
    # This is a scheduling problem. Since j and m are small, we can use the fact that
    # for each magistrate, the relative order of reviews assigned to them must be decided.
    # Actually, the problem is simpler: we just need to find the minimum makespan.
    # Given the constraints (j, m <= 10, total reviews <= 50), we can use a search.
    
    # State: (current_review_index_for_each_decree, available_time_for_each_magistrate)
    # This is not quite right because a magistrate might be busy.
    # Let's use a priority queue or simple recursion with memoization.
    
    memo = {}

    def get_min_time(progress, mag_free_time):
        state = (tuple(progress), tuple(mag_free_time))
        if state in memo:
            return memo[state]
        
        # Check if all finished
        if all(progress[i] == len(decrees[i]) for i in range(j)):
            return max(mag_free_time)
        
        res = float('inf')
        # Try to pick any decree that has a next review ready
        for i in range(j):
            if progress[i] < len(decrees[i]):
                mag, dur = decrees[i][progress[i]]
                # The review can start at max(mag_free_time[mag-1], time_when_previous_review_finished)
                # But we don't track previous_review_finish_time in state.
                # Let's redefine state: (progress, mag_free_time, decree_ready_time)
                pass
        return res

    # Given the constraints and the nature of the problem, this is a job shop scheduling problem.
    # With j, m <= 10, we can use a simple backtracking search with pruning.
    
    best_time = float('inf')
    
    def backtrack(progress, mag_free_time, decree_ready_time):
        nonlocal best_time
        if all(progress[i] == len(decrees[i]) for i in range(j)):
            best_time = min(best_time, max(mag_free_time))
            return
        
        if max(mag_free_time) >= best_time:
            return

        for i in range(j):
            if progress[i] < len(decrees[i]):
                mag, dur = decrees[i][progress[i]]
                start_time = max(mag_free_time[mag-1], decree_ready_time[i])
                end_time = start_time + dur
                
                new_progress = list(progress)
                new_progress[i] += 1
                new_mag_free = list(mag_free_time)
                new_mag_free[mag-1] = end_time
                new_decree_ready = list(decree_ready_time)
                new_decree_ready[i] = end_time
                
                backtrack(tuple(new_progress), tuple(new_mag_free), tuple(new_decree_ready))

    backtrack(tuple([0]*j), tuple([0]*m), tuple([0]*j))
    print(best_time)

solve()
