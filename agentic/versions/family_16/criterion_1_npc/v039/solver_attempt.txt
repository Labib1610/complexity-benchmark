import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    j = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    components = []
    for _ in range(j):
        o = int(input_data[idx])
        idx += 1
        tests = []
        for _ in range(o):
            f = int(input_data[idx])
            d = int(input_data[idx+1])
            tests.append((f, d))
            idx += 2
        components.append(tests)
    
    # State: (tuple of current test indices for each component)
    # We want to find the minimum makespan.
    # Since j is small (up to 10), we can use A* or Dijkstra.
    # However, the state space is potentially large.
    # Given the constraints and the nature of the problem (Job Shop Scheduling),
    # we can use a priority queue to explore the state space.
    
    # state: (current_test_indices_tuple)
    # We need to track when each component is available and when each facility is available.
    # This is a classic Job Shop Scheduling problem.
    # With j, m <= 10 and total tests <= 50, we can use a simple search.
    
    import itertools
    
    # Since we need the optimal schedule, and the search space is large,
    # we use a heuristic search.
    
    memo = {}
    
    def get_min_time(indices, comp_ready_time, fac_ready_time):
        state = (indices, comp_ready_time, fac_ready_time)
        if state in memo:
            return memo[state]
        
        if all(idx == len(components[i]) for i, idx in enumerate(indices)):
            return max(comp_ready_time)
        
        res = float('inf')
        # Try to schedule any available test
        for i in range(j):
            if indices[i] < len(components[i]):
                f, d = components[i][indices[i]]
                start_time = max(comp_ready_time[i], fac_ready_time[f-1])
                
                new_indices = list(indices)
                new_indices[i] += 1
                
                new_comp_ready = list(comp_ready_time)
                new_comp_ready[i] = start_time + d
                
                new_fac_ready = list(fac_ready_time)
                new_fac_ready[f-1] = start_time + d
                
                res = min(res, get_min_time(tuple(new_indices), tuple(new_comp_ready), tuple(new_fac_ready)))
        
        memo[state] = res
        return res

    # The recursive approach might be too slow. Let's use Dijkstra.
    pq = [(0, tuple([0]*j), tuple([0]*j), tuple([0]*m))]
    visited = {}
    
    while pq:
        time, indices, comp_ready, fac_ready = heapq.heappop(pq)
        
        if all(idx == len(components[i]) for i, idx in enumerate(indices)):
            print(time)
            return
        
        state = (indices, comp_ready, fac_ready)
        if visited.get(state, float('inf')) <= time:
            continue
        visited[state] = time
        
        for i in range(j):
            if indices[i] < len(components[i]):
                f, d = components[i][indices[i]]
                start_time = max(comp_ready[i], fac_ready[f-1])
                
                new_indices = list(indices)
                new_indices[i] += 1
                new_comp_ready = list(comp_ready)
                new_comp_ready[i] = start_time + d
                new_fac_ready = list(fac_ready)
                new_fac_ready[f-1] = start_time + d
                
                new_time = max(time, start_time + d)
                heapq.heappush(pq, (new_time, tuple(new_indices), tuple(new_comp_ready), tuple(new_fac_ready)))

solve()
