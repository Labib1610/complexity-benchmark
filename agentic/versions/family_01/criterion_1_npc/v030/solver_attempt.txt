```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning tier 1, 2, or 3
        for tier in range(1, 4):
            possible = True
            for v in adj[u]:
                if assignment[v] != 0 and abs(assignment[v] - tier) < 1:
                    # This condition |s_u - s_v| >= 1 is equivalent to s_u != s_v
                    # The problem states |s_u - s_v| >= 1, which is just s_u != s_v
                    # since s_u, s_v are integers.
                    if assignment[v] == tier:
                        possible = False
                        break
            
            if possible:
                assignment[u] = tier
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
