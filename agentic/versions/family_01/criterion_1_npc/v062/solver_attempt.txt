import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning phase 1, 2, or 3
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Heuristic: sort nodes by degree descending to fail faster
    nodes = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to process in order of degree
    mapping = [0] * n
    for i, node in enumerate(nodes):
        mapping[node] = i
    
    new_adj = [[] for _ in range(n)]
    for u in range(n):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    assignment = [0] * n
    
    def backtrack_mapped(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            valid = True
            for v in new_adj[u]:
                if assignment[v] == color:
                    valid = False
                    break
            if valid:
                assignment[u] = color
                if backtrack_mapped(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack_mapped(0):
        print("YES")
        res = [0] * n
        for i in range(n):
            res[nodes[i]] = assignment[i]
        print(*(res))
    else:
        print("NO")

solve()
