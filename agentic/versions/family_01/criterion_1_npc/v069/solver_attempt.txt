import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning portfolio 1, 2, or 3
        for p in range(1, 4):
            can_assign = True
            for neighbor in adj[u]:
                if assignment[neighbor] == p:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[u] = p
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Optimization: Sort nodes by degree descending to fail faster
    # However, for n=40, simple backtracking might be slow if not careful.
    # Given the constraints and the nature of 3-coloring, this is NP-complete.
    # But n=40 is small enough for optimized backtracking.
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
