import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning shift 1, 2, or 3
        for shift in range(1, 4):
            can_assign = True
            for v in range(u):
                if (adj[u] & (1 << v)) and assignment[v] == shift:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[u] = shift
                if backtrack(u + 1):
                    return True
        return False

    # Optimization: Sort nodes by degree descending to prune search tree faster
    # However, for n=40, simple backtracking with pruning is usually sufficient
    # if the graph is sparse or highly constrained.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
