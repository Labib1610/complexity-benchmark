```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_valid(u, zone):
        for v in adj[u]:
            if assignment[v] == zone:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning shard u to zone 1, 2, or 3
        for zone in [1, 2, 3]:
            if is_valid(u, zone):
                assignment[u] = zone
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # The problem is equivalent to 3-coloring a graph.
    # Since n <= 40, a simple backtracking with pruning is efficient enough
    # for most cases, but we can optimize by ordering nodes by degree.
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
