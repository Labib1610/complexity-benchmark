import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning ring 1, 2, or 3
        for ring in range(1, 4):
            can_assign = True
            for neighbor in adj[u]:
                if assignment[neighbor] == ring:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[u] = ring
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    # The problem asks for lexicographically smallest assignment.
    # The backtracking approach naturally tries 1, then 2, then 3 for each node
    # in order from 1 to n, which satisfies the requirement.
    # Given n <= 40 (5 servers * 8 cores), backtracking is efficient enough.
    sys.setrecursionlimit(2000)
    solve()
