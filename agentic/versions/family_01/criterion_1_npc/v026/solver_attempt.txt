import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_valid(node, color):
        for neighbor in adj[node]:
            if assignment[neighbor] == color:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
        
        # Heuristic: try colors 1, 2, 3
        for color in range(1, 4):
            if is_valid(node, color):
                assignment[node] = color
                if backtrack(node + 1):
                    return True
                assignment[node] = 0
        return False

    # Sort nodes by degree descending to improve backtracking performance
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to process in order of degree
    mapping = [0] * (n + 1)
    for i, node in enumerate(nodes):
        mapping[i + 1] = node
        
    # We need to keep track of assignments based on original indices
    final_assignment = [0] * (n + 1)
    
    def backtrack_mapped(idx):
        if idx > n:
            return True
        
        u = mapping[idx]
        for color in range(1, 4):
            if is_valid(u, color):
                assignment[u] = color
                if backtrack_mapped(idx + 1):
                    return True
                assignment[u] = 0
        return False

if __name__ == '__main__':
