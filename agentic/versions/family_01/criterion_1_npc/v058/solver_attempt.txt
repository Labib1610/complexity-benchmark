```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    tiers = [0] * (n + 1)
    
    def is_valid(node, tier):
        for neighbor in adj[node]:
            if tiers[neighbor] == tier:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
        
        for tier in [1, 2, 3]:
            if is_valid(node, tier):
                tiers[node] = tier
                if backtrack(node + 1):
                    return True
                tiers[node] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(tiers[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
