```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(worker_idx):
        if worker_idx == n:
            return True
        
        # Try assigning block 1, 2, then 3 to get lexicographically smallest
        for block in range(1, 4):
            can_assign = True
            for neighbor in adj[worker_idx]:
                if neighbor < worker_idx and assignment[neighbor] == block:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[worker_idx] = block
                if backtrack(worker_idx + 1):
                    return True
                assignment[worker_idx] = 0
        
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
