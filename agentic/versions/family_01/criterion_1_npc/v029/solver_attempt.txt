```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning 1, 2, 3
        neighbor_colors = {assignment[v] for v in adj[u] if assignment[v] != 0}
        for color in [1, 2, 3]:
            if color not in neighbor_colors:
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Optimization: Sort nodes by degree (heuristic)
    # But for n=40, simple backtracking with pruning is usually fine.
    # To handle potential worst cases, we can use a simple ordering.
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
