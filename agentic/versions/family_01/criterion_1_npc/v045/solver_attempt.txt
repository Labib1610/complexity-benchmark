```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # Sort nodes by degree descending to optimize backtracking (heuristic)
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    def backtrack(idx_in_nodes):
        if idx_in_nodes == n:
            return True
        
        u = nodes[idx_in_nodes]
        
        # Try assigning shift 1, 2, or 3
        for shift in range(1, 4):
            can_assign = True
            for neighbor in adj[u]:
                if assignment[neighbor] == shift:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[u] = shift
                if backtrack(idx_in_nodes + 1):
                    return True
                assignment[u] = 0
        
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
