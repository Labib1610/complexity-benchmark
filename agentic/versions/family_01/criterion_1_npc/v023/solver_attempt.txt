```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning colors 1, 2, 3
        neighbor_colors = {colors[v] for v in adj[u]}
        for c in [1, 2, 3]:
            if c not in neighbor_colors:
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # To optimize, process nodes in order of degree (heuristic)
    # But given n <= 40, simple backtracking is sufficient.
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
