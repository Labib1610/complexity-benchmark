import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def backtrack(delegate):
        if delegate > n:
            return True
        
        # Try chambers 1, 2, 3 in order for lexicographical smallest
        for chamber in range(1, 4):
            can_place = True
            for prev in range(1, delegate):
                if (adj[delegate] & (1 << prev)):
                    if assignment[prev] == chamber:
                        can_place = False
                        break
            
            if can_place:
                assignment[delegate] = chamber
                if backtrack(delegate + 1):
                    return True
        
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
