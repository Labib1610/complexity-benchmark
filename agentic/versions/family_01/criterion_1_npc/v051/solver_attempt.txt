```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning boom 1, 2, or 3
        for boom in range(1, 4):
            can_assign = True
            for neighbor in adj[u]:
                if assignment[neighbor] == boom:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[u] = boom
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # To optimize, process nodes in order of degree (heuristic)
    # However, with n=40, simple backtracking might be slow if not careful.
    # But 3-coloring is NP-complete, and n=40 is small enough for optimized backtracking.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
