import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # Since n <= 40, we use meet-in-the-middle or backtracking with pruning.
    # Given the constraints and the nature of 3-coloring, 
    # simple backtracking with bitmask pruning is efficient enough for n=40.
    
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Find which states are forbidden for process u
        forbidden = 0
        for v in range(u):
            if (adj[u] >> v) & 1:
                forbidden |= (1 << (assignment[v] - 1))
        
        for state in range(1, 4):
            if not (forbidden & (1 << (state - 1))):
                assignment[u] = state
                if backtrack(u + 1):
                    return True
        return False

    # Heuristic: order nodes by degree (most constrained first)
    # But for n=40, simple backtracking is often sufficient.
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
