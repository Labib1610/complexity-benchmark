import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem is equivalent to 3-coloring a graph.
    # Since n <= 40, we can use backtracking with pruning.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        # Heuristic: try colors in order
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # To speed up, sort nodes by degree (descending)
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to process in order of degree
    mapping = {node: i for i, node in enumerate(nodes)}
    new_adj = [[] for _ in range(n)]
    for u in range(1, n + 1):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    res_colors = [0] * n
    
    def backtrack_fast(u):
        if u == n:
            return True
        
        for c in [1, 2, 3]:
            possible = True
            for v in new_adj[u]:
                if res_colors[v] == c:
                    possible = False
                    break
            if possible:
                res_colors[u] = c
                if backtrack_fast(u + 1):
                    return True
                res_colors[u] = 0
        return False

    if backtrack_fast(0):
        print("YES")
        final_ans = [0] * n
        for i in range(n):
            final_ans[nodes[i] - 1] = res_colors[i]
        print(*(final_ans))
    else:
        print("NO")

solve()
