```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning stage 1, 2, or 3
        for stage in [1, 2, 3]:
            possible = True
            for neighbor in adj[u]:
                if assignment[neighbor] == stage:
                    possible = False
                    break
            
            if possible:
                assignment[u] = stage
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Heuristic: sort nodes by degree descending to prune search space
    # However, for n=40, simple backtracking might be slow.
    # But 3-coloring is NP-complete, and n=40 is small enough for 
    # optimized backtracking or the fact that constraints are loose.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
