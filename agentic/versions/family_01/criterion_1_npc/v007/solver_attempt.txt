```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    def is_valid(u, shift):
        for v in adj[u]:
            if assignment[v] == shift:
                return False
        return True

    def backtrack(u, current_sum):
        if u == n:
            return current_sum <= 3 * n
        
        for shift in [1, 2, 3]:
            if is_valid(u, shift):
                # Check if adding this shift keeps the sum constraint possible
                # Remaining workers (n - 1 - u) must have at least shift 1
                if current_sum + shift + (n - 1 - u) <= 3 * n:
                    assignment[u] = shift
                    if backtrack(u + 1, current_sum + shift):
                        return True
                    assignment[u] = 0
        return False

    if backtrack(0, 0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
