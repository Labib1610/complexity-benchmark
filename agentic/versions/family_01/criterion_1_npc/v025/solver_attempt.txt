import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
        idx += 2

    # Constraints:
    # 1. Toxicity: No two toxic species in same greenhouse.
    # 2. Water: Max 450L per greenhouse. Each species 10L -> Max 45 species per greenhouse.
    #    Since n <= 40, this is always satisfied (40 * 10 = 400 <= 450).
    # 3. Filtration: Pairs in different greenhouses <= 800.
    #    Since m <= n(n-1)/2 = 780, this is always satisfied.
    # So we only need to satisfy the Toxicity constraint (3-coloring).

    assignment = [0] * n

    def is_safe(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning greenhouse 1, 2, or 3
        for color in [1, 2, 3]:
            if is_safe(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
