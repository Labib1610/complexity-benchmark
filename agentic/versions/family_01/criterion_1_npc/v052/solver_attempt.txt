import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Memoization for the feasibility check
    memo = {}

    def can_complete(worker_idx, assignments):
        # assignments is a tuple of length n, 0 if unassigned
        state = (worker_idx, assignments)
        if state in memo:
            return memo[state]
        
        if worker_idx > n:
            return True
        
        # Try assigning worker_idx to phase 1, 2, or 3
        for phase in range(1, 4):
            possible = True
            for neighbor in adj[worker_idx]:
                if neighbor < worker_idx and assignments[neighbor - 1] == phase:
                    possible = False
                    break
            
            if possible:
                new_assignments = list(assignments)
                new_assignments[worker_idx - 1] = phase
                if can_complete(worker_idx + 1, tuple(new_assignments)):
                    memo[state] = True
                    return True
        
        memo[state] = False
        return False

    # Check if any solution exists at all
    if not can_complete(1, tuple([0] * n)):
        print("NO")
        return

    # Build the schedule greedily
    print("YES")
    current_assignments = [0] * n
    for i in range(1, n + 1):
        for phase in range(1, 4):
            # Check if this phase is valid for current worker
            valid_phase = True
            for neighbor in adj[i]:
                if neighbor < i and current_assignments[neighbor - 1] == phase:
                    valid_phase = False
                    break
            
            if valid_phase:
                # Check if this choice allows completion
                temp_assignments = list(current_assignments)
                temp_assignments[i - 1] = phase
                # Clear memo for each step to ensure fresh check
                memo = {}
                if can_complete(i + 1, tuple(temp_assignments)):
                    current_assignments[i - 1] = phase
                    break
    
    print(*(current_assignments))

solve()
