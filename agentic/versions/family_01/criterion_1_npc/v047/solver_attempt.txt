import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    def is_safe(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning rotation 1, 2, or 3
        for color in [1, 2, 3]:
            if is_safe(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # The constraints (max degree < 40, max clique < 41) and n <= 40
    # suggest that a simple backtracking approach will be efficient enough.
    # We can optimize by ordering nodes by degree (most constrained first).
    
    nodes = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    
    # Re-map nodes to process in order of constraints
    mapping = [0] * n
    for i, node in enumerate(nodes):
        mapping[node] = i
    
    new_adj = [[] for _ in range(n)]
    for u in range(n):
        for v in adj[u]:
            new_adj[mapping[u]].append(mapping[v])
            
    final_assignment = [0] * n
    
    def backtrack_optimized(u):
        if u == n:
            return True
        
        for color in [1, 2, 3]:
            safe = True
            for v in new_adj[u]:
                if final_assignment[v] == color:
                    safe = False
                    break
            if safe:
                final_assignment[u] = color
                if backtrack_optimized(u + 1):
                    return True
                final_assignment[u] = 0
        return False

    if backtrack_optimized(0):
        print("YES")
        res = [0] * n
        for i in range(n):
            res[nodes[i]] = final_assignment[i]
        print(*(res))
    else:
        print("NO")

solve()
