import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # We need to partition the graph into 3 independent sets.
    # Since n <= 40, we can use backtracking with pruning.
    # To optimize, we try to assign each node to shift 1, 2, or 3.
    
    assignment = [0] * n

    def can_assign(u, color):
        mask = adj[u]
        for i in range(u):
            if (mask & (1 << i)) and assignment[i] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning to shift 1, 2, then 3
        for color in [1, 2, 3]:
            if can_assign(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
