```python
import sys

# The problem asks to determine if a graph is 3-colorable.
# Since n is small (n <= 40), we can use backtracking with pruning.
# 3-coloring is NP-complete, but for n=40, a well-optimized backtracking
# algorithm will be efficient enough.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # Heuristic: sort nodes by degree (descending) to fail faster
    nodes = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
    
    def is_safe(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(idx_in_nodes):
        if idx_in_nodes == n:
            return True
        
        u = nodes[idx_in_nodes]
        
        # Try colors 1, 2, 3
        for color in range(1, 4):
            if is_safe(u, color):
                assignment[u] = color
                if backtrack(idx_in_nodes + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    # Increase recursion depth for safety, though n=40 is small
    sys.setrecursionlimit(2000)
    solve()
```
