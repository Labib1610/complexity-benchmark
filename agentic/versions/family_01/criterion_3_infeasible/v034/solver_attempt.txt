import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c,
    # we iterate through all possible a, then b, then check if c exists.
    # Since we need the smallest, we sort the adjacency lists.
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    # Iterate through all edges (a, b) as potential candidates for the first two nodes
    # To ensure lexicographical order, we iterate a from 1 to n, then b in sorted_adj[a]
    for a in range(1, n + 1):
        for b in sorted_adj[a]:
            # Now we need a c such that b < c, c is in adj[a], and c is in adj[b]
            # Intersection of sorted_adj[a] and sorted_adj[b]
            # Since we need c > b, we can just iterate through sorted_adj[a]
            # and check if it's in sorted_adj[b]
            
            # Optimization: use set for O(1) lookup for the second neighbor
            adj_b = adj[b]
            for c in sorted_adj[a]:
                if c > b:
                    if c in adj_b:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
