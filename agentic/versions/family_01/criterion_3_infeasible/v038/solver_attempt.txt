```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triple
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then check if c exists.
    # Since we need the smallest a, then b, then c:
    # We can iterate through all edges (u, v) as potential (a, b).
    # For each edge (u, v), we look for a common neighbor w such that
    # w > v (to maintain a < b < c).
    
    # To optimize, we can iterate through all u from 1 to n,
    # then all v in adj[u] where v > u,
    # then check for common neighbors w in adj[u] and adj[v] where w > v.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a < b < c. Since a is fixed, we look for c > b.
            # c must be in both adj[a] and adj[b].
            # To find the smallest c, we intersect the sorted lists.
            neighbors_b = adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
