import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, check if there exists c > b such that c is in adj[a] and adj[b]
    
    # Optimization: sort adjacency lists to iterate c efficiently
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider neighbors b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # We need a c > b such that c is in adj[a] and adj[b]
            # Intersection of sorted_adj[a] and sorted_adj[b]
            # Since we need the smallest c, we can just iterate
            # To make it faster, use set intersection or pointer
            set_a = adj[a]
            set_b = adj[b]
            
            # Find common neighbors c > b
            # Iterate through the smaller set to find common elements
            if len(set_a) < len(set_b):
                for c in set_a:
                    if c > b and c in set_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in set_b:
                    if c > b and c in set_a:
                        print(f"{a} {b} {c}")
                        return

solve()
