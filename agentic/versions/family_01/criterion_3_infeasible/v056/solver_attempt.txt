```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find a triangle (a, b, c) such that a < b < c
    # and (a,b), (b,c), (a,c) are all in the restriction set.
    # To minimize (a, b, c) lexicographically:
    # Iterate a from 1 to n-2
    #   Iterate b from a+1 to n-1
    #     If b is in adj[a]:
    #       Check if there exists c > b such that c is in adj[a] and adj[b]
    
    for a in range(1, n + 1):
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Find common neighbors of a and b that are > b
            # Intersection of neighbors_a[i+1:] and adj[b]
            # To optimize, use set intersection
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
```
