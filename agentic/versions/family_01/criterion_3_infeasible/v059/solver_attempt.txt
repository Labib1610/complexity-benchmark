import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we look for 'b' > 'a' such that 'b' is connected to 'a'.
    # Then we check if there exists 'c' > 'b' such that 'c' is connected to both 'a' and 'b'.
    
    # Sort edges to help find the smallest lexicographical triplet
    edges.sort()
    
    # Pre-sort adjacency lists to iterate in order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Iterate through all possible 'b' > 'a'
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need a 'c' > 'b' such that c is in adj[a] and c is in adj[b]
            # Since adj lists are sorted, we can use set intersection or two-pointer
            # Given the constraints and the need for lexicographical smallest,
            # we check the intersection of neighbors of a and b.
            
            set_b = set(adj[b])
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
