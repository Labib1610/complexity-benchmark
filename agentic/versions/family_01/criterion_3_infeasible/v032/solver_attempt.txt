import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we look at all neighbors b of a.
    # If b > a, we check if there exists c > b such that
    # c is a neighbor of a AND c is a neighbor of b.
    
    # Sort neighbors to ensure we find the smallest lexicographical triplet
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c such that c > b, c in adj[a], and c in adj[b]
            # Since adj[a] and adj[b] are sorted, we can use two pointers
            # or set intersection.
            # Given the constraints and the requirement for the smallest,
            # we can iterate through neighbors of b that are > b.
            
            set_a = set(neighbors_a)
            for c in adj[b]:
                if c > b and c in set_a:
                    print(f"{a} {b} {c}")
                    return

solve()
