import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # Sort edges to help find the lexicographically smallest triplet
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all edges (a, b) and check if there exists
    # a node c such that (a, c) and (b, c) are edges.
    # Since we want the smallest (a, b, c), we iterate a from 1 to n,
    # then b from a+1 to n, then check for common neighbors.
    
    # Pre-sort adjacency lists for consistent iteration
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        for b in adj[a]:
            # Now we have an edge (a, b). We need to find the smallest c > b
            # such that c is in adj[a] and c is in adj[b].
            # Since adj[a] and adj[b] are sorted, we can use two pointers
            # or set intersection.
            
            set_b = set(adj[b])
            for c in adj[a]:
                if c > b and c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
