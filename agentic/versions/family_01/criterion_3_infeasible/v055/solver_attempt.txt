import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c
    # We iterate through a from 1 to n-2
    # Then b from a+1 to n-1
    # Then check if there exists c > b such that c is in adj[a] and adj[b]
    
    # Sort adjacency lists to potentially speed up intersection
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Optimization: only consider b that are connected to a
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # We need a c such that c > b and c is in adj[a] and adj[b]
            # Intersection of neighbors_a and neighbors_b
            # Since we need the smallest c, we can iterate through sorted neighbors
            # of the node with the smaller degree to be faster
            
            # Using sets for O(1) lookup
            if len(adj[a]) < len(adj[b]):
                for c in neighbors_a:
                    if c > b and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                neighbors_b = sorted_adj[b]
                for c in neighbors_b:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
