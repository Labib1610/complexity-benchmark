import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c,
    # we iterate through all possible 'a' from 1 to n-2.
    # For a fixed 'a', we look for 'b' > 'a' such that 'b' is connected to 'a'.
    # Then we look for 'c' > 'b' such that 'c' is connected to both 'a' and 'b'.
    
    # Sort edges to help find candidates efficiently
    edges.sort()
    
    # Pre-sort adjacency lists to ensure we check 'b' and 'c' in increasing order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        # Candidates for b are neighbors of a that are > a
        neighbors_a = [x for x in adj[a] if x > a]
        for b in neighbors_a:
            # Candidates for c are neighbors of both a and b that are > b
            # We can intersect the sorted lists adj[a] and adj[b]
            # Since they are sorted, we can do this efficiently
            set_a = set(adj[a])
            for c in adj[b]:
                if c > b and c in set_a:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
