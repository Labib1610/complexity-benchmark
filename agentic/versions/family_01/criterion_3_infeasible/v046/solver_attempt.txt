```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c),
    # we iterate through sorted edges (a, b) and check if there exists
    # a node c such that (a, c) and (b, c) are edges.
    # Since we want the smallest a, then b, then c, we iterate
    # through edges (a, b) in order.
    
    for a, b in edges:
        # We need to find c such that a < b < c or a < c < b or c < a < b
        # But the problem asks for a < b < c.
        # So we look for c > b such that (a, c) and (b, c) exist.
        
        # To optimize, intersect the adjacency sets of a and b
        common = adj[a].intersection(adj[b])
        
        # We need c > b
        candidates = [c for c in common if c > b]
        if candidates:
            c = min(candidates)
            print(f"{a} {b} {c}")
            return

if __name__ == '__main__':
    solve()
```
