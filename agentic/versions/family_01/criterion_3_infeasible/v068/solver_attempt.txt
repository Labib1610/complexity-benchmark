import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Optimization: sort adjacency lists to iterate efficiently
    sorted_adj = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        if len(neighbors_a) < 2:
            continue
            
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a < b < c.
            # Check intersection of neighbors_a and neighbors_b
            # Only consider elements > b
            set_a = adj[a]
            set_b = adj[b]
            
            # Iterate through the smaller set to find common neighbors
            if len(set_a) < len(set_b):
                for c in set_a:
                    if c > b and c in set_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in set_b:
                    if c > b and c in set_a:
                        print(f"{a} {b} {c}")
                        return

solve()
