import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c
    # We iterate through possible a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    # For a fixed a, we iterate b in adj[a] such that b > a.
    # Then we check if there exists c in adj[a] intersect adj[b] such that c > b.
    
    # Sort adjacency lists to help with lexicographical order
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # To optimize, we can iterate through all edges (u, v) and check for a common neighbor w.
    # But we need the lexicographically smallest (a, b, c).
    # Let's iterate a from 1 to n.
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # For a fixed a, we want the smallest b > a that is connected to a.
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need the smallest c > b such that c is connected to both a and b.
            # Since adj[a] and adj[b] are sorted, we can find common elements efficiently.
            # We only care about c > b.
            
            # Optimization: use set intersection or two pointers
            # Since we want the smallest c, we can iterate through neighbors of a
            # that are > b and check if they are in neighbors of b.
            
            set_b = set(adj[b])
            for c in neighbors_a:
                if c > b and c in set_b:
                    print(f"{a} {b} {c}")
                    return

solve()
