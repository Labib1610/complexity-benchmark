import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find the lexicographically smallest triplet (a, b, c)
    # such that a < b < c and (a,b), (b,c), (a,c) are edges.
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Get neighbors of a that are > a
        neighbors_a = [v for v in adj[a] if v > a]
        neighbors_a.sort()
        
        # Iterate b in neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # We need c > b such that c is in adj[a] and adj[b]
            # To keep it lexicographically smallest, we check neighbors of b
            # that are > b and also in neighbors_a
            neighbors_b = adj[b]
            
            # Find common neighbors of a and b that are > b
            # We can iterate through the smaller set to optimize
            if len(neighbors_a) < len(neighbors_b):
                for c in neighbors_a:
                    if c > b and c in neighbors_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                # Sort neighbors_b to ensure we find the smallest c first
                sorted_neighbors_b = sorted([v for v in neighbors_b if v > b])
                for c in sorted_neighbors_b:
                    if c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
