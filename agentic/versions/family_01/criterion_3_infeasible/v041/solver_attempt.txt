```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triplet
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # all edges (a, b) and check if there exists a c > b such that
    # (a, c) and (b, c) are also edges.
    # Since we want the smallest a, then b, then c, we iterate through
    # sorted edges (a, b) and check common neighbors.
    
    best = None
    
    for a in range(1, n + 1):
        # Optimization: if we already found a triplet with a smaller 'a',
        # we don't need to check larger 'a's.
        if best and a > best[0]:
            break
            
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if best and (a > best[0] or (a == best[0] and b >= best[1])):
                continue
            
            # Find common neighbors c > b
            # We only need to check neighbors of b that are also in neighbors_a
            neighbors_b = adj[b]
            for c in neighbors_a[i+1:]:
                if c in neighbors_b:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
                    # Since we want the smallest c for a fixed (a, b),
                    # and we iterate c in increasing order, we can break here
                    break
                    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

solve()
```
