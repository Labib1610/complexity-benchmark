import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest trio
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists a c > b
    # such that (a, c) and (b, c) are edges.
    # To ensure lexicographical order, we iterate a from 1 to n,
    # then b from a+1 to n, then check common neighbors.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Find common neighbors c > b
            # We want the smallest c, so we iterate through sorted neighbors of a
            # that are also in neighbors of b
            neighbors_b = adj[b]
            for c in neighbors_a:
                if c > b and c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
