import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n
    # Iterate b in sorted(adj[a])
    # Check if there exists c in adj[a] such that c > b and c in adj[b]
    
    # Pre-sort adjacency lists to ensure we find the lexicographically smallest
    sorted_adj = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c > b such that c is in adj[a] and c is in adj[b]
            # Since neighbors_a is sorted, we look for c in neighbors_a[i+1:]
            # and check if c is in adj[b]
            
            # Optimization: use set lookup for adj[b]
            adj_b = adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
