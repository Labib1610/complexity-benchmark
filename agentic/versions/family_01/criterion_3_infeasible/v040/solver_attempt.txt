import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find the lexicographically smallest triple
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # possible values of a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2.
    for a in range(1, n - 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        # For each pair (b, c) in neighbors_a, check if b and c are connected
        # To ensure lexicographical order, we iterate b then c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Optimization: if we already found a triple with a smaller 'a' or 'b',
            # we would have returned. Since we want the smallest (a, b, c),
            # we check if any c > b exists in both adj[a] and adj[b].
            
            # Intersection of neighbors of a and neighbors of b
            # Both sets are sorted if we maintain them, but here we just check
            # neighbors of b that are > b
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
