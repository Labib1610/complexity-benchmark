import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need the lexicographically smallest triad (a, b, c) with a < b < c.
    # Iterate through all possible a from 1 to n-2
    for a in range(1, n - 1):
        # Get neighbors of a that are > a
        neighbors_a = [v for v in adj[a] if v > a]
        neighbors_a.sort()
        
        # Iterate through all pairs (b, c) from neighbors_a
        # Since we want the smallest (a, b, c), we iterate b then c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Optimization: if b is already larger than the best found, we could stop,
            # but since we iterate a, b, c in order, the first one we find is the answer.
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
