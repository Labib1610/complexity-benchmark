```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        idx += 2
        
    # We need the lexicographically smallest triple (a, b, c)
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted(list(adj[a]))
        
        # Iterate b from neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c such that c is in neighbors_a and c is in adj[b]
            # Since we want the smallest triple, we check c in sorted order
            # c must be > b
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
