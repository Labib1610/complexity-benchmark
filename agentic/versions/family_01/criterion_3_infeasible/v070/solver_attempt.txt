import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to help find lexicographically smallest triad
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c), we iterate through
    # sorted edges (a, b) and check if there exists a c such that
    # (a, c) and (b, c) are in the grievance records.
    # Since we want smallest a, then b, then c, we iterate through edges
    # and for each edge (a, b), we check common neighbors.
    
    # Optimization: iterate through edges (a, b) and find common neighbors
    # To ensure lexicographical order, we can iterate through all possible a,
    # then all neighbors b of a, then check if there is a c > b such that
    # c is in both adj[a] and adj[b].
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need a c > b such that c is in adj[a] and adj[b]
            # To be efficient, intersect the sets
            common = adj[a].intersection(adj[b])
            valid_c = [c for c in common if c > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

solve()
