```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest triple (a, b, c)
    # We iterate through possible a, then b, then c.
    # A triple (a, b, c) with a < b < c is a clique if:
    # b in adj[a], c in adj[a], and c in adj[b].
    
    # Sort edges to help find candidates efficiently
    # We want the smallest a, then smallest b, then smallest c.
    
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted(list(adj[a]))
        
        # For each pair of neighbors (b, c) of a, check if (b, c) is an edge
        # To keep it lexicographically smallest, we iterate b then c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We only care about b > a (already guaranteed by neighbors_a)
            
            # Check neighbors of b that are > b
            # We need to find a c such that c > b and c is in adj[a] and c is in adj[b]
            # This is equivalent to finding c in (adj[a] intersection adj[b]) where c > b
            
            # Optimization: intersect sets
            common = adj[a].intersection(adj[b])
            if common:
                c = min(common)
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
```
