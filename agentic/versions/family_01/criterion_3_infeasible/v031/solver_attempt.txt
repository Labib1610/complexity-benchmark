import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need the lexicographically smallest triplet (a, b, c) with a < b < c
    # such that a-b, a-c, and b-c are all edges.
    # Iterate a from 1 to n-2
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        # Iterate b from neighbors of a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a c such that c is in adj[a] and c is in adj[b]
            # To ensure lexicographical order, we check candidates for c
            # c must be > b
            common = adj[a].intersection(adj[b])
            valid_c = [x for x in common if x > b]
            
            if valid_c:
                c = min(valid_c)
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
