import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # We need to find the lexicographically smallest (a, b, c) with a < b < c
    # such that (a, b), (b, c), (a, c) are all edges.
    # Iterate through possible 'a' from 1 to n
    for a in range(1, n + 1):
        # For a fixed 'a', we want to find b, c such that b, c are in adj[a]
        # and b < c and (b, c) is an edge.
        # To minimize (a, b, c), we should iterate b in increasing order
        # and then c in increasing order.
        neighbors_a = sorted(list(adj[a]))
        
        # Optimization: check if any pair (b, c) in neighbors_a is connected
        # Since we want the smallest (a, b, c), we iterate b and c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We only care about b > a, which is guaranteed since neighbors_a > a
            # Now check if there exists c > b such that c is in neighbors_a AND c is in adj[b]
            # To keep it efficient, intersect adj[b] and neighbors_a
            # Only check c > b
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
