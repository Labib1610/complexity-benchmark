import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all possible a from 1 to n-2
    for a in range(1, n + 1):
        # Get neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        # Iterate through neighbors b of a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # We need a c such that c > b and c is a neighbor of both a and b
            # To ensure lexicographical order, we check neighbors of b that are > b
            # and also in neighbors_a
            neighbors_b = adj[b]
            
            # Find common neighbors c > b
            common = []
            for c in neighbors_a:
                if c > b and c in neighbors_b:
                    common.append(c)
            
            if common:
                print(f"{a} {b} {min(common)}")
                return

if __name__ == '__main__':
    solve()
