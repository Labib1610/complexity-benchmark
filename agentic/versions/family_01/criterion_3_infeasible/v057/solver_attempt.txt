import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and adj[b]
    
    # Sort adjacency lists to make finding c easier
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need to find c > b such that c is in both adj[a] and adj[b]
            # Since neighbors_a and neighbors_b are sorted, we can use intersection
            # or just iterate.
            neighbors_b = sorted_adj[b]
            
            # Use two pointers to find common elements > b
            ptr_a = i + 1
            ptr_b = 0
            
            # Optimization: only check neighbors of b that are > b
            # and neighbors of a that are > b
            
            # Actually, just intersecting the sets is efficient enough for N=2000
            # But to be lexicographically smallest, we want the smallest c.
            # Let's intersect the two sets and find the smallest element > b.
            
            set_a = adj[a]
            set_b = adj[b]
            
            # Find common neighbors
            common = set_a.intersection(set_b)
            
            best_c = float('inf')
            found = False
            for c in common:
                if c > b:
                    if c < best_c:
                        best_c = c
                        found = True
            
            if found:
                print(f"{a} {b} {best_c}")
                return

solve()
