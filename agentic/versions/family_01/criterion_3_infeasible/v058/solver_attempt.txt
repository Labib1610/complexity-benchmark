import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # Iterate through a from 1 to n
    # Iterate through b in adj[a] where b > a
    # Iterate through c in adj[a] where c > b
    # Check if c is in adj[b]
    
    # Sort adjacency lists to ensure we find the lexicographically smallest
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need to check if there exists c > b such that c is in adj[a] and c is in adj[b]
            # This is equivalent to finding the intersection of neighbors_a[i+1:] and neighbors_b
            # Since we want the smallest c, we iterate through sorted neighbors_a
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
