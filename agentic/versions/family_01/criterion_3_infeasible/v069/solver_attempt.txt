import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # Sort edges to iterate in lexicographical order
    edges.sort()
    
    # To find the lexicographically smallest triad (a, b, c) with a < b < c:
    # We iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we look for 'b' in adj[a] (where b > a).
    # Then we check if there exists 'c' such that c in adj[a] and c in adj[b] (where c > b).
    
    # To satisfy the requirement: smallest a, then smallest b, then smallest c.
    # We iterate a from 1 to n.
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need to find c > b such that c is in adj[a] and c is in adj[b]
            # Intersection of neighbors_a[i+1:] and adj[b]
            # Since we want the smallest c, we can iterate through sorted neighbors_a
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
