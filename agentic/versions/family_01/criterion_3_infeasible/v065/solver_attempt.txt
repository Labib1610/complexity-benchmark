import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We need to find the lexicographically smallest triplet (a, b, c)
    # such that a < b < c and (a,b), (a,c), (b,c) are all edges.
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Get neighbors of a, sorted to ensure we check b and c in order
        neighbors_a = sorted(list(adj[a]))
        
        # Iterate b from neighbors of a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need c such that c > b and c is in both adj[a] and adj[b]
            # To find the smallest c, we intersect the sets
            common = adj[a].intersection(adj[b])
            
            # Filter for c > b
            valid_c = [x for x in common if x > b]
            
            if valid_c:
                c = min(valid_c)
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
