import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest plan, we iterate from 1 to n.
    # For each connected component, we try assigning state 1 to the smallest
    # node in that component first.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from node i
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard bipartite check
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a connected component. We need to decide whether to
            # flip the colors to make the sequence lexicographically smallest.
            # The current assignment is one valid 2-coloring.
            # Let c1 be the set of nodes with color 1, c2 be the set with color 2.
            # We want the smallest node in the component to have color 1.
            
            min_node = min(component)
            if color[min_node] == 2:
                # Flip colors for this component
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
