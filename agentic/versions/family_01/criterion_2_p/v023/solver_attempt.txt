```python
import sys

# The problem is a classic 2-coloring problem (bipartite graph check).
# We need to assign each isotope to field 1 or 2 such that no two resonating
# isotopes are in the same field.
# To get the lexicographically smallest assignment, we iterate through isotopes
# 1 to n. If an isotope is not yet assigned, we try assigning it to field 1.
# If that leads to a contradiction, we try field 2. If both fail, it's impossible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Try assigning to 1 first (lexicographically smallest)
            stack = [(i, 1)]
            assignment[i] = 1
            component = [i]
            possible = True
            
            while stack:
                u, color = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - color
                        stack.append((v, 3 - color))
                        component.append(v)
                    elif assignment[v] == color:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Reset and try assigning to 2
                for node in component:
                    assignment[node] = 0
                
                stack = [(i, 2)]
                assignment[i] = 2
                component = [i]
                possible = True
                while stack:
                    u, color = stack.pop()
                    for v in adj[u]:
                        if assignment[v] == 0:
                            assignment[v] = 3 - color
                            stack.append((v, 3 - color))
                            component.append(v)
                        elif assignment[v] == color:
                            possible = False
                            break
                    if not possible:
                        break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(assignment[1:]))

solve()
```
