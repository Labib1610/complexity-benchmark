```python
import sys

# The problem asks for a 2-coloring of a graph such that the difference 
# between the number of nodes in each color is at most n.
# Since the difference constraint |count1 - count2| <= n is always satisfied 
# for any n (as |count1 - count2| <= n is always true for any partition of n),
# the problem reduces to checking if the graph is bipartite.
# To get the lexicographically smallest assignment, we process nodes 1 to n.
# If a node is not visited, we perform BFS/DFS to color its connected component.
# To ensure lexicographical smallest, we prefer color 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # The problem asks for the lexicographically smallest assignment.
    # Our current coloring is determined by the order of traversal.
    # To ensure lexicographical smallest, we need to check if we can swap 
    # colors in each component.
    
    # Actually, for each component, we have two choices:
    # 1. Keep current coloring
    # 2. Swap 1s and 2s
    # We want to minimize the sequence (s1, s2, ..., sn).
    # This means for each component, we pick the coloring that makes the 
    # first node in that component have color 1.
    
    # Re-run to ensure lexicographical smallest:
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Force color 1 for the first node of the component
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
