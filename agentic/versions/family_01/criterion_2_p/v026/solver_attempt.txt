import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is not assigned, we try assigning it to 1.
    # If that fails, we try 2.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [(i, 1)]
            assignment[i] = 1
            component = []
            
            # We need to check if this component is bipartite.
            # Since we want the lexicographically smallest, we prioritize 1.
            # However, for a connected component, once the first node is fixed,
            # the rest are determined.
            
            q = [i]
            assignment[i] = 1
            comp_nodes = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                        comp_nodes.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")

            # The lexicographical requirement applies to the whole sequence.
            # If we have a component, we can either assign the first node to 1 or 2.
            # If we assign the first node to 1, the rest are fixed.
            # If we assign the first node to 2, the rest are fixed (swapped).
            # We compare the two possibilities for the first node of the component.
            
            # Actually, the lexicographical smallest is achieved by always trying 1 first.
            # If the component is already colored, we don't need to do anything.
