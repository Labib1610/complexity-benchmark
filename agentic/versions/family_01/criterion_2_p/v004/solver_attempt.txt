import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want the lexicographically smallest assignment.
    # This means we prefer worker 1 to be in shift 1, then worker 2 in shift 1, etc.
    # This is a 2-coloring problem.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component.
            # To get lexicographically smallest, we try to assign 1 to the first node.
            stack = [(i, 1)]
            color[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                # Sort neighbors to process them in a deterministic way if needed,
                # but here we just need to ensure we don't violate the 2-coloring.
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
