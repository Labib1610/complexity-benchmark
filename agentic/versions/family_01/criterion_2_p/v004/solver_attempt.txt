```python
import sys

# The problem asks for a 2-coloring of a graph with n vertices and m edges.
# This is equivalent to checking if the graph is bipartite.
# If it is bipartite, we need the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for worker 1, then worker 2, etc.
# We can achieve this by iterating from 1 to n and assigning 1 if possible,
# otherwise assigning 2, while maintaining consistency via BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 to the current node
            # if it hasn't been colored yet.
            stack = [i]
            assignment[i] = 1
            
            while stack:
                u = stack.pop()
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
