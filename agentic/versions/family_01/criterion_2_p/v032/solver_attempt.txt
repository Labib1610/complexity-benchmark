import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unassigned, 1 = code 1, 2 = code 2
    color = [0] * (n + 1)
    
    # To minimize lexicographically, we try to assign 1 to the smallest index
    # if it's not already assigned.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [i]
            color[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process nodes in a way that respects the lexicographical requirement
            # Actually, for each component, we have two choices:
            # 1. Start with color[i] = 1
            # 2. Start with color[i] = 2
            # We need to check if both are valid and pick the one that results in
            # a lexicographically smaller sequence.
            
            comp_nodes = []
            q = [i]
            comp_nodes.append(i)
            visited = {i}
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if v not in visited:
                        visited.add(v)
                        q.append(v)
                        comp_nodes.append(v)
            
            # Try assigning 1 to the smallest index in the component
            # If that fails, try 2.
            def check(start_node, start_color):
                temp_color = {node: 0 for node in comp_nodes}
                temp_color[start_node] = start_color
                stack = [start_node]
                while stack:
                    u = stack.pop()
                    for v in adj[u]:
                        if temp_color[v] == 0:
                            temp_color[v] = 3 - temp_color[u]
                            stack.append(v)
                        elif temp_color[v] == temp_color[u]:
                            return None
                return temp_color

            res1 = check(i, 1)
            res2 = check(i, 2)
            
            if res1 is None and res2 is None:
                print("NO")
                return
            
            # Compare res1 and res2 lexicographically
            # We only care about the values at indices in comp_nodes
            # Sort comp_nodes to compare lexicographically
            comp_nodes.sort()
            
            best = None
            if res1 is not None and res2 is not None:
                for node in comp_nodes:
                    if res1[node] < res2[node]:
                        best = res1
                        break
                    elif res2[node] < res1[node]:
                        best = res2
                        break
                if best is None: best = res1
            else:
                best = res1 if res1 is not None else res2
            
            for node in comp_nodes:
                color[node] = best[node]
                
    print("YES")
    print(*(color[1:]))

solve()
