import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We need the lexicographically smallest sequence.
    # This means for each connected component, we want to assign 1 to the 
    # smallest node index if possible.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from node i.
            # To get lexicographically smallest, we prefer assigning 1 to i.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # The component is bipartite. We have two possible colorings:
            # 1. The one we just found.
            # 2. The flipped one (1 becomes 2, 2 becomes 1).
            # We want the one that makes the sequence lexicographically smallest.
            # Since we process nodes 1 to n, we check the first node in the 
            # component (the one with the smallest index).
            
            # Find the smallest index in this component
            min_node = i
            # Actually, the component nodes are already visited. 
            # We just need to check if flipping improves the result.
            # The lexicographical requirement is on the sequence s_1, s_2, ..., s_n.
            # For each component, we pick the coloring that assigns 1 to the 
            # smallest index node in that component.
            
            # Find the smallest node index in this component
            comp_nodes = [node for node, c in component]
            min_node = min(comp_nodes)
            
            # If color[min_node] is 2, we flip the whole component
            if color[min_node] == 2:
                for node in comp_nodes:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
