import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The condition "sum of power modes is 3" means if p[u] = 1, p[v] = 2, and vice versa.
    # This is a standard 2-coloring problem (bipartite graph check).
    # To get the lexicographically smallest sequence, we want p[i] = 1 whenever possible.
    
    color = [0] * (n + 1) # 0: unvisited, 1: mode 1, 2: mode 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component.
            # To make it lexicographically smallest, we prefer color 1 for the smallest node.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a stack for iterative DFS
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # After coloring the component, check if we could have swapped 1s and 2s
            # to make the sequence lexicographically smaller.
            # Actually, for each component, we have two choices:
            # Choice A: node i has color 1
            # Choice B: node i has color 2
            # We pick the one that results in a smaller value at the first node index.
            # Since we iterate i from 1 to n, the first node encountered in a component
            # is the smallest index in that component. We always want that to be 1.
            # So color[i] = 1 is always the best choice for the first node of a component.
            
    print("YES")
    print(*(color[1:]))

solve()
