import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition s_u + s_v = 3 means if s_u = 1, s_v = 2, and vice versa.
    # This is a standard 2-coloring problem on a graph.
    # To get the lexicographically smallest sequence, we prefer 1 over 2.
    
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS from i. To be lexicographically smallest,
            # we assign 1 to the first node of each connected component.
            stack = [(i, 1)]
            colors[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                # Neighbors should be colored with 3 - c
                target_c = 3 - c
                
                # To ensure lexicographical order, we process neighbors
                # but the coloring is forced by the bipartite constraint.
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = target_c
                        stack.append((v, target_c))
                    elif colors[v] != target_c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
