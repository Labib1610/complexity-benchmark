import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition |s_u - s_v| = 1 means s_u != s_v.
    # This is a standard bipartite graph coloring problem.
    # We want the lexicographically smallest assignment.
    # This means for each connected component, we want to assign 1 to the 
    # smallest index node if possible.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i. To make it lexicographically smallest,
            # we prefer assigning 1 to the smallest node in the component.
            # Actually, for each component, we have two choices:
            # 1. color[i] = 1
            # 2. color[i] = 2
            # We pick the one that results in a lexicographically smaller sequence.
            
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # We can either keep it as is, or flip all 1s to 2s and 2s to 1s.
            # We want the smallest index in the component to have 1 if possible.
            # Let min_node be the smallest index in this component.
            # If color[min_node] == 2, we should flip the whole component.
            
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
