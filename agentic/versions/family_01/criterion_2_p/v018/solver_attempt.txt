import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest assignment, we want s_1=1, then s_2=1 if possible, etc.
    # This is a 2-coloring problem. For each connected component, we have two choices
    # for the first node visited. We pick the choice that results in a smaller 
    # lexicographical sequence.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Component nodes
            component = []
            stack = [i]
            assignment[i] = 1
            component.append(i)
            
            # Standard BFS/DFS to color
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # Check if flipping this component makes it lexicographically smaller
            # Find the first node in this component
            first_node = min(component)
            if assignment[first_node] == 2:
                # Flip
                for node in component:
                    assignment[node] = 3 - assignment[node]
                    
    print("YES")
    print(*(assignment[1:]))

solve()
