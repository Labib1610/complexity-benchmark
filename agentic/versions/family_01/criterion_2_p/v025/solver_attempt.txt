import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == 0:
            comp = []
            stack = [(i, 1)]
            color[i] = 1
            comp.append((i, 1))
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        comp.append((v, 3 - c))
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            c1 = sum(1 for node, c in comp if c == 1)
            c2 = sum(1 for node, c in comp if c == 2)
            components.append((c1, c2, comp))
            
    # We need sum of identifiers S such that n < S < 2n + 1.
    # Let x be the number of speakers assigned to side 1.
    # S = x * 1 + (n - x) * 2 = 2n - x.
    # n < 2n - x < 2n + 1  =>  -n < -x < 1  =>  -1 < x < n.
    # Since x is the number of speakers on side 1, 0 <= x <= n.
    # So we need 1 <= x <= n-1.
    
    # Each component has two choices:
    # Choice A: side 1 gets c1, side 2 gets c2. (x += c1)
    # Choice B: side 1 gets c2, side 2 gets c1. (x += c2)
    # Let diff = c1 - c2. If we choose A, x changes by c1. If B, x changes by c2.
    # Let base = sum of all c2. Then x = base + sum of (chosen_c1 - chosen_c2).
    # Let d_i = c1_i - c2_i. We want 1 <= base + sum(s_i * d_i) <= n-1
    # where s_i is 1 if we pick A, 0 if we pick B.
    
    base = sum(comp[1] for comp in components)
    diffs = [comp[0] - comp[1] for comp in components]
    
    # DP to find reachable x
    # dp[i] = which choice was made for component i to reach sum x
    # Since n is 2e5, use bitset-like approach or simple DP
    reachable = {base}
    history = {base: []}
    
    for i, d in enumerate(diffs):
        new_reachable = {}
        for val in reachable:
            if val + d not in new_reachable:
                new_reachable[val + d] = (i, 1)
            if val not in new_reachable:
                new_reachable[val] = (i, 0)
        # This is still potentially slow, but we only need one valid x
        # Actually, we need lexicographically smallest assignment.
        # To get lexicographically smallest, we want side 1 to have as many 
        # small-indexed nodes as possible.
        
    # Re-evaluating: Lexicographical smallest means we want 1s at 1, 2, 3...
    # This is a variation of the partition problem.
    # Given the constraints and the requirement, we can use DP.
    
    dp = {base: []}
    for i, (c1, c2, nodes) in enumerate(components):
        new_dp = {}
        d = c1 - c2
        for val, path in dp.items():
            if val + d not in new_dp:
                new_dp[val + d] = path + [1]
            if val not in new_dp:
                new_dp[val] = path + [0]
        dp = new_dp
        
    for x in range(1, n):
        if x in dp:
            print("YES")
            assignment = [0] * (n + 1)
            choices = dp[x]
            for i, choice in enumerate(choices):
                c1, c2, nodes = components[i]
                if choice == 1:
                    for node, c in nodes:
                        assignment[node] = c
                else:
                    for node, c in nodes:
                        assignment[node] = 3 - c
            print(*(assignment[1:]))
            return
            
    print("NO")

solve()
