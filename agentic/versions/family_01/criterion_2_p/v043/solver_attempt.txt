import sys

# The problem states that for each constraint (u, v), the sum of their modes must be 3.
# Since modes are 1 or 2, this means mode(u) + mode(v) = 3.
# This implies mode(u) != mode(v).
# This is a standard bipartite graph coloring problem.
# We want the lexicographically smallest configuration, meaning we want 1s as early as possible.
# For each connected component, we can pick the mode of the first node (the one with the smallest index)
# to be 1, and then propagate the constraints. If that leads to a contradiction, the graph is not bipartite.
# If it is bipartite, we have two choices for each component:
# Choice A: Smallest index node = 1
# Choice B: Smallest index node = 2
# To get the lexicographically smallest result, we compare the two choices for each component
# and pick the one that results in a smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # 1. color[i] = 1 (current)
            # 2. color[i] = 2 (flip all)
            
            # Check which one is lexicographically smaller
            # We only care about the first node in the component that appears in the sequence
            # Actually, we want the smallest value at the smallest index.
            # Since we iterate 1 to n, the first node in the component is always the smallest index.
            # So we just need to check if setting color[i] = 1 is better than color[i] = 2.
            # If color[i] = 1, the first node gets 1. If color[i] = 2, the first node gets 2.
            # 1 is always better than 2. So we always prefer color[i] = 1.
            
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
