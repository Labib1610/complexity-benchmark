import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition "sum of codes must be 3" means if worker u has code c,
    # worker v must have code 3-c. This is a 2-coloring problem.
    # 1 -> 2, 2 -> 1.
    
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we want to assign 1
    # to the smallest index possible.
    # We iterate 1 to n. If a node is uncolored, we try assigning it 1.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [i]
            color[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process the component
            comp_nodes = [i]
            idx_comp = 0
            while idx_comp < len(comp_nodes):
                u = comp_nodes[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        comp_nodes.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
