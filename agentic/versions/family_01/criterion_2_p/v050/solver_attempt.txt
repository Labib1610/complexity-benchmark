import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a worker is not assigned, we try assigning them to 1 first.
    # This is a standard 2-coloring problem on a graph.
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [i]
            assignment[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process nodes in a way that respects lexicographical order
            # However, for 2-coloring, the choice of 1 or 2 for the first node
            # in a component determines the coloring for the whole component.
            # To be lexicographically smallest, we always prefer 1 for the 
            # smallest unassigned node in the component.
            
            q = [i]
            while q:
                u = q.pop()
                component.append(u)
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
