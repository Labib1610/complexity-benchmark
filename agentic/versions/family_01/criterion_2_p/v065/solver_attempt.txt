import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If an employee is not assigned, we try assigning 1. If that fails, 2.
    # Since it's a bipartite check, we use DFS.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start a new component
            stack = [(i, 1)]
            assignment[i] = 1
            component = []
            
            # Use a stack to perform DFS for bipartiteness
            # We need to keep track of the component to check for conflicts
            # and to ensure we pick the lexicographically smallest assignment.
            # Actually, for each component, there are only two ways to color it.
            # We pick the one that makes the first node in the component 1.
            
            comp_nodes = []
            q = [i]
            assignment[i] = 1
            comp_nodes.append(i)
            
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                        comp_nodes.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # The current assignment for this component is fixed by the first node
            # being 1. Is this the lexicographically smallest?
            # For a component, if we have nodes {v1, v2, ...} sorted,
            # we want the smallest assignment.
            # The two possible colorings are (c1, c2, ...) and (3-c1, 3-c2, ...).
            # We compare them at the first index where they differ.
            
            # Actually, the problem asks for the lexicographically smallest 
            # sequence for the whole array. This means for each component,
            # we should pick the coloring that results in a smaller value 
            # at the smallest index in that component.
            
            first_node = min(comp_nodes)
            if assignment[first_node] == 2:
                for node in comp_nodes:
                    assignment[node] = 3 - assignment[node]
                    
    print("YES")
    print(*(assignment[1:]))

solve()
