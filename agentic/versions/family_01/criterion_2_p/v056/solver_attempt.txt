import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] = 0 (unvisited), 1 (vault 1), 2 (vault 2)
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # For each connected component, we try assigning the first node to 1.
    # If that fails, we try 2.
    
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            # to perform a check if needed
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # The component is bipartite. 
            # We have two choices for the coloring:
            # Choice A: color[i] = 1
            # Choice B: color[i] = 2 (flip all colors in component)
            # We want the lexicographically smallest sequence.
            # Since we process nodes 1..n, we want the smallest value at the 
            # smallest index.
            
            # Find the smallest index in this component
            first_node = min(component)
            
            # If we set color[first_node] = 1, is it valid? Yes.
            # If we set color[first_node] = 2, is it valid? Yes.
            # To be lexicographically smallest, we want the smallest node 
            # in the component to have the smallest possible color (1).
            
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
