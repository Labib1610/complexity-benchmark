import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # To get the lexicographically smallest assignment, we want to assign 1
    # to as many early machines as possible.
    # This is a 2-coloring problem. We process components one by one.
    # For each component, we have two possible colorings (swap 1s and 2s).
    # We choose the one that makes the first machine in the component 1.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if the current coloring is the lexicographically smallest
            # for this component. The first node 'i' is already 1.
            # If we had assigned 2 to 'i', the whole component would be flipped.
            # Since we want the smallest sequence, we prefer 1 at the earliest
            # possible index. Since 'i' is the smallest index in this component,
            # color[i]=1 is already optimal.
            
    print("YES")
    print(*(color[1:]))

solve()
