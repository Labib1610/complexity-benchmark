```python
import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color[i] = 0 (unvisited), 1 (Enzyme 1), 2 (Enzyme 2)
    color = [0] * (n + 1)
    
    # To satisfy the priority condition:
    # We want to assign 1 to the first gene in each connected component if possible.
    # Since we iterate from 1 to n, if a component is unvisited, we start DFS
    # and assign 1 to the starting node.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assigning 1 to i
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```
