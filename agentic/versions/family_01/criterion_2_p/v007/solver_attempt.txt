def solve():
    input_data = sys.stdin.read().split()

    adj = [[] for _ in range(n + 1)]
    idx = 2

        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is not colored, we start a BFS/DFS to color its component.
    # To ensure lexicographical smallest, we try assigning 1 first.
    
    for i in range(1, n + 1):
        if color[i] == 0:

            # To get lexicographically smallest, we always try color 1 for the
