import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is not colored, we try assigning it 1.
    # Since it's a bipartite check, we use DFS.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start DFS for this connected component
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # We need to collect the component to ensure we can process it
            # However, standard bipartite coloring is sufficient.
            # To ensure lexicographical smallest, we must be careful.
            # Actually, for a fixed component, the choice of the first node's 
            # color (1 or 2) determines the rest. To be lexicographically 
            # smallest, we always prefer 1 for the smallest index in the component.
            
            q = [i]
            color[i] = 1
            while q:
                u = q.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
