import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem is a 2-coloring problem on a graph.
    # We want the lexicographically smallest assignment (1 or 2).
    # For each connected component, we can pick the color of the first node
    # (the one with the smallest index) to be 1, which forces the colors
    # of all other nodes in that component.
    
    color = [0] * (n + 1)
    result = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we must ensure that for each
            # component, the smallest index node is assigned 1.
            # Since we iterate 1 to n, the first uncolored node we encounter
            # is always the smallest in its component.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            for node, c in component:
                result[node] = c
                
    print("YES")
    print(*(result[1:]))

if __name__ == '__main__':
    solve()
