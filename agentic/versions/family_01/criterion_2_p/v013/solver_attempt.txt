import sys

# The problem asks for a 2-coloring of a graph such that the resulting 
# sequence (s_1, s_2, ..., s_n) is lexicographically smallest.
# Lexicographically smallest means we want s_1 = 1 if possible, 
# then s_2 = 1 if possible, and so on.
# This is a standard 2-coloring problem (bipartite check).
# For each connected component, we have two choices for coloring:
# (color1, color2) or (color2, color1).
# To make the sequence lexicographically smallest, for each component,
# we identify the smallest index i in that component. We want s_i = 1.
# If we set s_i = 1, all other nodes in the component are forced.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Find the smallest index in this component
            min_node = min(component)
            
            # We want s[min_node] = 1.
            # If color[min_node] is 1, we keep the current coloring.
            # If color[min_node] is 2, we flip all colors in this component.
            if color[min_node] == 1:
                for node in component:
                    res[node] = color[node]
            else:
                for node in component:
                    res[node] = 3 - color[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
