import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest assignment, we want b_1=1, then b_2=1 if possible, etc.
    # This is a 2-coloring problem.
    # We iterate 1 to n. If node i is not colored, we try to color it 1.
    # If that fails, we try 2. If both fail, output NO.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To ensure lexicographical smallest, we must be careful.
            # Actually, for each connected component, there are at most 2 ways to color it.
            # We pick the one that makes the first node in the component 1.
            # If that's not possible, we pick 2.
            
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if this coloring is the lexicographically smallest for this component.
            # The two possible colorings are:
            # 1. Current color[v]
            # 2. 3 - color[v]
            # We want the one that results in a smaller sequence.
            # Since we process nodes 1..n, we just need to ensure the first node
            # in the component (the smallest index) gets 1 if possible.
            # Our current logic already sets color[i] = 1 for the smallest i in component.
            
    print("YES")
    print(*(color[1:]))

solve()
