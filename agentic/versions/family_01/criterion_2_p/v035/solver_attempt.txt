import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(10**6)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # 2-coloring problem: check if bipartite
    # To get lexicographically smallest, we try to assign 1 to node i if possible.
    # Since it's a 2-coloring problem, once a component is colored, 
    # the other color is fixed. We want the smallest assignment.
    
    color = [0] * (n + 1)
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we prefer color 1 for the smallest node
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard DFS to find all nodes in component
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a component. We can either use the current coloring
            # or flip it (1->2, 2->1).
            # We want the smallest assignment.
            # Let's check the smallest index in this component.
            # If we assign 1 to the smallest index, is it better?
            # Actually, for each component, we have two choices:
            # 1. Keep current colors
            # 2. Flip colors
            # We want the smallest assignment lexicographically.
            # This means for the smallest node 's' in the component,
            # we want assignment[s] = 1.
            
            s = min(component)
            if color[s] == 2:
                # Flip
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                assignment[node] = color[node]
                
    print("YES")
    print(*(assignment[1:]))

solve()
