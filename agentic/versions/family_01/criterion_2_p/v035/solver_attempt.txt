import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is not assigned, we try assigning it to shift 1.
    # If that fails, we try shift 2.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the connected component
            component = []
            stack = [i]
            assignment[i] = 1
            component.append(i)
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process components
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # After coloring this component, we need to ensure it's the 
            # lexicographically smallest. Since we want the smallest sequence,
            # for each component, we check if flipping the colors results in 
            # a smaller sequence.
            # Actually, the greedy approach of assigning 1 to the first 
            # unvisited node in the component is correct for lexicographical order.
            
    print("YES")
    print(*(assignment[1:]))

solve()
