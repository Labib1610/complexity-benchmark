    if not input_data:

    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2

        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We need to assign 1 or 2 to each node such that no two adjacent nodes have the same value.
    # This is equivalent to 2-coloring the graph.
    # To make the sequence lexicographically smallest, we iterate 1 to n.
    # If a node is not colored, we try assigning 1. If that fails, we try 2.
    # Since we want the lexicographically smallest sequence, we greedily pick 1 whenever possible.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:

            # To make the sequence lexicographically smallest, we should try to set
            # the first node of each component to 1.
            
            q = [i]
