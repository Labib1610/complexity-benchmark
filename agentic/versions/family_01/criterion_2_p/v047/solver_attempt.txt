```python
import sys

# The problem asks for a 2-coloring of a graph such that all edges connect
# vertices of different colors. This is equivalent to checking if the graph
# is bipartite. The conditions given:
# 1. u and v cannot be in the same period (bipartite condition).
# 2. sum(c_x for x in P1) + sum(c_y for y in P2) = 2m.
#    Since each edge (u, v) contributes 1 to c_u and 1 to c_v, the sum of all
#    c_i is 2m. This condition is always satisfied for any partition.
# 3. k = m. This means all edges must connect different periods, which is
#    exactly the definition of a bipartite graph.
# Thus, the problem is simply to check if the graph is bipartite and find
# the lexicographically smallest 2-coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a node is uncolored, we assign it color 1 (the smallest possible).
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
