import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest assignment, we want to assign 1
    # to as many early nodes as possible.
    # This is a 2-coloring problem. For each connected component, we have 
    # two possible colorings (swap 1s and 2s). We choose the one that makes
    # the first node in the component 1 if possible.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            assignment[i] = 1
            component.append(i)
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to keep track of the component nodes
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # Check if the current assignment is the lexicographically smallest
            # for this component. The first node 'i' is already assigned 1.
            # If we swapped, 'i' would be 2, which is lexicographically larger.
            # So we keep it as is.
            
    print("YES")
    print(*(assignment[1:]))

solve()
