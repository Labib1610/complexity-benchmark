```python
import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # We want the lexicographically smallest sequence.
    # This means we prefer s_i = 1 whenever possible.
    # This is a 2-coloring problem.
    # For each connected component, we can pick the color of the first node
    # (the one with the smallest index) to be 1.
    
    color = [0] * (n + 1)
    result = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, set color[i] = 1
            stack = [i]
            color[i] = 1
            component = []
            
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring the component, we check if this assignment is 
            # lexicographically smallest. Since we want s_i = 1 for the smallest
            # index i in the component, and we already set color[i] = 1,
            # this is the best we can do for this component.
            for node in component:
                result[node] = color[node]
                
    print("YES")
    print(*(result[1:]))

if __name__ == '__main__':
    solve()
```
