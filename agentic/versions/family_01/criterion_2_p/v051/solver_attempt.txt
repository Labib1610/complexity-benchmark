import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If a node is not colored, we try assigning it color 1.
    # If that leads to a conflict, we try color 2.
    # Since we want the smallest sequence, we prioritize color 1 for each component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this connected component
            # We want the smallest lexicographical assignment.
            # For each component, we have two choices:
            # 1. Start node i with color 1
            # 2. Start node i with color 2
            # We check if both are valid, then pick the one that results in 
            # a lexicographically smaller sequence.
            
            # Actually, for a bipartite graph, once the color of one node is fixed,
            # the colors of all nodes in that component are fixed.
            # We just need to check if assigning color 1 to the smallest node
            # in the component is valid.
            
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            possible = True
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                print("NO")
                return
                
    print("YES")
    print(*(color[1:]))

solve()
