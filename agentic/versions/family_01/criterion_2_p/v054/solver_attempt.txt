import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # state: 0 = unvisited, 1 = clockwise, 2 = counter-clockwise
    state = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a component is not visited, we try assigning state 1 to the smallest
    # gear index in that component to keep it lexicographically smallest.
    
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS for this component
            # We want the smallest gear in this component to be 1
            stack = [(i, 1)]
            state[i] = 1
            component = []
            
            # Use a list to keep track of nodes in this component
            # to verify bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = 3 - state[u]
                        q.append(v)
                    elif state[v] == state[u]:
                        print("NO")
                        return

solve()
