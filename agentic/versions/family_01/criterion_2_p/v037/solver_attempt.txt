import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a worker is not assigned, we try assigning 1, then 2.
    # This is a standard 2-coloring problem (bipartite check).
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [i]
            assignment[i] = 1
            
            # We need to process the component. To ensure lexicographical 
            # smallest, we must be careful. However, for a bipartite graph,
            # once the first node of a component is fixed, the rest are 
            # determined. To get the smallest, we should always try to 
            # assign 1 to the smallest unassigned node in the component.
            
            component = []
            q = [i]
            assignment[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component.append(u)
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # The component is colored. Check if this is the lexicographically
            # smallest. Actually, for a bipartite component, there are only 
            # two ways to color it (swap 1s and 2s). We pick the one that 
            # makes the first node (the smallest index in the component) 1.
            # This is already handled by setting assignment[i] = 1.
            
    print("YES")
    print(*(assignment[1:]))

solve()
