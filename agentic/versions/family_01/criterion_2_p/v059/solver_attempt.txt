import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # state: 0 = unassigned, 1 = state 1, 2 = state 2
    state = [0] * (n + 1)
    
    # To satisfy the lexicographical requirement:
    # We iterate from 1 to n. If a node is unassigned, we assign it state 1.
    # This greedy approach works because the graph is bipartite.
    # If a component is not yet colored, assigning the first node 1 is 
    # always optimal for the lexicographical requirement.
    
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [i]
            state[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = 3 - state[u]
                        stack.append(v)
                    elif state[v] == state[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(state[1:]))

if __name__ == '__main__':
    solve()
