import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we want employee 1 to be in group 1.
    # Then for each component, we want the smallest node to be in group 1.
    # If a component is bipartite, there are two ways to color it (swapping 1s and 2s).
    # We choose the one that results in a smaller sequence.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            component = []
            stack = [i]
            assignment[i] = 1
            component.append(i)
            
            # Standard BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # Check if swapping colors for this component gives a lexicographically smaller result
            # We only need to check the first node in the component that is not yet fixed.
            # Actually, for each component, we want the smallest index in the component to be 1.
            # If the current assignment has assignment[min_node] == 2, we flip all.
            min_node = min(component)
            if assignment[min_node] == 2:
                for node in component:
                    assignment[node] = 3 - assignment[node]
                    
    print("YES")
    print(*(assignment[1:]))

solve()
