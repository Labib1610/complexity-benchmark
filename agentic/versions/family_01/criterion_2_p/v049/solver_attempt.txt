import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # To get the lexicographically smallest sequence, we want s[i] = 1 if possible.
    # This is a 2-coloring problem. For each connected component, we have two choices
    # (swap 1s and 2s). We choose the one that makes the first node in the component
    # (the one with the smallest index) have shift 1.
    
    shift = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if shift[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            shift[i] = 1
            component.append(i)
            
            # Use a stack for iterative DFS to avoid recursion limits
            idx_stack = 0
            while idx_stack < len(component):
                u = component[idx_stack]
                idx_stack += 1
                
                for v in adj[u]:
                    if shift[v] == 0:
                        shift[v] = 3 - shift[u]
                        component.append(v)
                    elif shift[v] == shift[u]:
                        print("NO")
                        return
            
            # After coloring the component, check if we need to flip
            # We want the smallest index in this component to be 1.
            # The smallest index is the first one we visited (i).
            # Since we set shift[i] = 1, it is already optimal.
            
    print("YES")
    print(*(shift[1:]))

solve()
