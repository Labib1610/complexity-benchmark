import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a worker is not colored, we try assigning them to shift 1.
    # This is a standard 2-coloring problem (bipartite check).
    # Since we want the lexicographically smallest sequence, we greedily
    # assign 1 to the current node if possible.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [i]
            color[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process the component
            comp_nodes = []
            q = [i]
            color[i] = 1
            while q:
                u = q.pop()
                comp_nodes.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring the component, we check if we could have made
            # a better choice. Actually, for a bipartite component, there are
            # only two ways to color it (swap 1s and 2s).
            # We check which one is lexicographically smaller.
            
            # Collect nodes in this component
            # The current coloring is one valid assignment.
            # The other valid assignment is swapping 1s and 2s.
            
            # To compare, we look at the first node in the component (smallest ID)
            # and ensure it gets 1 if possible.
            # Since we iterate 1 to n, the first node we encounter in a component
            # is the smallest ID. If we assign it 1, we are optimal.
            
    print("YES")
    print(*(color[1:]))

solve()
