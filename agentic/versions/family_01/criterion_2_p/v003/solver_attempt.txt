import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is not colored, we start a BFS/DFS.
    # To ensure lexicographical smallest, we greedily assign 1 if possible.
    # However, for a bipartite graph, the components are independent.
    # Within each component, once the first node is fixed, the rest are determined.
    # To get the lexicographically smallest, we try assigning 1 to the first
    # unvisited node in the component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # We need to check if this component is bipartite
            # and assign colors.
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # The component is bipartite. We have two choices:
            # 1. Keep current coloring
            # 2. Flip all colors (1->2, 2->1)
            # We want the lexicographically smallest result.
            # Compare the two options for the first node in the component.
            # Actually, we just need to ensure the smallest index in the 
            # component gets the smallest possible value (1).
            # Since we iterate 1..n, the first node i we encounter is the 
            # smallest in its component. We already assigned it 1.
            # This is always the lexicographically smallest choice for this component.
            
    print("YES")
    print(*(color[1:]))

solve()
