import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We need to assign 1 or 2 to each node.
    # This is a 2-coloring problem.
    # To satisfy the lexicographical requirement, we iterate 1 to n.
    # If a node is not colored, we try assigning 1. If that fails, try 2.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Collect component nodes
            q = [i]
            visited_idx = 0
            while visited_idx < len(q):
                u = q[visited_idx]
                visited_idx += 1
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # The greedy requirement: "prioritizing unit 1 as early as possible"
            # For a connected component, there are only two possible colorings:
            # C and 3-C. We pick the one that makes the first node in the 
            # component (the one with the smallest index) 1 if possible.
            # Actually, the problem asks for the lexicographically smallest 
            # sequence (c1, c2, ..., cn).
            # Since we process 1 to n, if we encounter an uncolored node,
            # we MUST assign it 1 to be lexicographically smallest.
            # This forces the coloring of the entire component.
            
    print("YES")
    print(*(color[1:]))

solve()
