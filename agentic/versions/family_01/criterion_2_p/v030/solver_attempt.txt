import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # We need to assign 1 or 2. 
    # Constraint: u, v must have different values (1+2=3).
    # This is a standard bipartite coloring problem.
    # To get the lexicographically smallest assignment, we want 
    # worker 1 to be 1 if possible, then worker 2, etc.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i. To make it lexicographically smallest,
            # we prefer assigning 1 to the smallest node in the component.
            # Actually, for each connected component, we have two choices:
            # 1. Node i gets 1, its neighbors get 2, etc.
            # 2. Node i gets 2, its neighbors get 1, etc.
            # We pick the one that results in a smaller value for the smallest index node.
            
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if we should flip the colors in this component
            # We want the smallest index in the component to have 1 if possible.
            # Since we process nodes 1 to n, the first node we encounter in a 
            # component is always the smallest index in that component.
            # So color[i] = 1 is already the lexicographically best choice.
            
    print("YES")
    print(*(color[1:]))

solve()
