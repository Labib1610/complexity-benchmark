import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unvisited, 1 = branch 1, 2 = branch 2
    color = [0] * (n + 1)
    
    # To minimize lexicographically, we iterate 1 to n.
    # For each connected component, we try assigning the first node to 1.
    # If that fails, we try 2.
    
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Use a stack to find all nodes in the component
            # and check for bipartiteness
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid bipartite component.
            # We need to decide whether to keep the current coloring
            # or flip it (1->2, 2->1) to minimize lexicographically.
            # We compare the two options for the first node in the component
            # that appears in the sorted order of nodes.
            
            # Actually, the rule is: minimize branch for 1, then 2, etc.
            # This means for each component, we want the smallest possible
            # branch number for the smallest index node in that component.
            
            # Find the smallest index in this component
            min_node = min(component)
            
            # If we keep current coloring, branch[min_node] = color[min_node]
            # If we flip, branch[min_node] = 3 - color[min_node]
            # We want branch[min_node] to be 1 if possible.
            
            if color[min_node] == 2:
                # Flip the whole component
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
