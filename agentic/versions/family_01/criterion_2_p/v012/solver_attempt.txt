import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unvisited, 1 = direction 1, 2 = direction 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a component is not visited, we start a DFS.
    # To make it lexicographically smallest, we prefer assigning 1 to the 
    # smallest index in the component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assigning color 1 to i
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process nodes in a way that respects lexicographical order
            # Actually, for lexicographical smallest, we just need to ensure
            # that for each connected component, the first node encountered
            # gets 1.
            
            q = [i]
            color[i] = 1
            while q:
                u = q.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
