import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # For each connected component, we try assigning the first node to 1.
    # If that leads to a conflict, we try 2.
    # Actually, for a bipartite graph, once the first node of a component is fixed,
    # all others are determined. To make it lexicographically smallest,
    # for each component, we check if assigning the smallest index node to 1
    # is valid. If it is, we do it. If not, we assign it to 2.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            assignment[i] = 1
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            # to verify bipartiteness
            idx_comp = 0
            is_bipartite = True
            
            # Standard BFS to color
            q = [i]
            assignment[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                    elif assignment[v] == assignment[u]:
                        is_bipartite = False
                        break
                if not is_bipartite:
                    break
            
            if not is_bipartite:
                print("NO")
                return
                
    print("YES")
    print(*(assignment[1:]))

solve()
