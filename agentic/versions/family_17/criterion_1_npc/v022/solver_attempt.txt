```python
import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    q = int(data[0])
    t = int(data[1])
    
    n = 3 * q
    by_first = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(t):
        a = int(data[idx])
        b = int(data[idx + 1])
        c = int(data[idx + 2])
        idx += 3
        by_first[a].append((b, c))
        
    used = bytearray(n + 1)
    ans = []
    
    def search(cur):
        while cur <= n and used[cur]:
            cur += 1
        if cur > n:
            return True
        
        for b, c in by_first[cur]:
            if not used[b] and not used[c]:
                used[b] = 1
                used[c] = 1
                ans.append((cur, b, c))
                if search(cur + 1):
                    return True
                ans.pop()
                used[b] = 0
                used[c] = 0
        return False

    sys.setrecursionlimit(2000)
    if search(1):
        print("YES")
        out = []
        for a, b, c in ans:
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
