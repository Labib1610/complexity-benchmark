import sys

# Increase recursion depth for deep search trees
sys.setrecursionlimit(5000)

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q
    
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        triples.append((a, b, c))
        idx += 3

    # Dancing Links (DLX) setup
    max_nodes = n + 1 + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row = [0] * max_nodes
    sz = [0] * (n + 1)

    # Initialize column headers
    for i in range(n + 1):
        L[i] = (i - 1) % (n + 1)
        R[i] = (i + 1) % (n + 1)
        U[i] = i
        D[i] = i
        sz[i] = 0

    node_cnt = n + 1

    for r_idx, (a, b, c) in enumerate(triples):
        nodes = []
        for x in (a, b, c):
            if x < 1 or x > n:
                continue
            node = node_cnt
            node_cnt += 1
            col[node] = x
            row[node] = r_idx
            sz[x] += 1
            
            # Insert into column x
            D[node] = x
            U[node] = U[x]
            D[U[x]] = node
            U[x] = node
            nodes.append(node)
        
        if len(nodes) == 3:
            n1, n2, n3 = nodes
            L[n1] = n3; R[n1] = n2
            L[n2] = n1; R[n2] = n3
            L[n3] = n2; R[n3] = n1

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                sz[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                sz[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum size (MRV heuristic)
        c = R[0]
        min_sz = sz[c]
        cur = R[c]
        while cur != 0:
            if sz[cur] < min_sz:
                min_sz = sz[cur]
                c = cur
            cur = R[cur]

        if min_sz == 0:
            return False

        cover(c)
        i = D[c]
        while i != c:
            solution.append(row[i])
            j = R[i]
            while j != i:
                cover(col[j])
                j = R[j]

            if search():
                return True

            solution.pop()
            j = L[i]
            while j != i:
                uncover(col[j])
                j = L[j]
            i = D[i]

        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_idx in solution:
            a, b, c = triples[r_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    main()
