import sys

# Increase recursion depth for search
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    N = 3 * q
    
    # Store original triples to output later
    triples = []
    idx = 2
    for r in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triples.append((a, b, c))
    
    # DLX structures
    max_nodes = 1 + N + 3 * t + 5
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row_id = [0] * max_nodes
    col_size = [0] * (N + 1)
    
    for i in range(N + 1):
        L[i] = (i - 1 + N + 1) % (N + 1)
        R[i] = (i + 1) % (N + 1)
        U[i] = i
        D[i] = i
    
    node_cnt = N + 1
    for r, (a, b, c) in enumerate(triples):
        first_node = node_cnt
        for x in (a, b, c):
            cur = node_cnt
            node_cnt += 1
            col[cur] = x
            row_id[cur] = r
            col_size[x] += 1
            
            U[cur] = U[x]
            D[cur] = x
            D[U[x]] = cur
            U[x] = cur
        
        L[first_node] = first_node + 2
        R[first_node] = first_node + 1
        L[first_node + 1] = first_node
        R[first_node + 1] = first_node + 2
        L[first_node + 2] = first_node + 1
        R[first_node + 2] = first_node

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                col_size[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                col_size[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        c = R[0]
        min_size = col_size[c]
        curr = R[c]
        while curr != 0:
            if col_size[curr] < min_size:
                min_size = col_size[curr]
                c = curr
                if min_size == 0:
                    break
            curr = R[curr]
        
        if min_size == 0:
            return False
        
        cover(c)
        r_node = D[c]
        while r_node != c:
            solution.append(row_id[r_node])
            j = R[r_node]
            while j != r_node:
                cover(col[j])
                j = R[j]
            
            if search():
                return True
            
            solution.pop()
            j = L[r_node]
            while j != r_node:
                uncover(col[j])
                j = L[j]
            r_node = D[r_node]
        
        uncover(c)
        return False

    if search():
        print("YES")
        out = []
        for r in solution:
            a, b, c = triples[r]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
