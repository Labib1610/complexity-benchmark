import sys

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    num_cols = 3 * q
    
    # Read triples
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triples.append((a, b, c))
        
    # DLX using flat arrays for high performance in Python
    # 0 is the root node
    # 1 .. num_cols are column headers
    # > num_cols are data nodes
    
    # Estimate total nodes: 1 + num_cols + 3 * t
    max_nodes = 1 + num_cols + 3 * t + 10
    
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    row_id = [0] * max_nodes
    col_size = [0] * (num_cols + 1)
    
    # Initialize column headers
    for i in range(num_cols + 1):
        L[i] = (i - 1) % (num_cols + 1)
        R[i] = (i + 1) % (num_cols + 1)
        U[i] = i
        D[i] = i
        C[i] = i
        
    node_count = num_cols + 1
    
    # Add rows
    for r, (a, b, c) in enumerate(triples):
        row_nodes = []
        for col in (a, b, c):
            node = node_count
            node_count += 1
            
            C[node] = col
            row_id[node] = r
            col_size[col] += 1
            
            # Insert vertically at bottom of column
            u = U[col]
            U[node] = u
            D[node] = col
            D[u] = node
            U[col] = node
            
            row_nodes.append(node)
            
        # Connect horizontally
        n0, n1, n2 = row_nodes
        L[n0] = n2; R[n0] = n1
        L[n1] = n0; R[n1] = n2
        L[n2] = n1; R[n2] = n0

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                col_size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                col_size[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum size (MRV heuristic)
        min_size = 1000000000
        best_c = -1
        c = R[0]
        while c != 0:
            sz = col_size[c]
            if sz < min_size:
                min_size = sz
                best_c = c
                if min_size <= 1:
                    break
            c = R[c]
            
        if min_size == 0 or best_c == -1:
            return False
        
        cover(best_c)
        r = D[best_c]
        while r != best_c:
            solution.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
                
            if search():
                return True
                
            solution.pop()
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            r = D[r]
            
        uncover(best_c)
        return False

    if search():
        sys.stdout.write("YES\n")
        out = []
        for r_idx in solution:
            a, b, c = triples[r_idx]
            out.append(f"{a} {b} {c}\n")
        sys.stdout.write("".join(out))
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
