import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q

    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx + 1]) - 1
        c = int(input_data[idx + 2]) - 1
        idx += 3
        triples.append((a, b, c))

    # Element to triples map
    elem_to_triples = [[] for _ in range(n)]
    for i, (a, b, c) in enumerate(triples):
        elem_to_triples[a].append(i)
        elem_to_triples[b].append(i)
        elem_to_triples[c].append(i)

    # Check for elements with 0 triples
    for x in range(n):
        if not elem_to_triples[x]:
            print("NO")
            return

    deg = [len(elem_to_triples[x]) for x in range(n)]
    triple_active = [True] * t
    covered = [False] * n

    solution = []

    def backtrack(uncovered_count):
        if uncovered_count == 0:
            return True

        # Choose uncovered element with minimum degree
        min_deg = float('inf')
        best_x = -1

        for x in range(n):
            if not covered[x]:
                d = deg[x]
                if d < min_deg:
                    min_deg = d
                    best_x = x
                    if d == 0:
                        break

        if min_deg == 0:
            return False

        # Try each active triple covering best_x
        candidates = [tr for tr in elem_to_triples[best_x] if triple_active[tr]]

        for tr in candidates:
            # Select triple tr
            u, v, w = triples[tr]
            
            # Find all active triples that conflict with tr
            # (i.e., contain u, v, or w)
            deactivated = []
            for elem in (u, v, w):
                for conflict_tr in elem_to_triples[elem]:
                    if triple_active[conflict_tr]:
                        triple_active[conflict_tr] = False
                        deactivated.append(conflict_tr)

            for d_tr in deactivated:
                for elem in triples[d_tr]:
                    deg[elem] -= 1

            covered[u] = covered[v] = covered[w] = True
            solution.append(tr)

            if backtrack(uncovered_count - 3):
                return True

            # Undo
            solution.pop()
            covered[u] = covered[v] = covered[w] = False

            for d_tr in deactivated:
                triple_active[d_tr] = True
                for elem in triples[d_tr]:
                    deg[elem] += 1

        return False

    if backtrack(n):
        print("YES")
        for tr in solution:
            a, b, c = triples[tr]
            print(f"{a + 1} {b + 1} {c + 1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
