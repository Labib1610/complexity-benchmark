import sys


def solve():
    sys.setrecursionlimit(2000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])

    triplets = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        triplets.append((a, b, c))

    unique_triplets = list(set(triplets))

    masks = []
    item_to_opts = [[] for _ in range(3 * q + 1)]
    for i, (a, b, c) in enumerate(unique_triplets):
        m = (1 << a) | (1 << b) | (1 << c)
        masks.append(m)
        item_to_opts[a].append(i)
        item_to_opts[b].append(i)
        item_to_opts[c].append(i)

    chosen = []

    def search(covered_mask, num_chosen):
        if num_chosen == q:
            return True

        initial_len = len(chosen)

        while True:
            best_opts = None
            min_candidates = 10**9

            for u in range(1, 3 * q + 1):
                if not (covered_mask & (1 << u)):
                    valid = [
                        opt
                        for opt in item_to_opts[u]
                        if not (masks[opt] & covered_mask)
                    ]
                    cnt = len(valid)
                    if cnt == 0:
                        del chosen[initial_len:]
                        return False
                    if cnt < min_candidates:
                        min_candidates = cnt
                        best_opts = valid
                        if cnt == 1:
                            break

            if min_candidates == 1:
                opt = best_opts[0]
                covered_mask |= masks[opt]
                chosen.append(opt)
                num_chosen += 1
                if num_chosen == q:
                    return True
            else:
                break

        for opt in best_opts:
            if not (masks[opt] & covered_mask):
                chosen.append(opt)
                if search(covered_mask | masks[opt], num_chosen + 1):
                    return True
                chosen.pop()

        del chosen[initial_len:]
        return False

    if search(0, 0):
        print("YES")
        for opt in chosen:
            a, b, c = unique_triplets[opt]
            print(f"{a} {b} {c}")
    else:
        print("NO")


if __name__ == "__main__":
    solve()
