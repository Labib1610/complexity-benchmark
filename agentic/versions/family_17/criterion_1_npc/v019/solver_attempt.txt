import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    num_elems = 3 * q
    idx = 2
    raw_triples = []
    for _ in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx+1]) - 1
        c = int(input_data[idx+2]) - 1
        idx += 3
        raw_triples.append((a, b, c))
    
    # Remove duplicates
    triples = list(set(raw_triples))
    m = len(triples)
    
    elem_to_triples = [[] for _ in range(num_elems)]
    for i, (a, b, c) in enumerate(triples):
        elem_to_triples[a].append(i)
        elem_to_triples[b].append(i)
        elem_to_triples[c].append(i)
    
    # Check if any element has 0 triples
    for i in range(num_elems):
        if not elem_to_triples[i]:
            print("NO")
            return
            
    covered = [False] * num_elems
    triple_dead = [False] * m
    deg = [len(elem_to_triples[i]) for i in range(num_elems)]
    
    solution = []
    
    def cover_triple(t_idx):
        # Disables this triple and removes all conflicting triples
        # Returns list of newly disabled triples and list of newly covered elements
        a, b, c = triples[t_idx]
        covered_now = []
        for elem in (a, b, c):
            covered[elem] = True
            covered_now.append(elem)
            
        disabled_now = []
        for elem in (a, b, c):
            for other_t in elem_to_triples[elem]:
                if not triple_dead[other_t]:
                    triple_dead[other_t] = True
                    disabled_now.append(other_t)
                    # Decrease degrees of all elements in other_t
                    oa, ob, oc = triples[other_t]
                    deg[oa] -= 1
                    deg[ob] -= 1
                    deg[oc] -= 1
        return covered_now, disabled_now

    def uncover_triple(t_idx, covered_now, disabled_now):
        for other_t in disabled_now:
            triple_dead[other_t] = False
            oa, ob, oc = triples[other_t]
            deg[oa] += 1
            deg[ob] += 1
            deg[oc] += 1
        for elem in covered_now:
            covered[elem] = False

    def search(elements_left):
        if elements_left == 0:
            return True
        
        # Unit propagation / Find column with minimum degree
        # If any uncovered element has degree 0, prune immediately.
        # If any uncovered element has degree 1, pick it immediately.
        best_col = -1
        best_deg = 1000000
        
        for c in range(num_elems):
            if not covered[c]:
                d = deg[c]
                if d == 0:
                    return False
                if d < best_deg:
                    best_deg = d
                    best_col = c
                    if d == 1:
                        break  # Can't get better than 1
        
        # Try all triples covering best_col
        # Make a copy of candidates since deg/state will be modified
        candidates = [t_idx for t_idx in elem_to_triples[best_col] if not triple_dead[t_idx]]
        
        for t_idx in candidates:
            cov_now, dis_now = cover_triple(t_idx)
            solution.append(t_idx)
            
            if search(elements_left - 3):
                return True
                
            solution.pop()
            uncover_triple(t_idx, cov_now, dis_now)
            
        return False

    if search(num_elems):
        print("YES")
        out = []
        for t_idx in solution:
            a, b, c = triples[t_idx]
            out.append(f"{a + 1} {b + 1} {c + 1}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
