import sys

# Increase recursion depth for search
sys.setrecursionlimit(5000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])

    N = 3 * q
    idx = 2
    triads = []
    for i in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        triads.append((a, b, c))

    # DLX structures
    # Root is node 0, column headers are 1 .. N
    L = [(i - 1) % (N + 1) for i in range(N + 1)]
    R = [(i + 1) % (N + 1) for i in range(N + 1)]
    U = list(range(N + 1))
    D = list(range(N + 1))
    C = list(range(N + 1))
    row = [0] * (N + 1)
    size = [0] * (N + 1)

    sz = N + 1
    for r_idx, (a, b, c) in enumerate(triads):
        first = sz
        for col in (a, b, c):
            node = sz
            C.append(col)
            row.append(r_idx)
            size[col] += 1
            u = U[col]
            U.append(u)
            D.append(col)
            D[u] = node
            U[col] = node
            sz += 1

        L.append(first + 2)
        R.append(first + 1)
        L.append(first)
        R.append(first + 2)
        L.append(first + 1)
        R.append(first)

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True

        c = R[0]
        min_size = size[c]
        curr = R[c]
        while curr != 0:
            if size[curr] < min_size:
                min_size = size[curr]
                c = curr
                if min_size == 0:
                    break
            curr = R[curr]

        if min_size == 0:
            return False

        cover(c)
        r_node = D[c]
        while r_node != c:
            solution.append(row[r_node])
            j = R[r_node]
            while j != r_node:
                cover(C[j])
                j = R[j]

            if search():
                return True

            solution.pop()
            j = L[r_node]
            while j != r_node:
                uncover(C[j])
                j = L[j]

            r_node = D[r_node]

        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_idx in solution:
            a, b, c = triads[r_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
