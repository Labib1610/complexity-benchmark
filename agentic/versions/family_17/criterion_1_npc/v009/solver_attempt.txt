```python
import sys

# Increase recursion depth for search
sys.setrecursionlimit(2000)


def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])

    m = 3 * q  # number of elements

    # Read triples and deduplicate
    triples = []
    seen = set()
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        trip = (a, b, c)
        if trip not in seen:
            seen.add(trip)
            triples.append(trip)

    # DLX Setup
    max_nodes = m + 1 + len(triples) * 3 + 10
    U = [0] * max_nodes
    D = [0] * max_nodes
    L = [0] * max_nodes
    R = [0] * max_nodes
    C = [0] * max_nodes
    row_id = [0] * max_nodes
    size = [0] * (m + 1)

    for i in range(m + 1):
        U[i] = i
        D[i] = i
        L[i] = i - 1
        R[i] = i + 1
    L[0] = m
    R[m] = 0

    node_count = m

    for k, (a, b, c) in enumerate(triples):
        row_nodes = []
        for col in (a, b, c):
            node_count += 1
            n = node_count
            C[n] = col
            row_id[n] = k
            size[col] += 1

            # Insert vertically at bottom of column
            U[n] = U[col]
            D[n] = col
            D[U[col]] = n
            U[col] = n
            row_nodes.append(n)

        n1, n2, n3 = row_nodes
        L[n1] = n3
        R[n1] = n2
        L[n2] = n1
        R[n2] = n3
        L[n3] = n2
        R[n3] = n1

    # If any element has no triples, impossible
    for i in range(1, m + 1):
        if size[i] == 0:
            sys.stdout.write("NO\n")
            return

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True

        # Choose column with min size
        c = R[0]
        min_size = size[c]
        cur = R[c]
        while cur != 0:
            if size[cur] < min_size:
                min_size = size[cur]
                c = cur
            cur = R[cur]

        if min_size == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            solution.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]

            if search():
                return True

            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            solution.pop()
            r = D[r]

        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_id in solution:
            a, b, c = triples[r_id]
            out.append(f"{a} {b} {c}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")


if __name__ == "__main__":
    main()
```
