import sys

# Increase recursion depth
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    n = 3 * q
    triples = []
    triples_of = [[] for _ in range(n + 1)]
    
    idx = 2
    for i in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triples.append((a, b, c))
        triples_of[a].append(i)
        triples_of[b].append(i)
        triples_of[c].append(i)
    
    # Check if any element has 0 triples
    for i in range(1, n + 1):
        if not triples_of[i]:
            print("NO")
            return
            
    used = [False] * (n + 1)
    chosen = []
    
    def search(covered_count):
        if covered_count == n:
            return True
        
        # Pick element with minimum remaining candidate triples (MRV)
        best_elem = -1
        min_candidates = 999999
        best_valid_triples = None
        
        for i in range(1, n + 1):
            if not used[i]:
                # Count valid triples
                valid = []
                for tri_idx in triples_of[i]:
                    a, b, c = triples[tri_idx]
                    if not used[a] and not used[b] and not used[c]:
                        valid.append(tri_idx)
                
                if len(valid) == 0:
                    return False
                
                if len(valid) < min_candidates:
                    min_candidates = len(valid)
                    best_elem = i
                    best_valid_triples = valid
                    if min_candidates == 1:
                        break
        
        for tri_idx in best_valid_triples:
            a, b, c = triples[tri_idx]
            if not used[a] and not used[b] and not used[c]:
                used[a] = used[b] = used[c] = True
                chosen.append(tri_idx)
                
                if search(covered_count + 3):
                    return True
                
                chosen.pop()
                used[a] = used[b] = used[c] = False
                
        return False

    if search(0):
        print("YES")
        out = []
        for idx in chosen:
            a, b, c = triples[idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
