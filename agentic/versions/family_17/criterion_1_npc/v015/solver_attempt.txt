import sys


def solve():
    sys.setrecursionlimit(10000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])
    n_cols = 3 * q

    triples = []
    idx = 2
    for _ in range(t):
        triples.append(
            (
                int(input_data[idx]),
                int(input_data[idx + 1]),
                int(input_data[idx + 2]),
            )
        )
        idx += 3

    # DLX structures
    max_nodes = n_cols + 1 + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    row_id = [0] * max_nodes
    S = [0] * (n_cols + 1)

    # Initialize column headers
    for i in range(n_cols + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
        S[i] = 0
    L[0] = n_cols
    R[n_cols] = 0

    node_cnt = n_cols + 1

    # Insert rows
    for r, (c1, c2, c3) in enumerate(triples):
        nodes = [node_cnt, node_cnt + 1, node_cnt + 2]
        node_cnt += 3

        cols = (c1, c2, c3)
        for i in range(3):
            n = nodes[i]
            c = cols[i]
            C[n] = c
            row_id[n] = r

            # Horizontal linking
            L[n] = nodes[(i - 1) % 3]
            R[n] = nodes[(i + 1) % 3]

            # Vertical linking
            U[n] = U[c]
            D[n] = c
            D[U[c]] = n
            U[c] = n
            S[c] += 1

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True

        # Choose column with minimum size (MRV)
        c = R[0]
        min_s = S[c]
        temp = R[c]
        while temp != 0:
            if S[temp] < min_s:
                min_s = S[temp]
                c = temp
            temp = R[temp]

        if min_s == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            ans.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]

            if search():
                return True

            ans.pop()
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            r = D[r]

        uncover(c)
        return False

    if search():
        print("YES")
        out = []
        for r in ans:
            a, b, c = triples[r]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
