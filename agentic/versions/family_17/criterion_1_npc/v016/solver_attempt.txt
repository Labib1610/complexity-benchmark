import sys

# Increase recursion depth just in case
sys.setrecursionlimit(10000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q

    triplets = []
    idx = 2
    for i in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        triplets.append((a, b, c))

    # DLX Implementation
    # Columns: 1 to n (0 is root)
    max_nodes = n + 1 + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row = [0] * max_nodes
    size = [0] * (n + 1)

    # Initialize column headers
    for i in range(n + 1):
        L[i] = (i - 1) % (n + 1)
        R[i] = (i + 1) % (n + 1)
        U[i] = i
        D[i] = i

    node_count = n

    for r_idx, (a, b, c) in enumerate(triplets):
        first_node = 0
        for c_idx in (a, b, c):
            node_count += 1
            cur = node_count
            col[cur] = c_idx
            row[cur] = r_idx
            size[c_idx] += 1

            # Insert into column list (at the bottom)
            D[cur] = c_idx
            U[cur] = U[c_idx]
            D[U[c_idx]] = cur
            U[c_idx] = cur

            # Insert into row list
            if first_node == 0:
                first_node = cur
                L[cur] = cur
                R[cur] = cur
            else:
                L[cur] = L[first_node]
                R[cur] = first_node
                R[L[first_node]] = cur
                L[first_node] = cur

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True

        c = R[0]
        min_size = size[c]
        curr = R[c]
        while curr != 0:
            if size[curr] < min_size:
                min_size = size[curr]
                c = curr
                if min_size == 0:
                    break
            curr = R[curr]

        if min_size == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            solution.append(row[r])
            j = R[r]
            while j != r:
                cover(col[j])
                j = R[j]
            if search():
                return True
            j = L[r]
            while j != r:
                uncover(col[j])
                j = L[j]
            solution.pop()
            r = D[r]
        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_idx in solution:
            a, b, c = triplets[r_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
