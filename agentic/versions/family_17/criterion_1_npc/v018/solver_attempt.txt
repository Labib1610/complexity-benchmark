```python
import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q

    trios = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        trios.append((a, b, c))

    max_nodes = 1 + n + t * 3 + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    row_of_node = [0] * max_nodes
    col_sz = [0] * (n + 1)

    for i in range(n + 1):
        L[i] = (i - 1) % (n + 1)
        R[i] = (i + 1) % (n + 1)
        U[i] = i
        D[i] = i
        C[i] = i

    node_cnt = n + 1

    for r_idx, (a, b, c) in enumerate(trios):
        row_nodes = []
        for col in (a, b, c):
            cur = node_cnt
            node_cnt += 1
            C[cur] = col
            row_of_node[cur] = r_idx
            col_sz[col] += 1

            U[cur] = U[col]
            D[cur] = col
            D[U[col]] = cur
            U[col] = cur

            row_nodes.append(cur)

        L[row_nodes[0]] = row_nodes[2]
        R[row_nodes[0]] = row_nodes[1]
        L[row_nodes[1]] = row_nodes[0]
        R[row_nodes[1]] = row_nodes[2]
        L[row_nodes[2]] = row_nodes[1]
        R[row_nodes[2]] = row_nodes[0]

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                col_sz[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                col_sz[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True

        c = R[0]
        min_sz = col_sz[c]
        curr = R[0]
        while curr != 0:
            if col_sz[curr] < min_sz:
                min_sz = col_sz[curr]
                c = curr
            curr = R[curr]

        if min_sz == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            solution.append(row_of_node[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]

            if search():
                return True

            solution.pop()
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            r = D[r]

        uncover(c)
        return False

    sys.setrecursionlimit(2000)
    if search():
        out = ["YES"]
        for r_idx in solution:
            a, b, c = trios[r_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
```
