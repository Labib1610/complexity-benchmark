import sys

# Increase recursion depth for search
sys.setrecursionlimit(3000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q
    
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        triples.append((a, b, c))
        idx += 3
        
    max_nodes = n + 1 + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    row = [0] * max_nodes
    size = [0] * (n + 1)
    
    # Initialize column headers
    for c in range(n + 1):
        L[c] = (c - 1) % (n + 1)
        R[c] = (c + 1) % (n + 1)
        U[c] = c
        D[c] = c
        size[c] = 0
        
    node_cnt = n
    for r, (a, b, c_val) in enumerate(triples):
        row_nodes = []
        for col in (a, b, c_val):
            node_cnt += 1
            i = node_cnt
            C[i] = col
            row[i] = r
            
            # Insert at bottom of column col
            U[i] = U[col]
            D[i] = col
            D[U[col]] = i
            U[col] = i
            size[col] += 1
            row_nodes.append(i)
            
        # Horizontally link the 3 nodes in this row
        L[row_nodes[0]] = row_nodes[2]
        R[row_nodes[0]] = row_nodes[1]
        L[row_nodes[1]] = row_nodes[0]
        R[row_nodes[1]] = row_nodes[2]
        L[row_nodes[2]] = row_nodes[1]
        R[row_nodes[2]] = row_nodes[0]

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum size
        c = R[0]
        min_sz = size[c]
        curr = R[c]
        while curr != 0:
            if size[curr] < min_sz:
                min_sz = size[curr]
                c = curr
            curr = R[curr]
            
        if min_sz == 0:
            return False
        
        cover(c)
        r_node = D[c]
        while r_node != c:
            ans.append(row[r_node])
            j = R[r_node]
            while j != r_node:
                cover(C[j])
                j = R[j]
            
            if search():
                return True
            
            ans.pop()
            j = L[r_node]
            while j != r_node:
                uncover(C[j])
                j = L[j]
            r_node = D[r_node]
            
        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_idx in ans:
            a, b, c_val = triples[r_idx]
            out.append(f"{a} {b} {c_val}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
