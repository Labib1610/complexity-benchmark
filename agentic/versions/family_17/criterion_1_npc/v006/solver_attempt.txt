import sys

# Increase recursion depth for deep search trees
sys.setrecursionlimit(20000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q
    
    trios = []
    idx = 2
    for i in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx+1]) - 1
        c = int(input_data[idx+2]) - 1
        idx += 3
        trios.append((a, b, c))
        
    # Build exact cover using DLX (Dancing Links)
    # Columns: 0 to n-1 (operators)
    # Rows: 0 to t-1 (trios)
    
    # DLX Data structures
    # Node 0 is the root (header).
    # Nodes 1 to n are column headers.
    # Subsequent nodes are matrix entries.
    
    MAX_NODES = n + 1 + 3 * t + 5
    L = [0] * MAX_NODES
    R = [0] * MAX_NODES
    U = [0] * MAX_NODES
    D = [0] * MAX_NODES
    C = [0] * MAX_NODES
    ROW = [0] * MAX_NODES
    SZ = [0] * (n + 1)
    
    # Initialize column headers
    for i in range(n + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
        C[i] = i
        SZ[i] = 0
    L[0] = n
    R[n] = 0
    
    node_cnt = n + 1
    
    for r, (a, b, c) in enumerate(trios):
        first = node_cnt
        for col in (a + 1, b + 1, c + 1):
            cur = node_cnt
            node_cnt += 1
            
            C[cur] = col
            ROW[cur] = r
            SZ[col] += 1
            
            # Insert into column
            U[cur] = U[col]
            D[cur] = col
            D[U[col]] = cur
            U[col] = cur
            
            # Insert into row (circular)
            if cur == first:
                L[cur] = cur
                R[cur] = cur
            else:
                L[cur] = L[first]
                R[cur] = first
                R[L[first]] = cur
                L[first] = cur

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                SZ[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                SZ[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum size (MRV heuristic)
        c = R[0]
        min_sz = SZ[c]
        curr = R[c]
        while curr != 0:
            if SZ[curr] < min_sz:
                min_sz = SZ[curr]
                c = curr
            curr = R[curr]
            
        if min_sz == 0:
            return False
            
        cover(c)
        r = D[c]
        while r != c:
            solution.append(ROW[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
                
            if search():
                return True
                
            solution.pop()
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            r = D[r]
            
        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_idx in solution:
            a, b, c = trios[r_idx]
            out.append(f"{a + 1} {b + 1} {c + 1}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
