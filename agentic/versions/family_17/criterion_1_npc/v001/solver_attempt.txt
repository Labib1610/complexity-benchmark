```python
import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])

    triples = []
    seen = set()
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        triple = (a, b, c)
        if triple not in seen:
            seen.add(triple)
            triples.append(triple)

    n = 3 * q
    cand = [[] for _ in range(n + 1)]
    for i, (a, b, c) in enumerate(triples):
        cand[a].append(i)
        cand[b].append(i)
        cand[c].append(i)

    # Quick check: if any element has 0 candidate triples, impossible
    for u in range(1, n + 1):
        if not cand[u]:
            print("NO")
            return

    covered = [False] * (n + 1)
    chosen = []

    def solve(num_covered):
        if num_covered == n:
            return True

        best_u = -1
        min_cand = 1000000

        for u in range(1, n + 1):
            if not covered[u]:
                cnt = 0
                for tri_idx in cand[u]:
                    a, b, c = triples[tri_idx]
                    if not covered[a] and not covered[b] and not covered[c]:
                        cnt += 1
                if cnt < min_cand:
                    min_cand = cnt
                    best_u = u
                    if cnt <= 1:
                        break

        if min_cand == 0:
            return False

        for tri_idx in cand[best_u]:
            a, b, c = triples[tri_idx]
            if not covered[a] and not covered[b] and not covered[c]:
                covered[a] = covered[b] = covered[c] = True
                chosen.append(tri_idx)
                if solve(num_covered + 3):
                    return True
                chosen.pop()
                covered[a] = covered[b] = covered[c] = False

        return False

    if solve(0):
        print("YES")
        out = []
        for tri_idx in chosen:
            a, b, c = triples[tri_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    main()
```
