import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triples.append((a, b, c))
    
    n = 3 * q
    
    # Store for each element the indices of triples containing it
    elem_triples = [[] for _ in range(n + 1)]
    for i, (a, b, c) in enumerate(triples):
        elem_triples[a].append(i)
        elem_triples[b].append(i)
        elem_triples[c].append(i)
    
    # Quick feasibility check: every element must have at least one triple
    for i in range(1, n + 1):
        if not elem_triples[i]:
            print("NO")
            return

    covered = [False] * (n + 1)
    solution = []
    
    # DLX / MRV Backtracking
    def search(remaining_count):
        if remaining_count == 0:
            return True
        
        # Choose uncovered element with minimum valid candidate triples (MRV)
        best_elem = -1
        min_candidates = float('inf')
        best_triples = []
        
        for elem in range(1, n + 1):
            if not covered[elem]:
                # Find valid triples for this element
                valid = []
                for tr_idx in elem_triples[elem]:
                    a, b, c = triples[tr_idx]
                    if not covered[a] and not covered[b] and not covered[c]:
                        valid.append(tr_idx)
                
                c_len = len(valid)
                if c_len == 0:
                    return False
                if c_len < min_candidates:
                    min_candidates = c_len
                    best_elem = elem
                    best_triples = valid
                    if min_candidates == 1:
                        break
        
        for tr_idx in best_triples:
            a, b, c = triples[tr_idx]
            if covered[a] or covered[b] or covered[c]:
                continue
            covered[a] = covered[b] = covered[c] = True
            solution.append(tr_idx)
            
            if search(remaining_count - 1):
                return True
            
            solution.pop()
            covered[a] = covered[b] = covered[c] = False
            
        return False

    if search(q):
        print("YES")
        for tr_idx in solution:
            a, b, c = triples[tr_idx]
            print(f"{a} {b} {c}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
