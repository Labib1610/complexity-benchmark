An efficient Python 3 solution using Donald Knuth's **Dancing Links (Algorithm X)** technique for the Exact Cover problem.

```python
import sys

# Increase recursion depth for deep search trees
sys.setrecursionlimit(200000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q

    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        triples.append((a, b, c))

    # Dancing Links (DLX) Data Structures
    # Column 0 is the root.
    # Columns 1 .. n represent operators 1 .. 3q.
    max_nodes = 1 + n + 3 * t + 10

    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    row_id = [0] * max_nodes
    col_size = [0] * (n + 1)

    # Initialize column headers
    for i in range(n + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
    L[0] = n
    R[n] = 0

    node_count = n

    # Insert rows (triples)
    for r, (a, b, c) in enumerate(triples):
        first = 0
        for col in (a, b, c):
            node_count += 1
            node = node_count

            C[node] = col
            row_id[node] = r
            col_size[col] += 1

            # Insert into vertical circular list for column `col`
            U[node] = U[col]
            D[node] = col
            D[U[col]] = node
            U[col] = node

            # Insert into horizontal circular list for this row
            if first == 0:
                first = node
                L[node] = node
                R[node] = node
            else:
                L[node] = L[first]
                R[node] = first
                R[L[first]] = node
                L[first] = node

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                col_size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                col_size[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True

        # Choose column with minimum size (MRV heuristic)
        c = R[0]
        min_sz = col_size[c]
        curr = R[0]
        while curr != 0:
            if col_size[curr] < min_sz:
                min_sz = col_size[curr]
                c = curr
            curr = R[curr]

        if min_sz == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            solution.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]

            if search():
                return True

            solution.pop()
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            r = D[r]

        uncover(c)
        return False

    if search():
        print("YES")
        out = []
        for r in solution:
            a, b, c = triples[r]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
```
